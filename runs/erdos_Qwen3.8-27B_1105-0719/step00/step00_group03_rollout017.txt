import numpy as np
from scipy.optimize import minimize
import time

def compute_score(h, n):
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(np.max(corr) * dx)

def logsumexp_objective(params, n, alpha):
    h = params.copy()
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def logsumexp_gradient(params, n, alpha):
    h = params
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        m_r = r - (n - 1)
        wr = w[r] * dx
        if wr < 1e-15:
            continue
        lo1 = max(0, m_r)
        hi1 = min(n - 1, n + m_r - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wr * (1 - h[idx - m_r])
        lo2 = max(0, -m_r)
        hi2 = min(n - 1, n - 1 - m_r)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= wr * h[idx + m_r]
    return grad

def project_to_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    # First clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust to meet sum constraint
    # Use water-filling / bisection on the Lagrange multiplier
    def sum_at_lambda(lam):
        return np.clip(h + lam, 0, 1).sum()
    
    # We need sum_at_lambda(lam) = target
    # sum_at_lambda is monotonically increasing in lam
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_at_lambda(mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

        if nn not in n_values and nn >= 10:

        for _ in range(3):
            pert = rng.random(n) * 0.3 - 0.15
            h0_pert = project_to_feasible(h0 + pert, n)
            starts.append(h0_pert)
        
        for h_start in starts:
            if elapsed() > budget_s - 60:
                break
            
            for alpha in alphas:
                if elapsed() > budget_s - 60:
                    break
                
                # SLSQP optimization
                try:
                    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                            'jac': lambda x: np.ones(n)}]
                    bounds = [(0, 1)] * n
                    
                    res = minimize(

                        bounds=bounds,

                        options={'maxiter': 200, 'ftol': 1e-12}
                    )
                    
                    if res.success or res.fun < best_c5:
                        h_opt = project_to_feasible(res.x, n)
                        c5 = compute_score(h_opt, n)
                        if c5 < best_c5:

                            best_h = h_opt.copy()
                            if n != best_n:
                                best_n = n
                except Exception:
                    pass

    dx = 2.0 / n
    q = 1.0 - h

    m = np.max(corr)

    active_indices = np.where(active)[0]
    for r in active_indices:
        m_r = r - (n - 1)
        lo1 = max(0, m_r)
        hi1 = min(n - 1, n + m_r - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += dx * (1.0 - h[idx - m_r])
        lo2 = max(0, -m_r)
        hi2 = min(n - 1, n - 1 - m_r)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[lo2:hi2 + 1] -= dx * h[idx + m_r]
    h_new = project_feasible(h - step_size * grad, n)
    return h_new, max_val


def binary_refine(h, n, max_iter=2000):
    """Try to improve a binary-like solution by flipping values"""
    h = h.copy()
    # Round to binary
    h = (h > 0.5).astype(float)
    h = project_feasible(h, n)
    current_score = compute_score(h, n)
    
    for it in range(max_iter):
        dx = 2.0 / n
        q = 1.0 - h
        corr = np.correlate(h, q, mode="full") * dx
        max_val = np.max(corr)
        if max_val <= current_score + 1e-10:
            break
        # Find position to flip (1->0 increases score, 0->1 decreases)
        # Try flipping a 1 to 0 where it contributes most to max lag
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_flip = None
        best_score = current_score
        # Try flipping some 1s to 0
        for i in ones[:20]:
            h[i] = 1.0 - h[i]
            s = compute_score(h, n)
            if s < best_score:
                best_score = s
                best_flip = i
            h[i] = 1.0 - h[i]
        # Try flipping some 0s to 1
        for i in zeros[:20]:
            h[i] = 1.0 - h[i]
            s = compute_score(h, n)
            if s < best_score:
                best_score = s
                best_flip = i
            h[i] = 1.0 - h[i]
        if best_flip is None:
            break
        h[best_flip] = 1.0 - h[best_flip]
        h = project_feasible(h, n)
        current_score = compute_score(h, n)
    return h, current_score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial values
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        best_h = np.array(globals()['initial_h_values'], dtype=float)
        best_n = len(best_h)
    else:
        best_n = 45
        best_h = np.full(best_n, 0.5)

    best_c5 = compute_score(best_h, best_n)

    # N values to try
    n_candidates = set()
    n_candidates.add(best_n)
    for offset in [-3, -2, -1, 0, 1, 2, 3, 5, 7, 10]:
        nn = best_n + offset
        if 10 <= nn <= 100:
            n_candidates.add(nn)
    n_candidates = sorted(n_candidates)

    alpha_schedule = [5, 10, 20, 50, 100, 200, 500]

    bounds_template = [(0.0, 1.0)]

    for n in n_candidates:
        if time.time() - t0 > budget_s - 30:
            break

        # Build starts
        starts = []

        # Interpolated from best
        if n != len(best_h):
            h_interp = np.interp(
                np.linspace(0, len(best_h) - 1, n),
                np.arange(len(best_h)), best_h)
        else:
            h_interp = best_h.copy()
        starts.append(project_feasible(h_interp, n))

        # Random starts
        for _ in range(5):
            h_rand = (rng.random(n) > 0.5).astype(float)
            h_rand = project_feasible(h_rand, n)
            starts.append(h_rand)

        # Structured: low-discrepancy-like
        h_struct = np.zeros(n)
        for i in range(n):
            h_struct[i] = 1.0 if (i * 0.618033988749895) % 1.0 > 0.5 else 0.0
        starts.append(project_feasible(h_struct, n))

        for h0 in starts:
            if time.time() - t0 > budget_s - 15:
                break

            current = h0.copy()
            for alpha in alpha_schedule:
                if time.time() - t0 > budget_s - 15:
                    break
                try:
                    bounds = [(0.0, 1.0)] * n
                    res = minimize(
                        lse_obj, current.copy(), args=(n, alpha),
                        jac=lse_grad,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': constr_eq,
                                     'jac': constr_eq_jac}],
                        options={'maxiter': 500, 'ftol': 1e-15}
                    )
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_opt = project_feasible(res.x, n)
                        c5 = compute_score(h_opt, n)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_opt.copy()
                            best_n = n
                        current = h_opt
                except Exception:
                    pass

            # Subgradient polishing
            if time.time() - t0 > budget_s - 30:
                break
            for sg_iter in range(500):
                if time.time() - t0 > budget_s - 15:
                    break
                step = 0.01 / (1.0 + sg_iter * 0.01)
                current, _ = subgradient_step(current, n, step)
                c5 = compute_score(current, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
                    best_n = n

            # Binary refinement
            if time.time() - t0 > budget_s - 30:
                break
            h_bin, c5_bin = binary_refine(current, n, max_iter=500)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n

    # Final polish on best solution
    if time.time() - t0 < budget_s - 10:
        # More SLSQP iterations with high alpha
        for alpha in [500, 1000, 2000]:
            if time.time() - t0 > budget_s - 10:
                break
            try:
                bounds = [(0.0, 1.0)] * best_n
                res = minimize(
                    lse_obj, best_h.copy(), args=(best_n, alpha),
                    jac=lse_grad, method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constr_eq,
                                 'jac': constr_eq_jac}],
                    options={'maxiter': 1000, 'ftol': 1e-16}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_opt = project_feasible(res.x, best_n)
                    c5 = compute_score(h_opt, best_n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
            except Exception:
                pass

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_score(best_h, best_n)

    return (best_h, best_c5, best_n)
