import numpy as np
import time


def project_feasible(y, target_sum):
    """Project y to {h: 0<=h<=1, sum(h)=target_sum} via scalar shift bisection."""
    lo = -y.max() - 3.0
    hi = 1.0 - y.min() + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y + mid, 0.0, 1.0)) < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    res = target_sum - h.sum()
    if abs(res) > 1e-12:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += res / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_lse(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*corr_k)) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sc = beta * corr - beta * corr.max()
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        a = max(0, s)
        b = min(n - 1, n - 1 + s)
        if a <= b:
            idx = np.arange(a, b + 1) - s
            g[a:b + 1] += w[k] * (1.0 - h[idx])
        a = max(0, -s)
        b = min(n - 1, n - 1 - s)
        if a <= b:
            idx = np.arange(a, b + 1) + s
            g[a:b + 1] -= w[k] * h[idx]
    return g
