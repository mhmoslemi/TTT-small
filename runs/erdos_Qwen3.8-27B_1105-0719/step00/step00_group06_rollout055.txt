import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(prof))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-fill to hit target
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase; add to entries < 1
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            add = diff / count
            if add > 1.0 - h[mask].min():
                h[mask] = 1.0
            else:
                h[mask] += add
        else:
            # Need to decrease; subtract from entries > 0
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            sub = -diff / count
            if sub > h[mask].min():
                h[mask] = 0.0
            else:
                h[mask] -= sub
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Distribute to any entry with slack
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    # Compute profile
    one_minus = 1.0 - x
    prof = np.correlate(x, one_minus, mode='full') * dx
    # Log-sum-exp (stable)
    pmax = np.max(prof)
    shifted = (prof - pmax) / tau
    log_sum_exp = pmax + tau * np.log(np.sum(np.exp(shifted)))
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            # ∂ψ_s/∂h[j] = dx * [(1-h[j+s])*1(0≤j≤n-1-s) - h[j-s]*1(s≤j≤n-1)]
            j_range1 = np.arange(0, n - s)
            if len(j_range1) > 0:
                grad[j_range1] += w * dx * (1.0 - x[j_range1 + s])
            j_range2 = np.arange(s, n)
            if len(j_range2) > 0:
                grad[j_range2] -= w * dx * x[j_range2 - s]
        else:
            a = -s
            # ∂ψ_{-a}/∂h[j] = dx * [(1-h[j-a])*1(a≤j≤n-1) - h[j+a]*1(0≤j≤n-1-a)]
            j_range1 = np.arange(a, n)
            if len(j_range1) > 0:
                grad[j_range1] += w * dx * (1.0 - x[j_range1 - a])
            j_range2 = np.arange(0, n - a)
            if len(j_range2) > 0:
                grad[j_range2] -= w * dx * x[j_range2 + a]
    return log_sum_exp, grad

def make_slsqp_func(tau):
    def func(x):
        val, _ = smooth_obj_and_grad(x, tau)
        return val
    def jac(x):
        _, g = smooth_obj_and_grad(x, tau)
        return g
    return func, jac

def run_slsqp(h0, n, deadline, tau_list):
    h = h0.copy()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    best_h = h.copy()
    best_val = exact_objective(h)
    
    for tau in tau_list:
        if time.time() > deadline - 5:
            break
        func, jac = make_slsqp_func(tau)
        try:
            res = minimize(func, h, method='SLSQP', jac=jac, bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
            if res.success or res.fun < best_val:
                h_new = project_box_sum(res.x)
                val = exact_objective(h_new)
                if val < best_val - 1e-14:
                    best_val = val
                    best_h = h_new.copy()
                h = project_box_sum(res.x)
        except Exception:
            pass
    return best_h, best_val

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_search(b, deadline, max_iters=5000):
    if b is None:
        return None, None
    n = len(b)
    cur_val = exact_objective(b)
    best_val = cur_val
    best_b = b.copy()
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        improved = False
        rng = np.random.RandomState(it)
        # Try a few random swaps
        for _ in range(20):
            if time.time() > deadline - 3:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            oi, oz = ones[i], zeros[j]
            b[oi] = 0.0
            b[oz] = 1.0
            val = exact_objective(b)
            if val < best_val - 1e-14:
                best_val = val
                best_b = b.copy()
                improved = True
            else:
                b[oi] = 1.0
                b[oz] = 0.0
                # swap in lists
            # Update lists
            if b[oz] == 1.0:
                ones[i] = oz
                zeros[j] = oi
            else:
                # reverted
                pass
        if not improved and it > 0 and it % 100 == 0:
            # Try systematic local search
            break
    return best_b, best_val

    n = len(h)
    noise = rng.randn(n) * scale
    h_new = h + noise

_target_sum = 70.0

    prof = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(prof)
    shifted = (prof - pmax) / _tau
    e = np.exp(shifted)
    se = np.sum(e)

def transfer_to_size(parent, n):

    h = np.interp(xn, xp, parent)
    return project_box_sum(h)

    sizes = sorted(set(sizes), key=lambda x: (x != parent_n, abs(x - parent_n)))

    tau_full = [0.01, 0.003, 0.001, 0.0003, 0.0001, 0.00003]
    tau_short = [0.003, 0.001, 0.0003, 0.0001]

    # Phase 1: Main optimization on parent size
    if time.time() > deadline - 30:
        pass
    else:
        h_t = transfer_to_size(parent, parent_n)
        h1, v1 = multi_start_slq(h_t, parent_n, deadline, rng, n_restarts=2)
        if v1 < best_val - 1e-14:
            best_val = v1
            best_h = h1.copy()
            best_n = parent_n

        b = binary_threshold(h1)
        if b is not None:
            bb, bv = binary_local_search(b, deadline, rng, max_rounds=500)
            if bv < best_val - 1e-14:
                best_val = bv
                best_h = bb.copy()
                best_n = parent_n

        # More restarts if time allows
        for i in range(4):
            if time.time() > deadline - 30:
                break
            h_p = project_box_sum(best_h + rng.randn(best_n) * 0.03)
            h_r, v_r = run_slsqp(h_p, best_n, deadline, tau_full)
            if v_r < best_val - 1e-14:
                best_val = v_r
                best_h = h_r.copy()
            b2 = binary_threshold(h_r)
            if b2 is not None:
                bb2, bv2 = binary_local_search(b2, deadline, rng, max_rounds=200)
                if bv2 < best_val - 1e-14:
                    best_val = bv2
                    best_h = bb2.copy()
                    best_n = best_n

    # Phase 2: Try other sizes
    for n in sizes:
        if n == parent_n:
            continue
        if time.time() > deadline - 25:
            break

        h_trans = transfer_to_size(parent, n)
        starts = [h_trans]
        # Add alternating pattern
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        if n % 2 == 1:
            h_alt[-1] = 0.0
        starts.append(project_box_sum(h_alt))
        # Sinusoidal
        xs = np.linspace(0, 2, n)
        h_sin = 0.5 + 0.35 * np.sin(np.pi * xs + 0.3 * rng.rand())
        starts.append(project_box_sum(h_sin))

        for h_s in starts[:3]:
            if time.time() > deadline - 20:
                break
            h_o, v_o = run_slsqp(h_s, n, deadline, tau_short, max_iter_per_tau=150)
            if v_o < best_val - 1e-14:
                best_val = v_o
                best_h = h_o.copy()
                best_n = n
            b = binary_threshold(h_o)
            if b is not None:
                bb, bv = binary_local_search(b, deadline, rng, max_rounds=300)
                if bv < best_val - 1e-14:
                    best_val = bv
                    best_h = bb.copy()
                    best_n = n

    # Phase 3: Final polish on best solution
    if time.time() < deadline - 15:
        n_b = len(best_h)
        h_p = project_box_sum(best_h + rng.randn(n_b) * 0.01)
        h_f, v_f = run_slsqp(h_p, n_b, deadline, [0.001, 0.0003, 0.0001, 0.00003], max_iter_per_tau=150)
        if v_f < best_val - 1e-14:
            best_val = v_f
            best_h = h_f.copy()
        b = binary_threshold(best_h)
        if b is not None:
            bb, bv = binary_local_search(b, deadline, rng, max_rounds=600)
            if bv < best_val - 1e-14:
                best_val = bv
                best_h = bb.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_val = exact_objective(best_h)

    return best_h, best_val, len(best_h)
