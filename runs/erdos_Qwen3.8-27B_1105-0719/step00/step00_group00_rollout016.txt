def optimize_gradient(h, max_time_s=300, top_k=50):
    """Gradient-guided pair search. Returns (h, c5)."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()
    n_delta = 10
    
    while time.time() - start < max_time_s:
        corr = compute_corr(h)
        c5 = float(np.max(corr) * dx)
        
        # Find active lags
        c_max = np.max(corr)
        active = corr > c_max - 0.01 * c_max
        n_active = np.sum(active)
        
        # Compute gradient for active lags
        # For lag offset d (0-indexed in corr: index k corresponds to offset d = k-(n-1))
        # grad[j] = sum over active d of (contribution of h[j] to C(d))
        # C(d) = sum_i h[i]*g[i+d], so dC[d]/dh[j] = g[j+d] - h[j-d] (valid terms)
        
        grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            d = k - (n - 1)  # lag offset
            # h[j] appears as first factor when i=j: g[j+d] = 1-h[j+d]
            # h[j] appears as second factor (in g) when i+d=j, i=j-d: -h[j-d]
            # g[j+d] term: valid if 0 <= j+d < n
            j1 = j = np.arange(n)
            valid1 = (j + d >= 0) & (j + d < n)
            if np.any(valid1):
                grad[valid1] += (1.0 - h[j[valid1] + d]) / n_active
            # -h[j-d] term: valid if 0 <= j-d < n
            valid2 = (j - d >= 0) & (j - d < n)
            if np.any(valid2):
                grad[valid2] -= h[j[valid2] - d] / n_active
        
        # Sort pairs by gradient difference
        # We want to move from high grad to low grad (decrease grad difference)
        # Actually, we want to decrease the objective.
        # Direction: h[a] += delta, h[b] -= delta
        # Change in objective ≈ delta * (grad[a] - grad[b]) (for small delta)
        # To decrease: if grad[a] > grad[b], use delta < 0 (i.e., move from a to b)
        # If grad[a] < grad[b], use delta > 0 (move from b to a)
        # Best pairs have large |grad[a] - grad[b]|
        
        # Compute gradient differences for all pairs
        # grad_diff[a,b] = grad[a] - grad[b]
        # We want pairs where moving mass in the right direction helps
        
        # Instead of all pairs, just try top_k based on gradient
        # For each a, find b that maximizes the "descent"
        # Descent direction for pair (a,b): if grad[a] > grad[b], decrease h[a], increase h[b]
        #     i.e., delta < 0 in our convention (h[a]+=delta means h[a] decreases)
        #     Equivalent to transfer from a to b
        # if grad[a] < grad[b], increase h[a], decrease h[b] (delta > 0)
        #     Transfer from b to a
        
        best_c5 = c5
        best_move = None
        
        # Try top pairs
        # Compute all pairwise gradient differences
        g = grad
        diff_matrix = g[:, None] - g[None, :]  # diff_matrix[a,b] = g[a] - g[b]
        
        # Sort by |diff|
        # Get top_k pairs
        iu = np.triu_indices(n, 1)
        diffs = diff_matrix[iu]
        sorted_idx = np.argsort(-np.abs(diffs))[:top_k]
        
        for idx in sorted_idx:
            a, b = iu[0][idx], iu[1][idx]
            lo = max(-h[a], h[b] - 1.0) + 1e-12
            hi = min(1.0 - h[a], h[b]) - 1e-12
            if lo >= hi:
                continue
            
            ha, hb = h[a], h[b]
            deltas = np.linspace(lo, hi, n_delta)
            
            for d in deltas:
                h[a], h[b] = ha + d, hb - d
                c5_new = float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_move = (a, b, d)
            
            h[a], h[b] = ha, hb
        
        if best_move is None:
            break
        a, b, d = best_move
        h[a] += d
        h[b] -= d
        h = np.clip(h, 0, 1)
        c5 = best_c5
    
    return h, c5

def adjust_sum(h, n):
    """Adjust h so that sum(h) = n/2 exactly, while keeping 0<=h<=1"""
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n_mask = np.sum(mask)
        if n_mask == 0:
            break
        h[mask] += diff / n_mask
        h = np.clip(h, 0, 1)
    return h

def upsample(h, n_new):
    """Upsample h to n_new points using step function (nearest) interpolation"""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    # Each old cell spans n_new/n_old new cells
    # Use simple repetition + remainder
    ratio = n_new / n_old
    indices = np.floor(np.arange(n_new) * n_old / n_new).astype(int)
    h_new = h[indices]
    h_new = adjust_sum(h_new, n_new)
    return h_new
