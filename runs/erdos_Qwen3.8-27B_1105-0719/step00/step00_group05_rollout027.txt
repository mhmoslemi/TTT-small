import numpy as np
from scipy.optimize import minimize_scalar
import time

def project_feasible(z, target_sum):
    """Project z onto {0 <= h <= 1, sum(h) = target_sum} using bisection on lambda."""
    n = len(z)
    lo, hi = -max(z) - 1, 1 - min(z) + 1
    
    def f(lam):
        return np.sum(np.clip(z + lam, 0, 1)) - target_sum
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2
    h = np.clip(z + lam, 0, 1)
    
    # Fix rounding
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Find indices with room
        if diff > 0:
            idx = np.where(h < 1 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            h[idx] = np.clip(h[idx] + per, 0, 1)
        # Final fix
        diff = target_sum - np.sum(h)
        idx2 = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
        if len(idx2) > 0 and abs(diff) > 1e-15:
            h[idx2[0]] += diff
    return h

def evaluate_exact(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    """Compute smooth max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx  # shape (2n-1,)
    
    # Softmax weights
    v_max = np.max(vals)
    exp_vals = np.exp(beta * (vals - v_max))
    p = exp_vals / np.sum(exp_vals)
    smooth_val = v_max + np.log(np.sum(exp_vals)) / beta
    
    # Gradient
    g = np.zeros(n)
    # corr has 2n-1 entries, indexed from lag -(n-1) to +(n-1)
    # corr[n-1 + k] corresponds to lag k
    for k in range(-(n-1), n):

        if pk < 1e-15:
            continue
        if k == 0:
            g += pk * (1 - 2*h)
        elif k > 0:
            # c_k = sum_{i=0}^{n-k-1} h_i * (1-h_{i+k})
            # dc_k/dh_j = (1-h_{j+k}) for j < n-k
            #             + (-h_{j-k}) for j >= k and j-k < n-k
            m = k
            # First part: j in [0, n-m)
            first = np.zeros(n)
            first[:n-m] = 1 - h[m:]
            # Second part: j in [m, n)
            second = np.zeros(n)
            second[m:] = -h[:n-m]
            g += pk * (first + second)
        else:
            m = -k
            # c_-m = sum_{i=0}^{n-m-1} h_{i+m} * (1-h_i)

            first[m:] = 1 - h[:n-m]
            second = np.zeros(n)
            second[:n-m] = -h[m:]
            g += pk * (first + second)
    
    return smooth_val, g

            # d/dh[j] of sum_i h[i]*(1-h[i+m]) [for positive lag]
            #   = (1-h[j+m]) if j+m < n  (j is first factor)
            #   + (-h[j-m]) if j-m >= 0   (j is second factor)
            # d/dh[j] of sum_i h[i+m]*(1-h[i]) [for negative lag]
            #   = (1-h[j-m]) if j-m >= 0   (j+m is first factor, j is i+m)
            #   + (-h[j+m]) if j+m < n      (j is second factor)
            
            if lag > 0:
                # c = sum_{i=0}^{n-m-1} h[i] * (1-h[i+m])
                part1 = np.zeros(n)
                part1[:n-m] = 1 - h[m:]
                part2 = np.zeros(n)
                part2[m:] = -h[:n-m]
                g += pk * dx * (part1 + part2)
            else:
                # c = sum_{i=0}^{n-m-1} h[i+m] * (1-h[i])
                part1 = np.zeros(n)
                part1[m:] = 1 - h[:n-m]
                part2 = np.zeros(n)
                part2[:n-m] = -h[m:]
                g += pk * dx * (part1 + part2)
    
    return smooth_val, g
