import numpy as np

def exact_objective(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    """Project h onto {0<=h_i<=1, sum(h_i)=target}"""
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = 0.5 * n
    # Replace non-finite
    h[~np.isfinite(h)] = 0.0
    # Clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # Water-fill to achieve target sum
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            # Need to subtract
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_profile_formula(h):
    """Compute the profile using explicit formula (for gradient alignment)."""
    n = len(h)
    dx = 2.0 / n
    # lag s ranges from -(n-1) to (n-1)
    # output index k = n-1+s
    out = np.zeros(2*n - 1)
    for s in range(0, n):
        # s >= 0: sum_{i=0}^{n-s-1} h[i+s] * (1-h[i])
        if s == 0:
            out[n-1] = np.sum(h * (1.0 - h)) * dx
        else:
            out[n-1+s] = np.sum(h[s:] * (1.0 - h[:n-s])) * dx
    for a in range(1, n):
        # s = -a: sum_{i=0}^{n-a-1} h[i] * (1-h[i+a])
        out[n-1-a] = np.sum(h[:n-a] * (1.0 - h[a:])) * dx
    return out

def smooth_objective_and_grad(h, tau):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    # Compute profile
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    P_max = np.max(profile)
    shifted = (profile - P_max) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        w_sum = 1e-300
    sigma_tau = tau * np.log(w_sum)
    value = P_max + sigma_tau
    # Gradient: weighted sum of individual profile gradients
    grad = np.zeros(n)
    # For each lag, compute gradient contribution
    # Weights normalized
    w = weights / w_sum
    for s in range(0, n):
        if s == 0:
            # d profile[n-1]/dh[j] = (1-2*h[j])*dx
            g = (1.0 - 2.0 * h) * dx
            grad += w[n-1] * g
        else:
            # profile[n-1+s] = sum_{i=0}^{n-s-1} h[i+s]*(1-h[i])
            # d/dh[j]: 
            #   -h[j+s]*dx for j in [0, n-s-1]
            #   +(1-h[j-s])*dx for j in [s, n-1]
            g = np.zeros(n)
            g[:n-s] -= h[s:] * dx  # from -h[i] term... wait
            # Actually: profile = sum h[i+s]*(1-h[i])
            # d/dh[j] = (1-h[j-s])*dx if j >= s (h[j-s] is the h[i+s] term)
            #           -h[j+s]*dx if j <= n-s-1 (h[j+s] is the h[i] term... no)
            # Let me redo: profile = sum_{i=0}^{n-s-1} h[i+s]*(1-h[i])
            # Term i contributes h[i+s]*1 - h[i+s]*h[i]
            # d/dh[j]:
            #   From h[i+s]: 1 if i+s=j, i.e., i=j-s, valid if 0<=j-s<=n-s-1, i.e., s<=j<=n-1
            #   From -h[i+s]*h[i]: -h[i] if i+s=j (j-s=i), and -h[i+s] if i=j
            #   So: (1-h[j-s])*dx if s<=j<=n-1, and -h[j]*dx if 0<=j<=n-s-1
            #   Wait, that's not right either. Let me be more careful.
            #   profile = sum_i h[i+s] - sum_i h[i+s]*h[i]
            #   d/dh[j] of sum_i h[i+s] = 1 if j-s is a valid i (s<=j<=n-1), else 0
            #   d/dh[j] of sum_i h[i+s]*h[i]:
            #     h[j+s] if i=j (valid if 0<=j<=n-s-1)  [h[i] = h[j], h[i+s] = h[j+s]]
            #     h[j-s] if i+s=j, i=j-s (valid if s<=j<=n-1)  [h[i+s]=h[j], h[i]=h[j-s]]
            #   So d profile/dh[j] = dx * [1*(s<=j<=n-1) - h[j+s]*(0<=j<=n-s-1) - h[j-s]*(s<=j<=n-1)]
            #                        = dx * [(1-h[j-s])*(s<=j<=n-1) - h[j+s]*(0<=j<=n-s-1)]
            # For j in [s, n-1]: contribution = (1-h[j-s])*dx
            # For j in [0, n-s-1]: contribution = -h[j+s]*dx
            # For j in [s, n-s-1] (overlap): contribution = (1-h[j-s])*dx - h[j+s]*dx
            # For j in [0, s-1]: contribution = -h[j+s]*dx
            # For j in [n-s, n-1]: contribution = (1-h[j-s])*dx
            g[:n-s] -= h[s:] * dx
            g[s:] += (1.0 - h[:n-s]) * dx
            grad += w[n-1+s] * g
    for a in range(1, n):
        # profile[n-1-a] = sum_{i=0}^{n-a-1} h[i]*(1-h[i+a])
        # Similarly: d/dh[j] = dx * [(1-h[j+a])*(0<=j<=n-a-1) - h[j-a]*(a<=j<=n-1)]
        g = np.zeros(n)
        g[:n-a] += (1.0 - h[a:]) * dx
        g[a:] -= h[:n-a] * dx
        grad += w[n-1-a] * g
    return value, grad

def binary_threshold(h):
    """Convert to binary vector with exactly n/2 ones (for even n)."""
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    # Top target values become 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_search(b, max_rounds=200, rng=None):

    n = len(b)
    if n % 2 != 0:
        return b
    target = n // 2
    b = b.copy()

    zeros = np.where(b <= 0.5)[0]

    if len(ones) > target:
        excess = len(ones) - target
        for _ in range(excess):
            b[ones[-1]] = 0
            ones = ones[:-1]
            zeros = np.append(zeros, ones[-1] + 1)  # hmm this is messy
    # Simpler: just threshold
    if len(ones) != target:
        b = binary_threshold(b)
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b <= 0.5)[0]
    
    cur_c5, _ = exact_objective(b)
    best_c5 = cur_c5
    best_b = b.copy()
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    improved = True
    round_num = 0
    while improved and round_num < max_rounds:
        improved = False
        round_num += 1
        # Try all pairs (or a subset for large n)

        else:

            pairs = []
            for _ in range(min(500, len(ones) * len(zeros))):

                j = zeros[rng.randint(len(zeros))]
                if i != j:

            if i == j:

            if b[i] <= 0.5 or b[j] > 0.5:

                b[i] = 1
                b[j] = 0
            # Update ones/zeros
            ones = np.where(b > 0.5)[0]
            zeros = np.where(b <= 0.5)[0]

def slsqp_optimize(h0, tau_list, max_iter_per_tau=100, deadline=None):
    """Run SLSQP with log-sum-exp smoothing, decreasing tau."""
    n = len(h0)
    best_h = h0.copy()
    best_c5, _ = exact_objective(best_h)
    
    bounds = [(0.0, 1.0)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([0.5 * n])
    
    for tau in tau_list:
        if deadline is not None and time.time() > deadline - 10:
            break
        
        def obj_func(x):
            # Penalty for constraint violation
            viol = 0.0
            sum_err = np.sum(x) - 0.5 * n
            viol += 1e4 * sum_err * sum_err
            out_of_box = np.sum(np.maximum(0, x - 1.0)**2) + np.sum(np.maximum(0, -x)**2)
            viol += 1e4 * out_of_box
            try:
                val, _ = smooth_objective_and_grad(x, tau)
            except Exception:
                val = 10.0
            return val + viol
        
        def grad_func(x):
            try:
                _, g = smooth_objective_and_grad(x, tau)
            except Exception:
                g = np.zeros(n)
            # Add penalty gradient
            sum_err = np.sum(x) - 0.5 * n
            g = g + 2e4 * sum_err * np.ones(n)
            g = g + 2e4 * np.where(x > 1.0, x - 1.0, 0.0) - 2e4 * np.where(x < 0.0, x, 0.0)
            return g
        
        def eq_func(x):
            return np.sum(x) - 0.5 * n
        
        def eq_grad(x):
            return np.ones(n)
        
        try:
            res = minimize(
                obj_func, best_h.copy(), method='SLSQP',
                jac=grad_func, bounds=bounds,
                constraints=[{'type': 'eq', 'fun': eq_func, 'jac': eq_grad}],
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-9, 'disp': False}
            )
            # Project result
            candidate = project_box_sum(res.x)
            c5, _ = exact_objective(candidate)
            if c5 < best_c5 - 1e-12:
                best_h = candidate
                best_c5 = c5
        except Exception:
            pass
    
    return best_h, best_c5

def random_perturb(h, scale, rng):
    """Add random perturbation and project."""
    h_new = h + rng.randn(len(h)) * scale
    return project_box_sum(h_new)

def interpolate(h1, h2, gamma):
    """Interpolate between two vectors."""
    return (1 - gamma) * h1 + gamma * h2

def forge_starts(n, rng):
    """Generate diverse starting vectors of length n."""
    starts = []
    # Uniform
    starts.append(np.full(n, 0.5))
    # Alternating binary
    b = np.zeros(n)
    b[::2] = 1.0
    starts.append(project_box_sum(b))
    # Complement
    b2 = 1.0 - b
    starts.append(project_box_sum(b2))
    # Random binary
    for _ in range(3):
        rb = (rng.rand(n) < 0.5).astype(float)
        starts.append(project_box_sum(rb))
    # Sinusoidal
    x = np.linspace(0, 2*np.pi, n, endpoint=False)
    starts.append(project_box_sum(0.5 + 0.4*np.sin(x)))
    starts.append(project_box_sum(0.5 + 0.4*np.cos(2*x)))
    # Block pattern
    block = n // 4
    b3 = np.zeros(n)
    b3[:block] = 1
    b3[2*block:3*block] = 1
    starts.append(project_box_sum(b3))
    return starts

def transfer_parent(parent, target_n):
    """Transfer parent vector to different size via interpolation."""
    n_parent = len(parent)
    if target_n == n_parent:
        return parent.copy()
    # Linear interpolation
    x_parent = np.linspace(0, 1, n_parent, endpoint=False) + 0.5/n_parent
    x_new = np.linspace(0, 1, target_n, endpoint=False) + 0.5/target_n
    new_vals = np.interp(x_new, x_parent, parent)
    return project_box_sum(new_vals)
