def compute_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_rows(h):

def project_box_hyperplane(h, target):

def lse_smooth_max(values, beta):
    """Log-sum-exp smooth maximum"""
    v = values - np.max(values)
    return np.max(values) + np.log(np.sum(np.exp(beta * v))) / beta

def lse_grad(values, grad_components, beta):
    """Gradient of LSE smooth max: sum w_t * g_t"""
    v = values - np.max(values)
    weights = np.exp(beta * v)
    weights /= weights.sum()
    # grad_components has shape (num_rows, dim)
    return weights @ grad_components

def correlate_grad_row(h, d, n):
    """
    Gradient of R_d(h) = sum_i h[i] * (1-h[i+d]) w.r.t. h
    where the sum is over valid indices.
    For the np.correlate convention:
    corr[t] = sum_i h[i] * (1-h[i + (n-1-t)])
    So lag d = n-1-t, meaning:
    corr[t] = sum_i h[i] * (1-h[i+d]) where i ranges so that 0 <= i < n and 0 <= i+d < n
    """
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        # sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        valid = slice(0, n - d)
        # d/dh[i] of h[i]*(1-h[i+d]) = (1-h[i+d])
        g[:n-d] += (1.0 - h[d:])
        # d/dh[i+d] of h[i]*(1-h[i+d]) = -h[i]
        g[d:] += -h[:n-d]
    else:
        q = -d
        # sum_{i=q}^{n-1} h[i] * (1-h[i-q])
        # Let j = i-q, so sum_{j=0}^{n-1-q} h[j+q] * (1-h[j])
        # d/dh[j+q] = (1-h[j])
        g[q:] += (1.0 - h[:n-q])
        # d/dh[j] = -h[j+q]
        g[:n-q] += -h[q:]
    return g

    grads = np.zeros((2*n-1, n))
    for t in range(2*n-1):
        d = (n - 1) - t
        grads[t] = correlate_grad_row(h, d, n)
    return corr, grads

def lse_smoothed_obj_and_grad(h, beta):
    """Compute LSE smoothed objective and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr, grads = compute_all_correlate_grads(h, n)
    obj = lse_smooth_max(corr, beta) * dx
    grad = lse_grad(corr, grads, beta) * dx
    return obj, grad

    """Reduce gradient from full h-space to a-space (symmetric coordinates)"""
    m = n // 2
    # h[i] = a[i] for i < m
    # h[n-1-i] = 1 - a[i] for i < m (odd: h[m] = 0.5, excluded)
    # dF/da[i] = dF/dh[i] - dF/dh[n-1-i]
    grad_a = grad_h[:m].copy()
    for i in range(m):
        grad_a[i] -= grad_h[n-1-i]
    return grad_a

            cur_h = np.clip(cur_h + rng.normal(0, 0.02, n), 0, 1)
            cur_h = project_box(cur_h, target)
            cur_c = compute_c5(cur_h)
            if cur_c > best_c * 1.005:
                cur_h = best_h.copy()
                cur_c = best_c
            no_improve = 0

    return best_h, best_c

def try_n_value(h_best, n_best, n_target, time_budget, rng):
    if n_target == n_best or n_target < 10:
        return None, None
    # Interpolate
    x_old = np.arange(n_best, dtype=float) / n_best * 2.0
    x_new = np.arange(n_target, dtype=float) / n_target * 2.0
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_box(h_interp, n_target / 2.0)
    h_interp = sym_h(h_interp)
    h_interp = project_box(h_interp, n_target / 2.0)
    c = compute_c5(h_interp)
    
    # Local optimize in symmetric coords
    m = n_target // 2
    a = h_interp[:m].copy()
    a_best, c_best = optimize_symmetric(a, n_target, time_budget, rng)
    h_best_t = h_from_a(a_best, n_target)
    c_best = compute_c5(h_best_t)
    if c_best > c:
        h_best_t = h_interp
        c_best = c
    return h_best_t, c_best

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    
    rng = np.random.default_rng(seed)
    t_start = time.time()
    reserve = 5.0
    
    # Get initial h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float).ravel()
        n_init = len(h_init)
    else:
        h_init = np.full(95, 0.5)
        n_init = 95
    
    h_init = np.clip(h_init, 0, 1)
    h_init = project_box(h_init, n_init / 2.0)
    
    # Symmetrize
    h_sym = sym_h(h_init)
    h_sym = project_box(h_sym, n_init / 2.0)
    
    c_orig = compute_c5(h_init)
    c_sym = compute_c5(h_sym)
    best_c = min(c_orig, c_sym)
    best_h = h_init.copy() if c_orig <= c_sym else h_sym.copy()
    best_n = n_init
    
    # Work in symmetric coordinates
    m = n_init // 2
    h_work = sym_h(h_init)
    h_work = project_box(h_work, n_init / 2.0)
    a_start = h_work[:m].copy()
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 1: LSE gradient descent (40% of budget)
    t1 = min(remaining * 0.35, 300)
    if t1 > 10:
        a_opt, c_opt = optimize_symmetric(a_start, n_init, t1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_from_a(a_opt, n_init)
            best_n = n_init
        a_start = a_opt
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 2: Coordinate descent (25% of budget)
    t2 = min(remaining * 0.25, 200)
    if t2 > 10:
        a_cd, c_cd = coordinate_descent_sym(a_start, n_init, t2, rng)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_from_a(a_cd, n_init)
            best_n = n_init
        a_start = a_cd
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 3: Full-space local search on best solution (15%)
    t3 = min(remaining * 0.15, 100)
    if t3 > 5:
        h_best_now, c_best_now = full_space_local(best_h, best_n, t3, rng)
        if c_best_now < best_c:
            best_c = c_best_now
            best_h = h_best_now
            best_n = best_n
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 4: Try different n values (20%)
    t4_total = min(remaining * 0.2, 120)
    if t4_total > 10:
        n_candidates = []
        for dn in [1, 2, 3, 5, 7, 10, 15]:
            if best_n - dn >= 15:
                n_candidates.append(best_n - dn)
            if best_n + dn <= 150:
                n_candidates.append(best_n + dn)
        # Also try some fixed small values
        for nv in [31, 47, 63, 79]:
            if nv >= 15 and nv != best_n:
                n_candidates.append(nv)
        n_candidates = sorted(set(n_candidates))[:8]
        
        t_per_n = t4_total / max(len(n_candidates), 1)
        for nc in n_candidates:
            if time.time() - t_start > budget_s - reserve:
                break
            h_t, c_t = try_n_value(best_h, best_n, nc, min(t_per_n, remaining * 0.3), rng)
            if h_t is not None and c_t < best_c:
                best_c = c_t
                best_h = h_t
                best_n = nc
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 5: Nelder-Mead polish (15%)
    t5 = min(remaining * 0.15, 60)
    if t5 > 5:
        m_best = best_n // 2
        h_sym_best = sym_h(best_h)
        h_sym_best = project_box(h_sym_best, best_n / 2.0)
        a_best = h_sym_best[:m_best].copy()
        
        def obj_a.av(a_vec):
            return compute_c5(h_from_a(a_vec, best_n))
        
        try:
            res = minimize(obj_a.av, a_best, method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-11,
                                    'adaptive': True})
            if res.fun < best_c:
                best_c = res.fun
                best_h = h_from_a(res.x, best_n)
                best_n = best_n
        except Exception:
            pass
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - reserve
    
    # Phase 6: Final micro-refinement (remaining)
    t6 = min(remaining, 30)
    if t6 > 3:
        n_cur = len(best_h)
        m_cur = n_cur // 2
        h_sym_fin = sym_h(best_h)
        h_sym_fin = project_box(h_sym_fin, n_cur / 2.0)
        a_fin = h_sym_fin[:m_cur].copy()
        a_fin, c_fin = optimize_symmetric(a_fin, n_cur, t6, rng)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_from_a(a_fin, n_cur)
            best_n = n_cur
    
    # Final validation
    best_h = np.clip(np.asarray(best_h, dtype=float).ravel(), 0.0, 1.0)
    best_h = project_box(best_h, len(best_h) / 2.0)
    final_c = compute_c5(best_h)
    
    if not np.isfinite(final_c) or final_c < 0:
        # Fallback to constant
        best_h = np.full(n_init, 0.5)
        final_c = compute_c5(best_h)
    
    return (best_h, float(final_c), len(best_h))
