import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_C(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return max(c) * dx

def project_feasible(u, n):
    """Water-filling projection onto {x: 0<=x_i<=1, sum x_i = n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo = -max(u) - 2.0
    hi = 1.0 - min(u) + 2.0
    # Actually: we need sum clip(u_i + lambda, 0, 1) = target
    # At lambda = lo: all clip = 0, sum = 0 < target
    # At lambda = hi: all clip = 1, sum = n > target
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(u + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(u + (lo + hi) / 2.0, 0, 1)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual to coordinates with most slack
        if residual > 0:
            # Add residual to coordinates where x < 1
            idx = np.where(x < 1.0 - 1e-12)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                x[idx] = np.minimum(x[idx] + per, 1.0)
        else:
            idx = np.where(x > 1e-12)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                x[idx] = np.maximum(x[idx] + per, 0.0)
    return x

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))

def feasible_from_logit(y, n):
    """Convert unconstrained y to feasible h with sum = n/2"""
    target = n / 2.0
    # Solve sum sigmoid(y_i + lambda) = target for lambda
    lo = -np.max(y) - 60.0
    hi = -np.min(y) + 60.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = sigmoid(y + lam)
    # Small correction
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        h = project_feasible(h, n)
    return h, lam

def logit_from_h(h):
    eps = 1e-9
    h_clip = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clip / (1.0 - h_clip))

def directional_values(h):
    """Compute all 2n-1 directional values"""
    n = len(h)
    # forward[m] = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) for m=0..n-1
    # backward[m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i]) for m=1..n-1
    vals = np.empty(2 * n - 1)
    vals[0] = np.sum(h * (1.0 - h))  # m=0
    for m in range(1, n):
        vals[2*m - 1] = np.sum(h[:n-m] * (1.0 - h[m:]))  # forward[m]
        vals[2*m] = np.sum(h[m:] * (1.0 - h[:n-m]))  # backward[m]
    return vals

def smooth_objective_and_grad(h, beta):
    """Compute smooth surrogate F_beta and its gradient G = dF/dh"""
    vals = directional_values(h)
    v_max = np.max(vals)
    exp_vals = np.exp(beta * (vals - v_max))
    weights = exp_vals / np.sum(exp_vals)
    F = v_max + np.log(np.sum(exp_vals)) / beta
    
    # Gradient G = sum_v w_v * (d v / dh)
    n = len(h)
    G = np.zeros(n)
    # m=0: v = sum h_i(1-h_i), dv/dh_i = 1-2h_i
    G += weights[0] * (1.0 - 2.0 * h)
    
    for m in range(1, n):
        nm = n - m
        # forward[m]: sum_{i=0}^{nm-1} h[i]*(1-h[i+m])
        # d/dh[i] = (1-h[i+m]) for i in [0,nm-1]
        # d/dh[i+m] = -h[i] for i in [0,nm-1]
        G[:nm] += weights[2*m-1] * (1.0 - h[m:])
        G[m:] += weights[2*m-1] * (-h[:nm])
        
        # backward[m]: sum_{i=0}^{nm-1} h[i+m]*(1-h[i])
        # d/dh[i] = -h[i+m] for i in [0,nm-1]
        # d/dh[i+m] = (1-h[i]) for i in [0,nm-1]
        G[:nm] += weights[2*m] * (-h[m:])
        G[m:] += weights[2*m] * (1.0 - h[:nm])
    
    # Scale: the actual objective is max(vals) * dx, but for gradient we need
    # d(max(vals)*dx)/dh = dx * d(max(vals))/dh
    # Since we're using the smooth surrogate of max(vals), the gradient of the
    # smooth surrogate of max(vals)*dx is dx * G. But since we're minimizing
    # F_beta which approximates max(vals), and the actual objective is max(vals)*dx,
    # the gradient of the actual smooth objective is dx * G.
    # However, since dx is a positive constant, minimizing F_beta is equivalent
    # to minimizing F_beta * dx, and the gradient direction is the same.
    # For BFGS, we can use G as-is (constant factor doesn't affect direction).
    # Actually, we should include dx for correct scaling in BFGS.
    dx = 2.0 / n
    G *= dx
    F *= dx
    
    return F, G

def transformed_objective(y, beta, n):
    """Objective in logit space"""
    h, lam = feasible_from_logit(y, n)
    F, G = smooth_objective_and_grad(h, beta)
    
    # Chain rule: df/dy_j = h_j(1-h_j) * (G_j - B)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-12:
        grad = np.zeros(n)
    else:
        B = np.dot(G, m) / M
        grad = m * (G - B)
    
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
        F = 1e6
    
    return F, grad

def pair_polish(h, max_iters=20, time_limit=30.0):
    """Local pair-transfer optimization"""
    global _BEST_H, _BEST_C
    h = h.copy()
    n = len(h)
    c_cur = exact_C(h)
    start_time = time.time()
    rng = np.random.default_rng(12345)
    
    stagnation = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if time.time() > _DEADLINE - 30:
            break
        
        # Choose a pair
        # Use gradient-based selection
        _, G = smooth_objective_and_grad(h, 200.0)
        # Top coordinates by |G|
        top_idx = np.argsort(-np.abs(G))[:min(12, n)]
        a = rng.choice(top_idx)
        b = rng.choice(top_idx)
        if a == b:
            b = (a + 1) % n
        
        # Find valid range for transfer
        # h[a] + t in [0,1], h[b] - t in [0,1]
        lo = max(0.0, h[b] - 1.0)  # h[b] - t >= 0 => t <= h[b]; h[a] + t <= 1 => t <= 1 - h[a]
        # Actually: h[a] + t <= 1 => t <= 1 - h[a]
        # h[a] + t >= 0 => t >= -h[a]
        # h[b] - t <= 1 => t >= h[b] - 1
        # h[b] - t >= 0 => t <= h[b]
        lo = max(-h[a], h[b] - 1.0)
        hi = min(1.0 - h[a], h[b])
        
        if hi - lo < 1e-12:
            continue
        
        # Grid search + refine
        n_grid = 15
        t_vals = np.linspace(lo, hi, n_grid)
        best_t = 0.0
        best_c = c_cur
        
        for t in t_vals:
            h_try = h.copy()
            h_try[a] += t
            h_try[b] -= t
            c_try = exact_C(h_try)
            if c_try < best_c:
                best_c = c_try
                best_t = t
        
        # Refine with bounded optimization
        if best_t > lo + 1e-10 and best_t < hi - 1e-10:
            # Find bracket
            for i in range(len(t_vals) - 1):
                if abs(t_vals[i] - best_t) < (t_vals[1] - t_vals[0]) * 0.6:
                    left = max(lo, t_vals[max(0, i-1)])
                    right = min(hi, t_vals[min(len(t_vals)-1, i+1)])
                    break
            else:
                left, right = lo, hi
            
            def obj(t):
                h_try = h.copy()
                h_try[a] += t
                h_try[b] -= t
                return exact_C(h_try)
            
            try:
                res = minimize_scalar(obj, bounds=(left, right), method='bounded',
                                     options={'xatol': 1e-10, 'maxiter': 30})
                if res.success and res.fun < best_c:
                    best_c = res.fun
                    best_t = res.x
            except Exception:
                pass
        
        if best_c < c_cur - 1e-14:
            h[a] += best_t
            h[b] -= best_t
            c_cur = best_c
            stagnation = 0
            record_candidate(h)
        else:
            stagnation += 1
            if stagnation > 5:
                break
    
    record_candidate(h)
    return h

        h_try = h.copy()
        h_try[i] += t
        h_try[j] -= t
        c_try = exact_C(h_try)
        if c_try < c_cur or rng.random() < np.exp(-(c_try - c_cur) / max(temperature, 1e-10)):
            h = h_try
            c_cur = c_try
            record_candidate(h)
    record_candidate(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + 0.95 * budget_s
    _STATE['deadline'] = deadline

    # Get initial values
    n_default = 42
    h_init = None
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        if h0.ndim == 1 and len(h0) > 10:
            n_default = len(h0)
            h0 = np.clip(h0, 0, 1)
            h0 = project_feasible(h0, n_default)
            h_init = h0
    except (NameError, Exception):
        pass

    if h_init is None:
        h_init = np.full(n_default, 0.5)

    _init_state(deadline, len(h_init))
    _STATE['best_h'] = h_init.copy()
    _STATE['best_c'] = exact_C(h_init)
    n = len(h_init)

    # Generate diverse starting points
    starts = [h_init.copy()]
    starts.append(h_init[::-1].copy())
    for s in [1, 2, 3]:
        starts.append(np.roll(h_init, s).copy())
    x = np.arange(n) / n
    for freq in [1.0, 2.0, 3.0, 4.0]:
        for phase in [0.0, np.pi / 3]:
            pert = 0.06 * np.sin(2 * np.pi * freq * x + phase)
            h_pert = h_init + pert
            h_pert = project_feasible(h_pert, n)
            starts.append(h_pert)
    rng = np.random.default_rng(seed)
    for _ in range(4):
        h_rand = h_init + 0.15 * rng.standard_normal(n)
        h_rand = project_feasible(h_rand, n)
        starts.append(h_rand)

    # Evaluate and sort starts
    scored_starts = []
    for s in starts:
        c_s = exact_C(s)
        if np.isfinite(c_s):
            scored_starts.append((c_s, s))
    scored_starts.sort(key=lambda t: t[0])

    # Phase 1: BFGS from top starts
    betas = [80.0, 200.0, 500.0]
    max_iter_bfgs = 100 if n <= 50 else 70
    n_starts_to_try = min(len(scored_starts), 8)

    for si in range(n_starts_to_try):
        if time.time() > deadline - 80:
            break
        _, h_start = scored_starts[si]
        y0 = logit_from_h(h_start)
        y0 = np.clip(y0, -25, 25)

        for beta in betas:
            if time.time() > deadline - 80:
                break
            _STATE['n_opt'] = n
            _STATE['beta'] = beta
            try:
                res = minimize(
                    _obj_func,
                    y0,
                    jac=_grad_func,
                    method='BFGS',
                    options={'maxiter': max_iter_bfgs, 'gtol': 5e-5, 'disp': False}
                )
                h_res = feasible_from_logit(res.x, n)
                h_res = project_feasible(h_res, n)
                record_candidate(h_res)
                y0 = logit_from_h(h_res)
                y0 = np.clip(y0, -25, 25)
            except Exception:
                pass

        # Pair polish after BFGS for this start
        if time.time() < deadline - 60:
            h_pol = pair_polish(h_res, max_iters=12, time_limit=12.0)
            record_candidate(h_pol)

    # Phase 2: Intensive pair polish on current best
    if time.time() < deadline - 40:
        h_best = _STATE['best_h'].copy()
        for _ in range(3):
            if time.time() > deadline - 40:
                break
            h_best = pair_polish(h_best, max_iters=25, time_limit=25.0)
            record_candidate(h_best)

    # Phase 3: Random walk exploration
    if time.time() < deadline - 30:
        h_rw = _STATE['best_h'].copy()
        h_rw = random_walk_polish(h_rw, n_steps=300, time_limit=30.0)
        record_candidate(h_rw)

    # Phase 4: Try different dimensions
    if time.time() < deadline - 50:
        for n_try in [n, n - 6, n + 6, n - 12, n + 12, n - 18, n + 18]:
            if n_try < 16 or n_try > 80 or n_try == n:
                continue
            if time.time() > deadline - 50:
                break
            x_old = (np.arange(n) + 0.5) / n
            x_new = (np.arange(n_try) + 0.5) / n_try
            h_interp = np.interp(x_new, x_old, _STATE['best_h'])
            h_interp = project_feasible(h_interp, n_try)
            record_candidate(h_interp)

            # Optimize in new dimension
            if time.time() > deadline - 50:
                break
            for beta in [150.0, 400.0]:
                if time.time() > deadline - 50:
                    break
                _STATE['n_opt'] = n_try
                _STATE['beta'] = beta
                y0 = logit_from_h(h_interp)
                y0 = np.clip(y0, -25, 25)
                try:
                    res = minimize(
                        _obj_func,
                        y0,
                        jac=_grad_func,
                        method='BFGS',
                        options={'maxiter': 70, 'gtol': 5e-5, 'disp': False}
                    )
                    h_res = feasible_from_logit(res.x, n_try)
                    h_res = project_feasible(h_res, n_try)
                    record_candidate(h_res)
                    h_interp = h_res
                except Exception:
                    pass
            if time.time() < deadline - 50:
                h_interp = pair_polish(h_interp, max_iters=10, time_limit=10.0)
                record_candidate(h_interp)

    # Phase 5: Final polish
    if time.time() < deadline - 15:
        h_final = _STATE['best_h'].copy()
        h_final = pair_polish(h_final, max_iters=20, time_limit=15.0)
        record_candidate(h_final)

    # Final validation
    h_out = _STATE['best_h'].copy()
    h_out = np.clip(h_out, 0, 1)
    h_out = project_feasible(h_out, len(h_out))
    c_out = exact_C(h_out)
    n_out = len(h_out)

    if not np.isfinite(c_out):
        h_out = np.full(n_default, 0.5)
        c_out = exact_C(h_out)
        n_out = n_default

    return h_out, c_out, n_out
