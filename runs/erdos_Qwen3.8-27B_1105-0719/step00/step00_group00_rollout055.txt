import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    """Returns the full correlation array (without dx scaling)."""
    return np.correlate(h, 1.0 - h, mode="full")

def greedy_local_search_binary(h, max_sweeps=20000, deadline=None):
    """
    Greedy local search for binary h. Uses vectorized delta computation.
    Returns (h, c5_value).
    """
    n = len(h)
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    c = compute_corr_full(h)
    current_max = c.max()
    
    # Precompute index arrays for F computation
    idx_p = np.arange(n)[:, None] + lags[None, :]  # (n, 2n-1)
    idx_m = np.arange(n)[:, None] - lags[None, :]
    valid_p = (idx_p >= 0) & (idx_p < n)
    valid_m = (idx_m >= 0) & (idx_m < n)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Compute F for all positions
        # F[p] = h[p+lags]*[valid] - (1-h[p-lags])*[valid]
        h_clipped = h[idx_p.clip(0, n-1)]  # (n, 2n-1)
        h_m_clipped = h[idx_m.clip(0, n-1)]
        F = np.where(valid_p, h_clipped, 0.0) - np.where(valid_m, 1.0 - h_m_clipped, 0.0)
        
        # F shape: (n, 2n-1)
        F_ones = F[ones_idx]  # (n_ones, 2n-1)
        F_zeros = F[zeros_idx]  # (n_zeros, 2n-1)
        
        # For each pair (i, j): new_c = c + F[i] - F[j], new_max = new_c.max()
        # Vectorized: M[a, b] = max(c + F_ones[a] - F_zeros[b])
        # M shape: (n_ones, n_zeros)
        # Compute: c[None, None, :] + F_ones[:, None, :] - F_zeros[None, :, :]
        # Then max over last axis
        
        # For memory efficiency with large n, process in batches
        batch_size = max(1, 5000000 // (n_zeros * n_lags))
        all_maxes = np.full((len(ones_idx), len(zeros_idx)), current_max + 1.0)
        
        for start in range(0, len(ones_idx), batch_size):
            end = min(start + batch_size, len(ones_idx))
            M = c[None, None, :] + F_ones[start:end, None, :] - F_zeros[None, :, :]
            all_maxes[start:end, :] = M.max(axis=2)
        
        flat_idx = all_maxes.argmin()
        best_ia, best_jb = np.unravel_index(flat_idx, all_maxes.shape)
        best_new_max = all_maxes[best_ia, best_jb]
        
        if best_new_max >= current_max - 1e-15:
            break
        
        i = ones_idx[best_ia]
        j = zeros_idx[best_jb]
        h[i] = 0.0
        h[j] = 1.0
        c = c + F[i] - F[j]
        current_max = best_new_max
    
    return h, current_max * (2.0 / n)

def simulated_annealing_binary(h, n_iters=100000, T_start=0.05, T_end=0.005, 
                                deadline=None, rng=None):
    """Simulated annealing with swap moves for binary h."""
    if rng is None:
        rng = np.random.RandomState()
    
    n = len(h)
    h_cur = h.copy()
    c_cur = compute_corr_full(h_cur)
    max_cur = c_cur.max()
    
    best_h = h_cur.copy()
    best_max = max_cur
    
    # Precompute for delta
    lags = np.arange(-(n-1), n)
    idx_p = np.arange(n)[:, None] + lags[None, :]
    idx_m = np.arange(n)[:, None] - lags[None, :]
    valid_p = (idx_p >= 0) & (idx_p < n)
    valid_m = (idx_m >= 0) & (idx_m < n)
    
    ones_set = set(np.where(h_cur == 1)[0].tolist())
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / max(n_iters, 1))
        
        # Random swap
        if len(ones_set) == 0:
            break
        i = rng.choice(list(ones_set))  # This is slow; optimize later
        j_candidates = np.where(h_cur == 0)[0]
        j = j_candidates[rng.randint(len(j_candidates))]
        
        # Compute delta
        F_i = np.where(valid_p[i], h_cur[idx_p[i].clip(0, n-1)], 0.0) - \
              np.where(valid_m[i], 1.0 - h_cur[idx_m[i].clip(0, n-1)], 0.0)
        F_j = np.where(valid_p[j], h_cur[idx_p[j].clip(0, n-1)], 0.0) - \
              np.where(valid_m[j], 1.0 - h_cur[idx_m[j].clip(0, n-1)], 0.0)
        
        new_c = c_cur + F_i - F_j
        new_max = new_c.max()
        delta = new_max - max_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            c_cur = new_c
            max_cur = new_max
            ones_set.remove(i)

import time

def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(np.max(corr) * dx)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _project_feasible(y, target):
    """Project y onto {h: 0<=h[i]<=1, sum(h)=target} using bisection."""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual
    res = target - h.sum()
    if abs(res) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += res / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _greedy_binary(h, max_sweeps=50000, deadline=None):
    """Greedy local search for binary h. Returns (h, c5)."""
    n = len(h)
    n_lags = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    
    c = _corr_full(h)
    current_max = c.max()
    
    # Precompute valid index masks
    pp = np.arange(n)[:, None] + lags[None, :]
    pm = np.arange(n)[:, None] - lags[None, :]
    vp = (pp >= 0) & (pp < n)
    vm = (pm >= 0) & (pm < n)
    pp_c = np.clip(pp, 0, n - 1)
    pm_c = np.clip(pm, 0, n - 1)
    
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        
        # F[p] = h[p+lag]*[valid] - (1-h[p-lag])*[valid]
        F = np.where(vp, h[pp_c], 0.0) - np.where(vm, 1.0 - h[pm_c], 0.0)
        
        Fo = F[oi]  # (n_ones, 2n-1)
        Fz = F[zi]  # (n_zeros, 2n-1)
        
        # M[a,b] = max(c + Fo[a] - Fz[b])

            allmax[s:e, :] = M.max(axis=2)

        h[zi[jb]] = 1.0
        c = c + F[oi[ia]] - F[zi[jb]]

def _continuous_refine(h, n_iters=2000, deadline=None):
    """Refine a solution in continuous space using subgradient method."""
    n = len(h)
    lags_arr = np.arange(-(n - 1), n)
    target = n / 2.0
    
    h = h.copy().astype(float)
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        c = _corr_full(h)
        active = np.where(c == c.max())[0]
        
        # Subgradient of c[k*] w.r.t. h[p]
        # ∂c[k*]/∂h[p] = (1-h[p-k*])*[valid] - h[p+k*]*[valid]
        # For the active lags, average the subgradients
        k_star = lags_arr[active]  # (n_active,)
        
        # Compute subgradient for each active lag, average them
        # For lag k: grad[p] = (1-h[p-k])*[0<=p-k<n] - h[p+k]*[0<=p+k<n]
        # Vectorized:
        # For each active lag k:
        #   grad_k[p] = (1 - h[p-k]) if 0<=p-k<n else 0, minus h[p+k] if 0<=p+k<n else 0
        
        subgrad = np.zeros(n)
        for k in k_star:
            # h[p+k] for valid p: p in [max(0,-k), min(n-1,n-1-k)]
            p_idx = np.arange(n)
            pk = p_idx + k
            pm = p_idx - k
            # (1-h[p-k]) for valid p-k
            mask_pm = (pm >= 0) & (pm < n)
            subgrad[mask_pm] += (1.0 - h[pm[mask_pm]])
            # -h[p+k] for valid p+k
            mask_pk = (pk >= 0) & (pk < n)
            subgrad[mask_pk] -= h[pk[mask_pk]]
        
        subgrad /= len(k_star)
        
        # Step size (simple decrease)
        alpha = 5.0 / (it + 10)
        
        # Gradient step
        h_new = h - alpha * subgrad
        
        # Project onto feasible set
        h_new = _project_feasible(h_new, target)

            h_new = _project_feasible(h_new, target)
