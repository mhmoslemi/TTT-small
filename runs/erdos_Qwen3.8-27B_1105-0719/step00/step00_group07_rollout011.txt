    s_lo = np.sum(_sigmoid(theta + lo))
    s_hi = np.sum(_sigmoid(theta + hi))
    if target_sum <= s_lo:
        return _sigmoid(theta + lo)
    if target_sum >= s_hi:
        return _sigmoid(theta + hi)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(_sigmoid(theta + mid)) > target_sum:
            hi = mid
        else:
            lo = mid
    return _sigmoid(theta + 0.5 * (lo + hi))

def _h_to_theta(h):
    eps = 1e-6
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))

def _get_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _lse_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = beta * (corr - M)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    total = np.sum(exp_s)
    lse_val = (M + (max_s + np.log(total)) / beta) * dx
    w = exp_s / total
    grad_h = np.zeros(n)
    for k in range(-n + 1, n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        if k >= 0:
            m1 = np.arange(k, n)
            grad_h[m1] += wk * (1.0 - h[m1 - k])
            m2 = np.arange(0, n - k)
            grad_h[m2] -= wk * h[m2 + k]
        else:
            q = -k
            m1 = np.arange(0, n - q)
            grad_h[m1] += wk * (1.0 - h[m1 + q])
            m2 = np.arange(q, n)
            grad_h[m2] -= wk * h[m2 - q]
    return lse_val, grad_h * dx

def _obj_func(theta):
    h = _theta_to_h(theta, _OPT['target_sum'])
    val, grad_h = _lse_val_grad(h, _OPT['beta'])
    s = h * (1.0 - h)
    S = s.sum()
    if S < 1e-15:
        return val, np.zeros_like(theta)
    ghs = np.dot(grad_h, s)
    grad_theta = s * (grad_h - ghs / S)
    return val, grad_theta

def _fix_mass(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argmax(1.0 - h)
        h[idx] = max(0.0, h[idx] - diff)
    else:
        idx = np.argmin(h)
        h[idx] = min(1.0, h[idx] - diff)
    return np.clip(h, 0.0, 1.0)

def _project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for _ in range(n):
            order = np.argsort(1.0 - h)
            idx = order[-1]
            slk = 1.0 - h[idx]
            rem = min(diff, slk)
            h[idx] -= rem
            diff -= rem
            if diff < 1e-14:
                break
    else:
        for _ in range(n):
            order = np.argsort(h)
            idx = order[-1]
            slk = h[idx]
            rem = min(-diff, slk)
            h[idx] -= rem
            diff += rem
            if diff > -1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.ndim == 1 and init.size > 0 and np.all(np.isfinite(init)):
            h = _project_feasible(init.copy())
            best_h = h
            best_c5 = _get_c5(h)
            best_n = len(h)
    except Exception:
        pass

    if best_h is None:
        n = 95
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = 0.5 + 0.3 * np.cos(np.pi * x)
        h = _project_feasible(h)
        best_h = h
        best_c5 = _get_c5(h)
        best_n = n

    def _remaining():
        return budget_s - (time.time() - t0)

    def _run_lbfgs(theta, n_pts, target_sum, beta, maxiter=300):
        _OPT['target_sum'] = target_sum
        _OPT['beta'] = beta
        bounds = [(-20.0, 20.0)] * n_pts
        try:
            res = minimize(_obj_func, theta, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': maxiter * 3})
            return res.x
        except Exception:
            return theta

    def _try_update(h_cand):
        nonlocal best_h, best_c5, best_n
        if h_cand is None:
            return False
        c = _get_c5(h_cand)
        if np.isfinite(c) and c < best_c5 - 1e-12:
            best_c5 = c
            best_h = h_cand.copy()
            best_n = len(h_cand)
            return True
        return False

    # Phase 1: Progressive LSE optimization
    theta = _h_to_theta(best_h)
    n_cur = best_n
    target_sum = n_cur / 2.0

    betas = [6, 12, 24, 48, 96, 192, 384, 768, 1536, 3072]
    for beta in betas:
        if _remaining() < 8:
            break
        maxiter = 400 if beta <= 64 else 250
        theta = _run_lbfgs(theta, n_cur, target_sum, beta, maxiter)
        h_new = _theta_to_h(theta, target_sum)
        _try_update(h_new)

    # Phase 2: Coordinate descent
    if _remaining() > 12:
        theta = _h_to_theta(best_h)
        target_sum = best_n / 2.0
        n_cur = best_n
        sizes = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003]
        improved = True
        max_sweeps = 8
        for _sweep in range(max_sweeps):
            if _remaining() < 8:
                break
            improved = False
            for coord in range(n_cur):
                if _remaining() < 3:
                    break
                for size in sizes:
                    for sign in (1, -1):
                        nt = theta.copy()
                        nt[coord] = float(np.clip(nt[coord] + sign * size, -20, 20))
                        ht = _theta_to_h(nt, target_sum)
                        ct = _get_c5(ht)
                        if np.isfinite(ct) and ct < best_c5 - 1e-12:
                            best_c5 = ct
                            best_h = ht.copy()
                            theta = nt.copy()
                            improved = True
                            break
                    else:
                        continue
                    break

    # Phase 3: Random perturbations (simulated annealing style)
    if _remaining() > 15:
        n_cur = best_n
        target_sum = n_cur / 2.0
        theta = _h_to_theta(best_h)
        n_attempts = 3000
        for i in range(n_attempts):
            if _remaining() < 5:
                break
            frac = i / max(n_attempts, 1)
            scale = 0.4 * (1.0 - frac) + 0.03
            noise = rng.normal(0.0, scale, size=n_cur)
            nt = np.clip(theta + noise, -20.0, 20.0)
            ht = _theta_to_h(nt, target_sum)
            ct = _get_c5(ht)
            if np.isfinite(ct) and ct < best_c5 - 1e-12:
                best_c5 = ct
                best_h = ht.copy()
                theta = nt.copy()

    # Phase 4: Try complement transformation
    if _remaining() > 15:
        comp = 1.0 - best_h[::-1]
        comp = _project_feasible(comp)
        if _try_update(comp):
            pass

    # Phase 5: Try different n values via interpolation
    if _remaining() > 30:
        for test_n in [55, 71, 87, 103, 119, 135, 151, 167, 183, 199, 215, 231, 247, 263, 279, 295]:
            if _remaining() < 20:
                break
            if test_n == best_n:
                continue
            x_old = np.linspace(0.0, 2.0, best_n)
            x_new = np.linspace(0.0, 2.0, test_n)
            h_interp = np.interp(x_new, x_old, best_h)
            tgt = test_n / 2.0
            h_interp = _project_feasible(h_interp)
            tt = _h_to_theta(h_interp)
            for beta in [32, 128, 512]:
                if _remaining() < 8:
                    break
                tt = _run_lbfgs(tt, test_n, tgt, beta, 200)
                ht = _theta_to_h(tt, tgt)
                ct = _get_c5(ht)
                if np.isfinite(ct) and (ct < best_c5 - 1e-12 or (abs(ct - best_c5) < 1e-10 and test_n < best_n)):
                    best_c5 = ct
                    best_h = ht.copy()
                    best_n = test_n

    # Phase 6: Final fine-tuning coordinate descent
    if _remaining() > 10:
        theta = _h_to_theta(best_h)
        target_sum = best_n / 2.0
        n_cur = best_n
        fine_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for _sweep in range(3):
            if _remaining() < 5:
                break
            for coord in range(n_cur):
                if _remaining() < 2:
                    break
                for size in fine_sizes:
                    for sign in (1, -1):
                        nt = theta.copy()
                        nt[coord] = float(np.clip(nt[coord] + sign * size, -20, 20))
                        ht = _theta_to_h(nt, target_sum)
                        ct = _get_c5(ht)
                        if np.isfinite(ct) and ct < best_c5 - 1e-12:
                            best_c5 = ct
                            best_h = ht.copy()
                            theta = nt.copy()
                            break
                    else:
                        continue
                    break

    # Final validation
    final_h = np.asarray(best_h, dtype=np.float64).copy()
    final_n = len(final_h)
    final_c5 = _get_c5(final_h)

    return (final_h, final_c5, final_n)
