    n = len(h)

    # result[k] = sum_i h[i+k] * (1-h[i]) for valid i
    
    # For each k, find affected i values

    # i.e., i ∈ {p, q, p-k, q-k}
    
    alpha = np.zeros(2*n-1)
    beta = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # Affected i values
        affected = set()
        for i_candidate in [p, q, p-k, q-k]:
            if 0 <= i_candidate < n and 0 <= i_candidate + k < n:
                affected.add(i_candidate)

            # Original term: h[ip] * (1-h[i])
            # New values:
            h_ip_new = h[ip] - (tau if ip == p else 0) + (tau if ip == q else 0)
            h_i_new = h[i] - (tau if i == p else 0) + (tau if i == q else 0)

            # Change = alpha*tau + beta*tau²
            if i in affected and len(affected) > 1:
