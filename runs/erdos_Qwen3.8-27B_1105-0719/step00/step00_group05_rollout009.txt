import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

_BEST_H = None
_BEST_C = float('inf')
_DEADLINE = 0.0

def _exact_C(h):
    n = len(h)
    if n == 0:
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def _is_feasible(h, tol=1e-8):
    n = len(h)
    if n == 0:
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol * n:
        return False
    return True

def _project_feasible(u):
    n = len(u)
    target = n / 2.0
    lo = -1.0 - float(np.max(u))
    hi = 1.0 - float(np.min(u))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(u + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    res = np.clip(u + lam, 0.0, 1.0)
    diff = target - np.sum(res)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - res))
        else:
            idx = int(np.argmin(res))
        res[idx] = np.clip(res[idx] + diff, 0.0, 1.0)
    return res

def _sig(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    et = np.exp(t[~pos])
    out[~pos] = et / (1.0 + et)
    return out

def _logistic_to_feasible(y):
    n = len(y)
    target = n / 2.0
    lo = -float(np.max(y)) - 40.0
    hi = -float(np.min(y)) + 40.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(_sig(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    return _sig(y + 0.5 * (lo + hi))

def _logit_of_h(h):
    eps = 1e-8
    hs = np.clip(h, eps, 1.0 - eps)
    return np.log(hs) - np.log1p(-hs)

def _smooth_obj_grad(y, beta):
    global _DEADLINE

    n = len(h)
    dx = 2.0 / n
    m_i = h * (1.0 - h)
    M = float(np.sum(m_i))
    if M < 1e-14:
        return 1e10, np.zeros(n)
    
    # v_0 and its gradient
    v0 = dx * float(np.sum(m_i))
    G = dx * (1.0 - 2.0 * h)
    vals = [v0]
    
    for k in range(1, n):
        pp = dx * float(np.dot(h[:n-k], 1.0 - h[k:]))
        vals.append(pp)
        G[:n-k] += dx * (1.0 - h[k:])
        G[k:] -= dx * h[:n-k]
        pm = dx * float(np.dot(h[k:], 1.0 - h[:n-k]))
        vals.append(pm)
        G[k:] += dx * (1.0 - h[:n-k])
        G[:n-k] -= dx * h[k:]
    
    vals = np.array(vals)
    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    es = np.exp(shifted)
    w = es / float(np.sum(es))
    
    F = vmax + (1.0 / beta) * float(np.log(np.sum(es)))
    
    # Chain rule
    B = float(np.dot(G, m_i)) / M

    if not np.all(np.isfinite(grad)):
        return F, np.zeros(n)

    return F, grad

def _bfgs(y_init, beta, max_iter=200):
    global _DEADLINE
    if time.time() > _DEADLINE - 25:
        return y_init
    try:
        res = minimize(_smooth_obj_grad, y_init, args=(beta,),
                       method='BFGS', jac=True,
                       options={'maxiter': max_iter, 'gtol': 1e-7})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except:
        pass
    return y_init

def _interp(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = (np.arange(n_old) + 0.5) / n_old
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project_feasible(np.interp(xn, xo, h_old))
