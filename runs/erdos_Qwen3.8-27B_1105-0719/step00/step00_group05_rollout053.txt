def sigmoid(x):
    return 0.5 + 0.5 * np.tanh(0.5 * x)

def solve_lambda(y, n):
    """Find lambda such that sum(sigmoid(y_i + lambda)) = n/2."""
    target = n / 2.0
    def phi(lam):
        return np.sum(sigmoid(y + lam)) - target
    lo = -np.max(y) - 50
    hi = -np.min(y) + 50
    for _ in range(100):
        mid = (lo + hi) / 2
        if phi(mid) > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def logit_to_h(y):
    n = len(y)
    lam = solve_lambda(y, n)
    return sigmoid(y + lam)

def h_to_logit(h, eps=1e-7):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def exact_C(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n < 1:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(u):
    u = np.asarray(u, dtype=float).copy()
    n = len(u)
    target = n / 2.0
    lo = -u.max() - 2.0
    hi = -u.min() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(u + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(u + (lo + hi) / 2.0, 0.0, 1.0)
    delta = target - h.sum()
    if abs(delta) > 1e-13:
        if delta > 0:
            for i in np.argsort(-h):
                step = min(1.0 - h[i], delta)
                if step > 0:
                    h[i] += step
                    delta -= step
                if delta <= 0:
                    break
        else:
            for i in np.argsort(h):
                step = max(-h[i], delta)
                if step < 0:
                    h[i] += step
                    delta -= step
                if delta >= 0:
                    break
    return h


def sigmoid(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def solve_lambda(y, n):
    target = n / 2.0
    lo = -float(np.max(y)) - 60.0
    hi = -float(np.min(y)) + 60.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = float(np.sum(sigmoid(y + mid)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0


def logit_to_h(y):
    n = len(y)
    lam = solve_lambda(y, n)
    return sigmoid(y + lam)


def h_to_logit(h, eps=1e-8):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(c))
    exp_c = np.exp(beta * (c - cmax))
    sum_exp = float(np.sum(exp_c))
    w = exp_c / sum_exp
    obj = (cmax + np.log(sum_exp)) / beta * dx
    grad = np.zeros(n)
    for m in range(-(n - 1), n):
        k = n - 1 + m
        wk = w[k]
        if wk < 1e-16:
            continue
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] += wk * (-h[:n - m])
        else:
            s = -m
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
    grad *= dx
    return obj, grad


def trans_obj_grad(y, beta):
    h = logit_to_h(y)
    obj, gh = smooth_obj_grad(h, beta)
    m = h * (1.0 - h)
    M = float(np.sum(m))
    if M < 1e-14:
        return obj, np.zeros_like(y)
    B = float(np.dot(gh, m)) / M
    gy = m * (gh - B)
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(y)
    return obj, gy


def pair_polish(h, max_iter=200, deadline=None, rng=None):
    h = h.copy()
    n = len(h)
    c_cur = exact_C(h)
    if rng is None:
        rng = np.random.default_rng(0)
    for _ in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        _, gh = smooth_obj_grad(h, 300.0)
        top_idx = np.argsort(-np.abs(gh))[:min(12, n)]
        improved = False
        for i in range(len(top_idx)):
            if deadline is not None and time.time() > deadline:
                break
            a = int(top_idx[i])
            for j in range(len(top_idx)):
                if i == j:
                    continue
                b = int(top_idx[j])
                lo = max(-h[a], h[b] - 1.0)
                hi = min(1.0 - h[a], h[b])
                if hi - lo < 1e-12:
                    continue
                ts = np.linspace(lo, hi, 21)
                best_t = 0.0
                best_val = c_cur
                for t in ts:
                    if abs(t) < 1e-14:
                        continue
                    h[a] = h[a] + t
                    h[b] = h[b] - t
                    if h[a] < -1e-12 or h[a] > 1.0 + 1e-12 or h[b] < -1e-12 or h[b] > 1.0 + 1e-12:
                        h[a] -= t
                        h[b] += t
                        continue
                    val = exact_C(h)
                    if val < best_val:
                        best_val = val
                        best_t = t
                    h[a] -= t
                    h[b] += t
                if best_t != 0.0 and best_val < c_cur - 1e-14:
                    h[a] += best_t
                    h[b] -= best_t
                    c_cur = best_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try random pairs
            for _ in range(5):
                if deadline is not None and time.time() > deadline:
                    break
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
                lo = max(-h[a], h[b] - 1.0)
                hi = min(1.0 - h[a], h[b])

                    continue
                ts = np.linspace(lo, hi, 13)
                best_t = 0.0
                best_val = c_cur
                for t in ts:
                    if abs(t) < 1e-14:
                        continue
                    h[a] += t
                    h[b] -= t
                    if h[a] < -1e-12 or h[a] > 1.0 + 1e-12 or h[b] < -1e-12 or h[b] > 1.0 + 1e-12:
                        h[a] -= t
                        h[b] += t

                    val = exact_C(h)
                    if val < best_val:
                        best_val = val
                        best_t = t
                    h[a] -= t
                    h[b] += t

                    h[a] += best_t
                    h[b] -= best_t
                    c_cur = best_val

def slsqp_smooth(h0, beta, n):
    from scipy.optimize import minimize as scmin
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj_grad(x, beta)[0]

    def jacc(x):
        return smooth_obj_grad(x, beta)[1]

    try:
        res = scmin(obj, h0, method='SLSQP', jac=jacc, bounds=bounds,
                    constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
        return res.x
    except Exception:
        return h0
