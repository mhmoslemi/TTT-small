def solve_primal_lp(h_curr, n):
    """Solve LP: min t s.t. sum_i h_new[i]*b[pair_k(i)]*dx <= t for all k"""
    dx = 2.0 / n
    b = 1.0 - h_curr
    M = 2 * n - 1
    A = np.zeros((M, n + 1))
    for k in range(M):
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                A[k, i] = dx * b[i + lag]
        else:
            for i in range(-lag, n):
                A[k, i] = dx * b[i + lag]
        A[k, n] = -1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c, A_ub=A, b_ub=np.zeros(M), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

        for _ in range(200):

            if s_mid > target:

            else:

        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        for _ in range(200):

            s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s_mid < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def build_lp_matrix(h_fixed, n, optimize_first):
    """Build LP constraint matrix.
    If optimize_first=True: optimizing h_new, h_fixed is b=1-h_curr
      constraint: sum_i h_new[i] * h_fixed[pair_k(i)] * dx <= t
    If optimize_first=False: optimizing b_new, h_fixed is h_curr
      constraint: sum_i h_fixed[i] * b_new[pair_k(i)] * dx <= t
    """
    dx = 2.0 / n
    M = 2 * n - 1
    A = np.zeros((M, n + 1))
    for k in range(M):
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                j = i + lag
                if optimize_first:
                    # h_new[i] * h_fixed[j]
                    A[k, i] = dx * h_fixed[j]
                else:
                    # h_fixed[i] * b_new[j]
                    A[k, j] += dx * h_fixed[i]
        else:
            for i in range(-lag, n):
                j = i + lag
                if optimize_first:
                    # h_new[i] * h_fixed[j]
                    A[k, i] = dx * h_fixed[j]
                else:
                    # h_fixed[i] * b_new[j]
                    A[k, j] += dx * h_fixed[i]
        A[k, n] = -1.0
    return A

def solve_lp(h_fixed, n, optimize_first):
    """Solve the LP subproblem. Returns candidate h_new or None."""
    A = build_lp_matrix(h_fixed, n, optimize_first)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            x = res.x[:n]
            if optimize_first:
                return x
            else:
                return 1.0 - x
    except:
        pass
    return None

def pair_descent(h, n, max_rounds=5, rng=None):
    """Pair coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    current_score = evaluate_c5(h)
    for round in range(max_rounds):
        improved = False
        # Try all pairs (or a subset for speed)
        indices = list(range(n))
        if n > 40:
            rng.shuffle(indices)
        for ii in range(len(indices)):
            for jj in range(ii+1, len(indices)):
                p, q = indices[ii], indices[jj]
                S = h[p] + h[q]
                lo = max(0.0, S - 1.0)
                hi = min(1.0, S)
                if hi - lo < 1e-10:
                    continue
                best_v = h[p]
                best_local = current_score
                # Try a few points + minimize_scalar
                try:
                    res = minimize_scalar(lambda v: _pair_obj(v, h, n, p, q, S), 
                                          bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-8})
                    if res.fun < best_local:
                        best_v = res.x
                        best_local = res.fun
                except:
                    pass
                if best_local < current_score - 1e-12:
                    h[p] = best_v
                    h[q] = S - best_v
                    current_score = best_local
                    improved = True
        if not improved:
            break
    return h, current_score

def minimize_max_linear_vec(A, s, lo, hi, n_iter=60):
    """Vectorized: minimize max_k (A[k] + s[k]*v) over v in [lo, hi]."""
    val_lo = np.max(A + s * lo)
    val_hi = np.max(A + s * hi)
    t_lo = min(val_lo, val_hi)
    t_hi = max(val_lo, val_hi)
    
    for _ in range(n_iter):
        t_mid = (t_lo + t_hi) / 2.0
        # v must satisfy: A[k] + s[k]*v <= t_mid for all k
        v_low = lo
        v_up = hi
        feasible = True
        
        mask_pos = s > 1e-15
        mask_neg = s < -1e-15
        mask_zero = ~mask_pos & ~mask_neg
        
        if np.any(mask_zero) and np.max(A[mask_zero]) > t_mid:
            feasible = False
        elif True:
            if np.any(mask_pos):
                ub = np.min((t_mid - A[mask_pos]) / s[mask_pos])
                v_up = min(v_up, ub)
            if np.any(mask_neg):
                lb = np.max((t_mid - A[mask_neg]) / s[mask_neg])
                v_low = max(v_low, lb)
            if v_low > v_up:
                feasible = False
        
        if feasible:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = (t_lo + t_hi) / 2.0
    v_low = lo
    v_up = hi
    mask_pos = s > 1e-15
    mask_neg = s < -1e-15
    if np.any(mask_pos):
        v_up = min(v_up, np.min((t_opt - A[mask_pos]) / s[mask_pos]))
    if np.any(mask_neg):
        v_low = max(v_low, np.max((t_opt - A[mask_neg]) / s[mask_neg]))
    v_opt = max(lo, min(hi, (v_low + v_up) / 2.0))
    return v_opt, t_opt

def compute_c_vec(h, n, p, q):
    dx = 2.0 / n
    lags = np.arange(2*n-1) - (n-1)
    M = 2*n-1
    c = np.zeros(M)
    
    # h[q-L] term: index = q - lags[k]
    idx = q - lags
    valid = (idx >= 0) & (idx < n)
    c[valid] += h[idx[valid]]
    
    # -h[p-L] term
    idx = p - lags
    valid = (idx >= 0) & (idx < n)
    c[valid] -= h[idx[valid]]
    
    # h[q+L] term
    idx = q + lags
    valid = (idx >= 0) & (idx < n)
    c[valid] += h[idx[valid]]
    
    # -h[p+L] term
    idx = p + lags
    valid = (idx >= 0) & (idx < n)
    c[valid] -= h[idx[valid]]
    
    return dx * c

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        lo_d, hi_d = 0.0, 1.0
        for _ in range(200):
            mid = (lo_d + hi_d) / 2.0
            if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
                lo_d = mid
            else:
                hi_d = mid
        return np.clip(h - (lo_d + hi_d) / 2.0, 0.0, 1.0)
    else:
        lo_d, hi_d = 0.0, 1.0
        for _ in range(200):
            mid = (lo_d + hi_d) / 2.0
            if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
                lo_d = mid
            else:
                hi_d = mid
        return np.clip(h + (lo_d + hi_d) / 2.0, 0.0, 1.0)

def minimize_max_linear_vec(A, s, lo, hi, n_iter=50):
    val_lo = np.max(A + s * lo)
    val_hi = np.max(A + s * hi)
    t_lo = min(val_lo, val_hi)
    t_hi = max(val_lo, val_hi)
    if t_hi - t_lo < 1e-15:
        return (lo + hi) / 2.0, t_hi
    for _ in range(n_iter):
        t_mid = (t_lo + t_hi) / 2.0
        v_low = lo
        v_up = hi
        feasible = True
        mp = s > 1e-14
        mn = s < -1e-14
        mz = ~mp & ~mn
        if np.any(mz):
            if np.max(A[mz]) > t_mid:
                feasible = False
        if feasible:
            if np.any(mp):
                v_up = min(v_up, np.min((t_mid - A[mp]) / s[mp]))
            if np.any(mn):
                v_low = max(v_low, np.max((t_mid - A[mn]) / s[mn]))
            if v_low > v_up:
                feasible = False
        if feasible:
            t_hi = t_mid
        else:
            t_lo = t_mid
    t_opt = (t_lo + t_hi) / 2.0
    v_low = lo
    v_up = hi
    mp = s > 1e-14
    mn = s < -1e-14
    if np.any(mp):
        v_up = min(v_up, np.min((t_opt - A[mp]) / s[mp]))
    if np.any(mn):
        v_low = max(v_low, np.max((t_opt - A[mn]) / s[mn]))
    v_opt = max(lo, min(hi, (v_low + v_up) / 2.0))
    return v_opt, t_opt
