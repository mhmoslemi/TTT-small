    F = np.max(c)

    c = compute_c5_values(x, n)
    F = np.max(c)
    shifted = alpha * (c - F)
    max_shifted = np.max(shifted)
    val = F + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha
    return val

    F = np.max(c)
    shifted = alpha * (c - F)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    dx = 2.0 / n

                if 0 <= i < n:

    dx = 2.0 / n

    return corr * dx

def compute_c5(h, n):
    return np.max(compute_c5_values(h, n))

def project_to_box_sum(x, target):
    """Project x onto {z : 0 <= z_i <= 1, sum(z) = target}."""
    if abs(np.sum(x) - target) < 1e-12 and np.all(x >= -1e-12) and np.all(x <= 1 + 1e-12):
        return np.clip(x, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(free) > 0:
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)

    c = compute_c5_values(x, n)
    F = np.max(c)
    shifted = alpha * (c - F)
    m = np.max(shifted)
    return F + np.log(np.sum(np.exp(shifted - m))) / alpha

    c = compute_c5_values(x, n)
    F = np.max(c)
    shifted = alpha * (c - F)
    m = np.max(shifted)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_values(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def compute_c5(h, n):
    return np.max(compute_c5_values(h, n))


def project_to_box_sum(x, target):
    if abs(np.sum(x) - target) < 1e-13 and np.all(x >= -1e-13) and np.all(x <= 1.0 + 1e-13):
        return np.clip(x, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(150):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = np.sum(free)
        if cnt > 0:
            result[free] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result


def lse_obj(x, n, alpha, target):
    c = compute_c5_values(x, n)
    F = np.max(c)
    shifted = alpha * (c - F)
    m = np.max(shifted)
    return F + np.log(np.sum(np.exp(shifted - m))) / alpha


def lse_grad(x, n, alpha, target):
    c = compute_c5_values(x, n)
    F = np.max(c)
    shifted = alpha * (c - F)
    m = np.max(shifted)
    exp_s = np.exp(shifted - m)
    w = exp_s / np.sum(exp_s)
    dx = 2.0 / n
    nc = len(c)
    k_vals = np.arange(nc) - (n - 1)
    g = np.zeros(n)
    for p in range(n):
        s1 = 0.0
        s2 = 0.0
        for idx in range(nc):
            widx = w[idx]
            if widx < 1e-14:
                continue
            kv = k_vals[idx]
            if kv <= p and p <= n - 1 + kv:
                j = p - kv
                s1 += widx * (1.0 - x[j])
            if -kv <= p and p <= n - 1 - kv:
                i = p + kv
                s2 += widx * x[i]
        g[p] = dx * (s1 - s2)
    return g


def constr_sum(x, n, alpha, target):
    return np.sum(x) - target


def constr_sum_jac(x, n, alpha, target):
    return np.ones(n)


def coord_polish(h, n, time_limit, start_time):

    best = h.copy()
    best_c5 = compute_c5(best, n)

    for rnd in range(300):
        if time.time() - start_time > time_limit - 15:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in deltas:
                    if best[i] >= d and best[j] + d <= 1.0 + 1e-12:
                        best[i] -= d
                        best[j] += d
                        c5_try = compute_c5(best, n)
                        if c5_try < best_c5 - 1e-13:

                            improved = True
                        else:

                            best[i] -= d
