def project_balanced(y):
    """Project y onto [0,1]^n with sum = n/2"""
    n = len(y)
    target = n / 2.0
    # Binary search for lambda such that sum(clip(y + lambda, 0, 1)) = target
    lo = -max(y.max(), 1.0) - 1
    hi = 1.0 - min(y.min(), 0.0) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    x = np.clip(y + lam, 0, 1)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # Find indices with room
        if residual > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        # Distribute residual
        if len(idx) > 0:
            per = residual / len(idx)
            x[idx] += per
            x = np.clip(x, 0, 1)
    # Final fix
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        if len(idx) > 0:
            # Add to one element
            i = idx[0]
            x[i] = np.clip(x[i] + residual, 0, 1)
    return x

    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx
    return scores

def compute_gradient(h, weights):
    """Compute weighted gradient of correlation scores"""
    n = len(h)
    dx = 2.0 / n
    g1 = np.zeros(n)
    g2 = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w == 0:
            continue
        k = r - (n - 1)
        # First term: (1-h[m-k]) for m where 0 <= m-k < n
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            g1[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
        # Second term: h[m+k] for m where 0 <= m+k < n
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            g2[lo2:hi2] += w * h[lo2 + k:hi2 + k]
    return dx * (g1 - g2)

def random_transfer(h, rng):
    """Random small transfer"""
    n = len(h)
    i = int(rng.randint(0, n))
    j = int(rng.randint(0, n))
    if i == j:
        return None
    amount = h[i] * rng.uniform(0.01, 0.3)
    if amount < 1e-8:
        return None
    if h[j] + amount > 1.0 + 1e-12:
        amount = 1.0 - h[j]
        if amount < 1e-8:
            return None
    new_h = h.copy()
    new_h[i] -= amount
    new_h[j] += amount
    return new_h

def resample(h, n_new):
    """Resample h to new resolution"""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # For each new cell, compute the area-weighted average of old cells
    new_h = np.zeros(n_new)
    for i_new in range(n_new):
        x_start = i_new * dx_new
        x_end = (i_new + 1) * dx_new
        # Find overlapping old cells
        i_start_old = int(np.floor(x_start / dx_old))
        i_end_old = int(np.ceil(x_end / dx_old))
        area = 0.0
        total_len = 0.0
        for i_old in range(max(0, i_start_old), min(n_old, i_end_old + 1)):
            old_start = i_old * dx_old
            old_end = (i_old + 1) * dx_old
            overlap_start = max(x_start, old_start)
            overlap_end = min(x_end, old_end)
            if overlap_end > overlap_start:

                total_len += (overlap_end - overlap_start)
        if total_len > 0:
            new_h[i_new] = area / total_len
    return project_balanced(new_h)

∫(h - h²) dx = 1 - ∫h² dx

∫h² dx = ∫h dx = 1, so the k=0 term = 1-1 = 0.

For a binary h with a specific pattern, the k=0 term is 0, and we need to minimize the max over other lags.

import numpy as np
import time

def project_balanced(y):
    n = len(y)
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y + lam, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        count = mask.sum()
        if count > 0:
            x = x.copy()
            x[mask] += residual / count
            x = np.clip(x, 0.0, 1.0)
    return x

def exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx

def compute_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if abs(w) < 1e-15:
            continue
        k = r - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            g[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            g[lo2:hi2] -= w * h[lo2 + k:hi2 + k]
    return g * dx

def active_pair_move(h, scores, rng):
    n = len(h)
    max_s = scores.max()
    if max_s < 1e-15:
        return None
    active = np.where(scores >= 0.90 * max_s)[0]
    if len(active) == 0:
        active = np.where(scores >= 0.80 * max_s)[0]
    if len(active) == 0:
        return None
    r = int(rng.integers(len(active)))
    r = int(active[r])
    k = r - (n - 1)
    if k == 0:
        return None
    if k > 0:
        max_i = n - k
        if max_i <= 0:
            return None
        i = int(rng.integers(max_i))
        amt = min(h[i], 1.0 - h[i + k])
        if amt < 1e-10:
            return None
        amt *= rng.uniform(0.05, 0.4)
        new_h = h.copy()
        new_h[i] -= amt
        new_h[i + k] += amt
        return new_h
    else:
        d = -k
        max_i = n - d
        if max_i <= 0:
            return None
        i = int(rng.integers(max_i))
        j = i + d
        amt = min(h[j], 1.0 - h[i])
        if amt < 1e-10:
            return None
        amt *= rng.uniform(0.05, 0.4)
        new_h = h.copy()
        new_h[j] -= amt
        new_h[i] += amt
        return new_h

def random_transfer(h, rng):
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    amt = h[i] * rng.uniform(0.02, 0.25)
    if amt < 1e-10:
        return None
    if h[j] + amt > 1.0:
        amt = max(0.0, 1.0 - h[j])
        if amt < 1e-10:
            return None
    new_h = h.copy()
    new_h[i] -= amt
    new_h[j] += amt
    return new_h

def pairwise_swap(h, rng):
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    if h[j] == h[i]:
        return None
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def make_initial(n, mode=0, rng=None):
    h = np.zeros(n)
    x = np.arange(n) / n * 2.0
    if mode == 0:
        h[:] = 0.5 + 0.3 * np.cos(np.pi * x)
    elif mode == 1:
        h[:] = 0.5 + 0.35 * np.cos(2.0 * np.pi * x)
    elif mode == 2:
        h[:n // 2] = 1.0
        h[n // 2:] = 0.0
    elif mode == 3:
        h[:] = 0.5 + 0.3 * np.cos(3.0 * np.pi * x)
    elif mode == 4:
        h[:] = 0.5 + 0.25 * np.cos(4.0 * np.pi * x)
    elif mode == 5:
        h[:] = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(2.0 * np.pi * x)
    elif mode == 6:
        # Near-binary random
        if rng is not None:
            h[:] = (rng.random(n) > 0.5).astype(float)
    elif mode == 7:
        # Two-bump
        h[:] = 0.5 + 0.4 * np.cos(2.0 * np.pi * x) * (1.0 - 0.3 * np.cos(4.0 * np.pi * x))
    else:
        h[:] = 0.5
    return project_balanced(h)

def local_optimize(h, rng, time_limit, aggressive=False):
    t_start = time.time()
    n = len(h)
    scores = exact_scores(h)
    c_cur = scores.max()
    c_best = c_cur
    h_best = h.copy()
    T = max(0.003, 0.008 * c_cur)
    T_floor = 0.00005
    stall = 0
    max_stall = 300 if not aggressive else 150

    for _ in range(200000):
        if time.time() - t_start > time_limit:
            break
        if stall > max_stall:
            break

        u = rng.random()
        if u < 0.35:
            max_s = scores.max()
            weights = np.zeros(2 * n - 1)
            threshold = max_s * 0.93
            active = np.where(scores >= threshold)[0]
            if len(active) > 0:
                weights[active] = 1.0 / len(active)
            beta = 5.0
            soft = np.exp(-beta * (max_s - scores) / max(max_s, 1e-10))
            ss = soft.sum()
            if ss > 1e-15:
                weights += 0.25 * soft / ss

            g = compute_gradient(h, weights)
            g_max = np.abs(g).max()
            if g_max < 1e-13:
                T *= 0.97
                stall += 1
                continue

            accepted = False
            for step in [0.4, 0.2, 0.1, 0.05, 0.025, 0.01, 0.005]:
                candidate = project_balanced(h - step * g / g_max)
                ns = exact_scores(candidate)
                nc = ns.max()
                if nc < c_cur - 1e-11:
                    h = candidate
                    scores = ns
                    c_cur = nc
                    if nc < c_best - 1e-11:
                        c_best = nc
                        h_best = candidate.copy()
                        T = max(T, 0.003)
                    accepted = True
                    stall = 0
                    break
                elif nc < c_cur + T * 2 and rng.random() < np.exp(-(nc - c_cur) / max(T, T_floor)):
                    h = candidate
                    scores = ns
                    c_cur = nc
                    if nc < c_best - 1e-11:
                        c_best = nc
                        h_best = candidate.copy()
                    accepted = True
                    break
            if not accepted:
                T *= 0.985
                stall += 1
        elif u < 0.72:
            candidate = active_pair_move(h, scores, rng)
            if candidate is not None:
                ns = exact_scores(candidate)
                nc = ns.max()
                if nc < c_cur - 1e-11 or (nc < c_cur + T and rng.random() < np.exp(-(nc - c_cur) / max(T, T_floor))):
                    h = candidate
                    scores = ns
                    c_cur = nc
                    if nc < c_best - 1e-11:
                        c_best = nc
                        h_best = candidate.copy()
                        stall = 0
                else:
                    stall += 1
                    T = max(T * 0.998, T_floor)
            else:
                stall += 1
        elif u < 0.88:
            candidate = random_transfer(h, rng)
            if candidate is not None:
                ns = exact_scores(candidate)
                nc = ns.max()
                if nc < c_cur - 1e-11 or (nc < c_cur + T and rng.random() < np.exp(-(nc - c_cur) / max(T, T_floor))):
                    h = candidate
                    scores = ns
                    c_cur = nc
                    if nc < c_best - 1e-11:
                        c_best = nc
                        h_best = candidate.copy()
                        stall = 0
                else:
                    T = max(T * 0.999, T_floor)
            else:
                stall += 1
        else:
            candidate = pairwise_swap(h, rng)
            if candidate is not None:
                ns = exact_scores(candidate)
                nc = ns.max()
                if nc < c_cur - 1e-11 or (nc < c_cur + T and rng.random() < np.exp(-(nc - c_cur) / max(T, T_floor))):
                    h = candidate
                    scores = ns
                    c_cur = nc
                    if nc < c_best - 1e-11:
                        c_best = nc
                        h_best = candidate.copy()
                        stall = 0
                else:
                    T = max(T * 0.999, T_floor)
            else:
                stall += 1

        T = max(T, T_floor)

    # Final greedy finishing pass
    if time.time() - t_start < time_limit:
        t_fin = time.time()
        h_fin = h_best.copy()
        s_fin = exact_scores(h_fin)
        c_fin = s_fin.max()
        for _ in range(5000):
            if time.time() - t_fin > max(5, time_limit - (t_start - t_start)):
                break
            if time.time() - t_start > time_limit:
                break
            candidate = active_pair_move(h_fin, s_fin, rng)
            if candidate is None:
                candidate = random_transfer(h_fin, rng)
            if candidate is None:
                break
            ns = exact_scores(candidate)
            nc = ns.max()
            if nc < c_fin - 1e-12:
                h_fin = candidate
                s_fin = ns
                c_fin = nc
            else:
                candidate2 = pairwise_swap(h_fin, rng)
                if candidate2 is not None:
                    ns2 = exact_scores(candidate2)
                    nc2 = ns2.max()
                    if nc2 < c_fin - 1e-12:
                        h_fin = candidate2
                        s_fin = ns2
                        c_fin = nc2
        if c_fin < c_best:
            c_best = c_fin
            h_best = h_fin.copy()

    return h_best, c_best

def resample_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    new_h = np.zeros(n_new)
    for i in range(n_new):
        x0 = i * dx_new
        x1 = (i + 1) * dx_new
        i0 = max(0, int(np.floor(x0 / dx_old)))
        i1 = min(n_old - 1, int(np.ceil(x1 / dx_old)) - 1)
        if i1 < i0:
            new_h[i] = h[max(0, min(n_old - 1, int(x0 / dx_old)))]
            continue
        total_w = 0.0
        val = 0.0
        for j in range(i0, i1 + 1):
            js = j * dx_old
            je = (j + 1) * dx_old
            os_ = max(x0, js)
            oe = min(x1, je)
            if oe > os_:
                w = oe - os_
                val += h[j] * w
                total_w += w
        if total_w > 1e-15:
            new_h[i] = val / total_w
    return project_balanced(new_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Build initial pool
    pool = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        if len(h0) > 0:
            h0 = project_balanced(h0)
            c0 = exact_scores(h0).max()
            pool.append((h0, c0, len(h0)))

    # Add algorithmic starts at the same resolution
    n_start = len(pool[0][0]) if pool else 45
    if 'initial_h_values' not in globals() or initial_h_values is None:
        n_start = 45

    for mode in range(8):
        h_m = make_initial(n_start, mode, rng)
        c_m = exact_scores(h_m).max()
        pool.append((h_m, c_m, n_start))

    # Sort by objective
    pool.sort(key=lambda x: x[1])
    best_h = pool[0][0].copy()
    best_c = pool[0][1]
    best_n = pool[0][2]

    # Phase 1: Optimize top candidates (60% of time)
    n_opt = min(3, len(pool))
    for idx in range(n_opt):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.65:
            break
        remaining = budget_s * 0.65 - elapsed
        if remaining < 10:
            break
        h_cand = pool[idx][0].copy()
        t_alloc = min(remaining * 0.6, 300)
        h_opt, c_opt = local_optimize(h_cand, rng, t_alloc, aggressive=(idx == 0))
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 2: Try resampling to other resolutions (20% of time)
    if time.time() - t0 < budget_s * 0.82:
        remain = budget_s * 0.82 - (time.time() - t0)
        if remain > 20:
            # Higher resolution
            n_hi = min(best_n * 2, 200)
            if n_hi != best_n:
                h_hi = resample_to(best_h, n_hi)
                c_hi = exact_scores(h_hi).max()
                if c_hi < best_c + 0.02 and remain > 30:
                    h_hi_opt, c_hi_opt = local_optimize(h_hi, rng, min(remain * 0.4, 120))
                    if c_hi_opt < best_c:
                        best_h = h_hi_opt
                        best_c = c_hi_opt
                        best_n = n_hi

            # Lower resolution
            if time.time() - t0 < budget_s * 0.85:
                n_lo = max(20, best_n // 2)
                if n_lo != best_n and n_lo >= 20:
                    h_lo = resample_to(best_h, n_lo)
                    c_lo = exact_scores(h_lo).max()
                    if c_lo < best_c + 0.02:
                        t_rem = budget_s * 0.85 - (time.time() - t0)
                        h_lo_opt, c_lo_opt = local_optimize(h_lo, rng, min(t_rem, 60))
                        if c_lo_opt < best_c:
                            best_h = h_lo_opt
                            best_c = c_lo_opt
                            best_n = n_lo

            # Slightly different resolutions
            for n_try in [best_n + 5, best_n + 10, best_n - 5]:
                if time.time() - t0 > budget_s * 0.88:
                    break
                if n_try < 15 or n_try > 250 or n_try == best_n:
                    continue
                h_try = resample_to(best_h, n_try)
                c_try = exact_scores(h_try).max()
                if c_try < best_c:
                    best_h = h_try.copy()
                    best_c = c_try
                    best_n = n_try
                    # Quick optimize
                    t_rem = budget_s * 0.88 - (time.time() - t0)
                    if t_rem > 10:
                        h_try_opt, c_try_opt = local_optimize(h_try, rng, min(t_rem, 60))
                        if c_try_opt < best_c:
                            best_h = h_try_opt
                            best_c = c_try_opt

    # Phase 3: Random restarts with different structures (15% of time)
    if time.time() - t0 < budget_s * 0.92:
        remain = budget_s * 0.92 - (time.time() - t0)
        restarts = 5
        t_per = max(5, remain / restarts)
        for rs in range(restarts):
            if time.time() - t0 > budget_s * 0.95:
                break
            mode = int(rng.integers(10))
            h_r = make_initial(best_n, mode, rng)
            # Perturb
            h_r = project_balanced(h_r + 0.1 * rng.standard_normal(best_n))
            c_r = exact_scores(h_r).max()
            t_alloc = min(t_per, budget_s - (time.time() - t0) - 15)
            if t_alloc < 5:

            h_r_opt, c_r_opt = local_optimize(h_r, rng, t_alloc)

                best_h = h_r_opt
                best_c = c_r_opt

    # Phase 4: Final polishing on best solution
    if time.time() - t0 < budget_s - 10:
        t_rem = budget_s - (time.time() - t0) - 5
        if t_rem > 5:
            h_final, c_final = local_optimize(best_h.copy(), rng, min(t_rem, 60), aggressive=True)
            if c_final < best_c:
                best_h = h_final
                best_c = c_final

    # Final validation
    final_scores = exact_scores(best_h)
    best_c = final_scores.max()
    # Ensure feasibility

        final_scores = exact_scores(best_h)
        best_c = final_scores.max()
    if np.any(best_h < -1e-10) or np.any(best_h > 1 + 1e-10):
        best_h = np.clip(best_h, 0, 1)
        best_h = project_balanced(best_h)
        final_scores = exact_scores(best_h)
        best_c = final_scores.max()

    return (best_h, best_c, len(best_h))
