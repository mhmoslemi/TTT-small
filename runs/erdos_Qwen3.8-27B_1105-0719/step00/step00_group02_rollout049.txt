def evaluate_c5(h):
    n = len(h)
    if n < 1:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(y, target):
    n = len(y)
    # Binary search for lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += r / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj_grad(h, tau):
    """Return (smoothed_value, gradient) of log-sum-exp smoothed C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx  # shape (2n-1,)
    g_max = float(np.max(g))
    exps = np.exp((g - g_max) / tau)
    total = float(exps.sum())
    w = exps / total  # normalized weights
    S = tau * (g_max / tau + np.log(total))  # = g_max + tau*log(total)
    
    # Gradient
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        l = p - (n - 1)  # lag
        wt = w[p] * dx
        # Term 1: (1-h[i-l]) for i in [max(0,l), min(n-1, n-1+l)]
        i1_lo = max(0, l)
        i1_hi = min(n - 1, n - 1 + l)
        if i1_lo <= i1_hi:
            grad[i1_lo:i1_hi+1] += wt * (1.0 - h[i1_lo - l:i1_hi - l + 1])
        # Term 2: -h[i+l] for i in [max(0,-l), min(n-1, n-1-l)]
        i2_lo = max(0, -l)
        i2_hi = min(n - 1, n - 1 - l)
        if i2_lo <= i2_hi:
            grad[i2_lo:i2_hi+1] -= wt * h[i2_lo + l:i2_hi + l + 1]
    
    return S, grad

    n = len(h)

            i = idx // (n - 1)
            j = idx % (n - 1)

            d_max = min(h[i], 1.0 - h[j])

                    h[i] -= d
