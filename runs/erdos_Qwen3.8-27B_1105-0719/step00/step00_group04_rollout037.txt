def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search for theta
    s = np.sort(h)
    # We need sum(clamp(h_i - theta, 0, 1)) = n/2
    # = sum_{h_i > theta} (h_i - theta) + count_{0 < h_i - theta < 1} * 0 + ...
    # Actually clamp(x, 0, 1) = min(max(x, 0), 1)
    # clamp(h_i - theta, 0, 1) = h_i - theta if theta <= h_i <= theta+1
    #                         = 0 if h_i < theta
    #                         = 1 if h_i > theta+1

    lo, hi = h.min() - 1, h.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.clip(h - theta, 0, 1)

import numpy as np
import time

def compute_correlations(h, n):
    """Compute F_k = dx * correlate_k for all k"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_gradient(h, n, tau):
    """Compute gradient of softmax surrogate"""
    dx = 2.0 / n
    F = compute_correlations(h, n)  # shape (2n-1,)
    
    # Softmax weights
    max_F = F.max()
    exp_F = np.exp((F - max_F) / tau)
    w = exp_F / exp_F.sum()  # shape (2n-1,)
    
    # For each k, lag L_k = |k - (n-1)|
    # dF_k/dh[j] = dx * [(1-h[j-L]) * 1_{L<=j} - h[j+L] * 1_{j+L<n}]
    
    # Group by lag L:
    # W_L = sum of w_k for k with |k-(n-1)| = L
    W = np.zeros(n)  # W[0] = w[n-1], W[L] = w[n-1+L] + w[n-1-L] for L>=1
    W[0] = w[n-1]
    for L in range(1, n):
        W[L] = w[n-1+L] + w[n-1-L]
    
    # Now compute gradient
    # dS/dh[j] = dx * sum_{L=0}^{n-1} W_L * [(1-h[j-L]) * 1_{L<=j} - h[j+L] * 1_{j+L<n}]
    
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for L in range(0, min(j, n-1) + 1):
            val += W[L] * (1.0 - h[j-L])
        for L in range(0, min(n-1-j, n-1) + 1):
            val -= W[L] * h[j+L]
        grad[j] = dx * val
    
    return grad

    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection for theta
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.clip(h - mid, 0.0, 1.0)
    # Fix floating point: adjust to get exact sum
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        count = mask.sum()
        if count > 0:

            result = np.clip(result, 0.0, 1.0)

def compute_weights(F, tau):
    """Compute softmax weights"""
    max_F = F.max()
    exp_F = np.exp((F - max_F) / tau)
    return exp_F / exp_F.sum()

def compute_gradient(h, n, tau):
    """Compute gradient of softmax surrogate S_tau(h)"""
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    w = compute_weights(F, tau)
    
    # Build W_L
    W = np.zeros(n)
    W[0] = w[n-1]
    for L in range(1, n):
        W[L] = w[n-1+L] + w[n-1-L]
    
    # Compute gradient using convolution/correlation
    conv = np.convolve(W, h, mode='full')
    corr = np.correlate(h, W, mode='full')
    cumsum_W = np.cumsum(W)
    
    grad = dx * (cumsum_W[:n] - conv[:n] - corr[n-1:2*n-1])
    return grad

So the question becomes: what 0-1 sequence (or near-0-1) of length n with exactly n/2 ones minimizes the maximum over all lags of the number of "1 followed by 0" occurrences at that lag?

This is related to the concept of a "balanced sequence" or "Christoffel word" in combinatorics.

A Christoffel word of length n with k ones is a sequence that is as evenly distributed as possible. It has the property that for every prefix, the number of ones is either floor(k*i/n) or ceil(k*i/n).

For n=70, k=35: the "perfectly balanced" sequence would alternate 1,0,1,0,... but that has period 2, and for lag 2, F_2 = 0 (perfect correlation), but for lag 1, F_1 = dx * 35 = (2/70)*35 = 1. That's terrible!

So alternating is the worst for odd lags. We want a sequence that's well-distributed at ALL lags.

This is essentially the problem of finding a set A ⊂ {0,...,n-1} of size n/2 that minimizes max_L |A ∩ (A+L)^c| * (2/n) (approximately, for small L).

This is related to the "difference set" problem. A (v, k, λ) difference set has the property that every nonzero difference appears exactly λ times. If we could find such a set with v=n, k=n/2, then |A ∩ (A+L)^c| would be constant for all L, giving a uniform bound.

For a difference set: |A ∩ (A+L)| = k - λ for all L ≠ 0 (I need to check the exact parameterization).

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.clip(h - theta, 0.0, 1.0)

def compute_gradient(h, n, tau):
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    max_F = F.max()
    exp_F = np.exp((F - max_F) / tau)
    w = exp_F / exp_F.sum()
    
    W = np.zeros(n)
    W[0] = w[n-1]
    for L in range(1, n):
        W[L] = w[n-1+L] + w[n-1-L]
    
    conv = np.convolve(W, h, mode='full')
    corr = np.correlate(h, W, mode='full')
    cumsum_W = np.cumsum(W)
    
    return dx * (cumsum_W[:n] - conv[:n] - corr[n-1:2*n-1])

def smooth_optimize(h, n, tau_s, tau_e, steps, lr_s, lr_e):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    h_cur = h.copy()
    
    for step in range(steps):
        t = step / max(steps - 1, 1)
        tau = tau_s * (tau_e / tau_s) ** t
        lr = lr_s * (lr_e / lr_s) ** t
        
        grad = compute_gradient(h_cur, n, tau)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_c5

def polish(h, n, max_time=30):

                    if h[j] - sign*delta > 1.0 + 1e-10 or h[j] - sign*delta < -1e-10:
                        continue
                    h[i] += sign * delta
                    h[j] -= sign * delta
                    new_c5 = compute_c5(h, n)
                    if new_c5 < current_c5 - 1e-12:
                        current_c5 = new_c5
                        improved = True
                        break
                    h[i] -= sign * delta
                    h[j] += sign * delta
                if improved:
                    break
            if not improved:
                break
    
    return h, current_c5

The Lipschitz constant of the gradient: The Hessian of F_k is a matrix with entries that are 0 or -dx (for the quadratic terms). So ||∇²F_k|| ≤ O(dx * degree) where degree is the number of terms, which is at most n. So ||∇²F_k|| ≤ O(1) roughly.

For the smoothed surrogate, the Hessian includes the softmax weight variation, which adds terms of order 1/tau. So the Lipschitz constant is roughly O(1 + 1/tau).

A safe step size would be lr ≈ 1 / (Lipschitz) ≈ tau / (1 + tau) ≈ tau for small tau.
