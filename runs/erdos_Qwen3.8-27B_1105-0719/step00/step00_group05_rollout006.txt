import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def valid_check(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return False, float('inf')
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return False, float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False, float('inf')
    return True, exact_objective(h)

def project_feasible(h_raw):
    n = len(h_raw)
    target = n / 2.0
    h = h_raw.astype(np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = h_proj < 1.0 - 1e-15
        else:
            mask = h_proj > 1e-15
        if np.any(mask):
            h_proj[mask] += residual / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_weighted_gradient(h, weights):
    n = len(h)
    g = np.zeros(n)
    w0 = weights[n - 1]
    if abs(w0) > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = weights[n - 1 + m]
        wm = weights[n - 1 - m]
        if abs(wp) < 1e-15 and abs(wm) < 1e-15:
            continue
        if abs(wp) > 1e-15:
            g[m:] += wp * (1.0 - h[:n - m])
            g[:n - m] += wp * (-h[m:])
        if abs(wm) > 1e-15:
            g[:n - m] += wm * (1.0 - h[m:])
            g[m:] += wm * (-h[:n - m])
    return g

def softmax_weights(vals, temperature):
    L = len(vals)
    v = vals.astype(np.float64)
    v_shifted = v - np.max(v)
    exps = np.exp(v_shifted / temperature)
    total = np.sum(exps)
    if total < 1e-30:
        return np.ones(L) / L
    return exps / total

def topk_weights(vals, k, decay):
    L = len(vals)
    order = np.argsort(-vals, kind='stable')
    w = np.zeros(L)
    kk = min(k, L)
    for i in range(kk):
        w[order[i]] = np.exp(-decay * i)
    total = np.sum(w)
    if total < 1e-15:
        return np.ones(L) / L
    return w / total

def projected_step(h, g, alpha):
    g_proj = g - np.mean(g)
    h_try = project_feasible(h - alpha * g_proj)
    return h_try

def sqp_subproblem(h, g, delta, rho):
    n = len(h)
    M = build_hessian_reg(h, delta, rho)
    
    def obj(d):
        return g @ d + 0.5 * (d @ M @ d)
    
    def obj_grad(d):
        return g + M @ d
    
    bounds = [(max(-h[i], -delta), min(1.0 - h[i], delta)) for i in range(n)]
    cons = [{'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)}]
    d0 = np.zeros(n)
    res = minimize(obj, d0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 50, 'ftol': 1e-12})
    if res.success or np.all(np.isfinite(res.x)):
        d = res.x
        dsum = np.sum(d)
        d -= dsum / n
        return d
    return None

def build_hessian_reg(h, delta, rho):
    n = len(h)
    M = np.zeros((n, n))
    for i in range(n):
        M[i, i] = 1.0
    for m in range(1, min(n, 6)):
        w = 1.0 / m
        for i in range(n - m):
            M[i, i] += w
            M[i + m, i + m] += w
            M[i, i + m] -= w
            M[i + m, i] -= w
    M += rho * np.eye(n)
    return M

def random_perturb(h, scale, rng):
    n = len(h)
    noise = rng.standard_normal(n) * scale
    noise -= np.mean(noise)
    return project_feasible(h + noise)

def paired_swap(h, i, j, amount, rng):
    h_new = h.copy()
    h_new[i] = np.clip(h_new[i] - amount, 0.0, 1.0)
    h_new[j] = np.clip(h_new[j] + amount, 0.0, 1.0)
    return h_new

def sqp_optimize(h_start, c_start, budget_s, start_time, seed, best_state):
    rng = np.random.RandomState(seed)
    h = h_start.astype(np.float64).copy()
    c_local = c_start
    n = len(h)
    
    temperatures = [0.3, 0.5, 0.8, 1.2, 2.0]
    k_tops = [8, 12, 16, 20]
    decays = [0.0, 0.3, 0.6]
    alphas_base = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002]
    
    no_improve = 0
    iter_count = 0
    param_idx = 0
    
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        
        iter_count += 1
        
        # Choose parameters
        temp_idx = param_idx % len(temperatures)
        k_idx = (param_idx // len(temperatures)) % len(k_tops)
        d_idx = (param_idx // (len(temperatures) * len(k_tops))) % len(decays)
        param_idx += 1
        
        # Compute weights
        vals = np.correlate(h, 1.0 - h, mode='full')
        if param_idx % 3 == 0:
            weights = softmax_weights(vals, temperatures[temp_idx])
        elif param_idx % 3 == 1:
            weights = topk_weights(vals, k_tops[k_idx], decays[d_idx])
        else:
            w1 = softmax_weights(vals, temperatures[temp_idx])
            w2 = topk_weights(vals, k_tops[k_idx], 0.0)
            weights = 0.5 * w1 + 0.5 * w2
        
        # Compute gradient
        g = compute_weighted_gradient(h, weights)
        g_norm = np.linalg.norm(g)
        
        if g_norm < 1e-12:
            # Gradient nearly zero, try random perturbation
            h_try = random_perturb(h, 0.02, rng)
            c_try = exact_objective(h_try)
            if c_try < c_local:
                h, c_local = h_try, c_try
                no_improve = 0
                record_best(h, c_local, n, best_state)
            else:
                no_improve += 1
                if no_improve > 10:
                    h_try = random_perturb(h, 0.05, rng)
                    c_try = exact_objective(h_try)
                    if c_try < c_start + 5e-4:
                        h, c_local = h_try, c_try
                    no_improve = 0
            continue
        
        # Try SLSQP subproblem
        delta = 0.03
        rho = max(0.01, g_norm / (delta * 10))
        d = sqp_subproblem(h, g, delta, rho)
        found_step = False
        
        if d is not None and np.linalg.norm(d) > 1e-12:
            # Line search with SLSQP direction
            d_scale = np.linalg.norm(d)
            for alpha in alphas_base:
                step = alpha * d / max(1.0, d_scale)
                h_try = h + step
                h_try = project_feasible(h_try)
                c_try = exact_objective(h_try)
                if c_try < c_local - 1e-14:
                    h, c_local = h_try, c_try
                    no_improve = 0
                    found_step = True
                    record_best(h, c_local, n, best_state)
                    break
        
        # If SLSQP didn't help, try projected gradient
        if not found_step:
            g_proj = g - np.mean(g)
            gpn = np.linalg.norm(g_proj)
            if gpn > 1e-12:
                for alpha in alphas_base:
                    h_try = project_feasible(h - alpha * g_proj / gpn)
                    c_try = exact_objective(h_try)
                    if c_try < c_local - 1e-14:
                        h, c_local = h_try, c_try
                        no_improve = 0
                        record_best(h, c_local, n, best_state)
                        break
        
        if not found_step:
            no_improve += 1
            if no_improve > 6:
                # Random escape
                scale = 0.01 + 0.02 * min(no_improve, 5)
                h_try = random_perturb(h, scale, rng)
                c_try = exact_objective(h_try)
                if c_try < c_local:
                    h, c_local = h_try, c_try
                    no_improve = 0
                    record_best(h, c_local, n, best_state)
                elif c_try < c_start + 3e-4:
                    h, c_local = h_try, c_try
                    no_improve = 0
                else:
                    no_improve = 0  # reset to try again
        
        if best_state['done']:
            break
    
    return h, c_local

def record_best(h, c, n, best_state):
    if c < best_state['best_c'] - 1e-14:
        best_state['best_c'] = c
        best_state['best_h'] = h.copy()
        best_state['best_n'] = n
        if c <= best_state['target']:
            best_state['done'] = True
    elif abs(c - best_state['best_c']) <= 1e-14 and n < best_state['best_n']:
        best_state['best_c'] = c
        best_state['best_h'] = h.copy()
        best_state['best_n'] = n
        if c <= best_state['target']:
            best_state['done'] = True

def make_fallback_h(n):
    h = np.full(n, 0.5)
    return h

def resize_interpolate(h, new_n):
    n0 = len(h)
    if n0 == new_n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x1, x0, h)
    return project_feasible(h_new)
