def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to decrease: reduce elements > 0
            mask = h > 1e-10
            count = np.sum(mask)
            if count == 0:
                break
            delta = diff / count
            h[mask] -= delta
        else:
            # Need to increase: increase elements < 1
            mask = h < 1.0 - 1e-10
            count = np.sum(mask)
            if count == 0:
                break
            delta = -diff / count
            h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    h = np.clip(h, 0.0, 1.0)

For n=50: M <= 9.52, so M <= 9.

The "density" of ones is always n/2 / n = 1/2. The question is whether we can find binary sequences with lower max lag values for smaller n.

Actually, the sum constraint is: sum of all result[k] = n^2/4 (for binary with n/2 ones).
With 2n-1 lag values, at least 3 of which are 0:

For n=70: average ≈ 8.75, so max >= 9.
For n=50: average ≈ 6.25, so max >= 7.

The lower bound scales as n/8, and C5 = M * 2/n >= (n/8) * 2/n = 1/4 = 0.25.

So the constant function is bad. We need variation in h to keep the lag values low.

def _binary_sa(b, n, rng, time_budget):
    dx = 2.0 / n
    b = b.copy()
    current = _eval(b, n)
    best = current
    best_b = b.copy()
    
    T = max(0.02, dx * 2)  # temperature in objective units
    T_min = 0.0001
    
    start = time.time()
    stuck_count = 0
    max_stuck = 2000
    
    while (time.time() - start) < time_budget:
        # Pick random pair
        p = rng.integers(n)
        q = rng.integers(n)
        if b[p] == b[q]:
            continue
        
        # Swap
        b[p], b[q] = b[q], b[p]
        new_val = _eval(b, n)
        
        delta = new_val - current
        if delta <= 0 or (rng.random() < np.exp(-delta / T)):
            current = new_val
            if current < best:
                best = current
                best_b = b.copy()
                stuck_count = 0
            T = max(T * 0.9995, T_min)
        else:
            b[p], b[q] = b[q], b[p]  # Undo
            stuck_count += 1
        
        if stuck_count > max_stuck:
            # Restart with perturbation
            for _ in range(20):
                p = rng.integers(n)
                q = rng.integers(n)
                if b[p] != b[q]:
                    b[p], b[q] = b[q], b[p]
            current = _eval(b, n)
            T = max(0.02, dx * 3)
            stuck_count = 0
    
    return best_b, best
