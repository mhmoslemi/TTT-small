def _params_to_h(params, n, target_sum):
    h = np.empty(n)
    h[:n-1] = params
    h[n-1] = target_sum - params.sum()
    if h[n-1] < -1e-8 or h[n-1] > 1.0 + 1e-8:
        return None
    return np.clip(h, 0.0, 1.0)

    h = _params_to_h(params, n, target_sum)
    if h is None:
        return 1e10
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(logsumexp(alpha * corr) / alpha * dx)

def _c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _cd_obj(t, h, p, q, dx):
    ht = h.copy()
    ht[p] += t
    ht[q] -= t
    return float(np.max(np.correlate(ht, 1.0 - ht, mode="full")) * dx)

def _fix_feasibility(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    h = h + (target_sum - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
