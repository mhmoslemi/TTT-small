```python
import numpy as np
import time

def get_now():
    return time.perf_counter()

def evaluate_c5(h):
    """Compute C5 bound for given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def compute_correlations(h):
    """Compute all correlation values."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_box_sum(x, mass):
    """Project x onto {0≤v≤1, sum(v)=mass}."""
    # Bisection on t: find t such that sum(clip(x-t, 0, 1)) = mass
    lo = x.min() - 1.0
    hi = x.max()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > mass:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def fw_vertex(g, mass):
    """Find vertex of K that minimizes g·y.
    K = {y: 0≤y≤1, sum(y)=mass}
    Sort g, assign 1 to smallest entries."""
    n = len(g)
    order = np.argsort(g)
    y = np.zeros(n)
    # mass should be integer for a pure vertex, but handle fractional
    int_mass = int(mass)
    frac = mass - int_mass
    # Assign 1 to the int_mass smallest
    for i in range(int_mass):
        y[order[i]] = 1.0
    if frac > 1e-10:
        y[order[int_mass]] = frac
    return y

def compute_subgradient(h, c, top_k=5, temperature=0.05):
    """Compute a subgradient of C(h) = max_k c_k.
    Uses softmax weighting over the top lags."""
    n = len(h)
    # c has length 2n-1
    # Weight lags by softmax
    weights = np.exp((c - c.max()) / temperature)
    weights = weights / weights.sum()
    
    # For each lag index m (0 to 2n-2), k = m - (n-1)
    # ∂c_k/∂h[j] = (1-h[j-k])*I[0≤j-k≤n-1] - h[j+k]*I[0≤j+k≤n-1]
    
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)
        w = weights[m]
        # First term: (1-h[j-k]) for valid j-k
        # j-k ranges from 0 to n-1, so j ranges from k to n-1+k
        # But j must be in [0, n-1]
        j_start = max(0, k)
        j_end = min(n - 1, n - 1 + k)
        if j_start <= j_end:
            jk = np.arange(j_start, j_end + 1) - k  # j-k values
            g[j_start:j_end+1] += w * (1.0 - h[jk])
        
        # Second term: -h[j+k] for valid j+k
        # j+k ranges from 0 to n-1, so j ranges from -k to n-1-k
        # But j must be in [0, n-1]
        j_start2 = max(0, -k)
        j_end2 = min(n - 1, n - 1 - k)
        if j_start2 <= j_end2:
            jk = np.arange(j_start2, j_end2 + 1) + k  # j+k values
            g[j_start2:j_end2+1] -= w * h[jk]
    
    return g
```

```python
    """Compute C5 = max_k correlate(h, 1-h, 'full')_k * dx."""

    """Project x onto {0<=v<=1, sum(v)=mass} via bisection."""
    lo = float(x.min()) - 1.0
    hi = float(x.max())

        s = np.sum(np.clip(x - mid, 0.0, 1.0))

        else:

            g[idx] += w * (1.0 - h[idx - k])

        if k >= 0:

            idx = np.arange(0, n - k)
            g[idx] -= w * h[idx + k]
        else:

            idx = np.arange(-k, n)
            g[idx] -= w * h[idx + k]

            best_g = lo

            best_g = hi
```

def get_now():
    return time.perf_counter()

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def eval_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(x, mass):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x))
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > mass:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def fw_vertex(g, mass):
    n = len(g)
    order = np.argsort(g)
    y = np.zeros(n)
    im = int(mass)
    fr = mass - im
    if im > 0:
        y[order[:im]] = 1.0
    if fr > 1e-12 and im < n:
        y[order[im]] = fr
    return y

def subgradient(h, cvec, temp):
    n = len(h)
    w = np.exp((cvec - cvec.max()) / temp)
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k == 0:
                g += wm * (1.0 - h)
                g -= wm * h
            elif k < n:
                g[k:] += wm * (1.0 - h[:n-k])
                g[:n-k] -= wm * h[k:]
        else:
            l = -k
            if 0 < l < n:
                g[:n-l] += wm * (1.0 - h[l:])
                g[l:] -= wm * h[:n-l]
    return g

def do_fw_step(h, c_best, mass, temp):
    cvec = eval_full(h)
    g = subgradient(h, cvec, temp)
    s = fw_vertex(g, mass)

    best_t = 0.0

        ht = (1.0 - t) * h + t * s
        c = eval_c5(ht)
        if c < best_c:
            best_c = c
            best_t = t
    if best_t == 0.0:
        return h, c_best
    ht = (1.0 - best_t) * h + best_t * s
    # Refine around best_t
    lo = max(0.0, best_t - 0.1)
    hi = min(1.0, best_t + 0.1)

        m1 = lo + (hi - lo) * 0.33
        m2 = hi - (hi - lo) * 0.33
        c1 = eval_c5((1 - m1) * h + m1 * s)
        c2 = eval_c5((1 - m2) * h + m2 * s)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
        mid = 0.5 * (lo + hi)

        if cm < best_c:
            best_c = cm
            best_t = mid
    ht = (1.0 - best_t) * h + best_t * s
    return ht, best_c

def local_search(h, c_best, mass, time_limit, seed=0):
    n = len(h)
    rng = np.random.default_rng(seed)
    start = get_now()
    best_h = h.copy()
    best_c = c_best
    
    while get_now() - start < time_limit:
        # Random pair swap
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Transfer from i to j (or j to i)
        if h[i] > h[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        delta_max = h[src] - h[dst]
        if delta_max < 1e-8:
            continue
        # Try several transfer amounts
        improved = False
        for frac in [0.5, 1.0]:
            d = delta_max * frac

            ht[dst] += d
            c = eval_c5(ht)
            if c < best_c - 1e-12:
                h = ht
                best_c = c
                improved = True
                break
        if improved:
            best_h = h.copy()
            best_c = c
    
    return best_h, best_c
