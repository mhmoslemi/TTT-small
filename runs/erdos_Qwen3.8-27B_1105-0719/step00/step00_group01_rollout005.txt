    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.argsort(-(1.0 - x))
        else:
            idx = np.argsort(x)
        for i in idx:
            if abs(r) < 1e-14:

                r -= d
            else:

                x[i] -= d
                r += d

def grad_lag(h, d, n, dx):
    grad = np.zeros(n)
    js = max(0, -d)
    je = min(n, n - d)
    if js >= je:
        return grad
    m = np.arange(js, je)
    first = m + d
    grad[first] += dx * (1.0 - h[m])
    grad[m] -= dx * h[first]
    return grad

def smooth_grad(h, F, beta, n, dx):
    M = np.max(F)
    sh = np.clip(beta * (F - M), -500.0, 0.0)
    ev = np.exp(sh)
    w = ev / ev.sum()
    grad = np.zeros(n)

        if w[ell] < 1e-12:
            continue

        grad += w[ell] * grad_lag(h, d, n, dx)
    return grad
