def active_set_step(h, n_active=3, step_size=0.001):
    """Take a small step to reduce the top active constraints."""
    n = len(h)
    vals = np.correlate(h, 1.0 - h, mode='full')
    # Find top indices
    top_idx = np.argsort(vals)[-n_active:]
    # Compute subgradient of the average of top constraints
    g = np.zeros(n)
    for i in top_idx:
        m = i - (n - 1)
        if m >= 0:
            for t in range(n - m):
                g[t + m] += (1.0 - h[t]) / n_active
                g[t] -= h[t + m] / n_active
        else:
            mp = -m
            for t in range(mp, n):
                g[t - mp] += (1.0 - h[t]) / n_active
                g[t] -= h[t - mp] / n_active
    # Project gradient onto feasible direction (sum = 0)
    g = g - np.mean(g)
    # Scale
    scale = step_size / (np.max(np.abs(g)) + 1e-15)
    h_new = h + scale * g
    return project_feasible(h_new)
