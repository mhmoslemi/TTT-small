import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_box_sum(x, target_sum):
    x = np.array(x, dtype=float, copy=True)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            slack = 1.0 - x
            ts = slack.sum()
            if ts > 1e-15:
                x += residual * slack / ts
        else:
            ts = x.sum()
            if ts > 1e-15:
                x += residual * x / ts
        x = np.clip(x, 0.0, 1.0)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmax(x))
        x[idx] += residual
        x[idx] = min(max(x[idx], 0.0), 1.0)
    return x


def compute_soft_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    c = corr_scaled - corr_scaled.max()
    w = np.exp(c / tau)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-15:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            m = -k
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n - m]
    return grad * dx


def smooth_descent(current, current_c, tau_start, max_iter, rng, target_sum, deadline):
    n = len(current)
    tau = tau_start
    alpha = 0.05
    best_local = current.copy()
    best_local_c = current_c
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() > deadline:
            break
        if time.perf_counter() > deadline + it * 0:
            pass
        grad = compute_soft_gradient(current, tau)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        improved = False
        for _ in range(20):
            candidate = project_box_sum(current + alpha * grad, target_sum)
            candidate_c = evaluate_c5(candidate)
            if candidate_c < current_c - 1e-11:
                current = candidate
                current_c = candidate_c
                improved = True
                if current_c < best_local_c:
                    best_local = current.copy()
                    best_local_c = current_c
                break
            alpha *= 0.5
        if not improved:
            d = rng.standard_normal(n)
            d -= d.mean()
            d /= np.linalg.norm(d) + 1e-15
            candidate = project_box_sum(current + alpha * d, target_sum)
            candidate_c = evaluate_c5(candidate)
            if candidate_c < current_c - 1e-11:
                current = candidate
                current_c = candidate_c
                if current_c < best_local_c:
                    best_local = current.copy()
                    best_local_c = current_c
            else:
                alpha *= 0.4
        if current_c < best_local_c - 1e-12:
            stall = 0
        else:
            stall += 1
        if stall > 80:
            alpha *= 0.7
            if alpha < 1e-5:
                break
        tau *= 0.92
        if tau < 0.0005:
            tau = 0.0005
    return best_local, best_local_c


def pair_descent(current, current_c, rng, target_sum, deadline, max_passes=6):
    n = len(current)
    best = current.copy()
    best_c = current_c
    fractions = np.array([0.5, 0.25, 0.1, 0.75, 0.15, 0.35, 0.6, 0.2])
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    for p in range(max_passes):
        if time.perf_counter() > deadline:
            break
        rng.shuffle(pairs)
        any_improved = False
        for (i, j) in pairs:
            if time.perf_counter() > deadline:
                break
            for (src, dst) in [(i, j), (j, i)]:
                cap = min(best[src], 1.0 - best[dst])
                if cap < 1e-7:
                    continue
                for frac in fractions:
                    d = frac * cap
                    if d < 1e-8:
                        continue
                    trial = best.copy()
                    trial[src] -= d
                    trial[dst] += d
                    tc = evaluate_c5(trial)
                    if tc < best_c - 1e-11:
                        best = trial
                        best_c = tc
                        any_improved = True
                        break
    return best, best_c


def sa_phase(current, current_c, rng, target_sum, deadline, max_iters=20000):
    n = len(current)
    best = current.copy()
    best_c = current_c
    cur = current.copy()
    cur_c = current_c
    temp = max(0.002, 0.08 * cur_c)
    min_temp = 1e-6
    scale = 0.04
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        r = rng.random()
        if r < 0.65:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            src, dst = int(i), int(j)
            cap = min(cur[src], 1.0 - cur[dst])
            if cap < 1e-7:
                src, dst = int(j), int(i)
                cap = min(cur[src], 1.0 - cur[dst])
                if cap < 1e-7:
                    continue
            d = rng.uniform(0.05, 1.0) * cap * scale
            d = min(d, cap)
            trial = cur.copy()
            trial[src] -= d
            trial[dst] += d
        else:
            d = rng.standard_normal(n) * scale
            d -= d.mean()
            trial = project_box_sum(cur + d, target_sum)
        tc = evaluate_c5(trial)
        if tc <= cur_c:
            cur = trial
            cur_c = tc
            if tc < best_c:
                best = cur.copy()
                best_c = tc
        else:
            delta = tc - cur_c
            if delta < 10 * temp:
                if rng.random() < np.exp(-delta / temp):
                    cur = trial
                    cur_c = tc
        if it % 500 == 499:
            temp *= 0.97
            scale *= 0.995
        if temp < min_temp:
            break
    return best, best_c


def try_all_small_transfers(h, c):
    n = len(h)
    best = h.copy()
    best_c = c
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            cap = min(best[i], 1.0 - best[j])
            if cap < 1e-8:
                continue
            for frac in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
                d = frac * cap
                if d < 1e-9:
                    continue
                trial = best.copy()
                trial[i] -= d
                trial[j] += d
                tc = evaluate_c5(trial)
                if tc < best_c - 1e-11:
                    best = trial
                    best_c = tc
    return best, best_c

    rng = np.random.default_rng(seed)

            best_h = h.copy()
            best_c = c
            best_n = len(h)

    best_c = evaluate_c5(best_h)

    starts.append(best_h.copy())
    for s in range(4):
        noise = rng.standard_normal(n) * (0.01 + 0.005 * s)
        pert = project_box_sum(best_h + noise, target_sum)
        pc = evaluate_c5(pert)
        if pc < best_c:
            try_global_update(pert, pc)
        starts.append(pert)

    # Try a few different resolutions from interpolation

        if time.perf_counter() > deadline_total - 120:
            break
        if n_try == n:
            continue
        x_grid = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_grid, best_h, period=2.0)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        c_try = evaluate_c5(h_interp)
        if c_try < best_c:
            try_global_update(h_interp, c_try)
            starts.append(h_interp.copy())

    # Main optimization loop over starts
    for start_idx, start_h in enumerate(starts):
        sn = len(start_h)
        st_sum = sn / 2.0
        sc = evaluate_c5(start_h)
        if time.perf_counter() > deadline_total - 30:

        if time.perf_counter() < ph1_deadline:
            tau0 = max(0.003, 0.06 * sc)
            res_h, res_c = smooth_descent(start_h, sc, tau0, 2000, rng, st_sum, ph1_deadline)
            if res_c < best_c:
                try_global_update(res_h, res_c)
            start_h, sc = res_h, res_c

        # Phase 2: Pair descent

        if time.perf_counter() < ph2_deadline:

            if res_c < best_c:
                try_global_update(res_h, res_c)
            start_h, sc = res_h, res_c

        # Phase 3: SA

            if res_c < best_c:
                try_global_update(res_h, res_c)
            start_h, sc = res_h, res_c

        # Phase 4: Final small transfers

            res_h, res_c = try_all_small_transfers(start_h, sc)
            if res_c < best_c:
                try_global_update(res_h, res_c)

    # Extra: if we still have time, do another round of SA from best

        extra_deadline = deadline_total
        res_h, res_c = sa_phase(best_h, best_c, rng, best_n / 2.0, extra_deadline, max_iters=30000)

            try_global_update(res_h, res_c)

    # Final polish
    if time.perf_counter() < deadline_total:
        res_h, res_c = try_all_small_transfers(best_h, best_c)
        if res_c < best_c:
            try_global_update(res_h, res_c)

    # Final validation
    final_h = best_h.copy()
    final_n = best_n
    final_c = evaluate_c5(final_h)
    target = final_n / 2.0
    if abs(final_h.sum() - target) > 1e-6:
        final_h = project_box_sum(final_h, target)
        final_c = evaluate_c5(final_h)
    if final_h.min() < -1e-9 or final_h.max() > 1.0 + 1e-9:
        final_h = np.clip(final_h, 0.0, 1.0)
        final_h = project_box_sum(final_h, final_n / 2.0)
        final_c = evaluate_c5(final_h)

    return final_h, final_c, final_n
