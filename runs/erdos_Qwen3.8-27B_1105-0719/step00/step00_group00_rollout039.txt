def compute_R(h):

    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full") * dx
    return R

    """Project onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:

    lo, hi = -2.0, 2.0

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_R(h, active_indices):
    """Compute average subgradient of R[s] for s in active_indices."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    grad = np.zeros(n)
    for s in active_indices:
        lag = s - n + 1
        for j in range(n):
            val = 0.0
            ip = j - lag
            if 0 <= ip < n:
                val += g[ip]
            ip2 = j + lag
            if 0 <= ip2 < n:
                val -= h[ip2]
            grad[j] += dx * val
    grad /= len(active_indices)
    return grad

    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_R(h).max()
    
    step_size = 0.05
    patience = 0

            patience = 0

For the subgradient method:

- We take a step in the negative subgradient direction, then project.
- Adaptive step size: start with a reasonable value, reduce when no improvement.

The projection is the key geometric operation. The feasible set is the intersection of the hypercube [0,1]^n and the hyperplane sum(h) = n/2. The projection onto this set can be done as:
1. Compute unconstrained projection p = h - alpha * grad (the "desired" point)
2. Project p onto [0,1]^n ∩ {sum = n/2}

The projection onto [0,1]^n is just clipping. The projection onto the hyperplane with box constraints is the "water-filling" problem, solvable by bisection.

    return np.correlate(h, 1.0 - h, mode="full") * dx

    return compute_R(h).max()

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(10):
        if np.clip(h - lo, 0, 1).sum() < target_sum:
            lo -= 1.0
        else:
            break
    for _ in range(10):
        if np.clip(h - hi, 0, 1).sum() > target_sum:
            hi += 1.0
        else:
            break

        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:

        else:

import numpy as np
import time


def compute_R(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_value(h):
    return compute_R(h).max()


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(10):
        if np.clip(h - lo, 0.0, 1.0).sum() < target_sum:
            lo -= 1.0
        else:
            break
    for _ in range(10):
        if np.clip(h - hi, 0.0, 1.0).sum() > target_sum:
            hi += 1.0
        else:
            break
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def soft_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    grad = np.zeros(n)
    for s in range(len(weights)):
        w = weights[s]
        if w < 1e-8:
            continue
        lag = s - n + 1
        if lag >= 0:
            grad[lag:] += w * dx * g[:n - lag]
            grad[:n - lag] -= w * dx * h[lag:]
        else:
            kl = -lag
            grad[:n - kl] += w * dx * g[kl:]
            grad[kl:] -= w * dx * h[:n - kl]
    return grad


def optimize_h(h, time_up, max_iters=5000):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_value(h)
    step_size = 0.04
    tau_start = 0.06
    tau_end = 0.002
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time_up():
            break

        R = compute_R(h)
        c5 = R.max()

        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            h = best_h + np.random.uniform(-0.04, 0.04, n)
            h = project_feasible(h, target_sum)
            stall = 0
            step_size = 0.03
            if stall > 8:
                break

        frac = min(1.0, it / (max_iters * 0.6))
        tau = tau_start * (tau_end / tau_start) ** frac

        R_shifted = (R - c5) / tau
        weights = np.exp(R_shifted - R_shifted.max())
        weights /= weights.sum()

        grad = soft_gradient(h, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        accepted = False
        for fac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = h - step_size * fac * grad
            h_try = project_feasible(h_try, target_sum)
            if c5_value(h_try) < c5 - 1e-12:
                h = h_try
                accepted = True
                break

        if not accepted:
            step_size *= 0.85
            if step_size < 0.0003:
                break

    return best_h, best_c5


def local_search(h, time_up, max_rounds=3000, seed=0):
    n = len(h)
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = c5_value(h)

    for rd in range(max_rounds):
        if time_up():
            break
        scale = 0.02 + 0.06 * rng.exponential(1.0)
        k = rng.randint(1, min(6, n))
        idx = rng.choice(n, k, replace=False)
        h_try = best_h.copy()
        h_try[idx] += rng.uniform(-scale, scale, k)
        h_try = project_feasible(h_try, target_sum)
        c5_try = c5_value(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()

    return best_h, best_c5

        fixed_sum = target_sum - (h.sum() - h[idx].sum())

            h_tmp = h.copy()
            h_tmp[idx] = x
            return c5_value(h_tmp)

        x0 = h[idx].copy()
        if abs(h[idx].sum() - fixed_sum) < 1e-8:
            pass
        from scipy.optimize import minimize as sp_min
        cons = {'type': 'eq', 'fun': lambda x: x.sum() - fixed_sum}
        bounds = [(0.0, 1.0)] * block_size
        res = sp_min(obj, x0, method='Nelder-Mead', bounds=bounds,
                     options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
        if res.success and res.fun < best_c5 - 1e-10:
            best_h[idx] = np.clip(res.x, 0, 1)
            best_h = project_feasible(best_h, target_sum)
            best_c5 = c5_value(best_h)

    return best_h, best_c5
