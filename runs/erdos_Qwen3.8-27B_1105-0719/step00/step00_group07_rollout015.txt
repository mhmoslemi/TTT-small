    """Project x onto {y: 0<=y<=1, sum(y)=target}"""
    n = len(x)
    # Bisection on lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix residual
    residual = target - proj.sum()
    if abs(residual) > 1e-10:
        # Distribute residual
        if residual > 0:
            idx = np.where(proj < 1 - 1e-10)[0]
        else:
            idx = np.where(proj > 1e-10)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            if per > 1 - proj[idx].max():
                per = (1 - proj[idx].max()) / 2
            if per < -proj[idx].min():
                per = -proj[idx].min() / 2
            proj[idx] += per
            # Re-fix

                    proj[i] = min(1, proj[i] + residual2)

                    proj[i] = max(0, proj[i] + residual2)

    max_iters = 200000
    it = 0
    while time.perf_counter() - t0 < time_budget and it < max_iters:
        it += 1
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        candidate = best_h.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0
        s, _ = compute_c5(candidate)
        if s < best_score - 1e-13:
            best_h = candidate
            best_score = s
    return best_h, best_score

def random_perturb(h, n, rng, scale):
    target = n / 2.0
    noise = rng.normal(0, scale, size=n)
    return project_onto_feasible(h + noise, target)

def try_resolution(best_h, best_score, best_n, n_new, time_per_try, rng):
    if n_new < 10 or n_new > 600 or n_new == best_n:
        return best_h, best_score, best_n
    target_new = n_new / 2.0
    if n_new < best_n:
        ratio = best_n / n_new
        if abs(ratio - round(ratio)) < 1e-6:
            g = int(round(ratio))
            h_new = best_h[:g * n_new].reshape(n_new, g).mean(axis=1)
        else:
            x_new = np.linspace(0, 1, n_new, endpoint=False)
            x_old = np.linspace(0, 1, best_n, endpoint=False)
            h_new = np.interp(x_new, x_old, best_h)
    else:
        x_new = np.linspace(0, 1, n_new, endpoint=False)
        x_old = np.linspace(0, 1, best_n, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
    h_new = project_onto_feasible(h_new, target_new)
    s, _ = compute_c5(h_new)
    if s < best_score:
        h_new, s = local_search(h_new, time_per_try, n_active_base=5)
        return h_new, s, n_new
    return best_h, best_score, best_n

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.perf_counter()
    deadline = t_start + budget_s - 3.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 0:
        n = len(h0)
        target = n / 2.0
        best_h = project_onto_feasible(h0, target)
    else:
        n = 95
        target = n / 2.0
        best_h = np.full(n, 0.5)

    best_score, _ = compute_c5(best_h)
    best_n = n

    # Phase 1: LP-based local search from initial point
    if time.perf_counter() < deadline - 30:
        tb = min(300, (deadline - time.perf_counter()) * 0.35)
        h_new, s_new = local_search(best_h, tb, use_lp=True, n_active_base=5)
        if s_new < best_score:
            best_h, best_score = h_new, s_new

    # Phase 2: Symmetry transforms + re-search
    for transform_name in ['sym', 'flip', 'anti_sym']:
        if time.perf_counter() >= deadline - 20:
            break
        if transform_name == 'sym':
            h_try = symmetrize(best_h)
        elif transform_name == 'flip':
            h_try = flip_complement(best_h)
        else:
            h_try = 0.5 * (best_h - flip_complement(best_h))
            h_try = project_onto_feasible(h_try, target)
        h_try = project_onto_feasible(h_try, target)
        s_try, _ = compute_c5(h_try)
        if s_try < best_score + 0.001:
            tb = min(100, (deadline - time.perf_counter()) * 0.15)
            if tb > 10:
                h_new, s_new = local_search(h_try, tb, use_lp=True, n_active_base=5)
                if s_new < best_score:
                    best_h, best_score = h_new, s_new

    # Phase 3: Binary swap search
    if time.perf_counter() < deadline - 30:
        tb = min(150, (deadline - time.perf_counter()) * 0.2)
        h_bin, s_bin = binary_swap_search(best_h, tb, rng)
        if s_bin < best_score:
            best_h, best_score = h_bin, s_new
            # Polish with LP search
            if time.perf_counter() < deadline - 10:
                tb2 = min(60, (deadline - time.perf_counter()) * 0.15)
                h_new, s_new = local_search(h_bin, tb2, use_lp=True, n_active_base=8)
                if s_new < best_score:
                    best_h, best_score = h_new, s_new

    # Phase 4: Random restarts with local search
    if time.perf_counter() < deadline - 30:
        max_restarts = 30
        for r in range(max_restarts):
            if time.perf_counter() >= deadline - 15:
                break
            scale = 0.02 + 0.03 * rng.random()
            h_pert = random_perturb(best_h, n, rng, scale)
            tb = min(40, (deadline - time.perf_counter()) * 0.2)
            if tb < 5:
                break
            h_new, s_new = local_search(h_pert, tb, use_lp=(r % 2 == 0), n_active_base=4)
            if s_new < best_score:
                best_h, best_score = h_new, s_new

    # Phase 5: Try different resolutions
    if time.perf_counter() < deadline - 20:
        res_candidates = []
        for factor in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            n_try = int(round(n * factor))
            if n_try >= 20 and n_try <= 400:
                res_candidates.append(n_try)
        # Deduplicate and ensure we don't repeat
        res_candidates = sorted(set(res_candidates))
        for n_try in res_candidates:
            if time.perf_counter() >= deadline - 15:
                break
            tb = min(60, (deadline - time.perf_counter()) * 0.2)
            if tb < 5:
                break
            best_h, best_score, best_n = try_resolution(
                best_h, best_score, best_n, n_try, tb, rng)

    # Phase 6: Final intensive polish
    if time.perf_counter() < deadline - 5:
        tb = deadline - time.perf_counter() - 3
        if tb > 5:
            h_new, s_new = local_search(best_h, tb, use_lp=True, n_active_base=8)
            if s_new < best_score:
                best_h, best_score = h_new, s_new

    # Final feasibility check
    target = best_n / 2.0
    best_h = project_onto_feasible(best_h, target)
    best_score, _ = compute_c5(best_h)
    # Ensure sum is exact
    assert abs(best_h.sum() - target) < 1e-6, f"Sum mismatch: {best_h.sum()} vs {target}"
    assert np.all(best_h >= -1e-9) and np.all(best_h <= 1 + 1e-9)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_feasible(best_h, target)
    best_score, _ = compute_c5(best_h)

    return best_h, best_score, best_n
