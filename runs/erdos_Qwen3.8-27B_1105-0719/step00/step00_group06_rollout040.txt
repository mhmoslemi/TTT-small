            cnt = int(np.sum(mask))

_state = {}

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = float(np.sum(h))
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            mx = float(h[mask].min())
            step = min(diff / cnt, mx)
            h[mask] -= step
        else:
            mask = h < 1
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            mx = float((1.0 - h[mask]).min())
            step = min(-diff / cnt, mx)
            h[mask] += step
    return h

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(overlaps))
    exps = np.exp((overlaps - M) / tau)
    W = float(np.sum(exps))
    obj = M + tau * np.log(W)
    weights = exps / W
    grad = np.zeros(n)
    n1 = n - 1
    for i in range(2 * n - 1):
        wi = weights[i]
        if wi < 1e-14:
            continue
        s = i - n1
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            a = -s
            grad[:n - a] += wi * (1.0 - h[a:])
            grad[a:] -= wi * h[:n - a]
    return obj, grad

    n = _state['n']
    tau = _state['tau']
    val, g = smooth_obj_grad(x, tau)
    s = float(np.sum(x)) - n / 2.0
    pen = 1e6 * s * s
    below = np.maximum(0.0, -x)
    above = np.maximum(0.0, x - 1.0)
    pen += 1e6 * (float(np.sum(below * below)) + float(np.sum(above * above)))
    return val + pen
