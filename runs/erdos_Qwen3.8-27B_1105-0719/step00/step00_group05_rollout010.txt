```python
def project_feasible(z, target_sum):
    n = len(z)
    # Project z onto {0 <= x <= 1, sum(x) = target_sum}
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2

        if s < target_sum:

        else:

                result[i] += add
                diff -= add

                result[i] -= sub
                diff += sub
```

```python
def compute_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    weights = exp_vals / exp_vals.sum()  # length 2n-1
    
    grad = np.zeros(n)
    # k ranges from 0 to 2n-2, actual lag m = k - (n-1)
    for k_idx in range(2 * n - 1):
        m = k_idx - (n - 1)
        w = weights[k_idx] * dx
        if abs(w) < 1e-14:
            continue
        if m >= 0:
            # ∂c(m)/∂h[j] = (1-h[j-m]) * [m ≤ j] - h[j+m] * [j ≤ n-1-m]
            # For j in m..n-1: add (1-h[j-m])
            grad[m:] += w * (1.0 - h[:n-m])
            # For j in 0..n-1-m: subtract h[j+m]
            grad[:n-m] -= w * h[m:]
        else:
            # m < 0, let p = -m > 0
            p = -m
            # ∂c(-p)/∂h[j] = (1-h[j+p]) * [j ≤ n-1-p] - h[j-p] * [j ≥ p]
            # For j in 0..n-1-p: add (1-h[j+p])
            grad[:n-p] += w * (1.0 - h[p:])
            # For j in p..n-1: subtract h[j-p]
            grad[p:] -= w * h[:n-p]
    return grad
```

```python
    n = len(z)
    # Binary search for lambda such that sum(clip(z + lambda, 0, 1)) = target_sum
    lo, hi = -max(z.max(), 0) - 1, max(1 - z.min(), 0) + 1

    """Projected smooth gradient descent"""
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2.0
    best_c = evaluate_c5(h)
    start_time = time.time()
    
    # Remove mean of gradient to preserve sum constraint approximately
    # (projection will fix it anyway)
    
    alpha = 0.01
    stall_count = 0
    max_stall = 30
    
    for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Compute gradient
        g = compute_smooth_gradient(h, beta)
        # Center gradient (remove mean to help with sum constraint)
        g_centered = g - g.mean()
        
        # Adaptive step
        gnorm = np.linalg.norm(g_centered)
        if gnorm < 1e-12:
            break
        
        # Try multiple step sizes
        accepted = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_trial = h - a * g_centered
            h_trial = project_feasible(h_trial, target_sum)
            c_trial = evaluate_c5(h_trial)
            if c_trial < best_c - 1e-10:
                h = h_trial
                best_c = c_trial
                alpha = min(alpha * 1.1, 0.1)
                stall_count = 0
                accepted = True
                break
        
        if not accepted:
            alpha *= 0.7
            stall_count += 1
            if stall_count > max_stall:

            # Try random perturbation

                c_trial = evaluate_c5(h_trial)
                if c_trial < best_c + 0.001:  # Allow slight worsening
                    if c_trial < best_c:
                        best_c = c_trial
                    h = h_trial
                    stall_count = 0
        
        # Increase beta periodically

            beta = min(beta * 1.3, 1000.0)
    
    return h, best_c
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(z, target_sum):
    n = len(z)
    lo, hi = -z.max() - 1.0, 1.0 - z.min() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - result
            idx = np.argsort(-room)
            for i in idx:
                a = min(diff, room[i])
                if a > 0:
                    result[i] += a
                    diff -= a
                    if diff <= 1e-13:
                        break
        else:
            room = result
            idx = np.argsort(room)
            for i in idx:
                a = min(-diff, room[i])
                if a > 0:
                    result[i] -= a
                    diff += a
                    if diff >= -1e-13:
                        break
    return result


def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ev = np.exp(beta * (corr - mc))
    w = ev / ev.sum()
    grad = np.zeros(n)

            continue

            grad[m:] += wt * (1.0 - h[:L])

        else:
            p = -m
            L = n - p
```
