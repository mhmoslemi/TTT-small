def evaluate_c5(h):
    """Compute the C5 objective for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def fix_sum(h, target_sum):
    """Project h to have the correct sum while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum; add to elements with room
        mask = h < 1
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
    else:
        # Need to decrease sum
        mask = h > 0
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free  # diff is negative
    return np.clip(h, 0, 1)
