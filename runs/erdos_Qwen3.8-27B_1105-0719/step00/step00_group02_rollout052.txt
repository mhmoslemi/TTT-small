    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s_y = y.sum()
        if s_y > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Exact repair for floating point
    residual = s - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = y < 1.0 - 1e-15

        d = scale * cap * (0.5 + rng.random())
        d = min(d, cap)
        h_cur[i] -= d
        h_cur[j] += d
        c_new = eval_c5(h_cur)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            c_cur = c_new
            if c_cur < best_c:
                best_h = h_cur.copy()
                best_c = c_cur
            temp *= 0.999
            if delta < 0:
                scale = min(scale * 1.1, 1.0)
        else:
            h_cur[i] += d
            h_cur[j] -= d
            temp *= 0.995
            scale = max(scale * 0.9, 0.01)
    
    return h_cur, c_cur, best_h, best_c


def make_fallback(n, seed):
    """Create a reasonable fallback vector"""
    rng = np.random.default_rng(seed)
    x = 0.5 + 0.2 * np.cos(2 * np.pi * np.arange(n) / n) + rng.standard_normal(n) * 0.05
    return project_box_sum(x, n / 2.0)
