import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def enforce_constraints(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            per = diff / cnt
            h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
        # Final fix
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            idx = int(np.argmax(h)) if diff2 < 0 else int(np.argmin(h))
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def compute_delta(h, p, q, shifts, n):
    """Compute d[j] such that c_new[j] ≈ c[j] + tau * d[j]."""
    d = np.zeros(2 * n - 1)
    
    k = shifts  # array of lag values
    
    # Term 1: -(1-h[p-k]) for valid index
    idx1 = p - k
    v1 = (idx1 >= 0) & (idx1 < n)
    d[v1] -= (1.0 - h[idx1[v1]])
    
    # Term 2: +(1-h[q-k]) for valid index
    idx2 = q - k
    v2 = (idx2 >= 0) & (idx2 < n)
    d[v2] += (1.0 - h[idx2[v2]])
    
    # Term 3: +h[p+k] for valid index
    idx3 = p + k
    v3 = (idx3 >= 0) & (idx3 < n)
    d[v3] += h[idx3[v3]]
    
    # Term 4: -h[q+k] for valid index
    idx4 = q + k
    v4 = (idx4 >= 0) & (idx4 < n)
    d[v4] -= h[idx4[v4]]
    
    return d

def find_optimal_tau(corr, d, tau_lo, tau_hi):
    """Find tau in [tau_lo, tau_hi] that minimizes max(corr + tau*d)."""
    current_max = np.max(corr)
    
    # Active set: lags near the max
    margin = 0.005
    active = np.where(corr >= current_max - margin)[0]
    
    if len(active) == 0:
        active = np.where(corr >= current_max - 0.01)[0]
    
    candidates = [tau_lo, tau_hi]
    
    # Breakpoints among active lags
    for i in range(len(active)):
        for j in range(i + 1, len(active)):
            ai, aj = int(active[i]), int(active[j])
            denom = d[ai] - d[aj]
            if abs(denom) > 1e-10:
                tau_bp = (corr[aj] - corr[ai]) / denom
                if tau_lo < tau_bp < tau_hi:
                    candidates.append(tau_bp)
    
    # Evaluate all candidates
    best_tau = 0.0
    best_val = current_max
    for tau in candidates:
        tau_f = float(np.clip(tau, tau_lo, tau_hi))
        val = float(np.max(corr + tau_f * d))
        if val < best_val:
            best_val = val
            best_tau = tau_f
    
    return best_tau, best_val
