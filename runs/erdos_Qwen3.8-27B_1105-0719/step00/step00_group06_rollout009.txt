def repair_h(h):
    """Enforce 0<=h<=1 and sum(h)=n/2 exactly."""
    h = np.array(h, dtype=np.float64).copy()
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to fix sum
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add mass
            mask = h < 1.0 - 1e-14
            if not mask.any():
                break
            per = diff / mask.sum()
            add = np.minimum(per, 1.0 - h[mask])
            h[mask] += add
        else:
            # Need to remove mass
            mask = h > 1e-14
            if not mask.any():
                break
            per = -diff / mask.sum()
            sub = np.minimum(per, h[mask])
            h[mask] -= sub
    h = np.clip(h, 0.0, 1.0)
    return h

def score_h(h):

    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = (3.0 - np.sqrt(5.0)) / 2.0

    d = a + invphi * (b - a)

    fd = f(d)

            break

            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:

            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)

def gradient_profile(h):
    """Compute gradient of each lag's overlap w.r.t. each h coordinate.
    Returns array of shape (2n-1, n) where row k is gradient of lag k's value."""

    # F_s = sum_{valid r} h[r'] * (1-h[r''})

    # valid j: 0 <= j < n and 0 <= j+k-(n-1) < n
    
    # Gradient w.r.t. h[i]:
    # h[i] appears as first factor when j=i: contributes one_minus_h[i + k - (n-1)] if valid
    # h[i] appears in (1-h[...]) when i = j + k - (n-1), i.e., j = i - k + (n-1): contributes -h[j] if valid
    
    grads = np.zeros((2*n - 1, n))

            # h[i] in second factor: need j = i - offset, 0 <= j < n
            j = i - offset
            if 0 <= j < n:
                grads[k, i] -= h[j]
    return grads
