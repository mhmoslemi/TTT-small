def sanitize_h(h, n_target=None):

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def sanitize_h(h):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target_sum = n / 2.0
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr > 1e-15:
                h += diff * room / tr
            else:
                h += diff / n
        else:
            cap = h.copy()
            tc = np.sum(cap)
            if tc > 1e-15:
                h += diff * cap / tc
            else:
                h += diff / n
        h = np.clip(h, 0.0, 1.0)
        residual = target_sum - np.sum(h)
        if residual > 1e-14:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if residual <= 1e-15:
                    break
        elif residual < -1e-14:
            idx = np.argsort(h)[::-1]
            residual = -residual
            for i in idx:
                rem = min(residual, h[i])
                h[i] -= rem
                residual -= rem
                if residual <= 1e-15:
                    break
    return h
