def objective_exact(h):

    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(F)), F

def project_feasible(z, target_sum):
    n = len(z)
    z = np.array(z, dtype=float).copy()
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result

def compute_lag_gradient(h, lag, dx, N):
    d = lag - (N - 1)
    g = np.zeros(N)
    m_start = max(0, -d)
    m_end = min(N, N - d)
    if m_start >= m_end:
        return g
    m_idx = np.arange(m_start, m_end)
    first_idx = m_idx + d
    second_idx = m_idx
    g[first_idx] += dx * (1.0 - h[second_idx])
    g[second_idx] -= dx * h[first_idx]
    return g

def softmax_objective(F, beta):
    M = np.max(F)
    shifted = beta * (F - M)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    s = np.sum(exps)
    phi = M + (max_s + np.log(s)) / beta
    weights = exps / s
    return phi, weights

def softmax_grad(h, beta, weights, dx, N):
    grad = np.zeros(N)
    for lag in range(2 * N - 1):
        w = weights[lag]
        if w < 1e-12:
            continue
        g = compute_lag_gradient(h, lag, dx, N)
        grad += w * g
    return grad

            if accepted:
                stuck = 0
                alpha = min(alpha * 1.1, 0.2)
            else:
                stuck += 1
                alpha *= 0.5
                if stuck > 15 or alpha < 1e-12:
                    break
    return best_local, best_c

def active_lp_descent(h0, dx, N, target_sum, deadline, max_rounds=100):
    from scipy.optimize import linprog
    h = h0.copy()
    c, F = objective_exact(h)
    best_local = h.copy()
    best_c = c
    radius = 0.4
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c, F = objective_exact(h)
        M = np.max(F)
        active = np.where(F >= M - 0.002)[0]
        if len(active) == 0:
            active = np.array([np.argmax(F)])
        if len(active) > 30:
            active = active[np.argsort(F[active])[-30:]]
        n_vars = N + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for lag in active:
            g = compute_lag_gradient(h, lag, dx, N)
            row = np.zeros(n_vars)
            row[:N] = g
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-F[lag] + np.dot(g, h))
        if len(A_rows) == 0:
            break
        A_ineq = np.array(A_rows)
        b_ineq = np.array(b_rows)
        lb = np.maximum(0.0, h - radius)
        ub = np.minimum(1.0, h + radius)
        lb_full = np.append(lb, c - radius)
        ub_full = np.append(ub, c + radius)
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :N] = 1.0
        b_eq = np.array([target_sum])
        try:
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=list(zip(lb_full, ub_full)), method='highs')
        except Exception:
            break
        if res is None or not res.success:
            radius *= 0.5
            if radius < 0.002:
                break
            continue
        x_new = res.x[:N]
        x_new = project_feasible(x_new, target_sum)
        c_new, _ = objective_exact(x_new)
        if c_new < c - 1e-11:
            h = x_new
            c = c_new
            if c < best_c - 1e-11:
                best_c = c
                best_local = h.copy()
            radius = min(radius * 1.3, 1.0)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                x_try = h + frac * (x_new - h)
                x_try = project_feasible(x_try, target_sum)
                c_try, _ = objective_exact(x_try)
                if c_try < c - 1e-11:
                    h = x_try
                    c = c_try
                    if c < best_c - 1e-11:
                        best_c = c
                        best_local = h.copy()
                    improved = True
                    break
            if not improved:
                radius *= 0.5
                if radius < 0.002:
                    break
    return best_local, best_c

def active_steepest(h0, dx, N, target_sum, deadline, max_it=200):
    h = h0.copy()
    c, _ = objective_exact(h)
    best_local = h.copy()
    best_c = c
    alpha = 0.05
    for it in range(max_it):
        if time.time() > deadline:
            break
        _, F = objective_exact(h)
        M = np.max(F)
        active = np.where(F >= M - 0.001)[0]
        if len(active) == 0:
            active = np.array([np.argmax(F)])
        if len(active) > 20:
            active = active[np.argsort(F[active])[-20:]]
        grad = np.zeros(N)
        for lag in active:
            grad += compute_lag_gradient(h, lag, dx, N)
        grad /= len(active)
        grad_tan = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break
        grad_tan /= gnorm
        accepted = False
        a = alpha
        for _ in range(40):
            step = h - a * gnorm * grad_tan
            step = project_feasible(step, target_sum)
            c_step, _ = objective_exact(step)
            if c_step <= c - 1e-13:
                h = step
                c = c_step
                accepted = True
                if c < best_c - 1e-12:
                    best_c = c
                    best_local = h.copy()
                break
            a *= 0.5
        if accepted:
            alpha = min(alpha * 1.05, 0.15)
        else:
            alpha *= 0.5
            if alpha < 1e-13:
                break
    return best_local, best_c
