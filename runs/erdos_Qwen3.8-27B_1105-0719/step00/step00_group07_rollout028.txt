    n = len(h)

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_sym_h(a, n):
    m = (n - 1) // 2
    h = np.empty(n)
    h[:m] = a
    h[m] = 0.5
    h[m+1:] = 1.0 - a[::-1]
    return h

def smooth_max_obj(a, n, beta):
    h = make_sym_h(a, n)
    raw = np.correlate(h, 1.0 - h, mode="full")
    shift = np.max(raw)
    return shift + np.log(np.sum(np.exp(beta * (raw - shift)))) / beta

def smooth_max_grad(a, n, beta):
    h = make_sym_h(a, n)
    raw = np.correlate(h, 1.0 - h, mode="full")
    shift = np.max(raw)
    exp_vals = np.exp(beta * (raw - shift))
    w = exp_vals / np.sum(exp_vals)
    
    m = (n - 1) // 2
    grad_h = np.zeros(n)
    
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        d = (n - 1) - t
        if d >= 0:
            max_i = n - d
            grad_h[:max_i] += wt * (1.0 - h[d:])
            grad_h[d:] -= wt * h[:max_i]
        else:

            max_i = n - q
            grad_h[q:] += wt * (1.0 - h[:max_i])
            grad_h[:max_i] -= wt * h[q:]

    grad_a = grad_h[:m] - grad_h[n-1:m-1:-1]

    m = len(a)

    current_h = make_sym_h(current_a, n)
    current_c = compute_c5(current_h)
    best_a = current_a.copy()

            break
        
        # Random move
        n_changes = rng.randint(1, 4)
        idx = rng.choice(m, size=n_changes, replace=False)
        new_a = current_a.copy()
        for i in idx:
            delta = rng.uniform(-step_size, step_size)
            new_a[i] = np.clip(new_a[i] + delta, 0, 1)
        
        new_h = make_sym_h(new_a, n)
        new_c = compute_c5(new_h)

            current_h = new_h

            current_a = new_a
            current_h = new_h
            current_c = new_c

        if it % 50 == 0:
            step_size *= 0.95
    
    return best_a, best_c
