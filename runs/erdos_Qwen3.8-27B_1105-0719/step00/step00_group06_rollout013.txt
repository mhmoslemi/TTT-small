                    h = h_candidate.copy()
        except Exception:
            pass

    # Phase 4: Binary search after continuous optimization
    if best_n % 2 == 0 and time.time() < deadline - 20:
        b, c = binary_search_1swap(best_h, deadline)
        if c < best_c5 - 1e-12:
            best_h = b
            best_c5 = c
            best_n = len(b)

    # Phase 5: Try SLSQP from binary solution with perturbation
    if time.time() < deadline - 60:
        h_pert = best_h.copy()
        h_pert = h_pert + rng.randn(len(h_pert)) * 0.05
        h_pert = project_box_sum(h_pert)
        n_p = len(h_pert)
        _con_n = n_p
        bounds_p = [(0.0, 1.0)] * n_p
        conver_p = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        for tau in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.time() > deadline - 30:
                break
            try:
                res = minimize(slsqp_fun, h_pert, jac=slsqp_fun_grad, args=(tau,),
                               method='SLSQP', bounds=bounds_p, constraints=conver_p,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    hc = project_box_sum(res.x)
                    cc = exact_c5(hc)
                    if cc < best_c5 - 1e-12:
                        best_h = hc
                        best_c5 = cc
                        best_n = len(hc)
                        h_pert = hc.copy()
            except Exception:
                pass

    # Phase 6: Try other resolutions
    if time.time() < deadline - 60:
        for n_try in [best_n - 4, best_n + 4, best_n - 8, best_n + 8]:
            if n_try < 20 or n_try > 200 or time.time() > deadline - 30:
                continue
            if n_try % 2 != 0:
                n_try += 1
            h_try = resample_to_n(best_h, n_try, rng)
            _con_n = n_try
            bounds_t = [(0.0, 1.0)] * n_try
            cons_t = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
            for tau in [0.02, 0.01, 0.005, 0.001]:
                if time.time() > deadline - 20:
                    break
                try:
                    res = minimize(slsqp_fun, h_try, jac=slsqp_fun_grad, args=(tau,),
                                   method='SLSQP', bounds=bounds_t, constraints=cons_t,
                                   options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        hc = project_box_sum(res.x)
                        cc = exact_c5(hc)
                        if cc < best_c5 - 1e-12:
                            best_h = hc
                            best_c5 = cc
                            best_n = len(hc)
                            h_try = hc.copy()
                except Exception:
                    pass
            # Binary search on this resolution
            if n_try % 2 == 0 and time.time() < deadline - 15:
                b, c = binary_search_1swap(best_h, deadline)
                if c < best_c5 - 1e-12:
                    best_h = b
                    best_c5 = c
                    best_n = len(b)

    # Phase 7: Final binary refinement
    if best_n % 2 == 0 and time.time() < deadline - 5:
        b, c = binary_search_1swap(best_h, deadline)
        if c < best_c5 - 1e-12:
            best_h = b
            best_c5 = c
            best_n = len(b)

    if time.time() < deadline - 3:
        b, c = binary_search_2swap(best_h, deadline, max_rounds=1)
        if c < best_c5 - 1e-12:
            best_h = b
            best_c5 = c
            best_n = len(b)

    # Final validation
    best_h = project_box_sum(best_h)
    best_c5 = exact_c5(best_h)
    best_n = len(best_h)

    return (best_h, best_c5, best_n)
