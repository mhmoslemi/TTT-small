def pair_line_search(h, i, j, dx, n_grid=50):
    """Find best t for moving mass from j to i (or vice versa)."""
    # t > 0 means h[i] += t, h[j] -= t
    # t < 0 means h[i] -= |t|, h[j] += |t|
    t_min = max(-1.0, -h[i])  # h[i] + t >= 0 => t >= -h[i], and t >= -1
    t_max = min(1.0, h[j])    # h[j] - t >= 0 => t <= h[j], and t <= 1

        return 0.0

    ts = np.linspace(t_min, t_max, n_grid)

        h_new[j] -= t
        val = c5_eval(h_new, dx)
        if val < best_val:
            best_val = val
            best_t = t

            h_new[j] -= t

Actually, for a pair move h[i] += t, h[j] -= t, the change in the correlation at lag k is:
- Terms involving h[i]: h[i] appears in c_k as h[i]*(1-h[i-k]) (when i-k is valid) 
  Wait, let me re-derive.

- The term where m+k = i, i.e., m = i-k: h[i-k]*(1-h[i]) becomes h[i-k]*(1-h[i]-t)

     = t * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]

where we take terms as 0 when indices are out of range.

So Δc_k = t * [1 - h[i+k] - h[i-k] - 1 + h[j+k] + h[j-k]]
         = t * [h[j+k] + h[j-k] - h[i+k] - h[i-k]]

This is LINEAR in t for each k! So c_k(t) = c_k(0) + t * a_k where a_k = h[j+k] + h[j-k] - h[i+k] - h[i-k] (with out-of-range terms = 0).

Therefore C5(t) = max_k [dx * (c_k(0) + t * a_k)] which is the max of linear functions of t, i.e., a convex piecewise linear function!

This means I can find the minimum in closed form or with a very efficient method:
- The minimum of max_k (a_k * t + b_k) over t in [t_min, t_max]
- This is a 1D convex optimization problem
- I can solve it by finding where the upper envelope is minimized

The minimum of a convex piecewise linear function occurs where the slope changes sign (from negative to positive), or at a boundary.

The slope of the upper envelope at t is: max of a_k over those k where c_k(0) + t*a_k = C5(0) + t * slope... 

Actually, a simpler approach: just do a binary search / ternary search since it's convex. Or even simpler: evaluate at the "breakpoints" which are t values where two lines cross.

But the simplest efficient approach: since it's convex and 1D, I can use scipy.optimize.minimize_scalar with bounds, or just do a golden section search with exact evaluation (which now only requires O(N) per eval since I precomputed a_k and b_k).

Even better: I can evaluate C5(t) in O(N) time:
C5(t) = dx * max_k [c_k(0) + t * a_k]

where c_k(0) is the original correlation and a_k is precomputed for the pair.

This makes each line search O(N * n_evals) instead of O(N^2 * n_evals). Huge speedup!

    return np.correlate(h, 1.0 - h, mode='full') * dx

def proj_onto_simplex(h, target):
    h = np.asarray(h, dtype=np.float64)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    if abs(r) > 1e-13:

        if mask.sum() > 0:

import numpy as np
import time
from scipy.optimize import minimize

def c5_raw(h):
    """Return dx * max(correlate(h, 1-h, full))"""
    # Will be called with h of known length
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def c5_eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj_onto_simplex(h, target):
    h = np.asarray(h, dtype=np.float64)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    res = np.clip(h - lam, 0.0, 1.0)
    r = target - res.sum()
    if abs(r) > 1e-13:
        mask = (res < 1.0) if r > 0 else (res > 0.0)
        if mask.sum() > 0:
            res[mask] += r / mask.sum()
    return np.clip(res, 0.0, 1.0)

def compute_a(h, i, j, N):
    """Compute a_d for pair (i,j): a_d = h[j+d]+h[j-d]-h[i+d]-h[i-d]"""
    a = np.zeros(2 * N - 1)
    for d in range(-(N-1), N):
        idx = d + N - 1
        val = 0.0
        if 0 <= j + d < N:
            val += h[j + d]
        if 0 <= j - d < N:
            val += h[j - d]
        if 0 <= i + d < N:
            val -= h[i + d]
        if 0 <= i - d < N:
            val -= h[i - d]
        a[idx] = val
    return a

def compute_q(i, j, N):
    """Compute q_d for pair (i,j): quadratic coefficient"""
    q = np.zeros(2 * N - 1)
    q[N-1] = 2.0  # d=0
    d_diff = j - i  # lag j-i
    if d_diff != 0:
        q[d_diff + N - 1] = 1.0
        q[-d_diff + N - 1] = 1.0
    return q

def pair_search_exact(raw_corr, a, q, t_min, t_max, n_grid=20):
    """Find t minimizing max(raw_corr + t*a - t^2*q) over [t_min, t_max]"""
    if t_max - t_min < 1e-14:
        return 0.0
    
    def f(t):
        return np.max(raw_corr + t * a - t * t * q)
    
    # Grid search
    ts = np.linspace(t_min, t_max, n_grid)
    best_val = f(0.0)
    best_t = 0.0
    for t in ts:
        val = f(t)
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    
    # Refine around best_t with golden section
    if best_t > t_min + 1e-10 and best_t < t_max - 1e-10:
        width = (t_max - t_min) / n_grid
        lo_t = max(t_min, best_t - 2*width)
        hi_t = min(t_max, best_t + 2*width)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = lo_t + gr * (hi_t - lo_t)
        d = lo_t + (1.0 - gr) * (hi_t - lo_t)
        fc = f(c)
        fd = f(d)
        for _ in range(40):
            if fc < fd:
                hi_t = d
                d = c; fd = fc
                c = lo_t + gr * (hi_t - lo_t)
                fc = f(c)
            else:
                lo_t = c
                c = d; fc = fd
                d = lo_t + (1.0 - gr) * (hi_t - lo_t)
                fd = f(d)
        t_ref = (lo_t + hi_t) / 2.0
        val_ref = f(t_ref)
        if val_ref < best_val - 1e-15:
            best_val = val_ref
            best_t = t_ref
    
    # Check boundaries
    for t in [t_min, t_max]:
        if t_min - 1e-10 <= t <= t_max + 1e-10:
            val = f(t)
            if val < best_val - 1e-15:
                best_val = val
                best_t = t
    
    return best_t
