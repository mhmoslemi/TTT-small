    """Compute all 2n-1 correlation values (before multiplying by dx)."""

    b = 1.0 - h

        h[m] = 0.5

        h[m:] = 1.0 - a[::-1]

    """Returns (objective, grad_a) using smooth max approximation."""
    h = make_sym_h(a, n)
    b = 1.0 - h
    n = len(h)
    # Compute all correlation values
    corr = np.correlate(h, b, mode="full")
    # Smooth max
    cmax = np.max(corr)
    # Shift for numerical stability
    shifted = beta * (corr - cmax)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    obj = cmax + np.log(np.sum(exp_shifted)) / beta

    grad_h = np.zeros(n)

import numpy as np
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_sym_h(a, n):
    m = n // 2
    h = np.empty(n, dtype=np.float64)
    if n % 2 == 1:
        h[:m] = a
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    else:
        h[:m] = a
        h[m:] = 1.0 - a[::-1]
    return h

def sym_obj_grad(a, n, beta):
    m = n // 2
    h = make_sym_h(a, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    obj = cmax + np.log(total) / beta
    
    grad_h = np.zeros(n)
    for d in range(n):
        wp = weights[d + n - 1]
        if wp > 1e-13:
            grad_h[:n-d] += wp * (1.0 - h[d:])
            grad_h[d:] -= wp * h[:n-d]
        if d > 0:
            wn = weights[n - 1 - d]
            if wn > 1e-13:
                grad_h[d:] += wn * (1.0 - h[:n-d])
                grad_h[:n-d] -= wn * h[d:]
    
    idx = np.arange(m)
    grad_a = grad_h[idx] - grad_h[n-1-idx]
    
    return obj, grad_a

def local_search(a, n, t0, budget, rng, max_time_frac=0.7):
    m = n // 2
    h = make_sym_h(a, n)
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    cur_a = a.copy()
    
    beta = 20.0
    alpha = 0.05
    max_time = t0 + budget * max_time_frac
    stagnant = 0
    
    while time.time() < max_time:
        obj, grad = sym_obj_grad(cur_a, n, beta)
        
        if np.all(np.abs(grad) < 1e-10):
            break
        
        # Direction
        direction = -grad
        gnorm = np.linalg.norm(direction, np.inf)
        if gnorm < 1e-15:
            break
        direction /= gnorm
        
        # Line search
        improved = False
        for _ in range(25):
            step = alpha
            new_a = cur_a + step * direction
            new_a = np.clip(new_a, 0, 1)
            new_h = make_sym_h(new_a, n)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-14:
                cur_a = new_a
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        
        if not improved:
            # Try random perturbation
            stagnant += 1
            if stagnant > 10:
                # Reset with larger step
                alpha = min(alpha * 2, 0.5)
                stagnant = 0
                # Random kick
                kick = rng.uniform(-0.02, 0.02, m)
                new_a = np.clip(cur_a + kick, 0, 1)
                new_h = make_sym_h(new_a, n)
                new_c5 = compute_c5(new_h)
                if new_c5 < cur_c5:
                    cur_a = new_a
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                continue
            stagnant += 1
            continue
        
        stagnant = 0
        
        # Increase beta periodically
        beta = min(beta * 1.1, 500.0)
        # Slightly increase alpha
        alpha = min(alpha * 1.05, 0.3)
    
    # Final exact computation
    final_h = make_sym_h(cur_a, n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        cur_a = cur_a  # already good
    
    return cur_a, best_c5

        h[m+1:] = 1.0 - a[::-1]
    else:
        h[:m] = a
        h[m:] = 1.0 - a[::-1]
    return h

def sym_obj_grad(a, n, beta):
    m = n // 2
    h = make_sym_h(a, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    obj = cmax + np.log(total) / beta
    grad_h = np.zeros(n)
    for d in range(n):
        wp = weights[d + n - 1]
        if wp > 1e-13:
            grad_h[:n-d] += wp * (1.0 - h[d:])
            grad_h[d:] -= wp * h[:n-d]
        if d > 0:
            wn = weights[n - 1 - d]
            if wn > 1e-13:
                grad_h[d:] += wn * (1.0 - h[:n-d])
                grad_h[:n-d] -= wn * h[d:]
    idx = np.arange(m)
    grad_a = grad_h[idx] - grad_h[n-1-idx]
    return obj, grad_a

def project_mass_box(h, target):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h.copy()
        if np.sum(slack) < 1e-15:
            break
        adj = diff * slack / np.sum(slack)
        h = np.clip(h + adj, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h
