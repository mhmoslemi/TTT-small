    m = np.max(a)
    exp_args = (a - m) / tau
    exp_vals = np.exp(exp_args)
    exp_sum = exp_vals.sum()
    smooth_val = m + tau * np.log(exp_sum)
    W = exp_vals / exp_sum  # weights over lags, length 2n-1
    # Gradient computation
    grad = np.zeros(n)
    for k in range(n):
        idx_a = np.arange(n) - k + (n - 1)
        term_a = np.dot(W[idx_a], (1.0 - h))
        idx_b = k - np.arange(n) + (n - 1)
        term_b = np.dot(W[idx_b], h)
        grad[k] = dx * (term_a - term_b)
    return smooth_val, grad

def tangent_project(grad, h, n):
    """Project gradient onto the affine constraint sum(h)=n/2 (remove mean)."""
    return grad - np.mean(grad)

def smooth_descent(h, n, s, start_tau, time_limit, rng):
    """Smooth gradient descent with backtracking line search.
    Returns (best_h, best_c, time_spent)."""
    t0 = time.perf_counter()
    tau = start_tau
    alpha = 0.05
    best_h = h.copy()
    best_c = evaluate(h)
    cur_h = h.copy()
    cur_c = best_c
    no_improve = 0
    max_no_improve = 80
    
    while True:
        elapsed = time.perf_counter() - t0
        if elapsed > time_limit:
            break
        sv, grad = smooth_val_and_grad(cur_h, tau)
        g = tangent_project(grad, cur_h, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        g = -g / gnorm
        # Line search
        accepted = False
        for attempt in range(30):
            step = alpha * (0.8 ** attempt)
            cand = cur_h + step * g
            cand = project_box_sum(cand, s)
            cand_c = evaluate(cand)
            if cand_c < cur_c - 1e-12:
                # Also check smooth improvement for acceptance
                cur_h = cand
                cur_c = cand_c
                accepted = True
                if cur_c < best_c - 1e-14:
                    best_h = cur_h.copy()
                    best_c = cur_c
                    no_improve = 0
                else:
                    no_improve += 1
                break
        if not accepted:
            alpha *= 0.5
            no_improve += 1
        else:
            alpha = min(alpha * 1.1, 0.5)
        
        if no_improve >= max_no_improve:
            tau *= 1.5
            alpha = 0.05
            no_improve = 0
            # Small random perturbation to escape
            perturb = rng.standard_normal(n)
            perturb -= np.mean(perturb)
            cur_h = project_box_sum(cur_h + 0.01 * perturb, s)
            cur_c = evaluate(cur_h)
        
        if tau > best_c * 0.5:
            break
    
    # Final: use best found
    if best_c > evaluate(best_h) + 1e-12:
        best_c = evaluate(best_h)
    return best_h, best_c

def pair_transfer_search(h, n, s, time_limit, rng, max_passes=100):
    """Exact pair-transfer local search. Returns (best_h, best_c)."""
    t0 = time.perf_counter()
    best_h = h.copy()
    best_c = evaluate(h)
    cur_h = h.copy()
    cur_c = best_c
    
    improvements = 0
    for p in range(max_passes):
        found = False
        elapsed = time.perf_counter() - t0
        if elapsed > time_limit:
            break
        # Shuffle pairs
        pairs = list(range(n))
        rng.shuffle(pairs)
        for i in pairs:
            if time.perf_counter() - t0 > time_limit:
                break
            for j in pairs:
                if i == j:
                    continue
                # Try transfer i -> j
                cap = min(cur_h[i], 1.0 - cur_h[j])
                if cap < 1e-10:
                    continue
                for frac in (0.5, 0.3, 0.1, 0.05):
                    d = cap * frac
                    if d < 1e-10:
                        continue
                    cur_h[i] -= d
                    cur_h[j] += d
                    new_c = evaluate(cur_h)
                    if new_c < cur_c - 1e-12:
                        cur_c = new_c
                        found = True
                        improvements += 1
                        if cur_c < best_c - 1e-14:
                            best_c = cur_c
                            best_h = cur_h.copy()
                        break
                    else:
                        cur_h[i] += d
                        cur_h[j] -= d
        if not found:
            break
        found = False
        # Reverse direction
        for i in pairs:
            if time.perf_counter() - t0 > time_limit:
                break
            for j in pairs:
                if i == j:
                    continue
                cap = min(cur_h[j], 1.0 - cur_h[i])
                if cap < 1e-10:
                    continue
                for frac in (0.5, 0.3, 0.1, 0.05):
                    d = cap * frac
                    if d < 1e-10:
                        continue
                    cur_h[j] -= d
                    cur_h[i] += d
                    new_c = evaluate(cur_h)
                    if new_c < cur_c - 1e-12:
                        cur_c = new_c
                        found = True
                        improvements += 1
                        if cur_c < best_c - 1e-14:
                            best_c = cur_c
                            best_h = cur_h.copy()
                        break
                    else:
                        cur_h[j] += d
                        cur_h[i] -= d
        if not found:
            break
    
    return best_h, best_c

def simulated_annealing(h, n, s, time_limit, rng):
    """SA with pair transfers. Returns (best_h, best_c)."""
    t0 = time.perf_counter()
    best_h = h.copy()
    best_c = evaluate(h)
    cur_h = h.copy()
    cur_c = best_c
    temp = max(0.01, 0.05 * best_c)
    temp_min = 1e-6
    step_scale = 0.15
    total_proposals = 0
    accepted = 0
    
    while temp > temp_min:
        batch = 50
        for _ in range(batch):
            if time.perf_counter() - t0 > time_limit:
                break
            # Propose: random pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            direction = rng.random() < 0.5
            if direction:
                donor, recip = i, j
            else:
                donor, recip = j, i
            cap = min(cur_h[donor], 1.0 - cur_h[recip])
            if cap < 1e-10:
                continue
            d = cap * step_scale * (0.3 + 0.7 * rng.random())
            if d < 1e-12:
                continue
            cur_h[donor] -= d
            cur_h[recip] += d
            new_c = evaluate(cur_h)
            total_proposals += 1
            delta = new_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
                cur_c = new_c
                accepted += 1
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    best_h = cur_h.copy()
            else:
                cur_h[donor] += d
                cur_h[recip] -= d
        temp *= 0.92
        # Adapt step scale
        if total_proposals > 100:
            acc_rate = accepted / max(total_proposals, 1)
            if acc_rate < 0.1:
                step_scale *= 0.7
            elif acc_rate > 0.6:
                step_scale = min(step_scale * 1.3, 0.4)
            total_proposals = 0
            accepted = 0
    
    return best_h, best_c

def random_perturb_restart(h, n, s, rng, scale=0.02):
    """Generate a perturbed feasible candidate."""
    noise = rng.standard_normal(n) * scale
    noise -= np.mean(noise)
    cand = h + noise
    cand = project_box_sum(cand, s)
    return cand

def interpolate_to_n(h_from, n_to):
    """Interpolate h from its current length to n_to samples over [0,2]."""
    n_from = len(h_from)
    if n_from == n_to:
        return h_from.copy()
    x_from = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    x_to = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    h_interp = np.interp(x_to, x_from, np.concatenate([h_from, [h_from[0]]]), period=2.0)
    # Actually, for step functions, linear interpolation on centers is better
    h_interp = np.interp(x_to, x_from, h_from)
    s_to = n_to / 2.0
    h_interp = project_box_sum(h_interp, s_to)
    return h_interp

def run(seed=42, budget_s=1100, **kwargs):
    t_global = time.perf_counter()
    rng = np.random.default_rng(seed)
    
    # Get parent
    parent = None
    if 'initial_h_values' in dir() or True:
        try:
            pv = np.asarray(initial_h_values, dtype=np.float64).ravel()
            if pv.size > 0 and np.all(np.isfinite(pv)):
                n_p = len(pv)
                s_p = n_p / 2.0
                parent = project_box_sum(pv, s_p)
        except (NameError, AttributeError):
            parent = None
    
    if parent is None or len(parent) < 4:
        # Fallback: construct a simple feasible vector
        n_default = 47
        s_default = n_default / 2.0
        parent = 0.5 + 0.15 * np.cos(2 * np.pi * np.arange(n_default) / n_default)
        parent = project_box_sum(parent, s_default)
    
    n_main = len(parent)
    s_main = n_main / 2.0
    best_h = parent.copy()
    best_c = evaluate(best_h)
    
    # Quick return if budget is tiny
    if budget_s < 10:
        return best_h.tolist(), best_c, n_main
    
    # Phase 1: Smooth gradient descent on main resolution
    time_left = budget_s - (time.perf_counter() - t_global) - 10
    if time_left > 5:
        t_limit = time_left * 0.35
        start_tau = max(0.005, 0.08 * best_c)
        h1, c1 = smooth_descent(parent, n_main, s_main, start_tau, t_limit, rng)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    
    # Phase 2: Pair transfer on main resolution
    time_left = budget_s - (time.perf_counter() - t_global) - 10
    if time_left > 5:
        t_limit = time_left * 0.25
        h2, c2 = pair_transfer_search(best_h, n_main, s_main, t_limit, rng)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 3: SA on main resolution
    time_left = budget_s - (time.perf_counter() - t_global) - 10
    if time_left > 5:
        t_limit = min(time_left * 0.25, 120)
        h3, c3 = simulated_annealing(best_h, n_main, s_main, t_limit, rng)
        if c3 < best_c:
            best_h = h3
            best_c = c3
    
    # Phase 4: Multi-restart with perturbations
    time_left = budget_s - (time.perf_counter() - t_global) - 10
    n_restarts = 3
    if time_left > 20:
        per_restart = time_left / (n_restarts + 1)
        for ri in range(n_restarts):
            if time.perf_counter() - t_global > budget_s - 10:
                break
            cand = random_perturb_restart(best_h, n_main, s_main, rng, scale=0.01 + 0.01 * ri)
            cand_c = evaluate(cand)
            # Quick polish
            if per_restart > 10:
                h_r, c_r = pair_transfer_search(cand, n_main, s_main, per_restart * 0.5, rng)
                if c_r < best_c:
                    best_h = h_r
                    best_c = c_r
            elif cand_c < best_c:
                best_h = cand
                best_c = cand_c
    
    # Phase 5: Try alternative resolutions
    time_left = budget_s - (time.perf_counter() - t_global) - 10
    if time_left > 20 and best_c > 0.380:
        for n_try in [n_main * 2, max(8, n_main // 2)]:
            if time.perf_counter() - t_global > budget_s - 10:
                break
            if n_try < 8 or n_try > 300:
                continue
            s_try = n_try / 2.0
            h_try = interpolate_to_n(best_h, n_try)
            c_try = evaluate(h_try)
            if time.perf_counter() - t_global > budget_s - 10:
                continue
            t_lim = min(time_left * 0.3, 80)
            if t_lim > 5:
                h_try, c_try = pair_transfer_search(h_try, n_try, s_try, t_lim, rng)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try
    
    # Phase 6: Final polish with fine pair search
    time_left = budget_s - (time.perf_counter() - t_global) - 5
    if time_left > 5:
        n_f = len(best_h)
        s_f = n_f / 2.0
        h_f, c_f = pair_transfer_search(best_h, n_f, s_f, time_left, rng, max_passes=50)
        if c_f < best_c:
            best_h = h_f
            best_c = c_f
    
    # Final validation
    n_final = len(best_h)
    s_final = n_final / 2.0
    # Ensure feasibility
    if np.any(best_h < -1e-9) or np.any(best_h > 1 + 1e-9):
        best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, s_final)
    best_c = evaluate(best_h)
    
    return best_h.tolist(), best_c, n_final
