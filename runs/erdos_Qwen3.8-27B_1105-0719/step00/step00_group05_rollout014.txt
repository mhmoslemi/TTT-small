    lb = np.maximum(-h, -delta)
    ub = np.minimum(1.0 - h, delta)
    
    # Solve: min g^T d + 0.5*rho*||d||^2 s.t. sum(d)=0, lb <= d <= ub
    # KKT: g + rho*d + lambda*1 - mu_lower + mu_upper = 0
    # d_j = clip(-g_j/rho - lambda, lb_j, ub_j)
    
    def sum_d(lam):
        d = (-g / rho - lam)
        d = np.clip(d, lb, ub)
        return np.sum(d)
    
    # Bisection on lambda
    # sum_d is decreasing in lambda
    lo, hi = -10.0, 10.0
    # Find valid range
    while sum_d(lo) < 0:
        lo -= abs(lo) + 1
    while sum_d(hi) > 0:
        hi += abs(hi) + 1

        mid = (lo + hi) / 2
        if sum_d(mid) > 0:
            lo = mid
        else:
            hi = mid

            g[m:] -= w * h[:n-m]
    return g

def soft_obj_and_grad(h, beta):
    """Log-sum-exp smooth approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * dx * corr
    max_s = float(np.max(scaled))
    exp_s = np.exp(scaled - max_s)
    sum_exp = float(np.sum(exp_s))
    obj = (max_s + np.log(sum_exp)) / beta
    softmax_w = exp_s / sum_exp
    grad = compute_gradient_weighted(h, softmax_w)
    # Note: grad is for the weighted sum of corr, not for obj
    # obj = (1/beta) * log(sum exp(beta*dx*corr_p))
    # d(obj)/dh = (1/beta) * sum_p exp(beta*dx*corr_p) * dx * d(corr_p)/dh / sum exp(...)
    #            = dx * sum_p softmax_p * d(corr_p)/dh
    #            = dx * compute_gradient_weighted(h, softmax_w)
    grad = dx * grad
    return float(obj), grad

def solve_trust_region(g, h, delta, rho):
    """Solve min g^T d + 0.5*rho*||d||^2 s.t. sum(d)=0, lb<=d<=ub."""
    n = len(h)
    lb = np.maximum(-h, -delta)
    ub = np.minimum(1.0 - h, delta)
    
    def sum_at(lam):
        d = (-g / rho) - lam
        return float(np.sum(np.clip(d, lb, ub)))
    
    lo, hi = -50.0, 50.0
    # Expand
    while sum_at(lo) < 0:
        lo *= 2.0
        if abs(lo) > 1e6:
            break
    while sum_at(hi) > 0:
        hi *= 2.0
        if abs(hi) > 1e6:
            break
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if sum_at(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(-g / rho - lam, lb, ub)

def top_k_weights(corr, K, decay=0.2):
    n_corr = len(corr)
    K = min(K, n_corr)
    weights = np.zeros(n_corr)
    idx = np.argsort(corr)[::-1][:K]
    for rank, i in enumerate(idx):
        weights[i] = np.exp(-decay * rank)
    weights /= weights.sum()
    return weights

def sqp_refine(h0, time_limit=100.0, K=10, decay=0.15, delta_init=0.04, rho_init=0.5):
    """Trust-region SQP using top-K weights."""
    h = h0.copy()
    c_cur = exact_objective(h)
    t0 = time.time()
    delta = delta_init
    rho = rho_init
    stall = 0
    
    while time.time() - t0 < time_limit and delta > 1e-8:
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = top_k_weights(corr, K, decay)
        g = compute_gradient_weighted(h, w)
        if np.linalg.norm(g) < 1e-14:
            break
        
        d = solve_trust_region(g, h, delta, rho)
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_new = project_feasible(h + alpha * d)
            c_new = exact_objective(h_new)
            if c_new < c_cur - 1e-14:
                h = h_new
                c_cur = c_new
                improved = True
                delta = min(delta * 1.1, 0.2)
                break
        if not improved:
            delta *= 0.5
            stall += 1
            if stall > 10:
                break
            # Try random perturbation
            if stall % 3 == 0:
                rng = np.random.RandomState(stall * 12345)
                pert = rng.normal(0, delta, len(h))
                h = project_feasible(h + pert)
                c_cur = exact_objective(h)
    
    return h, c_cur

def smooth_optimize(h0, beta_start=50.0, beta_end=2000.0, n_anneal=8, time_per_beta=15.0):
    """Annealed log-sum-exp optimization."""
    h = h0.copy()
    log_betas = np.linspace(np.log(beta_start), np.log(beta_end), n_anneal)
    
    for bi in range(n_anneal):
        beta = float(np.exp(log_betas[bi]))
        t_beta_start = time.time()
        
        def obj(h_):
            o, _ = soft_obj_and_grad(h_, beta)
            return o
        
        def grad(h_):
            _, gr = soft_obj_and_grad(h_, beta)
            return gr
        
        def eq_fun(h_):
            return np.array([np.sum(h_) - len(h_) / 2.0])
        
        def eq_jac(h_):
            return np.ones((1, len(h_)))
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * len(h),
                          constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                          options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
            if res.success and is_feasible(res.x):
                h_new = project_feasible(res.x)
                c_new = exact_objective(h_new)
                c_old = exact_objective(h)
                if c_new <= c_old + 1e-10:
                    h = h_new
        except Exception:
            pass
        
        # Check time for this beta level
        if time.time() - t_beta_start > time_per_beta * 2:
            continue
    
    return h, exact_objective(h)

def make_perturbed(h0, seed, scale=0.02):
    """Create a perturbed feasible vector."""
    rng = np.random.RandomState(seed)
    d = rng.normal(0, scale, len(h0))
    return project_feasible(h0 + d)

def resize_h(h0, new_n):
    """Interpolate h0 to new_n points."""
    n0 = len(h0)
    if n0 == new_n:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x1, x0, h0)
    return project_feasible(h_new)
