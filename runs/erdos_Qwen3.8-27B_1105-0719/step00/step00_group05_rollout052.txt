def resize_h(h, n_new):
    """Resize h from its current length to n_new using interpolation."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_OPT_N = 0
_OPT_BETA = 1.0
_OPT_DEADLINE = 0.0
_BEST_H = None
_BEST_C = float('inf')


def exact_C(h):
    n = len(h)
    if n <= 1:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def stable_sigmoid(t):
    t = np.asarray(t, dtype=float)
    out = np.empty_like(t)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    e = np.exp(t[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def feasible_from_logit(y, n):
    target = n / 2.0
    y_max = np.max(y)
    y_min = np.min(y)
    lo = -y_max - 50.0
    hi = -y_min + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(stable_sigmoid(y + mid))
        if s > target:
            hi = mid
        else:
            lo = mid
    return stable_sigmoid(y + (lo + hi) / 2.0)


def logit_from_h(h, eps=1e-10):
    h_clip = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clip / (1.0 - h_clip))


def project_feasible(u, n):
    h = np.array(u, dtype=float).copy()
    target = n / 2.0
    lo = -h.max() - 2.0
    hi = 1.0 - h.min() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    delta = target - result.sum()
    if abs(delta) > 1e-13:
        if delta > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for idx in order:
            if delta > 0:
                room = 1.0 - result[idx]
                add = min(room, delta)
                result[idx] += add
                delta -= add
            else:
                room = result[idx]
                sub = min(room, -delta)
                result[idx] -= sub
                delta += sub
            if abs(delta) < 1e-15:
                break
    return result


def record_candidate(h, n):
    global _BEST_H, _BEST_C
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return
    h_proj = project_feasible(h, n)
    c = exact_C(h_proj)
    if np.isfinite(c) and c < _BEST_C:
        _BEST_C = c
        _BEST_H = h_proj.copy()


def smooth_obj_and_grad(y):
    n = _OPT_N
    beta = _OPT_BETA
    dx = 2.0 / n
    h = feasible_from_logit(y, n)

    num_dir = 1 + 2 * (n - 1)
    v = np.empty(num_dir)
    G = np.zeros(n)

    v[0] = dx * np.sum(h * (1.0 - h))
    G += dx * (1.0 - 2.0 * h)

    idx = 1
    for m in range(1, n):
        hp = h[:n - m]
        hm = h[m:]
        v[idx] = dx * np.sum(hp * (1.0 - hm))
        v[idx + 1] = dx * np.sum(hm * (1.0 - hp))
        G[:n - m] += dx * (1.0 - hm)
        G[m:] += dx * (-hp)
        G[:n - m] += dx * (-hm)
        G[m:] += dx * (1.0 - hp)
        idx += 2

    v_max = np.max(v)
    shifted = np.clip(beta * (v - v_max), -500, 500)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    if sum_exp < 1e-300 or not np.isfinite(sum_exp):
        w = np.zeros(num_dir)
        mask = np.isclose(v, v_max, atol=1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        else:
            w[0] = 1.0
    else:
        w = exp_v / sum_exp

    F = v_max + (1.0 / beta) * np.log(max(sum_exp, 1e-300))

    m_i = h * (1.0 - h)
    M = np.sum(m_i)
    if M < 1e-12:
        gw = np.zeros(n)
    else:
        B = np.dot(G, m_i) / M
        gw = m_i * (G - B)
        if not np.all(np.isfinite(gw)):
            gw = np.zeros(n)
        gw_max = np.max(np.abs(gw))
        if gw_max > 100.0:
            gw = gw / gw_max * 100.0

    if _OPT_DEADLINE > 0 and time.time() > _OPT_DEADLINE:
        pass

    c = F
    if c < _BEST_C:
        c_exact = exact_C(h)
        if c_exact < _BEST_C:
            _BEST_C = c_exact
            _BEST_H = h.copy()

    return float(F), gw


def try_bfgs(h_start, n, beta, max_iter, deadline):
    global _OPT_N, _OPT_BETA, _OPT_DEADLINE
    y = logit_from_h(h_start)
    y = np.clip(y, -30.0, 30.0)
    _OPT_N = n
    _OPT_BETA = beta
    _OPT_DEADLINE = deadline
    try:
        result = minimize(
            smooth_obj_and_grad,
            y,
            method='BFGS',
            jac=True,
            options={'maxiter': max_iter, 'gtol': 1e-5, 'disp': False}
        )
        h_end = feasible_from_logit(result.x, n)
    except Exception:
        h_end = h_start.copy()
    h_end = project_feasible(h_end, n)
    record_candidate(h_end, n)
    return h_end


def pair_polish(h, n, max_pairs=60, max_moves=20, deadline=0.0, seed=42):
    global _BEST_H, _BEST_C
    h = project_feasible(h, n).copy()
    c_cur = exact_C(h)
    n_evals = 0
    max_evals = 50000

    rng = np.random.default_rng(seed)

    for move_iter in range(max_moves):
        if time.time() >= deadline or n_evals > max_evals:
            break
        best_c = c_cur
        best_t = 0.0
        best_pair = None

        # Build pair list: active lag pairs + random
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top few lags
        top_lags = np.argsort(corr)[-8:]
        pairs = set()
        for tl in top_lags:
            k = tl - (n - 1)
            if 0 <= k < n:
                for i in range(n - k):
                    pairs.add((i, i + k))
            elif -n < k < 0:
                for i in range(n + k):
                    pairs.add((i - k, i))
        for _ in range(20):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a != b:
                pairs.add((min(a, b), max(a, b)))
        pair_list = list(pairs)
        rng.shuffle(pair_list)
        pair_list = pair_list[:max_pairs]

        for (a, b) in pair_list:
            if n_evals > max_evals or time.time() >= deadline:
                break
            lo_t = max(-h[a], h[b] - 1.0)
            hi_t = min(1.0 - h[a], h[b])
            if hi_t - lo_t < 1e-10:
                continue
            grid = np.linspace(lo_t, hi_t, 13)
            local_best_t = 0.0
            local_best_c = c_cur
            for t in grid:
                n_evals += 1
                h[a] += t
                h[b] -= t
                if h[a] < -1e-9 or h[a] > 1 + 1e-9 or h[b] < -1e-9 or h[b] > 1 + 1e-9:
                    h[a] -= t
                    h[b] += t
                    continue
                c_trial = exact_C(h)
                n_evals += 0
                if c_trial < local_best_c:
                    local_best_c = c_trial
                    local_best_t = t
                h[a] -= t
                h[b] += t
            if local_best_c < best_c - 1e-14:
                best_c = local_best_c
                best_t = local_best_t
                best_pair = (a, b)

        if best_pair is None:
            break

        a, b = best_pair
        lo_t = max(-h[a], h[b] - 1.0)
        hi_t = min(1.0 - h[a], h[b])
        width = max((hi_t - lo_t) / 13.0, 1e-5)
        fine_lo = max(lo_t, best_t - width)
        fine_hi = min(hi_t, best_t + width)
        fine_grid = np.linspace(fine_lo, fine_hi, 25)
        for t in fine_grid:
            n_evals += 1
            if time.time() >= deadline:
                break
            h[a] += t
            h[b] -= t
            if h[a] < -1e-9 or h[a] > 1 + 1e-9:
                h[a] -= t
                h[b] += t
                continue
            c_trial = exact_C(h)
            if c_trial < best_c:
                best_c = c_trial
                best_t = t
            h[a] -= t
            h[b] += t

        h[a] += best_t
        h[b] -= best_t
        h = project_feasible(h, n)
        c_cur = exact_C(h)
        record_candidate(h, n)

    return h, c_cur


def random_search(h0, n, n_iters, deadline, seed=0):
    global _BEST_H, _BEST_C
    h = h0.copy()
    c_cur = exact_C(h)
    rng = np.random.default_rng(seed)
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        scale = 0.05 * max(1.0 - it / float(n_iters), 0.1)
        h_trial = project_feasible(h + scale * rng.standard_normal(n), n)
        c_trial = exact_C(h_trial)
        if c_trial < c_cur:
            h = h_trial
            c_cur = c_trial
            record_candidate(h, n)
    return h, c_cur


def generate_starts(h_parent, n, rng):
    starts = [h_parent.copy()]
    starts.append(h_parent[::-1].copy())
    for s in [1, 2, n // 2, n // 3]:
        starts.append(np.roll(h_parent, s).copy())
    for amp in [0.08, 0.15, 0.25]:
        x = np.arange(n)
        pert = amp * np.sin(2 * np.pi * x / n)
        starts.append(project_feasible(h_parent + pert, n))
        pert2 = amp * np.cos(2 * np.pi * x / n)
        starts.append(project_feasible(h_parent + pert2, n))
        pert3 = amp * np.sin(4 * np.pi * x / n)
        starts.append(project_feasible(h_parent + pert3, n))
    for scale in [0.03, 0.07, 0.12, 0.2]:
        starts.append(project_feasible(h_parent + scale * rng.standard_normal(n), n))
    starts.append(np.full(n, 0.5))
    # Clip all
    valid = []
    for s in starts:
        sp = project_feasible(s, n)
        if np.all(sp >= -1e-10) and np.all(sp <= 1 + 0.01):
            valid.append(sp)
    return valid


def run(seed=42, budget_s=1100, **kwargs):
    global _BEST_H, _BEST_C, _OPT_DEADLINE

    start_time = time.time()
    deadline = start_time + 0.92 * budget_s

    # Get parent
    n_parent = 42
    h_parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h_parent = np.array(initial_h_values, dtype=float).copy()
            n_parent = len(h_parent)
            if h_parent.size == 0 or not np.all(np.isfinite(h_parent)):
                h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is None:
        h_parent = np.full(n_parent, 0.5)

    h_parent = project_feasible(h_parent, n_parent)
    c_parent = exact_C(h_parent)
    _BEST_C = c_parent
    _BEST_H = h_parent.copy()

    rng = np.random.default_rng(seed)

    # Generate diverse starts
    starts = generate_starts(h_parent, n_parent, rng)
    # Sort by exact C
    start_scores = sorted(range(len(starts)), key=lambda i: exact_C(starts[i]))
    starts_sorted = [starts[i] for i in start_scores]

    betas = [40.0, 80.0, 160.0, 320.0, 600.0]

    # Phase 1: BFGS from sorted starts, with chaining
    for s_idx in range(len(starts_sorted)):
        if time.time() >= deadline - 150:
            break
        if _BEST_C <= 0.3807:
            break
        h_cur = starts_sorted[s_idx].copy()
        for beta in betas:
            if time.time() >= deadline - 150:
                break
            if _BEST_C <= 0.3807:
                break
            max_iter = 120 if n_parent <= 50 else 60
            h_cur = try_bfgs(h_cur, n_parent, beta, max_iter, deadline)
        # Pair polish after BFGS chain
        if time.time() < deadline - 100:
            h_pol, c_pol = pair_polish(h_cur, n_parent, max_pairs=60,
                                       max_moves=15, deadline=deadline - 50,
                                       seed=seed + s_idx)
            record_candidate(h_pol, n_parent)
            h_cur = h_pol

    # Phase 2: More BFGS from current best with different betas
    if time.time() < deadline - 100 and _BEST_H is not None:
        for beta in [200.0, 500.0, 100.0]:
            if time.time() >= deadline - 80:
                break
            h_end = try_bfgs(_BEST_H, n_parent, beta, 100, deadline)
            if time.time() < deadline - 40:
                h_pol, _ = pair_polish(h_end, n_parent, max_pairs=80,
                                       max_moves=12, deadline=deadline - 20,
                                       seed=seed + 100 + int(beta))
                record_candidate(h_pol, n_parent)

    # Phase 3: Aggressive pair polish from best
    if time.time() < deadline - 30 and _BEST_H is not None:
        for pseed in range(5):
            if time.time() >= deadline - 10:
                break
            h_pol, _ = pair_polish(_BEST_H, n_parent, max_pairs=100,
                                   max_moves=25, deadline=deadline - 5,
                                   seed=seed + 200 + pseed)
            record_candidate(h_pol, n_parent)

    # Phase 4: Random search
    if time.time() < deadline - 20 and _BEST_H is not None:
        h_rs, _ = random_search(_BEST_H, n_parent, n_iters=500,
                                deadline=deadline - 5, seed=seed + 500)
        record_candidate(h_rs, n_parent)

    # Phase 5: Final pair polish
    if time.time() < deadline - 5 and _BEST_H is not None:
        h_final_pol, _ = pair_polish(_BEST_H, n_parent, max_pairs=120,
                                     max_moves=20, deadline=deadline,
                                     seed=seed + 999)
        record_candidate(h_final_pol, n_parent)

    # Final validation
    h_return = _BEST_H.copy() if _BEST_H is not None else h_parent.copy()
    h_return = project_feasible(h_return, n_parent)
    c_return = exact_C(h_return)

    if not np.isfinite(c_return) or c_return > c_parent:
        h_return = h_parent.copy()
        c_return = c_parent

    return h_return, float(c_return), n_parent
