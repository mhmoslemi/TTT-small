def project_to_box_sum(x, target):
    """Project x onto {0 <= x_i <= 1, sum(x) = target} via bisection on lambda."""
    x = np.array(x, dtype=float).copy()
    n = len(x)

    def _f(lam):
        return np.sum(np.clip(x - lam, 0.0, 1.0)) - target

    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if _f(mid) > 0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break

    lam = (lo + hi) / 2.0
    x_proj = np.clip(x - lam, 0.0, 1.0)

    diff = target - np.sum(x_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x_proj < 1.0 - 1e-12
        else:
            mask = x_proj > 1e-12
        if mask.any():
            x_proj[mask] += diff / mask.sum()
            x_proj = np.clip(x_proj, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(out * dx))


def lsq_value_and_grad(h, alpha, n):
    """Compute log-sum-exp surrogate value and gradient."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    out = np.correlate(h, one_minus_h, mode='full')
    s = out * dx

    max_s = np.max(s)
    exp_s = np.exp(alpha * (s - max_s))
    sum_exp = np.sum(exp_s)
    phi = max_s + np.log(sum_exp) / alpha
    w = exp_s / sum_exp

    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        k = r - (n - 1)
        # ds_r/dx_m: term (1-h[m-k]) where 0<=m-k<n
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:

        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] += w[r] * dx * (-h[lo2 + k:hi2 + k])

    return phi, grad
