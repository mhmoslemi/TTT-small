def project_to_box_simplex(x, mass):
    """Project x onto {z: 0 <= z_i <= 1, sum(z) = mass}"""
    n = len(x)
    # Use bisection on lambda for x - lambda clipped to [0,1]
    lo, hi = x.min() - 1, x.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1).sum()
        if s > mass:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    result = np.clip(x - lam, 0, 1)
    # Fix residual
    residual = mass - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            caps = np.maximum(0, 1 - result)
            cap_sum = caps.sum()
            if cap_sum > 1e-15:
                result += residual * caps / cap_sum
                result = np.minimum(result, 1.0)
        else:
            floors = np.maximum(0, result)
            floor_sum = floors.sum()
            if floor_sum > 1e-15:
                result += residual * floors / floor_sum
                result = np.maximum(result, 0.0)
    return result

def exact_score(h, n):

    return np.max(c) * dx

def compute_line_coeffs(h, u, n):

    q1 = np.correlate(u, one_minus_h, mode='full') - np.correlate(h, u, mode='full')
    q2 = -np.correlate(u, u, mode='full')
    return q0, q1, q2

            if nm > 1e-10:
                dirs.append(u / nm)
    
    # Subgradient
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = float(corr.max())
    act = np.where(corr > mv * 0.9)[0]
    if len(act) > 0:
        u = np.zeros(n)
        for k in act[:8]:
            s = k - (n - 1)
            for i in range(n):
                if 0 <= i + s < n:
                    u[i] += (1.0 - h[i+s])
                if s != 0 and 0 <= i - s < n:
                    u[i] -= h[i-s]
        u -= u.mean()
        nm = np.linalg.norm(u)
        if nm > 1e-10:
            dirs.append(-u / nm)
    
    for (ph, _) in pool[:5]:
        u = ph - h
        u -= u.mean()
        nm = np.linalg.norm(u)
        if nm > 1e-10:
            dirs.append(u / nm)
    
    return dirs
