from scipy.optimize import minimize, linprog
from scipy.interpolate import interp1d
import time

def compute_raw(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    raw = compute_raw(h)
    return float(dx * np.max(raw))

def project_feasible(x, target_sum):
    """Project x onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        clipped = np.clip(x - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x_proj = np.clip(x - mid, 0.0, 1.0)
    # Fix exact sum
    residual = target_sum - x_proj.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - x_proj)[::-1]
        else:
            idx = np.argsort(x_proj)
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - x_proj[i])
                x_proj[i] += add
                residual -= add
            else:
                rem = min(-residual, x_proj[i])
                x_proj[i] -= rem
                residual += rem
            if abs(residual) < 1e-12:
                break
    return x_proj

def smooth_objective_and_grad(h, T, penalty_weight=1e4):
    n = len(h)
    target_sum = n / 2.0
    raw = compute_raw(h)
    
    # Log-sum-exp
    raw_max = raw.max()
    shifted = (raw - raw_max) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = raw_max + T * np.log(np.sum(np.exp(shifted)))
    
    # Penalty for sum constraint
    sum_penalty = penalty_weight * (h.sum() - target_sum) ** 2
    val = log_sum + sum_penalty
    
    # Gradient of log-sum-exp
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for j in range(len(raw)):
        if weights[j] < 1e-15:
            continue
        m = j - (n - 1)
        if m >= 0:
            grad[:n-m] += weights[j] * (1.0 - h[m:])
            grad[m:] += weights[j] * (-h[:n-m])
        else:
            r = -m
            grad[r:] += weights[j] * (1.0 - h[:n-r])
            grad[:n-r] += weights[j] * (-h[r:])
    
    # Gradient of penalty
    grad += 2.0 * penalty_weight * (h.sum() - target_sum)
    
    return val, grad

def local_optimize(h0, n_temps=8, iters_per_temp=200, penalty_weight=1e4):
    """Run log-sum-exp smoothed optimization with decreasing temperature."""
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum).astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T_start = 0.05
    T_end = 0.005
    temps = np.geomspace(T_start, T_end, n_temps)
    
    for T in temps:
        result = minimize(
            lambda x: smooth_objective_and_grad(x, T, penalty_weight)[0],
            h,
            jac=lambda x: smooth_objective_and_grad(x, T, penalty_weight)[1],
            method='L-BFGS-B',
            bounds=[(0, 1)] * n,
            options={'maxiter': iters_per_temp, 'ftol': 1e-10, 'gtol': 1e-6}
        )
        h_new = project_feasible(result.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def resample_h(h, new_n):

    old_n = len(h)
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    x_old = (np.arange(old_n) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new

For coordinate descent: given all h[i] for i ≠ k, the objective is:
max_j raw[j](h) where raw[j] is linear in h[k].

So we're minimizing a piecewise linear function of h[k] (the max of linear functions). The minimum is at a "kink" point or at the boundary [0,1].

The kink points are where two of the raw[j] values are equal. For each pair (j1, j2):
raw[j1](h[k]) = raw[j2](h[k])
=> (raw[j1]_base + a1*h[k]) = (raw[j2]_base + a2*h[k])
=> h[k] = (raw[j2]_base - raw[j1]_base) / (a1 - a2)

where a1, a2 are the coefficients of h[k] in raw[j1] and raw[j2] respectively.

This is expensive to compute exactly (O((2n-1)^2) pairs), but we can approximate by just checking the active set.
