                amt = min(diff2, 1.0 - proj[idx])
                proj[idx] += amt
                diff2 -= amt
            else:
                amt = min(-diff2, proj[idx])
                proj[idx] -= amt
                diff2 += amt
            if abs(diff2) < 1e-13:
                break
    return proj

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_raw_lags(h):
    n = len(h)
    raw = np.zeros(2 * n - 1)
    for p in range(-(n - 1), n):
        idx = p + (n - 1)
        if p >= 0:
            raw[idx] = np.dot(h[:n - p], 1.0 - h[p:])
        else:
            q = -p
            raw[idx] = np.dot(h[q:], 1.0 - h[:n - q])
    return raw

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = compute_raw_lags(h)
    max_raw = np.max(raw)
    exps = np.exp((raw - max_raw) / tau)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for p_idx in range(2 * n - 1):
        w = weights[p_idx]
        if w < 1e-14:
            continue
        p = p_idx - (n - 1)
        if p >= 0:
            m = n - p
            grad[:m] += w * (1.0 - h[p:])
            grad[p:] += w * (-h[:m])
        else:
            q = -p
            m = n - q
            grad[q:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[q:])
    return dx * grad

def smooth_phase(h, tau0, max_steps, deadline):
    n = len(h)
    current = h.copy()
    current_c = eval_c5(current)
    best = current.copy()
    best_c = current_c
    tau = tau0
    alpha = 0.02
    no_improve = 0

    for step in range(max_steps):
        if time.perf_counter() > deadline:
            break
        g = smooth_grad(current, tau)
        g_tan = g - np.mean(g)
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-10:
            tau *= 0.7
            no_improve += 5
            if no_improve > 50:
                break
            continue

        cand = current - alpha * g_tan
        cand = project_box_sum(cand, n)
        cand_c = eval_c5(cand)

        if cand_c < current_c - 1e-12:
            current = cand
            current_c = cand_c
            if current_c < best_c:
                best = current.copy()
                best_c = current_c
            no_improve = 0
            alpha = min(alpha * 1.15, 0.08)
        else:
            no_improve += 1
            alpha *= 0.7
            if alpha < 0.002:
                tau *= 0.8
                alpha = 0.015
            if no_improve > 80:
                break

        if step % 80 == 0 and step > 0:
            tau = max(tau * 0.95, 0.001)

    return best, best_c

def pair_phase(h, max_passes, deadline, rng):
    n = len(h)
    current = h.copy()
    current_c = eval_c5(current)
    fractions = np.array([0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0])

    for path in range(max_passes):
        if time.perf_counter() > deadline:
            break
        improved = False
        pair_list = []
        for i in range(n):
            for j in range(i + 1, n):
                pair_list.append((i, j))
        rng.shuffle(pair_list)

        for (i, j) in pair_list:
            if time.perf_counter() > deadline:
                return current, current_c
            for sign in [1, -1]:
                donor, recv = (j, i) if sign == 1 else (i, j)
                cap = min(current[donor], 1.0 - current[recv])
                if cap < 1e-8:
                    continue
                best_d = None
                best_c_trial = current_c
                for frac in fractions:
                    d = frac * cap
                    if d < 1e-9:
                        continue
                    trial = current.copy()
                    trial[donor] -= d
                    trial[recv] += d
                    if trial[donor] < -1e-12 or trial[recv] > 1.0 + 1e-12:
                        continue
                    c = eval_c5(trial)
                    if c < best_c_trial - 1e-12:
                        best_c_trial = c
                        best_d = d
                if best_d is not None:
                    current[donor] -= best_d
                    current[recv] += best_d
                    current_c = best_c_trial
                    improved = True

    return current, current_c

def sa_phase(h, max_steps, deadline, seed):
    n = len(h)
    current = h.copy()
    current_c = eval_c5(current)
    best = current.copy()
    best_c = current_c
    rng = np.random.default_rng(seed)
    temp = current_c * 0.03
    scale = 0.04

    for step in range(max_steps):
        if time.perf_counter() > deadline:
            break
        if step % 100 == 0:
            temp = max(temp * 0.97, 1e-8)
            scale = max(scale * 0.98, 0.005)

        if rng.random() < 0.65:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if rng.random() < 0.5:
                donor, recv = i, j
            else:
                donor, recv = j, i
            cap = min(current[donor], 1.0 - current[recv])
            if cap < 1e-7:
                continue
            d = rng.uniform(0.01, 1.0) * cap
            trial = current.copy()
            trial[donor] -= d
            trial[recv] += d
        elif rng.random() < 0.5:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            k = rng.integers(0, n)
            if k in (i, j):
                continue
            d1 = rng.uniform(0.01, 0.5) * min(current[i], 1.0 - current[k])
            d2 = rng.uniform(0.01, 0.5) * min(current[j], 1.0 - current[k])
            trial = current.copy()
            trial[i] -= d1
            trial[j] -= d2
            trial[k] += d1 + d2
            if trial[i] < -1e-12 or trial[j] < -1e-12 or trial[k] > 1.0 + 1e-12:
                continue
        else:
            noise = rng.standard_normal(n) * scale
            trial = project_box_sum(current + noise, n)

        c = eval_c5(trial)
        delta = c - current_c
        if delta <= 1e-12 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            current = trial
            current_c = c
            if c < best_c:
                best = current.copy()
                best_c = c

    return best, best_c

def perturb_restart(h, rng, magnitude):
    n = len(h)
    noise = rng.standard_normal(n) * magnitude
    cand = project_box_sum(h + noise, n)
    return cand

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline_hard = t0 + budget_s

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
    except (NameError, AttributeError):
        parent = None

    if parent is None or len(parent) < 8:
        n = 47
        k = np.arange(n)
        parent = 0.5 + 0.15 * np.cos(2 * np.pi * 3 * k / n) + 0.08 * np.sin(2 * np.pi * 7 * k / n)
        parent = project_box_sum(parent, n)

    n = len(parent)
    parent = project_box_sum(parent, n)
    parent_c = eval_c5(parent)

    best_h = parent.copy()
    best_c = parent_c
    rng_main = np.random.default_rng(seed)

    # Phase 1: Smooth gradient from parent
    dl = t0 + budget_s * 0.2
    h1, c1 = smooth_phase(parent, tau0=0.012, max_steps=5000, deadline=dl)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Pair polish
    dl = t0 + budget_s * 0.35
    h2, c2 = pair_phase(best_h, max_passes=4, deadline=dl, rng=np.random.default_rng(seed + 1))
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: SA
    dl = t0 + budget_s * 0.5
    h3, c3 = sa_phase(best_h, max_steps=5000, deadline=dl, seed=seed + 2)
    if c3 < best_c:
        best_h, best_c = h3, c3

    # Phase 4: Smooth again
    dl = t0 + budget_s * 0.62
    h4, c4 = smooth_phase(best_h, tau0=0.008, max_steps=5000, deadline=dl)
    if c4 < best_c:
        best_h, best_c = h4, c4

    # Phase 5: Pair polish again
    dl = t0 + budget_s * 0.72
    h5, c5 = pair_phase(best_h, max_passes=5, deadline=dl, rng=np.random.default_rng(seed + 5))
    if c5 < best_c:
        best_h, best_c = h5, c5

    # Phase 6: Restart from perturbed parent
    dl = t0 + budget_s * 0.82
    for restart in range(3):
        if time.perf_counter() > deadline_hard - 60:
            break
        perturb = perturb_restart(best_h, rng_main, 0.015)
        hp, cp = smooth_phase(perturb, tau0=0.01, max_steps=2000, deadline=dl)
        hp, cp = pair_phase(hp, max_passes=3, deadline=dl, rng=np.random.default_rng(seed + 10 + restart))
        if cp < best_c:
            best_h, best_c = hp, cp

    # Phase 7: Final SA
    dl = t0 + budget_s * 0.92
    h7, c7 = sa_phase(best_h, max_steps=3000, deadline=dl, seed=seed + 42)
    if c7 < best_c:
        best_h, best_c = h7, c7

    # Phase 8: Final pair polish
    dl = t0 + budget_s * 0.97
    h8, c8 = pair_phase(best_h, max_passes=4, deadline=dl, rng=np.random.default_rng(seed + 99))
    if c8 < best_c:
        best_h, best_c = h8, c8

    # Try coarser resolution
    remaining = deadline_hard - time.perf_counter()
    if remaining > 40:
        n_coarse = max(16, (n * 2) // 3)
        if n_coarse % 2 == 0 and n_coarse < n:
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_coarse)
            h_coarse = np.interp(x_new, x_old, best_h)
            h_coarse = project_box_sum(h_coarse, n_coarse)
            hc = eval_c5(h_coarse)
            dl2 = t0 + budget_s * 0.99
            h_coarse, hc = smooth_phase(h_coarse, tau0=0.02, max_steps=1500, deadline=dl2)
            h_coarse, hc = pair_phase(h_coarse, max_passes=3, deadline=dl2, rng=np.random.default_rng(seed + 200))
            if hc < best_c:
                best_h, best_c = h_coarse, hc

    # Final validation
    best_h = project_box_sum(best_h, len(best_h))
    final_c = eval_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h.copy(), best_c, len(best_h)
