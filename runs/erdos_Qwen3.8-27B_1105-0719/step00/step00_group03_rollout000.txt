from scipy.signal import correlate

_pbag = {}

def _constraint_fun(x):
    return np.sum(x) - _pbag['target']

def _constraint_jac(x):
    return np.ones(_pbag['n'])

def _obj_fun(x):
    return _lsq_obj_and_grad(x, _pbag['alpha'], _pbag['n'])[0]

def _grad_fun(x):
    return _lsq_obj_and_grad(x, _pbag['alpha'], _pbag['n'])[1]

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    vh = 1.0 - h
    corr = correlate(h, vh, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, n):
    """Euclidean projection onto {0<=x_i<=1, sum(x_i)=n/2}."""
    target = n / 2.0
    s = np.sum(x)
    if abs(s - target) < 1e-14:
        return np.clip(x, 0.0, 1.0)
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if np.sum(y) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def _lsq_obj_and_grad(x, alpha, n):
    """Log-sum-exp smoothed max correlation and its gradient."""
    h = x
    vh = 1.0 - h
    dx = 2.0 / n
    corr = correlate(h, vh, mode='full') * dx
    n_lags = 2 * n - 1

    F = np.max(corr)
    shifted = alpha * (corr - F)
    m_s = np.max(shifted)
    exps = np.exp(shifted - m_s)
    s_exp = np.sum(exps)
    obj = F + (m_s + np.log(s_exp)) / alpha

    weights = exps / s_exp
    grad = np.zeros(n)

    for r in range(n_lags):
        l = r - (n - 1)
        w = weights[r] * dx
        lo1 = max(0, l)
        hi1 = min(n - 1, n - 1 + l)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - l:hi1 - l + 1])
        lo2 = max(0, -l)
        hi2 = min(n - 1, n - 1 - l)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += w * (-h[lo2 + l:hi2 + l + 1])

    return obj, grad

    _pbag['n'] = n
    _pbag['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n

    res = minimize(_obj_fun, h0, jac=_grad_fun, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res
