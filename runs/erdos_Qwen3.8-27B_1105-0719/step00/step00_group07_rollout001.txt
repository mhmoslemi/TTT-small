import numpy as np
from scipy.optimize import minimize
import time

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -30, 30)))

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lse_value_grad_h(h, beta):
    corr = get_corr(h)
    n = len(h)
    M = np.max(corr)
    shifted = beta * (corr - M)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    value = M + np.log(np.sum(exp_vals)) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-13:
            continue
        l = k - (n - 1)
        w = weights[k]
        if l >= 0:
            grad[l:] += w * (1.0 - h[:n - l])
            grad[:n - l] += w * (-h[l:])
        else:
            q = -l
            grad[q:] += w * (1.0 - h[:n - q])
            grad[:n - q] += w * (-h[q:])
    return value, grad

def theta_to_h(theta, target_sum, c_prev=None):
    if c_prev is None:
        c = 0.0
    else:
        c = c_prev
    for _ in range(20):
        h = sigmoid(theta + c)
        f = np.sum(h) - target_sum
        fp = np.sum(h * (1.0 - h))
        if fp < 1e-15:
            break
        c_new = c - f / fp
        if abs(c_new - c) < 1e-14:
            c = c_new
            break
        c = np.clip(c_new, -50, 50)
    h = sigmoid(theta + c)
    return h, c

def h_to_theta(h):
    eps = 1e-7
    hc = np.clip(h, eps, 1.0 - eps)
    theta = np.log(hc / (1.0 - hc))
    theta -= np.mean(theta)
    return theta

def grad_h_to_grad_theta(grad_h, h):
    s = h * (1.0 - h)
    S = np.sum(s)
    if S < 1e-15:
        return np.zeros_like(h)
    return s * grad_h - (s / S) * np.dot(s, grad_h)

def project_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, target_sum)

def exact_coordinate_descent(h, target_sum, rng, max_time, best_c5):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    sizes = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    sweeps = 0
    while time.time() - t0 < max_time:
        improved = False
        for j in range(n):
            if time.time() - t0 > max_time:
                break
            for size in sizes:
                for sign in [1, -1]:
                    h_new = h.copy()
                    h_new[j] += sign * size
                    if h_new[j] < 0 or h_new[j] > 1:
                        continue
                    # Adjust sum: spread difference to random other coord
                    diff = target_sum - np.sum(h_new)
                    if abs(diff) < 1e-14:
                        pass
                    else:
                        k = rng.integers(0, n)
                        while k == j:
                            k = rng.integers(0, n)
                        h_new[k] -= diff
                        if h_new[k] < -0.01 or h_new[k] > 1.01:
                            continue
                        h_new = np.clip(h_new, 0.0, 1.0)
                        # Re-fix sum approximately
                        diff2 = target_sum - np.sum(h_new)
                        if abs(diff2) > 1e-10:
                            continue
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        h = h_new.copy()
                        best_c5 = c5_new
                        improved = True
                        # Shrink remaining sizes for this coord
                        break
        sweeps += 1
        if not improved and sweeps > 5:
            break
    return h, best_c5

def generate_candidates(n, target_sum, rng, seed):
    """Generate diverse starting candidates algorithmically."""
    candidates = []
    x = np.linspace(0, 2, n)
    # Cosine families
    for a in [0.2, 0.35, 0.5, 0.65, 0.8]:

            for f in [1, 2, 3]:
                h = 0.5 + a * np.cos(np.pi * f * x / 2.0 + phase)
                h = make_feasible(h, target_sum)
                candidates.append(h)
    # Step functions
    for pos1 in np.linspace(0.1, 0.9, 5):
        for pos2 in np.linspace(0.1, 0.9, 5):
            if abs(pos1 - pos2) < 0.1:
                continue
            if pos1 > pos2:
                pos1, pos2 = pos2, pos1
            h = np.zeros(n)
            h[x < pos1] = 1.0
            h[(x >= pos1) & (x < pos2)] = 0.3
            h[x >= pos2] = 0.8
            # Scale to meet constraint
            h = make_feasible(h, target_sum)
            candidates.append(h)
    # Random smooth

            h += am * np.sin(np.pi * fr * x + ph)
        h = make_feasible(h, target_sum)
        candidates.append(h)
    return candidates
