    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    need = target - cur_sum
    
    if abs(need) < 1e-14:
        return h
    
    # Bisection on uniform shift
    if need > 0:
        lo, hi = -1e-15, 1.0
    else:
        lo, hi = -1.0, 1e-15
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        shifted = np.clip(h + mid, 0.0, 1.0)
        if np.sum(shifted) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in order:
            if diff > 0:
                adj = min(diff, 1.0 - result[i])
            else:
                adj = max(diff, -result[i])
            result[i] += adj
            diff -= adj
            if abs(diff) < 1e-13:
                break
    
    return result

def pair_optimize(h, p, q, dx, n_grid=50, n_golden=30):
    """Optimize pair (p,q) in-place. Returns new score if improved, else None."""
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return None
    
    old_p, old_q = h[p], h[q]
    
    # Grid search
    grid = np.linspace(lo, hi, n_grid)
    best_score = np.inf
    best_idx = 0
    
    for i in range(n_grid):
        a = grid[i]
        h[p] = a
        h[q] = s - a
        sc = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if sc < best_score:
            best_score = sc
            best_idx = i
    
    # Restore
    h[p] = old_p
    h[q] = old_q
    
    current_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    if best_score >= current_score - 1e-14:
        return None
    
    # Golden section in neighborhood of best grid point
    spread = max(2, n_grid // 15)
    a_lo = grid[max(0, best_idx - spread)]
    a_hi = grid[min(n_grid - 1, best_idx + spread)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_golden):
        a1 = a_lo + gr * (a_hi - a_lo)
        a2 = a_hi - gr * (a_hi - a_lo)
        
        h[p] = a1
        h[q] = s - a1
        s1 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        
        h[p] = a2
        h[q] = s - a2
        s2 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        
        if s1 <= s2:
            a_hi = a2
        else:
            a_lo = a1
    
    h[p] = old_p
    h[q] = old_q
    
    a_star = (a_lo + a_hi) / 2.0
    h[p] = a_star
    h[q] = s - a_star
    final_score = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    
    if final_score < current_score - 1e-14:
        return final_score
    else:
        h[p] = old_p
        h[q] = old_q
        return None
