    h = np.clip(h, 0, 1).astype(float)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute diff evenly
    if diff > 0:

        mask = h < 1.0
        count = mask.sum()
        if count == 0:
            return h

def binary_all_swaps(b, n):

    zeros = np.where(b == 0)[0]

            base[i] = 0
            base[j] = 1
            s = eval_c5(base)
            base[j] = 0
            base[i] = 1
            results.append((s, i, j))

def binary_steepest(b, n, max_rounds=500, time_out=30.0):

    best_score = eval_c5(b)
    start = time.time()

            if s < best_sc:
                best_h, best_sc = b.copy(), s
            # Follow up with steepest from SA result
            rem2 = deadline - time.time()
            if rem2 > 10:
                dl2 = time.time() + min(15.0, rem2 * 0.5)
                b2, s2 = bin_steepest(b, best_n, 3000, dl2)
                if s2 < best_sc:
                    best_h, best_sc = b2, s
