```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(y, target):

    n = len(y)

            x[m] += res / m.sum()
            x = np.clip(x, 0.0, 1.0)
    # Final fix
    res = target - x.sum()
    if abs(res) > 1e-15:
        idx = int(np.argmax(x)) if res > 0 else int(np.argmin(x))
        x[idx] += res
    return x

def lse_grad(h, corr, weights):
    """Compute gradient of LSE of corr values w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        p = k - (n - 1)
        wk = weights[k]
        if wk < 1e-14:
            continue
        i_lo = max(0, -p)
        i_hi = min(n - 1, n - 1 - p)
        for i in range(i_lo, i_hi + 1):
            j = i + p
            grad[i] += dx * wk * (1.0 - h[j])
            grad[j] -= dx * wk * h[i]
    return grad

def phase_lse(h, time_limit, rng):
    """LSE smoothed gradient descent."""
    n = len(h)
    target = n * 0.5
    h = project_box_sum(h, target)
    c_best = evaluate_c5(h)
    h_best = h.copy()
    
    tau = 0.02
    alpha = 0.05
    t0 = time.perf_counter()
    stall = 0
    
    while time.perf_counter() - t0 < time_limit:
        if tau < 1e-7:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        cmax = np.max(corr)
        exps = np.exp((corr - cmax) / tau)
        w = exps / exps.sum()
        
        g = lse_grad(h, corr, w)
        g -= g.mean()  # project to sum=0 direction
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            stall += 5
        else:
            d = -g / gn
            accepted = False
            a = alpha
            for _ in range(15):
                cand = project_box_sum(h + a * d, target)
                cc = evaluate_c5(cand)
                if cc < c_best - 1e-11:
                    h = cand
                    c_best = cc
                    h_best = h.copy()
                    accepted = True
                    break
                a *= 0.5
            if not accepted:
                stall += 1
                # Occasional perturbation
                if stall % 5 == 0:
                    h = project_box_sum(h + 0.02 * rng.standard_normal(n), target)
                    cc = evaluate_c5(h)
                    if cc < c_best:
                        c_best = cc
                        h_best = h.copy()
        
        if stall > 40:
            tau *= 0.7
            stall = 0
            alpha = max(alpha * 0.9, 0.005)
    
    return h_best, c_best

def phase_pairs(h, time_limit, rng):
    """Pair transfer coordinate descent."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    c_best = evaluate_c5(h)
    h_best = h.copy()
    
    t0 = time.perf_counter()
    fracs = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.5])
    no_improve_passes = 0
    
    while time.perf_counter() - t0 < time_limit:
        improved = False
        # Shuffle pairs for diversity
        indices = list(range(n))
        rng.shuffle(indices)
        
        for ii in range(n):
            i = indices[ii]
            for jj in range(ii + 1, n):
                j = indices[jj]
                if time.perf_counter() - t0 > time_limit:
                    return h_best, c_best
                
                # i -> j
                cap = min(h[i], 1.0 - h[j])
                if cap > 1e-6:
                    for f in fracs:
                        d = cap * f
                        if d < 1e-7:
                            continue
                        h[i] -= d
                        h[j] += d
                        cn = evaluate_c5(h)
                        if cn < c_best - 1e-11:
                            c_best = cn
                            h_best = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                
                # j -> i
                cap = min(h[j], 1.0 - h[i])
                if cap > 1e-6:
                    for f in fracs:
                        d = cap * f
                        if d < 1e-7:
                            continue
                        h[j] -= d
                        h[i] += d
                        cn = evaluate_c5(h)
                        if cn < c_best - 1e-11:
                            c_best = cn
                            h_best = h.copy()
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break
        else:
            no_improve_passes = 0
    
    return h_best, c_best

def phase_sa(h, time_limit, rng):
    """Simulated annealing."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    c_cur = evaluate_c5(h)
    c_best = c_cur
    h_best = h.copy()
    temp = 0.003
    scale = 0.03
    t0 = time.perf_counter()
    it = 0
    
    while time.perf_counter() - t0 < time_limit:
        it += 1
        if it % 200 == 0:
            temp = max(temp * 0.95, 1e-7)
        
        if rng.random() < 0.6:
            # Pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if rng.random() < 0.5:
                d = rng.uniform(0.002, 0.1) * min(h[i], 1.0 - h[j])
            else:
                d = -rng.uniform(0.002, 0.1) * min(h[j], 1.0 - h[i])
            if abs(d) < 1e-9:
                continue
            h[i] += d
            h[j] -= d
            # Check bounds (pair move should preserve them if d is clamped)
            if h[i] < -1e-12 or h[i] > 1 + 1e-12 or h[j] < -1e-12 or h[j] > 1 + 1e-12:
                h[i] -= d
                h[j] += d
                continue
            cn = evaluate_c5(h)
            if cn <= c_cur or rng.random() < np.exp(-(cn - c_cur) / temp):
                c_cur = cn
                if cn < c_best - 1e-11:
                    c_best = cn
                    h_best = h.copy()
            else:
                h[i] -= d
                h[j] += d
        else:
            # Projected random step
            noise = scale * rng.standard_normal(n)
            cand = project_box_sum(h + noise, target)
            cn = evaluate_c5(cand)
            if cn <= c_cur or rng.random() < np.exp(-(cn - c_cur) / temp):
                h = cand
                c_cur = cn
                if cn < c_best - 1e-11:
                    c_best = cn
                    h_best = h.copy()
    
    return h_best, c_best
```

```python
def compute_grad(h, weights, dx):
    N = len(h)
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        p = i - (N - 1)
        wi = weights[i]
        if wi < 1e-14:
            continue
        # First term: t from max(0,p) to min(N-1, N-1+p)
        t_lo1 = max(0, p)
        t_hi1 = min(N - 1, N - 1 + p)
        if t_lo1 <= t_hi1:
            t1 = np.arange(t_lo1, t_hi1 + 1)
            m1 = t1 - p  # = n values, 0<=m1<N
            grad[t1] += dx * wi * (1.0 - h[m1])
        
        # Second term: t from max(0,-p) to min(N-1, N-1-p)
        t_lo2 = max(0, -p)
        t_hi2 = min(N - 1, N - 1 - p)
        if t_lo2 <= t_hi2:
            t2 = np.arange(t_lo2, t_hi2 + 1)
            m2 = t2 + p  # = n+p values, 0<=m2<N
            grad[t2] -= dx * wi * h[m2]
    
    return grad
```
