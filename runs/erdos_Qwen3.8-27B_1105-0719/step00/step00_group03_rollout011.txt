def adam_polish(h0, n, dx, target, alpha, n_iter=20000, lr=0.005):
    h = h0.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2 = 0.9, 0.999
    best = compute_c5(h)
    best_h = h.copy()
    for t in range(n_iter):
        _, grad = lse_obj_and_grad(h, alpha, dx)
        m1 = b1 * m1 + (1.0 - b1) * grad
        m2 = b2 * m2 + (1.0 - b2) * grad * grad
        m1_hat = m1 / (1.0 - b1 ** (t + 1))
        m2_hat = m2 / (1.0 - b2 ** (t + 1))
        h_new = h - lr * m1_hat / (np.sqrt(m2_hat) + 1e-8)
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best:
            best = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best

def quasiperiodic_init(n, seed_offset=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = 2.0 * (np.arange(n) + 0.5) / n
    h = np.where(np.mod(x * phi * 2.0 + seed_offset, 1.0) < 0.5, 1.0, 0.0)
    return h

def interpolate_h(h_src, n_src, n_dst):
    x_src = 2.0 * (np.arange(n_src) + 0.5) / n_src
    x_dst = 2.0 * (np.arange(n_dst) + 0.5) / n_dst
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = budget_s - 30

    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n

    # Phase 1: SLSQP with increasing alpha from parent
    alphas_main = [5, 10, 20, 50, 100, 200, 500, 1000, 3000]
    for alpha in alphas_main:
        if time.time() - t0 > deadline:
            break
        for attempt in range(3):
            if time.time() - t0 > deadline:
                break
            if attempt == 0:
                h0 = best_h.copy()
            else:
                h0 = best_h + rng.uniform(-0.02, 0.02, n)
                h0 = project_feasible(h0, target)
            try:
                h_res = try_slsqp(h0, n, dx, target, alpha, maxiter=800)
                h_res = project_feasible(h_res, target)
                c5 = compute_c5(h_res)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_res.copy()
            except Exception:
                pass

    # Phase 2: Adam polishing on current best
    if time.time() - t0 < deadline:
        for alpha in [200, 1000, 5000]:
            if time.time() - t0 > deadline:
                break
            h_pol, c5_pol = adam_polish(best_h, n, dx, target, alpha,
                                         n_iter=15000, lr=0.003)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 3: Try different n values
    for n_try in [30, 36, 40, 44, 48, 54, 60, 80]:
        if time.time() - t0 > deadline:
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0

        h_init = interpolate_h(best_h, best_n, n_try)
        h_init = project_feasible(h_init, target_t)

        for alpha in [50, 200, 1000]:
            if time.time() - t0 > deadline:
                break
            try:
                h_res = try_slsqp(h_init, n_try, dx_t, target_t, alpha, maxiter=500)
                h_res = project_feasible(h_res, target_t)
                c5 = compute_c5(h_res)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_res.copy()
                    best_n = n_try
            except Exception:
                pass

        # Adam polish on this n
        if time.time() - t0 < deadline:
            h_pol, c5_pol = adam_polish(h_init, n_try, dx_t, target_t, 500,
                                         n_iter=10000, lr=0.003)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_try

    # Phase 4: Random restarts with quasiperiodic inits
    for r in range(15):
        if time.time() - t0 > deadline:
            break
        h_qp = quasiperiodic_init(best_n, seed_offset=r * 0.1)
        h_qp = project_feasible(h_qp, best_n / 2.0)
        dx_b = 2.0 / best_n
        target_b = best_n / 2.0
        try:
            h_res = try_slsqp(h_qp, best_n, dx_b, target_b, 200, maxiter=500)
            h_res = project_feasible(h_res, target_b)
            c5 = compute_c5(h_res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_res.copy()
        except Exception:
            pass

    # Phase 5: Final SLSQP + Adam polish on the best
    if time.time() - t0 < deadline:
        dx_b = 2.0 / best_n
        target_b = best_n / 2.0
        for alpha in [1000, 5000, 20000]:
            if time.time() - t0 > deadline:
                break
            try:
                h_res = try_slsqp(best_h, best_n, dx_b, target_b, alpha, maxiter=500)
                h_res = project_feasible(h_res, target_b)
                c5 = compute_c5(h_res)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_res.copy()
            except Exception:
                pass

    if time.time() - t0 < deadline:
        h_pol, c5_pol = adam_polish(best_h, best_n, 2.0 / best_n, best_n / 2.0,
                                     5000, n_iter=20000, lr=0.002)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5, best_n
