    return corr * dx

    scores = compute_scores(x)
    F = np.max(scores)
    shifted = alpha * (scores - F)
    # For numerical stability
    log_sum = np.log(np.sum(np.exp(shifted)))
    return F + log_sum / alpha

def logsumexp_gradient(x, alpha):
    n = len(x)
    scores = compute_scores(x)
    F = np.max(scores)
    shifted = alpha * (scores - F)
    max_shifted = np.max(shifted)  # should be 0 since we subtracted F
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    dx = 2.0 / n
    q = 1.0 - x
    grad = np.zeros(n)

        if w < 1e-14:

    n_best = len(best_h)
    n_restarts = 20
    for i in range(n_restarts):
        if get_elapsed() > budget_s - 15:
            break
        scale = 0.03 + 0.04 * (i % 5)
        pert = rng.randn(n_best) * scale
        current_h = project_feasible(best_h + pert, n_best)
        alpha_r = 200 + 200 * (i % 4)
        obj = ObjFunc(alpha_r)
        cons = ConsFunc(n_best / 2.0)
        bounds = [(0.0, 1.0)] * n_best
        try:
            result = minimize(
                obj, current_h,
                jac=obj.gradient,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons.fun, 'jac': cons.jac}],
                options={'maxiter': 400, 'ftol': 1e-15}
            )
            candidate = project_feasible(result.x, n_best)
            c5_cand = exact_c5(candidate)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = candidate.copy()
        except Exception:
            pass

    # Try different n values
    for n_try in [30, 50, 60, 45]:
        if get_elapsed() > budget_s - 30:
            break
        if n_try == len(best_h):
            continue
        n_pts_old = len(best_h)
        x_old = np.linspace(0.0, 2.0, n_pts_old, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n_try)
        for alpha_t in [50, 200, 500]:
            if get_elapsed() > budget_s - 15:
                break
            obj = ObjFunc(alpha_t)
            cons = ConsFunc(n_try / 2.0)
            bounds = [(0.0, 1.0)] * n_try
            try:
                result = minimize(
                    obj, h_try,
                    jac=obj.gradient,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': cons.fun, 'jac': cons.jac}],
                    options={'maxiter': 300, 'ftol': 1e-15}
                )
                candidate = project_feasible(result.x, n_try)
                c5_cand = exact_c5(candidate)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = candidate.copy()
            except Exception:
                pass

    # Final polish with high alpha on best solution
    if get_elapsed() < budget_s - 10:
        n_f = len(best_h)
        for alpha_f in [1000, 5000, 10000]:
            if get_elapsed() > budget_s - 5:
                break
            obj = ObjFunc(alpha_f)
            cons = ConsFunc(n_f / 2.0)
            bounds = [(0.0, 1.0)] * n_f
            try:
                result = minimize(
                    obj, best_h,
                    jac=obj.gradient,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': cons.fun, 'jac': cons.jac}],
                    options={'maxiter': 500, 'ftol': 1e-16}
                )
                candidate = project_feasible(result.x, n_f)
                c5_cand = exact_c5(candidate)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = candidate.copy()
            except Exception:
                pass

    final_c5 = exact_c5(best_h)
    return (best_h, final_c5, len(best_h))
