def c5_value(h):

    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

    """Log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    M = np.max(corr)
    # For numerical stability
    exp_vals = np.exp(alpha * (corr - M))
    log_sum = M + np.log(np.sum(exp_vals)) / alpha
    return dx * log_sum

def lse_gradient(h, alpha):
