```python
def repair_sum(h, target_sum):
    """Adjust h to have exact target_sum while staying in [0,1]"""
    h = np.clip(h, 0.0, 1.0)

        space = h
        total_space = space.sum()
        if total_space < 1e-12:
            return h
        h += diff * space / total_space
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment for any residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        # Add/subtract from a single element
        if residual > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def line_search_pair(h, base_corr, i, j, dx):
    """Given current h and base correlation, find best t for direction e_i - e_j"""
    n = len(h)
    # Valid t range
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi - 1e-12:
        return 0.0, float('inf')
    
    # Compute linear coefficient
    # lin[p] for p in 0..2n-2
    lin = np.zeros(2*n - 1)
    
    # correlate(d, 1-h, 'full')[p] where d = e_i - e_j
    # = (1-h)[i + p - n + 1] - (1-h)[j + p - n + 1] (valid indices)
    for p in range(2*n - 1):
        idx1 = i + p - n + 1
        idx2 = j + p - n + 1
        val = 0.0
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])
        if 0 <= idx2 < n:
            val -= (1.0 - h[idx2])
        lin[p] = val
    
    # correlate(h, d, 'full')[p] where d = e_i - e_j
    # = h[i - p + n - 1] - h[j - p + n - 1] (valid indices)
    for p in range(2*n - 1):
        idx1 = i - p + n - 1
        idx2 = j - p + n - 1
        val = 0.0
        if 0 <= idx1 < n:
            val -= h[idx1]  # subtract because lin = correlate(d,1-h) - correlate(h,d)
        if 0 <= idx2 < n:
            val += h[idx2]
        lin[p] += val
    
    # Quadratic coefficient: -correlate(d,d)
    p_center = n - 1
    p_pos = j - i + n - 1
    p_neg = i - j + n - 1
    quad_entries = [(p_center, -2.0)]
    if 0 <= p_pos < 2*n - 1 and p_pos != p_center:
        quad_entries.append((p_pos, 1.0))
    if 0 <= p_neg < 2*n - 1 and p_neg != p_center:
        quad_entries.append((p_neg, 1.0))
    # Handle case where p_pos == p_neg (only if i==j, but we skip that)
    # Handle case where p_pos == p_center (i==j, skip)
    
    # Evaluate objective at parameter t
    def eval_obj(t, bc, ln, qe):
        # max_k (bc[k] + t*ln[k] + t²*qe[k])
        base = bc + t * ln
        mx = base.max()
        for (p, q) in qe:
            v = base[p] + t*t*q
            if v > mx:
                mx = v
        return mx * dx
    
    # Coarse grid search
    n_grid = 30
    ts = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_val = eval_obj(0, base_corr, lin, quad_entries)  # current value
    
    for t in ts:
        val = eval_obj(t, base_corr, lin, quad_entries)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best_t
    if best_t > t_lo + 1e-10 and best_t < t_hi - 1e-10:
        # Fine grid around best_t
        margin = (t_hi - t_lo) / n_grid
        ts_fine = np.linspace(max(t_lo, best_t - margin), min(t_hi, best_t + margin), 30)
        for t in ts_fine:
            val = eval_obj(t, base_corr, lin, quad_entries)
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val
```

Actually, for n=63, 2*n-1=125. The Python loop over 125 iterations per pair, for 2000 pairs per sweep, is 250,000 iterations. Each iteration does a few comparisons and additions. This should be fast enough (maybe 0.1-0.5s per sweep).

I need to be smarter. Let me vectorize the t-evaluation.

For a fixed pair (i,j), I have:
- lin: array of length 2n-1
- base_corr: array of length 2n-1
- quad entries: 3 (position, coefficient) pairs

For multiple t values at once, I can compute:
values[k, t_idx] = base_corr[k] + t[t_idx] * lin[k] + t[t_idx]² * quad[k]

This is a (2n-1) × n_grid matrix, and I take the max over axis 0.

```python
def repair_sum(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = (1.0 - h)
        total_room = room.sum()
        if total_room < 1e-12:
            return h
        h += diff * room / total_room
    else:
        space = h
        total_space = space.sum()
        if total_space < 1e-12:
            return h
        h += diff * space / total_space
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += residual

def compute_lin(h, i, j, n):

    # (1-h)[i + p - n + 1] for valid p
    # i + p - n + 1 >= 0 => p >= n - 1 - i
    # i + p - n + 1 < n => p < 2n - 1 - i
    p_start = max(0, n - 1 - i)
    p_end = min(2 * n - 1, 2 * n - 1 - i)
    if p_start < p_end:
        lin[p_start:p_end] += (1.0 - h[0:p_end - p_start])

    # i - p + n - 1 >= 0 => p <= i + n - 1
```

Wait, I have a bug in compute_lin. Let me re-derive.

numpy.correlate(a, v, 'full')[p] = sum_m a[m] * v[m + p - (len(v)-1)]

For correlate(d, 1-h, 'full')[p]:
a = d, v = 1-h, len(v) = n
= sum_m d[m] * (1-h)[m + p - n + 1]
= d[i] * (1-h)[i + p - n + 1] + d[j] * (1-h)[j + p - n + 1]
= (1-h)[i + p - n + 1] - (1-h)[j + p - n + 1]

For the first term (1-h)[i + p - n + 1] to be valid: 0 ≤ i + p - n + 1 < n
⟹ n - 1 - i ≤ p ≤ n - 1 + i... wait:
i + p - n + 1 ≥ 0 ⟹ p ≥ n - 1 - i
i + p - n + 1 < n ⟹ p < 2n - 1 - i ⟹ p ≤ 2n - 2 - i

```python
def compute_lin(h, i, j, n):
    lin = np.zeros(2 * n - 1)
    
    # Part 1: (1-h)[i+p-n+1] for valid p
    # Valid when 0 <= i+p-n+1 < n
    # p from n-1-i to 2n-2-i
    ps = max(0, n - 1 - i)
    pe = min(2 * n - 1, 2 * n - 1 - i)
    if ps < pe:
        # index = i + p - n + 1, when p=ps: index = i+ps-n+1
        idx0 = i + ps - n + 1
        lin[ps:pe] += (1.0 - h[idx0:idx0+(pe-ps)])
    
    # Part 2: -(1-h)[j+p-n+1]
    ps = max(0, n - 1 - j)
    pe = min(2 * n - 1, 2 * n - 1 - j)
    if ps < pe:
        idx0 = j + ps - n + 1
        lin[ps:pe] -= (1.0 - h[idx0:idx0+(pe-ps)])
    
    # Part 3: -h[i-p+n-1]
    # Valid when 0 <= i-p+n-1 < n
    # p from i to i+n-1
    ps = max(0, i)
    pe = min(2 * n - 1, i + n)
    if ps < pe:
        # index = i - p + n - 1, when p=ps: index = i-ps+n-1
        # when p=pe-1: index = i-(pe-1)+n-1
        # The index decreases as p increases
        idx_at_ps = i - ps + n - 1
        idx_at_pe_minus1 = i - (pe - 1) + n - 1
        # We need h[idx_at_ps], h[idx_at_ps-1], ..., h[idx_at_pe_minus1]
        # That is h[idx_at_pe_minus1 : idx_at_ps+1][::-1]
        lin[ps:pe] -= h[idx_at_pe_minus1:idx_at_ps+1][::-1]
    
    # Part 4: +h[j-p+n-1]
    ps = max(0, j)
    pe = min(2 * n - 1, j + n)
    if ps < pe:
        idx_at_ps = j - ps + n - 1
        idx_at_pe_minus1 = j - (pe - 1) + n - 1
        lin[ps:pe] += h[idx_at_pe_minus1:idx_at_ps+1][::-1]
    
    return lin
```
