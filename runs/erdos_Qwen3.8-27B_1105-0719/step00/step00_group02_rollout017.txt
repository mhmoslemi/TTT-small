    max_lag = np.max(lags)

            js = np.arange(j_lo2, j_hi2 + 1)

    exps = np.exp((lags - max_lag) / tau)
    w = exps / np.sum(exps)
    s_val = max_lag + tau * np.log(np.sum(exps))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        p = k - (n - 1)
        j1_lo = max(0, p)
        j1_hi = min(n - 1, p + n - 1)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi + 1)
            grad[js] += wk * dx * (1.0 - h[js - p])
        j2_lo = max(0, -p)
        j2_hi = min(n - 1, n - 1 - p)
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi + 1)
            grad[js] -= wk * dx * h[js + p]
    return s_val, grad

def is_feasible(h, n, tol=1e-6):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def smooth_descent(h, c, n, max_time, start_time, total_budget, seed, best):
    rng = np.random.default_rng(seed)
    h = h.copy()
    c = c
    tau = max(2e-5, 0.08 * c)
    alpha = 0.02
    no_improve = 0
    max_stall = 300
    minimum_tau = 5e-6

    while True:
        elapsed = time.perf_counter() - start_time
        if elapsed >= total_budget - 5:
            break
        if elapsed - (total_budget - max_time) > max_time:
            break
        if no_improve >= max_stall:
            tau *= 0.7
            if tau <= minimum_tau:
                break
            no_improve = 0
            alpha = 0.02

        s_val, grad = soft_max_grad(h, tau)
        if not np.all(np.isfinite(grad)):
            break
        grad_t = grad - np.mean(grad)
        gn = np.linalg.norm(grad_t)
        if gn < 1e-12:
            noise = rng.standard_normal(n)
            noise -= np.mean(noise)
            direction = noise / max(1e-12, np.linalg.norm(noise))
        else:
            direction = -grad_t / gn

        accepted = False
        for _ in range(12):
            h_new = project_box_sum(h + alpha * direction, n / 2.0)
            h_new = repair_exact(h_new, n)
            c_new = evaluate_c5(h_new)
            if np.isfinite(c_new) and c_new < c - 5e-11:
                h = h_new
                c = c_new
                accepted = True
                no_improve = 0
                if c < best[1] - 1e-12:
                    best[0] = h.copy()
                    best[1] = c
                break
            alpha *= 0.5
            if alpha < 1e-9:
                break
        if not accepted:
            no_improve += 1
            alpha = max(0.005, alpha * 0.7)

    return h, c

def pair_polish(h, c, n, max_time, start_time, total_budget, seed, best):
    rng = np.random.default_rng(seed)
    h = h.copy()
    c = c
    fractions = [0.01, 0.03, 0.08, 0.15, 0.3, 0.6]
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    rng.shuffle(pairs)
    passes = 0
    max_passes = 50
    stall = 0

    while passes < max_passes:
        elapsed = time.perf_counter() - start_time
        if elapsed >= total_budget - 5:
            break
        if elapsed - (total_budget - max_time) > max_time:
            break

        improved = False
        for pi, (i, j) in enumerate(pairs):
            if pi % 200 == 0 and (time.perf_counter() - start_time) >= total_budget - 5:
                break
            for f in fractions:
                delta = min(h[i], 1.0 - h[j]) * f
                if delta < 1e-11:
                    continue
                h[i] -= delta
                h[j] += delta
                h = repair_exact(h, n)
                c_new = evaluate_c5(h)
                if np.isfinite(c_new) and c_new < c - 5e-11:
                    c = c_new
                    improved = True
                    stall = 0
                    if c < best[1] - 1e-12:
                        best[0] = h.copy()
                        best[1] = c
                    break
                else:
                    h[i] += delta
                    h[j] -= delta
                    h = repair_exact(h, n)
            for f in fractions:
                delta = min(h[j], 1.0 - h[i]) * f
                if delta < 1e-11:
                    continue
                h[j] -= delta
                h[i] += delta
                h = repair_exact(h, n)
                c_new = evaluate_c5(h)
                if np.isfinite(c_new) and c_new < c - 5e-11:
                    c = c_new
                    improved = True
                    stall = 0
                    if c < best[1] - 1e-12:
                        best[0] = h.copy()
                        best[1] = c
                    break
                else:
                    h[j] += delta
                    h[i] -= delta
                    h = repair_exact(h, n)

        passes += 1
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        rng.shuffle(pairs)

    return h, c

def sa_explore(h, c, n, max_time, start_time, total_budget, seed, best):
    rng = np.random.default_rng(seed)
    h = h.copy()
    c = c
    temp = 0.015
    delta_scale = 0.04
    min_temp = 1e-7

    while True:
        elapsed = time.perf_counter() - start_time
        if elapsed >= total_budget - 5:
            break
        if elapsed - (total_budget - max_time) > max_time:
            break

        move_type = rng.random()
        if move_type < 0.65:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            direction = 1 if rng.random() < 0.5 else -1
            donor = i if direction > 0 else j
            recv = j if direction > 0 else i
            cap = min(h[donor], 1.0 - h[recv])
            if cap < 1e-11:
                continue
            delta = delta_scale * cap * rng.uniform(0.3, 1.5)
            delta = min(delta, cap)
            h[donor] -= delta
            h[recv] += delta
        elif move_type < 0.85:
            noise = rng.standard_normal(n)
            noise -= np.mean(noise)
            h = project_box_sum(h + delta_scale * noise, n / 2.0)
        else:
            i = rng.integers(n)
            h[i] += (rng.random() - 0.5) * 0.1
            h = project_box_sum(h, n / 2.0)

        h = repair_exact(h, n)
        c_new = evaluate_c5(h)
        if not np.isfinite(c_new):
            h = best[0].copy()
            c = best[1]
            continue

        dc = c_new - c
        if dc < -5e-11:
            c = c_new
            if c < best[1] - 1e-12:
                best[0] = h.copy()
                best[1] = c
        else:
            accept = False
            if temp > min_temp:
                ratio = -dc / max(temp, 1e-12)
                if ratio < 50:
                    accept = rng.random() < np.exp(-ratio)
                else:
                    accept = False
            if accept:
                c = c_new
            else:
                h = best[0].copy()
                c = best[1]

        temp *= 0.998
        if temp < min_temp:
            temp = min_temp
            delta_scale *= 0.95
            if delta_scale < 0.005:
                break

    return h, c

def try_resolution(parent_h, parent_n, target_n, seed, best):
    if target_n < 8 or target_n > 300:
        return
    t = np.linspace(0.0, 2.0, target_n, endpoint=True)
    parent_t = np.linspace(0.0, 2.0, parent_n, endpoint=True)
    h_interp = np.interp(t, parent_t, parent_h)
    h_interp = project_box_sum(h_interp, target_n / 2.0)
    h_interp = repair_exact(h_interp, target_n)
    c = evaluate_c5(h_interp)
    if np.isfinite(c) and c < best[1] - 1e-12:
        best[0] = h_interp.copy()
        best[1] = c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    try:
        parent = np.asarray(initial_h_values, dtype=float).flatten().copy()
    except Exception:
        parent = None

    if parent is not None and len(parent) >= 8:
        n = len(parent)
        parent = project_box_sum(parent, n / 2.0)
        parent = repair_exact(parent, n)
        c_parent = evaluate_c5(parent)
        if not np.isfinite(c_parent):
            parent = np.full(47, 0.5)
            n = 47
            c_parent = evaluate_c5(parent)
    else:
        n = 47
        parent = np.full(n, 0.5)
        c_parent = evaluate_c5(parent)

    best = [parent.copy(), c_parent]

    # Try nearby resolutions quickly
    for target_n in [n - 5, n - 3, n + 3, n + 5, n + 10]:
        try_resolution(parent, n, target_n, seed, best)
        if time.perf_counter() - t_start > 10:
            break

    # If parent was at a different n than best, adjust
    best_n = len(best[0])
    base_h = best[0].copy()
    base_c = best[1]

    # Phase 1: Smooth gradient descent (main workhorse)
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    limit1 = min(400, 0.40 * time_left)
    base_h, base_c = smooth_descent(base_h, base_c, best_n, limit1, t_start, budget_s, seed + 1, best)
    if base_c < best[1]:
        best[0] = base_h.copy()
        best[1] = base_c

    # Phase 2: Pair polish
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    limit2 = min(250, 0.25 * time_left)
    base_h, base_c = pair_polish(best[0].copy(), best[1], best_n, limit2, t_start, budget_s, seed + 2, best)
    if base_c < best[1]:
        best[0] = base_h.copy()
        best[1] = base_c

    # Phase 3: SA exploration
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    limit3 = max(30, 0.25 * time_left)
    base_h, base_c = sa_explore(best[0].copy(), best[1], best_n, limit3, t_start, budget_s, seed + 3, best)
    if base_c < best[1]:
        best[0] = base_h.copy()
        best[1] = base_c

    # Phase 4: More smooth polish (with smaller tau)
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    if time_left > 30:
        limit4 = min(150, 0.20 * time_left)
        base_h, base_c = smooth_descent(best[0].copy(), best[1], best_n, limit4, t_start, budget_s, seed + 4, best)
        if base_c < best[1]:
            best[0] = base_h.copy()
            best[1] = base_c

    # Phase 5: Final pair polish
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    if time_left > 20:
        limit5 = max(10, time_left - 15)
        base_h, base_c = pair_polish(best[0].copy(), best[1], best_n, limit5, t_start, budget_s, seed + 5, best)
        if base_c < best[1]:
            best[0] = base_h.copy()
            best[1] = base_c

    # Phase 6: Final micro-smooth
    elapsed = time.perf_counter() - t_start
    time_left = budget_s - elapsed
    if time_left > 15:
        limit6 = max(5, time_left - 10)
        base_h, base_c = smooth_descent(best[0].copy(), best[1], best_n, limit6, t_start, budget_s, seed + 6, best)
        if base_c < best[1]:
            best[0] = base_h.copy()
            best[1] = base_c

    # Final validation
    h_final = np.clip(best[0], 0.0, 1.0)
    n_final = len(h_final)
    h_final = repair_exact(h_final, n_final)
    c_final = evaluate_c5(h_final)

    if not np.isfinite(c_final) or not is_feasible(h_final, n_final):
        h_final = parent
        c_final = c_parent
        n_final = len(parent)

    return h_final, c_final, n_final
