    target = n / 2.0

    n = len(h)
    l = k - n + 1
    g = np.zeros(n)

        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    else:
        l2 = -l
        g[l2:] += (1.0 - h[:n-l2])

    c_max = np.max(c)
    s = alpha * (c - c_max)
    m = np.max(s)
    log_sum = m + np.log(np.sum(np.exp(s - m)))
    return c_max + log_sum / alpha

def ls_exp_grad(h, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)

    m = np.max(s)

    w = exp_s / np.sum(exp_s)

        if w[k] > 1e-12:
            grad += w[k] * grad_c_k(h, k) * dx
    return grad

def paired_exchange_optimize(h, n, max_iter=5000):
    """Optimize using paired exchanges (sum-preserving direction)."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

        # h[i] - t, h[j] + t, for t in range
        t_min = max(-h[i], -(1-h[j]))
        t_max = min(h[i], 1-h[j])
        
        # 1D optimization: find t that minimizes C5
        # Use golden section search or scipy
        best_t = 0
        best_val = best_c5
        
        # Sample a few points
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            t = t_min + frac * (t_max - t_min)
            if abs(t) < 1e-10:

            h_trial[i] -= t
            h_trial[j] += t
            val = compute_c5(h_trial)
            if val < best_val:

                best_t = t
        
        if best_t != 0:

            h[j] += best_t

                best_h = h.copy()

def block_optimize(h_init, n, max_iter=5000, seed=43):
    """Optimize using block transfers."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    block_size = max(2, n // 10)
    
    for it in range(max_iter):
        # Pick two non-overlapping blocks
        start1 = rng.randint(0, n - 2*block_size)
        start2 = start1 + block_size + rng.randint(0, n - 2*start1 - 2*block_size)
        
        if start2 + block_size > n:
            continue
        
        idx1 = range(start1, start1 + block_size)
        idx2 = range(start2, start2 + block_size)
        
        # Try transferring mass from block1 to block2
        m1 = np.sum(h[list(idx1)])
        m2 = np.sum(h[list(idx2)])
        total = m1 + m2
        
        # Transfer fraction: make block1 have fraction f of total
        for f in [0.3, 0.4, 0.5, 0.6, 0.7]:
            target_m1 = f * total
            delta = m1 - target_m1  # amount to transfer from block1 to block2
            
            # Check feasibility
            h1_vals = h[list(idx1)]
            h2_vals = h[list(idx2)]
            
            if delta > 0:
                # Transfer from block1 to block2
                max_transfer = np.min(h1_vals)  # can't go below 0
                max_transfer2 = np.min(1.0 - h2_vals)  # can't go above 1
                max_t = min(max_transfer, max_transfer2)
                t = min(delta, max_t)
                if t < 1e-10:
                    continue
                h[list(idx1)] -= t / block_size
                h[list(idx2)] += t / block_size
            else:
                # Transfer from block2 to block1
                t = -delta
                max_transfer = np.min(h2_vals)
                max_transfer2 = np.min(1.0 - h1_vals)
                max_t = min(max_transfer, max_transfer2)
                t = min(t, max_t)
                if t < 1e-10:
                    continue
                h[list(idx2)] -= t / block_size
                h[list(idx1)] += t / block_size
            
            val = compute_c5(h)
            if val < best_c5:
                best_c5 = val
                best_h = h.copy()
            else:
                # Revert
                if delta > 0:
                    h[list(idx1)] += t / block_size
                    h[list(idx2)] -= t / block_size
                else:
                    h[list(idx2)] += t / block_size
                    h[list(idx1)] -= t / block_size
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(v, n):
    target = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c_k(h, k):
    n = len(h)
    l = k - n + 1
    g = np.zeros(n)
    if l >= 0:
        g[:n-l] += (1.0 - h[l:])
        g[l:] -= h[:n-l]
    else:
        l2 = -l
        g[l2:] += (1.0 - h[:n-l2])
        g[:n-l2] -= h[l2:]
    return g

def slsqp_optimize(h_init, n, alpha=100, maxiter=200):

        cm = np.max(c)
        s = alpha * (c - cm)
        sm = np.max(s)
        return cm + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

    def obj_grad(h):
        c = compute_c5_vec(h)
        cm = np.max(c)
        s = alpha * (c - cm)
        sm = np.max(s)
        es = np.exp(s - sm)
        w = es / np.sum(es)
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] > 1e-10:
                grad += w[k] * grad_c_k(h, k) * (2.0 / n)
        return grad

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,

                       bounds=bounds, constraints=cons,

            val = compute_c5(h)
            if val < best_c5:
                best_c5 = val
                best_h = h.copy()
            else:
                if delta < 0:
                    h[i1] += t
                    h[i2] -= t
                else:
                    h[i2] += t
                    h[i1] -= t
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: coordinate descent
    if tl() > 90:
        h1, c5_1 = coordinate_optimize(best_h, n0, max_iter=30000, seed=seed)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: SLSQP
    if tl() > 90:
        h2, c5_2 = slsqp_optimize(best_h, n0, alpha=50, maxiter=300)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        if tl() > 60:
            h2b, c5_2b = slsqp_optimize(best_h, n0, alpha=200, maxiter=200)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b

    # Phase 3: subgradient + SLSQP
    if tl() > 60:
        h3, c5_3 = subgradient_method(best_h, n0, max_iter=3000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
        if tl() > 30:
            h3b, c5_3b = slsqp_optimize(h3, n0, alpha=100, maxiter=200)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b

    # Phase 4: more coordinate descent
    if tl() > 60:
        h4, c5_4 = coordinate_optimize(best_h, n0, max_iter=30000, seed=seed + 1)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: block optimization
    if tl() > 60:
        h5, c5_5 = block_optimize(best_h, n0, max_iter=5000, seed=seed + 2)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5
        if tl() > 30:
            h5b, c5_5b = slsqp_optimize(h5, n0, alpha=100, maxiter=100)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b

    # Phase 6: try other n values
    for n_try in [55, 60, 65, 70, 75, 80, 90, 100]:
        if tl() < 40:
            break
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, n_try)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_feasible(hi, n_try)
        ht, ct = slsqp_optimize(hi, n_try, alpha=50, maxiter=150)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht
        if tl() < 20:
            break
        ht2, ct2 = coordinate_optimize(ht, n_try, max_iter=10000, seed=seed + n_try)
        if ct2 < best_c5:
            best_c5 = ct2
            best_h = ht2

    # Phase 7: random perturbations + local opt
    nf = len(best_h)
    for attempt in range(40):
        if tl() < 15:
            break
        h_p = best_h + rng.normal(0, 0.02, nf)
        h_p = project_feasible(h_p, nf)
        hr, cr = slsqp_optimize(h_p, nf, alpha=100, maxiter=80)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr

    # Final
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, nf)
