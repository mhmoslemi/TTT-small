    L = 2 * n - 1
    lags = np.arange(-(n-1), n)

        if l >= 0:
            if l <= a:
                coeff += g[a - l]
            if l <= b:
                coeff -= g[b - l]
            if l <= n - 1 - a:
                coeff -= h[a + l]
            if l <= n - 1 - b:
                coeff += h[b + l]

                coeff -= g[b + m]
            if m <= a:
                coeff -= h[a - m]
            if m <= b:
                coeff += h[b - m]

    return D
