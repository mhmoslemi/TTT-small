def objective_exact(h):
    """Compute C5 exactly using the evaluator formula."""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(F), F

def project_to_feasible(z, target_sum):
    """Project z onto {x in [0,1]^N : sum(x) = target_sum}."""
    n = len(z)
    z = np.clip(np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0), 0, 1)
    # Bisection for lambda: sum(clip(z_i - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - 0.5*(lo+hi), 0, 1)
    # Exact correction
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual
        slack_up = 1.0 - result
        slack_down = result
        if residual > 0:
            # Need to add
            total_slack = np.sum(slack_up[slack_up > 1e-12])
            if total_slack > abs(residual):
                idx = np.where(slack_up > 1e-12)[0]
                frac = residual / total_slack
                result[idx] = np.minimum(result[idx] + frac * slack_up[idx], 1.0)
            else:
                result += residual / len(result)
                result = np.clip(result, 0, 1)
        else:
            total_slack = np.sum(slack_down[slack_down > 1e-12])
            if total_slack > abs(residual):
                idx = np.where(slack_down > 1e-12)[0]
                frac = -residual / total_slack
                result[idx] = np.maximum(result[idx] - frac * slack_down[idx], 0.0)
            else:
                result += residual / len(result)
                result = np.clip(result, 0, 1)
    return result

def smooth_objective_and_grad(h, beta):
    """Compute soft-max smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    Fmax = np.max(F)
    shifted = beta * (F - Fmax)
    # Stable softmax weights
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total
    Phi = Fmax + np.log(total) / beta
    # Analytic gradient
    grad = np.zeros(n)
    for ell in range(2 * n - 1):
        if w[ell] < 1e-14:
            continue
        d = ell - (n - 1)  # lag offset
        js = max(0, -d)
        je = min(n, n - d)
        if js >= je:
            continue
        m_idx = np.arange(js, je)
        first_idx = m_idx + d
        # d(first) = dx * w * (1 - h[m])
        # d(second) = -dx * w * h[first]
        grad[first_idx] += dx * w[ell] * (1.0 - h[m_idx])
        grad[m_idx] -= dx * w[ell] * h[first_idx]
    return Phi, grad
