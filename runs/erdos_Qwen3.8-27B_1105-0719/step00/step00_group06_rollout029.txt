def make_feasible(h, n):
    """Ensure h is feasible: in [0,1] and sum = n/2."""
    h = np.array(h, dtype=float).flatten()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Water-filling to adjust sum
    diff = target - s
    if diff > 0:
        # Need to increase sum: add to elements < 1
        order = np.argsort(1.0 - h)  # elements with most room first
        remaining = diff
        for i in order:
            can_add = 1.0 - h[i]
            add = min(can_add, remaining)
            h[i] += add
            remaining -= add
            if remaining < 1e-12:
                break
    else:
        # Need to decrease sum: remove from elements > 0
        order = np.argsort(h)  # elements with least value first (hardest to remove)
        # Actually, remove from elements with most value first
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            can_remove = h[i]
            remove = min(can_remove, remaining)
            h[i] -= remove
            remaining -= remove
            if remaining < 1e-12:
                break
    return h

def pair_line_minimize(h, u, v):
    """Minimize C5 along line h[u]+=s, h[v]-=s. Returns (new_h, new_c5)."""
    h_orig_u = h[u]
    h_orig_v = h[v]
    
    s_min = max(-h_orig_u, h_orig_v - 1.0) + 1e-13
    s_max = min(1.0 - h_orig_u, h_orig_v) - 1e-13
    
    if s_max <= s_min:
        return h, evaluate_c5(h)
    
    def obj(s):
        h[u] = h_orig_u + s
        h[v] = h_orig_v - s
        return evaluate_c5(h)
    
    # Evaluate at endpoints and current point
    c5_at_0 = evaluate_c5(h)
    
    result = minimize_scalar(obj, bounds=(s_min, s_max), method='bounded',
                             options={'xatol': 1e-12})
    
    if result.fun < c5_at_0:
        h[u] = h_orig_u + result.x
        h[v] = h_orig_v - result.x
        return h, float(result.fun)
    else:
        return h, c5_at_0

def local_optimize_pairs(h, seed, max_rounds=50, pairs_per_round=100):
    """Local optimization via random pair exchanges."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    
    for round_idx in range(max_rounds):
        improved = False
        for _ in range(pairs_per_round):
            u = rng.randint(n)
            v = rng.randint(n)
            if u == v:
                continue
            h, new_c5 = pair_line_minimize(h, u, v)
            if new_c5 < best_c5:
                best_c5 = new_c5
                improved = True
            c5 = new_c5
        if not improved:
            break
    
    return h, best_c5

One more thought: the "pair line search" approach where we minimize along h[u]+=s, h[v]-=s is essentially a "2-coordinate" update. An alternative is "1-coordinate" update where we just change h[u] to a new value (and adjust one other coordinate to maintain the sum). But the pair approach is cleaner.

Actually, there's an even better approach: since the objective is convex along the line, and we know it's affine everywhere except at 2 specific k values where it's quadratic, we can compute the exact minimum analytically!

For a given pair (u,v) with difference d = u-v:
- For k ≠ ±d: c'[k](s) = c[k] + a_k * s (affine, where a_k is computed from h)
- For k = d: c'[k](s) = c[k] + b_d * s + s² (quadratic)
- For k = -d: c'[k](s) = c[k] + b_{-d} * s + s² (quadratic)

The max of these is a piecewise function. The minimum occurs either:
- At a vertex where two or more pieces are equal and the max switches
- At the vertex of one of the quadratics (if it's the active piece)
- At a boundary of [s_min, s_max]

This could be computed in O(n) time per pair (no need for iterative 1D search). But for simplicity and correctness, I'll use scipy's minimize_scalar.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).flatten()[:n].copy()
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)))
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass: prefer elements with room (close to 0)
        room = 1.0 - h
        # Spread proportionally to room
        if room.sum() > 0:
            h += diff * room / room.sum()
        else:
            h += diff / n
    else:
        # Need to remove mass: prefer elements with excess (close to 1)
        mass = h.copy()
        if mass.sum() > 0:
            h += diff * mass / mass.sum()
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final correction: distribute residual evenly among non-saturated
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += residual / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def pair_line_minimize(h, u, v):

    hv = h[v]

    s_hi = min(1.0 - hu, hv)
    margin = 1e-14
    s_lo += margin
    s_hi -= margin
    if s_hi <= s_lo:
        return h, evaluate_c5(h)

        h[u] = hu + s
        h[v] = hv - s
