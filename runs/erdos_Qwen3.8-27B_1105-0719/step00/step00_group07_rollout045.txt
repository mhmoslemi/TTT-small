def compute_c5(h):

    return (2.0/n) * R.max()

    h[ones] = 1

    best_max = current_max
    best_h = h.copy()

            if current_max < best_max:

import numpy as np
import time

def compute_R(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    R = compute_R(h)
    return (2.0 / n) * np.max(R)

def project_box_sum(y, target):
    """Project y onto {x : 0<=x_i<=1, sum x_i = target}"""
    n = len(y)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    r = target - x.sum()
    if r > 1e-10:
        i = 0
        while r > 1e-12 and i < n:
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
            i += 1
    elif r < -1e-10:
        i = 0
        while r < -1e-12 and i < n:
            rem = min(-r, x[i])
            x[i] -= rem
            r += rem
            i += 1
    return x

def lse_grad(h, beta):
    """Compute gradient of log-sum-exp of R values."""
    n = len(h)
    R = compute_R(h)
    m = R.max()
    shifted = np.clip(beta * (R - m), -300, 300)
    ev = np.exp(shifted)
    w = ev / ev.sum()
    
    grad = np.zeros(n)
    for t in range(len(R)):
        wt = w[t]
        if wt < 1e-8:
            continue
        l = t - (n - 1)
        # dR_t/dh[p] = (1-h[p-l]) * [0<=p-l<n] - h[p+l] * [0<=p+l<n]
        # Sum over p of first term: for p in [max(0,l), min(n-1, l+n-1)]
        p1_lo = max(0, l)
        p1_hi = min(n - 1, l + n - 1)
        if p1_lo <= p1_hi:
            idx = p1_lo - l  # this is the index into h for the (1-h[...]) part
            idx2 = idx + (p1_hi - p1_lo + 1)
            grad[p1_lo:p1_hi + 1] += wt * (1.0 - h[idx:idx2])
        # Sum over p of second term: for p in [max(0,-l), min(n-1, n-1-l)]
        p2_lo = max(0, -l)
        p2_hi = min(n - 1, n - 1 - l)
        if p2_lo <= p2_hi:
            idx = p2_lo + l
            idx2 = idx + (p2_hi - p2_lo + 1)
            grad[p2_lo:p2_hi + 1] -= wt * h[idx:idx2]
    return grad

            lr = lr * 0.5
            patience = 0

        h[:s] = 1

        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[rng.randint(ones_idx.size)]
        j = zeros_idx[rng.randint(zeros_idx.size)]
        
        # Swap
        h[i] = 0
        h[j] = 1
        R_new = compute_R(h)
        new_max = R_new.max()
        delta = new_max - cur_max
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            R = R_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0

    improved = True
    while improved:
        improved = False
        R = compute_R(best_h)
        cur_max = R.max()
        ones_idx = np.where(best_h == 1)[0]
        zeros_idx = np.where(best_h == 0)[0]

                best_h[i] = 0

                R_t = compute_R(best_h)
                if R_t.max() < cur_max:
                    cur_max = R_t.max()
                    improved = True
                    break
                else:
                    best_h[i] = 1
                    best_h[j] = 0
    
    dx = 2.0 / n
    return best_h, dx * best_max
