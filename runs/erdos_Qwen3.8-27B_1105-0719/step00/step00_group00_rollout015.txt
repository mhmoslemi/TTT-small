    """For pair (a,b), compute B_m and D_m for each lag m."""
    # Lags range from -(n-1) to n-1
    # C_m = sum_{i valid} h[i] * (1-h[i+m])
    # After perturbation: h[a]+delta, h[b]-delta
    
    B = {}  # coefficient of delta
    D = {}  # coefficient of delta^2
    
    for m in range(-(n-1), n):
        b_coeff = 0.0
        d_coeff = 0.0
        for i in range(max(0, -m), min(n, n-m)):
            # valid pair (i, i+m)
            im = i + m
            sa = 1.0 if i == a else 0.0
            sb = 1.0 if i == b else 0.0
            ta = 1.0 if im == a else 0.0
            tb = 1.0 if im == b else 0.0
            if sa + sb + ta + tb > 0:
                b_coeff += h[i] * (tb - ta) + (sa - sb) * (1 - h[im])
                d_coeff += (sa - sb) * (tb - ta)
        if b_coeff != 0.0 or d_coeff != 0.0:
            B[m] = b_coeff
            D[m] = d_coeff
    
    return B, D

    """Upsample h by linear interpolation, then adjust to maintain sum constraint."""

    # Simple approach: repeat each value `factor` times
    h_new = np.repeat(h, factor)
    # Adjust sum
    target_sum = new_n / 2.0
    current_sum = np.sum(h_new)
    diff = target_sum - current_sum
    h_new += diff / new_n
    # Clip
    h_new = np.clip(h_new, 0, 1)
    # Re-adjust sum (clipping may have changed it)

        # Redistribute
        s = np.sum(h_new)
        h_new += (target_sum - s) / new_n
        h_new = np.clip(h_new, 0, 1)
    return h_new
