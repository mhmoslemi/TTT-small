        return mid, obj_func(mid)

def compute_gradient_profile(h, n):
    """Compute gradient of each lag's overlap with respect to h.
    Returns array of shape (2n-1, n) where grad[k, i] = d(overlap at lag k)/d(h[i])
    """
    grad = np.zeros((2*n-1, n))
    # For lag s (shift), where s ranges from -(n-1) to n-1
    # In np.correlate output, index k corresponds to shift s = k - (n-1)
    for s in range(-(n-1), n):
        k = s + (n-1)  # index in correlate output
