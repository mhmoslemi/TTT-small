```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return dx * np.max(corr)

def correlation_values(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    return np.correlate(h, q, mode="full") * dx

def lse(h, beta=100.0):
    """Log-sum-exp surrogate for max of correlation values"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    # Shift for numerical stability
    shifted = corr - max_c
    return max_c + np.log(np.exp(beta * shifted).sum()) / beta

def lse_grad(h, beta=100.0):
    """Gradient of log-sum-exp surrogate"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = corr - max_c
    weights = beta * np.exp(beta * shifted) / np.exp(beta * shifted).sum()
    # Now need gradient of each corr[k] w.r.t. h
    # This is expensive... let's use finite differences for the gradient
    return None  # We'll use numerical gradient

def optimize_lseb(h0, n, beta=200.0, maxiter=500, tol=1e-10):
    """Optimize using L-BFGS-B with reduced coordinates"""
    target = n / 2.0
    
    def obj(z):
        h = np.ones(n)
        h[:-1] = z
        h[-1] = target - np.sum(z)
        if h[-1] < 0 or h[-1] > 1:
            return 1e6
        return lse(h, beta)
    
    # Bounds for z: 0 <= z[i] <= 1, and 0 <= target - sum(z) <= 1
    # The last constraint is handled by the objective penalty
    bounds = [(0, 1)] * (n-1)
    
    result = minimize(obj, h0[:-1], method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8})
    
    h = np.ones(n)
    h[:-1] = result.x
    h[-1] = target - np.sum(h[:-1])
    h = np.clip(h, 0, 1)
    return h, result.fun

def pair_transfer(h, n_passes=5):
    """Local optimization via pair transfers"""
    n = len(h)
    h = h.copy()
    current = evaluate_c5(h)
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                # Try h[i] + delta, h[j] - delta
                max_d = min(1.0 - h[i], h[j])
                if max_d < 1e-12:
                    continue
                # Try a few delta values
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    d = max_d * frac
                    h_new = h.copy()
                    h_new[i] = h[i] + d
                    h_new[j] = h[j] - d
                    if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                        s = evaluate_c5(h_new)
                        if s < current - 1e-12:
                            h = h_new
                            current = s
                            improved = True
                            break
        if not improved:
            break
    return h

def project_to_feasible(y, n):
    target = n / 2.0
```

```python
def correlation_gradient(h, weights):
    """
    Compute sum_k weights[k] * dc_k/dh
    where k indexes the full correlation output (0 to 2n-2)
    and weights[k] is the LSE weight for lag k-(n-1).
    """
    n = len(h)
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        k = lag_idx - (n - 1)
        w = weights[lag_idx]
        if w < 1e-14:
            continue
        if k >= 0:
            # (1-h[j-k]) for j in [k, n-1]
            if k < n:
                js = np.arange(k, n)
                grad[js] += w * (1.0 - h[js - k])
            # -h[j+k] for j in [0, n-1-k]
            if k < n:
                js = np.arange(0, n - k)
                grad[js] -= w * h[js + k]
        else:
            m = -k
            # (1-h[j+m]) for j in [0, n-1-m]
            if m < n:
                js = np.arange(0, n - m)
                grad[js] += w * (1.0 - h[js + m])
            # -h[j-m] for j in [m, n-1]
            if m < n:
                js = np.arange(m, n)
                grad[js] -= w * h[js - m]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def project_to_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.any():
            h[free] += residual / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_grad(z, n, beta):
    target = n / 2.0
    h = np.empty(n)
    h[:-1] = z
    h[-1] = target - np.sum(z)
    if h[-1] < -0.5 or h[-1] > 1.5:
        return 1e6, np.zeros(n - 1)
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    c_max = np.max(corr)
    shifted = np.clip((corr - c_max) * beta, -700.0, 700.0)
    exp_s = np.exp(shifted)
    lse_val = c_max + np.log(exp_s.sum()) / beta
    weights = beta * exp_s / exp_s.sum()
    grad_h = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        k = lag_idx - (n - 1)
        w = weights[lag_idx]
        if w < 1e-14:
            continue
        if k >= 0:
            if k < n:
                js = np.arange(k, n)
                grad_h[js] += w * (1.0 - h[js - k])
            if k < n:
                js = np.arange(0, n - k)
                grad_h[js] -= w * h[js + k]
        else:
            m = -k
            if m < n:
                js = np.arange(0, n - m)
                grad_h[js] += w * (1.0 - h[js + m])
            if m < n:
                js = np.arange(m, n)
                grad_h[js] -= w * h[js - m]
    grad_z = grad_h[:-1] - grad_h[n - 1]
    return lse_val, grad_z


def optimize_lseb(h0, n, betas, maxiter):
    h = h0.copy()
    for beta in betas:
        z0 = h[:-1].copy()
        result = minimize(
            lse_obj_grad, z0, args=(n, beta),
            jac=True, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * (n - 1),
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        h_new = np.empty(n)
        h_new[:-1] = result.x
        h_new[-1] = n / 2.0 - np.sum(result.x)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, n)
        if evaluate_c5(h_new) <= evaluate_c5(h) + 1e-10:
            h = h_new
    return h


def pair_transfer_refine(h, n_passes=3):
    n = len(h)
    h = h.copy()
    current = evaluate_c5(h)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                max_d = min(1.0 - h[i], h[j])
                if max_d < 1e-12:
                    continue
                best_d = 0.0
                best_s = current
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    d = max_d * frac
                    h[i] += d
                    h[j] -= d
                    s = evaluate_c5(h)
                    if s < best_s:
                        best_s = s
                        best_d = d
                    h[i] -= d
                    h[j] += d
                if best_d > 0:
                    h[i] += best_d
                    h[j] -= best_d
                    current = best_s
                    improved = True
                max_d2 = min(1.0 - h[j], h[i])
                if max_d2 < 1e-12:
                    continue
                best_d2 = 0.0
                best_s2 = current
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    d = max_d2 * frac
                    h[i] -= d
                    h[j] += d
                    s = evaluate_c5(h)
                    if s < best_s2:
                        best_s2 = s
                        best_d2 = d
                    h[i] += d
                    h[j] -= d
                if best_d2 > 0:
                    h[i] -= best_d2
                    h[j] += best_d2
                    current = best_s2
                    improved = True
        if not improved:
            break
    return h


def random_feasible(n, rng):
    y = rng.uniform(0, 1, n)
    return project_to_feasible(y, n)


def block_start(n, rng):
    n_blocks = rng.randint(4, 10)
    h = np.zeros(n)
    gaps = rng.randint(1, 4, n_blocks)
    sizes = rng.randint(1, max(2, n // (n_blocks * 2) + 1), n_blocks)
    pos = 0
    for b in range(n_blocks):
        h[pos:pos + sizes[b]] = 1.0
        pos += sizes[b] + gaps[b]
        if pos >= n:
            break
    return project_to_feasible(h, n)


def sine_start(n, rng):
    freq = rng.uniform(2.0, 8.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n) * (2.0 / n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project_to_feasible(h, n)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_to_feasible(h_interp, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = project_to_feasible(np.full(45, 0.5), 45)

    best_n = len(best_h)
    best_score = evaluate_c5(best_h)

    n_list = sorted(set([30, 36, 45, 60, 75, 90, 120]))
    if best_n not in n_list:
        n_list.insert(0, best_n)

    for n in n_list:
        if time.time() - t0 > budget_s - 90:
            break

        n_starts = max(4, int(40.0 * 45.0 / n))
        if n > 90:
            n_starts = max(3, n_starts - 2)

        for si in range(n_starts):
            if time.time() - t0 > budget_s - 30:
                break
            elapsed = time.time() - t0
            remaining = budget_s - elapsed - 20
            if remaining < 30:
                break

            if si == 0:
                h = interpolate_h(best_h, n)
            elif si == 1:
                h = random_feasible(n, rng)
            elif si == 2:
                h = block_start(n, rng)
            elif si == 3:
                h = sine_start(n, rng)
            else:
                h_base = interpolate_h(best_h, n)
                noise_scale = 0.03 + 0.01 * si
                h = h_base + rng.normal(0, noise_scale, n)
                h = project_to_feasible(h, n)

            # Multi-stage optimization
            maxiter = int(200 * remaining / 300) if remaining > 0 else 50
            maxiter = min(max(50, maxiter), 400)

            if n <= 60:
                betas = [30, 60, 100, 200, 400, 800]
            elif n <= 90:
                betas = [30, 60, 100, 200, 400]
            else:
                betas = [30, 60, 100, 200]

            h = optimize_lseb(h, n, betas, maxiter)

            if time.time() - t0 > budget_s - 15:
                break

            # Pair transfer refinement
            n_passes = 3 if n <= 60 else 2
            h = pair_transfer_refine(h, n_passes)

            score = evaluate_c5(h)
            if score < best_score - 1e-10:
                best_score = score
                best_h = h.copy()
                best_n = n

            # Annealing step: perturb and re-optimize if time allows
            if time.time() - t0 < budget_s - 60:
                h_anneal = h + rng.normal(0, 0.02, n)
                h_anneal = project_to_feasible(h_anneal, n)
                h_anneal = optimize_lseb(h_anneal, n, [100, 300, 600], min(100, maxiter // 2))
                h_anneal = pair_transfer_refine(h_anneal, 2)
                score_a = evaluate_c5(h_anneal)
                if score_a < best_score - 1e-10:
                    best_score = score_a
                    best_h = h_anneal.copy()
                    best_n = n

    # Final polish on the best solution
    if time.time() - t0 < budget_s - 10:
        best_h = pair_transfer_refine(best_h, 5)
        best_score = evaluate_c5(best_h)

    final_score = evaluate_c5(best_h)
    return best_h, final_score, len(best_h)
```
