def grad_single(h, d):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N, dtype=float)
    if N == 0: return g
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    if d > 0:
        if d >= N: return g
        idx1 = np.arange(d, N)
        if idx1.size:
            g[idx1] += dx * (1.0 - h[idx1 - d])
        idx2 = np.arange(0, N - d)
        if idx2.size:
            g[idx2] -= dx * h[idx2 + d]
    else:
        m = -d
        if m >= N: return g
        idx1 = np.arange(0, N - m)
        if idx1.size:
            g[idx1] += dx * (1.0 - h[idx1 + m])
        idx2 = np.arange(m, N)
        if idx2.size:
            g[idx2] -= dx * h[idx2 - m]
    return g
```

def objective_all(h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0:
        return float("inf"), np.array([], dtype=float)
    N = h.size
    if not np.all(np.isfinite(h)):
        return float("inf"), np.full(2 * N - 1, float("inf"), dtype=float)
    F = np.correlate(h, 1.0 - h, mode="full") * (2.0 / N)
    if F.size == 0 or not np.all(np.isfinite(F)):
        return float("inf"), F
    return float(np.max(F)), F


def project_feasible(x, target):
    x = np.asarray(x, dtype=float).copy()
    N = x.size
    if N == 0:
        return x
    if not np.all(np.isfinite(x)):

    lo = float(np.min(x)) - 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def grad_single(h, d):
    N = h.size
    if N == 0:
        return np.array([], dtype=float)
    dx = 2.0 / N
    g = np.zeros(N, dtype=float)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    if d > 0:
        if d >= N:
            return g
        idx1 = np.arange(d, N)
        if idx1.size:
            g[idx1] += dx * (1.0 - h[idx1 - d])
        idx2 = np.arange(0, N - d)
        if idx2.size:
            g[idx2] -= dx * h[idx2 + d]
    else:
        m = -d
        if m >= N:
            return g
        idx1 = np.arange(0, N - m)
        if idx1.size:
            g[idx1] += dx * (1.0 - h[idx1 + m])
        idx2 = np.arange(m, N)
        if idx2.size:
            g[idx2] -= dx * h[idx2 - m]
    return g


def determine_correlate_sign():
    N = 6
    rng = np.random.default_rng(12345)
    h = rng.uniform(0.25, 0.75, N)
    base = N - 1
    eps = 1e-5
    err1 = 0.0
    err2 = 0.0
    cnt = 0
    for p in range(N):
        hp = h.copy()
        hm = h.copy()
        hp[p] += eps
        hm[p] -= eps
        Fp = objective_all(hp)[1]
        Fm = objective_all(hm)[1]
        if Fp.size != 2 * N - 1 or Fm.size != 2 * N - 1:
            break
        num = (Fp - Fm) / (2.0 * eps)
        for k in range(2 * N - 1):
            d1 = k - base
            d2 = base - k
            g1 = grad_single(h, d1)
            g2 = grad_single(h, d2)
            v1 = g1[p] - num[k]
            v2 = g2[p] - num[k]
            if np.isfinite(v1):
                err1 += v1 * v1
            if np.isfinite(v2):
                err2 += v2 * v2
            cnt += 1
    if cnt == 0:
        return 1
    return 1 if err1 <= err2 else -1


def lse_phi_weights(h, beta):
    c, F = objective_all(h)
    n = F.size
    if n == 0 or not np.isfinite(c):
        return float("inf"), np.array([], dtype=float), F
    maxF = float(np.max(F))
    z = float(beta) * (F - maxF)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if not np.isfinite(s) or s <= 0.0:
        w = np.full(n, 1.0 / float(n))
        phi = maxF
    else:
        w = ez / s
        phi = maxF + math.log(s) / float(beta)
    return phi, w, F


def lse_grad(h, weights):
    h = np.asarray(h, dtype=float)
    N = h.size
    g = np.zeros(N, dtype=float)
    if N == 0 or weights.size == 0:
        return g
    base = N - 1
    for k in range(weights.size):
        w = float(weights[k])
        if w <= 1e-12:
            continue
        d = int(_CORR_SIGN * (int(k) - base))
        g += w * grad_single(h, d)
    return g


def lse_beta_pass(h, target, deadline, beta, max_iter=100):
    cur = np.asarray(h, dtype=float).copy()
    cur_c, F = objective_all(cur)
    if not np.isfinite(cur_c):
        return cur, cur_c
    best = cur.copy()
    best_c = cur_c
    no_accept = 0
    for _ in range(max_iter):
        if _now() > deadline:
            break
        phi, w, F = lse_phi_weights(cur, beta)
        if not np.isfinite(phi):
            break
        g = lse_grad(cur, w)
        g = g - np.mean(g)
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-11:
            break
        direction = -g / gn
        alpha = 0.2
        accepted = False
        for _ in range(35):
            cand = project_feasible(cur + alpha * direction, target)
            phi_new, _, F_new = lse_phi_weights(cand, beta)
            c_new = float(np.max(F_new)) if F_new.size else float("inf")
            tol = 1e-10 * max(1.0, abs(phi))
            if np.isfinite(phi_new) and (phi_new < phi - tol or c_new < cur_c - 1e-10):
                cur = cand
                phi = phi_new
                F = F_new
                cur_c = c_new
                accepted = True
                if cur_c < best_c - 1e-12:
                    best = cur.copy()
                    best_c = cur_c
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        if not accepted:
            no_accept += 1
            if no_accept > 8:
                break
        else:
            no_accept = 0
    return best, best_c


def lse_refine(h0, target, deadline, betas=BETA_SCHEDULE):
    best = np.asarray(h0, dtype=float).copy()
    best_c, _ = objective_all(best)
    if not np.isfinite(best_c):
        return best, best_c
    N = best.size
    mi = 110 if N < 120 else 35
    for beta in betas:
        if _now() > deadline:
            break
        h_new, c_new = lse_beta_pass(best, target, min(deadline, _now() + max(5.0, 0.08 * max(60.0, deadline - _now() + 1.0))), beta, mi)
        if np.isfinite(c_new) and c_new < best_c - 1e-12:
            best = h_new
            best_c = c_new
    return best, best_c


def make_variants(h, N, rng, target):
    vals = []
    vals.append(h.copy())
    vals.append(0.5 * (h + h[::-1]))
    if N > 1:
        vals.append(np.roll(h, 1))
        vals.append(np.roll(h, -1))
    if N > 2:
        vals.append(0.2 * np.roll(h, 1) + 0.6 * h + 0.2 * np.roll(h, -1))
    for amp in (0.02, 0.05, 0.10, 0.15):
        vals.append(h + amp * rng.normal(0.0, 1.0, N))
    out = []
    for v in vals:
        p = project_feasible(v, target)
        c, F = objective_all(p)
        if np.isfinite(c):
            out.append((c, p, F))
    return out


def select_active_indices(F, top_k=40, tau=1e-4):
    if F.size == 0 or not np.all(np.isfinite(F)):
        return np.array([], dtype=int)
    total = F.size
    k = max(1, min(int(top_k), total))
    top = np.argsort(F)[-k:]
    if tau <= 0.0:
        return top
    cmax = float(np.max(F))
    near = np.where(F >= cmax - tau)[0]
    if near.size <= k:
        combined = np.unique(np.concatenate((near, top)))
        if combined.size > 2 * k:
            order = np.argsort(F[combined])[-(2 * k):]
            combined = combined[order]
        return combined
    order = np.argsort(F[near])[-k:]
    return near[order]


def active_lp_candidate(h, target, active, radius):
    N = h.size
    if N == 0 or active.size == 0:
        return None
    F = objective_all(h)[1]
    if F.size == 0 or not np.all(np.isfinite(F)):
        return None
    base = N - 1
    m = active.size
    rows = m + 2 * N
    A_ub = np.zeros((rows, N + 1), dtype=float)
    b_ub = np.empty(rows, dtype=float)
    r = 0
    for kk in active:
        d = int(_CORR_SIGN * (int(kk) - base))
        g = grad_single(h, d)
        gdot = float(np.dot(g, h))
        A_ub[r, :N] = g
        A_ub[r, N] = -1.0
        b_ub[r] = gdot - float(F[kk])
        r += 1
    for i in range(N):
        A_ub[r, i] = 1.0
        b_ub[r] = h[i] + radius
        r += 1
        A_ub[r, i] = -1.0
        b_ub[r] = -h[i] + radius
        r += 1
    c = np.zeros(N + 1, dtype=float)
    c[N] = 1.0
    A_eq = np.zeros((1, N + 1), dtype=float)
    A_eq[0, :N] = 1.0
    b_eq = np.array([float(target)], dtype=float)
    bounds = [(0.0, 1.0)] * N + [(-1.0, 1.0)]
    res = None
    try:
        res = so.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method="highs")
    except Exception:
        try:
            res = so.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method="revised simplex")
        except Exception:
            res = None
    if res is None or not getattr(res, "success", False):
        return None
    x = np.asarray(res.x[:N], dtype=float)
    if x.size != N or not np.all(np.isfinite(x)):
        return None
    return project_feasible(x, target)


def active_lp_refine(h0, target, rng, deadline, top_k=40, tau=1e-4, max_rounds=500):
    cur = np.asarray(h0, dtype=float).copy()
    N = cur.size
    if N == 0:
        return cur, float("inf")
    if N > 150:
        top_k = min(20, top_k)
        max_rounds = min(max_rounds, 120)
    cur_c, cur_F = objective_all(cur)
    if not np.isfinite(cur_c):
        return cur, cur_c
    best = cur.copy()
    best_c = cur_c
    radius = 0.15
    stall = 0
    perturb = 0
    for _ in range(max_rounds):
        if _now() > deadline:
            break
        active = select_active_indices(cur_F, top_k, tau)
        if active.size == 0:
            break
        cand = active_lp_candidate(cur, target, active, radius)
        if cand is None:
            radius *= 0.6
            if radius < 1e-5:
                break
            continue
        cand_c, cand_F = objective_all(cand)
        if not np.isfinite(cand_c):
            radius *= 0.6
            continue
        if cand_c < cur_c - 1e-10:
            cur = cand.copy()
            cur_c = cand_c
            cur_F = cand_F
            stall = 0
            if cur_c < best_c - 1e-12:
                best = cur.copy()
                best_c = cur_c
            move = float(np.linalg.norm(cand - cur))
            if move < 1e-5:
                radius *= 0.75
        else:
            stall += 1
            if stall >= 5:
                amp = 0.005 * (perturb + 1)
                amp = min(0.1, amp)
                cur = project_feasible(best + amp * rng.normal(0.0, 1.0, N), target)
                cur_c, cur_F = objective_all(cur)
                if np.isfinite(cur_c) and cur_c < best_c - 1e-12:
                    best = cur.copy()
                    best_c = cur_c
                radius = max(0.02, min(0.2, radius * 0.8))
                stall = 0
                perturb += 1
                if perturb > 18:
                    break
            else:
                radius *= 0.9
        if radius < 1e-6:
            radius = 0.05
            cur = project_feasible(best + 0.02 * rng.normal(0.0, 1.0, N), target)
            cur_c, cur_F = objective_all(cur)
    best = project_feasible(best, target)
    bc, _ = objective_all(best)
    if np.isfinite(bc) and bc < best_c:
        best_c = bc
    return best, best_c


def gradient_pair_polish(h0, target, deadline, rng, max_sweeps=120):
    cur = np.asarray(h0, dtype=float).copy()
    N = cur.size
    if N == 0:
        return cur, float("inf")
    if N > 150:
        max_sweeps = min(max_sweeps, 40)
    cur_c, cur_F = objective_all(cur)
    if not np.isfinite(cur_c):
        return cur, cur_c
    best = cur.copy()
    best_c = cur_c
    no_improve = 0
    base = N - 1
    for sweep in range(max_sweeps):
        if _now() > deadline:
            break
        active = select_active_indices(cur_F, 25, 1e-5)
        g = np.zeros(N, dtype=float)
        if active.size:
            Fac = cur_F[active]
            maxc = float(np.max(Fac))
            z = 50.0 * (Fac - maxc)
            ez = np.exp(z)
            s = float(np.sum(ez))
            if np.isfinite(s) and s > 0.0:
                w = ez / s
            else:
                w = np.full(active.size, 1.0 / float(active.size))
            for ii, kk in enumerate(active):
                wi = float(w[ii])
                if wi <= 1e-12:
                    continue
                d = int(_CORR_SIGN * (int(kk) - base))
                g += wi * grad_single(cur, d)
        g = g - np.mean(g)
        improved = False
        if np.all(np.isfinite(g)):
            diff = g[:, None] - g[None, :]
            order = np.argsort(diff.ravel())[::-1]
            ii_arr, jj_arr = np.divmod(order, N)
            considered = 0
            for pos in range(ii_arr.size):
                if considered >= 15:
                    break
                i = int(ii_arr[pos])
                j = int(jj_arr[pos])
                if i == j:
                    continue
                max_step = min(float(cur[i]), float(1.0 - cur[j]))
                if max_step <= 1e-13:
                    continue
                considered += 1
                bt = 0.0
                bc_local = cur_c
                bf_local = cur_F
                for frac in PAIR_FRACS:
                    t = frac * max_step
                    if t <= 1e-14:
                        continue
                    new = cur.copy()
                    new[i] -= t
                    new[j] += t
                    c, Fn = objective_all(new)
                    if np.isfinite(c) and c < bc_local - 1e-12:
                        bc_local = c
                        bt = t
                        bf_local = Fn
                if bt > 0.0:
                    cur = cur.copy()
                    cur[i] -= bt
                    cur[j] += bt
                    cur_c = bc_local
                    cur_F = bf_local
                    improved = True
                    if cur_c < best_c - 1e-12:
                        best = cur.copy()
                        best_c = cur_c
                    break

        if not improved:
            rand_improved = False
            for _ in range(20):
                if _now() > deadline or N < 2:
                    break
                i = int(rng.integers(N))
                j = int(rng.integers(N))
                if i == j:
                    continue
                if rng.random() < 0.5:
                    max_step = min(float(cur[i]), float(1.0 - cur[j]))
                else:
                    max_step = min(float(cur[j]), float(1.0 - cur[i]))
                    i, j = j, i
                if max_step <= 1e-13:
                    continue
                bt = 0.0
                bc_local = cur_c
                bf_local = cur_F
                for frac in PAIR_FRACS:
                    t = frac * max_step
                    if t <= 1e-14:
                        continue
                    new = cur.copy()
                    new[i] -= t
                    new[j] += t
                    c, Fn = objective_all(new)
                    if np.isfinite(c) and c < bc_local - 1e-12:
                        bc_local = c
                        bt = t
                        bf_local = Fn
                if bt > 0.0:
                    cur = cur.copy()
                    cur[i] -= bt
                    cur[j] += bt
                    cur_c = bc_local
                    cur_F = bf_local
                    rand_improved = True
                    if cur_c < best_c - 1e-12:
                        best = cur.copy()
                        best_c = cur_c
                    break
            if rand_improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 5:
                    break
                amp = 0.002 * no_improve
                cur = project_feasible(best + amp * rng.normal(0.0, 1.0, N), target)
                cur_c, cur_F = objective_all(cur)
                if np.isfinite(cur_c) and cur_c < best_c - 1e-12:
                    best = cur.copy()
                    best_c = cur_c
        else:
            no_improve = 0
        if sweep and sweep % 25 == 0:
            cur = project_feasible(cur, target)
            cc, cF = objective_all(cur)
            if np.isfinite(cc):
                cur_c, cur_F = cc, cF
                if cc < best_c - 1e-12:
                    best = cur.copy()
                    best_c = cc
    best = project_feasible(best, target)
    bc, _ = objective_all(best)
    if np.isfinite(bc) and bc < best_c:
        best_c = bc
    return best, best_c


def random_local_search(h0, target, deadline, rng, max_iters=300000):
    cur = np.asarray(h0, dtype=float).copy()
    N = cur.size
    if N == 0:
        return cur, float("inf")
    if N > 150:
        max_iters = min(max_iters, 60000)
    cur_c, cur_F = objective_all(cur)
    if not np.isfinite(cur_c):
        return cur, cur_c
    best = cur.copy()
    best_c = cur_c
    it = 0
    stall = 0
    perturb = 0
    while it < max_iters and _now() < deadline:
        it += 1
        if it % 512 == 0:
            if _now() > deadline:
                break
            if abs(float(np.sum(cur)) - target) > 1e-10:
                cur = project_feasible(cur, target)
                cur_c, cur_F = objective_all(cur)
                if np.isfinite(cur_c) and cur_c < best_c - 1e-12:
                    best = cur.copy()
                    best_c = cur_c
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        if rng.random() < 0.5:
            max_step = min(float(cur[i]), float(1.0 - cur[j]))
        else:
            max_step = min(float(cur[j]), float(1.0 - cur[i]))
            i, j = j, i
        if max_step <= 1e-13:
            stall += 1
            continue
        u = rng.random()
        if u < 0.5:
            frac = 10.0 ** rng.uniform(-5.0, -2.0)
        else:
            frac = 10.0 ** rng.uniform(-2.0, -0.3)
        t = frac * max_step
        if t <= 1e-15:
            continue
        new = cur.copy()
        new[i] -= t
        new[j] += t
        c, Fn = objective_all(new)
        if not np.isfinite(c):
            continue
        if c < cur_c - 1e-12:
            cur = new
            cur_c = c
            cur_F = Fn
            stall = 0
        else:
            stall += 1
        if c < best_c - 1e-12:
            best = new.copy()
            best_c = c
            stall = 0
        if stall > 20000:
            amp = 0.01 * (perturb + 1)
            amp = min(0.1, amp)
            cur = project_feasible(best + amp * rng.normal(0.0, 1.0, N), target)
            cur_c, cur_F = objective_all(cur)
            if np.isfinite(cur_c) and cur_c < best_c - 1e-12:
                best = cur.copy()
                best_c = cur_c
            stall = 0
            perturb += 1
            if perturb > 25:
                break
    best = project_feasible(best, target)
    bc, _ = objective_all(best)
    if np.isfinite(bc) and bc < best_c:
        best_c = bc
    return best, best_c


def slsqp_fun(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return 1e6
    if x.min() < -1e-4 or x.max() > 1.0 + 1e-4:
        return 1e6
    phi, w, F = lse_phi_weights(x, _SLSQ_BETA)
    if not np.isfinite(phi):
        return 1e6
    return float(phi)


def slsqp_jac(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return np.array([], dtype=float)
    phi, w, F = lse_phi_weights(x, _SLSQ_BETA)
    if not np.isfinite(phi):
        return np.zeros_like(x)
    g = lse_grad(x, w)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(x)
    return g


def slsqp_eq(x):
    x = np.asarray(x, dtype=float)
    return float(np.sum(x) - _SLSQ_TARGET)


def slsqp_eq_jac(x):
    x = np.asarray(x, dtype=float)
    return np.ones_like(x)


def slsqp_refine(h0, target, deadline):
    global _SLSQ_BETA, _SLSQ_TARGET
    best = np.asarray(h0, dtype=float).copy()
    best_c, _ = objective_all(best)
    N = best.size
    if N == 0:
        return best, best_c
    for beta in (64.0, 128.0, 256.0):
        if _now() > deadline - 15.0:
            break
        _SLSQ_BETA = float(beta)
        _SLSQ_TARGET = float(target)
        try:
            res = so.minimize(
                slsqp_fun,
                best.copy(),
                jac=slsqp_jac,
                method="SLSQP",
                bounds=[(0.0, 1.0)] * N,
                constraints=[{"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac}],
                options={"maxiter": 60, "ftol": 1e-8}
            )
        except Exception:
            continue
        if res is None or getattr(res, "x", None) is None:
            continue
        x = np.asarray(res.x, dtype=float)
        if x.size != N or not np.all(np.isfinite(x)):
            continue
        xp = project_feasible(x, target)
        xc, _ = objective_all(xp)
        if np.isfinite(xc) and xc < best_c - 1e-12:
            best = xp.copy()
            best_c = xc
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _CORR_SIGN, _SLSQ_BETA, _SLSQ_TARGET

    t0 = time.time()
    total = float(budget_s)
    if total <= 0.0:
        total = 1.0

    best = None
    best_c = float("inf")
    N = 63
    target = 0.5 * float(N)

    try:
        try:
            _CORR_SIGN = determine_correlate_sign()
        except Exception:
            _CORR_SIGN = 1

        h, N = prepare_initial(seed)
        target = 0.5 * float(N)
        best = h.copy()
        best_c, _ = objective_all(best)
        if not np.isfinite(best_c):
            best = project_feasible(np.full(N, 0.5), target)
            best_c, _ = objective_all(best)

        rng = np.random.default_rng(seed)

        var_list = make_variants(best, N, rng, target)
        for c, p, F in var_list:
            if c < best_c - 1e-12:
                best = p.copy()
                best_c = c

        if total > 60:
            main_deadline = t0 + total - 30.0
            final_deadline = t0 + total - 10.0
        else:
            main_deadline = t0 + 0.75 * total
            final_deadline = t0 + 0.95 * total
        main_deadline = max(main_deadline, t0 + 1.0)
        final_deadline = max(final_deadline, t0 + 0.5)

        lse_end = min(main_deadline, t0 + 0.20 * total)
        if lse_end > _now() + 10.0:
            hb, cb = lse_refine(best, target, lse_end)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        slq_end = min(main_deadline, t0 + 0.35 * total)
        if slq_end > _now() + 25.0:
            hb, cb = slsqp_refine(best, target, slq_end)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        lp_end = min(main_deadline, t0 + 0.75 * total)
        if lp_end > _now() + 20.0:
            hb, cb = active_lp_refine(best, target, rng, lp_end)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        gp_end = min(main_deadline, t0 + 0.88 * total)
        if gp_end > _now() + 10.0:
            hb, cb = gradient_pair_polish(best, target, gp_end, rng, max_sweeps=80)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        rs_end = main_deadline
        if rs_end > _now() + 10.0:
            hb, cb = random_local_search(best, target, rs_end, rng)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        if final_deadline > _now() + 8.0:
            hb, cb = active_lp_refine(best, target, rng, final_deadline,
                                      top_k=20, tau=1e-5, max_rounds=120)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        if final_deadline > _now() + 5.0:
            hb, cb = gradient_pair_polish(best, target, final_deadline, rng, max_sweeps=30)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        if final_deadline > _now() + 3.0:
            hb, cb = random_local_search(best, target, final_deadline, rng, max_iters=80000)
            if np.isfinite(cb) and cb < best_c - 1e-12:
                best = hb.copy()
                best_c = cb

        best = project_feasible(best, target)
        final_c, _ = objective_all(best)
        if not np.isfinite(final_c):
            best = project_feasible(np.full(N, 0.5), target)
            final_c, _ = objective_all(best)
        if not np.isfinite(final_c):
            final_c = TARGET_C + 0.1
        return best.copy(), float(final_c), int(best.size)

    except Exception:
        if best is not None and np.all(np.isfinite(best)) and best.size >= 3:
            N = best.size
        else:
            try:
                best, N = prepare_initial(seed)
            except Exception:
                N = 63
                best = fallback_h(N, seed)
        target = 0.5 * float(N)
        best = project_feasible(best, target)
        c, _ = objective_all(best)
        if not np.isfinite(c):
            best = project_feasible(np.full(N, 0.5), target)
            c, _ = objective_all(best)
        if not np.isfinite(c):
            c = TARGET_C + 0.1
        return best.copy(), float(c), int(best.size)
