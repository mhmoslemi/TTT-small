    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _con_fun(h):
    return np.sum(h) - _g_target_sum

def _con_jac(h):
    return np.ones(len(h))

def _obj(h):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    return max_c + np.log(np.sum(np.exp(shifted))) / s

    for _ in range(max_iterations):
