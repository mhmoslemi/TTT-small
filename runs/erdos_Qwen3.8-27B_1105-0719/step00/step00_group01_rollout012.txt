        return time.time() - t0

    # Phase 1: Progressive SLSQP on initial solution
    alphas_phase1 = [5, 10, 20, 50, 100]
    h_work = h_best.copy()
    for alpha in alphas_phase1:
        if elapsed() > budget_s * 0.25:
            break
        h_work = slsqp_step(h_work, n, target, dx, alpha, maxiter=500)
        c5 = compute_c5(h_work, dx)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_work.copy()

    # Phase 2: Higher alpha refinement
    for alpha in [200, 500, 1000]:
        if elapsed() > budget_s * 0.45:
            break
        h_work = slsqp_step(h_best, n, target, dx, alpha, maxiter=400)
        c5 = compute_c5(h_work, dx)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_work.copy()

    # Phase 3: Random restarts
    n_restarts = 20
    for r in range(n_restarts):
        if elapsed() > budget_s * 0.65:
            break
        scale = max(0.02, 0.15 - 0.005 * r)
        h_start = h_best + scale * np.random.randn(n)
        h_start = fix_sum(h_start, target)
        h_opt = slsqp_step(h_start, n, target, dx, 30, maxiter=200)
        h_opt = slsqp_step(h_opt, n, target, dx, 200, maxiter=200)
        c5 = compute_c5(h_opt, dx)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_opt.copy()

    # Phase 4: Fine grid
    if elapsed() < budget_s * 0.4:
        n_f = n * 2
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        x_old = np.arange(n) * dx
        x_new = np.arange(n_f) * dx_f
        h_f = np.interp(x_new, x_old, h_best)
        h_f = fix_sum(h_f, target_f)
        c5_f = compute_c5(h_f, dx_f)

        if elapsed() < budget_s * 0.8:
            for alpha in [10, 30, 100]:
                if elapsed() > budget_s * 0.85:
                    break
                h_f = slsqp_step(h_f, n_f, target_f, dx_f, alpha, maxiter=400)
            c5_f = compute_c5(h_f, dx_f)

        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 5: Final polish
    if elapsed() < budget_s * 0.9:
        n_p = len(h_best)
        target_p = n_p / 2.0
        dx_p = 2.0 / n_p
        h_p = slsqp_step(h_best, n_p, target_p, dx_p, 500, maxiter=300)
        h_p = slsqp_step(h_p, n_p, target_p, dx_p, 1000, maxiter=200)
        c5_p = compute_c5(h_p, dx_p)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p

    return h_best, c5_best, len(h_best)
