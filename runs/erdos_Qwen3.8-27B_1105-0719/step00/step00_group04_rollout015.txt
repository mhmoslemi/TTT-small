import numpy as np
from scipy.optimize import linprog
import time
import math

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def project_feasible(h):
    h = np.array(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = np.sum(free)
        if nfree == 0:
            if diff > 0:
                # increase some values at 0
                mask = h < 0.5
                if not np.any(mask):
                    mask = np.ones(n, dtype=bool)
                nmask = np.sum(mask)
                h[mask] += diff / nmask
            else:
                mask = h > 0.5
                if not np.any(mask):
                    mask = np.ones(n, dtype=bool)
                nmask = np.sum(mask)
                h[mask] += diff / nmask
        else:
            h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def build_A_matrix(n):
    """Build the incidence structure for np.correlate(h, 1-h, mode='full').
    Returns list of (i, j) pairs for each output index k.
    output[k] = sum h[i] * (1-h)[j] for (i,j) in pairs[k]
    """
    pairs = []
    for k in range(2 * n - 1):
        s = k - (n - 1)  # shift
        i_start = max(0, -s)
        i_end = min(n - 1, n - 1 - s)
        idx = [(i, i + s) for i in range(i_start, i_end + 1)]
        pairs.append(idx)
    return pairs

def lp_optimize_h(h_fixed_b, n, max_time=5.0):
    """Fix b = 1 - h_current, solve LP to find h minimizing max_k sum h[i]*b[i+s]."""
    b = 1.0 - h_fixed_b
    dx = 2.0 / n
    pairs = build_A_matrix(n)
    
    # Variables: h[0..n-1], t
    # Minimize t
    # Constraints: sum_i h[i] * b[j] <= t/dx for each k
    #              sum_i h[i] = n/2
    #              0 <= h[i] <= 1
    
    nvars = n + 1
    c = np.zeros(nvars)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # sum_i h[i] * b[j] - t/dx <= 0 for each k
    M = 2 * n - 1
    A_ineq = np.zeros((M, nvars))
    b_ineq = np.zeros(M)
    
    for k in range(M):
        for (i, j) in pairs[k]:
            A_ineq[k, i] = b[j]
        A_ineq[k, n] = -1.0 / dx
    
    # Equality: sum h[i] = n/2
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = project_feasible(res.x[:n])
            return h_new
    except:
        pass
    return None

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def project_feasible(h):
    h = np.array(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            if diff > 0:
                idx = np.argsort(h)[:max(1, -int(diff))]
            else:
                idx = np.argsort(h)[-max(1, int(diff)):]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_lp_matrices(n, fixed_arr, optimize_first):
    """Build LP matrices.
    If optimize_first: optimize h with b=1-fixed fixed. A[m,i] = b[i-shift]
    If not: optimize b with h=fixed fixed. A[m,j] = h[j+shift]
    """
    M = 2 * n - 1
    A = np.zeros((M, n))
    dx = 2.0 / n
    if optimize_first:
        b = 1.0 - fixed_arr
        for m in range(M):
            s = m - (n - 1)
            for i in range(n):
                j = i - s
                if 0 <= j < n:
                    A[m, i] = b[j]
    else:
        h = fixed_arr
        for m in range(M):
            s = m - (n - 1)
            for j in range(n):
                i = j + s
                if 0 <= i < n:
                    A[m, j] = h[i]
    A = A * dx
    return A

        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        w = np.exp(beta * (corr - np.max(corr)))
        w /= np.sum(w)
        # Subgradient of smoothed max w.r.t. h
        grad = np.zeros(n)
        M = 2 * n - 1
        for m in range(M):
            s = m - (n - 1)
            if w[m] < 1e-10:
                continue
            for i in range(n):
                j = i - s
                if 0 <= j < n:
                    grad[i] += w[m] * dx * (1.0 - h[j])
                i2 = i + s
                if 0 <= i2 < n:
                    grad[i] -= w[m] * dx * h[i2]
        lr_eff = lr / (1.0 + step * 0.01)
        h = h - lr_eff * grad
        h = project_feasible(h)
        c5 = exact_c5(h)
        if c5 < current:
            current = c5
        else:
            h = h * 0.5 + 0.5 * h  # step back (keep old)
    return h, current

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    # Initialize from parent
    h_best = np.array(initial_h_values, dtype=float).copy()
    h_best = project_feasible(h_best)
    c5_best = exact_c5(h_best)
    n_best = len(h_best)

    # Phase 1: LP alternating on initial grid
    h_cur = h_best.copy()
    n = n_best
    for rnd in range(300):
        if time_left() < 120:
            break
        # LP: optimize h with b fixed
        h_lp = lp_solve(h_cur, n, optimize_first=True)
        if h_lp is not None:
            h_blend, c5_blend = blend_and_evaluate(h_cur, h_lp, 20)
            if c5_blend < exact_c5(h_cur):
                h_cur = h_blend
        # LP: optimize b with h fixed
        h_lp2 = lp_solve(h_cur, n, optimize_first=False)
        if h_lp2 is not None:
            h_blend2, c5_blend2 = blend_and_evaluate(h_cur, h_lp2, 20)
            if c5_blend2 < exact_c5(h_cur):
                h_cur = h_blend2
        # Track global best
        c5_cur = exact_c5(h_cur)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h_cur.copy()
            n_best = n
        # Occasionally do pair optimization
        if rnd % 20 == 19:
            h_pair, c5_pair = pair_optimize(h_cur, n_max_iter=30, n_pairs=30, rng=rng)
            if c5_pair < exact_c5(h_cur):
                h_cur = h_pair
            c5_cur = exact_c5(h_cur)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
                n_best = n

    # Phase 2: Try different grid sizes
    n_candidates = [40, 50, 60, 80, 90, 100, 120, 140, 160, 180, 200]
    for n_try in n_candidates:
        if time_left() < 150:
            break
        if n_try == n_best:
            continue
        h_try = interpolate_h(h_best, n_best, n_try)
        c5_try = exact_c5(h_try)
        # LP alternating on new grid
        for rnd in range(80):
            if time_left() < 60:
                break
            h_lp = lp_solve(h_try, n_try, optimize_first=True)
            if h_lp is not None:
                h_blend, c5_blend = blend_and_evaluate(h_try, h_lp, 15)
                if c5_blend < c5_try:
                    h_try = h_blend
                    c5_try = c5_blend
            h_lp2 = lp_solve(h_try, n_try, optimize_first=False)
            if h_lp2 is not None:
                h_blend2, c5_blend2 = blend_and_evaluate(h_try, h_lp2, 15)
                if c5_blend2 < c5_try:
                    h_try = h_blend2
                    c5_try = c5_blend2
        # Pair optimization
        if time_left() > 60:
            h_try, c5_try = pair_optimize(h_try, n_max_iter=40, n_pairs=40, rng=rng)
        if c5_try < c5_best - 1e-8:
            c5_best = c5_try
            h_best = h_try.copy()
            n_best = n_try

    # Phase 3: Smoothed gradient descent on best
    if time_left() > 60:
        h_gd, c5_gd = smoothed_gradient_descent(h_best, n_steps=300, lr=0.05, beta=30.0, rng=rng)
        if c5_gd < c5_best:
            h_best = h_gd
            c5_best = c5_gd
            n_best = len(h_gd)

    # Phase 4: Final aggressive pair optimization
    if time_left() > 30:
        h_best, c5_best = pair_optimize(h_best, n_max_iter=200, n_pairs=60, rng=rng)

    # Phase 5: Try a few random perturbations
    while time_left() > 40:
        h_pert = h_best + rng.randn(len(h_best)) * 0.05
        h_pert = project_feasible(h_pert)
        c5_pert = exact_c5(h_pert)
        if c5_pert < c5_best + 0.01:  # near current best
            h_pert, c5_pert = pair_optimize(h_pert, n_max_iter=50, n_pairs=30, rng=rng)
            h_lp = lp_solve(h_pert, len(h_pert), optimize_first=True)
            if h_lp is not None:
                h_pert, c5_pert = blend_and_evaluate(h_pert, h_lp, 15)
            h_lp2 = lp_solve(h_pert, len(h_pert), optimize_first=False)
            if h_lp2 is not None:
                h_pert, c5_pert = blend_and_evaluate(h_pert, h_lp2, 15)
            if c5_pert < c5_best:
                c5_best = c5_pert
                h_best = h_pert
                n_best = len(h_best)

    # Final verification
    h_best = project_feasible(h_best)
    c5_final = exact_c5(h_best)
    return h_best, c5_final, len(h_best)
