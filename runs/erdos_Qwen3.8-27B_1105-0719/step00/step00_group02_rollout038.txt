    """Find intersection of all {x: a_k x² + b_k x + c_k ≤ T} with [low, high]."""
    n = len(a)
    # Start with the full interval
    lo = low
    hi = high
    
    for k in range(n):
        ak, bk, ck = a[k], b[k], c[k]
        # Solve a_k x² + b_k x + (c_k - T) ≤ 0
        d_k = ck - T
        
        if abs(ak) < 1e-14:

            if abs(bk) < 1e-14:
                # Constant: d_k ≤ 0
                if d_k > 1e-12:

def get_raw_corr(h):

def get_score(h, dx):

def sum_project(h, n):
    """Project h to have exact sum n/2, respecting [0,1] bounds."""
    h = h.copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Distribute the difference
    # Find how many can absorb
    # Simple approach: distribute uniformly, then clip
    step = diff / n
    h += step
    # Clip and redistribute
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Find indices that can absorb more
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)

            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

    curr_max = float(curr_raw.max())
    
    # 3-point evaluation
    x0 = low
    x1 = (low + high) / 2.0
    x2 = high
    
    ht = h.copy()
    ht[i] = x0; ht[j] = S - x0
    c0 = get_raw_corr(ht)
    ht[i] = x1; ht[j] = S - x1
    c1 = get_raw_corr(ht)
    ht[i] = x2; ht[j] = S - x2
    c2 = get_raw_corr(ht)
    
    V = np.array([[x0*x0, x0, 1.0], [x1*x1, x1, 1.0], [x2*x2, x2, x2*0+1.0]])
    V_inv = np.linalg.inv(V)
    coeffs = V_inv @ np.stack([c0, c1, c2])
    a, b, c = coeffs[0].copy(), coeffs[1].copy(), coeffs[2].copy()
    
    # Grid search + refinement
    xg = np.linspace(low, high, 250)
    xx = xg[:, None]
    M = a[None,:] * (xx*xx) + b[None,:] * xx + c[None,:]
    fv = M.max(axis=1)
    best_idx = np.argmin(fv)
    best_x = xg[best_idx]
    best_v = fv[best_idx]
    
    # Refine
    for _ in range(5):
        span = (high - low) / 500
        lo_r = max(low, best_x - span)
        hi_r = min(high, best_x + span)
        if hi_r - lo_r < 1e-13:
            break
        xg2 = np.linspace(lo_r, hi_r, 80)
        xx2 = xg2[:, None]
        M2 = a[None,:] * (xx2*xx2) + b[None,:] * xx2 + c[None,:]
        fv2 = M2.max(axis=1)
        best_idx2 = np.argmin(fv2)
        best_x = xg2[best_idx2]
        best_v = fv2[best_idx2]
    
    if best_v >= curr_max - 1e-13:
        return None
    
    # Verify with actual evaluation
    ht[i] = best_x
    ht[j] = S - best_x
    actual_max = float(get_raw_corr(ht).max())
    if actual_max < curr_max - 1e-12:
        return (best_x, S - best_x)
    return None
