        if residual > 0:
            slack = 1.0 - result
            ts = np.sum(slack)
            if ts > 1e-15:
                result += residual * (slack / ts)
        else:
            ts = np.sum(result)
            if ts > 1e-15:
                result += residual * (result / ts)
    result = np.clip(result, 0, 1)
    residual2 = s - np.sum(result)
    if abs(residual2) > 1e-14:
        if residual2 > 0:
            idx = np.argmax(1.0 - result)
            result[idx] = min(1.0, result[idx] + residual2)
        else:
            idx = np.argmax(result)
            result[idx] = max(0.0, result[idx] + residual2)
    return result


def repair_exact(h, s):
    h = h.copy()
    residual = s - np.sum(h)
    if abs(residual) < 1e-14:
        return h
    if residual > 0:
        slack = 1.0 - h
        ts = np.sum(slack)
        if ts > 1e-15:
            h += residual * (slack / ts)
    else:
        ts = np.sum(h)
        if ts > 1e-15:
            h += residual * (h / ts)
    h = np.clip(h, 0, 1)
    residual2 = s - np.sum(h)
    if abs(residual2) > 1e-14:
        if residual2 > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + residual2)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual2)
    return h


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        p = k - (n - 1)
        # d corr[k]/dh[m] = (1-h[m-p])*[m-p valid] - h[m+p]*[m+p valid]
        # Contribution to grad from (1-h[m-p]) term: m ranges where m-p in [0,n-1]
        # m in [p, p+n-1] intersect [0, n-1]
        lo1 = max(0, p)
        hi1 = min(n, p + n)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - p:hi1 - p])
        # Contribution from -h[m+p] term: m ranges where m+p in [0,n-1]
        lo2 = max(0, -p)
        hi2 = min(n, n - p)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + p:hi2 + p]
    return grad


def smooth_descent(h_init, n, s, time_limit, seed, best_c):
    rng = np.random.default_rng(seed)
    t0 = time.perf_counter()
    current_h = h_init.copy()
    current_c = best_c
    local_best_h = current_h.copy()
    local_best_c = best_c

    tau = max(0.003, 0.08 * best_c)
    alpha = 0.02
    stall = 0

    while time.perf_counter() - t0 < time_limit:
        grad = compute_lse_gradient(current_h, tau)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            stall += 1
        else:
            direction = -grad / gnorm
            improved = False
            for trial in range(15):
                step = alpha * (0.6 ** trial)
                cand = project_box_sum(current_h + step * direction, s)
                cand_c = evaluate_c5(cand)
                if cand_c < current_c - 1e-12:
                    current_h = cand
                    current_c = cand_c
                    if current_c < local_best_c:
                        local_best_c = current_c
                        local_best_h = current_h.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                alpha = min(0.1, alpha * 1.05)
            else:
                stall += 1
                alpha *= 0.7

        if stall > 40:
            tau *= 0.8
            if tau < 0.0005:
                break
            if rng.random() < 0.6:
                pert = rng.standard_normal(n) * 0.015
                pert -= np.mean(pert)
                cand = project_box_sum(current_h + pert, s)
                cand_c = evaluate_c5(cand)
                if cand_c < local_best_c + tau * 0.5:
                    current_h = cand
                    current_c = cand_c
            stall = 0

    return local_best_h, local_best_c


def pair_transfer(h_init, n, s, time_limit, seed, best_c, fine=True):
    rng = np.random.default_rng(seed)
    t0 = time.perf_counter()
    current_h = h_init.copy()
    current_c = best_c
    local_best_h = current_h.copy()
    local_best_c = best_c

    if fine:
        fracs = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35, 0.55, 0.8, 1.0]
    else:
        fracs = [0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]

    no_pass = 0
    while time.perf_counter() - t0 < time_limit and no_pass < 8:
        order = rng.permutation(n)
        ni = 0
        for ii in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            i = order[ii]
            for jj in range(ii + 1, n):
                j = order[jj]
                # i -> j
                cap = min(current_h[i], 1.0 - current_h[j])
                if cap > 1e-12:
                    for f in fracs:
                        d = cap * f
                        if d < 1e-13:
                            break
                        oi, oj = current_h[i], current_h[j]
                        current_h[i] = oi - d
                        current_h[j] = oj + d
                        cc = evaluate_c5(current_h)
                        if cc < current_c - 1e-13:
                            current_c = cc
                            if cc < local_best_c:
                                local_best_c = cc
                                local_best_h = current_h.copy()
                            ni += 1
                        else:
                            current_h[i] = oi
                            current_h[j] = oj
                # j -> i
                cap = min(current_h[j], 1.0 - current_h[i])
                if cap > 1e-12:
                    for f in fracs:
                        d = cap * f
                        if d < 1e-13:
                            break
                        oi, oj = current_h[i], current_h[j]
                        current_h[i] = oi + d
                        current_h[j] = oj - d
                        cc = evaluate_c5(current_h)
                        if cc < current_c - 1e-13:
                            current_c = cc
                            if cc < local_best_c:
                                local_best_c = cc
                                local_best_h = current_h.copy()
                            ni += 1
                        else:
                            current_h[i] = oi
                            current_h[j] = oj
        no_pass = 0 if ni > 0 else no_pass + 1

    return local_best_h, local_best_c


def sa_optimize(h_init, n, s, time_limit, seed, best_c):
    rng = np.random.default_rng(seed)
    t0 = time.perf_counter()
    current_h = h_init.copy()
    current_c = best_c
    local_best_h = current_h.copy()
    local_best_c = best_c
    temp = max(0.002, 0.04 * best_c)
    scale = 0.04
    acc_count = 0
    total_count = 0

    while time.perf_counter() - t0 < time_limit:
        total_count += 1
        r = rng.random()
        saved = None
        if r < 0.55:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            cap = min(current_h[i], 1.0 - current_h[j])
            if cap < 1e-11:
                cap = min(current_h[j], 1.0 - current_h[i])
                if cap < 1e-11:
                    continue
                i, j = j, i
            d = cap * rng.uniform(0.01, 1.0)
            sign = 1.0 if rng.random() < 0.5 else -1.0
            saved_i, saved_j = current_h[i], current_h[j]
            current_h[i] = saved_i - sign * d
            current_h[j] = saved_j + sign * d
            saved = (i, j, saved_i, saved_j)
        elif r < 0.8:
            saved = current_h.copy()
            pert = rng.standard_normal(n) * scale * rng.uniform(0.3, 1.5)
            pert -= np.mean(pert)
            current_h = project_box_sum(current_h + pert, s)
        else:
            i = rng.integers(0, n)
            saved = current_h.copy()
            current_h = project_box_sum(current_h + rng.uniform(-scale, scale) * (np.eye(n)[i] - 1.0/n * np.ones(n)), s)

        cc = evaluate_c5(current_h)
        delta = cc - current_c
        if delta <= 0 or (temp > 1e-10 and rng.random() < np.exp(-delta / temp)):
            current_c = cc
            acc_count += 1
            if cc < local_best_c:
                local_best_c = cc
                local_best_h = current_h.copy()
                temp = max(0.0005, temp * 0.995)
        else:
            if saved is not None:
                if isinstance(saved, tuple):
                    current_h[saved[0]] = saved[2]
                    current_h[saved[1]] = saved[3]
                else:
                    current_h = saved.copy()

        if total_count % 200 == 0:
            rate = acc_count / max(1, total_count)
            if rate < 0.1:
                scale *= 0.7
            elif rate > 0.7:
                scale *= 1.2
            scale = min(0.2, max(0.005, scale))
            temp *= 0.98
            acc_count = 0
            total_count = 0

    return local_best_h, local_best_c


def coordinate_fine(h_init, n, s, time_limit, seed, best_c):
    """Very fine single-coordinate adjustments."""
    rng = np.random.default_rng(seed)
    t0 = time.perf_counter()
    current_h = h_init.copy()
    current_c = best_c
    local_best_h = current_h.copy()
    local_best_c = best_c

    no_improve = 0
    while time.perf_counter() - t0 < time_limit and no_improve < 20:
        order = rng.permutation(n)
        ni = 0
        for ii in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            i = order[ii]
            # Try small adjustments
            for delta in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                for sign in [1, -1]:
                    new_val = current_h[i] + sign * delta
                    if new_val < 0 or new_val > 1:
                        continue
                    saved = current_h[i]
                    # Apply and fix sum
                    current_h[i] = new_val
                    residual = s - np.sum(current_h)
                    # Fix by adjusting other coordinates proportionally
                    others = np.arange(n) != i
                    if residual > 0:
                        slack = 1.0 - current_h[others]
                        ts = np.sum(slack)
                        if ts < abs(residual):
                            current_h[i] = saved
                            continue
                        current_h[others] += residual * (slack / ts)
                    else:
                        ts = np.sum(current_h[others])
                        if ts < abs(residual):
                            current_h[i] = saved
                            continue
                        current_h[others] += residual * (current_h[others] / ts)
                    current_h = np.clip(current_h, 0, 1)
                    cc = evaluate_c5(current_h)
                    if cc < current_c - 1e-13:
                        current_c = cc
                        if cc < local_best_c:
                            local_best_c = cc
                            local_best_h = current_h.copy()
                        ni += 1
                        break
                    else:
                        current_h[i] = saved
                        # Need to re-fix the array; just restore
                        current_h = repair_exact(current_h, s)
                if cc < current_c:
                    break
        no_improve = 0 if ni > 0 else no_improve + 1

    return local_best_h, local_best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def elapsed():
        return time.perf_counter() - t_start

    def remaining():
        return budget_s - elapsed()

    # Get initial values
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel().copy()
        n = len(parent)
    except (NameError, AttributeError, ValueError):
        n = 47
        k = np.arange(n)
        parent = 0.5 + 0.25 * np.cos(2 * np.pi * k / n) + 0.1 * np.sin(4 * np.pi * k / n)

    s = n / 2.0
    parent = project_box_sum(parent, s)
    parent = repair_exact(parent, s)
    parent_c = evaluate_c5(parent)

    best_h = parent.copy()
    best_c = parent_c
    best_n = n

    if remaining() < 5:
        return best_h, best_c, best_n

    # Phase 1: Smooth descent
    t_alloc = 0.35 * budget_s
    if remaining() > t_alloc + 10:
        h_s, c_s = smooth_descent(best_h, n, s, t_alloc, seed, best_c)
        if c_s < best_c:
            best_h = h_s.copy()
            best_c = c_s

    if remaining() < 10:
        return best_h, best_c, best_n

    # Phase 2: Pair transfer (fine)
    t_alloc = 0.25 * budget_s
    if remaining() > t_alloc + 10:
        h_p, c_p = pair_transfer(best_h, n, s, t_alloc, seed + 100, best_c, fine=True)
        if c_p < best_c:
            best_h = h_p.copy()
            best_c = c_p

    if remaining() < 10:
        return best_h, best_c, best_n

    # Phase 3: SA
    t_alloc = min(0.25 * budget_s, remaining() * 0.5)
    if t_alloc > 5:
        h_a, c_a = sa_optimize(best_h, n, s, t_alloc, seed + 200, best_c)
        if c_a < best_c:
            best_h = h_a.copy()
            best_c = c_a

    if remaining() < 10:
        return best_h, best_c, best_n

    # Phase 4: More pair transfer fine
    t_alloc = min(0.15 * budget_s, remaining() * 0.5)
    if t_alloc > 5:
        h_p2, c_p2 = pair_transfer(best_h, n, s, t_alloc, seed + 300, best_c, fine=True)
        if c_p2 < best_c:
            best_h = h_p2.copy()
            best_c = c_p2

    if remaining() < 10:
        return best_h, best_c, best_n

    # Phase 5: Coordinate fine
    t_alloc = max(0, remaining() - 10)
    if t_alloc > 5:
        h_cf, c_cf = coordinate_fine(best_h, n, s, t_alloc, seed + 400, best_c)
        if c_cf < best_c:
            best_h = h_cf.copy()
            best_c = c_cf

    # Final verification
    best_h = repair_exact(best_h, s)
    final_c = evaluate_c5(best_h)
    if not np.isfinite(final_c) or final_c < 0:
        best_h = parent.copy()
        final_c = parent_c

    return best_h, final_c, best_n
