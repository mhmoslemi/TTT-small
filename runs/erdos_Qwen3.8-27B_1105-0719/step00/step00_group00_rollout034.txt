    n = len(h)

    shifted = np.clip(shifted, -500, 500)

    sum_exp = np.sum(exp_shifted)
    obj = dx * (max_corr + np.log(sum_exp) / eps)

            if 0 <= k + d < n:
                g_k += (-h[k + d]) * weights[s]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = (corr - max_c) * eps
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    return dx * (max_c + np.log(np.sum(e)) / eps)

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = (corr - max_c) * eps
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < 1e-15:
            continue
        d = s - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n - 1, n - 1 + d)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += ws * (1.0 - h[idx - d])
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += ws * (-h[idx + d])
    return dx * grad

            h += diff / len(h)

def slsqp_optimize(h0, target_sum, eps, maxiter=1000):
    n = len(h0)
    
    def obj(h):
        return smooth_obj(h, eps)
    
    def grad(h):
        return smooth_grad(h, eps)
    
    def con_fun(h):
        return np.sum(h) - target_sum
    
    def con_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def binary_local_search(h_bin, max_sweeps=50, rng=None):
    n = len(h_bin)
    h = np.array(h_bin, dtype=float).copy()
    best = compute_c5(h)
    
    if rng is None:
        rng = np.random
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for sweep in range(max_sweeps):
        improved = False
        # Shuffle for diversity
        if sweep > 0:
            ones = rng.permutation(ones)
            zeros = rng.permutation(zeros)
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < best - 1e-15:
                    best = new_c5
                    improved = True
                    # Update lists
                    ones = np.delete(ones, np.where(ones == i)[0][0])
                    zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                    ones = np.append(ones, j)
                    zeros = np.append(zeros, i)
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            break
    
    return h, best

        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        order = np.argsort(-h_interp)
        h_bin_int = np.zeros(n_try)
        half = n_try // 2
        h_bin_int[order[:half]] = 1.0
        inits.append(h_bin_int.copy())

        for h_init in inits:
            if time_left() < 25:
                break

            h_ls, c5_ls = binary_local_search(h_init, max_iters=20000, rng=rng)

            c5_candidate = c5_ls
            h_candidate = h_ls.copy()

            for eps2 in [20.0, 50.0, 100.0]:
                if time_left() < 15:
                    break
                h_pol = slsqp_optimize(h_candidate, target_try, eps=eps2, maxiter=1500)
                h_pol = project_feasible(h_pol, target_try)
                c5_pol = compute_c5(h_pol)
                if c5_pol < c5_candidate:
                    c5_candidate = c5_pol
                    h_candidate = h_pol

            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate

    # Phase 4: Final polish at best n
    n_best = len(best_h)
    target_best = n_best / 2.0
    for eps in [50.0, 100.0, 200.0, 500.0]:
        if time_left() < 10:
            break
        h_new = slsqp_optimize(best_h, target_best, eps=eps, maxiter=3000)
        h_new = project_feasible(h_new, target_best)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5, len(best_h)
