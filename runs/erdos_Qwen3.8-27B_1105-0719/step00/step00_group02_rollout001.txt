import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def fix_sum(h, target_sum):
    h = h.astype(np.float64).copy()
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        n_free = int(np.sum(mask))
        h[mask] = np.clip(h[mask] + diff / n_free, 0.0, 1.0)
    return h


def optimize_pair_fine(h, i, j):
    S = h[i] + h[j]
    low = max(0.0, S - 1.0)
    high = min(1.0, S)
    if high - low < 1e-12:
        return 0.0
    orig_val = evaluate_c5(h)
    best_x = h[i]
    best_val = orig_val
    n_coarse = 40
    step = (high - low) / n_coarse
    for k in range(n_coarse + 1):
        x = low + k * step
        h[i] = x
        h[j] = S - x
        v = evaluate_c5(h)
        if v < best_val:
            best_val = v
            best_x = x
    if low + 0.5 * step < best_x < high - 0.5 * step:
        lo_f = max(low, best_x - 2 * step)
        hi_f = min(high, best_x + 2 * step)
        if hi_f - lo_f > 1e-12:
            n_fine = 120
            for k in range(n_fine + 1):
                x = lo_f + (hi_f - lo_f) * k / n_fine
                h[i] = x
                h[j] = S - x
                v = evaluate_c5(h)
                if v < best_val:
                    best_val = v
                    best_x = x
    h[i] = best_x
    h[j] = S - best_x
    return orig_val - best_val


def full_pass(h, order):
    n = len(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    if order == 0:
        pass
    elif order == 1:
        pairs = list(reversed(pairs))
    else:
        rng_local = np.random.default_rng(order + 1000)
        rng_local.shuffle(pairs)
    _, _ = full_pass_inner(h, pairs)
    return h


def full_pass_inner(h, pairs):
    total_imp = 0.0
    for (i, j) in pairs:
        imp = optimize_pair_fine(h, i, j)
        total_imp += imp
    return h, total_imp
