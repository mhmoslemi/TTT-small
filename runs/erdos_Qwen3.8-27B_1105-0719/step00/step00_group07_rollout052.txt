import numpy as np
import time

def project_box(y, target):
    """Project y onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(y)
    target = max(0, min(n, target))
    # Binary search for lambda
    lo = y.min() - 1 - target
    hi = y.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    x = np.clip(y - lam, 0, 1)
    # Fix residual
    r = target - x.sum()
    if r > 1e-12:
        # Add to indices with slack
        order = np.argsort(x)  # smallest first = most slack
        for idx in order:
            if r <= 0: break
            add = min(r, 1 - x[idx])
            x[idx] += add
            r -= add
    elif r < -1e-12:
        order = np.argsort(-x)  # largest first
        for idx in order:
            if r >= 0: break
            sub = min(-r, x[idx])
            x[idx] -= sub
            r += sub
    return x

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return dx * c.max()

def smoothed_max_and_grad(c, beta):
    """Compute log-sum-exp smoothed max and its gradient w.r.t. c."""
    m = c.max()
    exps = np.exp(beta * (c - m))
    total = exps.sum()
    F = m + np.log(total) / beta
    w = exps / total
    return F, w

    n = len(h)
    # w has length 2n-1
    # grad[p] = sum_j w[n-1+p-j]*(1-h[j]) - sum_j w[n-1+j-p]*h[j]

def project_box(y, target):
    """Project y onto {x: 0<=x<=1, sum(x)=target}"""
    n_pts = len(y)
    target = float(np.clip(target, 0, n_pts))
    lo = y.min() - target - 1
    hi = y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    r = target - x.sum()
    if r > 1e-10:
        order = np.argsort(x)
        for idx in order:
            if r <= 0:
                break
            add = min(r, 1.0 - x[idx])
            x[idx] += add
            r -= add
    elif r < -1e-10:
        order = np.argsort(-x)
        for idx in order:
            if r >= 0:
                break

            x[idx] -= sub

def continuous_optimize(h0, t_limit):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best = h.copy()
    best_val = get_c5(h)
    t0 = time.time()
    beta = 25.0
    
    while time.time() - t0 < t_limit:
        lr = 0.08
        stuck = 0
        while stuck < 40 and time.time() - t0 < t_limit:
            g = grad_smoothed(h, beta)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-10:
                break
            improved = False
            for _ in range(25):
                h_try = project_box(h - lr * g, target)
                v = get_c5(h_try)
                if v < best_val - 1e-12:
                    h = h_try
                    best_val = v
                    best = h.copy()
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                stuck += 1
                lr = 0.15
        beta = min(beta * 1.4, 300.0)
        if beta >= 300.0:
            break
    
    return best, best_val

def bin_sa(n, t_limit, seed=0, init=None):
    rng = np.random.RandomState(seed)
    half = n // 2
    
    if init is not None:
        h = (init > 0.5).astype(np.float64)
        cnt = int(h.sum())
        if cnt > half:
            pos = np.where(h == 1)[0]
            rng.shuffle(pos)
            h[pos[:cnt - half]] = 0.0
        elif cnt < half:
            pos = np.where(h == 0)[0]
            rng.shuffle(pos)
            h[pos[:half - cnt]] = 1.0
    else:
        h = np.zeros(n)
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    
    c = get_c(h)
    cost = c.max()
    best_h = h.copy()
    best_cost = cost
    T = 4.0
    t0 = time.time()
    count = 0
    
    while time.time() - t0 < t_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c = get_c(h)
        new_cost = int(c.max())
        d = new_cost - cost
        
        if d <= 0 or (T > 0.01 and rng.random() < np.exp(-d / T)):
            cost = new_cost
            if cost < best_cost:
                best_cost = cost
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        count += 1
        if count % 500 == 0:
            T *= 0.97
        if T < 0.02:
            T = 0.02
    
    return best_h, best_cost

def local_search(h, max_rounds=50):
    n = len(h)
    c5_max = get_c(h).max()
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if improved:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c_new = get_c(h)
                m_new = c_new.max()
                if m_new < c5_max:
                    c5_max = m_new
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, c5_max

    rng = np.random.RandomState(seed)
    half = n // 2
    h = np.zeros(n)
    # Use a mix of structured and random
    # Quadratic residue-like pattern
    p = n // 2
    for i in range(n):
        if i < n and (i * i) % n != 0:
            h[i] = 1.0
    # Fix to have exactly half ones
    cnt = int(h.sum())
    if cnt > half:
        pos = np.where(h == 1)[0]
        rng.shuffle(pos)
        h[pos[:cnt - half]] = 0.0
    elif cnt < half:
        pos = np.where(h == 0)[0]
        rng.shuffle(pos)
        h[pos[:half - cnt]] = 1.0
    return h
