    # Bisection on lambda: h_proj[i] = clip(h[i] - lambda, 0, 1)
    # sum(h_proj) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    # Fix any numerical issues
    diff = target_sum - h_proj.sum()

    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(h, target_sum):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:

import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_set(h, target_sum):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = beta * corr * dx
    s_max = float(np.max(s))
    e = np.exp(s - s_max)
    w = e / float(np.sum(e))
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-10:
            continue
        lag = j - (n - 1)
        a = max(0, lag)
        b = min(n - 1, n - 1 + lag)
        if a <= b:
            g[a:b+1] += wj * (1.0 - h[a - lag:b - lag + 1])
        c = max(0, -lag)
        d = min(n - 1, n - 1 - lag)
        if c <= d:
            g[c:d+1] -= wj * h[c + lag:d + lag + 1]
    return dx * g
