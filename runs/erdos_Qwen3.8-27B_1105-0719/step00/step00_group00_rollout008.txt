    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    return float(np.max(c) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def random_feasible(n, rng):
    return project_feasible(rng.uniform(0.0, 1.0, size=n), n)

def optimize_slsqp(h_init, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, "full")
        return float(np.max(c)) * dx
    def con_fun(x):
        return float(np.sum(x) - n / 2.0)
    cons = [{'type': 'eq', 'fun': con_fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        h_opt = project_feasible(h_init, n)
    return h_opt, compute_c5(h_opt)

def optimize_lsmmax(h_init, n, betas=(15, 40, 80, 150, 300), maxiter=200):
    h = h_init.copy()
    for beta in betas:
        def obj_ls(x):
            c = np.correlate(x, 1.0 - x, "full")
            cmax = np.max(c)
            scaled = beta * (c - cmax)
            scaled = np.clip(scaled, -500, 500)
            val = cmax + np.log(np.sum(np.exp(scaled))) / beta
            return float(val) * (2.0 / n)
        def con_fun(x):
            return float(np.sum(x) - n / 2.0)
        cons = [{'type': 'eq', 'fun': con_fun}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_ls, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h = project_feasible(np.clip(res.x, 0, 1), n)
        except Exception:
            pass
    return h, compute_c5(h)

def pair_local_search(h, n, max_rounds=100):
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        c = np.correlate(h, 1.0 - h, "full")

                continue

                    continue

def basin_hopping(h_init, n, rng, n_inner=3, time_limit=15):
    t0 = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
