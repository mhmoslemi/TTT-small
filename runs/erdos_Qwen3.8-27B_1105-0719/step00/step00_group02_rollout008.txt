1. We need to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx
2. The constraint is ∫₀² h(x) dx = 1
3. We're discretizing with n_points samples over [0,2]
4. Current record: C₅ ≤ 0.38085
5. Target: C₅ ≤ 0.3808

The strategy describes an exact-direction constrained descent method where:
- We maintain feasibility (0 ≤ h_i ≤ 1, sum(h) = n/2)
- We generate balanced directions (sum(u) = 0)
- We move along rays h(t) = h + t*u
- We solve the 1D max-quadratic problem exactly along each ray
- We accept if the exact score improves

Let me implement this step by step:

1. Start from `initial_h_values` (n=47)
2. Generate diverse directions (random, low-frequency, active-lag-derived, elite-difference)
3. For each direction, compute the feasible interval [t_low, t_high]
4. Enumerate candidate points (endpoints, stationary points, pairwise intersections)
5. Evaluate the max-quadratic at all candidates
6. Accept the best if it improves the exact score

Key implementation details:
- `np.correlate(h, 1-h, mode="full")` gives 2n-1 values
- The score is `dx * max(those values)`
- Along a ray h+t*u: corr(h+t*u, 1-h-t*u) = q0 + t*q1 + t^2*q2
  where q0 = corr(h, 1-h), q1 = corr(u, 1-h) + corr(h, -u) = corr(u,1-h) - corr(h,u)
  and q2 = corr(u, -u) = -corr(u, u)

Let me think about the quadratic coefficients more carefully:
- corr(h+t*u, 1-(h+t*u)) = corr(h+t*u, 1-h-t*u)
- By bilinearity: = corr(h, 1-h) + t*corr(u, 1-h) + t*corr(h, -u) + t^2*corr(u, -u)
- = corr(h, 1-h) + t*(corr(u, 1-h) - corr(h, u)) + t^2*(-corr(u, u))

So:
- q0 = np.correlate(h, 1-h, "full")
- q1 = np.correlate(u, 1-h, "full") - np.correlate(h, u, "full")
- q2 = -np.correlate(u, u, "full")

For the 1D max-quadratic solve, I need to find candidate t values:
1. Endpoints t_low, t_high
2. For each a: stationary point of q0[a] + q1[a]*t + q2[a]*t^2, i.e., t = -q1[a]/(2*q2[a]) if q2[a] != 0
3. For each pair (a,b): solve (q0[a]-q0[b]) + (q1[a]-q1[b])*t + (q2[a]-q2[b])*t^2 = 0

import numpy as np
from scipy.optimize import minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_to_k(vec, mass):
    """Project vector to [0,1]^n with sum = mass using bisection on lambda"""
    n = len(vec)
    vec = np.clip(vec, 0.0, 1.0)
    # Bisection on lambda: x_i = clip(vec_i - lambda, 0, 1)
    lo, hi = vec.min() - 1.0, vec.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        x = np.clip(vec - mid, 0.0, 1.0)
        s = x.sum()
        if s > mass:
            lo = mid
        else:
            hi = mid
    x = np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = mass - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            caps = np.maximum(0.0, 1.0 - x)
            total_cap = caps.sum()
            if total_cap > 0:
                x += residual * caps / total_cap
                x = np.clip(x, 0.0, 1.0)
        else:
            affords = np.maximum(0.0, x)
            total_aff = affords.sum()
            if total_aff > 0:
                x += residual * affords / total_aff
                x = np.clip(x, 0.0, 1.0)
    # Final tiny repair
    residual = mass - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            caps = np.maximum(0.0, 1.0 - x)
            total_cap = caps.sum()
            if total_cap > 0:
                x += residual * caps / total_cap
        else:
            affords = np.maximum(0.0, x)
            total_aff = affords.sum()
            if total_aff > 0:
                x += residual * affords / total_aff
    return x

def compute_ray_coefficients(h, u):
    """Compute q0, q1, q2 for the ray h + t*u"""
    one_minus_h = 1.0 - h
    one_minus_hu = 1.0 - h - u
    q0 = np.correlate(h, one_minus_h, mode="full")
    q1 = np.correlate(u, one_minus_h, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")
    return q0, q1, q2

def feasible_interval(h, u, eps=1e-14):
    """Compute [t_low, t_high] such that 0 <= h + t*u <= 1"""
    t_low = 0.0
    t_high = 0.0
    # For each i where u_i > 0: t >= -h_i/u_i, t <= (1-h_i)/u_i
    # For each i where u_i < 0: t >= (1-h_i)/u_i, t <= -h_i/u_i
    pos = u > eps
    neg = u < -eps
    if np.any(pos):
        t_low = max(t_low, np.max(-h[pos] / u[pos]))
        t_high = min(t_high, np.min((1.0 - h[pos]) / u[pos]))
    if np.any(neg):
        t_low = max(t_low, np.max((1.0 - h[neg]) / u[neg]))
        t_high = min(t_high, np.min(-h[neg] / u[neg]))
    # Clamp to contain 0
    t_low = min(t_low, 0.0)
    t_high = max(t_high, 0.0)
    return t_low, t_high

def solve_max_quadratic(q0, q1, q2, t_low, t_high, n_chunks=50000):
    """Solve min_t max_a (q0[a] + q1[a]*t + q2[a]*t^2) over [t_low, t_high]"""
    n_lags = len(q0)
    width = t_high - t_low
    if width <= 1e-15:
        return 0.0
    
    candidates = [t_low, t_high, 0.0]
    
    # Stationary points of individual quadratics
    for a in range(n_lags):
        if abs(q2[a]) > 1e-15:
            t_star = -q1[a] / (2.0 * q2[a])
            if t_low - 1e-10 < t_star < t_high + 1e-10:
                candidates.append(t_star)
    
    # Pairwise intersections
    # For efficiency, process in batches
    for a in range(n_lags):
        for b in range(a + 1, n_lags):
            da = q0[a] - q0[b]
            db = q1[a] - q1[b]
            dc = q2[a] - q2[b]
            if abs(dc) > 1e-14:
                disc = db * db - 4.0 * dc * da
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2.0 * dc)
                    t2 = (-db - sqrt_disc) / (2.0 * dc)
                    if t_low - 1e-10 < t1 < t_high + 1e-10:
                        candidates.append(t1)
                    if t_low - 1e-10 < t2 < t_high + 1e-10:
                        candidates.append(t2)
            elif abs(db) > 1e-14:
                t1 = -da / db
                if t_low - 1e-10 < t1 < t_high + 1e-10:
                    candidates.append(t1)
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_low, t_high)
    
    # Deduplicate
    candidate_set = set()
    for c in candidates:
        candidate_set.add(round(c, 12))
    candidates = np.array(sorted(candidate_set))
    
    if len(candidates) == 0:
        return 0.0
    
    # Evaluate F_raw(t) = max_a (q0[a] + q1[a]*t + q2[a]*t^2) at all candidates
    # Process in chunks
    best_t = 0.0
    best_val = np.inf
    for i in range(0, len(candidates), n_chunks):
        batch = candidates[i:i+n_chunks]
        # Evaluate all lags at all batch points
        vals = q0[None, :] + q1[None, :] * batch[:, None] + q2[None, :] * batch[:, None] ** 2
        max_vals = vals.max(axis=1)
        idx = np.argmin(max_vals)
        if max_vals[idx] < best_val:
            best_val = max_vals[idx]
            best_t = batch[idx]
    
    return best_t

def generate_directions(h, n, rng, active_lags=None):
    """Generate a batch of diverse balanced directions"""
    directions = []
    
    # Random balanced directions
    for _ in range(3):
        u = rng.standard_normal(n)
        u -= u.mean()
        u_norm = np.linalg.norm(u)
        if u_norm > 1e-10:
            directions.append(u / u_norm)
    
    # Low-frequency directions
    idx = np.arange(n)
    for k in [1, 2, 3]:
        u_cos = np.cos(2.0 * np.pi * k * idx / n)
        u_cos -= u_cos.mean()
        u_sin = np.sin(2.0 * np.pi * k * idx / n)
        u_sin -= u_sin.mean()
        for u in [u_cos, u_sin]:
            u_norm = np.linalg.norm(u)
            if u_norm > 1e-10:
                directions.append(u / u_norm)
    
    # Active-lag derived directions
    if active_lags is not None and len(active_lags) > 0:
        # For each active lag, try to decrease it
        # The gradient of lag k w.r.t. h is: (1-h) shifted by k minus h shifted by k
        # Simplified: use the correlation structure
        for lag in active_lags[:5]:
            u = rng.standard_normal(n)
            # Bias towards reducing the active lag
            # For lag k in full correlation, it involves h[i] and (1-h)[i-k+offset]
            # Keep it simple: just add random perturbation
            u -= u.mean()
            u_norm = np.linalg.norm(u)
            if u_norm > 1e-10:
                directions.append(u / u_norm)
    
    # Pair moves
    for _ in range(5):
        i, j = rng.choice(n, size=2, replace=False)
        u = np.zeros(n)
        u[i] = 1.0
        u[j] = -1.0
        directions.append(u)
    
    return directions

def solve_max_quadratic_vectorized(q0, q1, q2, t_low, t_high):
    n_lags = len(q0)
    width = t_high - t_low
    if width <= 1e-15:
        return 0.0
    
    # Build all pairwise differences
    # DA[i,j] = q0[i] - q0[j], etc.
    # Shape: (n_lags, n_lags)
    DA = q0[:, None] - q0[None, :]
    DB = q1[:, None] - q1[None, :]
    DC = q2[:, None] - q2[None, :]
    
    # Only look at upper triangle (i < j)
    # Discriminant: DB^2 - 4*DC*DA
    disc = DB * DB - 4.0 * DC * DA
    
    # Find valid pairs (upper triangle, disc >= 0, |DC| > threshold)
    i_idx, j_idx = np.triu_indices(n_lags, k=1)
    da = DA[i_idx, j_idx]
    db = DB[i_idx, j_idx]
    dc = DC[i_idx, j_idx]
    d = disc[i_idx, j_idx]
    
    # Quadratic case
    valid_q = (np.abs(dc) > 1e-14) & (d >= 0)
    if np.any(valid_q):
        sqrt_d = np.sqrt(d[valid_q])
        t1 = (-db[valid_q] + sqrt_d) / (2.0 * dc[valid_q])
        t2 = (-db[valid_q] - sqrt_d) / (2.0 * dc[valid_q])
        candidates = np.concatenate([t1, t2])
    else:
        candidates = np.array([])
    
    # Linear case
    valid_l = (np.abs(dc) <= 1e-14) & (np.abs(db) > 1e-14)
    if np.any(valid_l):
        t_lin = -da[valid_l] / db[valid_l]
        candidates = np.concatenate([candidates, t_lin])
    
    # Stationary points
    valid_s = np.abs(q2) > 1e-15
    if np.any(valid_s):
        t_star = -q1[valid_s] / (2.0 * q2[valid_s])
        candidates = np.concatenate([candidates, t_star])
    
    # Endpoints and 0
    candidates = np.concatenate([candidates, [t_low, t_high, 0.0]])
    
    # Filter to feasible interval
    candidates = candidates[~np.isnan(candidates) & ~np.isinf(candidates)]
    candidates = candidates[(candidates >= t_low - 1e-10) & (candidates <= t_high + 1e-10)]
    candidates = np.clip(candidates, t_low, t_high)
    
    if len(candidates) == 0:
        return 0.0
    
    # Deduplicate
    candidates = np.unique(np.round(candidates, 10))
    
    # Evaluate
    best_t = candidates[0]
    best_val = np.inf
    chunk_size = 5000
    for i in range(0, len(candidates), chunk_size):
        batch = candidates[i:i+chunk_size]
        vals = q0[None, :] + q1[None, :] * batch[:, None] + q2[None, :] * batch[:, None] ** 2
        max_vals = vals.max(axis=1)
        idx = np.argmin(max_vals)
        if max_vals[idx] < best_val:
            best_val = max_vals[idx]
            best_t = batch[idx]
    
    return best_t
