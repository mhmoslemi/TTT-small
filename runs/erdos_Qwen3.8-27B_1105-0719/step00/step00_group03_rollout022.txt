    """Compute B_m for m = 0, ..., n-1"""
    B = np.zeros(n)
    # For m=0: B_0 is special (quadratic), handle separately
    # For m > 0:
    for m in range(1, n):

        if p + m < n: b += h[p+m]
        if p - m >= 0: b += h[p-m]
        if q + m < n: b -= h[q+m]
        if q - m >= 0: b -= h[q-m]

def c5_eval(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(full)

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Simple approach: clip and adjust
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Adjust: distribute the difference
    diff = target - s
    if diff > 0:
        # Need to add mass: add to positions with most room
        slack = 1.0 - h
        order = np.argsort(-slack)
        remaining = diff
        for i in order:
            add = min(remaining, slack[i])
            h[i] += add
            remaining -= add
            if remaining < 1e-12:
                break
    else:
        # Need to remove mass
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining < 1e-12:

def interp_h(h, n_old, n_new):
    """Interpolate h from n_old to n_new points, preserving integral"""
    # h is piecewise constant on [0, 2] with n_old intervals

    dx_new = 2.0 / n_new

    x_new = (np.arange(n_new) + 0.5) * dx_new
    
    # Find which old intervals each new point falls in

    # Actually, for piecewise constant, the value at a point is the value of the interval containing it
    # But for better interpolation, use the average of overlapping intervals
    
    # Simple approach: use np.interp on the piecewise linear interpolation of the step function

    h_interp = np.interp(x_new, x_old, h)
