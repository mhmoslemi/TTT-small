    if s < 1e-15:

def _lse_grad_h(h, beta):
    """Compute LSE value and gradient wrt h."""
    raw = _raw_corr(h)
    n = len(h)
    max_raw = raw.max()
    shifted = beta * (raw - max_raw)
    shifted = np.minimum(shifted, 0.0)  # clip negative overflow
    exp_v = np.exp(shifted)
    denom = exp_v.sum()
    if denom < 1e-300:
        return max_raw, np.zeros(n)
    w = exp_v / denom
    lse_val = max_raw + np.log(denom) / beta
    
    grad_h = np.zeros(n)
    for lag in range(-(n-1), n):
        idx = lag + (n - 1)
        wt = w[idx]
        if wt < 1e-14:
            continue
        if lag >= 0:
            grad_h[lag:] += wt * (1.0 - h[:n-lag])
            grad_h[:n-lag] += wt * (-h[lag:])
        else:
            q = -lag
            grad_h[:n-q] += wt * (1.0 - h[q:])
            grad_h[q:] += wt * (-h[:n-q])
    return lse_val, grad_h
