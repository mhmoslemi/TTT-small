def project_balanced(y, tol=1e-12):
    n = len(y)
    target = n / 2.0
    # Project onto {x: 0<=x_i<=1, sum(x)=target}
    # Use bisection on the Lagrange multiplier
    lo = -1.0
    hi = 1.0
    # Expand if needed
    for _ in range(20):
        if np.sum(np.clip(y + hi, 0, 1)) < target:
            hi *= 2
        elif np.sum(np.clip(y + lo, 0, 1)) > target:
            lo = lo * 2
        else:
            break
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + hi, 0, 1)  # use hi since sum(hi) >= target
    # Exact correction
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        # Spread residual over interior points
        interior = np.where((x > 1e-10) & (x < 1 - 1e-10))[0]
        if len(interior) > 0:
            per = residual / len(interior)
            x[interior] += per
            x = np.clip(x, 0, 1)
            # Final tiny correction
            residual2 = target - np.sum(x)
            if abs(residual2) > 1e-15:
                idx = np.where((x > 0) & (x < 1))[0]
                if len(idx) > 0:
                    x[idx[0]] += residual2
                    x = np.clip(x, 0, 1)
    return x

            abs(np.sum(h) - target) < tol)

def downsample(h, n_new):
    """Downsample h from n_old to n_new using area integration."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # For each new cell, integrate over overlapping old cells
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New cell j covers [j*dx_new, (j+1)*dx_new] in physical coords
        # Old cells covering this range:
        old_start = max(0, int(j * dx_new / dx_old))
        old_end = min(n_old, int(np.ceil((j+1) * dx_new / dx_old)))
        for i in range(old_start, old_end):
            # Overlap between old cell [i*dx_old, (i+1)*dx_old] and new cell [j*dx_new, (j+1)*dx_new]
            lo = max(i * dx_old, j * dx_new)
            hi = min((i+1) * dx_old, (j+1) * dx_new)
            if hi > lo:
                h_new[j] += h[i] * (hi - lo) / dx_new
    return h_new

def upsample(h, n_new):
    """Upsample h from n_old to n_new (piecewise constant)."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    # Map each new cell to the old cell it falls in
    # New cell j covers [j*dx_new, (j+1)*dx_new]
    # This might span multiple old cells, but for up sampling (n_new > n_old)
    # each new cell is smaller, so it fits in at most 2 old cells
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j+1) * dx_new
        i_start = int(x_lo / dx_old)
        i_end = min(n_old - 1, int((x_hi - 1e-15) / dx_old))
        # Distribute
        total_weight = 0.0
        val = 0.0
        for i in range(i_start, i_end + 1):
            cell_lo = i * dx_old
            cell_hi = (i+1) * dx_old
            overlap_lo = max(x_lo, cell_lo)
            overlap_hi = min(x_hi, cell_hi)
            if overlap_hi > overlap_lo:
                w = (overlap_hi - overlap_lo) / dx_new
                val += h[i] * w
                total_weight += w
        h_new[j] = val / total_weight if total_weight > 0 else h[min(i_start, n_old-1)]
    return h_new

def resample(h, n_new):
    """Resample h to n_new points."""
    if n_new == len(h):
        return h.copy()
    if n_new > len(h):
        return upsample(h, n_new)
    else:
        return downsample(h, n_new)

def annealing_step(h, rng, temperature):

    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i

    max_amt = h[src]  # can't go below 0

    if max_amt < 1e-10:

    if h_new[src] < -1e-10 or h_new[dst] > 1 + 1e-10:
