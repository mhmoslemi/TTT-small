def do_pair_pass(h, pairs, max_pairs=None):
    n = len(h)
    current_c = exact_score(h)
    improved = False
    n_try = len(pairs) if max_pairs is None else min(max_pairs, len(pairs))
    for idx in range(n_try):
        p, q = pairs[idx]
        a_new, b_new = pair_solve(h, p, q)
        if abs(a_new - h[p]) > 1e-13:
            h[p] = a_new
            h[q] = b_new
            new_c = exact_score(h)
            if new_c < current_c - 1e-14:
                current_c = new_c
                improved = True
    return h, current_c, improved

def pair_descent(h, start_time, time_limit, rng):
    n = len(h)
    h = h.copy()
    current_c = exact_score(h)
    best_c = current_c
    best_h = h.copy()
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    no_improve = 0
    pass_num = 0
    while True:
        elapsed = time.perf_counter() - start_time
        if elapsed > time_limit:
            break
        perm = rng.permutation(len(pairs))
        shuffled_pairs = [(pairs[k][0], pairs[k][1]) for k in perm]
        h, current_c, improved = do_pair_pass(h, shuffled_pairs)
        pass_num += 1
        if improved:
            no_improve = 0
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 2:
                break
    return best_h, best_c

def try_direction(h, start_time, time_limit, rng):

    dx = 2.0 / n

    C = float(np.max(prof))

    active_mask = active_vals >= thresh
    if np.sum(active_mask) < 2:
        active_idx = top_idx[:max(2, k // 4)]
    else:

    for j in range(ma):

            hp = h.copy(); hm = h.copy()
            hp[i] = min(1.0, h[i] + eps)
            hm[i] = max(0.0, h[i] - eps)

            if ae < 1e-12:
                hp[i] += eps; hm[i] -= eps; ae = 2 * eps
            pp = np.correlate(hp, 1.0 - hp, mode='full')[idx] * dx
            pm = np.correlate(hm, 1.0 - hm, mode='full')[idx] * dx
            grad[j, i] = (pp - pm) / ae
    c_vec = np.zeros(n + 1); c_vec[-1] = 1.0
    A_ub = np.zeros((ma, n + 1))
    A_ub[:, :n] = grad; A_ub[:, -1] = -1.0

    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(-5.0, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res is not None and res.success:

    except Exception:

def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)

    h_new = np.interp(xt, xs, h_src, left=h_src[0], right=h_src[-1])
    return h_new

    h_best_local, c_best_local = pair_descent(h_init, t_start, max(0, t_phase1_end - t_start), rng)
    update_best(h_best_local, c_best_local)

    # Phase 2: Try direction-based improvements
    elapsed = time.perf_counter() - t_start
    if elapsed < budget_s * 0.80:
        h_dir, c_dir = try_direction(h_best_local, t_start, t_phase1_end + budget_s * 0.15, rng)
        update_best(h_dir, c_dir)
        if c_dir < c_best_local:
            h2, c2 = pair_descent(h_dir, t_start, max(0, t_phase1_end + budget_s * 0.1 - elapsed), rng)
            update_best(h2, c2)

    # Phase 3: Try different dimensions
    elapsed = time.perf_counter() - t_start
    if elapsed < budget_s * 0.82:
        n_list = []
        if n != 64: n_list.append(64)
        if n != 80: n_list.append(80)
        if n != 96: n_list.append(96)
        for n_try in n_list:
            elapsed = time.perf_counter() - t_start
            if elapsed > budget_s * 0.85:
                break
            if n_try < 20 or n_try > 150:
                continue
            h_interp = interpolate_h(h_best_local, n_try)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            c_interp = exact_score(h_interp)
            t_local_end = min(t_deadline, t_start + budget_s * (0.85 + 0.05 * (n_list.index(n_try) + 1) / max(1, len(n_list))))
            if t_local_end - t_start > 5:
                h_opt, c_opt = pair_descent(h_interp, t_start, max(1, t_local_end - elapsed), rng)
                update_best(h_opt, c_opt)
            else:
                update_best(h_interp, c_interp)

    # Phase 4: More pair descent on the best found
    elapsed = time.perf_counter() - t_start
    if elapsed < t_deadline - 10:
        h_refined, c_refined = pair_descent(best_h, t_start, max(1, t_deadline - elapsed - 5), rng)
        update_best(h_refined, c_refined)

    # Phase 5: Micro-perturbation to escape plateaus
    elapsed = time.perf_counter() - t_start
    if elapsed < t_deadline - 5 and best_c > 0.3807:
        for trial in range(5):
            elapsed = time.perf_counter() - t_start
            if elapsed > t_deadline - 5:
                break
            h_pert = best_h + 0.01 * (rng.random(len(best_h)) - 0.5)
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            c_pert = exact_score(h_pert)
            t_end = min(t_deadline - 2, t_start + budget_s * 0.92)
            if t_end - elapsed > 10:
                h_po, c_po = pair_descent(h_pert, t_start, max(1, t_end - elapsed), rng)
                update_best(h_po, c_po)
            else:
                update_best(h_pert, c_pert)

    # Final validation
    final_h = best_h.copy()
    final_n = len(final_h)
    final_h = project_feasible(final_h, final_n / 2.0)
    final_c = exact_score(final_h)
    if not np.isfinite(final_c):
        final_h = np.ones(64) * 0.5
        final_n = 64
        final_c = 0.25

    return (np.asarray(final_h, dtype=np.float64), float(final_c), int(final_n))
