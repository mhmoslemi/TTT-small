import numpy as np
import time
import math

def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo = -z.max() - 1.0
    hi = -z.min() + 1.0
    
    def sum_at(lam):
        return np.clip(z + lam, 0, 1).sum()
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sum_at(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0, 1)
    
    # Fix summation error
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute equally
        h = h + diff / n
        h = np.clip(h, 0, 1)
        # If still off, adjust individual elements
        diff = target - h.sum()
        idx = np.argsort(np.abs(h - 0.5))[::-1]  # adjust elements furthest from 0.5 first? 
        # Actually, adjust elements with most room
        for i in range(n):
            if diff > 0:
                room = 1.0 - h[i]
                adj = min(diff, room)
                h[i] += adj
                diff -= adj
            elif diff < 0:
                room = h[i]
                adj = min(-diff, room)
                h[i] -= adj
                diff += adj
            if abs(diff) < 1e-15:
                break
    
    return h

def compute_smooth_and_grad(h, beta):
    """Compute smooth surrogate Q_beta and its gradient."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    
    # Compute all correlation values
    # r[center] = sum h[i]*(1-h[i])
    # r[center+m] = sum h[i+m]*(1-h[i]) for m=1..n-1
    # r[center-m] = sum h[i]*(1-h[i+m]) for m=1..n-1
    
    r = np.zeros(2 * n - 1)
    r[center] = np.sum(h * (1.0 - h))
    
    for m in range(1, n):
        r[center + m] = np.sum(h[m:] * (1.0 - h[:-m]))
        r[center - m] = np.sum(h[:-m] * (1.0 - h[m:]))
    
    s = r * dx  # actual overlap values
    
    # Softmax weights (stable)
    s_max = np.max(s)
    exp_s = np.exp(beta * (s - s_max))
    p = exp_s / np.sum(exp_s)
    
    Q = (s_max + np.log(np.sum(exp_s)) / beta)
    
    # Gradient
    grad = np.zeros(n)
    
    # Center term: p[center] * (1 - 2*h)
    grad += p[center] * dx * (1.0 - 2.0 * h)
    
    for m in range(1, n):
        # r[center+m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # dr/dh_j = (1-h[j-m]) * [m<=j<=n-1] - h[j+m] * [0<=j<=n-1-m]
        pm = p[center + m] * dx
        # For j from m to n-1: contribution (1-h[j-m])
        if m < n:
            grad[m:] += pm * (1.0 - h[:-m])
            # For j from 0 to n-1-m: contribution -h[j+m]
            grad[:n-m] += pm * (-h[m:])
        
        # r[center-m] = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # dr/dh_j = (1-h[j+m]) * [0<=j<=n-1-m] - h[j-m] * [m<=j<=n-1]
        pb = p[center - m] * dx
        if m < n:
            grad[:n-m] += pb * (1.0 - h[m:])
            grad[m:] += pb * (-h[:-m])
    
    return Q, grad

def smooth_descent(h, max_iters, beta, time_limit, start_time):
    """Projected smooth minimax descent."""
    n = len(h)
    h = h.copy()
    best_c = evaluate_c5(h)
    best_h = h.copy()
    
    step = 0.1
    failed = 0

        grad_centered = grad - np.mean(grad)

            break

def pair_line_search(h, i, j, n_samples=20):
    """Do a line search along the pair (i,j) transfer direction."""
    n = len(h)
    best_c = evaluate_c5(h)
    best_h = h.copy()
    
    # Feasible range for eps
    lo = max(-h[i], h[j] - 1.0)

    if hi - lo < 1e-14:
        return h, best_c
    
    # Sample points along the interval
    eps_values = np.linspace(lo, hi, n_samples)
    
    for eps in eps_values:
        new_h = h.copy()
        new_h[i] = h[i] + eps
        new_h[j] = h[j] - eps
        
        if new_h[i] < -1e-12 or new_h[i] > 1 + 1e-12:
            continue

            continue

            continue

def interpolate_h(h_old, n_new):
    """Interpolate h from n_old points to n_new points."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    if n_old == 1:
        return np.full(n_new, h_old[0])
    
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    target = 0.5 * n
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(z + mid, 0.0, 1.0)
        if h.sum() < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(z + lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        for i in range(n):
            if diff > 1e-15:
                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
            elif diff < -1e-15:
                adj = min(-diff, h[i])
                h[i] -= adj
                diff += adj
            else:

    return h

def smooth_value_and_grad(h, beta):

    dx = 2.0 / n

    r = np.zeros(2 * n - 1)

    for m in range(1, n):
        r[center + m] = np.dot(h[m:], 1.0 - h[:-m])
        r[center - m] = np.dot(h[:-m], 1.0 - h[m:])
    s = r * dx
    s_max = s.max()
    exp_s = np.exp(beta * (s - s_max))
    total = exp_s.sum()
    p = exp_s / total
    Q = s_max + np.log(total) / beta
    grad = np.zeros(n)
    grad += (p[center] * dx) * (1.0 - 2.0 * h)
    for m in range(1, n):
        pm = p[center + m] * dx
        pb = p[center - m] * dx

        grad[:n - m] -= pm * h[m:]
        grad[:n - m] += pb * (1.0 - h[m:])
