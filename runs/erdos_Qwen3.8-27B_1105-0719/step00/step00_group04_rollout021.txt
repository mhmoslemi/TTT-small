def optimize_coordinate_analytical(h, j):
    n = len(h)
    dx = 2.0 / n
    
    # Current correlation
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Coefficient of h[j] in each correlation output element
    # output[k] = sum_i h[i + (k-(n-1))] * (1-h[i]) * dx
    # Let l = k - (n-1), so output[k] = sum_i h[i+l] * (1-h[i]) * dx
    # h[j] appears when i+l = j (i = j-l, need 0<=j-l<n) or i = j (need 0<=j+l<n)
    
    coeffs = np.zeros(2*n-1)
    for k in range(2*n-1):
        l = k - (n-1)
        c = 0.0
        if 0 <= j - l < n:
            c += (1.0 - h[j - l])
        if 0 <= j + l < n:
            c -= h[j + l]
        coeffs[k] = c
    
    # val_k(t) = corr[k] + (t - h[j]) * coeffs[k]
    # = (corr[k] - h[j]*coeffs[k]) + t * coeffs[k]
    # = A[k] + t * B[k]
    A = corr - h[j] * coeffs
    B = coeffs
    
    # Find min of max_k (A[k] + t*B[k]) for t in [0,1]
    # This is a 1D convex minimization
    
    # Approach: check endpoints and all pairwise intersections
    candidates = [0.0, 1.0]
    
    active = np.abs(B) > 1e-10
    idx = np.where(active)[0]
    m = len(idx)
    
    for i in range(m):
        for j2 in range(i+1, m):
            denom = B[idx[i]] - B[idx[j2]]
            if abs(denom) > 1e-10:
                t_c = (A[idx[j2]] - A[idx[i]]) / denom
                if 0.0 <= t_c <= 1.0:
                    candidates.append(t_c)
    
    best_t = h[j]
    best_val = np.max(A + B * best_t)
    
    for t in candidates:
        val = np.max(A + B * t)
        if val < best_val:
            best_val = val
            best_t = t
    
    h[j] = best_t
    return best_val

from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h):
    """Project h to satisfy sum = n/2 and 0 <= h <= 1."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Need to increase some values
            mask = h < 1.0
            n_up = mask.sum()
            if n_up > 0:
                h[mask] += diff / n_up
                h = np.clip(h, 0, 1)
        else:
            mask = h > 0.0
            n_down = mask.sum()
            if n_down > 0:
                h[mask] += diff / n_down
                h = np.clip(h, 0, 1)

def coordinate_sweep(h, verbose=False):
    """One full coordinate descent sweep."""
    n = len(h)
    dx = 2.0 / n
    
    for j in range(n):
        # Compute current correlation
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Coefficient of h[j] in each output element
        coeffs = np.zeros(2*n - 1)
        for k in range(2*n - 1):
            l = k - (n - 1)
            c = 0.0
            if 0 <= j - l < n:
                c += (1.0 - h[j - l])
            if 0 <= j + l < n:
                c -= h[j + l]
            coeffs[k] = c
        
        A = corr - h[j] * coeffs
        B = coeffs
        
        # Find candidates: endpoints + pairwise intersections
        candidates = [0.0, 1.0]
        
        active_idx = np.where(np.abs(B) > 1e-10)[0]
        m = len(active_idx)
        
        for i in range(m):
            for j2 in range(i + 1, m):
                denom = B[active_idx[i]] - B[active_idx[j2]]
                if abs(denom) > 1e-10:
                    t_c = (A[active_idx[j2]] - A[active_idx[i]]) / denom
                    if 0.0 <= t_c <= 1.0:
                        candidates.append(t_c)
        
        best_t = h[j]
        best_val = np.max(A + B * best_t)
        
        for t in candidates:
            val = np.max(A + B * t)
            if val < best_val - 1e-14:
                best_val = val
                best_t = t
        
        h[j] = best_t
    
    return h

        B = coeffs

def optimize_1d_with_quadratic(A_lin, B_lin, c0):
    """
    Minimize max{c0 + t - t^2, max_i (A_lin[i] + B_lin[i]*t)} over t in [0,1].
    A_lin, B_lin: arrays of intercepts and slopes for the linear functions
    c0: constant for the quadratic (c0 + t - t^2)
    """
    candidates = [0.0, 1.0, 0.5]  # endpoints and quadratic vertex
    
    # Quadratic: q(t) = c0 + t - t^2
    # Lines: l_i(t) = A_lin[i] + B_lin[i]*t
    
    # Intersections of quadratic with each line:
    # c0 + t - t^2 = A_i + B_i*t
    # t^2 + (B_i - 1)*t + (A_i - c0) = 0
    for i in range(len(A_lin)):
        a_coef = 1.0
        b_coef = B_lin[i] - 1.0
        c_coef = A_lin[i] - c0
        disc = b_coef**2 - 4*a_coef*c_coef
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            t1 = (-b_coef - sqrt_disc) / (2*a_coef)
            t2 = (-b_coef + sqrt_disc) / (2*a_coef)
            if 0.0 <= t1 <= 1.0:
                candidates.append(t1)
            if 0.0 <= t2 <= 1.0:
                candidates.append(t2)
    
    # Intersections of pairs of lines:
    for i in range(len(A_lin)):
        for j in range(i+1, len(A_lin)):
            denom = B_lin[i] - B_lin[j]
            if abs(denom) > 1e-10:
                t = (A_lin[j] - A_lin[i]) / denom
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    
    # Evaluate at all candidates
    best_t = 0.5
    best_val = float('inf')
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        q = c0 + t - t*t
        lines = A_lin + B_lin * t
        val = max(q, np.max(lines)) if len(lines) > 0 else q
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def coordinate_sweep(h, rng=None):
    """One coordinate descent sweep."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)  # lags from -(n-1) to (n-1)
    
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    
    for j in order:
        # Current correlation (unscaled by dx for now, we'll multiply at end)
        corr_raw = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
        
        # Coefficients of h[j] in each lag (for l != 0, linear)
        coeffs = np.zeros(n_lags)
        for idx in range(n_lags):
            l = lags[idx]
            if l == 0:
                continue  # handle separately
            c = 0.0
            # h[j] as h[i+l]: i = j-l, need 0 <= j-l < n
            if 0 <= j - l < n:
                c += (1.0 - h[j - l])
            # h[j] as 1-h[i]: i = j, need 0 <= j+l < n
            if 0 <= j + l < n:
                c -= h[j + l]
            coeffs[idx] = c
        
        # For lags l != 0: new_value = corr_raw[idx] + (t - h[j]) * coeffs[idx]
        # = (corr_raw[idx] - h[j]*coeffs[idx]) + t * coeffs[idx]
        non_zero_mask = np.arange(n_lags) != (n-1)  # lag 0 is at index n-1
        A_lin = corr_raw[non_zero_mask] - h[j] * coeffs[non_zero_mask]
        B_lin = coeffs[non_zero_mask]
        
        # For lag 0 (index n-1): new_value = (corr_raw[n-1] - h[j]*(1-h[j])) + t*(1-t)
        c0 = corr_raw[n-1] - h[j] * (1.0 - h[j])
        
        # Now solve: minimize max{c0 + t - t^2, max_i(A_lin[i] + B_lin[i]*t)} over t in [0,1]
        best_t = h[j]
        best_val = float('inf')
        
        candidates = [0.0, 1.0, 0.5]
        
        # Intersections of quadratic with lines
        for i in range(len(A_lin)):
            # c0 + t - t^2 = A_lin[i] + B_lin[i]*t
            # t^2 + (B_lin[i] - 1)*t + (A_lin[i] - c0) = 0
            a_q = 1.0
            b_q = B_lin[i] - 1.0
            c_q = A_lin[i] - c0
            disc = b_q*b_q - 4.0*a_q*c_q
            if disc >= 0:
                sqrt_d = np.sqrt(disc)
                t1 = (-b_q - sqrt_d) / 2.0
                t2 = (-b_q + sqrt_d) / 2.0
                if 0.0 <= t1 <= 1.0:
                    candidates.append(t1)
                if 0.0 <= t2 <= 1.0:
                    candidates.append(t2)
        
        # Intersections of pairs of lines
        m = len(A_lin)
        for i in range(m):
            for j2 in range(i+1, m):
                denom = B_lin[i] - B_lin[j2]
                if abs(denom) > 1e-12:
                    t_c = (A_lin[j2] - A_lin[i]) / denom
                    if 0.0 <= t_c <= 1.0:
                        candidates.append(t_c)
        
        for t in candidates:
            t = float(np.clip(t, 0.0, 1.0))
            q_val = c0 + t - t*t
            if len(A_lin) > 0:
                lin_vals = A_lin + B_lin * t
                val = max(q_val, np.max(lin_vals))
            else:
                val = q_val
            if val < best_val:
                best_val = val
                best_t = t
        
        h[j] = best_t
    
    return h
