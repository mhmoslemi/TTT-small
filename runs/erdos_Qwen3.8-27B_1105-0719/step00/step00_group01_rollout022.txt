import numpy as np

def project_onto_feasible(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    
    def sum_clipped(lam):
        return np.sum(np.clip(v - lam, 0, 1))
    
    # Bisection for lambda
    lo, hi = v.min() - 2, v.max() + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(v - lam, 0, 1)
    
    # Fix rounding to ensure exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        idx = np.argsort(h)  # indices where h is closest to 0 or 1
        # Add to those with room
        if diff > 0:
            room = 1 - h
            order = np.argsort(-room)
            for i in order:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:

        else:

            order = np.argsort(-room)
            for i in order:
                sub = min(-diff, room[i])
                h[i] -= sub
                diff += sub

    grad *= dx
    return grad

def subgradient_direction(h, n):
    """Compute the subgradient of the exact max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    tol = 1e-10 * max_val + 1e-15

    grad *= dx / n_active
    return grad, max_val * dx

def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h, kind='stable')
            for i in order:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    break
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:

                diff += sub
