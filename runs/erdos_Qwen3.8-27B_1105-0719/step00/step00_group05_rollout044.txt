def _sig(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    e = np.exp(t[~pos])
    out[~pos] = e / (1.0 + e)
    return out

def _logit_of(h, eps=1e-10):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def _h_from_y(y):
    n = len(y)
    target = n / 2.0
    lo = -np.max(y) - 80.0
    hi = -np.min(y) + 80.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(_sig(y + mid)) < target:
            lo = mid
        else:
            hi = mid
    return _sig(y + 0.5 * (lo + hi))

def _exact_c(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def _validate(h):
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-7:
        return False
    return True

def _project(h):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(100):

def _smooth_obj(y, beta):
    h = _h_from_y(y)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    bc_max = np.max(bc)
    e = np.exp(bc - bc_max)
    w = e / np.sum(e)
    F = (np.max(corr) + (1.0 / beta) * np.log(np.sum(e))) * dx
    G = np.zeros(n)
    for p in range(len(corr)):
        if w[p] < 1e-14:
            continue
        lag = p - (n - 1)
        wp = w[p]
        if lag >= 0:
            G[lag:] += wp * (1.0 - h[:n - lag])
            G[:n - lag] -= wp * h[lag:]
        else:
            k = -lag
            G[:n - k] += wp * (1.0 - h[k:])
            G[k:] -= wp * h[:n - k]
    G *= dx
    _record(h)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-15:
        return F, np.zeros(n)
    B = np.dot(G, m) / M
    return F, m * (G - B)
