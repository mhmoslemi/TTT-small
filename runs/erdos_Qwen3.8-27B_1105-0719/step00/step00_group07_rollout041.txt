import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def evaluate_c5(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def make_binary_from_h(h):
    """Convert h to a binary sequence with correct sum, minimizing distance."""
    n = len(h)
    target = n / 2.0
    # Sort by h value, assign 1s to top n/2 (if n even)
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(h)[::-1]
        binary = np.zeros(n)
        binary[idx[:k]] = 1.0
        return binary
    else:
        # For odd n, we need sum = n/2 which is not integer
        # Keep the fractional part
        k = n // 2
        idx = np.argsort(h)[::-1]
        binary = np.zeros(n)
        binary[idx[:k]] = 1.0
        # The (k+1)-th element gets the fractional part
        binary[idx[k]] = 2 * target - k  # = 0.5 for n=95, target=47.5
        return binary

def smooth_max(corr, beta):
    """Log-sum-exp approximation of max."""
    m = np.max(corr)
    if beta * (np.max(corr) - m) < -700:
        return m
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def compute_grad_smooth(h, beta):
    """Compute gradient of smooth max of correlations w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    # Gradient of each correlation w.r.t. h
    # corr has length 2n-1
    # corr[n-1+l] = sum_{i=0}^{n-1-l} h[i+l]*(1-h[i]) for l >= 0
    # corr[n-1-l] = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l]) for l >= 0
    # corr[n-1] = sum h[i]*(1-h[i])
    
    grad = np.zeros(n)
    
    # Lag 0: d/dh[j] sum h[i](1-h[i]) = 1-2h[j]
    grad += weights[n-1] * (1 - 2*h)
    
    for l in range(1, n):
        # Positive lag: corr[n-1+l] = sum_{i=0}^{n-1-l} h[i+l]*(1-h[i])
        # d/dh[j] = (1-h[j-l])[l<=j] - h[j+l][j<=n-1-l]
        w_pos = weights[n-1+l]
        if l <= n-1:
            # Term from (1-h[j-l]): valid for j >= l
            if l <= n-1:
                grad[l:] += w_pos * (1 - h[:n-l])
            # Term from -h[j+l]: valid for j <= n-1-l
            if n-1-l >= 0:
                grad[:n-l] -= w_pos * h[l:]
        
        # Negative lag: corr[n-1-l] = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
        # d/dh[j] = (1-h[j+l])[j<=n-1-l] - h[j-l][l<=j]
        w_neg = weights[n-1-l]
        if n-1-l >= 0:
            grad[:n-l] += w_neg * (1 - h[l:])
            grad[l:] -= w_neg * h[:n-l]
    
    return dx * grad

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target_sum - np.sum(hp)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t*t)) / 2.0

        # Check improvement
        val = evaluate_c5(h)
        if val < best_val - 1e-12:

            best_h = h.copy()
    
    return best_h, best_val

            if val < best_val - 1e-12:

                best_val = val
                improved = True
