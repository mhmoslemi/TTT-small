Δc(tau) = δ * [(1-h[j-tau])*1_{0≤j-tau<n} - h[j+tau]*1_{0≤j+tau<n} - (1-h[i-tau])*1_{0≤i-tau<n} + h[i+tau]*1_{0≤i+tau<n}]

This is O(n) to compute for all lags (since there are 2n-1 lags and for each we just check a few indices). Actually it's O(1) per lag since we just look up 4 values. So total is O(n) per pair trial, which is the same as the correlate.

But we can precompute the per-lag coefficient for a given pair (i,j):
coeff[tau] = (1-h[j-tau])*1_{valid} - h[j+tau]*1_{valid} - (1-h[i-tau])*1_{valid} + h[i+tau]*1_{valid}

Then c_new(tau) = c_old(tau) + delta * coeff[tau]

And we want to minimize max_tau c_new(tau) = max(c_old + delta * coeff).

This is a 1D problem: find delta in [0, max_delta] that minimizes max(c_old + delta * coeff).

The function f(delta) = max_t (c_old[t] + delta * coeff[t]) is convex in delta (max of linear functions). So we can find the minimum by:
- The minimum is where the derivative changes sign
- Or we can do bisection on the subgradient

Actually, for a convex piecewise-linear function, the minimum is at a "kink" or where 0 is in the subdifferential. We can find it by:
- Computing the breakpoints where the argmax changes
- Or just doing a ternary-like search (but it's not unimodal in general... wait, it IS convex, so it IS unimodal)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_c_values(h):
    """Get all lag values (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(h, mass):
    h = np.array(h, dtype=np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > mass:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    diff = mass - np.sum(result)
    if abs(diff) > 1e-10:
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(interior):
            result[interior] += diff / np.count_nonzero(interior)
            result = np.clip(result, 0.0, 1.0)
    return result

def extreme_point_min(weights, mass):
    n = len(weights)
    order = np.argsort(weights, kind='stable')
    x = np.zeros(n)
    remaining = mass
    for i in order:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0 - 1e-12:
            x[i] = 1.0
            remaining -= 1.0
        else:
            x[i] = remaining
            remaining = 0.0
    return x

def compute_fw_gradient(h, top_indices, top_weights):
    """Compute gradient of sum_k top_weights[k] * c[top_indices[k]] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    js = np.arange(n)
    for idx, k_idx in enumerate(top_indices):
        w = top_weights[idx]
        if abs(w) < 1e-15:
            continue
        tau = k_idx - (n - 1)
        # d c(tau) / dh[j] = (1-h[j-tau])*1_{0<=j-tau<n} - h[j+tau]*1_{0<=j+tau<n}
        jt = js - tau
        mask1 = (jt >= 0) & (jt < n)
        grad[mask1] += w * (1.0 - h[jt[mask1]])
        
        jtau = js + tau
        mask2 = (jtau >= 0) & (jtau < n)
        grad[mask2] -= w * h[jtau[mask2]]
    return grad

def fw_step(h, c_vals, mass, K, beta):
    """One FW step."""
    n = len(h)
    sorted_idx = np.argsort(c_vals)[::-1]
    top_indices = sorted_idx[:K]
    top_vals = c_vals[top_indices]
    c_max = top_vals[0]
    
    # Softmax weights
    spread = c_max - top_vals[-1]
    if spread < 1e-12:
        top_weights = np.ones(K) / K
    else:
        top_weights = np.exp(-beta * (c_max - top_vals) / spread)
        top_weights = top_weights / np.sum(top_weights)
    
    # Compute gradient
    grad = compute_fw_gradient(h, top_indices, top_weights)
    
    # Find extreme point
    h_ext = extreme_point_min(grad, mass)
    
    if np.max(np.abs(h_ext - h)) < 1e-10:
        return h, evaluate_c5(h)
    
    # Line search
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    n_trials = 40
    gammas = np.linspace(0.0, 1.0, n_trials)
    vals = np.zeros(n_trials)
    for i, g in enumerate(gammas):
        candidate = (1.0 - g) * h + g * h_ext
        candidate = project_box_sum(candidate, mass)
        vals[i] = evaluate_c5(candidate)
    
    best_i = np.argmin(vals)
    if vals[best_i] < best_val - 1e-12:
        best_val = vals[best_i]
        best_h = (1.0 - gammas[best_i]) * h + gammas[best_i] * h_ext
        best_h = project_box_sum(best_h, mass)
        
        # Refine
        if 0 < best_i < n_trials - 1:
            g_lo = gammas[max(0, best_i - 2)]
            g_hi = gammas[min(n_trials - 1, best_i + 2)]
            fine_gammas = np.linspace(g_lo, g_hi, 20)
            for g in fine_gammas:
                candidate = (1.0 - g) * h + g * h_ext
                candidate = project_box_sum(candidate, mass)
                v = evaluate_c5(candidate)
                if v < best_val:
                    best_val = v
                    best_h = candidate.copy()
    
    return best_h, best_val

def pair_optimal_delta(c_vals, coeff, max_delta):
    """Find delta in [0, max_delta] minimizing max(c_vals + delta * coeff)."""
    if max_delta < 1e-12:
        return 0.0
    # f(delta) = max(c_vals + delta * coeff) is convex
    lo, hi = 0.0, max_delta
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c_vals + m1 * coeff)
        f2 = np.max(c_vals + m2 * coeff)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    start = time.perf_counter()
    best_h = h.copy()
    best_c = c_vals.copy()
    best_val = np.max(best_c)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_c_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(h, mass):
    h = np.array(h, dtype=np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > mass:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    diff = mass - np.sum(result)
    if abs(diff) > 1e-10:
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(interior):
            result[interior] += diff / np.count_nonzero(interior)
            result = np.clip(result, 0.0, 1.0)
    return result

def extreme_point_min(weights, mass):
    n = len(weights)
    order = np.argsort(weights, kind='stable')
    x = np.zeros(n)
    remaining = mass
    for i in order:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0 - 1e-12:
            x[i] = 1.0
            remaining -= 1.0
        else:
            x[i] = remaining
            remaining = 0.0
    return x

def compute_fw_gradient(h, top_indices, top_weights):
    n = len(h)
    grad = np.zeros(n)
    js = np.arange(n)
    for idx in range(len(top_indices)):
        w = top_weights[idx]
        if abs(w) < 1e-15:
            continue
        tau = top_indices[idx] - (n - 1)
        jt = js - tau
        mask1 = (jt >= 0) & (jt < n)
        grad[mask1] += w * (1.0 - h[jt[mask1]])
        jtau = js + tau
        mask2 = (jtau >= 0) & (jtau < n)
        grad[mask2] -= w * h[jtau[mask2]]
    return grad

def fw_step(h, c_vals, mass, K, beta):
    n = len(h)
    sorted_idx = np.argsort(c_vals)[::-1]
    top_indices = sorted_idx[:K]
    top_vals = c_vals[top_indices]
    c_max = top_vals[0]
    spread = c_max - top_vals[-1]
    if spread < 1e-12:
        top_weights = np.ones(K) / K
    else:
        top_weights = np.exp(-beta * (c_max - top_vals) / spread)
        top_weights = top_weights / np.sum(top_weights)
    grad = compute_fw_gradient(h, top_indices, top_weights)
    h_ext = extreme_point_min(grad, mass)
    if np.max(np.abs(h_ext - h)) < 1e-10:
        return h, evaluate_c5(h)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    n_trials = 30
    gammas = np.linspace(0.01, 0.99, n_trials)
    vals = np.full(n_trials, np.inf)
    for i, g in enumerate(gammas):
        candidate = (1.0 - g) * h + g * h_ext
        candidate = project_box_sum(candidate, mass)
        vals[i] = evaluate_c5(candidate)
    best_i = np.argmin(vals)
    if vals[best_i] < best_val - 1e-12:
        best_val = vals[best_i]
        best_h = (1.0 - gammas[best_i]) * h + gammas[best_i] * h_ext
        best_h = project_box_sum(best_h, mass)
        if 1 <= best_i <= n_trials - 2:
            g_lo = max(0.0, gammas[best_i] - 2 * (gammas[1] - gammas[0]))
            g_hi = min(1.0, gammas[best_i] + 2 * (gammas[1] - gammas[0]))
            fine = np.linspace(g_lo, g_hi, 15)
            for g in fine:
                candidate = (1.0 - g) * h + g * h_ext
                candidate = project_box_sum(candidate, mass)
                v = evaluate_c5(candidate)
                if v < best_val:
                    best_val = v
                    best_h = candidate.copy()
    return best_h, best_val

def pair_opt_delta(c_vals, coeff, max_delta):
    if max_delta < 1e-12:
        return 0.0
    lo, hi = 0.0, max_delta
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c_vals + m1 * coeff)
        f2 = np.max(c_vals + m2 * coeff)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def pair_polish(h, c_vals, n_pairs, rng, time_limit):
    n = len(h)
    start = time.perf_counter()
    best_h = h.copy()
    best_c = c_vals.copy()
    best_val = float(np.max(best_c))
    k_idx = np.arange(2 * n - 1)
    tau_arr = k_idx - (n - 1)
    for _ in range(n_pairs):
        if time.perf_counter() - start > time_limit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        coeff = np.zeros(2 * n - 1)
        jt = j - tau_arr
        m1 = (jt >= 0) & (jt < n)
        coeff[m1] += (1.0 - best_h[jt[m1]])
        jtau = j + tau_arr
        m2 = (jtau >= 0) & (jtau < n)
        coeff[m2] -= best_h[jtau[m2]]
        it = i - tau_arr
        m3 = (it >= 0) & (it < n)
        coeff[m3] -= (1.0 - best_h[it[m3]])
        itau = i + tau_arr
        m4 = (itau >= 0) & (itau < n)
        coeff[m4] += best_h[itau[m4]]
        md = min(best_h[i], 1.0 - best_h[j])
        if md > 1e-12:
            d = pair_opt_delta(best_c, coeff, md)
            if d > 1e-10:
                nc = best_c + d * coeff
                nv = np.max(nc)
                if nv < best_val - 1e-12:
                    best_h[i] -= d
                    best_h[j] += d
                    best_c = nc
                    best_val = nv
        md2 = min(best_h[j], 1.0 - best_h[i])
        if md2 > 1e-12:
            d = pair_opt_delta(best_c, -coeff, md2)
            if d > 1e-10:
                nc = best_c + d * (-coeff)
                nv = np.max(nc)
                if nv < best_val - 1e-12:
                    best_h[j] -= d
                    best_h[i] += d
                    best_c = nc
                    best_val = nv
    return best_h, best_val

def transfer_to_resolution(h, n_new):
