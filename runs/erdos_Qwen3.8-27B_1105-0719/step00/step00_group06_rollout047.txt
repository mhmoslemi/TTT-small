- If j <= n-2 (j appears in h[j]): contributes dx * (1-h[j+1])
- If j >= 1 (j appears in (1-h[j]) where j-1 is the other index): contributes dx * (-h[j-1])

So: ∂Φ_{-1}/∂h_j = dx*(1-h[j+1]) for j=0,...,n-2, minus dx*h[j-1] for j=1,...,n-1.

In vector form:
- grad[:n-1] += dx * (1 - h[1:])  (for j=0 to n-2, h[j+1] = h[1] to h[n-1])
- grad[1:] -= dx * h[:n-1]  (for j=1 to n-1, h[j-1] = h[0] to h[n-2])

My formula for a=1: grad[:n-a] += dx*(1-h[a:]) and grad[a:] -= dx*h[:n-a]
→ grad[:n-1] += dx*(1-h[1:]) and grad[1:] -= dx*h[:n-1] ✓

        h3, c3 = _slsqp_run(best_h, deadline,
                           tau_list=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 5: Binary swap again
    if time.time() < deadline - 50:
        hb2 = _binary_swap(best_h, rng, max_rounds=1500)
        cb2 = _exact_c5(hb2)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2

    # Phase 6: Random descent again
    if time.time() < deadline - 60:
        h4, c4 = _random_descent(best_h, rng, deadline - 20)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 7: Multiple perturbations + re-optimize
    if time.time() < deadline - 60:
        for ps in [0.003, 0.007, 0.012, 0.02]:
            if time.time() > deadline - 45:
                break
            h_pert = _project(best_h + ps * rng.randn(n))
            h5, c5 = _slsqp_run(h_pert, deadline,
                               tau_list=[0.01, 0.004, 0.0015, 0.0006])
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 8: Try smaller n for better bound
    if time.time() < deadline - 80:
        for target_n in [64, 56, 48]:
            if time.time() > deadline - 40:
                break
            h_small = _resample(best_h, target_n)
            h6, c6 = _slsqp_run(h_small, deadline,
                               tau_list=[0.01, 0.005, 0.002, 0.0008])
            hb6 = _binary_swap(h6, rng, max_rounds=500)
            c6b = _exact_c5(hb6)
            if c6b < c6:
                h6 = hb6
                c6 = c6b
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 9: Final polish - short SLSQP + binary
    if time.time() < deadline - 30:
        h7, c7 = _slsqp_run(best_h, deadline,
                           tau_list=[0.003, 0.001, 0.0004])
        if c7 < best_c:
            best_c = c7
            best_h = h7
        hb7 = _binary_swap(best_h, rng, max_rounds=500)
        cb7 = _exact_c5(hb7)
        if cb7 < best_c:
            best_c = cb7
            best_h = hb7

    # Final validation
    best_h = _project(best_h)
    final_c = _exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    # Ensure feasibility
    assert abs(np.sum(best_h) - len(best_h) / 2.0) < 1e-8, "Sum constraint violated"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10), "Box constraint violated"

    return best_h, best_c, len(best_h)
