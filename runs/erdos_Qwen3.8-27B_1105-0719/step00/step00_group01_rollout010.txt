def _eval_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _feasible_interval(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = -1e18
    t_hi = 1e18
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_lo >= t_hi:
        return 0.0, 0.0
    return float(t_lo), float(t_hi)

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) < 1e-12:
        return h
    if residual > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            add = min(residual, room[i])
            h[i] += add
            residual -= add
            if residual < 1e-12:
                break
    else:
        order = np.argsort(h)
        for i in order:
            rem = min(-residual, h[i])
            h[i] -= rem
            residual += rem
            if residual > -1e-12:
                break
    return h

            d -= np.mean(d)
            norm = np.linalg.norm(d)
            if norm < 1e-14:
                continue
            d *= rng.uniform(0.1, 0.4) / norm

        # Line search
        grid = 50 if phase <= 1 else 40
        if cur_c5 < 0.3809:
            grid = max(grid, 60)
        t_best = _line_search(h, d, n_grid=grid)

        if abs(t_best) > 1e-12:
            h_new = h + t_best * d
            h_new = _fix_sum(h_new, target_sum)
            c_new = _eval_c5(h_new)
            if c_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c_new
                no_improve = 0
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Binarization pressure every few hundred iterations
        if iter_count % 300 == 0 and no_improve > 500:
            # Push fractional values towards 0 or 1
            frac_mask = (h > 0.05) & (h < 0.95)
            if np.any(frac_mask):
                frac_idx = np.where(frac_mask)[0]
                for fi in frac_idx[:10]:
                    # Try pushing hi towards 0 or 1
                    target_val = 0.0 if h[fi] < 0.5 else 1.0
                    delta = target_val - h[fi]
                    # Find a partner to absorb
                    if delta > 0:
                        # Need to decrease another value
                        candidates = np.where(h > 0.1)[0]
                    else:
                        candidates = np.where(h < 0.9)[0]
                    candidates = candidates[candidates != fi]
                    if len(candidates) > 0:
                        pi = candidates[rng.randint(len(candidates))]
                        d2 = np.zeros(n)
                        d2[fi] = delta
                        d2[pi] = -delta
                        d2 -= np.mean(d2)
                        t2 = _line_search(h, d2, n_grid=30)
                        h_new = h + t2 * d2
                        h_new = _fix_sum(h_new, target_sum)
                        c_new = _eval_c5(h_new)
                        if c_new < cur_c5 - 1e-12:
                            h = h_new
                            cur_c5 = c_new
                            if c_new < best_c5:
                                best_c5 = c_new
                                best_h = h.copy()
                            no_improve = 0

        # Periodic restart from best
        if no_improve > max_no_improve:
            if rng.random() < 0.4:
                h = best_h.copy()
                cur_c5 = best_c5
            no_improve = 0
            max_no_improve = max(500, max_no_improve * 3 // 4)

        # Check target
        if best_c5 <= 0.3807:
            break

    # Final output
    h_out = best_h.copy()
    c5_out = _eval_c5(h_out)
    h_out = np.clip(h_out, 0.0, 1.0)
    n_out = len(h_out)
    return (h_out, c5_out, n_out)
