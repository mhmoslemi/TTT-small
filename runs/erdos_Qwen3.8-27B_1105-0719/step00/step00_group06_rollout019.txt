    """Create a random balanced binary vector with exactly n/2 ones."""
    n_half = n // 2
    idx = rng.permutation(n)
    b = np.zeros(n)
    b[idx[:n_half]] = 1.0

    """Convert binary to continuous: 0->epsilon, 1->1-epsilon"""
    return epsilon + (1.0 - 2.0 * epsilon) * b

def pair_line_search(h, u, v):

    s_lo = max(-h[u], h[v] - 1.0)
    s_hi = min(1.0 - h[u], h[v])

        return 0.0

        h[u] += s
        h[v] -= s

        h[v] += s
