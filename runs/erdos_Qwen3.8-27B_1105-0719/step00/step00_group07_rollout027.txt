def solve_c(theta, target):

    n = len(h)
    omh = 1.0 - h

    for q in range(n):
        corr[q] = np.dot(h[q:], omh[:n-q])

    """Compute LSE value and gradient w.r.t. h."""
    n = len(h)
    corr = compute_corr_values(h)
    # LSE
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)

    omh = 1.0 - h
    gh = np.convolve(w, omh, mode='full')[:n] - np.correlate(h, w, mode='full')[:n]

    val = max_c + np.log(s) / beta

            return theta_obj_grad(th, target, beta)

                    best_h = h_new.copy()
                    best_c5 = c5_new
                else:
                    best_theta = res.x.copy()
        except Exception:
            pass
    return best_theta, best_h, best_c5

                    th_new[j] += sign * ss

def cos_init(n, target, amp, freq, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + amp * np.cos(np.pi * freq * x + phase)

    hc = np.clip(h, eps, 1.0 - eps)

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return np.clip(h_new, 0.0, 1.0)

def adjust_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:

    mask = (h > 0.01) & (h < 0.99)

        return np.clip(h, 0.0, 1.0)
    h = h.copy()
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)
