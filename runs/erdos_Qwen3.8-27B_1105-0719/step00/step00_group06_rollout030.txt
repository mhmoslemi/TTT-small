def compute_gradients(h, n):
    """Compute gradient matrix: grad[k, i] = dP[k]/dh[i] for all k, i"""
    dx = 2.0 / n
    grad = np.zeros((2*n-1, n))
    # Center: k = n-1, m = 0
    grad[n-1, :] = 1.0 - 2.0 * h
    # Positive lags: k = n-1+s, m = s
    for s in range(1, n):
        k = n - 1 + s
        valid = np.arange(s, n)
        grad[k, valid] += (1.0 - h[valid - s])
        valid2 = np.arange(0, n - s)
        grad[k, valid2] -= h[valid2 + s]
    # Negative lags: k = n-1-s, m = -s
    for s in range(1, n):
        k = n - 1 - s
        valid = np.arange(0, n - s)
        grad[k, valid] += (1.0 - h[valid + s])
        valid2 = np.arange(s, n)
        grad[k, valid2] -= h[valid2 - s]
    grad *= dx
    return grad

        grad[k2, s:] -= h[:n-s] * dx
    return grad

def _lp_step(h, n, profile, grad, active_k, step_budget, lam_move):
    """
    Solve LP to find descent direction.
    Returns (W, X, t_val) or None if infeasible.
    """
    num_active = len(active_k)
    # Variables: W(n), X(n), t(1) → total 2n+1
    c_vec = np.concatenate([np.full(n, lam_move), np.full(n, lam_move), [1.0]])
    
    # Inequality constraints: grad[k,:]·(W-X) - t ≤ -P[k]
    # → -grad[k,:]·W + grad[k,:]·X + t ≥ P[k]
    # In A_ub form: grad[k,:]·W - grad[k,:]·X - t ≤ -P[k]
    A_ub_rows = []
    b_ub = []
    for k in active_k:
        row = np.concatenate([grad[k, :].copy(), -grad[k, :].copy(), [-1.0]])
        A_ub_rows.append(row)
        b_ub.append(-profile[k])
    
    if A_ub_rows:
        A_ub = np.column_stack(A_ub_rows)
        b_ub = np.array(b_ub)
    else:
        A_ub = np.zeros((1, 2*n+1))
        b_ub = np.array([0.0])
    
    # Equality: sum(W) - sum(X) = 0
    A_eq = np.zeros((1, 2*n+1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds
    W_bounds = [(0.0, 1.0 - h[i]) for i in range(n)]
    X_bounds = [(0.0, h[i]) for i in range(n)]
    t_bound = (profile.max() - step_budget, None)  # t can't exceed current max
    bounds = W_bounds + X_bounds + [t_bound]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            W = res.x[:n]
            X = res.x[n:2*n]
            t_val = res.x[2*n]
            return W, X, t_val
    except:
        pass
    return None

def _binary_local_search(h, n, max_iters=50, time_limit=None, start_time=None):
    """
    Perform binary local search: round h to binary, try all single swaps.
    Returns improved h.
    """
    # Round to binary
    m = n // 2
    # Sort by value, take top m
    h_bin = (h >= 0.5).astype(np.float64)
    # Fix sum
    ones_count = h_bin.sum()
    if ones_count != m:
        order = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[order[:m]] = 1.0
    
    current = h_bin.copy()
    current_score = _evaluate(current, n)
    
    for iter in range(max_iters):
        if time_limit and (start_time is not None):
            if time.time() - start_time > time_limit:
                break
        improved = False
        best_swap = None
        best_s = current_score
        
        ones = np.where(current == 1.0)[0]
        zeros = np.where(current == 0.0)[0]
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                current[i] = 0.0
                current[j] = 1.0
                s = _evaluate(current, n)
                current[i] = 1.0
                current[j] = 0.0
                if s < best_s - 1e-12:
                    best_s = s
                    best_swap = (i, j)
                    improved = True
        
        if not improved:
            break
        current[best_swap[0]] = 0.0
        current[best_swap[1]] = 1.0
        current_score = best_s
    
    return current, current_score

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break

            h[mask] += step
