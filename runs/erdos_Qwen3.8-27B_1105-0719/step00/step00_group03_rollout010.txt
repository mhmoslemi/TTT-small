def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(scores), scores

def project_balanced(y, tol=1e-10):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    n = len(y)
    target = n / 2.0
    # Bisection on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        # Find entries not at bounds
        interior = (x > 1e-10) & (x < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            adjust = residual / n_int
            x[interior] += adjust
            x = np.clip(x, 0.0, 1.0)
        else:
            # All at bounds, need to adjust
            if residual > 0:
                at_zero = np.where(x == 0.0)[0]
                for idx in at_zero:
                    amt = min(residual, 1.0)
                    x[idx] += amt
                    residual -= amt
                    if abs(residual) < 1e-12:
                        break
            else:
                at_one = np.where(x == 1.0)[0]
                for idx in at_one:
                    amt = min(-residual, 1.0)
                    x[idx] -= amt
                    residual += amt
                    if abs(residual) < 1e-12:
                        break
    return x

def compute_gradient(h, scores, weights):
    """Compute gradient of dx * sum_r weights[r] * scores[r] w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    # Build W matrix: W[i,j] = weights[n-1+i-j]
    W = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            W[i, j] = weights[n - 1 + i - j]
    grad = dx * (W @ (1.0 - h) - W.T @ h)
    return grad

def active_weights(scores, tol=1e-10):
    """Get weights for active (max) lags"""
    mx = np.max(scores)
    mask = scores >= mx - tol
    w = np.zeros(len(scores))
    w[mask] = 1.0 / np.sum(mask)
    return w

def soft_weights(scores, beta=5.0):
    """Get soft weights based on softmax"""
    mx = np.max(scores)
    if mx < 1e-15:
        return np.ones(len(scores)) / len(scores)
    w = np.exp(-beta * (mx - scores) / mx)
    w /= np.sum(w)
    return w

def is_feasible(h, tol=1e-8):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(np.sum(h) - n/2.0) > tol:

    return True

def pair_transfer(h, k, rng):
    """Perform a single pair transfer along lag k. Returns (new_h, amount) or (None, 0)."""
    n = len(h)
    if k >= 0:
        # pairs (i, i+k) for i in 0..n-k-1
        n_pairs = n - k
        if n_pairs <= 0:
            return None, 0.0
        i = rng.integers(0, n_pairs)
        j = i + k
        # Transfer: decrease h[i], increase h[j] or vice versa
        if rng.random() < 0.5:
            # decrease h[i], increase h[j]
            amt = min(h[i], 1.0 - h[j])
        else:
            # increase h[i], decrease h[j]
            amt = min(1.0 - h[i], h[j])
        if amt < 1e-12:
            return None, 0.0
        # Random fraction
        frac = rng.uniform(0.1, 1.0)
        amt *= frac
        new_h = h.copy()
        if new_h[i] >= h[i]:
            pass  # already same
        # Determine direction
        if h[j] < h[i]:
            # transfer from i to j: decrease i, increase j
            new_h[i] -= amt
            new_h[j] += amt
        else:
            # transfer from j to i: decrease j, increase i
            new_h[j] -= amt
            new_h[i] += amt
        return new_h, amt
    else:
        return pair_transfer(h, -k, rng)  # symmetric

    grad = compute_gradient(h, scores, weights)
    # Normalize
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    # Randomize direction slightly
    noise_scale = 0.01
    grad_noisy = grad + noise_scale * np.random.randn(len(grad)) * gn
    # Projected step
    candidates = []

        new_h = h - scale * grad_noisy
        new_h = project_balanced(new_h)
        candidates.append(new_h)

    """Upsample h to n_new points using area averaging"""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        # New cell [i*2/n_new, (i+1)*2/n_new)
        x_start = i * 2.0 / n_new
        x_end = (i + 1) * 2.0 / n_new
        # Find overlapping old cells
        # Old cell j covers [j*2/n_old, (j+1)*2/n_old)
        j_start = max(0, int(np.floor(x_start / dx_old)))
        j_end = min(n_old - 1, int(np.floor((x_end - 1e-12) / dx_old)))
        if j_start > j_end:
            h_new[i] = 0.0
            continue
        total_area = 0.0
        for j in range(j_start, j_end + 1):
            cell_start = j * dx_old
            cell_end = (j + 1) * dx_old
            overlap_start = max(x_start, cell_start)
            overlap_end = min(x_end, cell_end)
            if overlap_end > overlap_start:
                total_area += h[j] * (overlap_end - overlap_start)
        h_new[i] = total_area / dx_new
    # Project to maintain constraints
    return project_balanced(h_new)

def downsample(h, n_new):
    """Downsample h to n_new points using area averaging"""
    return upsample(h, n_new)  # same function works both ways

    return h_bin
