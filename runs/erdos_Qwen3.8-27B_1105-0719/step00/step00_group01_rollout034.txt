```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def subgradient_descent(h, n_iter=1000, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1-h, mode="full")
        max_val = np.max(corr)
        # Find active lags
        active = np.where(corr >= max_val - 1e-4)[0]
        
        # Compute subgradient
        sg = np.zeros(n)
        for m in active:
            L = m - (n-1)
            for j in range(n):
                t1 = 1-h[j-L] if 0 <= j-L < n else 0
                t2 = -h[j+L] if 0 <= j+L < n else 0
                sg[j] += t1 + t2
        sg /= len(active)
        
        # Make zero-sum
        sg -= np.mean(sg)
        
        # Step
        new_h = h - lr * sg
        new_h = np.clip(new_h, 0, 1)
        # Rescale to maintain sum
        target_sum = n / 2.0
        if new_h.sum() > 0:
            new_h *= target_sum / new_h.sum()
            new_h = np.clip(new_h, 0, 1)
        
        new_val = compute_c5(new_h)
        if new_val < best_val:
            best_val = new_val
            best_h = new_h.copy()
        
        h = new_h
        # Decrease learning rate
        lr *= 0.999
    
    return best_h, best_val
```

```python
        sg = get_subgrad(h)
        sg -= np.mean(sg)

        # Sort by subgradient to find best pairs
        order = np.argsort(sg)
        i_cand = order[0]
        j_cand = order[-1]

        if i_cand == j_cand:
            i_cand = order[0]
            j_cand = order[len(order)//2]
            if i_cand == j_cand:
                break

        d, v = pair_line_search(h, i_cand, j_cand, n, n_trials=12)
        if abs(d) > 1e-10:
            if d > 0:
                h[i_cand] += d
                h[j_cand] -= d
            else:
                h[j_cand] += (-d)
                h[i_cand] -= (-d)
            cur_val = v
            stagnant = 0
            if v < best_val:
                best_val = v
                best_h = h.copy()
        else:
            # Try second and third best pairs
            improved = False
            for trial in range(3):
                if trial == 0:
                    ii, jj = order[0], order[-1]
                elif trial == 1:
                    ii, jj = order[1], order[-2] if len(order) > 2 else (order[0], order[-1])
                else:
                    ii, jj = order[2], order[-3] if len(order) > 3 else (order[0], order[-1])
                if ii == jj:
                    continue
                d, v = pair_line_search(h, ii, jj, n, n_trials=8)
                if abs(d) > 1e-10:
                    if d > 0:
                        h[ii] += d
                        h[jj] -= d
                    else:
                        h[jj] += (-d)
                        h[ii] -= (-d)
                    cur_val = v
                    improved = True
                    stagnant = 0
                    if v < best_val:
                        best_val = v
                        best_h = h.copy()
                    break

            if not improved:
                # Random pair attempts
                found = False
                for _ in range(15):
                    ii = rng.integers(n)
                    jj = rng.integers(n)
                    if ii == jj:
                        continue
                    d, v = pair_line_search(h, ii, jj, n, n_trials=6)
                    if abs(d) > 1e-10:
                        if d > 0:
                            h[ii] += d
                            h[jj] -= d
                        else:
                            h[jj] += (-d)
                            h[ii] -= (-d)
                        cur_val = v
                        found = True
                        stagnant = 0
                        if v < best_val:
                            best_val = v
                            best_h = h.copy()
                        break

                if not found:
                    stagnant += 1
                    if stagnant > 200:
                        # Annealing restart
                        noise = rng.normal(0.0, 0.02, n)
                        noise -= np.mean(noise)
                        new_h = best_h + noise
                        new_h = np.clip(new_h, 0.0, 1.0)
                        new_h = fix_sum(new_h, n)
                        if npc_all_valid(new_h):
                            h = new_h.copy()
                            cur_val = compute_c5(h)
                        stagnant = 0

    # Phase 3: Fine polishing
    if time.time() < deadline - 2.0:
        h = best_h.copy()
        cur_val = best_val
        no_improve = 0
        while time.time() < deadline - 2.0 and no_improve < 50:
            sg = get_subgrad(h, tol=1e-6)
            sg -= np.mean(sg)
            order = np.argsort(sg)
            improved = False
            for trial in range(5):
                if len(order) <= trial + 1:
                    break
                ii = order[trial]
                jj = order[-1 - trial]
                if ii == jj:
                    continue
                d, v = pair_line_search(h, ii, jj, n, n_trials=20)
                if abs(d) > 1e-12:
                    if d > 0:
                        h[ii] += d
                        h[jj] -= d
                    else:
                        h[jj] += (-d)
                        h[ii] -= (-d)
                    cur_val = v
                    improved = True
                    if v < best_val:
                        best_val = v
                        best_h = h.copy()
            if not improved:
                no_improve += 1
    return best_h, best_val, n

def npc_all_valid(h):
    return np.all(h >= 0.0) and np.all(h <= 1.0)
```
