    lo = max(-h[a], h[b] - 1)

    """Generate a random feasible h with sum = n/2."""
    # Start with uniform and add random perturbation

            h_opt, c5_opt, _ = local_optimize(h_try, time_limit=time_limit_r, start_time=start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 4: Final polish on best

            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Ensure final sum constraint
    best_h = fix_sum(best_h)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
