            if best_h[i] + d < 0 or best_h[i] + d > 1:
                continue
            if best_h[j] - d < 0 or best_h[j] - d > 1:
                continue
            ht = best_h.copy()
            ht[i] += d
            ht[j] -= d
            ht = _project_box_sum(ht, mass)
            c5t, _ = _compute_c5(ht, dx)
            if c5t < best_c5 - 1e-13:
                best_h = ht
                best_c5 = c5t
                break
        # Also try random pair
        if rng.random() < 0.3:
            i2 = rng.integers(n)
            j2 = rng.integers(n)
            if i2 != j2:
                d2 = 0.05 if best_h[i2] < 0.5 else -0.05
                if 0 <= best_h[i2] + d2 <= 1 and 0 <= best_h[j2] - d2 <= 1:
                    ht2 = best_h.copy()
                    ht2[i2] += d2
                    ht2[j2] -= d2
                    c5t2, _ = _compute_c5(ht2, dx)
                    if c5t2 < best_c5 - 1e-13:
                        best_h = ht2
                        best_c5 = c5t2
    return best_h, best_c5

def _random_restart(h, n, mass, scale, rng):
    hp = h + rng.normal(0, scale, n)
    hp = _project_box_sum(hp, mass)
    c5p, _ = _compute_c5(hp, dx)
    return hp, c5p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = _time.perf_counter()
    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - (_time.perf_counter() - t0)

    # Get initial
    h0 = np.asarray(initial_h_values, dtype=np.float64).flatten()
    n = len(h0)
    dx = 2.0 / n
    mass = n / 2.0
    h = _project_box_sum(h0, mass)
    c5_best, _ = _compute_c5(h, dx)
    h_best = h.copy()

    # Phase 1: FW with annealed temperature
    temps = [0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
    for temp in temps:
        if tleft() < 60:
            break
        stall = 0
        max_it = 3000
        for it in range(max_it):
            if tleft() < 60:
                break
            if stall > 800:
                break
            h_new, c5_new = _fw_step(h, n, mass, dx, temp, rng)
            if c5_new < c5_best - 1e-13:
                h = h_new
                c5_best = c5_new
                h_best = h.copy()
                stall = 0
            else:
                stall += 1
                if stall % 100 == 0:
                    h_p, c5_p = _random_restart(h, n, mass, 0.08, rng)
                    if c5_p < c5_best:
                        h = h_p
                        c5_best = c5_p
                        h_best = h.copy()
                        stall = 0
        # After each temp, do some gradient steps
        if tleft() < 60:
            break
        step = 0.02
        for it in range(3000):
            if tleft() < 60:
                break
            if step < 1e-7:
                break
            h_g, c5_g = _gradient_step(h, n, mass, dx, step, temp)
            if c5_g < c5_best - 1e-13:
                h = h_g
                c5_best = c5_g
                h_best = h.copy()
                step = min(step * 1.05, 0.08)
            else:
                step *= 0.5

    # Phase 2: Pair moves
    if tleft() > 60:
        h, c5_best = _pair_moves(h, n, mass, dx, rng, max_iter=10000)
        if c5_best < 0:
            c5_best, _ = _compute_c5(h, dx)
        h_best = h.copy()

    # Phase 3: More FW with smaller temp
    if tleft() > 90:
        for temp in [0.003, 0.002, 0.001, 0.0005]:
            if tleft() < 60:
                break
            stall = 0
            for it in range(5000):
                if tleft() < 60:
                    break
                if stall > 500:
                    break
                h_new, c5_new = _fw_step(h, n, mass, dx, temp, rng)
                if c5_new < c5_best - 1e-14:
                    h = h_new
                    c5_best = c5_new
                    h_best = h.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall % 200 == 0:
                        h_p, c5_p = _random_restart(h, n, mass, 0.03, rng)
                        if c5_p < c5_best:
                            h = h_p
                            c5_best = c5_p
                            h_best = h.copy()
                            stall = 0

    # Phase 4: Gradient polish
    if tleft() > 60:
        step = 0.005
        for it in range(10000):
            if tleft() < 30:
                break
            if step < 1e-8:
                break
            h_g, c5_g = _gradient_step(h, n, mass, dx, step, 0.005)
            if c5_g < c5_best - 1e-14:
                h = h_g
                c5_best = c5_g
                h_best = h.copy()
                step = min(step * 1.02, 0.02)
            else:
                step *= 0.5

    # Phase 5: Pair moves again
    if tleft() > 60:
        h, c5_new = _pair_moves(h, n, mass, dx, rng, max_iter=8000)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()

    # Phase 6: Try nearby resolutions
    if tleft() > 120:
        for n_try in [n + 2, n - 2, n + 4, n - 4, n + 6]:
            if tleft() < 90 or n_try < 20 or n_try > 200:
                continue
            dx_t = 2.0 / n_try
            mass_t = n_try / 2.0
            if n_try > n:
                old_x = np.linspace(0, 2.0, n)
                new_x = np.linspace(0, 2.0, n_try)
                h_t = np.interp(new_x, old_x, h_best)
            else:
                idx = np.linspace(0, n - 1, n_try).astype(int)
                h_t = h_best[idx]
            h_t = _project_box_sum(h_t, mass_t)
            c5_t, _ = _compute_c5(h_t, dx_t)
            if c5_t < c5_best:
                # Optimize at this resolution
                h_c = h_t
                for temp in [0.05, 0.02, 0.01, 0.005]:
                    if tleft() < 60:
                        break
                    for it in range(3000):
                        if tleft() < 60:
                            break
                        h_n2, c5_n2 = _fw_step(h_c, n_try, mass_t, dx_t, temp, rng)
                        if c5_n2 < c5_best - 1e-13:
                            h_c = h_n2
                            c5_best = c5_n2
                            h_best = h_c.copy()
                if tleft() > 60:
                    step2 = 0.01
                    for it in range(3000):
                        if tleft() < 30 or step2 < 1e-8:
                            break
                        h_g2, c5_g2 = _gradient_step(h_c, n_try, mass_t, dx_t, step2, 0.01)
                        if c5_g2 < c5_best - 1e-13:
                            h_c = h_g2
                            c5_best = c5_g2
                            h_best = h_c.copy()
                            step2 = min(step2 * 1.05, 0.05)
                        else:
                            step2 *= 0.5
                n = n_try
                dx = dx_t

    # Final
    h_final = _project_box_sum(h_best, n / 2.0)
    c5_final, _ = _compute_c5(h_final, dx)

    # Verify with evaluate_erdos_solution if available
    try:
        if 'evaluate_erdos_solution' in globals() and callable(globals()['evaluate_erdos_solution']):
            c5_check = evaluate_erdos_solution(h_final)
            if c5_check < c5_final:
                c5_final = c5_check
    except Exception:
        pass

    return h_final, c5_final, n
