def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

    """Optimize h using pairwise coordinate exchanges."""
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    start = time.time()

        iteration += 1
        improved = False
