import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def line_search(h, u, v):
    t_min = max(-h[u], h[v] - 1.0) + 1e-14
    t_max = min(1.0 - h[u], h[v]) - 1e-14
    if t_min >= t_max:
        return
    hu = h[u]
    hv = h[v]
    def obj(t):
        h[u] = hu + t
        h[v] = hv - t
        return evaluate(h)
    res = minimize_scalar(obj, bounds=(t_min, t_max), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 150})
    h[u] = hu + res.x
    h[v] = hv - res.x

def optimize(h, deadline, max_passes=500):
    n = len(h)
    best_c5 = evaluate(h)
    best_h = h.copy()
    stall = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        perm = np.random.permutation(n)
        for pi in range(n):
            i = perm[pi]
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_c5
                if (h[i] < 1e-11 and h[j] < 1e-11) or (h[i] > 1 - 1e-11 and h[j] > 1 - 1e-11):
                    continue
                line_search(h, i, j)
                c5 = evaluate(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c5

def make_staircase(n, num_steps):
    h = np.zeros(n)
    for i in range(n):
        frac = i / n
        step = min(int(frac * num_steps), num_steps - 1)
        h[i] = (step + 1) / num_steps
    t = n / 2.0
    s = h.sum()
    if s > 0:
        h = np.clip(h * t / s, 0, 1)
    return h

def make_periodic(n, period, duty=0.5):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < (period * duty) else 0.0
    target = n // 2
    cnt = int(h.sum())
    if cnt > target:
        idx = np.where(h == 1)[0]
        h[idx[:cnt - target]] = 0
    elif cnt < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - cnt]] = 1
    return h

def relax_binary(b, alpha):
    return alpha + (1.0 - 2.0 * alpha) * b

def make_seeds(n, rng, best_h):
    seeds = []
    for ns in [2, 3, 4, 5, 6, 8, 10, 12, 15, 20]:
        seeds.append(make_staircase(n, ns))
    for period in [2, 3, 4, 5, 6, 7, 8]:
        if period <= n:
            seeds.append(make_periodic(n, period))
    for _ in range(12):
        b = np.zeros(n)
        ones = rng.choice(n, size=n // 2, replace=False)
        b[ones] = 1
        for alpha in [0.05, 0.1, 0.2, 0.35]:
            seeds.append(relax_binary(b, alpha))
    for _ in range(8):
        hp = best_h + rng.normal(0, 0.05, n)
        hp = np.clip(hp, 0, 1)
        diff = n / 2.0 - hp.sum()
        hp = np.clip(hp + diff / n, 0, 1)
        seeds.append(hp)
    for _ in range(4):
        x = np.linspace(0, 1, n)
        h = np.zeros(n)
        for k in range(6):
            h += rng.normal(0, 1) * np.sin((k + 1) * 2 * np.pi * x + rng.uniform(0, 6.28))
        h = (h - h.min()) / (h.max() - h.min() + 1e-12)
        h = np.clip(h * (n / 2.0) / h.sum(), 0, 1)
        seeds.append(h)
    return seeds

def upsample(h, factor):
    n = len(h)
    new_n = n * factor
    x_old = np.arange(n) / n
    x_new = np.arange(new_n) / new_n
    new_h = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    diff = target - new_h.sum()
    new_h = np.clip(new_h + diff / new_n, 0, 1)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate(best_h)

    # Phase 1: Intensive optimization of initial solution
    dl1 = t0 + budget_s * 0.35
    h, c5 = optimize(h0.copy(), dl1, max_passes=300)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # Phase 2: Upsampled optimization
    if time.time() < deadline - 120:
        for factor in [2, 3]:
            if time.time() > deadline - 60:
                break
            h_up = upsample(best_h, factor)
            n_up = len(h_up)
            dl = min(time.time() + (deadline - time.time()) * 0.4, deadline - 30)
            h, c5 = optimize(h_up, dl, max_passes=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Phase 3: Diverse seeds
    if time.time() < deadline - 60:
        seeds = make_seeds(n, rng, best_h)
        time_left = deadline - time.time() - 30
        time_per = max(time_left / len(seeds), 4.0)
        for hs in seeds:
            if time.time() > deadline - 25:
                break
            c5_s = evaluate(hs)
            if c5_s < best_c5 + 0.06:
                dl = time.time() + min(time_per, deadline - time.time() - 5)
                h, c5 = optimize(hs.copy(), dl, max_passes=40)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h

    # Phase 4: Iterated local search on best
    if time.time() < deadline - 20:
        n_iters = max(1, int((deadline - time.time() - 5) / 30))
        for _ in range(n_iters):
            if time.time() > deadline - 5:
                break
            scale = 0.03
            hp = best_h + rng.normal(0, scale, n)
            hp = np.clip(hp, 0, 1)
            diff = n / 2.0 - hp.sum()
            hp = np.clip(hp + diff / n, 0, 1)
            dl = min(time.time() + 25, deadline - 3)
            h, c5 = optimize(hp, dl, max_passes=50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Final refinement
    if time.time() < deadline:
        h, c5 = optimize(best_h.copy(), deadline, max_passes=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Ensure exact feasibility
    target = len(best_h) / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target - best_h.sum()
    if abs(diff) > 1e-14:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate(best_h)
    return best_h, final_c5, len(best_h)
