def project_box_sum(y, target):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(y)

def project_box_sum(y, target):
    n = len(y)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0).copy()
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:

                    add = min(res, 1.0 - x[i])
                    x[i] += add
