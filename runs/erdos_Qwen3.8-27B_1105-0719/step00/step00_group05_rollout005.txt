def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h_raw):
    n = len(h_raw)
    target = n / 2.0
    h = np.clip(h_raw.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo = -max(h_raw.max(), 1.0) - 2.0
    hi = max(1.0 - h_raw.min(), 1.0) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_test = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if s_test < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h_raw + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                add = min(residual, room[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h

def weighted_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    w0 = weights[n - 1]
    if abs(w0) > 1e-16:
        g += w0 * (1.0 - 2.0 * h)
    for m in range(1, n):
        wm = weights[n - 1 + m]
        if abs(wm) > 1e-16:
            g[m:n] += wm * (1.0 - h[:n - m])
            g[:n - m] -= wm * h[m:n]
    for m in range(1, n):
        wm = weights[n - 1 - m]
        if abs(wm) > 1e-16:
            g[:n - m] += wm * (1.0 - h[m:n])
            g[m:n] -= wm * h[:n - m]
    return g

def smooth_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = tau * (corr - mx)
    log_sum = mx + (1.0 / tau) * np.log(np.sum(np.exp(shifted)))
    return log_sum * dx

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = tau * (corr - mx)
    exp_vals = np.exp(shifted)
    softmax_w = exp_vals / np.sum(exp_vals)
    g = weighted_grad(h, softmax_w)
    return g * dx

def projected_gradient_step(h, grad, step_size):
    n = len(h)
    d = -grad
    d -= np.mean(d)
    h_new = h + step_size * d
    h_new = project_feasible(h_new)

def local_refine(h0, tau, n_iter, step_init, rng):

    step = step_init
