def exact_obj(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h):
    """Project to feasible set {0<=h<=1, sum(h)=len(h)/2}."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-13:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += r / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient_all(h, weights):
    """
    Compute sum_k weights[k] * dV_k/dh
    where weights has length 2n-1 (same indexing as np.correlate full output).
    Index j corresponds to:
      j < n-1: V(n-1-j) forward
      j = n-1: V(0)
      j > n-1: V(-(j-(n-1))) backward
    """
    n = len(h)
    g = np.zeros(n)
    # V(0)
    g += weights[n-1] * (1.0 - 2.0 * h)
    # Forward: j from 0 to n-2, m = n-1-j (so m = n-1, n-2, ..., 1)
    for j in range(n - 1):
        wj = weights[j]
        if abs(wj) < 1e-16:
            continue
        m = n - 1 - j
        nm = n - m
        g[:nm] += wj * (1.0 - h[m:])
        g[m:] -= wj * h[:nm]
    # Backward: j from n to 2n-2, m = j-(n-1) (so m = 1, 2, ..., n-1)
    for j in range(n, 2 * n - 1):
        wj = weights[j]
        if abs(wj) < 1e-16:
            continue
        m = j - (n - 1)
        nm = n - m
        g[m:] += wj * (1.0 - h[:nm])
        g[:nm] -= wj * h[m:]
    return g

def smooth_obj_grad(h, beta):
    """Smooth approximation of C(h) and its gradient using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    vmax = np.max(vals)
    shifted = vals - vmax
    exp_s = np.exp(beta * shifted)
    sum_e = np.sum(exp_s)
    f = dx * (vmax + np.log(sum_e) / beta)
    w = exp_s / sum_e
    g = lag_gradient_all(h, w) * dx
    return f, g
