import numpy as np
from scipy.optimize import minimize
from scipy import stats
import time

def evaluate_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    if n == 0:
        return np.inf
    h = np.asarray(h, dtype=np.float64)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def make_feasible(h_raw):
    """Project h_raw to feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.asarray(h_raw, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    
    # Water fill to achieve exact sum
    # Binary search on lambda: sum(clip(h + lambda, 0, 1)) = target
    lo = -h.max() - 1
    hi = 1 - h.min() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    
    # Final correction for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute correction
        if diff > 0:
            # Add to entries with most room
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            # Remove from entries with most value
            order = np.argsort(h)
            diff = -diff
            for i in order:
                rem = min(diff, h[i])
                h[i] -= rem
                diff -= rem
                if abs(diff) < 1e-14:
                    break
    
    return h

def directional_values(h):
    """Compute all directional overlap values (before dx scaling)."""
    c = np.correlate(h, 1.0 - h, mode="full")
    return c

def build_top_weights(c_vals, K, decay=0.3):
    """Build weights for top-K directional values."""
    n_lags = len(c_vals)
    sorted_idx = np.argsort(-c_vals)
    top_idx = sorted_idx[:K]
    
    weights = np.zeros(n_lags)
    for rank, idx in enumerate(top_idx):
        weights[idx] = np.exp(-decay * rank)
    
    total = weights.sum()
    if total > 1e-15:
        weights /= total
    else:
        weights = np.ones(n_lags) / n_lags
    return weights

def weighted_gradient(h, weights):
    """Compute gradient of sum_j weights[j] * c[j] where c = correlate(h, 1-h, full).
    
    c[j] = sum_i h[i+lag] * (1-h[i]) where lag = j - (n-1)
    """
    n = len(h)
    g = np.zeros(n)
    n_lags = 2 * n - 1
    
    for j in range(n_lags):
        w = weights[j]
        if abs(w) < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            # c[j] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            # d/dh[p] = (1-h[p-lag]) for lag<=p<=n-1 minus h[p+lag] for 0<=p<=n-1-lag
            for i in range(n - lag):
                # Contribution from h[i+lag] term: (1-h[i]) to index i+lag
                g[i + lag] += w * (1.0 - h[i])
                # Contribution from (1-h[i]) term: -h[i+lag] to index i
                g[i] -= w * h[i + lag]
        else:
            m = -lag
            # c[j] = sum_{i=m}^{n-1} h[i-m] * (1-h[i])
            # d/dh[p] = (1-h[p+m]) for 0<=p<=n-1-m minus h[p-m] for m<=p<=n-1
            for i in range(m, n):
                # Contribution from h[i-m] term: (1-h[i]) to index i-m
                g[i - m] += w * (1.0 - h[i])
                # Contribution from (1-h[i]) term: -h[i-m] to index i
                g[i] -= w * h[i - m]
    
    return g

def solve_trust_region(h, grad, radius, equality_target):
    """Solve: min grad^T d + 0.5*||d||^2/radius s.t. sum(d)=0, |d_i|<=radius, 0<=h_i+d_i<=1"""
    n = len(h)
    rho = 1.0 / max(radius, 1e-6)
    
    def objective(d):
        return grad @ d + 0.5 * rho * d @ d
    
    def gradient(d):
        return grad + rho * d
    
    # Bounds for d
    lo = np.maximum(-h, -radius)
    hi = np.minimum(1.0 - h, radius)
    
    # Ensure feasible start
    d0 = np.zeros(n)
    
    cons = [{'type': 'eq', 'fun': lambda d: np.sum(d) - 0.0, 'jac': lambda d: np.ones(n)}]
    
    bounds = list(zip(lo, hi))
    
    res = minimize(objective, d0, jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-12})
    
    if res.success and np.isfinite(res.fun):
        return res.x
    else:
        # Fallback: projected negative gradient
        d = -grad
        d -= np.mean(d)  # enforce sum = 0
        d = np.clip(d, lo, hi)
        d -= np.mean(d)  # re-project after clipping
        norm = np.linalg.norm(d)
        if norm > radius:
            d *= radius / norm
        return d

    """Vectorized gradient of sum_j weights[j] * correlate(h, 1-h, 'full')[j]"""
    n = len(h)
    g = np.zeros(n)
    
    for j in range(2 * n - 1):
        w = weights[j]
        if abs(w) < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            g[lag:] += w * (1.0 - h[:n-lag])
            g[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            g[:n-m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n-m]

import numpy as np

import time

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    h = np.asarray(h, dtype=np.float64)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def make_feasible(h_raw):
    h = np.asarray(h_raw, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                add = min(diff, room[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            diff = -diff
            order = np.argsort(h)
            for idx in order:
                rem = min(diff, h[idx])
                h[idx] -= rem
                diff -= rem
                if abs(diff) < 1e-14:
                    break
    return h

def weighted_gradient_vec(h, weights):
    n = len(h)
    g = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if abs(w) < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n - m]
    return g

def build_top_weights(c_vals, K, decay=0.3):
    n_lags = len(c_vals)
    K = min(K, n_lags)
    sorted_idx = np.argsort(-c_vals)
    top_idx = sorted_idx[:K]
    weights = np.zeros(n_lags)
    for rank, idx in enumerate(top_idx):
        weights[idx] = np.exp(-decay * rank)
    total = weights.sum()
    if total > 1e-15:
        weights /= total
    else:
        weights = np.ones(n_lags) / n_lags
    return weights

def direct_slsqp(h0, K, decay, maxiter=100, time_fn=None):
    n = len(h0)
    target = n / 2.0

    def get_weights(h_cur):
        c = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        return build_top_weights(c, K, decay)

    w = get_weights(h0)

    def obj(h_cur):
        c = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        return float(w @ c)

    def grad(h_cur):
        return weighted_gradient_vec(h_cur, w)

    constr = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
               'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=constr, options={'maxiter': maxiter, 'ftol': 1e-10})

    if res.success or np.isfinite(res.fun):
        h_new = make_feasible(res.x)
        c_new = evaluate_c5(h_new)
        if c_new < evaluate_c5(h0) + 1e-10:
            return h_new, c_new
    return None, None

def trust_region_step(h, grad, radius):
    n = len(h)
    rho = 1.0 / max(radius, 1e-8)

    def obj(d):
        return float(grad @ d + 0.5 * rho * d @ d)

    def grad_d(d):
        return grad + rho * d

    lo = np.maximum(-h, -radius)
    hi = np.minimum(1.0 - h, radius)
    bounds = list(zip(lo, hi))
    constr = [{'type': 'eq', 'fun': lambda d: np.sum(d),
               'jac': lambda d: np.ones(len(d))}]

    d0 = np.zeros(n)
    try:
        res = minimize(obj, d0, jac=grad_d, method='SLSQP', bounds=bounds,
                       constraints=constr, options={'maxiter': 100, 'ftol': 1e-12})
        if np.isfinite(res.fun):
            return res.x
    except Exception:
        pass
    d = -grad
    d -= np.mean(d)
    scale = np.max(np.abs(d))
    if scale > radius:
        d *= radius / scale
    d = np.clip(d, lo, hi)
    d -= np.mean(d)
    return d
