from scipy.optimize import minimize

def evaluate_solution(h):
    """Evaluate the exact C5 bound for a given step function."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

def compute_lag_values(h):
    """Compute all lag values s_r = dx * correlate(h, 1-h, full)[r]."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return corr * dx

def smooth_objective_and_grad(h, alpha):
    """Compute the log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    s = compute_lag_values(h)
    s_max = np.max(s)
    alpha_s = alpha * (s - s_max)
    # Clip for numerical stability
    alpha_s = np.clip(alpha_s, -500, 500)
    exp_sum = np.sum(np.exp(alpha_s))
    obj = s_max + np.log(exp_sum) / alpha
    
    # Weights
    weights = np.exp(alpha_s) / exp_sum
    
    # Gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        wr = weights[r] * dx
        if abs(wr) < 1e-15:
            continue
        for m in range(n):
            term = 0.0
            if 0 <= m + k < n:
                term += (1.0 - h[m + k])
            if 0 <= m - k < n:
                term -= h[m - k]
            if term != 0.0:
                grad[m] += wr * term
    return obj, grad

def project_to_feasible(h, target):
    """Project h to the feasible set {0 <= x <= 1, sum(x) = target}."""
    n = len(h)

    # Clip to [0, 1]

    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    return np.correlate(h, q, mode='full') * dx

def lse_objective(h, alpha):
    s = compute_lag_values(h)
    s_max = np.max(s)
    if alpha < 1e-10:
        return float(s_max)
    a = alpha * (s - s_max)
    a = np.clip(a, -500, 500)
    return float(s_max + np.log(np.sum(np.exp(a))) / alpha)

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    s = compute_lag_values(h)
    s_max = np.max(s)
    a = alpha * (s - s_max)
    a = np.clip(a, -500, 500)
    exp_a = np.exp(a)
    exp_sum = np.sum(exp_a)
    weights = exp_a / exp_sum

            if 0 <= j2 < n:

    s_max = np.max(s)
    a = alpha * (s - s_max)
    a = np.clip(a, -500.0, 500.0)
    return float(s_max + np.log(np.sum(np.exp(a))) / alpha)

    n = len(h)
    dx = 2.0 / n
    s = compute_lag_values(h)
    s_max = np.max(s)
    a = alpha * (s - s_max)
    a = np.clip(a, -500.0, 500.0)
    exp_a = np.exp(a)
    weights = exp_a / np.sum(exp_a)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):

        wr = weights[r] * dx
        if abs(wr) < 1e-18:
            continue
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += wr * (1.0 - h[k:])
                grad[k:] -= wr * h[:n-k]
        else:
            if n + k > 0:
                grad[-k:] += wr * (1.0 - h[:n+k])
                grad[:n+k] -= wr * h[-k:]
    return grad

        free = (x > 1e-15) & (x < 1.0 - 1e-15)

        if n_free == 0:
            break
        x[free] += diff / n_free
        x = np.clip(x, 0.0, 1.0)
    return x

                        improved = True

    s_max = np.max(s)
    a = alpha * (s - s_max)
    a = np.clip(a, -500.0, 500.0)
    exp_a = np.exp(a)
    total = np.sum(exp_a)
    weights = exp_a / total

    grad = np.zeros(n)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        wr = weights[r] * dx
        if abs(wr) < 1e-18:
            continue
        if k >= 0:
            m = n - k
            if m > 0:
                grad[:m] += wr * (1.0 - h[k:])
                grad[k:] -= wr * h[:m]
        else:
            m = n + k
            if m > 0:
                grad[-k:] += wr * (1.0 - h[:m])
                grad[:m] -= wr * h[-k:]
    return grad

def lse_fun(h, alpha):
    return lse_objective(h, alpha)

def lse_jac(h, alpha):
    return lse_gradient(h, alpha)

def sum_constraint(h, target):
    return np.sum(h) - target

def sum_constraint_jac(h, target):
    return np.ones(len(h))

def project_feasible(h, target):
    n = len(h)
    x = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(x)
        if abs(diff) < 1e-15:
            break
        free = (x > 1e-15) & (x < 1.0 - 1e-15)
        n_free = np.sum(free)
        if n_free == 0:
            break
        x[free] += diff / n_free
        x = np.clip(x, 0.0, 1.0)
    return x

def pair_local_search(h, target, time_limit=60.0):
    n = len(h)
    h = project_feasible(h, target).copy()
    best_c5 = evaluate_solution(h)
    start = time.time()

    eps_levels = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]

    for eps in eps_levels:
        if time.time() - start > time_limit:
            break
        for sweep in range(30):
            if time.time() - start > time_limit:
                break
            improved = False
            for i in range(n):
                if h[i] < eps:
                    continue
                for j in range(n):
                    if i == j or h[j] > 1.0 - eps:
                        continue
                    h[i] -= eps
                    h[j] += eps
                    c5 = evaluate_solution(h)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        improved = True
                    else:
                        h[i] += eps
                        h[j] -= eps
            if not improved:
                break
    return h, best_c5

def gradient_refine(h, target, alpha, n_steps=2000, time_limit=30.0):
    n = len(h)
    h = project_feasible(h, target).copy()
    best_c5 = evaluate_solution(h)
    start = time.time()
    step = 0.01

    for _ in range(n_steps):
        if time.time() - start > time_limit:
            break
        g = lse_gradient(h, alpha)
        h_new = h - step * g
        h_new = project_feasible(h_new, target)
        c5_new = evaluate_solution(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.5
            if step < 1e-9:
                step = 0.005
    return h, best_c5

def time_left(start, budget):
    return budget - (time.time() - start)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    hv = globals().get('initial_h_values', None)
    if hv is not None:
        h0 = np.array(hv, dtype=float)
        n0 = len(h0)
    else:
        n0 = 45
        h0 = np.full(n0, 0.5)

    target0 = n0 / 2.0
    h0 = project_feasible(h0, target0)
    best_h = h0.copy()
    best_c5 = evaluate_solution(h0)

    # Phase 1: SLSQP with progressive alpha from initial point
    alphas = [10, 30, 100, 300, 1000]
    h_cur = h0.copy()

    for alpha in alphas:
        if time_left(start_time, budget_s) < 30:
            break
        for attempt in range(3):
            if time_left(start_time, budget_s) < 15:
                break
            if attempt == 0:
                h_start = h_cur.copy()
            elif attempt == 1:
                h_start = h_cur + rng.randn(n0) * 0.01
                h_start = project_feasible(h_start, target0)
            else:
                h_start = h_cur + rng.randn(n0) * 0.05
                h_start = project_feasible(h_start, target0)

            try:
                res = minimize(
                    lse_fun, h_start, args=(alpha,),
                    jac=lse_jac, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n0,
                    constraints=[{'type': 'eq', 'fun': sum_constraint,
                                  'jac': sum_constraint_jac, 'args': (target0,)}],
                    options={'maxiter': 500, 'ftol': 1e-14}
                )
                if res.x is not None:
                    h_try = project_feasible(res.x, target0)
                    c5_try = evaluate_solution(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h_cur = h_try.copy()
            except Exception:
                pass

    # Phase 2: Try other n values
    for n_try in [36, 40, 48, 50, 54, 60, 72]:
        if time_left(start_time, budget_s) < 60:
            break
        n_best = len(best_h)
        if n_best == n_try:
            h_cur = best_h.copy()
        else:
            xo = np.linspace(0, 2, n_best, endpoint=False)
            xn = np.linspace(0, 2, n_try, endpoint=False)
            h_cur = np.interp(xn, xo, best_h)
            h_cur = project_feasible(h_cur, n_try / 2.0)

        target = n_try / 2.0

        for alpha in [30, 100, 300, 1000]:
            if time_left(start_time, budget_s) < 25:
                break
            try:
                res = minimize(
                    lse_fun, h_cur.copy(), args=(alpha,),
                    jac=lse_jac, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n_try,
                    constraints=[{'type': 'eq', 'fun': sum_constraint,
                                  'jac': sum_constraint_jac, 'args': (target,)}],
                    options={'maxiter': 400, 'ftol': 1e-14}
                )
                if res.x is not None:
                    h_try = project_feasible(res.x, target)
                    c5_try = evaluate_solution(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h_cur = h_try.copy()
            except Exception:
                pass

    # Phase 3: Gradient refinement on best solution found so far
    remaining = time_left(start_time, budget_s)
    if remaining > 40:
        n_best = len(best_h)
        target = n_best / 2.0
        h_ref, c5_ref = gradient_refine(best_h, target, alpha=500,
                                         n_steps=3000, time_limit=min(60, remaining - 20))
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Phase 4: Pair local search
    remaining = time_left(start_time, budget_s)
    if remaining > 30:
        n_best = len(best_h)
        target = n_best / 2.0
        h_ls, c5_ls = pair_local_search(best_h, target,
                                         time_limit=min(90, remaining - 20))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # Phase 5: Final SLSQP polish at very high alpha
    remaining = time_left(start_time, budget_s)
    if remaining > 20:
        n_best = len(best_h)
        target = n_best / 2.0
        for alpha in [3000, 10000]:
            if time_left(start_time, budget_s) < 15:
                break
            try:
                res = minimize(
                    lse_fun, best_h.copy(), args=(alpha,),
                    jac=lse_jac, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n_best,
                    constraints=[{'type': 'eq', 'fun': sum_constraint,
                                  'jac': sum_constraint_jac, 'args': (target,)}],
                    options={'maxiter': 300, 'ftol': 1e-15}
                )
                if res.x is not None:
                    h_try = project_feasible(res.x, target)
                    c5_try = evaluate_solution(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
            except Exception:
                pass

    # Final feasibility check
    n_best = len(best_h)
    best_h = project_feasible(best_h, n_best / 2.0)
    best_c5 = evaluate_solution(best_h)

    return (best_h, best_c5, n_best)
