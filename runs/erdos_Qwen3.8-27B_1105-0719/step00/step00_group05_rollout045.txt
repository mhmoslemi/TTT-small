    G = np.zeros(n)
    # k goes from 0 to 2n-2, offset = k - (n-1)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        offset = k - (n-1)

            G += w[k] * dx * (1 - 2*h)

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def exact_C(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c))

def is_feasible(h, tol=1e-8):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    return (np.all(h >= -tol) and np.all(h <= 1+tol) and abs(np.sum(h) - n/2.0) < tol)

def project_feasible(u, n):
    """Project u onto {h: 0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # Find lambda such that sum(clip(u+lambda, 0, 1)) = target
    lo, hi = -1.0 - u.max() - 1, 1.0 + 1 - u.min()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(u + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(u + (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        # Distribute residual to coordinates with most room
        if resid > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        # Add/distribute to first coordinate with room
        for idx in order:
            room = min(resid, 1.0 - h[idx]) if resid > 0 else max(resid, -h[idx])
            if abs(room) < 1e-15:
                break
            h[idx] += room
            resid -= room
            if abs(resid) < 1e-14:
                break
    return h

def sigmoid(x):
    x = np.clip(x, -500, 500)
    return 1.0 / (1.0 + np.exp(-x))

def feasible_from_logit(y, n):
    """Map y to feasible h via sigmoid(y + lambda) with sum constraint."""
    target = n / 2.0
    # Find lambda: sum(sigmoid(y + lambda)) = target
    lo = -y.max() - 50.0
    hi = -y.min() + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = sigmoid(y + (lo+hi)/2.0)
    # Small correction
    resid = target - h.sum()
    if abs(resid) > 1e-13:
        # Add to coordinate with most room
        idx = np.argmax(1-h) if resid > 0 else np.argmin(h - 0)
        # Actually distribute properly
        h = project_feasible(h, n)
    return h

def logit_of_h(h, eps=1e-10):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def full_correlation_values(h, n):
    """Return the 2n-1 correlation values (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_directive_gradient(h, w, dx, n):
    """Compute dF/dh given weights w over full correlation."""
    G = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        offset = k - (n - 1)
        if offset == 0:
            G += wk * dx * (1.0 - 2.0 * h)
        elif offset > 0:
            valid = n - offset
            G[:valid] += wk * dx * (1.0 - h[offset:offset+valid])
            G[offset:] += wk * dx * (-h[:valid])
        else:
            m = -offset
            valid = n - m
            G[m:] += wk * dx * (1.0 - h[:valid])
            G[:valid] += wk * dx * (-h[m:])
    return G

def smooth_obj_and_grad(h, beta):
    """Compute log-sum-exp smoothed objective and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    vals = full_correlation_values(h, n) * dx
    vmax = np.max(vals)
    shifted = beta * (vals - vmax)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    F = vmax + np.log(exp_shifted.sum()) / beta
    G = compute_directive_gradient(h, w, dx, n)
    return F, G

def transformed_obj(y, beta, n):
    """Objective and gradient for BFGS in y-space."""
    h = feasible_from_logit(y, n)
    F, G = smooth_obj_and_grad(h, beta)
    if not np.isfinite(F):
        return 1e6, np.zeros_like(y)
    m = h * (1.0 - h)
    M = m.sum()
    if M < 1e-10:
        return F, np.zeros_like(y)
    B = np.dot(G, m) / M
    gw = m * (G - B)
    # Clip for numerical safety
    gw = np.clip(gw, -1e6, 1e6)
    if not np.all(np.isfinite(gw)):
        gw = np.zeros_like(y)
    return F, gw

def pair_polish(h, time_limit, rng):
    """Local pair-transfer optimization."""
    n = len(h)
    h = h.copy()
    c_cur = exact_C(h)
    max_pairs = 80
    max_iter = 30
    stagnation = 0
    
    for it in range(max_iter):

            break

        pairs = []
        # Top-coordinate pairs based on current values
        idx = np.argsort(h)

            a = rng.integers(n)
            b = rng.integers(n)
            if a != b:
                pairs.append((a, b))
        # Add structured pairs (high-low)
        for i in range(min(n//2, 10)):
            pairs.append((idx[i], idx[n-1-i]))
        
        improved = False
        for (a, b) in pairs:
            if time.time() > time_limit:
                break
            # Find valid range for t
            lo_t = max(0.0, h[a] + 0.0)  # h[a]+t >= 0 always if t>=0
            # h[a] + t <= 1 => t <= 1 - h[a]
            # h[b] - t >= 0 => t <= h[b]
            # h[a] + t >= 0 => t >= -h[a] (always true since h[a]>=0)
            # h[b] - t <= 1 => t >= h[b] - 1 (always true since h[b]<=1)
            lo_t = max(-h[a], h[b] - 1.0)
            hi_t = min(1.0 - h[a], h[b])

                h_test[b] -= t
                if h_test[a] < -1e-12 or h_test[a] > 1+1e-12:

                        continue

                    if c_test < best_c - 1e-14:

                h[b] -= best_t
                h = np.clip(h, 0.0, 1.0)
                # Fix sum

                c_cur = exact_C(h)
                improved = True
        if not improved:
            stagnation += 1
            if stagnation >= 3:
                break
        else:

def record_candidate(h, n):

    global _best_h, _best_c
    if not np.all(np.isfinite(h)):

    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    if not is_feasible(h):

    c = exact_C(h)
    if np.isfinite(c) and c < _best_c:

def resize_h(h, new_n):
    """Resize h to new_n points via interpolation."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, new_n)
    return h_new
