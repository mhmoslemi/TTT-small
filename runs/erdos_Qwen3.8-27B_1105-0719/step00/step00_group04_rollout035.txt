    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    """Ensure 0<=h<=1 and sum(h) = n/2."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Need to add mass; add to coordinates not at 1
            room = 1.0 - h
            total_room = np.sum(room[room > 1e-12])
            if total_room < 1e-12:
                break
            h += room * (diff / total_room)
        else:
            # Need to remove mass; remove from coordinates not at 0
            space = h
            total_space = np.sum(space[space > 1e-12])
            if total_space < 1e-12:
                break
            h -= space * (-diff / total_space)

from scipy.optimize import linprog
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-12:
                break
            h += room * (diff / total)
        else:
            space = np.maximum(h, 0.0)
            total = space.sum()
            if total < 1e-12:
                break
            h -= space * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_A_for_b(b, n):
    """A such that np.correlate(x, b, 'full')*dx = A @ x."""
    dx = 2.0 / n
    cols = []
    idx = np.arange(n)
    for j in range(n):
        e = (idx == j).astype(float)
        cols.append(np.correlate(e, b, mode='full') * dx)
    return np.column_stack(cols)

def build_A_for_a(a, n):
    """B such that np.correlate(a, y, 'full')*dx = B @ y."""
    dx = 2.0 / n
    cols = []
    idx = np.arange(n)
    for j in range(n):
        e = (idx == j).astype(float)
        cols.append(np.correlate(a, e, mode='full') * dx)
    return np.column_stack(cols)

def solve_lp(A, n):
    """Solve min t s.t. A@x - t <= 0, sum(x)=n/2, 0<=x<=1."""
    N = A.shape[0]
    A_ub = np.hstack([A, -np.eye(N)])
    b_ub = np.zeros(N)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, 10.0)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return project_feasible(res.x[:n])
    return None

def lp_alternate(h, max_iters=300, time_budget=400):
    n = len(h)
    h = project_feasible(h.copy())
    c5_cur = evaluate_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        
        # Primal
        b = 1.0 - h
        A = build_A_for_b(b, n)
        cand = solve_lp(A, n)
        if cand is not None:
            c5_new = evaluate_c5(cand)
            if c5_new < c5_cur - 1e-12:
                h, c5_cur = cand, c5_new
            else:
                for alpha in (0.5, 0.3, 0.1):
                    blend = project_feasible(alpha * cand + (1 - alpha) * h)
                    c5_b = evaluate_c5(blend)
                    if c5_b < c5_cur - 1e-12:
                        h, c5_cur = blend, c5_b
                        break
        
        # Dual
        B = build_A_for_a(h, n)
        b_opt = solve_lp(B, n)
        if b_opt is not None:
            cand = project_feasible(1.0 - b_opt)
            c5_new = evaluate_c5(cand)
            if c5_new < c5_cur - 1e-12:
                h, c5_cur = cand, c5_new
            else:
                for alpha in (0.5, 0.3, 0.1):
                    blend = project_feasible(alpha * cand + (1 - alpha) * h)
                    c5_b = evaluate_c5(blend)
                    if c5_b < c5_cur - 1e-12:
                        h, c5_cur = blend, c5_b
                        break
    
    return h, c5_cur

def pair_descent(h, time_budget=100, n_random_trials=500):

            if hi - lo < 1e-10:

            best_a = None

                h[p], h[q] = best_a, S - best_a
                c5_cur = best_val

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            space = np.maximum(h, 0.0)
            total = space.sum()
            if total < 1e-14:
                break
            h -= space * (-diff / total)
        h = np.clip(h, 0.0, 1.0)
    # Final micro-correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            cands = np.where(h < 1.0 - 1e-14)[0]
        else:
            cands = np.where(h > 1e-14)[0]
        if len(cands) > 0:
            h[cands[0]] += diff
    return np.clip(h, 0.0, 1.0)

def build_corr_matrix_fixed_second(b, n):
    """A such that np.correlate(x, b, 'full')*dx = A @ x for any x."""
    dx = 2.0 / n
    idx = np.arange(n)
    cols = np.zeros((2 * n - 1, n))
    for j in range(n):
        e = (idx == j).astype(float)
        cols[:, j] = np.correlate(e, b, mode='full') * dx
    return cols

def build_corr_matrix_fixed_first(a, n):
    """B such that np.correlate(a, y, 'full')*dx = B @ y for any y."""
    dx = 2.0 / n
    idx = np.arange(n)
    cols = np.zeros((2 * n - 1, n))
    for j in range(n):
        e = (idx == j).astype(float)
        cols[:, j] = np.correlate(a, e, mode='full') * dx
    return cols

def solve_lp_subproblem(A, n):
    """Solve: min t s.t. A@x - t <= 0, sum(x)=n/2, 0<=x<=1."""
    N = A.shape[0]
    A_ub = np.hstack([A, -np.eye(N)])
    b_ub = np.zeros(N)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-2.0, 10.0)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return project_feasible(res.x[:n])
    return None

def lp_alternate(h, max_iters=400, time_budget=400):
    """Alternating LP: fix complement, solve LP for the other side."""
    n = len(h)
    h = project_feasible(h.copy())
    c5_cur = evaluate_c5(h)
    start = time.time()
    stall_count = 0

    for it in range(max_iters):
        if time.time() - start > time_budget:
            break

        # --- Primal LP: fix b=1-h, optimize h' ---
        b = 1.0 - h
        A = build_corr_matrix_fixed_second(b, n)
        cand = solve_lp_subproblem(A, n)
        accepted = False
        if cand is not None:
            c5_new = evaluate_c5(cand)
            if c5_new < c5_cur - 1e-13:
                h, c5_cur = cand, c5_new
                accepted = True
            else:
                for alpha in (0.6, 0.4, 0.2, 0.8):
                    blend = project_feasible(alpha * cand + (1.0 - alpha) * h)
                    c5_b = evaluate_c5(blend)
                    if c5_b < c5_cur - 1e-13:
                        h, c5_cur = blend, c5_b
                        accepted = True
                        break

        # --- Dual LP: fix h, optimize b, set h'=1-b ---
        B = build_corr_matrix_fixed_first(h, n)
        b_opt = solve_lp_subproblem(B, n)
        if b_opt is not None:
            cand_d = project_feasible(1.0 - b_opt)
            c5_new = evaluate_c5(cand_d)
            if c5_new < c5_cur - 1e-13:
                h, c5_cur = cand_d, c5_new
                accepted = True
            else:
                for alpha in (0.6, 0.4, 0.2, 0.8):
                    blend = project_feasible(alpha * cand_d + (1.0 - alpha) * h)
                    c5_b = evaluate_c5(blend)
                    if c5_b < c5_cur - 1e-13:
                        h, c5_cur = blend, c5_b
                        accepted = True
                        break

        if not accepted:
            stall_count += 1
            if stall_count > 20:
                break
        else:
            stall_count = 0

    return h, c5_cur

def pair_descent(h, time_budget=100, n_trials=400):
    """Pair coordinate descent with grid + refinement."""
    n = len(h)
    h = project_feasible(h.copy())
    c5_cur = evaluate_c5(h)
    start = time.time()
    dx = 2.0 / n
    n_rounds_done = 0

    while time.time() - start < time_budget:
        improved = False
        for trial in range(n_trials):
            if time.time() - start > time_budget:
                break
            p = np.random.randint(n)
            q = np.random.randint(n)
            if p == q:
                continue
            S = h[p] + h[q]
            lo = max(0.0, S - 1.0)
            hi = min(1.0, S)
            if hi - lo < 1e-11:
                continue

            hp_save, hq_save = h[p], h[q]
            best_a = None
            best_val = c5_cur - 1e-15

            # Coarse grid (10 points)
            step = (hi - lo) / 10.0
            for k in range(11):
                a_val = lo + k * step
                h[p], h[q] = a_val, S - a_val
                val = evaluate_c5(h)
                if val < best_val:
                    best_val = val
                    best_a = a_val

            h[p], h[q] = hp_save, hq_save

            if best_a is not None:
                # Fine grid around best (10 points)
                fine_lo = max(lo, best_a - step)
                fine_hi = min(hi, best_a + step)
                fine_step = (fine_hi - fine_lo) / 10.0
                for k in range(11):
                    a_val = fine_lo + k * fine_step
                    h[p], h[q] = a_val, S - a_val
                    val = evaluate_c5(h)
                    if val < best_val:
                        best_val = val
                        best_a = a_val
                h[p], h[q] = best_a, S - best_a
                c5_cur = evaluate_c5(h)
                improved = True
            # else already restored

        n_rounds_done += 1
        if not improved and n_rounds_done > 2:
            break

    return h, c5_cur

def interpolate_h(h_src, n_tgt):
    """Interpolate from source grid to target grid size."""
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_feasible(h_src.copy())
    x_s = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_t = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h_t = np.interp(x_t, x_s, h_src)
    return project_feasible(h_t)

def random_feasible(n, seed_val):
    """Generate a random feasible h of size n."""
    rng = np.random.RandomState(seed_val)
    h = rng.uniform(0, 1, n)
    return project_feasible(h)
