def toscore(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * dx)

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(proj < 1.0 - 1e-7)[0]
        else:
            free = np.where(proj > 1e-7)[0]
        if len(free) > 0:
            proj[free] += diff / len(free)
            proj[free] = np.clip(proj[free], 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        idx = int(np.argmax(1.0 - proj)) if diff > 0 else int(np.argmin(proj))
        proj[idx] = float(np.clip(proj[idx] + diff, 0.0, 1.0))
    return proj

def grad_row(h, t):
    n = len(h)
    d = (n - 1) - t
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:

        g[d:] += -h[:n - d]
    else:
        r = -d
        g[r:] += (1.0 - h[:n - r])
        g[:n - r] += -h[r:]
    return g

        A_ub[idx, 1:] = g
        A_ub[idx, 0] = -1.0
        b_ub[idx] = np.dot(g, h) - raw[t]
    A_ub[m, 0] = 1.0
    b_ub[m] = np.max(raw) + 0.1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    bounds = [(None, None)]
    for i in range(n):
        lo = max(0.0, h[i] - rho)
        hi = min(1.0, h[i] + rho)
        bounds.append((lo, hi))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[n / 2.0],
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[1:]
    except Exception:
        pass
    return None
