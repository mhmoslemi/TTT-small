    return 1.0 / (1.0 + np.exp(-x))

def find_c(theta, target):
    # Find c such that sum(sigmoid(theta_i + c)) = target
    # Bisection
    lo, hi = -50, 50
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(sigmoid(theta + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2

    """Compute dLSE/dh given h and weights w (length 2n-1)."""
    n = len(h)
    grad = np.zeros(n)
    # First half: i = 0 to n-1
    for i in range(n):
        if w[i] < 1e-15:

        wi = w[i]

        for k in range(i + 1):
