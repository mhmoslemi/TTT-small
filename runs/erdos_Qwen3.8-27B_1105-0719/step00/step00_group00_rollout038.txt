from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C₅ = max of correlation * dx"""
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def project_onto_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint using bisection on lambda
    lo, hi = -10, 10
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    return h_proj

def subgradient_c5(h, dx):
    """Compute subgradient of C₅ at h. Returns gradient vector."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full') * dx
    max_val = np.max(corr)
    # Find active lags (within tolerance)
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for idx in active:
        L = idx - (n - 1)  # the actual lag
        for i in range(n):
            # h[i] appears in correlation at lag L as:
            # 1 - h[i-L] (if valid) - h[i+L] (if valid)
            val = 1.0
            if i - L >= 0 and i - L < n:
                val -= h[i - L]
            if i + L >= 0 and i + L < n:
                val -= h[i + L]
            grad[i] += val
    
    grad /= len(active)
    return grad, max_val

def compute_c5(h, dx):
    """Compute C₅ = max of correlate(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def compute_c5_vec(h, dx):
    """Compute all correlation values * dx"""
    return np.correlate(h, 1-h, mode='full') * dx

def project_onto_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:

def compute_subgrad(h, dx):

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_onto_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_lag(h, L):
    N = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(N)
    if L > 0:
        M = L
        if M < N:
            grad[M:N] += (1.0 - h[:N - M])
            grad[:N - M] -= h[M:N]
    else:
        M = -L
        if M < N:
            grad[:N - M] += (1.0 - h[M:N])
            grad[M:N] -= h[:N - M]
    return grad

def soft_gradient(h, dx, epsilon=0.01):
    N = len(h)
    corr_raw = np.correlate(h, 1 - h, mode='full')
    scaled = corr_raw / epsilon
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(N)
    active = np.where(weights > 1e-6)[0]
    for idx in active:
        L = idx - (N - 1)
        grad += weights[idx] * grad_for_lag(h, L)
    c5_val = np.max(corr_raw) * dx
    return grad, c5_val

def subgradient_optimize(h, dx, target_sum, max_iters=3000, seed_offset=0, start_time=0, time_limit=1100):
    rng = np.random.RandomState(12345 + seed_offset)
    N = len(h)
    h = project_onto_feasible(h.copy(), target_sum)
    c5_val = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_val

    alpha = 0.05
    momentum = np.zeros(N)
    beta = 0.85
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit - 5:
            break
        grad, c5_val = soft_gradient(h, dx, epsilon=0.005)
        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum + grad_centered
        h_new = h - alpha * momentum
        h_new = project_onto_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        alpha *= 0.9995
        if no_improve > 300:
            alpha *= 0.5
            momentum *= 0.1
            no_improve = 0
        if alpha < 1e-10:
            break
    return best_h, best_c5

def local_search(h, dx, target_sum, rng, max_iters=5000, start_time=0, time_limit=1100):
    N = len(h)
    h = h.copy()
    c5_val = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_val

    for it in range(max_iters):

        elif move == 1:
            i, j = rng.choice(N, size=2, replace=False)

            h_trial[i] = np.clip(h_trial[i] + d, 0, 1)
            h_trial[j] = np.clip(h_trial[j] - d, 0, 1)

            i, j = rng.choice(N, size=2, replace=False)
            d = rng.uniform(0.1, 0.5)

            h_trial[j] = np.clip(h_trial[j] + d, 0, 1)

    best_b = b.copy()

                c5_val = c5_trial
