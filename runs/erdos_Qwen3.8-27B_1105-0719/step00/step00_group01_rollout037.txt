import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum):
    """Project to [0,1]^n with exact sum."""
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Distribute positive mass to positions with most room
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-12:
            return h
        h += room * (diff / total_room)
    else:
        # Remove mass from positions with most value
        total_val = h.sum()
        if total_val < 1e-12:
            return h
        h *= (1.0 + diff / total_val)
        h = np.clip(h, 0.0, 1.0)
        # Fix residual
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            # Distribute to positions at 0 or 1
            if diff2 > 0:
                mask = h < 1.0
                if mask.any():
                    h[mask] += diff2 / mask.sum()
            else:
                mask = h > 0.0
                if mask.any():
                    h[mask] += diff2 / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final fix: adjust one element
    residual = target_sum - h.sum()
    if abs(residual) > 1e-15:
        # Find an element that can absorb the residual
        if residual > 0:
            candidates = np.where(h < 1.0 - abs(residual))[0]
        else:
            candidates = np.where(h > abs(residual))[0]
        if len(candidates) > 0:
            h[candidates[0]] += residual
            h = np.clip(h, 0, 1)
    return h

def pair_line_search(h, i, j, n_grid=40):
    """Find best t for h[i]+=t, h[j]-=t that minimizes C5."""
    n = len(h)
    dx = 2.0 / n
    
    # Bounds
    t_low = max(-h[i], h[j] - 1.0)
    t_high = min(1.0 - h[i], h[j])
    
    if t_low >= t_high - 1e-12:
        return h.copy(), evaluate_c5(h)
    
    # Precompute base correlation
    base = np.correlate(h, 1.0 - h, mode="full")
    
    # Direction
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    
    # Linear coefficient: correlate(h, -d) + correlate(d, 1-h)
    lin = np.correlate(h, -d, mode="full") + np.correlate(d, 1.0 - h, mode="full")
    
    # Quadratic coefficient: correlate(d, -d)
    quad = np.correlate(d, -d, mode="full")
    
    # Grid search
    ts = np.linspace(t_low, t_high, n_grid)
    vals = np.maximum.reduce([
        base[k] + ts * lin[k] + ts**2 * quad[k] for k in range(len(base))
    ]) * dx
    # Actually, vectorize:
    # vals[k] = base[k] + ts*lin[k] + ts^2*quad[k], take max over k
    mat = base[np.newaxis, :] + ts[:, np.newaxis] * lin[np.newaxis, :] + (ts[:, np.newaxis])**2 * quad[np.newaxis, :]
    vals = mat.max(axis=1) * dx
    
    idx = np.argmin(vals)
    best_t = ts[idx]
    best_val = vals[idx]
    
    # Refine with ternary search
    # Find bracket
    lo = ts[max(0, idx-1)]
    hi = ts[min(len(ts)-1, idx+1)]
    for _ in range(20):
        m1 = lo + (hi - lo) * 0.333
        m2 = hi - (hi - lo) * 0.333
        v1 = (base + m1 * lin + m1**2 * quad).max() * dx
        v2 = (base + m2 * lin + m2**2 * quad).max() * dx
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    best_t = (lo + hi) / 2.0
    best_val = (base + best_t * lin + best_t**2 * quad).max() * dx
    
    h_new = h.copy()
    h_new[i] += best_t
    h_new[j] -= best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, best_val

def make_feasible(h, target_sum):

        else:
            idx = int(np.argmin(h))

def pair_line_search(h, i, j, n_grid=50):
    n = len(h)
    dx = 2.0 / n
    
    t_low = max(-h[i], h[j] - 1.0)
    t_high = min(1.0 - h[i], h[j])

        return h, evaluate_c5(h)
    
    one_minus_h = 1.0 - h
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    
    C0 = np.correlate(h, one_minus_h, mode="full")
    C1 = np.correlate(h, -d, mode="full") + np.correlate(d, one_minus_h, mode="full")
    C2 = np.correlate(d, -d, mode="full")
    
    ts = np.linspace(t_low, t_high, n_grid)
    vals = C0[None, :] + ts[:, None] * C1[None, :] + (ts[:, None])**2 * C2[None, :]
    obj = vals.max(axis=1) * dx
    
    idx = int(np.argmin(obj))
    
    # Refine with golden section
    lo = ts[max(0, idx-1)]
    hi = ts[min(len(ts)-1, idx+1)]
    for _ in range(25):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = (C0 + m1 * C1 + m1*m1 * C2).max()
        v2 = (C0 + m2 * C1 + m2*m2 * C2).max()

    h_new[i] += best_t
    h_new[j] -= best_t
    h_new = np.clip(h_new, 0.0, 1.0)
