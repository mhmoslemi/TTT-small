import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        # Spread among all non-bound variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    """Compute all correlation values (the raw sums, not multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj_grad(h, beta):
    """Return objective and gradient for the softmax-smoothed C5."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute softmax weights
    c_max = np.max(c)
    shifted = (c - c_max) / dx * beta
    # Clip to avoid underflow
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Objective: dx * (c_max/dx + log(sum exp)/beta) = c_max + (dx/beta)*log(sum exp)
    obj = c_max + (dx / beta) * np.log(np.sum(exp_vals))
    
    # Gradient
    g = np.zeros(n)
    n_corr = 2 * n - 1
    center = n - 1
    
    for p in range(n_corr):
        wp = w[p]
        if wp < 1e-14:
            continue
        lag = p - center
        m = abs(lag)
        if m == 0:
            g += wp * (1.0 - 2.0 * h)
        elif lag > 0:
            # c_m = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
            for i in range(n - m):
                g[i + m] += wp * (1.0 - h[i])
                g[i] += wp * (-h[i + m])
        else:
            # c_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            for i in range(n - m):
                g[i] += wp * (1.0 - h[i + m])
                g[i + m] += wp * (-h[i])
    
    return dx * obj, dx * g

    return np.sum(h) - len(h) / 2.0

def sum_constraint_jac(h):
    return np.ones(len(h))

            break

            o, _ = softmax_obj_grad(x, beta)
            return o

            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                h0 = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c, n

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    shifted = (c - cmax) * beta / dx
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    return cmax + dx * np.log(np.sum(e)) / beta

    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def optimize_slsqp(h_start, beta, n, maxiter=400):
    h = np.asarray(h_start, dtype=np.float64).copy()

        bounds=bounds,
        constraints=cons,

    return h_out


def optimize_trust(h_start, beta, n, maxiter=1000):
