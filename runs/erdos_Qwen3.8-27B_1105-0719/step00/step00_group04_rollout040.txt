```python
import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum} via bisection on shift."""
    n = len(h)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def smooth_max_grad(h, dx, tau):
    """Compute smooth max surrogate value and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Stable log-sum-exp
    m = corr.max()
    exp_shifted = np.exp((corr - m) / tau)
    logsum = m + tau * np.log(exp_shifted.sum())
    weights = exp_shifted / exp_shifted.sum()
    
    # Gradient computation via convolutions
    # first_term[j] = sum_t (1-h[t]) * weights[n-1+j-t] for valid t
    # = np.convolve(1-h, weights, "full")[n-1 : 2n-1]
    conv1 = np.convolve(1.0 - h, weights, "full")
    first_term = conv1[n-1 : 2*n-1]
    
    # second_term[j] = sum_t h[t] * weights[2n-2-(n-1+j-t)] = sum_t h[t] * weights[n-1-j+t]
    # = np.convolve(h, weights[::-1], "full")[n-1 : 2n-1]
    conv2 = np.convolve(h, weights[::-1], "full")
    second_term = conv2[n-1 : 2*n-1]
    
    grad = dx * (first_term - second_term)
    return logsum, weights, grad

def optimize_from(h0, tau_start, tau_end, n_iters, momentum=0.9):
    """Projected gradient descent with momentum and tau annealing."""
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    
    # Initial projection
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_exact = compute_c5(h)
    
    v = np.zeros(n)  # momentum
    lr = 1.0  # will adapt
    
    for it in range(n_iters):
        # Anneal tau
        progress = it / max(n_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        obj, weights, grad = smooth_max_grad(h, dx, tau)
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project
        h_new = project_feasible(h_new, target_sum)
        
        # Check if objective improved (using smooth surrogate)
        obj_new, _, _ = smooth_max_grad(h_new, dx, tau)
        if obj_new > obj * 1.01 + 1e-8:
            # Step too large, reduce lr
            lr *= 0.5
            v = np.zeros(n)
            continue
        
        h = h_new
        
        # Evaluate exact
        exact = compute_c5(h)
        if exact < best_exact:
            best_exact = exact
            best_h = h.copy()
        
        # Adaptive lr: if improvement is good, increase
        if it % 10 == 0 and it > 0:
            lr = min(lr * 1.1, 5.0)
    
    return best_h, best_exact
```

```python
import numpy as np
import time

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = h.min() - 3.0, h.max() + 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def smooth_max_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    exp_s = np.exp((corr - m) / tau)
    Z = exp_s.sum()
    obj = m + tau * np.log(Z)
    w = exp_s / Z
    c1 = np.convolve(1.0 - h, w, "full")
    c2 = np.convolve(h, w[::-1], "full")
    grad = dx * (c1[n-1:2*n-1] - c2[n-1:2*n-1])
    return obj, grad

def run_projected_grad(h0, time_limit, tau_start=0.05, tau_end=0.001, 
                       n_phases=5, momentum=0.85, verbose=False):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t0 = time.time()
    
    # Total iterations spread across phases
    iters_per_phase = 500
    total_iters = n_phases * iters_per_phase
    
    for phase in range(n_phases):
        progress = phase / max(n_phases - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = 0.5 * tau / dx  # scale lr with tau
        
        for it in range(iters_per_phase):
            if time.time() - t0 > time_limit:
                return best_h, best_c5
            
            obj, grad = smooth_max_grad(h, dx, tau)
            
            # Nesterov step
            v_new = momentum * v - lr * grad
            h_trial = h + v_new
            h_trial = project_feasible(h_trial, target)
            
            # Quick check: evaluate at trial point
            obj_trial, _ = smooth_max_grad(h_trial, dx, tau)

                h = h_trial
            else:

                h = project_feasible(h + v, target)
            
            exact = compute_c5(h)
            if exact < best_c5:
                best_c5 = exact
                best_h = h.copy()
```

```python
import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}."""
    lo, hi = h.min() - 3.0, h.max() + 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)

def smooth_max_grad(h, dx, tau):
    """Compute smooth max surrogate and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    exp_s = np.exp((corr - m) / tau)
    Z = exp_s.sum()
    obj = m + tau * np.log(Z)
    w = exp_s / Z
    c1 = np.convolve(1.0 - h, w, "full")
    c2 = np.convolve(h, w[::-1], "full")
    grad = dx * (c1[n-1:2*n-1] - c2[n-1:2*n-1])
    return obj, grad

def optimize_projected(h0, time_budget):
    """Projected gradient descent with tau annealing and backtracking."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    schedule = [
        (0.08, 600, 3.0),
        (0.04, 600, 2.5),
        (0.02, 600, 2.0),
        (0.01, 600, 1.5),
        (0.005, 600, 1.2),
        (0.002, 600, 0.8),
        (0.001, 600, 0.5),
        (0.0005, 600, 0.3),
    ]

    for tau, n_iters, base_lr in schedule:
        if time.time() - t0 > time_budget:
            break
        lr = base_lr
        for it in range(n_iters):
            if time.time() - t0 > time_budget:
                break
            obj, grad = smooth_max_grad(h, dx, tau)
            step = lr
            h_new = h
            for _ in range(6):
                h_cand = project_feasible(h - step * grad, target)
                obj_cand, _ = smooth_max_grad(h_cand, dx, tau)
                if obj_cand < obj:
                    h_new = h_cand
                    break
                step *= 0.5
            h = h_new
            exact = compute_c5(h)
            if exact < best_c5:
                best_c5 = exact
                best_h = h.copy()
            # Slowly decay lr within phase
            if it % 100 == 99:
                lr *= 0.85

    return best_h, best_c5

def upsample_seq(h, n_new):
    """Upsample sequence h to length n_new via linear interpolation."""
    n = len(h)
    indices = np.arange(n_new) * n / n_new
    return np.interp(indices, np.arange(n), h)

def random_feasible(n, rng):
    """Generate a random feasible vector."""
    h = rng.random(n)
    return project_feasible(h, n / 2.0)

def block_pattern(n, n_blocks, rng):
    """Generate a block-like pattern."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        if rng.random() < 0.5:
            h[i*block_size:(i+1)*block_size] = 1.0
    return project_feasible(h, n / 2.0)
```
