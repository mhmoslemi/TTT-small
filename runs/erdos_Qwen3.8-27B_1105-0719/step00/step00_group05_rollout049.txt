import numpy as np

def exact_C(h):
    """Compute the exact C5 value for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Project h to feasible set: 0<=h_i<=1, sum(h)=n/2."""
    n = len(h)
    T = n / 2.0
    h = np.array(h, dtype=np.float64)
    # Water filling: find lambda such that sum(clip(h_i + lambda, 0, 1)) = T
    lo, hi = -max(np.abs(h).max(), 1.0) - 2.0, max(np.abs(h).max(), 1.0) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    delta = T - np.sum(h_proj)
    if abs(delta) > 1e-12:
        # Distribute among coordinates with room
        if delta > 0:
            order = np.argsort(1.0 - h_proj)  # most room first
        else:
            order = np.argsort(h_proj)  # most value first
        for idx in order:
            if delta > 0:
                step = min(1.0 - h_proj[idx], delta)
            else:
                step = max(h_proj[idx] + delta, delta)
            h_proj[idx] += step
            delta -= step
            if abs(delta) < 1e-14:
                break
    return h_proj

def sigmoid(t):
    """Stable sigmoid."""
    t = np.clip(t, -500, 500)
    return np.where(t >= 0, 1/(1+np.exp(-t)), np.exp(t)/(1+np.exp(t)))

def logit(t):
    """Stable logit (inverse sigmoid)."""
    t = np.clip(t, 1e-15, 1-1e-15)
    return np.log(t / (1 - t))

def feasible_from_logit(y):
    """Convert logit vector y to feasible h with sum = n/2."""
    n = len(y)
    T = n / 2.0
    # Find lambda such that sum sigmoid(y_i + lambda) = T
    # sigmoid(y_i + lambda) is increasing in lambda
    lo = -np.max(y) - 50.0
    hi = -np.min(y) + 50.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(sigmoid(y + mid))
        if s < T:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = sigmoid(y + lam)
    # Fix tiny residual
    delta = T - np.sum(h)
    if abs(delta) > 1e-13:
        # Small correction
        h = project_feasible(h)
    return h

def compute_directional_values(h):
    """Compute all directional overlap values v_l = dx * correlate at each lag."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx, corr

def smooth_objective_and_grad_logit(y, beta):
    """Compute smoothed objective and gradient with respect to y (logit space)."""
    n = len(y)
    h = feasible_from_logit(y)
    
    # Compute directional values
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    v = corr * dx  # shape (2n-1,)
    
    # Smoothed max
    v_max = np.max(v)
    v_shifted = beta * (v - v_max)
    # Softmax weights
    exp_v = np.exp(v_shifted - np.max(v_shifted))
    w = exp_v / np.sum(exp_v)
    F = v_max + (1.0/beta) * (np.max(v_shifted) + np.log(np.sum(exp_v)))
    
    # Gradient of smooth max with respect to h
    # G_a = sum_l w_l * dv_l/dh_a
    # v_l = dx * corr_l where corr_l = sum_i h[i] * (1-h)[i - (n-1) + l]
    # Actually let me compute this more carefully
    
    # np.correlate(h, 1-h, mode="full")[k] for k=0,...,2n-2
    # = sum_{i} h[i+k-(n-1)] * (1-h)[i] where indices are valid
    # Let lag = k - (n-1), so lag ranges from -(n-1) to (n-1)
    # correlate value at lag k: sum_i h[i+k] * (1-h[i]) for valid i
    
    # dv/dh_a = dx * sum_l w_l * dv_l/dh_a
    # v_l = dx * sum_i h[i+lag] * (1-h[i])
    # dv_l/dh_a = dx * [(1-h[a-lag]) if a-lag valid + (-h[a]) if a valid as (1-h) term]
    # More precisely:
    # v_l = dx * sum_i h[i+lag_l] * (1-h[i]) where lag_l = l - (n-1)
    # dh_a appears in:
    #   - h[a] term: need a+lag_l in [0,n-1], i.e., a = i, so i=a, need a+lag_l valid
    #     contribution: dx * (1-h[a+lag_l]) if 0 <= a+lag_l < n
    #   - (1-h[a]) term: need a-lag_l in [0,n-1], i.e., i+lag_l=a so i=a-lag_l
    #     contribution: dx * (-h[a-lag_l]) if 0 <= a-lag_l < n
    
    # This is O(n * (2n-1)) = O(n^2) per gradient evaluation
    # For n=42, that's about 3500 operations, very fast
    
    G = np.zeros(n)
    nfull = 2 * n - 1
    for l in range(nfull):
        lag = l - (n - 1)
        wl = w[l] * dx
        # Contribution from h[a] where a + lag in [0, n-1]
        # a ranges from max(0, -lag) to min(n-1, n-1-lag)
        a_start = max(0, -lag)
        a_end = min(n, n - lag)  # exclusive
        if a_end > a_start:
            G[a_start:a_end] += wl * (1.0 - h[a_start + lag : a_end + lag])
        # Contribution from (1-h[a]) where a - lag in [0, n-1]
        # a ranges from max(0, lag) to min(n-1, n-1+lag)
        a_start2 = max(0, lag)
        a_end2 = min(n, n + lag)  # exclusive
        if a_end2 > a_start2:
            G[a_start2:a_end2] += wl * (-h[a_start2 - lag : a_end2 - lag])
    
    # Chain rule: dF/dy_j = h_j(1-h_j) * G_j - h_j(1-h_j)/M * sum_a h_a(1-h_a)*G_a
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-15:
        grad_y = np.zeros(n)
    else:
        B = np.dot(G, m) / M
        grad_y = m * (G - B)
    
    # Replace any non-finite values
    if not np.all(np.isfinite(grad_y)):
        grad_y = np.where(np.isfinite(grad_y), grad_y, 0.0)
    
    return F, grad_y

def _smooth_obj_grad_h(h, beta):
    """Smoothed objective and gradient with respect to h (not logit)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr * dx
    v_max = np.max(v)
    v_shifted = beta * (v - v_max)
    exp_v = np.exp(v_shifted - np.max(v_shifted))
    w = exp_v / np.sum(exp_v)
    F = v_max + (1.0/beta) * np.log(np.sum(exp_v))
    
    # Gradient
    G = np.zeros(n)
    nfull = 2 * n - 1
    for l in range(nfull):
        lag = l - (n - 1)
        wl = w[l] * dx
        a_start = max(0, -lag)
        a_end = min(n, n - lag)
        if a_end > a_start:
            G[a_start:a_end] += wl * (1.0 - h[a_start + lag:a_end + lag])
        a_start2 = max(0, lag)
        a_end2 = min(n, n + lag)
        if a_end2 > a_start2:
            G[a_start2:a_end2] += wl * (-h[a_start2 - lag:a_end2 - lag])
    
    return F, G

def bfgs_optimize(h_start, beta, max_iter=100):
    """Run BFGS from h_start using logit parameterization."""
    n = len(h_start)
    h_safe = np.clip(h_start, 1e-7, 1-1e-7)
    y0 = logit(h_safe)
    
    def obj_func(y):
        if time.time() > _deadline - 30:
            raise ValueError("time up")
        F, g = smooth_objective_and_grad_logit(y, beta)
        return F, g
    
    try:
        res = minimize(obj_func, y0, method='BFGS', jac=True,
                       options={'maxiter': max_iter, 'gtol': 1e-6, 'disp': False})
        h_end = feasible_from_logit(res.x)
        h_end = project_feasible(h_end)
        return h_end
    except (ValueError, Exception):
        return h_start

def interpolate_h(h, new_n):
    """Interpolate h to new number of points."""
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_n_vals = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_n_vals, old_x, h)
    h_new = project_feasible(h_new)
    return h_new

        if wl < 1e-16:

        a_lo = max(0, -lag)
        a_hi = min(n, n - lag)
        if a_hi > a_lo:
            G[a_lo:a_hi] += wl * (1.0 - h[a_lo + lag:a_hi + lag])
        a_lo2 = max(0, lag)
        a_hi2 = min(n, n + lag)
        if a_hi2 > a_lo2:
            G[a_lo2:a_hi2] += wl * (-h[a_lo2 - lag:a_hi2 - lag])
    return F, G


def smooth_obj_grad_logit(y, beta):
    n = len(y)
    h = feasible_from_logit(y)
    F_h, G = smooth_obj_grad_h(h, beta)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-14:
        return F_h, np.zeros(n)
    B = np.dot(G, m) / M
    grad_y = m * (G - B)
    if not np.all(np.isfinite(grad_y)):
        grad_y = np.nan_to_num(grad_y)
    if not np.isfinite(F_h):
        F_h = 1e6
    return F_h, grad_y


def bfgs_objective(y):
    if time.time() > _deadline - 15:
        raise OverflowError("deadline")
    F, g = smooth_obj_grad_logit(y, _current_beta)
    if not np.isfinite(F):
        return 1e6, np.zeros_like(y)
    return F, g


def bfgs_run(h_start, beta, max_iter=150):
    global _current_beta
    _current_beta = beta
    h_safe = np.clip(h_start, 1e-8, 1.0 - 1e-8)
    y0 = logit_stable(h_safe)
    try:
        res = minimize(bfgs_objective, y0, method='BFGS', jac=True,
                       options={'maxiter': max_iter, 'gtol': 1e-5})
        h_end = feasible_from_logit(res.x)
        return project_feasible(h_end)
    except Exception:
        return h_start


def pair_line_min(h, a, b):
    """Find best t for h[a]+=t, h[b]-=t using grid+refine."""
    cur_c = exact_C(h)
    lo_t = max(0.0, h[b] - 1.0) - h[a]
    hi_t = min(h[b], 1.0 - h[a])
    if hi_t - lo_t < 1e-14:
        return 0.0, cur_c
    ts = np.linspace(lo_t, hi_t, 40)
    best_t = 0.0
    best_c = cur_c
    for t in ts:
        h_try = h.copy()
        h_try[a] += t
        h_try[b] -= t
        if h_try[a] < -1e-12 or h_try[a] > 1 + 1e-12:
            continue
        if h_try[b] < -1e-12 or h_try[b] > 1 + 1e-12:
            continue
        c = exact_C(h_try)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
    return best_t, best_c


def pair_polish(h, max_rounds=40, time_limit=30.0):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    cur_c = exact_C(h)
    t_end = min(time.time() + time_limit, _deadline - 10)
    stagnant = 0
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        # Compute gradient to find active coordinates
        _, G = smooth_obj_grad_h(h, 200.0)
        abs_G = np.abs(G)
        top_idx = np.argsort(-abs_G)[:10]
        pairs = []
        for i in range(len(top_idx)):
            for j in range(i + 1, min(i + 4, len(top_idx))):
                pairs.append((int(top_idx[i]), int(top_idx[j])))
        # Add random pairs
        for _ in range(min(15, n)):
            a = int(_rng.integers(0, n))
            b = int(_rng.integers(0, n))
            if a != b:
                pairs.append((min(a, b), max(a, b)))
        # Also include (0,1), (0,n-1) as structural pairs
        if n > 2:
            pairs.append((0, n - 1))
            pairs.append((0, 1))
        best_move = None
        best_move_c = cur_c
        for (a, b) in pairs:
            if time.time() > t_end:
                break
            t, c = pair_line_min(h, a, b)
            if c < best_move_c - 1e-14:
                best_move_c = c
                best_move = (a, b, t)
        if best_move is not None:
            a, b, t = best_move
            h[a] += t
            h[b] -= t
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h)
            cur_c = exact_C(h)
            stagnant = 0
        else:
            # Try random small moves
            improved = False
            for _ in range(8):
                a = int(_rng.integers(0, n))
                b = int(_rng.integers(0, n))
                if a == b:
                    continue
                t = _rng.uniform(-0.03, 0.03)
                h_try = h.copy()
                h_try[a] += t
                h_try[b] -= t
                if np.min(h_try) >= -1e-10 and np.max(h_try) <= 1 + 1e-10:
                    h_try = np.clip(h_try, 0, 1)
                    c_try = exact_C(h_try)
                    if c_try < cur_c - 1e-14:
                        h = h_try
                        cur_c = c_try
                        improved = True
                        break
            if improved:
                stagnant = 0
            else:
                stagnant += 1
                if stagnant >= 3:
                    break
    return h, cur_c


def interpolate_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    global _best_h, _best_c, _start_time, _deadline, _rng

    _start_time = time.time()
    _deadline = _start_time + 0.92 * budget_s
    _rng = np.random.default_rng(seed)
    _best_h = None
    _best_c = np.inf

    target = 0.3808

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).flatten()
    except (NameError, Exception):
        h_init = np.full(42, 0.5)

    n0 = len(h_init)
    h_init = project_feasible(h_init)
    record_candidate(h_init)

    if _best_c <= target:
        return (_best_h.copy(), _best_c, len(_best_h))

    # Build diverse starting points
    starts = [h_init.copy()]

    # Reversed
    starts.append(h_init[::-1].copy())

    # Shifts
    for s in [1, 2, 3, 5]:
        starts.append(np.roll(h_init, s).copy())

    # Algorithmic perturbations
    n = n0
    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        for amp in [0.03, 0.08, 0.12]:
            h_var = h_init + amp * np.sin(2 * np.pi * freq * x / n)
            h_var = project_feasible(h_var)
            starts.append(h_var)
    for freq in [1, 2, 4]:
        for amp in [0.03, 0.06]:
            h_var = h_init + amp * np.cos(2 * np.pi * freq * x / n)
            h_var = project_feasible(h_var)
            starts.append(h_var)

    # Random perturbations
    for scale in [0.02, 0.05, 0.1]:
        for _ in range(3):
            h_var = h_init + scale * _rng.standard_normal(n)
            h_var = project_feasible(h_var)
            starts.append(h_var)

    # BFGS phase
    betas = [50.0, 100.0, 200.0, 400.0, 700.0]

    for si, h_s in enumerate(starts):
        if time.time() > _deadline - 100:
            break
        for beta in betas:
            if time.time() > _deadline - 80:
                break
            remaining = max(30, _deadline - time.time() - 60)
            max_iter = min(200, max(40, int(remaining / 4)))
            h_opt = bfgs_run(h_s, beta, max_iter=max_iter)
            record_candidate(h_opt)
            if _best_c <= target:
                return (_best_h.copy(), _best_c, len(_best_h))
            # Quick pair polish
            if time.time() < _deadline - 50:
                hl, cl = pair_polish(h_opt, max_rounds=10,
                                     time_limit=min(15, max(3, _deadline - time.time() - 40)))
                record_candidate(hl)
                if _best_c <= target:
                    return (_best_h.copy(), _best_c, len(_best_h))

    # Intensive pair polish from best
    if _best_h is not None and time.time() < _deadline - 20:
        remaining = max(5, _deadline - time.time() - 10)
        hp, cp = pair_polish(_best_h, max_rounds=80, time_limit=remaining * 0.6)
        record_candidate(hp)
        if _best_c <= target:
            return (_best_h.copy(), _best_c, len(_best_h))

    # Try alternate dimensions
    if _best_h is not None and time.time() < _deadline - 15:
        for new_n in [max(20, n0 - 8), n0 + 8, max(16, int(n0 * 0.7))]:
            if new_n == n0 or time.time() > _deadline - 12:
                continue
            h_new = interpolate_h(_best_h, new_n)
            record_candidate(h_new)
            if time.time() < _deadline - 12:
                h_opt = bfgs_run(h_new, 200.0, max_iter=60)
                record_candidate(h_opt)
            if _best_c <= target:
                return (_best_h.copy(), _best_c, len(_best_h))

    # Final polish
    if _best_h is not None and time.time() < _deadline - 5:
        remaining = max(3, _deadline - time.time() - 3)
        hf, cf = pair_polish(_best_h, max_rounds=50, time_limit=remaining)
        record_candidate(hf)

    # Return best
    if _best_h is not None:
        h_ret = project_feasible(_best_h)
        c_ret = exact_C(h_ret)
        return (h_ret, c_ret, len(h_ret))
    h_fb = np.full(42, 0.5)
    return (h_fb, exact_C(h_fb), 42)
