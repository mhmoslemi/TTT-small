def _interval_intersection_with_quad(a_min, a_max, A, B, C_minus_T):
    """Given current feasible interval [a_min, a_max], return new feasible set
    after adding constraint A*x^2 + B*x + C_minus_T <= 0."""
    if abs(A) < 1e-15:
        # Linear
        if abs(B) < 1e-15:
            if C_minus_T > 0:
                return []  # infeasible
            return [(a_min, a_max)]
        if B > 0:
            root = -C_minus_T / B
            return [(a_min, min(a_max, root))] if a_min <= min(a_max, root) else []
        else:
            root = -C_minus_T / B
            return [(max(a_min, root), a_max)] if max(a_min, root) <= a_max else []
    else:
        disc = B*B - 4*A*C_minus_T
        if disc < -1e-12:
            if A > 0:
                return []  # Always infeasible
            else:
                return [(a_min, a_max)]  # Always feasible
        if disc < 1e-12:
            r = -B / (2*A)
            if A > 0:
                return [(r, r)] if a_min <= r <= a_max else []
            else:
                # Feasible everywhere except at r (measure zero), so effectively all
                return [(a_min, a_max)]
        sqrt_d = np.sqrt(disc)
        r1 = (-B - sqrt_d) / (2*A)
        r2 = (-B + sqrt_d) / (2*A)
        r_lo = min(r1, r2)
        r_hi = max(r1, r2)
        if A > 0:
            # Feasible between roots: [r_lo, r_hi] ∩ [a_min, a_max]
            new_min = max(a_min, r_lo)
            new_max = min(a_max, r_hi)
            if new_min <= new_max:
                return [(new_min, new_max)]
            return []
        else:
            # Feasible outside roots: (-inf, r_lo] ∪ [r_hi, inf)
            # Intersect with [a_min, a_max]:
            result = []
            # Left part: [a_min, min(a_max, r_lo)]
            if a_min <= min(a_max, r_lo):
                result.append((a_min, min(a_max, r_lo)))
            # Right part: [max(a_min, r_hi), a_max]

                result.append((max(a_min, r_hi), a_max))
