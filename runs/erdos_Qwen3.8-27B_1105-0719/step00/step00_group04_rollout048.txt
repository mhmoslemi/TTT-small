import time

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(h):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Add mass
        mask = h < 1.0
        capacities = np.where(mask, 1.0 - h, 0.0)
        total_cap = capacities.sum()
        if total_cap < diff:
            # Can't satisfy, just set to target proportionally
            h = h / h.sum() * target if h.sum() > 0 else np.full(n, 0.5)
            return h
        scale = diff / total_cap
        h = h + scale * capacities
    else:
        # Remove mass
        mask = h > 0.0
        avail = np.where(mask, h, 0.0)
        total_avail = avail.sum()
        if total_avail < -diff:
            h = h / h.sum() * target if h.sum() > 0 else np.full(n, 0.5)
            return h
        scale = -diff / total_avail
        h = h - scale * avail
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += residual if residual > 0 else residual
        h = np.clip(h, 0.0, 1.0)

    grads = np.zeros((2*n-1, n))

            grads[m, 0:n-d] += dx * (1.0 - h[d:n])
            grads[m, d:n] += -dx * h[0:n-d]

        grad = weights @ G  # shape (n,)

def interpolate_to_length(h, new_n):
    """Interpolate step function to new number of points."""
    n = len(h)
    if n == new_n:

    x_old = (np.arange(n) + 0.5) * (2.0/n)
    x_new = (np.arange(new_n) + 0.5) * (2.0/new_n)
    # Piecewise constant interpolation: for each new point, find which old bin it falls in
    h_new = np.zeros(new_n)
    for i in range(new_n):

        old_idx = min(old_idx, n-1)

def slsqp_epigraph(h_init, max_iter=200, beta=0.0):

    n = len(h_init)
    dx = 2.0 / n
    h0 = repair_feasible(h_init.copy())
    c0 = exact_score(h0)
    t0 = c0 + 0.01
    
    x0 = np.concatenate([h0, [t0]])
    
    def obj(x):

        t = x[n]
        val = t
        if beta > 0:
            val += beta * np.sum(h * (1.0 - h))
        return val
    
    def obj_grad(x):

        g = np.zeros(n+1)
        g[n] = 1.0
        if beta > 0:
            g[:n] = beta * (1.0 - 2.0*h)
        return g
    
    def con_eq(x):
        return x[:n].sum() - n/2.0
    
    def con_eq_grad(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g

    # We'll use a single vector-valued constraint

        h = x[:n]
        t = x[n]
        prof = np.correlate(h, 1.0-h, mode='full') * dx
        return t - prof
    
    def con_ineq_jac(x):
        h = x[:n]
        G = lag_gradient_matrix(h)
        # Jacobian of (t - prof) w.r.t. x
        # d/dh: -G[m,a], d/dt: 1
        J = np.zeros((2*n-1, n+1))
        J[:, n] = 1.0
        J[:, :n] = -G
        return J

    bounds = [(0,1)]*n + [(0, 1.0)]
    
    result = minimize(obj, x0, jac=obj_grad, method='SLSQP',

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def profile_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(h):
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        cap[h >= 1.0] = 0.0
        total_cap = cap.sum()
        if total_cap > 0:
            h += (diff / total_cap) * cap
        else:
            h[:] = 0.5
    else:
        avail = h.copy()
        avail[h <= 0.0] = 0.0
        total_avail = avail.sum()
        if total_avail > 0:
            h -= (-diff / total_avail) * avail
        else:
            h[:] = 0.5
    h = np.clip(h, 0.0, 1.0)
    # Final correction on one element
    res = target - h.sum()
    if abs(res) > 1e-14:
        if res > 0:
            idx = np.argmax(1.0 - h)
            add = min(res, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            rem = min(-res, h[idx])
            h[idx] -= rem
    h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient_matrix(h):
    """Returns (2n-1, n) matrix of gradients."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n - 1, n))
    for m in range(2*n - 1):
        delta = m - (n - 1)
        if delta == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            G[m, delta:n] += dx * (1.0 - h[0:n-delta])
            G[m, 0:n-delta] += -dx * h[delta:n]
        else:
            d = -delta
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
    return G

class EpigraphProb:
    def __init__(self, n, beta=0.0):
        self.n = n
        self.dx = 2.0 / n
        self.beta = beta
    
    def f(self, x):
        return x[self.n] + self.beta * np.sum(x[:self.n] * (1.0 - x[:self.n]))
    
    def f_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        if self.beta > 0:
            g[:self.n] = self.beta * (1.0 - 2.0 * x[:self.n])
        return g

    def h_eq_jac(self, x):

        return g

        t = x[self.n]
        prof = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return t - prof
    
    def h_ineq_jac(self, x):
        h = x[:self.n]
        G = lag_gradient_matrix(h)
        J = np.zeros((2*self.n - 1, self.n + 1))
        J[:, self.n] = 1.0
        J[:, :self.n] = -G
        return J

def slsqp_solve(h_init, beta=0.0, max_iter=200):
    n = len(h_init)
    h0 = repair_feasible(h_init.copy())
    c0 = exact_score(h0)
    t0 = c0 + 0.05
    x0 = np.concatenate([h0, [t0]])
    
    prob = EpigraphProb(n, beta)
    
    constraints = [
        {'type': 'eq', 'fun': prob.h_eq, 'jac': prob.h_eq_jac},
        {'type': 'ineq', 'fun': prob.h_ineq, 'jac': prob.h_ineq_jac},
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    try:
        res = minimize(prob.f, x0, jac=prob.f_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n].copy())
    except Exception:
        pass
    return h0

def active_descent(h_init, n_active=20, max_iter=300):
    h = repair_feasible(h_init.copy())
    best = h.copy()
    best_c = exact_score(h)
    n = len(h)
    
    for it in range(max_iter):
        prof = profile_full(h)
        c = prof.max()
        if c < best_c - 1e-13:

            best = h.copy()

        gaps = prof[top_idx] - prof.min()

        grad = (G[top_idx] * w[:, None]).sum(axis=0)

            norm = np.linalg.norm(g_proj)

Note F(k) + F(-k) = (number of i where h[i]≠h[i+k]) * dx.

So for binary h, the problem is to arrange n/2 ones and n/2 zeros to minimize the maximum number of "inversions" at any lag.

This is related to the "discrepancy" of a binary sequence at all lags. The optimal arrangement would be one where the autocorrelation is as flat as possible.

If h is random (each position independently 0 or 1 with probability 1/2), then for lag k, the expected number of "0→1" transitions is (n-k)*1/4, and the expected F(k) = (n-k)/4 * 2/n = (n-k)/(2n). For n=70 and k=0, this gives about 70/(2*70) = 0.5. But we need the maximum over all k, and there are 2n-1 = 139 lags. The maximum would be around 0.5 + a few standard deviations.

But the current best is 0.38085, which is much less than 0.5. So the optimal arrangement is very structured, not random.

For a periodic sequence with period p, the lags that are multiples of p would have F(k) = 0 (if the period is such that h(i) = h(i+p) for all valid i). Other lags would have some value.
