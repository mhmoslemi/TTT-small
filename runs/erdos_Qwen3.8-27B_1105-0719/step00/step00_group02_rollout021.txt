def evaluate_c5(h):
    """Compute the C5 value from h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_valid_h(n, rng):
    """Create a random valid h with sum = n/2."""
    h = rng.uniform(0, 1, n)
    # Project to sum = n/2
    target = n / 2.0
    s = h.sum()
    if s > 0:
        h = h / s * target
    h = np.clip(h, 0, 1)
    # Fix sum
    _fix_sum(h, target)
    return h

def _fix_sum(h, target):
    """Fix the sum of h to be target, clipping to [0,1]."""
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase some entries
            can_inc = h < 1.0 - 1e-12
            if not can_inc.any():
                break
            room = (1.0 - h[can_inc])
            total_room = room.sum()
            if total_room < diff:
                h[can_inc] += room  # saturate all
            else:
                frac = diff / total_room
                h[can_inc] += room * frac
        else:
            # Need to decrease some entries
            can_dec = h > 1e-12
            if not can_dec.any():
                break
            avail = h[can_dec]
            total_avail = avail.sum()
            if total_avail < -diff:
                h[can_dec] = 0.0
            else:
                frac = (-diff) / total_avail
                h[can_dec] -= avail * frac
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute equally among non-saturated entries
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

def pair_minimax(h, i, j):
    """
    Optimize pair (i,j) keeping sum S = h[i] + h[j] fixed.
    Returns the optimal x = h[i] value.
    """
    S = h[i] + h[j]
    low = max(0.0, S - 1.0)
    high = min(1.0, S)
    if high - low < 1e-14:
        return h[i]

    x0 = low
    x1 = (low + high) / 2.0
    x2 = high
    
    h_copy = h.copy()
    
    h_copy[i] = x0
    h_copy[j] = S - x0
    c0 = np.correlate(h_copy, 1.0 - h_copy, mode="full")
    
    h_copy[i] = x1
    h_copy[j] = S - x1
    c1 = np.correlate(h_copy, 1.0 - h_copy, mode="full")
    
    h_copy[i] = x2
    h_copy[j] = S - x2
    c2 = np.correlate(h_copy, 1.0 - h_copy, mode="full")
    
    # For each lag, find quadratic coefficients a*x^2 + b*x + c
    # Using 3 points: x0, x1, x2
    # Lagrange interpolation
    n_lags = len(c0)
    a = np.zeros(n_lags)
    b = np.zeros(n_lags)
    c = np.zeros(n_lags)

    # L0(x) = (x-x1)(x-x2) / ((x0-x1)(x0-x2))
    # L1(x) = (x-x0)(x-x2) / ((x1-x0)(x1-x2))
    # L2(x) = (x-x0)(x-x1) / ((x2-x0)(x2-x1))
    
    d01 = x0 - x1
    d02 = x0 - x2
    d10 = x1 - x0
    d12 = x1 - x2
    d20 = x2 - x0
    d21 = x2 - x1

    a = c0/(d01*d02) + c1/(d10*d12) + c2/(d20*d21)

    b = -c0*(x1+x2)/(d01*d02) - c1*(x0+x2)/(d10*d12) - c2*(x0+x1)/(d20*d21)

    c = c0*x1*x2/(d01*d02) + c1*x0*x2/(d10*d12) + c2*x0*x1/(d20*d21)

    # i.e., for each lag, x must be in the solution set of a_l*x^2 + b_l*x + (c_l - T) <= 0

        # Returns (feasible, x_candidate)
        # For each lag, find the interval of x where quadratic <= T
        lo = low
        hi = high
        for l in range(n_lags):
            al = a[l]
            bl = b[l]
            cl = c[l] - T
            
            if abs(al) < 1e-15:

                    hi = min(hi, x_max)
                else:

                        # Unless the quadratic is always <= T in our range
                        # Check at midpoint
                        mid = (lo + hi) / 2
                        if al*mid*mid + bl*mid + cl > 1e-12:
                            return False, None
                        continue
                    else:
                        # Always negative, always feasible
                        continue
                if disc < 0:
                    disc = 0
                sqrt_disc = np.sqrt(disc)
                r1 = (-bl - sqrt_disc) / (2*al)
                r2 = (-bl + sqrt_disc) / (2*al)
                if r1 > r2:
                    r1, r2 = r2, r1
                
                if al > 0:
                    # Feasible between r1 and r2
                    lo = max(lo, r1)
                    hi = min(hi, r2)
                else:
                    # Feasible outside [r1, r2]
                    # This means we need x <= r1 or x >= r2
                    # This is a union of two intervals, which complicates things

                    if hi <= r1 + 1e-12:
                        # All in x <= r1 side, ok

                        pass

                        # For simplicity, just keep the larger feasible part
                        left_len = r1 - lo
                        right_len = hi - r2
                        if left_len >= right_len:
                            hi = r1
                        else:
                            lo = r2
            
            if lo > hi + 1e-12:
                return False, None
        
        if lo > hi:
            return False, None
        x_mid = (lo + hi) / 2
        return True, x_mid
    
    # Current best threshold
    T_lo = 0.0
    T_hi = max(np.max(c0), np.max(c1), np.max(c2))
    
    # Actually, we want to minimize T = max_lag quadratic(x)
    # Bisection: find smallest T such that feasible(T) is True
    x_best = h[i]  # default
    c_best = evaluate_c5(h)

        if ok:

            x_best = x_cand
        else:

    h_copy[j] = S - x_best
    h_copy = np.clip(h_copy, 0, 1)
    _fix_sum(h_copy, len(h)/2.0)
    
    return x_best, h_copy

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    """Ensure sum(h) == target with 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = 1.0 - h
            room[~mask] = 0
            total = room.sum()
            if total <= 0:
                break
            scale = min(1.0, diff / total)
            h += room * scale
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            avail = h
            avail[~mask] = 0
            total = avail.sum()
            if total <= 0:
                break
            scale = min(1.0, (-diff) / total)
            h -= avail * scale
    h = np.clip(h, 0.0, 1.0)
    return h

def _check_feasible(T, a, b, c, low, high):
    """Check if exists x in [low, high] with a_l*x^2 + b_l*x + (c_l - T) <= 0 for all l."""
    n_lags = len(a)
    intervals = [(low, high)]
    
    for l in range(n_lags):
        al = a[l]
        bl = b[l]
        cl = c[l] - T
        
        new_intervals = []
        for (lo_i, hi_i) in intervals:
            if abs(al) < 1e-15:
                if abs(bl) < 1e-15:
                    if cl <= 1e-12:
                        new_intervals.append((lo_i, hi_i))
                else:
                    root = -cl / bl
                    if bl > 0:
                        if hi_i <= root + 1e-12:
                            new_intervals.append((lo_i, hi_i))
                        elif lo_i < root:
                            new_intervals.append((lo_i, root))
                    else:
                        if lo_i >= root - 1e-12:
                            new_intervals.append((lo_i, hi_i))
                        elif hi_i > root:
                            new_intervals.append((root, hi_i))
            else:
                disc = bl * bl - 4.0 * al * cl
                if disc < -1e-12:
                    if al < 0:
                        new_intervals.append((lo_i, hi_i))
                    # else: infeasible
                else:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    r1 = (-bl - sq) / (2.0 * al)
                    r2 = (-bl + sq) / (2.0 * al)
                    if r1 > r2:
                        r1, r2 = r2, r1
                    if al > 0:
                        lo_new = max(lo_i, r1)
                        hi_new = min(hi_i, r2)
                        if lo_new <= hi_new + 1e-12:
                            new_intervals.append((lo_new, hi_new))
                    else:
                        # Feasible outside [r1, r2]
                        if lo_i < r1 - 1e-12:
                            new_intervals.append((lo_i, min(hi_i, r1)))
                        if hi_i > r2 + 1e-12:
                            new_intervals.append((max(lo_i, r2), hi_i))
                        # If the interval is entirely between r1 and r2, it's infeasible
                        # If it only partially overlaps, we handled above
        
        if not new_intervals:
            return False, None
        intervals = new_intervals
        if len(intervals) > 20:
            # Too many intervals, do a grid check
            n_grid = 50
            xs = np.linspace(low, high, n_grid)
            for x in xs:
                if np.all(a * x * x + b * x + c <= T + 1e-12):
                    return True, x
            return False, None
    
    # Choose midpoint of largest interval
    best = max(intervals, key=lambda iv: iv[1] - iv[0])
    x_mid = (best[0] + best[1]) / 2.0
    # Clamp
    x_mid = max(low, min(high, x_mid))
    return True, x_mid

But this is the same as m=i term! Because m=i=0 and m=j-L=0.

So I'm double-counting! The term m=0 appears both as "m=i" and as "m=j-L" since j-L = 1-1 = 0 = i.

This is a problem. When i = j-L (i.e., L = j-i), the terms m=i and m=j-L are the same term.

Similarly, when i+L = j (i.e., L = j-i), the term m=i has h[i+L] = h[j] which is being changed.

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            total = room.sum()
            if total <= 1e-15:
                break
            frac = min(1.0, diff / total)
            h += room * frac
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            avail = np.where(mask, h, 0.0)
            total = avail.sum()
            if total <= 1e-15:
                break
            frac = min(1.0, (-diff) / total)
            h -= avail * frac
    h = np.clip(h, 0.0, 1.0)
    return h

def _solve_pair(h, i, j):
    """Solve minimax for pair (i,j). Returns (x_opt, h_new) or None."""
    n = len(h)
    S = h[i] + h[j]
    low = max(0.0, S - 1.0)
    high = min(1.0, S)
    if high - low < 1e-13:
        return None
    
    x0 = low
    x1 = low + 0.5 * (high - low)
    x2 = high
    
    h_tmp = h.copy()
    h_tmp[i] = x0; h_tmp[j] = S - x0
    c0 = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    
    h_tmp[i] = x1; h_tmp[j] = S - x1
    c1 = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    
    h_tmp[i] = x2; h_tmp[j] = S - x2
    c2 = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    
    # Quadratic coefficients via Lagrange
    d01 = x0 - x1; d02 = x0 - x2
    d10 = x1 - x0; d12 = x1 - x2
    d20 = x2 - x0; d21 = x2 - x1
    
    inv0 = 1.0 / (d01 * d02)
    inv1 = 1.0 / (d10 * d12)
    inv2 = 1.0 / (d20 * d21)
    
    a = c0 * inv0 + c1 * inv1 + c2 * inv2
    b = -(c0 * (x1 + x2) * inv0 + c1 * (x0 + x2) * inv1 + c2 * (x0 + x1) * inv2)
    c = c0 * x1 * x2 * inv0 + c1 * x0 * x2 * inv1 + c2 * x0 * x1 * inv2
    
    # Bisection
    T_lo = 0.0
    T_hi = max(c0.max(), c1.max(), c2.max()) + 1e-10
    x_best = h[i]  # start with current value
    
    for _ in range(22):
        T_mid = 0.5 * (T_lo + T_hi)
        ok, x_cand = _feasible(T_mid, a, b, c, low, high)
        if ok:
            T_hi = T_mid
            if x_cand is not None:
                x_best = x_cand
        else:
            T_lo = T_mid
        if T_hi - T_lo < 1e-14:
            break
    
    h_new = h.copy()
    h_new[i] = x_best
    h_new[j] = S - x_best
    h_new = _fix_sum(h_new, n / 2.0)
    return x_best, h_new

def _feasible(T, a, b, c, low, high):
    n_lags = len(a)
    intervals = [(low, high)]
    for l in range(n_lags):
        al = a[l]; bl = b[l]; cl = c[l] - T
        new_iv = []
        for (lo_i, hi_i) in intervals:
            if hi_i - lo_i < 1e-14:
                # Single point, check directly
                val = al*lo_i*lo_i + bl*lo_i + cl
                if val <= 1e-12:
                    new_iv.append((lo_i, hi_i))
                continue
            if abs(al) < 1e-15:
                if abs(bl) < 1e-15:
                    if cl <= 1e-12:
                        new_iv.append((lo_i, hi_i))
                else:
                    root = -cl / bl
                    if bl > 0:
                        if hi_i <= root + 1e-12:
                            new_iv.append((lo_i, hi_i))
                        elif lo_i < root - 1e-12:
                            new_iv.append((lo_i, root))
                    else:
                        if lo_i >= root - 1e-12:
                            new_iv.append((lo_i, hi_i))
                        elif hi_i > root + 1e-12:
                            new_iv.append((root, hi_i))
            else:
                disc = bl*bl - 4.0*al*cl
                if disc < -1e-12:
                    if al < 0:
                        new_iv.append((lo_i, hi_i))
                else:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    r1 = (-bl - sq) / (2.0*al)
                    r2 = (-bl + sq) / (2.0*al)
                    if r1 > r2:
                        r1, r2 = r2, r1
                    if al > 0:
                        lo_new = max(lo_i, r1)
                        hi_new = min(hi_i, r2)
                        if lo_new <= hi_new + 1e-12:
                            new_iv.append((lo_new, hi_new))
                    else:
                        if lo_i < r1 - 1e-12:
                            new_iv.append((lo_i, min(hi_i, r1)))
                        if hi_i > r2 + 1e-12:
                            new_iv.append((max(lo_i, r2), hi_i))
        if not new_iv:
            return False, None
        intervals = new_iv
        if len(intervals) > 10:
            # Fallback: grid search
            xs = np.linspace(low, high, 100)
            for x in xs:
                if np.all(a*x*x + b*x + c <= T + 1e-12):
                    return True, float(x)
            return False, None
    if not intervals:
        return False, None
    best = max(intervals, key=lambda t: t[1] - t[0])
    xm = 0.5 * (best[0] + best[1])
    xm = max(low, min(high, xm))
    return True, float(xm)

def _optimize_pairs(h, target, time_limit, seed=0):
    """Run pair optimization passes until convergence or time limit."""
    start = time.time()
    n = len(h)
    current_score = _eval_c5(h)
    
    all_pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    pass_num = 0
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        if pass_num >= 200:
            break
        
        # Choose pair ordering
        if pass_num % 2 == 0:
            pairs = all_pairs
        else:
            pairs = all_pairs[::-1]
        
        improved = False
        for (i, j) in pairs:
            elapsed = time.time() - start
            if elapsed > time_limit:
                break
            
            result = _solve_pair(h, i, j)
            if result is None:
                continue
            x_opt, h_new = result
            s_new = _eval_c5(h_new)
            if s_new < current_score - 1e-12:
                h = h_new
                current_score = s_new
                improved = True
        
        pass_num += 1
        if not improved:
            break
    
    return h, current_score
