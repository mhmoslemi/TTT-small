import numpy as np
import time

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def repair_feasibility(h, n):
    h = h.astype(np.float64).copy()
    mass = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        r = mass - np.sum(h)
        if abs(r) < 1e-12:
            break
        if r > 0:
            cap = np.maximum(0.0, 1.0 - h)
            tc = np.sum(cap)
            if tc < 1e-14:
                break
            h += r * cap / tc
        else:
            aff = np.maximum(1e-14, h)
            ta = np.sum(aff)
            if ta < 1e-14:
                break
            h += r * aff / ta
        h = np.clip(h, 0.0, 1.0)
    return h

def feasible_interval(h, u):
    n = len(h)
    tiny = 1e-14
    t_lo = 0.0
    t_hi = 0.0
    nz = np.abs(u) > tiny
    if not np.any(nz):
        return None
    idx = np.where(nz)[0]
    pos = idx[u[idx] > 0]
    neg = idx[u[idx] < 0]
    if len(pos) > 0:
        t_lo = max(t_lo, np.max(-h[pos] / u[pos]))
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / u[pos]))
    if len(neg) > 0:
        t_lo = max(t_lo, np.max((1.0 - h[neg]) / u[neg]))
        t_hi = min(t_hi, np.min(-h[neg] / u[neg]))
    if t_lo > 0 and t_lo < 1e-10:
        t_lo = 0.0
    if t_hi < 0 and t_hi > -1e-10:
        t_hi = 0.0
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    if t_hi - t_lo < 1e-14:
        return None
    return (t_lo, t_hi)

def solve_ray(h, u, n):
    interval = feasible_interval(h, u)
    if interval is None:
        return 0.0, exact_score(h, n)
    t_lo, t_hi = interval
    if t_hi - t_lo < 1e-13:
        return 0.0, exact_score(h, n)

    dx = 2.0 / n
    g = 1.0 - h
    q0 = np.correlate(h, g, mode="full")
    q1 = np.correlate(u, g, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")

    n_lags = len(q0)
    candidates = [t_lo, t_hi, 0.0]

    tiny = 1e-14
    # Single quadratic stationary points
    nz_q2 = np.abs(q2) > tiny
    if np.any(nz_q2):
        t_stan = -q1[nz_q2] / (2.0 * q2[nz_q2])
        valid = (t_stan >= t_lo) & (t_stan <= t_hi)
        candidates.extend(t_stan[valid].tolist())

    # Pairwise intersections - use all lags for thoroughness
    # C(93,2) = 4278 pairs
    active = np.arange(n_lags)
    n_act = len(active)
    for i in range(n_act):
        for j in range(i + 1, n_act):
            a, b = active[i], active[j]
            dq0 = q0[a] - q0[b]
            dq1 = q1[a] - q1[b]
            dq2 = q2[a] - q2[b]
            if abs(dq2) > tiny:
                disc = dq1 * dq1 - 4.0 * dq2 * dq0
                if disc >= 0:
                    sqrt_d = np.sqrt(disc)
                    t1 = (-dq1 + sqrt_d) / (2.0 * dq2)
                    t2 = (-dq1 - sqrt_d) / (2.0 * dq2)
                    if t_lo <= t1 <= t_hi:
                        candidates.append(t1)
                    if t_lo <= t2 <= t_hi:
                        candidates.append(t2)
            elif abs(dq1) > tiny:
                t = -dq0 / dq1
                if t_lo <= t <= t_hi:
                    candidates.append(t)

    candidates = np.array(candidates, dtype=np.float64)
    candidates = candidates[(candidates >= t_lo - 1e-12) & (candidates <= t_hi + 1e-12)]
    candidates = np.clip(candidates, t_lo, t_hi)

    if len(candidates) == 0:
        return 0.0, exact_score(h, n)

    # Deduplicate
    candidates = np.unique(np.round(candidates, 10))

    # Evaluate all candidates vectorized in chunks
    best_t = 0.0
    best_raw = np.max(q0)  # value at t=0

    chunk_size = 2000
    for start in range(0, len(candidates), chunk_size):
        batch = candidates[start:start + chunk_size]
        scores = np.max(q0[None, :] + batch[:, None] * q1[None, :] + (batch[:, None] ** 2) * q2[None, :], axis=1)
        bi = np.argmin(scores)
        if scores[bi] < best_raw - 1e-15:
            best_raw = scores[bi]
            best_t = batch[bi]

    # Refine around best
    eps = 1e-9 * max(1.0, abs(best_t), t_hi - t_lo)
    for dt in [-eps, eps]:
        t = best_t + dt
        if t_lo <= t <= t_hi:
            f = q0 + t * q1 + t * t * q2
            s = np.max(f)
            if s < best_raw - 1e-15:
                best_raw = s
                best_t = t

    return best_t, best_raw * dx

def gen_random_dir(n, rng):
    u = rng.standard_normal(n)
    u -= np.mean(u)
    nrm = np.linalg.norm(u)
    if nrm < 1e-12:
        return None
    return u / nrm

def gen_sine_dir(n, rng):
    k = rng.integers(1, n // 3 + 1)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    if rng.random() < 0.5:
        u = np.cos(2 * np.pi * k * t / n + phase)
    else:
        u = np.sin(2 * np.pi * k * t / n + phase)
    u -= np.mean(u)
    nrm = np.linalg.norm(u)
    if nrm < 1e-12:
        return None
    return u / nrm

def gen_gradient_dir(h, n):
    """Direction from negative sum of active lag gradients."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_val = np.max(corr)
    if max_val < 1e-10:
        return None
    threshold = max_val - 0.02 * max_val
    active = np.where(corr >= threshold)[0]

    u = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        # dS/dh[j] = (1-h[j-lag]) * [0<=j-lag<n] - h[j+lag] * [0<=j+lag<n]
        idx = np.arange(n)
        src_mask = (idx - lag >= 0) & (idx - lag < n)
        tgt_mask = (idx + lag >= 0) & (idx + lag < n)
        u[src_mask] += (1.0 - h[idx[src_mask] - lag])
        u[tgt_mask] -= h[idx[tgt_mask] + lag]

    u = -u  # negative gradient direction
    u -= np.mean(u)
    nrm = np.linalg.norm(u)
    if nrm < 1e-12:
        return None
    return u / nrm

def gen_pair_dir(h, n, rng):
    """Structured two-group transfer direction."""
    split = rng.integers(1, n - 1)
    u = np.zeros(n)
    group_a = np.arange(split)
    group_b = np.arange(split, n)
    # Transfer mass from high-to-low within each group, considering the other group
    for i in group_a:
        u[i] = -(1.0 - h[i])
    for i in group_b:
        u[i] = h[i]
    u -= np.mean(u)
    nrm = np.linalg.norm(u)
    if nrm < 1e-12:
        return None
    # Randomize
    w = rng.standard_normal(n)
    u = 0.7 * u / nrm + 0.3 * w / (np.linalg.norm(w) + 1e-12)
    u -= np.mean(u)
    nrm = np.linalg.norm(u)
    if nrm < 1e-12:
        return None
    return u / nrm
