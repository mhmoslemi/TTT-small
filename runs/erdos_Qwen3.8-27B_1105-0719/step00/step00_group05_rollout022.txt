    idx = np.argmin([pair_evaluate(h, p, q, e) for e in grid])
    a = max(lo, grid[max(0, idx - 1)])
    b = min(hi, grid[min(len(grid) - 1, idx + 1)])
    if b - a < 1e-12:
        pass
    else:
        gr = 0.618033988749895
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = pair_evaluate(h, p, q, c1)
        f2 = pair_evaluate(h, p, q, c2)
        for _ in range(40):
            if f1 < f2:
                b = c2
                c2 = c1
                f2 = f1
                c1 = b - gr * (b - a)
                f1 = pair_evaluate(h, p, q, c1)
            else:
                a = c1
                c1 = c2
                f1 = f2
                c2 = a + gr * (b - a)
                f2 = pair_evaluate(h, p, q, c2)
        best_eps = 0.5 * (a + b)
        best_c = min(pair_evaluate(h, p, q, best_eps), best_c)
    if best_c < evaluate_exact(h) - 1e-15:
        h_new = h.copy()
        h_new[p] += best_eps
        h_new[q] -= best_eps
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new, best_c
    return h, best_c

def projected_descent(h, n, max_iters, deadline, beta, rng, best_h, best_c):
    h = project_feasible(h.copy(), n / 2.0)
    cur_c = evaluate_exact(h)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    step = 0.05
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        Q, grad = smooth_max_and_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        direction = -grad / gnorm
        found = False
        for _ in range(15):
            candidate = project_feasible(h + step * direction, n / 2.0)
            c_new = evaluate_exact(candidate)
            if c_new < cur_c - 1e-14:
                h = candidate
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                found = True
                break
            step *= 0.5
        if not found:
            step *= 0.5
            if step < 1e-10:
                break
        beta *= 1.02
        if beta > 5000:
            beta = 5000
    return best_h, best_c

def pair_search(h, n, max_rounds, deadline, rng, best_h, best_c):
    h = h.copy()
    cur_c = evaluate_exact(h)
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline - 10:
            break
        Q, grad = smooth_mag = smooth_max_and_grad(h, 200.0)
        gmag = np.abs(grad)
        order = np.argsort(-gmag)
        n_pairs = min(30, n // 2)
        pairs_done = set()
        improved_round = False
        for _ in range(n_pairs * 3):
            if time.time() > deadline - 10:
                break
            if len(order) < 2:
                break
            p = order[rng.randint(len(order))]
            q = order[rng.randint(len(order))]
            if p == q:
                continue
            key = (min(p, q), max(p, q))
            if key in pairs_done:
                continue
            pairs_done.add(key)
            h_new, c_new = try_pair(h, p, q, n)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                improved_round = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
        if not improved_round:
            for _ in range(20):
                if time.time() > deadline - 10:
                    break
                p = rng.randint(n)
                q = rng.randint(n)
                if p == q:
                    continue
                h_new, c_new = try_pair(h, p, q, n)
                if c_new < cur_c - 1e-15:
                    h = h_new
                    cur_c = c_new
                    improved_round = True
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
            if not improved_round:
                break
    return best_h, best_c

def interpolate_to_dim(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x0 = (np.arange(n) + 0.5) / n
    x1 = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x1, x0, h)
    return project_feasible(h_new, new_n / 2.0)

def shift_circular(h, s):
    n = len(h)
    return np.roll(h, s)

def add_noise(h, amplitude, rng):
    n = len(h)
    noise = rng.randn(n) * amplitude
    h_new = project_feasible(h + noise, n / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c = evaluate_exact(h0)

    if best_c <= 0.3808:
        return best_h, best_c, n0

    # Phase 1: Projected gradient descent at native dimension
    if time.time() < deadline - 60:
        h_work = h0.copy()
        best_h, best_c = projected_descent(h_work, n0, 2000, deadline, 80.0, rng, best_h, best_c)

    # Phase 2: Pair search at native dimension
    if time.time() < deadline - 60:
        best_h, best_c = pair_search(best_h, n0, 100, deadline, rng, best_h, best_c)

    # Phase 3: Additional gradient descent with different beta
    if time.time() < deadline - 60:
        h_work = best_h.copy()
        best_h, best_c = projected_descent(h_work, n0, 1500, deadline, 40.0, rng, best_h, best_c)

    # Phase 4: More pair search
    if time.time() < deadline - 60:
        best_h, best_c = pair_search(best_h, n0, 80, deadline, rng, best_h, best_c)

    # Phase 5: Try shifted versions
    if time.time() < deadline - 60:
        for s in [1, 2, 3, n0 // 3, n0 // 2]:
            if time.time() > deadline - 30:
                break
            h_shift = shift_circular(best_h, s)
            h_shift = project_feasible(h_shift, n0 / 2.0)
            c_shift = evaluate_exact(h_shift)
            if c_shift < best_c:
                best_c = c_shift
                best_h = h_shift.copy()
            if c_shift < best_c + 0.001 and time.time() < deadline - 30:
                best_h, best_c = projected_descent(h_shift, n0, 500, deadline, 100.0, rng, best_h, best_c)

    # Phase 6: Multi-dimension exploration
    dims_to_try = []
    for d in [n0 - 4, n0 + 4, n0 + 8, n0 - 8, n0 + 12]:
        if 16 <= d <= 80 and d != n0:
            dims_to_try.append(d)
    dims_to_try = sorted(set(dims_to_try))

    for dim in dims_to_try:
        if time.time() > deadline - 40:
            break
        h_dim = interpolate_to_dim(best_h, dim)
        c_dim = evaluate_exact(h_dim)
        if c_dim < best_c:
            best_c = c_dim
            best_h = h_dim.copy()
        if time.time() < deadline - 40:
            best_h, best_c = projected_descent(h_dim, dim, 800, deadline, 60.0, rng, best_h, best_c)
        if time.time() < deadline - 30:
            best_h, best_c = pair_search(best_h, dim, 40, deadline, rng, best_h, best_c)

    # Phase 7: Random restarts
    if time.time() < deadline - 40:
        for restart in range(5):
            if time.time() > deadline - 20:
                break
            h_rand = add_noise(best_h, 0.15, rng)
            c_rand = evaluate_exact(h_rand)
            if c_rand < best_c:
                best_c = c_rand
                best_h = h_rand.copy()
            if time.time() < deadline - 20:
                h_work = h_rand.copy()
                best_h, best_c = projected_descent(h_work, n0, 600, deadline, 50.0, rng, best_h, best_c)
                best_h, best_c = pair_search(best_h, n0, 30, deadline, rng, best_h, best_c)

    # Phase 8: Final intensive pair search on best
    if time.time() < deadline - 15:
        best_h, best_c = pair_search(best_h, n0, 60, deadline, rng, best_h, best_c)

    # Phase 9: Final polish
    if time.time() < deadline - 10:
        h_work = best_h.copy()
        best_h, best_c = projected_descent(h_work, n0, 500, deadline, 150.0, rng, best_h, best_c)
        best_h, best_c = pair_search(best_h, n0, 30, deadline, rng, best_h, best_c)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    residual = n_final / 2.0 - best_h.sum()
    if abs(residual) > 1e-12:
        best_h = project_feasible(best_h, n_final / 2.0)
    best_c = evaluate_exact(best_h)

    return best_h, best_c, n_final
