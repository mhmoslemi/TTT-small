        h0 = None

    if h0 is None or len(h0) < 2:
        n0 = 40
        h0 = np.full(n0, 0.5)
    else:
        h0 = project_feasible(h0, len(h0) / 2.0)
        n0 = len(h0)

    best_h = h0.copy()
    best_c5 = exact_c5(best_h)
    best_n = n0

    # Phase 1: Thorough optimization at parent resolution
    if time.time() < deadline - 60:
        best_h, best_c5 = coord_pass(best_h, best_c5, deadline, n_grid=15, do_golden=True, max_passes=300)
        best_n = len(best_h)

    # Phase 2: Fine polish at best resolution
    if time.time() < deadline - 40:
        best_h, best_c5 = fine_polish(best_h, best_c5, deadline, n_grid=40)
        best_n = len(best_h)

    # Phase 3: Try other resolutions
    candidate_res = [24, 30, 32, 36, 40, 45, 48, 50, 55, 60, 64, 72, 80, 100, 120]
    candidate_res = sorted(set(r for r in candidate_res if r != best_n and r >= 20),
                          key=lambda r: abs(r - best_n))

    for n_try in candidate_res:
        if time.time() > deadline - 80:
            break
        # Interpolate from best solution
        h_try = interp_to_res(best_h, n_try)
        c5_try = exact_c5(h_try)
        # Optimize at this resolution
        h_try, c5_try = coord_pass(h_try, c5_try, deadline, n_grid=12, do_golden=True, max_passes=200)
        # Polish
        if time.time() < deadline - 30:
            h_try, c5_try = fine_polish(h_try, c5_try, deadline, n_grid=30)
        if c5_try < best_c5 - 5e-6:
            best_h = h_try.copy()
            best_c5 = c5_try
            best_n = n_try

    # Phase 4: Random restarts with perturbation
    if time.time() < deadline - 40:
        for attempt in range(5):
            if time.time() > deadline - 20:
                break
            # Different perturbation sizes
            sigma = 0.03 + 0.02 * attempt
            h_r = best_h + rng.randn(len(best_h)) * sigma
            h_r = project_feasible(h_r, len(h_r) / 2.0)
            c5_r = exact_c5(h_r)
            # Quick optimize
            h_r, c5_r = coord_pass(h_r, c5_r, deadline, n_grid=10, do_golden=False, max_passes=100)
            if c5_r < best_c5 - 1e-6:
                # Do more optimization on this good find
                if time.time() < deadline - 25:
                    h_r, c5_r = fine_polish(h_r, c5_r, deadline, n_grid=35)
                if c5_r < best_c5 - 1e-6:
                    best_h = h_r.copy()
                    best_c5 = c5_r
                    best_n = len(h_r)

    # Phase 5: Try completely different random initializations at small n
    for n_rand in [24, 30, 36, 40]:
        if time.time() > deadline - 50:
            break
        for _ in range(2):
            if time.time() > deadline - 30:
                break
            h_rand = rng.rand(n_rand)
            h_rand = project_feasible(h_rand, n_rand / 2.0)
            c5_rand = exact_c5(h_rand)
            h_rand, c5_rand = coord_pass(h_rand, c5_rand, deadline, n_grid=15, do_golden=True, max_passes=150)
            if time.time() < deadline - 20:
                h_rand, c5_rand = fine_polish(h_rand, c5_rand, deadline, n_grid=35)
            if c5_rand < best_c5 - 5e-6:
                best_h = h_rand.copy()
                best_c5 = c5_rand
                best_n = n_rand

    # Phase 6: Final fine polish
    if time.time() < deadline - 10:
        best_h, best_c5 = fine_polish(best_h, best_c5, deadline, n_grid=50)
        best_n = len(best_h)

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = exact_c5(best_h)
    best_n = len(best_h)

    return (best_h, best_c5, best_n)
