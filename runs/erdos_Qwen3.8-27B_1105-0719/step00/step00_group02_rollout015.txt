                x = np.clip(x, 0.0, 1.0)
    return x

def repair_exact(h, s):
    h = np.clip(h.copy(), 0.0, 1.0)
    resid = s - h.sum()
    if abs(resid) > 1e-14:
        if resid > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            slack = (1.0 - h) if resid > 0 else h
            slack = slack[mask]
            if slack.sum() > 1e-15:
                h[mask] += resid * slack / slack.sum()
            else:
                idx = np.where(mask)[0]
                h[idx[0]] += resid
    return np.clip(h, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def compute_lse_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    R_max = corr.max()
    exp_vals = np.exp((corr - R_max) / tau)
    total = exp_vals.sum()
    if total < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(corr))] = 1.0
        return R_max + tau * np.log(1.0), w
    w = exp_vals / total
    lse = R_max + tau * np.log(total)
    return lse, w

def compute_smooth_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for idx in range(2*n - 1):
            ws = w[idx]
            if ws < 1e-18:
                continue
            s_val = idx - (n - 1)
            js = i - s_val
            if 0 <= js < n:
                s1 += ws * (1.0 - h[js])
            jp = i + s_val
            if 0 <= jp < n:
                s2 += ws * h[jp]
        grad[i] = dx * (s1 - s2)
    return grad

def polish_lse(h, s, c_best, tau_init, time_limit, max_iters=5000):
    """LSE-smoothed projected gradient descent."""
    n = len(h)
    t_start = time.perf_counter()
    h_cur = h.copy()
    c_cur = c_best
    tau = tau_init
    best_h = h_cur.copy()
    best_c = c_cur
    
    alpha = 0.05
    stale = 0
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > time_limit:
            break
        
        lse_val, w = compute_lse_weights(h_cur, tau)
        grad = compute_smooth_grad(h_cur, w)
        
        # Project gradient onto sum=0 subspace (tangent to constraint)
        grad_tan = grad - grad.mean()
        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-12:
            break
        
        # Try several step sizes
        improved = False
        for trial in range(8):
            alpha_t = alpha * (0.5 ** trial)
            if alpha_t < 1e-8:
                break
            h_try = project_box_sum(h_cur - alpha_t * grad_tan, s)
            h_try = repair_exact(h_try, s)
            c_try = eval_c5(h_try)
            if c_try < c_cur - 1e-12:
                h_cur = h_try
                c_cur = c_try
                alpha = min(alpha * 1.1, 0.5)
                improved = True
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h_cur.copy()
                break
            elif c_try < c_cur + tau * 0.5:
                # Accept as "smooth improvement" (secondary criterion)
                # Only if LSE decreased
                lse_new, _ = compute_lse_weights(h_try, tau)
                if lse_new < lse_val - 1e-10:
                    h_cur = h_try
                    c_cur = c_try
                    alpha = min(alpha * 1.05, 0.5)
                    improved = True
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
                    break
        
        if not improved:
            stale += 1
            alpha *= 0.7
            if stale > 30:
                tau *= 0.85
                stale = 0
                if tau < 1e-6:
                    break
        else:
            stale = max(0, stale - 1)
    
    return best_h, best_c

def polish_pairs(h, s, c_best, time_limit, max_passes=20):
    """Exact pair-transfer local search."""
    n = len(h)

                    h_cur = h_try
                    c_cur = c_try
                    any_improve = True
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
                    break
            
            # Try transfer j -> i
            for f in fractions:
                d = f * min(h_cur[j], 1.0 - h_cur[i])

                    continue
                h_try = h_cur.copy()
                h_try[j] -= d

def polish_sa(h, s, c_best, time_limit, temp_init=None):
    """Simulated annealing with pair transfers."""
    n = len(h)
    t_start = time.perf_counter()
    rng = np.random.default_rng(123)
    h_cur = h.copy()
    c_cur = c_best
    best_h = h_cur.copy()
    best_c = c_cur
    
    if temp_init is None:
        temp_init = c_best * 0.1
    temp = temp_init
    step_scale = 0.3
    
    no_improve_count = 0
    
    max_iters = 200000
    for it in range(max_iters):
        if time.perf_counter() - t_start > time_limit:
            break
        
        # Propose: pick random pair and transfer amount
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Random amount
        cap = min(h_cur[i], 1.0 - h_cur[j])
        if cap < 1e-8:
            # Try reverse
            cap = min(h_cur[j], 1.0 - h_cur[i])
            if cap < 1e-8:
                continue
            # Swap direction
            i, j = j, i
        
        d = rng.uniform(0.01, 1.0) * cap * step_scale
        if d < 1e-10:
            continue
        
        h_try = h_cur.copy()
        h_try[i] -= d
        h_try[j] += d
        c_try = eval_c5(h_try)
        diff = c_try - c_cur
        
        if diff < 0:
            # Improvement
            h_cur = h_try
            c_cur = c_try
            no_improve_count = 0
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            # Metropolis acceptance
            if temp > 1e-10:
                log_accept = -diff / temp
                if log_accept > 0 or np.log(rng.random()) < log_accept:
                    h_cur = h_try
                    c_cur = c_try
            no_improve_count += 1
        
        # Adapt temperature
        if it % 1000 == 0:
            if no_improve_count > 500:
                temp *= 0.9
            elif no_improve_count < 100:
                temp = min(temp * 1.1, temp_init)
    
    return best_h, best_c

def project_box_sum(y, s):
    n = len(y)
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        t = float(x.sum())
        if t > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    resid = s - float(x.sum())
    if abs(resid) > 1e-13:
        interior = (x > 1e-8) & (x < 1.0 - 1e-8)
        if interior.any():
            x[interior] += resid / float(interior.sum())
            x = np.clip(x, 0.0, 1.0)
    return x

def repair_exact(h, s):
    h = np.clip(h.copy(), 0.0, 1.0)
    resid = s - float(h.sum())
    if abs(resid) > 1e-14:
        if resid > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            slack = (1.0 - h) if resid > 0 else h
            slack = slack[mask]
            total_slack = float(slack.sum())
            if total_slack > 1e-15:
                h[mask] += resid * slack / total_slack
    return np.clip(h, 0.0, 1.0)

    n = len(h)

    R_max = float(corr.max())
    exp_vals = np.exp((corr - R_max) / tau)
    total = float(exp_vals.sum())

    return w
