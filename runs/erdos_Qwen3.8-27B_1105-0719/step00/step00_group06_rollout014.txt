import time

def exact_c5(h):
    """Compute the exact C5 value for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def logsumexp_objective(h, alpha):
    """Smooth approximation to max of overlaps."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    # Subtract max for numerical stability
    pmax = np.max(profile)
    if pmax < 1e-15:
        return 0.0
    shifted = profile - pmax
    lse = pmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return lse

def logsumexp_gradient(h, alpha):
    """Gradient of the log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlap values (unscaled)
    one_minus_h = 1.0 - h
    profile_raw = np.correlate(h, one_minus_h, mode='full')  # unscaled
    profile = profile_raw * dx
    
    # Softmax weights
    pmax = np.max(profile)
    if pmax < 1e-15:
        return np.zeros(n)
    shifted = alpha * (profile - pmax)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights = weights / wsum  # normalized weights
    
    # Compute gradient: sum_k weights[k] * d(overlap_k)/dh
    # profile[k] corresponds to lag (k - (n-1))
    # For lag k (0-indexed in profile = lag k-(n-1)):
    # We need d/dh[j] of overlap at that lag
    
    grad = np.zeros(n)
    for idx in range(len(profile_raw)):
        k = idx - (n - 1)  # actual lag
        if weights[idx] < 1e-15:
            continue
        w = weights[idx] * dx
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            # overlap_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j] = (1-h[j+k]) for 0<=j<=n-1-k, minus h[j-k] for k<=j<=n-1
            # First term: j from 0 to n-1-k
            if n - 1 - k >= 0:
                grad[0:n-k] += w * (1.0 - h[k:n])
            # Second term: j from k to n-1
            grad[k:n] -= w * h[0:n-k]
        else:

            a = -k
            # overlap = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i])

            if n - 1 - a >= 0:
                grad[0:n-a] += w * (1.0 - h[a:n])
                grad[a:n] -= w * h[0:n-a]

    """Project h onto [0,1]^n with sum = target."""
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    # Water-filling to achieve exact sum
    target = float(target)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to add: add to elements < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to subtract: subtract from elements > 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(h0, n, alpha_start, alpha_end, max_iter, budget_time, t_start):
    """Optimize h using SLSQP with log-sum-exp smoothing."""
    target = n / 2.0
    h = project_box_sum(h0.copy(), target)
    best_c5 = exact_c5(h)
    best_h = h.copy()
    
    n_constraints = [{"type": "eq", "fun": lambda x: np.sum(x) - target, "jac": lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return logsumexp_objective(x, alpha_current[0])
    
    def obj_grad(x):
        return logsumexp_gradient(x, alpha_current[0])
    
    alphas = np.geomspace(alpha_start, alpha_end, max_iter // 30 + 1)
    for alpha in alphas:
        if time.time() - t_start > budget_time:
            break
        alpha_current = [alpha]
        
        def obj_a(x):
            return logsumexp_objective(x, alpha_current[0])

            return logsumexp_gradient(x, alpha_current[0])
        
        try:

                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass

∂/∂h[j] = h[j-a] (if a ≤ j ≤ n-1, j appears as h[i+a]) + h[j] (if 0 ≤ j ≤ n-1-a, j appears as h[i])

So ∂overlap_{-a}/∂h[j] = 1_{a≤j≤n-1} - h[j-a]*1_{a≤j≤n-1} - h[j]*1_{0≤j≤n-1-a}
= (1-h[j-a])*1_{a≤j≤n-1} - h[j]*1_{0≤j≤n-1-a}

OK so for the full gradient:
∂F/∂h[j] = dx * sum_k w_k * ∂overlap_k/∂h[j]

where w_k are the softmax weights.

Let me code this up properly. The profile array from np.correlate has index j corresponding to lag k = j - (n-1).

For profile index idx, lag k = idx - (n-1):
- k = 0 (idx = n-1): ∂/∂h[j] = 1 - 2h[j]
- k > 0 (idx > n-1): ∂/∂h[j] = (1-h[j+k]) for j < n-k, minus h[j-k] for j ≥ k
- k < 0 (idx < n-1), a = -k: ∂/∂h[j] = (1-h[j-a]) for j ≥ a, minus h[j] for j < n-a
