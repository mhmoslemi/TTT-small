```python
def binary_sa(h0, n_rounds=100000, start_temp=1.0, end_temp=0.01):

    # Fix if needed

        np.random.shuffle(idx)
        h[idx[:int(target-ones)]] = 1

    best_max = np.max(best_corr)

    current_max = best_max

        temp = start_temp * (end_temp/start_temp) ** (round_num / n_rounds)

        a = np.random.randint(n)
        b = np.random.randint(n)
        if a == b or h[a] == h[b]:
            continue
        
        # Compute change in correlation
        # Swap h[a] 1->0, h[b] 0->1 (or vice versa)
        da = -1 if h[a] > h[b] else 1
        db = -da
        
        # Recompute correlation (fast for small n)
        h[a] += da
        h[b] += db
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = np.max(new_corr)

            current_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                best_corr = new_corr.copy()
        else:
            h[a] -= da
            h[b] -= db
    
    return best_h, (2.0/n) * best_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_corr(h):
    """Compute the full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def logsumexp_max(corr, beta):
    """Smooth approximation to max(corr)."""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_objective(h, target, beta):
    """Smooth objective with constraint penalty."""
    corr = compute_corr(h)
    obj = logsumexp_max(corr, beta)
    # Penalty for constraint violation
    pen = 100.0 * (np.sum(h) - target) ** 2
    return obj + pen

def smooth_gradient(h, target, beta):
    """Gradient of smooth objective."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        k = t - (n - 1)
        wt = w[t]
        if wt < 1e-12:
            continue
        # First term: (1-h[i-k]) for valid i
        if k >= 0:
            i_start, i_end = k, n - 1
        else:
            i_start, i_end = 0, n - 1 + k
        if i_start <= i_end:
            grad[i_start:i_end+1] += wt * (1.0 - h[i_start - k : i_end - k + 1])
        
        # Second term: -h[i+k] for valid i
        if k >= 0:
            i_start2, i_end2 = 0, n - 1 - k
        else:
            i_start2, i_end2 = -k, n - 1
        if i_start2 <= i_end2:
            grad[i_start2:i_end2+1] += -wt * h[i_start2 + k : i_end2 + k + 1]
    
    # Constraint gradient
    grad += 200.0 * (np.sum(h) - target) / n
    return grad

def local_optimize(h0, target, max_iter=5000, beta_init=20.0, beta_final=100.0):
    """Local optimization with increasing beta."""
    h = h0.astype(float).copy()
    n = len(h)
    
    # Anneal beta
    n_anneal = 5
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for stage in range(n_anneal):
        beta = beta_init * (beta_final / beta_init) ** (stage / max(n_anneal - 1, 1))
        res = minimize(
            smooth_objective, h, args=(target, beta),
            jac=smooth_gradient,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter // n_anneal, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        h = np.clip(res.x, 0.0, 1.0)
        # Fix sum
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

def binary_hill_climb(h0, target, n_iters=200000, rnd=None):
    """Hill climbing on binary (or near-binary) sequences."""
    if rnd is None:
        rnd = np.random
    n = len(h0)
    h = h0.astype(float).copy()
    
    # Make it (nearly) binary
    # Round, then fix sum
    h_bin = (h >= 0.5).astype(float)
    ones = int(np.sum(h_bin))
    if ones > target:
        idx_ones = np.where(h_bin == 1)[0]
        idx_zeros = np.where(h_bin == 0)[0]
        need_remove = ones - target
        # Remove from positions that are most "willing" to be 0
        r = h[idx_ones]
        order = np.argsort(r)[:need_remove]
        h_bin[idx_ones[order]] = 0
    elif ones < target:
        idx_zeros = np.where(h_bin == 0)[0]
        need_add = target - ones
        r = h[idx_zeros]
        order = np.argsort(r, reverse=True)[:need_add]
        h_bin[idx_zeros[order]] = 1
    
    h = h_bin.copy()
    corr = compute_corr(h)
    curr_max = np.max(corr)
    best_h = h.copy()
    best_max = curr_max
    
    for it in range(n_iters):
        # Random swap
        a = rnd.randint(n)
        b = rnd.randint(n)
        if a == b or h[a] == h[b]:
            continue
        
        # Try swap
        h[a], h[b] = h[b], h[a]
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        
        if new_max <= curr_max:
            curr_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Revert
            h[a], h[b] = h[b], h[a]
    
    return best_h, (2.0 / n) * best_max

    n = len(h0)

    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        perm = rnd.permutation(n - ones)
        h_bin[idx[perm[:target-ones]]] = 1
    
    h = h_bin.copy()
    corr = compute_corr(h)
    curr_max = np.max(corr)
    best_h = h.copy()
    best_max = curr_max
    
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / n_iters)
        a = rnd.randint(n)
        b = rnd.randint(n)
        if a == b or h[a] == h[b]:
            continue
        
        h[a], h[b] = h[b], h[a]
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        delta = new_max - curr_max
        
        if delta <= 0 or rnd.random() < np.exp(-delta / max(t, 1e-10)):
            curr_max = new_max
            corr = new_corr

        perm = rnd.permutation(n - ones)
        h[idx[perm[:target-ones]]] = 1
    
    return h
```

Here's another idea: coordinate descent. For each coordinate i, find the optimal value of h[i] given all other coordinates fixed. This is a 1D optimization problem (though non-convex in general).

For a fixed h[-i] (all coordinates except i), the objective is:
max_t R_t(h) where R_t depends linearly on h[i] (for each t, R_t is linear in h[i]).

So we're looking at max of linear functions of h[i], which is a piecewise-linear convex function in h[i]. The minimum of a convex function over [0,1] is either at a breakpoint (where two R_t values are equal) or at a boundary.

The breakpoints occur where R_t(h) = R_s(h) for some t ≠ s, as h[i] varies.

This is actually a nice structure! For each coordinate, I can:
1. Compute all R_t as functions of h[i] (linear)
2. Find the breakpoints
3. Evaluate the max at each breakpoint and at 0, 1
4. Choose the value that minimizes the max

This is O(n) breakpoints per coordinate (at most), and O(n) coordinates, so O(n²) per full sweep. For n=95, that's ~9000 operations per sweep. Very fast!

```python
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(h, target, beta):
    """Returns (objective, gradient) for smooth approximation."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = np.sum(w)
    w /= w_sum
    obj = m + np.log(w_sum) / beta
    
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        k = t - (n - 1)
        wt = w[t]
        if wt < 1e-14:
            continue
        # d R_t / d h[i] for first index
        if k >= 0:
            i_lo, i_hi = k, n - 1
        else:
            i_lo, i_hi = 0, n - 1 + k
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += wt * (1.0 - h[i_lo - k : i_hi - k + 1])
        # d R_t / d h[i] for second index
        if k >= 0:
            i_lo2, i_hi2 = 0, n - 1 - k
        else:
            i_lo2, i_hi2 = -k, n - 1
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] += -wt * h[i_lo2 + k : i_hi2 + k + 1]
    
    # Add sum constraint as penalty
    pen = 500.0 * (np.sum(h) - target)
    obj += 0.5 * pen * (np.sum(h) - target)
    grad += pen
    
    return obj, grad

def local_optimize(h0, target, time_limit=60.0):
    """Multi-stage local optimization with increasing beta."""
    h = h0.astype(float).copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    t_start = time.time()
    
    betas = [5, 15, 40, 100, 200]
    for beta in betas:
        if time.time() - t_start > time_limit:
            break
        
        def obj(h):
            o, _ = smooth_obj_grad(h, target, beta)
            return o
        
        def grad(h):
            _, g = smooth_obj_grad(h, target, beta)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1)] * n,
                          options={'maxiter': 1000, 'ftol': 1e-14, 'gtol': 1e-12})
            h = np.clip(res.x, 0, 1)
            # Fix sum
            diff = target - np.sum(h)
            h = h + diff / n
            h = np.clip(h, 0, 1)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except:
            break
    
    return best_h, best_c5

def binary_sa(h_init, n, target, iters, t0, t1, rng):
    """Simulated annealing for binary sequences."""
    h = (h_init >= 0.5).astype(np.float64)
    # Fix count
    ones = int(h.sum())
    if ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:ones - target]]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:target - ones]]] = 1
    
    corr = compute_corr(h)
    curr = corr.max()
    best = curr
    best_h = h.copy()
    ones_idx = np.where(h == 1)[0].copy()
    zeros_idx = np.where(h == 0)[0].copy()
    
    for i in range(iters):
        t = t0 * (t1 / t0) ** (i / max(iters - 1, 1))
        # Pick random 1 and 0 to swap
        a = ones_idx[rng.integers(len(ones_idx))]
        b = zeros_idx[rng.integers(len(zeros_idx))]
        
        h[a], h[b] = 1 - h[a], 1 - h[b]
        new_corr = compute_corr(h)
        new_val = new_corr.max()
        delta = new_val - curr
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            curr = new_val
            corr = new_corr
            # Update index sets

            zeros_idx = np.where(h == 0)[0]
            if new_val < best:
                best = new_val
                best_h = h.copy()
        else:
            h[a], h[b] = 1 - h[a], 1 - h[b]
    
    return best_h, 2.0 / n * best
```
