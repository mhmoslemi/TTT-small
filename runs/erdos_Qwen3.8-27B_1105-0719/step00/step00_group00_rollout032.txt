```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def gradient_optimize(h0, dx, time_limit=30):
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    
    for alpha in [20, 50, 100, 200]:
        if time.time() - t_start > time_limit:
            break
        
        def obj(x):
            corr = np.correlate(x, 1-x, mode="full")
            return smooth_max(corr * dx, alpha)
        
        def obj_grad(x):
            corr = np.correlate(x, 1-x, mode="full")
            vals = corr * dx
            m = np.max(vals)
            weights = np.exp(alpha * (vals - m))
            weights /= np.sum(weights)
            
            # Compute gradient
            grad = np.zeros(n)
            for k in range(2*n-1):
                lag = k - (n-1)
                w = weights[k] * dx
                if lag >= 0:
                    # overlap = sum_{i=0}^{n-1-lag} x[i] * (1-x[i+lag])
                    # d/dx[j]: (1-x[j+lag]) for j < n-lag, minus x[j-lag] for j >= lag
                    if n - 1 - lag >= 0:
                        grad[:n-lag] += w * (1 - x[lag:])  # (1-x[i+lag]) for i=0..n-1-lag
                        grad[lag:] -= w * x[:n-lag]  # -x[i] for i=0..n-1-lag
                else:
                    lag_abs = -lag
                    # overlap = sum_{i=lag_abs}^{n-1} x[i] * (1-x[i-lag_abs])
                    # d/dx[j]: (1-x[j-lag_abs]) for j >= lag_abs, minus x[j+lag_abs] for j < n-lag_abs
                    grad[lag_abs:] += w * (1 - x[:n-lag_abs])
                    grad[:n-lag_abs] -= w * x[lag_abs:]
            return grad
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
        bounds = [(0.0, 1.0)] * n
        
        result = minimize(obj, h, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-12})
        h = np.clip(result.x, 0, 1)
        # Fix sum
        s = h.sum()
        if s > 0:
            h = h / s * (n/2.0)
            h = np.clip(h, 0, 1)
    
    return h
```
