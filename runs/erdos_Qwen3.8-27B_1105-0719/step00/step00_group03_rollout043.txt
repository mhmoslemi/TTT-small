import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def optimize_pair(h, p, q, n_samples=100):
    """Find optimal tau for pair (p, q). Returns (best_tau, best_c5)."""
    n = len(h)
    dx = 2.0 / n
    
    # Current correlations
    C = np.correlate(h, 1.0 - h, mode="full")
    
    # Compute A[lag] for each lag
    # lag ranges from -(n-1) to n-1
    lags = np.arange(-(n-1), n)  # length 2n-1
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    for i, lag in enumerate(lags):
        a_val = 0.0
        if 0 <= q - lag < n:
            a_val += (1.0 - h[q - lag])
        if 0 <= p - lag < n:
            a_val -= (1.0 - h[p - lag])
        if 0 <= p + lag < n:
            a_val += h[p + lag]
        if 0 <= q + lag < n:
            a_val -= h[q + lag]
        A[i] = a_val
        
        if lag == 0:
            B[i] = -2.0
        elif lag == q - p or lag == p - q:
            B[i] = 1.0
    
    # Feasible range for tau
    tau_min = -h[p]  # h[p] - tau >= 0 → tau <= h[p]... wait
    # h[p] new = h[p] - tau >= 0 → tau <= h[p]
    # h[q] new = h[q] + tau >= 0 → tau >= -h[q]
    # h[p] - tau <= 1 → tau >= h[p] - 1 (usually not binding since h[p] <= 1)
    # h[q] + tau <= 1 → tau <= 1 - h[q]
    tau_lo = max(-h[q], h[p] - 1)  # usually -h[q]
    tau_hi = min(h[p], 1 - h[q])
    
    if tau_lo >= tau_hi:
        return 0, None
    
    # Sample tau values
    taus = np.linspace(tau_lo, tau_hi, n_samples)
    
    # For each tau, compute max(C + tau*A + B*tau^2)
    # Vectorize: (n_samples, 2n-1) matrix
    tau_col = taus[:, np.newaxis]  # (n_samples, 1)
    vals = C[np.newaxis, :] + tau_col * A[np.newaxis, :] + B[np.newaxis, :] * tau_col**2
    f_vals = np.max(vals, axis=1) * dx
    
    best_idx = np.argmin(f_vals)
    best_tau = taus[best_idx]
    best_c5 = f_vals[best_idx]
    
    return best_tau, best_c5

    idx = p - lags
    m = (idx >= 0) & (idx < n)

    idx = p + lags
    m = (idx >= 0) & (idx < n)

    m = (idx >= 0) & (idx < n)

    B = np.zeros(2*n-1)
    B[n-1] = -2.0
    d = q - p
    r1 = d + (n-1)
    r2 = -d + (n-1)
    if 0 <= r1 < 2*n-1: B[r1] = 1.0
    if 0 <= r2 < 2*n-1: B[r2] = 1.0

        return 0.0, float(np.max(C_raw))
    
    # Coarse search
    taus = np.linspace(lo, hi, n_samp_coarse)
    t = taus[:, None]
    vals = C_raw[None, :] + t * A[None, :] + (t*t) * B[None, :]
    fmax = vals.max(axis=1)
    i_coarse = int(fmax.argmin())
    
    # Fine search around best coarse
    tau_center = taus[i_coarse]
    half_width = (hi - lo) / n_samp_coarse
    fine_lo = max(lo, tau_center - half_width)
    fine_hi = min(hi, tau_center + half_width)
    if fine_hi - fine_lo > 1e-14:
        taus_fine = np.linspace(fine_lo, fine_hi, n_samp_fine)
        t = taus_fine[:, None]
        vals = C_raw[None, :] + t * A[None, :] + (t*t) * B[None, :]
        fmax_fine = vals.max(axis=1)
        i_fine = int(fmax_fine.argmin())
        best_tau = taus_fine[i_fine]
        best_val = fmax_fine[i_fine]
    else:
        best_tau = tau_center
        best_val = fmax[i_coarse]
    
    return best_tau, float(best_val)

def optimize(h_in, t0, budget_s, rng, max_sweeps=200, n_samp=200, max_pairs_per_sweep=None):
    n = len(h_in)
    h = h_in.copy()
    C = get_C_raw(h)
    c5 = float(np.max(C)) * 2.0 / n
    dx = 2.0 / n
    
    total_pairs = n * (n - 1) // 2
    if max_pairs_per_sweep is None:
        max_pairs_per_sweep = total_pairs
    n_pairs = min(total_pairs, max_pairs_per_sweep)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        
        if n_pairs >= total_pairs:
            pairs_list = [(i, j) for i in range(n) for j in range(i+1, n)]
            rng.shuffle(pairs_list)
        else:
            pairs_list = []
            attempts = 0
            while len(pairs_list) < n_pairs and attempts < n_pairs * 4:
                p = rng.randint(n)
                q = rng.randint(n)
                if p != q:
                    pairs_list.append((p, q))
                attempts += 1
        
        for p, q in pairs_list:
            if time.time() - t0 > budget_s - 10:
                break
            tau, new_max = pair_best_tau(h, p, q, C, n_samp)
            new_c5 = new_max * dx
            if new_c5 < c5 - 1e-12 and abs(tau) > 1e-14:
                h[p] -= tau
                h[q] += tau
                np.clip(h, 0.0, 1.0, out=h)
                c5 = new_c5
                C = get_C_raw(h)
                improved = True
        
        if not improved:
            break
    
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h0 = np.full(45, 1.0/45)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    best_n = len(h0)
    
    # Phase 1: Optimize at initial n with full pair search
    h1, c5_1 = optimize(h0, t0, budget_s, rng, max_sweeps=500, n_samp=250)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c5_1, len(h1)
    
    # Phase 2: Try larger n values
    n_targets = [48, 50, 54, 60, 64, 72, 80, 90, 100, 112, 120, 144, 160, 180, 200, 240, 288, 320, 400, 500]
    for n in n_targets:
        if time.time() - t0 > budget_s - 40:
            break
        if n <= best_n and n > 46:
            continue
        
        # Interpolate from best
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = fix_sum(h_interp, n / 2.0)
        
        c5_int = evaluate_c5(h_interp)
        if c5_int < best_c5:
            best_h, best_c5, best_n = h_interp.copy(), c5_int, n
        
        # Determine optimization settings based on n
        if n <= 64:
            max_sw = 300
            n_samp_use = 200
            max_pairs_use = None  # all pairs
        elif n <= 128:
            max_sw = 200
            n_samp_use = 150
            max_pairs_use = 15000
        elif n <= 256:
            max_sw = 100
            n_samp_use = 100
            max_pairs_use = 10000
        else:
            max_sw = 50
            n_samp_use = 80
            max_pairs_use = 6000
        
        h_opt, c5_opt = optimize(h_interp, t0, budget_s, rng,
                                  max_sweeps=max_sw, n_samp=n_samp_use,
                                  max_pairs_per_sweep=max_pairs_use)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
    
    # Phase 3: Random restarts at best_n
    if time.time() - t0 < budget_s - 30:
        for trial in range(50):
            if time.time() - t0 > budget_s - 15:
                break
            n = best_n
            h_pert = best_h.copy()
            n_noise = max(3, n // 6)
            for _ in range(n_noise):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    tau = rng.uniform(-0.12, 0.12)
                    h_pert[i] -= tau
                    h_pert[j] += tau
            np.clip(h_pert, 0.0, 1.0, out=h_pert)
            h_pert = fix_sum(h_pert, n / 2.0)
            
            if n <= 64:
                h_opt, c5_opt = optimize(h_pert, t0, budget_s, rng,
                                          max_sweeps=150, n_samp=200)
            else:
                h_opt, c5_opt = optimize(h_pert, t0, budget_s, rng,
                                          max_sweeps=80, n_samp=100,
                                          max_pairs_per_sweep=8000)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
    
    # Phase 4: Final polish at best_n with more sweeps
    if time.time() - t0 < budget_s - 15:
        h_final, c5_final = optimize(best_h, t0, budget_s, rng,
                                      max_sweeps=200, n_samp=250)
        if c5_final < best_c5:
            best_h, best_c5, best_n = h_final.copy(), c5_final, len(h_final)
    
    return best_h, best_c5, best_n
