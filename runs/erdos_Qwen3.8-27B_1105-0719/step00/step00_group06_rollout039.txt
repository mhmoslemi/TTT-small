def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode='full') * dx
    return result.max(), result

def project_feasible(h, target_sum):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    diff = target_sum - current_sum
    if abs(diff) > 1e-12:
        # Distribute evenly
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Iterate to fix
        for _ in range(100):
            current_sum = h.sum()
            diff = target_sum - current_sum
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                # Need to increase; add to elements < 1
                mask = h < 1.0
                if mask.sum() == 0:
                    break
                add = diff / mask.sum()
                h[mask] = np.minimum(h[mask] + add, 1.0)
            else:
                # Need to decrease; subtract from elements > 0
                mask = h > 0.0
                if mask.sum() == 0:
                    break
                sub = -diff / mask.sum()
                h[mask] = np.maximum(h[mask] - sub, 0.0)
    # Final exact fix
    current_sum = h.sum()
    residual = target_sum - current_sum
    if abs(residual) > 1e-15:
        # Add residual to one element
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lag_values(h):
    """Compute all lag values of the correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr  # length 2n-1, centered at index n-1

def compute_gradients(h, dx):
    """Compute gradients of all lag functions.
    
    Returns array of shape (2n-1, n) where row i is gradient of lag (i - (n-1)).
    """
    n = len(h)
    grad = np.zeros((2*n - 1, n))
    
    # Lag 0 (at index n-1): C_0 = dx * sum h[i]*(1-h[i])
    # dC_0/dh[j] = dx * (1 - 2*h[j])
    grad[n-1] = dx * (1.0 - 2.0 * h)
    
    # Positive lags: s = 1, ..., n-1 (at indices n-1+s)
    # C_s = dx * sum_{i=0}^{n-s-1} h[i+s] * (1-h[i])
    for s in range(1, n):
        idx = n - 1 + s
        # dC_s/dh[j] = dx * [(1-h[j-s]) if j >= s] - dx * [h[j+s] if j <= n-s-1]
        for j in range(s, n):
            grad[idx, j] += dx * (1.0 - h[j-s])
        for j in range(0, n - s):
            grad[idx, j] -= dx * h[j + s]
    
    # Negative lags: s = 1, ..., n-1 (at indices n-1-s)
    # C_{-s} = dx * sum_{i=0}^{n-s-1} h[i] * (1-h[i+s])
    for s in range(1, n):
        idx = n - 1 - s
        # dC_{-s}/dh[j] = dx * [(1-h[j+s]) if j <= n-s-1] - dx * [h[j-s] if j >= s]
        for j in range(0, n - s):
            grad[idx, j] += dx * (1.0 - h[j + s])
        for j in range(s, n):
            grad[idx, j] -= dx * h[j - s]
    
    return grad

from scipy.optimize import linprog
import time


def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            if not m.any():
                break
            h[m] = np.minimum(h[m] + d / m.sum(), 1.0)
        else:
            m = h > 0.0
            if not m.any():
                break
            h[m] = np.maximum(h[m] + d / m.sum(), 0.0)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        h[np.argmin(np.abs(h - 0.5))] += r
    return h


def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1 - 2 * h)
    for s in range(1, n):
        G[n - 1 + s, s:] += dx * (1 - h[:n - s])
        G[n - 1 + s, :n - s] -= dx * h[s:]
        G[n - 1 - s, :n - s] += dx * (1 - h[s:])
        G[n - 1 - s, s:] -= dx * h[:n - s]
    return G


def _lp_dir(h, G, corr, budget, reg=1e-4):
    n = len(h)
    nv = 2 * n + 1
    c = np.concatenate([np.full(n, reg), np.full(n, reg), [1.0]])
    nl = 2 * n - 1
    A = np.zeros((nl + 1, nv))
    b = np.zeros(nl + 1)
    A[:nl, :n] = G
    A[:nl, n:2 * n] = -G
    A[:nl, -1] = -1.0
    b[:nl] = -corr
    A[nl, :2 * n] = 1.0
    b[nl] = budget
    Aeq = np.zeros((1, nv))
    Aeq[0, :n] = 1.0
    Aeq[0, n:2 * n] = -1.0
    bnd = []
    for j in range(n):
        bnd.append((0.0, 1.0 - h[j]))
    for j in range(n):
        bnd.append((0.0, h[j]))
    bnd.append((0.0, None))
    res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0],
                  bounds=bnd, method='highs')
    if res.status == 0:
        v = res.x[:n] - res.x[n:2 * n]
        if np.abs(v).max() > 1e-9:
            return v
    return None


def _descend(h, n, tlim, sd):
    rng = np.random.default_rng(sd)

    best = _eval(h)
    hb = h.copy()

    while time.time() - t0 < tlim:

        c5 = corr.max()
        if c5 < best - 1e-11:

        else:

        G = _grad(h)

def _binsearch(h, n, tlim):
    t0 = time.time()
    tgt = n // 2
    thr = np.sort(h)[::-1][tgt - 1] if tgt < n else 0.0
    b = (h >= thr).astype(float)
    if b.sum() > tgt:

        # Keep the ones with highest h values

    best = _eval(b)

        if bp:

            improved = True
        if not improved:

    return b, _eval(b)
