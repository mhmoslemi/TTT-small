                grad[s:] += dx * (1.0 - h[:n-s])  # from h[t+s] term
                grad[:n-s] -= dx * h[s:]  # from (1-h[t]) term

                # c = sum_{t=0}^{n-1-s2} h[t]*(1-h[t+s2])
                grad[:n-s2] += dx * (1.0 - h[s2:])  # from h[t] term

    src_centers = (np.arange(n_from) + 0.5) * (2.0 / n_from)

        src_hi = int(np.ceil(hi / src_width)) - 1
        src_lo = max(0, src_lo)
        src_hi = min(n_from - 1, src_hi)

        for i in range(src_lo, src_hi + 1):

            src_bin_hi = (i + 1) * src_width

                b[i] = 0.0
                b[j] = 1.0
                new_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
                delta = new_max - curr_max
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    best_new_max = new_max
                b[i] = bi
                b[j] = 0.0

        if best_swap is not None and best_delta < -1e-12:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_max = best_new_max
            best_val = curr_max * dx
            best_b = b.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                T = 1.0 / (1.0 + 0.05 * stall)
                for _ in range(3):
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
                    b[i] = 0.0
                    b[j] = 1.0
                    new_val = float(np.max(np.correlate(b, 1.0 - b, mode='full'))) * dx
                    delta_val = new_val - best_val
                    if delta_val < 0 or np.random.random() < np.exp(-delta_val / (T * dx)):
                        if new_val < best_val:
                            best_val = new_val
                            best_b = b.copy()
                        curr_max = np.max(np.correlate(b, 1.0 - b, mode='full'))
                        stall = 0
                        break
                    b[i] = 1.0
                    b[j] = 0.0
    return best_b, best_val

def continuous_refine(h0, deadline, rng):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(np.float64)
    best_val = eval_c5(h)
    best_h = h.copy()
    lr = 0.5
    no_improve = 0

    while time.time() < deadline - 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(np.max(corr))
        threshold = max_val * 0.92
        active_idx = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for k in active_idx:
            s = int(k) - (n - 1)
            if s == 0:
                grad += dx * (1.0 - 2.0 * h)
            elif s > 0:
                grad[s:] += dx * (1.0 - h[:n - s])
                grad[:n - s] -= dx * h[s:]
            else:
                s2 = -s
                grad[:n - s2] += dx * (1.0 - h[s2:])
                grad[s2:] -= dx * h[:n - s2]
        grad /= max(1, len(active_idx))

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            lr *= 0.5
            no_improve += 1
            if no_improve > 50:
                break
            continue

        h_new = h - lr * grad
        h_new = project_box_sum(h_new, n)
        new_val = eval_c5(h_new)

        if new_val < best_val - 1e-11:
            best_val = new_val
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            lr *= 0.7
            no_improve += 1
            if no_improve > 80:
                if time.time() > deadline - 1.0:
                    break
                h = h + rng.normal(0, 0.03, n)
                h = project_box_sum(h, n)
                lr = 0.2
                no_improve = 0

    return best_h, best_val

def interpolate_to_length(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    src_width = 2.0 / n_from
    tgt_width = 2.0 / n_to
    h_out = np.zeros(n_to)
    for j in range(n_to):
        lo = j * tgt_width
        hi = (j + 1) * tgt_width
        src_lo = max(0, int(lo / src_width))
        src_hi = min(n_from - 1, int(np.ceil(hi / src_width)) - 1)
        total_w = 0.0
        wval = 0.0
        for i in range(src_lo, src_hi + 1):
            s_lo = max(lo, i * src_width)
            s_hi = min(hi, (i + 1) * src_width)
            if s_hi > s_lo:
                w = s_hi - s_lo
                wval += w * h[i]
                total_w += w
        h_out[j] = wval / total_w if total_w > 0 else h[src_lo]
    return project_box_sum(h_out, n_to)

def structured_init(n, rng):
    b = np.zeros(n, dtype=np.float64)
    m = n // 2
    step = n / m
    for i in range(m):
        pos = int(round(i * step + rng.random() * 0.5)) % n
        b[pos] = 1.0
    # Fix if duplicates
    if b.sum() != m:
        b = np.zeros(n, dtype=np.float64)
        b[rng.choice(n, m, replace=False)] = 1.0
    return b
