def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_to_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)

def _lse_obj(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    sum_exps = float(np.sum(exps))
    lse_val = m + np.log(sum_exps) / beta
    obj = lse_val * dx
    if penalty > 0:
        target = n / 2.0
        sum_h = float(np.sum(h))
        obj += penalty * (sum_h - target) ** 2
    return obj
