def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def compute_c5_vector(h):
    """Returns the vector of correlation values * dx (before taking max)"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def pair_exchange(h, p, q):
    """Find optimal τ for moving mass from p to q.
    h[p] -= τ, h[q] += τ
    Returns (best_tau, best_c5)
    """
    n = len(h)
    dx = 2.0 / n
    c = compute_c5_vector(h)  # shape (2n-1,)
    
    # Feasible range for τ: -h[q] ≤ τ ≤ min(h[p], 1-h[q])
    tau_min = -h[q]
    tau_max = min(h[p], 1.0 - h[q])
    if tau_min >= tau_max:
        return None, compute_c5(h)
    
    # Compute a[k] for each lag k (indexed 0 to 2n-2 in the full correlation)
    # The full correlation has index r = k + (n-1) where k is the lag
    # So lag k = r - (n-1)
    # For lag k: Δc[k] = -τ*(1-f[p+k]) (if 0≤p+k<n) + τ*(1-f[q+k]) (if 0≤q+k<n) 
    #                + τ*f[p-k] (if 0≤p-k<n) - τ*f[q-k] (if 0≤q-k<n)
    
    a = np.zeros(2*n - 1)
    for r in range(2*n - 1):
        k = r - (n - 1)  # lag
        val = 0.0
        if 0 <= p + k < n:
            val -= (1.0 - h[p + k])
        if 0 <= q + k < n:
            val += (1.0 - h[q + k])
        if 0 <= p - k < n:
            val += h[p - k]
        if 0 <= q - k < n:
            val -= h[q - k]
        a[r] = val
    
    # Now minimize max_r (c[r] + τ*a[r]) over τ ∈ [tau_min, tau_max]
    # Find all candidate τ values
    candidates = [tau_min, tau_max]
    
    # Intersection points
    for i in range(2*n - 1):
        for j in range(i + 1, 2*n - 1):
            if a[i] != a[j]:
                tau = (c[j] - c[i]) / (a[i] - a[j])
                if tau_min <= tau <= tau_max:
                    candidates.append(tau)
    
    # Also add the current max points
    max_r = np.argmax(c)
    # τ where c[max_r] + τ*a[max_r] = c[r] for each r
    # Already covered by intersection points above
    
    best_obj = np.inf
    best_tau = 0.0
    for tau in candidates:

        if obj < best_obj:

            best_tau = tau

import numpy as np
import time


def compute_corr_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_value(h):
    return np.max(compute_corr_vector(h))


def compute_a_vec(h, p, q, n):
    k = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    idx1 = p + k
    mask1 = (idx1 >= 0) & (idx1 < n)
    a[mask1] -= (1.0 - h[idx1[mask1]])
    idx2 = q + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    a[mask2] += (1.0 - h[idx2[mask2]])
    idx3 = p - k
    mask3 = (idx3 >= 0) & (idx3 < n)
    a[mask3] += h[idx3[mask3]]
    idx4 = q - k
    mask4 = (idx4 >= 0) & (idx4 < n)
    a[mask4] -= h[idx4[mask4]]
    return a


def find_optimal_tau(c, a, tau_min, tau_max):
    if tau_min > tau_max - 1e-15:
        return 0.0, np.max(c)
    best_obj = np.max(c + tau_min * a)
    best_tau = tau_min
    obj_hi = np.max(c + tau_max * a)
    if obj_hi < best_obj:
        best_obj = obj_hi
        best_tau = tau_max
    lo, hi = tau_min, tau_max
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = c + mid * a
        mx = vals.max()
        if mx < best_obj:
            best_obj = mx
            best_tau = mid
        active = vals >= mx - 1e-12
        a_min = a[active].min()
        a_max = a[active].max()
        if a_min <= 0 and a_max >= 0:
            break
        elif a_max <= 0:
            lo = mid
        else:
            hi = mid
    return best_tau, best_obj
