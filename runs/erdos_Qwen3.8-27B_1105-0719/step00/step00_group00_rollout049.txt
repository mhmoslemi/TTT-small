```python
def subgradient_step(h, n, step_size, active_threshold=0.05):
    """One subgradient step for minimizing max_k corr[k]."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    
    # Find active lags
    active = corr > cmax - active_threshold * cmax
    
    # Compute subgradient (average over active lags)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if active[k]:
            lag = k - (n - 1)
            for j in range(n):
                g = 0.0
                jk = j - lag
                if 0 <= jk < n:
                    g += (1.0 - h[jk])
                jl = j + lag
                if 0 <= jl < n:
                    g -= h[jl]
                grad[j] += g
    grad /= np.sum(active)
    
    # Step
    h_new = h - step_size * grad
    
    # Project to feasible set
    h_new = project_feasible(h_new, n)
    return h_new, cmax
```

Actually, I just realized I should also try the approach of using the analytical gradient I derived earlier. That would make SLSQP much faster.

Let me implement the analytical gradient of the LSE objective:

The LSE objective is: f(h) = cmax + (1/beta) * log(sum_k exp(beta * (corr[k] - cmax)))

The gradient involves:
∂f/∂h[j] = sum_k w_k * ∂corr[k]/∂h[j]

where w_k = exp(beta*(corr[k]-cmax)) / sum_l exp(beta*(corr[l]-cmax))

And ∂corr[k]/∂h[j] = (1-h[j-lag]) * 1_{valid} - h[j+lag] * 1_{valid} as derived earlier.

This is O(n * (2n-1)) = O(2n^2) per gradient evaluation, which is about the same as the numerical gradient (n evaluations × O(n) per evaluation = O(n^2)). But the analytical gradient avoids the n function evaluations, so it's faster in practice.

Let me implement it:

```python
def lse_obj_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    shifted = beta * (corr - cmax)
    obj = cmax + np.log(np.sum(np.exp(shifted))) / beta
    
    # Weights
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        # ∂corr[k]/∂h[j] = (1-h[j-lag])*1_{0<=j-lag<n} - h[j+lag]*1_{0<=j+lag<n}
        if lag >= 0:
            # j-lag >= 0 => j >= lag
            grad[lag:] += w[k] * (1.0 - h[:n-lag])
            # j+lag < n => j < n-lag
            grad[:n-lag] -= w[k] * h[lag:]
        else:
            # lag < 0, let l = -lag > 0
            l = -lag
            # j-lag = j+l >= 0 (always) and j+l < n => j < n-l
            grad[:n-l] += w[k] * (1.0 - h[l:])
            # j+lag = j-l >= 0 => j >= l, and j-l < n (always for j<n, l>0)
            grad[l:] -= w[k] * h[:n-l]
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += resid / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    shifted = beta * (corr - cmax)
    obj = cmax + np.log(np.sum(np.exp(shifted))) / beta
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n - l]
    return obj, grad

def slsqp_optimize(h0, n, betas=None, max_iter=200):
    if betas is None:
        betas = [3, 5, 10, 20, 50, 100, 200]
    h_curr = h0.copy()
    constraints = [{'type': 'eq',
                    'fun': lambda h: h.sum() - n / 2.0,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def obj(h):
            return lse_obj_grad(h, beta)[0]
        def grad(h):
            return lse_obj_grad(h, beta)[1]
        try:
            res = minimize(obj, h_curr, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            if compute_score(h_new) <= compute_score(h_curr) + 1e-12:
                h_curr = h_new
            else:
                # Still use it as starting point for next beta
                h_curr = h_new
        except Exception:
            pass
    return h_curr

def local_shake(h, n, rng, n_shakes=500):
    """Try small random perturbations and keep improvements."""
    best = h.copy()
    best_score = compute_score(best)
    for _ in range(n_shakes):
        h_test = best.copy()
        # Perturb 1-3 random coordinates
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        h_test[idx] += rng.normal(0, 0.05, n_pert)
        h_test = project_feasible(h_test, n)
        s = compute_score(h_test)
        if s < best_score - 1e-12:
            best = h_test
            best_score = s
    return best, best_score
```
