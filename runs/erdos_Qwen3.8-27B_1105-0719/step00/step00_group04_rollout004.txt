import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair_feasible(vec, target_sum):
    x = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    n = len(x)
    rem = target_sum - x.sum()
    if abs(rem) < 1e-12:
        return x
    if rem > 0:
        capacity = 1.0 - x
        order = np.argsort(-capacity)
        added = 0.0
        for idx in order:
            if added >= rem - 1e-13:
                break
            d = min(capacity[idx], rem - added)
            x[idx] += d
            added += d
    else:
        order = np.argsort(x)
        removed = 0.0
        for idx in order:
            if removed >= -rem - 1e-13:
                break
            d = min(x[idx], -rem - removed)
            x[idx] -= d
            removed += d
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if 0 < x[i] < 1:
                x[i] += diff
                x[i] = np.clip(x[i], 0.0, 1.0)
                break
    return x

def interpolate_to_dim(src, target_dim):
    n_src = len(src)
    n_tgt = target_dim
    if n_src == n_tgt:

    h_tgt = np.interp(x_tgt, x_ext, h_ext)
    return np.clip(h_tgt, 0.0, 1.0)

        self.active_lags = active_lags

            G[k, self.n] = 1.0

def solve_epigraph_active(h_start, n, max_iter, active_count, timeout_s=60.0):
    dx = 2.0 / n
    target = n / 2.0
    M = 2 * n - 1
    t0 = time.perf_counter()

    profile = np.correlate(h_start, 1.0 - h_start, mode="full") * dx
    n_active = min(active_count, M)
    active_idx = np.argsort(-profile)[:n_active]
    best_h = h_start.copy()
    best_score = exact_score(best_h)

    for iteration in range(5):
        if time.perf_counter() - t0 > timeout_s:
            break
        if time.perf_counter() - t0 < timeout_s - 5:
            pass
        else:
            break

        prob = EpigraphProblem(n, active_idx)
        bounds = [(0.0, 1.0)] * n + [(0.0, best_score + 0.5)]
        constraints = [
            {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
            {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
        ]
        x0 = np.concatenate([best_h, [best_score + 0.01]])

        result = minimize(prob.objective, x0, jac=prob.objective_jac,
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-12})

        if result.x is not None and not np.any(np.isnan(result.x)):
            h_sol = repair_feasible(result.x[:n], target)
            s = exact_score(h_sol)
            if s < best_score - 1e-14:
                best_h = h_sol.copy()
                best_score = s

        # Update active set
        profile = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        n_active = min(active_count, M)
        active_idx = np.argsort(-profile)[:n_active]

    return best_h, best_score

def local_pair_search(h, n, time_limit=5.0):
    target = n / 2.0
    h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]

    while time.perf_counter() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            for j in range(n):
                if i == j:
                    continue
                if h[i] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                if h[i] - h[j] < 0.01:
                    continue
                for ss in step_sizes:
                    d = ss
                    if d > h[i] - 1e-10 or d > 1.0 - h[j] - 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    s = exact_score(h)
                    if s < best_score - 1e-13:
                        best_score = s
                        improved = True
                        break
                    else:
                        h[i] += d
                        h[j] -= d
        if not improved:
            break
    return h, best_score

def random_local_search(h, n, rng, time_limit=5.0):
    target = n / 2.0
    h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()

    while time.perf_counter() - t0 < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if h[i] <= 1e-8 or 1.0 - h[j] <= 1e-8:
            continue
        direction = 1.0 if h[i] > h[j] else -1.0
        ss = rng.choice([0.01, 0.005, 0.002, 0.001, 0.0005])
        d = direction * ss
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
        s = exact_score(h)
        if s < best_score - 1e-13:
            best_score = s

    return h, best_score

def smooth_local(h, n, time_limit=5.0, top_k=20):
    """Gradient-based local search on active lags."""
    dx = 2.0 / n
    target = n / 2.0
    M = 2 * n - 1
    h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    lr = 0.005

    for _ in range(2000):
        if time.perf_counter() - t0 > time_limit:
            break
        profile = np.correlate(h, 1.0 - h, mode="full") * dx
        active_mask = np.zeros(M, dtype=bool)
        active_idx = np.argsort(-profile)[:top_k]
        active_mask[active_idx] = True

        G = np.zeros(n)
        for m in range(M):
            if not active_mask[m]:
                continue
            delta = m - (n - 1)
            for a in range(n):
                if 0 <= a + delta < n:
                    G[a] += (1.0 - h[a + delta])
                if 0 <= a - delta < n:
                    G[a] -= h[a - delta]
        G *= dx
        G -= G.mean()

        h_new = h - lr * G
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_feasible(h_new, target)
        s = exact_score(h_new)
        if s < best_score - 1e-12:
            h = h_new
            best_score = s
            lr = min(lr * 1.3, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-6:
                break

    return h, best_score
