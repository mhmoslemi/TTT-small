def continuous_refine(h0, budget_s, rng):
    """Refine a continuous solution using projected gradient descent."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy().astype(np.float64)
    # Ensure exact feasibility
    h = project_box(h, target)
    
    best_h = h.copy()
    best_val = c5_value(h)
    best_corr_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    
    start_time = time.time()
    lr = 0.1
    beta = 20.0
    
    while time.time() - start_time < budget_s:
        # Compute log-sum-exp objective
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = float(np.max(corr))
        shifted = beta * (corr - m)
        shifted = shifted - np.max(shifted)
        logsumexp_obj = m + float(np.sum(np.exp(shifted))) / beta
        
        # Compute gradient
        weights = np.exp(shifted)
        weights = weights / np.sum(weights)
        
        grad = np.zeros(n)
        for t in range(2 * n - 1):
            j = t - (n - 1)
            w = weights[t]
            if w < 1e-12:
                continue
            # First term: p in [max(0,j), min(n-1, n-1+j)] for j>=0, or [0, n-1+j] for j<0
            # Combined: p in [max(0, j), min(n-1, n-1+j)]
            p_lo = max(0, j)
            p_hi = min(n - 1, n - 1 + j)
            if p_lo <= p_hi:
                ps = np.arange(p_lo, p_hi + 1)
                grad[ps] += w * (1.0 - h[ps - j])
            
            # Second term: p in [max(0, -j), min(n-1, n-1-j)]
            p_lo2 = max(0, -j)
            p_hi2 = min(n - 1, n - 1 - j)
            if p_lo2 <= p_hi2:
                ps = np.arange(p_lo2, p_hi2 + 1)
                grad[ps] += -w * h[ps + j]
        
        # Projected gradient step
        h_new = project_box(h - lr * grad, target)
        
        # Check if improved
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_corr_max = float(np.max(corr_new))
        new_val = 2.0 / n * new_corr_max
        
        if new_val < best_val:
            best_val = new_val
            best_corr_max = new_corr_max
            best_h = h_new.copy()
            h = h_new
            # Slightly increase lr
            lr = min(lr * 1.05, 1.0)
        else:
            # Decrease learning rate
            lr = lr * 0.95
            # Occasionally increase beta for sharper max
            if rng.random() < 0.01:
                beta = min(beta * 1.5, 200.0)
        
        # Adaptive beta: increase as we get closer
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.5:
            beta = min(beta * 1.01, 100.0)
    
    return best_h, best_val
