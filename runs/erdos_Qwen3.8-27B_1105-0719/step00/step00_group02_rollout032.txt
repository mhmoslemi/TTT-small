from scipy.optimize import minimize_scalar

def evaluate_c5(h):

    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def fw_step(h, mass, active_lags, alpha=1.0):
    """Frank-Wolfe step: linearize at active lags, solve LP, line search."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute gradient of the weighted combination of active lags
    # g[m] = sum over active k of w_k * dc[k]/dh[m]
    # For simplicity, use the active lag with maximum c[k]
    # Or use a weighted combination
    
    # Actually, let's compute gradient of the max by using all lags with weight proportional to how "active" they are
    
    # Simpler: just use the single active lag
    k_active = np.argmax(np.correlate(h, 1-h, mode="full"))
    # k_active is the index in the full correlation, actual lag = k_active - (n-1)
    lag = k_active - (n - 1)
    
    # Compute gradient of c[lag] w.r.t. h
    g = np.zeros(n)
    for m in range(n):
        # Term 1: (1-h[m-lag]) if 0 <= m-lag <= n-1
        if 0 <= m - lag <= n - 1:
            g[m] += (1 - h[m - lag])
        # Term 2: -h[m+lag] if 0 <= m+lag <= n-1
        if 0 <= m + lag <= n - 1:
            g[m] -= h[m + lag]
    
    # Solve min g·y subject to 0 <= y <= 1, sum(y) = mass
    # Sort g, set y=1 for smallest mass values
    order = np.argsort(g)
    y = np.zeros(n)
    y[order[:int(mass)]] = 1.0
    # If mass is not integer, handle fractional
    if int(mass) < mass < int(mass) + 1:
        y[order[int(mass)]] = mass - int(mass)
    
    # Line search
    def obj(t):
        h_t = (1 - t) * h + t * y
        return evaluate_c5(h_t)
    
    result = minimize_scalar(obj, bounds=[0, 1], method='bounded', options={'xatol': 1e-6})
    t_best = result.x
    
    h_new = (1 - t_best) * h + t_best * y
    return h_new, obj(t_best)

from scipy.optimize import minimize_scalar
import time

def get_now():
    return time.perf_counter()

def evaluate_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1 - h[:n-lag])
        if n - lag > 0:
            g[:n-lag] -= h[lag:]
    else:
        g[:n+lag] += (1 - h[-lag:])
        g[-lag:] -= h[:n+lag]
    return g

def solve_linear_subproblem(g, mass):
    """Solve min g·y s.t. 0<=y<=1, sum(y)=mass. Returns y."""
    n = len(g)
    order = np.argsort(g)
    y = np.zeros(n)
    k = int(mass)
    y[order[:k]] = 1.0
    frac = mass - k
    if frac > 1e-12:
        y[order[k]] = frac
    return y

def fw_line_search(h, y, mass):
    """Line search along h + t*(y-h). Returns best t and improved h."""
    n = len(h)
    
    def obj(t):
        h_t = (1-t)*h + t*y
        return evaluate_c5(h_t)
    
    # Coarse search
    best_t = 0
    best_val = obj(0)
    for t_candidate in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        v = obj(t_candidate)
        if v < best_val:
            best_val = v
            best_t = t_candidate
    
    # Fine search around best_t
    if best_t > 0:
        lo = max(0, best_t - 0.1)
        hi = min(1, best_t + 0.1)
        result = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-8})
        if result.fun < best_val:
            best_val = result.fun
            best_t = result.x
    
    h_new = (1-best_t)*h + best_t*y
    # Project to ensure exact feasibility
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum
    current_sum = np.sum(h_new)
    target = mass
    diff = target - current_sum
    if abs(diff) > 1e-10:

        n = len(h_new)
        if diff > 0:

            can_add = 1 - h_new
            order = np.argsort(-can_add)  # most room first
            remaining = diff
            for i in order:
                add = min(remaining, can_add[i])
                h_new[i] += add
                remaining -= add
                if remaining <= 1e-12:
                    break
        else:

            order = np.argsort(h_new)  # smallest first (least to remove)
            # Actually remove from largest
            order = np.argsort(-h_new)
            remaining = -diff
            for i in order:
                rem = min(remaining, h_new[i])
                h_new[i] -= rem
                remaining -= rem
                if remaining <= 1e-12:
                    break

    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta

            g[lag:] += (1.0 - h[:n - lag])

    else:
        if n > -lag:
            g[:n + lag] += (1.0 - h[-lag:])
            g[-lag:] -= h[:n + lag]

    n = len(g)
    order = np.argsort(g)
    y = np.zeros(n)
    k = int(mass)

            best_t = float(res.x)

            for _ in range(20):
                if remaining() < 30:
                    break
                h_up_new, _, _ = fw_iteration(h_up, mass_up, n_active=5, temp_scale=6.0, rng=rng)
                v_new = evaluate_c5(h_up_new)
                if v_new < evaluate_c5(h_up) - 1e-12:
                    h_up = h_up_new
            v_final = evaluate_c5(h_up)
            if v_final < best_c5:
                best_c5 = v_final
                best_h = h_up.copy()

    # Phase 4: Final pair polish on best
    if remaining() > 5:
        n_b = len(best_h)
        mass_b = n_b / 2.0
        h_final, v_final, _ = pair_polish(best_h, mass_b, rng, n_trials=500)
        if v_final < best_c5:
            best_c5 = v_final
            best_h = h_final

    return best_h, best_c5, len(best_h)
