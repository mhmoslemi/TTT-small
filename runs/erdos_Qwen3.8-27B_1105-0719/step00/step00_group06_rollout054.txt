import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def compute_profile(h):
    """Compute the full correlation profile (without dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, tau):
    """Log-sum-exp smooth approximation of max(profile * dx)."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(profile)
    if m < 1e-30:
        return tau * np.log(len(profile))
    shifted = (profile - m) / tau
    shifted = np.clip(shifted, -500, 500)  # prevent overflow
    lse = m + tau * np.log(np.sum(np.exp(shifted)))
    return lse

def smooth_grad(h, tau):
    """Gradient of the log-sum-exp smooth objective."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(profile)
    shifted = (profile - m) / tau
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # profile[k] corresponds to lag m_k = k - (n-1)
    # profile[k] = sum_i h[i + m_k] * (1 - h[i]) * dx
    # d(profile[k])/d(h[j]) = dx * [(1-h[j-m_k]) * I(0<=j-m_k<n) - h[j+m_k] * I(0<=j+m_k<n)]
    
    lags = np.arange(-(n-1), n)  # lags[m] = m, shape (2n-1,)
    # lags[k] = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        m_k = lags[k]
        # Contribution from h[j] as h[i+m] where i = j - m
        # (1 - h[j - m_k]) for valid j - m_k
        # Contribution from h[j] as h[i] where i = j
        # -h[j + m_k] for valid j + m_k
        
        # For first term: j - m_k in [0, n-1] => j in [m_k, m_k + n - 1]
        j_start1 = max(0, m_k)
        j_end1 = min(n - 1, m_k + n - 1)
        if j_start1 <= j_end1:
            idx = np.arange(j_start1, j_end1 + 1)
            grad[idx] += w * dx * (1.0 - h[idx - m_k])

        j_start2 = max(0, -m_k)
        j_end2 = min(n - 1, n - 1 - m_k)
        if j_start2 <= j_end2:
            idx = np.arange(j_start2, j_end2 + 1)
            grad[idx] -= w * dx * h[idx + m_k]
    
    return grad
