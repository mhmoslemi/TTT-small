    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        # Log-sum-exp
        corr_scaled = alpha * corr
        max_corr = np.max(corr_scaled)
        result = (max_corr + np.log(np.sum(np.exp(corr_scaled - max_corr)))) / alpha
        return result
    
    def jac(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        corr_scaled = alpha * corr
        max_corr = np.max(corr_scaled)
        weights = np.exp(corr_scaled - max_corr)
        weights /= np.sum(weights)
        # Gradient of log-sum-exp = weighted average of gradients
        grad = np.zeros(n)
        s_vals = np.arange(len(corr)) - (n - 1)
        for idx, w in enumerate(weights):
            if w < 1e-10:
                continue
            s = s_vals[idx]

                i_range = range(0, n - s)
            else:
                i_range = range(-s, n)

                grad[i] += w * (1.0 - h[i + s])
                if i - s >= 0 and i - s < n:
                    grad[i] -= w * h[i - s]

def project_to_simplex_box(y, target):

def make_T(h, n):
    """Precompute T matrix: T[i,k] = h[i-s_k] + h[i+s_k] with 0 padding."""
    s_vals = np.arange(2 * n - 1) - (n - 1)
    T = np.zeros((n, 2 * n - 1))
    for i in range(n):
        idx_p = i + s_vals
        idx_m = i - s_vals
        Tp = np.zeros(2 * n - 1)
        Tm = np.zeros(2 * n - 1)

        mask_m = (idx_m >= 0) & (idx_m < n)
        Tp[mask_p] = h[idx_p[mask_p]]
        Tm[mask_m] = h[idx_m[mask_m]]

def slsqp_optimize(h0, alpha=20.0, maxiter=500):
