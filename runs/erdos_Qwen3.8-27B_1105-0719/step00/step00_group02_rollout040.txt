def pair_optimize_vec(h, i, j):
    n = len(h)
    S = h[i] + h[j]
    log = max(0.0, S - 1.0)
    hih = min(1.0, S)

        return None

    d_lo = log - h_old_i
    d_hi = hih - h_old_i
    if d_hi - d_lo < 1e-12:
        return None

    n_lags = 2 * n - 1
    C0 = np.correlate(h, 1 - h, mode='full')

        result[mask] = h[idx[mask]]

def _fix_sum(h):
    """Fix sum of h to n/2, maintaining 0<=h<=1."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:

    r = np.arange(n_lags) - (n - 1)

        h_c = h_curr.copy()
        for sw in range(nsweeps):
            if time.time() > t_limit:
                break
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            imp = False
            for i, j in pairs:
                if time.time() > t_limit:
                    return h_c
                hn = _pair_solve(h_c, i, j)
                if hn is not None:
                    h_c = hn
                    imp = True
            h_c = _fix_sum(h_c)
            sc = _eval(h_c)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_h = h_c.copy()
            if not imp:
                break
        return h_c
    
    # Phase 1: deterministic sweeps
    h_c = _sweeps(h0, 80, t0 + budget_s * 0.4)
    
    # Phase 2: SA
    t_sa_end = t0 + budget_s * 0.85
    temp = 0.005
    while time.time() < t_sa_end:
        h_p = best_h.copy()
        np_ = int(rng.integers(1, 8))
        for _ in range(np_):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            S = h_p[i] + h_p[j]
            lo = max(0.0, S - 1.0)
            hi = min(1.0, S)
            if hi - lo > 1e-8:
                d = rng.normal(0, (hi - lo) * 0.3)
                h_p[i] = float(np.clip(h_p[i] + d, lo, hi))
                h_p[j] = S - h_p[i]
        h_p = np.clip(h_p, 0, 1)
        h_p = _fix_sum(h_p)
        
        # Local opt from perturbed
        for _ in range(5):
            if time.time() > t_sa_end:
                break
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            for i, j in pairs:
                if time.time() > t_sa_end:
                    break
                hn = _pair_solve(h_p, i, j)
                if hn is not None:
                    h_p = hn
        h_p = _fix_sum
