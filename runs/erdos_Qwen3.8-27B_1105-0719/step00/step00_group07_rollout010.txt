import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, mode='full') * dx"""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(vals) * dx)

def compute_raw(h):
    """Compute all correlation values (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def make_sym_h(a, n):
    """Construct symmetric h from reduced coordinates a.
    For odd n: h[:m] = a, h[m] = 0.5, h[m+1:] = 1 - a[::-1]
    For even n: h[:m] = a, h[m:] = 1 - a[::-1]
    """
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    else:
        h[m:] = 1.0 - a[::-1]
    return h

def get_a_from_h(h, n):
    """Extract reduced coordinates a from a symmetric h."""
    m = n // 2
    return h[:m].copy()

def symmetrize(h):
    """Apply symmetrization T(h) = (h + (1-h)[::-1]) / 2"""
    n = len(h)
    T = (1.0 - h[::-1])
    return 0.5 * (h + T)

def project_box_hyperplane(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} using bisection on lambda."""
    x = np.clip(h.astype(float), 0.0, 1.0)
    n = len(x)
    if abs(np.sum(x) - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    idx = np.arange(n)
    if res > 1e-12:
        for i in idx:
            space = 1.0 - x[i]
            add = min(space, res)
            if add > 1e-14:
                x[i] += add
                res -= add
            if abs(res) < 1e-14:
                break
    elif res < -1e-12:
        for i in idx:
            room = x[i]
            sub = min(room, -res)
            if sub > 1e-14:
                x[i] -= sub
                res += sub
            if abs(res) < 1e-14:
                break
    return x

def compute_grad_a(h, n, beta):
    """Compute gradient of smooth-max objective in a-coordinates.
    Returns grad_a of length m = n//2.
    """
    m = n // 2
    raw = compute_raw(h)
    shift = np.max(raw)
    scaled = (raw - shift) / max(beta, 1e-10)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / np.sum(exp_scaled)
    u = 1.0 - 2.0 * h
    grad_h = np.correlate(w, u, mode='valid')
    # grad_a[j] = grad_h[j] - grad_h[n-1-j]
    indices = np.arange(m)
    grad_a = grad_h[indices] - grad_h[n - 1 - indices]
    return grad_a, raw, shift

def try_cvx_young(n, max_time=30):
    """Try CVXPY convex relaxation using Young's inequality majorant."""
    try:
        import cvxpy as cvx
    except ImportError:
        return None, None
    
    m = n // 2
    t0 = time.time()
    
    try:
        a = cvx.Variable(m)
        # Build h expression
        if n % 2 == 1:
            h_expr = cvx.hstack((a, 0.5, 1.0 - a[::-1]))
        else:
            h_expr = cvx.hstack((a, 1.0 - a[::-1]))
        
        # Build Young's inequality constraints for each lag
        # R_t <= U_t for each t, minimize max U_t
        t_var = cvx.Variable()
        constraints = [a >= 0, a <= 1]
        
        # For each correlation lag, add Young majorant constraint
        h = h_expr
        for t in range(2 * n - 1):
            # out[t] = sum_{i=max(0,t-n+1)}^{min(n-1,t)} h[i] * (1-h[t-i])
            i_lo = max(0, t - n + 1)
            i_hi = min(n - 1, t)
            if i_lo > i_hi:
                continue
            indices = range(i_lo, i_hi + 1)
            # Young: h[i]*(1-h[t-i]) <= (h[i]^2 + (1-h[t-i])^2) / 2
            # Sum: R_t <= 0.5*sum(h[i]^2) + 0.5*sum((1-h[t-i])^2)
            h_i_sq = cvx.sum_squares(cvx.vec(h[indices]))
            j_indices = [t - i for i in indices]
            one_minus_h_j_sq = cvx.sum_squares(1.0 - cvx.vec(h[j_indices]))
            U_t = 0.5 * h_i_sq + 0.5 * one_minus_h_j_sq
            constraints.append(U_t <= t_var)
        
        dx = 2.0 / n
        prob = cvx.Problem(cvx.Minimize(t_var * dx), constraints)
        
        # Try solvers
        solvers_to_try = ['ECOS', 'CVXPY', 'CLARABEL', 'SCS', 'MOSEK', 'OSQP', 'PDLP']
        a_val = None
        status = None
        
        for solver in solvers_to_try:
            if time.time() - t0 > max_time * 0.8:
                break
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.array(a.value, dtype=float).flatten()
                    status = prob.status
                    break
            except Exception:
                continue
        
        if a_val is not None and np.all(np.isfinite(a_val)):
            a_val = np.clip(a_val, 0.0, 1.0)

            c = compute_c5(h)
            return a_val, c
    except Exception:
        pass
    
    return None, None

def downsample_h(h, new_n):
    """Downsample h from one resolution to another, preserving the integral."""
    old_n = len(h)
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    # Interpolate
    x_old = np.arange(old_n) * dx_old + dx_old / 2.0
    x_new = np.arange(new_n) * dx_new + dx_new / 2.0
    h_interp = np.interp(x_new, x_old, h)
    # Project to have correct mass
    h_new = project_box_hyperplane(h_interp, new_n / 2.0)
    return h_new
