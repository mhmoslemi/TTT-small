def corr_grad(h, n):
    """Compute gradient of each correlation component w.r.t. h.
    Returns array of shape (2n-1, n) where result[m, j] = d(c[m])/d(h[j])
    """
    # For output index m, lag k = m - (n-1)
    # If k >= 0: c[m] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    #   d/dh[j] = (1-h[j-k])*[j>=k] - h[j+k]*[j<=n-1-k]
    # If k < 0, q=-k: c[m] = sum_{i=0}^{n-1-q} h[i]*(1-h[i+q])
    #   d/dh[j] = (1-h[j+q])*[j<=n-1-q] - h[j-q]*[j>=q]
    
    grad = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            L = k
            # (1-h[j-L]) for j >= L
            if L < n:
                jt = np.arange(L, n)
                grad[m, jt] += (1.0 - h[jt - L])
            # -h[j+L] for j <= n-1-L
            if L < n:
                jt = np.arange(0, n-L)
                grad[m, jt] -= h[jt + L]
        else:
            q = -k
            # (1-h[j+q]) for j <= n-1-q
            if q < n:
                jt = np.arange(0, n-q)
                grad[m, jt] += (1.0 - h[jt + q])
            # -h[j-q] for j >= q
            if q < n:
                jt = np.arange(q, n)
                grad[m, jt] -= h[jt - q]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sigmoid_vec(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def find_c_scalar(theta, target):
    lo, hi = -50.0, 50.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(sigmoid_vec(theta + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def theta_to_h(theta, target):
    c = find_c_scalar(theta, target)
    return sigmoid_vec(theta + c)

def h_to_theta(h):
    h = np.clip(h, 1e-12, 1.0 - 1e-12)
    return np.log(h / (1.0 - h))

def lse_value_grad_theta(theta, target, beta):
    h = theta_to_h(theta, target)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = float(np.max(corr))
    shifted = np.clip(beta * (corr - M), -700.0, 0.0)
    q = np.exp(shifted)
    sw = float(np.sum(q))
    lse_val = M + np.log(sw) / beta
    w = q / sw
    
    # Gradient of LSE w.r.t. h
    grad_h = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            L = k
            if L < n:
                grad_h[L:] += wm * (1.0 - h[:n-L])
                grad_h[:n-L] -= wm * h[L:]
        else:
            qq = -k
            if qq < n:
                grad_h[:n-qq] += wm * (1.0 - h[qq:])
                grad_h[qq:] -= wm * h[:n-qq]
    
    # Convert to theta gradient
    s = h * (1.0 - h)
    S = float(np.sum(s))
    if S < 1e-15:
        return lse_val, np.zeros(n)
    gs = float(np.dot(grad_h, s))
    grad_theta = s * (grad_h - gs / S)
    return lse_val, grad_theta

def bfgs_lse(theta, target, beta, max_iter=200):
    def obj_func(th):
        val, grad = lse_value_grad_theta(th, target, beta)
        return val, grad
    
    res = minimize(obj_func, theta, method='L-BFGS-B', 
                   options={'maxiter': max_iter, 'gtol': 1e-8})
    return res.x

def coordinate_descent(theta, target, n_iter, rng, theta_low=-15.0, theta_high=15.0):
    n = len(theta)
    theta = theta.copy()
    h = theta_to_h(theta, target)
    best_c5 = evaluate_c5(h)
    
    step_sizes = [3.0, 1.5, 0.75, 0.35, 0.15, 0.05, 0.01]

        direction = 1.0 if rng.random() < 0.5 else -1.0
        new_theta = theta.copy()
        new_val = np.clip(theta[j] + direction * size, theta_low, theta_high)
        new_theta[j] = new_val
        new_h = theta_to_h(new_theta, target)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5 - 1e-12:

            h = new_h
            best_c5 = new_c5

                new_theta2[j] = np.clip(theta[j] + direction * ss, theta_low, theta_high)
                new_h2 = theta_to_h(new_theta2, target)
                new_c5_2 = evaluate_c5(new_h2)
                if new_c5_2 < best_c5 - 1e-12:
                    theta = new_theta2
                    h = new_h2
                    best_c5 = new_c5_2
                    break
    
    return theta, h, best_c5

def interpolate_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])

        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def make_cosine_candidate(n, amp, freq, phase, rng):

def interpolate_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    target = n_new / 2.0
    diff = target - float(np.sum(h_new))
    h_new = h_new + diff / n_new
    h_new = np.clip(h_new, 0.0, 1.0)

    if abs(diff2) > 1e-10:

                    theta_cr = h_to_theta(best_h)
                    theta_cr, h_cr, c5_cr = coord_descent(theta_cr, tgt_final, n_cd_r, rng)
                    if c5_cr < best_c5:
                        best_c5 = c5_cr
                        best_h = h_cr.copy()

    # Phase 8: Final polish coordinate descent
    if remaining() > 30.0:
        n_polish = int(min(20000, remaining() * 6))
        n_fin = len(best_h)
        tgt_fin = n_fin / 2.0
        theta_fin = h_to_theta(best_h)
        theta_fin, h_fin, c5_fin = coord_descent(theta_fin, tgt_fin, n_polish, rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation and mass fix
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_mass(best_h)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
