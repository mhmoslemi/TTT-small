```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile

def project_box_sum(h, target=None):
    """Project h onto {0 <= x <= 1, sum(x) = target}"""
    n = len(h)
    x = np.array(h, dtype=np.float64).copy()
    if target is None:
        target = 0.5 * n
    
    # Replace non-finite values
    x[~np.isfinite(x)] = 0.0
    
    # Water-filling to project onto box + sum constraint
    for _ in range(100):
        # Clip to [0, 1]
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to add; add to elements < 1
            mask = x < 1.0
            count = mask.sum()
            if count == 0:
                break
            step = min(diff / count, 1.0 - x[mask])
            # Actually just do naive: add diff/count to all < 1
            x[mask] += diff / count
        else:
            # Need to subtract; subtract from elements > 0
            mask = x > 0.0
            count = mask.sum()
            if count == 0:
                break
            x[mask] += diff / count
    
    # Final clip and fix
    x = np.clip(x, 0.0, 1.0)
    # One more correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
            if mask.sum() > 0:
                x[mask] += diff / mask.sum()
        else:
            mask = x > 0.0
            if mask.sum() > 0:
                x[mask] += diff / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    
    return x

def compute_profile_and_grad(h, tau):
    """Compute smoothed log-sum-exp objective and its gradient.
    Returns (value, gradient). All measures in units of dx."""
    n = len(h)
    dx = 2.0 / n
    profile_raw = np.correlate(h, 1.0 - h, mode='full')
    profile = profile_raw * dx  # shape (2n-1,)
    
    M = np.max(profile)
    shifted = (profile - M) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 0)
    weights = np.exp(shifted)
    W = np.sum(weights)
    if W < 1e-300:
        W = 1.0
        weights = np.ones(2*n-1) / (2*n-1)
    w = weights / W
    
    value = M + tau * np.log(W)
    
    # Compute gradient
    # profile index m corresponds to shift s = m - (n-1)
    # For s >= 0: profile component = dx * sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
    #   grad w.r.t. h[i]: dx * [(1-h[i-s]) if s<=i<=n-1] - dx * [h[i+s] if i<=n-1-s]
    # For s < 0, a=-s: profile component = dx * sum_{j=0}^{n-1-a} h[j]*(1-h[j+a])
    #   grad w.r.t. h[i]: dx * [(1-h[i+a]) if i<=n-1-a] - dx * [h[i-a] if i>=a]
    
    grad = np.zeros(n)
    
    for idx in range(2*n-1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        if s >= 0:
            # g[i] += wi * dx * [(1-h[i-s]) if s<=i<=n-1] - wi * dx * [h[i+s] if i<=n-1-s]
            # Term 1: i ranges from s to n-1, value (1-h[i-s])
            if s < n:
                grad[s:] += wi * dx * (1.0 - h[:n-s])
            # Term 2: i ranges from 0 to n-1-s, subtract h[i+s]
            if s < n:
                grad[:n-s] -= wi * dx * h[s:]
        else:
            a = -s
            # g[i] += wi * dx * [(1-h[i+a]) if i<=n-1-a] - wi * dx * [h[i-a] if i>=a]
            # Term 1: i ranges from 0 to n-1-a, value (1-h[i+a])
            if a < n:
                grad[:n-a] += wi * dx * (1.0 - h[a:])
            # Term 2: i ranges from a to n-1, subtract h[i-a]
            if a < n:
                grad[a:] -= wi * dx * h[:-a]
    
    return value, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile

def project_box_sum(h, target=None):
    n = len(h)
    x = np.array(h, dtype=np.float64).copy()
    if target is None:
        target = 0.5 * n
    x[~np.isfinite(x)] = 0.0
    for _ in range(200):
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = x < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            x[mask] = np.clip(x[mask] + diff / cnt, 0, 1)
        else:
            mask = x > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            x[mask] = np.clip(x[mask] + diff / cnt, 0, 1)
    x = np.clip(x, 0.0, 1.0)
    return x

def lse_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    prof_raw = np.correlate(h, 1.0 - h, mode='full')
    prof = prof_raw * dx
    
    M = np.max(prof)
    shifted = (prof - M) / tau
    shifted = np.minimum(shifted, 0.0)
    weights = np.exp(shifted)
    W = np.sum(weights)
    if W < 1e-300:
        weights = np.ones(len(prof)) / len(prof)
        W = 1.0
    w = weights / W
    
    value = M + tau * np.log(W)
    
    grad = np.zeros(n)
    # Vectorized gradient computation
    # For each index i in [0, n):
    # grad[i] = dx * sum_{s=-n+1}^{n-1} w[n-1+s] * d(prof_s)/d(h[i])
    
    # Let's define w_pos[s] = w[n-1+s] for s = 0, 1, ..., n-1 (positive shifts)
    # and w_neg[a] = w[n-1-a] for a = 1, 2, ..., n-1 (negative shifts, a = -s)
    
    w_pos = w[n-1:2*n-1]  # w for shifts s = 0, 1, ..., n-1, length n
    w_neg = w[0:n-1][::-1]  # w for shifts s = -1, -2, ..., -(n-1), i.e., a = 1, 2, ..., n-1
    
    # For positive shift s (0-indexed in w_pos as index s):
    # Contribution to grad[i]:
    #   + w_pos[s] * (1 - h[i-s]) if i >= s
    #   - w_pos[s] * h[i+s] if i+s < n
    
    # For negative shift a (0-indexed in w_neg as index a-1):
    # Contribution to grad[i]:
    #   + w_neg[a-1] * (1 - h[i+a]) if i+a < n
    #   - w_neg[a-1] * h[i-a] if i >= a
    
    # Let's compute using convolutions or direct loops
    # For n=70, a loop is fine
    
    for i in range(n):
        g = 0.0
        # Positive shifts s = 0 to n-1
        # + w_pos[s]*(1-h[i-s]) if i>=s
        s_max = min(i, n-1)
        if s_max >= 0:
            s_range = np.arange(0, s_max+1)
            g += np.dot(w_pos[s_range], 1.0 - h[i - s_range])
        # - w_pos[s]*h[i+s] if i+s < n
        s_max2 = min(n-1-i, n-1)
        if s_max2 >= 0:
            s_range2 = np.arange(0, s_max2+1)
            g -= np.dot(w_pos[s_range2], h[i + s_range2])
        # Negative shifts a = 1 to n-1
        # + w_neg[a-1]*(1-h[i+a]) if i+a < n
        a_max = min(n-1-i, n-1)
        if a_max >= 1:
            a_range = np.arange(1, a_max+1)
            g += np.dot(w_neg[a_range-1], 1.0 - h[i + a_range])
        # - w_neg[a-1]*h[i-a] if i >= a
        a_max2 = min(i, n-1)
        if a_max2 >= 1:
            a_range2 = np.arange(1, a_max2+1)
            g -= np.dot(w_neg[a_range2-1], h[i - a_range2])
        
        grad[i] = g * dx
    
    return value, grad

def binary_threshold(h):
    """Create binary version: top n/2 entries become 1, rest 0"""
    n = len(h)
    target = n // 2
    if target * 2 != n:
        return None
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_swap_search(b, max_rounds=20000, rng=None):
    """Greedily swap 1s and 0s to reduce C5"""
    if b is None:
        return None
    n = len(b)
    best = b.copy()
    best_c5, _ = exact_objective(best)
    ones = np.where(best == 1.0)[0].tolist()
    zeros = np.where(best == 0.0)[0].tolist()
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    for round_i in range(max_rounds):
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_swaps = min(n_ones * n_zeros, 500)
        for _ in range(n_swaps):
            oi = rng.randint(n_ones)
            zi = rng.randint(n_zeros)
            o = ones[oi]
            z = zeros[zi]
            # Swap
            best[o] = 0.0
            best[z] = 1.0
            c5, _ = exact_objective(best)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
                ones[oi] = z
                zeros[zi] = o
            else:
                # Revert
                best[o] = 1.0
                best[z] = 0.0
        if not improved:
            # Try a few more deterministic
            break
    return best, best_c5

def random_local_search(h, n_iter=5000, rng=None):
    """Random perturbation with projection, accept if improves"""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    best = project_box_sum(h)
    best_c5, _ = exact_objective(best)
    
    for _ in range(n_iter):
        # Pick 2-5 random indices and perturb
        k = rng.randint(2, 6)
        idx = rng.choice(n, size=k, replace=False)
        perturb = rng.uniform(-0.3, 0.3, size=k)
        trial = best.copy()
        trial[idx] += perturb
        trial = project_box_sum(trial)
        c5, _ = exact_objective(trial)
        if c5 < best_c5 - 1e-12:
            best = trial
            best_c5 = c5
    
    return best, best_c5

def interpolate_pair(h1, h2, rng=None):
    """Try midpoints between two feasible vectors"""
    results = []
    alphas = [0.25, 0.5, 0.75]
    for a in alphas:
        h = (1-a)*h1 + a*h2
        h = project_box_sum(h)
        c5, _ = exact_objective(h)
        results.append((h.copy(), c5))
    return results
```

Actually, for a given pair (i, j) and a change of δ:
- h[i] changes from a to a+δ
- h[j] changes from b to b-δ

The change in c_raw[m] is:
Δc[m] = (dc[m]/dh[i]) * δ + (dc[m]/dh[j]) * (-δ) = (ci - cj) * δ

where ci = dc[m]/dh[i] and cj = dc[m]/dh[j] are constant (don't depend on δ, since c[m] is linear in each h[k]).

So I can compute all the (ci - cj) values once per pair, and then for any δ, the new profile is:
c_new[m] = c_old[m] + (ci[m] - cj[m]) * δ

This means I can evaluate the objective at any δ in O(n) (just compute the max of the updated profile), without calling correlate!

Let me precompute ci and cj for each m:

For shift s (m = n-1+s):
ci = dc[m]/dh[i]:
  If s >= 0: (1-h[i-s]) if i>=s and i-s < n, else 0; minus h[i+s] if i+s < n, else 0
  If s < 0, a=-s: (1-h[i+a]) if i+a < n, else 0; minus h[i-a] if i>=a, else 0

cj = dc[m]/dh[j]: similar with j instead of i.

So the slope for term m is: slope[m] = ci - cj

And c_new[m] = c_old[m] + slope[m] * δ

The objective at δ is: max_m (c_old[m] + slope[m]*δ) * dx

This is a convex piecewise-linear function of δ. I can find its minimum efficiently.

For a convex PL function f(δ) = max_m (a_m + b_m * δ), the minimum over [L, U] is at:
- δ = L or δ = U (boundaries), or
- A point where the argmax changes (a kink point)

The kink points are where a_m + b_m*δ = a_l + b_l*δ for some m ≠ l, i.e., δ = (a_l - a_m) / (b_m - b_l).

Since there are 2n-1 terms, there are O((2n)^2) = O(n^2) kink points. For n=70, that's about 19000 kinks. I can compute them all and check which ones fall in [L, U].

Actually, a simpler approach for convex PL minimization: the function f(δ) = max_m (a_m + b_m * δ) is minimized where the "upper envelope" has a local minimum. This happens where the leftmost and rightmost active terms have opposite slopes (or where a single term is active and has zero slope, etc.)

The most practical approach: 
1. Evaluate f at δ=L, δ=U, and at δ=0 (current point)
2. Find the δ where f is minimized by checking all kink points in [L, U]

For n=70, computing all kinks is O((2n)^2) = O(n^2) ≈ 20K operations, and evaluating f at each is O(n) = O(70). Total per pair: O(n^3) ≈ 100K. For 10000 pairs: 10^9, which is about 1-2 seconds in Python... might be a bit slow.

```python
def coord_pair_optimize(h, i, j):
    """Optimize pair (i,j) preserving sum. Returns updated h."""
    n = len(h)
    dx = 2.0 / n
    a, b = h[i], h[j]
    
    delta_min = max(-a, b - 1.0)
    delta_max = min(1.0 - a, b)
    if delta_min >= delta_max - 1e-15:
        return h.copy()
    
    prof_old = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute slopes
    slopes = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        s = m - (n - 1)
        ci = _coeff(h, i, s, n)
        cj = _coeff(h, j, s, n)
        slopes[m] = ci - cj
    
    # Coarse grid
    deltas = np.linspace(delta_min, delta_max, 100)
    prof_grid = prof_old[None, :] + slopes[None, :] * deltas[:, None]
    max_vals = (prof_grid.max(axis=1) * dx)
    best_idx = np.argmin(max_vals)
    best_delta = deltas[best_idx]
    
    # Refine
    d_gap = (delta_max - delta_min) / 99.0
    d_lo = max(delta_min, best_delta - d_gap)
    d_hi = min(delta_max, best_delta + d_gap)
    deltas2 = np.linspace(d_lo, d_hi, 200)
    prof_grid2 = prof_old[None, :] + slopes[None, :] * deltas2[:, None]
    max_vals2 = (prof_grid2.max(axis=1) * dx)
    best_idx2 = np.argmin(max_vals2)
    best_delta = deltas2[best_idx2]
    
    h_new = h.copy()
    h_new[i] = a + best_delta
    h_new[j] = b - best_delta
    return h_new
```
