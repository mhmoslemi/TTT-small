def exact_C(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def validate_h(h):
    h = np.asarray(h, dtype=np.float64)

def project_feasible(u):
    """Project u onto {0<=h<=1, sum(h)=n/2} using water-filling."""
    u = np.asarray(u, dtype=np.float64).copy()

def sigmoid(x):
    """Numerically stable sigmoid."""
    result = np.empty_like(x, dtype=np.float64)
    pos = x >= 0
    neg = ~pos
    result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[neg])
    result[neg] = ex / (1.0 + ex)
    return result

def transformed_objective(y, beta):
    """F_beta(h(y)) where h = feasible_from_logit(y). Returns (F, grad_y)."""
    h = feasible_from_logit(y)
    F, G = smoothed_objective_and_grad(h, beta)
    
    # Chain rule: dF/dy_j = s_j * (G_j - B)
    s = h * (1.0 - h)

    B = np.dot(G, s) / M
    grad_y = s * (G - B)
    
    # Safety
    if not np.all(np.isfinite(grad_y)):
        return F, np.zeros_like(y)
    
    return F, grad_y

def resize_h(h, new_n):

GS = {}

def exact_C(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def validate_h(h):
    h = np.asarray(h, dtype=np.float64)

    target = n / 2.0

def project_feasible(u):
    u = np.asarray(u, dtype=np.float64).copy()
    n = len(u)
