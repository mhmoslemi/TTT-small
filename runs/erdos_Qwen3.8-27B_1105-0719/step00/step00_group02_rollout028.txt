```python
    n = len(h)
    p = np.arange(-(n-1), n)  # lag indices, length 2n-1
    # d_p = -(1-h[i+p])*I(0≤i+p<n) + h[i-p]*I(0≤i-p<n) + (1-h[j+p])*I(0≤j+p<n) - h[j-p]*I(0≤j-p<n)
```

```python
    current_max = raw.max()

    fracs = np.array([0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0])
    
    start = time.perf_counter()
    n = len(h)

    while True:
        elapsed = time.perf_counter() - start
        if elapsed > time_limit:
            break
        
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                cap = min(h[i], 1.0 - h[j])
                if cap <= 1e-12:
                    continue
                d = compute_d(h, i, j)
                δs = fracs * cap
                # new_raws[k, p] = raw[p] + δs[k] * d[p]
                new_maxes = (raw[None, :] + δs[:, None] * d[None, :]).max(axis=1)

                if new_maxes[best_k] < current_max - tol:

                    δ = δs[best_k]
                    h[i] -= δ
                    h[j] += δ
                    raw = raw + δ * d
                    current_max = raw.max()
                    improved = True
    
        if not improved:
            break
    
    return h, current_max * dx
```

```python
def lse_gradient_step(h, tau, alpha, s):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    
    # Softmax weights
    shifted = raw - raw.max()
    w = np.exp(shifted / tau)
    w /= w.sum()
    # w[k] corresponds to lag p = k - (n-1)
    
    # Compute gradient
    # ∇S[a] = dx * [sum_q w[q-a] * (1-h[q]) - sum_q w[a-q] * h[q]]
    # In index form: w is indexed 0..2n-2, lag = idx - (n-1)
    # w[q-a] means w at index (q-a) + (n-1) = q + n-1-a
    # w[a-q] means w at index (a-q) + (n-1) = a + n-1-q
    
    omh = 1.0 - h
    grad = np.zeros(n)
    for a in range(n):
        # First sum: sum_{q=0}^{n-1} w[q+n-1-a] * omh[q]
        idx1 = np.arange(n-1-a, n-1-a+n)
        # Second sum: sum_{q=0}^{n-1} w[a+n-1-q] * h[q] = sum_{q=0}^{n-1} w[a+q] * h[n-1-q]
        # Actually: a+n-1-q for q=0..n-1 gives indices a+n-1, a+n-2, ..., a
        # So: sum of w[a:a+n][::-1] * h (reversed?)
        # Let me just do: idx2[q] = a + n - 1 - q for q=0..n-1
        idx2 = np.arange(a, a+n)[::-1]  # a+n-1, a+n-2, ..., a
        # Wait, arange(a, a+n) = [a, a+1, ..., a+n-1], reversed = [a+n-1, ..., a+1, a]
        # That's a+n-1-q for q=0,1,...,n-1. Yes!
        grad[a] = dx * (np.dot(w[idx1], omh) - np.dot(w[idx2], h))
    
    # Project gradient: subtract mean to stay in sum=const hyperplane
    grad_tangent = grad - grad.mean()
    
    # Step
    new_h = h - alpha * grad_tangent
    new_h = project_box_sum(new_h, s)
    
    return new_h
```
