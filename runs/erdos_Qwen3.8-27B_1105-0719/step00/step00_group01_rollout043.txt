            mask = h < (1.0 - 1e-15)
        else:
            mask = h > 1e-15
        if np.any(mask):
            h[mask] += r / float(np.count_nonzero(mask))
            np.clip(h, 0.0, 1.0, out=h)

    np.clip(h, 0.0, 1.0, out=h)
    return h


def exact_c(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def evaluate_feasible(h):
    arr = np.asarray(h, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    arr = arr.copy()
    n = len(arr)
    if n == 0:
        return arr, 0.0
    hh = repair_h(arr, n / 2.0)
    return hh, float(exact_c(hh))


def default_h(n):
    n = max(int(n), 8)
    x = (np.arange(n, dtype=np.float64) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.35 * np.cos(np.pi * x) + 0.10 * np.sin(2.0 * np.pi * x)
    return repair_h(h, n / 2.0)


def symmetry_and_shift_best(h):
    h = repair_h(np.asarray(h, dtype=np.float64), len(np.asarray(h)) / 2.0)
    n = len(h)
    if n <= 1:
        return h.copy(), float(exact_c(h))

    best_h = h.copy()
    best_c = float(exact_c(h))

    bases = [
        h,
        1.0 - h,
        h[::-1].copy(),
        1.0 - h[::-1],
        0.5 * (h + (1.0 - h[::-1])),
    ]

    for base in bases:
        base = np.asarray(base, dtype=np.float64)
        if base.shape != (n,):
            continue
        if not np.all(np.isfinite(base)):
            continue
        for k in range(n):
            if k == 0:
                cand = base
            else:
                cand = np.roll(base, k)
            ch = repair_h(cand, n / 2.0)
            cc = float(exact_c(ch))
            if cc < best_c - 1e-14:
                best_c = cc
                best_h = ch.copy()

    return best_h, best_c


def resample_h(h, new_n):
    new_n = max(int(new_n), 8)
    old = np.asarray(h, dtype=np.float64)
    if old.ndim != 1:
        old = old.ravel()
    old_n = len(old)
    if old_n == 0:
        return default_h(new_n)
    if new_n == old_n:
        return repair_h(old, new_n / 2.0)

    xo = (np.arange(old_n, dtype=np.float64) + 0.5) * (2.0 / float(old_n))
    xn = (np.arange(new_n, dtype=np.float64) + 0.5) * (2.0 / float(new_n))
    hh = np.interp(xn, xo, old, left=old[0], right=old[-1])
    return repair_h(hh, new_n / 2.0)


def build_pair_list(n):
    i_list = []
    j_list = []
    for i in range(n):
        for j in range(i + 1, n):
            i_list.append(i)
            j_list.append(j)
    return np.array(i_list, dtype=np.int64), np.array(j_list, dtype=np.int64)


def clean_direction(d):
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size == 0:
        return None
    if not np.all(np.isfinite(d)):
        d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    d = d - float(np.mean(d))
    maxabs = float(np.max(np.abs(d)))
    if not math.isfinite(maxabs) or maxabs <= 1e-15:
        return None
    return d / maxabs


def feasible_interval(h, d):
    tlo = -np.inf
    thi = np.inf
    pos = d > 1e-15
    neg = d < -1e-15

    if np.any(pos):
        tlo = max(tlo, float(np.max(-h[pos] / d[pos])))
        thi = min(thi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tlo = max(tlo, float(np.max((1.0 - h[neg]) / d[neg])))
        thi = min(thi, float(np.min(-h[neg] / d[neg])))

    if not math.isfinite(tlo):
        tlo = -1.0
    if not math.isfinite(thi):
        thi = 1.0

    if tlo > 0.0:
        tlo = 0.0
    if thi < 0.0:
        thi = 0.0
    if tlo > thi:
        return None
    return (float(tlo), float(thi))


def line_values(c0, A, B, ts):
    t = np.asarray(ts, dtype=np.float64)[:, None]
    vals = c0[None, :] + A[None, :] * t - B[None, :] * (t * t)
    return vals.max(axis=1)


def line_value_single(c0, A, B, t):
    return float(np.max(c0 + A * t - B * t * t))


def golden_min_line(c0, A, B, lo, hi, max_iter=24):
    if hi <= lo:
        return (float(lo), line_value_single(c0, A, B, lo))

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = line_value_single(c0, A, B, x1)
    f2 = line_value_single(c0, A, B, x2)

    for _ in range(max_iter):
        if hi - lo <= 1e-13:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = line_value_single(c0, A, B, x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = line_value_single(c0, A, B, x2)

    if f1 <= f2:
        return (float(x1), float(f1))
    return (float(x2), float(f2))


def line_search_values(h, d, current_corr, q0=None):
    interval = feasible_interval(h, d)
    if interval is None:
        return None
    tlo, thi = interval
    if thi - tlo <= 1e-14:
        c0max = float(np.max(current_corr))
        return (0.0, c0max, c0max)

    if q0 is None:
        q0 = 1.0 - h

    A = np.correlate(d, q0, mode="full") - np.correlate(h, d, mode="full")
    B = np.correlate(d, d, mode="full")
    c0 = current_corr
    c0max = float(np.max(c0))

    grid = 40
    ts = np.linspace(tlo, thi, grid)
    vals = line_values(c0, A, B, ts)
    vals = np.where(np.isfinite(vals), vals, 1e300)
    idx = int(np.argmin(vals))
    tbest = float(ts[idx])
    cbest = float(vals[idx])
    if c0max < cbest:
        tbest = 0.0
        cbest = c0max

    cell = (thi - tlo) / float(grid)
    for _ in range(2):
        lo = max(tlo, tbest - 3.0 * cell)
        hi = min(thi, tbest + 3.0 * cell)
        if hi - lo <= 1e-15:
            break
        m = 64
        ts2 = np.linspace(lo, hi, m)
        vals2 = line_values(c0, A, B, ts2)
        vals2 = np.where(np.isfinite(vals2), vals2, 1e300)
        idx2 = int(np.argmin(vals2))
        c2 = float(vals2[idx2])
        if c2 + 1e-13 < cbest:
            cbest = c2
            tbest = float(ts2[idx2])
            cell = (hi - lo) / float(m)
        else:
            cell *= 0.4
            if cell <= 0.0:
                cell = (thi - tlo) / float(grid)

    lo = max(tlo, tbest - max(cell, 1e-10))
    hi = min(thi, tbest + max(cell, 1e-10))
    if hi - lo > 1e-13:
        gt, gc = golden_min_line(c0, A, B, lo, hi, 24)
        if gc + 1e-13 < cbest:
            cbest = gc
            tbest = gt

    if c0max + 1e-13 < cbest:
        cbest = c0max
        tbest = 0.0

    return (float(tbest), float(cbest), c0max)


def try_direction(h, d, current_corr, current_c, q0):
    d2 = clean_direction(d)
    if d2 is None:
        return None

    res = line_search_values(h, d2, current_corr, q0)
    if res is None:
        return None

    tbest, cline, c0max = res
    if cline > c0max + 1e-9:
        return None
    if abs(tbest) < 1e-13:
        return None

    target = len(h) / 2.0
    factors = (1.0, 0.75, 0.5, 0.25)
    last_t = 0.0

    for f in factors:
        t = float(tbest * f)
        if abs(t - last_t) < 1e-14:
            continue
        last_t = t
        if abs(t) < 1e-13:
            continue
        hn = h + t * d2
        hn = repair_h(hn, target)
        c = float(exact_c(hn))
        if c < current_c - 1e-12:
            return hn, c

    return None


def active_gradient(h, active_indices):
    n = len(h)
    G = np.zeros(n, dtype=np.float64)
    center = n - 1
    for m in active_indices:
        m = int(m)
        if m < 0 or m >= 2 * n - 1:
            continue
        s = center - m
        if s >= 0:
            i = np.arange(0, n - s, dtype=np.int64)
        else:
            i = np.arange(-s, n, dtype=np.int64)
        ii = i + s
        G[i] += 1.0 - h[ii]
        G[ii] += -h[i]
    return G


def generate_active_directions(current, current_corr, rng, max_count=14):
    n = len(current)
    c0 = current_corr
    top_count = min(40, len(c0))
    top = np.argsort(c0)[-top_count:]
    top = top[::-1]

    dirs = []

    G = active_gradient(current, top[:16])
    cd = clean_direction(-G)
    if cd is not None:
        dirs.append(cd)

    for k in range(min(4, len(top))):
        G = active_gradient(current, top[k:k + 1])
        cd = clean_direction(-G)
        if cd is not None and len(dirs) < max_count:
            dirs.append(cd)

    sizes = (3, 5, 8)
    for size in sizes:
        for _ in range(3):
            cnt = min(size, len(top))
            if cnt <= 0:
                break
            sub = rng.choice(top, size=cnt, replace=False)
            G = active_gradient(current, sub)
            gstd = float(G.std())
            if gstd > 1e-14:
                d = -G + 0.08 * gstd * rng.normal(size=n)
            else:
                d = -G
            cd = clean_direction(d)
            if cd is not None and len(dirs) < max_count:
                dirs.append(cd)
            if len(dirs) >= max_count:
                break
        if len(dirs) >= max_count:
            break

    if len(dirs) < max_count:
        cd = clean_direction(rng.normal(size=n))
        if cd is not None:
            dirs.append(cd)

    return dirs[:max_count]


def perturb_h(base, rng, eps):
    n = len(base)
    if n == 0:
        return base.copy()
    h = base + float(eps) * rng.normal(size=n)
    return repair_h(h, n / 2.0)


def slsqp_objective(x):
    if time.time() > RUN_DEADLINE:
        return 1.0
    x = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return 1.0
    xc = np.clip(x, 0.0, 1.0)
    r = float(np.sum(xc) - RUN_TARGET_SUM)
    val = float(exact_c(xc))
    val += 1000.0 * r * r
    return val


def slsqp_constraint(x):
    x = np.asarray(x, dtype=np.float64)
    return float(np.sum(x) - RUN_TARGET_SUM)


def local_search(h0, seed, time_s, use_slsqp=True):
    global RUN_TARGET_SUM, RUN_DEADLINE

    start = time.time()
    arr = np.asarray(h0, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    arr = arr.copy()
    n = len(arr)
    target = n / 2.0
    arr = repair_h(arr, target)

    if time_s <= 0.0:
        return arr, float(exact_c(arr))

    deadline = start + max(5.0, float(time_s))
    RUN_TARGET_SUM = target

    best_h, best_c = symmetry_and_shift_best(arr)
    current = best_h.copy()
    current_c = float(best_c)
    current_corr = np.correlate(current, 1.0 - current, mode="full")
    current_q = 1.0 - current

    rng = np.random.default_rng(int(seed))
    pair_i, pair_j = build_pair_list(n)
    num_pairs = int(len(pair_i))
    d_pair = np.zeros(n, dtype=np.float64)

    no_improve = 0
    no_pass = 0

    while time.time() < deadline - 3.0:
        if best_c <= EARLY_TARGET:
            break

        if time.time() < deadline - 5.0:
            dirs = generate_active_directions(current, current_corr, rng, 14)
            for d in dirs:
                if time.time() > deadline - 3.0:
                    break
                res = try_direction(current, d, current_corr, current_c, current_q)
                if res is not None:
                    current, current_c = res
                    current_corr = np.correlate(current, 1.0 - current, mode="full")
                    current_q = 1.0 - current
                    no_improve = 0
                    if current_c < best_c - 1e-12:
                        best_h = current.copy()
                        best_c = float(current_c)
                else:
                    no_improve += 1

        if best_c <= EARLY_TARGET:
            break

        if time.time() < deadline - 8.0 and num_pairs > 0:
            perm = rng.permutation(num_pairs)
            pass_improved = False

            for ci, pi in enumerate(perm):
                if (ci & 63) == 0 and time.time() > deadline - 4.0:
                    break

                i = int(pair_i[pi])
                j = int(pair_j[pi])
                hi = current[i]
                hj = current[j]
                if (hi <= 1e-14 and hj <= 1e-14) or (hi >= 1.0 - 1e-14 and hj >= 1.0 - 1e-14):
                    continue

                d_pair.fill(0.0)
                d_pair[i] = 1.0
                d_pair[j] = -1.0

                res = try_direction(current, d_pair, current_corr, current_c, current_q)
                if res is not None:
                    current, current_c = res
                    current_corr = np.correlate(current, 1.0 - current, mode="full")
                    current_q = 1.0 - current
                    no_improve = 0
                    pass_improved = True
                    if current_c < best_c - 1e-12:
                        best_h = current.copy()
                        best_c = float(current_c)
                else:
                    no_improve += 1
                    if no_improve > 5000 and time.time() > deadline - 30.0:
                        break

            if pass_improved:
                no_pass = 0
            else:
                no_pass += 1

        if (no_pass >= 2 or no_improve > 800) and time.time() < deadline - 8.0:
            for it in range(12):
                if time.time() > deadline - 4.0:
                    break

                if it % 2 == 0:
                    d = rng.normal(size=n)
                else:
                    if n >= 3:
                        k = int(rng.integers(3, min(n, 12) + 1))
                    else:
                        k = max(1, n)
                    idx = rng.choice(n, size=k, replace=False)
                    w = rng.normal(size=k)
                    d = np.zeros(n, dtype=np.float64)
                    d[idx] = w

                res = try_direction(current, d, current_corr, current_c, current_q)
                if res is not None:
                    current, current_c = res
                    current_corr = np.correlate(current, 1.0 - current, mode="full")
                    current_q = 1.0 - current
                    no_improve = 0
                    if current_c < best_c - 1e-12:
                        best_h = current.copy()
                        best_c = float(current_c)
                else:
                    no_improve += 1

            if no_improve > 1000 or current_c > best_c + 1e-4:
                base = best_h if (current_c > best_c + 1e-5 or rng.random() < 0.4) else current
                eps = 0.05 if current_c <= best_c + 1e-5 else 0.08
                current = perturb_h(base, rng, eps)
                current_c = float(exact_c(current))
                current_corr = np.correlate(current, 1.0 - current, mode="full")
                current_q = 1.0 - current
                no_improve = 0
                no_pass = 0

                if current_c > best_c + 5e-4:
                    current = best_h.copy()
                    current_c = float(best_c)
                    current_corr = np.correlate(current, 1.0 - current, mode="full")
                    current_q = 1.0 - current

        if best_c <= EARLY_TARGET:
            break

    if use_slsqp and time.time() < deadline - 40.0 and n <= 90:
        RUN_DEADLINE = deadline - 2.0
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = {"type": "eq", "fun": slsqp_constraint}

        if time.time() < deadline - 60.0:
            attempts = [
                (best_h.copy(), 40, 1200),
                (perturb_h(best_h, rng, 0.03), 30, 800)
            ]
        else:
            attempts = [(best_h.copy(), 30, 800)]

        for x0, maxiter, maxfun in attempts:
            if time.time() > deadline - 20.0:
                break
            try:
                res = optimize.minimize(
                    slsqp_objective,
                    x0,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": int(maxiter), "ftol": 1e-8, "maxfun": int(maxfun), "disp": False}
                )
                if hasattr(res, "x") and np.all(np.isfinite(res.x)):
                    hr = repair_h(res.x, target)
                    cr = float(exact_c(hr))
                    if cr < best_c - 1e-12:
                        best_h = hr
                        best_c = cr
            except Exception:
                pass

    best_h = repair_h(best_h, target)
    best_c = float(exact_c(best_h))
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    total_start = time.time()
    budget = max(20.0, float(budget_s))
    total_deadline = total_start + budget - 3.0
    seed_int = int(seed) if seed is not None else 42

    init = globals().get("initial_h_values", None)
    if init is not None:
        arr = np.asarray(init, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size >= 8:
            h_init = arr.copy()
        else:
            h_init = default_h(63)
    else:
        h_init = default_h(63)

    if len(h_init) > 90:
        h_init = resample_h(h_init, 90)

    h_init = repair_h(h_init, len(h_init) / 2.0)
    c_init = float(exact_c(h_init))
    if c_init <= EARLY_TARGET:
        return h_init, c_init, int(len(h_init))

    if budget > 300.0:
        t_main = budget - 200.0
    else:
        t_main = budget - 25.0
    t_main = max(5.0, min(t_main, budget - 10.0))

    h_best, c_best = local_search(h_init, seed_int, t_main, use_slsqp=True)

    if time.time() < total_deadline - 30.0:
        current_len = int(len(h_best))
        seen = {current_len}
        candidate_ns = [
            current_len + 1,
            current_len - 1,
            64, 62, 60, 66, 58, 68, 56, 72, 80
        ]

        for new_n in candidate_ns:
            if time.time() >= total_deadline - 20.0:
                break
            new_n = int(new_n)
            if new_n in seen:
                continue
            if new_n < 16 or new_n > 90:
                continue
            seen.add(new_n)

            rem = total_deadline - time.time() - 10.0
            if rem <= 25.0:
                break

            alloc = min(45.0, 0.40 * rem)
            if alloc <= 10.0:
                break

            hr = resample_h(h_best, new_n)
            hr, cr = local_search(hr, seed_int + 1000 * new_n, alloc, use_slsqp=False)
            if cr < c_best - 1e-14:
                h_best = hr
                c_best = float(cr)

            if c_best <= EARLY_TARGET:
                break

    h_best = repair_h(h_best, len(h_best) / 2.0)
    c_best = float(exact_c(h_best))
    return h_best, c_best, int(len(h_best))
