def c5_evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2, using L2 projection."""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Bisection on lambda: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -1.0 - h.max()
    hi = 1.0 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual among elements with slack
        if residual > 0:
            # Need to add mass; use elements not at 1
            slack = 1.0 - result
            idx = np.where(slack > 1e-12)[0]
            if len(idx) > 0:
                # Add proportional to slack
                total_slack = slack[idx].sum()
                result[idx] += residual * slack[idx] / total_slack
                result[idx] = np.clip(result[idx], 0, 1)
        else:
            # Need to remove mass; use elements not at 0
            idx = np.where(result > 1e-12)[0]
            if len(idx) > 0:
                total = result[idx].sum()
                result[idx] += residual * result[idx] / total
                result[idx] = np.clip(result[idx], 0, 1)
    # Final exact fix
    residual = target - np.sum(result)
    if abs(residual) > 0:
        # Add to first element with room
        if residual > 0:
            for i in range(n):
                can_add = min(1.0 - result[i], residual)
                if can_add > 0:
                    result[i] += can_add
                    residual -= can_add
                    if abs(residual) < 1e-15:
                        break
        else:
            for i in range(n):
                can_remove = min(result[i], -residual)
                if can_remove > 0:
                    result[i] -= can_remove
                    residual += can_remove
                    if abs(residual) < 1e-15:
                        break
    return result

def optimize_pair(h, p, q, current_c):
    """Find optimal tau for transfer from q to p, and the resulting C value."""
    n = len(h)
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if lo >= hi:
        return h.copy(), current_c
    
    # Compute slopes for all lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    # slope[k] for lag k in 0..2n-2
    nm1 = n - 1
    slopes = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        s = 0.0
        ip = p + k - nm1
        if 0 <= ip < n:
            s += (1.0 - h[ip])
        ip2 = p - k + nm1
        if 0 <= ip2 < n:
            s -= h[ip2]
        iq = q + k - nm1
        if 0 <= iq < n:
            s -= (1.0 - h[iq])
        iq2 = q - k + nm1
        if 0 <= iq2 < n:
            s += h[iq2]
        slopes[k] = s
    
    # Find all breakpoints in [lo, hi]
    # tau_ij = (corr[j] - corr[i]) / (slopes[i] - slopes[j])
    dx = 2.0 / n
    corr_scaled = corr * dx  # This is what we're maximizing
    
    # Vectorized computation of all intersection points
    dc = corr_scaled[None, :] - corr_scaled[:, None]  # dc[i,j] = corr[j] - corr[i]
    ds = slopes[:, None] - slopes[None, :]  # ds[i,j] = slope[i] - slope[j]
    
    # Avoid division by zero

    tau_matrix = np.full_like(dc, np.nan)

    taus = tau_matrix[valid].flatten()
    taus = taus[(taus >= lo) & (taus <= hi)]

    candidates.add(lo)
    candidates.add(hi)
    if lo <= 0 <= hi:

    new_h = np.clip(new_h, 0, 1)
    
    # Verify exact

        if time.monotonic() >= hard_end:
            break
        tau = optimal_tau(h, p, q, corr_dx)
        if abs(tau) > 1e-12:
            new_h = h.copy()
            new_h[p] += tau
            new_h[q] -= tau
            new_h = np.clip(new_h, 0.0, 1.0)
            if abs(np.sum(new_h) - n / 2.0) < 1e-9:
                new_c = c5_evaluate(new_h)
                if new_c < current_c - 1e-13:
                    h = new_h
                    current_c = new_c
                    corr_dx = h * 0.0  # will recompute
                    corr_dx = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
                    improved = True
    return h, current_c, improved

def interpolate_to(h, target_n):
    old_n = len(h)
    old_dx = 2.0 / old_n
    new_dx = 2.0 / target_n
    new_h = np.zeros(target_n)
    for i in range(target_n):
        x0 = i * new_dx
        x1 = x0 + new_dx
        mass = 0.0
        for j in range(old_n):
            jx0 = j * old_dx
            jx1 = jx0 + old_dx
            overlap = min(x1, jx1) - max(x0, jx0)
            if overlap > 0:
                mass += h[j] * overlap
        new_h[i] = mass / new_dx
    new_h = np.clip(new_h, 0.0, 1.0)
    # Project to exact sum
    target = target_n / 2.0
    resid = target - np.sum(new_h)
    if abs(resid) > 1e-12:
        new_h = project_balanced(new_h, target_n)
    return new_h

def project_balanced(h, n):
    target = n / 2.0
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(result)
    if abs(resid) > 1e-12:
        if resid > 0:
            for i in range(n):
                add = min(1.0 - result[i], resid)
                if add > 0:
                    result[i] += add
                    resid -= add
                    if resid < 1e-14:
                        break
        else:
            for i in range(n):
                rem = min(result[i], -resid)
                if rem > 0:
                    result[i] -= rem
                    resid += rem
                    if -resid < 1e-14:
                        break
    return result

def random_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    return project_balanced(h, n)

def perturbed_initial(h, rng, strength):
    n = len(h)
    h_new = h + rng.normal(0, strength, n)
    return project_balanced(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.monotonic()
    hard_end = t_start + budget_s - 3.0
    
    rng = np.random.RandomState(seed)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = random_initial(45, seed)
    
    n = len(best_h)
    best_c = c5_evaluate(best_h)
    
    # Phase 1: Optimize at initial n
    h = best_h.copy()
    current_c = best_c
    for phase in range(3):
        for _ in range(50):
            if time.monotonic() >= hard_end - 200:
                break
            h, current_c, improved = full_pass(h, rng, t_start, hard_end - 200)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            if not improved and phase > 0:
                break
        # Perturb to escape local optimum
        if time.monotonic() < hard_end - 200 and phase < 2:
            h_pert = perturbed_initial(current_h if 'current_h' in dir() else h, rng, 0.05)
            h_pert = project_balanced(h_pert, n)
            c_pert = c5_evaluate(h_pert)
            if c_pert < current_c:
                h = h_pert
                current_c = c_pert
    
    # Continue with remaining time at current n
    while time.monotonic() < hard_end - 60:
        h, current_c, improved = full_pass(h, rng, t_start, hard_end - 60)
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        if not improved:
            break
    
    # Phase 2: Try higher resolution
    for target_n in [90, 120]:
        if time.monotonic() >= hard_end - 80:
            break
        h_interp = interpolate_to(best_h, target_n)
        c_interp = c5_evaluate(h_interp)
        if c_interp < best_c:
            best_c = c_interp
            best_h = h_interp.copy()
        # Optimize at new resolution
        h_new = h_interp.copy()
        while time.monotonic() < hard_end - 40:
            h_new, c_new, improved = full_pass(h_new, rng, t_start, hard_end - 40)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            if not improved:
                break
    
    # Phase 3: Final polish at best resolution
    h_final = best_h.copy()
    n_final = len(h_final)
    c_final = c5_evaluate(h_final)
    while time.monotonic() < hard_end - 5:
        h_final, c_final, improved = full_pass(h_final, rng, t_start, hard_end - 5)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
        if not improved:
            break
    
    # Return best
    final_c = c5_evaluate(best_h)
    return (best_h, final_c, len(best_h))
