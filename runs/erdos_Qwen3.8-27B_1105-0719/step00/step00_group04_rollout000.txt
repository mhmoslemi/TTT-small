import numpy as np
import time
import sys
from scipy.optimize import minimize

def exact_score(h):
    """Official evaluation: max of correlate * dx"""
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    """Get all correlation lags scaled by dx"""
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    """Get gradient matrix G where G[k, j] = d(F_k*dx)/dh[j]
    F_k is the k-th element of correlate(h, 1-h, 'full')"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1  # number of lags
    G = np.zeros((M, n))
    
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            # F = sum_{i=0}^{n-1-delta} h[i+delta]*(1-h[i])
            # dF/dh[j] = (1-h[j-delta])*[delta<=j<=n-1] - h[j+delta]*[0<=j<=n-1-delta]
            # First term: j from delta to n-1
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n-delta])
            # Second term: j from 0 to n-1-delta
            if delta < n:
                G[k, 0:n-delta] -= dx * h[delta:n]
        else:
            d = -delta
            # F = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dF/dh[j] = (1-h[j+d])*[0<=j<=n-1-d] - h[j-d]*[d<=j<=n-1]
            if d < n:
                G[k, 0:n-d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n-d]
    
    return G

def repair_feasible(h, target=None):
    """Ensure h is in [0,1] and sum(h) = target (= n/2 by default)"""
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass; add to entries with most room
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        # Need to remove mass
        need = -diff
        # Remove from entries with most value
        order = np.argsort(-h)
        remaining = need
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_epigraph_slsqp(h_start, t_start, beta=0.0, maxiter=80, time_limit=60):
    """Solve the epigraph NLP using SLSQP"""
    n = len(h_start)
    M = 2 * n - 1
    dx = 2.0 / n
    
    def objective(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def objective_jac(x):
        h = x[:n]
        t = x[n]
        grad = np.zeros(n + 1)
        grad[:n] = beta * (1.0 - 2.0 * h)
        grad[n] = 1.0
        return grad
    
    def constraints_fun(x):
        h = x[:n]
        t = x[n]
        # Equality: sum(h) - n/2 = 0
        eq_val = h.sum() - n / 2.0
        # Inequalities: t - F_k * dx >= 0 for all k
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        ineq_vals = t - prof
        return np.concatenate([[eq_val], ineq_vals])
    
    def constraints_jac(x):
        h = x[:n]
        t = x[n]
        G = get_lag_gradients(h)
        # Equality gradient: [1,1,...,1, 0]
        eq_grad = np.zeros(n + 1)
        eq_grad[:n] = 1.0
        # Inequality gradients: for constraint (t - F_k*dx >= 0):
        # d/dh[j] = -G[k,j], d/dt = 1
        ineq_grad = np.zeros((M, n + 1))
        ineq_grad[:, :n] = -G
        ineq_grad[:, n] = 1.0
        # Full Jacobian: rows are [eq_grad; ineq_grad rows]
        J = np.zeros((1 + M, n + 1))
        J[0, :] = eq_grad
        J[1:, :] = ineq_grad
        return J
    
    # Set up bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        start_time = time.perf_counter()
        result = minimize(
            objective, x0, jac=objective_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{
                'type': 'ineq',
                'fun': constraints_fun,
                'jac': constraints_jac
            }],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        elapsed = time.perf_counter() - start_time
    except Exception:
        return None, 0.0
    
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_sol = repair_feasible(result.x[:n])
        score = exact_score(h_sol)
        return h_sol, elapsed
    return None, time.perf_counter() - start_time

def local_gradient_descent(h_current, max_iters=40, time_limit=30):
    """Active-set gradient descent on the max of correlation lags"""
    n = len(h_current)
    h = h_current.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    start_time = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        
        prof = get_profile(h)
        score = np.max(prof)
        
        # Find active set (top lags)
        # Use top A lags
        A = max(5, min(30, n // 3))
        active_idx = np.argsort(prof)[::-1][:A]
        
        # Compute weighted average gradient of active lags
        G = get_lag_gradients(h)
        weights = prof[active_idx]  # weight by magnitude
        weights = weights / (weights.sum() + 1e-15)
        
        # Direction: negative weighted gradient, projected to sum-zero
        direction = -np.sum(weights[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()  # ensure sum zero
        
        # Normalize
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            break
        direction /= norm
        
        # Line search
        found = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = h + alpha * direction
            h_trial = repair_feasible(h_trial)
            score_trial = exact_score(h_trial)
            if score_trial < best_score - 1e-10:
                h = h_trial
                best_h = h.copy()
                best_score = score_trial
                found = True
                break
        
        if not found:
            break
    
    return best_h, best_score

def interpolate_to_dimension(vec, new_len):
    """Interpolate a step function to a new dimension"""
    n = len(vec)
    # Source centers
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    # Target centers
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    
    # Extend for interpolation: add boundary points
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    return h_new

def symmetrize(h):
    """Average h with its reversal (preserves feasibility)"""
    h_sym = 0.5 * (h + h[::-1].copy())
    h_sym = repair_feasible(h_sym)
    return h_sym

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def solve_epigraph(h_start, t_start, beta=0.0, maxiter=80, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M

    # Precompute which lags to use (top |nc| by current profile value)
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])

    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def local_descent(h_current, max_iters=50, time_limit=30):
    n = len(h_current)
    h = h_current.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()

        # Top-A active lags
        A = max(5, min(25, n // 3))
        active_idx = np.argsort(prof)[::-1][:A]
        G = get_lag_gradients(h)
        w = prof[active_idx].copy()
        w = np.maximum(w, 0)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum

        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm

        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(h + alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-11:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)
