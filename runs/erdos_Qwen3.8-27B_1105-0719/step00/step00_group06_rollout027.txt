import numpy as np

def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_feasible(h, target=None):
    """Make h feasible: 0 <= h[i] <= 1, sum(h) = target (= n/2)"""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    current_sum = h.sum()
    diff = target - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute the adjustment
    if diff > 0:
        # Need to increase: add to elements that are < 1
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        add = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + add, 1.0)
        # Re-adjust if clipping caused issue
        remaining = target - h.sum()
        if remaining > 1e-10:
            mask2 = h < 1.0
            if mask2.sum() > 0:
                h[mask2] += remaining / mask2.sum()
                h = np.minimum(h, 1.0)
    else:
        # Need to decrease: subtract from elements that are > 0
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        sub = -diff / mask.sum()
        h[mask] = np.maximum(h[mask] - sub, 0.0)
        remaining = h.sum() - target
        if remaining > 1e-10:
            mask2 = h > 0.0
            if mask2.sum() > 0:
                h[mask2] -= remaining / mask2.sum()
                h = np.maximum(h, 0.0)
    return h

def pair_exchange_step(h, u, v, n_lags=None):

    # Quadratic coeff: +1 if ∃i: i+L=u, i=v (i.e., L=u-v, valid); -1 if ∃i: i+L=v, i=u (i.e., L=v-u, valid)
    
    a = np.zeros(n_corr)
    quad_lags = []  # list of (index_in_corr, quadratic_coeff)
    
    for L in range(-(n-1), n):
        k = L + (n - 1)  # index in correlation array
        # Valid range of i: max(0, -L) <= i <= min(n-1, n-1-L)
        i_start = max(0, -L)

        # Check if u or v are in the product for any valid i

        return h, best_val, 0.0

    h_new[u] += best_s

def make_binary_seed(n, seed):
    """Create a balanced binary vector with n/2 ones."""
    rng = np.random.RandomState(seed)
    positions = rng.choice(n, size=n//2, replace=False)
    b = np.zeros(n)
    b[positions] = 1.0
    return b

def continuousize_binary(b, eps=0.05):
    """Convert binary to continuous: 0->eps, 1->1-eps, preserving sum."""
    n = len(b)
    h = b * (1.0 - eps) + (1.0 - b) * eps
    return h

def sturmian_seed(n, alpha, offset=0.0):
    """Sturmian/Beatty sequence: h[i] = 1 if frac(i*alpha + offset) < 0.5"""
    # Actually for balanced: use the property that sum = n/2
    # Simple approach: use a mechanical word
    i = np.arange(n)
    # Use a irrational slope for good distribution
    h = ((i * alpha + offset) % 1.0) < 0.5
    h = h.astype(float)
    return h

def interpolate_to_n(h, n_new):
    """Interpolate h to a new length n_new using linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h)
    return repair_feasible(h_new, n_new / 2.0)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] = np.min(h[mask] + diff / cnt, 1.0)
    else:
        mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] = np.max(h[mask] + diff / cnt, 0.0)
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += diff2 / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff2 / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

def find_optimal_s(h, u, v):
    """Find optimal s minimizing max correlation for h[u]+=s, h[v]-=s."""
    n = len(h)
    s_min = max(-h[u], h[v] - 1.0) + 1e-10
    s_max = min(1.0 - h[u], h[v]) - 1e-10
    if s_min >= s_max:
        return 0.0
    
    base_corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    
    # Compute linear and quadratic coefficients
    a = np.zeros(n_corr)
    q = np.zeros(n_corr)
    
    for L in range(-(n-1), n):
        k = L + n - 1
        i_lo = max(0, -L)
        i_hi = min(n-1, n-1-L)
        
        lin = 0.0
        # h[i+L] contributes +s if i+L=u (i=u-L), -s if i+L=v (i=v-L)
        # (1-h[i]) contributes -s if i=u, +s if i=v
        # Linear part: [δ(i+L=u) - δ(i+L=v)]*(1-h[i]) + h[i+L]*[δ(i=v) - δ(i=u)]
        
        ui = u - L
        if i_lo <= ui <= i_hi:
            lin += (1.0 - h[ui])
        vi = v - L
        if i_lo <= vi <= i_hi:
            lin -= (1.0 - h[vi])
        if i_lo <= u <= i_hi:
            lin -= h[u + L]
        if i_lo <= v <= i_hi:
            lin += h[v + L]
        a[k] = lin
        
        # Quadratic: from cross term [δ(i+L=u) - δ(i+L=v)] * [-δ(i=u) + δ(i=v)] * s^2
        # = δ(i+L=u)*δ(i=v) - δ(i+L=u)*δ(i=u) - δ(i+L=v)*δ(i=v) + δ(i+L=v)*δ(i=u)
        # δ(i+L=u)*δ(i=v): i=u-L and i=v, so u-L=v, L=u-v. And i=v must be valid.
        # δ(i+L=u)*δ(i=u): i=u-L and i=u, so L=0. And i=u must be valid.
        # δ(i+L=v)*δ(i=v): i=v-L and i=v, so L=0. And i=v must be valid.
        # δ(i+L=v)*δ(i=u): i=v-L and i=u, so v-L=u, L=v-u. And i=u must be valid.
        
        if L == 0:
            # δ(i+L=u)*δ(i=u) = 1 if i=u is valid, i.e., i_lo <= u <= i_hi
            if i_lo <= u <= i_hi:
                q[k] -= 1.0
            # δ(i+L=v)*δ(i=v) = 1 if i=v is valid
            if i_lo <= v <= i_hi:
                q[k] -= 1.0
            # δ(i+L=u)*δ(i=v): L=u-v must be 0, so u=v. But u≠v by assumption.
            # δ(i+L=v)*δ(i=u): L=v-u must be 0, so u=v. Same.
        elif L == u - v:
            # δ(i+L=u)*δ(i=v): i=v, L=u-v. Valid if i_lo <= v <= i_hi.
            if i_lo <= v <= i_hi:
                q[k] += 1.0
        elif L == v - u:
            # δ(i+L=v)*δ(i=u): i=u, L=v-u. Valid if i_lo <= u <= i_hi.
            if i_lo <= u <= i_hi:
                q[k] += 1.0
    
    # Minimize max_k(base_corr[k] + a[k]*s + q[k]*s^2) over [s_min, s_max]
    # This is convex, use ternary search
    
    def obj(s):
        return np.max(base_corr + a * s + q * s * s)
    
    # Check s=0
    f0 = obj(0.0)
    
    lo, hi = s_min, s_max

        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    
    s_opt = (lo + hi) / 2.0
    if obj(s_opt) < f0:
        return s_opt
    return 0.0

def find_optimal_s(h, u, v):
    n = len(h)
    s_min = max(-h[u], h[v] - 1.0) + 1e-10
    s_max = min(1.0 - h[u], h[v]) - 1e-10
    if s_min >= s_max:
        return 0.0
    
    base_corr = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    
    a = np.zeros(nc)
    q = np.zeros(nc)
    
    # Quadratic coefficients
    q[n-1] = -2.0  # lag 0 is at index n-1
    q[u - v + n - 1] = 1.0  # lag u-v
    q[v - u + n - 1] = 1.0  # lag v-u
    
    # Linear coefficients
    for L in range(-(n-1), n):
        i_lo = max(0, -L)
        i_hi = min(n-1, n-1-L)
        k = L + n - 1
        
        lin = 0.0
        ui = u - L
        if i_lo <= ui <= i_hi and 0 <= ui <= n-1:
            lin += (1.0 - h[ui])
        vi = v - L
        if i_lo <= vi <= i_hi and 0 <= vi <= n-1:
            lin -= (1.0 - h[vi])
        if i_lo <= u <= i_hi:
            lin -= h[u + L]
        if i_lo <= v <= i_hi:
            lin += h[v + L]
        a[k] = lin
    
    # Ternary search for min of max(base_corr + a*s + q*s^2)
    def obj(s):
        return np.max(base_corr + a*s + q*s*s)
    
    f0 = obj(0.0)
    lo, hi = s_min, s_max
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    s_opt = (lo + hi) / 2.0
    if obj(s_opt) < f0:
        return s_opt
    return 0.0
