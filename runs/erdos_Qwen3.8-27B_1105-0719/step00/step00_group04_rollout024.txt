        tau *= tau_decay
        lr *= lr_decay

But wait, `compute_c5` involves `np.correlate` which is O(n^2) for "full" mode. So each evaluation is O(n^2), and the local search is O(n^2 * n^2) = O(n^4) per full pass. For n=70, that's about 70^4 = 24 million. With max_iters=1000, that could be 24 billion operations. Too slow.

Let me optimize: instead of computing the full correlation each time, I can update it incrementally when swapping two positions.

When we swap positions p (1→0) and q (0→1):
- For each lag L, F(L) changes.
- The change depends on the neighbors of p and q at distance L.

Actually, let me think about this differently. The correlation at lag L (in the np.correlate sense, index k = L + n - 1) is:
F(L) = dx * sum_{i valid} h[i] * (1-h[i+L])

When we change h[p] from 1 to 0 and h[q] from 0 to 1:
- For any term involving h[p]: it contributes h[p]*(1-h[...]) = 1*(1-h[...]) before, 0*(1-h[...]) = 0 after. Change = -(1-h[...])
- For any term involving h[q] in the (1-h[...]) part: it contributes h[...]*(1-h[q]) = h[...]*1 before, h[...]*0 = 0 after. Change = -h[...]
- For any term involving h[q] as the first factor: 0*(1-h[...]) = 0 before, 1*(1-h[...]) after. Change = (1-h[...])
- For any term involving h[p] in the (1-h[...]) part: h[...]*(1-1) = 0 before, h[...]*(1-0) = h[...] after. Change = h[...]

This is getting complex. Let me just compute the delta for each lag.

For lag L (meaning we look at pairs (i, i+L)):
- h[p] appears as first factor when i=p: requires p+L in [0,n-1]
  Before: 1*(1-h[p+L]), After: 0*(1-h[p+L]) → delta = -(1-h[p+L])
- h[p] appears as second factor (in 1-h) when i+L=p, i=p-L: requires p-L in [0,n-1]
  Before: h[p-L]*(1-1) = 0, After: h[p-L]*(1-0) = h[p-L] → delta = h[p-L]
- h[q] appears as first factor when i=q: requires q+L in [0,n-1]
  Before: 0*(1-h[q+L]) = 0, After: 1*(1-h[q+L]) → delta = (1-h[q+L])
- h[q] appears as second factor when i+L=q, i=q-L: requires q-L in [0,n-1]
  Before: h[q-L]*(1-0) = h[q-L], After: h[q-L]*(1-1) = 0 → delta = -h[q-L]

So delta_F(L) = dx * [-(1-h[p+L])*I(p+L valid) + h[p-L]*I(p-L valid) + (1-h[q+L])*I(q+L valid) - h[q-L]*I(q-L valid)]

This is O(1) per lag, so O(n) per swap evaluation. For n=70, that's 70 operations per swap. With 35*35 = 1225 possible swaps, one full pass is about 86,000 operations. Very fast!

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full") * dx
    return float(np.max(corr))

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    return np.correlate(h, b, mode="full") * dx

def project_to_feasible(h, n):
    target = n / 2.0
    lo = h.min() - target
    hi = h.max() + target
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    return np.clip(h - alpha, 0.0, 1.0)

def surrogate_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full") * dx
    m = corr.max()
    exp_corr = np.exp((corr - m) / tau)
    Z = exp_corr.sum()
    obj = m + tau * np.log(Z)
    weights = exp_corr / Z
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = weights[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif L > 0:
            nL = n - L
            grad[:nL] += wk * dx * (1.0 - h[L:L+nL])
            grad[L:] += -wk * dx * h[:nL]
        else:
            L2 = -L
            nL = n - L2
            grad[L2:] += wk * dx * (1.0 - h[:nL])
            grad[:nL] += -wk * dx * h[L2:]
    
    return obj, grad

def smooth_optimize(h, n, time_limit, tau_init=0.05, tau_decay=0.995, lr_init=0.01, lr_decay=0.999):
    """Projected gradient descent on log-sum-exp surrogate."""
    start = time.time()
    h_curr = project_to_feasible(h.copy(), n)
    tau = tau_init
    lr = lr_init
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    
    while time.time() - start < time_limit:
        obj, grad = surrogate_and_grad(h_curr, tau)
        h_new = h_curr - lr * grad
        h_new = project_to_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
        tau = max(tau * tau_decay, 0.001)
        lr = max(lr * lr_decay, 0.0001)
    
    return best_h, best_c5

def binary_local_search(h_bin, n, max_passes=200):

    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_all_corrs(h)
    best_c5 = float(np.max(corr))
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    for _ in range(max_passes):
        improved = False
        for i in range(len(ones)):
            for j in range(len(zeros)):
                p, q = ones[i], zeros[j]
                if p == q:
                    continue
                # Compute delta for each lag
                # h[p]: 1->0, h[q]: 0->1
                # For lag L: delta = dx * [-(1-h[p+L])*I(p+L valid) + h[p-L]*I(p-L valid) 
                #                       + (1-h[q+L])*I(q+L valid) - h[q-L]*I(q-L valid)]
                # But we need to be careful: the "lag" in np.correlate terms
                # k = L + (n-1), where L is the actual shift
                
                # Let me use the direct formula:
                # corr[k] = sum_i h[i] * (1-h)[i + k - (n-1)]
                # Let L = k - (n-1)
                # corr[k] = sum_i h[i] * (1-h[i+L]) for valid i
                
                # Changing h[p] from 1 to 0:
                #   affects terms where i=p: (1-h[p+L]) -> 0, delta = -(1-h[p+L]) if p+L valid
                #   affects terms where i+L=p, i=p-L: h[p-L]*(1-1)=0 -> h[p-L]*(1-0)=h[p-L], delta = h[p-L] if p-L valid

                #   affects terms where i=q: 0*(1-h[q+L])=0 -> 1*(1-h[q+L]), delta = (1-h[q+L]) if q+L valid

                # Instead of computing per-lag, let me just compute the full correlation change
                # Actually for efficiency, let me precompute which lags are affected
                
                # Simpler: just recompute the max after the swap
                # This is O(n) for the correlate, which is fine for n=70
                h[p], h[q] = 0.0, 1.0
                new_corr = compute_all_corrs(h)
                new_max = float(np.max(new_corr))

                    corr = new_corr
                    ones[i] = q
                    zeros[j] = p

                    break
                else:
                    h[p], h[q] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full") * dx
    return float(np.max(corr))


def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    return np.correlate(h, b, mode="full") * dx


def project_to_feasible(h, n):
    target = n / 2.0
    lo = h.min() - target
    hi = h.max() + target
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    return np.clip(h - alpha, 0.0, 1.0)


def surrogate_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full") * dx
    m = corr.max()
    exp_corr = np.exp((corr - m) / tau)
    Z = exp_corr.sum()
    obj = m + tau * np.log(Z)
    weights = exp_corr / Z
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = weights[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif L > 0:
            nL = n - L
            grad[:nL] += wk * dx * (1.0 - h[L:L + nL])
            grad[L:] += -wk * dx * h[:nL]
        else:
            L2 = -L
            nL = n - L2
            grad[L2:] += wk * dx * (1.0 - h[:nL])
            grad[:nL] += -wk * dx * h[L2:]
    return obj, grad


def smooth_optimize(h, n, time_limit, tau_init=0.03, tau_decay=0.999, lr_init=0.02, lr_decay=0.9995):
    start = time.time()
    h_curr = project_to_feasible(h.copy(), n)
    tau = tau_init
    lr = lr_init
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    while time.time() - start < time_limit:
        obj, grad = surrogate_and_grad(h_curr, tau)
        h_new = h_curr - lr * grad
        h_new = project_to_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
        tau = max(tau * tau_decay, 0.0005)
        lr = max(lr * lr_decay, 0.00001)
    return best_h, best_c5


def binarize(h, n):
    k = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:k]] = 1.0
    return h_bin


def binary_local_search(h_bin, n, max_passes=500):
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    for _ in range(max_passes):
        improved = False
        for i in range(len(ones)):
            p = ones[i]
            for j in range(len(zeros)):
                q = zeros[j]
                if p == q:
                    continue
                h[p] = 0.0
                h[q] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    ones[i] = q
                    zeros[j] = p
                    improved = True
                    break
                else:
                    h[p] = 1.0
                    h[q] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5
