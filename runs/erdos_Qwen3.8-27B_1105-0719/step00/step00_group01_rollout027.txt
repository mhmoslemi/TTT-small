import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta

    def value(self, x):
        return logsumexp(self.beta * compute_corrs(x)) / self.beta * self.dx

class ConstrFunc:
    def __init__(self, target_sum, n):
        self.target_sum = target_sum
        self.n = n

    def fun(self, x):
        return x.sum() - self.target_sum

    def jac(self, x):
        return np.ones(self.n)


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    objf = ObjFunc(dx, beta)
    cf = ConstrFunc(target_sum, n)
    cons = [{'type': 'eq', 'fun': cf.fun, 'jac': cf.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf.value, h0, jac=objf.gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
