import numpy as np
import time
from scipy.optimize import minimize

def evaluate_score(h):
    """Exact score as defined by the problem."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def get_profile(h):
    """Get all lag values (times dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(vec):
    """Repair a vector to satisfy 0<=v<=1 and sum = n/2."""
    n = len(vec)
    x = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    # Iterative repair
    for _ in range(100):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum
            mask = x < 1.0
            if not mask.any():
                break
            per = diff / mask.sum()
            x[mask] = np.minimum(x[mask] + per, 1.0)
        else:
            # Need to decrease sum
            mask = x > 0.0
            if not mask.any():
                break
            per = diff / mask.sum()
            x[mask] = np.maximum(x[mask] + per, 0.0)
    # Final exact fix
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) > 1e-15:
        # Put remainder in a free coordinate
        diff = target - s
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                x[idx[0]] += min(diff, 1.0 - x[idx[0]])
        else:
            idx = np.where(x > 0.0)[0]
            if len(idx) > 0:
                x[idx[0]] += max(diff, -x[idx[0]])
    return x

def compute_lag_gradients(h):
    """
    Compute gradient matrix G of shape (M, n) where M = 2n-1.
    G[p, j] = d(F_p)/dh_j where F_p = dx * output[p]
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for p in range(M):
        lag = p - (n - 1)
        # d(F_p)/dh_j = dx * [(1-h[j-lag]) * I(0<=j-lag<n) - h[j+lag] * I(0<=j+lag<n)]
        j = np.arange(n)
        j1 = j - lag
        j2 = j + lag
        g = np.zeros(n)
        mask1 = (j1 >= 0) & (j1 < n)
        g[mask1] += dx * (1.0 - h[j1[mask1]])
        mask2 = (j2 >= 0) & (j2 < n)
        g[mask2] -= dx * h[j2[mask2]]
        G[p, :] = g
    return G

def interpolate_to_len(vec, new_len):
    """Interpolate a step function to a new number of samples."""
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    # Extend with boundary values for interpolation
    xp = np.concatenate([[0.0], x_src, [2.0]])
    fp = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, xp, fp)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

class EpigraphState:
    def __init__(self, h0, beta=0.0):
        self.n = len(h0)
        self.dx = 2.0 / self.n
        self.M = 2 * self.n - 1
        self.h0 = h0.copy()
        self.beta = beta
        self.lags = np.arange(self.M) - (self.n - 1)

    def fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        val = t
        if self.beta > 0:
            val += self.beta * np.sum(h * (1.0 - h))
        return val

    def jac(self, x):
        n = self.n
        h = x[:n]
        g = np.zeros(n + 1)
        if self.beta > 0:
            g[:n] = self.beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def constraint_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        profile = get_profile(h)
        # Equality constraint
        eq = np.sum(h) - self.n / 2.0
        # Inequality constraints: t - profile[p] >= 0
        ineq = t - profile
        return np.concatenate([[eq], ineq])

    def constraint_jac(self, x):
        h = x[:self.n]
        G = compute_lag_gradients(h)
        n = self.n
        M = self.M
        # Build Jacobian: (M+1, n+1)
        # Row 0: equality sum(h) - n/2 = 0
        # Rows 1..M: t - profile[p] >= 0
        J = np.zeros((M + 1, n + 1))
        J[0, :n] = 1.0
        for p in range(M):
            J[p + 1, :n] = -G[p, :]
            J[p + 1, n] = 1.0
        return J

def solve_epigraph(h0, beta=0.0, maxiter=80, ftol=1e-10):
    """Solve the epigraph NLP."""
    state = EpigraphState(h0, beta=beta)
    n = state.n
    M = state.M
    profile0 = get_profile(h0)
    t0 = float(np.max(profile0))
    
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = t0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.5)]
    
    # Build constraints in scipy format
    # We'll use a single constraint dict with fun returning array
    cons = {
        'type': 'ineq',
        'fun': state.constraint_fun,
        'jac': state.constraint_jac
    }
    
    # Actually, SLSQP needs separate equality and inequality.
    # Let's use two constraints.
    
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        return t - get_profile(h)
    
    def ineq_jac(x):
        h = x[:n]
        G = compute_lag_gradients(h)
        J = np.zeros((M, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    cons_eq = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    cons_ineq = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    
    result = minimize(
        state.fun, x0, method='SLSQP', jac=state.jac,
        bounds=bounds, constraints=[cons_eq, cons_ineq],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return result

def local_gradient_descent(h, max_steps=30, base_alpha=0.02):
    """Local descent using active lag gradients."""
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_score = evaluate_score(h)
    profile = get_profile(h)
    
    for step in range(max_steps):
        # Recompute profile
        profile = get_profile(best)
        score = float(np.max(profile))
        
        if score < best_score - 1e-13:
            best_score = score
            continue
        
        # Find active lags (near-maximum)
        threshold = score - 0.01
        active_idx = np.where(profile >= threshold)[0]
        
        if len(active_idx) == 0:
            active_idx = [np.argmax(profile)]
        
        # Compute average gradient of active lags
        G = compute_lag_gradients(best)
        g_active = G[active_idx, :].mean(axis=0)
        
        # Project to zero-sum direction
        g_active = g_active - g_active.mean()
        
        # Normalize
        norm = np.linalg.norm(g_active)
        if norm < 1e-12:
            break
        g_active = g_active / norm
        
        # Try several step sizes
        improved = False
        for alpha in [base_alpha, base_alpha * 0.5, base_alpha * 0.25, base_alpha * 0.1,
                      base_alpha * 2, base_alpha * 4]:
            h_new = best - alpha * g_active
            h_new = repair_feasible(h_new)
            s_new = evaluate_score(h_new)
            if s_new < best_score - 1e-14:
                best = h_new
                best_score = s_new
                improved = True
                break
        
        if not improved:
            base_alpha *= 0.5
            if base_alpha < 1e-6:
                break
    
    return best, best_score

def local_random_perturb(h, n_trials=20, eps=0.01):
    """Try random perturbations."""
    n = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.default_rng(12345)
    
    for _ in range(n_trials):

            s_new = evaluate_score(h_new)
            if s_new < best_score - 1e-14:

                best_score = s_new
                break

def coordinate_descent(h, n_rounds=5):

    n = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    
    for rnd in range(n_rounds):

                    improved_any = True
                    break

def evaluate_score(h):

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(vec):
    n = len(vec)
    x = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(200):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = x < 1.0
            if not mask.any():
                break
            per = min(diff / mask.sum(), 0.5)
            x[mask] = np.minimum(x[mask] + per, 1.0)
        else:
            mask = x > 0.0
            if not mask.any():
                break
            per = min(-diff / mask.sum(), 0.5)
            x[mask] = np.maximum(x[mask] - per, 0.0)
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                add = min(diff, 1.0 - x[idx[0]])
                x[idx[0]] += add
                diff -= add
                if diff > 1e-14 and len(idx) > 1:
                    x[idx[1]] += min(diff, 1.0 - x[idx[1]])
        else:

                x[idx[0]] -= sub
                diff += sub

    return x

def compute_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    j = np.arange(n)
    for p in range(M):
        lag = p - (n - 1)
        g = np.zeros(n)
        j1 = j - lag
        mask1 = (j1 >= 0) & (j1 < n)
        g[mask1] += dx * (1.0 - h[j1[mask1]])
        j2 = j + lag
        mask2 = (j2 >= 0) & (j2 < n)
        g[mask2] -= dx * h[j2[mask2]]
        G[p] = g
    return G

def interpolate_to_len(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xp = np.concatenate([[0.0], x_src, [2.0]])
    fp = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, xp, fp)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_epigraph(h0, beta=0.0, maxiter=60, ftol=1e-10):
    state = EpigraphState(h0, beta=beta)
    n = state.n
    M = state.M
    profile0 = get_profile(h0)
    t0 = float(np.max(profile0)) + 1e-10

    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = t0

    bounds = [(0.0, 1.0)] * n + [(0.0, max(1.0, t0 + 0.5))]

    cons_eq = {'type': 'eq', 'fun': state.eq_fun, 'jac': state.eq_jac}
    cons_ineq = {'type': 'ineq', 'fun': state.ineq_fun, 'jac': state.ineq_jac}
    constraints = [cons_eq, cons_ineq]

    try:
        result = minimize(
            state.fun, x0, method='SLSQP', jac=state.jac,
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return result
    except Exception:
        return None

def local_gradient_descent(h, max_steps=50, base_alpha=0.03):
    best = h.copy()
    best_score = evaluate_score(h)

    for step in range(max_steps):
        profile = get_profile(best)
        score = float(np.max(profile))
        if score < best_score:
            best_score = score

        threshold = score - 0.02
        active_idx = np.where(profile >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(profile)])

        G = compute_lag_gradients(best)
        g_active = G[active_idx, :].mean(axis=0)
        g_active = g_active - g_active.mean()
        norm = np.linalg.norm(g_active)
        if norm < 1e-12:
            break
        g_active = g_active / norm

        improved = False
        for alpha in [base_alpha, base_alpha * 0.5, base_alpha * 0.25,
                      base_alpha * 2.0, base_alpha * 4.0, base_alpha * 0.1]:
            h_new = best - alpha * g_active
            h_new = repair_feasible(h_new)
            s_new = evaluate_score(h_new)
            if s_new < best_score - 1e-14:
                best = h_new
                best_score = s_new
                improved = True
                break

        if not improved:
            base_alpha *= 0.5
            if base_alpha < 1e-7:
                break

    return best, best_score

def local_random_perturb(h, n_trials=30, eps=0.015, seed_val=42):
    best = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.default_rng(seed_val)
    n = len(best)

    for _ in range(n_trials):
        d = rng.standard_normal(n)
        d = d - d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            continue
        d = d / dn
        for alpha in [eps, eps * 0.5, eps * 2.0]:
            h_new = best + alpha * d
            h_new = repair_feasible(h_new)
            s_new = evaluate_score(h_new)
            if s_new < best_score - 1e-14:
                best = h_new
                best_score = s_new
                break

    return best, best_score

            for delta in deltas:
                h_new = best.copy()
                h_new[i] += delta
                if h_new[i] < -1e-12 or h_new[i] > 1.0 + 1e-12:
                    continue
                h_new = repair_feasible(h_new)
                s = evaluate_score(h_new)
                if s < best_score - 1e-14:
                    best = h_new
                    best_score = s
                    improved_any = True
                    break
        if not improved_any:
            break
    return best, best_score

def blend_candidates(h1, h2, n_points=20):
    best = h1.copy()
    best_score = evaluate_score(h1)
    for a in np.linspace(0.0, 1.0, n_points):
        h_mix = (1.0 - a) * h1 + a * h2
        h_mix = repair_feasible(h_mix)
        s = evaluate_score(h_mix)
        if s < best_score - 1e-14:
            best_score = s
            best = h_mix
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def elapsed():
        return time.perf_counter() - t_start

    def remaining():
        return budget_s - elapsed()

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).copy()
        if parent.size == 0 or not np.all(np.isfinite(parent)):
            parent = None
    except (NameError, Exception):
        parent = None

    if parent is not None:
        best_h = repair_feasible(parent)
        best_score = evaluate_score(best_h)
    else:
        best_h = np.full(70, 0.5)
        best_score = evaluate_score(best_h)

    # Phase 1: SLSQP on parent dimension with various betas
    betas = [0.0, 1e-5, 5e-5, 2e-4]
    for b in betas:
        if remaining() < 15:
            break
        result = solve_epigraph(best_h.copy(), beta=b, maxiter=50, ftol=1e-10)
        if result is not None:
            x_final = result.x
            h_new = repair_feasible(x_final[:len(best_h)])
            if np.all(np.isfinite(h_new)):
                s_new = evaluate_score(h_new)
                if s_new < best_score - 1e-14:
                    best_h = h_new
                    best_score = s_new

    # Phase 2: Local descent and random perturbations
    if remaining() > 15:
        h_descent, s_descent = local_gradient_descent(best_h.copy(), max_steps=40, base_alpha=0.03)
        if s_descent < best_score - 1e-14:
            best_h = h_descent
            best_score = s_descent

    if remaining() > 10:
        h_rand, s_rand = local_random_perturb(best_h.copy(), n_trials=25, eps=0.012, seed_val=seed)
        if s_rand < best_score - 1e-14:
            best_h = h_rand
            best_score = s_rand

    # Phase 3: Multi-dimension search
    cur_n = len(best_h)
    dim_candidates = []
    for d in [cur_n - 10, cur_n - 5, cur_n + 5, cur_n + 10, cur_n + 20, cur_n + 40, cur_n + 60]:
        if 40 <= d <= 220 and d != cur_n:
            dim_candidates.append(d)
    dim_candidates = sorted(set(dim_candidates))

    for d in dim_candidates:
        if remaining() < 25:
            break
        h_d = interpolate_to_len(best_h, d)
        s_d = evaluate_score(h_d)

        if remaining() > 30:
            result = solve_epigraph(h_d.copy(), beta=0.0, maxiter=45, ftol=1e-10)
            if result is not None:
                h_d2 = repair_feasible(result.x[:d])
                if np.all(np.isfinite(h_d2)):
                    s_d2 = evaluate_score(h_d2)
                    if s_d2 < s_d:
                        h_d = h_d2
                        s_d = s_d2

        if remaining() > 10:
            h_d, s_d = local_gradient_descent(h_d.copy(), max_steps=25, base_alpha=0.02)

        if s_d < best_score - 1e-14:
            best_h = h_d
            best_score = s_d

        # Blend with current best if both similar length
        if s_d < best_score + 0.005 and d == len(best_h) and remaining() > 5:
            h_blend, s_blend = blend_candidates(best_h, h_d, n_points=15)
            if s_blend < best_score - 1e-14:
                best_h = h_blend
                best_score = s_blend

    # Phase 4: Determination if we need more work
    if remaining() > 30 and best_score > 0.38079:
        # Try SLSQP from slightly perturbed starts
        rng = np.random.default_rng(seed + 1)
        for trial in range(4):
            if remaining() < 15:
                break
            pert = rng.standard_normal(len(best_h)) * 0.01
            pert = pert - pert.mean()
            h_start = repair_feasible(best_h + pert)
            result = solve_epigraph(h_start, beta=0.0, maxiter=40, ftol=1e-10)
            if result is not None:
                h_new = repair_feasible(result.x[:len(best_h)])
                if np.all(np.isfinite(h_new)):
                    s_new = evaluate_score(h_new)
                    if s_new < best_score - 1e-14:
                        best_h = h_new
                        best_score = s_new

    # Phase 5: Final polish
    if remaining() > 10:
        best_h, s = local_gradient_descent(best_h.copy(), max_steps=40, base_alpha=0.015)
        if s < best_score - 1e-14:
            best_score = s

    if remaining() > 10:
        best_h, s = coordinate_descent(best_h.copy(), n_rounds=2,
                                       deltas=[0.01, 0.005, 0.002, -0.01, -0.005, -0.002])
        if s < best_score - 1e-14:
            best_score = s

    if remaining() > 20:
        result = solve_epigraph(best_h.copy(), beta=0.0, maxiter=35, ftol=1e-11)
        if result is not None:
            h_new = repair_feasible(result.x[:len(best_h)])
            if np.all(np.isfinite(h_new)):
                s_new = evaluate_score(h_new)
                if s_new < best_score - 1e-14:
                    best_h = h_new
                    best_score = s_new

    # Final verification
    best_h = repair_feasible(best_h)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
