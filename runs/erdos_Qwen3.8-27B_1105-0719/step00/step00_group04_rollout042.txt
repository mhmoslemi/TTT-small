def _epi_obj(x):
    h = x[:-1]
    t = x[-1]
    if _G_BETA == 0.0:
        return float(t)
    return float(t + _G_BETA * np.sum(h * (1.0 - h)))

def _epi_jac(x):
    h = x[:-1]
    n = len(x)
    g = np.empty(n, dtype=float)
    if _G_BETA == 0.0:
        g[:-1] = 0.0
    else:
        g[:-1] = _G_BETA * (1.0 - 2.0 * h)
    g[-1] = 1.0

def _eq_fun(x):

def _eq_jac(x):

def _ineq_fun(x):
    h = x[:-1]

import time
import numpy as np
from scipy.optimize import minimize

_ROW_CACHE = {}
_G_N = 0
_G_BETA = 0.0
_G_TGT = 0.0
_G_ROWS = None


def get_rows(n):
    rows = _ROW_CACHE.get(n)
    if rows is None:
        rows = []
        for d in range(-(n - 1), n):
            if d >= 0:
                i = np.arange(0, n - d)
                j = np.arange(d, n)
            else:
                L = -d
                i = np.arange(L, n)
                j = np.arange(0, n - L)
            rows.append((d, i, j))
        _ROW_CACHE[n] = rows
    return rows


def exact_score(h):
    try:
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size < 1:
            return float("inf")
        if not np.all(np.isfinite(h)):
            return float("inf")
        dx = 2.0 / float(h.size)
        return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    except Exception:
        return float("inf")


def repair_feasible(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.ndim != 1 or h.size == 0:
        return h
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    np.clip(h, 0.0, 1.0, out=h)
    n = h.size
    if target is None:
        target = 0.5 * float(n)
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    for _ in range(60):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            cap = 1.0 - h
            capsum = cap.sum()
            if capsum <= 1e-15:
                break
            h = h + cap * (diff / capsum)
            np.clip(h, 0.0, 1.0, out=h)
        else:
            cap = h
            capsum = cap.sum()
            if capsum <= 1e-15:
                break
            h = h - cap * (-diff / capsum)
            np.clip(h, 0.0, 1.0, out=h)

    for _ in range(n + 5):
        s = h.sum()
        diff = target - s
        if abs(diff) <= 1e-12:

        if diff > 0.0:
            k = int(np.argmin(h))

        else:
            k = int(np.argmax(h))

            if dec <= 0.0:

            h[k] -= dec

def direct_profile(h):
    n = h.size
    rows = get_rows(n)
    dx = 2.0 / float(n)
    P = np.empty(len(rows), dtype=float)
    omh = 1.0 - h
    for r in range(len(rows)):
        i = rows[r][1]
        j = rows[r][2]
        P[r] = dx * float(np.dot(h[i], omh[j]))
    return P


def grad_profile_selected(h, idx):
    n = h.size
    dx = 2.0 / float(n)
    rows = get_rows(n)
    G = np.zeros((len(idx), n), dtype=float)
    omh = 1.0 - h
    for k, r in enumerate(idx):
        i = rows[int(r)][1]
        j = rows[int(r)][2]
        G[k, i] += dx * omh[j]
        G[k, j] -= dx * h[i]
    return G


def _epi_obj(x):
    h = x[:-1]
    t = x[-1]
    if _G_BETA == 0.0:
        return float(t)
    return float(t + _G_BETA * np.sum(h * (1.0 - h)))


def _epi_jac(x):
    n = len(x)
    g = np.empty(n, dtype=float)
    if _G_BETA == 0.0:
        g[:-1] = 0.0
    else:
        g[:-1] = _G_BETA * (1.0 - 2.0 * x[:-1])
    g[-1] = 1.0
    return g


def _eq_fun(x):
    return float(np.sum(x[:-1]) - _G_TGT)


def _eq_jac(x):
    n = len(x)
    g = np.zeros(n, dtype=float)
    g[:-1] = -1.0
    return g


def _ineq_fun(x):
    h = x[:-1]
    t = x[-1]
    rows = _G_ROWS
    m = len(rows)
    if m == 0:
        return np.empty(0, dtype=float)
    P = np.empty(m, dtype=float)
    dx = 2.0 / float(_G_N)
    omh = 1.0 - h
    for r in range(m):
        i = rows[r][1]
        j = rows[r][2]
        P[r] = dx * float(np.dot(h[i], omh[j]))
    return t - P


def _ineq_jac(x):

    rows = _G_ROWS
    m = len(rows)
    nvar = len(x)
    J = np.zeros((m, nvar), dtype=float)

        j = rows[r][2]
        J[r, i] += dx * omh[j]
        J[r, j] -= dx * h[i]

def prepare_slsqp_state(n, beta, active_idx):
    global _G_N, _G_BETA, _G_TGT, _G_ROWS
    _G_N = int(n)
    _G_BETA = float(beta)

def run_slsqp(h, beta=0.0, maxiter=200, active_count=None, ftol=1e-10):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n < 2:
        return None
    P = direct_profile(h)
    C = float(P.max())
    if not np.isfinite(C):
        return None
    M = int(P.size)
    if active_count is None:
        if n <= 70:
            active_count = M
        else:
            active_count = min(M, 60)
    active_count = int(min(max(1, active_count), M))
    idx = np.argsort(P)[-active_count:]
    prepare_slsqp_state(n, float(beta), idx)

    t0 = C + 1e-8
    t_ub = max(C + 1.0, t0 + 1e-6)
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = t0
    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, t_ub))
    cons = (
        {"type": "eq", "fun": _eq_fun, "jac": _eq_jac},
        {"type": "ineq", "fun": _ineq_fun, "jac": _ineq_jac},
    )
    try:
        res = minimize(
            _epi_obj,
            x0,
            jac=_epi_jac,
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": int(max(1, min(int(maxiter), 1000))), "ftol": float(ftol), "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
    except Exception:
        return None

    hnew = x[:n]
    if not np.all(np.isfinite(hnew)):
        return None

    best_h = h.copy()
    best_s = exact_score(best_h)
    for alpha in (0.0, 0.2, 0.35, 0.5, 0.7, 1.0):
        cand = (1.0 - alpha) * h + alpha * hnew
        hr = repair_feasible(cand, 0.5 * float(n))
        s = exact_score(hr)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_h = hr
    return (best_h, float(best_s))


def local_active_descent(h, max_iters=8, active_count=18,
                         alphas=(0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005)):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n < 2:
        return h, exact_score(h)
    s = exact_score(h)
    for _ in range(int(max_iters)):
        if not np.isfinite(s):
            break
        P = direct_profile(h)
        C = float(P.max())
        A = int(min(max(1, active_count), P.size))
        idx = np.argsort(P)[-A:]
        G = grad_profile_selected(h, idx)
        p = P[idx]
        w = np.maximum(p - (C - 0.01), 1e-6)
        ws = float(w.sum())
        if ws <= 0.0:
            break
        w = w / ws
        g = (G * w[:, None]).sum(axis=0)
        g = g - g.mean()
        norm = float(np.linalg.norm(g))
        if norm < 1e-12:
            break
        d = -g / norm
        improved = False
        for a in alphas:
            hb = h + float(a) * d
            hr = repair_feasible(hb, 0.5 * float(n))
            s2 = exact_score(hr)
            if np.isfinite(s2) and s2 < s - 1e-13:
                h = hr
                s = s2
                improved = True
                break
        if not improved:
            break
    return h, float(s)


def pair_local_search(h, max_rounds=6, pair_count=25,
                      delta_fracs=(0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125),
                      active_count=18):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n < 2:
        return h, exact_score(h)
    s = exact_score(h)
    eye = np.eye(n, dtype=bool)
    evals = 0
    max_evals = 1600

    for _ in range(int(max_rounds)):
        if evals >= max_evals:
            break
        P = direct_profile(h)
        C = float(P.max())
        A = int(min(max(1, active_count), P.size))
        idx = np.argsort(P)[-A:]
        G = grad_profile_selected(h, idx)
        p = P[idx]
        w = np.maximum(p - (C - 0.01), 1e-6)
        ws = float(w.sum())
        if ws <= 0.0:
            break
        w = w / ws
        g = (G * w[:, None]).sum(axis=0)

        D = g[:, None] - g[None, :]
        maxd = np.minimum(1.0 - h[:, None], h[None, :])
        mask = (D < -1e-10) & (maxd > 1e-8) & (~eye)
        if not np.any(mask):
            break
        positions = np.argwhere(mask)
        pot = D[mask] * maxd[mask]
        order = np.argsort(pot)[:int(pair_count)]
        improved = False
        for oi in order:
            ii = int(positions[oi, 0])
            jj = int(positions[oi, 1])
            md = float(maxd[ii, jj])
            for frac in delta_fracs:
                d = float(frac) * md
                if d <= 1e-10:
                    continue
                hb = h.copy()
                hb[ii] += d
                hb[jj] -= d
                np.clip(hb, 0.0, 1.0, out=hb)
                s2 = exact_score(hb)
                evals += 1
                if not np.isfinite(s2) or s2 >= s - 1e-13:
                    continue
                hr = repair_feasible(hb, 0.5 * float(n))
                s2 = exact_score(hr)
                if np.isfinite(s2) and s2 < s - 1e-13:
                    h = hr
                    s = s2
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, float(s)


def random_pair_search(h, rng, trials=300, max_delta=0.05):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n < 2:
        return h, exact_score(h)
    s = exact_score(h)
    mdelta = float(max_delta)
    for _ in range(int(trials)):
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a == b:
            continue
        done = False
        md = min(1.0 - h[a], h[b])
        if md > 1e-10:
            d = mdelta * float(rng.random()) * float(md)
            if d > 1e-10:
                hb = h.copy()
                hb[a] += d
                hb[b] -= d
                np.clip(hb, 0.0, 1.0, out=hb)
                s2 = exact_score(hb)
                if np.isfinite(s2) and s2 < s - 1e-13:
                    hr = repair_feasible(hb, 0.5 * float(n))
                    s2 = exact_score(hr)
                    if np.isfinite(s2) and s2 < s - 1e-13:
                        h = hr
                        s = s2
                        done = True
        if not done:
            md = min(1.0 - h[b], h[a])
            if md > 1e-10:
                d = mdelta * float(rng.random()) * float(md)
                if d > 1e-10:
                    hb = h.copy()
                    hb[b] += d
                    hb[a] -= d
                    np.clip(hb, 0.0, 1.0, out=hb)
                    s2 = exact_score(hb)
                    if np.isfinite(s2) and s2 < s - 1e-13:
                        hr = repair_feasible(hb, 0.5 * float(n))
                        s2 = exact_score(hr)
                        if np.isfinite(s2) and s2 < s - 1e-13:
                            h = hr
                            s = s2
    return h, float(s)


def symmetrize(h):
    h = np.asarray(h, dtype=float)
    return 0.5 * (h + h[::-1])


def transfer_to_length(h, m):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    m = int(m)
    if m < 2:
        return None
    if m == n:
        return h.copy()
    sw = 2.0 / float(n)
    tw = 2.0 / float(m)
    src_edges = np.arange(n + 1, dtype=float) * sw
    out = np.empty(m, dtype=float)
    for t in range(m):
        a = float(t) * tw
        b = float(t + 1) * tw
        lo = np.maximum(a, src_edges[:-1])
        hi = np.minimum(b, src_edges[1:])
        w = np.clip(hi - lo, 0.0, None)
        out[t] = float(np.dot(h, w)) / tw
    return repair_feasible(out, 0.5 * float(m))


def structured_candidate(n, m):
    n = int(n)
    if n < 2:
        return None
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.5 * np.cos(np.pi * float(m) * x)
    return repair_feasible(h, 0.5 * float(n))


def random_fourier_candidate(n, rng, modes=8):
    n = int(n)
    if n < 2:
        return None
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = 0.5 * np.ones(n, dtype=float)
    for m in range(1, int(modes) + 1):
        a = float(rng.normal(scale=0.25 / float(m)))
        b = float(rng.normal(scale=0.25 / float(m)))
        h = h + a * np.cos(np.pi * float(m) * x) + b * np.sin(np.pi * float(m) * x)
    h = np.clip(h, 0.0, 1.0)
    return repair_feasible(h, 0.5 * float(n))


def safe_eval_array(arr):
    try:
        h = np.asarray(arr, dtype=float)
        if h.ndim != 1 or h.size < 2:
            return None
        h = repair_feasible(h, 0.5 * float(h.size))
        if not np.all(np.isfinite(h)):
            return None
        s = exact_score(h)
        if not np.isfinite(s):
            return None
        return (h, float(s), int(h.size))
    except Exception:
        return None


def update_best(best, cand):
    if cand is None:
        return best
    h, s, n = cand
    if s < best[1] - 1e-13 or (abs(s - best[1]) <= 1e-13 and n < best[2]):
        best[0] = h
        best[1] = float(s)
        best[2] = int(n)
    return best


def add_candidate(cands, best, cand):
    if cand is None:
        return best
    best = update_best(best, cand)
    cands.append((cand[0].copy(), float(cand[1]), int(cand[2])))
    return best


def process_candidate(cands, best, arr):
    cand = safe_eval_array(arr)
    return add_candidate(cands, best, cand)


def sort_indices_by_cand(cands):
    idxs = []
    for i in range(len(cands)):
        pos = 0
        while pos < len(idxs):
            j = idxs[pos]
            ci = cands[i]
            cj = cands[j]
            if ci[1] < cj[1] - 1e-15 or (abs(ci[1] - cj[1]) <= 1e-15 and ci[2] < cj[2]):
                break
            pos += 1
        idxs.insert(pos, i)
    return idxs


def select_slsqp_candidates(cands, max_count=3):
    if len(cands) == 0:
        return []
    max_count = int(max_count)
    idxs = sort_indices_by_cand(cands)
    sel = [idxs[0]]
    seen = set()
    seen.add(int(cands[sel[0]][2]))
    for i in idxs[1:]:
        if len(sel) >= max_count:
            break
        n = int(cands[i][2])
        if n in seen:
            continue
        sel.append(i)
        seen.add(n)
    if len(sel) < max_count:
        for i in idxs:
            if len(sel) >= max_count:
                break
            already = False
            for j in sel:
                if i == j:
                    already = True
                    break
            if not already:
                sel.append(i)
    return [cands[i] for i in sel]


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    budget_s = float(budget_s)
    if budget_s <= 0.0:
        budget_s = 1e-3
    rng = np.random.default_rng(seed)

    par = None
    try:
        par = np.array(initial_h_values, dtype=float, copy=True)
    except Exception:
        par = None

    if par is None or par.ndim != 1 or par.size == 0:
        par = np.full(70, 0.5, dtype=float)
    if not np.all(np.isfinite(par)):
        par = np.full(70, 0.5, dtype=float)

    par = repair_feasible(par, 0.5 * float(par.size))
    best_s = exact_score(par)
    if not np.isfinite(best_s):
        par = np.full(70, 0.5, dtype=float)
        best_s = exact_score(par)

    best = [par, float(best_s), int(par.size)]
    cands = []
    best = process_candidate(cands, best, par)

    TARGET = 0.3808
    rem = budget_s - (time.perf_counter() - t_start)
    if best[1] <= TARGET or rem < 5.0:
        return (best[0], best[1], best[2])

    if rem > 30.0:
        hloc, sloc = local_active_descent(par, max_iters=5, active_count=24)
        best = process_candidate(cands, best, hloc)
        hp, sp = pair_local_search(par, max_rounds=3, pair_count=20, active_count=24)
        best = process_candidate(cands, best, hp)
        rem = budget_s - (time.perf_counter() - t_start)
        if rem > 30.0 and best[1] > TARGET:
            hr, sr = random_pair_search(best[0], rng, trials=200, max_delta=0.03)
            best = process_candidate(cands, best, hr)
    if best[1] <= TARGET:
        return (best[0], best[1], best[2])

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 220.0:
        bb = int(best[2])
        for m in (1, 2, 3):
            rem = budget_s - (time.perf_counter() - t_start)
            if rem < 60.0 or best[1] <= TARGET:
                break
            hstr = structured_candidate(bb, m)
            if hstr is not None:
                best = process_candidate(cands, best, hstr)
                rem = budget_s - (time.perf_counter() - t_start)
                if rem > 100.0:
                    hl, sl = local_active_descent(hstr, max_iters=2, active_count=16)
                    best = process_candidate(cands, best, hl)
                    hp, sp = pair_local_search(hstr, max_rounds=1, pair_count=10, active_count=16)
                    best = process_candidate(cands, best, hp)

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 250.0:
        for _ in range(2):
            rem = budget_s - (time.perf_counter() - t_start)
            if rem < 120.0 or best[1] <= TARGET:
                break
            hrf = random_fourier_candidate(int(best[2]), rng, modes=8)
            if hrf is not None:
                best = process_candidate(cands, best, hrf)
                hl, sl = local_active_descent(hrf, max_iters=2, active_count=16)
                best = process_candidate(cands, best, hl)

    base = int(best[2])
    dims = set()
    for delta in (-30, -20, -10, -5, 0, 5, 10, 20, 40, 70):
        nn = base + delta
        if 24 <= nn <= 200:
            dims.add(int(nn))
    if 2 * base <= 200:
        dims.add(int(2 * base))
    if base // 2 >= 24:
        dims.add(int(base // 2))

    for nn in sorted(dims):
        rem = budget_s - (time.perf_counter() - t_start)
        if rem < 25.0 or best[1] <= TARGET:
            break
        if nn < 2:
            continue
        if nn == int(best[2]):
            h_orig = best[0].copy()
        else:
            h_orig = transfer_to_length(best[0], int(nn))
        if h_orig is None:
            continue

        c1 = safe_eval_array(h_orig)
        c2 = safe_eval_array(symmetrize(h_orig))

        h_dim = None
        s_dim = float("inf")
        if c1 is not None:
            best = add_candidate(cands, best, c1)
            h_dim = c1[0]
            s_dim = c1[1]
        if c2 is not None:
            best = add_candidate(cands, best, c2)
            if h_dim is None or c2[1] < s_dim:
                h_dim = c2[0]
                s_dim = c2[1]

        if h_dim is not None:
            rem = budget_s - (time.perf_counter() - t_start)
            if rem > 60.0:
                hl, sl = local_active_descent(h_dim, max_iters=3, active_count=18)
                best = process_candidate(cands, best, hl)
                rem = budget_s - (time.perf_counter() - t_start)
                if rem > 45.0:
                    hp, sp = pair_local_search(h_dim, max_rounds=2, pair_count=12, active_count=18)
                    best = process_candidate(cands, best, hp)

    if not cands:
        cands.append((best[0].copy(), float(best[1]), int(best[2])))

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 100.0 and best[1] > TARGET:
        max_count = 3
        if rem < 500.0:
            max_count = 2
        if rem < 250.0:
            max_count = 1
        selected = select_slsqp_candidates(cands, max_count=max_count)
        for item in selected:
            rem = budget_s - (time.perf_counter() - t_start)
            if rem < 60.0 or best[1] <= TARGET:
                break
            hstart = item[0]
            nstart = int(hstart.size)
            if nstart <= 80:
                base_max = 250
            elif nstart <= 120:
                base_max = 160
            else:
                base_max = 100
            maxiter = int(min(base_max, max(25, int(rem / 3.0))))
            res = run_slsqp(hstart, beta=0.0, maxiter=maxiter, active_count=None)
            if res is not None:
                best = process_candidate(cands, best, res[0])
                res_h = res[0]
                rem = budget_s - (time.perf_counter() - t_start)
                if rem > 70.0:
                    hl, sl = local_active_descent(res_h, max_iters=4, active_count=20)
                    best = process_candidate(cands, best, hl)
                    rem = budget_s - (time.perf_counter() - t_start)
                    if rem > 50.0:
                        hp, sp = pair_local_search(res_h, max_rounds=3, pair_count=18, active_count=20)
                        best = process_candidate(cands, best, hp)

            rem = budget_s - (time.perf_counter() - t_start)
            if rem > 180.0 and best[1] > TARGET:
                maxiter2 = int(min(90, max(20, int(rem / 8.0))))
                active_count2 = int(min(2 * int(best[0].size) - 1, 40))
                res2 = run_slsqp(best[0], beta=0.0, maxiter=maxiter2, active_count=active_count2)
                if res2 is not None:
                    best = process_candidate(cands, best, res2[0])

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 120.0 and best[1] > TARGET:
        for _ in range(2):
            rem = budget_s - (time.perf_counter() - t_start)
            if rem < 100.0 or best[1] <= TARGET:
                break
            nbest = int(best[0].size)
            maxiter = int(min(120, max(25, int(rem / 6.0))))
            ac = int(min(2 * nbest - 1, 45))
            res = run_slsqp(best[0], beta=0.0, maxiter=maxiter, active_count=ac)
            if res is not None:
                best = process_candidate(cands, best, res[0])
                hl, sl = local_active_descent(res[0], max_iters=3, active_count=20)
                best = process_candidate(cands, best, hl)
                rem = budget_s - (time.perf_counter() - t_start)
                if rem > 45.0:
                    hp, sp = pair_local_search(res[0], max_rounds=3, pair_count=18, active_count=20)
                    best = process_candidate(cands, best, hp)

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 90.0 and best[1] > TARGET:
        nbest = int(best[0].size)
        maxiter = int(min(100, max(25, int(rem / 8.0))))
        res = run_slsqp(best[0], beta=1e-5, maxiter=maxiter, active_count=int(min(2 * nbest - 1, 55)))
        if res is not None:
            best = process_candidate(cands, best, res[0])
            hl, sl = local_active_descent(res[0], max_iters=3, active_count=20)
            best = process_candidate(cands, best, hl)

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 20.0:
        hl, sl = local_active_descent(best[0], max_iters=5, active_count=24)
        best = process_candidate(cands, best, hl)
        hp, sp = pair_local_search(best[0], max_rounds=4, pair_count=20, active_count=24)
        best = process_candidate(cands, best, hp)

    rem = budget_s - (time.perf_counter() - t_start)
    if rem > 10.0:
        hr, sr = random_pair_search(best[0], rng, trials=300, max_delta=0.02)
        best = process_candidate(cands, best, hr)

    best = process_candidate(cands, best, symmetrize(best[0]))
    return (best[0], float(best[1]), int(best[2]))
