where w_k = exp(β*c_k*dx) / sum_j exp(β*c_j*dx) are the softmax weights.

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx

    max_v = np.max(vals)
    log_sum = max_v + np.log(np.sum(np.exp(beta * (vals - max_v))))

            tl = min(per_resolve, budget_s - 15 - (get_now_time() - t_start))
            if tl < 3:
                continue
            h_r, c5_r = fw_optimize(h_try, n_try, max_steps=2000, beta=20.0, time_limit=tl)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, n_try

    # Phase 6: Final pair polish on best
    time_left = budget_s - (get_now_time() - t_start)
    if time_left > 10:
        t6 = min(60, time_left * 0.4)
        h_fp, c5_fp = pair_swap_search(best_h, best_n, time_limit=t6, rng=rng)
        if c5_fp < best_c5:
            best_h, best_c5, best_n = h_fp, c5_fp, best_n

    # Phase 7: One more SLSQP if time
    time_left = budget_s - (get_now_time() - t_start)
    if time_left > 20:
        h_f2, c5_f2 = slsqp_refine(best_h, best_n, time_limit=min(60, time_left - 5), beta=50.0)
        if c5_f2 < best_c5:
            best_h, best_c5, best_n = h_f2, c5_f2, best_n

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
