def gradient_c5(h, n, alpha=None):
    """Compute gradient of log-sum-exp of correlations, or subgradient of max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    if alpha is None:
        # Subgradient: pick the max correlation
        max_idx = np.argmax(corr)
        p = np.zeros(2*n-1)
        p[max_idx] = 1.0
    else:
        max_c = corr.max()
        exp_c = np.exp(alpha * (corr - max_c))
        p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        w = p[m]
        if w < 1e-15:
            continue
        k = m - (n-1)
        # Positions j where j+k is valid (i.e., 0<=j+k<n)
        if k >= 0:
            js = np.arange(n-k)  # j from 0 to n-1-k
        else:
            js = np.arange(-k, n)  # j from -k to n-1
        
        # First term: w * (1 - h[j+k])
        grad[js] += w * (1.0 - h[js + k])
        
        # Second term: -w * h[j-k] where 0<=j-k<n
        jm = js - k
        valid = (jm >= 0) & (jm < n)
        grad[js[valid]] -= w * h[jm[valid]]
    
    grad *= dx
    return grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def c5_value(h, n):
    return float(compute_correlations(h, n).max())

def lse_objective(h, n, alpha):
    corr = compute_correlations(h, n)
    mc = corr.max()
    if alpha < 1e-6:
        return mc, np.zeros_like(h)
    exp_c = np.exp(alpha * (corr - mc))
    val = mc + np.log(exp_c.sum()) / alpha
    p = exp_c / exp_c.sum()
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)

        w = p[m]
        if w < 1e-15:
            continue

        if k >= 0:
            js = np.arange(n - k)
        else:
            js = np.arange(-k, n)
        grad[js] += w * (1.0 - h[js + k])
        jm = js - k
        valid = (jm >= 0) & (jm < n)
        grad[js[valid]] -= w * h[jm[valid]]
    grad *= dx
    return val, grad

def make_feasible(h, n):

    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Iterative adjustment
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase; find elements with room
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:

    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr


def c5_value(h, n):
    return float(compute_correlations(h, n).max())


def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_grad(h, n, alpha):
    corr = compute_correlations(h, n)
    mc = corr.max()
    if alpha < 1e-6:
        p = np.zeros(2 * n - 1)
        p[np.argmax(corr)] = 1.0
    else:
        exp_c = np.exp(alpha * (corr - mc))
        p = exp_c / exp_c.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            js = np.arange(n - k)
        else:
            js = np.arange(-k, n)
        grad[js] += w * (1.0 - h[js + k])
        jm = js - k
        valid = (jm >= 0) & (jm < n)
        grad[js[valid]] -= w * h[jm[valid]]
    grad *= dx
    return grad


def lse_val(h, n, alpha):
    corr = compute_correlations(h, n)
    mc = corr.max()
    if alpha < 1e-6:
        return mc
    return mc + np.log(np.sum(np.exp(alpha * (corr - mc)))) / alpha

        self.n = n
        self.alpha = alpha

    def jac(self, h):
        return lse_grad(h, self.n, self.alpha)

    def con_fun(self, h):
        return h.sum() - self.n / 2.0

    def con_jac(self, h):
        return np.ones(self.n)


def optimize_lse(h0, n, max_time, max_iter=80):
    h = h0.copy()
    h = make_feasible(h, n)
    start_t = time.time()
    alphas = [5, 10, 20, 50, 100, 200, 500]
    for alpha in alphas:
        if time.time() - start_t > max_time:
            break
        prob = LSEProblem(n, alpha)
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': prob.con_fun, 'jac': prob.con_jac}]
        try:
            res = minimize(prob.fun, h, jac=prob.jac, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': 1e-10})
            h_new = make_feasible(res.x, n)
            if c5_value(h_new, n) <= c5_value(h, n) + 1e-8:
                h = h_new
        except Exception:
            pass
    return h
