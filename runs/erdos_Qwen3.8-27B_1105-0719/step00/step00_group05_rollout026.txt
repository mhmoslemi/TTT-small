```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Water fill: adjust to meet sum constraint
    # If diff > 0, add to elements that can increase
    # If diff < 0, subtract from elements that can decrease
    if diff > 0:
        # Sort by room to increase (1-h[i]) descending
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            add = min(room[i], remaining)
            h[i] += add
            remaining -= add
            if remaining < 1e-15:
                break
    else:
        # Sort by value ascending (elements closer to 0 first)
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            sub = min(h[i], remaining)
            h[i] -= sub
            remaining -= sub
            if remaining < 1e-15:
                break
    return h

def lag_gradient(h, lag_idx, n):
    """Compute gradient of V_lag where lag_idx is the index in np.correlate output."""
    # lag_idx ranges from 0 to 2*n-2
    # Actual lag = lag_idx - (n-1)
    lag = lag_idx - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        # V_lag = sum_{i=0}^{n-lag-1} h[i+lag] * (1-h[i])
        # d/dh[j] = (1-h[j-lag]) for j >= lag, j <= n-1
        #          - h[j+lag] for j <= n-lag-1
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        k = -lag  # k > 0
        # V_{-k} = sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
        # d/dh[j] = (1-h[j+k]) for j <= n-k-1
        #          - h[j-k] for j >= k
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

def weighted_lag_objective(h, active_indices, weights, n):
    """Compute weighted sum of active lag values."""
    lags = np.correlate(h, 1.0 - h, 'full')
    return float(np.dot(weights, lags[active_indices]))

def weighted_lag_gradient(h, active_indices, weights, n):
    """Compute gradient of weighted sum of active lag values."""
    g = np.zeros(n)
    for idx, w in zip(active_indices, weights):
        g += w * lag_gradient(h, idx, n)
    return g
```

```python
def project_feasible(h, n):
```
