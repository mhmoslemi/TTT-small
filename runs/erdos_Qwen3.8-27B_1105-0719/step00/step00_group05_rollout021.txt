import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_C(h):
    """Compute the exact C5 value for a step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(u, n):
    """Project u onto {0<=x<=1, sum(x)=n/2} using water-filling."""
    target = n / 2.0
    # Bisection on shift
    lo = -u.max() - 1.0
    hi = -u.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(u + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(u + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-15:
        # Distribute residual to coordinates with most slack
        if residual > 0:
            order = np.argsort(-(u + (lo+hi)/2.0))  # most room to increase
        else:
            order = np.argsort(u + (lo+hi)/2.0)  # most room to decrease
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - x[idx])
                x[idx] += add
                residual -= add
            else:
                sub = min(-residual, x[idx])
                x[idx] -= sub
                residual += sub
            if abs(residual) < 1e-15:
                break
    return x

def sigmoid(t):
    """Numerically stable sigmoid."""
    t = np.clip(t, -500, 500)
    return np.where(t >= 0, 1.0/(1.0+np.exp(-t)), np.exp(t)/(1.0+np.exp(t)))

def feasible_from_logit(y):
    """Map unconstrained y to feasible h via logistic manifold."""
    n = len(y)
    target = n / 2.0
    # Solve for lambda such that sum(sigmoid(y + lambda)) = target
    # As lambda -> -inf, sum -> 0; as lambda -> +inf, sum -> n
    lo = -np.max(y) - 50.0
    hi = -np.min(y) + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sigmoid(y + mid).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    grad_y = m * (G - B)

    return F, grad_y
