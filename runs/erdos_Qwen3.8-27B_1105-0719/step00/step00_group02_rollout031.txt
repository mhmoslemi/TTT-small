    # h[j] appears when i=j: contributes (1-h[j-d]) (if j-d in [0,n))
    # h[j] appears in (1-h[i-d]) term when i-d=j, i.e., i=j+d: contributes -h[j+d] (if j+d in [0,n))
    
    # So: d(corr_val(d))/dh[j] = (1-h[j-d]) * [0<=j-d<n] - h[j+d] * [0<=j+d<n]
    
    coord_weights = np.zeros(n)
    for d in range(-(n-1), n):
        w = lag_weights[d + n - 1]
        if w < 1e-10:
            continue
        for j in range(n):
            g = 0.0
            jd = j - d
            if 0 <= jd < n:
                g += (1.0 - h[jd])
            jp = j + d
            if 0 <= jp < n:
                g -= h[jp]
            coord_weights[j] += w * g
    
    return coord_weights

Let me also think about pair moves. A pair move transfers mass from one coordinate to another:
- h[i] += delta, h[j] -= delta (or vice versa)
- Need to maintain 0 ≤ h[k] ≤ 1
- Delta is bounded by min(h[j], 1-h[i]) for the first direction

For a pair (i,j) and a transfer amount t:
- New h[i] = h[i] + t, h[j] = h[j] - t
- Constraint: -h[i] ≤ t ≤ 1-h[i] and h[j]-1 ≤ t ≤ h[j]
- So t ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]

We want to find t that minimizes C₅. Since C₅ is the max of quadratic functions of t, we can try a grid search.

Actually, for a fixed pair (i,j), the correlation values are all quadratic in t (since each correlation involves at most two terms with h[i] or h[j]). So the max of quadratics is a piecewise quadratic function, and we can find its minimum by checking the critical points of each quadratic and the breakpoints.

But that's complex. A simpler approach: just try a grid of t values and pick the best.

def get_now():
    return time.perf_counter()

def evaluate(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    target = n / 2.0
    # Water-filling projection
    lo, hi = x.min() - 2, x.max() + 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
            if mask.sum() > 0:
                proj[mask] = np.minimum(proj[mask] + diff / mask.sum(), 1.0)
        else:
            mask = proj > 1e-10
            if mask.sum() > 0:
                proj[mask] = np.maximum(proj[mask] + diff / mask.sum(), 0.0)
    return proj

def extreme_point(weights, n):
    target = n / 2.0
    idx = np.argsort(weights, kind='stable')
    x = np.zeros(n)
    k = int(n // 2)
    x[idx[:k]] = 1.0
    rem = target - k
    if rem > 1e-10:
        x[idx[k]] = rem
    return x

def compute_coord_weights(h, n, scale_param=0.03):
    """Compute gradient-based weights for FW step."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    scale = max(cmax * scale_param, 1e-8)
    lag_w = np.exp((corr - cmax) / scale)
    lag_w /= lag_w.sum()

    # d(corr_val(d))/dh[j] = (1-h[j-d]) * [0<=j-d<n] - h[j+d] * [0<=j+d<n]

            jd = j - d

                g += (1.0 - h[jd])
            jp = j + d
            if 0 <= jp < n:
                g -= h[jp]
            coord_w[j] += wl * g
    return coord_w

def line_search(h0, h1, n, n_coarse=40, n_fine=20):
    """Two-stage line search."""
    best_t = 0.0
    best_val = evaluate(h0, n)
    
    # Coarse grid
    ts = np.linspace(0, 1, n_coarse)
    vals = np.array([evaluate((1-t)*h0 + t*h1, n) for t in ts])
    i = np.argmin(vals)
    if vals[i] < best_val:
        best_val = vals[i]
        best_t = ts[i]
    
    # Fine grid around best
    if 0 < i < n_coarse - 1:
        t_lo, t_hi = ts[i-1], ts[i+1]
        ts2 = np.linspace(t_lo, t_hi, n_fine)
        vals2 = np.array([evaluate((1-t)*h0 + t*h1, n) for t in ts2])
        i2 = np.argmin(vals2)
        if vals2[i2] < best_val:
            best_val = vals2[i2]
            best_t = ts2[i2]
    
    return best_t, best_val

def pair_move(h, n, i, j, n_grid=30):
    """Try transferring mass between coordinates i and j."""
    # h[j] -> h[i]: h[i]+t, h[j]-t
    # t in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
    t_lo = max(-h[i], h[j] - 1)
    t_hi = min(1 - h[i], h[j])
    if t_hi - t_lo < 1e-10:
        return h, evaluate(h, n)
    
    best_t = 0.0
    best_val = evaluate(h, n)
    ts = np.linspace(t_lo, t_hi, n_grid)
    for t in ts:
        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        val = evaluate(h_new, n)
        if val < best_val:
            best_val = val
            best_t = t
    if best_t != 0:
        h_new = h.copy()
        h_new[i] += best_t
        h_new[j] -= best_t
        return h_new, best_val
    return h, best_val

def random_feasible(n, rng):

For the subgradient approach:

- If there are multiple argmax lags, we can use a convex combination of their gradients.

import numpy as np
import time

def get_now():
    return time.perf_counter()

def evaluate(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    target = n / 2.0
    lo, hi = float(x.min()) - 2.0, float(x.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                proj[mask] = np.minimum(proj[mask] + diff / cnt, 1.0)
        else:
            mask = proj > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                proj[mask] = np.maximum(proj[mask] + diff / cnt, 0.0)
    return proj

def extreme_point(weights, n):
    target = n / 2.0
    idx = np.argsort(weights, kind='stable')
    x = np.zeros(n)
    k = n // 2
    x[idx[:k]] = 1.0
    rem = target - k
    if rem > 1e-10:
        x[idx[k]] = rem
    return x

def grad_lag(h, n, d):
    """Compute gradient of corr at lag d w.r.t. h."""
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        # corr[d] = sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
        # d/dh[j]: 
        #   from h[j+d] term (j from d to n-1): (1-h[j-d])
        #   from (1-h[j]) term (j from 0 to n-1-d): -h[j+d]
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
    else:
        dd = -d
        # corr[-dd] = sum_{j=0}^{n-1-dd} h[j]*(1-h[j+dd])
        # d/dh[j]:
        #   from h[j] term (j from 0 to n-1-dd): (1-h[j+dd])
        #   from (1-h[j+dd]) term (j from dd to n-1): -h[j-dd]
        g[:n-dd] += (1.0 - h[dd:])
        g[dd:] -= h[:n-dd]
    return g

def compute_weights(h, n, scale=0.03, use_subgrad=False):
    """Compute coordinate weights for FW direction."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    coords = np.zeros(n)
    
    if use_subgrad:
        # Use subgradient: gradient of top few lags
        top_k = min(5, 2*n-1)
        top_indices = np.argsort(corr)[-top_k:]
        for idx in top_indices:
            d = idx - (n - 1)
            coords += grad_lag(h, n, d)
    else:
        # Soft-max weighted
        scale_val = max(cmax * scale, 1e-8)
        lag_w = np.exp((corr - cmax) / scale_val)
        lag_w /= lag_w.sum()
        for i in range(2*n - 1):
            if lag_w[i] < 1e-10:
                continue
            d = i - (n - 1)
            coords += lag_w[i] * grad_lag(h, n, d)
    
    return coords

def line_search(h0, h1, n, n_coarse=40, n_fine=20):
    best_t = 0.0
    best_val = evaluate(h0, n)
    ts = np.linspace(0, 1, n_coarse)
    vals = np.empty(n_coarse)
    for i in range(n_coarse):
        t = ts[i]
        if t == 0:
            vals[i] = best_val
        else:
            vals[i] = evaluate((1-t)*h0 + t*h1, n)
    i_min = int(np.argmin(vals))
    if vals[i_min] < best_val:
        best_val = float(vals[i_min])
        best_t = float(ts[i_min])
    if 0 < i_min < n_coarse - 1:
        ts2 = np.linspace(ts[i_min-1], ts[i_min+1], n_fine)
        vals2 = np.empty(n_fine)
        for i in range(n_fine):
            t = ts2[i]
            vals2[i] = evaluate((1-t)*h0 + t*h1, n)
        i2 = int(np.argmin(vals2))
        if vals2[i2] < best_val:
            best_val = float(vals2[i2])
            best_t = float(ts2[i2])
    return best_t, best_val

def pair_move(h, n, i, j, n_grid=30):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-10:
        return h, evaluate(h, n)
    best_t = 0.0
    h_cur = h.copy()
    best_val = evaluate(h_cur, n)
    ts = np.linspace(t_lo, t_hi, n_grid)
    for t in ts:
        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        v = evaluate(h_new, n)
        if v < best_val:
            best_val = v
            best_t = t
    if abs(best_t) > 1e-10:
        h_new = h.copy()
        h_new[i] += best_t
        h_new[j] -= best_t
        return h_new, best_val
    return h, best_val

def random_binary(n, rng):
    pos = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0

    x2 = 2.0 * (np.arange(n2) + 0.5) / n2
    h2 = np.interp(x2, x1, h)
    return project_feasible(h2, n2)
