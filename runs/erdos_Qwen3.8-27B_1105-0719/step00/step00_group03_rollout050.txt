    n = len(h)

    """Smoothed log-sum-exp objective + constraint penalty."""
    n = len(x)
    dx = 2.0 / n
    scores = compute_scores(x, dx)

    shifted = alpha * (scores - F)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    log_sum = F + np.log(np.sum(np.exp(shifted))) / alpha
    return log_sum

    n = len(x)
    dx = 2.0 / n
    scores = compute_scores(x, dx)
    F = np.max(scores)
    shifted = alpha * (scores - F)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)

            if 0 <= j_pk < n:
                g -= x[j_pk]
            grad[j] += w * dx * g

def project_feasible(x, target):

    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - target) < 1e-12:
        return x
    # Use bisection to find lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo+hi)/2, 0.0, 1.0)

            proj[mask] += adj
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    """Try to improve by optimizing coordinates p and q together."""
    n = len(x)
    S = target - (np.sum(x) - x[p] - x[q])
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if lo >= hi:
        return x, False

    # where t = x[p], x[q] = S - t

def compute_correlation(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    """Compute C5 = max of correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlation(h)) * dx

def pair_sweep(x, target, dx, max_sweeps=50):
    """Perform pair coordinate descent sweeps."""
    n = len(x)
    R = 2 * n - 1
    best = compute_c5(x)
    
    for sweep in range(max_sweeps):
        improved = False
        # Precompute s = correlation * dx
        s = compute_correlation(x) * dx
        
        # Precompute cp for all p and all k
        ks = np.arange(-(n-1), n)
        # cp[p, r] where r indexes ks
        # cp[p, r] = (1-x[p-ks[r]]) * I(0<=p-ks[r]<n) - x[p+ks[r]] * I(0<=p+ks[r]<n)
        cp_mat = np.zeros((n, R))
        for p in range(n):
            v1 = (p - ks >= 0) & (p - ks < n)
            cp_mat[p, v1] = (1.0 - x[p - ks[v1]])
            v2 = (p + ks >= 0) & (p + ks < n)
            cp_mat[p, v2] -= x[p + ks[v2]]
        
        for p in range(n):
            for q in range(p + 1, n):
                S = target - (np.sum(x) - x[p] - x[q])
                lo = max(0.0, S - 1.0)
                hi = min(1.0, S)
                if lo >= hi - 1e-12:
                    continue
                
                a = dx * (cp_mat[p] - cp_mat[q])
                rest = s / dx - cp_mat[p] * x[p] - cp_mat[q] * x[q]
                b = dx * (cp_mat[q] * S + rest)
                
                # Minimize max_r (a[r]*t + b[r]) for t in [lo, hi]

                    x_old = (x[p], x[q])
                    x[p] = t_best
                    x[q] = S - t_best
                    new_c5 = np.max(compute_correlation(x)) * dx
                    if new_c5 < best - 1e-12:
                        best = new_c5
                        improved = True
                        # Update s and cp_mat
                        s = compute_correlation(x) * dx
                        for p2 in range(n):
                            v1 = (p2 - ks >= 0) & (p2 - ks < n)
                            cp_mat[p2, v1] = (1.0 - x[p2 - ks[v1]])
                            v2 = (p2 + ks >= 0) & (p2 + ks < n)
                            cp_mat[p2, v2] -= x[p2 + ks[v2]]
                    else:
                        x[p], x[q] = x_old
        
        if not improved:
            break
    
    return x, best

def minimize_max_linear(a, b, lo, hi):
    """Minimize max_k (a[k]*t + b[k]) for t in [lo, hi]."""
    R = len(a)
    
    # Collect candidate points
    da = a[:, None] - a[None, :]
    db = b[:, None] - b[None, :]
    mask = np.abs(da) > 1e-15
    with np.errstate(divide='ignore', invalid='ignore'):

    valid = (t_mat >= lo) & (t_mat <= hi) & (~np.isnan(t_mat))
    candidates = np.concatenate([[lo, hi], t_mat[valid]])
    candidates = np.unique(np.round(candidates, 14))
    
    # Evaluate at all candidates
    vals = a[:, None] * candidates[None, :] + b[:, None]
    maxes = np.max(vals, axis=0)
    
    return candidates[np.argmin(maxes)]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def minimize_max_linear(a, b, lo, hi):
    R = len(a)
    if R == 0:
        return (lo + hi) / 2.0
    mid = (lo + hi) / 2.0
    best_val = np.max(a * mid + b)
    best_t = mid
    # Try endpoints
    for t in [(lo,), (hi,)]:
        v = np.max(a * t[0] + b)
        if v < best_val - 1e-15:
            best_val = v
            best_t = t[0]
    # Compute breakpoints
    da = a[:, None] - a[None, :]
    db = b[:, None] - b[None, :]
    mask = np.abs(da) > 1e-14
    if not np.any(mask):
        return best_t
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = np.where(mask, -db / da, np.nan)
    valid = (t_mat >= lo + 1e-14) & (t_mat <= hi - 1e-14)
    if not np.any(valid):
        return best_t
    candidates = t_mat[valid]
    # Deduplicate approximately
    candidates = np.unique(np.round(candidates, 12))
    if len(candidates) == 0:
        return best_t
    vals = a[:, None] * candidates[None, :] + b[:, None]
    maxes = np.max(vals, axis=0)
    idx = np.argmin(maxes)
    if maxes[idx] < best_val - 1e-15:
        return candidates[idx]
    return best_t

def pair_descent(x, target, max_sweeps=300, time_limit=None, start_time=None):

    dx = 2.0 / n
    R = 2 * n - 1
    best = compute_c5(x)
    ks = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        s = np.correlate(x, 1.0 - x, mode='full') * dx
        
        cp_mat = np.zeros((n, R))
        for p in range(n):
            v1 = (p - ks >= 0) & (p - ks < n)
            cp_mat[p, v1] = (1.0 - x[p - ks[v1]])

                    continue

                rest = s / dx - cp_mat[p] * x[p] - cp_mat[q] * x[q]
                b_arr = dx * (cp_mat[q] * S + rest)

def slsqp_optimize(x0, target, alpha, maxiter=150):
    n = len(x0)
    dx = 2.0 / n
    R = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    
    def objective(x):
        s = np.correlate(x, 1.0 - x, mode='full') * dx
        F = np.max(s)
        shifted = np.clip(alpha * (s - F), -300, 300)
        return F + np.log(np.sum(np.exp(shifted))) / alpha
    
    def gradient(x):
        s = np.correlate(x, 1.0 - x, mode='full') * dx
        F = np.max(s)
        shifted = np.clip(alpha * (s - F), -300, 300)
        ev = np.exp(shifted)
        w = ev / np.sum(ev)
        grad = np.zeros(n)
        for r in range(R):
            if w[r] < 1e-10:
                continue
            k = r - (n - 1)
            v1 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
            v2 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            g = np.zeros(n)
            g[v1] += (1.0 - x[np.arange(n)[v1] - k])
            g[v2] -= x[np.arange(n)[v2] + k]
            grad += w[r] * dx * g
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, x0, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return res.x
    except:
        pass
    return x0
