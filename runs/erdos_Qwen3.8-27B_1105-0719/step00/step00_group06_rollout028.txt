import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def refine_continuous(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.01, 0.004, 0.0015, 0.0006, 0.0002, 0.00008, 0.00003]
    for tau in taus:
        if time.time() > deadline - 6:
            break
        h_new = slsqp_optimize(h, tau, maxiter=min(400, n * 5 + 50))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_swap_search(b, deadline, max_rounds=300):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c


def binary_block_swap_search(b, deadline, max_rounds=50):
    n = len(b)
    curr_c = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_move = None
        for block_len in [2, 3, 4]:
            for i in range(0, n - block_len + 1):
                block_src = b[i:i + block_len]
                if np.sum(block_src) == 0 or np.sum(block_src) == block_len:
                    continue
                for j in range(0, n - block_len + 1):
                    if j == i or abs(i - j) < block_len:
                        continue
                    block_dst = b[j:j + block_len]
                    if np.sum(block_dst) == np.sum(block_src):
                        continue
                    old_src = b[i:i + block_len].copy()
                    old_dst = b[j:j + block_len].copy()
                    b[i:i + block_len] = old_dst
                    b[j:j + block_len] = old_src
                    c = exact_objective(b)
                    improve = curr_c - c
                    if improve > best_improve:
                        best_improve = improve
                        best_move = (i, j, block_len, old_src, old_dst)
                    b[i:i + block_len] = old_src
                    b[j:j + block_len] = old_dst
        if best_move is not None:
            i, j, bl, old_src, old_dst = best_move
            b[i:i + bl] = old_dst
            b[j:j + bl] = old_src
            curr_c = exact_objective(b)
        else:
            break
    return b, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)


def interpolate_pair(ha, hb, gamma):
    h = (1.0 - gamma) * ha + gamma * hb
    return project_box_sum(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
                best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        best_h = make_start(70, rng, 0)
        best_c = exact_objective(best_h)
        best_n = 70

    # Build search sizes
    parent_n = len(parent_h) if parent_h is not None else 70
    sizes = [parent_n]
    for s in [70, 64, 60, 56, 52, 48, 44, 40, 36, 32, 28, 24]:
        if s not in sizes:
            sizes.append(s)
    sizes = [s for s in sizes if 8 <= s <= 200]
    # Remove duplicates preserving order
    seen = set()
    sizes_unique = []
    for s in sizes:
        if s not in seen:
            seen.add(s)
            sizes_unique.append(s)
    sizes = sizes_unique

    # Phase 1: Continuous refinement from multiple starts
    for n in sizes:
        if time.time() > deadline - 25:
            break

        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.015, rng))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.04, rng))
        for mode in range(6):
            cands.append(make_start(n, rng, mode))

        # Score candidates
        scored = []
        for c in cands:
            c_score = exact_objective(c)
            scored.append((c_score, c))
        scored.sort(key=lambda x: x[0])

        # Refine top candidates
        n_top = min(3, len(scored))
        local_best = [scored[0][1], scored[0][0]]
        for idx in range(n_top):
            if time.time() > deadline - 18:
                break
            h0 = scored[idx][1]
            h_ref, c_ref = refine_continuous(h0, deadline)
            if c_ref < local_best[1]:
                local_best = [h_ref, c_ref]
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n

        # Try binary threshold + swap search
        if n % 2 == 0:
            if time.time() > deadline - 15:
                break
            b = binary_threshold(local_best[0])
            if b is not None:
                c_b_init = exact_objective(b)
                if c_b_init < best_c - 1e-12:
                    best_c = c_b_init
                    best_h = b.copy()
                    best_n = n
                b, c_b = binary_swap_search(b, deadline, max_rounds=200)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
                    best_n = n
                # Also try block swaps
                if time.time() < deadline - 8:
                    b2, c_b2 = binary_block_swap_search(b, deadline, max_rounds=30)
                    if c_b2 < best_c - 1e-12:
                        best_c = c_b2
                        best_h = b2.copy()
                        best_n = n

            # Try binary from second best too
            if n > 1 and scored[1][0] < 0.39:
                b2 = binary_threshold(scored[1][1])
                if b2 is not None and time.time() < deadline - 8:
                    b2, c_b2 = binary_swap_search(b2, deadline, max_rounds=100)
                    if c_b2 < best_c - 1e-12:
                        best_c = c_b2
                        best_h = b2.copy()
                        best_n = n

    # Phase 2: If we have a good solution, try to improve with more iterations
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            # Already binary, do more swap rounds
            b, c_b = binary_swap_search(best_h.copy(), deadline, max_rounds=300)
            if c_b < best_c - 1e-12:
                best_c = c_b
                best_h = b.copy()
        else:
            # Continuous, do more LSE refinement
            h_ref, c_ref = refine_continuous(best_h.copy(), deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
            # Try binary from best continuous
            if n_best % 2 == 0:
                b = binary_threshold(best_h)
                if b is not None:
                    b, c_b = binary_swap_search(b, deadline, max_rounds=200)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()

    # Phase 3: Interpolation between different-size solutions (same size only)
    if time.time() < deadline - 5 and best_h is not None:
        n_b = len(best_h)
        if parent_h is not None and len(parent_h) == n_b:
            for gamma in [0.3, 0.5, 0.7]:
                h_interp = interpolate_pair(best_h, parent_h, gamma)
                c_interp = exact_objective(h_interp)
                if c_interp < best_c - 1e-12:
                    best_c = c_interp
                    best_h = h_interp.copy()
                if n_b % 2 == 0:
                    b = binary_threshold(h_interp)
                    if b is not None:
                        b, c_b = binary_swap_search(b, deadline, max_rounds=50)
                        if c_b < best_c - 1e-12:
                            best_c = c_b
                            best_h = b.copy()

    # Final: try one more round of SLSQP on best
    if time.time() < deadline - 4:
        n_final = len(best_h)
        h_final = slsqp_optimize(best_h, 0.0001, maxiter=200)
        c_final = exact_objective(h_final)
        if c_final < best_c - 1e-12:
            best_c = c_final
            best_h = h_final.copy()
        if n_final % 2 == 0:
            b = binary_threshold(best_h)
            if b is not None:
                b, c_b = binary_swap_search(b, deadline, max_rounds=100)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()

    # Final validation and cleanup
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    # Safety: ensure feasibility
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
