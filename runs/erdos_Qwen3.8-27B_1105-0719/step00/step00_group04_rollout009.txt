        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:

def interpolate_to_n(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    dx_old = 2.0 / n
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    x_ext = np.concatenate([[0.0], x_old, [2.0]])
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_new, x_ext, h_ext)
    return np.clip(h_new, 0.0, 1.0)


class EpigraphNLP:
    def __init__(self, n, dx):
        self.n = n
        self.dx = dx
        self.offsets = np.arange(-(n - 1), n)

    def _profile(self, h):
        return np.correlate(h, 1.0 - h, mode="full") * self.dx

    def _profile_grad(self, h):
        n = self.n
        dx = self.dx
        M = 2 * n - 1
        G = np.zeros((M, n))
        for m in range(M):
            delta = int(self.offsets[m])
            if delta >= 0:
                G[m, :n - delta] += dx * (1.0 - h[delta:])
                G[m, delta:] -= dx * h[:n - delta]
            else:
                d = -delta
                G[m, d:] += dx * (1.0 - h[:n - d])
                G[m, :n - d] -= dx * h[d:]
        return G

        return x[self.n]

        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def ineq_con(self, x):
        h = x[:self.n]
        t = x[self.n]
        return t - self._profile(h)

    def ineq_con_grad(self, x):
        h = x[:self.n]
        G = self._profile_grad(h)
        return np.hstack([-G, np.ones((len(G), 1))])


def solve_epigraph(h_start, maxiter=400, time_limit=90.0):
    t0 = time.perf_counter()
    n = len(h_start)
    dx = 2.0 / n
    state = EpigraphNLP(n, dx)
    t_init = exact_score(h_start) + 0.005
    x0 = np.zeros(n + 1)
    x0[:n] = h_start
    x0[n] = t_init
    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 0.2)]
    constraints = [
        {'type': 'eq', 'fun': state.eq_con, 'jac': state.eq_con_grad},
        {'type': 'ineq', 'fun': state.ineq_con, 'jac': state.ineq_con_grad}
    ]
    try:
        res = minimize(
            state.objective, x0, jac=state.obj_grad,
            bounds=bounds, constraints=constraints,
            method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12}
        )
    except Exception:
        return None
    if time.perf_counter() - t0 > time_limit:
        return None
    if res.x is not None and np.all(np.isfinite(res.x)):
        return repair_to_feasible(res.x[:n])
    return None


def active_gradient_descent(h, n_steps=200, margin=0.004):
    n = len(h)
    dx = 2.0 / n
    state = EpigraphNLP(n, dx)
    best_h = h.copy()
    best_score = exact_score(h)
    alphas = np.array([0.02, 0.01, 0.005, 0.002, 0.001, 0.005, 0.003, 0.001, 0.0005])
    for step in range(n_steps):
        P = state._profile(best_h)
        C = np.max(P)
        active = P >= C - margin
        if not active.any():
            active[np.argmax(P)] = True
        G = state._profile_grad(best_h)
        D = G[active, :].mean(axis=0)
        D = D - D.mean()
        dn = np.max(np.abs(D))
        if dn < 1e-12:
            break
        D = D / dn * 0.01
        improved = False
        for a in alphas:
            h_new = best_h + a * D
            h_new = repair_to_feasible(h_new)
            s = exact_score(h_new)
            if s < best_score - 1e-14:
                best_h = h_new
                best_score = s
                improved = True
                break
        if not improved:
            margin *= 0.6
            if margin < 5e-5:
                break
    return best_h, best_score


def symmetrize(h):
    h_rev = h[::-1].copy()
    h_sym = 0.5 * (h + h_rev)
    return repair_to_feasible(h_sym)


def perturbed_start(h, seed, scale=0.01):
    rng = np.random.RandomState(seed)
    z = rng.uniform(-1, 1, len(h))
    z = z - z.mean()
    h_new = h + scale * z
    return repair_to_feasible(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        h_raw = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError):
        h_raw = np.full(70, 0.5)

    best_h = repair_to_feasible(h_raw)
    best_score = exact_score(best_h)

    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Stage 1: SLSQP from initial
    if time_left() > 120:
        h_sol = solve_epigraph(best_h, maxiter=600, time_limit=min(100, time_left() - 60))
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-13:
                best_h = h_sol
                best_score = s

    # Stage 2: Gradient descent
    if time_left() > 30:
        h_gd, s_gd = active_gradient_descent(best_h, n_steps=300, margin=0.004)
        if s_gd < best_score - 1e-13:
            best_h = h_gd
            best_score = s_gd

    # Stage 3: SLSQP from improved
    if time_left() > 120:
        h_sol2 = solve_epigraph(best_h, maxiter=600, time_limit=min(100, time_left() - 60))
        if h_sol2 is not None:
            s2 = exact_score(h_sol2)
            if s2 < best_score - 1e-13:
                best_h = h_sol2
                best_score = s2

    # Stage 4: Gradient descent again
    if time_left() > 30:
        h_gd2, s_gd2 = active_gradient_descent(best_h, n_steps=300, margin=0.002)
        if s_gd2 < best_score - 1e-13:
            best_h = h_gd2
            best_score = s_gd2

    # Stage 5: Perturbed SLSQP starts
    if time_left() > 200:
        n_cur = len(best_h)
        for k in range(3):
            if time_left() < 80:
                break
            h_pert = perturbed_start(best_h, seed + k * 100, scale=0.005)
            h_sol3 = solve_epigraph(h_pert, maxiter=400, time_limit=min(70, time_left() - 40))
            if h_sol3 is not None:
                s3 = exact_score(h_sol3)
                if s3 < best_score - 1e-13:
                    best_h = h_sol3
                    best_score = s3

    # Stage 6: Gradient descent polish
    if time_left() > 30:
        h_gd3, s_gd3 = active_gradient_descent(best_h, n_steps=400, margin=0.001)
        if s_gd3 < best_score - 1e-13:
            best_h = h_gd3
            best_score = s_gd3

    # Stage 7: Resolution trial (double)
    n_cur = len(best_h)
    if time_left() > 200 and n_cur <= 90:
        n_new = min(2 * n_cur, 140)
        h_interp = repair_to_feasible(interpolate_to_n(best_h, n_new))
        if time_left() > 120:
            h_sol4 = solve_epigraph(h_interp, maxiter=500, time_limit=min(90, time_left() - 60))
            if h_sol4 is not None:
                s4 = exact_score(h_sol4)
                if s4 < best_score - 1e-13:
                    best_h = h_sol4
                    best_score = s4
        if time_left() > 30:
            h_gd4, s_gd4 = active_gradient_descent(h_interp, n_steps=200, margin=0.003)
            if s_gd4 < best_score - 1e-13:
                best_h = h_gd4
                best_score = s_gd4

    # Stage 8: One more SLSQP + GD cycle
    if time_left() > 100:
        h_sol5 = solve_epigraph(best_h, maxiter=500, time_limit=min(70, time_left() - 30))
        if h_sol5 is not None:
            s5 = exact_score(h_sol5)
            if s5 < best_score - 1e-13:
                best_h = h_sol5
                best_score = s5

    if time_left() > 30:
        h_gd5, s_gd5 = active_gradient_descent(best_h, n_steps=300, margin=0.001)
        if s_gd5 < best_score - 1e-13:
            best_h = h_gd5
            best_score = s_gd5

    # Stage 9: Final micro gradient descent
    if time_left() > 15:
        h_gd6, s_gd6 = active_gradient_descent(best_h, n_steps=200, margin=0.0005)
        if s_gd6 < best_score - 1e-14:
            best_h = h_gd6
            best_score = s_gd6

    best_h = repair_to_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
