def project_box_sum(x, s):
    """Project x onto {0<=x<=1, sum(x)=s}"""
    n = len(x)
    x = np.array(x, dtype=float)
    
    # Bisection on lambda: x_i(lambda) = clip(x_i - lambda, 0, 1)
    # sum(x_i(lambda)) is decreasing in lambda
    lo = x.min() - 1.0  # ensures all x_i - lo > 1
    hi = x.max() + 1.0  # ensures all x_i - hi < 0
    
    # We want sum(clip(x_i - lam, 0, 1)) = s
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        total = proj.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Exact repair: fix the sum
    residual = s - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass; add to positions with value < 1
            slacks = 1.0 - result
            slacks[result >= 1.0 - 1e-14] = 0.0
            if slacks.sum() > 1e-14:
                result += residual * slacks / slacks.sum()
            else:
                # All at 1, shouldn't happen if s <= n
                result[np.argsort(result)[::-1][:max(1, int(abs(residual)))]] += np.sign(residual) * 1e-14
        else:
            # Need to remove mass; remove from positions with value > 0
            masses = result.copy()
            masses[result <= 1e-14] = 0.0
            if masses.sum() > 1e-14:
                result -= abs(residual) * masses / masses.sum()
    
    # Final clip (just in case of floating point)
    result = np.clip(result, 0.0, 1.0)
    # Re-fix sum after clip
    residual = s - result.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            slacks = 1.0 - result
            slacks[result >= 1.0 - 1e-14] = 0.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    max_corr = np.max(corr)
    exponents = (corr - max_corr) / tau
    weights = np.exp(exponents)
    weights /= weights.sum()

    # Gradient: d(corr[k])/d(h[i]) = (1-h[i-k]) * [i-k valid] - h[i+k] * [i+k valid]
    # where valid means 0 <= i-k < n and 0 <= i+k < n AND the corresponding n is valid

    grad = np.zeros(n)

            i2 = n_val      # index in h for h[n_val]
            grad[i1] += w * (1.0 - h[i2])
            grad[i2] -= w * h[i1]

            cap2 = min(h[j], 1.0 - h[i])
            if cap2 < 1e-9:
                continue
            i, j = j, i
            cap = cap2
        d = cap * rng.uniform(0.02, 1.0)
        h[i] -= d
        h[j] += d
        new_c = eval_c5(h)
        n_total += 1
        delta = new_c - current_c
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            current_c = new_c
            n_accept += 1
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            h[i] += d
            h[j] -= d
        if n_total % 1000 == 0:
            rate = n_accept / max(1, n_total)
            if rate < 0.1:
                temp *= 0.9
            elif rate > 0.6:
                temp = min(temp * 1.1, 0.05)
            n_accept = 0
            n_total = 0
            if temp < 1e-7:
                break
    return best_h, best_c

def gradient_descent(h, n, time_limit, seed):
    rng = np.random.default_rng(seed)
    current_c = eval_c5(h)
    best_h = h.copy()
    best_c = current_c
    tau = max(0.003, 0.1 * current_c)
    alpha = 0.03
    no_improve = 0
    start_t = time.perf_counter()
    while time.perf_counter() - start_t < time_limit:
        grad = smooth_gradient(h, tau)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-12:
            tau *= 0.7
            no_improve += 5
            if no_improve > 50:
                break
            continue
        direction = -grad_t / gnorm
        candidate = h + alpha * direction
        candidate = project_box_sum(candidate, n / 2.0)
        cand_c = eval_c5(candidate)
        if cand_c < current_c - 1e-13:
            h = candidate
            current_c = cand_c
            no_improve = 0
            alpha = min(alpha * 1.15, 0.2)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            alpha *= 0.5
            no_improve += 1
        if no_improve > 40:
            tau *= 0.65
            alpha = 0.03
            no_improve = 0
            if tau < 0.0005:
                break
    return best_h, best_c

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_box_sum(h_interp, n_dst / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    deadline = t0 + budget_s - 10.0

    # Get parent
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).ravel().copy()
    
    if parent is None or len(parent) < 4:
        n = 47
        parent = np.linspace(0.0, 1.0, n)
        parent = project_box_sum(parent, n / 2.0)
    
    n_parent = len(parent)
    parent = project_box_sum(parent, n_parent / 2.0)
    parent_c = eval_c5(parent)
    
    best_h = parent.copy()
    best_c = parent_c
    best_n = n_parent
    
    rng = np.random.default_rng(seed)
    
    # Phase A: Gradient descent on parent resolution
    remaining = deadline - time.perf_counter()
    if remaining > 30:
        alloc = min(remaining * 0.25, 180.0)
        h_try, c_try = gradient_descent(parent.copy(), n_parent, alloc, seed + 1)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            best_n = n_parent
    
    # Phase B: Pair transfer descent
    remaining = deadline - time.perf_counter()
    if remaining > 20:
        alloc = min(remaining * 0.25, 150.0)
        tlim = max(5.0, alloc)
        h_try, c_try = pair_transfer_descend(best_h.copy(), best_n, tlim, max_evals=300000)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    
    # Phase C: SA
    remaining = deadline - time.perf_counter()
    if remaining > 20:
        alloc = min(remaining * 0.3, 200.0)
        h_try, c_try = sa_optimize(best_h.copy(), best_n, max(5.0, alloc), seed + 10)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    
    # Phase D: More pair transfer + gradient cycles
    for cycle in range(6):
        remaining = deadline - time.perf_counter()
        if remaining < 15:
            break
        alloc = remaining * 0.12
        if alloc < 5:
            break
        # Gradient
        if remaining > 20:
            h_try, c_try = gradient_descent(best_h.copy(), best_n, min(alloc, remaining * 0.4), seed + 100 + cycle)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        # Pair transfer
        remaining = deadline - time.perf_counter()
        if remaining > 10:
            h_try, c_try = pair_transfer_descend(best_h.copy(), best_n, min(alloc, remaining * 0.4), max_evals=150000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        # SA
        remaining = deadline - time.perf_counter()
        if remaining > 10:
            h_try, c_try = sa_optimize(best_h.copy(), best_n, min(alloc * 0.5, remaining * 0.3), seed + 200 + cycle, max_iter=100000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
    
    # Phase E: Try perturbed restarts
    remaining = deadline - time.perf_counter()
    for restart in range(4):
        if remaining < 15:
            break
        alloc = remaining * 0.18
        noise_scale = 0.01 * (restart + 1)
        h_pert = best_h + noise_scale * rng.standard_normal(best_n)
        h_pert = project_box_sum(h_pert, best_n / 2.0)
        h_try, c_try = gradient_descent(h_pert, best_n, alloc * 0.4, seed + 500 + restart)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
        remaining = deadline - time.perf_counter()
        if remaining > 10 and c_try < best_c:
            h_try, c_try = pair_transfer_descend(h_try, best_n, alloc * 0.3, max_evals=80000)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
    
    # Phase F: Try different resolutions
    remaining = deadline - time.perf_counter()
    if remaining > 30:
        for n_try in [45, 49, 53, 56]:
            if remaining < 20:
                break
            h_interp = interpolate_h(best_h, n_try)
            alloc = min(remaining * 0.15, 60.0)
            h_try, c_try = pair_transfer_descend(h_interp, n_try, max(5.0, alloc * 0.5), max_evals=50000)
            remaining = deadline - time.perf_counter()
            if remaining > 10:
                h_try2, c_try2 = gradient_descent(h_try, n_try, max(5.0, alloc * 0.3), seed + 800 + n_try)
                if c_try2 < c_try:
                    h_try, c_try = h_try2, c_try2
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                best_n = n_try
            remaining = deadline - time.perf_counter()
    
    # Final polish
    remaining = deadline - time.perf_counter()
    if remaining > 10:
        h_try, c_try = pair_transfer_descend(best_h.copy(), best_n, max(5.0, remaining * 0.6), max_evals=100000)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    
    # Final validation
    best_h = np.array(best_h, dtype=float).ravel()
    best_n = len(best_h)
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c = eval_c5(best_h)
    
    # Ensure we keep the best verified result
    if final_c > best_c + 1e-12:
        # Recompute parent as fallback
        best_h = parent.copy()
        best_n = n_parent
        final_c = parent_c
    else:
        best_c = final_c
    
    return best_h, best_c, best_n
