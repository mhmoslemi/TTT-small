import numpy as np
from scipy.optimize import minimize
import time

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def _project(y, target):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.any():
            idx = np.where(mask)[0]
            i = idx[np.argmin(np.abs(x[idx] - 0.5))]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    F = np.max(c)
    shifted = alpha * (c - F)
    ms = np.max(shifted)
    exp_s = np.exp(shifted - ms)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = F + (1.0 / alpha) * (np.log(sum_exp) + ms)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-18:
            continue
        wk = w[r] * dx
        for j in range(n):
            idx1 = j - r + n - 1
            idx2 = j + r - n + 1
            ct = 0.0
            if 0 <= idx1 < n:
                ct += (1.0 - h[idx1])
            if 0 <= idx2 < n:
                ct -= h[idx2]
            grad[j] += wk * ct
    return obj, grad

def _proj_grad(h, target, alphas, iters_per_alpha, lr_scale):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_val = _evaluate(h)
    for alpha in alphas:
        lr = lr_scale / alpha
        for it in range(iters_per_alpha):
            obj, grad = _smooth_obj_grad(h, alpha)
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            h = h_new
            val = _evaluate(h)
            if val < best_val:
                best_val = val
                best = h.copy()
    return best, best_val

def _slsqp_optimize(h, target, alpha):
    n = len(h)
    def obj_func(x):
        o, _ = _smooth_obj_grad(x, alpha)
        return o
    def grad_func(x):
        _, g = _smooth_obj_grad(x, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-12})
    if res.success:
        x = _project(res.x, target)
        return x, _evaluate(x)
    return h, _evaluate(h)

def _structured_binary(n):
    """Create balanced binary sequences with structure (e.g., blocks, alternating)."""
    target = n // 2
    h = np.zeros(n)
    # Try a few structured patterns
    # Pattern 1: blocks of 1s then 0s (tune block length)
    # Pattern 2: alternating
    # Pattern 3: golden ratio distribution
    # Return the best of a few patterns
    candidates = []
    
    # Alternating
    h1 = np.zeros(n)
    h1[0::2] = 1.0
    if h1.sum() != target:
        h1[-1] = 1.0 - h1[-1]  # adjust
        if h1.sum() != target:
            # find a pair to fix
            if h1.sum() > target:
                h1[np.where(h1==1.0)[0][0]] = 0
            else:
                h1[np.where(h1==0.0)[0][0]] = 1
    candidates.append(h1.copy())
    
    # Golden ratio spacing
    from math import floor
    phi = (1 + 5**0.5) / 2
    h2 = np.zeros(n)
    for k in range(target):
        pos = int(floor((k + 0.5) * n / target)) % n
        h2[pos] = 1.0
    candidates.append(h2.copy())
    
    # Equally spaced
    h3 = np.zeros(n)
    for k in range(target):

        h3[pos] = 1.0
    candidates.append(h3.copy())
    
    # Random
    h4 = np.zeros(n)
    idx = np.arange(n)

def _refine_continuous(h, target, time_budget):
    """Refine a continuous solution using projected gradient then SLSQP."""
    n = len(h)
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    h2, v2 = _proj_grad(h, target, alphas, 200, 0.5)
    if v2 < _evaluate(h):

    for alpha in [50.0, 100.0, 200.0, 400.0]:

        if v3 < _evaluate(h):

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        target0 = n0 / 2.0
    else:
        n0 = 46
        h0 = np.zeros(n0)
        h0[::2] = 1.0
        target0 = n0 // 2
    
    best_h = h0.copy()
    best_c5 = _evaluate(h0)
    
    # Phase 1: Refine the initial continuous solution
    if time_left() > 60:
        h1, v1 = _refine_continuous(h0.copy(), target0, time_left)
        if v1 < best_c5:
            best_c5 = v1
            best_h = h1.copy()
    
    # Phase 2: Try different even n with binary/local search
    n_values_to_try = []
    # Include even values around the initial n
    for n_try in [32, 36, 40, 42, 44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80]:
        if n_try != n0:
            n_values_to_try.append(n_try)
    
    for n_try in n_values_to_try:
        if time_left() < 30:
            break
        target = n_try // 2
        dx = 2.0 / n_try
        
        # Try structured starts
        candidates = _structured_binary(n_try)
        # Add some random ones
        for _ in range(3):
            h_rand = np.zeros(n_try)
            idx = rng.choice(n_try, size=target, replace=False)
            h_rand[idx] = 1.0
            candidates.append(h_rand)
        
        for cand in candidates:
            if time_left() < 20:
                break
            # Quick local search
            t_for_ls = min(60, time_left() - 20)
            if t_for_ls <= 10:
                break
            # Estimate iterations
            max_iters = int(t_for_ls * 5000)  # ~5000 iters per second target
            max_iters = min(max_iters, 50000)
            h_best_ls, v_ls = _binary_local_search(n_try, rng, max_iters)
            if v_ls < best_c5:
                best_c5 = v_ls
                best_h = h_best_ls.copy()
            
            # If promising, try SA for a bit more
            if v_ls < best_c5 + 0.005 and time_left() > 40:
                max_sa = int(min(30, time_left() - 20) * 3000)
                max_sa = min(max_sa, 30000)
                h_best_sa, v_sa = _binary_sa(n_try, rng, max_sa)
                if v_sa < best_c5:
                    best_c5 = v_sa
                    best_h = h_best_sa.copy()
    
    # Phase 3: If we have a binary best, try continuous refinement at same n
    if len(best_h) % 2 == 0 and time_left() > 60:
        target = len(best_h) // 2.0
        h_cont, v_cont = _refine_continuous(best_h.copy(), target, time_left)
        if v_cont < best_c5:
            best_c5 = v_cont
            best_h = h_cont.copy()
    
    # Phase 4: Multi-start continuous optimization at various n
    if time_left() > 90:
        for n_try in [44, 46, 48, 50, 52, 56, 60]:
            if time_left() < 30:
                break
            target = n_try / 2.0
            # Start from a good binary sequence
            h_start = np.zeros(n_try)
            # Use golden ratio placement
            for k in range(n_try // 2):
                pos = int(round((k + 0.5) * n_try / (n_try // 2))) % n_try
                h_start[pos] = 1.0
            if h_start.sum() != n_try // 2:
                h_start = _project(h_start, n_try // 2)
            
            h_opt, v_opt = _refine_continuous(h_start, target, time_left)
            if v_opt < best_c5:
                best_c5 = v_opt
                best_h = h_opt.copy()
    
    # Phase 5: Final coordinate descent refinement
    if time_left() > 15:
        n = len(best_h)
        target = n / 2.0
        h_cd = best_h.copy()
        cur_val = _evaluate(h_cd)
        improved = True
        while improved and time_left() > 5:
            improved = False
            for j in range(n):
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    h_try = h_cd.copy()
                    # Move mass from j to another coordinate to maintain sum
                    # Simpler: just try small perturbations of individual components
                    # and project
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = _project(h_try, target)
                    v_try = _evaluate(h_try)
                    if v_try < cur_val - 1e-12:
                        cur_val = v_try
                        h_cd = h_try
                        improved = True
                        break
        if cur_val < best_c5:
            best_c5 = cur_val
            best_h = h_cd.copy()
    
    return best_h, best_c5, len(best_h)
