                pairs.append((min(a, b), max(a, b)))
        improved = False
        for (a, b) in pairs[:40]:
            if time.time() > deadline:
                break
            lo_t = max(-h[a], h[b] - 1.0)
            hi_t = min(1.0 - h[a], h[b])
            if hi_t - lo_t < 1e-10:
                continue
            best_t = 0.0
            best_c = C_best
            grid = np.linspace(lo_t, hi_t, 13)
            for t in grid:
                c_test = pair_line_func_setup(h, a, b, t)
                if c_test < best_c:
                    best_c = c_test
                    best_t = t
            if best_t == 0.0:
                continue
            span = (hi_t - lo_t) / 13.0
            left = max(lo_t, best_t - span)
            right = min(hi_t, best_t + span)
            if right > left + 1e-10:
                _PAIR_H = h
                _PAIR_A = a
                _PAIR_B = b
                try:
                    res = minimize_scalar(pair_line_func, bounds=(left, right),
                                          method='bounded', options={'xatol': 1e-9, 'maxiter': 18})
                    if res.success and res.fun < best_c:
                        best_t = res.x
                        best_c = res.fun
                except Exception:
                    pass
            h[a] += best_t
            h[b] -= best_t
            h = np.clip(h, 0.0, 1.0)
            delta = n / 2.0 - h.sum()
            if abs(delta) > 1e-12:
                h = project_feasible(h)
                best_c = exact_C(h)
            if best_c < C_best - 1e-12:
                C_best = best_c
                improved = True
        if not improved:
            break
    return h, C_best

def pair_line_func_setup(h, a, b, t):
    h_test = h.copy()
    h_test[a] += t
    h_test[b] -= t
    if h_test[a] < -1e-12 or h_test[a] > 1.0 + 1e-12:
        return 1.0
    if h_test[b] < -1e-12 or h_test[b] > 1.0 + 1e-12:
        return 1.0
    return exact_C(h_test)

def bfgs_optimize(h_start, deadline, rng):
    global BEST_H, BEST_C, _CURRENT_BETA
    h_start = project_feasible(h_start)
    if not is_feasible(h_start):
        return
    betas = [50.0, 120.0, 300.0, 700.0]
    y_cur = logit_from_h(h_start, eps=1e-7)
    for beta in betas:
        if time.time() > deadline - 30:
            break
        _CURRENT_BETA = beta
        try:
            maxit = 200 if len(h_start) <= 50 else 120
            res = minimize(transformed_obj_grad, y_cur, method='BFGS',
                           jac=True, options={'maxiter': maxit, 'gtol': 1e-4})
            h_end = feasible_from_logit(res.x)
            if is_feasible(h_end):
                c_end = exact_C(h_end)
                if c_end < BEST_C - 1e-12:
                    BEST_H = h_end.copy()
                    BEST_C = c_end
                y_cur = logit_from_h(h_end, eps=1e-7)
        except Exception:
            pass
    _CURRENT_BETA = 100.0

def resize_h(h, new_n):
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    global BEST_H, BEST_C, DEADLINE
    rng = np.random.default_rng(seed)
    start_time = time.time()
    DEADLINE = start_time + 0.92 * budget_s

    # Get parent
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.ndim != 1 or len(h0) == 0:
            h0 = None
    except Exception:
        h0 = None

    if h0 is not None:
        n0 = len(h0)
        if not is_feasible(h0):
            h0 = project_feasible(h0)
    else:
        n0 = 42
        h0 = np.full(n0, 0.5)

    BEST_H = h0.copy()
    BEST_C = exact_C(h0)
    n = n0

    # Build diverse starting points
    starts = [h0.copy(), h0[::-1].copy()]
    for s in range(1, min(4, n // 5 + 1)):
        starts.append(np.roll(h0, s).copy())
    # Complement
    h_comp = 1.0 - h0
    h_comp = project_feasible(h_comp)
    starts.append(h_comp)
    # Noisy
    for _ in range(5):
        h_noisy = h0 + rng.normal(0, 0.03, n)
        starts.append(project_feasible(h_noisy))
    # Sine-based
    t_arr = (np.arange(n) + 0.5) / n
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.35]:
            h_sin = 0.5 + amp * np.sin(2.0 * np.pi * freq * t_arr)
            starts.append(project_feasible(h_sin))

    # BFGS on each start
    for i, h_s in enumerate(starts):
        if time.time() > DEADLINE - 90:
            break
        bfgs_optimize(h_s, DEADLINE, rng)
        # Pair polish after each BFGS round
        if time.time() > DEADLINE - 60:
            break
        h_pol, C_pol = pair_polish(BEST_H, BEST_C, DEADLINE - 45, rng, max_moves=25)
        if C_pol < BEST_C - 1e-12:
            BEST_C = C_pol
            BEST_H = h_pol.copy()

    # Try other dimensions
    try_dims = []
    for dn in [-10, -6, -2, 2, 4, 8]:
        nd = n + dn
        if 12 <= nd <= 70 and nd != n:
            try_dims.append(nd)
    for nd in try_dims:
        if time.time() > DEADLINE - 120:
            break
        h_resized = resize_h(h0, nd)
        c_resized = exact_C(h_resized)
        if c_resized < BEST_C + 0.02:
            bfgs_optimize(h_resized, DEADLINE, rng)
            if time.time() > DEADLINE - 60:
                break
            h_pol, C_pol = pair_polish(BEST_H, BEST_C, DEADLINE - 45, rng, max_moves=30)
            if C_pol < BEST_C - 1e-12:
                BEST_C = C_pol
                BEST_H = h_pol.copy()

    # Final intensive pair polish
    if time.time() < DEADLINE - 20:
        h_pol, C_pol = pair_polish(BEST_H, BEST_C, DEADLINE, rng, max_moves=50)
        if C_pol < BEST_C - 1e-12:
            BEST_C = C_pol
            BEST_H = h_pol.copy()

    # One more BFGS from best
    if time.time() < DEADLINE - 30:
        bfgs_optimize(BEST_H, DEADLINE, rng)

    # Final validation
    h_final = BEST_H.copy()
    h_final = project_feasible(h_final)
    c_final = exact_C(h_final)
    if c_final <= BEST_C + 1e-12:
        BEST_H = h_final
        BEST_C = c_final

    return BEST_H, float(BEST_C), int(len(BEST_H))
