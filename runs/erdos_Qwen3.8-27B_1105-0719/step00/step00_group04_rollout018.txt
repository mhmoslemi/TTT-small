def solve_lp_for_h(v, n):
    """Solve LP: min t s.t. for each lag k, dx*sum h[i]*v[j] <= t, sum(h)=n/2, 0<=h<=1"""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    # Build lag constraints
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            # Pairs: h[i+k] * v[i] for i = 0, ..., n-k-1
            coeff = np.zeros(n)
            for i in range(n - k):
                coeff[i + k] = dx * v[i]
        else:
            d = -k
            # Pairs: h[i-d] * v[i] for i = d, ..., n-1
            coeff = np.zeros(n)
            for i in range(d, n):
                coeff[i - d] = dx * v[i]
        constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.ECOS)
    
    return h_var.value, t_var.value
