def _eval_c5(h):

    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

def _fix_sum(h, target_sum):
    """Fix h to have exact sum = target_sum while staying in [0,1]."""
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    if not mask.any():
        return h
    n_mask = mask.sum()
    # Check if we can achieve the target
    if diff > 0:
        max_add = (1.0 - h[mask]).sum()
        if diff > max_add:
            diff = max_add
    else:
        max_sub = h[mask].sum()
        if -diff > max_sub:
            diff = -max_sub
    h[mask] += diff / n_mask
    return np.clip(h, 0.0, 1.0)

    g = 1.0 - h
    n_lags = 2 * n - 1
