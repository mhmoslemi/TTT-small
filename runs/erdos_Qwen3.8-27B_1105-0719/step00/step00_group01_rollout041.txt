    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, lr):
    """One step of projected subgradient descent."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    j = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if j == 0:
        grad[:] = 1.0 - 2.0 * h
    elif j > 0:
        grad[:n-j] += (1.0 - h[j:])
        grad[j:] += (-h[:n-j])
    else:
        m = -j
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += (-h[m:])
    
    return h - lr * grad

    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        lr = max(lr_min, lr_init * (1.0 + 0.001 * it) ** (-0.7))
        h_new = subgradient_step(h, lr)
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:

            best_h = h_new.copy()
        h = h_new

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.sum(corr ** p) ** (1.0 / p))

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    p_inv = 1.0 / p
    base = (np.sum(corr ** p)) ** (p_inv - 1.0)
    weights = corr ** (p - 1.0) * dx  # (2n-1,)

        else:
            m = -j
            grad[m:] += w * (1.0 - h[:n-m])

def slsqp_lp(h, n, p=10.0, maxiter=200):
    """SLSQP with Lp objective."""
    target = n / 2.0
    h = project_feasible(h, target)
    
    def obj(x):
        return lp_value(x, p)
    
    def grad(x):
        return lp_grad(x, p)
    
    def eq_con(x):
        return x.sum() - target
    
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_feasible(res.x, target)
        c5 = compute_c5(h_opt)
        return h_opt, c5
