```python
import time

def compute_c5(h):
    """Compute the Erdős C5 value for a given h array."""
    n = len(h)
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1 - h, mode="full")
    return np.max(scores)

def lse_values(h, alpha, n):
    """Compute log-sum-exp smooth approximation of max score."""
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1 - h, mode="full")
    m = np.max(scores)
    return m + np.log(np.sum(np.exp(alpha * (scores - m)))) / alpha

def lse_grad(h, alpha, n):
    """Gradient of log-sum-exp smooth approximation."""
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1 - h, mode="full")
    m = np.max(scores)
    w = np.exp(alpha * (scores - m))
    w = w / np.sum(w)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r] * dx
        if wr < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[d:] += wr * (1 - h[:n-d])
                grad[:n-d] -= wr * h[d:]
        else:
            dd = -d
            if n - dd > 0:
                grad[:n-dd] += wr * (1 - h[dd:])
                grad[dd:] -= wr * h[:n-dd]
    return grad

        else:
```

```python
    m = np.max(scores)
    return m + np.log(np.sum(np.exp(alpha * (scores - m)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1.0 - h, mode="full")
    m = np.max(scores)
```

```python
def polish_subgradient(h, n_iters=200, lr_init=0.01):

        c5 = scores[r_max]

        top_k = max(1, min(5, len(scores)))
        idx = np.argsort(scores)[-top_k:]

            best_c5 = c5_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

# Global state for optimizer callbacks
_state = {}

def _obj_fun(x):
    alpha = _state['alpha']
    target = _state['target']
    mu = _state['mu']
    val, grad = lse_value_and_grad(x, alpha)
    s = np.sum(x) - target
    return val + mu * s * s

def _obj_jac(x):
    alpha = _state['alpha']
    target = _state['target']
    mu = _state['mu']
    n = _state['n']
    val, grad = lse_value_and_grad(x, alpha)
    s = np.sum(x) - target
    return grad + 2.0 * mu * s * np.ones(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(scores))

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")

def lse_value_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    scores = dx * np.correlate(h, 1.0 - h, mode="full")
    m = np.max(scores)
    a = np.clip(alpha * (scores - m), -500.0, 500.0)
    w = np.exp(a)
    w_sum = np.sum(w)
    val = m + np.log(w_sum) / alpha
    w = w / w_sum
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r] * dx
        if wr < 1e-16:
            continue
        d = r - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[d:] += wr * (1.0 - h[:n-d])
                grad[:n-d] -= wr * h[d:]
        else:
            dd = -d
            if n - dd > 0:
                grad[:n-dd] += wr * (1.0 - h[dd:])
                grad[dd:] -= wr * h[:n-dd]
    return val, grad

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def score_single_grad(h, r, n):
    dx = 2.0 / n
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            grad[d:] += dx * (1.0 - h[:n-d])
            grad[:n-d] -= dx * h[d:]
    else:
        dd = -d
        if n - dd > 0:
            grad[:n-dd] += dx * (1.0 - h[dd:])
            grad[dd:] -= dx * h[:n-dd]
    return grad

                          bounds=bounds,

def polish_subgradient(h, n_iters=500, lr_init=0.005):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    best_lr = lr
    
    for it in range(n_iters):
        scores = compute_scores(best_h)
        max_score = scores.max()
        # Weighted combination of top scores
        shifted = (scores - max_score) * 20.0
        shifted = np.clip(shifted, -500.0, 0.0)
        w = np.exp(shifted)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-10:
                continue
            grad += w[r] * score_single_grad(best_h, r, n)
        
        h_new = project_feasible(best_h - lr * grad, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            best_lr = lr
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-6:
                lr = best_lr
                # Apply small random perturbation to escape
                rng = np.random.RandomState(it)
                pert = rng.normal(0, 0.01, size=n)
                h_try = project_feasible(best_h + pert, target)
                if compute_c5(h_try) < best_c5:
                    best_h = h_try
                    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5

def make_random_init(n, rng):
    h = rng.uniform(0, 1, size=n)
    return project_feasible(h, n / 2.0)

def make_binary_init(n, rng):
    h = rng.randint(0, 2, size=n).astype(float)
    return project_feasible(h, n / 2.0)

def make_block_init(n, rng):
    bs = rng.choice([2, 3, 4, 5, 7, 10])
    h = np.zeros(n)
    pos = 0
    val = 1.0
    while pos < n:
        end = min(pos + int(bs), n)
        h[pos:end] = val
        pos = end
        val = 1.0 - val
    return project_feasible(h, n / 2.0)
```
