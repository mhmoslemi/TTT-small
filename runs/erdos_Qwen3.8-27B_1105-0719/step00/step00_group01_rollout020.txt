import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def make_step_h(n):
    h = np.zeros(n)
    h[:n//2] = 1.0
    if n % 2 == 1:
        h[n//2] = 0.5
    return h

def slsqp_active(h0, n, max_rounds=10, maxiter_per_round=200, active_tol=0.01):
    """SLSQP with active set iteration."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    
    for round_num in range(max_rounds):
        overlaps = compute_all_overlaps(h)
        t = np.max(overlaps)
        
        # Active set: constraints within active_tol of max
        active = np.where(overlaps > t - active_tol)[0]
        
        # Build and solve
        x0 = np.concatenate([h, [t]])
        
        def obj(x): return x[n]
        def obj_jac(x):
            g = np.zeros(n + 1); g[n] = 1.0; return g
        
        cons = [{
            'type': 'eq',
            'fun': lambda x: np.sum(x[:n]) - target_sum,
            'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
        }]
        
        for m in active:
            lag = m - (n - 1)
            def ineq_fun(x, m=m):
                hh = x[:n]
                return np.correlate(hh, 1.0-hh, mode="full")[m] * dx - x[n]
            def ineq_jac(x, m=m, lag=lag):
                hh = x[:n]
                g = np.zeros(n + 1)
                if lag >= 0:
                    if n - lag > 0:
                        g[lag:] += (1.0 - hh[:n-lag]) * dx
                        g[:n-lag] -= hh[lag:] * dx
                else:
                    alag = -lag
                    g[:n-alag] += (1.0 - hh[alag:]) * dx
                    g[alag:] -= hh[:n-alag] * dx
                g[n] = -1.0
                return g
            cons.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter_per_round, 'ftol': 1e-14})
        
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        # Fix sum
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h_new < 1.0
                if mask.sum() > 0:
                    h_new[mask] = np.minimum(1.0, h_new[mask] + diff / mask.sum())
            else:
                mask = h_new > 0.0
                if mask.sum() > 0:
                    h_new[mask] = np.maximum(0.0, h_new[mask] + diff / mask.sum())
        
        t_new = np.max(compute_all_overlaps(h_new))
        if t_new >= t - 1e-10:
            # No improvement, break
            break
        h = h_new
    
    return h, compute_c5(h)

        except Exception as e:

    # Try smaller n values with interpolated initial points
    n_candidates = sorted(set([
        n_init - 6, n_init - 12, n_init - 18,

    for n_try in n_candidates:

        # Interpolate from best solution
        h_interp = np.interp(
            np.linspace(0, 2, n_try, endpoint=False),

                if mask.sum() > 0:

                    h_interp = np.minimum(1.0, h_interp)
            else:

                    h_interp = np.maximum(0.0, h_interp)

        if c5_interp < best_c5:
            best_h = h_interp.copy()
            best_c5 = c5_interp
            best_n = n_try
            print(f"Interpolated n={n_try}: C5={c5_interp:.8f}")
        
        if remaining() > 30:
            try:
                h_opt, c5_opt = slsqp_active(h_interp, n_try, max_rounds=5, active_tol=0.02)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n = n_try
                    print(f"SLSQP n={n_try}: C5={c5_opt:.8f} ({elapsed():.1f}s)")
            except Exception as e:
                pass
    
    # Final SLSQP polish on best
    if remaining() > 30:

        except Exception as e:
            pass
    
    return best_h, best_c5, best_n
