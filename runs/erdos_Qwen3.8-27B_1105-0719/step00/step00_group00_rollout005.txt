def lse_smooth(corr, beta):
    max_c = np.max(corr)
    if beta <= 0:
        return max_c
    shifted = corr - max_c
    return max_c + (1.0/beta) * np.log(np.sum(np.exp(beta * shifted)))

def objective_and_grad(h, beta):
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    
    # LSE value
    max_c = np.max(corr)
    shifted = corr - max_c
    lse_val = max_c + (1.0/beta) * np.log(np.sum(np.exp(beta * shifted)))
    
    # Weights for gradient
    w = np.exp(beta * shifted) / np.sum(np.exp(beta * shifted))
    
    # Gradient
    grad = np.zeros(n)
    for s in range(2*n - 1):
        offset = s - (n - 1)
        ws = w[s]
        
        # d(corr[s])/d(h[i]) = (1-h[i+offset]) if 0<=i+offset<n - h[i-offset] if 0<=i-offset<n
        # Part 1: (1-h[i+offset]) for valid i
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:
            grad[lo1:hi1] += ws * (1.0 - h[lo1+offset:hi1+offset])
        
        # Part 2: -h[i-offset] for valid i
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= ws * h[lo2-offset:hi2-offset]
    
    return lse_val, grad

    """Project y onto {0<=x<=1, sum(x)=target_sum} using bisection on threshold."""
    # Find t such that sum(clip(y - t, 0, 1)) = target_sum
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(y - mid, 0, 1))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2
    x = np.clip(y - t, 0, 1)
    
    # Fix residual
    residual = target_sum - x.sum()
    if abs(residual) > 1e-10:
        # Distribute residual among non-boundary elements
        free = (x > 1e-10) & (x < 1 - 1e-10)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0, 1)
    
    return x

            return g[:n-1] - g[n-1]

                        h[i] = hi_val + d
                        h[j] = hj_val - d
                        v = compute_c5(h)
                        if v < best_val:
                            best_val = v
                            best_delta = d
                        h[i] = hi_val
                        h[j] = hj_val
                    if hi_val - d > 1e-12 and hj_val + d < 1.0 - 1e-12:
                        h[i] = hi_val - d
                        h[j] = hj_val + d
                        v = compute_c5(h)
                        if v < best_val:
                            best_val = v
                            best_delta = -d
                        h[i] = hi_val
                        h[j] = hj_val
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    current_c5 = best_val
                    improved = True
        if not improved:
            break
    if abs(h.sum() - target_sum) > 1e-10:
        h = project_to_feasible(h, target_sum)
    return h, compute_c5(h)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_to_feasible(h_dst, n_dst / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def deadline(frac=1.0):
        return elapsed() > budget_s * frac

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_sum_init = n_init / 2.0

    h_init = project_to_feasible(h_init, target_sum_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: LSE optimization on initial n
    betas = [5, 10, 20, 50, 100, 200, 400]
    h_opt = optimize_lse(h_init, n_init, betas, max_iter=300)
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt

    # Phase 2: Coarse pair search
    if not deadline(0.15):
        h_ps, c5_ps = pair_search(best_h, n_init, max_passes=8)
        if c5_ps < best_c5:
            best_h = h_ps.copy()
            best_c5 = c5_ps

    # Phase 3: LSE again on improved point
    if not deadline(0.25):
        betas2 = [50, 100, 200, 400, 800]
        h_opt2 = optimize_lse(best_h, n_init, betas2, max_iter=300)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_h = h_opt2.copy()
            best_c5 = c5_opt2

    # Phase 4: Fine pair search
    if not deadline(0.35):
        h_fp, c5_fp = fine_pair_search(best_h, n_init, max_passes=8)
        if c5_fp < best_c5:
            best_h = h_fp.copy()
            best_c5 = c5_fp

    # Phase 5: Try different n values
    n_candidates = [50, 55, 60, 64, 72, 80]
    for n_try in n_candidates:
        if deadline(0.75):
            break
        h_start = interpolate_h(best_h, n_init, n_try)
        betas_try = [10, 30, 80, 200]
        h_try = optimize_lse(h_start, n_try, betas_try, max_iter=200)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
        # Quick pair search on new n
        if not deadline(0.78):
            h_try2, c5_try2 = pair_search(h_try, n_try, max_passes=4)
            if c5_try2 < best_c5:
                best_h = h_try2.copy()
                best_c5 = c5_try2

    # Phase 6: Multi-start on best n
    n_best = len(best_h)
    for i in range(15):
        if deadline(0.95):
            break
        rng = np.random.RandomState(seed + 500 + i)
        # Perturbation of best solution
        scale = 0.03 + 0.02 * i
        h_pert = best_h + scale * rng.randn(n_best)
        h_pert = project_to_feasible(h_pert, n_best / 2.0)
        betas_ms = [20, 60, 150, 400]
        h_ms = optimize_lse(h_pert, n_best, betas_ms, max_iter=150)
        c5_ms = compute_c5(h_ms)
        if c5_ms < best_c5:
            best_h = h_ms.copy()
            best_c5 = c5_ms

    # Phase 7: Final fine-tuning
    if not deadline(0.98):
        h_final, c5_final = fine_pair_search(best_h, len(best_h), max_passes=5)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    # Ensure exact feasibility
    n_final = len(best_h)
    if abs(best_h.sum() - n_final / 2.0) > 1e-10:
        best_h = project_to_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
