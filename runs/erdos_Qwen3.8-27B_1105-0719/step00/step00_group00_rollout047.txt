    lo = y.min() - 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.any():
            h[mask] += res / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, beta):
    m = corr.max()
    if m < 1e-15:
        return m
    return m + np.log(np.mean(np.exp((corr - m) * beta))) / beta

def make_binary(h, n):
    """Make h binary with sum exactly n/2"""
    h = np.round(h).clip(0, 1).astype(float)
    target = n / 2
    diff = int(round(target - h.sum()))
    if diff > 0:
        zeros = np.where(h < 0.5)[0]
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h > 0.5)[0]
        h[ones[:(-diff)]] = 0.0
    return h

def binary_local_search(h, n, max_rounds=50, time_limit=None, t0=None):
    """Local search on binary sequences via swaps"""
    h = h.copy()
    curr = compute_c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for rnd in range(max_rounds):
        if t0 is not None and time.time() - t0 > time_limit:
            break
        improved = False
        best_improvement = 0
        best_pair = None
        
        one_list = list(ones)
        zero_list = list(zeros)
        
        for i in zero_list:
            for j in one_list:
                # Swap: h[i] 0->1, h[j] 1->0
                h[i] = 1.0
                h[j] = 0.0
                new_c5 = compute_c5(h)
                h[i] = 0.0
                h[j] = 1.0
                
                imp = curr - new_c5
                if imp > best_improvement:
                    best_improvement = imp
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            ones.add(i)
            ones.discard(j)
            zeros.discard(i)
            zeros.add(j)
            curr -= best_improvement
            improved = True
        
        if not improved:
            break
    
    return h, curr

def continuous_refine(h, n, beta_values, max_iter=500, t0=None, time_limit=None):
    """Refine using L-BFGS-B on smooth objective"""
    h = h.astype(float).copy()
    
    for beta in beta_values:
        if t0 is not None and time.time() - t0 > time_limit:
            break
        
        z0 = h[:n-1].copy()
        
        def obj(z):
            hf = np.empty(n)
            hf[:n-1] = z
            hf[n-1] = n / 2.0 - z.sum()
            if hf.min() < -0.1 or hf.max() > 1.1:
                penalty = 10.0 * (max(0, -hf.min()) + max(0, hf.max() - 1.0))
                return 100.0 + penalty
            corr = np.correlate(hf, 1.0 - hf, mode="full")
            return smooth_max(corr, beta)
        
        def grad(z):
            hf = np.empty(n)
            hf[:n-1] = z
            hf[n-1] = n / 2.0 - z.sum()
            hf = np.clip(hf, 0.0, 1.0)
            corr = np.correlate(hf, 1.0 - hf, mode="full")
            m = corr.max()
            weights = np.exp((corr - m) * beta)
            weights /= weights.sum()
            # Gradient of smooth_max w.r.t. corr
            # d(smooth_max)/d(corr[k]) = weights[k]
            # d(corr[k])/d(h[j]) for each j
            g = np.zeros(n)
            for k_idx in range(len(corr)):
                k = k_idx - (n - 1)  # actual lag
                w = weights[k_idx]
                if w < 1e-15:
                    continue
                # d(c_k)/d(h[j]) = (1-h[j-k]) * [0<=j-k<n] - h[j+k] * [0<=j+k<n]
                # For j = 0..n-1:
                # Term 1: 1-h[j-k] if 0 <= j-k < n
                # Term 2: -h[j+k] if 0 <= j+k < n
                jk = np.arange(n) - k  # j-k for each j
                valid1 = (jk >= 0) & (jk < n)
                g[valid1] += w * (1.0 - hf[jk[valid1]])
                jk2 = np.arange(n) + k  # j+k for each j
                valid2 = (jk2 >= 0) & (jk2 < n)
                g[valid2] -= w * hf[jk2[valid2]]
            
            # Chain rule: h[j] for j < n-1 maps to z[j], h[n-1] = n/2 - sum(z)
            grad_z = g[:n-1] - g[n-1]
            return grad_z
        
        res = minimize(obj, z0, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * (n-1),
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        
        h_opt = np.empty(n)
        h_opt[:n-1] = res.x
        h_opt[n-1] = n / 2.0 - res.x.sum()
        h_opt = project_feasible(h_opt, n)
        h = h_opt
    
    return h
