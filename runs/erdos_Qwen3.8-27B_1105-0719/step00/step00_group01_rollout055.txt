import numpy as np
import time
from scipy.optimize import linprog

def objective_and_grads(h):
    """Compute all lag values and per-lag gradients."""
    N = len(h)
    dx = 2.0 / N
    h1 = 1.0 - h
    F = np.correlate(h, h1, mode="full") * dx
    # Total lags = 2N-1
    return F, dx

def compute_grad_for_lag(h, d, dx):
    """Gradient of F_d with respect to h.
    F_d = dx * sum_m h[m+d] * (1 - h[m])
    where m ranges over valid indices.
    d = lag - (N-1)  (so d can be negative or positive)
    """
    N = len(h)
    grad = np.zeros(N)
    # Valid m range: max(0, -d) <= m <= min(N-1, N-1-d)
    m_start = max(0, -d)
    m_end = min(N-1, N-1-d)  # inclusive
    if m_start > m_end:
        return grad
    for m in range(m_start, m_end + 1):
        idx_first = m + d  # index in h for h[m+d]
        idx_second = m     # index in h for h[m]
        # d/dh[idx_first] of h[idx_first]*(1-h[idx_second]) = (1-h[idx_second])
        # d/dh[idx_second] of h[idx_first]*(1-h[idx_second]) = -h[idx_first]
        grad[idx_first] += dx * (1.0 - h[idx_second])
        grad[idx_second] += -dx * h[idx_first]
    return grad

def smooth_objective(h, beta, F=None):
    """Compute softmax surrogate Phi(h) and weights."""
    if F is None:
        F, _ = objective_and_grads(h)
    Fmax = np.max(F)
    shifted = beta * (F - Fmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    Phi = Fmax + np.log(total) / beta
    weights = exp_vals / total
    return Phi, weights

def smooth_gradient(h, beta, F=None, weights=None):
    """Gradient of softmax surrogate."""
    N = len(h)
    dx = 2.0 / N
    if F is None:
        F, _ = objective_and_grads(h)
    if weights is None:
        _, weights = smooth_objective(h, beta, F)
    
    grad = np.zeros(N)
    num_lags = len(F)  # 2N-1
    for lag_idx in range(num_lags):
        if weights[lag_idx] < 1e-14:
            continue
        d = lag_idx - (N - 1)
        grad_l = compute_grad_for_lag(h, d, dx)
        grad += weights[lag_idx] * grad_l
    return grad

def project_to_feasible(h, target_sum):
    """Project h onto [0,1]^N with sum = target_sum using bisection on water-filling."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    # h = clip(h, 0, 1) first
    h = np.clip(h, 0.0, 1.0)
    
    # Bisection for lambda such that sum(clip(h[i] - lambda, 0, 1)) = target_sum
    # Actually standard projection: minimize ||x - z||^2 s.t. 0<=x<=1, sum(x)=target
    # Solution: x_i = clip(z_i - lambda, 0, 1) where lambda chosen so sum = target
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    
    # Fix any tiny rounding to make sum exact
    s = x.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # Distribute to non-boundary elements
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.any():
            n_mask = mask.sum()
            x[mask] += diff / n_mask
            x = np.clip(x, 0.0, 1.0)
            # Final tiny fix
            s2 = x.sum()
            diff2 = target_sum - s2
            if abs(diff2) > 1e-15:
                idx = int(np.argmax(mask.astype(int)))
                x[idx] += diff2
                x = np.clip(x, 0.0, 1.0)
    
    return x

def two_point_transfer(h, target_sum, rng, max_delta=0.05):
    """Random transfer between two points preserving sum."""
    N = len(h)
    h = h.copy()
    i = rng.integers(N)
    j = rng.integers(N)
    if i == j:
        return h
    # Determine valid delta range
    if h[i] > h[j]:
        delta_max = min(h[i], max_delta)
        delta_max = min(delta_max, max_delta)
        delta = rng.uniform(1e-6, delta_max)
        h[i] -= delta
        h[j] += delta
    else:
        delta_max = min(1.0 - h[j], max_delta)
        delta_max = min(delta_max, max_delta)
        delta = rng.uniform(1e-6, delta_max)
        h[j] -= delta
        h[i] += delta
    # Ensure bounds
    h = np.clip(h, 0.0, 1.0)
    # Fix sum exactly
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_F(h):
    """Compute all lag values F_lag = dx * correlate(h, 1-h, 'full')[lag]"""
    N = len(h)

    return np.correlate(h, 1.0 - h, mode="full") * dx, dx

def compute_weighted_gradient(h, weights, dx):
    """Gradient of sum_lag weights[lag] * F_lag."""
    N = len(h)
    grad = np.zeros(N)
    for d in range(-(N-1), N):
        lag_idx = d + (N-1)
        w = weights[lag_idx]
        if abs(w) < 1e-15:
            continue
        m_start = max(0, -d)
        m_end = min(N-1, N-1-d)
        if m_start > m_end:

        idx0 = np.arange(m_start, m_end+1)  # m indices
        idx1 = idx0 + d  # m+d indices
        grad[idx1] += w * dx * (1.0 - h[idx0])
        grad[idx0] += -w * dx * h[idx1]

import numpy as np
import time
from scipy.optimize import minimize

def compute_F(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_grad(h, d, dx):
    N = len(h)
    g = np.zeros(N)
    ms = max(0, -d)
    me = min(N - 1, N - 1 - d)
    if ms > me:
        return g
    i0 = np.arange(ms, me + 1)
    i1 = i0 + d
    g[i1] += dx * (1.0 - h[i0])
    g[i0] += -dx * h[i1]
    return g

def soft_phi_weights(h, beta):
    F = compute_F(h)
    mx = F.max()
    s = np.clip(beta * (F - mx), -700, 700)
    e = np.exp(s)
    tot = e.sum()
    if tot < 1e-300:
        tot = 1.0
    w = e / tot
    phi = mx + np.log(tot) / beta
    return phi, w, F

def soft_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    _, w, F = soft_phi_weights(h, beta)
    g = np.zeros(N)
    for k in range(len(F)):
        if w[k] < 1e-15:
            continue
        d = k - (N - 1)
        g += w[k] * lag_grad(h, d, dx)
    return g

def proj(z, target):

    lo = z.min() - 2.0

            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    if abs(diff) > 1e-13:
