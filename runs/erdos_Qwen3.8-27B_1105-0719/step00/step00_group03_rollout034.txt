C(h) = max_k dx * sum_i h_i * (1 - h_{i+k})

This is a max of linear-in-h terms (each term is bilinear in h). The subgradient of the max is the gradient of the active (maximizing) terms.

For a specific lag k with index r = n-1+k:
f_k(h) = dx * sum_i h_i * (1 - h_{i+k})

The gradient with respect to h_j:
- h_j appears in f_k when j = i (coefficient: dx * (1 - h_{j+k})) or j = i+k (coefficient: -dx * h_{j-k})
- So ∂f_k/∂h_j = dx * (1 - h_{j+k}) - dx * h_{j-k} (where terms are 0 if index out of range)

import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(scores.max()), scores

def is_feasible(h, tol=1e-8):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if h.min() < -tol or h.max() > 1.0 + tol:
        return False
    target = n / 2.0
    if abs(h.sum() - target) > tol:
        return False
    return True

def project_balanced(y):
    n = len(y)
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):

                cands = np.argsort(x)

                cands = np.argsort(x)[::-1]

                if residual > 0:

                    if amt > 0:

                        residual -= amt
                else:
                    amt = min(x[idx], -residual)
                    if amt > 0:
                        x[idx] -= amt
                        residual += amt
                if abs(residual) < 1e-12:
                    break
    return x

def compute_gradient(h, scores):
    """Compute subgradient using active (maximizing) lags."""
    n = len(h)
    dx = 2.0 / n
    max_val = scores.max()
    tol = max(1e-10, max_val * 1e-6)
    active = np.where(scores >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for r in active:
        k = r - (n - 1)
        # ∂f_k/∂h_j = dx * (1-h_{j-k})*[0<=j-k<n] - dx * h_{j+k}*[0<=j+k<n]
        # First term: j-k valid means 0 <= j-k < n, so j ranges from max(0,k) to min(n-1, n-1+k)
        # Second term: j+k valid means 0 <= j+k < n, so j ranges from max(0,-k) to min(n-1, n-1-k)
        
        if k >= 0:
            # First term: j-k in [0,n-1] => j in [k, n-1]
            j1 = np.arange(k, n)
            grad[j1] += dx * (1.0 - h[j1 - k])
            # Second term: j+k in [0,n-1] => j in [0, n-1-k]
            j2 = np.arange(0, n - k)
            grad[j2] -= dx * h[j2 + k]
        else:
            d = -k
            # First term: j-k = j+d in [0,n-1] => j in [0, n-1-d]
            j1 = np.arange(0, n - d)
            grad[j1] += dx * (1.0 - h[j1 + d])
            # Second term: j+k = j-d in [0,n-1] => j in [d, n-1]
            j2 = np.arange(d, n)
            grad[j2] -= dx * h[j2 - d]
    
    grad /= len(active)
    return grad

def resample(h, n_new):
    """Resample h of length n_old to length n_new using area preservation."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # For each new cell [i*dx_new, (i+1)*dx_new), compute the integral of h_old over that interval
    # h_old is piecewise constant on [j*dx_old, (j+1)*dx_old) with value h[j]
    
    new_h = np.zeros(n_new)
    for i in range(n_new):
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        # Find overlapping old cells
        j_start = int(x_start / dx_old)
        j_end = int(np.ceil(x_end / dx_old)) - 1
        j_start = max(0, j_start)
        j_end = min(n_old - 1, j_end)
        
        total = 0.0
        for j in range(j_start, j_end + 1):
            overlap_start = max(x_start, j * dx_old)
            overlap_end = min(x_end, (j + 1) * dx_old)
            if overlap_end > overlap_start:
                total += h[j] * (overlap_end - overlap_start)
        new_h[i] = total / dx_new
    
    return project_balanced(new_h)

def gradient_descent_step(h, scores, eta):
    """One step of projected gradient descent."""
    grad = compute_gradient(h, scores)
    h_new = h - eta * grad
    return project_balanced(h_new)

def gradient_descent(h, max_iter=1000, eta=None, tol=1e-8, time_limit=None, t_start=None):
    """Run projected gradient descent. Returns best h and best score."""
    n = len(h)
    c_best, _ = exact_score(h)
    h_best = h.copy()
    scores = None
    
    if eta is None:
        eta = 0.1 / n
    
    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        if scores is None:
            _, scores = exact_score(h)
        
        grad = compute_gradient(h, scores)
        gnorm = np.linalg.norm(grad)
        if gnorm < tol:
            break
        
        # Line search
        best_h = h
        best_c = c_best
        for trial in range(10):
            h_try = project_balanced(h - eta * grad)
            c_try, _ = exact_score(h_try)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try
            else:
                break
            eta *= 0.8
        eta = max(eta, 1e-6)
        
        h = best_h
        if best_c < c_best:
            c_best = best_c
            h_best = best_h.copy()
        scores = None  # Recompute
    
    return h_best, c_best

def anneal_optimize(h, time_limit, seed=42):
    """Optimize h using simulated annealing + gradient descent."""
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    
    h_best, c_best = exact_score(h)
    h_best = h.copy()
    h_cur = h.copy()
    
    T_start = max(0.01, 0.05 * c_best)
    T_end = 0.001
    T = T_start
    
    t_start = time.time()
    iter_count = 0
    
    while time.time() - t_start < time_limit:
        iter_count += 1
        # Decide move type
        u = rng.random()
        if u < 0.5:
            # Gradient step
            _, scores = exact_score(h_cur)
            grad = compute_gradient(h_cur, scores)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-10:
                eta = 0.5 / (gnorm * n + 1e-10)
                eta = min(eta, 0.5)
                h_try = project_balanced(h_cur - eta * grad)
            else:
                break
        elif u < 0.8:
            # Random pair transfer
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            amt = 0.05 * rng.random() * abs(h_cur[i] - 0.5) + 0.01
            if rng.random() < 0.5:
                amt = -amt
            h_try = h_cur.copy()
            h_try[i] += amt
            h_try[j] -= amt
            h_try = project_balanced(h_try)
        else:
            # Random perturbation
            idx = rng.randint(n)
            delta = 0.1 * rng.randn()
            h_try = h_cur.copy()
            h_try[idx] += delta
            h_try = project_balanced(h_try)
        
        c_try, _ = exact_score(h_try)
        c_cur, _ = exact_score(h_cur)
        
        # Acceptance criterion
        delta_c = c_try - c_cur
        if delta_c < 0 or rng.random() < np.exp(-abs(delta_c) / max(T, 1e-10)):
            h_cur = h_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_try.copy()
        
        # Cool down
        T = max(T * 0.999, T_end)
        
        # Occasionally do a mini gradient descent
        if iter_count % 50 == 0 and time.time() - t_start < time_limit - 5:
            h_cur, c_gd = gradient_descent(h_cur, max_iter=50, time_limit=3, t_start=time.time())
            if c_gd < c_best:
                c_best = c_gd
                h_best = h_cur.copy()
    
    return h_best, c_best

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(scores)), scores

def project_balanced(y):
    n = len(y)
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        interior = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            x[interior] += residual / len(interior)
            x = np.clip(x, 0.0, 1.0)
        else:
            if residual > 0:
                idxs = np.where(x < 1.0)[0]
            else:
                idxs = np.where(x > 0.0)[0]
            for idx in idxs:
                if residual > 0:
                    amt = min(1.0 - x[idx], residual)
                    x[idx] += amt
                    residual -= amt
                else:
                    amt = min(x[idx], -residual)
                    x[idx] -= amt
                    residual += amt
                if abs(residual) < 1e-13:
                    break
    return x

def compute_gradient(h, scores):
    n = len(h)
    dx = 2.0 / n
    max_val = scores.max()
    tol = max(1e-14, max_val * 1e-7)
    active = np.where(scores >= max_val - tol)[0]
    grad = np.zeros(n)
    for r in active:
        k = r - (n - 1)
        if k >= 0:
            grad[k:] += dx * (1.0 - h[:n - k])
            grad[:n - k] -= dx * h[k:]
        else:
            d = -k
            grad[:n - d] += dx * (1.0 - h[d:])
            grad[d:] -= dx * h[:n - d]
    grad /= len(active)
    return grad

def resample(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    new_h = np.zeros(n_new)
    for i in range(n_new):
        x0 = i * dx_new
        x1 = (i + 1) * dx_new
        j0 = max(0, int(x0 / dx_old))
        j1 = min(n_old, int(np.ceil(x1 / dx_old - 1e-12)))
        area = 0.0
        for j in range(j0, j1):
            cs = j * dx_old
            ce = (j + 1) * dx_old
            ov = max(0.0, min(x1, ce) - max(x0, cs))
            area += h[j] * ov
        new_h[i] = area / dx_new
    return project_balanced(new_h)

def gradient_descent(h, max_iter=5000, time_limit=60.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    n = len(h)
    c_cur, scores = exact_score(h)
    c_best = c_cur
    h_best = h.copy()
    eta = 1.0 / (n * 0.5)
    stagnant = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        grad = compute_gradient(h, scores)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        accepted = False
        for _ in range(25):
            h_try = project_balanced(h - eta * grad)
            c_try, scores_try = exact_score(h_try)
            if c_try < c_cur - 1e-13:
                h = h_try
                scores = scores_try
                c_cur = c_try
                if c_cur < c_best:
                    c_best = c_cur
                    h_best = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                eta = min(eta * 1.15, 2.0)
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            if stagnant > 50:
                h_try = project_balanced(h + 0.15 * np.random.randn(n))
                c_try, scores_try = exact_score(h_try)
                if c_try < c_best + 0.02:
                    h = h_try
                    scores = scores_try
                    c_cur = c_try
                stagnant = 0
            eta = max(eta, 0.01 / n)
    return h_best, c_best

def anneal(h, time_limit=100.0, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    c_cur, scores_cur = exact_score(h)
    c_best = c_cur
    h_best = h.copy()
    h_cur = h.copy()
    T = max(0.008, 0.04 * c_cur)
    T_min = 0.0003
    t_start = time.time()
    no_improve = 0
    while time.time() - t_start < time_limit:
        u = rng.random()
        if u < 0.45:
            grad = compute_gradient(h_cur, scores_cur)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                eta = 0.8 / (gnorm * np.sqrt(n) + 1e-10)
                eta = min(max(eta, 1e-5), 1.0)
                h_try = project_balanced(h_cur - eta * grad)
            else:
                break
        elif u < 0.70:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(0.05, 0.4) * rng.choice([1.0, -1.0])
            h_try = h_cur.copy()
            h_try[i] -= delta
            h_try[j] += delta
            h_try = project_balanced(h_try)
        elif u < 0.85:
            mask = np.abs(h_cur - 0.5) < 0.3
            if mask.sum() >= 2:
                idxs = np.where(mask)[0]
                i, j = rng.choice(idxs, size=2, replace=False)
                h_try = h_cur.copy()
                swap = h_try[i]
                h_try[i] = h_try[j]
                h_try[j] = swap
                h_try = project_balanced(h_try)
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                h_try = h_cur.copy()
                h_try[i] += rng.uniform(-0.2, 0.2)
                h_try[j] -= rng.uniform(-0.2, 0.2)
                h_try = project_balanced(h_try)
        else:
            n_p = max(1, n // 15)
            idxs = rng.choice(n, size=n_p, replace=False)
            h_try = h_cur.copy()
            h_try[idxs] += rng.uniform(-0.3, 0.3, size=n_p)

        c_try, scores_try = exact_score(h_try)
        dc = c_try - c_cur
        if dc < 0 or rng.random() < np.exp(-max(dc, 0.0) / max(T, 1e-12)):
            h_cur = h_try
            scores_cur = scores_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
                h_best = h_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T * 0.9997, T_min)
        if no_improve > 300:
            h_cur = project_balanced(h_best + 0.1 * rng.randn(n))
            c_cur, scores_cur = exact_score(h_cur)
            no_improve = 0
            T = max(T * 2.0, 0.01)
    return h_best, c_best
