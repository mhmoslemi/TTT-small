            res = solver.solve(best_h, t0, maxiter=maxiter, ftol=1e-13)
            if res.x is not None and res.x.shape[0] == n_parent + 1:
                h_sol = repair_feasible(res.x[:n_parent], n_parent)
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s_sol
        except Exception:
            pass

    # Stage 2: Multiple perturbed SLSQP starts
    elapsed = time.perf_counter() - t_start
    n_pert = 15
    for attempt in range(n_pert):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 40:
            break

        eps_scale = 0.015 + 0.005 * (attempt % 5)
        d = rng.standard_normal(best_n)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            continue
        d /= dn
        h_pert = repair_feasible(best_h + eps_scale * d, best_n)

        try:
            solver = EpiSolver(best_n, beta=0.0)
            t0 = max(exact_score(h_pert), 0.001)
            maxiter = min(100, max(30, int((budget_s - elapsed) * 0.15)))
            res = solver.solve(h_pert, t0, maxiter=maxiter, ftol=1e-12)
            if res.x is not None and res.x.shape[0] == best_n + 1:
                h_sol = repair_feasible(res.x[:best_n], best_n)
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s_sol
        except Exception:
            pass

    # Stage 3: SLSQP with regularization
    elapsed = time.perf_counter() - t_start
    for beta in [1e-5, 5e-5, 1e-4]:
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 30:
            break
        try:
            solver = EpiSolver(best_n, beta=beta)
            t0 = max(best_score, 0.001)
            maxiter = min(100, max(30, int((budget_s - elapsed) * 0.2)))
            res = solver.solve(best_h, t0, maxiter=maxiter, ftol=1e-12)
            if res.x is not None and res.x.shape[0] == best_n + 1:
                h_sol = repair_feasible(res.x[:best_n], best_n)
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s_sol
        except Exception:
            pass

    # Stage 4: Dimension transfer
    elapsed = time.perf_counter() - t_start
    dim_candidates = []
    if best_n <= 110:
        dim_candidates.append(best_n + 20)
        dim_candidates.append(best_n + 40)
    if best_n > 70:
        dim_candidates.append(best_n - 10)
        dim_candidates.append(best_n - 20)
    if best_n > 80:
        dim_candidates.append(best_n - 30)
    dim_candidates = [d for d in dim_candidates if 40 <= d <= 140]
    # Remove duplicates, keep unique
    seen = set()
    dim_unique = []
    for d in dim_candidates:
        if d not in seen and d != best_n:
            seen.add(d)
            dim_unique.append(d)

    for new_n in dim_unique[:4]:
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 40:
            break
        try:
            h_interp = interpolate_to_n(best_h, new_n)
            s_interp = exact_score(h_interp)
            if s_interp < best_score - 1e-13:
                best_h = h_interp
                best_score = s_interp
                best_n = new_n
            elif s_interp <= best_score + 1e-13:
                best_h = h_interp
                best_score = s_interp
                best_n = new_n
        except Exception:
            continue

        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 30:
            continue
        try:
            solver = EpiSolver(best_n, beta=0.0)
            t0 = max(best_score, 0.001)
            maxiter = min(100, max(30, int((budget_s - elapsed) * 0.3)))
            res = solver.solve(best_h, t0, maxiter=maxiter, ftol=1e-12)
            if res.x is not None and res.x.shape[0] == best_n + 1:
                h_sol = repair_feasible(res.x[:best_n], best_n)
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s_sol
        except Exception:
            pass

    # Stage 5: Local active descent polish
    elapsed = time.perf_counter() - t_start
    if budget_s - elapsed > 20:
        try:
            h_loc, s_loc = local_active_descent(best_h, max_iters=40, seed_offset=seed)
            if s_loc < best_score - 1e-13:
                best_h = h_loc
                best_score = s_loc
        except Exception:
            pass

    # Stage 6: Final perturbation polish
    elapsed = time.perf_counter() - t_start
    if budget_s - elapsed > 15:
        for _ in range(15):
            elapsed = time.perf_counter() - t_start
            if budget_s - elapsed < 8:
                break
            d = rng.standard_normal(best_n)
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                continue
            d /= dn
            for alpha in [0.005, 0.002, 0.001]:
                h_try = repair_feasible(best_h + alpha * d, best_n)
                s_try = exact_score(h_try)
                if s_try < best_score - 1e-13:
                    best_h = h_try
                    best_score = s_try
                    break

    # Final validation
    best_h = repair_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if not np.isfinite(final_score) or final_score > 1.0:
        best_h = repair_feasible(parent, n_parent)
        final_score = parent_score
        best_n = n_parent

    return (best_h, final_score, best_n)
