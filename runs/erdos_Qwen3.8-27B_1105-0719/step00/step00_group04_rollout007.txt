    G = np.zeros((M, n))

            h_sm = repair_feasible(h_sm)
            c_sm = exact_score(h_sm)
            if c_sm < best_c:
                best_h = h_sm
                best_c = c_sm
            # Also try from perturbed start
            pert = rng.randn(n_parent) * 0.03
            h_try = repair_feasible(best_h + pert)
            h_sm = solve_smooth_max(h_try, alpha=alpha_val, maxiter=200)
            h_sm = repair_feasible(h_sm)
            c_sm = exact_score(h_sm)
            if c_sm < best_c:
                best_h = h_sm
                best_c = c_sm
    
    # Stage 3: Local refinement
    if time_left() > 20:
        t_rem = time_left()
        alloc = min(60.0, t_rem * 0.5)
        h_loc, c_loc = local_refine_gradient(best_h, time_limit=alloc)
        if c_loc < best_c:
            best_h = h_loc
            best_c = c_loc
    
    if time_left() > 20:
        t_rem = time.perf_counter() - t_global_start
        alloc = min(40.0, (budget_s - t_rem) * 0.3)
        h_loc, c_loc = local_refine(best_h, time_limit=alloc)
        if c_loc < best_c:
            best_h = h_loc
            best_c = c_loc
    
    # Stage 4: Try different resolutions
    if time_left() > 90:
        trial_ns = []
        for dn in [-10, -6, -4, 4, 6, 10, 14]:
            nn = n_parent + dn
            if 40 <= nn <= 140:
                trial_ns.append(nn)
        for nn in trial_ns:
            if time_left() < 40:
                break
            h_try = repair_feasible(interpolate_to_n(best_h, nn))
            h_sl = solve_slsqp(h_try, maxiter=150)
            h_sl = repair_feasible(h_sl)
            c_sl = exact_score(h_sl)
            if c_sl < best_c - 1e-9:
                best_h = h_sl
                best_c = c_sl
            elif c_sl < best_c + 1e-9 and nn < len(best_h):
                # Prefer smaller if tied
                best_h = h_sl
                best_c = c_sl
    
    # Stage 5: Final SLSQP polish from best
    if time_left() > 40:
        h_final = solve_slsqp(best_h, maxiter=300)
        h_final = repair_feasible(h_final)
        c_final = exact_score(h_final)
        if c_final < best_c:
            best_h = h_final
            best_c = c_final
    
    # Stage 6: Final gradient local refine
    if time_left() > 15:
        alloc = min(30.0, time_left() * 0.5)
        h_loc, c_loc = local_refine_gradient(best_h, time_limit=alloc)
        if c_loc < best_c:
            best_h = h_loc
            best_c = c_loc
    
    # Final repair and evaluation
    best_h = repair_feasible(best_h)
    best_c = exact_score(best_h)
    
    return (best_h, best_c, len(best_h))
