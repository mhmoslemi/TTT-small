def project_box_sum(x, s):
    x = np.asarray(x, dtype=np.float64).ravel().copy()
    n = len(x)

    for _ in range(60):

        if np.sum(y) > s:

        else:

    return float(np.max(corr) * dx)

def project_box_sum(x, s):
    x = np.asarray(x, dtype=np.float64).ravel().copy()
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        if np.sum(y) > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    resid = s - np.sum(y)
    if abs(resid) > 1e-13:
        if resid > 0:
            slack = 1.0 - y
            m = slack > 1e-10
            sm = np.sum(slack[m])
            if sm > 1e-14:
                y[m] += resid * slack[m] / sm
        else:
            m = y > 1e-10
            sm = np.sum(y[m])
            if sm > 1e-14:
                y[m] += resid * y[m] / sm
    return np.clip(y, 0.0, 1.0)

def lse_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    e = np.exp((corr - cmax) / tau)
    w = e / np.sum(e)
    val = cmax + tau * np.log(np.sum(e))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            a = np.arange(sh, n)
            b = np.arange(n - sh)
            np.add.at(grad, a, (1.0 - h[b]) * wk)
            np.add.at(grad, b, -h[a] * wk)
        else:
            t = -sh
            a = np.arange(n - t)
            b = np.arange(t, n)
            np.add.at(grad, a, (1.0 - h[b]) * wk)
            np.add.at(grad, b, -h[a] * wk)
    return val, grad

def smooth_phase(h0, time_budget, rng):
    n = len(h0)
    s = n / 2.0
    h = h0.copy()
    c = eval_score(h)
    best_h = h.copy()
    best_c = c
    tau = max(0.001, 0.05 * c)
    alpha = 0.01
    t0 = time.perf_counter()
    stall = 0
    while time.perf_counter() - t0 < time_budget:
        val, grad = lse_value_grad(h, tau)
        gt = grad - np.mean(grad)
        gn = np.linalg.norm(gt)
        if gn < 1e-13:
            break
        h_new = project_box_sum(h - alpha * gt, s)
        c_new = eval_score(h_new)
        if c_new < c - 1e-12:
            h = h_new
            c = c_new
            stall = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
            alpha = min(alpha * 1.05, 0.06)
        else:
            alpha *= 0.8
            stall += 1
            if stall > 50:
                tau *= 0.7
                alpha = 0.01
                stall = 0
                if tau < 1e-5:
                    break
    return best_h, best_c

def pair_phase(h0, time_budget, rng):
    n = len(h0)
    h = h0.copy()
    c = eval_score(h)
    best_h = h.copy()
    best_c = c
    t0 = time.perf_counter()
    fracs = [0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.25, 0.5]
    stall_count = 0
    while time.perf_counter() - t0 < time_budget:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = order[ii]
            if time.perf_counter() - t0 > time_budget:
                break
            for jj in range(n):
                j = order[jj]
                if j == i:
                    continue
                cap = min(h[i], 1.0 - h[j])
                if cap < 1e-8:
                    continue
                for f in fracs:
                    d = f * cap
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    c_new = eval_score(h)
                    if c_new < c - 1e-12:
                        c = c_new
                        if c < best_c:
                            best_c = c
                            best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i] += d
                        h[j] -= d
        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0
    return best_h, best_c

def sa_phase(h0, time_budget, rng):
    n = len(h0)
    h = h0.copy()
    c = eval_score(h)
    best_h = h.copy()
    best_c = c
    t0 = time.perf_counter()
    T = max(0.0005, 0.02 * c)
    no_improve = 0
    while time.perf_counter() - t0 < time_budget:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        cap = min(h[i], 1.0 - h[j])
        if cap < 1e-8:
            continue
        amt = rng.uniform(0.005, 0.5) * cap
        if amt < 1e-10:
            continue
        h[i] -= amt
        h[j] += amt
        c_new = eval_score(h)
        delta = c_new - c
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] += amt
            h[j] -= amt
        T *= 0.9998
        if no_improve > 5000:
            break
    return best_h, best_c
