```python
def _dir_gradient(h, n, n_top=3):
    """Direction that decreases the top-n_top lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top lags
    top_indices = np.argsort(corr)[-n_top:]
    grad = np.zeros(n)
    for idx in top_indices:
        k = idx - (n - 1)  # lag
        g = np.zeros(n)
        if k >= 0:
            # f = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            # df/dh[j] = (1-h[j-k]) [if j>=k] - h[j+k] [if j+k<n]
            for j in range(n):
                val = 0.0
                if j >= k:
                    val += (1.0 - h[j - k])
                if j + k < n:
                    val -= h[j + k]
                g[j] = val
        else:
            # k < 0, let k' = -k
            kp = -k
            # f = sum_{i=0}^{n-1-kp} h[i] * (1-h[i+kp])
            # df/dh[j] = (1-h[j+kp]) [if j+kp<n] - h[j-kp] [if j>=kp]
            for j in range(n):
                val = 0.0
                if j + kp < n:
                    val += (1.0 - h[j + kp])
                if j >= kp:
                    val -= h[j - kp]
                g[j] = val
        grad -= g
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm
```

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return dx * float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _repair(h, mass):
    h = np.clip(h, 0.0, 1.0)
    r = mass - h.sum()
    if abs(r) < 1e-12:
        return h
    if r > 0:
        slack = 1.0 - h
        s = slack.sum()
        if s < 1e-14:
            return h
        h = h + r * slack / s
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h + r * h / s
    return np.clip(h, 0.0, 1.0)

def _feasible_interval(h, u):
    n = len(h)
    t_lo = 0.0
    t_hi = 0.0
    for i in range(n):
        ui = u[i]
        if abs(ui) < 1e-16:
            continue
        if ui > 0:
            lo = -h[i] / ui
            hi = (1.0 - h[i]) / ui
        else:
            hi = -h[i] / ui
            lo = (1.0 - h[i]) / ui
        if lo > t_lo:
            t_lo = lo
        if hi < t_hi:
            t_hi = hi
    if t_lo > 0:
        t_lo = 0.0
    if t_hi < 0:
        t_hi = 0.0
    return t_lo, t_hi

def _solve_ray(q0, q1, q2, t_lo, t_hi):
    L = len(q0)
    cands = [t_lo, t_hi]
    
    # Stationary points of individual quadratics
    mask = np.abs(q2) > 1e-14
    if np.any(mask):
        ts = -q1[mask] / (2.0 * q2[mask])
        valid = (ts >= t_lo - 1e-9) & (ts <= t_hi + 1e-9)
        if np.any(valid):
            cands.extend(np.clip(ts[valid], t_lo, t_hi).tolist())
    
    # Pairwise intersections
    ia = np.arange(L)
    for a in ia:
        da = q0[a] - q0[a+1:]
        db = q1[a] - q1[a+1:]
        dc = q2[a] - q2[a+1:]
        # Quadratic
        qm = np.abs(dc) > 1e-14
        if np.any(qm):
            disc = db[qm]**2 - 4.0 * dc[qm] * da[qm]
            vd = disc >= -1e-10
            if np.any(vd):
                d = np.maximum(0.0, disc[qm][vd])
                sq = np.sqrt(d)
                a_coeffs = dc[qm][vd]
                b_coeffs = db[qm][vd]
                t1 = (-b_coeffs + sq) / (2.0 * a_coeffs)
                t2 = (-b_coeffs - sq) / (2.0 * a_coeffs)
                for ta in [t1, t2]:
                    vt = (ta >= t_lo - 1e-9) & (ta <= t_hi + 1e-9)
                    if np.any(vt):
                        cands.extend(np.clip(ta[vt], t_lo, t_hi).tolist())
        # Linear
        lm = (np.abs(dc) <= 1e-14) & (np.abs(db) > 1e-14)
        if np.any(lm):
            tl = -da[lm] / db[lm]
            vt = (tl >= t_lo - 1e-9) & (tl <= t_hi + 1e-9)
            if np.any(vt):
                cands.extend(np.clip(tl[vt], t_lo, t_hi).tolist())
    
    # Dense samples for safety
    nd = max(256, 4 * L)

    scores = vals.max(axis=1)
    idx = scores.argmin()
    return cands[idx], scores[idx]

def _dir_gradient(h, n, n_top=3):
    corr = np.correlate(h, 1.0 - h, mode='full')

                    v -= h[j - kp]
                g[j] = v

    return grad / norm

    m2 = min(m1 + 1 + int(rng.integers(0, max(2, n // 7))), max(m1 + 2, n // 2))
    p1 = rng.uniform(0.0, 2.0 * np.pi)
    p2 = rng.uniform(0.0, 2.0 * np.pi)
    w1 = rng.uniform(0.5, 1.0)
    x = np.arange(n, dtype=np.float64)
    d = w1 * np.cos(2.0 * np.pi * m1 * x / n + p1) + np.sin(2.0 * np.pi * m2 * x / n + p2)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    return d / norm

def _make_fallback(n, rng):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + 0.25 * np.cos(2.0 * np.pi * x / n) + 0.1 * np.cos(4.0 * np.pi * x / n + 0.7)
    h = np.clip(h, 0.0, 1.0)
    mass = n / 2.0
    h = _repair(h, mass)
    return h

def _interp_n(h_old, n_old, n_new):
    x_old = (np.arange(n_old, dtype=np.float64) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new, dtype=np.float64) + 0.5) * (2.0 / n_new)
    h_ext = np.concatenate([h_old[-1:], h_old, h_old[:1]])
    x_ext = np.concatenate([x_old - 2.0 / n_old, x_old, x_old + 2.0 / n_old])
    h_new = np.interp(x_new, x_ext, h_ext)
    mass = n_new / 2.0
    h_new = _repair(h_new, mass)
    return h_new

def _optimize_n(h, n, t_start, deadline, rng, seed_offset=0):
    rng2 = np.random.default_rng(seed + seed_offset) if False else rng
    mass = n / 2.0
    h = _repair(h.copy(), mass)
    best_h = h.copy()
    best_c = _eval(h)
    cur_h = h.copy()
    cur_c = best_c
    pool = [(cur_h.copy(), cur_c)]
    stagnation = 0
    while time.perf_counter() < deadline:
        for _ in range(10):
            if time.perf_counter() > deadline:
                break
            r = rng2.random()
            if r < 0.25:
                d = _dir_random(rng2, n)
            elif r < 0.45:
                d = _dir_sinusoidal(rng2, n)
            elif r < 0.60:
                d = _dir_translation(cur_h, rng2, n)
            elif r < 0.75:
                d = _dir_gradient(cur_h, n)
            elif r < 0.85 and len(pool) >= 2:
                i = int(rng2.integers(0, len(pool)))
                j = int(rng2.integers(0, len(pool)))
                if i != j:
                    d = pool[i][0] - pool[j][0]
                    d -= d.mean()
                    nm = np.linalg.norm(d)
                    d = d / nm if nm > 1e-15 else _dir_random(rng2, n)
                else:
                    d = _dir_random(rng2, n)
            else:
                d = _dir_multifreq(rng2, n)
            if d is None:
                d = _dir_random(rng2, n)
            h_new, c_new = _optimize_ray(cur_h, d, n)
            if c_new < cur_c - 1e-10:
                cur_h = h_new
                cur_c = c_new
                stagnation = 0
            else:
                stagnation += 1
            if c_new < best_c - 1e-10:
                best_h = h_new.copy()
                best_c = c_new
                pool.append((best_h.copy(), best_c))
                if len(pool) > 30:
                    pool.pop(0)
        if stagnation > 60 and len(pool) > 1:
            idx = int(rng2.integers(0, len(pool)))
            cur_h = pool[idx][0].copy()
            cur_c = _eval(cur_h)
            pert = rng2.standard_normal(n) * 0.03
            pert -= pert.mean()
            cur_h = _repair(cur_h + pert, mass)
            cur_c = _eval(cur_h)
            stagnation = 0
    return best_h, best_c, n

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + max(5.0, budget_s - 3.0)
    rng = np.random.default_rng(seed)
    
    try:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        h_init = _make_fallback(47, rng)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c = _eval(h_init)
    best_n = n0
    
    # Phase 1: Optimize parent n
    t_phase2 = t_start + 0.65 * (budget_s - 5.0)
    best_h, best_c, best_n = _optimize_n(h_init, n0, t_start, t_phase2, rng)
    
    # Phase 2: Try other n values
    t_end = deadline
    alt_ns = [n0 - 4, n0 - 2, n0 + 2, n0 + 4, n0 - 6, n0 + 6]
    alt_ns = [x for x in alt_ns if x >= 17 and x <= 80 and x != n0]
    # Deduplicate and limit
    seen = set()
    alt_ns = [x for x in alt_ns if x not in seen and not seen.add(x)][:5]
    
    if alt_ns and time.perf_counter() < t_end:
        per_trial = (t_end - time.perf_counter()) / len(alt_ns)
        for ai, an in enumerate(alt_ns):
            trial_deadline = min(t_end, time.perf_counter() + per_trial)
            if trial_deadline <= time.perf_counter() + 2:
                continue
            h_trial = _interp_n(best_h, best_n, an)
            hh, cc, nn = _optimize_n(h_trial, an, t_start, trial_deadline, rng)
            if cc < best_c - 1e-10:
                best_h = hh
                best_c = cc
                best_n = nn
    
    return (best_h, best_c, best_n)
