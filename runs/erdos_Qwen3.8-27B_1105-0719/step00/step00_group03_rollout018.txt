def gradient_descent_refine(h, rng, max_iter=2000, time_limit=30.0):

    n = len(h)
    h = project_balanced(h.copy())
    scores, c_val = exact_scores(h)
    best_h = h.copy()
    best_c = c_val
    t0 = time.time()
    
    # Adaptive epsilon
    eps = 0.05
    # Adaptive step size
    step0 = 0.1
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        scores, c_val = exact_scores(h)
        if c_val < best_c - 1e-10:
            best_c = c_val
            best_h = h.copy()
        
        grad = active_gradient(h, scores, epsilon=eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:

        # Line search
        g = -grad / gnorm
        improved = False
        for trial in range(8):

        else:
            step0 *= 0.5
            eps = min(eps * 1.3, 0.1)

                break

def resample(h_old, n_new):
    """Resample function from n_old to n_new points using area preservation."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    h_new = np.zeros(n_new)
    for i in range(n_new):
        # New cell [i*dx_new, (i+1)*dx_new]
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        # Find overlapping old cells
        old_start = int(x_start / dx_old)
        old_end = int(np.ceil(x_end / dx_old)) - 1
        old_start = max(0, old_start)
        old_end = min(n_old - 1, old_end)
        
        total_area = 0.0
        for j in range(old_start, old_end + 1):
            cell_start = j * dx_old
            cell_end = (j + 1) * dx_old
            overlap_start = max(x_start, cell_start)
            overlap_end = min(x_end, cell_end)
            if overlap_end > overlap_start:
                total_area += h_old[j] * (overlap_end - overlap_start)
        
        h_new[i] = total_area / dx_new
    
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_balanced(h_new)
    return h_new

    for pos in positions:
        width = rng.integers(1, max(2, n // (2 * n_blocks)))
        for j in range(pos, min(n, pos + width)):
            h[j] = 1.0
    # Add some randomness
    h += rng.uniform(0, 0.3, n)
    h = np.clip(h, 0, 1)
    h = project_balanced(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return max(0, budget_s - elapsed() - 3.0)
    
    # Get initial values
    best_h = None
    best_c = np.inf
    best_n = 0
    
    initial = None
    try:
        initial = initial_h_values
    except NameError:
        initial = None
    
    if initial is not None and len(initial) > 0:
        h0 = np.array(initial, dtype=float).copy()
        h0 = project_balanced(h0)
        scores, c0 = exact_scores(h0)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
            best_n = len(h0)
    else:
        # Generate initial
        n0 = 45
        h0 = generate_random_start(n0, rng)
        scores, c0 = exact_scores(h0)
        best_c = c0
        best_h = h0.copy()
        best_n = n0
    
    print(f"Initial: n={best_n}, C={best_c:.6f}")
    
    # Phase 1: Binary local search on initial
    if remaining() > 15:
        t_lim = min(15, remaining() - 2)
        b, bc = swap_search_binary(best_h, rng, max_iter=8000, time_limit=t_lim)
        if bc < best_c:
            best_c = bc
            best_h = b
            best_n = len(b)
            print(f"After binary search: C={best_c:.6f}")
    
    # Phase 2: Gradient descent refinement
    if remaining() > 20:
        t_lim = min(25, remaining() - 2)
        h_g, c_g = gradient_descent_refine(best_h, rng, max_iter=3000, time_limit=t_lim)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
            best_n = len(h_g)
            print(f"After gradient descent: C={best_c:.6f}")
    
    # Phase 3: Simulated annealing
    if remaining() > 50:
        t_lim = min(200, remaining() - 5)
        h_sa, c_sa = simulated_annealing(best_h, rng, t_start=0.03, t_end=0.0005,
                                          cooling=0.9995, max_iter=100000, time_limit=t_lim)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
            best_n = len(h_sa)
            print(f"After SA round 1: C={best_c:.6f}")
    
    # Phase 4: Try different resolutions
    if remaining() > 60:
        n_current = best_n
        resolutions = [n_current]
        if n_current <= 50:
            resolutions.extend([64, 80, 100, 128, 160])
        elif n_current <= 100:
            resolutions.extend([128, 160, 200, 256])
        else:
            resolutions.extend([int(n_current * 0.75), int(n_current * 1.25)])
        resolutions = [r for r in resolutions if r % 2 == 0 and 10 <= r <= 300]
        
        for n_new in resolutions:
            if remaining() < 30:
                break
            if n_new == n_current:
                continue
            h_res = resample(best_h, n_new)
            scores, c_res = exact_scores(h_res)
            print(f"  Resampled to n={n_new}: C={c_res:.6f}")
            
            # Quick binary search on resampled
            if remaining() > 20:
                b, bc = swap_search_binary(h_res, rng, max_iter=3000, time_limit=min(10, remaining()-3))
                if bc < c_res:
                    c_res = bc
                    h_res = b
                    print(f"  After binary on n={n_new}: C={c_res:.6f}")
            
            # Quick gradient
            if remaining() > 25:
                h_g2, c_g2 = gradient_descent_refine(h_res, rng, max_iter=1500, time_limit=min(15, remaining()-3))
                if c_g2 < c_res:
                    c_res = c_g2
                    h_res = h_g2
                    print(f"  After gradient on n={n_new}: C={c_res:.6f}")
            
            if c_res < best_c:
                best_c = c_res
                best_h = h_res
                best_n = n_new
                print(f"  New best: n={n_new}, C={best_c:.6f}")
    
    # Phase 5: More SA if time
    if remaining() > 40:
        # Always use the best we have
        t_lim = min(remaining() - 5, 150)
        h_sa2, c_sa2 = simulated_annealing(best_h, rng, t_start=0.015, t_end=0.0003,
                                            cooling=0.9997, max_iter=100000, time_limit=t_lim)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2
            best_n = len(h_sa2)
            print(f"After SA round 2: C={best_c:.6f}")
    
    # Phase 6: Final binary polish
    if remaining() > 10:
        b, bc = swap_search_binary(best_h, rng, max_iter=10000, time_limit=min(8, remaining()-2))
        if bc < best_c:
            best_c = bc
            best_h = b
            best_n = len(b)
            print(f"Final binary polish: C={best_c:.6f}")
    
    # Final gradient polish
    if remaining() > 8:
        h_final, c_final = gradient_descent_refine(best_h, rng, max_iter=2000, time_limit=min(5, remaining()-2))
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
            best_n = len(h_final)
            print(f"Final gradient: C={best_c:.6f}")
    
    # Final verification
    final_h = project_balanced(best_h.copy())
    scores_f, c_f = exact_scores(final_h)
    if c_f <= best_c + 1e-10:
        best_h = final_h
        best_c = c_f
    else:
        # Shouldn't happen, but be safe
        best_h = project_balanced(best_h)
        _, best_c = exact_scores(best_h)
    
    print(f"FINAL: n={len(best_h)}, C5={best_c:.6f}, time={elapsed():.1f}s")
    return best_h, best_c, len(best_h)
