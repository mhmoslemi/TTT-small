```python
        best_h = h.copy()
    
    # Phase 2: Multi-start with random perturbations
    while time.time() < deadline:
        # Perturb
        h = best_h.copy()
        n_pert = rng.integers(1, 5)
        for _ in range(n_pert):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j: continue
            S = h[i] + h[j]
            lo = max(0, S-1); hi = min(1, S)
            if hi - lo > 0.01:
                h[i] = rng.uniform(lo, hi)
                h[j] = S - h[i]
        
        cur_c5 = _eval_c5(h)
        
        # Optimize
        time_left = deadline - time.time()
        if time_left < 1: break
        max_opt_time = min(time_left * 0.8, budget_s * 0.15)
        cur_c5 = _optimize_pairs(h, cur_c5=cur_c5, deadline=deadline, max_time=max_opt_time)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    final_c5 = _eval_c5(best_h)
    return (best_h, final_c5, n)
```

This is cleaner. The `_optimize_pairs` function runs the deterministic pairwise optimization for a limited time and returns the best C5 found.
