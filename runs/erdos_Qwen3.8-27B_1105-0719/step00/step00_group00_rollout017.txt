One more consideration: the subgradient method for min-max problems. The standard approach is:
- At iteration t, compute g_t = ∇f_{i_t}(x_t) where i_t is an active (near-max) component
- Update: x_{t+1} = P(x_t - α_t * g_t) where P is projection onto feasible set

import numpy as np
from scipy.optimize import minimize
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h):
    n = len(h)
    return float(np.max(_compute_corr(h))) * 2.0 / n

def _project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += residual / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _corr_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[k:] += (1.0 - h[:nk])
            grad[:nk] -= h[k:]
    else:
        nk = -k
        nnk = n - nk
        if nnk > 0:
            grad[:nnk] += (1.0 - h[nk:])
            grad[nk:] -= h[:nnk]
    return grad

def _subgradient_obj(x, n, beta):
    c = _compute_corr(x)
    scaled = c * beta
    mc = scaled.max()
    return float((mc + np.log(np.sum(np.exp(scaled - mc)))) / beta)

def _subgradient_grad(x, n, beta):
    c = _compute_corr(x)
    scaled = c * beta
    mc = scaled.max()
    ex = np.exp(scaled - mc)
    w = ex / ex.sum()
    grad = np.zeros(n)
    nfull = len(c)
    for mi in range(nfull):
        if w[mi] > 1e-10:
            k = mi - (n - 1)
            grad += w[mi] * _corr_gradient(x, k)
    return grad

def _penalty_obj(x, n, beta, mu):
    return _subgradient_obj(x, n, beta) + mu * (x.sum() - n / 2.0) ** 2

def _penalty_grad(x, n, beta, mu):
    return _subgradient_grad(x, n, beta) + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)

def _run_subgradient(h, n, time_fn, time_limit, lr_scale=1.0):
    momentum = np.zeros(n)
    beta_m = 0.85
    best_local = h.copy()
    best_c5 = _compute_c5(h)
    phase = 0
    while time_fn() < time_limit:
        lr = 0.06 * lr_scale / (1.0 + 0.4 * phase)
        for _ in range(30000):
            if time_fn() >= time_limit:
                break
            corr = _compute_corr(h)
            top_n = min(3, len(corr))
            top_idx = np.argsort(corr)[-top_n:]
            w = np.exp(corr[top_idx] * 25.0)
            w /= w.sum()
            grad = np.zeros(n)
            for wi, mi in zip(w, top_idx):
                k = mi - (n - 1)
                grad += wi * _corr_gradient(h, k)
            momentum = beta_m * momentum + (1.0 - beta_m) * grad
            h_new = _project_feasible(h - lr * momentum, n)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_local = h_new.copy()
            h = h_new
        phase += 1
        momentum = np.zeros(n)
    return best_local, best_c5

def _make_block_structure(n):
    half = n // 2
    h = np.zeros(n)
    h[:half] = 1.0
