```python
def compute_lse_grad(n, dx, weights, h):
    """Compute gradient of LSE-smoothed objective."""
    # weights has length 2n-1
    # (A*h)[j] = sum_{t=0}^{n-1} w[n-1-j+t] * h[t]
    # (B*h)[j] = sum_{t=0}^{n-1} w[n-1+j-t] * h[t]
    
    # Build index matrices
    j = np.arange(n)  # 0, 1, ..., n-1
    t = np.arange(n)  # 0, 1, ..., n-1
    
    # For A: index = n-1-j+t, shape (n, n), j is row, t is col
    idx_A = (n-1) - j[:, None] + t[None, :]  # shape (n, n)
    
    # For B: index = n-1+j-t, shape (n, n)
    idx_B = (n-1) + j[:, None] - t[None, :]  # shape (n, n)
    
    Ah = weights[idx_A] @ h  # shape (n,)
    Bh = weights[idx_B] @ h  # shape (n,)
    
    W_total = np.sum(weights)
    grad = dx * (W_total - Ah - Bh)
    return grad
```

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def project_box_sum(h):
    n = len(h)
    target = n / 2.0
    x = np.array(h, dtype=np.float64).copy()
    x[~np.isfinite(x)] = 0.0
    for _ in range(300):
        diff = target - np.sum(x)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = x < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            x[mask] = np.minimum(1.0, x[mask] + diff / cnt)
        else:
            mask = x > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            x[mask] = np.maximum(0.0, x[mask] + diff / cnt)
    return np.clip(x, 0.0, 1.0)


def lse_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(profile)
    shifted = np.clip((profile - m) / tau, -500, 0)
    return m + tau * np.log(np.sum(np.exp(shifted)))


def lse_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(profile)
    shifted = np.clip((profile - m) / tau, -500, 0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    j_idx = np.arange(n)
    t_idx = np.arange(n)
    idx_A = (n - 1) - j_idx[:, None] + t_idx[None, :]
    idx_B = (n - 1) + j_idx[:, None] - t_idx[None, :]
    Ah = w[idx_A] @ x
    Bh = w[idx_B] @ x
    return dx * (1.0 - Ah - Bh)


def slsqp_optimize(h0, taus, deadline):
    """Multi-stage SLSQP with decreasing tau."""
    n = len(h0)
    h = np.array(h0, dtype=np.float64).copy()
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = exact_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    for tau in taus:
        if time.time() > deadline - 5.0:
            break
        
        def obj(x):
            return lse_obj(x, tau)
        
        def obj_grad(x):
            return lse_grad(x, tau)
        
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or ( hasattr(res, 'x') and res.x is not None):
                h_new = project_box_sum(res.x)
                c_new = exact_c5(h_new)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c


def binary_swap_search(h, max_rounds=200, deadline=None):

    n = len(h)
    if n % 2 != 0:
        return h.copy(), exact_c5(h)
