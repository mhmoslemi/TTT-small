    h = np.clip(h, 0, 1)

    lo, hi = -1.0, 1.0

def _slsqp_polish(h, target, maxiter=30):
    n = len(h)
    
    def obj(x):
        return _exact_score(np.clip(x, 0, 1))
    
    def grad(x):
        hc = np.clip(x, 0, 1)
        n = len(hc)
        corr = np.correlate(hc, 1.0-hc, mode="full")
        max_idx = int(np.argmax(corr))
        L = max_idx - (n-1)
        g = np.zeros(n)
        for m in range(n):
            if 0 <= m+L < n:
                g[m] += (1.0 - hc[m+L])
            if 0 <= m-L < n:
                g[m] -= hc[m-L]
        return g * (2.0/n)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.success or res.fun < _exact_score(h):
        h_new = _project(res.x, target)
        if _exact_score(h_new) <= _exact_score(h) + 1e-12:
            return h_new
    return h

        h_s = _slsqp_polish(best_h.copy(), target, maxiter=40)
        update_best(h_s)

    # Phase 3: Multiple diverse restarts
    starts = [_random_binary, _start_step, _start_tri, _start_cos, _start_sin, _start_mixed]
    attempts = 0
    while time.perf_counter() < hard_deadline - 30 and attempts < 60:
        start_fn = starts[attempts % len(starts)]
        h_s = start_fn(n_base, rng)
        attempts += 1
        h_opt = _pair_descent(h_s, rng, hard_deadline - 25)
        prev_c = _exact_score(h_opt)
        update_best(h_opt)
        if _exact_score(h_opt) < best_c + 0.01 and time.perf_counter() < hard_deadline - 10:
            h_sl = _slsqp_polish(h_opt.copy(), n_base / 2.0, maxiter=30)
            update_best(h_sl)
        # Mutation on current best occasionally
        if attempts % 5 == 0 and time.perf_counter() < hard_deadline - 20:
            h_mut = _mutate(best_h.copy(), rng, n_mut=2)
            h_mut_opt = _pair_descent(h_mut, rng, hard_deadline - 25)
            update_best(h_mut_opt)

    # Phase 4: Dimension transfers
    for n_tr in [n_base, max(10, n_base - 6), n_base + 6, 64, 80, 90]:
        if n_tr < 10 or n_tr > 200 or time.perf_counter() > hard_deadline - 15:
            continue
        if n_tr == n_base:
            continue
        h_tr = _interpolate(best_h, n_tr)
        h_tr_opt = _pair_descent(h_tr, rng, hard_deadline - 10)
        update_best(h_tr_opt)
        if time.perf_counter() < hard_deadline - 8:
            h_tr_sl = _slsqp_polish(h_tr_opt.copy(), n_tr / 2.0, maxiter=25)
            update_best(h_tr_sl)

    # Final: ensure feasibility and recompute
    final_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_score(final_h)
    if final_c > best_c + 1e-8:
        final_h = best_h.copy()
        final_c = best_c

    return final_h, final_c, len(final_h)
