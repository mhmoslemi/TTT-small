    """Compute C5 for a given h vector."""

def project_balanced(a, target):
    """Project a onto [0,1]^n with sum = target using L2 projection."""
    n = len(a)
    a = np.clip(a, 0, 1)
    # Bisection on lambda: x = clip(a + lambda, 0, 1), find lambda such that sum(x) = target
    lo = -1.0 - a.max()
    hi = 1.0 - a.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(a + mid, 0, 1)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(a + (lo + hi) / 2, 0, 1)
    # Fix any residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual among free variables
        if residual > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
        else:
            free = np.where(x > 1e-12)[0]
        if len(free) > 0:
            per = residual / len(free)
            x[free] = np.clip(x[free] + per, 0, 1)
    return x

    # h[p] += tau, h[q] -= tau
    # Constraints: 0 <= h[p]+tau <= 1, 0 <= h[q]-tau <= 1

    h_try[p] += c_pt

    fd = c5_evaluator(h_try)
    h_try[p] = h[p]
    h_try[q] = h[q]

            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)

- Term involving p as i: h[p+k]*(1-h[p]) = h[p+k]*0 = 0 (if p+k is valid)
- Term involving q as i+k: h[q]*(1-h[q-k]) = 0*(1-h[q-k]) = 0
- Term involving q as i: h[q+k]*(1-h[q]) = h[q+k]*1 = h[q+k] (if q+k is valid)

After the swap:
- Term involving p as i+k: h[p]*(1-h[p-k]) = 0*(1-h[p-k]) = 0
- Term involving p as i: h[p+k]*(1-h[p]) = h[p+k]*1 = h[p+k]
- Term involving q as i+k: h[q]*(1-h[q-k]) = 1*(1-h[q-k]) = 1-h[q-k]
- Term involving q as i: h[q+k]*(1-h[q]) = h[q+k]*0 = 0

So the change in c_k is:
Δc_k = [0 + h[p+k] + (1-h[q-k]) + 0] - [(1-h[p-k]) + 0 + 0 + h[q+k]]
= h[p+k] + 1 - h[q-k] - 1 + h[p-k] - h[q+k]
= h[p+k] + h[p-k] - h[q-k] - h[q+k]

(where we interpret h[index] = 0 if index is out of bounds)

So Δc_k = h[p+k] + h[p-k] - h[q-k] - h[q+k]
