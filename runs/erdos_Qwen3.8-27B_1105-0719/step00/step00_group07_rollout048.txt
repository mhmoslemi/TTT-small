def projected_gradient_descent(h, n, betas, max_iters, time_limit):
    """Run projected gradient descent with log-sum-exp."""
    start = time.time()
    target = n / 2.0
    h = h.copy()
    
    for beta in betas:
        for it in range(max_iters):
            if time.time() - start > time_limit:
                return h
            obj, grad = lse_obj_grad(h, beta)
            
            # Backtracking line search
            step = 1.0
            obj_current = obj
            for _ in range(20):
                h_new = project_box_hyperplane(h - step * grad, target)
                obj_new = lse_obj_only(h_new, beta)
                if obj_new < obj_current:
                    h = h_new
                    break
                step *= 0.5
            
            # Check convergence
            if np.linalg.norm(grad) < 1e-8:
                break
    
    return h

def project_box_hyperplane(y, target):
    n = len(y)
    target = float(max(0.0, min(float(n), target)))
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - float(np.sum(x))
    if r > 1e-12:
        order = np.argsort(1.0 - x)[::-1]
        for idx in order:
            add = min(r, 1.0 - x[idx])
            if add <= 0:

    m = float(np.max(counts))
    shifted = beta * (counts - m)
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    total = float(np.sum(exp_s))
    weights = exp_s / total
    
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        w = weights[t]
        if w < 1e-14:
            continue
        k = t - (n - 1)
        if k >= 0:
            j_min, j_max = 0, n - 1 - k
        else:
            j_min, j_max = -k, n - 1
        
        # Term 1: (1-x[i-k]) for i where i-k in [j_min, j_max]
        i_lo = max(0, j_min + k)
        i_hi = min(n - 1, j_max + k)

            grad[i] += w * (1.0 - x[i - k])

        i_lo2 = max(0, j_min)

        for i in range(i_lo2, i_hi2 + 1):
            ki = i + k
            if 0 <= ki < n:

    return grad
