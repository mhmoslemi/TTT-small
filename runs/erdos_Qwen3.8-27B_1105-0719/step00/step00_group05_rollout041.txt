import time

def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(y, n):
    """Project y onto {x : 0 <= x <= 1, sum(x) = n/2}."""
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix numerical issues
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_of_active(h, n, s):
    """Compute subgradient of f_s at h, where s is the shift index in corr (0 to 2n-2)."""
    dx = 2.0 / n
    # The shift in terms of the actual offset is: tau = s - (n-1)
    tau = s - (n - 1)
    grad = np.zeros(n)
    if tau >= 0:
        # f_tau = dx * sum_{i=tau}^{n-1} h[i] * (1-h[i-tau])
        # d/dh[j]:
        # j appears as h[j] when j >= tau and j-tau < n (always true): contribution (1-h[j-tau])*dx
        # j appears in (1-h[i-tau]) when i-tau = j, i.e., i = j+tau: contribution -h[j+tau]*dx if j+tau < n
        valid_first = np.arange(n) >= tau
        grad[valid_first] += (1.0 - h[np.arange(n)[valid_first] - tau]) * dx
        valid_second = np.arange(n - tau)  # j from 0 to n-tau-1, j+tau < n
        grad[valid_second] += (-h[valid_second + tau]) * dx
    else:
        tau_abs = -tau
        # f_{-tau_abs} = dx * sum_{i=0}^{n-tau_abs-1} h[i] * (1-h[i+tau_abs])
        # d/dh[j]:
        # j appears as h[j] when j < n-tau_abs: contribution (1-h[j+tau_abs])*dx
        # j appears in (1-h[i+tau_abs]) when i+tau_abs = j, i.e., i = j-tau_abs: contribution -h[j-tau_abs]*dx if j >= tau_abs
        valid_first = np.arange(n - tau_abs)
        grad[valid_first] += (1.0 - h[valid_first + tau_abs]) * dx
        valid_second = np.arange(tau_abs, n)
        grad[valid_second] += (-h[valid_second - tau_abs]) * dx
    return grad

        max_idx = np.argmax(corr)
        current_c5 = corr[max_idx] * dx

def project_feasible(y, n):
    target = n / 2.0

def get_subgradient(h, n, s_index):
    dx = 2.0 / n

    else:
        ta = -tau

        grad[ta:] += (-h[:n - ta])
    return grad * dx

        x = np.linspace(0, 2 * np.pi * freq, n, endpoint=False)
        h = 0.5 + 0.3 * np.cos(x)
        h = project_feasible(h, n)
        inits.append(h.copy())
    return inits
