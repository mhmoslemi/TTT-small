import numpy as np
from scipy.optimize import minimize, linprog
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def repair_feasible(h, target=None):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
        space = (1.0 - h)[mask]
        total = space.sum()
        if total > 0:
            h[mask] += diff * space / total
    else:
        mask = h > 0.0
        headroom = h[mask]
        total = headroom.sum()
        if total > 0:
            h[mask] += diff * headroom / total
    h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_dst = (np.arange(new_n) + 0.5) * 2.0 / new_n
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_dst, x_ext, h_ext)
    return repair_feasible(h_new, new_n / 2.0)


def lag_gradient(h, delta):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if delta == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif 0 < delta < n:
        grad[0:n-delta] += dx * (1.0 - h[delta:n])
        grad[delta:n] -= dx * h[0:n-delta]
    elif -n < delta < 0:
        d = -delta
        grad[d:n] += dx * (1.0 - h[0:n-d])
        grad[0:n-d] -= dx * h[d:n]
    return grad


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


class NLPProblem:
    def __init__(self, n, t_upper):
        self.n = n
        self.t_upper = t_upper
    
    def obj(self, x):
        return x[self.n]
    
    def obj_jac(self, x):
        j = np.zeros(self.n + 1)
        j[self.n] = 1.0
        return j
    
    def eq_con(self, x):
        return x[:self.n].sum() - self.n / 2.0
    
    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j
    
    def ineq_con(self, x):
        h = x[:self.n]
        return x[self.n] - compute_profile(h)
    
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        M = 2 * n - 1
        j = np.zeros((M, n + 1))
        j[:, n] = 1.0
        for m in range(M):
            delta = m - (n - 1)
            j[m, :n] = -lag_gradient(h, delta)
        return j


def solve_slsqp(h0, t_upper, maxiter=200):
    n = len(h0)
    prob = NLPProblem(n, t_upper)
    x0 = np.concatenate([h0, [t_upper]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_upper)]
    constraints = [
        {'type': 'eq', 'fun': prob.eq_con, 'jac': prob.eq_jac},
        {'type': 'ineq', 'fun': prob.ineq_con, 'jac': prob.ineq_jac}
    ]
    result = minimize(prob.obj, x0, method='SLSQP',
                      jac=prob.obj_jac,
                      bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    if result.x is not None and np.all(np.isfinite(result.x)):
        return repair_feasible(result.x[:n])
    return None


def solve_slp(h0, max_iters=20):
    """Sequential LP: linearize and solve LP subproblems"""
    n = len(h0)
    h_cur = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(h0)
    
    for it in range(max_iters):
        prof = compute_profile(h_cur)
        cur_score = float(prof.max())
        
        # Build LP
        M = 2 * n - 1
        # c: minimize t (last variable)
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # Inequality: grad_m · h - t <= grad_m · h_cur - F_m(h_cur) for all m
        A_ub = np.zeros((M, n + 1))
        b_ub = np.zeros(M)
        for m in range(M):
            delta = m - (n - 1)
            g = lag_gradient(h_cur, delta)
            A_ub[m, :n] = g
            A_ub[m, n] = -1.0
            b_ub[m] = g @ h_cur - prof[m]
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, cur_score + 0.5)]
        
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs')
            if result.success and result.x is not None:
                h_lp = repair_feasible(result.x[:n])
                s_lp = exact_score(h_lp)
                if s_lp < best_s - 1e-12:
                    best_h = h_lp.copy()
                    best_s = s_lp
                    h_cur = h_lp.copy()
                else:
                    # Try interpolation
                    for alpha in [0.5, 0.25, 0.75]:
                        h_interp = (1 - alpha) * h_cur + alpha * result.x[:n]
                        h_interp = repair_feasible(h_interp)

                            best_s = s_interp

                            break
                    else:
                        break  # No improvement, stop
            else:
                break
        except Exception:
            break
    
    return best_h, best_s


def active_descent(h0, max_iters=50):
    """Local descent using negative gradient of active lags"""
    n = len(h0)
    h_cur = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(h0)
    
    for it in range(max_iters):
        prof = compute_profile(h_cur)
        cur_score = float(prof.max())
        
        # Top active lags
        A = min(20, 2 * n - 1)
        top_idx = np.argsort(prof)[::-1][:A]
        
        # Negative average gradient
        grad_sum = np.zeros(n)
        for idx in top_idx:
            delta = idx - (n - 1)
            grad_sum += lag_gradient(h_cur, delta)
        grad_sum /= A
        grad_sum -= grad_sum.mean()  # zero mean
        
        gnorm = np.linalg.norm(grad_sum)
        if gnorm < 1e-14:
            break
        direction = -grad_sum / gnorm
        
        # Line search
        improved = False
        for alpha in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            h_trial = h_cur + alpha * direction
            h_trial = repair_feasible(h_trial)
            if not np.all(np.isfinite(h_trial)):
                continue
            s = exact_score(h_trial)
            if s < cur_score - 1e-13:
                h_cur = h_trial
                if s < best_s - 1e-13:
                    best_h = h_trial.copy()
                    best_s = s
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_s
