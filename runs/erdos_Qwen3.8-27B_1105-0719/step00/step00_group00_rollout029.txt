from scipy.optimize import minimize, linprog

    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(R)

def project_onto_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}."""
    # Water-filling / bisection
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Fine-tune: distribute remaining mass
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:

        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.any(free):
            x[free] += diff / np.sum(free)
            x = np.clip(x, 0.0, 1.0)

def compute_subgradient(h):
    """Compute subgradient of C5 at h. Returns (grad, max_lag, max_val)."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    # Find all lags achieving (near) the max
    max_val = np.max(R)
    # Use all lags within a small tolerance of max
    tolerance = 0.01 * max_val if max_val > 0 else 0.01
    active = R >= max_val - tolerance
    
    grad = np.zeros(n)
    for s in np.where(active)[0]:
        lag = s - (n - 1)
        for i in range(n):
            g = 0.0
            ni = i - lag
            if 0 <= ni < n:
                g += (1.0 - h[ni])
            nj = i + lag
            if 0 <= nj < n:
                g -= h[nj]
            grad[i] += g
    grad *= dx
    return grad, np.argmax(R), max_val * dx

    """One subgradient step with projection."""
    grad, _, _ = compute_subgradient(h)
    h_new = h - lr * grad
    h_new = project_onto_feasible(h_new, target_sum)
    return h_new

def smooth_c5(h, tau=20.0):
    """Compute softmax-smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    R_scaled = tau * dx * R
    # For numerical stability
    R_shifted = R_scaled - np.max(R_scaled)
    weights = np.exp(R_shifted)
    weights /= np.sum(weights)
    
    val = (1.0/tau) * (np.log(np.sum(weights)) + np.max(R_scaled))
    
    # Gradient: weighted average of dR_s/dh
    grad = np.zeros(n)
    for s in range(len(R)):
        if weights[s] < 1e-10:
            continue
        lag = s - (n - 1)
        for i in range(n):
            g = 0.0
            ni = i - lag
            if 0 <= ni < n:
                g += (1.0 - h[ni])
            nj = i + lag
            if 0 <= nj < n:
                g -= h[nj]
            grad[i] += weights[s] * dx * g
    return val, grad

def lp_direction(h, target_sum, tol=0.01):
    """Find a direction that decreases all near-active lags.
    Returns direction d or None if at local optimum."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = np.max(R)
    active = np.where(R >= max_R - tol * max_R)[0]
    
    # Compute gradients for active lags
    grads = np.zeros((len(active), n))
    for idx, s in enumerate(active):
        lag = s - (n - 1)
        for i in range(n):
            val = 0.0
            ni = i - lag
            if 0 <= ni < n:
                val += (1.0 - h[ni])
            nj = i + lag
            if 0 <= nj < n:
                val -= h[nj]
            grads[idx, i] = val
    
    # LP: find d such that grads @ d <= -1 (decrease all active by at least 1/dx)
    # subject to sum(d) = 0, -eps <= d_i <= eps
    # Actually, let's just check if there exists d with grads @ d < 0 for all active
    # and sum(d) = 0.
    
    # Use scipy linprog
    from scipy.optimize import linprog
    m = len(active)
    # Minimize 0
    c = np.zeros(n)
    # Inequality constraints: grads @ d <= -0.01 (want strict decrease)
    A_ub = grads
    b_ub = np.full(m, -0.01)
    # Equality: sum(d) = 0
    A_eq = np.array([[1.0]*n])
    b_eq = [0.0]
    # Bounds: -0.1 <= d_i <= 0.1 (small step)
    bounds = [(-0.1, 0.1)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        # Normalize
        d = res.x
        d_norm = np.max(np.abs(d))
        if d_norm > 1e-10:
            d = d / d_norm * 0.05
        return d
    return None

    lo, hi = -2.0, 2.0

        if s > target_sum:
