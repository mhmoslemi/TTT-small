    """Perform LSE-approximated gradient descent in symmetric coordinates."""
    best_c = compute_c5(make_h_from_a(a, n))
    best_a = a.copy()
    h = make_h_from_a(a, n)
    
    for beta in [beta_start * (beta_mult ** i) for i in range(int(math.log(beta_max/beta_start, beta_mult)) + 1)]:
        for it in range(max_iter):
            raw = raw_lags(h)
            max_raw = raw.max()
            exp_raw = np.exp(raw - max_raw)

            # Line search: try decreasing step sizes
            best_step_c = None
            best_step_a = None
            for lr in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:

            else:

def lse_descent(a, n, total_time, rng):
    """LSE-based gradient descent in symmetric coordinates."""
    m = n // 2
    best_a = a.copy()
    best_c = compute_c5(make_h_from_a(a, n))
    h = make_h_from_a(a, n)
    
    betas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    start_time = time.time()
    
    for beta in betas:
        stall_count = 0
        while time.time() - start_time < total_time and stall_count < 30:
            raw = raw_lags(h)
            max_raw = raw.max()
            exp_raw = np.exp(beta * (raw - max_raw))
            weights = exp_raw / exp_raw.sum()
            
            grad_h = compute_full_gradient(h, weights)
            grad_a = grad_h[:m] - grad_h[n-1:n-1-m:-1]
            
            # Check if gradient is small
            if np.abs(grad_a).max() < 1e-10:
                stall_count += 10
                continue
            
            # Line search
            improved = False
            for lr in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
                new_a = np.clip(a - lr * grad_a, 0.0, 1.0)
                new_h = make_h_from_a(new_a, n)
                c = compute_c5(new_h)
                if c < best_c - 1e-12:
                    a = new_a
                    h = new_h
                    best_c = c
                    best_a = a.copy()
                    improved = True
                    break
            
            if not improved:
                stall_count += 1
    
    return best_a, best_c, make_h_from_a(best_a, n)

def coordinate_descent(a, n, max_iter, rng):
    """Simple coordinate descent in symmetric coordinates."""
    m = n // 2
    best_c = compute_c5(make_h_from_a(a, n))
    best_a = a.copy()
    
    for it in range(max_iter):
        improved = False
        for i in range(m):
            for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                new_a = a.copy()
                new_a[i] = np.clip(a[i] + delta, 0.0, 1.0)
                c = compute_c5(make_h_from_a(new_a, n))
                if c < best_c - 1e-12:
                    a = new_a
                    best_c = c
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best_a, best_c, make_h_from_a(best_a, n)

def random_perturb_search(a, n, num_trials, step_sizes, rng):
    """Try random perturbations of a."""
    m = n // 2
    best_a = a.copy()
    best_c = compute_c5(make_h_from_a(a, n))
    
    for trial in range(num_trials):
        new_a = a.copy()
        # Perturb 1-3 random coordinates
        num_perturb = rng.integers(1, 4)
        indices = rng.choice(m, size=num_perturb, replace=False)
        for idx in indices:
            step = step_sizes * (1.0 if rng.random() < 0.5 else -1.0)
            new_a[idx] = np.clip(a[idx] + step, 0.0, 1.0)
        c = compute_c5(make_h_from_a(new_a, n))
        if c < best_c - 1e-12:
            best_a = new_a.copy()
            best_c = c
    
    return best_a, best_c, make_h_from_a(best_a, n)

def project_mass_box(h, target):

    n = len(h)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - h_proj.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h_proj < 1.0 - 1e-10)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                h_proj[idx] = np.minimum(h_proj[idx] + per, 1.0)
        else:
            idx = np.where(h_proj > 1e-10)[0]
            if len(idx) > 0:
                per = residual / len(idx)
                h_proj[idx] = np.maximum(h_proj[idx] + per, 0.0)
    return h_proj

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)
    # Fix residual
    res = target - h.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.minimum(h[mask] + res / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.maximum(h[mask] + res / cnt, 0.0)
    return h

def grad_of_lse(h, weights):
    """Compute gradient of sum_k weights[k]*corr[k] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        lo1 = max(0, k)
        hi1 = min(n, k + n)

        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2+k:hi2+k])
    return grad
