```python
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x_proj = np.clip(x - lam, 0.0, 1.0)
    # Exact repair
    diff = target_sum - x_proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            slacks = 1.0 - x_proj
        else:
            slacks = x_proj
        total = slacks.sum()
        if total > 1e-14:
            x_proj += diff * slacks / total
            x_proj = np.clip(x_proj, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    R = corr * dx  # length 2n-1
    R_max = R.max()
    log_sum = R_max + tau * np.log(np.sum(np.exp((R - R_max) / tau)))
    weights = np.exp((R - R_max) / tau)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx]
        if w < 1e-12:
            continue
        k = lag_idx - (n - 1)  # actual lag
        # dR_k/dh[m] = dx * [(1-h[m-k]) if 0<=m-k<n - h[m+k] if 0<=m+k<n]

        m1_end = min(n, k + n)
        if m1_end > m1_start:

        # Term 2: m+k in [0,n-1] → m in [-k, -k+n-1] ∩ [0,n-1]

        if m2_end > m2_start:

def simulated_annealing(h, target_sum, time_limit, seed):

    best_c = c_cur

    n_total = 0
    
    while time.perf_counter() - start_time < time_limit:
        batch_size = 500
        if rng.random() < 0.5:
            # Pair transfer
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)

            if delta_max < 1e-8:
                temp *= 0.99
                continue

            h_orig = (h_cur[i], h_cur[j])
            h_cur[i] -= delta
            h_cur[j] += delta

        if c_new < best_c:

        if c_new <= c_cur:
            h_cur = h_new.copy() if not isinstance(h_new, type(None)) else h_cur  # already modified in place for pair
            # For pair transfer, h_cur is already modified
            # For random step, need to copy
            if isinstance(h_new, type(None)):
                pass  # pair transfer already done in place
            else:
                h_cur = h_new
            c_cur = c_new
            n_accepted += 1
        else:
            if temp > 1e-6 and rng.random() < np.exp(-(c_new - c_cur) / temp):
                if isinstance(h_new, type(None)):
                    pass  # already modified in place
                else:
                    h_cur = h_new
                c_cur = c_new
                n_accepted += 1
            else:
                if isinstance(h_new, type(None)):
                    h_cur[i] = h_orig[0]
                    h_cur[j] = h_orig[1]
                
        # Adapt parameters
        if n_total % 200 == 0:
            rate = n_accepted / 200.0
            if rate < 0.3:
                step_scale *= 0.7
            elif rate > 0.7:
                step_scale *= 1.3
            step_scale = min(step_scale, 0.5)
            n_accepted = 0
            temp *= 0.97
    
    return best_h, best_c
```

```python
    t3 = min(150.0, 0.18 * budget_s)
    if tleft() > t3:
        h3, c3 = sa_optimize(bh, ts, t3, seed + 2)
        if c3 < bc:
            bh, bc = h3, c3

    # Phase 4: Restart with noise + re-optimize
    rng = np.random.default_rng(seed + 3)
    for ns in [0.015, 0.04, 0.08]:
        if tleft() < 20.0:
            break
        h_noisy = project_box_sum(bh + rng.standard_normal(n) * ns, ts)
        tt = min(80.0, 0.3 * tleft())
        if tt < 10.0:
            break
        h4, c4 = smooth_descend(h_noisy, ts, tt, seed + 10 + int(ns * 100))
        if c4 < bc:
            bh, bc = h4, c4
        if tleft() > 15.0 and c4 < bc:
            h4b, c4b = pair_polish(bh, ts, min(40.0, tleft() - 5.0), seed + 50)
            if c4b < bc:
                bh, bc = h4b, c4b

    # Phase 5: Final pair polish on best
    if tleft() > 8.0:
        h5, c5 = pair_polish(bh, ts, tleft() - 3.0, seed + 99)
        if c5 < bc:
            bh, bc = h5, c5

    # Phase 6: One more smooth pass on final best
    if tleft() > 10.0:
        h6, c6 = smooth_descend(bh, ts, tleft() - 3.0, seed + 200)
        if c6 < bc:
            bh, bc = h6, c6

    # Final validation and repair
    bf = project_box_sum(bh, ts)
    cf = eval_c5(bf)
    if cf <= bc + 1e-12:
        bh, bc = bf, cf
    else:
        # Keep original if repair worsened
        pass

    return bh, bc, n
```
