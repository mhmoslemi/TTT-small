import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def pair_search(h, n, max_sweeps=100, tol=1e-10):
    """Do pair transfer optimization."""
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = evaluate(h, n)
    
    for sweep in range(max_sweeps):
        improved = False
        # Precompute correlation
        corr = np.correlate(h, 1-h, mode="full")
        c5 = np.max(corr) * dx
        
        # For each pair (a, b) with a < b
        for a in range(n):
            for b in range(a+1, n):
                # Feasible range for delta
                lo = max(-h[a], h[b] - 1.0)
                hi = min(1.0 - h[a], h[b])
                if lo >= hi - 1e-15:
                    continue
                
                # Compute A_l and B_l for each lag
                # shift_l = l - (n-1)
                shifts = np.arange(2*n-1) - (n-1)
                
                # For each lag, compute the linear and quadratic coefficients
                A = np.zeros(2*n-1)
                B = np.zeros(2*n-1)
                
                for l in range(2*n-1):
                    s = shifts[l]  # shift
                    # Case: a - shift = valid index for h[a] to appear
                    n1 = a - s
                    # Case: b + shift = valid index for (1-h[b]) to appear  
                    n2 = b
                    # Check if it's the cross term
                    if a - b == s:
                        # Cross term: delta*(1-h[b]+h[a]) + delta^2
                        if 0 <= b < n and 0 <= a < n:  # always true
                            A[l] = 1.0 - h[b] + h[a]
                            B[l] = 1.0
                    else:
                        # Term 1: h[a] * (1-h[a-s]) if 0 <= a-s < n
                        if 0 <= a - s < n and (a - s) != b:
                            A[l] += 1.0 - h[a - s]
                        # Term 2: h[b+s] * (1-h[b]) if 0 <= b+s < n
                        if 0 <= b + s < n and (b + s) != a:
                            A[l] += h[b + s]
                
                # Minimize max_l (corr[l] + A[l]*d + B[l]*d^2) * dx
                # Subject to lo <= d <= hi
                
                def obj(d):
                    new_corr = corr + A * d + B * d * d
                    return np.max(new_corr)
                
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded', 
                                     options={'xatol': 1e-10})
                if res.fun < c5 - 1e-12:
                    h[a] += res.x
                    h[b] -= res.x
                    c5 = res.fun * dx
                    corr = np.correlate(h, 1-h, mode="full")
                    improved = True
                    best_c5 = min(best_c5, c5)
        
        if not improved:
            break
    
    return h, best_c5

from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def pair_transfer_optimize(h, n, time_limit, max_sweeps=50):
    """Optimize using pair transfers."""
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    dx = 2.0 / n
    shifts = np.arange(2*n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        c5_base = np.max(corr)
        
        for a in range(n):
            for b in range(n):
                if a == b:
                    continue
                lo = max(-best_h[a], best_h[b] - 1.0)
                hi = min(1.0 - best_h[a], best_h[b])
                if lo >= hi - 1e-12:
                    continue
                
                # Compute A and B
                j_a = a - shifts
                i_b = b + shifts
                A = np.zeros(2*n - 1)

                valid_ib = (i_b >= 0) & (i_b < n) & (i_b != a)
                
                A[valid_ja] = 1.0 - best_h[j_a[valid_ja]]
                A[valid_ib] += best_h[i_b[valid_ib]]

                if np.any(cross_mask):
                    A[cross_mask] = 1.0 - best_h[b] + best_h[a]
                B = cross_mask.astype(float)
                
                # Minimize
                def obj(d):
                    return np.max(corr + A * d + B * d * d)

def interpolate_to_n(h_old, n_old, n_new):

    target = n_new / 2.0
    h_new = h_new + (target - h_new.sum()) / n_new

        else:

            h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0, 1)
    return h_new

  These are the same product if a = b+s and a-s = b. The first gives s = a-b, the second gives s = a-b. So yes, they're the same when s = a-b. In that case, i = a = b+s and j = a-s = b. So the product is h[a]*(1-h[b]). ✓

- (a, a+s) = (b-s, b) iff a=b-s and a+s=b iff s=b-a and s=b-a. So same when s=b-a. Product is h[b]*(1-h[a]).

- (a-s, a) = (a, a+s) iff a-s=a and a=a+s → s=0. In that case, it's the product h[a]*(1-h[a]) (the self-correlation term).

- (b-s, b) = (b, b+s) iff s=0. Product h[b]*(1-h[b]).

- (a-s, a) = (b, b+s) as we showed: only when s=a-b, and then it's the cross product.

- (a, a+s) = (b-s, b): only when s=b-a, cross product.

- (a-s, a) = (b-s, b): a=b, impossible.
- (a, a+s) = (b, b+s): a=b, impossible.

So the distinct products involving a or b at shift s are:
1. h[a]*(1-h[a-s]) if 0≤a-s<n and (a-s)≠... well, this is always a valid distinct product.
   (It equals h[b]*(1-h[b-s]) only if a=b, impossible.)
   (It equals h[b+s]*(1-h[b]) only if s=a-b, in which case it IS the cross product.)
   (It equals h[a+s]*(1-h[a]) only if s=0, in which case it's the self product h[a]*(1-h[a]).)

2. h[a+s]*(1-h[a]) if 0≤a+s<n.
   (It equals h[b]*(1-h[b+s]) only if s=b-a, the other cross product.)
   (It equals h[a]*(1-h[a-s]) only if s=0.)

3. h[b]*(1-h[b-s]) if 0≤b-s<n.
   (Equals h[b+s]*(1-h[b]) only if s=0.)
   (Equals h[a]*(1-h[a-s]) only if a=b, impossible.)

4. h[b+s]*(1-h[b]) if 0≤b+s<n.
   (Equals h[b]*(1-h[b-s]) only if s=0.)
   (Equals h[a]*(1-h[a-s]) only if s=a-b.)

- s=0: Products are h[a](1-h[a]), h[b](1-h[b]) (the self-terms).

  Total change: d(1-2h[a]+2h[b]-1) - 2d² = 2d(h[b]-h[a]) - 2d²

    So change = -d*h[a+s] = -d*h[2a-b].
