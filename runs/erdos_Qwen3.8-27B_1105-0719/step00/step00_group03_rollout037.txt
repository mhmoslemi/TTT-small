import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def coordinate_step(h, i, j):
    """Try to improve by moving mass from i to j (or j to i)."""
    n = len(h)
    dx = 2.0 / n
    best_c = evaluate(h)
    
    # Direction 1: decrease h[i] by t, increase h[j] by t
    # t in [0, min(h[i], 1-h[j])]

    if t_max > 1e-12:
        def obj(t):
            h_new = h.copy()
            h_new[i] -= t
            h_new[j] += t
            c = np.correlate(h_new, 1 - h_new, mode='full') * dx
            return np.max(c)
        res = minimize_scalar(obj, bounds=(0, t_max), method='bounded', options={'xatol': 1e-10})
        if res.fun < best_c - 1e-12:
            h[i] -= res.x
            h[j] += res.x
            best_c = res.fun
    
    # Direction 2: increase h[i] by t, decrease h[j] by t
    t_max = min(1 - h[i], h[j])

    """Minimize convex f on [a, b] using golden section search."""

            c = d
