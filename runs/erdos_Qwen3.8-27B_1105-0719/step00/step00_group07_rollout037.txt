import time
from scipy.optimize import minimize

def compute_correlation(h):
    """Compute np.correlate(h, 1-h, mode='full')"""
    return np.correlate(h, 1.0 - h, mode="full")

def lse_surr(R, beta):
    """Log-sum-exp surrogate for max(R)"""
    m = np.max(R)
    return m + np.log(np.sum(np.exp(beta * (R - m)))) / beta

    """Gradient of log-sum-exp surrogate w.r.t. h"""
    n = len(h)
    m = np.max(R)
    w = np.exp(beta * (R - m))
    w /= np.sum(w)
    # w has length 2n-1
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        lag = t - (n - 1)
        if w[t] < 1e-15:

        if lag >= 0:

            # and i ranges from lag to n-1 (i-lag valid)
            # Combine: i from 0 to n-1
            # Term 1: (1-h[i+lag]) for i in [0, n-1-lag]

            idx1 = np.arange(0, n - lag)
            grad[idx1] += w[t] * (1.0 - h[idx1 + lag])

            grad[idx2] -= w[t] * h[idx2 - lag]

            grad[idx2] -= w[t] * h[idx2 - lag]  # idx2-lag = idx2+lag2

    s = np.sum(x)

def compute_R(h):

def obj_value(h):
    R = compute_R(h)
    return np.max(R) * 2.0 / len(h)

def project(h, target):
    """Project onto {0<=x<=1, sum=x_target}"""
    n = len(h)
    lo = h.min() - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # fix residual
    r = target - x.sum()
    if r > 0:
        for i in range(n):
            if r <= 1e-15: break
            d = min(1 - x[i], r)
            x[i] += d
            r -= d
    elif r < 0:
        for i in range(n):
            if r >= -1e-15: break
            d = min(x[i], -r)
            x[i] -= d
            r += d
    return x

    R = compute_R(h)
    m = R.max()

    sw = w.sum()

import numpy as np
import time

def compute_R(h):
    """Compute correlation vector."""
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h):
    """Compute C5 = max(R) * dx."""
    return np.max(compute_R(h)) * 2.0 / len(h)

def project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(h)
    # Bisection on shift parameter
    lo = h.min() - 1.5
    hi = h.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    r = target - x.sum()
    if r > 1e-14:
        for i in range(n):
            if r <= 1e-14:
                break
            d = min(1.0 - x[i], r)
            x[i] += d
            r -= d
    elif r < -1e-14:
        for i in range(n):
            if r >= -1e-14:
                break
            d = min(x[i], -r)
            x[i] -= d
            r += d
    return x

def grad_lse(h, beta):
    """Gradient of log-sum-exp(h) surrogate for max(R)."""
    n = len(h)
    R = compute_R(h)
    m = R.max()
    exponents = beta * (R - m)
    w = np.exp(exponents)
    sw = w.sum()
    if sw == 0:
        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        lag = t - (n - 1)
        if lag >= 0:
            # (1-h[i+lag]) for i=0..n-1-lag
            grad[:n - lag] += wt * (1.0 - h[lag:n])
            # -h[i-lag] for i=lag..n-1
            grad[lag:] -= wt * h[:n - lag]
        else:
            l2 = -lag
            # (1-h[i+lag]) = (1-h[i-l2]) for i=l2..n-1
            grad[l2:] += wt * (1.0 - h[:n - l2])
            # -h[i-lag] = -h[i+l2] for i=0..n-1-l2
            grad[:n - l2] -= wt * h[l2:]
    return grad

def optimize_continuous(h0, max_time=100.0):
    """Optimize continuous h using projected gradient descent with LSE."""
    n = len(h0)
    target = n / 2.0
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_val = c5_value(h)
    
    beta = 50.0
    lr = 0.01
    start = time.time()
    
    while time.time() - start < max_time:
        # Compute gradient
        g = grad_lse(h, beta)
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            # Increase beta and reset
            beta *= 2
            continue
        
        # Gradient step with backtracking
        for _ in range(20):
            h_new = project_box_sum(h - lr * g, target)
            v_new = c5_value(h_new)
            if v_new < best_val + 1e-10:
                h = h_new
                break
            lr *= 0.5
        
        if c5_value(h) < best_val:
            best_val = c5_value(h)
            best_h = h.copy()
        
        # Adapt beta: increase as we converge
        if best_val < 0.381:
            beta = min(beta * 1.05, 500.0)
        elif beta < 100:
            beta = min(beta * 1.1, 100.0)
        
        # Reset learning rate periodically
        if int(time.time() - start) % 5 == 0:
            lr = max(lr, 0.005)
    
    return best_h, best_val

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0/1)."""
    t = np.zeros(n, dtype=np.int64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def binary_c5(h):
    """Compute C5 for binary h (values are 0 or 1)."""
    R = compute_R(h.astype(float))
    return R.max() * 2.0 / len(h)

def anneal_binary(n, max_iters=5000000, T0=1.0, T1=0.01, seed=42):
    """Simulated annealing for binary sequence optimization."""
    rng = np.random.default_rng(seed)
    m = n // 2
    
    # Start from Thue-Morse (balanced, low autocorrelation)
    if n & (n - 1) == 0:  # power of 2
        h = thue_morse(n).copy()
    else:
        # Use Thue-Morse prefix if possible, otherwise random
        h = thue_morse(max(n, 64))[:n].copy()
        # Ensure exactly m ones
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > m:
            # Remove extra ones
            extra = ones[m:]
            for i in extra:
                h[i] = 0
        elif len(ones) < m:
            # Add needed ones
            needed = m - len(ones)
            to_add = zeros[:needed]
            h[to_add] = 1
    
    # Verify
    assert h.sum() == m, f"Sum is {h.sum()}, expected {m}"
    
    R = compute_R(h.astype(float))
    curr_max = R.max()
    best_h = h.copy()
    best_max = curr_max
    
    T = T0
    
    for it in range(max_iters):
        # Pick a random 1 position and a random 0 position, swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.integers(0, len(ones))]
        b = zeros[rng.integers(0, len(zeros))]
        
        # Compute delta using incremental update
        # h[a]: 1->0, h[b]: 0->1
        # delta_R[t] for each lag
        old_a, old_b = 1, 0
        new_a, new_b = 0, 1
        
        # We need to compute the new max efficiently
        # Full recompute is O(n) with correlate, incremental is also O(n)
        # Let's just recompute (it's fast for n~100-128)
        h[a] = new_a
        h[b] = new_b
        R_new = compute_R(h.astype(float))
        new_max = R_new.max()
        
        delta = new_max - curr_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[a] = old_a
            h[b] = old_b
        
        # Cool down
        if it % 100 == 0:
            T = max(T1, T * 0.999)
    
    return best_h.astype(float), best_max * 2.0 / n

def project_box_sum(h, target):
    n = len(h)
    lo = h.min() - 1.5
    hi = h.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5

        if s > target:
            lo = mid
        else:

            r -= d

        else:
            h[a] = 1.0
            h[b] = 0.0
        it += 1
        if it % 200 == 0:
            T = max(0.005, T * 0.998)
    return best_h, best_max * 2.0 / n


def anneal_binary_ssq(n, max_iters, seed, time_limit):
    rng = np.random.default_rng(seed)
    h = make_binary_initial(n, seed)
    R = compute_R(h)
    curr_e = float(R @ R)
    best_max = float(R.max())
    best_h = h.copy()
    T = 5.0
    t0 = time.time()
    it = 0
    while it < max_iters and time.time() - t0 < time_limit:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        a = ones_idx[rng.integers(0, len(ones_idx))]
        b = zeros_idx[rng.integers(0, len(zeros_idx))]
        if a == b:
            it += 1
            continue
        h[a] = 0.0
        h[b] = 1.0
        R_new = compute_R(h)
        new_e = float(R_new @ R_new)
        new_max = float(R_new.max())
        delta = new_e - curr_e
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_e = new_e
            R = R_new
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        it += 1
        if it % 500 == 0:
            T = max(0.5, T * 0.995)
    return best_h, best_max * 2.0 / n


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 1: Continuous optimization of initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        max_t1 = min(180.0, budget_s * 0.22)
        h_opt, c5_opt = optimize_continuous(h0, max_t1)
        best_c5 = c5_opt
        best_h = h_opt
        best_n = n0
    except (NameError, AttributeError, Exception):
        pass

    # Phase 2: Binary annealing
    n_list = [100, 102, 104, 106, 108, 110, 112, 116, 120, 128, 144, 160]
    for idx, n in enumerate(n_list):
        if time.time() - start_time > budget_s * 0.90:
            break
        remaining = budget_s * 0.90 - (time.time() - start_time)
        n_left = len(n_list) - idx
        time_per_n = min(35.0, remaining / max(1, n_left))
        for method_idx in range(2):
            if time.time() - start_time > budget_s * 0.90:
                break
            t_lim = time_per_n * 0.45
            for attempt in range(2):
                if time.time() - start_time > budget_s * 0.90:
                    break
                s = seed + n * 7919 + attempt * 104729 + method_idx * 51031
                if method_idx == 0:
                    h_bin, c5_bin = anneal_binary(n, 10000000, s, t_lim)
                else:
                    h_bin, c5_bin = anneal_binary_ssq(n, 10000000, s, t_lim)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin
                    best_n = n

    # Phase 3: Polish continuous best if time remains
    if best_h is not None and (time.time() - start_time) < budget_s * 0.93:
        remaining = budget_s * 0.96 - (time.time() - start_time)
        if remaining > 15:
            is_bin = np.allclose(best_h, np.round(best_h), atol=1e-10)
            if not is_bin:
                h_pol, c5_pol = optimize_continuous(best_h, remaining)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    if best_h is None:
        n = 100
        h = np.zeros(n)
        h[::2] = 1.0
        best_h = h
        best_c5 = get_c5(h)
        best_n = n

    return (best_h, best_c5, best_n)
