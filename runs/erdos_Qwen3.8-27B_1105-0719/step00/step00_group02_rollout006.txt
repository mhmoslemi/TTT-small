```python
import numpy as np
import time

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_k(vec, mass):
    """Project vector onto {x: 0<=x<=1, sum(x)=mass}"""
    x = np.clip(vec.copy(), 0.0, 1.0)
    # Binary search on lambda for clip(y - lambda, 0, 1) to sum to mass
    lo, hi = -1.0, 1.0  # lambda range
    target = mass
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(vec - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    residual = mass - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            slack = np.maximum(0, 1.0 - x)
            if slack.sum() > 0:
                x += residual * slack / slack.sum()
        else:
            afford = np.maximum(0, x)
            if afford.sum() > 0:
                x += residual * afford / afford.sum()
    return np.clip(x, 0.0, 1.0)

def repair_feasibility(x, mass):
    """Small repair for floating point issues"""
    x = x.copy()
    x = np.clip(x, 0.0, 1.0)
    residual = mass - x.sum()
    tol = 1e-10
    if abs(residual) > tol:
        if residual > 0:
            slack = np.maximum(1e-15, 1.0 - x)
            x += residual * slack / slack.sum()
        else:
            afford = np.maximum(1e-15, x)
            x += residual * afford / afford.sum()
    x = np.clip(x, 0.0, 1.0)
    return x

def solve_ray(h, u, n):
    """Find best t along h + t*u minimizing max of lag correlations"""
    g = 1.0 - h
    L = 2 * n - 1
    
    # Compute q0, q1, q2
    q0 = np.correlate(h, g, mode="full")
    q1 = np.correlate(u, g, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")
    
    # Find feasible interval
    eps = 1e-14
    t_low = 0.0
    t_high = 0.0
    for i in range(n):
        if u[i] > eps:
            t_low = max(t_low, -h[i] / u[i])
            t_high = min(t_high, (1.0 - h[i]) / u[i])
        elif u[i] < -eps:
            t_low = max(t_low, (1.0 - h[i]) / u[i])
            t_high = min(t_high, -h[i] / u[i])
    
    # Ensure 0 is in interval
    t_low = min(t_low, 0.0)
    t_high = max(t_high, 0.0)
    
    if t_high - t_low < 1e-12:
        return (0.0, q0.max())
    
    # Generate candidate t values
    candidates = [t_low, t_high, 0.0]
    
    # Stationary points of individual quadratics
    for a in range(L):
        if abs(q2[a]) > 1e-14:
            t_stat = -q1[a] / (2.0 * q2[a])
            if t_low <= t_stat <= t_high:
                candidates.append(t_stat)
    
    # Pairwise intersection points
    for a in range(L):
        for b in range(a + 1, L):
            da = q2[a] - q2[b]
            db = q1[a] - q1[b]
            dc = q0[a] - q0[b]
            if abs(da) > 1e-14:
                disc = db * db - 4.0 * da * dc
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2.0 * da)
                    t2 = (-db - sqrt_disc) / (2.0 * da)
                    if t_low - 1e-10 <= t1 <= t_high + 1e-10:
                        candidates.append(np.clip(t1, t_low, t_high))
                    if t_low - 1e-10 <= t2 <= t_high + 1e-10:
                        candidates.append(np.clip(t2, t_low, t_high))
            elif abs(db) > 1e-14:
                t_lin = -dc / db
                if t_low - 1e-10 <= t_lin <= t_high + 1e-10:
                    candidates.append(np.clip(t_lin, t_low, t_high))
    
    candidates = np.array(candidates)
    candidates = np.clip(candidates, t_low, t_high)
    
    # Evaluate max of quadratics at all candidates
    # F(t) = max_a (q0[a] + q1[a]*t + q2[a]*t^2)
    best_t = 0.0
    best_val = q0.max()  # value at t=0
    
    # Process in chunks to avoid memory issues
    t = candidates.reshape(-1, 1)  # shape (M, 1)
    q0v = q0.reshape(1, -1)  # shape (1, L)
    q1v = q1.reshape(1, -1)
    q2v = q2.reshape(1, -1)
    
    vals = q0v + t * q1v + t * t * q2v  # shape (M, L)
    max_vals = vals.max(axis=1)
    
    idx = np.argmin(max_vals)
    best_t = candidates[idx]
    best_val = max_vals[idx]
    
    return (best_t, best_val)
```

However, there's a subtlety: the maximum of a collection of functions (convex or not) has its minimum at either:

2. A point where one of the functions has a local minimum (q2[a] > 0 and its derivative is zero)
3. A point where two or more functions are equal and the max "changes" (nonsmooth point)

For case 2, we only need stationary points of quadratics with q2[a] > 0 (minima of convex quadratics).

import numpy as np

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_k(vec, mass):
    # Project onto {x: 0<=x_i<=1, sum(x)=mass}
    x = vec.astype(np.float64).copy()
    # Binary search on shift
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    target = mass
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(x - lam, 0.0, 1.0)
    # Fine adjustment
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            slack = 1.0 - x
            slack = np.where(slack > 1e-12, slack, 1.0)
            x += (resid / slack.sum()) * slack
        else:
            aff = np.where(x > 1e-12, x, 1.0)
            x += (resid / aff.sum()) * aff
    return np.clip(x, 0.0, 1.0)

def repair(x, mass):
    x = np.clip(x, 0.0, 1.0)
    resid = mass - x.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            slack = np.maximum(1e-15, 1.0 - x)
            x += resid * slack / slack.sum()
        else:
            aff = np.maximum(1e-15, x)
            x += resid * aff / aff.sum()
    return np.clip(x, 0.0, 1.0)

def solve_ray(h, u, n):
    """Return (best_t, best_score) along h + t*u"""
    g = 1.0 - h
    L = 2 * n - 1
    q0 = np.correlate(h, g, mode="full")
    q1 = np.correlate(u, g, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")
    
    # Feasible interval
    eps = 1e-14
    t_lo = 0.0
    t_hi = 0.0
    for i in range(n):
        if u[i] > eps:
            t_lo = max(t_lo, -h[i] / u[i])
            t_hi = min(t_hi, (1.0 - h[i]) / u[i])
        elif u[i] < -eps:
            t_lo = max(t_lo, (1.0 - h[i]) / u[i])
            t_hi = min(t_hi, -h[i] / u[i])
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(q0.max())
    
    # Candidate t values
    cands = [0.0, t_lo, t_hi]
    
    # Stationary points (where q2[a] != 0)
    for a in range(L):
        if abs(q2[a]) > 1e-14:
            ts = -q1[a] / (2.0 * q2[a])
            if t_lo <= ts <= t_hi:
                cands.append(ts)
    
    # Pairwise intersections
    for a in range(L):
        for b in range(a+1, L):
            A = q2[a] - q2[b]
            B = q1[a] - q1[b]
            C = q0[a] - q0[b]
            if abs(A) > 1e-14:
                disc = B*B - 4.0*A*C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    t1 = (-B + sd) / (2.0*A)
                    t2 = (-B - sd) / (2.0*A)
                    for t in (t1, t2):
                        if t_lo - 1e-10 <= t <= t_hi + 1e-10:
                            cands.append(np.clip(t, t_lo, t_hi))
            elif abs(B) > 1e-14:
                tl = -C / B
                if t_lo - 1e-10 <= tl <= t_hi + 1e-10:
                    cands.append(np.clip(tl, t_lo, t_hi))
    
    cands = np.array(cands)
    cands = np.clip(cands, t_lo, t_hi)
    
    # Evaluate
    t = cands[:, np.newaxis]  # (M, 1)
    vals = q0[np.newaxis, :] + t * q1[np.newaxis, :] + t*t * q2[np.newaxis, :]
    max_vals = vals.max(axis=1)
    idx = int(np.argmin(max_vals))
    return float(cands[idx]), float(max_vals[idx])
