                    smooth_obj_grad,
                    h_pert,
                    args=(200,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 150, 'ftol': 1e-13}
                )
                h_new = project_feasible(res.x, n_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
