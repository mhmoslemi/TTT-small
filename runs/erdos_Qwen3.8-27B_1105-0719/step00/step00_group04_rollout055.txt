class WeightedActiveProblem:
  def __init__(self,n,deltas,weights,mu):
     self.n=n; self.deltas=np.asarray(deltas,int); self.weights=np.asarray(weights,float); self.mu=mu
  def fun(self,x):
     h=x
     prof=profile_values(h,self.deltas)
     val=float(np.dot(self.weights,prof))
     s=float(h.sum())-0.5*self.n

def repair_to_feasible(v):
    v = np.asarray(v, dtype=float).copy()
    if v.ndim != 1:
        v = v.ravel()
    if v.size == 0:
        v = np.array([0.5, 0.5], dtype=float)

        if abs(rem) <= 1e-13:
            break

def row_value(h, d):
    n = int(h.size)
    dx = 2.0 / n
    d = int(d)
    if d >= 0:
        cnt = n - d
        if cnt <= 0:
            return 0.0
        return float(dx * np.dot(h[:cnt], 1.0 - h[d:d + cnt]))
    else:
        m = -d
        cnt = n - m
        if cnt <= 0:
            return 0.0
        return float(dx * np.dot(h[m:m + cnt], 1.0 - h[:cnt]))


def profile_values(h, deltas):

    out = np.empty(deltas.size, dtype=float)
    for k, d in enumerate(deltas):
        out[k] = row_value(h, int(d))
    return out


def row_value_gradient(h, d):

    g = np.zeros(n, dtype=float)
    d = int(d)

def transfer_linear(src, m):

        64, 72, 80, 90, 100, 110, 120, 130
    ]
    cand = []
    for d0 in wanted:
        d = int(d0)
        if d < 30 or d > 150:
            continue
        if d % 2:
            d += 1
        if d < 30:
            continue
        if d not in cand:
            cand.append(d)

    if base not in cand:
        cand.append(base)

    pairs = []
    for i, d in enumerate(cand):
        pairs.append((abs(d - base), i, d))
    pairs.sort()
    cand = [p[2] for p in pairs]

    if base in cand:
        cand.remove(base)
    cand = [base] + cand
    return cand


def lp_direction(h, grads, alpha_max=0.1, scale=1.0):
    h = np.asarray(h, dtype=float)
    grads = np.asarray(grads, dtype=float)
    n = int(h.size)
    A = int(grads.shape[0])
    if A == 0 or n == 0 or alpha_max <= 0.0 or scale <= 0.0:
        return None
    if not np.all(np.isfinite(grads)):
        return None

    nv = 2 * n + 1
    nr = A + 2 * n + 1
    A_ub = np.zeros((nr, nv), dtype=float)
    b_ub = np.zeros(nr, dtype=float)

    A_ub[:A, :n] = grads
    A_ub[:A, n:2 * n] = -grads
    A_ub[:A, 2 * n] = 1.0
    b_ub[:A] = 0.0

    row = A
    idx = np.arange(n)

    A_ub[row:row + n, idx] = 1.0
    A_ub[row:row + n, n + idx] = -1.0
    b_ub[row:row + n] = (1.0 - h) / alpha_max
    row += n

    A_ub[row:row + n, idx] = -1.0
    A_ub[row:row + n, n + idx] = 1.0
    b_ub[row:row + n] = h / alpha_max
    row += n

    A_ub[row, :2 * n] = 1.0
    b_ub[row] = scale

    c = np.zeros(nv, dtype=float)
    c[-1] = -1.0

    A_eq = np.zeros((1, nv), dtype=float)
    A_eq[0, :n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    b_eq = np.zeros(1, dtype=float)

    bounds = [(0.0, None)] * (2 * n) + [(0.0, 10.0)]
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None
    gamma = float(res.x[-1])
    if not np.isfinite(gamma) or gamma <= 1e-9:
        return None

    d = res.x[:n] - res.x[n:2 * n]
    if not np.all(np.isfinite(d)):
        return None
    return d


def try_direction(h, h_score, d, holder, alphas):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float) if d is not None else None
    if d is None or d.size != h.size or not np.all(np.isfinite(d)):
        return h, float(h_score), False

    nrm = float(np.sum(np.abs(d)))
    if nrm <= 1e-13:
        return h, float(h_score), False

    d = d - float(d.mean())
    best_h = h
    best_s = float(h_score)
    improved = False

    for a in alphas:
        a = float(a)
        if a <= 0.0:
            continue
        cand = h + a * d
        cand = repair_to_feasible(cand)
        sc = exact_c5(cand)
        if sc < best_s - 1e-12:
            best_s = float(sc)
            best_h = cand
            improved = True

    if improved:
        holder.maybe_improve(best_h)
    return best_h, best_s, improved


def local_lp_search(h0, holder, t0, budget, max_iters=300, max_time_s=120.0, rng=None):
    h = repair_to_feasible(h0)
    local_score = exact_c5(h)
    holder.maybe_improve(h)
    t_local = time.perf_counter()

    for _ in range(int(max_iters)):
        if time_left_sec(t0, budget) < 3.0:
            break
        if (time.perf_counter() - t_local) >= float(max_time_s):
            break

        prof, G = profile_and_grad(h)
        local_score = exact_c5(h)
        order = np.argsort(prof)[::-1]
        M = int(prof.size)

        counts = (1, 2, 3, 5, 8, 12, 20)
        improved = False

        for cnt in counts:
            if cnt > M:
                break
            rows = order[:cnt]

            nz_rows = []
            for r in rows:
                if float(np.sum(np.abs(G[r]))) > 1e-10:
                    nz_rows.append(int(r))

            if nz_rows:
                d = lp_direction(h, G[nz_rows], alpha_max=0.10, scale=1.0)
                alphas = (0.10, 0.06, 0.035, 0.018, 0.009, 0.0045)
                h, local_score, imp = try_direction(h, local_score, d, holder, alphas)
                if imp:
                    improved = True
                    break

            mean_rows = nz_rows if len(nz_rows) > 0 else rows
            if len(mean_rows) > 0:
                d = -G[mean_rows].mean(axis=0)
                alphas = (0.05, 0.03, 0.016, 0.008, 0.004, 0.002)
                h, local_score, imp = try_direction(h, local_score, d, holder, alphas)
                if imp:
                    improved = True
                    break

        if not improved and rng is not None:
            z = rng.standard_normal(h.size)
            z = z - float(z.mean())
            nz = float(np.sum(np.abs(z)))
            if nz > 1e-12:
                z = z / nz
                alphas = (0.04, 0.02, 0.01, 0.005, 0.0025)
                h, local_score, imp = try_direction(h, local_score, z, holder, alphas)
                if imp:
                    improved = True

        if not improved:
            break

    holder.maybe_improve(h)
    return h, local_score


class EpigraphProblem:
    def __init__(self, n, beta=0.0, deltas=None):
        self.n = int(n)
        self.beta = float(beta)
        if deltas is None:
            self.deltas = np.arange(-(self.n - 1), self.n, dtype=int)
        else:
            self.deltas = np.asarray(deltas, dtype=int).ravel()

    def fun(self, x):
        h = x[:-1]
        t = x[-1]
        prof = profile_values(h, self.deltas)
        val = float(t)
        if self.beta != 0.0:
            val += self.beta * float(np.dot(h, 1.0 - h))
        return val

    def jac(self, x):
        h = x[:-1]
        _, G = profile_subset_grad(h, self.deltas)
        g = np.zeros(x.size, dtype=float)
        if self.beta != 0.0:
            g[:-1] += self.beta * (1.0 - 2.0 * h)
        g[-1] = 1.0
        return g

    def con_fun(self, x):
        h = x[:-1]
        t = x[-1]
        prof = profile_values(h, self.deltas)
        return t - prof

    def con_jac(self, x):
        h = x[:-1]
        _, G = profile_subset_grad(h, self.deltas)
        J = np.empty((G.shape[0], x.size), dtype=float)
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    def eq_fun(self, x):
        return float(x[:-1].sum() - 0.5 * self.n)

    def eq_jac(self, x):
        g = np.zeros(x.size, dtype=float)
        g[:-1] = 1.0
        return g


def slsqp_maxiter(n, remaining):
    n = int(n)
    remaining = float(remaining)
    if remaining < 20.0:
        return 0
    if n <= 70:
        cap = 80
    elif n <= 95:
        cap = 55
    else:
        cap = 35
    if remaining < 60.0:
        return max(10, min(25, int(remaining * 0.4)))
    if remaining < 180.0:
        return max(20, min(50, int(remaining * 0.25)))
    return max(30, min(cap, int(remaining * 0.15)))


def run_slsqp(h0, holder, t0, budget, maxiter=50, active_k=0, betas=(0.0,)):
    h_best = repair_to_feasible(h0)
    if h_best is None or h_best.size == 0:
        return h_best
    best_score = exact_c5(h_best)
    if not np.isfinite(best_score):
        return h_best

    n = int(h_best.size)
    if maxiter <= 0:
        return h_best

    if active_k <= 0:
        if n <= 90:
            k = None
        else:
            k = 22
    else:
        k = int(active_k)

    for beta in betas:
        if time_left_sec(t0, budget) < 8.0:
            break

        if k is None:
            deltas = None
        else:
            all_d = np.arange(-(n - 1), n, dtype=int)
            prof = profile_values(h_best, all_d)
            kk = min(int(prof.size), max(5, int(k)))
            order = np.argsort(prof)[::-1]
            deltas = all_d[order[:kk]]

        h_start = h_best.copy()
        s0 = exact_c5(h_start)
        if not np.isfinite(s0):
            continue

        try:
            prob = EpigraphProblem(n, float(beta), deltas)
            x0 = np.empty(n + 1, dtype=float)
            x0[:n] = h_start
            x0[n] = s0 + 1e-6

            bounds = [(0.0, 1.0)] * n + [(0.0, s0 + 0.25)]
            constraints = [
                {"type": "eq", "fun": prob.eq_fun, "jac": prob.eq_jac},
                {"type": "ineq", "fun": prob.con_fun, "jac": prob.con_jac}
            ]

            res = optimize.minimize(
                prob.fun,
                x0,
                jac=prob.jac,
                bounds=bounds,
                constraints=constraints,
                method="SLSQP",
                options={"maxiter": int(maxiter), "ftol": 1e-9, "disp": False}
            )

            if res is not None and hasattr(res, "x") and res.x is not None:
                h_sol = repair_to_feasible(res.x[:n])
                if h_sol is not None and h_sol.size == n:
                    alphas = (1.0, 0.9, 0.8, 0.7, 0.55, 0.4, 0.25, 0.12, 0.05)
                    for a in alphas:
                        if time_left_sec(t0, budget) < 1.0:
                            break
                        cand = (1.0 - float(a)) * h_start + float(a) * h_sol
                        holder.maybe_improve(cand)

        except Exception:
            pass

        if (
            holder.h is not None
            and holder.h.size == n
            and holder.score < best_score - 1e-10
        ):
            h_best = holder.h.copy()
            best_score = float(holder.score)

    return h_best


class WeightedActiveProblem:
    def __init__(self, n, deltas, weights, mu):
        self.n = int(n)
        self.deltas = np.asarray(deltas, dtype=int).ravel()
        self.weights = np.asarray(weights, dtype=float).ravel()
        self.mu = float(mu)

    def fun(self, x):
        h = x
        prof = profile_values(h, self.deltas)
        val = float(np.dot(self.weights, prof))
        s = float(h.sum()) - 0.5 * self.n
        val += 0.5 * self.mu * s * s
        return val

    def jac(self, x):
        h = x
        _, G = profile_subset_grad(h, self.deltas)
        grad = np.dot(self.weights, G)
        s = float(h.sum()) - 0.5 * self.n
        grad = grad + self.mu * s * np.ones(self.n, dtype=float)
        return grad


def run_weighted_smooth(h0, holder, t0, budget, max_cycles=4, max_time_s=120.0):
    h = repair_to_feasible(h0)
    local_score = exact_c5(h)
    holder.maybe_improve(h)
    t_local = time.perf_counter()
    n = int(h.size)

    all_d = np.arange(-(n - 1), n, dtype=int)

    for _ in range(int(max_cycles)):
        if time_left_sec(t0, budget) < 6.0:
            break
        if (time.perf_counter() - t_local) >= float(max_time_s):
            break

        prof = profile_values(h, all_d)
        sc = float(np.max(prof))
        order = np.argsort(prof)[::-1]

        for k in (3, 5, 8, 12):
            if int(k) > int(prof.size):
                continue
            if time_left_sec(t0, budget) < 6.0:
                break
            if (time.perf_counter() - t_local) >= float(max_time_s):
                break

            rows = order[:k]
            gaps = sc - prof[rows]
            w = 1.0 / (gaps + 1e-6)
            wsum = float(np.sum(w))
            if wsum <= 0.0:
                w = np.ones(k, dtype=float) / float(k)
            else:
                w = w / wsum

            deltas = all_d[rows]

            for mu in (10.0, 100.0):
                if time_left_sec(t0, budget) < 5.0:
                    break
                if (time.perf_counter() - t_local) >= float(max_time_s):
                    break
                try:
                    prob = WeightedActiveProblem(n, deltas, w, mu)
                    res = optimize.minimize(
                        prob.fun,
                        h,
                        jac=prob.jac,
                        bounds=[(0.0, 1.0)] * n,
                        method="L-BFGS-B",
                        options={"maxiter": 70, "ftol": 1e-12, "disp": 0}
                    )
                    if (
                        res is not None
                        and hasattr(res, "x")
                        and res.x is not None
                        and np.all(np.isfinite(res.x))
                    ):
                        h_sol = repair_to_feasible(res.x)
                        if h_sol is not None and h_sol.size == n:
                            alphas = (1.0, 0.8, 0.6, 0.4, 0.2)
                            for a in alphas:
                                holder.maybe_improve((1.0 - float(a)) * h + float(a) * h_sol)
                            if holder.h is not None and holder.h.size == n and holder.score < local_score - 1e-11:
                                h = holder.h.copy()
                                local_score = float(holder.score)
                except Exception:
                    pass

    holder.maybe_improve(h)
    return h, local_score


def binary_local_search(h0, holder, t0, budget, max_proposals=10000, rng=None):
    if rng is None:
        rng = np.random.default_rng(987654)

    current = binarize_top_half(repair_to_feasible(h0))
    current = repair_to_feasible(current)
    current_score = exact_c5(current)
    holder.maybe_improve(current)

    temp = 0.005
    count = 0

    while count < int(max_proposals) and time_left_sec(t0, budget) > 1.0:
        ones = np.where(current >= 0.9)[0]
        zeros = np.where(current <= 0.1)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = int(ones[rng.integers(0, ones.size)])
        j = int(zeros[rng.integers(0, zeros.size)])
        if i == j:
            count += 1
            continue

        current[i] = 0.0
        current[j] = 1.0
        new = repair_to_feasible(current)
        ns = exact_c5(new)

        accept = False
        if ns < current_score:
            accept = True
        else:
            delta = ns - current_score
            if delta < 0.06 and temp > 1e-7:
                if math.exp(-delta / temp) > float(rng.random()):
                    accept = True

        if accept:
            current = new
            current_score = float(ns)
            holder.maybe_improve(current)
        else:
            current[i] = 1.0
            current[j] = 0.0

        temp *= 0.998
        count += 1

    holder.maybe_improve(current)
    return current


def get_parent():
    try:
        arr = np.asarray(initial_h_values, dtype=float)
    except Exception:
        return None
    if arr is None:
        return None
    arr = arr.ravel()
    if arr.size == 0:
        return None
    return arr


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    budget = max(20.0, float(budget_s))
    rng = np.random.default_rng(int(seed))

    holder = Best()

    parent = get_parent()
    if parent is None:
        parent = np.full(70, 0.5, dtype=float)
    parent = repair_to_feasible(parent)
    holder.maybe_improve(parent)

    if holder.h is None:
        holder.h = parent
        holder.score = exact_c5(parent)
        holder.n = parent.size

    dims = make_dims(holder.h.size)

    # Quick feasible transformations and dimensional transfers.
    for m in dims:
        if time_left_sec(t0, budget) < 8.0:
            break
        m = int(m)
        base = holder.h

        if m == base.size:
            holder.maybe_improve(symmetrize(base))
            nm = base.size
            shifts = (1, max(1, nm // 7), max(1, nm // 5), max(1, nm // 3), nm // 2)
            for s in shifts:
                if 0 < s < nm:
                    holder.maybe_improve(cyclic_shift(base, s))
            holder.maybe_improve(binarize_top_half(base))
        else:
            h_lin = repair_to_feasible(transfer_linear(base, m))
            h_ne = repair_to_feasible(transfer_nearest(base, m))
            holder.maybe_improve(h_lin)
            holder.maybe_improve(h_ne)
            holder.maybe_improve(symmetrize(h_lin))
            holder.maybe_improve(symmetrize(h_ne))
            holder.maybe_improve(binarize_top_half(h_lin))
            holder.maybe_improve(binarize_top_half(h_ne))

    # Primary local active-descent refinement.
    if time_left_sec(t0, budget) > 20.0:
        max_lp = min(250.0, max(25.0, time_left_sec(t0, budget) * 0.25))
        local_lp_search(holder.h, holder, t0, budget, max_iters=800, max_time_s=max_lp, rng=rng)

    # SLSQP epigraph, active subset and full if dimension is modest.
    if time_left_sec(t0, budget) > 60.0:
        n = int(holder.h.size)
        mi = slsqp_maxiter(n, time_left_sec(t0, budget))
        if mi > 0:
            if n <= 95:
                run_slsqp(holder.h, holder, t0, budget, maxiter=mi, active_k=18, betas=(0.0,))
            else:
                run_slsqp(holder.h, holder, t0, budget, maxiter=mi, active_k=25, betas=(0.0,))

        if time_left_sec(t0, budget) > 50.0:
            n = int(holder.h.size)
            mi = slsqp_maxiter(n, time_left_sec(t0, budget))
            if mi > 0:
                if n <= 85:
                    run_slsqp(holder.h, holder, t0, budget, maxiter=mi, active_k=0, betas=(0.0,))
                else:
                    run_slsqp(holder.h, holder, t0, budget, maxiter=mi, active_k=22, betas=(0.0,))

        if time_left_sec(t0, budget) > 50.0:
            n = int(holder.h.size)
            mi = slsqp_maxiter(n, time_left_sec(t0, budget))
            if mi > 0:
                run_slsqp(holder.h, holder, t0, budget, maxiter=max(15, mi // 2),
                          active_k=10, betas=(0.0, 1e-5))

    # Smooth active-row minimization.
    if time_left_sec(t0, budget) > 50.0:
        max_ws = min(180.0, max(20.0, time_left_sec(t0, budget) * 0.25))
        run_weighted_smooth(holder.h, holder, t0, budget, max_cycles=5, max_time_s=max_ws)

    # Random and symmetry-based starts at the current scale.
    starts = []
    for scale in (0.03, 0.06, 0.10, 0.15):
        if time_left_sec(t0, budget) < 60.0:
            break
        starts.append(random_perturb_start(holder.h, rng, scale))

    if time_left_sec(t0, budget) > 50.0:
        starts.append(random_smooth_start(holder.h.size, rng))
    if time_left_sec(t0, budget) > 50.0:
        starts.append(random_uniform_start(holder.h.size, rng))
    if time_left_sec(t0, budget) > 50.0:
        starts.append(random_balanced_binary(holder.h.size, rng))
    if time_left_sec(t0, budget) > 50.0:
        starts.append(binarize_top_half(holder.h))

    for hst in starts:
        if time_left_sec(t0, budget) < 55.0:
            break

        holder.maybe_improve(hst)
        max_lp = min(25.0, max(8.0, time_left_sec(t0, budget) * 0.2))
        h_local, _ = local_lp_search(hst, holder, t0, budget, max_iters=250,
                                     max_time_s=max_lp, rng=rng)

        if time_left_sec(t0, budget) > 35.0 and h_local is not None:
            n = int(h_local.size)
            mi = slsqp_maxiter(n, time_left_sec(t0, budget))
            if n <= 100 and mi > 0:
                run_slsqp(h_local, holder, t0, budget, maxiter=max(15, mi // 2),
                          active_k=18, betas=(0.0,))

    # Binary local search / annealing.
    if time_left_sec(t0, budget) > 70.0:
        binary_local_search(holder.h, holder, t0, budget, max_proposals=12000, rng=rng)
    if time_left_sec(t0, budget) > 50.0:
        rb = random_balanced_binary(holder.h.size, rng)
        binary_local_search(rb, holder, t0, budget, max_proposals=8000, rng=rng)

    # Try neighboring dimensions using the best current vector.
    for idx, m in enumerate(dims):
        if idx == 0:
            continue
        if time_left_sec(t0, budget) < 70.0:
            break
        m = int(m)

        h_lin = repair_to_feasible(transfer_linear(holder.h, m))
        h_ne = repair_to_feasible(transfer_nearest(holder.h, m))
        holder.maybe_improve(h_lin)
        holder.maybe_improve(h_ne)

        for hst in (h_lin, h_ne):
            max_lp = min(22.0, max(6.0, time_left_sec(t0, budget) * 0.18))
            h_local, _ = local_lp_search(hst, holder, t0, budget, max_iters=200,
                                         max_time_s=max_lp, rng=rng)
            if time_left_sec(t0, budget) > 40.0 and h_local is not None:
                n = int(h_local.size)
                mi = slsqp_maxiter(n, time_left_sec(t0, budget))
                if n <= 95 and mi > 0:
                    run_slsqp(h_local, holder, t0, budget, maxiter=max(12, mi // 2),
                              active_k=18, betas=(0.0,))

    # Final polish of the best vector.
    if time_left_sec(t0, budget) > 25.0:
        max_final_lp = min(160.0, max(15.0, time_left_sec(t0, budget) * 0.35))
        local_lp_search(holder.h, holder, t0, budget, max_iters=800,
                        max_time_s=max_final_lp, rng=rng)

    if time_left_sec(t0, budget) > 18.0:
        n = int(holder.h.size)
        mi = slsqp_maxiter(n, time_left_sec(t0, budget))
        if n <= 90 and mi > 0:
            run_slsqp(holder.h, holder, t0, budget, maxiter=max(10, mi // 2),
                      active_k=0, betas=(0.0,))

    final_h = repair_to_feasible(holder.h)
    if final_h is None or final_h.size == 0:
        final_h = np.full(70, 0.5, dtype=float)

    final_score = exact_c5(final_h)
    if not np.isfinite(final_score):
        final_h = np.full(70, 0.5, dtype=float)
        final_score = exact_c5(final_h)

    return final_h, float(final_score), int(final_h.size)
