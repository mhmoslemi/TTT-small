def project_onto_simplex_box(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float((corr * dx).max())

def compute_gradient(h, dx, tau):
    """Compute gradient of softmax-smoothed C5 objective"""
    n = len(h)
    g = np.correlate(h, 1 - h, mode='full') * dx
    
    # Softmax weights
    g_max = g.max()
    exp_g = np.exp(tau * (g - g_max))
    exp_sum = exp_g.sum()
    weights = exp_g / exp_sum
    
    # Gradient computation using convolution
    T1 = np.convolve(weights, 1 - h, 'full')[n-1:2*n-1]
    T2 = np.correlate(weights, h, 'full')[n-1:2*n-1][::-1]
    grad = dx * (T1 - T2)
    
    return grad, g.max()

def project_onto_set(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}"""
    if h.sum() == target_sum:
        return np.clip(h, 0, 1)
    lo, hi = -1.0, 1.0
    lo -= h.max()
    hi += h.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_objective(h, dx):
    """C5 = max_k (correlate(h, 1-h, full) * dx)"""

    return corr.max(), corr
