def grad_max_correlation(h, n, m_star=None):
    """Compute gradient of c[m_star] * dx with respect to h.
    m_star is the index of the max lag. If None, compute for argmax."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    if m_star is None:
        m_star = int(np.argmax(c))
    k = m_star - (n - 1)  # the shift
    grad = np.zeros(n)
    # For each j, contribution from h[n+k] term: (1-h[j-k]) if j-k in [0,n)
    # And from h[n] term: -h[j+k] if j+k in [0,n)

    lo1 = max(0, k)  # j-k >= 0 → j >= k
    hi1 = min(n, n + k)  # j-k < n → j < n+k

        if 0 <= nj < n:

    lo2 = max(0, -k)  # j+k >= 0 → j >= -k

    for j in range(lo2, hi2):

        if 0 <= nj < n:
            grad[j] += -h[nj]
    grad *= dx
    return grad, m_star, c

def grad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    
    grad = np.zeros(n)

    if k >= 0:
        # j ranges from k to n-1, j-k ranges from 0 to n-1-k
        grad[k:] += (1 - h[:n-k])
    else:
        # j ranges from 0 to n-1+k, j-k ranges from -k to n-1
        grad[:n+k] += (1 - h[-k:])
    
    # Term 2: -h[j+k] for valid j+k
    if k >= 0:
        # j ranges from 0 to n-1-k, j+k ranges from k to n-1
        grad[:n-k] += -h[k:]
    else:
        # j ranges from -k to n-1, j+k ranges from 0 to n-1+k
        grad[-k:] += -h[:n+k]
    
    grad *= dx
    return grad, m_star, c

            m = c.max()
            shifted = alpha * (c - m)
            sm = shifted.max()
            smooth = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
            penalty = 1e6 * (x.sum() - n/2.0)**2
            return smooth + penalty
        
        def grad(x):
            c = np.correlate(x, 1-x, mode='full') * (2.0/n)
            m = c.max()

            m_star = int(np.argmax(c))
            k = m_star - (n-1)
            g = np.zeros(n)
            if k >= 0:
                g[k:] += (1-x[:n-k])
                g[:n-k] += -x[k:]
            else:
                g[:n+k] += (1-x[-k:])
                g[-k:] += -x[:n+k]
            g *= (2.0/n)
            # Penalty gradient
            g += 2e6 * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', jac=grad,
                          bounds=[(0,1)]*n,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] += -h[k:]
    else:
        grad[:n + k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n + k]
    grad *= dx
    return grad, m_star, c * dx

def subgradient_momentum(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0
    mom = 0.85
    v = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        grad, m_star, cdx = grad_at_max(h, n)
        m = cdx.max()
        v = mom * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 3.0)
        else:
            h = project_feasible(h - 0.5 * v, n)
            v *= 0.5
            lr *= 0.92
            stall += 1
            if stall > 500:
                h = project_feasible(best_h + 0.03 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.5
                stall = 0
        if lr < 1e-6:
            break
    return best_h, best_c

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e5 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_swap_search(h, n, deadline, rng, max_iter=50000):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    temp = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_bin.copy()
            ones.remove(j)
            ones.append(i)
            zeros = [x for x in range(n) if h_bin[x] == 0]
            temp = max(temp * 0.999, 1e-6)
        else:
            delta = c_new - best_c
            accept = rng.random() < np.exp(-delta / max(temp, 1e-8))
            if accept:
                ones.remove(j)
                ones.append(i)
                zeros = [x for x in range(n) if h_bin[x] == 0]
                best_h = h_bin.copy()
                best_c = c_new
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
        temp *= 0.9995
        if temp < 1e-7:
            temp = 1e-7
    return best_h, best_c

def continuous_refine(h, n, deadline, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(n - 1, i, -1):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
    return best_h, best_c
