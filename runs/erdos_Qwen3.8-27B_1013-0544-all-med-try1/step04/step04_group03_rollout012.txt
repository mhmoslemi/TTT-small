def smart_binary_optimize(h, t0, budget, max_iter=5000000):
    n = len(h)
    h = h.copy()
    best_max = float(np.max(np.correlate(h, 1-h, mode="full")))
    best_h = h.copy()
    
    rng = np.random.default_rng(12345)
    
    # Temperature schedule
    T0 = 0.5
    T_end = 0.001
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 30:
            break
        
        # Pick a 1 and a 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        
        if a == b:
            continue
        
        # Try the swap
        h[a], h[b] = h[b], h[a]
        new_corr = np.correlate(h, 1-h, mode="full")
        new_max = float(np.max(new_corr))
        
        # Accept/reject
        frac = it / max_iter
        T = T0 * (T_end / T0) ** frac
        if new_max < best_max or (new_max < best_max):
            pass  # always accept improvement
        elif new_max <= best_max:
            pass  # neutral move, accept
        else:
            # Worse, accept with probability exp(-(new_max - best_max)/T)
            if rng.random() >= np.exp(-(new_max - best_max) / T):
                h[a], h[b] = h[b], h[a]  # reject
                continue
        
        if new_max < best_max:
            best_max = new_max
            best_h = h.copy()
    
    return best_h, best_max

    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def resample(h, n_new):
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max_obj(x, beta, n):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta) * (2.0 / n)

def _sum_fun(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones(x.shape)

def _slsqp_refine(h0, n, t0, budget):
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    best = h0.copy()
    best_val = compute_c5(h0)
    for beta in [500, 2000, 8000, 30000]:
        remaining = budget - (time.time() - t0)
        if remaining < 8:
            break
        try:
            r = minimize(smooth_max_obj, best, args=(beta, n), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 4000, 'ftol': 1e-16})
            candidate = project_h(r.x, target)
            val = compute_c5(candidate)
            if val < best_val:
                best = candidate
                best_val = val
        except Exception:
            pass
    return best

def _binary_sa(h_init, n, t0, budget, seed):
    h = h_init.astype(float).copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    cur_ones = int(np.sum(h))
    n_half = n // 2
    if cur_ones > n_half:
        oi = np.where(h == 1)[0]
        rng0 = np.random.default_rng(seed)
        for i in range(cur_ones - n_half):
            h[oi[i]] = 0.0
    elif cur_ones < n_half:
        zi = np.where(h == 0)[0]
        rng0 = np.random.default_rng(seed)
        for i in range(n_half - cur_ones):
            h[zi[i]] = 1.0

    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    best_max = float(corr.max())
    best_h = h.copy()
    rng = np.random.default_rng(seed + 7)
    k_arr = np.arange(-(n - 1), n, dtype=np.int64)
    deadline = t0 + budget

    # Phase 1: High temp exploration
    T = 3.0
    cool1 = 0.99992
    stall = 0
    while time.time() < deadline and T > 0.3:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        a = int(oi[rng.integers(len(oi))])
        b = int(zi[rng.integers(len(zi))])
        # Compute delta vectorized
        delta = np.zeros(2 * n - 1)
        ak = a + k_arr
        m1 = (ak >= 0) & (ak < n)
        delta[m1] += h[ak[m1]]
        amk = a - k_arr
        m2 = (amk >= 0) & (amk < n)
        delta[m2] += h[amk[m2]] - 1.0
        bk = b + k_arr
        m3 = (bk >= 0) & (bk < n)
        delta[m3] -= h[bk[m3]]
        bmk = b - k_arr
        m4 = (bmk >= 0) & (bmk < n)
        delta[m4] += 1.0 - h[bmk[m4]]
        new_corr = corr + delta
        new_max = float(new_corr.max())
        if new_max <= best_max:
            h[a], h[b] = h[b], h[a]
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if rng.random() < np.exp((best_max - new_max) / T):
                h[a], h[b] = h[b], h[a]
                corr = new_corr
            stall += 1
        T *= cool1
        if stall > 50000:
            break

    # Phase 2: Low temp exploitation (only accept improvements or neutral)
    remaining_iters = 0
    while time.time() < deadline and remaining_iters < 5000000:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        a = int(oi[rng.integers(len(oi))])
        b = int(zi[rng.integers(len(zi))])
        delta = np.zeros(2 * n - 1)
        ak = a + k_arr
        m1 = (ak >= 0) & (ak < n)
        delta[m1] += h[ak[m1]]
        amk = a - k_arr
        m2 = (amk >= 0) & (amk < n)
        delta[m2] += h[amk[m2]] - 1.0
        bk = b + k_arr
        m3 = (bk >= 0) & (bk < n)
        delta[m3] -= h[bk[m3]]
        bmk = b - k_arr
        m4 = (bmk >= 0) & (bmk < n)
        delta[m4] += 1.0 - h[bmk[m4]]
        new_corr = corr + delta
        new_max = float(new_corr.max())
        if new_max < best_max:
            h[a], h[b] = h[b], h[a]
            corr = new_corr
            best_max = new_max
            best_h = h.copy()
        remaining_iters += 1
        if remaining_iters % 100000 == 0:
            # Check if we're stuck
            pass

    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
        h_parent[:192] = 0.75
        h_parent[192:] = 0.25
        h_parent = project_h(h_parent, 192.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.default_rng(seed)

    # Try multiple resolutions
    n_candidates = [len(h_parent), 360, 384, 420, 480]
    seen = set()
    n_list = []
    for x in n_candidates:
        if x not in seen and x < 1000:
            seen.add(x)
            n_list.append(x)

    for n in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 40:
            break
        alloc = min(remaining - 30, 180)
        if alloc < 20:
            break

        # Initialize
        if n == len(h_parent):
            h0 = h_parent.copy()
        else:
            h0 = resample(h_parent, n)
            h0 = project_h(h0, n / 2.0)

        h_bin = make_binary(h0, n)

        # Binary SA
        h_bin, best_corr = _binary_sa(h_bin, n, t0, alloc, seed + n * 13)
        c5_bin = best_corr * (2.0 / n)

        # Continuous refinement from best binary
        h_cont = h_bin + 0.03 * rng.standard_normal(n)
        h_cont = project_h(h_cont, n / 2.0)
        refine_budget = budget_s - (time.time() - t0)
        if refine_budget > 15:
            h_cont = _slsqp_refine(h_cont, n, t0, min(refine_budget - 10, 80))
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
            elif c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
        elif c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Final polish with remaining time
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        n_cur = len(best_h)
        for scale in [0.02, 0.01, 0.04]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            h_try = best_h + scale * rng.standard_normal(n_cur)
            h_try = project_h(h_try, n_cur / 2.0)
            h_try = _slsqp_refine(h_try, n_cur, t0, min(remaining - 5, 40))
            val = compute_c5(h_try)
            if val < best_c5:
                best_c5 = val
                best_h = h_try.copy()

    return best_h, best_c5, len(best_h)
