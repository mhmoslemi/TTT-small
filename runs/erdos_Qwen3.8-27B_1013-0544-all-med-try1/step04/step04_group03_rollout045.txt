def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        free = (h > 1e-12) & (h < 1.0 - 1e-12)

def subgrad_opt(h, target, n, t0, budget, n_iter=15000, top_k=10):
    cur = h.copy()
    best = compute_c5(cur)
    step = 0.08
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        lags = compute_lags(cur)
        top_idx = np.argsort(lags)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[:n-lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n-lag]
            elif lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n-la])
                grad[:n-la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = max(step * 0.9995, 0.001)
        trial = project_h(cur - step * grad / gn, target)
        c_try = compute_c5(trial)
        if c_try < best - 1e-15:
            best = c_try
            stall = 0
        else:
            stall += 1
            step *= 0.85
        cur = trial
        if stall > 500:
            break
    return cur, best

def coord_polish(h, target, n, t0, budget, max_sweeps=80):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                return cur, best_c5
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    cur = trial
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def periodic_init(n, period, duty, amp, phase):
    h = np.zeros(n)
    for i in range(n):
        t = (i + phase) % period
        if t < period * duty:
            h[i] = amp
        else:
            h[i] = max(0.0, 1.0 - amp)
    return project_h(h, n / 2.0)

def anneal_continuous(h, target, n, t0, budget, n_iter=50000):
    cur = h.copy()
    best = compute_c5(cur)
    best_h = cur.copy()
    rng = np.random.default_rng(12345)
    temp = 0.05
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        temp = 0.05 * (0.9995 ** it)
        j = rng.integers(n)
        delta = rng.standard_normal() * 0.05
        trial = cur.copy()
        trial[j] += delta
        trial = project_h(trial, target)
        c_try = compute_c5(trial)
        if c_try < best or rng.random() < np.exp(-(c_try - best) / max(temp, 1e-15)):
            cur = trial
            if c_try < best:
                best = c_try
                best_h = cur.copy()
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(256, 0.5)
        h_parent[:128] = 0.7
        h_parent = project_h(h_parent, 128.0)

    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    # Phase 1: Diverse periodic initializations at multiple resolutions
    for n_try in [64, 96, 128]:
        if budget_s - (time.time() - t0) < 60:
            break
        target_try = n_try / 2.0
        for period in [4, 6, 8, 12, 16, 24]:
            for duty in [0.3, 0.4, 0.5]:
                for phase in [0, 1]:
                    if budget_s - (time.time() - t0) < 20:
                        break
                    h_try = periodic_init(n_try, period, duty, 0.85, phase)
                    h_try = slsqp_opt(h_try, n_try, target_try, 300, maxiter=1500)
                    h_try, _ = frank_wolfe(h_try, target_try, n_try, t0, budget_s, n_iter=3000, top_k=4)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 0.001:
                        best_h = project_h(resample(h_try, n_parent), n_parent / 2.0)
                        best_c5 = compute_c5(best_h)
                        best_n = n_parent

    # Phase 2: Frank-Wolfe primary optimization at parent resolution
    n_cur = best_n
    target = n_cur / 2.0
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = frank_wolfe(best_h, target, n_cur, t0, budget_s, n_iter=15000, top_k=8)

    # Phase 3: SLSQP refinement with increasing sharpness
    for beta in [200, 800, 3000, 10000, 30000]:
        if budget_s - (time.time() - t0) < 15:
            break
        best_h = slsqp_opt(best_h, n_cur, target, beta, maxiter=8000)
        best_c5 = compute_c5(best_h)

    # Phase 4: Subgradient with Polyak-like step
    if budget_s - (time.time() - t0) > 40:
        best_h, best_c5 = subgrad_opt(best_h, target, n_cur, t0, budget_s, n_iter=15000, top_k=12)

    # Phase 5: More Frank-Wolfe after SLSQP
    if budget_s - (time.time() - t0) > 40:
        best_h, best_c5 = frank_wolfe(best_h, target, n_cur, t0, budget_s, n_iter=10000, top_k=6)

    # Phase 6: Multi-perturbation restart
    if budget_s - (time.time() - t0) > 80:
        scales = [0.06, 0.03, 0.1, 0.02, 0.05, 0.08, 0.015, 0.04]
        for att in range(8):
            if budget_s - (time.time() - t0) < 25:
                break
            hp = best_h + rng.standard_normal(n_cur) * scales[att]
            hp = project_h(hp, target)
            hp = frank_wolfe(hp, target, n_cur, t0, budget_s, n_iter=5000, top_k=6)
            hp = slsqp_opt(hp, n_cur, target, 3000, maxiter=4000)
            hp, _ = subgrad_opt(hp, target, n_cur, t0, budget_s, n_iter=3000, top_k=8)
            hp = slsqp_opt(hp, n_cur, target, 10000, maxiter=4000)
            c_try = compute_c5(hp)
            if c_try < best_c5:
                best_h = hp.copy()
                best_c5 = c_try

    # Phase 7: Try slightly higher resolution
    if budget_s - (time.time() - t0) > 150:
        n_hr = n_cur + 64
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample(best_h, n_hr), target_hr)
        cur_hr = frank_wolfe(cur_hr, target_hr, n_hr, t0, budget_s, n_iter=5000, top_k=6)
        for beta in [500, 3000, 15000]:
            if budget_s - (time.time() - t0) < 20:
                break
            cur_hr = slsqp_opt(cur_hr, n_hr, target_hr, beta, maxiter=5000)
        cur_hr = subgrad_opt(cur_hr, target_hr, n_hr, t0, budget_s, n_iter=3000, top_k=8)[0]
        c_try = compute_c5(cur_hr)
        if c_try < best_c5:
            best_h = cur_hr.copy()
            best_c5 = c_try
            best_n = n_hr
            n_cur = n_hr
            target = target_hr

    # Phase 8: Continuous annealing
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = anneal_continuous(best_h, target, n_cur, t0, budget_s, n_iter=30000)

    # Phase 9: Final SLSQP polish
    if budget_s - (time.time() - t0) > 15:
        for beta in [50000, 150000]:
            if budget_s - (time.time() - t0) < 8:
                break
            best_h = slsqp_opt(best_h, n_cur, target, beta, maxiter=10000)
            best_c5 = compute_c5(best_h)

    # Phase 10: Coordinate descent final polish
    if budget_s - (time.time() - t0) > 10:
        best_h, best_c5 = coord_polish(best_h, target, n_cur, t0, budget_s, max_sweeps=200)

    # Final check
    if budget_s - (time.time() - t0) > 5:
        best_h = slsqp_opt(best_h, n_cur, target, 100000, maxiter=5000)
        best_c5 = compute_c5(best_h)
        best_h = project_h(best_h, target)

    return best_h, best_c5, best_n
