```python
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

    return val
```

- lag = k - (n-1)
- For lag >= 0: sum_{i=0}^{n-1-lag} a[i+lag] * b[i]
- For lag < 0: sum_{i=0}^{n-1+lag} a[i] * b[i-lag]

- For lag >= 0: sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
- For lag < 0: sum_{i=0}^{n-1+lag} h[i] * (1-h[i-lag])

```
grad[:n - lag] -= w * h[lag:]       # -h[j+lag] for j <= n-1-lag
```

Yes, that's correct.

For lag < 0 (let m = -lag > 0):
f_{-m} = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])

For a fixed j:
- h[j] appears in h[i] when i=j. Requires 0 ≤ j ≤ n-1-m.
- h[j] appears in (1-h[i+m]) when i+m=j, i=j-m. Requires 0 ≤ j-m ≤ n-1-m, i.e., m ≤ j ≤ n-1.

df_{-m}/dh[j] = (1-h[j-m]) * [m ≤ j ≤ n-1]... wait, no.

Let me redo: f_{-m} = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])

- h[j] in h[i]: i=j, requires 0 ≤ j ≤ n-1-m. Contribution: (1-h[j+m])
- h[j] in (1-h[i+m]): i=j-m, requires 0 ≤ j-m ≤ n-1-m, i.e., m ≤ j ≤ n-1. Contribution: -h[j-m]

So: df_{-m}/dh[j] = (1-h[j+m]) * [j ≤ n-1-m] - h[j-m] * [j ≥ m]

In vector form:
grad[:n-m] += (1 - h[m:])  # (1-h[j+m]) for j <= n-1-m
grad[m:] -= h[:n-m]        # -h[j-m] for j >= m

This matches:

```
grad[:n - m2] += w * (1.0 - h[m2:])
grad[m2:] -= w * h[:n - m2]
```

One more idea: **Use the `cvxpy` solver**. The problem can be formulated as:
minimize t
subject to: h(x) * (1 - h(x+k)) ≤ t for all k (approximately, since this is not linear)

Actually, this isn't linear because of the h*(1-h) term. But if we fix h to be 0-1, it becomes a different problem.

For the continuous case, the problem is non-convex (h*(1-h) is concave in h, so the constraint is non-convex). So cvxpy won't directly help.

However, we could use a different formulation. Note that:
h*(1-h) = 1/4 - (h-1/2)²

So maximizing h*(1-h) is equivalent to minimizing (h-1/2)². But we're trying to minimize the max of these, which is equivalent to maximizing the min of (h-1/2)². This is still non-convex.

```python
import numpy as np
from scipy.optimize import minimize
import time

_state = {'beta': 100.0, 'target': 250.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def _obj(x):
    val, _ = smooth_max_and_grad(x, _state['beta'])
    return val

def _grad(x):
    _, g = smooth_max_and_grad(x, _state['beta'])
    return g

def _con_eq(x):
    return x.sum() - _state['target']

def _con_eq_jac(x):
    return np.ones(len(x))

def _lbfgs_obj(x):
    base, _ = smooth_max_and_grad(x, _state['beta'])
    p = _state['penalty']
    lam = _state['lam']
    c = x.sum() - _state['target']
    return base + 0.5 * p * c * c + lam * c

def _lbfgs_grad(x):
    _, g = smooth_max_and_grad(x, _state['beta'])
    p = _state['penalty']
    lam = _state['lam']
    c = x.sum() - _state['target']
    return g + (p * c + lam) * np.ones(len(x))

def _slsqp_run(h0, beta, target, maxiter, time_left_fn):
    _state['beta'] = beta
    _state['target'] = target
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds, jac=_grad,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new
    except Exception:
        return h0.copy()

def _lbfgs_aug(h0, beta, target, time_left_fn):
    _state['beta'] = beta
    _state['target'] = target
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    mu = 500.0
    lam = 0.0
    h_cur = h0.copy()
    for _ in range(20):
        if time_left_fn() < 30:
            break
        _state['penalty'] = mu
        _state['lam'] = lam
        try:
            res = minimize(_lbfgs_obj, h_cur, method='L-BFGS-B', bounds=bounds,
                           jac=_lbfgs_grad, options={'maxiter': 200, 'ftol': 1e-19,
                                                     'gtol': 1e-12, 'maxfun': 2000})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_h(h_new, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
            c = h_new.sum() - target
            lam = lam + mu * c
            mu *= 4.0
        except Exception:
            pass
    return h_cur

def _adam_pass(h, target, beta, lr_init, n_iter, time_left_fn):
    n = len(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0
    best_local = compute_c5(h)
    stall = 0
    for _ in range(n_iter):
        if time_left_fn() < 10:
            break
        val, grad = smooth_max_and_grad(h, beta)
        t += 1
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad ** 2
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        h_new = project_h(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        c5_new = compute_c5(h_new)
        if c5_new < best_local - 1e-16:
            h = h_new
            best_local = c5_new
            stall = 0
            lr = min(lr * 1.05, lr_init)
        else:
            stall += 1
            lr *= 0.85
            if stall > 200:
                break
    return h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    def tl():
        return budget_s - (time.time() - start)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP at initial resolution with increasing beta
    for beta in [2000, 10000, 50000, 200000]:
        if tl() < 80:
            break
        h_new = _slsqp_run(best_h, beta, target, 4000, tl)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: L-BFGS-B augmented Lagrangian at current resolution
    if tl() > 100:
        h_new = _lbfgs_aug(best_h, 100000, target, tl)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Increase resolution to 1000
    if tl() > 120:
        n_new = 1000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_h(h_hi, n_new / 2.0)
        t_hi = n_new / 2.0
        for beta in [5000, 50000, 200000]:
            if tl() < 80:
                break
            h_trial = _slsqp_run(h_hi, beta, t_hi, 3000, tl)
            c5_trial = compute_c5(h_trial)
            if c5_trial < compute_c5(h_hi) - 1e-15:
                h_hi = h_trial
        c5_final = compute_c5(h_hi)
        if c5_final < best_c5 - 1e-15:
            best_h, best_c5 = h_hi, c5_final
            n = n_new
            target = t_hi

    # Phase 4: Try resolution 2000
    if tl() > 150:
        n_new = 2000
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_h(h_hi, n_new / 2.0)
        t_hi = n_new / 2.0
        for beta in [10000, 100000]:
            if tl() < 80:
                break
            h_trial = _slsqp_run(h_hi, beta, t_hi, 2000, tl)
            c5_trial = compute_c5(h_trial)
            if c5_trial < compute_c5(h_hi) - 1e-15:
                h_hi = h_trial
        c5_final = compute_c5(h_hi)
        if c5_final < best_c5 - 1e-15:
            best_h, best_c5 = h_hi, c5_final
            n = n_new
            target = t_hi

    # Phase 5: L-BFGS-B at final resolution
    if tl() > 80:
        h_new = _lbfgs_aug(best_h, 200000, target, tl)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Adam fine-tuning
    if tl() > 40:
        h_new, c5_new = _adam_pass(best_h, target, 500000, 0.002, 500, tl)
        if c5_new < best_c5 - 1e-16:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Final SLSQP polish with very high beta
    for beta in [500000, 2000000]:
        if tl() < 30:
            break
        h_new = _slsqp_run(best_h, beta, target, 3000, tl)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_h, best_c5 = h_new, c5_new

    # Final projection to ensure feasibility
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
