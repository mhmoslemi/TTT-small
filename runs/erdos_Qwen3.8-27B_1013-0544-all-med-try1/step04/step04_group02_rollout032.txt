def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def topk_smooth_max(x, k, beta):
    """Smooth max focusing on top-k elements."""
    m = np.max(x)
    # Get top k
    idx = np.argsort(x)[-k:]
    top = x[idx]
    shifted = beta * (top - top.max())
    return top.max() + np.log(np.sum(np.exp(shifted))) / beta

def project_to_simplex_box(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:

        else:
            step = max(step, -(h[mask].min() + 1e-12))

        h = np.clip(h, 0.0, 1.0)

        if r != pn and 150 <= r <= 600:
            resolutions.append(r)
    for d in [-40, -30, -20, 20, 30, 40]:
        r = pn + d
        if 150 <= r <= 600 and r not in resolutions:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))
    n_res = len(resolutions)
    for ri, n in enumerate(resolutions):
        if time.time() > d2:
            break
        src_n = best_n if best_h is not None else pn
        src_h = best_h if best_h is not None else parent
        x_old = np.linspace(0, 2, src_n, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_new, x_old, src_h)
        h0 = project_feasible(h0, n / 2.0)
        remaining = d2 - time.time()
        alloc = remaining / max(n_res - ri, 1) * 0.85
        d_alloc = time.time() + alloc
        h0, c5n = active_set_solve(h0, n, d_alloc, seed + 500 + n, K_start=15, K_end=3)
        h0, c5n = subgrad_polish(h0, n, d_alloc, seed + 600 + n)
        try_update(h0, c5n, n)

    # Phase 3: High-precision refinement (25%)
    d3 = t0 + budget_s * 0.95
    if best_h is not None and time.time() < d3:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        c5 = compute_c5(best_h)
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (ts,)}]

        for K in [12, 8, 5, 3, 2]:
            if time.time() > d3:
                break
            corr = get_corr(best_h)
            active = np.argsort(corr)[-K:]
            for beta in [200, 2000, 20000, 100000]:
                if time.time() > d3:
                    break
                for attempt in range(2):
                    if time.time() > d3:
                        break
                    x0 = best_h.copy()
                    if attempt > 0:
                        x0 = project_feasible(x0 + np.random.randn(n) * 0.003, ts)
                    try:
                        res = minimize(smooth_max_obj, x0, args=(active, beta, dx, n),
                                     jac=smooth_max_grad, method='SLSQP',
                                     bounds=bounds, constraints=cons,
                                     options={'maxiter': 15000, 'ftol': 1e-16})
                        h_new = project_feasible(res.x, ts)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5:
                            best_h = h_new
                            c5 = c5_new
                            break
                    except Exception:
                        pass

        best_h, best_c5 = subgrad_polish(best_h, best_n, d3, seed + 999)

    # Phase 4: Final micro-polish (5%)
    d4 = t0 + budget_s * 0.99
    if best_h is not None and time.time() < d4:
        best_h, best_c5 = subgrad_polish(best_h, best_n, d4, seed + 2000)

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
