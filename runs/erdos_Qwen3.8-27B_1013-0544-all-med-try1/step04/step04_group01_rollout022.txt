```python
import time

    c = np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def lse_obj_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    q = np.correlate(h, 1 - h, mode='full') * dx
    m = q.max()
    sh = alpha * (q - m)
    ms = sh.max()
    es = np.exp(sh - ms)
    Z = es.sum()
    w = es / Z
    f = m + (ms + np.log(Z)) / alpha
    f += penalty * (h.sum() - n / 2.0) ** 2
    
    grad = np.zeros(n)
    # w has length 2n-1, w[k] corresponds to shift s = k-(n-1)
    for j in range(n):
        # First part: sum_s w[s] * (1-h[j-s]) for 0<=j-s<n
        # t = j-s, s = j-t, w stored at index j-t+n-1

        valid = (w_idx >= 0) & (w_idx < 2 * n - 1)
        if valid.any():
            grad[j] += np.sum(w[w_idx[valid]] * (1 - h[t_idx[valid]]))
        # Second part: -sum_s w[s] * h[j+s] for 0<=j+s<n
        # t = j+s, s = t-j, w stored at index t-j+n-1
        t_lo2 = max(0, j - n + 1)
        t_idx2 = np.arange(t_lo2, n)

    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return f, grad

        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8}
    )
    return project_feasible(res.x, n)

def pair_search_fast(h, n, deadline, deltas, max_passes=200):
    best_c = compute_c5(h, n)
    n_sh = 2 * n - 1
    for d in deltas:
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):

                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                            improved = True
                        else:

                    hi2 = h[i] - d

                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

                            improved = True
            if not improved:

    return h, best_c

def local_search(h, n, deadline):
    """Adaptive-step local search with single and pair moves."""
    best_c = compute_c5(h, n)
    base_deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

                                improved = True
                            else:

    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, penalty):
    dx = 2.0 / n
    q = np.correlate(h, 1 - h, mode='full') * dx
    m = q.max()
    sh = alpha * (q - m)
    ms = sh.max()
    es = np.exp(sh - ms)
    Z = es.sum()
    w = es / Z
    f = m + (ms + np.log(Z)) / alpha
    f += penalty * (h.sum() - n / 2.0) ** 2
    J, T = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    W1 = w[J + n - 1 - T]
    W2 = w[T - J + n - 1]
    grad = (W1 @ (1.0 - h)) - (W2 @ h)
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return f, grad

def lse_lbf(h0, n, alpha, penalty, maxiter=600):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return lse_obj_grad(x, n, alpha, penalty)[0]
    def jac(x):
        return lse_obj_grad(x, n, alpha, penalty)[1]
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
    return project_feasible(res.x, n)

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                hi = h[i] + d
                hi2 = h[i] - d
                for j in range(i + 1, n):
                    if time.time() > deadline - 1:
                        return h, best_c
                    if hi <= 1.0 and h[j] - d >= 0.0:
                        h[i], h[j] = hi, h[j] - d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            return h, best_c
                        h[j] += d
                    if hi2 >= 0.0 and h[j] + d <= 1.0:
                        h[i], h[j] = hi2, h[j] + d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            return h, best_c
                        h[j] -= d
                h[i] = h[i]  # no net change
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                hi = h[i] + d
                hi2 = h[i] - d
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if j == i:
                        continue
                    if hi <= 1.0 and h[j] - d >= 0.0:
                        h[i], h[j] = hi, h[j] - d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            return h, best_c
                        h[j] += d
                    if hi2 >= 0.0 and h[j] + d <= 1.0:
                        h[i], h[j] = hi2, h[j] + d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            return h, best_c
                        h[j] -= d
    return h, best_c

def optimize_from(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [50, 200, 1000, 5000, 20000, 80000, 300000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = lse_lbf(best_h, n, a, penalty=1e5, maxiter=800)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 10:
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, min(deadline - 5, time.time() + 40),
                                         [0.03, 0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = near_pair_search(best_h, n, min(deadline - 5, time.time() + 40),
                                              [0.03, 0.01, 0.005, 0.002, 0.001], radius=50)
    if time.time() < deadline - 8:
        ht = lse_lbf(best_h, n, 1000000, penalty=1e6, maxiter=1000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 3:
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 1, [0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.005, 0.002, 0.001, 0.0005], radius=40)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 150
    if h_init is not None:
        alloc = min(350, (deadline - t0) * 0.40)
        h_opt, c_opt = optimize_from(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150, 200, 250, 300]:
            if n_try == n_init:
                continue
            if time.time() > deadline - 80:
                break
            alloc = min(120, (deadline - time.time()) * 0.18)
            if alloc < 40:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_from(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 15
        n_restarts = max(2, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_from(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.003, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.003, 0.001, 0.0005, 0.0002], radius=30)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
```
