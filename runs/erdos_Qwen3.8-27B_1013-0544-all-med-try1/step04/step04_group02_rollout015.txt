import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_binary(n, seed=0):
    N = 1
    while N < n:
        N *= 2
    seq = np.zeros(N, dtype=np.int8)
    length = 1
    while length < N:
        seq[length:2*length] = 1 - seq[:length]
        length *= 2
    h = seq[:n].astype(float)
    target = n // 2
    n_ones = int(h.sum())
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        flip = rng.permutation(ones)[:n_ones - target]
        h[flip] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        flip = rng.permutation(zeros)[:target - n_ones]
        h[flip] = 1
    return h

def golden_ratio_binary(n, seed=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    rng = np.random.RandomState(seed)
    offsets = rng.random()
    h = np.zeros(n)
    num_ones = n // 2
    positions = np.floor(offsets + np.arange(num_ones) * phi).astype(int) % n
    h[positions] = 1.0
    if h.sum() != num_ones:
        h = thue_morse_binary(n, seed)
    return h

def subgrad_at_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        return grad
    if lag > 0:
        m = n - lag
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def optimize_subgrad(h, n, deadline, seed, n_lags=8, lr_init=0.004, binary_push=0.0):
    rng = np.random.RandomState(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    velocity = np.zeros(n)
    lr = lr_init
    stagnant = 0
    momentum = 0.82
    for it in range(80000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(n_lags, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        temp = 8.0
        w = np.exp(temp * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(top_k):
            lag = top_idx[j] - (n - 1)
            if lag == 0:
                continue
            grad += w[j] * subgrad_at_lag(h, lag, n)
        velocity = momentum * velocity - lr * grad
        h_new = h + velocity
        if binary_push > 0:
            h_new = h_new * (1.0 - binary_push) + (h_new > 0.5).astype(float) * binary_push
        h_new = project_feasible(h_new, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            h = h_new
            c5 = c_new
            if c_new < best_c5:
                best_h = h.copy()
                best_c5 = c_new
            stagnant = 0
            lr = min(lr * 1.005, 0.008)
        else:
            stagnant += 1
            velocity *= 0.4
            lr *= 0.90
            if stagnant > 400 and lr < 5e-6:
                break
    return best_h, best_c5

def slsqp_optimize(h, n, deadline, seed):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    c5 = compute_c5(h)
    beta = 20000
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500.0, 500.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    def con_fun(x):
        return x.sum() - ts
    def con_jac(x):
        return np.ones_like(x)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    for p in [0.0, 0.005, 0.015]:
        if time.time() > deadline:
            break
        x0 = h.copy()
        if p > 0:
            x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
        except Exception:
            pass
    return h, c5
