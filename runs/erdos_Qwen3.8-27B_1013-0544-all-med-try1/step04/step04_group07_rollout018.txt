def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def swap_delta(corr, h, i, j, n):
    """Compute the delta in corr when h[i] goes 1->0 and h[j] goes 0->1."""
    # corr has length 2n-1, indexed from lag -(n-1) to n-1
    # corr[k] corresponds to lag k - (n-1)
    delta = np.zeros(2*n-1)
    
    # Effect of h[i]: 1 -> 0
    # For lag k >= 0 (index k + n - 1 in corr):
    #   term h[t]*(1-h[t+k]) with t=i: exists if i <= n-1-k, i.e., k <= n-1-i
    #   delta: -(1-h[i+k]) for k in [0, n-1-i]
    #   term h[t]*(1-h[t+k]) with t+k=i, i.e., t=i-k: exists if i-k >= 0, i.e., k <= i
    #   delta: -h[i-k] for k in [0, min(i, n-1)]
    
    # For lag k < 0 (let l=-k, index l in corr, i.e., index n-1-l):
    #   a_{-l} = sum_{t=l}^{n-1} h[t]*(1-h[t-l])
    #   term with t=i: exists if i >= l, i.e., l <= i
    #   delta: -(1-h[i-l]) for l in [0, i]
    #   term with t-l=i, i.e., t=i+l: exists if i+l <= n-1, i.e., l <= n-1-i
    #   delta: -h[i+l] for l in [0, min(n-1-i, n-1)]
    
    for k in range(0, n - i):  # lag k, index k + n - 1
        delta[k + n - 1] -= (1 - h[i + k])
    for k in range(0, min(i + 1, n)):  # lag k, t = i-k
        if i - k >= 0:
            delta[k + n - 1] -= h[i - k]
    
    for l in range(0, min(i + 1, n)):  # lag -l, index n-1-l
        delta[n - 1 - l] -= (1 - h[i - l])
    for l in range(0, min(n - i, n)):  # lag -l, t = i+l
        if i + l < n:
            delta[n - 1 - l] -= h[i + l]
    
    # Effect of h[j]: 0 -> 1
    # Reverse of the above with i replaced by j
    for k in range(0, n - j):
        delta[k + n - 1] += (1 - h[j + k])
    for k in range(0, min(j + 1, n)):
        if j - k >= 0:
            delta[k + n - 1] += h[j - k]

    for l in range(0, min(n - j, n)):
        if j + l < n:
            delta[n - 1 - l] += h[j + l]
    
    return delta

Wait, I need to double-check the gradient computation. The smooth max of c(x) = corr(x) is:
φ(x) = max_k c_k(x) + (1/β) log(sum_k exp(β(c_k(x) - max_j c_j(x))))

The gradient is:
∇φ(x) = sum_k w_k ∇c_k(x)

where w_k = exp(β(c_k - max)) / sum_j exp(β(c_j - max))

And c_k(x) = sum_t x[t] * (1-x[t+k]) (for lag k ≥ 0, over valid t)

∇c_k(x)[i] = 
- (1-x[i+k]) if i is in the "first factor" range and i+k < n
- -x[i-k] if i is in the "second factor" range (i.e., i = t+k for some valid t, so t = i-k)
- Both if both conditions hold

Let me be more careful:
c_k = sum_{t=0}^{n-1-k} x[t] * (1-x[t+k]) for k ≥ 0

∂c_k/∂x[i] = 
- (1-x[i+k]) if i ≤ n-1-k (i is a valid "t")
- -x[i-k] if i ≥ k (i = t+k for t = i-k, and t ≥ 0 means i ≥ k, and t ≤ n-1-k means i-k ≤ n-1-k, i.e., i ≤ n-1, which is always true)
- Sum of both if both conditions hold

So:
∂c_k/∂x[i] = (1-x[i+k]) * [i ≤ n-1-k] - x[i-k] * [i ≥ k]

For k < 0, let l = -k > 0:
c_{-l} = sum_{t=l}^{n-1} x[t] * (1-x[t-l])

∂c_{-l}/∂x[i] = (1-x[i-l]) * [i ≥ l] - x[i+l] * [i ≤ n-1-l]

This is correct.

Now, for the full gradient:
∇φ(x)[i] = (2/n) * sum_k w_k * ∂c_k/∂x[i]

This is O(n²) to compute (n lags × n positions), but for n=384, that's ~147K operations, which is fast.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta, n):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    exps = np.exp(np.clip(beta * (c - m), -500, 500))
    return (m + np.log(exps.sum() / len(c)) / beta) * (2.0 / n)

def smooth_max_grad(x, beta, n):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    exps = np.exp(np.clip(beta * (c - m), -500, 500))
    w = exps / exps.sum()
    g = np.zeros(n)
    # Vectorized gradient computation
    # For each lag k (index k + n - 1 in c):
    #   if k >= 0: ∂c/∂x[i] = (1-x[i+k])*(i<=n-1-k) - x[i-k]*(i>=k)
    #   if k < 0: l=-k, ∂c/∂x[i] = (1-x[i-l])*(i>=l) - x[i+l]*(i<=n-1-l)
    
    for idx in np.where(w > 1e-8)[0]:
        ww = w[idx]
        lag = int(idx) - (n - 1)
        if lag >= 0:
            # First factor: positions 0 to n-1-lag
            g[:n - lag] += ww * (1 - x[lag:])
            # Second factor: positions lag to n-1
            g[lag:] += ww * (-x[:n - lag])
        else:
            la = -lag
            # First factor: positions la to n-1
            g[la:] += ww * (1 - x[:n - la])
            # Second factor: positions 0 to n-1-la
            g[:n - la] += ww * (-x[la:])
    
    return g * (2.0 / n)

def sum_constraint(x, target):
    return x.sum() - target

def slsqp_smooth(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        result = minimize(lambda x: smooth_max_obj(x, beta, n), h0,
                         jac=lambda x: smooth_max_grad(x, beta, n),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(result.x, target)

    corr = np.correlate(h, 1.0 - h, mode="full")
    curr_max = float(corr.max())
    best_h = h.copy()
    best_max = curr_max
    rng = np.random.RandomState(rng_seed)
    
    no_improve = 0
    while time.time() - t0 < budget - 60:
        # Rebuild position lists every 1000 steps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for _ in range(1000):
            if time.time() - t0 > budget - 60:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode="full")
            new_max = float(new_corr.max())
            
            if new_max <= curr_max:
                curr_max = new_max
                corr = new_corr
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        
        if no_improve > 20000:
            # Kick: accept 3 random swaps
            for _ in range(3):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode="full")
            curr_max = float(corr.max())
            no_improve = 0
    
    return best_h, best_max
