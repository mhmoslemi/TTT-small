    c5 = compute_c5(h0)
    if betas is None:
        betas = [50, 200, 500, 2000, 10000, 50000, 200000]
    for beta in betas:
        if time.time() > deadline:
            break
        perts = [0.0, 0.003, 0.008, 0.015]
        for p in perts:
            if time.time() > deadline:
                break
            x0 = h0.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_grad_smooth, x0, args=(dx, beta), method='SLSQP',
                               jac=True, bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h0 = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h0, c5

def subgradient_polish(h, n, deadline, seed, lr_init=0.003):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = lr_init
    stagnant = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(8, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        temp = 5.0
        w = np.exp(temp * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            if lag >= 0:
                idx = np.arange(n - lag)
                grad[idx] += w[j] * (1.0 - h[idx + lag])
                grad[idx + lag] -= w[j] * h[idx]
            else:
                la = -lag
                idx = np.arange(n - la)
                grad[idx + la] += w[j] * (1.0 - h[idx])
                grad[idx] -= w[j] * h[idx + la]
        velocity = 0.8 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.003, 0.008)
        else:
            stagnant += 1
            velocity *= 0.3
            lr *= 0.85
        if stagnant > 400 and lr < 3e-6:
            break
    return h, c5

def coordinate_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    order = np.random.permutation(n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in order:
            if time.time() > deadline:
                return h, c5
            old_val = h[i]
            # Try a few candidate values around old_val
            candidates = [old_val]
            for delta in [0.001, -0.001, 0.005, -0.005, 0.02, -0.02]:
                v = old_val + delta
                if 1e-6 < v < 1.0 - 1e-6:
                    candidates.append(v)
            # Also try the endpoints if close
            if old_val < 0.1:
                candidates.append(0.0)
            if old_val > 0.9:
                candidates.append(1.0)
            best_val = old_val
            best_c = c5
            for v in candidates:
                h[i] = v
                h[i] = np.clip(v, 0, 1)
                c_trial = compute_c5(h)
                if c_trial < best_c:
                    best_c = c_trial
                    best_val = h[i]
            h[i] = best_val
            if best_c < c5 - 1e-12:
                c5 = best_c
                improved = True
            else:
                h[i] = old_val
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Main optimization at parent resolution (50% time)
    d1 = t0 + budget_s * 0.50
    h1 = parent.copy()
    h1, c1 = optimize_slsqp(h1, pn, d1, seed)
    h1, c1 = subgradient_polish(h1, pn, d1, seed + 1)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Try reversed
    if time.time() < d1:
        h_rev = project_feasible(parent[::-1].copy(), pn / 2.0)
        c_rev = compute_c5(h_rev)
        if c_rev < best_c5:
            h2, c2 = optimize_slsqp(h_rev, pn, d1, seed + 200)
            h2, c2 = subgradient_polish(h2, pn, d1, seed + 201)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c2, pn

    # Phase 2: Other resolutions (25% time)
    d2 = t0 + budget_s * 0.75
    x_old = np.linspace(0, 2, pn, endpoint=False)
    for n_try in [pn - 20, pn + 20, pn - 50, pn + 50, 250, 400, 500]:
        if time.time() > d2:
            break
        if n_try < 80 or n_try == pn or n_try > 600:
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h0 = np.interp(x_new, x_old, parent, period=None)
        h0 = project_feasible(h0, n_try / 2.0)
        h0, c0 = optimize_slsqp(h0, n_try, d2, seed + n_try,
                                 betas=[100, 500, 2000, 10000, 50000])
        h0, c0 = subgradient_polish(h0, n_try, d2, seed + n_try + 1)
        if c0 < best_c5:
            best_h, best_c5, best_n = h0.copy(), c0, n_try

    # Phase 3: Coordinate polish + high-beta refinement (15% time)
    d3 = t0 + budget_s * 0.90
    if best_h is not None and time.time() < d3:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        h_cur = best_h.copy()
        c_cur = best_c5
        for beta in [100000, 500000, 2000000]:
            if time.time() > d3:
                break
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
            try:
                res = minimize(obj_grad_smooth, h_cur, args=(dx, beta), method='SLSQP',
                               jac=True, bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c_new = compute_c5(h_new)
                if c_new < c_cur:
                    h_cur = h_new
                    c_cur = c_new
            except Exception:
                pass
        best_h, best_c5 = h_cur, c_cur
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 4: Final coordinate + subgradient polish (10% time)
    d4 = t0 + budget_s * 0.98
    if best_h is not None and time.time() < d4:
        best_h, best_c5 = coordinate_polish(best_h, best_n, d4, seed + 500)
        best_h, best_c5 = subgradient_polish(best_h, best_n, d4, seed + 600, lr_init=0.001)

    # Final
    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
