```python
import cvxpy as cp

def linearized_step(h, epsilon, n_constraints=20):
    """Solve a linearized subproblem to get a descent direction."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_constraints]
    
    # Compute gradients at h for the top correlations
    grads = []
    for k in top_idx:
        L = int(k) - (n - 1)
        g = np.zeros(n)
        if L == 0:
            g[:] = 1.0 - 2.0 * h
        elif 0 < L < n:
            g[L:] = 1.0 - h[:-L]
            g[:n-L] -= h[L:]
        elif L < 0 and abs(L) < n:
            m2 = -L
            g[:n-m2] = 1.0 - h[m2:]
            g[m2:] -= h[:n-m2]
        grads.append(g)
    
    # Linearized constraints: c(k) + grad_k · (x - h) ≤ t
    # i.e., grad_k · x ≤ t - c(k) + grad_k · h
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2.0]
    for k, g in zip(top_idx, grads):
        c_k = corr[k] * dx
        constraints.append(g @ x * dx <= t - c_k + g @ h * dx)
    
    # Also add a trust region
    constraints.append(cp.norm(x - h) <= epsilon)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=1000)
        if x.value is not None:
            return np.clip(x.value, 0, 1)
    except:
        pass
    
    return h
```

For example:
- h(x) = 0.5 + a*sin(2πx) + b*sin(4πx) + ... (Fourier series)
- h(x) = indicator of a carefully chosen set

With a few terms, this is a low-dimensional optimization problem.

Actually, I think the most promising approach is:

1. **Start from initial_h_values** and do a very aggressive optimization
2. **Try perturbations**: add random noise of various scales and re-optimize
3. **Use a different optimization algorithm** (e.g., trust-region, or L-BFGS-B with the smooth max)

Let me try L-BFGS-B with the smooth max. L-BFGS-B is generally faster than SLSQP for large problems because it doesn't form the full Hessian approximation.

Wait, but we have an equality constraint (sum = n/2). L-BFGS-B doesn't handle equality constraints directly. We can either:
- Eliminate one variable (express h[n-1] = target - sum(h[0:n-1]))
- Use a penalty method

The penalty approach: add λ*(sum(h) - target)² to the objective. With a large λ, this enforces the constraint approximately.

Or better: parameterize h[0:n-1] freely and set h[n-1] = target - sum(h[0:n-1]). Then we only need box constraints on h[0:n-1] (and h[n-1] must also be in [0,1], which adds another constraint).

Actually, the cleanest approach for L-BFGS-B is:
- Variables: h[0], ..., h[n-2] (n-1 variables)
- Constraint: sum(h[0:n-1]) ∈ [target - 1, target] (so that h[n-1] = target - sum(h[0:n-1]) ∈ [0,1])
- This is a box constraint on the sum, which L-BFGS-B can't handle directly.

Alternative: just use the penalty method. f_penalty(h) = f_smooth_max(h) + λ*(sum(h) - target)². For large λ, this is close to the constrained problem.

Actually, the simplest is to just use SLSQP but with more iterations and tighter tolerances. Or use the trust-constr method.

Let me just go with a practical approach:

1. SLSQP with various betas (as before but more aggressive)
2. L-BFGS-B with penalty (faster per iteration)
3. Coordinate-wise pairwise descent
4. Multi-start

Let me also add a **restarting strategy**: if we haven't improved in a while, perturb the solution and restart.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -400, 400)
    ev = np.exp(shifted)
    se = ev.sum()
    w = ev / se
    val = (m + np.log(se) / beta) * dx
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wi = w[k]
        L = int(k) - (n - 1)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < L < n:
            grad[L:] += wi * (1.0 - h[:-L])
            grad[:n-L] -= wi * h[L:]
        else:
            m2 = -L
            if m2 < n:
                grad[:n-m2] += wi * (1.0 - h[m2:])
                grad[m2:] -= wi * h[:n-m2]
    return val, grad * dx

def topk_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        L = int(k) - (n - 1)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < L < n:
            grad[L:] += (1.0 - h[:-L])
            grad[:n-L] -= h[L:]
        else:
            m2 = -L
            if m2 < n:
                grad[:n-m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n-m2]
    return grad * dx / top_k, top_idx

def slsqp_once(h0, target, beta, maxiter=5000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val_grad(x, beta)[0]
    def jac(x):
        return smooth_max_val_grad(x, beta)[1]
    def ceq(x):
        return x.sum() - target
    def ceq_j(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, jac=jac,
                   constraints=[{'type': 'eq', 'fun': ceq, 'jac': ceq_j}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_h(res.x, target)

def lbfgsb_once(h0, target, beta, maxiter=20000, lam=1e4):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smooth_max_val_grad(x, beta)
        s = x.sum() - target
        return v + lam * s * s
    def jac(x):
        _, g = smooth_max_val_grad(x, beta)
        s = x.sum() - target
        return g + 2 * lam * s * np.ones(n)
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, jac=jac,
                   options={'maxiter': maxiter, 'gtol': 1e-10})
    return project_h(res.x, target)
