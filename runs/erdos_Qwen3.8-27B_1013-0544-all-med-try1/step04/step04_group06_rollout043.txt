The second term is quadratic (bilinear), so the problem is non-convex.

            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_at_lags(h, lags, weights=None):

    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(lags)):
        w = 1.0 if weights is None else weights[i]

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def grad_at_lags(h, lags, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(lags)):
        w = weights[i]
        lag = int(lags[i]) - (n - 1)
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w * (1.0 - h[:-lag])
            g[:n - lag] -= w * h[lag:]

            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n - m]
    return g * dx

    return idx, w

        g = grad_at_lags(best, idx, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            v = g
        else:
            v /= vn
        imp = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
            ht = project_sum(best - step * v, target)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best, best_c5, imp = ht, c5t, True
                break
        if not imp:
            pat += 1
            if pat > 500:
                break
        else:
            pat = 0
    return best, best_c5


def coord_local(h, target, sweeps):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            if best[i] < 1e-6 or best[i] > 1.0 - 1e-6:
                continue
            for d in [0.015, -0.015, 0.006, -0.006, 0.002, -0.002]:
                ht = best.copy()
                ht[i] = np.clip(best[i] + d, 0.0, 1.0)
                ht = project_sum(ht, target)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-15:
                    best, best_c5, imp = ht, c5t, True
                    break
        if not imp:
            break
    return best, best_c5
