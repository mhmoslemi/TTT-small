import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    """Gradient of sum_j h[j]*(1-h[j+lag]) with respect to h."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:lag+m])  # wait, h[lag:] has length n-lag = m

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_grad_vector(h, i, n):
    """Compute d(corr)/d(h[i]) as a vector of length 2n-1."""
    g = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        L = p - (n - 1)
        if L >= 0:
            if 0 <= i <= n - L - 1:
                g[p] += (1.0 - h[i + L])
            if L <= i <= n - 1 and L > 0:
                g[p] -= h[i - L]
        else:
            M = -L
            if M <= i <= n - 1:
                g[p] += (1.0 - h[i - M])
            if 0 <= i <= n - M - 1 and M > 0:
                g[p] -= h[i + M]
    return g

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, deadline, max_passes=10000):
    ts = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stagnant = 0
    
    for it in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            g = get_grad_vector(h, i, n)
            
            # Find best delta in [-h[i], 1-h[i]]
            dm = -h[i]
            dM = 1.0 - h[i]
            
            # Evaluate at candidates
            cands = np.array([dm, 0.0, dM])
            # Add crossing points of top-3
            top3 = np.argsort(corr)[-3:]
            for a in range(3):
                for b in range(a + 1, 3):
                    if abs(g[top3[a]] - g[top3[b]]) > 1e-10:
                        dc = (corr[top3[b]] - corr[top3[a]]) / (g[top3[a]] - g[top3[b]])
                        if dm < dc < dM:
                            cands = np.append(cands, dc)
            
            best_d = 0.0
            best_v = np.max(corr + 0 * g)
            for d in cands:
                d = np.clip(d, dm, dM)
                v = np.max(corr + d * g)
                if v < best_v:
                    best_v = v
                    best_d = d
            
            if abs(best_d) > 1e-10:
                h[i] += best_d
                h = project_feasible(h, ts)
                improved = True
        
        if improved:
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_h = h.copy()
                best_c5 = c5_new
                stagnant = 0
            else:
                h = best_h.copy()
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 20:
            break
    
    return best_h, best_c5

An alternative: use a "paired" update where we increase h[i] by δ and decrease h[j] by δ (for some j) to maintain the sum exactly. This avoids the projection step.

For the paired update, the objective as a function of δ is:
f(δ) = max_p (corr[p] + δ * (g_i[p] - g_j[p]))

This is still a max of linear functions in δ, so it's convex and can be minimized exactly.

But which j to pair with i? Options:
- Random j
- j that minimizes the impact (|g_i - g_j| is small)
- j = (i+1) % n (adjacent)
- j that maximizes h[j] (to keep the change small in relative terms)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_i(h, i, n):
    """Compute d(corr)/d(h[i]) for all lags."""
    g = np.zeros(2 * n - 1)
    p = np.arange(2 * n - 1)
    L = p - (n - 1)
    
    # First factor: h[i] as h[j] in the sum
    # L >= 0: valid if i <= n-1-L, i.e., L <= n-1-i

    if m1.any():
        g[m1] += (1.0 - h[i + L[m1]])
    # L < 0: M = -L, valid if i >= M = -L, i.e., L >= -i

            break
        improved = False

                return best_h, best_c5

def subgrad_step(h, n, alpha, momentum_buf):
    """One subgradient step with momentum."""
    ts = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = min(10, len(corr) // 5)
    top_idx = np.argsort(corr)[-top_k:]
    w = np.exp(8.0 * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    for j, p in enumerate(top_idx):

        # Compute gradient of corr[p] w.r.t. h vector
