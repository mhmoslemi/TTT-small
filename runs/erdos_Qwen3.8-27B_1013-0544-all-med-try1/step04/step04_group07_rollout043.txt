```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0, 1).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0, 1)
    return h

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    t = int(round(target))
    diff = t - int(z.sum())
    if diff > 0:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        z[zeros[order[:diff]]] = 1.0
    elif diff < 0:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:(-diff)]]] = 0.0
    return z

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_max(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta*(corr-m), -500, 500))) / len(corr)) / beta

def make_sml_obj(beta, dx):
    def obj(h):
        return smooth_max(np.correlate(h, 1.0-h, mode="full"), beta) * dx
    return obj

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, target, beta, maxiter, ftol):
    n = len(h0)
    dx = 2.0/n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type':'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(make_sml_obj(beta, dx), h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum(r.x, target)
    except:
        return h0

def local_search(h0, t0, time_limit, rng=None):
    """Iterated local search with perturbation."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    while time.time() < time_limit:
        # Local search: try random swaps, keep improvements
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        n_stall = 0
        
        while time.time() < time_limit and n_stall < 200:
            j = rng.choice(ones)
            k = rng.choice(zeros)
            h[j], h[k] = 0, 1
            c_new = compute_c5(h)
            if c_new < c5 - 1e-15:
                c5 = c_new
                improved = True
                n_stall = 0
                if c5 < best_c5:
                    best_h, best_c5 = h.copy(), c5
            else:
                h[j], h[k] = 1, 0
                n_stall += 1
        
        if not improved:
            # Perturbation: flip a few random pairs
            n_flip = rng.randint(3, 10)
            for _ in range(n_flip):
                j = rng.choice(ones)
                k = rng.choice(zeros)
                h[j], h[k] = 0, 1
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
    
    return best_h, best_c5

def greedy_construction(n, seed):
    """Greedily construct a binary sequence with low max correlation."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Start with random assignment
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1
    
    # Greedy improvement: repeatedly find the best swap
    for step in range(200):
        best_c5 = compute_c5(h)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_delta = 1e-10
        
        # Try a subset of swaps
        n_tries = min(2000, len(ones)*len(zeros))
        for _ in range(n_tries):
            j = rng.choice(ones)
            k = rng.choice(zeros)
            h[j], h[k] = 0, 1
            c = compute_c5(h)
            h[j], h[k] = 1, 0
            if c < best_c5 - best_delta:
                best_delta = best_c5 - c
                best_swap = (j, k)
        
        if best_swap is None:
            break
        j, k = best_swap
        h[j], h[k] = 0, 1
    
    return h, compute_c5(h)
