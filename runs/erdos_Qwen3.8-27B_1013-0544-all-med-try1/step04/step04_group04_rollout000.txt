                W[lag] += weights[k]
    grad_A = np.correlate(1.0 - h, W, 'full')[:n]
    grad_B = np.convolve(W, h, 'full')[:n]
    grad = (grad_A - grad_B) * dx
    return val, grad

def slsqp_obj(x, beta, top_k):
    v, _ = smooth_max_grad(x, beta, top_k)
    return v

def slsqp_grad_fn(x, beta, top_k):
    _, g = smooth_max_grad(x, beta, top_k)
    return g

def do_slsqp_fast(h0, betas, maxiter_each, time_fn, top_k_list=None):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(len(x))}]
    for i, beta in enumerate(betas):
        if time_fn() < 15:
            break
        tk = top_k_list[i] if top_k_list else None
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP',
                          bounds=bounds, jac=slsqp_grad_fn,
                          constraints=constraints, args=(beta, tk),
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
        if time_fn() < 15:
            h_cur = project_h(h_cur, target)
            c5_cur = compute_c5(h_cur)
    return h_cur, c5_cur

def proj_grad_descent(h, target, n_iter, time_fn, top_k_init, top_k_max,
                      step_init, step_min, perturb_every, perturb_scale):
    n = len(h)
    dx = 2.0 / n
    no_imp = 0
    best_c5 = compute_c5(h)
    step = step_init
    top_k = top_k_init
    for it in range(n_iter):
        if time_fn() < 3:
            break
        val, grad = smooth_max_grad(h, 500.0, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        # Project gradient to be orthogonal to sum constraint
        grad = grad - np.mean(grad)
        found = False
        s = step
        for _ in range(20):
            ht = project_h(h - s * grad, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                h = ht
                best_c5 = c5t
                found = True
                step = min(s * 1.5, step_init)
                no_imp = 0
                break
            s *= 0.5
        if not found:
            no_imp += 1
            step = max(s, step_min)
            # Adaptive top_k
            top_k = min(top_k_max, top_k_init + no_imp // 50)
            if no_imp % perturb_every == perturb_every - 1:
                sc = perturb_scale * (1 + no_imp // perturb_every * 0.5)
                noise = np.random.randn(n) * sc
                noise -= np.mean(noise)
                ht = project_h(h + noise, target)
                if compute_c5(ht) < best_c5:
                    h = ht
                    best_c5 = compute_c5(h)
                    no_imp = 0
            if no_imp > 2000:
                break
    return h, best_c5

def try_fourier_start(n, n_terms, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_terms + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = project_h(h, n / 2.0)
    return h

def try_block_start(n, n_blocks, rng):
    vals = np.sort(rng.dirichlet(np.ones(n_blocks) * 3))
    h = np.repeat(vals, n // n_blocks + 1)[:n]
    h = project_h(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: SLSQP with FFT-based gradient, increasing beta
    if time_left() > 60:
        betas1 = [50, 200, 500, 2000, 8000, 30000]
        tks1 = [10, 8, 6, 5, 4, 3]
        h_new, c5_new = do_slsqp_fast(best_h, betas1, 3000, time_left, tks1)
        if c5_new < best_c5:
            best_h, best_c5, n_best = h_new, c5_new, len(h_new)

    # Phase 2: Projected gradient with adaptive top-k
    if time_left() > 60:
        tgt = len(best_h) / 2.0
        best_h, best_c5 = proj_grad_descent(
            best_h, tgt, 20000, time_left,
            top_k_init=5, top_k_max=30,
            step_init=0.05, step_min=0.0001,
            perturb_every=300, perturb_scale=0.008)

    # Phase 3: Multi-resolution exploration
    for n_res in [200, 300, 400, 500, 600, 800]:
        if time_left() < 80:
            break
        if n_res == n_best:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_res, endpoint=False)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        betas2 = [200, 1000, 5000, 20000]
        tks2 = [8, 6, 5, 4]
        h_nr, c5_nr = do_slsqp_fast(h_res, betas2, 2500, time_left, tks2)
        if c5_nr < best_c5:
            best_h, best_c5, n_best = h_nr, c5_nr, n_res

    # Phase 4: Alternative starts
    for n_alt in [200, 300, 400]:
        if time_left() < 120:
            break
        for init_fn, kwargs in [
            (try_fourier_start, {'n_terms': 5}),
            (try_fourier_start, {'n_terms': 8}),
            (try_block_start, {'n_blocks': 6}),
            (try_block_start, {'n_blocks': 10}),
        ]:
            if time_left() < 40:
                break
            h_alt = init_fn(n_alt, rng, **kwargs)
            c5_alt = compute_c5(h_alt)
            if c5_alt > best_c5 + 0.05:
                continue
            betas3 = [100, 500, 2000, 10000]
            tks3 = [10, 7, 5, 4]
            h_new, c5_new = do_slsqp_fast(h_alt, betas3, 2000, time_left, tks3)
            if c5_new < best_c5:
                best_h, best_c5, n_best = h_new, c5_new, n_alt
                break

    # Phase 5: Long gradient descent polish
    if time_left() > 40:
        tgt = len(best_h) / 2.0
        best_h, best_c5 = proj_grad_descent(
            best_h, tgt, 30000, time_left,
            top_k_init=4, top_k_max=20,
            step_init=0.02, step_min=0.0001,
            perturb_every=500, perturb_scale=0.005)

    # Phase 6: Final SLSQP polish
    if time_left() > 20:
        betas4 = [20000, 50000, 100000]
        tks4 = [4, 3, 3]
        h_new, c5_new = do_slsqp_fast(best_h, betas4, 5000, time_left, tks4)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final projection and validation
    tgt = len(best_h) / 2.0
    best_h = project_h(best_h, tgt)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    diff = tgt - best_h.sum()
    if abs(diff) > 1e-12:
        free = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if free.sum() > 0:
            best_h[free] += diff / free.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
