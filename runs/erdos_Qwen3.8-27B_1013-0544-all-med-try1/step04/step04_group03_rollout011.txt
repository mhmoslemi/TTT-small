        if lag < n:

            for i in range(n - lag):

            for i in range(lag, n):

        for i in range(n - la):

        for i in range(la, n):

    remaining = budget_s - (time.time() - t0)
    for n_hr in [256, 512]:
        if remaining < 50:
            break
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample_fn(best_h, n_hr), target_hr)
        for beta in [500, 2000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            cur_hr = do_slsqp(cur_hr, n_hr, target_hr, beta, maxiter=5000)
            cur_hr, c_try = targeted_reduction(cur_hr, target_hr, n_hr, t0, budget_s, max_iter=3000)
            if c_try < best_c5:
                best_h = cur_hr.copy()
                best_c5 = c_try
                best_n = n_hr
                n_cur = n_hr
                target = target_hr

    # Phase 7: Perturbation restarts
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 200)
    for att in range(5):
        if remaining < 30:
            break
        scale = 0.02 * (att + 1)
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp = do_slsqp(hp, n_cur, target, 2000, maxiter=4000)
        hp, c_try = targeted_reduction(hp, target, n_cur, t0, budget_s, max_iter=4000)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try
        remaining = budget_s - (time.time() - t0)

    # Phase 8: Final targeted + polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = targeted_reduction(best_h, target, n_cur, t0, budget_s, max_iter=8000)
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h = do_slsqp(best_h, n_cur, target, 20000, maxiter=6000, ftol=1e-16)
        best_c5 = compute_c5(best_h)
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coord_descent_fine(best_h, target, n_cur, 150, t0, budget_s)

    return best_h, best_c5, best_n
