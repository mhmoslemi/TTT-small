def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h_in, beta, n):
    dx = 2.0 / n
    h = np.clip(h_in, 0, 1)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    exps = np.exp(np.clip(beta * (c - m), -500, 500))
    sw = np.sum(exps)
    weights = exps / sw
    obj = (m + np.log(sw / len(c)) / beta) * dx
    grad = np.zeros(n)
    for i in range(len(c)):
        lag = i - (n - 1)
        w = weights[i] * dx
        if lag >= 0:

            grad[lag:] += -w * h[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] += -w * h[la:]
    return obj, grad
