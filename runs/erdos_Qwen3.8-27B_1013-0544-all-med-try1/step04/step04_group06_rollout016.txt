        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])

        bc = beta * (tv - mm)
        e = np.exp(bc)
        se = e.sum()

            g += w[j] * grad_lag(xc, lag, n)

    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
