from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]

- Maintain a distribution over the active lags
- At each step, update h to reduce the weighted average of the active lags
- Also update the distribution to put more weight on the currently worst lags

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_val_grad(h, beta):
    """Smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(x)
    s = e.sum()
    w = e / s
    val = (m + np.log(s) / beta) * dx
    # Gradient
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        wi = w[idx] * dx
        lag = idx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:n-lag] -= wi * h[lag:]
        else:
            m2 = -lag
            grad[:n-m2] += wi * (1.0 - h[m2:])
            grad[m2:] -= wi * h[:n-m2]
    return val, grad

def lbfgsb_optimize(h0, target, beta, maxiter, time_fn):
    """L-BFGS-B with penalty for sum constraint."""
    n = len(h0)
    lam = 1000.0  # penalty parameter
    
    def obj(x):
        v, g = smax_val_grad(x, beta)
        s = (x.sum() - target)
        v += lam * s * s
        g = g + 2.0 * lam * s * np.ones(n)
        return v, g
    
    res = minimize(obj, h0, method='L-BFGS-B', jac=True,
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    h_new = project_h(res.x, target)
    return h_new, compute_c5(h_new)

def slsqp_optimize(h0, target, beta, maxiter, time_fn):
    """SLSQP with exact sum constraint."""
    n = len(h0)
    
    def obj(x):
        v, _ = smax_val_grad(x, beta)
        return v
    
    def grad(x):
        _, g = smax_val_grad(x, beta)
        return g
    
    def con(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=[(0,1)]*n,
                   constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = project_h(res.x, target)
    return h_new, compute_c5(h_new)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def expand_sym(h_half, n):
    h = np.empty(n)
    hh = len(h_half)
    h[:hh] = h_half
    h[hh:] = 1.0 - h_half[::-1]
    return h

def smax_full(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(x)
    s = e.sum()
    w = e / s
    val = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        wi = w[idx] * dx
        lag = idx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += wi * (1.0 - h[m2:])
            grad[m2:] -= wi * h[:n - m2]
    return val, grad

    val, gf = smax_full(h, beta)
    hh = len(h_half)
    gh = gf[:hh] - gf[hh:][::-1]

        def cj(x):
            return np.ones(n_f)

        try:
            res = minimize(of, best_h, method='SLSQP', jac=gf_,
                           bounds=[(0, 1)] * n_f,
                           constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                           options={'maxiter': 6000, 'ftol': 1e-15})
            h_new = project_h(res.x, tgt_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
