def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = int(round(target - z.sum()))
    if diff > 0:
        zeros = np.where(z == 0)[0]
        z[zeros[np.argsort(h[zeros])[::-1]][:diff]] = 1.0
    elif diff < 0:
        ones = np.where(z == 1)[0]
        z[ones[np.argsort(h[ones])[:diff]]] = 0.0
    return z

def swap_delta(corr, h, a, b, n):
    """Compute the change in corr for each lag when swapping a(1->0) and b(0->1)."""
    # Returns the full delta array
    delta = np.zeros(2*n-1)
    # a as source: 1->0
    d_start = max(0, -(a))  # d such that 0 <= a+d < n, i.e., -a <= d < n-a
    d_end = min(n-1, n-1-a)
    for d in range(d_start, d_end+1):
        m = d + n - 1
        delta[m] += h[a+d] - 1.0  # remove (1-h[a+d]) = h[a+d]-1
    # b as source: 0->1
    d_start = max(-(b), -(n-1))
    d_end = min(n-1-b, n-1)
    for d in range(d_start, d_end+1):
        m = d + n - 1
        delta[m] += 1.0 - h[b+d]
    # a as target: (1-h[a]) 0->1
    d_start = max(-(n-1), a-(n-1))  # d such that 0 <= a-d < n, i.e., a-n+1 <= d <= a
    d_end = min(a, n-1)
    for d in range(d_start, d_end+1):
        m = d + n - 1
        i = a - d
        if h[i] == 1:
            delta[m] += 1.0
    # b as target: (1-h[b]) 1->0
    d_start = max(-(n-1), b-(n-1))
    d_end = min(b, n-1)
    for d in range(d_start, d_end+1):
        m = d + n - 1
        i = b - d
        if h[i] == 1:
            delta[m] -= 1.0
    return delta

def swap_delta_vec(h, a, b, n):
    """Vectorized computation of correlation change for swap (a: 1->0, b: 0->1)."""
    delta = np.zeros(2*n-1)
    
    # a as source (1->0): for d where 0<=a+d<n, corr[n-1+d] -= (1-h[a+d])
    # d ranges from max(0,-a) to min(n-1-a, n-1)
    d_min_a = max(0, -a)
    d_max_a = min(n-1-a, n-1)
    if d_min_a <= d_max_a:
        m_idx = np.arange(d_min_a, d_max_a+1) + n - 1
        targets = a + np.arange(d_min_a, d_max_a+1)
        delta[m_idx] -= (1.0 - h[targets])
    
    # b as source (0->1): for d where 0<=b+d<n, corr[n-1+d] += (1-h[b+d])
    d_min_b = max(0, -b)
    d_max_b = min(n-1-b, n-1)
    if d_min_b <= d_max_b:
        m_idx = np.arange(d_min_b, d_max_b+1) + n - 1
        targets = b + np.arange(d_min_b, d_max_b+1)
        delta[m_idx] += (1.0 - h[targets])
    
    # a as target: (1-h[a]) goes 0->1. For d where 0<=a-d<n (i.e., a-n+1<=d<=a),
    # corr[n-1+d] += h[a-d] (the source position a-d)
    d_min_t = max(0-(n-1)+1, a-(n-1))  # d >= -(n-1), and a-d >= 0, so d <= a
    d_max_t = min(a, n-1)
    d_min_t = max(d_min_t, a-(n-1))
    if d_min_t <= d_max_t:
        ds = np.arange(d_min_t, d_max_t+1)
        m_idx = ds + n - 1
        sources = a - ds
        delta[m_idx] += h[sources]
    
    # b as target: (1-h[b]) goes 1->0. For d where 0<=b-d<n,
    # corr[n-1+d] -= h[b-d]
    d_min_t = max(b-(n-1), -(n-1))
    d_max_t = min(b, n-1)
    if d_min_t <= d_max_t:
        ds = np.arange(d_min_t, d_max_t+1)
        m_idx = ds + n - 1
        sources = b - ds
        delta[m_idx] -= h[sources]
    
    return delta
