import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_top(x, indices, beta):
    """Smooth max of x at given indices"""
    vals = x[indices]
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -400, 400)
    return m + np.log(np.sum(np.exp(shifted)) / len(vals)) / beta

def active_set_slsqp(h, target, dx, n_active, beta, max_iter, time_fn):
    """SLSQP optimizing smooth max of top n_active lags"""
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    
    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max_top(c, top_idx, beta) * dx
    
    def con_fn(h_in):
        return h_in.sum() - target
    
    def jac_fn(h_in):
        return np.ones(n)
    
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new
    except Exception:
        return h

def smooth_max_and_grad(h, active_indices, beta, dx):
    """Compute smooth max of correlations at active_indices and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr[active_indices] * dx
    m = np.max(vals)
    shifted = np.clip(beta * (vals - m), -400, 400)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    sm = m + np.log(Z) / beta
    
    # Gradient
    grad = np.zeros(n)
    weights = exps / Z
    for wi, idx in zip(weights, active_indices):
        d = int(idx) - (n - 1)
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            # c(d) = sum_i h[i]*(1-h[i-d])
            # dc/dh[j] = (1-h[j-d]) [j-d>=0] - h[j+d] [j+d<n]
            grad[:n-d] += wi * (1.0 - h[d:])   # (1-h[j-d]) for j>=d
            grad[d:] -= wi * h[:n-d]             # -h[j+d] for j<n-d
        else:
            d2 = -d
            # c(-d2) = sum_i h[i]*(1-h[i+d2])
            # dc/dh[j] = (1-h[j+d2]) [j+d2<n] - h[j-d2] [j-d2>=0]
            grad[:n-d2] += wi * (1.0 - h[d2:])   # (1-h[j+d2]) for j<n-d2
            grad[d2:] -= wi * h[:n-d2]           # -h[j-d2] for j>=d2
    
    return sm, grad

    mv = np.max(vals)
    shifted = np.clip(beta * (vals - mv), -500, 500)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    obj_val = mv + np.log(Z) / beta
    weights = exps / Z
    grad = np.zeros(n)
    for m in range(M):
        wi = weights[m]
        d = int(active_idx[m]) - (n - 1)
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:n-d]
        else:
            d2 = -d
            grad[d2:] += wi * (1.0 - h[:n-d2])
            grad[:n-d2] -= wi * h[d2:]
    return obj_val, grad

def slsqp_call(h, target, dx, active_idx, beta, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    def fun(x):
        v, _ = active_obj_grad(x, active_idx, beta, dx)
        return v

    def jac(x):
        _, g = active_obj_grad(x, active_idx, beta, dx)
        return g

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-20})
        if hasattr(res, 'x') and res.x is not None:
            return project_h(res.x, target)
    except Exception:
        pass
    return h

def optimize_resolution(h, target, dx, time_fn):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)

    # Stage 1: Full active set, increasing beta
    for beta in [30, 100, 400, 1500, 5000, 20000]:
        if time_fn() < 25:
            break
        na = min(300, 2*n-2)
        active = get_active_lags(best, na)
        h_new = slsqp_call(best, target, dx, active, beta, 4000, time_fn)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new

    # Stage 2: Narrow active set, high beta
    for na in [15, 30, 60, 120, 250]:
        if time_fn() < 15:
            break
        if na >= 2*n-2:
            na = 2*n-2
        active = get_active_lags(best, na)
        for beta in [10000, 50000, 150000]:
            if time_fn() < 10:
                break
            h_new = slsqp_call(best, target, dx, active, beta, 3000, time_fn)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at original resolution
    n0 = len(best_h)
    best_h, best_c5 = optimize_resolution(best_h, n0/2.0, 2.0/n0, rem)

    # Phase 2: Try other resolutions
    for n_try in [300, 500, 600, 800]:
        if rem() < 100:
            break
        target_t = n_try / 2.0
        dx_t = 2.0 / n_try
        x_src = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_t = project_h(np.interp(x_dst, x_src, best_h), target_t)
        h_t, c5_t = optimize_resolution(h_t, target_t, dx_t, rem)
        if c5_t < best_c5:
            best_h = h_t
            best_c5 = c5_t

    # Phase 3: Perturbation + re-optimization
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    dx_cur = 2.0 / n_cur

    for i in range(50):
        if rem() < 45:
            break
        scale = 0.015 + 0.008 * (i % 10)
        h_pert = best_h + np.random.randn(n_cur) * scale
        h_pert = project_h(h_pert, target_cur)
        h_pert, c5_pert = optimize_resolution(h_pert, target_cur, dx_cur, rem)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert

    # Phase 4: Final high-precision polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f

    for beta in [50000, 100000, 200000, 500000]:
        if rem() < 12:
            break
        active = get_active_lags(best_h, 20)
        best_h = slsqp_call(best_h, target_f, dx_f, active, beta, 5000, rem)
        best_c5 = compute_c5(best_h)

    # Phase 5: One more full sweep at best resolution
    if rem() > 60:
        best_h, best_c5 = optimize_resolution(best_h, target_f, dx_f, rem)

    return best_h, best_c5, len(best_h)
