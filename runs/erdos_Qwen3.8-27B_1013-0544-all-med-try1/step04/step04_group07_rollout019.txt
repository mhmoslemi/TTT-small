import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_h(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0: free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new: return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def sa_optimize(h0, n, target, t0, budget, seed, n_iter=200000):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    blocks = [1, 2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]
    T = 0.015
    T_min = 1e-10
    cool = 1.0 - 2.0 / max(n_iter, 1)  # reaches T_min near end
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3: break
        mt = rng.random()
        b = blocks[rng.randint(len(blocks))]
        if b > n // 2: b = max(1, n // 4)
        if mt < 0.35:
            # Block swap
            if 2 * b > n: continue
            i = rng.randint(0, n - b)
            j = rng.randint(0, n - b)
            if abs(i - j) < b: continue
            trial = h.copy()
            trial[i:i+b], trial[j:j+b] = trial[j:j+b].copy(), trial[i:i+b].copy()
            c = compute_c5(trial)
        elif mt < 0.6:
            # Block complement
            i = rng.randint(0, n - b + 1)
            trial = h.copy()
            trial[i:i+b] = 1.0 - trial[i:i+b]
            trial = project_h(trial, target)
            c = compute_c5(trial)
        elif mt < 0.85:
            # Block perturb
            i = rng.randint(0, n - b + 1)
            trial = h.copy()
            trial[i:i+b] += rng.uniform(-0.15, 0.15, size=b)

            c = compute_c5(trial)

            j = rng.randint(n)
            trial[j] += rng.choice([-0.5, -0.2, 0.2, 0.5])
            trial = project_h(trial, target)
            c = compute_c5(trial)

            if c < best_c5:

                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        T = max(T * cool, T_min)
    return best_h, best_c5

def slsqp_refine(h0, n, target, t0, budget, beta=2000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(np.clip(beta*(c-m), -500, 500)))) / len(c)) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
        return project_h(r.x, target)
    except:
        return h0

def coord_descent(h0, target, n, t0, budget, n_sweeps=100):
    h = h0.copy()
    best = compute_c5(h)
    steps = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 1: break
        improved = False
        for j in range(n):
            for s in steps:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + s, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best - 1e-16:
                    h, best, improved = trial, c, True
                    break
        if not improved: break
    return h, best

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_h(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0: free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new: return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def sa_optimize(h0, n, target, t0, budget, seed=42, n_iter=500000):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    # Block sizes: mix of small and large
    blocks = [1, 2, 3, 4, 6, 8, 12, 16, 20, 24, 32, 40, 48, 64]
    blocks = [b for b in blocks if b <= n // 3]
    if not blocks: blocks = [1]
    T = 0.01
    T_min = 1e-12
    # Geometric cooling: T * cool^n_iter = T_min => cool = (T_min/T)^(1/n_iter)
    cool = (T_min / T) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() - t0 > budget - 2: break
        r = rng.random()
        b = blocks[rng.randint(len(blocks))]
        if r < 0.4:
            if 2 * b > n: continue
            i = rng.randint(0, n - b)
            j = rng.randint(0, n - b)
            if abs(i - j) < b: continue
            trial = h.copy()
            trial[i:i+b], trial[j:j+b] = trial[j:j+b].copy(), trial[i:i+b].copy()
        elif r < 0.65:
            i = rng.randint(0, n - b + 1)
            trial = h.copy()
            trial[i:i+b] = 1.0 - trial[i:i+b]
            trial = project_h(trial, target)
        elif r < 0.85:
            i = rng.randint(0, n - b + 1)
            trial = h.copy()
            trial[i:i+b] += rng.uniform(-0.1, 0.1, size=b)
            trial = project_h(trial, target)
        else:
            trial = h.copy()
            j = rng.randint(n)
            trial[j] += rng.choice([-0.5, -0.25, -0.1, 0.1, 0.25, 0.5])
            trial = project_h(trial, target)
        
        c = compute_c5(trial)
        delta = c - best_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            h = trial
            if c < best_c5:
                best_h, best_c5 = h.copy(), c
        T = max(T * cool, T_min)
    return best_h, best_c5

def slsqp_refine(h0, n, target, t0, budget, beta=5000, maxiter=2000):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(np.clip(beta*(c-m), -500, 500)))) / len(c)) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except:
        return h0

def fine_coord_descent(h0, target, n, t0, budget):
    h = h0.copy()
    best = compute_c5(h)
    # Adaptive step sizes
    for scale in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t0 > budget - 1: break
        for _ in range(3):  # 3 sweeps per scale
            if time.time() - t0 > budget - 1: break
            improved = False
            for j in range(n):
                for s in [scale, -scale]:
                    trial = h.copy()
                    trial[j] = np.clip(trial[j] + s, 0, 1)
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best - 1e-16:
                        h, best, improved = trial, c, True
                        break
                if time.time() - t0 > budget - 0.5: break
            if not improved: break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):

        if cur_c5 < best_c5:
