def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] = 1.0
        g[lag:] -= h[:n-lag]
        g[:n-lag] -= h[lag:]

        g[m:] = 1.0
        g[m:] -= h[:n-m]

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    """One active set LP iteration. Returns new h and success flag."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:m_active]
    lags = [int(i) - (n - 1) for i in top_idx]
    
    # Build LP
    # Variables: [h_0, ..., h_{n-1}, t] (n+1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for lag in lags:
        g = compute_grad_at_lag(h, lag)
        idx = lag + (n - 1)
        f_val = corr[idx]
        # Constraint: dx * g^T h - t <= dx * (g^T h* - f_val)

        row[:n] = dx * g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(dx * (np.dot(g, h) - f_val))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

            return h_new, True, res.x[-1]

        pass

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] = 1.0
        g[lag:] -= h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[m:] = 1.0
        g[m:] -= h[:n - m]
        g[:n - m] -= h[m:]
    return g

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def lp_solve(h, target, dx, m_active, threshold=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if threshold is not None:
        max_c = np.max(corr)
        active_mask = corr >= (max_c - threshold)
        if active_mask.sum() > m_active:
            top_idx = np.argsort(corr)[::-1][:m_active]
            active_mask = np.zeros(len(corr), dtype=bool)
            active_mask[top_idx] = True
    else:
        top_idx = np.argsort(corr)[::-1][:m_active]
        active_mask = np.zeros(len(corr), dtype=bool)
        active_mask[top_idx] = True
    lags = [int(i) - (n - 1) for i in np.where(active_mask)[0]]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for lag in lags:
        g = compute_grad_at_lag(h, lag)
        idx = lag + (n - 1)
        f_val = corr[idx]
        row = np.zeros(n + 1)
        row[:n] = dx * g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(dx * (np.dot(g, h) - f_val))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def lp_phase(h, target, time_fn, max_iters=2000, m_active=100):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 5:
            break
        h_lp, t_lp = lp_solve(current, target, dx, m_active)
        if h_lp is None:
            break
        h_lp = project_h(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < best_c5 - 1e-14:
            best_h = h_lp
            best_c5 = c5_lp
            no_improve = 0
        else:
            no_improve += 1
        
        current = 0.7 * h_lp + 0.3 * current
        current = project_h(current, target)

                no_improve = 0

                m_active = max(80, m_active - 20)
                no_improve = 0
            if no_improve > 200:
                break
    return best_h, best_c5

def subgrad_optimize(h, target, time_fn, max_iters=15000, m_factor=15,
                     step_list=None, perturb_every=250, perturb_scale=0.004, momentum=0.88):
    if step_list is None:
        step_list = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_factor, max(3, n // 10))
        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = np.zeros(n)
        for lag in lags:
            g += compute_grad_at_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and no_improve % perturb_every == 0:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                if compute_c5(ht) < best_c5:
                    best = ht
                    best_c5 = compute_c5(best)
                    vel *= 0.3
                    no_improve = 0
            if no_improve > perturb_every * 12:
                break
    return best, best_c5

def slsqqpolish(h, target, time_fn, beta, max_iter=8000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    def con(h_in):
        return h_in.sum() - target
    def jac(h_in):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                      options={'maxiter': max_iter, 'ftol': 1e-20})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
