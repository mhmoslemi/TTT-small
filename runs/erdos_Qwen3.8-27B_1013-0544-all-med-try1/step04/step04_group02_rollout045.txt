import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_optimize(h, n, deadline, seed, betas=None):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    if betas is None:
        betas = [50, 200, 1000, 5000, 20000, 80000]
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.003, 0.01]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 4000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def subgrad_optimize(h, n, deadline, seed, top_k=5, temp=15.0, lr0=0.006, momentum0=0.85):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = lr0
    mom = momentum0
    stagnant = 0
    it = 0
    while time.time() < deadline:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(temp * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * subgrad_for_lag(h, lag, n)
        vel = mom * vel - lr * grad
        h_new = project_feasible(h + vel, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.003, 0.012)
            mom = min(mom + 0.0005, 0.92)
        else:
            stagnant += 1
            vel *= 0.5
            lr *= 0.92
            mom = max(mom - 0.01, 0.7)
            if stagnant % 200 == 0:
                top_k = min(top_k + 1, 12)
                temp *= 1.3
            if stagnant > 400 and lr < 3e-5:
                h = project_feasible(h + np.random.randn(n) * 0.005, ts)
                c5 = compute_c5(h)
                stagnant = 0
                lr = lr0 * 0.5
                top_k = 5
                temp = 15.0
    return h, c5

def interp_to_resolution(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: SLSQP from parent (30%)
    d1 = t0 + budget_s * 0.30
    h1, c1 = slsqp_optimize(parent.copy(), pn, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Try reversed
    if time.time() < d1:
        h_rev = parent[::-1].copy()
        h_rev = project_feasible(h_rev, pn / 2.0)
        c_rev = compute_c5(h_rev)
        if c_rev < best_c5:
            h2, c2 = slsqp_optimize(h_rev, pn, d1, seed + 7)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c2, pn

    # Phase 2: Subgradient polish (40%)
    d2 = t0 + budget_s * 0.70
    if best_h is not None:
        best_h, best_c5 = subgrad_optimize(best_h, best_n, d2, seed + 100)

    # Phase 2b: SLSQP from subgradient result
    if time.time() < d2:
        h3, c3 = slsqp_optimize(best_h.copy(), best_n, d2, seed + 200,
                                betas=[20000, 80000, 200000])
        if c3 < best_c5:
            best_h, best_c5, best_n = h3.copy(), c3, best_n

    # Phase 3: Multi-resolution (18%)
    d3 = t0 + budget_s * 0.88
    candidates = []
    for delta in [-40, -20, 20, 40, -60, 60]:
        nr = best_n + delta
        if 150 <= nr <= 600 and nr != best_n:
            candidates.append(nr)
    for nr in [250, 300, 400, 450]:
        if nr != best_n and nr not in candidates:
            candidates.append(nr)
    candidates = sorted(set(candidates))

    for nr in candidates:
        if time.time() > d3:
            break
        h0 = interp_to_resolution(best_h, nr)
        h0 = project_feasible(h0, nr / 2.0)
        c5_n = compute_c5(h0)
        h0, c5_n = slsqp_optimize(h0, nr, d3, seed + nr, betas=[500, 5000, 50000])
        h0, c5_n = subgrad_optimize(h0, nr, min(d3, time.time() + 30), seed + nr + 50)
        if c5_n < best_c5:
            best_h, best_c5, best_n = h0.copy(), c5_n, nr

    # Phase 4: Final polish (12%)
    d4 = t0 + budget_s * 0.98
    if best_h is not None and time.time() < d4:
        # Very careful subgradient
        best_h, best_c5 = subgrad_optimize(best_h, best_n, d4, seed + 900,
                                           top_k=8, temp=30.0, lr0=0.002, momentum0=0.80)
        # SLSQP with very high beta
        if time.time() < d4:
            h4, c4 = slsqp_optimize(best_h, best_n, d4, seed + 950,
                                    betas=[100000, 500000])
            if c4 < best_c5:
                best_h, best_c5 = h4, c4

    # Final check
    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
