from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def weighted_subgrad(h, corr, m_lags, w_beta):
    n = len(h)
    top_idx = np.argsort(corr)[::-1][:m_lags]
    lags = (top_idx - (n - 1)).astype(int)
    diffs = corr[top_idx[0]] - corr[top_idx]
    weights = np.exp(-w_beta * diffs)
    weights /= weights.sum()
    grad = np.zeros(n)
    for i in range(len(lags)):
        w = weights[i]
        lag = lags[i]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_quick(h, target, time_fn, betas, max_iter):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 10:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def opt_pass(h, target, time_fn, max_iters, m_lags, w_beta, step0, decay_rate,
             mu_start, mu_end, perturb_every, perturb_scale, use_nesterov):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    x = h.copy()
    vel = np.zeros(n)
    no_improve = 0
    rng = np.random.RandomState(int(best_c5 * 1e6) % 100000)
    for it in range(max_iters):
        if time_fn() < 3:
            break
        frac = it / max(1, max_iters)
        mu = mu_start + (mu_end - mu_start) * frac
        step = step0 * (decay_rate ** (it / 300.0))
        if step < 1e-5:
            step = 1e-5
        if use_nesterov:
            y = x + mu * vel
            y = np.clip(y, 0.0, 1.0)
        else:
            y = x
        corr = np.correlate(y, 1.0 - y, mode="full")
        g = weighted_subgrad(y, corr, m_lags, w_beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        found = False
        s = step
        for _ in range(10):
            x_new = project_h(y - s * g, target)
            c5_new = compute_c5(x_new)
            if c5_new < best_c5 - 1e-16:
                if use_nesterov:
                    vel = (x_new - x) / (1.0 + mu)
                else:
                    vel = 0.9 * vel + 0.1 * (x_new - x)
                x = x_new
                best = x_new.copy()
                best_c5 = c5_new
                no_improve = 0
                found = True
                break
            s *= 0.6
        if not found:
            no_improve += 1
            if no_improve >= perturb_every:
                scale = perturb_scale * (0.5 + 0.5 * rng.rand())
                p = rng.randn(n) * scale
                x_p = project_h(x + p, target)
                c5_p = compute_c5(x_p)
                if c5_p < best_c5:
                    x = x_p
                    best = x_p.copy()
                    best_c5 = c5_p
                    vel = np.zeros(n)
                    no_improve = 0
            if no_improve > perturb_every * 20:
                break
    return best, best_c5

def interpolate_resize(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 400
        best_h = np.full(n, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP cleanup
    best_h, best_c5 = slsqp_quick(best_h, target, rem, [200, 800, 3000], 2500)

    # Phase 2: Main subgradient pass (weighted, Nesterov, many lags)
    best_h, best_c5 = opt_pass(
        best_h, target, rem,
        max_iters=50000, m_lags=35, w_beta=50,
        step0=0.018, decay_rate=0.65,
        mu_start=0.70, mu_end=0.92,
        perturb_every=400, perturb_scale=0.007,
        use_nesterov=True
    )

    # Phase 3: Second pass with more lags and slower decay
    if rem() > 120:
        h3, c5_3 = opt_pass(
            best_h, target, rem,
            max_iters=40000, m_lags=50, w_beta=100,
            step0=0.008, decay_rate=0.7,
            mu_start=0.85, mu_end=0.95,
            perturb_every=500, perturb_scale=0.004,
            use_nesterov=True
        )
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Third pass - non-Nesterov with moderate momentum
    if rem() > 100:
        h4, c5_4 = opt_pass(
            best_h, target, rem,
            max_iters=35000, m_lags=40, w_beta=30,
            step0=0.012, decay_rate=0.68,
            mu_start=0.50, mu_end=0.80,
            perturb_every=350, perturb_scale=0.005,
            use_nesterov=False
        )
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Try resolution increase to n+200
    n_cur = len(best_h)
    if rem() > 180:
        n2 = n_cur + 200
        if n2 % 2 == 1:
            n2 += 1
        target2 = n2 / 2.0
        h2 = project_h(interpolate_resize(best_h, n2), target2)
        c5_2 = compute_c5(h2)
        if c5_2 <= best_c5 + 0.001:
            if rem() > 60:
                h2, c5_2 = slsqp_quick(h2, target2, rem, [300, 1500], 2000)
            if rem() > 40:
                h2, c5_2 = opt_pass(
                    h2, target2, rem,
                    max_iters=25000, m_lags=45, w_beta=60,
                    step0=0.008, decay_rate=0.65,
                    mu_start=0.80, mu_end=0.93,
                    perturb_every=400, perturb_scale=0.004,
                    use_nesterov=True
                )
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2
                n_cur = n2
                target = target2

    # Phase 6: Final fine-tuning with very small steps
    n_f = len(best_h)
    target_f = n_f / 2.0
    if rem() > 15:
        best_h, best_c5 = opt_pass(
            best_h, target_f, rem,
            max_iters=25000, m_lags=55, w_beta=200,
            step0=0.003, decay_rate=0.55,
            mu_start=0.90, mu_end=0.96,
            perturb_every=600, perturb_scale=0.002,
            use_nesterov=True
        )

    # Final validation
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
