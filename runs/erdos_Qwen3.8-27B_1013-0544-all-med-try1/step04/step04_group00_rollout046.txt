```python
def compute_autocorr_fft(h):
    """Compute linear autocorrelation using FFT."""
    n = len(h)
    size = 2 * n
    H = np.fft.rfft(h, size)
    acorr = np.fft.irfft(H * np.conj(H), size)
    # acorr[0:n] gives autocorrelation at lags 0, 1, ..., n-1
    # acorr[n:] gives autocorrelation at lags -(n-1), ..., -1 (by symmetry)
    # Full: lags -(n-1) to (n-1)
    full = np.concatenate([acorr[1:][::-1], acorr[:n]])
    return full  # full[k-(n-1)] = A(k) for k = -(n-1), ..., n-1
```

def slsqp_soft_max(h, n, alpha, maxiter=500, penalty=1e6):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_and_grad(x):
        hh = np.clip(x, 0.0, 1.0)
        s = hh.sum()
        g = np.correlate(hh, 1 - hh, mode='full') * dx
        mg = g.max()
        sh = alpha * (g - mg)
        ms = sh.max()
        p = np.exp(sh - ms)
        p /= p.sum()
        obj_val = mg + (ms + np.log(p.sum())) / alpha + penalty * (s - target) ** 2

                grad += w * grad_g_at_tau(hh, n, tau)
