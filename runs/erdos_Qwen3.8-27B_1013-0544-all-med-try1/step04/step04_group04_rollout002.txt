            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            m2 = -lag
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
    return grad

def _obj(x):
    n = _g_n[0]
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip(_g_beta[0] * (corr - m), -500, 0)
    val = (m + np.log(np.sum(np.exp(sh))) / _g_beta[0]) * dx
    if _g_penalty[0] > 0:
        val += _g_penalty[0] * (x.sum() - _g_target[0]) ** 2
    return val

def _grad(x):
    corr, weights, m = _corr_weights(x, _g_beta[0])
    g = _grad_weights(x, weights)
    if _g_penalty[0] > 0:
        g += 2.0 * _g_penalty[0] * (x.sum() - _g_target[0])
    return g

def _con_eq(x):
    return x.sum() - _g_target[0]

def _con_eq_jac(x):
    return np.ones(_g_n[0])

def do_slsqp(h0, beta, maxiter):
    n = len(h0)
    _g_beta[0] = beta
    _g_n[0] = n
    _g_target[0] = n / 2.0
    _g_penalty[0] = 0.0
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h0, method='SLSQP', jac=_grad,
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def do_lbfgsb(h0, beta, penalty, maxiter):
    n = len(h0)
    _g_beta[0] = beta
    _g_n[0] = n
    _g_target[0] = n / 2.0
    _g_penalty[0] = penalty
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h0, method='L-BFGS-B', jac=_grad,
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return res.x

def do_momentum(h0, n_iter, time_fn, beta=50000):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.85
    base_step = 0.008
    c5_cur = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 5:
            break
        _, weights, _ = _corr_weights(h, beta)
        g = _grad_weights(h, weights)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = mu * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break
        v = v / vn
        alpha = base_step * min(1.0, 10.0 / (it + 1))
        found = False
        for trial in range(8):
            ht = project_h(h - alpha * v, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-16:
                h = ht
                c5_cur = c5t
                found = True
                no_imp = 0
                break
            alpha *= 0.5
        if not found:
            no_imp += 1
            if no_imp > 500:
                sc = 0.003 * min(3, no_imp // 500 + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                c5t = compute_c5(ht)
                if c5t < c5_cur:
                    h = ht
                    c5_cur = c5t
                    no_imp = 0
                else:
                    break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(500, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup with moderate beta
    for beta in [200, 1000, 5000]:
        if tl() < 40:
            break
        h_new = do_slsqp(best_h, beta, maxiter=2000)
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: L-BFGS-B with increasing beta and penalty
    for beta in [5000, 20000, 50000, 100000, 300000]:
        if tl() < 60:
            break
        for pen in [100, 1000, 10000]:
            if tl() < 30:
                break
            h_new = do_lbfgsb(best_h, beta, pen, maxiter=3000)
            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 3: Momentum projected gradient
    if tl() > 100:
        h_new, c5_new = do_momentum(best_h, 60000, tl, beta=50000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Perturbation + re-optimize
    for trial in range(8):
        if tl() < 60:
            break
        scale = 0.002 * (1 + trial * 0.5)
        h_pert = project_h(best_h + np.random.randn(n) * scale, target)
        c5_pert = compute_c5(h_pert)
        h_new = do_lbfgsb(h_pert, 50000, 10000, maxiter=2000)
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Final SLSQP polish with very high beta
    if tl() > 30:
        for beta in [100000, 500000]:
            if tl() < 20:
                break
            h_new = do_slsqp(best_h, beta, maxiter=3000)
            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Ultra-fine L-BFGS-B
    if tl() > 20:
        for beta in [500000, 1000000]:
            if tl() < 15:
                break
            h_new = do_lbfgsb(best_h, beta, 100000, maxiter=5000)
            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
