import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += (1.0 - h[m2:])
        g[m2:] -= h[:n - m2]
    return g

def slsqp_softmax(h_init, n, beta, maxiter=2000, ftol=1e-14):
    target = n / 2.0
    bounds = [(0, 1)] * n
    dx = 2.0 / n

    def obj_func(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        return (m + np.log(e.sum() / len(corr)) / beta) * dx

    def grad_func(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        nc = len(corr)
        for j in range(nc):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            g += w[j] * lag_grad(h, lag, n)
        return g * dx

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(obj_func, h_init, jac=grad_func, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def subgrad_momentum(h, target, t0, budget, max_iters=30000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    momentum = np.zeros(n)
    step = 0.025
    n_active = 5
    no_improve = 0

    for it in range(max_iters):
        if budget - (time.time() - t0) < 5:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m = np.max(corr)
        k = min(n_active, len(corr))
        top = np.argsort(corr)[::-1][:k]
        # Soft-max weights over active lags
        diffs = m - corr[top]
        weights = np.exp(-30.0 * diffs)
        ws = weights.sum()
        g = np.zeros(n)
        for i, idx in enumerate(top):
            lag = int(idx) - (n - 1)
            g += (weights[i] / ws) * lag_grad(best, lag, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        # Momentum
        momentum = 0.82 * momentum - step * g
        ht = project_sum(best + momentum, target)
        c5t = compute_c5(ht)
        if c5t < best_c5 - 1e-15:
            best = ht
            best_c5 = c5t
            no_improve = 0
            if it % 2000 == 1999 and n_active < 18:
                n_active += 1
        else:
            no_improve += 1
            if no_improve % 150 == 149:
                step *= 0.75
                momentum *= 0.4
            if no_improve > 1500 and it % 400 == 399:
                scale = 0.003 + 0.003 * (no_improve // 2000)
                pert = np.random.randn(n) * scale
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                    momentum = np.zeros(n)
                    step = 0.015
            if no_improve > 8000:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP with gradual beta increase at initial resolution
    betas = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 15000]
    for beta in betas:
        if tl() < 120:
            break
        maxit = 2500 if beta <= 500 else 1500
        h_new = slsqp_softmax(best_h, n, beta, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_momentum(best_h, target, t0, budget_s, max_iters=1500)

    # Phase 2: Intensive subgradient polishing at current resolution
    if tl() > 150:
        best_h, best_c5 = subgrad_momentum(best_h, target, t0, budget_s, max_iters=60000)

    # Phase 3: Try increasing resolution
    for n_try in [n + 80, n + 160]:
        if tl() < 250:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        for beta in [100, 500, 2000, 8000]:
            if tl() < 120:
                break
            h_new = slsqp_softmax(h_interp, n_try, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n = n_try
                target = n_try / 2.0
            h_interp = h_new
        if tl() > 100:
            best_h, best_c5 = subgrad_momentum(best_h, target, t0, budget_s, max_iters=30000)

    # Phase 4: Multi-start at best resolution
    n_best = len(best_h)
    tgt_best = n_best / 2.0
    starts = [
        best_h.copy(),
        project_sum(best_h + np.random.randn(n_best) * 0.03, tgt_best),
        project_sum(best_h + np.random.randn(n_best) * 0.08, tgt_best),
    ]
    # Add a structured perturbation
    flip = n_best // 6
    h_struct = best_h.copy()
    h_struct[:flip] = 1.0 - h_struct[:flip]
    starts.append(project_sum(h_struct, tgt_best))

    for h_start in starts:
        if tl() < 100:
            break
        for beta in [300, 2000]:
            if tl() < 60:
                break
            h_new = slsqp_softmax(h_start, n_best, beta, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_start = h_new
        if tl() > 60:
            best_h, best_c5 = subgrad_momentum(best_h, tgt_best, t0, budget_s, max_iters=10000)

    # Phase 5: Final high-beta SLSQP
    if tl() > 40:
        h_new = slsqp_softmax(best_h, len(best_h), 50000, maxiter=4000, ftol=1e-16)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Final micro-polish with very small steps
    n_f = len(best_h)
    tgt_f = n_f / 2.0
    for it in range(20000):
        if tl() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        m = np.max(corr)
        top = np.argsort(corr)[::-1][:6]
        g = np.zeros(n_f)
        for idx in top:
            lag = int(idx) - (n_f - 1)
            g += lag_grad(best_h, lag, n_f)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for s in [0.005, 0.002, 0.001, 0.0004, 0.0001, 0.00003]:
            ht = project_sum(best_h - s * g, tgt_f)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                improved = True
                break
        if not improved and it > 5000:
            break

    return best_h, best_c5, len(best_h)
