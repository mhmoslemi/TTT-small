def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            # Spread among all
            h += diff / len(h)
            h = np.clip(h, 0, 1)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def weighted_subgrad(h, n, top_frac=0.15):
    """Subgradient using top fraction of lags with exponential weighting."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    # Weight by how close each lag is to the max
    weights = np.exp(-30 * (mc - corr) / max(mc, 1e-10))
    # Normalize
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = (corr == mc).astype(float)
        wsum = weights.sum()
    weights /= wsum
    g = np.zeros(n)
    for i in range(len(corr)):
        if weights[i] < 1e-10:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                g[:n-lag] += weights[i] * (1.0 - h[lag:])
                g[lag:] -= weights[i] * h[:n-lag]
        else:
            la = -lag
            if la < n:
                g[la:] += weights[i] * (1.0 - h[:n-la])
                g[:n-la] -= weights[i] * h[la:]
    return g, corr, mc

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")

    shifted = beta * (corr - m)

    return (m + logsum / beta) * dx

def sum_obj(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def run_slsqp(h0, n, target, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_obj, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except:
        return h0

    step = step0

    return best_h, best_c5

    # Adaptive deltas: start large, get smaller

def clip01(x):
    return max(0.0, min(1.0, x))

def make_binary(h, target):
    """Snap to binary, maintaining sum constraint."""
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 0.5:
        # Need more 1s
        order = np.argsort(h)[::-1]
        for i in range(int(diff)):
            if i >= n: break
            z[order[i]] = 1.0

        for i in range(int(-diff)):
            if i >= n: break
            z[order[i]] = 0.0
    return project_h(z, target)

def make_vander_corput(n):
    """Low-discrepancy based pattern."""
    x = np.zeros(n)
    for i in range(n):
        # van der Corput sequence
        f = 1.0
        val = 0.0
        j = i + 1
        while j > 0:
            val += (j % 2) * f
            j //= 2
            f /= 2
        x[i] = val
    # Threshold at 0.5 to get binary with density ~0.5
    h = (x < 0.5).astype(float)
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.log(np.sum(np.exp(np.clip(beta*(corr-m), -50, 50))))
    return (m + s/beta) * dx

def sq_eq(x, t):
    return x.sum() - t

def sq_jac(x, t):
    return np.ones(x.shape)

def run_slsqp(h0, n, target, beta, maxiter):
    b = Bounds(np.zeros(n), np.ones(n))
    c = [{'type':'eq','fun':sq_eq,'jac':sq_jac,'args':(target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=b, constraints=c, options={'maxiter':maxiter,'ftol':1e-14})
        return project_h(r.x, target)
    except:
        return h0

def wsubgrad(h, n, scale=50):
    corr = np.correlate(h, 1.0-h, mode="full")
    m = np.max(corr)
    w = np.exp(-scale*(m-corr)/max(m,1e-10))
    ws = w.sum()
    if ws < 1e-15:
        w = (corr==m).astype(float); ws = w.sum()
    w /= ws
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-8: continue
        ll = k - (n-1)
        if ll >= 0:
            g[:n-ll] += w[k]*(1-h[ll:])
            g[ll:] -= w[k]*h[:n-ll]
        else:
            a = -ll
            g[a:] += w[k]*(1-h[:n-a])
            g[:n-a] -= w[k]*h[a:]
    return g, m

def subgrad_opt(h0, n, target, n_iter, step0, t0, budget):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time()-t0 > budget-3: break
        g, m = wsubgrad(h, n)
        gn = np.linalg.norm(g)
        if gn < 1e-12: break
        g /= gn
        h_new = project_h(h - step*g, target)
        c = compute_c5(h_new)
        if c < best_c5-1e-12:
            best_h, best_c5, stall = h_new.copy(), c, 0
            step = min(step*1.005, 0.08)
        else:
            stall += 1
            if stall > 80:
                step *= 0.6; stall = 0
        h = h_new
    return best_h, best_c5

def coord_opt(h0, target, n, t0, budget, sweeps=20):
    best_c5 = compute_c5(h0)
    h = h0.copy()
    for sw in range(sweeps):
        if time.time()-t0 > budget-2: break
        imp = False
        sc = 1.0/(1.0+0.25*sw)
        ds = (0.08*sc, -0.08*sc, 0.04*sc, -0.04*sc, 0.02*sc, -0.02*sc, 0.01*sc, -0.01*sc)
        for j in range(n):

                if c < best_c5-1e-13:
                    h, best_c5, imp = t, c, True
                    break
        if not imp and sw > 4: break
    return h, best_c5

def swap_opt(h0, target, n, t0, budget, n_iters=3000):

    return h, best_c5

    z = np.zeros(len(h))
    idx = np.argsort(h)[::-1]
    z[idx[:int(target)]] = 1.0
    return project_h(z, target)

def make_periodic(n, period, phase=0):
    x = np.arange(n)
    h = (((x+phase) % period) < period//2).astype(float)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    top = [inits[i] for i in scored[:6]]

    return best_h, best_c5, n
