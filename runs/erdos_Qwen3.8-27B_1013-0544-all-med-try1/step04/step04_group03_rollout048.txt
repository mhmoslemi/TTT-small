import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_to(h, n_new, target):
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_h(h_new, target)

def thue_morse_bin(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Ensure exactly n//2 ones
    n_ones = t.sum()
    need = n // 2
    if n_ones < need:
        zeros = np.where(t == 0)[0]
        np.random.shuffle(zeros)
        for i in range(need - n_ones):
            t[zeros[i]] = 1
    elif n_ones > need:
        ones = np.where(t == 1)[0]
        np.random.shuffle(ones)
        for i in range(n_ones - need):
            t[ones[i]] = 0
    return t.astype(float)

def structured_pattern(n, pattern_type, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    if pattern_type == 0:
        # Alternating blocks
        block = max(n // 8, 2)
        for i in range(n):
            h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    elif pattern_type == 1:
        # Sine-based
        h = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    elif pattern_type == 2:
        # Two-scale pattern
        period = max(n // 4, 4)
        for p in range(0, n, period):
            end = min(p + period, n)
            mid = p + period // 2
            h[p:mid] = 1.0
            h[mid:end] = 0.0
    else:
        # Random balanced
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    return project_h(h, n / 2.0)

def binary_local_search(h_bin, t0, budget_s, max_rounds=3000, seed=42):
    n = len(h_bin)
    h = h_bin.copy()
    cur_corr = get_corr(h)
    cur_max = float(np.max(cur_corr))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(seed)
    temp = 0.3
    no_improve = 0
    round_count = 0

    for rnd in range(max_rounds):
        if budget_s - (time.time() - t0) < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Try random swaps
        n_tries = min(500, len(ones) * len(zeros))
        best_delta = 0
        best_a, best_b = -1, -1

        for _ in range(n_tries):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            if a == b:
                continue
            h[a] = 0.0
            h[b] = 1.0
            new_corr = get_corr(h)
            new_max = float(np.max(new_corr))
            h[a] = 1.0
            h[b] = 0.0
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_a, best_b = a, b

        accepted = False
        if best_a >= 0:
            if best_delta < -1e-10:
                accepted = True
            elif rng.random() < np.exp(-best_delta / max(temp, 1e-12)):
                accepted = True

        if accepted:
            h[best_a] = 0.0
            h[best_b] = 1.0
            cur_corr = get_corr(h)
            cur_max = float(np.max(cur_corr))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        temp *= 0.998
        if no_improve > 80:
            break
        round_count += 1

    return best_h, best_max

def directed_swap_search(h_bin, t0, budget_s):

    n = len(h_bin)

def smooth_max(x, beta):
    m = np.max(x)
    xc = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xc)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_con(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or r.fun < slsqp_obj(h0, beta, n):
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def subgrad_refine(h, target, n, n_iter, t0, budget_s):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = 0.01
    momentum = np.zeros(n)
    mom_factor = 0.7
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3:
            break
        corr = get_corr(cur)
        top_k = 8
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        momentum = mom_factor * momentum - step * grad
        trial = project_h(cur + momentum, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-14:
            cur = trial
            best_c5 = c_try
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.7
            momentum *= 0.5

    return cur, best_c5
