```python
def binary_local_search(h_bin, max_sweeps=100, t0=None, budget=None, verbose=False):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    R = np.correlate(h, 1-h, mode="full")
    cur_max = R.max()
    
    for sweep in range(max_sweeps):
        if t0 is not None and budget is not None:
            if budget - (time.time() - t0) < 5:
                break
        
        D = compute_influence(h)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improvement = 0
        best_a, best_b = -1, -1
        best_R_new = None
        
        for a in ones:
            # For this a, compute delta for all b in zeros
            # ΔR_b = D[zeros, :][:, b] - D[:, a]  -- wait, D is indexed as D[position, lag]
            # ΔR at lag k = D[b, k] - D[a, k] for all k
            # R_new_b = R + D[b,:] - D[a,:]
            delta = D[zeros, :] - D[a, :]  # shape: (n_zeros, nlags)
            R_new = R[None, :] + delta  # broadcast: (n_zeros, nlags)
            maxes = R_new.max(axis=1)  # shape: (n_zeros,)
            best_b_idx = maxes.argmin()
            if maxes[best_b_idx] < cur_max - best_improvement:
                best_improvement = cur_max - maxes[best_b_idx]
                best_a = a
                best_b = zeros[best_b_idx]
                best_R_new = R_new[best_b_idx]
        
        if best_a == -1:
            break  # No improvement possible
        
        # Perform the swap
        h[best_a] = 0
        h[best_b] = 1
        R = best_R_new
        cur_max = R.max()
        if verbose:
            print(f"Sweep {sweep}: max={cur_max}")
    
    return h, cur_max
```

```python
                best_improvement = cur_max - maxes[bi]
                best_a = a
                best_b = zeros[bi]
                best_R_new = R_new[bi]
        
        if best_a == -1:
            break
        
        h[best_a] = 0.0
        h[best_b] = 1.0
        R = best_R_new
        cur_max = R.max()
    
    return h, cur_max

def smooth_max_obj(h, beta, n, R_cache=None):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * (2.0 / n)

def slsqp_refine(h0, n, target, beta, maxiter=10000, ftol=1e-15):
    def obj(x):
        return smooth_max_obj(x, beta, n)
    
    def con(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample_h(h, n_new):
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)
```
