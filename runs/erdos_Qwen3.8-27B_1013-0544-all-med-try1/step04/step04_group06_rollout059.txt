    n = len(h)
    corr = fft_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]

    g_prev = None

            break
        g = subgrad_active(best, n_active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

            ys = np.dot(s_prev, s_prev)

                    step = abs(ys) / sg  # BB1

                best_c5 = c5t
                s_prev = s_new
                g_prev = g.copy()

                break

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fft_corr(h):
    n = len(h)
    b = 1.0 - h
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(H * np.conj(B), N)[:size]

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_at_lag(h, lag):
    n = len(h)
    if lag == 0:
        return np.sum(h * (1 - h))
    elif lag > 0:
        return np.sum(h[:n-lag] * (1 - h[lag:]))
    else:
        m2 = -lag
        return np.sum(h[m2:] * (1 - h[:n-m2]))

def slsqp_minmax(h_init, n, active_indices, maxiter=5000):
    """SLSQP: minimize t s.t. t >= corr[j] for j in active, sum(h)=target, 0<=h<=1"""
    target = n / 2.0
    nact = len(active_indices)
    lags = [int(j) - (n - 1) for j in active_indices]

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def make_cons_and_jac(lag):
        def fun(x):
            h = x[:n]
            t = x[n]
            return t - corr_at_lag(h, lag)
        def jac(x):

                g[m2:] = -(1.0 - h[:n-m2])
                g[:n-m2] = -h[m2:]
            return g
        return fun, jac

        cons.append({'type': 'ineq', 'fun': f, 'jac': j})

        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fft_corr(h):
    n = len(h)
    b = 1.0 - h
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(H * np.conj(B), N)[:size]

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_active(h, k):
    corr = fft_corr(h)
    return np.argsort(corr)[::-1][:k]

def grad_corr_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        m2 = -lag
        g[m2:] += (1.0 - h[:n - m2])
        g[:n - m2] -= h[m2:]
    return g

def corr_at(h, lag):
    n = len(h)
    if lag == 0:
        return np.sum(h * (1.0 - h))
    elif lag > 0:
        return np.sum(h[:n - lag] * (1.0 - h[lag:]))
    else:
        m2 = -lag
        return np.sum(h[m2:] * (1.0 - h[:n - m2]))

def slsqp_minmax(h_init, n, active_idx, maxiter=4000):
    target = n / 2.0
    lags = [int(j) - (n - 1) for j in active_idx]
    na = len(lags)

    def fun_obj(x):
        return x[n]

    def jac_obj(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = []
    for lag in lags:
        def make_f(l):
            def cf(x):
                return x[n] - corr_at(x[:n], l)
            def cj(x):
                h = x[:n]
                g = np.zeros(n + 1)
                g[n] = 1.0
                gl = -grad_corr_lag(h, l)
                g[:n] = gl
                return g
            return cf, cj
        cf, cj = make_f(lag)
        cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})

    def eq_f(x):
        return np.sum(x[:n]) - target
    def eq_j(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_f, 'jac': eq_j})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    init_t = max(corr_at(h_init, l) for l in lags) + 0.3
    x0 = np.concatenate([h_init, [init_t]])

    res = minimize(fun_obj, x0, jac=jac_obj, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(res.x[:n], target)

def subgrad_polish(h, target, n_act, max_iter, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_imp = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = fft_corr(best)
        top = np.argsort(corr)[::-1][:n_act]
        g = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            g += grad_corr_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp % 300 == 299:
                pert = np.random.randn(n) * 0.004
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_imp = 0
            if no_imp > 4000:
                break
        else:
            no_imp = 0
    return best, best_c5
