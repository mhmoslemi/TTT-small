import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * subgrad_for_lag(h, lag, n)
    return grad * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_optimize(h, n, deadline, seed, betas):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in betas:
        if time.time() > deadline:
            break
        for p in [0.0, 0.01]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def prox_grad_polish(h, n, deadline, seed):
    """Projected subgradient with adaptive step size."""
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    c5_init = c5
    # Polyak step size: step_t = (f_t - f*) / ||g_t||^2
    # We approximate f* by best_c5
    for t in range(1, 50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top-k weighted subgradient
        top_k = min(20, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(20.0 * (corr[top_idx] - corr.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * subgrad_for_lag(h, lag, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Step size: use (f - f_est) / ||g||^2 with f_est = best_c5
        step = max(0.01 * (c5 - best_c5) / (gnorm**2 * (2.0/n)), 1e-4 / t)
        step = min(step, 0.5 / gnorm) * (2.0 / n)
        h_new = project_feasible(h - step * grad, ts)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
        h = h_new
        c5 = c_new
        if t % 100 == 0 and c5 > c5_init + 0.01:
            h = best_h.copy()
            c5 = best_c5
            break
    return best_h, best_c5

def make_block_h(n, block_frac):
    """Create a block construction: alternating 1s and 0s."""
    h = np.zeros(n)
    block = int(n * block_frac)
    for i in range(0, n, 2 * block):
        h[i:i+block] = 1.0
    return project_feasible(h, n / 2.0)

def make_ramp_h(n):
    """Create a ramp: linearly increasing then wrapping."""
    h = np.arange(n) / n
    return project_feasible(h, n / 2.0)

    h = np.zeros(n)
    b = int(n * frac / 2)
    h[:b] = 1.0
