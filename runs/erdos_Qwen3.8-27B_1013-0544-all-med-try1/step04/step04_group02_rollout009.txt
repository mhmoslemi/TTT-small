import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sm_obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 50)
    obj = (m + np.log(np.sum(np.exp(sh))) / beta) * dx
    w = np.exp(sh)
    w /= w.sum()
    active = np.where(w > 1e-8)[0]
    n = len(h)
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += w[k] * (1.0 - h[idx + lag])
            grad[idx + lag] -= w[k] * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += w[k] * (1.0 - h[idx])
            grad[idx] -= w[k] * h[idx + la]
    return obj, grad * dx

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(150):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        mask = h < 1.0 - 1e-13 if d > 0 else h > 1e-13
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp(h, n, beta, maxit, deadline):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    try:

                       jac=lambda x: sm_obj_grad(x, dx, beta)[1],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        if time.time() < deadline:
            return project(res.x, ts)
    except Exception:
        pass
    return project(h, ts)
