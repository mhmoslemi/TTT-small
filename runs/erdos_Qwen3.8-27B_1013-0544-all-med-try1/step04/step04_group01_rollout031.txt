def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_logsumexp(x, n, alpha):

    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.03 * (r / max(1, n_restarts - 1))
            hp = best_h + amp * np.sin(freq * np.pi * x + phase)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.25)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

        # Also try random restart
        if time.time() < deadline - 30:
            for r in range(3):
                if time.time() > deadline - 30:
                    break
                hp = best_h + rng.normal(0, 0.03, n)
                hp = project_feasible(hp, n)
                alloc = min(30, (deadline - time.time() - 25) * 0.3)
                if alloc < 15:
                    break
                h_opt, c_opt = optimize_single(hp, n, time.time() + alloc)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 4: Final intensive local search
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        # Subgradient search
        if time.time() < deadline - 5:
            best_h, best_c = subgradient_pair_search(best_h, n, deadline - 3, base_step=0.005)
        # Very fine pair search
        if time.time() < deadline - 2:
            best_h, best_c = fast_pair_search(best_h, n, deadline - 1,
                                              [0.001, 0.0005, 0.0002, 0.0001])
        # Final SLSQP
        if time.time() < deadline - 0.5:
            ht = slsqp_logsumexp(best_h, n, 20000000, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_h, best_c = ht, c

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
