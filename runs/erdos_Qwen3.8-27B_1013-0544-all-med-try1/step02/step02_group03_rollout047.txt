    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_worst_lags(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices by value
    idx = np.argsort(corr)[::-1][:top_k]
    return idx, corr

def subgrad_for_lag(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            grad[j1] += wk * (1.0 - h[j1 + lag])
            j2 = np.arange(lag, n)
            grad[j2] -= wk * h[j2 - lag]
        else:
            la = -lag
            j1 = np.arange(la, n)
            grad[j1] += wk * (1.0 - h[j1 - la])
            j2 = np.arange(0, n - la)
            grad[j2] -= wk * h[j2 + la]
    return grad * dx

    return smooth_max_val(np.correlate(h, 1.0 - h, mode='full'), beta) * dx

def jac_func(h, beta):
    return grad_smooth_max(h, beta)

def con_fun(h, target_sum):
    return h.sum() - target_sum

def con_jac(h, target_sum):
    return np.ones(len(h))

def subgrad_for_lag(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
