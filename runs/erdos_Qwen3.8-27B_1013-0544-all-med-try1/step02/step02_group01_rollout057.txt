def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr_full(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])

        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

    """Smooth max over all lags."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    val = (m + np.log(sw) / beta) * dx
    return val, corr, sw

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:

        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

    """SLSQP with smooth max, optionally restricted to active lags."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    j = np.arange(n)
    
    def obj(x):
        corr = compute_corr_full(x)
        if active_lags is not None:
            corr_use = corr[active_lags]
        else:
            corr_use = corr
        m = np.max(corr_use)
        exps = np.exp(beta * (corr_use - m))
        sw = np.sum(exps)
        return (m + np.log(sw) / beta) * dx
    
    def objg(x):
        corr = compute_corr_full(x)
        if active_lags is not None:
            corr_use = corr[active_lags]
        else:
            corr_use = corr
        m = np.max(corr_use)
        exps = np.exp(beta * (corr_use - m))
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        if active_lags is not None:
            indices = active_lags
            weights = w
        else:
            indices = np.arange(len(corr))
            weights = w
        for i, k in enumerate(indices):
            wk = weights[i]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += wk * (1.0 - x[js[ms]])
            if mp.any():
                grad[mp] += wk * (-x[jp[mp]])
        grad *= dx
        return grad

        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def active_set_optimize(h, maxiter_per_round, beta, n_rounds=5, top_k=3):
    """Iteratively identify active lags and optimize."""

    dx = 2.0 / n

    current_val = compute_c5(current)
    
    for r in range(n_rounds):

        max_corr = np.max(corr)
        # Select top-k lags with some margin
        thresholds = np.sort(corr)[-max(top_k, 1):]
        threshold = thresholds[0] - 0.01 * max_corr
        active = np.where(corr >= threshold)[0]

        val_new = compute_c5(h_new)
        
        if val_new < current_val:
            current = h_new
            current_val = val_new
        else:
            # Try with all lags
            h_new2 = slsqp_optimize(current, maxiter_per_round, beta)
            val_new2 = compute_c5(h_new2)
            if val_new2 < current_val:
                current = h_new2
                current_val = val_new2
            else:
                break
    
    return current, current_val

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr_full(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):

    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_slsqp_n = 0
_slsqp_beta = 1.0
_slsqp_active = None
_slsqp_target = 0.0
_slsqp_dx = 0.0
_slsqp_j = None

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x):
    dx = _slsqp_dx
    corr = compute_corr_full(x)
    if _slsqp_active is not None:
        corr_use = corr[_slsqp_active]
    else:
        corr_use = corr
    m = np.max(corr_use)
    beta = _slsqp_beta
    exps = np.exp(beta * (corr_use - m))
    sw = np.sum(exps)
    return (m + np.log(sw) / beta) * dx

def _slsqp_objg(x):
    n = _slsqp_n
    dx = _slsqp_dx
    j = _slsqp_j
    corr = compute_corr_full(x)
    if _slsqp_active is not None:
        corr_use = corr[_slsqp_active]
        indices = _slsqp_active
    else:
        corr_use = corr
        indices = np.arange(len(corr))
    m = np.max(corr_use)
    beta = _slsqp_beta
    exps = np.exp(beta * (corr_use - m))
    sw = np.sum(exps)
    w = exps / sw
    grad = np.zeros(n)
    for i in range(len(indices)):
        wk = w[i]
        if wk < 1e-12:
            continue
        k = indices[i]
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - x[js[ms]])
        if mp.any():
            grad[mp] += wk * (-x[jp[mp]])
    grad *= dx
    return grad

def _sum_func(x):
    return x.sum() - _slsqp_target

def _sum_jac(x):
    return np.ones(_slsqp_n)

def slsqp_optimize(h, maxiter, beta, active_lags=None):
    global _slsqp_n, _slsqp_beta, _slsqp_active, _slsqp_target, _slsqp_dx, _slsqp_j
    n = len(h)
    _slsqp_n = n
    _slsqp_beta = beta
    _slsqp_active = active_lags
    _slsqp_target = n / 2.0
    _slsqp_dx = 2.0 / n
    _slsqp_j = np.arange(n)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_func, 'jac': _sum_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)

def active_set_optimize(h, maxiter_per_round, beta, n_rounds, top_k):
    n = len(h)
    current = h.copy()
    current_val = compute_c5(current)
    for r in range(n_rounds):
        corr = compute_corr_full(current)
        max_corr = np.max(corr)
        sorted_corr = np.sort(corr)
        threshold = sorted_corr[-top_k] - 0.005 * max_corr
        active = np.where(corr >= threshold)[0]
        h_new = slsqp_optimize(current, maxiter_per_round, beta, active_lags=active)
        val_new = compute_c5(h_new)
        if val_new < current_val:
            current = h_new
            current_val = val_new
        else:
            h_new2 = slsqp_optimize(current, maxiter_per_round, beta)
            val_new2 = compute_c5(h_new2)
            if val_new2 < current_val:
                current = h_new2
                current_val = val_new2
            else:
                break
    return current, current_val

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr_full(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():

        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
