def solve_lp_relaxation(h_fixed_complement, n):
    """Solve the LP relaxation with frozen complement."""
    # h_fixed_complement = 1 - h_current, shape (n,)
    g = h_fixed_complement  # this is 1 - h_current
    
    # Build constraint matrix for lags
    n_lags = 2 * n - 1
    # A shape: (n_lags, n) where A[m, j] = g[j - lag_m]
    A = np.zeros((n_lags, n))
    for m in range(n_lags):
        lag = m - (n - 1)
        for j in range(n):
            idx = j - lag
            if 0 <= idx < n:
                A[m, j] = g[idx]
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize t: c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ h - t <= 0 for each lag
    # i.e., [A, -I] @ [h, t] <= 0
    A_ub = np.hstack([A, -np.eye(n_lags)])
    b_ub = np.zeros(n_lags)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)


def solve_lp(n, g):
    n_lags = 2 * n - 1
    M = np.zeros((n_lags, n))
    for m in range(n_lags):
        lag = m - (n - 1)
        for j in range(n):
            idx = j - lag
            if 0 <= idx < n:
                M[m, j] = g[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.eye(n_lags)])
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def subgrad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    if L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        L = -L
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    return g


def compute_subgradient(h, tol=1e-3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    active = np.where(corr >= mx - tol)[0]
    g = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        g += subgrad_lag(h, lag)
    g /= len(active)
    return g, mx


def dca_phase(h, n, max_iters, time_fn):
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time_fn() < 200:
            break
        g_frozen = 1.0 - h
        h_lp = solve_lp(n, g_frozen)
        if h_lp is None:
            break
        best_t = 0.0
        best_c5_trial = best_c5
        for tm in [1.0, 0.8, 0.6, 0.4, 0.2]:
            h_try = (1.0 - tm) * h + tm * h_lp
            h_try = project_h(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5_trial:
                best_c5_trial = c5_try
                best_t = tm
        if best_t > 0:
            h = project_h((1.0 - best_t) * h + best_t * h_lp, target_sum)
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        else:
            if it > 5:
                break
    return best_h, best_c5


def subgrad_phase(h, n, max_iters, time_fn, rng, init_tol=1e-3):
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tol_list = [init_tol, init_tol * 0.1, init_tol * 0.01, init_tol * 0.001]
    tol_idx = 0
    stagnation = 0
    step_sizes = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    for it in range(max_iters):
        if time_fn() < 60:
            break
        g, mx = compute_subgradient(h, tol=tol_list[tol_idx])
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            if tol_idx < len(tol_list) - 1:
                tol_idx += 1
                continue
            h = project_h(h + rng.randn(n) * 0.02, target_sum)
            tol_idx = 0
            stagnation = 0
            continue
        g_dir = g / gn
        found = False
        for s in step_sizes:
            h_try = project_h(h - s * g_dir, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                found = True
                break
        if not found:
            stagnation += 1
            if stagnation > 30:
                scale = 0.03 + 0.01 * min(stagnation / 50.0, 1.0)
                h_pert = project_h(h + rng.randn(n) * scale, target_sum)
                c5_p = compute_c5(h_pert)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
                    h = h_pert
                else:
                    h = best_h.copy()
                tol_idx = min(tol_idx + 1, len(tol_list) - 1)
                stagnation = 0
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    h0 = project_h(h_init, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: DCA with LP (fast convergence)
    h1, c5_1 = dca_phase(h0, n_points, 300, time_left)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient refinement
    h2, c5_2 = subgrad_phase(best_h, n_points, 3000, time_left, rng, init_tol=1e-3)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-start with perturbations
    attempts = 0
    while time_left() > 80:
        attempts += 1
        scale = 0.04 + 0.02 * np.exp(-attempts / 8.0)
        h_pert = project_h(best_h + rng.randn(n_points) * scale, target_sum)
        # Short DCA from perturbed start
        h_a, c5_a = dca_phase(h_pert, n_points, 80, time_left)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
        # Short subgrad from perturbed start
        h_b, c5_b = subgrad_phase(h_pert, n_points, 500, time_left, rng, init_tol=1e-2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: Final polish with tight tolerance
    if time_left() > 30:
        h3, c5_3 = subgrad_phase(best_h, n_points, 2000, time_left, rng, init_tol=1e-5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Final projection and verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
