        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def cd_sweep(h, n_sweeps=200):
    """Coordinate descent sweep with exact objective."""
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    for _ in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        any_change = False
        
        for i in range(n):
            old_val = h[i]
            # Determine search range
            if old_val < 0.05:
                lo, hi = 0.0, 0.5
            elif old_val > 0.95:
                lo, hi = 0.5, 1.0
            else:
                lo = max(0.0, old_val - 0.4)
                hi = min(1.0, old_val + 0.4)
            
            candidates = np.linspace(lo, hi, 25)
            best_t = old_val
            best_obj = np.inf
            
            for t in candidates:
                if abs(t - old_val) < 1e-10:
                    continue
                delta = t - old_val
                new_corr = corr.copy()
                
                # First factor: m in [i, min(2n-2, i+n-1)]
                m_end1 = min(2*n - 2, i + n - 1)
                for m in range(i, m_end1 + 1):
                    n_idx = i - m + n - 1
                    if 0 <= n_idx < n:
                        if m == n - 1:
                            new_corr[m] = corr[m] - old_val*(1.0 - old_val) + t*(1.0 - t)
                        else:
                            new_corr[m] = corr[m] + delta * (1.0 - h[n_idx])
                
                # Second factor: m in [n-1-i, n-2]
                if n - 1 - i <= n - 2:
                    for m in range(n - 1 - i, n - 1):
                        h_idx = i + m - n + 1
                        if 0 <= h_idx < n:
                            if m == n - 1:
                                pass  # already handled
                            else:
                                new_corr[m] = corr[m] - delta * h[h_idx]
                
                obj = np.max(new_corr)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            if abs(best_t - old_val) > 1e-10:
                h[i] = best_t
                corr = np.correlate(h, 1.0 - h, mode="full")
                any_change = True
        
        # Project
        h = project_h(h, target_sum)
        if not any_change:
            break
    
    return h

def slsqp_optimize(h, beta, maxiter=3000):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    
    def con_fun(h):
        return np.sum(h) - target_sum
    
    def con_jac(h):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(res.x, target_sum)

def make_binary_pattern(n, period, fill):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    pattern = [1.0]*int(fill*period) + [0.0]*(period - int(fill*period))
    for i in range(n):
        h[i] = pattern[i % period]
    return h

def make_binary_random(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.random(n) < 0.5).astype(float)
    # Adjust to get exact sum n/2
    target = n // 2
    ones = np.sum(h == 1.0)
    if ones > target:
        # Need to flip some 1s to 0s
        one_indices = np.where(h == 1.0)[0]
        to_flip = rng.choice(one_indices, ones - target, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        zero_indices = np.where(h == 0.0)[0]
        to_flip = rng.choice(zero_indices, target - ones, replace=False)
        h[to_flip] = 1.0
    return h

def slsqp_refine(h, beta=500, maxiter=2000):

    target_sum = n / 2.0

    def obj_func(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        return smooth_max(corr, beta) * dx

        return np.sum(hv) - target_sum

def make_binary_random(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.random(n) < 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h == 1.0))
    if ones > target:
        idx = np.where(h == 1.0)[0]
        flips = rng.choice(idx, ones - target, replace=False)
        h[flips] = 0.0
    elif ones < target:
        idx = np.where(h == 0.0)[0]
        flips = rng.choice(idx, target - ones, replace=False)
        h[flips] = 1.0
    return h

def make_periodic(n, period, fill):
    h = np.zeros(n)
    k = int(round(fill * period))
    k = max(1, min(k, period - 1))
    for i in range(n):
        h[i] = 1.0 if (i % period) < k else 0.0
    return h

def make_ramp(n, frac=0.6):
    h = np.ones(n)
    n_on = int(frac * n)
    h[n_on:] = 0.0
    for i in range(max(0, n_on - 5), min(n, n_on + 5)):
        h[i] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 70

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        target0 = n0 / 2.0
        h0 = project_h(h0, target0)
        c5_0 = compute_c5(h0)
        best_h, best_c5, best_n = h0.copy(), c5_0, n0
        result = cd_optimize(h0, n_sweeps=400, time_fn=elapsed, time_limit=budget_s * 0.25)
        if result[1] < best_c5:
            best_h, best_c5 = result[0].copy(), result[1]
        if tleft() > 80:
            for beta in [300, 800, 1500]:
                if tleft() < 30:
                    break
                h_sl = slsqp_refine(best_h, beta)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_h, best_c5 = h_sl.copy(), c5_sl
                    result = cd_optimize(best_h, n_sweeps=100, time_fn=elapsed, time_limit=budget_s * 0.5)
                    if result[1] < best_c5:
                        best_h, best_c5 = result[0].copy(), result[1]

    n_to_try = [n for n in [80, 90, 100, 120, 140, 60] if n != best_n]
    for n_pts in n_to_try:
        if tleft() < 50:
            break
        target = n_pts / 2.0
        inits = []
        for s in range(4):
            inits.append(make_binary_random(n_pts, seed + s * 137))
        for period in [5, 7, 9, 11, 13, 17, 2, 3, 4]:
            for fill in [0.3, 0.4, 0.5, 0.6]:
                inits.append(make_periodic(n_pts, period, fill))
        inits.append(make_ramp(n_pts, 0.55))
        inits.append(make_ramp(n_pts, 0.45))
        x = np.linspace(0, 2, n_pts)
        inits.append(np.clip(0.5 + 0.45 * np.sin(2 * np.pi * x * 3.0 / 2.0), 0, 1))
        inits.append(np.clip(0.5 + 0.45 * np.sin(2 * np.pi * x * 5.0 / 2.0 + 1.0), 0, 1))
        valid_inits = []
        for h_init in inits:
            h_init = project_h(h_init, target)
            c5_init = compute_c5(h_init)
            valid_inits.append((c5_init, h_init))
        valid_inits.sort(key=lambda x: x[0])
        max_tries = min(len(valid_inits), 6)
        for c5_init, h_init in valid_inits[:max_tries]:
            if tleft() < 30:
                break
            h_cd, c5_cd = cd_optimize(h_init, n_sweeps=200, time_fn=elapsed, time_limit=budget_s * 0.85)
            if c5_cd < best_c5:
                best_h, best_c5, best_n = h_cd.copy(), c5_cd, n_pts
                if tleft() > 40:
                    h_sl = slsqp_refine(h_cd, 600)
                    c5_sl = compute_c5(h_sl)
                    if c5_sl < best_c5:
                        best_h, best_c5 = h_sl.copy(), c5_sl
                        h_cd2, c5_cd2 = cd_optimize(best_h, n_sweeps=80, time_fn=elapsed, time_limit=budget_s * 0.9)
                        if c5_cd2 < best_c5:
                            best_h, best_c5 = h_cd2.copy(), c5_cd2

    if tleft() > 20:
        for beta in [800, 1500, 3000]:
            if tleft() < 10:
                break
            h_sl = slsqp_refine(best_h, beta)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_h, best_c5 = h_sl.copy(), c5_sl
    if tleft() > 10:
        result = cd_optimize(best_h, n_sweeps=200, time_fn=elapsed, time_limit=budget_s * 0.99)
        if result[1] < best_c5:
            best_h, best_c5 = result[0].copy(), result[1]

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
