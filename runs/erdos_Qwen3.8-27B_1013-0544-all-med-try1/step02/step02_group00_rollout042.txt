def project_h(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_step(h, lam, max_sweeps=200):
    """One round of coordinate descent with Lagrange multiplier lam."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    for sweep in range(max_sweeps):
        for j in range(n):
            old = h[j]
            a = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):

                c = 0.0
                j_minus = j - lag
                j_plus = j + lag
                if 0 <= j_minus < n:
                    c += 1.0 - h[j_minus]
                if 0 <= j_plus < n:
                    c -= h[j_plus]
                a[idx] = c
            b = corr - a * old
            a_eff = a + lam
            # Minimize max(b + a_eff * t) for t in [0,1] via ternary search
            lo, hi = 0.0, 1.0

  - There exists j with a_j > 0 and b_j + a_j * t* = g(t*)

    lam = 0.5 * (lam_lo + lam_hi)

    stall = 0
    for phase in range(10000):
        if time.time() - t0 > budget_s - 15:
            break

        # Periodically re-calibrate lambda
        if phase % 25 == 0 and phase > 0:
            lam = 0.5 * (lam_lo + lam_hi)
            ht = coord_descent(best_h, lam, 12)
            s = ht.sum()
            if s > target + 0.03:
                lam_lo = lam
            elif s < target - 0.03:
                lam_hi = lam
            lam = 0.5 * (lam_lo + lam_hi)

        # Main coordinate descent
        ht = coord_descent(best_h, lam, 50)
        ht = project_h(ht, target)
        c5 = compute_c5(ht)

        if c5 < best_c5 - 1e-13:
            best_h = ht.copy()
            best_c5 = c5
            stall = 0
        else:
            stall += 1

        # Escape local optima with smooth perturbation
        if stall >= 15:
            if time.time() - t0 > budget_s - 25:
                break
            pert_type = phase % 3
            if pert_type == 0:
                # Smooth sinusoidal perturbation
                freq = np.random.uniform(1.0, 6.0)
                ph = np.random.uniform(0, 2 * np.pi)
                amp = np.random.uniform(0.02, 0.07)
                hp = best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
            elif pert_type == 1:
                # Random sparse perturbation
                k = max(3, n // 12)
                idx = np.random.choice(n, k, replace=False)
                hp = best_h.copy()
                hp[idx] += np.random.uniform(-0.06, 0.06, k)
            else:
                # Block perturbation
                block_size = max(2, n // 8)
                start = np.random.randint(0, n - block_size)
                hp = best_h.copy()
                hp[start:start + block_size] += np.random.uniform(-0.04, 0.04)
            hp = project_h(hp, target)
            c5p = compute_c5(hp)
            if c5p < best_c5 + 0.004:
                best_h = hp
            stall = 0

    # Try multiple structured starting points
    if time.time() - t0 < budget_s - 50:
        starts = []
        for freq in [1, 2, 3, 4, 5, 7, 9, 13]:
            s = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n)
            starts.append(project_h(s, target))
        for freq in [1, 2, 3]:
            s = 0.5 + 0.35 * np.cos(2 * np.pi * freq * np.arange(n) / n)
            starts.append(project_h(s, target))
        alt = np.where(np.arange(n) % 2 == 0, 0.62, 0.38)
        starts.append(project_h(alt, target))

        for hs in starts:
            if time.time() - t0 > budget_s - 20:
                break
            ht = coord_descent(hs, lam, 40)
            ht = project_h(ht, target)
            c5 = compute_c5(ht)
            if c5 < best_c5:
                best_h = ht.copy()
                best_c5 = c5

    # Final polish with more sweeps
    if time.time() - t0 < budget_s - 15:
        ht = coord_descent(best_h, lam, 100)
        ht = project_h(ht, target)
        c5 = compute_c5(ht)
        if c5 < best_c5:
            best_h = ht.copy()
            best_c5 = c5

    return best_h, best_c5, n
