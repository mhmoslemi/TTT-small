```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    """Compute c[L] for L=0,...,n-1 (the unique lags)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # corr[n-1+L] = c[L] for L=0,...,n-1
    return corr[n-1:]  # shape (n,)

def find_min_1d(slopes, intercepts):
    """
    Find t in [0,1] minimizing max_i (slopes[i]*t + intercepts[i]).
    slopes, intercepts: 1D arrays of length m.
    """
    # Evaluate at endpoints
    f0 = np.max(intercepts)
    f1 = np.max(slopes + intercepts)

    n = len(slopes)
    # Vectorized: for each pair (i,j), t_ij = (b_j - b_i) / (a_i - a_j)
```

```python
def find_min_1d_vec(slopes, intercepts):
    n = len(slopes)
    # All pairwise intersections
    # A[i,j] = slopes[i] - slopes[j]
    # B[i,j] = intercepts[j] - intercepts[i]
    A = slopes[:, None] - slopes[None, :]  # (n, n)
    B = intercepts[None, :] - intercepts[:, None]  # (n, n)
    # t_ij = B[i,j] / A[i,j] where A[i,j] != 0
    valid = np.abs(A) > 1e-14
    T = np.where(valid, B / np.where(valid, A, 1.0), -1.0)
    # Filter to [0, 1]
    mask = (T >= 0) & (T <= 1) & valid & (np.arange(n)[:,None] < np.arange(n)[None,:])  # i < j to avoid duplicates
    candidates = T[mask]
    candidates = np.concatenate([candidates, [0.0, 1.0]])
    # Evaluate f at each candidate
    # f(t) = max_L (slopes[L]*t + intercepts[L])
    # Vectorized: create a matrix (num_candidates, n), take max along axis 1
    F = candidates[:, None] * slopes[None, :] + intercepts[None, :]
    f_vals = F.max(axis=1)
    idx = f_vals.argmin()
    return float(candidates[idx])
```

```python
def subgradient_c(h, k_worst):
    """Compute subgradient of c[k_worst] w.r.t. h."""
    n = len(h)
    L = k_worst  # lag (0-indexed, so actual lag = k_worst)
    grad = np.zeros(n)
    # c[L] = sum_{i=0}^{n-1-L} h[i+L](1-h[i])
    for i in range(n - L):
        grad[i+L] += (1 - h[i])
        grad[i] += (-h[i+L])
    return grad

    # At alpha=0: c[L](0) = c[L] (current values)
    # c[L](alpha) = c[L](0) + alpha * (dc[L]/dh) · direction
    # dc[L]/dh is the gradient of c[L] w.r.t. h
    
    # For each L, the slope of c[L] in the direction 'direction' is:
    # slope_L = grad_c_L · direction
    # where grad_c_L[i+L] = 1-h[i], grad_c_L[i] = -h[i+L]
    
    n = len(h)
    c_current = compute_lags(h)

    slopes = np.zeros(n)
    for L in range(n):
        # slope_L = sum of grad_c_L * direction
        # grad_c_L has: (1-h[i]) at position i+L, and -h[i+L] at position i
        s = 0.0
        for i in range(n - L):
            s += (1 - h[i]) * direction[i+L] + (-h[i+L]) * direction[i]
        slopes[L] = s
    
    intercepts = c_current

    # Subject to: 0 <= h + alpha*direction <= 1 and sum(h + alpha*direction) = sum(h)
    # The box constraint gives alpha bounds.
    
    # For box constraint: 0 <= h[j] + alpha*direction[j] <= 1
    # If direction[j] > 0: alpha <= (1-h[j])/direction[j]
    # If direction[j] < 0: alpha >= (0-h[j])/direction[j] = -h[j]/direction[j] (which is positive since direction[j]<0)
    # Wait: 0 <= h[j] + alpha*direction[j]
    # If direction[j] < 0: h[j] + alpha*direction[j] >= 0 => alpha <= h[j]/(-direction[j])
    # If direction[j] > 0: 1 >= h[j] + alpha*direction[j] => alpha <= (1-h[j])/direction[j]
    
    alpha_max = 0.1  # Default
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    
    alpha_max = max(alpha_max, 0.0)
    
    # Now find alpha in [0, alpha_max] minimizing max_L (slopes[L]*alpha + intercepts[L])
    # Same as find_min_1d but with bounds [0, alpha_max]
    
    # Use the same vectorized approach
    candidates = [0.0, alpha_max]
    s = slopes
    b = intercepts
    m = len(s)
    A = s[:, None] - s[None, :]
    B = b[None, :] - b[:, None]
    valid = np.abs(A) > 1e-14
    T = np.where(valid, B / np.where(valid, A, 1.0), -2.0)
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    mask = (T >= 0) & (T <= alpha_max) & valid & tri
    candidates.extend(T[mask].tolist())
    candidates = np.array(candidates)
    
    F = candidates[:, None] * s[None, :] + b[None, :]
    f_vals = F.max(axis=1)
    idx = f_vals.argmin()
    return float(candidates[idx])
```

Hmm, this is getting complex. Let me simplify and just use the coordinate descent approach, which is cleaner and already handles the 1D subproblem exactly.

Let me also think about whether I should try the "block" approach: optimize 2 or 3 coordinates simultaneously. This would help escape 1D local minima.

For 2 coordinates (j, k):
c[L](h_j, h_k) = a_Lj * h_j + a_Lk * h_k + b_L (bilinear in h_j, h_k if j and k are related by lag L, otherwise linear).

The objective max_L c[L](h_j, h_k) is the max of bilinear functions, which is not convex. So exact 2D minimization is harder.

Let me stick with 1D coordinate descent + random restarts + SLSQP polish.

One more thought: I can also try **cyclic coordinate descent** (always process coordinates in the same order) vs **random order** (shuffle the order each sweep). Random order might help escape some traps.

    n = len(h)
