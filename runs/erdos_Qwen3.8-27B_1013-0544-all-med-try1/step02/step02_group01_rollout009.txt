def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):

        top_idx = np.argsort(corr)[-top_k:]
        m = corr[top_idx[0]]  # min of top-k as reference

        sw = np.sum(exps)
        w = exps / sw
        val = (m + np.log(sw) / beta) * dx
        grad = np.zeros(n)
        j = np.arange(n)
        for idx in range(len(top_idx)):

                continue
            s = k - (n - 1)

def exact_subgrad(h, top_k=10):
    """Compute a subgradient of C5 at h, using top-k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    j = np.arange(n)
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    grad /= len(top_idx)
    grad *= dx
    return grad

def adam_exsubgrad(h, max_iters, lr, top_k, noise_schedule=None):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(1, max_iters + 1):
        grad = exact_subgrad(x, top_k)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x_new = x - step
        x_new = project_feasible(x_new, target)
        
        # Periodic perturbation
        if noise_schedule and t % noise_schedule[0] == 0:
            x_new += noise_schedule[1] * np.random.randn(n)
            x_new = project_feasible(x_new, target)
        
        x = x_new
        c = compute_c5(x)
        if c < best_v:
            best_v = c
            best_x = x.copy()
    
    return best_x, best_v

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def exact_subgrad(h, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    j = np.arange(n)
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    grad /= len(top_idx)
    grad *= dx
    return grad

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def adam_exsubgrad(h, max_iters, lr, top_k=10, b1=0.9, b2=0.999, noise_every=0, noise_scale=0.0):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    patience = 0
    for t in range(1, max_iters + 1):
        grad = exact_subgrad(x, top_k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x_new = x - step
        if noise_every > 0 and t % noise_every == 0:
            x_new += noise_scale * np.random.randn(n)
        x_new = project_feasible(x_new, target)
        x = x_new
        c = compute_c5(x)
        if c < best_v - 1e-10:
            best_v = c
            best_x = x.copy()
            patience = 0
        else:
            patience += 1
    return best_x, best_v

def slsqp_refine(h, maxiter=300, beta=100):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def project_half(h_half, target_half):
    """Project half-vector to feasible set: 0<=x<=1, sum=target_half."""
    h_half = np.clip(h_half, 0.0, 1.0)
    diff = target_half - h_half.sum()
    if abs(diff) > 1e-12:
        h_half = h_half + diff / len(h_half)

def smoothmax_val_grad_sym(h_half, n, beta, top_k=None):
    h = half_to_full(h_half, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        ref = corr[top_idx[0]]
    else:
        top_idx = np.arange(len(corr))
        ref = np.max(corr)
    exps = np.exp(beta * (corr[top_idx] - ref))
    sw = np.sum(exps)
    w = exps / sw
    val = (ref + np.log(sw) / beta) * dx
    grad_full = np.zeros(n)
    j = np.arange(n)
    for i in range(len(top_idx)):
        k = top_idx[i]
        wk = w[i]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad_full[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad_full[mp] += wk * (-h[jp[mp]])
    grad_full *= dx
    # Chain rule to get gradient w.r.t. h_half
    grad_half = grad_full[:n//2] + grad_full[n//2:][::-1]
    return val, grad_half
