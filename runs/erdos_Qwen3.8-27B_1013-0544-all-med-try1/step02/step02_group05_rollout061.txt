def coord_descent(h, target_sum, max_sweeps=20):
    n = len(h)
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        improved = False
        for p in range(n):
            save = h[p]
            h[p] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full")
            h[p] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode="full")
            h[p] = save
            a = corr1 - corr0
            c = corr0
            # Find optimal x in [0,1] minimizing max_j(c[j] + a[j]*x)
            # Try current value, boundaries, and breakpoints
            candidates = {0.0, 1.0, save}
            # Sample breakpoints (subset for speed)
            for i1 in range(0, num_lags, 3):
                for i2 in range(i1+3, num_lags, 3):
                    if abs(a[i2] - a[i1]) > 1e-12:
                        xbp = (c[i1] - c[i2]) / (a[i2] - a[i1])
                        if -0.01 <= xbp <= 1.01:
                            candidates.add(float(np.clip(xbp, 0, 1)))
            best_x = save
            best_val = float('max(c[j] + a[j]*x for j in range(num_lags))')  # max(c + a*save)
            best_val = float(np.max(c + a * save))
            for xc in candidates:
                xc = float(np.clip(xc, 0, 1))
                val = float(np.max(c + a * xc))
                if val < best_val - 1e-12:
                    best_val = val
                    best_x = xc
            if best_x != save:
                h[p] = best_x
                improved = True
        h = water_filling_project(h, target_sum)
        if not improved:
            break
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def water_filling_project(x, S):
    n = len(x)
    lo, hi = -S, n - S
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > S:

        else:

        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

        if c5_fin2 < best_c5:

            best_h = h_fin2.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
