    if abs(s - target_sum) > 1e-12 and s > 0:

    return delta, new_h, new_corr

def obj_fn(x, beta, dx):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    return (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * dx

def con_fun(x):
    return np.sum(x) - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def do_cd_phase(h_best, best_c5, n, dx, target_sum, t_start, t_end, rng, restart_thresh):
    h_cd = h_best.copy()
    corr_cd = get_corr(h_cd)
    stale = 0
    while time.time() < t_end:
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        delta, h_new, corr_new = cd_step(h_cd, corr_cd, i, j, n)
        if abs(delta) > 1e-13:
            h_cd, corr_cd = h_new, corr_new
            c5 = float(np.max(corr_cd) * dx)
            if c5 < best_c5:
                h_best, best_c5 = h_cd.copy(), c5
                stale = 0
            else:
                stale += 1
        else:
            stale += 1
        if stale > restart_thresh:
            scale = 0.005 + 0.005 * (stale // (2 * restart_thresh))
            h_cd = h_best + rng.standard_normal(n) * scale
            h_cd = project_affine_box(h_cd, target_sum)
            corr_cd = get_corr(h_cd)
            stale = 0
    return h_best, best_c5

def do_slsqp(h_best, n, t_limit, beta_list):
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    for beta in beta_list:
        if time.time() > t_limit:
            break
        try:
            res = minimize(obj_fn, h_best, args=(beta, dx), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_affine_box(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < compute_c5(h_best):
                h_best = h_new.copy()
        except:
            pass
    return h_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)

    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_affine_box(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: CD (50% budget)
    p1_end = start_time + budget_s * 0.50
    best_h, best_c5 = do_cd_phase(best_h, best_c5, n, dx, target_sum,
                                   start_time, p1_end, rng, 1500)

    # Phase 2: SLSQP polish (20% budget)
    p2_end = start_time + budget_s * 0.70
    best_h = do_slsqp(best_h, n, p2_end, [60, 120, 250, 500])

    # Phase 3: CD again (15% budget)
    p3_end = start_time + budget_s * 0.85
    best_h, best_c5 = do_cd_phase(best_h, best_c5, n, dx, target_sum,
                                   time.time(), p3_end, rng, 3000)

    # Phase 4: Final SLSQP (10% budget)
    p4_end = start_time + budget_s * 0.95
    best_h = do_slsqp(best_h, n, p4_end, [600, 800])

    # Phase 5: Tiny CD crunch (3% budget)
    p5_end = start_time + budget_s * 0.98
    best_h, best_c5 = do_cd_phase(best_h, best_c5, n, dx, target_sum,
                                   time.time(), p5_end, rng, 5000)

    # Final validation
    best_h = project_affine_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
