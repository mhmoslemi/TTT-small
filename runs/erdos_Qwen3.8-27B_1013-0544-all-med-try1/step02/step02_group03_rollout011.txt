def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50.0, 50.0)

def top_m_soft(x, m, beta):
    """Approximate average of top m values using soft weighting"""
    sorted_idx = np.argsort(x)[::-1]
    if m >= len(x):
        return np.mean(x)
    threshold = x[sorted_idx[m-1]]
    # Soft weight: sigmoid-shaped around threshold
    w = 1.0 / (1.0 + np.exp(-beta * (x - threshold)))
    # Ensure at least m elements have significant weight
    return np.sum(w * x) / np.sum(w)

                    res = minimize(obj_res, h_t, method='SLSQP', bounds=bnds_t,
                                   constraints=con_t, options={'maxiter': 2500, 'ftol': 1e-13})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        h_t = h_new.copy()
                except Exception:
                    pass

    # Phase 5: Random restarts from best
    if remain() > 60:
        for r in range(5):
            if remain() < 12:
                break
            x0 = project_feasible(best_h + np.random.randn(len(best_h)) * 0.02, target_sum)
            bnds = [(0.0, 1.0)] * len(best_h)
            con = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
                   'jac': lambda h: np.ones(len(best_h))}]
            try:
                def obj_rr(h):
                    c = np.correlate(h, 1.0 - h, mode='full')
                    return smooth_max(c, 2000) * dx
                res = minimize(obj_rr, x0, method='SLSQP', bounds=bnds,
                               constraints=con, options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final ultra-high-beta polish
    if remain() > 12:
        bnds = [(0.0, 1.0)] * n
        con = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
               'jac': lambda h: np.ones(n)}]
        for beta in [10000, 30000, 80000]:
            if remain() < 6:
                break
            try:
                def obj_fin(h):
                    c = np.correlate(h, 1.0 - h, mode='full')
                    return smooth_max(c, beta) * dx
                res = minimize(obj_fin, best_h, method='SLSQP', bounds=bnds,
                               constraints=con, options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5, len(best_h)
