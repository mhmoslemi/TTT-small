            corr = np.correlate(h, 1.0 - h, mode="full")
            top_k = np.argsort(corr)[-min(4, len(corr)):]
            for kk in top_k:
                lag = int(kk) - (n - 1)
                h = subgradient_step(h, n, lag, 0.006)
            h += np.random.randn(n) * 0.003
        elif strat == 3:
            blen = np.random.randint(4, max(5, n // 5))
            bs = np.random.randint(0, n - blen)
            h[bs:bs + blen] = 0.5 + 0.3 * np.random.randn(blen)
        elif strat == 4:
            f = np.random.uniform(1, 5)
            ph = np.random.uniform(0, 2 * np.pi)
            h += 0.035 * np.sin(2 * np.pi * f * np.arange(n) * 2.0 / n + ph)
        elif strat == 5:
            n_swaps = np.random.randint(2, max(3, n // 10))
            for _ in range(n_swaps):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h += np.random.randn(n) * 0.005
        else:
            # Perturb top-3 worst coordinates
            corr = np.correlate(h, 1.0 - h, mode="full")
            h += np.random.randn(n) * 0.05
            h[np.random.choice(n, 3, replace=False)] = np.random.uniform(0, 1, 3)

        h = project_feasible(h, target_sum)

        for beta in [600, 2000]:
            if remaining() < 30:
                break
            lam = min(beta * 2.0, 4000.0)
            try:
                h_new = run_lbfgsb(h, n, target_sum, beta, lam, 300, 1e-14)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5, h = h_new, c, h_new
            except Exception:
                pass

    # Phase 4: High-precision final polish
    if remaining() > 20:
        for beta in [5000, 10000]:
            if remaining() < 12:
                break
            lam = min(beta * 1.0, 5000.0)
            try:
                h_new = run_lbfgsb(best_h, n, target_sum, beta, lam, 3000, 1e-16)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 5: Try SLSQP with equality constraint
    if remaining() > 40:
        try:
            def slsqp_obj(h):
                n_local = len(h)
                dx = 2.0 / n_local
                corr = np.correlate(h, 1.0 - h, mode="full")
                max_c = corr.max()
                exp_arg = 2000.0 * (corr - max_c)
                return (max_c + np.log(np.sum(np.exp(exp_arg))) / 2000.0) * dx

            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                     'jac': lambda x: np.ones(n)}]
            bnds = [(0.0, 1.0)] * n
            r = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(r.x, target_sum)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
