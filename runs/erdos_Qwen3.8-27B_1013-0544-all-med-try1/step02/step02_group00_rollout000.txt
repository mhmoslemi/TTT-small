        def _obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        return _obj

    def _con(x_in):
        return np.sum(x_in) - target_sum

    def _jac(x_in):
        return np.ones(x_in.shape)

    constraints = [{'type': 'eq', 'fun': _con, 'jac': _jac}]

    for beta in [30, 60, 120, 300, 700, 1500]:
        if time_left() < 30:
            break
        try:
            res = minimize(_obj_factory(beta), best_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    # Phase 3: More subgradient with different top_k and smaller steps
    for top_k in [2, 4, 5]:
        if time_left() < 20:
            break
        h, c5 = subgradient_phase(best_h, target_sum, 40000, 0.02, top_k, time_left)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Perturbation-based exploration
    if time_left() > 40:
        for attempt in range(20):
            if time_left() < 15:
                break
            # Random Gaussian perturbation
            scale = 0.05 * (1.0 - attempt / 20.0)
            h_pert = project_h(best_h + np.random.randn(n_base) * scale, target_sum)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_h, best_c5 = h_pert, c5_pert
            # Also try: flip a random subset
            mask = np.random.random(n_base) < 0.1
            h_flip = best_h.copy()
            h_flip[mask] = 1.0 - h_flip[mask]
            h_flip = project_h(h_flip, target_sum)
            c5_flip = compute_c5(h_flip)
            if c5_flip < best_c5:
                best_h, best_c5 = h_flip, c5_flip
                # Follow up with subgradient
                h, c5 = subgradient_phase(best_h, target_sum, 10000, 0.03, 3, time_left)
                if c5 < best_c5:
                    best_h, best_c5 = h, c5

    # Phase 5: Final SLSQP polish
    if time_left() > 20:
        for beta in [2000, 5000]:
            if time_left() < 15:
                break
            try:
                res = minimize(_obj_factory(beta), best_h, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    # Phase 6: Final tiny subgradient steps
    if time_left() > 5:
        h, c5 = subgradient_phase(best_h, target_sum, 50000, 0.005, 4, time_left)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, n_base
