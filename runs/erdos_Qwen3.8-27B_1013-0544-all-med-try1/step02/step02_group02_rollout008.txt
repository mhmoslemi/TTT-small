import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)

def smoo_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    ss = s.max()
    if ss > 500:
        v = np.sum(np.exp(s - ss))
    else:
        v = np.sum(np.exp(s))
    return m + np.log(v) / alpha

def smoo_obj_jac(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    ss = s.max()
    if ss > 500:
        w = np.exp(s - ss)
        w /= w.sum()
    else:
        w = np.exp(s)
        w /= w.sum()
    # Gradient of w[k] * c[k] w.r.t. h
    # c[k] = dx * sum_i h[i]*(1-h[i+k-n+1]) for appropriate range
    # d(c[k])/d(h[j]) involves h[j] and its partners
    # For simplicity, use numerical-free approach: just return zeros (SLSQP will use finite diff)
    return None
