from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            if d > 0:
                h[np.argmin(h)] = min(1.0, h[np.argmin(h)] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
        else:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _obj(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2

    c = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(c)[-K:]
    return smooth_max(c[idx], beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2

            best_v = orig
            best_f = _obj(h, n, target_sum, 200.0, penalty)

                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                bs = np.random.randint(0, n - blen)
                h[bs:bs + blen] = np.random.uniform(0.2, 0.8, blen)
            elif strat == 3:
                w = np.random.uniform(2, 8)
                h += 0.02 * np.sin(w * np.arange(n) * 2.0 / n + np.random.uniform(0, 6.28))
            elif strat == 4:
                # Antisymmetric perturbation (helps break symmetry)
                h[:n // 2] += 0.03 * np.random.randn(n // 2)
                h[n // 2:] -= 0.03 * np.random.randn(n - n // 2)
            elif strat == 5:
                # Block swap
                blen = np.random.randint(5, max(6, n // 4))
                b1 = np.random.randint(0, n - 2 * blen)
                b2 = b1 + np.random.randint(blen + 1, n - b1 - blen + 1)
                h[b1:b1 + blen], h[b2:b2 + blen] = h[b2:b2 + blen].copy(), h[b1:b1 + blen].copy()
            elif strat == 6:
                # Push toward binary
                h = np.where(h > 0.5, np.minimum(1.0, h + 0.1), np.maximum(0.0, h - 0.1))
                h += np.random.randn(n) * 0.01
            else:
                # Randomize a fraction
                frac = np.random.uniform(0.2, 0.4)
                mask = np.random.rand(n) < frac
                h[mask] = np.random.uniform(0, 1, mask.sum())

            h = project_box_sum(h, ts)
            for beta, lam in [(300, 200), (600, 400)]:
                if tl() < 25:
                    break
                try:
                    r = minimize(_obj, h, args=(n, ts, beta, lam),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11})
                    hn = project_box_sum(r.x, ts)
                    cn = compute_c5(hn)
                    if cn < best_c5:
                        best_c5, best_h = cn, hn
                        h = hn.copy()
                except Exception:
                    pass

        # Phase 6: Final coordinate descent + high-beta polish
        if tl() > 30:
            hn = coord_descent(best_h, n, ts, sweeps=15, inner_iters=12, penalty=1000.0)
            cn = compute_c5(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

        # Phase 7: Ultra-high precision L-BFGS-B
        for beta, lam in [(2000, 2000), (4000, 4000)]:
            if tl() < 12:
                break
            try:
                r = minimize(_obj, best_h, args=(n, ts, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14})
                hn = project_box_sum(r.x, ts)
                cn = compute_c5(hn)
                if cn < best_c5:
                    best_c5, best_h = cn, hn
            except Exception:
                pass

    else:
        # No initial values: try multiple n
        for n in [70, 100]:
            if tl() < 80:
                break
            ts = n / 2.0
            bnds = [(0.0, 1.0)] * n
            inits = [_blocky(n, b, seed=b) for b in [3, 4, 5, 6, 8]]
            inits += [_sine_init(n, f, 0.3, 0, ts) for f in [1, 2, 3, 4]]
            inits.append(project_box_sum(np.random.uniform(0, 1, n), ts))

            for h0 in inits:
                if tl() < 30:
                    break
                c0 = compute_c5(h0)
                if c0 < best_c5:
                    best_c5, best_h, best_n = c0, h0, n
                for beta, lam, mi in [(100, 80, 400), (400, 300, 600)]:
                    if tl() < 15:
                        break
                    try:
                        r = minimize(_obj, h0, args=(n, ts, beta, lam),
                                     method='L-BFGS-B', bounds=bnds,
                                     options={'maxiter': mi, 'ftol': 1e-15})
                        hn = project_box_sum(r.x, ts)
                        cn = compute_c5(hn)
                        if cn < best_c5:
                            best_c5, best_h, best_n = cn, hn, n
                            h0 = hn.copy()
                    except Exception:
                        pass

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
