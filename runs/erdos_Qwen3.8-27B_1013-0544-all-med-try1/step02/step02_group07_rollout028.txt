So: df(k)/dh[m] = (1-h[m-k'])*(k' <= m <= n-1) - h[m+k']*(m <= n-1-k')  [for k < 0, k'=-k>0]

def compute_grad_active(h, active_lags):
    n = len(h)
    grad = np.zeros(n)
    for k in active_lags:
        if k >= 0:
            # df/dh[m] = (1-h[m+k]) for m=0..n-1-k, minus h[m-k] for m=k..n-1
            g = np.zeros(n)
            g[:n-k] += (1.0 - h[k:])  # (1-h[m+k]) for m=0 to n-1-k
            g[k:] -= h[:n-k]          # -h[m-k] for m=k to n-1
        else:
            kpn = -k  # k' = -k > 0
            g = np.zeros(n)
            g[kpn:] += (1.0 - h[:n-kpn])  # (1-h[m-k']) for m=k' to n-1
            g[:n-kpn] -= h[kpn:]          # -h[m+k'] for m=0 to n-1-k'
        grad += g
    grad /= len(active_lags)
    return grad

def grad_overlap_active(h, active_idx):

    n = len(h)

        g = np.zeros(n)

            g[:n-kp] -= h[kp:]
