def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    sw = exps.sum()
    val = (m + np.log(sw)) / beta
    w = exps / sw
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        k_idx = s + n - 1
        wk = w[k_idx] * dx
        if wk < 1e-13:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    return val, grad


def _lbfgs_obj(x, beta, mu, n):
    target = n / 2.0
    v, _ = smooth_max_val_grad(x, beta)
    diff = x.sum() - target
    return v + 0.5 * mu * diff * diff


def _lbfgs_jac(x, beta, mu, n):
    target = n / 2.0
    _, g = smooth_max_val_grad(x, beta)
    diff = x.sum() - target
    return g + mu * diff * np.ones(n)


def lbfgs_optimize(h0, beta, mu, maxiter):
    n = len(h0)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(_lbfgs_obj, h0, args=(beta, mu, n), jac=_lbfgs_jac,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'maxls': 50, 'maxfun': 10 * maxiter})
    return project_feasible(res.x, target)


def topk_subgrad_step(h, top_k, lr):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        s = k_idx - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 - s:hi2 - s]
    grad = grad / top_k * dx
    h_new = project_feasible(h - lr * grad, target)
    return h_new


def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n_best = len(h_best)

    # Phase 1: Beta annealing with L-BFGS-B at current resolution
    for beta in [30, 50, 80, 120, 180, 250, 350]:
        if tl() < 80:
            break
        mu = 20.0
        h_try = lbfgs_optimize(h_best, beta, mu, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_best, best_c5 = h_try, c5_try

    # Phase 2: Fine-tune with high beta
    if tl() > 80:
        for beta in [400, 600]:
            if tl() < 60:
                break
            h_try = lbfgs_optimize(h_best, beta, 50.0, maxiter=500)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_best, best_c5 = h_try, c5_try

    # Phase 3: Top-k subgradient refinement (handles sharp features)
    if tl() > 60:
        h_sub = h_best.copy()
        lr = 0.04
        for it in range(3000):
            if tl() < 30:
                break
            h_sub = topk_subgrad_step(h_sub, top_k=5, lr=lr)
            lr *= 0.9997
            c5_sub = compute_c5(h_sub)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                h_best = h_sub.copy()
        # L-BFGS polish after subgradient
        if tl() > 40:
            h_try = lbfgs_optimize(h_best, 200, 30.0, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_best, best_c5 = h_try, c5_try

    # Phase 4: Random restarts to escape local minima
    n_cur = len(h_best)
    for trial in range(8):
        if tl() < 100:
            break
        scale = 0.03 * (1.0 + 0.5 * trial)
        noise = np.random.randn(n_cur) * scale
        h_pert = project_feasible(h_best + noise, n_cur / 2.0)
        h_try = lbfgs_optimize(h_pert, 100, 30.0, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_best, best_c5 = h_try, c5_try
            n_cur = len(h_best)

    # Phase 5: Coarser resolution (might find better structure)
    if tl() > 150:
        h_coarse = interp_to_n(h_best, 100)
        for beta in [50, 150, 300]:
            if tl() < 100:
                break
            h_try = lbfgs_optimize(h_coarse, beta, 30.0, maxiter=400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_best, best_c5 = h_try, c5_try
                h_coarse = h_best.copy()

    # Phase 6: Higher resolution
    if tl() > 200:
        h_fine = interp_to_n(h_best, 300)
        for beta in [50, 150, 300]:
            if tl() < 120:
                break
            h_try = lbfgs_optimize(h_fine, beta, 30.0, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_best, best_c5 = h_try, c5_try
                h_fine = h_best.copy()

    # Phase 7: Final high-precision polish at best resolution
    if tl() > 60:
        n_fin = len(h_best)
        if n_fin > 400:
            h_best = interp_to_n(h_best, 300)
            n_fin = 300
        h_final = lbfgs_optimize(h_best, 500, 80.0, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            h_best, best_c5 = h_final, c5_final
        # One more subgradient pass
        if tl() > 30:
            h_sg = h_best.copy()
            lr = 0.02
            for it in range(2000):
                if tl() < 20:
                    break
                h_sg = topk_subgrad_step(h_sg, top_k=3, lr=lr)
                lr *= 0.9998
                c5_sg = compute_c5(h_sg)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    h_best = h_sg.copy()
            h_try = lbfgs_optimize(h_best, 500, 80.0, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_best, best_c5 = h_try, c5_try

    return h_best, best_c5, len(h_best)
