import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    smooth_val = m + log_sum / beta
    weights = exp_shifted / exp_shifted.sum()
    return smooth_val, weights

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        lag = j - (n - 1)
        # dcorr[j]/dh[i] = (1-h[i-lag])*[0<=i-lag<n] - h[i+lag]*[0<=i+lag<n]
        if lag >= 0:
            # i-lag valid: i from lag to n-1
            grad[lag:] += w * (1.0 - h[:n - lag])
            # i+lag valid: i from 0 to n-1-lag
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            # i-lag = i+l valid: i from 0 to n-1-l
            grad[:n - l] += w * (1.0 - h[l:])
            # i+lag = i-l valid: i from l to n-1
            grad[l:] -= w * h[:n - l]
    return grad
