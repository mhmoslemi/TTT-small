def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_init, n, n_iters, seed=0):
    """Simulated annealing for binary arrangement"""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    corr = compute_corr(h.astype(float))
    curr_max = np.max(corr)
    best_h = h.copy()
    best_max = curr_max
    
    T = 3.0
    cool = (0.05 / T) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h.astype(float))
        new_max = np.max(new_corr)
        
        delta = new_max - curr_max
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            curr_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
        
        T *= cool
    
    return best_h

def greedy_refine(h, n, max_rounds=50):
    """Greedy: try all swaps, keep the best"""
    h = h.copy()
    corr = compute_corr(h.astype(float))
    curr_max = np.max(corr)
    best_h = h.copy()
    best_max = curr_max
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for round_ in range(max_rounds):
        improved = False
        best_delta = 0
        best_swap = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                new_corr = compute_corr(h.astype(float))
                new_max = np.max(new_corr)
                delta = curr_max - new_max
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    best_new_max = new_max
                    best_new_corr = new_corr.copy()
                h[i], h[j] = h[j], h[i]
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            curr_max = best_new_max
            corr = best_new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            best_h = h.copy()
            best_max = curr_max
            improved = True
        
        if not improved:
            break
    
    return best_h, best_max * 2.0 / n

def lp_relax(h_bin, n, target_sum):
    """LP relaxation around binary point"""
    corr_bin = compute_corr(h_bin.astype(float))
    # Linearize: O(lag) ≈ O_bin(lag) + grad_lag · (h - h_bin)
    # For each lag m (index in full corr array), grad[m, i] = d O(m) / d h[i]
    
    n_lags = len(corr_bin)
    # Compute gradient for each lag
    # O(m) where lag = m - (n-1)
    # dO/dh[i] = (1-h[i+lag])*[valid] - h[i-lag]*[valid]
    
    # For efficiency, compute the gradient matrix
    # grad[m, i] for m in range(n_lags), i in range(n)
    lags = np.arange(n_lags) - (n - 1)  # actual lag values
    
    # For each position i and each lag, compute the gradient contribution
    # dO(lag)/dh[i] = (1-h[i+lag])*[0<=i+lag<n] - h[i-lag]*[0<=i-lag<n]
    
    # Build the constraint matrix A where A[lag, i] = dO(lag)/dh[i]
    A = np.zeros((n_lags, n))
    for m in range(n_lags):
        lag = lags[m]
        for i in range(n):
            g = 0.0
            if 0 <= i + lag < n:
                g += (1.0 - h_bin[i + lag])
            if 0 <= i - lag < n:
                g -= h_bin[i - lag]
            A[m, i] = g
    
    # LP: minimize t
    # s.t. corr_bin[m] + A[m, :] @ eps <= t for all m
    #      sum(eps) = 0
    #      -h_bin[i] <= eps[i] <= 1 - h_bin[i]
    
    # Variables: eps[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A @ eps - t <= -corr_bin
    # i.e., A @ eps + (-1)*t <= -corr_bin
    A_ub = np.column_stack([A, -np.ones(n_lags)])
    b_ub = -corr_bin
    
    # Equality: sum(eps) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h_bin[i] <= eps[i] <= 1-h_bin[i], t >= 0
    bounds = [(-h_bin[i], 1.0 - h_bin[i]) for i in range(n)]
    bounds.append((0, None))
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        eps = res.x[:n]
        h_new = h_bin.astype(float) + eps
        h_new = project_h(h_new, target_sum)
        c5 = compute_c5(h_new)
        return h_new, c5
    return None, None
