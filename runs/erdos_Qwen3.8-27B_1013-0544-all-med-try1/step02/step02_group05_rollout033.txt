            t_sub = min(time.time() + time_per / max(1, len(starts)) * 1.6, t_limit)
            best_h, best_c5, best_n = optimize_resolution(h0, n, t_sub, best_h, best_c5, best_n)

    # Final high-precision polish
    if best_h is not None and time.time() < t_start + budget_s - 15:
        n_f = best_n
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        cons_f = make_constraint(target_f)

        for beta in [400, 600, 1000]:
            if time.time() > t_start + budget_s - 8:
                break
            obj = ObjSmooth(dx_f, beta)
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=[cons_f], options={'maxiter': 2000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

        c = compute_corr(best_h)
        sidx = np.argsort(c)[::-1]
        for top_k in [2, 3]:
            if time.time() > t_start + budget_s - 5:
                break
            obj = ObjActive(dx_f, sidx[:top_k], 500)
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=[cons_f], options={'maxiter': 1500, 'ftol': 1e-16})
                h_new = project_h(res.x, target_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

        h_bin = try_binarize(best_h, target_f)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
            for beta in [200, 400]:
                if time.time() > t_start + budget_s - 3:
                    break
                obj = ObjSmooth(dx_f, beta)
                try:
                    res = minimize(obj, best_h, method='SLSQP', bounds=bounds_f,
                                   constraints=[cons_f], options={'maxiter': 1000, 'ftol': 1e-16})
                    h_new = project_h(res.x, target_f)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new, c5
                except Exception:
                    pass

    return best_h, best_c5, best_n
