                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 3 * pop_size:
            # Inject diversity
            for k in range(3):
                idx = int(np.argmax(fit))
                pop[idx] = project_feasible(rng.uniform(0, 1, n), n)
                fit[idx] = compute_c5(pop[idx], n)
            stagnant = 0
    
    return best_h, best_f

def two_point_polish(h, n, t0, budget):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    for step in steps:
        if budget - (time.time() - t0) < 15:
            break
        for _pass in range(30):
            if budget - (time.time() - t0) < 10:
                break
            improved = False
            for i in range(n):
                if budget - (time.time() - t0) < 10:
                    break
                for j in range(i + 1, n):
                    if budget - (time.time() - t0) < 10:
                        break
                    for sgn in [1.0, -1.0]:
                        hi = best_h[i] + sgn * step
                        hj = best_h[j] - sgn * step
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            oi, oj = best_h[i], best_h[j]
                            best_h[i], best_h[j] = hi, hj
                            c = compute_c5(best_h, n)
                            if c < best_c5:
                                best_c5 = c
                                improved = True
                            else:
                                best_h[i], best_h[j] = oi, oj
            if not improved:
                break
    return best_h, best_c5
