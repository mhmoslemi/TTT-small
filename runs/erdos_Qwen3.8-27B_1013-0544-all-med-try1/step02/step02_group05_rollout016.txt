import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_fixed_v(v, n, target_sum):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [sum(h_var) == target_sum, h_var >= 0, h_var <= 1]
    for k in range(2 * n - 1):
        offset = k - n + 1
        coeff = np.zeros(n)
        for j in range(n):
            vi = j - offset
            if 0 <= vi < n:
                coeff[j] = v[vi]
        if coeff.sum() > 1e-15:
            cons.append(cp.sum(h_var * coeff) <= t_var / dx)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.asarray(h_var.value, dtype=float)

def solve_lp_fixed_h(h, n, target_sum):
