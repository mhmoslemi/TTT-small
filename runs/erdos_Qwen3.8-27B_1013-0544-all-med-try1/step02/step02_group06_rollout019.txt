import numpy as np

import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _objective_fn(h, n, ts, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - ts) ** 2

    n = len(h)

            ak += (1.0 - h[j + k])
        if 0 <= j - k < n:

def coord_search(h, j):
    """Exact 1D minimization of max correlation by varying h[j]"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = 2 * n - 1
    a = np.zeros(m)
    jp = j  # shorthand
    for idx in range(m):
        k = idx - (n - 1)
        ak = 0.0
        if 0 <= jp + k < n:
            ak += (1.0 - h[jp + k])
        if 0 <= jp - k < n:
            ak -= h[jp - k]
        a[idx] = ak
    b = corr - a * h[j]
    # Collect breakpoints
    cands = [0.0, 1.0]
    for i in range(m):
        ai = a[i]
        bi = b[i]
        for j2 in range(i + 1, m):
            da = ai - a[j2]
            if abs(da) > 1e-14:
                t = (b[j2] - bi) / da
                if 0.0 < t < 1.0:
                    cands.append(t)
    best_t = h[j]
    best_val = np.max(a * best_t + b)
    for t in cands:
        val = np.max(a * t + b)
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    return best_t


def coord_descent(h, ts, max_passes=300):
    """Coordinate descent with exact 1D search + projection"""
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    no_improve = 0
    for p in range(max_passes):
        improved = False
        for j in range(n):
            h_cur[j] = coord_search(h_cur, j)
            h_cur = project_box_sum(h_cur, ts)
            c5 = compute_c5(h_cur)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    h_cur = project_box_sum(h_cur, ts)
    return h_cur, compute_c5(h_cur)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B warm-up with increasing beta
    for beta, lam, maxit in [(50, 50, 200), (200, 150, 300), (500, 300, 400)]:
        if tl() < 40:
            break
        try:
            r = minimize(_obj_fn, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 2: Coordinate descent (exact 1D subproblem)
    if tl() > 60:
        h_cd, c_cd = coord_descent(best_h, ts, max_passes=200)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 3: High-beta L-BFGS-B refinement
    for beta, lam, maxit in [(1500, 800, 600), (4000, 2000, 800)]:
        if tl() < 30:
            break
        try:
            r = minimize(_obj_fn, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 4: More coordinate descent
    if tl() > 40:
        h_cd, c_cd = coord_descent(best_h, ts, max_passes=300)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 5: Diverse restarts from best with perturbation + coord descent
    it = 0
    while tl() > 50:
        it += 1
        h = best_h.copy()
        scale = max(0.003, 0.03 * np.exp(-it / 10.0))
        h += np.random.randn(n) * scale
        h = project_box_sum(h, ts)
        # Quick L-BFGS-B
        try:
            r = minimize(_obj_fn, h, args=(n, ts, 1000, 500),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12})
            h = project_box_sum(r.x, ts)
        except Exception:
            pass
        # Coordinate descent from here
        if tl() > 30:
            h_cd, c_cd = coord_descent(h, ts, max_passes=100)
            if c_cd < best_c5:
                best_h, best_c5 = h_cd, c_cd

    # Phase 6: Final ultra-precise coordinate descent
    if tl() > 15:
        h_cd, c_cd = coord_descent(best_h, ts, max_passes=500)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 7: Final L-BFGS-B polish
    if tl() > 10:
        try:
            r = minimize(_obj_fn, best_h, args=(n, ts, 20000, 10000),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Final coordinate descent polish
    if tl() > 5:
        h_cd, c_cd = coord_descent(best_h, ts, max_passes=200)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
