import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exp_s = np.exp(beta * shifted)
    w = exp_s / exp_s.sum()
    val = (m + np.log(exp_s.sum()) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif 0 < d < n:
            grad[:n-d] += w[k] * (1.0 - h[d:])
            grad[d:] -= w[k] * h[:n-d]
        elif -n < d < 0:
            dp = -d
            grad[dp:] += w[k] * (1.0 - h[:n-dp])
            grad[:n-dp] -= w[k] * h[dp:]
    return val * 1.0, grad * dx

def con_fun(x):
    return np.sum(x) - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g = 1.0 - 2.0 * h
    elif 0 < d < n:
        g[:n-d] += 1.0 - h[d:]
        g[d:] -= h[:n-d]
    elif -n < d < 0:
        dp = -d
        g[dp:] += 1.0 - h[:n-dp]
        g[:n-dp] -= h[dp:]
    h_new = h - alpha * g
    return project_h(h_new, n / 2.0)

def coordinate_polish(h, target_sum, max_rounds=200):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            save = h_cur[j]
            for trial in [0.0, 1.0, save + 0.05, save - 0.05, save + 0.02, save - 0.02]:
                if trial < 0 or trial > 1:
                    continue
                h_cur[j] = trial
                h_try = project_h(h_cur, target_sum)
                c = compute_c5(h_try)
                if c < c5_cur - 1e-12:
                    c5_cur = c
                    h_cur = h_try
                    improved = True
                    break
            h_cur[j] = save
        if not improved:
            break
    return project_h(h_cur, target_sum), c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:

    candidates = []
    # Add initial
    n0 = len(h_init)
    h0 = project_h(h_init, n0 / 2.0)
    candidates.append(h0)

    # Try other n values with structured init
    for n_try in [60, 80, 90, 100, 120]:
        if tleft() < 20:
            break
        alpha = (np.sqrt(5) - 1) / 2.0
        h_struct = np.array([1.0 if ((i * alpha) % 1.0) < 0.5 else 0.0 for i in range(n_try)])
        h_struct = project_h(h_struct, n_try / 2.0)
        candidates.append(h_struct)
        alpha2 = np.sqrt(2.0) / 2.0
        h_struct2 = np.array([1.0 if ((i * alpha2) % 1.0) < 0.5 else 0.0 for i in range(n_try)])
        h_struct2 = project_h(h_struct2, n_try / 2.0)
        candidates.append(h_struct2)

    best_h = None
    best_c5 = float('inf')

    for h_cand in candidates:
        if tleft() < 60:
            break
        n = len(h_cand)
        ts = n / 2.0
        h_cur = project_h(h_cand, ts)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

        # SLSQP with exact gradient, sweeping beta
        bounds = Bounds(np.zeros(n), np.ones(n))
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        for beta in [40, 80, 150, 300, 600, 1200, 2500]:
            if tleft() < 20:

            def obj(x):
                return smooth_max_and_grad(x, beta)[0]

                if c5_sub < c5_cur:
                    h_cur, c5_cur = h_sub, c5_sub
                alpha *= 0.98

        # Coordinate polish
        if tleft() > 20:

            best_h, best_c5 = h_cur.copy(), c5_cur

    # Final intensive polish on best solution
    n_best = len(best_h)
    ts_best = n_best / 2.0
    bounds = Bounds(np.zeros(n_best), np.ones(n_best))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        try:
            res = minimize(obj_final, best_h, jac=grd_final, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, ts_best)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Coordinate polish on final
    if tleft() > 10:
        best_h, best_c5 = coordinate_polish(best_h, ts_best, max_rounds=200)

    # Subgradient final
    if tleft() > 10:
        alpha = 0.01
        for it in range(200):

            h_sub = subgradient_step(best_h, alpha)

            alpha *= 0.97

    best_h = project_h(best_h, ts_best)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
