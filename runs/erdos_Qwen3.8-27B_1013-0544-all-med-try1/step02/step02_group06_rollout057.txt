                h[i], h[j] = h[j], h[i]
        elif s == 6:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    exps = np.exp(beta * (c - m))
    s = exps.sum()
    probs = exps / s
    val = (m + np.log(s) / beta) * dx
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        w = probs[k + n - 1] * dx
        if w < 1e-18:
            continue
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            g[lo1:hi1] += w * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            g[lo2:hi2] += w * (-h[lo2 - k:hi2 - k])
    return val, g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    def polish(h, beta, maxiter=400):
        def fun(x):
            val, grad = smooth_max_val_grad(x, beta)
            pen = x.sum() - ts
            val += 200.0 * pen * pen
            grad += 400.0 * pen * np.ones(n)
            return val, grad
        try:
            r = minimize(fun, h, jac=True, method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': maxiter, 'ftol': 1e-18})
            return project_box_sum(r.x, ts)
        except Exception:
            return h

    # Phase 1: Annealed L-BFGS-B with mirror symmetry
    for beta in [30, 80, 200, 500, 1500, 4000, 10000]:
        if tl() < 20:
            break
        h = polish(best_h, beta)
        c = compute_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c
        if tl() < 15:
            continue
        h_mirror = best_h[::-1].copy()
        h_mirror = project_box_sum(h_mirror, ts)
        h_m = polish(h_mirror, beta)
        c_m = compute_c5(h_m)
        if c_m < best_c5:
            best_h, best_c5 = h_m, c_m

    # Phase 2: SLSQP with hard equality constraint
    if tl() > 40:
        for beta in [5000, 15000]:
            if tl() < 15:
                break
            try:
                def fun_s(x):
                    val, grad = smooth_max_val_grad(x, beta)
                    return val, grad
                cons = [{'type': 'eq',
                         'fun': lambda x: x.sum() - ts,
                         'jac': lambda x: np.ones(n)}]
                r = minimize(fun_s, best_h, jac=True, method='SLSQP',
                            bounds=bnds, constraints=cons,

                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)

            except Exception:

    # Phase 3: Binary swap local search

    h_bin = project_box_sum(h_bin, ts)

                    else:
                        h_s[i], h_s[j] = h_s[j], h_s[i]

    # Re-optimize from best binary start
    h_start = h_s.copy() if c_bin < best_c5 else best_h.copy()
    for beta in [3000, 10000, 30000]:

    rc = 0
    while tl() > 60:

        h = best_h.copy()

        elif s == 4:
            mask = np.random.random(n) < 0.25
            h[mask] = 1.0 - h[mask]
        elif s == 5:
            for _ in range(np.random.randint(3, 10)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
        elif s == 6:
            # Random column permutation
            perm = np.random.permutation(n)
            h = best_h[perm]
        else:
            h = np.random.uniform(0, 1, n)

        h = project_box_sum(h, ts)
        c = compute_c5(h)
        if c < best_c5:
