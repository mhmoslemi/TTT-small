The variable h[j] appears in this sum in two ways:
1. As h[i + lag_k] where i + lag_k = j, i.e., i = j - lag_k. Valid if 0 ≤ j-lag_k < n.
   Contribution: (1 - h[j-lag_k])
2. As (1 - h[i]) where i = j. Valid if 0 ≤ j < n (always true) AND j+lag_k is valid (0 ≤ j+lag_k < n).
   Contribution: -h[j+lag_k]

So: d corr[k] / d h[j] = (1-h[j-lag_k])·[0≤j-lag_k<n] - h[j+lag_k]·[0≤j+lag_k<n]

For a fixed j and all k (i.e., all lags):
lag ranges from -(n-1) to n-1.
- j - lag is valid when 0 ≤ j-lag ≤ n-1, i.e., j-(n-1) ≤ lag ≤ j
- j + lag is valid when 0 ≤ j+lag ≤ n-1, i.e., -j ≤ lag ≤ n-1-j

So:
d corr[k] / d h[j] = (1-h[j-lag])·[max(-(n-1),j-(n-1)) ≤ lag ≤ min(n-1,j)]

def coord_desc(h0, beta, gamma, max_sweeps, time_s, seed=0):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    t_start = time.time()
    n_corr = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    idx = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        sum_h = h.sum()
        for j in range(n):
            valid1 = (j - lags >= 0) & (j - lags < n)
            valid2 = (j + lags >= 0) & (j + lags < n)
            g = np.zeros(n_corr)
            if np.any(valid1):
                g[valid1] = 1.0 - h[j - lags[valid1]]
            if np.any(valid2):
                g[valid2] -= h[j + lags[valid2]]

            hi = 1.0 - h[j]
            u = 0.0
            for _ in range(12):
                val = corr + u * g
                m = val.max()
                sh = np.clip(beta * (val - m), -500, 500)

                hess = beta * dx * (pg2 - pg * pg) + 2.0 * gamma
                if hess < 1e-15:

                if abs(u_new - u) < 1e-13:
                    u = u_new
                    break
                u = u_new
            u = max(lo, min(hi, u))
            if abs(u) > 1e-14:
                corr += u * g
                h[j] += u
                sum_h += u

def topk_subgrad(h0, top_k, n_iter, lr_init, time_s):

    target = n / 2.0

        w /= w.sum()
        grad = np.zeros(n)
        for i in range(top_k):
