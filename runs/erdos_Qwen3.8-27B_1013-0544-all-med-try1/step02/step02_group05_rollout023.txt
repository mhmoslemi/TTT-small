        if curr_c5 < best_c5:
            best_h, best_c5, best_n = curr_h.copy(), curr_c5, n_points

        # Phase 1: SLSQP smooth max with increasing beta
        for beta in [50, 100, 200, 400]:
            if remaining() < 25:
                break
            _state['beta'] = float(beta)
            _state['dx'] = dx
            _state['target_sum'] = target_sum
            try:
                res = minimize(_obj_smooth, curr_h, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < curr_c5 - 1e-10:
                    curr_h, curr_c5 = h_new.copy(), c5
                    if curr_c5 < best_c5:
                        best_h, best_c5, best_n = curr_h.copy(), c5, n_points
            except Exception:
                pass

        # Phase 2: Active-lag focused optimization
        if remaining() > 40:
            for n_active in [3, 5, 8, 12]:
                if remaining() < 20:
                    break
                for beta in [200, 400, 800]:
                    if remaining() < 12:
                        break
                    corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
                    active_idx = np.argsort(corr)[::-1][:n_active]
                    _state['active_idx'] = active_idx
                    _state['beta'] = float(beta)
                    _state['dx'] = dx
                    _state['target_sum'] = target_sum
                    try:
                        res = minimize(_obj_active, curr_h, method='SLSQP', bounds=bounds,
                                       constraints=cons,
                                       options={'maxiter': 400, 'ftol': 1e-14})
                        h_new = project_h(res.x, target_sum)
                        c5 = compute_c5(h_new)
                        if c5 < curr_c5 - 1e-10:
                            curr_h, curr_c5 = h_new.copy(), c5
                            if curr_c5 < best_c5:
                                best_h, best_c5, best_n = curr_h.copy(), c5, n_points
                    except Exception:
                        pass

        # Phase 3: Weighted lag optimization (emphasize near-max lags)
        if remaining() > 30:
            for n_w in [10, 15, 20]:
                if remaining() < 15:
                    break
                corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
                sorted_c = np.sort(corr)[::-1]
                thresh = sorted_c[min(n_w, len(sorted_c)-1)] * 0.95
                weights = np.where(corr >= thresh, 1.0, 0.01)
                _state['weights'] = weights
                _state['beta'] = 150.0
                _state['dx'] = dx
                _state['target_sum'] = target_sum
                try:
                    res = minimize(_obj_weighted, curr_h, method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-13})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5 - 1e-10:
                        curr_h, curr_c5 = h_new.copy(), c5
                        if curr_c5 < best_c5:
                            best_h, best_c5, best_n = curr_h.copy(), c5, n_points
                except Exception:
                    pass

        # Phase 4: Diversification with structured perturbations
        if remaining() > 30:
            base = curr_h.copy()
            for trial in range(10):
                if remaining() < 12:
                    break
                # Different perturbation strategies
                if trial < 3:
                    scale = 0.04
                    h_rand = base + np.random.randn(n_points) * scale
                elif trial < 6:
                    # Perturb only a random subset
                    h_rand = base.copy()
                    idx = np.random.choice(n_points, size=n_points//3, replace=False)
                    h_rand[idx] += np.random.randn(len(idx)) * 0.08
                else:
                    # Block perturbation
                    h_rand = base.copy()
                    block = n_points // 4
                    start = np.random.randint(0, n_points - block)
                    h_rand[start:start+block] += np.random.randn(block) * 0.06

                h_rand = project_h(h_rand, target_sum)
                _state['beta'] = 150.0
                _state['dx'] = dx
                _state['target_sum'] = target_sum
                try:
                    res = minimize(_obj_smooth, h_rand, method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 250, 'ftol': 1e-13})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5 - 1e-10:
                        curr_h, curr_c5 = h_new.copy(), c5
                        base = h_new.copy()
                        if curr_c5 < best_c5:
                            best_h, best_c5, best_n = curr_h.copy(), c5, n_points
                except Exception:
                    pass

        # Update h_init for next resolution
        h_init = curr_h.copy()

    # Final high-precision polish
    if best_h is not None and remaining() > 10:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        _state['target_sum'] = ts
        cons_f = make_constraints()

        for beta in [500, 800, 1200]:
            if remaining() < 6:
                break
            _state['beta'] = float(beta)
            _state['dx'] = dx_f
            _state['target_sum'] = ts
            try:
                res = minimize(_obj_smooth, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=cons_f,
                               options={'maxiter': 800, 'ftol': 1e-15})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

        # Active-lag final push
        if remaining() > 8:
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_k = 5
            active_idx = np.argsort(corr)[::-1][:top_k]
            _state['active_idx'] = active_idx
            _state['beta'] = 1000.0
            _state['dx'] = dx_f
            _state['target_sum'] = ts
            try:
                res = minimize(_obj_active, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=cons_f,
                               options={'maxiter': 1000, 'ftol': 1e-15})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    if best_h is None:
        n_f = 70
        best_h = np.full(n_f, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n_f

    return best_h, best_c5, best_n
