- If j is in the second index position: j = i+m, so i = j-m, j ranges from m to n-1, contributes -h[j-m]

So:
∂corr[k]/∂h[j] = (1-h[j+m]) * 1[j ≤ n-1-m] + (-h[j-m]) * 1[j ≥ m]

For m > 0:
- j in [0, n-1-m]: (1-h[j+m]) - h[j-m] (if j ≥ m, which is j in [m, n-1-m])
- j in [0, m-1]: (1-h[j+m])
- j in [n-m, n-1]: -h[j-m]
- j in [m, n-1-m]: (1-h[j+m]) - h[j-m]

g_k = convolve(1-h, window) + convolve(-h, shifted_window)

Where window is a vector of ones of length n-m.

import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def all_corr_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m = n - 1 - k
        if m == 0:
            grads[k] = 1.0 - 2.0 * h
        elif m > 0:
            grads[k, :n - m] += (1.0 - h[m:])
            grads[k, m:] += (-h[:n - m])
        else:
            m2 = -m
            grads[k, m2:] += (1.0 - h[:n - m2])
            grads[k, :n - m2] += (-h[m2:])
    return grads


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    grads = all_corr_grads(h)
    return (weights @ grads) * dx

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best = compute_c5(h_curr)
    best_h = h_curr.copy()
    alpha = 1.0 / (0.5 + beta * 0.01)
    for t in range(n_iter):
        momentum = t / (t + 3.0)
        y = h_curr + momentum * (h_curr - h_prev)
        y = np.clip(y, 0, 1)
        grad = smooth_max_grad(y, beta)
        h_new = y - alpha * grad
        h_new = project_h(h_new, target_sum)

            best_h = h_new.copy()
        h_prev = h_curr
        h_curr = h_new
    return best_h, best

    n = len(h_init)
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init]])

    def objective(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    constraints = []
    for k in active_lags:
        def make_constr(k):
            def fun(x):
                hv = x[:n]
                tv = x[-1]
                c = np.correlate(hv, 1.0 - hv, mode="full")
                return tv - c[k] * dx
            def jac(x):
                hv = x[:n]
                g = np.zeros(n + 1)
                g[:n] = -dx * all_corr_grads(hv)[k]
                g[-1] = 1.0
                return g
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        constraints.append(make_constr(k))

    def eq_fun(x):
        return x[:n].sum() - target_sum
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})

    bounds = [(0, 1)] * n + [(0, 2.0)]
    try:

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.x is not None:
            h_new = project_h(res.x[:n], target_sum)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h0.copy()

    # Phase 5: Final high-precision SLSQP polish
    if time_left() > 20:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_lags = sorted(np.argsort(corr)[::-1][:12].tolist())
        h_new, c5_new = active_set_solve(best_h, top_lags, target_sum, dx, maxiter=10000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        # One more round with all top lags
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_lags2 = sorted(np.argsort(corr)[::-1][:15].tolist())
        h_new, c5_new = active_set_solve(best_h, top_lags2, target_sum, dx, maxiter=10000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
