def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-11) & (h < 1.0 - 1e-11)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def con_fun(x):
    return np.sum(x) - len(x) / 2.0


def make_obj(beta):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta)
    return obj


def slsqp_run(h0, beta, bounds, constraints, maxiter, ftol):
    try:
        res = minimize(make_obj(beta), h0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0


def generate_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def generate_sine(n, rng):
    x = np.linspace(0, 2, n)
    freq = rng.uniform(3, 10)
    phase = rng.uniform(0, 2 * np.pi)
    return np.clip(0.5 + 0.4 * np.sin(freq * x + phase), 0, 1)


def sa_lag_target(h_cur, h_new, rng):
    n = len(h_cur)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    top_idx = np.argsort(corr)[-min(4, len(corr)):]
    for k in top_idx:
        lag = k - (n - 1)
        if lag > 0:
            n_pair = n - lag
            if n_pair < 2:
                continue
            count = min(4, n_pair)
            idx = rng.choice(n_pair, count, replace=False)
            for i in idx:
                a, b = h_cur[i], h_cur[i + lag]
                if a > b:
                    h_new[i] -= 0.04 * rng.random()
                    h_new[i + lag] += 0.04 * rng.random()
                else:
                    h_new[i] += 0.03 * rng.random()
                    h_new[i + lag] -= 0.03 * rng.random()
        elif lag < 0:
            da = -lag
            n_pair = n - da
            if n_pair < 2:
                continue
            count = min(4, n_pair)
            idx = rng.choice(n_pair, count, replace=False)
            for i in idx:
                a, b = h_cur[i + da], h_cur[i]
                if a > b:
                    h_new[i + da] -= 0.04 * rng.random()
                    h_new[i] += 0.04 * rng.random()
                else:
                    h_new[i + da] += 0.03 * rng.random()
                    h_new[i] -= 0.03 * rng.random()
        else:
            idx = rng.choice(n, min(5, n), replace=False)
            for i in idx:
                if h_cur[i] > 0.5:
                    h_new[i] -= 0.05
                else:
                    h_new[i] += 0.05


def sa_block_swap(h_cur, h_new, n, rng):
    blen = rng.randint(2, min(8, n // 8 + 1))
    i = rng.randint(0, n - blen + 1)
    j = rng.randint(0, n - blen + 1)
    tries = 0
    while abs(j - i) < blen and tries < 15:
        j = rng.randint(0, n - blen + 1)
        tries += 1
    if abs(j - i) >= blen:
        tmp = h_new[i:i + blen].copy()
        h_new[i:i + blen] = h_new[j:j + blen]
        h_new[j:j + blen] = tmp


def sa_gaussian(h_cur, h_new, n, rng, T, T0):
    frac = 0.1 + 0.3 * (T / T0)
    mask = rng.random(n) < frac
    n_mask = max(1, int(mask.sum()))
    h_new[mask] += rng.randn(n_mask) * (0.02 + 0.06 * T / T0)


def sa_pair_swap(h_cur, h_new, n, rng):
    i, j = rng.choice(n, 2, replace=False)
    h_new[i], h_new[j] = h_new[j], h_new[i]


def sa_smooth(h_cur, h_new, n, rng):
    idx = rng.randint(1, n - 1)
    h_new[idx] = 0.5 * h_cur[idx] + 0.25 * (h_new[idx - 1] + h_new[idx + 1])


def pair_flip_search(h, target_sum, n, dx):
    best_c5 = compute_c5(h)
    improved = True
    while improved:
        improved = False
        h_sorted = np.argsort(h)
        high_idx = h_sorted[-n // 4:]
        low_idx = h_sorted[:n // 4]
        for i in high_idx:
            for j in low_idx:
                if abs(h[i] - h[j]) < 0.1:
                    continue
                h_try = h.copy()
                h_try[i] = min(1.0, h_try[i] + 0.5)
                h_try[j] = max(0.0, h_try[j] - 0.5)
                h_try = project_h(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-7:
                    h, best_c5 = h_try, c5_try
                    improved = True
                    break
            if improved:
                break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    n_points = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constraints = [{'type': 'eq', 'fun': con_fun}]

    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP from diverse starts
    starts = [best_h.copy()]
    for i in range(6):
        if i < 3:
            h_try = generate_binary(n_points, rng)
        else:
            h_try = generate_sine(n_points, rng)
        h_try = project_h(h_try, target_sum)
        if compute_c5(h_try) < best_c5 + 0.03:
            starts.append(h_try)

    for h_start in starts[:4]:
        if tleft() < 60:
            break
        h_cur = h_start.copy()
        for beta in [30, 80, 200, 500, 1000]:
            if tleft() < 25:
                break
            h_new = slsqp_run(h_cur, beta, bounds, constraints, 2000, 1e-12)
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
            h_cur = h_new

    # Phase 2: Intensive SA
    if tleft() > 100:
        h_cur = best_h.copy()
        c5_cur = best_c5
        T0 = 0.006
        T = T0
        cooling = 0.9992
        no_improve = 0
        max_no_improve = 250

        while tleft() > 80:
            h_new = h_cur.copy()
            r = rng.random()
            if r < 0.30:
                sa_lag_target(h_cur, h_new, rng)
            elif r < 0.50:
                sa_block_swap(h_cur, h_new, n_points, rng)
            elif r < 0.72:
                sa_gaussian(h_cur, h_new, n_points, rng, T, T0)
            elif r < 0.85:
                sa_pair_swap(h_cur, h_new, n_points, rng)
            else:
                sa_smooth(h_cur, h_new, n_points, rng)

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or rng.random() < np.exp(-min(400.0, delta / max(T, 1e-12))):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T *= cooling
            if no_improve > max_no_improve:
                T = max(T, 0.004)
                no_improve = 0

            if no_improve > max_no_improve * 3:
                h_cur = best_h.copy()
                c5_cur = best_c5
                T = 0.007
                no_improve = 0

    # Phase 3: Pair flip search
    if tleft() > 40:
        best_h, best_c5 = pair_flip_search(best_h, target_sum, n_points, dx)
        # Another SLSQP after flip search
        for beta in [300, 800]:
            if tleft() < 20:
                break
            h_new = slsqp_run(best_h, beta, bounds, constraints, 3000, 1e-13)
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new

    # Phase 4: Multi-restart polish
    if tleft() > 40:
        for i in range(12):
            if tleft() < 20:
                break
            h_try = best_h.copy()
            n_pert = rng.randint(3, 10)
            idx = rng.choice(n_points, n_pert, replace=False)
            h_try[idx] += rng.randn(n_pert) * 0.025
            h_try = project_h(h_try, target_sum)
            for beta in [200, 500, 1200]:
                if tleft() < 15:
                    break
                h_new = slsqp_run(h_try, beta, bounds, constraints, 2000, 1e-13)
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
                    break

    # Final verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
