    for _ in range(50):

        vals = c * t + b
        mx = np.max(vals)
        # Subgradient (use average of active c's for smoothness)
        active = vals >= mx - 1e-10
        sg = np.mean(c[active])
        if sg > 0:
            hi = t
        elif sg < 0:
            lo = t
        else:

    return (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def cd_sweep(h, f_all, k_shift, n, lam, order):
    for j in order:
        c = np.zeros(2*n-1)
        ma = (j - k_shift >= 0) & (j - k_shift < n)
        if ma.any():
            c[ma] += (1.0 - h[j - k_shift[ma]])
        mb = (j + k_shift >= 0) & (j + k_shift < n)
        if mb.any():
            c[mb] -= h[j + k_shift[mb]]
        old = h[j]
        b = f_all - c * old
        lo, hi = 0.0, 1.0
        for _ in range(50):
            t = (lo + hi) / 2.0
            vals = c * t + b
            mx = np.max(vals)
            active = vals >= mx - 1e-14
            sg = np.mean(c[active]) + lam
            if sg > 1e-14:
                hi = t
            elif sg < -1e-14:
                lo = t
            else:
                break
        new = (lo + hi) / 2.0
        f_all += c * (new - old)
        h[j] = new
    return h, f_all

def coord_descent(h, n, n_sweeps, lam=0.0, project_every=50):
    f_all = full_corr(h)
    k_shift = np.arange(2*n-1) - (n-1)
    target = n / 2.0
    for s in range(n_sweeps):
        order = np.random.permutation(n)
        h, f_all = cd_sweep(h, f_all, k_shift, n, lam, order)
        if (s + 1) % project_every == 0:
            h = project_box_sum(h, target)
            f_all = full_corr(h)
    return h

def block_cd(h, n, n_sweeps, block_size=2):
    """Block coordinate descent with 2D grid refinement."""
    target = n / 2.0
    k_shift = np.arange(2*n-1) - (n-1)
    for s in range(n_sweeps):
        np.random.shuffle(h) if False else None
        blocks = [np.random.permutation(n) for _ in range(1)]
        for j in blocks[0]:
            if block_size == 1:
                continue
            j2 = (j + 1) % n
            if j == j2:
                continue
            f_all = full_corr(h)
            c = np.zeros(2*n-1)
            for jj in [j, j2]:
                ma = (jj - k_shift >= 0) & (jj - k_shift < n)
                if ma.any():
                    c[ma] += (1.0 - h[jj - k_shift[ma]])
                mb = (jj + k_shift >= 0) & (jj + k_shift < n)
                if mb.any():
                    c[mb] -= h[jj + k_shift[mb]]
            # 2D grid search
            old_j, old_j2 = h[j], h[j2]
            grid = np.linspace(0, 1, 15)
            best_t = (old_j, old_j2)
            best_val = float(np.max(full_corr(h)))
            for t1 in np.concatenate([[old_j], grid]):
                for t2 in np.concatenate([[old_j2], grid]):
                    h[j], h[j2] = t1, t2
                    v = float(np.max(full_corr(h)))
                    if v < best_val - 1e-12:
                        best_val = v
                        best_t = (t1, t2)
            h[j], h[j2] = best_t
            if s % 30 == 0:
                h = project_box_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    target = n / 2.0
    best_h = project_box_sum(h_init, target)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Aggressive coordinate descent from initial
    h = best_h.copy()
    h = coord_descent(h, n, 2000)
    h = project_box_sum(h, target)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 2: Main loop with diverse restarts
    restart_count = 0
    while tl() > 50:
        strat = restart_count % 8
        h = best_h.copy()

        if strat == 0:
            # Pure CD
            h = coord_descent(h, n, 500)
        elif strat == 1:
            # Small noise + CD
            h += np.random.randn(n) * 0.02
            h = project_box_sum(h, target)
            h = coord_descent(h, n, 500)
        elif strat == 2:
            # Larger noise + CD
            h += np.random.randn(n) * 0.08
            h = project_box_sum(h, target)
            h = coord_descent(h, n, 400)
        elif strat == 3:
            # Binary restart
            h_new = np.zeros(n)
            ones = np.random.choice(n, n // 2, replace=False)
            h_new[ones] = 1.0
            h_new += np.random.randn(n) * 0.05
            h = project_box_sum(h_new, target)
            h = coord_descent(h, n, 500)
        elif strat == 4:
            # Block CD
            h = coord_descent(h, n, 200)
            h = block_cd(h, n, 200, block_size=2)
            h = coord_descent(h, n, 300)
        elif strat == 5:
            # Sinusoidal perturbation
            x = np.arange(n) * 2.0 / n
            w = np.random.uniform(2, 8) * np.pi
            ph = np.random.uniform(0, 2 * np.pi)
            h += 0.05 * np.sin(w * x + ph)
            h = project_box_sum(h, target)
            h = coord_descent(h, n, 500)
        elif strat == 6:
            # Inversion perturbation (flip a random block)
            bsz = np.random.randint(5, max(6, n // 4))
            bs = np.random.randint(0, n - bsz)
            h[bs:bs+bsz] = 1.0 - h[bs:bs+bsz]
            h = project_box_sum(h, target)
            h = coord_descent(h, n, 400)
        else:
            # Swap-based
            for _ in range(np.random.randint(3, 12)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h = coord_descent(h, n, 400)

        h = project_box_sum(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        restart_count += 1

    # Phase 3: SLSQP polish
    if tl() > 20:
        def obj(x):
            return float(np.max(full_corr(x)))
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            r = minimize(obj, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 4: Final CD
    if tl() > 10:
        h_new = coord_descent(best_h.copy(), n, 2000)
        h_new = project_box_sum(h_new, target)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 5: One more SLSQP if time
    if tl() > 15:
        try:
            c_pts = full_corr(best_h)
            mx = np.max(c_pts)
            active = c_pts > mx - 0.01 * mx
            def obj2(x):
                cc = full_corr(x)
                return float(np.mean(cc[active] ** 4) ** 0.25)
            r = minimize(obj2, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
