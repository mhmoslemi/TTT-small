    dx = 2.0 / n

def compute_corr(h):
    """Compute all correlation values."""
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_grad(x, beta):
    """Gradient of smooth_max with respect to x."""
    m = np.max(x)
    e = np.exp(beta * (x - m))
    w = e / e.sum()
    return w

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

    """
    Compute d_corr[k]/d_h[j] for all k, j.
    Returns a 2D array of shape (2n-1, n).
    corr[k] = sum_i h[i] * (1 - h[i+d]) where d = k - n + 1
    d/dh[j] corr[k] = (1-h[j+d])*[0<=j+d<n] - h[j-d]*[0<=j-d<n]

                cnt = int(mask.sum())
                if cnt > 0:
                    h_new[mask] += np.random.randn(cnt) * 0.04
            elif r < 0.82:
                corr = compute_corr(h_cur)
                k_max = int(np.argmax(corr))
                d = k_max - (n - 1)
                n_adj = np.random.randint(2, min(6, n // 5 + 1))
                if d > 0:
                    idx = np.random.choice(n - d, min(n_adj, n - d), replace=False)
                    for i in idx:
                        h_new[i] += np.random.uniform(-0.02, 0.05)
                        h_new[i + d] += np.random.uniform(-0.05, 0.02)
                elif d < 0:
                    da = -d
                    idx = np.random.choice(n - da, min(n_adj, n - da), replace=False)
                    for i in idx:
                        h_new[i + da] += np.random.uniform(-0.02, 0.05)
                        h_new[i] += np.random.uniform(-0.05, 0.02)
                else:
                    idx = np.random.choice(n, n_adj, replace=False)
                    h_new[idx] += np.random.randn(n_adj) * 0.03
            else:
                n_swaps = np.random.randint(1, 3)
                for _ in range(n_swaps):
                    i, j = np.random.choice(n, 2, replace=False)
                    h_new[i], h_new[j] = h_new[j], h_new[i]

            h_new = project_h(h_new, target)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or (T > 1e-6 and np.random.random() < np.exp(-min(300.0, delta / T))):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T *= cooling
            if no_improve > 600:
                T = max(T, 0.004)
                no_improve = 0

    # Phase 4: Multi-start SLSQP
    if tleft() > 60:
        for attempt in range(10):
            if tleft() < 20:
                break
            h0 = best_h.copy()
            n_pert = np.random.randint(3, 10)
            idx = np.random.choice(n, n_pert, replace=False)
            h0[idx] += np.random.randn(n_pert) * 0.025
            h0 = project_h(h0, target)
            beta_val = int(np.random.choice([150, 400, 800]))
            try:
                res = minimize(obj_fn, h0, args=(beta_val,), jac=obj_grad,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Final high-beta polish
    if tleft() > 25:
        for beta_val in [2000, 5000, 10000]:
            if tleft() < 12:
                break
            try:
                res = minimize(obj_fn, best_h, args=(beta_val,), jac=obj_grad,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final subgradient pass if time remains
    if tleft() > 40:
        holder = [best_c5, best_h.copy()]
        _, _ = subgradient_polish(best_h, target, 3000, t0, budget_s, holder)
        if holder[0] < best_c5:
            best_c5, best_h = holder[0], holder[1]

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
