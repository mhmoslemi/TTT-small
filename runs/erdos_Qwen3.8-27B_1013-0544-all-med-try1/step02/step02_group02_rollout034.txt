    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    mx = c.max()
    exp_c = np.exp(alpha * (c - mx))
    log_sum = mx + np.log(exp_c.sum()) / alpha
    # Softmax weights
    w = exp_c / exp_c.sum()  # shape (2n-1,)
    
    # Compute gradient
    # grad[i] = dx * sum_shift w[shift] * [(1-h[i+shift])*valid - h[i-shift]*valid]

    for m in range(2 * n - 1):
        shift = m - n + 1
        wm = w[m] * dx
        if shift >= 0:
            # (1-h[i+shift]) valid for i = 0, ..., n-1-shift
            grad[:n-shift] += wm * (1 - h[shift:])
            # -h[i-shift] valid for i = shift, ..., n-1

            # -h[i+shift] = -h[i-s] valid for i = s, ..., n-1
            grad[s:] -= wm * h[:n-s]

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 3.0
    for _ in range(200):
        tau = (lo + hi) / 2
        proj = np.clip(h - tau, 0, 1)
        s = proj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mx = c.max()
    exp_c = np.exp(alpha * (c - mx))
    val = mx + np.log(exp_c.sum()) / alpha
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - n + 1
        wm = w[m] * dx
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wm * (1.0 - h[i_lo + s:i_hi + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wm * h[i_lo2 - s:i_hi2 - s]
    return val, grad

def proj_grad(h, n, alpha, t0, budget_s, max_iter=50000, mom=0.85):

def slsqp(h, n, alpha, maxiter, ftol):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx2 = sh.max()
        if mx2 > 700:
            return float(m + (mx2 + np.log(np.sum(np.exp(sh - mx2)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def objg(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx2 = sh.max()
        es = np.exp(sh - mx2)
        w = es / es.sum()
        gr = np.zeros(n)
        for mm in range(2 * n - 1):
            s = mm - n + 1
            wm = w[mm] * dx
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_lo < i_hi:
                gr[i_lo:i_hi] += wm * (1.0 - x[i_lo + s:i_hi + s])
            i_lo2 = max(0, s)
            i_hi2 = min(n, n + s)
            if i_lo2 < i_hi2:
                gr[i_lo2:i_hi2] -= wm * x[i_lo2 - s:i_hi2 - s]
        return gr
    def sc(x):
        return np.sum(x) - n / 2.0
    def scj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]

                      options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

    # Phase 3: Pair optimization
    if rem() > 60:
        best_h, best_c5 = pair_opt(best_h, n, t0, budget_s, rng, 5, 400)

    # Phase 4: SLSQP
    if rem() > 60:
        ht = slsqp(best_h, n, 50000, 5000, 1e-15)
        c5 = compute_c5(ht, n)
        if c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 5: High-alpha projected gradient
    for alpha in [50000, 200000, 1000000]:
        if rem() < 50:
            break
        ht, ft = proj_grad(best_h, n, alpha, t0, budget_s,
                           max_iter=80000, mom=0.9)
        c5 = compute_c5(ht, n)
        if c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 6: Pair opt + SLSQP
    if rem() > 50:
        best_h, best_c5 = pair_opt(best_h, n, t0, budget_s, rng, 5, 500)
    if rem() > 40:
        ht = slsqp(best_h, n, 200000, 8000, 1e-16)
        c5 = compute_c5(ht, n)
        if c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 7: Very high alpha gradient
    for alpha in [5000000, 20000000]:
        if rem() < 40:
            break
        ht, ft = proj_grad(best_h, n, alpha, t0, budget_s,
                           max_iter=100000, mom=0.92)
        c5 = compute_c5(ht, n)
        if c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 8: Final pair optimization
    if rem() > 30:
        best_h, best_c5 = pair_opt(best_h, n, t0, budget_s, rng, 5, 600)

    # Phase 9: Final SLSQP
    if rem() > 20:
        ht = slsqp(best_h, n, 500000, 10000, 1e-18)
        c5 = compute_c5(ht, n)
        if c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 10: Ultra-fine pair opt
    if rem() > 15:
        best_h, best_c5 = pair_opt(best_h, n, t0, budget_s, rng, 3, 800)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
