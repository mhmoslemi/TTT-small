            ph1 = np.random.uniform(0, 2 * np.pi)
            ph2 = np.random.uniform(0, 2 * np.pi)
            trial += T * 0.4 * (np.sin(2 * np.pi * freq * x_grid / n + ph1) +
                                 np.cos(2 * np.pi * 2 * freq * x_grid / n + ph2))
        trial = project_h(trial, target_sum)
        c5t = compute_c5(trial)
        if c5t < cur_c5 or (T > 0.005 and np.random.random() < np.exp(-(c5t - cur_c5) / T)):
            cur = trial
            cur_c5 = c5t
            if cur_c5 < best_c5:
                best_h = cur.copy()
                best_c5 = cur_c5

    # Phase 4: Final fine subgradient + SLSQP
    if time_left() > 8:
        cur = best_h.copy()
        cur_c5 = best_c5
        for iteration in range(50000):
            if time_left() < 5:
                break
            corrs = compute_corr(cur)
            grad = subgradient_c5(cur, corrs, 10)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.008 / (1.0 + 0.0005 * iteration)
            trial = project_h(cur - step * grad, target_sum)
            c5t = compute_c5(trial)
            if c5t < cur_c5 - 1e-16:
                cur = trial
                cur_c5 = c5t
                if cur_c5 < best_c5:
                    best_h = cur.copy()
                    best_c5 = cur_c5

        if time_left() > 8:
            _PARAMS['beta'] = 5000
            try:
                res = minimize(slsqp_obj, cur, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    return best_h, best_c5, best_n
