import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def corr_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def con_fun(x):
    return np.sum(x) - len(x) / 2.0


def con_jac(x):
    return np.ones(len(x))


def lag_gradient(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        for i in range(m):
            grad[i] += (1.0 - h[i + d])
            grad[i + d] += (-h[i])
    else:
        aa = -d
        m = n - aa
        for i in range(m):
            grad[i + aa] += (1.0 - h[i])
            grad[i] += (-h[i + aa])
    return grad

            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
