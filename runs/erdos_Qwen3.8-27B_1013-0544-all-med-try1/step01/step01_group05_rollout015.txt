            best_h = h_new.copy()
        else:
            ss *= 0.93
            if ss < 1e-13:
                break
        h = h_new
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    remaining = budget_s - 5.0

    def tleft():
        return remaining - (time.time() - start_time)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent)
    except NameError:
        parent = None
        parent_n = 70

    best_h = None
    best_c5 = 0.5
    best_n = 70

    n_list = [parent_n]
    for cand in [50, 60, 80, 100]:
        if cand not in n_list:
            n_list.append(cand)
    n_list = sorted(set(n_list))

    n_final_reserve = 60.0
    per_n_time = (budget_s - 5.0 - n_final_reserve) / len(n_list)

    for n in n_list:
        if tleft() < 35.0:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        _g['dx'] = dx
        _g['target_sum'] = target_sum
        alloc = min(per_n_time, tleft() - 20.0)
        if alloc < 15.0:
            break

        inits = []
        if parent is not None:
            if len(parent) == n:
                inits.append(parent.copy())
            else:
                xp = np.linspace(0, 2, len(parent))
                xn = np.linspace(0, 2, n)
                inits.append(np.interp(xn, xp, parent))
        x = np.linspace(0, 2, n)
        inits.append(np.full(n, 0.5))
        for f in [1, 2, 3]:
            for a in [0.25, 0.4, 0.55]:
                for p in [0, 1.0, 2.0]:
                    hc = 0.5 + a * np.cos(np.pi * f * x + p)
                    hc = project_feasible(np.clip(hc, 0, 1), target_sum)
                    inits.append(hc)
        for _ in range(2):
            inits.append(project_feasible(rng.random(n), target_sum))

        unique_inits = [inits[0]]
        for h_i in inits[1:]:
            if all(np.linalg.norm(h_i - u) > 0.4 for u in unique_inits):
                unique_inits.append(h_i)
        inits = unique_inits[:14]
        if len(inits) > 14:
            inits = inits[:14]

        per_init = max(alloc / len(inits), 4.0)

        for h_init in inits:
            if tleft() < 20.0:
                break
            h_init = project_feasible(h_init, target_sum)
            c5_init = compute_c5(h_init, dx)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()
                best_n = n

            pg_time = per_init * 0.45
            if pg_time < 2.0:
                pg_time = 2.0
            h_pg, c5_pg = pg_optimize(h_init, n, dx, target_sum, pg_time)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n

            if tleft() > 15.0:
                try:
                    res = minimize(obj_fn, h_pg, method='SLSQP',
                                   bounds=[(0.0, 1.0)] * n,
                                   constraints=[{'type': 'eq', 'fun': eq_fn}],
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    if res.x is not None:
                        h_r = project_feasible(res.x, target_sum)
                        c5_r = compute_c5(h_r, dx)
                        if c5_r < best_c5:
                            best_c5 = c5_r
                            best_h = h_r.copy()
                            best_n = n
                except Exception:
                    pass

            if tleft() > 10.0:
                h_f, c5_f = fine_pg(best_h if best_n == n else h_pg,
                                    n, dx, target_sum, max_iters=3000)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
                    best_n = n

    # Final SLSQP on global best
    if tleft() > 25.0:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        _g['dx'] = dx
        _g['target_sum'] = target_sum
        try:
            res = minimize(obj_fn, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': eq_fn}],
                           options={'maxiter': 8000, 'ftol': 1e-16})
            if res.x is not None:
                h_r = project_feasible(res.x, target_sum)
                c5_r = compute_c5(h_r, dx)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r
        except Exception:
            pass

    # Ultra-fine
    if tleft() > 5.0:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h_f, c5_f = fine_pg(best_h, n, dx, target_sum, max_iters=20000, ss_init=5e-6, mu=0.2)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    return (best_h, best_c5, best_n)
