    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)

        w[np.argmax(corr)] = 1.0
    else:

    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:

    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for k_idx in active:
        k = k_idx - (n - 1)
        grad += w[k_idx] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def adam_optimize(h0, target_sum, deadline, rng, alpha_init=80.0, lr_init=0.008):
    n = len(h0)
    h = h0.astype(float).copy()
    m_ema = np.zeros(n)
    v_ema = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha = alpha_init
    stall = 0
    n_restarts = 0
    best_alpha = alpha_init

    while time.time() < deadline:
        grad = compute_grad(h, alpha)
        t += 1
        m_ema = beta1 * m_ema + (1 - beta1) * grad
        v_ema = beta2 * v_ema + (1 - beta2) * grad**2
        m_hat = m_ema / (1 - beta1**t)
        v_hat = v_ema / (1 - beta2**t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target_sum)
        c5 = compute_c5(h_new)

        if c5 < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5
            h = h_new
            stall = 0
            if lr < lr_init * 3:
                lr = min(lr * 1.005, lr_init * 3)
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.92
            if stall > 400:
                n_restarts += 1
                strategy = n_restarts % 4
                if strategy == 0:
                    noise_scale = 0.02 + 0.03 * min(n_restarts, 5)
                    h = project_box_sum(best_h + rng.randn(n) * noise_scale, target_sum)
                elif strategy == 1:
                    freq = rng.randint(1, 6)
                    h_pert = best_h + 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.rand() * 6.28)
                    h = project_box_sum(h_pert, target_sum)
                elif strategy == 2:
                    block = max(1, n // rng.randint(4, 12))
                    h_pert = best_h.copy()
                    start = rng.randint(0, n - block)
                    h_pert[start:start+block] += 0.05 * rng.rand()
                    h = project_box_sum(h_pert, target_sum)
                else:
                    h_bin = (best_h > 0.5).astype(float)
                    h = project_box_sum(0.2 * h_bin + 0.8 * best_h + rng.randn(n) * 0.015, target_sum)

                c5_check = compute_c5(h)
                if c5_check >= best_c5:
                    h = best_h.copy()

                m_ema = np.zeros(n)
                v_ema = np.zeros(n)
                t = 0
                lr = lr_init
                if n_restarts % 2 == 0:
                    alpha = min(alpha * 1.3, 600.0)
                else:
                    alpha = max(alpha * 0.7, 40.0)
                best_alpha = alpha
                stall = 0
                if n_restarts > 20:
                    break

    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def generate_random_h(n, target_sum, rng):
    h = rng.uniform(0.1, 0.9, size=n)
    return project_box_sum(h, target_sum)
