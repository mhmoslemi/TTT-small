    for mult in [2, 3]:
        n2 = n * mult
        if n2 > 720 or time.time() > p4:
            break
        dx2 = 2.0 / n2
        best_up_h = None
        best_up_c5 = 999.0
        for method in ['interp', 'repeat', 'avg']:
            h_up = project_feasible(upsample(best_h, n2, method), n2)
            c5_up = compute_c5(h_up, dx2)
            if c5_up < best_up_c5:
                best_up_c5 = c5_up
                best_up_h = h_up
        if best_up_h is not None and best_up_c5 < best_c5 + 0.02:
            sub_deadline = min(t0 + budget_s * 0.7, p4)
            h_m, c5_m = momentum_optimize(best_up_h, n2, dx2, lr=0.03, decay=0.85,
                                            max_iters=100000, deadline=sub_deadline)
            if c5_m < best_up_c5:
                best_up_h, best_up_c5 = h_m, c5_m
            if time.time() < p4:
                h_cd2, c5_cd2 = pairwise_cd(best_up_h, n2, dx2, p4, rng, max_steps=50000)
                if c5_cd2 < best_up_c5:
                    best_up_h, best_up_c5 = h_cd2, c5_cd2
            if best_up_c5 < best_c5:
                best_h, best_c5 = best_up_h, best_up_c5

    # Phase 5: Final SLSQP polishing with perturbations
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    p5 = t0 + budget_s - 10
    stuck = 0
    while time.time() < p5 and stuck < 5:
        h_new = do_slsqp(best_h, 500, 1e-15)
        c5_new = compute_c5(h_new, dx_f)
        if c5_new < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c5_new
            stuck = 0
        else:
            stuck += 1
            h_pert = best_h + rng.randn(n_f) * 0.004
            h_pert = project_feasible(h_pert, n_f)
            h_new = do_slsqp(h_pert, 500, 1e-15)
            c5_new = compute_c5(h_new, dx_f)
            if c5_new < best_c5 - 1e-15:
                best_h, best_c5 = h_new, c5_new
                stuck = 0

    return (best_h, best_c5, n_f)
