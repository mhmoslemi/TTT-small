def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_for_shift(h, L):

    g = np.zeros(n)
    if L == 0:
        g[:] = omh[:] - h[:]
    elif L > 0:
        g[L:] += omh[:n - L]
        g[:n - L] -= h[L:]
    else:
        L2 = -L
        g[:n - L2] += omh[L2:]
        g[L2:] -= h[:n - L2]
    return g

def smooth_obj_grad(h, dx, beta):

    max_scaled = np.max(scaled)
    log_sum = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
    obj = (c_max + log_sum / beta) * dx
    weights = np.exp(scaled - max_scaled)
    weights /= np.sum(weights)
    n = len(h)

        if w > 1e-10:
            L = int(idx) - n_half
            grad += w * grad_for_shift(h, L)
    grad *= dx
    return obj, grad

def build_B_matrix(h, n):
    n_corr = 2 * n - 1

        k = m - (n - 1)

        else:
            kp = -k
            for j in range(n - kp):
                B[m, j] = 1.0 - h[j + kp]
    return B

def solve_lp(h, n, target_sum):
    B = build_B_matrix(h, n)
    h_var = cp.Variable(n)
    t = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum, B @ h_var <= t]
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None and np.all(np.isfinite(h_var.value)):
                return np.asarray(h_var.value).copy()
        except Exception:
            continue
    return None

def active_set_grad(h, n, tol_frac=0.001, max_active=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))

    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += grad_for_shift(h, L)
    grad /= float(len(active))
    return grad
