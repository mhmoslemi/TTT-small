    best_c5 = compute_c5(best_h, dx)
    best_n = n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with increasing t
    for t in [15, 40, 100, 250, 600, 1500]:
        if elapsed() > budget_s - 250:
            break
        h_new = slsqp(best_h, dx, ts, t, maxiter=6000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Top-k objective variant
    for k in [3, 5, 8]:
        if elapsed() > budget_s - 200:
            break
        h_new = slsqp_topk(best_h, dx, ts, k=k, maxiter=4000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Follow up with smooth max
        h_new = slsqp(best_h, dx, ts, t=300, maxiter=4000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Nelder-Mead group descent
    if elapsed() < budget_s - 150:
        h_new, c5_new = nm_group_descent(best_h, dx, ts, group_size=4, n_rounds=40, step=0.05)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Follow up
        h_new = slsqp(best_h, dx, ts, t=200, maxiter=5000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Fourier-space simulated annealing
    if elapsed() < budget_s - 100:
        h_new, c5_new = fourier_sa(best_h, dx, ts, n_iters=6000, seed_val=77)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Refine after SA
        h_new = slsqp(best_h, dx, ts, t=300, maxiter=6000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Try n=80 (coarser)
    if elapsed() < budget_s - 80:
        n_lo = 80
        dx_lo = 2.0 / n_lo
        ts_lo = n_lo / 2.0
        h_lo = np.interp(np.linspace(0, 2, n_lo), np.linspace(0, 2, n), best_h)
        h_lo = fix_sum(h_lo, ts_lo)
        for t in [20, 80, 300, 1000]:
            if elapsed() > budget_s - 40:
                break
            h_lo = slsqp(h_lo, dx_lo, ts_lo, t, maxiter=4000)
        c5_lo = compute_c5(h_lo, dx_lo)
        if c5_lo < best_c5:
            best_h = h_lo
            best_c5 = c5_lo
            best_n = n_lo

    # Phase 6: Try n=200 (finer)
    if elapsed() < budget_s - 60:
        n_hi = 200
        dx_hi = 2.0 / n_hi
        ts_hi = n_hi / 2.0
        h_hi = np.interp(np.linspace(0, 2, n_hi), np.linspace(0, 2, best_n), best_h)
        h_hi = fix_sum(h_hi, ts_hi)
        for t in [30, 100, 400]:
            if elapsed() > budget_s - 30:
                break
            h_hi = slsqp(h_hi, dx_hi, ts_hi, t, maxiter=3000)
        c5_hi = compute_c5(h_hi, dx_hi)
        if c5_hi < best_c5:
            best_h = h_hi
            best_c5 = c5_hi
            best_n = n_hi

    # Phase 7: Final high-precision polish
    if elapsed() < budget_s - 5:
        n_f = best_n
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        for t in [400, 1000, 2500]:
            if elapsed() > budget_s - 2:
                break
            h_new = slsqp(best_h, dx_f, ts_f, t, maxiter=8000)
            c5_new = compute_c5(h_new, dx_f)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = np.clip(best_h, 0, 1)
    best_h = fix_sum(best_h, best_n / 2.0)
    dx_f = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
