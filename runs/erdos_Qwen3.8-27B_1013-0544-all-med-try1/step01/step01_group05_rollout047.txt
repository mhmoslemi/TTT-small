import numpy as np
from scipy.optimize import minimize
import time

def project_to_sum(x, target_sum):
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_grad(corr, h, dx, tau, target_sum, lam):
    n = len(h)
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    n_half = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if p[idx] < 1e-10:
            continue
        L = idx - n_half
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:
            g = np.zeros(n)
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            L2 = -L
            g[:n - L2] += (1.0 - h[L2:])
            g[L2:] -= h[:n - L2]
        grad += p[idx] * g
    grad *= dx
    sum_diff = np.sum(h) - target_sum
    grad += 2.0 * lam * sum_diff
    return grad

def lse_value(corr, tau, h, target_sum, lam):
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    return max_c + tau * np.log(np.sum(np.exp(shifted))) + lam * (np.sum(h) - target_sum) ** 2

def run_lse_phase(h_start, n, dx, target_sum, tau_schedule, max_iter, time_limit, start_time):
    best_h = h_start.copy()
    lam = 800.0
    for tau in tau_schedule:
        elapsed = time.time() - start_time
        if elapsed > time_limit - 10:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            return lse_value(corr, tau, x, target_sum, lam)
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            return lse_grad(corr, x, dx, tau, target_sum, lam)
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
            if res.x is not None:
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                if compute_c5(h_new, dx) < compute_c5(best_h, dx):
                    best_h = h_new
        except Exception:
            pass
    return best_h

def subgradient_refine(h, n, dx, target_sum, max_iters, start_time, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step_size = 0.0008
    n_half = n - 1
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        tol = max(1e-7, c_max * 5e-5)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 25:
            active = np.argsort(corr)[-25:]
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            if L == 0:
                g = 1.0 - 2.0 * h
            elif L > 0:
                g = np.zeros(n)
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                L2 = -L
                g[:n - L2] += (1.0 - h[L2:])
                g[L2:] -= h[:n - L2]
            grad += g
        grad = grad / len(active) * dx
        sum_diff = np.sum(h) - target_sum
        grad += 200.0 * sum_diff

        if c5_new < best_c5:

            best_h = h_new.copy()
            h = h_new

                h = best_h + rng.normal(0, 0.008, n)
