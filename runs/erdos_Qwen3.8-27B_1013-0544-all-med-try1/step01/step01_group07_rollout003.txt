import numpy as np

from scipy.optimize import minimize
from functools import partial

def _obj(x, t, dx):
    corr = np.correlate(x, 1 - x, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 0)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def _constraint(x, target_sum):
    return np.sum(x) - target_sum

def _sub_obj(x, parent_h, idx, t, dx):
    h_sub = parent_h.copy()
    h_sub[idx] = x
    corr = np.correlate(h_sub, 1 - h_sub, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 0)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        obj_fn = partial(_obj, t=t, dx=dx)
        cons_fn = partial(_constraint, target_sum=target_sum)
        res = minimize(obj_fn, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fn}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def group_descent(h, dx, target_sum, group_size=8, n_rounds=150, t=60, seed_val=12345):

    best_c5 = compute_c5(h, dx)

            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                stagnation = 0
            else:
                stagnation += 1
        except Exception:
            stagnation += 1
    return h, best_c5

def targeted_descent(h, dx, target_sum, n_rounds=40, t=40):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for r in range(n_rounds):
        corr = np.correlate(h, 1 - h, mode="full")
        worst_lag = np.argmax(corr)
        shift_idx = worst_lag - (n - 1)
        if shift_idx == 0:
            continue
        if shift_idx > 0:
            x1 = np.arange(n - shift_idx)
            x2 = x1 + shift_idx
        else:
            x2 = np.arange(n + shift_idx)
            x1 = x2 - shift_idx
        score = np.abs(h[x1] - h[x2])
        n_adj = min(12, len(score))

                h[b] += adj

    old_n = len(h)

    return np.interp(x_new, x_old, h)
