```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, tau):
    x_scaled = tau * x
    x_max = np.max(x_scaled)
    return float(x_max / tau + np.log(np.sum(np.exp(x_scaled - x_max))) / tau)

def smooth_max_weighted_grad(corr, tau):
    """Returns the soft-max value and the softmax weights (normalized to sum to 1)."""
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    weights = np.exp(shifted)
    weights_sum = np.sum(weights)
    value = x_max / tau + np.log(weights_sum) / tau
    probs = weights / weights_sum
    return value, probs

def corr_grad(corr_probs, h, n):
    """
    Compute gradient of smooth_max(np.correlate(h, 1-h, 'full'), tau) with respect to h.
    corr_probs: softmax weights for each lag (length 2n-1)
    h: the function values (length n)
    n: number of points
    Returns: gradient vector of length n
    """
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        p = corr_probs[k]
        if p < 1e-15:
            continue
        lag = k - (n - 1)
        # d/dh[j] corr[k] = (1-h[j-lag])*I(valid) - h[j+lag]*I(valid)
        # For term 1: (1-h[j-lag]) is nonzero when 0 <= j-lag < n, i.e., lag <= j < n+lag
        # For term 2: -h[j+lag] is nonzero when 0 <= j+lag < n, i.e., -lag <= j < n-lag
        
        # Term 1: contributes (1-h[n-lag...n+lag-1]) to grad[lag...n+lag-1] (clipped to [0,n))
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            # j ranges from lo1 to hi1-1, j-lag ranges from lo1-lag to hi1-1-lag
            indices = np.arange(lo1, hi1)
            grad[indices] += (1.0 - h[indices - lag]) * p
        
        # Term 2: contributes -h[n-jag...n-lag-1] to grad[-lag...n-lag-1] (clipped to [0,n))
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            indices = np.arange(lo2, hi2)
```

```python
def compute_objective_and_grad(h, tau, penalty, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Compute smooth max and weights
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    exp_shifted = np.exp(shifted)
    w_sum = np.exp(shifted).sum()
    value = (x_max + np.log(w_sum)) / tau
    W = exp_shifted / w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(W)):
        p = W[k]
        if p < 1e-12:
            continue
        lag = k - (n - 1)
        # First term: (1-h[j-lag]) for j where 0<=j-lag<n
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += p * (1.0 - h[idx - lag])
        # Second term: -h[j+lag] for j where 0<=j+lag<n
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] += -p * h[idx + lag]
    
    # Penalty gradient
    grad += 2.0 * penalty * (h.sum() - target)
    
    return value, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_and_grad(x, tau, penalty, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    xc = tau * corr
    x_max = np.max(xc)
    shifted = xc - x_max
    exp_s = np.exp(shifted)
    w_sum = exp_s.sum()
    value = (x_max + np.log(w_sum)) / tau + penalty * (x.sum() - target) ** 2
    W = exp_s / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        p = W[k]
        if p < 1e-14:
            continue
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += p * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] += -p * x[lo2 + lag:hi2 + lag]
    grad += 2.0 * penalty * (x.sum() - target)
    return value, grad

def objective(x, tau, penalty, target):
    v, _ = obj_and_grad(x, tau, penalty, target)
    return v

def gradient(x, tau, penalty, target):
    _, g = obj_and_grad(x, tau, penalty, target)
    return g

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(80, 0.5)

    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    bnds = [(0.0, 1.0)] * n

    def remain():
        return budget_s - (time.time() - t0)

    # Phase 1: Multi-stage annealing from parent
    tau_schedule = np.geomspace(5, 20000, 40)
    for i, tau in enumerate(tau_schedule):
        if remain() < 40:
            break
        pen = 50.0
        try:
            res = minimize(objective, best_h, jac=gradient,
                          args=(tau, pen, target),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = fix_sum(res.x, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            else:
                best_h = h_new
        except Exception:
            pass

        if remain() < 60:

        h_t = fix_sum(h_t, t_target)
        for tau in [10, 50, 200, 1000, 5000, 20000]:

                h_t = fix_sum(res.x, t_target)

            hp = best_h + np.random.normal(0, 0.02, hn)

        hp = np.clip(hp, 0.0, 1.0)

    if remain() > 30:
        hn3 = len(best_h)
        tgt3 = hn3 / 2.0
        tau_cd = 50000
        for sweep in range(min(5, int(remain() // 10))):
            if remain() < 10:
                break
            improved = True
            while improved and remain() > 10:
                improved = False
                for j in range(hn3):
                    old = best_h[j]
                    lo, hi = 0.0, 1.0
                    for _ in range(50):
                        mid = 0.5 * (lo + hi)
                        best_h[j] = old
                        v_lo, _ = obj_and_grad(best_h + (best_h - best_h).copy(), tau_cd, 0, tgt3)
                        best_h[j] = mid
                        v_mid, _ = obj_and_grad(best_h, tau_cd, 0, tgt3)

                        else:

                        improved = True
            c5_cd = evaluate_c5(best_h)
            if c5_cd < best_c5:
                best_c5 = c5_cd

    # Final fix
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
```
