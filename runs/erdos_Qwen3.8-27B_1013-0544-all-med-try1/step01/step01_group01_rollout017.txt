    return h

def objective_fn(h, n, target, beta, pw):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, beta)
    pen = pw * (h.sum() - target) ** 2
    return sm * dx + pen

def gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        ia = j - lag
        if 0 <= ia < n:
            g += (1.0 - h[ia])
        ib = j + lag
        if 0 <= ib < n:
            g -= h[ib]
        grad[j] = g
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_h(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing beta
    for beta in [50, 100, 200, 500, 1000, 2000, 4000]:
        if budget_s - (time.time() - t0) < 8:
            break
        try:
            res = minimize(objective_fn, best_h, args=(n, target, beta, 1e5),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Active-set local search with backtracking
    for rnd in range(200):
        if budget_s - (time.time() - t0) < 8:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[-6:]
        for idx in top_idx:
            if budget_s - (time.time() - t0) < 2:
                break
            lag = idx - (n - 1)
            grad = gradient_for_lag(best_h, lag)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                continue
            grad = grad / gnorm
            # Backtracking line search
            step = 0.05
            for _ls in range(30):
                h_try = project_h(best_h - step * grad, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
                    break
                step *= 0.6
            # Also try small positive step
            for step2 in [0.005, 0.002, 0.001]:
                h_try = project_h(best_h + step2 * grad, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
                    break

    # Phase 3: Diversified restarts
    cnt = 0
    while budget_s - (time.time() - t0) > 20:
        cnt += 1
        h0 = best_h.copy()
        s = cnt % 6
        if s == 0:
            scale = 0.04 + 0.08 * np.exp(-cnt / 50.0)
            h0 += np.random.randn(n) * scale
        elif s == 1:
            blen = np.random.randint(2, max(3, n // 5))
            bs = np.random.randint(0, n - blen)
            h0[bs:bs + blen] = np.random.random(blen) * 0.7 + 0.15
        elif s == 2:
            h0 = (best_h > 0.5).astype(float)
            h0 += np.random.randn(n) * 0.05
        elif s == 3:
            freq = np.random.randint(1, 5)
            h0 += 0.04 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + np.random.rand() * 2 * np.pi)
        elif s == 4:
            mask = np.random.random(n) < 0.2
            h0[mask] = np.random.random(int(mask.sum()))
        else:
            # Swap pairs
            for _ in range(np.random.randint(2, 6)):
                i1, i2 = np.random.choice(n, 2, replace=False)
                h0[i1], h0[i2] = h0[i2], h0[i1]

        h0 = project_h(h0, target)
        beta = 200 + 150 * (cnt % 4)
        try:
            res = minimize(objective_fn, h0, args=(n, target, beta, 1e5),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: Final high-precision polish
    if budget_s - (time.time() - t0) > 5:
        for beta in [500, 1000, 2000, 4000, 8000]:
            if budget_s - (time.time() - t0) < 3:
                break
            try:
                res = minimize(objective_fn, best_h, args=(n, target, beta, 1e6),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 15000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # One more active-set pass
    if budget_s - (time.time() - t0) > 3:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[-4:]
        for idx in top_idx:
            lag = idx - (n - 1)
            grad = gradient_for_lag(best_h, lag)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                continue
            grad /= gnorm
            step = 0.01
            for _ls in range(20):
                h_try = project_h(best_h - step * grad, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
                    break
                step *= 0.5

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
