    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 30:
            break
        if time.time() - last_polish > polish_interval:
            last_polish = time.time()
            h = best_h.copy()
            current_c5 = best_c5
            h_pol = slsqp_optimize(h, n, 5000, 300, 1e-14)
            c_pol = compute_c5(h_pol, n)
            if c_pol < best_c5:
                best_h = h_pol.copy()
                best_c5 = c_pol
                h = h_pol.copy()
                current_c5 = c_pol
            if budget_s - (time.time() - t0) < 30:
                break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, D) * rng.choice([1.0, -1.0])
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < 5.0 * E and rng.random() < np.exp(-delta / max(E, 1e-10))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

def fine_2pt(h, n, t0, budget_s, rng):
    best_c5 = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for base_d in steps:
        if budget_s - (time.time() - t0) < 5:
            break
        for i in range(n):
            if budget_s - (time.time() - t0) < 5:
                break
            for j in range(i + 1, n):
                for s in [1.0, -1.0]:
                    hi = h[i] + s * base_d
                    hj = h[j] - s * base_d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c5:
                            best_c5 = c
                        else:
                            h[i] -= s * base_d
                            h[j] += s * base_d
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 0: Multi-start with structured patterns
    x = np.linspace(0, 2, n)
    candidates = [best_h.copy()]
    alt = np.array([1.0 if (i % 2 == 0) else 0.0 for i in range(n)])
    candidates.append(project_feasible(alt, n))
    blk = np.array([1.0 if (i % 8 < 4) else 0.0 for i in range(n)])
    candidates.append(project_feasible(blk, n))
    sine = 0.5 + 0.4 * np.sin(np.pi * x)
    candidates.append(project_feasible(sine, n))
    sine2 = 0.5 + 0.4 * np.sin(2.0 * np.pi * x)
    candidates.append(project_feasible(sine2, n))
    rnd_bin = np.zeros(n)
    idx = rng.permutation(n)
    rnd_bin[idx[:n // 2]] = 1.0
    candidates.append(project_feasible(rnd_bin, n))

    for cand in candidates:
        if remaining() < 10:
            break
        c = compute_c5(cand, n)
        if c < best_c5:
            best_h = cand.copy()
            best_c5 = c

    # Phase 1: SLSQP continuation for fast local optimization
    for alpha in [500, 5000, 50000]:
        if remaining() < 10:
            break
        ht = slsqp_optimize(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 2: Simulated annealing - main exploration
    sa_configs = [
        (500000, 0.08, 1e-5, 0.4, 0.01),
        (400000, 0.04, 1e-6, 0.25, 0.005),
        (300000, 0.02, 1e-6, 0.15, 0.003),
    ]
    for n_iter, e_s, e_m, d_s, d_m in sa_configs:
        if remaining() < 80:
            break
        best_h, best_c5 = simulated_annealing(
            best_h, n, t0, budget_s, rng, n_iter, e_s, e_m, d_s, d_m)

    # Phase 3: SLSQP polish after SA
    for alpha in [10000, 100000, 500000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 4: Another SA round from polished point
    if remaining() > 120:
        best_h, best_c5 = simulated_annealing(
            best_h, n, t0, budget_s, rng, 200000, 0.01, 1e-6, 0.1, 0.002)

    # Phase 5: Random restarts with local optimization
    while remaining() > 80:
        scale = rng.choice([0.02, 0.05, 0.1, 0.15, 0.25])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        improved = False
        for alpha in [5000, 50000]:
            if remaining() < 15:
                break
            ht = slsqp_optimize(h_pert, n, alpha, 1000, 1e-14)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c
                improved = True
                break

    # Phase 6: Final SLSQP polish
    for alpha in [500000, 2000000]:
        if remaining() < 10:
            break
        ht = slsqp_optimize(best_h, n, alpha, 8000, 1e-20)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    # Phase 7: Fine 2-point optimization
    if remaining() > 10:
        best_h, best_c5 = fine_2pt(best_h, n, t0, budget_s, rng)

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
