    n = len(h)

def compute_corr(h, dx):

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def smooth_max_grad(x, beta):
    m = np.max(x)
    shifted = np.exp(beta * (x - m))

def project_h(h, target_sum):

def lag_gradient(h, lags, dx):
    """Compute gradient of sum of selected correlations w.r.t. h."""
    n = len(h)

            j2 = i - lag
            if 0 <= j2 < n:
                grad[i] -= h[j2] * dx
    return grad
