    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n_points):
            old_corr = np.correlate(h, 1.0 - h, mode="full")
            
            # For coordinate j, find optimal h[j]
            # c_l = (1-h[j+l]) * 1[0<=j+l<n] - h[j-l] * 1[0<=j-l<n]
            c = np.zeros(2*n_points-1)
            d = old_corr.copy()
            
            for lag in range(-(n_points-1), n_points):
                # lag corresponds to index lag + (n_points-1) in the correlation
                idx = lag + (n_points - 1)
                cj = 0.0
                if 0 <= j + lag < n_points:
                    cj += (1.0 - h[j + lag])
                if 0 <= j - lag < n_points:
                    cj -= h[j - lag]
                c[idx] = cj

            # Minimize max_idx (c[idx] * h_new + d[idx] - c[idx] * h[j]) over h_new in [0, 1]
            # = max_idx (c[idx] * (h_new - h[j]) + d[idx])

            # subject to -h[j] <= t <= 1 - h[j]
            
            # 1D convex minimization
            from scipy.optimize import minimize_scalar
            def f1d(t):
                return np.max(c * t + d)
            
            lo = -h[j]
            hi = 1.0 - h[j]
            if lo >= hi:
                continue
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded')
            h_new_j = h[j] + res.x
            h_new_j = np.clip(h_new_j, 0.0, 1.0)
            
            if abs(h_new_j - h[j]) > 1e-10:

                improved = True
        
        if not improved:
            break

    diff = target_sum - h.sum()

def find_1d_min(c, d, lo, hi):
    """Find t in [lo, hi] that minimizes max_i (c[i]*t + d[i])."""
    # Bisection on the subgradient
    for _ in range(60):  # ~1e-18 precision
        mid = 0.5 * (lo + hi)
        vals = c * mid + d
        active = (vals == vals.max())
        subgrad_min = c[active].min() if active.any() else 0.0
        subgrad_max = c[active].max() if active.any() else 0.0
        if subgrad_min > 0:
            hi = mid
        elif subgrad_max < 0:
            lo = mid
        else:
            break  # 0 is in subgradient, optimal
    return 0.5 * (lo + hi)
