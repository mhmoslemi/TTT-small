def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def grad_at_lag(h, k):
    """Gradient of correlation at lag k with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g * (2.0 / n)

def project(h, target_sum):
    """Project onto {0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_grad(h, alpha):
    """Compute smooth max gradient"""
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(len(h))
    n = len(h)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_at_lag(h, k)
    return grad

def optimize(h0, target_sum, deadline, rng, alpha_init=50):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_grad(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_at_lag(h, k)
    return grad * (2.0 / n)

def sharpen(h, t, target_sum):
    h_shift = h - 0.5
    return project(0.5 + 0.5 * np.tanh(t * h_shift), target_sum)

def smooth_h(h, target_sum, window=5):
    kernel = np.ones(window) / window
    h_s = np.convolve(h, kernel, mode='same')
    return project(h_s, target_sum)

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    lr = lr_init
    momentum = np.zeros_like(h)
    beta = 0.9
    stall = 0
    iters = 0
    
    while time.time() < deadline:
        iters += 1
        grad = smooth_grad(h, alpha)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - lr * momentum
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
            lr = min(lr * 1.01, 0.05)
        else:
            stall += 1
            lr *= 0.97
            if stall > 50:
                # Multiple escape strategies
                candidates = []
                # 1. Sharpen
                candidates.append(sharpen(h, 2.0 + rng.rand()*5.0, target_sum))
                # 2. Smooth
                candidates.append(smooth_h(h, target_sum, window=7))
                # 3. Random noise
                candidates.append(project(h + rng.randn(n)*0.03, target_sum))
                # 4. Sharpen best
                candidates.append(sharpen(best_h, 3.0 + rng.rand()*4.0, target_sum))
                # 5. Best + noise
                candidates.append(project(best_h + rng.randn(n)*0.02, target_sum))
                
                best_cand = h
                best_cand_c5 = best_c5

                    c = compute_c5(cand)
                    if c < best_cand_c5:
                        best_cand = cand
                        best_cand_c5 = c
                
                h = best_cand
                if best_cand_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = best_cand_c5
                
                alpha = min(alpha * 1.3, 400)
                lr = lr_init
                momentum = np.zeros_like(h)
                beta = max(0.8, beta * 0.93)
                stall = 0
    
    return best_h, best_c5
