The number of valid n values is n - |lag| (for |lag| < n, which is always true since lag ranges from -(n-1) to n-1).

= (sum of h over valid positions of n+lag) - autocorr[lag]

The first term is not simply n-|lag| unless h is constant. It's the sum of h values at positions {n+lag : n valid}, which is a shifted sum of h.

from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] -= diff / float(free.sum())
        else:
            h -= diff / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] -= diff / float(free.sum())
            np.clip(h, 0.0, 1.0, out=h)
    return h

def sobg(h, dx, beta, pen):
    """Smooth objective and gradient with penalty"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    bvals = beta * vals
    ls = logsumexp(bvals)
    obj = ls / beta
    w = np.exp(bvals - ls)
    s = h.sum() - n / 2.0
    obj += pen * s * s
    sg = 2.0 * pen * s
    wdx = w * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(n):
        g = 0.0
        for i in range(2 * n - 1):
            wi = wdx[i]
            if wi < 1e-16:
                continue
            lag = i - nm1
            j1 = m - lag
            if 0 <= j1 < n:
                g += wi * (1.0 - h[j1])
            j2 = m + lag
            if 0 <= j2 < n:
                g -= wi * h[j2]
        grad[m] = g + sg
    return obj, grad

    b = [(0.0, 1.0)] * n

                 bounds=b, constraints=c, options={'maxiter': mi, 'ftol': 1e-14})
    return project_feasible(r.x, n)
