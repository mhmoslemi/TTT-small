```python
def project_onto_feasible(x, target):
    """Project x onto {h: sum h = target, 0 <= h <= 1}"""
    n = len(x)
    # Bisection on threshold tau: h_i = clip(x_i - tau, 0, 1)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - mid, 0, 1)
    # Fix any tiny numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound variables
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h
```
