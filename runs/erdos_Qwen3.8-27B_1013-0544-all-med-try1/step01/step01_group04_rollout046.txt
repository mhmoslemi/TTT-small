            pop[worst] = best_h + rng.randn(n) * 0.15
            pop[worst] = project_feasible(pop[worst], n)
            obj[worst] = compute_c5(pop[worst], dx)
            if obj[worst] < best_c5:
                best_c5 = obj[worst]
                best_h = pop[worst].copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1a: DE at n=90 with exploratory params (30%)
    de_time1 = budget_s * 0.30
    rh, rc = de_run(best_h, n, dx, 70, de_time1, rng, best_c5)
    if rc < best_c5:
        best_h, best_c5 = rh, rc
    
    # Phase 1b: Second DE run with different params (20%)
    if elapsed() < budget_s * 0.50:
        de_time2 = budget_s * 0.50 - elapsed()
        rng2 = np.random.RandomState(seed + 123)
        rh2, rc2 = de_run(best_h, n, dx, 50, de_time2, rng2, best_c5)
        if rc2 < best_c5:
            best_h, best_c5 = rh2, rc2
    
    # Phase 2: SLSQP polishing (20%)
    p2_deadline = start_time + budget_s * 0.72
    attempt = 0
    while time.time() < p2_deadline:
        if attempt == 0:
            h_try = best_h.copy()
        else:
            progress = (time.time() - start_time - budget_s * 0.50) / max(1, budget_s * 0.22)
            progress = min(1.0, max(0.0, progress))
            scale = 0.06 * (1.0 - progress) + 0.005
            h_try = best_h + rng.randn(n) * scale
            h_try = project_feasible(h_try, n)
        
        h_new = slsqp_polish(h_try, n, dx, target_sum, max_iter=250)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        attempt += 1
    
    # Phase 3: Upsample to 2n and refine (28%)
    if elapsed() < budget_s * 0.96:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = np.repeat(best_h, 2).astype(float)
        h_up += rng.randn(n2) * 0.015
        h_up = project_feasible(h_up, n2)
        
        p3_deadline = start_time + budget_s * 0.96
        attempt3 = 0
        while time.time() < p3_deadline:
            if attempt3 == 0:
                h_try3 = h_up.copy()
            else:
                scale = 0.015
                h_try3 = h_up + rng.randn(n2) * scale
                h_try3 = project_feasible(h_try3, n2)
            
            h_new3 = slsqp_polish(h_try3, n2, dx2, target2, max_iter=120)
            c5_new3 = compute_c5(h_new3, dx2)
            if c5_new3 < best_c5 - 1e-14:
                best_h = h_new3
                best_c5 = c5_new3
                h_up = h_new3.copy()
            attempt3 += 1
    
    return (best_h, best_c5, len(best_h))
