    if tleft() > 250:
        n3 = 200
        dx3 = 2.0 / n3
        target3 = n3 / 2.0
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n3)
        h3 = np.interp(x_new, x_old, best_h)
        h3 = project_feasible(h3, target3)
        tb3 = min(150.0, tleft() - 50)
        h3_opt, c5_3 = local_opt(h3, target3, dx3, max_iter=15000,
                                  time_budget=tb3, t0=start_time)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            best_n = n3

    # Phase 4: Final refinement at best resolution
    if tleft() > 30:
        dx_b = 2.0 / best_n
        target_b = best_n / 2.0
        h_ref, c5_ref = local_opt(best_h, target_b, dx_b,
                                   max_iter=10000, time_budget=max(5.0, tleft() - 5),
                                   t0=start_time)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Phase 5: Try a few random restarts from best
    if tleft() > 20:
        for r in range(5):
            if tleft() < 5:
                break
            dx_b = 2.0 / best_n
            target_b = best_n / 2.0
            h_r = best_h + rng.normal(0, 0.01, best_n)
            h_r = project_feasible(h_r, target_b)
            h_ro, c5_ro = local_opt(h_r, target_b, dx_b, max_iter=3000,
                                     time_budget=min(3.0, tleft() - 3), t0=start_time)
            if c5_ro < best_c5:
                best_c5 = c5_ro
                best_h = h_ro

    return (best_h, best_c5, best_n)
