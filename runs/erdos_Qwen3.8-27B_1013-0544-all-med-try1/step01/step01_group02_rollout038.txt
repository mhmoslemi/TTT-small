import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h * (target_sum / total)
    return np.clip(h, 0.0, 1.0)

def solve_1d_max(base, coeff):
    """Find t in [0,1] minimizing max_p(base[p] + coeff[p]*t)."""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = base + coeff * mid
        max_val = vals.max()
        # Find active indices (within tolerance)
        tol = 1e-10 * max(1.0, abs(max_val))
        active = np.where(np.abs(vals - max_val) <= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        min_c = coeff[active].min()
        max_c = coeff[active].max()
        if min_c >= -1e-14:
            hi = mid
        elif max_c <= 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, max_iters=100, tol=1e-12):
    h = h.copy()
    for it in range(max_iters):
        max_change = 0.0
        result = np.correlate(h, 1 - h, mode='full')
        for j in range(n):
            # Compute derivative of result w.r.t. h[j]
            s = np.arange(2*n - 1) - (n - 1)
            i_min = np.maximum(0, s)
            i_max = np.minimum(n-1, n-1-s)

            I1 = ((j >= i_min) & (j <= i_max))
            js = j - s
            I1 = I1 & (js >= 0) & (js <= n-1)
            
            # I2: j+s is valid as i for shift s
            jps = j + s
            I2 = ((jps >= i_min) & (jps <= i_max) & (jps >= 0) & (jps <= n-1))
            
            # h[j-s] where valid
            h_js = np.zeros(2*n-1)
            h_js[I1 & (js >= 0) & (js <= n-1)] = h[js[I1 & (js >= 0) & (js <= n-1)]]
            
            # h[j+s] where valid
            h_jps = np.zeros(2*n-1)
            valid_jps = I2 & (jps >= 0) & (jps <= n-1)
            h_jps[valid_jps] = h[jps[valid_jps]]
            
            deriv = (1.0 - h_js).astype(float) * I1.astype(float) - h_jps * I2.astype(float)
            
            base = result - h[j] * deriv
            new_t = solve_1d_max(base, deriv)
            change = abs(new_t - h[j])
            if change > max_change:
                max_change = change
            h[j] = new_t
            result = np.correlate(h, 1 - h, mode='full')
        
        h = project_feasible(h, n)
        if max_change < tol:
            break
    return h

```python
def compute_deriv(h, j, n):
    """Compute d(output[p])/d(h[j]) for all p=0,...,2n-2."""
    k = np.arange(2*n-1) - (n-1)  # shift values
    
    # I_A: (k >= 0 and j >= k) or (k < 0 and j <= n-1+k)
    IA = np.zeros(2*n-1)
    pos_mask = k >= 0
    neg_mask = k < 0
    IA[pos_mask] = (j >= k[pos_mask]).astype(float)
    IA[neg_mask] = (j <= n-1+k[neg_mask]).astype(float)
    
    # I_B: (k >= 0 and j <= n-1-k) or (k < 0 and j >= -k)
    IB = np.zeros(2*n-1)
    IB[pos_mask] = (j <= n-1-k[pos_mask]).astype(float)
    IB[neg_mask] = (j >= -k[neg_mask]).astype(float)
    
    # h[j-k] where IA is 1
    jk = j - k
    h_jk = np.zeros(2*n-1)
    valid_jk = (jk >= 0) & (jk <= n-1) & (IA > 0)
    h_jk[valid_jk] = h[jk[valid_jk].astype(int)]
    
    # h[j+k] where IB is 1
    jpk = j + k
    h_jpk = np.zeros(2*n-1)
    valid_jpk = (jpk >= 0) & (jpk <= n-1) & (IB > 0)
    h_jpk[valid_jpk] = h[jpk[valid_jpk].astype(int)]
    
    return (1.0 - h_jk) * IA - h_jpk * IB
```

This is O(n) per coordinate (vectorized).

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h * (target_sum / total)
    return np.clip(h, 0.0, 1.0)

def solve_1d_max(base, coeff):
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        vals = base + coeff * mid
        max_val = vals.max()
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.where(np.abs(vals - max_val) <= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        min_c = coeff[active].min()
        max_c = coeff[active].max()
        if min_c >= -1e-14:
            hi = mid
        elif max_c <= 1e-14:
            lo = mid
        else:
            break

    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
