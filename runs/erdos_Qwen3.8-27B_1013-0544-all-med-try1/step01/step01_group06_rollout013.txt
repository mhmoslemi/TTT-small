```python
def delta_corr_single(h, i, delta):
    """Compute the change in corr[k] for all k when h[i] changes by delta."""
    n = len(h)
    # delta_corr[k] for k from -(n-1) to n-1
    # Full correlation has 2n-1 entries
    delta_corr = np.zeros(2 * n - 1)
    
    # Effect on h[i]*(1-h[i+k]) terms: for each k, if i and i+k are valid
    # h[i] appears in corr[k] at positions where i is the first index
    # corr[k] = sum_j h[j] * (1-h[j+k]) for valid j, j+k
    # h[i] appears as h[j] when j=i, contributing (1-h[i+k]) for lag k
    # h[i] appears as h[j+k] when j+k=i, i.e., j=i-k, contributing -h[i-k] for lag k
    
    # For lag k (where k ranges from -(n-1) to n-1):
    # h[i] contributes to corr[k] if:
    #   - As h[j]: j=i, need i+k in [0,n), i.e., 0<=i+k<n
    #   - As h[j+k]: j=i-k, need j in [0,n), i.e., 0<=i-k<n, and j+k=i in [0,n) (always true)
    
    # So for each valid k:
    # delta_corr[k] = delta * (1 - h[i+k]) [if i+k valid] - delta * h[i-k] [if i-k valid]
    
    for k in range(-(n-1), n):
        dc = 0.0
        j1 = i + k  # h[i] as first factor, need j1 valid
        if 0 <= j1 < n:
            dc += delta * (1.0 - h[j1])
        j2 = i - k  # h[i] as second factor (in 1-h[j+k]), need j2 valid
        if 0 <= j2 < n:
            dc -= delta * h[j2]
        delta_corr[k + n - 1] = dc
    return delta_corr
```

```python
    return grad * (2.0 / n)

def grad_exact_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def optimize(h0, target, deadline, rng, alpha0=80):
    n = len(h0)
    best_h = project_simplex(h0.copy(), target)
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    vel = np.zeros(n)
    lr = 0.003
    beta_m = 0.82
    alpha = alpha0
    no_improve = 0
    stall = 0

    while time.time() < deadline:
        grad = grad_smooth_max(cur_h, alpha)
        vel = beta_m * vel - lr * grad
        new_h = project_simplex(cur_h + vel, target)
        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5 - 1e-15:
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
            cur_h = new_h
            no_improve = 0
            if lr < 0.012:
                lr = min(lr * 1.025, 0.02)
        else:
            no_improve += 1
            vel *= 0.4
            lr = max(lr * 0.94, 0.0004)
            cur_h = project_simplex(best_h + rng.randn(n) * 0.002, target)
            if no_improve > 70:
                size = 0.008 + 0.012 * min(stall, 6)
                h_p = project_simplex(best_h + rng.randn(n) * size, target)
                c5_p = compute_c5(h_p)
                cur_h = h_p
                if c5_p < best_c5:
                    best_h = h_p
                    best_c5 = c5_p
                alpha = min(alpha * 1.35, 500.0)
                lr = 0.003
                vel = np.zeros(n)
                no_improve = 0
                stall += 1
                if stall > 18:
                    break
    return best_h, best_c5

def local_refine(h, target, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    top_M = 8
    while time.time() < deadline:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_idx = np.argsort(corr)[-top_M:]
        improved = False
        for idx in top_idx:
            k = idx - (n - 1)
            g = grad_exact_k(best_h, k)
            for lr_t in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                new_h = project_simplex(best_h - lr_t * g, target)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5 - 1e-15:
                    best_h = new_h
                    best_c5 = new_c5
                    improved = True
                    break
        if not improved:
            progressed = False
            for _ in range(20):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for delta in [0.02, 0.01, 0.005]:
                    new_h = best_h.copy()
                    if new_h[i] >= delta and new_h[j] <= 1.0 - delta:
                        new_h[i] -= delta
                        new_h[j] += delta
                        new_h = project_simplex(new_h, target)
                        nc = compute_c5(new_h)
                        if nc < best_c5 - 1e-15:
                            best_h = new_h
                            best_c5 = nc
                            progressed = True
                            break
            if not progressed:
                break
    return best_h, best_c5

def interp_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h)

def make_init(n, rng, style=0):
    if style == 0:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.1
    elif style == 1:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif style == 2:
        h = (np.sin(2 * np.pi * rng.uniform(1, 4) * np.arange(n) / n) + 1) / 2
    else:
        h = np.full(n, 0.5)
        h[:n // 4] = 0.8
        h[n // 4:n // 2] = 0.2
        h[n // 2:3 * n // 4] = 0.8
        h[3 * n // 4:] = 0.2
        h += rng.randn(n) * 0.05
    return project_simplex(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_init(200, rng, 0)

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization at initial resolution (45% budget)
    d1 = t0 + budget_s * 0.45
    best_h, best_c5 = optimize(best_h, n_init / 2.0, d1, rng, alpha0=70)

    # Phase 2: Multi-resolution exploration (35% budget)
    start2 = time.time()
    end2 = start2 + budget_s * 0.35
    n_base = len(best_h)
    res_list = []
    for dn in [-30, -15, 15, 30, 50, 80, 120]:
        nt = n_base + dn
        if 60 <= nt <= 600:
            res_list.append(nt)
    res_list = sorted(set(res_list))
    per_res = max((end2 - start2) / max(len(res_list), 1), 20.0)
    for nt in res_list:
        if time.time() >= end2 - 15:
            break
        h_int = project_simplex(interp_to(best_h, nt), nt / 2.0)
        d = min(time.time() + per_res, end2 - 5)
        h_opt, c5_opt = optimize(h_int, nt / 2.0, d, rng, alpha0=90)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Alternative initializations at best resolution (10% budget)
    n_best = len(best_h)
    start3 = time.time()
    end3 = start3 + budget_s * 0.10
    for style in [1, 2, 3]:
        if time.time() >= end3 - 10:
            break
        h_alt = make_init(n_best, rng, style)
        # Blend with current best for continuity
        h_blend = 0.5 * best_h + 0.5 * h_alt
        h_blend = project_simplex(h_blend, n_best / 2.0)
        d = min(time.time() + (end3 - start3) / 4, end3 - 5)
        h_opt, c5_opt = optimize(h_blend, n_best / 2.0, d, rng, alpha0=60)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 4: Local refinement on top lags (8% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        d4 = min(t0 + budget_s - 5, time.time() + remaining * 0.6)
        best_h, best_c5 = local_refine(best_h, n_best / 2.0, d4, rng)

    # Phase 5: Final gradient polish (remaining time)
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        d5 = t0 + budget_s - 3
        best_h, best_c5 = optimize(best_h, len(best_h) / 2.0, d5, rng, alpha0=200)

    best_h = project_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
