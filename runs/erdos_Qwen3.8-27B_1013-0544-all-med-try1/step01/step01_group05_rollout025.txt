        alpha = 8e-5
        momentum = np.zeros(n)
        for _ in range(5000):
            if tleft() < 2:
                break
            g = active_set_grad(h_f, dx, tol_frac=0.001, max_active=6)
            momentum = 0.5 * momentum - alpha * g
            h_new = project_feasible(h_f + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                alpha *= 0.93
                if alpha < 1e-11:
                    break
            h_f = h_new

    return (best_h, best_c5, best_n)
