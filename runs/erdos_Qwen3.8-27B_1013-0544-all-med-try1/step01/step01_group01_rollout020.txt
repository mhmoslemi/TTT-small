def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Add uniformly, then clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # If still off, distribute to non-boundary elements
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def gradient_active_lags(h, lags):
    """Compute averaged gradient of c(s) for given lags."""
    n = len(h)
    grad = np.zeros(n)
    for s in lags:
        # ∂c(s)/∂h[j] = (1-h[j-s])*1[j-s valid] - h[j+s]*1[j+s valid]
        j = np.arange(n)
        js = j - s
        jp = j + s
        mask_s = (js >= 0) & (js < n)
        mask_p = (jp >= 0) & (jp < n)
        grad[mask_s] += (1.0 - h[js[mask_s]])
        grad[mask_p] += -h[jp[mask_p]]
    grad /= len(lags)
    return grad

def nesterov_psd(h, max_iters, lr_init, lr_decay, top_k=3):
    """Nesterov-accelerated projected subgradient method."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h_curr)
    
    for t in range(max_iters):
        lr = lr_init * (lr_decay ** t)
        # Nesterov momentum
        momentum = (t + 0.5) / (t + 1.5)
        y = h_curr + momentum * (h_curr - h_prev)
        y = project_feasible(y, target)
        
        # Compute subgradient
        corr = compute_corr_full(y)
        # Get top-k lags
        top_indices = np.argsort(corr)[-top_k:]
        lags = top_indices - (n - 1)
        grad = gradient_active_lags(y, lags) * dx
        
        # Gradient step
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
    
    return best_h, best_val

def slsqp_optimize(h, maxiter=500, beta=200):
    """SLSQP with smooth max of correlations."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = compute_corr_full(x)
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
    
    def obj_grad(x):
        # Numerical gradient (expensive but accurate)
        eps = 1e-5
        g = np.zeros(n)
        for i in range(n):
            x_p = x.copy()
            x_p[i] += eps
            x_m = x.copy()
            x_m[i] -= eps
            g[i] = (obj(x_p) - obj(x_m)) / (2 * eps)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, target)

def interpolate_to_n(h, n_new):
    """Interpolate h from current resolution to n_new points."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 2/(2*n_old)  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 2/(2*n_new)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    target = n_new / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def nesterov_subgrad(h, max_iters, lr0, decay, top_k):

    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):

            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr_full(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
