def project_with_convolution(w, target=1.0):
    w_sum = np.sum(w)
    if w_sum <= 0:
        return np.ones_like(w) * (target / len(w))
    scale = target / w_sum
    return np.clip(w * scale, 0.0, 1.0)

def compute_envelope_gradients(w, dx=2.0/47):
    corr = np.correlate(w, 1.0-w, mode="full")
    norm_corr = (corr - np.min(corr)) / (np.max(corr) - np.min(corr))
    norm_corr[abs(norm_corr)>0.8] = 0.1
    return norm_corr * dx

def irwin_low_rank_kernel(x, center=None, scale=1.0):
    if center is None:
        center = np.median(x)

    return r
