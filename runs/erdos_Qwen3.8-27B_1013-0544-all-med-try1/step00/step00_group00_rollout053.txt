from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_constraint(h, n):
    """Project h to satisfy sum(h) = n/2, 0 <= h <= 1"""
    h = h.copy()
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Add uniformly to non-saturated elements
        can_add = h < 1.0 - 1e-12
        can_sub = h > 1e-12
        if diff > 0:
            if not np.any(can_add):
                break
            step = min(diff / np.sum(can_add), 1.0 - h[can_add]).min() if np.any(can_add) else 0
            h[can_add] += step
        else:
            if not np.any(can_sub):
                break
            step = min((-diff) / np.sum(can_sub), h[can_sub]).min() if np.any(can_sub) else 0
            h[can_sub] -= step
    return h

def smooth_max(corr, T):
    """Numerically stable log-sum-exp approximation of max"""
    c = T * corr
    c = c - np.max(c)
    return (np.max(c) + np.log(np.sum(np.exp(c)))) / T

        if T in [50, 200, 500]:
            hs = local_opt_slsqp(hc, n, bounds, T)
            c5s = compute_c5(hs, n)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = hs.copy()
                hc = hs.copy()

    # Phase 2: Basin hopping with random perturbations
    n_restarts = 30
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 80:
            break
        scale = 0.03 + 0.12 * max(0.0, 1.0 - i / n_restarts)
        hp = np.clip(best_h + rng.normal(0, scale, n), 0, 1)
        hp = project_sum(hp, n)
        # Multi-T local optimization
        for T in [50, 200, 1000]:
            hp = local_opt(hp, n, bounds, T, maxiter=2000)
        c5 = compute_c5(hp, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp.copy()

    # Phase 2b: Try structured perturbations (flip small regions)
    for i in range(15):
        if time.time() - t0 > budget_s - 60:
            break
        hp = best_h.copy()
        # Flip a random block
        start = rng.randint(0, max(1, n - 5))
        length = rng.randint(3, min(10, n - start))
        hp[start:start + length] = 1.0 - hp[start:start + length]
        hp = project_sum(hp, n)
        for T in [200, 1000]:
            hp = local_opt(hp, n, bounds, T, maxiter=2000)
        c5 = compute_c5(hp, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp.copy()

    # Phase 3: Final high-T polishing
    if time.time() - t0 < budget_s - 10:
        for T in [5000, 10000, 50000]:
            if time.time() - t0 > budget_s - 5:
                break
            hp = local_opt(best_h, n, bounds, T, maxiter=10000)
            c5 = compute_c5(hp, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
        # One more SLSQP polish
        hp = local_opt_slsqp(best_h, n, bounds, 5000, maxiter=5000)
        c5 = compute_c5(hp, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hp.copy()

    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
