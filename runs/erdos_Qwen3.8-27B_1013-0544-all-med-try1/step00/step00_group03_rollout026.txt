                break
            h_try = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                              np.linspace(0, 2, n_pts, endpoint=False), best_h)
            ts = n_try / 2.0
            h_try = project_to_box_sum(h_try, ts)
            for temp in [20.0, 8.0, 3.0]:
                if time.time() - t0 > time_limit - 80:
                    break
                og2 = make_obj_grad(n_try, temp)
                res2 = minimize(og2, h_try, method='L-BFGS-B',
                                bounds=[(0.0, 1.0)] * n_try,
                                options={'maxiter': 150, 'maxfun': 5000, 'ftol': 1e-12})
                h_opt2 = project_to_box_sum(res2.x, ts)
                c5_2 = compute_c5(h_opt2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2.copy()

    # Phase 4: Fine-tune with random local search on best
    if time.time() - t0 < time_limit:
        cur = best_h.copy()
        cur_n = len(cur)
        cur_s = cur_n / 2.0
        T = 0.003
        for _ in range(5000):
            if time.time() - t0 > time_limit:
                break
            hn = cur.copy()
            n_p = rng.randint(1, 4)
            idx = rng.choice(cur_n, n_p, replace=False)
            for i in idx:
                hn[i] += rng.uniform(-0.04, 0.04)
            hn = project_to_box_sum(hn, cur_s)
            c5n = compute_c5(hn)
            c5c = compute_c5(cur)
            if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-8)):
                cur = hn
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = cur.copy()
            T *= 0.999

    return (best_h, compute_c5(best_h), len(best_h))
