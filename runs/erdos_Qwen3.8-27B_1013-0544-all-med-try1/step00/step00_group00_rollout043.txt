import numpy as np

import time

_dx = 0.0

def compute_c5(h, dx):
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def c5_gradient(h, dx, n):
    raw = np.correlate(h, 1.0 - h, mode='full')
    c5_raw = np.max(raw)
    active = np.where(np.abs(raw - c5_raw) < 1e-8)[0]
    grad = np.zeros(n)
    m = np.arange(n)
    for j in active:
        idx1 = m + j - n + 1
        idx2 = m - j + n - 1
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)
        h2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
        grad += np.where(mask1, 1.0 - h1, 0.0) - np.where(mask2, h2, 0.0)
    grad /= len(active)
    return grad * dx

def _obj(h):
    return compute_c5(h, _dx)

def _objgrad(h):
    return c5_gradient(h, _dx, len(h))

def _sum_con(h):
    return np.sum(h) - len(h) / 2.0

def _sum_jac(h):
    return np.ones(len(h))

    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(raw)
    sh = raw - mx
    return float(mx + np.log(np.sum(np.exp(t * sh))) / t)

def _smooth_grad(h, dx, t):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(raw)
    sh = raw - mx
    w = np.exp(t * sh)
    w /= np.sum(w)
    grad = np.zeros(n)
    m = np.arange(n)
    for k in range(len(raw)):
        idx1 = m + k - n + 1
        idx2 = m - k + n - 1
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)

        gk = np.where(mask1, 1.0 - h1, 0.0) - np.where(mask2, h2, 0.0)
        grad += w[k] * gk
    return grad * dx

        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

        h_try = np.clip(h_try, 0, 1)
        h_try = project_to_constraint(h_try, target)
        r = _slsqp(h_try, bounds, cons, 5000, 1e-12)
        r = project_to_constraint(r, target)
        c5 = compute_c5(r, _dx)
        if c5 < best_c5:
            best_c5, best_h = c5, r

    # Phase 4: Smooth + SLSQP from best
    if elapsed() < budget_s * 0.9:
        for t in [50, 150]:
            if elapsed() > budget_s * 0.9:
                break
            r = _smooth_slsqp(best_h, _dx, t, bounds, cons, 5000, 1e-12)
            r = project_to_constraint(r, target)
            c5 = compute_c5(r, _dx)
            if c5 < best_c5:
                best_c5, best_h = c5, r

    # Phase 5: Final high-precision polish
    if elapsed() < budget_s * 0.98:
        r = _slsqp(best_h, bounds, cons, 20000, 1e-14)
        r = project_to_constraint(r, target)
        c5 = compute_c5(r, _dx)
        if c5 < best_c5:
            best_c5, best_h = c5, r

    return best_h, best_c5, n
