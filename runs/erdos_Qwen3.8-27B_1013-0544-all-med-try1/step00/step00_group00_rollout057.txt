def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    """Project h onto {sum(h) = n/2, 0 <= h <= 1} approximately"""
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Adjust only interior points
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0, 1)

    return h

def smooth_obj(x, n, tau, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - h_complement(x), mode="full") * dx
    m = np.max(corr)
    s = np.sum(np.exp(tau * (corr - m)))
    c5s = m + np.log(s) / tau
    viol = np.sum(x) - n / 2.0
    return c5s + penalty * viol * viol

def h_complement(x):
    return 1.0 - x

def smooth_obj_direct(x, n, tau, penalty):
    dx = 2.0 / n
    comp = 1.0 - x
    corr = np.correlate(x, comp, mode="full") * dx
    m = np.max(corr)
    c5s = m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau
    viol = np.sum(x) - n / 2.0
    return c5s + penalty * viol * viol

def try_optimize(x0, n, tau, penalty, bounds, maxiter, ftol, gtol):
    res = minimize(
        partial(smooth_obj_direct, n=n, tau=tau, penalty=penalty),
        x0, method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'maxfun': maxiter * 5}
    )
    return res.x

def coordinate_polish(h, n, c5_current, max_rounds=50, delta=0.02):
    best_h = h.copy()
    best_c5 = c5_current
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in [delta, -delta, 2*delta, -2*delta, 0.5*delta, -0.5*delta]:
                h_try = best_h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_sum(h_try, n)
                c5 = compute_c5(h_try)
                if c5 < best_c5 - 1e-15:
                    best_h = h_try
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def random_restart(h_base, n, scale, rng, tau, penalty, bounds, maxiter):
    h_pert = h_base + rng.randn(n) * scale
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_sum(h_pert, n)
    res = try_optimize(h_pert, n, tau, penalty, bounds, maxiter, 1e-12, 1e-10)
    if res is not None:
        hc = project_sum(res, n)
        c5 = compute_c5(hc)
        return hc, c5
    return None, None

def interpolate_h(h_old, n_old, n_new):
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old, period=1.0)
    return np.clip(h_new, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        h0 = np.ones(45) * 0.5

    n = len(h0)
    best_h = project_sum(h0.copy(), n)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Multi-scale L-BFGS-B
    taus = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for tau in taus:
        if time_left() < 30:
            break
        penalty = max(500.0, 100.0 * tau)
        res = try_optimize(best_h, n, tau, penalty, bounds, 2000, 1e-14, 1e-12)
        if res is not None:
            hc = project_sum(res, n)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_h = hc
                best_c5 = c5

    # Phase 2: Coordinate descent polish
    if time_left() > 30:
        best_h, best_c5 = coordinate_polish(best_h, n, best_c5, 80, 0.02)

    # Phase 3: Random restarts
    for r in range(40):
        if time_left() < 25:
            break
        scale = max(0.05, 0.2 * (1.0 - best_c5))
        hc, c5 = random_restart(best_h, n, scale, rng, 30.0, 1000.0, bounds, 300)
        if hc is not None and c5 < best_c5:
            best_h = hc
            best_c5 = c5
            # Quick polish after improvement
            if time_left() > 30:
                best_h, best_c5 = coordinate_polish(best_h, n, best_c5, 30, 0.01)

    # Phase 4: Fine-tune with high tau
    if time_left() > 40:
        for tau in [100.0, 200.0, 500.0, 1000.0]:
            if time_left() < 20:
                break
            penalty = max(2000.0, 50.0 * tau)
            res = try_optimize(best_h, n, tau, penalty, bounds, 1000, 1e-15, 1e-13)
            if res is not None:
                hc = project_sum(res, n)
                c5 = compute_c5(hc)
                if c5 < best_c5:
                    best_h = hc
                    best_c5 = c5

    # Phase 5: Try larger n
    for n2 in [n + 5, n + 10, n + 20]:
        if time_left() < 60:
            break
        h_interp = interpolate_h(best_h, n, n2)
        h_interp = project_sum(h_interp, n2)
        c5_interp = compute_c5(h_interp)
        bounds2 = [(0.0, 1.0)] * n2
        res = try_optimize(h_interp, n2, 20.0, 1000.0, bounds2, 1500, 1e-13, 1e-11)
        if res is not None:
            hc = project_sum(res, n2)
            c5 = compute_c5(hc)
            if c5 < best_c5:
                best_h = hc
                best_c5 = c5
                n = n2
                # Continue optimizing at this n
                if time_left() > 30:
                    res2 = try_optimize(best_h, n, 50.0, 2000.0, bounds2, 1000, 1e-14, 1e-12)
                    if res2 is not None:
                        hc2 = project_sum(res2, n)
                        c52 = compute_c5(hc2)
                        if c52 < best_c5:
                            best_h = hc2
                            best_c5 = c52

    # Phase 6: Final coordinate polish
    if time_left() > 15:
        best_h, best_c5 = coordinate_polish(best_h, n, best_c5, 40, 0.01)

    # Final verification
    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
