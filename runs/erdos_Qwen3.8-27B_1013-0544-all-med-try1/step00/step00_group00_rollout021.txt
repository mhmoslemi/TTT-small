    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

            h = h_new
            break
        h = h_new

        else:
            idx = np.argmin(h)

    lam = _state['lam']
    n = _state['n']

    h = _project(x, target)

    h = _project(x, target)
    return _c5_val(h)

    bounds = [(0.0, 1.0)] * n
    _state['target'] = target
    _state['n'] = n

    best_c5 = _c5_val(h)

import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _project(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = (target - np.sum(h)) / n
        h_new = np.clip(h + diff, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-14:
            h = h_new
            break
        h = h_new
    r = target - np.sum(h)
    if abs(r) > 1e-15:
        if r > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += r
    return h

def _obj_smooth(x):
    target = _state['target']
    lam = _state['lam']
    n = _state['n']
    dx = 2.0 / n
    h = _project(x, target)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    cs = np.clip((c - m) * lam, -100, 100)
    return m + np.log(np.sum(np.exp(cs))) / lam

        for i in range(n):

                lo = max(0.0, s - 1.0)

                    continue
                bi = h[i]
                grid = np.linspace(lo, hi, grid_size)
                best_v = _c5_val(h)

                    h[i] = ci
                    h[j] = s - ci
                    v = _c5_val(h)
                    if v < best_v:

                h[j] = s - best_ci

                    improved = True
                if best_v < best_c5:
                    best_c5 = best_v
        if not improved:
            break
