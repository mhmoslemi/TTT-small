def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full") * dx
    return np.max(c)

def water_filling(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    target_sum = float(target_sum)
    
    # Binary search for lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    
    # Fine-tune to get exact sum
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-15) & (x < 1.0 - 1e-15)
        n_free = free.sum()
        if n_free > 0:
            x[free] += diff / n_free
            # Clip again and redo if needed
            x = np.clip(x, 0.0, 1.0)
    
    return x

    n = len(h)
    sg = np.zeros(n)
    if k >= 0:
        j1 = np.arange(k, n)
        sg[j1] += (1.0 - h[j1 - k])
        j2 = np.arange(0, n - k)
        sg[j2] -= h[j2 + k]

        sg[j1] += (1.0 - h[j1 + m])
        j2 = np.arange(m, n)
        sg[j2] -= h[j2 - m]
    return sg

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-14:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        nf = free.sum()
        if nf > 0:
            x[free] += diff / nf
            np.clip(x, 0.0, 1.0, out=x)
    return x

def get_subgradient(h, k, n):
    sg = np.zeros(n)
    if k >= 0:
        sg[k:] += 1.0 - h[:n - k]
        sg[:n - k] -= h[k:]
    else:
        m = -k
        sg[:n - m] += 1.0 - h[m:]
        sg[m:] -= h[:n - m]
    return sg
