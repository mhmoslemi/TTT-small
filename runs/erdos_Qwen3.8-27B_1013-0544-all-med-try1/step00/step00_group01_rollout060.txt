def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)


def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return corr * dx


def gradient_worst(h):
    n = len(h)
    lags = compute_all_lags(h)
    k_star = int(np.argmax(lags))
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        ms = m - shift
        mp = m + shift
        if 0 <= ms < n:
            g += 1.0 - h[ms]
        if 0 <= mp < n:
            g -= h[mp]
        grad[m] = g
    return grad, shift

    target = n / 2.0

        p = np.clip(v - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

        d = np.random.uniform(0.01, min(max_d, 0.5))
        new_h = h_cur.copy()
        new_h[i] -= d
        new_h[j] += d
        new_c5 = compute_c5(new_h)

        delta = new_c5 - c5_cur
        if delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-delta / T)):
            h_cur = new_h
            c5_cur = new_c5
            if new_c5 < best_c5_local:
                best_c5_local = new_c5
                best_local = new_h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        T = max(T * 0.9998, 0.002)
        if stagnant > 3000:
            h_cur = best_local.copy()
            c5_cur = best_c5_local
            T = 0.004
            stagnant = 0

    return best_local, best_c5_local


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(47, 0.5)
    n0 = len(h0)
    h0 = project_to_feasible(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient descent (15% budget)
    deadline = t_start + budget_s * 0.15
    best_h, best_c5 = gradient_descent(best_h, n0, deadline)

    # Phase 2: Simulated annealing (45% budget)
    deadline = t_start + budget_s * 0.60
    best_h, best_c5 = simulated_anneal(best_h, n0, deadline)

    # Phase 3: Explore different n values (remaining)
    deadline = t_start + budget_s - 5
    n_tried = set([n0])
    n_order = list(range(n0 - 3, n0 + 30))
    np.random.shuffle(n_order)

    for nc in n_order:
        if time.time() >= deadline:
            break
        if nc < 12 or nc >= 200 or nc in n_tried:
            continue
        n_tried.add(nc)
        h_try = interpolate_h(best_h, nc)
        dl = min(deadline, time.time() + 8)
        h_t, c5_t = gradient_descent(h_try, nc, dl)
        if c5_t < best_c5:
            best_h = h_t
            best_c5 = c5_t
            dl2 = min(deadline, time.time() + 12)
            best_h, best_c5 = simulated_anneal(best_h, nc, dl2)

    # Final polish
    deadline = t_start + budget_s - 1
    best_h, best_c5 = gradient_descent(best_h, len(best_h), deadline)
    best_h = project_to_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
