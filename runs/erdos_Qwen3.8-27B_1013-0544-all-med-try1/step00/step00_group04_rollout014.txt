            h[j] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
                break
            h[i] = 1.0
            h[j] = 0.0
        if not improved:
            break
    return h, best_c5

def perturb(h, rng, strength):
    h_p = h.copy()
    k = rng.randint(1, strength + 1)
    for _ in range(k):
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_p[i], h_p[j] = 0.0, 1.0
    return h_p

def generate_initials(rng, n):
    inits = []
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    inits.append(h)
    for _ in range(5):
        h = np.zeros(n)
        nblocks = rng.randint(3, n // 3 + 1)
        boundaries = sorted(rng.choice(range(1, n), size=min(nblocks - 1, n - 2), replace=False))
        bounds = [0] + list(boundaries) + [n]
        val = rng.randint(0, 2)
        for a, b in zip(bounds[:-1], bounds[1:]):
            h[a:b] = val
            val = 1 - val
        inits.append(h)
    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    inits.append(h_alt)
    return inits

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = 42
    best_h = None
    best_c5 = float('inf')

    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == n:
            candidates.append(h_parent)
        else:
            h_resized = np.zeros(n)
            h_resized[:len(h_parent)] = h_parent
            candidates.append(h_resized)
    candidates.extend(generate_initials(rng, n))

    for h_init in candidates:
        if time.time() - start_time > budget_s - 120:
            break
        h_bin = make_binary(np.array(h_init, dtype=float), n)
        h_ls, c5_ls = local_search_quick(h_bin, max_passes=500)
        h_ls, c5_ls = local_search_best(h_ls, max_passes=100)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls

    # Iterated local search
    timers = [0.0]
    while time.time() - start_time < budget_s - 60:
        strength = rng.randint(2, 8)
        h_pert = perturb(best_h, rng, strength)
        h_ls, c5_ls = local_search_quick(h_pert, max_passes=400)
        h_ls, c5_ls = local_search_best(h_ls, max_passes=100)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls

    # Final deep local search
    if time.time() - start_time < budget_s - 10:
        best_h, best_c5 = local_search_best(best_h, max_passes=500)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
