import time

def compute_c5_and_corr(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5, corr

def project_to_feasible(h, n_target):
    """Project h onto {h: 0<=h[i]<=1, sum(h)=n_target}"""
    h = np.clip(h, 0, 1)
    # Iterate to meet sum constraint
    for _ in range(50):
        diff = n_target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Add/subtract to non-saturated entries
        if diff > 0:
            mask = h < 1
        else:
            mask = h > 0
        if not np.any(mask):
            break
        # Distribute evenly
        step = diff / np.sum(mask)
        step = min(step, 1e-3) if diff > 0 else max(step, -1e-3)
        h[mask] += step
        h = np.clip(h, 0, 1)

def _c5_grad(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-8)[0]
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] += (-h[k:])
        else:
            kk = -k
            grad[:n-kk] += (1.0 - h[kk:])
            grad[kk:] += (-h[:n-kk])
    grad /= len(active)
    return dx * grad

    h = h.copy()

        zeros = np.where(h == 0)[0]
        best_c5 = c5

        h[best_j] = 1
