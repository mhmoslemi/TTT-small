                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                no_improve = 0
                found = True
                break
            alpha *= 0.5
        if not found:
            no_improve += 1
            alpha = max(alpha * 2.0, 0.02)
            if no_improve % 25 == 0:
                smooth = max(smooth * 0.65, 0.003)
            if no_improve > 80:
                h = best_h + 0.08 * rng.randn(n)
                h = project_to_feasible(h, target_sum)
                no_improve = 0
                alpha = 0.1
                smooth = max(0.04, smooth)
        elif no_improve == 0 and best_c5 < 0.38:
            smooth = max(smooth * 0.9, 0.002)

    return best_h, best_c5

def polish_local(h, n, target_sum, time_limit, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deadline = time.time() + time_limit
    scale = 0.03

    while time.time() < deadline:
        trial = best_h + scale * rng.randn(n)
        trial = project_to_feasible(trial, target_sum)
        c5_trial = compute_c5(trial)
        if c5_trial < best_c5:
            best_h = trial
            best_c5 = c5_trial
            scale = max(scale * 0.9, 0.003)
        else:
            scale = min(scale * 1.1, 0.1)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or len(h0) == 0:
        n = 70
        h0 = np.full(n, 0.5)
    else:
        n = len(h0)

    target_sum = n / 2.0

    # Phase 1: Main gradient optimization from parent
    t_left = budget_s - (time.time() - start_time) - 60
    best_h, best_c5 = optimize_gradient(h0, n, target_sum, 0.05, max(t_left, 120), seed)

    # Phase 2: Polish with local random search
    t_left = budget_s - (time.time() - start_time) - 10
    if t_left > 20:
        best_h, best_c5 = polish_local(best_h, n, target_sum, min(t_left, 120), seed + 1)

    # Phase 3: Try a few random restarts if time permits
    t_left = budget_s - (time.time() - start_time) - 5
    restart_seed = seed + 100
    while t_left > 80:
        h_rand = np.clip(0.5 + 0.35 * np.random.randn(n), 0.0, 1.0)
        h_rand = project_to_feasible(h_rand, target_sum)
        t_alloc = min(t_left - 10, 150)
        h_try, c5_try = optimize_gradient(h_rand, n, target_sum, 0.1, t_alloc, restart_seed)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
        t_left = budget_s - (time.time() - start_time) - 5
        restart_seed += 7777

    # Final polish
    t_left = budget_s - (time.time() - start_time) - 2
    if t_left > 10:
        best_h, best_c5 = polish_local(best_h, n, target_sum, min(t_left, 60), seed + 999)

    return best_h, best_c5, n
