def targeted_swap(binary, n, rng):
    """Find a targeted swap that reduces the max correlation."""
    dx = 2.0 / n
    corr = np.correlate(binary, 1.0 - binary, mode="full")
    max_idx = np.argmax(corr)
    
    # Find positions involved in the max correlation
    # corr[max_idx] = sum_m binary[m + max_idx] * (1 - binary[m]) for valid m
    # This counts pairs (a,b) with a-b = max_idx-(n-1), h[a]=1, h[b]=0
    
    lag = max_idx - (n - 1)
    
    # Count how many times each position is involved
    involvement = np.zeros(n)
    if lag >= 0:
        for m in range(n - lag):
            if binary[m + lag] == 1 and binary[m] == 0:
                involvement[m + lag] += 1  # this 1 is "causing" the correlation
                involvement[m] += 1  # this 0 is "allowing" the correlation
    else:
        for m in range(-lag, n):
            if binary[m + lag] == 1 and binary[m] == 0:
                involvement[m + lag] += 1
                involvement[m] += 1
    
    # Best move: turn off a highly-involved 1 and turn on a highly-involved 0
    # (or turn on a 0 that's not involved and turn off a 1 that is)
    
    ones = np.where(binary == 1)[0]
    zeros = np.where(binary == 0)[0]
    
    # Score: we want to pick a 1 with high involvement and a 0 with high involvement
    # (turning off an involved 1 reduces the count, turning on an involved 0 also reduces it)
    
    best_score = -1
    best_i, best_j = -1, -1
    
    # For efficiency, only check top candidates
    top_ones = ones[np.argsort(-involvement[ones])[:min(10, len(ones))]]
    top_zeros = zeros[np.argsort(-involvement[zeros])[:min(10, len(zeros))]]
    
    for i in top_ones:
        for j in top_zeros:
            binary[i] = 0
            binary[j] = 1
            new_c5 = compute_c5_binary(binary, n)
            score = -new_c5
            binary[i] = 1
            binary[j] = 0
            if score > best_score:
                best_score = score
                best_i, best_j = i, j
    
    return best_i, best_j
