import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def to_01(h_init, n_target):
    if len(h_init) == n_target:
        h = h_init.copy()
    else:
        x = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n_target)
        h = np.interp(x_new, x, h_init)
    h_01 = (h >= 0.5).astype(float)
    target = n_target // 2
    num = int(h_01.sum())
    if num > target:
        idx = np.where(h_01 == 1)[0]
        order = idx[np.argsort(h[idx])]
        h_01[order[:num - target]] = 0
    elif num < target:
        idx = np.where(h_01 == 0)[0]
        order = idx[np.argsort(-h[idx])]
        h_01[order[:target - num]] = 1
    return h_01

def greedy_local_search(h, t_limit, t_start):
    n = len(h)
    cur = compute_c5(h)
    while time.time() - t_start < t_limit:
        improved = False
        best_delta = 0
        best_i = -1
        for i in range(n):
            h[i] = 1.0 - h[i]
            new_val = compute_c5(h)
            delta = cur - new_val
            if delta > best_delta:
                best_delta = delta
                best_i = i
            h[i] = 1.0 - h[i]
        if best_i >= 0:
            h[best_i] = 1.0 - h[best_i]
            cur = compute_c5(h)
            improved = True
        if not improved:
            break
    return h, cur

def sa_01(h, t_limit, t_start):
    n = len(h)
    cur = compute_c5(h)
    best = h.copy()
    best_val = cur
    temp = 0.008
    while time.time() - t_start < t_limit and temp > 1e-5:
        i = np.random.randint(n)
        h[i] = 1.0 - h[i]
        new_val = compute_c5(h)
        delta = new_val - cur
        if delta < 0 or (temp > 1e-6 and np.random.random() < np.exp(-delta / temp)):
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
        else:
            h[i] = 1.0 - h[i]
        temp *= 0.9997
        if temp < 0.001:
            temp = 0.001
    return best, best_val

def double_flip_sa(h, t_limit, t_start):
    n = len(h)
    cur = compute_c5(h)
    best = h.copy()
    best_val = cur
    temp = 0.01
    while time.time() - t_start < t_limit and temp > 1e-5:
        i, j = np.random.choice(n, size=2, replace=False)
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        new_val = compute_c5(h)
        delta = new_val - cur
        if delta < 0 or (temp > 1e-6 and np.random.random() < np.exp(-delta / temp)):
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        temp *= 0.9995
        if temp < 0.002:
            temp = 0.002
    return best, best_val

    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
