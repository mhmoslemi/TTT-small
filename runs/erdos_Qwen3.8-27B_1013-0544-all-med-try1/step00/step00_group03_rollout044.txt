import numpy as np
from scipy.optimize import minimize
import time

# Module-level state for helper functions
_DX = 1.0
_N = 1
_TARGET = 0.5
_ALPHA = 100.0


def repair(h, n):
    """Clip to [0,1] and rescale to satisfy sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * _DX)


def obj_smooth(x):
    """Smooth max approximation of C5."""
    corr = np.correlate(x, 1.0 - x, mode='full') * _DX
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(_ALPHA * (corr - m))))) / _ALPHA


def grad_smooth(x):
    """Numerical gradient of smooth objective."""
    eps = 1e-6
    g = np.zeros(_N)
    for i in range(_N):
        xp = x.copy()
        xp[i] += eps
        xm = x.copy()
        xm[i] -= eps
        g[i] = (obj_smooth(xp) - obj_smooth(xm)) / (2.0 * eps)
    return g


def constraint_fun(x):
    return np.sum(x) - _TARGET


def constraint_jac(x):
    return np.ones(_N)


def obj_repair(x):
    """Objective with built-in repair for Nelder-Mead."""
    h = repair(x, _N)
    return compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    global _DX, _N, _TARGET, _ALPHA
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = len(h0)
    else:
        n = 70
        h0 = np.full(n, 0.5)

    _N = n
    _DX = 2.0 / n
    _TARGET = n / 2.0

    h0 = repair(h0, n)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Nelder-Mead with repair (handles non-smoothness directly)
    nm_time_limit = budget_s * 0.45
    starts = [best_h.copy()]
    for _ in range(6):
        starts.append(repair(best_h + rng.randn(n) * 0.04, n))
    starts.append(repair(best_h + rng.randn(n) * 0.1, n))
    # Try a few structured perturbations
    for s in [(0.02, 0.06), (0.05, 0.02)]:
        h_p = best_h.copy()
        h_p[rng.choice(n, n // 3, replace=False)] += s[0]
        h_p[rng.choice(n, n // 3, replace=False)] -= s[1]
        starts.append(repair(h_p, n))

    for start in starts:
        if time.time() - t0 > nm_time_limit:
            break
        try:
            res = minimize(obj_repair, start, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-9,
                                    'fatol': 1e-11, 'adaptive': True})
            h_new = repair(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: SLSQP with smooth max (multiple alphas)
    slsqp_time_limit = budget_s * 0.75
    alphas = [50.0, 200.0, 500.0]
    for alpha in alphas:
        if time.time() - t0 > slsqp_time_limit:
            break
        _ALPHA = alpha
        constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        bounds = [(0.0, 1.0)] * n

        slsqp_starts = [best_h.copy()]
        for _ in range(4):
            slsqp_starts.append(repair(best_h + rng.randn(n) * 0.03, n))

        for start in slsqp_starts:
            if time.time() - t0 > slsqp_time_limit:
                break
            try:
                res = minimize(obj_smooth, start, method='SLSQP', jac=grad_smooth,
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 400, 'ftol': 1e-12})
                h_new = repair(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 3: Coordinate descent polish
    cd_time_limit = budget_s * 0.95
    improved = True
    while improved and (time.time() - t0) < cd_time_limit:
        improved = False
        for i in range(n):
            if time.time() - t0 > cd_time_limit:
                break
            base = best_h[i]
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02,
                          0.04, -0.04, 0.08, -0.08]:
                val = base + delta
                if val < 0.0 or val > 1.0:
                    continue
                h_try = best_h.copy()
                h_try[i] = val
                h_try = repair(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
                    break

    return best_h, best_c5, n
