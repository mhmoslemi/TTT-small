def _smooth_grad_vec(h, alpha, n, lambda_sum=100.0):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(vals)
    exp_vals = np.exp(alpha * (vals - max_val))
    total = np.sum(exp_vals)
    weights = exp_vals / total
    c5 = max_val + np.log(total) / alpha
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - n + 1
        w = weights[m] * dx
        if w < 1e-18:
            continue
        if k >= 0:
            # First term: j in [0, n-1-k]
            grad[:n-k] += w * (1.0 - h[k:])
            # Second term: j in [k, n-1]
            grad[k:] -= w * h[:n-k]
        else:
            jk = -k
            # First term: j in [jk, n-1]
            grad[jk:] += w * (1.0 - h[:n-jk])
            # Second term: j in [0, n-1-jk]
            grad[:n-jk] -= w * h[jk:]
    
    grad += 2.0 * lambda_sum * (np.sum(h) - n / 2.0)
    return c5, grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h):
    return float(np.max(_corr_values(h)))

def _smooth_obj_grad(h, alpha, n, lam=500.0):
    dx = 2.0 / n
    vals = _corr_values(h)
    vmax = np.max(vals)
    ev = np.exp(alpha * (vals - vmax))
    tot = np.sum(ev)
    w = ev / tot
    obj = vmax + np.log(tot) / alpha + lam * (np.sum(h) - n / 2.0) ** 2
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - n + 1
        wm = w[m] * dx
        if wm < 1e-18:
            continue
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
        else:
            jk = -k
            grad[jk:] += wm * (1.0 - h[:n-jk])
            grad[:n-jk] -= wm * h[jk:]
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return obj, grad

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(0.0, h[mask] + step)
    return h

def _optimize_h(h0, n, alpha, lam, maxiter=500, time_limit=60.0):
    target = n / 2.0
    h0 = _project(h0.copy(), target)
    t0 = time.time()
    def fun(x):
        return _smooth_obj_grad(x, alpha, n, lam)[0]
    def fun_and_grad(x):
        return _smooth_obj_grad(x, alpha, n, lam)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_and_grad, h0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                               'gtol': 1e-10, 'maxls': 50})
        h_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_opt = h0
    h_opt = _project(h_opt, target)
    return h_opt

def _gen_cosine(n, freqs, amps):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.ones(n) * 0.5
    for f, a in zip(freqs, amps):
        h += a * np.cos(np.pi * f * x)
    target = n / 2.0
    return _project(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        target_init = n_init / 2.0
        h_init = _project(h_init, target_init)
        c5_init = _c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
    else:
        n_init = 80
        best_h = np.full(n_init, 0.5)
        best_c5 = _c5(best_h)

    # Define initializations to try
    def get_inits(n):
        target = n / 2.0
        inits = []
        # Interpolate best
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            inits.append(_project(np.interp(x_new, x_old, best_h), target))
        # Uniform
        inits.append(np.full(n, 0.5))
        # Cosine bumps
        inits.append(_gen_cosine(n, [1], [0.3]))
        inits.append(_gen_cosine(n, [1], [0.15]))
        inits.append(_gen_cosine(n, [2], [0.2]))
        inits.append(_gen_cosine(n, [1, 2], [0.2, 0.1]))
        inits.append(_gen_cosine(n, [1, 3], [0.15, 0.1]))
        # Random
        inits.append(_project(0.5 + 0.3 * rng.randn(n), target))
        inits.append(_project(np.clip(rng.rand(n) * 0.8 + 0.1, 0, 1), target))
        return inits

    n_list = [80, 100, 120, 160]
    alphas = [50, 100, 200]

    for n in n_list:
        if time.time() - t_start > budget_s * 0.75:
            break
        inits = get_inits(n)
        for init in inits:
            for alpha in alphas:
                if time.time() - t_start > budget_s * 0.9:
                    break
                time_left = budget_s * 0.9 - (time.time() - t_start)
                if time_left < 5.0:
                    break
                tl = min(80.0, time_left * 0.5)
                h_opt = _optimize_h(init, n, alpha, lam=500.0, maxiter=300, time_limit=tl)
                c5_opt = _c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Polishing pass with high alpha on best solution found
    if time.time() - t_start < budget_s * 0.95:
        n_best = len(best_h)
        for alpha in [200, 400]:
            if time.time() - t_start > budget_s * 0.95:
                break
            h_pol = _optimize_h(best_h, n_best, alpha, lam=2000.0, maxiter=200, time_limit=30.0)
            c5_pol = _c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Try smaller n refinements
    for n_small in [64, 72, 80, 96]:
        if time.time() - t_start > budget_s * 0.97:
            break
        if n_small < len(best_h):
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_small, endpoint=False)
            h_try = _project(np.interp(x_new, x_old, best_h), n_small / 2.0)
            h_try = _optimize_h(h_try, n_small, 200, lam=2000.0, maxiter=200, time_limit=20.0)
            c5_try = _c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Final projection
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    best_c5 = _c5(best_h)

    return (best_h, best_c5, n_final)
