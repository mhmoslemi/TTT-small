def compute_objective_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    obj = float(max_val) * dx

    grad = np.zeros(n)
    
    one_minus_h = 1.0 - h
    for k in active:
        lag = k - (n - 1)
        # grad contribution: (1-h[j+lag]) - h[j-lag]
        # (1-h[j+lag]) is zero where j+lag is out of [0,n-1]
        # h[j-lag] is zero where j-lag is out of [0,n-1]

        term2 = np.zeros(n)
        if lag >= 0:
            term1[:n-lag] = one_minus_h[lag:]
            term2[lag:] = h[:n-lag]
        else:
            term1[-lag:] = one_minus_h[:n+lag]
            term2[:n+lag] = h[-lag:]
        grad += term1 - term2
    
    grad *= dx / len(active)
    return obj, grad

where c_k = correlate(h, 1-h, 'full')[k].

∂f_smooth/∂h[j] = dx * sum_k (w_k * ∂c_k/∂h[j])

f_smooth = (dx / T) * log_sum_exp
