import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

def subgradient_descent(h, dx, n_iters, start_time, time_frac):
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.02
    time_limit = start_time + time_frac
    
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        # Compute correlation
        corr = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(corr)
        s = max_idx - (n - 1)
        
        # Gradient
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            js = j - s
            if 0 <= js < n:
                g += (1.0 - h[js])
            jp = j + s
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] = g
        
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if it % 100 == 99:
            lr *= 0.95
    
    return best_h, best_c5

        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

def make_binary(h, rng):
    """Convert to binary while maintaining sum"""
    n = len(h)
    target = n // 2
    h = np.round(h).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while len(ones) > target:
        i = ones[rng.randint(len(ones))]
        h[i] = 0
        ones = np.where(h == 1)[0]
    while len(ones) < target:
        i = zeros[rng.randint(len(zeros))]
        h[i] = 1
        zeros = np.where(h == 0)[0]
    return h

def exhaustive_search(h, dx, time_limit, rng):
    """Exhaustive pairwise swap local search with kicks"""
    n = len(h)
    h = make_binary(h, rng)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    no_improve = 0
    
    while time.time() < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > time_limit:
                break
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break  # Restart with new configuration
                else:
                    h[i], h[j] = 1.0, 0.0
        
        if not improved:
            no_improve += 1
            # Kick: random perturbation
            k = rng.randint(1, 4)  # Flip k pairs
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= k and len(zeros) >= k:
                flip_ones = rng.choice(ones, size=k, replace=False)
                flip_zeros = rng.choice(zeros, size=k, replace=False)
                h[flip_ones] = 0.0
                h[flip_zeros] = 1.0
            
            if no_improve > 50:
                no_improve = 0
                # Larger kick
                k = rng.randint(3, 8)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) >= k and len(zeros) >= k:
                    flip_ones = rng.choice(ones, size=k, replace=False)
                    flip_zeros = rng.choice(zeros, size=k, replace=False)
                    h[flip_ones] = 0.0
                    h[flip_zeros] = 1.0
    
    return best_h, best_c5

def sgdescent(h, dx, n_iters, time_limit):
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.01
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        s = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            js = j - s
            if 0 <= js < n:
                g += 1.0 - h[js]
            jp = j + s
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] = g
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 199:
            lr *= 0.92
    return best_h, best_c5
