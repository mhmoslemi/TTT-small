```python
def compute_subgradient(h, n, active_shifts):
    grad = np.zeros(n)
    for k in active_shifts:
        # Term 1: add (1-h[j-k]) for valid j
        j_start = max(0, k)
        j_end = min(n, n + k)
        if j_start < j_end:
            grad[j_start:j_end] += (1 - h[j_start - k : j_end - k])
        # Term 2: subtract h[j+k] for valid j
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_start < j_end:
            grad[j_start:j_end] -= h[j_start + k : j_end + k]
    return grad / len(active_shifts)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, step_size):
    corr = compute_corr(h)
    max_val = np.max(corr)
    tol = 1e-8 * max(1.0, max_val)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += (1.0 - h[js - k:je - k])
        js = max(0, -k)
        je = min(n, n - k)
        if js < je:
            grad[js:je] -= h[js + k:je + k]
    grad /= len(active)
    return repair_h(h - step_size * grad, n)

def gradient_optimize(h, n, max_iters, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step_size = 0.15
    patience = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 300:
            break
        h_new = subgradient_step(h, n, step_size)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            step_size = min(step_size * 1.02, 0.4)
        elif c5_new < best_c5 + 1e-5:
            h = h_new
            patience += 1
        else:
            patience += 3
            step_size *= 0.93
        if it % 100 == 99 and patience > 80:
            step_size = 0.08
    return best_h, best_c5

def coord_descent(h, n, max_sweeps, deadline):
    h = h.copy()
    cur = evaluate_c5(h)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.08, -0.08]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            found = False
            for d in deltas:
                h_new = h.copy()
                h_new[i] = h[i] + d
                if h_new[i] < 0 or h_new[i] > 1:
                    continue
                h_new = repair_h(h_new, n)
                c5 = evaluate_c5(h_new)
                if c5 < cur - 1e-10:
                    h = h_new
                    cur = c5
                    improved = True
                    found = True
                    break
            if found:
                continue
        if not improved:
            break
    return h

def nm_objective(h, n):
    return evaluate_c5(repair_h(h, n))

def nelder_polish(h, n, deadline):
    remaining = deadline - time.time()
    if remaining < 5:
        return h
    max_iter = int(min(3000, remaining * 50))
    if max_iter < 50:
        return h
    result = minimize(nm_objective, h, args=(n,), method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-6, 'fatol': 1e-8,
                              'adaptive': True, 'maxfev': max_iter * (len(h) + 1)})
    return result.x

def try_update(best_h, best_c5, best_n, h, n):
    h = repair_h(h, n)
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        return h.copy(), c5, n
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0

    h_init = None
    if 'initial_h_values' in globals():
        h_init = np.array(globals()['initial_h_values'], dtype=float)

    # Phase 1: Optimize initial construction
    if h_init is not None:
        n = len(h_init)
        h_init = repair_h(h_init, n)
        best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_init, n)

        # Subgradient phase
        deadline = start_time + budget_s * 0.35
        h_grad, _ = gradient_optimize(h_init, n, 10000, deadline)
        best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_grad, n)

        # Nelder-Mead polish
        deadline = start_time + budget_s * 0.55
        h_nm = nelder_polish(best_h, n, deadline)
        best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_nm, n)

        # Coordinate descent
        deadline = start_time + budget_s * 0.68
        h_cd = coord_descent(best_h, n, 30, deadline)
        best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_cd, n)

    # Phase 2: Try other resolutions with different inits
    n_values = [40, 50, 60, 80, 100, 120, 150, 200, 250]
    remaining_frac = 0.85
    for n in n_values:
        deadline = start_time + budget_s * remaining_frac
        if time.time() > deadline - 15:
            break
        for init_type in range(2):
            if time.time() > deadline - 10:
                break
            if init_type == 0:
                h_rand = np.random.uniform(0.25, 0.75, n)
            else:
                # Block-like initialization
                h_rand = np.zeros(n)
                half = n // 2
                h_rand[:half] = 1.0
                h_rand += np.random.uniform(-0.15, 0.15, n)
            h_rand = repair_h(h_rand, n)
            d = max(deadline - 10, time.time() + 5)
            h_grad, _ = gradient_optimize(h_rand, n, 2000, d)
            best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_grad, n)
            # Quick NM if time
            if time.time() < deadline - 15 and best_n == n:
                h_nm = nelder_polish(best_h, n, start_time + budget_s * 0.82)
                best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_nm, n)

    # Phase 3: Final polish on best
    if best_h is not None:
        n = best_n
        deadline = start_time + budget_s * 0.93
        h_cd = coord_descent(best_h, n, 40, deadline)
        best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_cd, n)

        if time.time() < start_time + budget_s * 0.97:
            h_nm = nelder_polish(best_h, n, start_time + budget_s * 0.98)
            best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_nm, n)

        # Final coord descent
        if time.time() < start_time + budget_s * 0.99:
            h_cd = coord_descent(best_h, n, 20, start_time + budget_s * 0.995)
            best_h, best_c5, best_n = try_update(best_h, best_c5, best_n, h_cd, n)

    if best_h is None:
        n = 50
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
```
