import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - target - 1.0
    hi = h.max() + target + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_soft(h, tau, n):
    """Compute gradient of soft-max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    tc = tau * c
    tc -= np.max(tc)
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        weight = w[p]
        if weight < 1e-12:
            continue
        q = p - (n - 1)
        if q >= 0:
            grad[:n - q] += weight * (1.0 - h[q:n])
            grad[q:n] -= weight * h[:n - q]
        else:
            qa = -q
            grad[qa:n] += weight * (1.0 - h[:n - qa])
            grad[:n - qa] -= weight * h[qa:n]
    return dx * grad

def upsample(h, new_n):
    """Upsample h from len(h) to new_n using linear interpolation."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return h_new

def optimize_h(h, n, tau, lr, max_iters, time_limit, rng):
    """Optimize h using projected gradient descent with soft-max."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        
        grad = compute_gradient_soft(h, tau, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)

            best_h = h_new.copy()
        
        # Adaptive learning rate
        # If improvement is small, reduce lr and increase tau
        h = h_new
    
    return best_h, best_c5

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_weights(c, tau):
    tc = tau * c
    tc -= tc.max()
    e = np.exp(tc)
    return e / e.sum()

def compute_grad(h, n, tau):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    w = softmax_weights(c, tau)
    w_vec = w  # w is already length 2n-1, same as w_vec
    
    # T and S
    corr = np.correlate(h, w_vec, mode='full')  # length 3n-2
    conv = np.convolve(h, w_vec, mode='full')   # length 3n-2
    T = corr[3*n-3 : 2*n-3 : -1]  # length n
    S = conv[n-1 : 2*n-1]          # length n
    
    # A
    cumw = np.zeros(2*n)
    cumw[1:] = np.cumsum(w_vec)
    # A[k] = cumw[2n-1-k] - cumw[n-1-k]
    k = np.arange(n)
    A = cumw[2*n-1-k] - cumw[n-1-k]
    
    grad = (A - T - S) * dx
    return grad, c
```
