import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(y, n):
    """Project y onto {h: sum(h)=n/2, 0<=h<=1}"""
    target = n / 2.0
    lo, hi = -y.max() - 1, 1 - y.min()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2
    return np.clip(y + alpha, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    c_shifted = beta * (c - c.max())
    e = np.exp(c_shifted)
    return c.max() + np.log(e.sum()) / beta

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    m_vals = np.arange(2*n-1) - (n-1)
    c = np.correlate(h, 1 - h, mode="full") * dx
    
    e = np.exp(beta * (c - c.max()))
    w = e / e.sum()
    
    grad = np.zeros(n)
    for j in range(2*n-1):
        m = m_vals[j]
        wj = w[j] * dx
        if m == 0:
            grad += wj * (1 - 2*h)
        else:
            hm = np.zeros(n)
            hp = np.zeros(n)
            if m > 0:
                hm[m:] = h[:n-m]
                hp[:n-m] = h[m:]
            else:
                hm[:n+m] = h[-m:]
                hp[-m:] = h[:n+m]
            grad += wj * (1 - hm - hp)
    
    return grad

def projected_gradient_descent(h0, beta, n_iter=1000, lr=None):
    h = h0.copy()
    n = len(h)
    best_val = smooth_objective(h, beta)
    best_h = h.copy()
    
    if lr is None:
        lr = 1.0 / (beta * (2*n-1))
    
    for it in range(n_iter):
        grad = smooth_gradient(h, beta)
        h_new = h - lr * grad
        h_new = project_onto_feasible(h_new, n)
        
        val = smooth_objective(h_new, beta)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        # Simple backtracking if objective increases
        if val > smooth_objective(h, beta) * 1.01 + 1e-10:
            lr = lr * 0.5
        
        h = h_new
    
    return best_h

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    if h0 is not None:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, n):
    """Euclidean projection onto {h: sum(h)=n/2, 0<=h<=1}"""
    target = n / 2.0
    s0 = np.sum(np.clip(y, 0, 1))
    if abs(s0 - target) < 1e-12:
        return np.clip(y, 0, 1)
    lo = -np.max(y) - 0.5
    hi = 1.0 - np.min(y) + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0, 1)

def get_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def lse_obj(h, beta):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    c = c - c.max()
    return c.max() + np.log(np.sum(np.exp(beta*c)))/beta

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    c_s = c - c.max()
    e = np.exp(beta * c_s)
    w = e / e.sum()
    grad = np.zeros(n)
    m_base = n - 1
    for j in range(2*n - 1):
        m = j - m_base
        coeff = w[j] * dx
        if m == 0:
            grad += coeff * (1.0 - 2.0*h)
        elif m > 0:
            # h[k-m] valid for k>=m: positions m..n-1 get h[0..n-m-1]
            # h[k+m] valid for k<n-m: positions 0..n-m-1 get h[m..n-1]
            d = np.ones(n)
            d[m:] -= h[:n-m]      # subtract h[k-m] for k=m..n-1
            d[:n-m] -= h[m:]      # subtract h[k+m] for k=0..n-m-1
            grad += coeff * d
        else:
            p = -m
            d = np.ones(n)
            d[:n-p] -= h[p:]      # subtract h[k-p] = h[k+m] for k=0..n-p-1
            d[p:] -= h[:n-p]      # subtract h[k+p] = h[k-m] for k=p..n-1
            grad += coeff * d
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, n):
    target = n / 2.0
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def get_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c = c - c.max()
    return c.max() + np.log(np.sum(np.exp(beta * c))) / beta

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_s = c - c.max()
    e = np.exp(beta * c_s)
    w = e / e.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        coeff = w[j] * dx
        if m == 0:
            grad += coeff * (1.0 - 2.0 * h)
        elif m > 0:
            d = np.zeros(n)
            d[m:] = 1.0 - h[:n - m]
            d[:n - m] -= h[m:]
            grad += coeff * d
        else:
            p = -m
            d = np.zeros(n)
            d[:n - p] = 1.0 - h[p:]
            d[p:] -= h[:n - p]
            grad += coeff * d
    return grad

def pgd_optimize(h, beta, n, time_limit=30.0):

    t0 = time.time()

    best_h = h.copy()

    lr = 1.0 / (beta + 1.0)
    stall = 0
    while time.time() - t0 < time_limit:
        g = lse_grad(h, beta)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        f_new = lse_obj(h_new, beta)
        if f_new <= f + 1e-14:
            h = h_new
            f = f_new
            lr = min(lr * 1.03, 20.0)

                best_h = h.copy()
        else:
            lr *= 0.5
            stall += 1
            if lr < 1e-14 or stall > 300:
                break
    return best_h

def slsqp_optimize(h, beta, n, time_limit=60.0):
    t0 = time.time()
    def obj(x):
        return lse_obj(x, beta)
    def obj_grad(x):
        return lse_grad(x, beta)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.fun < obj(h):
            return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def interp_resize(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    pad = max(1, n_old // 5)
    h_ext = np.concatenate([h, h[:pad]])
    return np.interp(x_new, x_old, h_ext)
