import numpy as np
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def binarize(h, n_ones):
    n = len(h)
    h_bin = np.zeros(n)
    h_bin[np.argsort(-h, kind='stable')[:n_ones]] = 1.0
    return h_bin

def greedy_search(h, time_limit=10.0):
    c5 = evaluate(h)
    start = time.time()
    while time.time() - start < time_limit:
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        best_d, best_i, best_j = 1e-14, -1, -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                nc = evaluate(h)
                d = c5 - nc
                h[j] = 0.0
                if d > best_d:
                    best_d, best_i, best_j = d, i, j
            h[i] = 1.0
        if best_i < 0:
            break
        h[best_i], h[best_j] = 0.0, 1.0
        c5 = evaluate(h)
    return h, c5

def simulate_anneal(h, rng, time_limit=30.0):
    c5 = evaluate(h)
    start = time.time()
    bh, bc = h.copy(), c5
    T = 0.01
    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        a = int(rng.choice(ones))
        b = int(rng.choice(zeros))
        h[a], h[b] = 0.0, 1.0
        nc = evaluate(h)
        if nc < c5 or (T > 1e-9 and rng.random() < np.exp(-(nc - c5) / T)):
            c5 = nc
            if c5 < bc:
                bh, bc = h.copy(), c5
        else:
            h[a], h[b] = 1.0, 0.0
        T *= 0.9992
        if T < 1e-8:
            T = 1e-8
    return bh, bc
