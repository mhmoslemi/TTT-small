    best_obj = compute_c5(h)
    
    for T in T_list:
        lr = 0.01 / T  # Scale with temperature
        for it in range(max_iters):
            if time.time() > deadline:
                return best_h, best_obj
            
            obj, grad = compute_obj_and_grad(h, T)
            
            # Gradient descent
            h_new = h - lr * grad
            h_new = project_box_simplex(h_new, S)
            
            c5 = compute_c5(h_new)
            if c5 < best_obj:
                best_obj = c5
                best_h = h_new.copy()
            
            # Backtracking line search
            if c5 > obj:
                lr *= 0.5
                # Re-do with smaller lr
                h_new = h - lr * grad
                h_new = project_box_simplex(h_new, S)
            else:
                lr = min(lr * 1.1, 0.1)
            
            h = h_new
        
        # Cool down: already handled by T_list
    
    return best_h, best_obj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / np.sum(exp_c)
    obj = (c_max + np.log(np.sum(exp_c))) / T
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] += wk * (-h[m:])
        else:
            s = -m
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n - s])
    return obj, grad * dx

def project_box_simplex(x, S):

        if s > S:
            lo = mid
        else:
            hi = mid
