import numpy as np

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h):
    return float(np.max(compute_correlation(h))) * (2.0 / len(h))

def smooth_max_value(corr, temperature):
    scaled = corr / temperature
    max_s = float(np.max(scaled))
    exps = np.exp(scaled - max_s)
    total = float(np.sum(exps))
    return (max_s + np.log(total)) * temperature, exps / total

def grad_smooth_max(h, temperature):
    corr = compute_correlation(h)
    value, weights = smooth_max_value(corr, temperature)
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo > i_hi:
            continue
        js = np.arange(i_lo + s, i_hi + s + 1)
        grad[i_lo:i_hi + 1] += w * (1.0 - h[js])
        j_start = max(0, i_lo + s)
        j_end = min(n - 1, i_hi + s)
        if j_start <= j_end:
            h_idx = np.arange(j_start - s, j_end - s + 1)
            grad[j_start:j_end + 1] -= w * h[h_idx]
    return value, grad

def project_feasible(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:

        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
