    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx

def soft_max_objective(h, alpha=100):

    return np.log(np.sum(np.exp(alpha * vals))) / alpha

def soft_max_gradient(h, alpha=100):

    dx = 2.0 / n
    vals = c5_values(h)
    weights = np.exp(alpha * vals)
    weights /= np.sum(weights)

    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_onto_feasible(h, n, max_iter=100):
    """Project h onto {x: sum(x)=n/2, 0<=x<=1}"""
    h = h.copy()
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Add diff/n to non-clipped elements
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
    return h

def active_gradient(h, tol=0.001):

    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = vals.max()
    weights = np.zeros(2*n-1)
    active = vals >= max_val - tol
    weights[active] = 1.0 / np.sum(active)
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] == 0:
            continue
        lag = m - (n-1)
        # For this lag, contribution to grad[j]:
        # dx * (1-h[j+lag])*I(0<=j+lag<n) - dx*h[j-lag]*I(0<=j-lag<n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):
            h += diff / n
            continue
        h[free] += diff / np.sum(free)
    return h

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = vals.max()
    w = np.exp(alpha * (vals - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * dx * (1.0 - h[:n - lag])
                grad[:n - lag] -= wi * dx * h[lag:]
        else:
            k = -lag
            if k < n:
                grad[:n - k] += wi * dx * (1.0 - h[k:n])
                grad[k:] -= wi * dx * h[:n - k]
    return grad
