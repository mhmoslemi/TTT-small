import numpy as np
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def grad_lag(h, n, i):
    """Analytical gradient of correlation value at index i."""
    lag = i - (n - 1)
    grad = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        grad[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
    j_lo2 = max(0, -lag)
    j_hi2 = min(n - 1, n - 1 - lag)
    if j_lo2 <= j_hi2:
        grad[j_lo2:j_hi2 + 1] -= h[j_lo2 + lag:j_hi2 + lag + 1]
    return grad


def smooth_grad(h, n, weights):
    """Weighted sum of lag gradients (smooth max gradient)."""
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-8:
            continue
        lag = i - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j_lo2 = max(0, -lag)
        j_hi2 = min(n - 1, n - 1 - lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w * h[j_lo2 + lag:j_hi2 + lag + 1]
    return grad


def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection on shift."""
    target = float(n) / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def optimize_h(h, n, t_end, alpha_init=0.02, T_init=0.02):
    """Projected gradient with smooth-max and backtracking line search."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    alpha = alpha_init
    T = T_init
    dx = 2.0 / n
    stagnant = 0

    while time.time() < t_end:
        corr = compute_corr(h)
        active = int(np.argmax(corr))
        cur_val = float(corr[active]) * dx

        if cur_val < best_c5 - 1e-12:
            best_c5 = cur_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Smooth max weights
        shifted = corr - corr[active]
        weights = np.exp(shifted / T)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros(2 * n - 1)
            weights[active] = 1.0
        else:
            weights /= wsum

        grad = smooth_grad(h, n, weights)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search
        accepted = False
        a = alpha
        for _ in range(60):
            h_try = project_feasible(h - a * grad, n)
            c_try = float(np.max(compute_corr(h_try))) * dx
            if c_try <= cur_val - 1e-10:
                h = h_try
                accepted = True
                break
            a *= 0.5
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-12:
                h = project_feasible(h + 0.08 * np.random.randn(n), n)
                alpha = alpha_init
                T = max(T * 0.7, 0.002)
        else:
            alpha = min(alpha * 1.05, 0.5)

        # Anneal temperature
        if stagnant % 200 == 0 and stagnant > 0:
            T = max(T * 0.95, 0.002)
        if stagnant > 2000:
            h = project_feasible(h + 0.15 * np.random.randn(n), n)
            alpha = alpha_init
            T = T_init
            stagnant = 0

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 70
        h0 = np.clip(0.5 + 0.1 * np.random.randn(n0), 0, 1)
        h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: optimize the initial construction
    t_end1 = t0 + budget_s * 0.6
    h_opt, c5_opt = optimize_h(h0, n0, t_end1, alpha_init=0.02, T_init=0.02)
    if c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n0

    # Phase 2: try other resolutions with random starts
    t_end2 = t0 + budget_s * 0.85
    for n_try in [100, 140, 200]:
        if time.time() > t_end2:
            break
        h_rand = np.clip(0.5 + 0.1 * np.random.randn(n_try), 0, 1)
        h_rand = project_feasible(h_rand, n_try)
        t_end_try = min(time.time() + (budget_s * 0.85 - (time.time() - t0)) / 3.0, t_end2)
        h_opt2, c5_opt2 = optimize_h(h_rand, n_try, t_end_try, alpha_init=0.01, T_init=0.03)
        if c5_opt2 < best_c5:
            best_h, best_c5, best_n = h_opt2, c5_opt2, n_try

    # Phase 3: polish the best solution found
    remaining = budget_s - 5 - (time.time() - t0)
    if remaining > 30:
        h_pol, c5_pol = optimize_h(best_h, best_n, time.time() + remaining,
                                   alpha_init=0.005, T_init=0.005)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
