        rng.shuffle(positions)

    elif pattern_type == 1:
        # Gaussian-like concentration
        center = rng.uniform(0.2, 0.8) * n
        sigma = rng.uniform(0.1, 0.4) * n
        scores = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        scores += np.exp(-0.5 * ((np.arange(n) - (n - center)) / sigma) ** 2)
        scores += rng.random(n) * 0.5
        top = np.argsort(scores)[-target:]
        h[top] = 1.0
    elif pattern_type == 2:
        # Arithmetical progression based
        step = rng.randint(2, n // 2 + 1)
        start = rng.randint(n)
        positions = set()
        cur = start
        for _ in range(target):
            positions.add(cur % n)
            cur += step
        positions = list(positions)
        if len(positions) < target:
            remaining = [i for i in range(n) if i not in positions]
            positions.extend(rng.choice(remaining, target - len(positions), replace=False))
        h[positions[:target]] = 1.0
    else:
        # Random
        pos = rng.choice(n, target, replace=False)
        h[pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = np.inf
    
    # Get initial solution
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    # Ensure binary
    h0 = (h0 > 0.5).astype(float)
    # Ensure sum constraint
    target0 = n0 // 2
    if int(h0.sum()) != target0:
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        if len(ones) > target0:
            for idx in ones[-(len(ones) - target0)]:
                h0[idx] = 0
        elif len(ones) < target0:
            for idx in zeros[:target0 - len(ones)]:
                h0[idx] = 1
    
    # Phase 1: Greedy on initial
    best_h, best_c5 = greedy_local_search(h0, max_rounds=200)
    elapsed = time.time() - start_time
    time_remaining = budget_s - elapsed
    
    # Phase 2: SA on initial (multiple restarts with kicks)
    remaining = max(0, time_remaining - 10)
    n_sa = max(1, int(remaining / 40))
    for sa_i in range(n_sa):
        if time.time() - start_time > budget_s - 15:
            break
        h_trial = best_h.copy()
        # Slight perturbation before SA
        ones = list(np.where(h_trial == 1)[0])
        zeros = list(np.where(h_trial == 0)[0])
        n_kicks = rng.randint(1, 4)
        for _ in range(n_kicks):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_trial[i] = 0.0; h_trial[j] = 1.0
            ones[i] = j; zeros[j] = i
        h_new, c5_new = simulated_annealing(h_trial, rng, time_budget=min(35, time_remaining / 2))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        time_remaining = budget_s - (time.time() - start_time)
    
    # Phase 3: Try other n values
    other_ns = [36, 44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 100]
    for n in other_ns:
        if time.time() - start_time > budget_s - 20:
            break
        if n % 2 != 0:
            continue
        n_trials = 8
        for trial in range(n_trials):
            if time.time() - start_time > budget_s - 20:
                break
            h_trial = structured_init(n, rng)
            h_new, c5_new = greedy_local_search(h_trial, max_rounds=30)
            if c5_new < best_c5:
                # Further improve with SA
                sa_budget = min(15, max(3, budget_s - (time.time() - start_time) - 30))
                h_new, c5_new = simulated_annealing(h_new, rng, time_budget=sa_budget)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
    
    # Final polish
    if best_h is not None and time.time() - start_time < budget_s - 5:
        best_h, best_c5 = greedy_local_search(best_h, max_rounds=50)
    
    return best_h, float(best_c5), len(best_h)
