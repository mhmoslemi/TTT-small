            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 500:
            step = step_init * 0.5
            stall = 0
    return cur, best_local

def coord_opt(cur, target, n, t0, budget, max_sweeps=80):
    best_c5 = compute_c5(cur)
    cur_c5 = best_c5
    deltas = [0.05, -0.05, 0.025, -0.025, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                old = trial[j]
                trial[j] = old + d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < cur_c5 - 1e-15:
                    cur = trial
                    cur_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
    return cur, best_c5

def block_opt(cur, target, n, t0, budget, block_size=4, n_blocks=500):
    best_c5 = compute_c5(cur)
    for _ in range(n_blocks):
        if budget - (time.time() - t0) < 3:
            break
        b0 = np.random.randint(0, max(1, n - block_size))
        b1 = b0 + block_size
        bstart = cur[b0:b1].copy()
        be = bstart.mean()
        best_block = bstart.copy()
        best_block_c5 = best_c5
        for _ in range(50):
            if budget - (time.time() - t0) < 1:
                break
            pert = np.random.randn(block_size) * 0.03
            cand = bstart + pert
            cand = np.clip(cand, 0, 1)
            cand = cand + (be - cand.mean())
            cand = np.clip(cand, 0, 1)
            trial = cur.copy()
            trial[b0:b1] = cand
            trial = project_h(trial, target)
            c_try = compute_c5(trial)
            if c_try < best_block_c5 - 1e-15:
                best_block = cand.copy()
                best_block_c5 = c_try
        if best_block_c5 < best_c5 - 1e-15:
            cur[b0:b1] = best_block
            cur = project_h(cur, target)
            best_c5 = compute_c5(cur)
    return cur, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new),
                     np.linspace(0.0, 2.0, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(320, 0.5)

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Work at the parent resolution and a couple of alternatives
    resolutions = [best_n]
    for r in [280, 360]:
        if r != best_n:
            resolutions.append(r)

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 80:
            break
        target = n_pts / 2.0
        cur = project_h(resample_h(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 1: SLSQP smooth max with increasing sharpness
        for beta in [100, 300, 800, 2000, 5000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 20:
                break
            cur = slsqp_opt(cur, n_pts, target, beta, maxiter=5000, ftol=1e-15)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts
                cur_c5 = c

        # Phase 2: Subgradient with momentum
        remaining = budget_s - (time.time() - t0)
        if remaining > 30 and n_pts == best_n:
            cur, cur_c5 = subgrad_opt(cur, target, n_pts, t0, budget_s,
                                       n_iter=30000, step_init=0.02)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

        # Phase 3: Coordinate descent
        remaining = budget_s - (time.time() - t0)
        if remaining > 20 and n_pts == best_n:
            cur, cur_c5 = coord_opt(cur, target, n_pts, t0, budget_s, max_sweeps=100)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5

    # Focus on the best resolution found
    nf = best_n
    tf = nf / 2.0
    best_h = project_h(best_h, tf)

    # Final high-precision SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        for beta in [3000, 8000, 20000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            best_h = slsqp_opt(best_h, nf, tf, beta, maxiter=8000, ftol=1e-16)
            c = compute_c5(best_h)
            if c < best_c5:
                best_c5 = c

    # Final subgradient polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, c = subgrad_opt(best_h, tf, nf, t0, budget_s,
                                 n_iter=50000, step_init=0.005)
        if c < best_c5:
            best_c5 = c

    # Coordinate descent final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, c = coord_opt(best_h, tf, nf, t0, budget_s, max_sweeps=120)
        if c < best_c5:
            best_c5 = c

    # Block coordinate optimization
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, c = block_opt(best_h, tf, nf, t0, budget_s, block_size=4, n_blocks=300)
        if c < best_c5:
            best_c5 = c
        best_h, c = coord_opt(best_h, tf, nf, t0, budget_s, max_sweeps=60)
        if c < best_c5:
            best_c5 = c

    # Perturbation attempts to escape local optima
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        scales = [0.015, 0.01, 0.005, 0.003]
        for s in scales:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            for att in range(3):
                remaining = budget_s - (time.time() - t0)
                if remaining < 4:
                    break
                hp = project_h(best_h + np.random.randn(nf) * s, tf)
                hp = slsqp_opt(hp, nf, tf, 2000, maxiter=3000, ftol=1e-14)
                c = compute_c5(hp)
                if c < best_c5:
                    best_h = hp
                    best_c5 = c
                    hp, c2 = coord_opt(hp, tf, nf, t0, budget_s, max_sweeps=30)
                    if c2 < best_c5:
                        best_h, best_c5 = hp, c2

    # Final verification
    best_h = project_h(best_h, tf)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, nf
