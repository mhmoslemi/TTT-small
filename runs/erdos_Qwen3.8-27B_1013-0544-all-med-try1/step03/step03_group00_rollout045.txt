            h_local = lbfgsb_optimize(h_local, b, maxiter, pen, top_k=35)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()
    n = len(h_cur)

    # Phase 1: L-BFGS-B continuation with increasing beta
    for beta in [15, 40, 80, 150, 300, 500]:
        if remaining() < 50:
            break
        for _ in range(2):
            if remaining() < 25:
                break
            h_new = lbfgsb_optimize(h_cur, beta, 600, 500.0, top_k=40)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                h_best = h_new.copy()
            h_cur = h_new

    # Phase 2: SLSQP polish for tightness
    if remaining() > 100:
        for beta in [300, 500, 800]:
            if remaining() < 45:
                break
            h_new = slsqp_optimize(h_cur, beta, 500, top_k=None)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                h_best = h_new.copy()
            h_cur = h_new

    # Phase 3: Basin hopping with L-BFGS-B (many hops)
    h_best, best_c5 = basin_hop_lbfgsb(
        h_best, best_c5, n_hops=25,
        noise_base=0.02, noise_decay=0.82,
        beta_list=[150, 400], maxiter=400, pen=1000.0,
        seed=seed + 1, t0=t0, budget_s=budget_s)

    # Phase 4: Fine basin hopping
    if remaining() > 80:
        h_best, best_c5 = basin_hop_lbfgsb(
            h_best, best_c5, n_hops=15,
            noise_base=0.008, noise_decay=0.85,
            beta_list=[400, 700], maxiter=350, pen=2000.0,
            seed=seed + 100, t0=t0, budget_s=budget_s)

    # Phase 5: Try higher resolution
    if remaining() > 150:
        for n_try in [400, 300]:
            if remaining() < 80:
                break
            h_interp = interp_to_n(h_best, n_try)
            h_c = h_interp
            for beta in [50, 150, 400]:
                if remaining() < 40:
                    break
                h_c = lbfgsb_optimize(h_c, beta, 400, 800.0, top_k=50)
            c_try = compute_c5(h_c)
            if c_try < best_c5:
                best_c5 = c_try
                h_best = h_c.copy()
            if remaining() > 60:
                h_best, best_c5 = basin_hop_lbfgsb(
                    h_best, best_c5, n_hops=8,
                    noise_base=0.015, noise_decay=0.8,
                    beta_list=[200, 500], maxiter=300, pen=1000.0,
                    seed=seed + 200 + n_try, t0=t0, budget_s=budget_s)

    # Phase 6: Final SLSQP polish at best resolution
    if remaining() > 45:
        h_fin = slsqp_optimize(h_best, 600, 600, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 7: Ultra-fine basin hopping
    if remaining() > 50:
        h_best, best_c5 = basin_hop_lbfgsb(
            h_best, best_c5, n_hops=10,
            noise_base=0.005, noise_decay=0.88,
            beta_list=[600, 900], maxiter=300, pen=3000.0,
            seed=seed + 500, t0=t0, budget_s=budget_s)

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    final_c5 = compute_c5(h_best)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return h_best, best_c5, len(h_best)
