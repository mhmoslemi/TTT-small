The objective C₅ = max_k ∫h(x)(1-h(x+k))dx can be rewritten.

Let's define the auto-correlation: A(k) = ∫h(x)h(x+k)dx (over valid x)
And let B(k) = ∫h(x)dx (over valid x for shift k)

Then C₅ = max_k [B(k) - A(k)]

For the full domain [0,2] with h defined on [0,2]:
- B(k) = ∫_{max(0,-k)}^{min(2,2-k)} h(x) dx
- A(k) = ∫_{max(0,-k)}^{min(2,2-k)} h(x)h(x+k) dx

For k=0: B(0) = ∫₀² h = 1, A(0) = ∫₀² h² ≤ 1
So C₅(0) = 1 - ∫h²

To minimize C₅(0), we want to maximize ∫h², which means making h as "peaked" as possible. But h ≤ 1, so the maximum of ∫h² is 1 (when h=1 on a set of measure 1). But then for other k, the overlap might be larger.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (d / t) * h
    return np.clip(h, 0, 1)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        if sh > 0:
            grad[:n - sh] += wk * (1 - h[sh:])
            grad[sh:] -= wk * h[:n - sh]
        elif sh < 0:
            ss = -sh
            grad[ss:] += wk * (1 - h[:n - ss])
            grad[:n - ss] -= wk * h[ss:]
        else:
            grad += wk * (1 - 2 * h)
    return grad

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return float(m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha)

def pgd_optimize(h, n, t_end, alpha_start=200, alpha_max=500000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alpha = alpha_start
    lr = 0.3
    stall = 0
    while time.time() < t_end:
        g = compute_grad(h, n, alpha)
        c_cur = compute_c5(h, n)
        ok = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            ht = project_feasible(h - lr * f * g, n)
            ct = compute_c5(ht, n)
            if ct < c_cur - 1e-15:
                h = ht
                ok = True
                break
        if ok:
            stall = 0
            lr = min(lr * 1.05, 2.0)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.7
            if stall > 40:
                alpha = min(alpha * 2.5, alpha_max)
                lr = 0.2
                stall = 0
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=2000):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def jobj(x):
        return compute_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jobj, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, t_end, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > t_end:
            break
        progress = True
        while progress and time.time() < t_end:
            progress = False
            for i in range(n):
                if time.time() > t_end:
                    return h, best_c
                for j in range(i + 1, n):
                    for sgn in (1, -1):
                        hi = h[i] + sgn * d
                        hj = h[j] - sgn * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            oi, oj = h[i], h[j]
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                progress = True
                            else:
                                h[i], h[j] = oi, oj
    return h, best_c

def binary_push(h, n, rng):
    """Try pushing h towards binary and re-optimizing."""
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    target = n / 2
    cur = h_bin.sum()
    if cur > target:
        idx = np.where(h_bin == 1.0)[0]
        order = rng.permutation(len(idx))
        for i in range(int(cur - target)):
            h_bin[idx[order[i]]] = 0.0
    elif cur < target:
        idx = np.where(h_bin == 0.0)[0]
        order = rng.permutation(len(idx))
        for i in range(int(target - cur)):
            h_bin[idx[order[i]]] = 1.0
    return h_bin

def make_starts(n, rng, h_ref, n_ref):
    starts = []
    x = np.linspace(0, 2, n)
    if h_ref is not None:
        x_old = np.linspace(0, 2, n_ref)
        starts.append(project_feasible(np.interp(x, x_old, h_ref), n))
    starts.append(np.full(n, 0.5))
    for k in [1, 2, 3, 5, 7, 11]:
        starts.append(project_feasible(
            0.5 + 0.3 * np.sin(k * np.pi * x) + 0.1 * np.cos(2 * k * np.pi * x), n))
    for _ in range(5):
        starts.append(project_feasible(rng.uniform(0, 1, n), n))
    for period in [3, 4, 6, 7]:
        pat = np.zeros(n)
        for i in range(n):
            if (i % period) < period // 2:
                pat[i] = 1.0
        starts.append(project_feasible(pat, n))
    return starts

def optimize_n(h_start, n, t_end, rng):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # PGD phase with increasing alpha
    alpha_seq = [100, 500, 2000, 10000, 50000]
    for a in alpha_seq:
        if time.time() > t_end - 30:
            break
        ht, ct = pgd_optimize(best_h, n, min(time.time() + (t_end - time.time()) * 0.25, t_end - 30), a)
        if ct < best_c:
            best_c, best_h = ct, ht

    # SLSQP polish
    if time.time() < t_end - 20:
        h2 = slsqp_refine(best_h, n, 50000, 2000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Try a few diverse starts
    if time.time() < t_end - 40:
        t_sub = time.time() + (t_end - time.time()) * 0.3
        for s in make_starts(n, rng, best_h, n):
            if time.time() > t_sub:
                break
            cs = compute_c5(s, n)
            if cs < best_c:
                best_c, best_h = cs, s.copy()
            ht, ct = pgd_optimize(s, n, min(time.time() + 20, t_sub), 500)
            if ct < best_c:
                best_c, best_h = ct, ht
            if time.time() < t_sub - 5:
                h3 = slsqp_refine(ht, n, 20000, 1000)
                c3 = compute_c5(h3, n)
                if c3 < best_c:
                    best_c, best_h = c3, h3

    # Binary push experiment
    if time.time() < t_end - 30 and n <= 120:
        h_bin = binary_push(best_h, n, rng)
        c_bin = compute_c5(h_bin, n)
        h4, c4 = pgd_optimize(h_bin, n, min(time.time() + 30, t_end - 10), 500)
        if c4 < best_c:
            best_c, best_h = c4, h4

    return best_h, best_c
