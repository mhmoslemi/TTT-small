            grad[lag:] -= h[:n - lag]
        else:
            m = -lag
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad

    ones_idx = list(np.where(cur_b == 1)[0])
    zeros_idx = list(np.where(cur_b == 0)[0])
    
    max_rounds = 5000
    for rnd in range(max_rounds):
        if time.time() - t0 > limit_s:
            break
        improved = False
        # Try swapping: flip one 1 to 0 and one 0 to 1
        # Focus on positions that contribute to high lags
        top_lags = np.argsort(corr_full)[::-1][:20]
        candidate_ones = []
        for li in top_lags[:10]:
            lag = int(li) - (n - 1)
            if lag >= 0:
                cand = np.arange(0, n - lag)
                mask = (cur_b[cand] == 1) & (cur_b[cand + lag] == 0)
                candidate_ones.extend(cand[mask].tolist())
            else:
                m = -lag
                cand = np.arange(m, n)
                mask = (cur_b[cand] == 1) & (cur_b[cand - m] == 0)
                candidate_ones.extend(cand[mask].tolist())
        
        candidate_ones = list(set(candidate_ones))[:50]
        
        for i in candidate_ones:
            if time.time() - t0 > limit_s:
                break
            for j in zeros_idx:
                if j == i:
                    continue
                # Try swap
                cur_b[i] = 0.0
                cur_b[j] = 1.0
                new_corr = np.correlate(cur_b, 1.0 - cur_b, mode="full")
                new_max = float(new_corr.max())
                if new_max < best_max - 1e-12:
                    best_b = cur_b.copy()
                    best_max = new_max
                    ones_idx.remove(i)
                    ones_idx.append(j)
                    zeros_idx.remove(j)
                    zeros_idx.append(i)
                    corr_full = new_corr
                    improved = True
                    break
                else:
                    cur_b[i] = 1.0
                    cur_b[j] = 0.0
            if improved:
                break
        if not improved and rnd > 100:
            break
    final_c5 = compute_c5(best_b)
    return best_b, final_c5

def try_multiple_binary_thresholds(h, target, t0, limit_s):
    n = len(h)
    best_b, best_c5 = None, 1e9
    for t in np.linspace(0.3, 0.7, 17):
        if time.time() - t0 > limit_s:
            break
        b = (h >= t).astype(float)
        deficit = target - b.sum()
        if deficit > 0.5:
            zeros = np.where(b == 0)[0]
            order = zeros[np.argsort(h[zeros])[::-1]]
            b[order[:int(np.ceil(deficit))]] = 1.0
        elif deficit < -0.5:
            ones = np.where(b == 1)[0]
            order = ones[np.argsort(h[ones])]
            b[order[:int(np.ceil(-deficit))]] = 0.0
        if abs(b.sum() - target) < 1e-6:
            c5 = compute_c5(b)
            if c5 < best_c5:
                best_b, best_c5 = b.copy(), c5
    return best_b, best_c5

def interpolate_to(h, n_new):
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)
    
    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient at base resolution (~200s)
    limit1 = min(200, budget_s * 0.25)
    h1, c5_1 = subgrad_optimize(best_h, n_base / 2.0, t0, limit1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    
    # Phase 2: Interpolate to 2x and optimize (~250s)
    n2 = min(n_base * 2, 600)
    if budget_s - (time.time() - t0) > 60:
        limit2 = min(250, budget_s * 0.3 - (time.time() - t0))
        h2_init = project_h(interpolate_to(best_h, n2), n2 / 2.0)
        h2, c5_2 = subgrad_optimize(h2_init, n2 / 2.0, t0, limit2)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 3: Try a 3x resolution if time permits (~200s)
    n3 = min(n_base * 3, 900)
    if budget_s - (time.time() - t0) > 80:
        limit3 = min(200, budget_s * 0.45 - (time.time() - t0))
        h3_init = project_h(interpolate_to(best_h, n3), n3 / 2.0)
        h3, c5_3 = subgrad_optimize(h3_init, n3 / 2.0, t0, limit3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
    
    # Phase 4: Binary threshold search at best resolution
    n_best = len(best_h)
    if budget_s - (time.time() - t0) > 30:
        limit4 = min(60, budget_s - (time.time() - t0) - 10)
        hb, cb = try_multiple_binary_thresholds(best_h, n_best / 2.0, t0, limit4)
        if hb is not None and cb < best_c5:
            best_h, best_c5 = hb, cb
    
    # Phase 5: Binary local search on best binary or continuous solution
    if budget_s - (time.time() - t0) > 20:
        n_bs = len(best_h)
        # If best is already binary, use it; otherwise threshold at 0.5
        if np.all((best_h == 0) | (best_h == 1)):
            b_start = best_h.copy()
        else:
            b_start = (best_h >= 0.5).astype(float)
            deficit = n_bs / 2.0 - b_start.sum()
            if deficit > 0.5:
                z = np.where(b_start == 0)[0]
                o = z[np.argsort(best_h[z])[::-1]]
                b_start[o[:int(np.ceil(deficit))]] = 1.0
            elif deficit < -0.5:
                o = np.where(b_start == 1)[0]
                o = o[np.argsort(best_h[o])]
                b_start[o[:int(np.ceil(-deficit))]] = 0.0
        if abs(b_start.sum() - n_bs / 2.0) < 1e-6:
            limit5 = max(10, budget_s - (time.time() - t0) - 5)
            b_final, c5_bin = binary_local_search(b_start, n_bs / 2.0, t0, limit5)
            if b_final is not None and c5_bin < best_c5:
                best_h, best_c5 = b_final, c5_bin
    
    # Phase 6: Final subgradient polish on best solution
    if budget_s - (time.time() - t0) > 15:
        n_fin = len(best_h)
        limit6 = max(5, budget_s - (time.time() - t0) - 2)
        h_fin, c5_fin = subgrad_optimize(best_h, n_fin / 2.0, t0, limit6, max_iter=5000)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
        # Also try binary rounding of the polished solution
        if not np.all((best_h == 0) | (best_h == 1)):
            hb2, cb2 = try_multiple_binary_thresholds(best_h, n_fin / 2.0, t0, 5)
            if hb2 is not None and cb2 < best_c5:
                best_h, best_c5 = hb2, cb2
    
    # Ensure feasibility
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, int(len(best_h))
