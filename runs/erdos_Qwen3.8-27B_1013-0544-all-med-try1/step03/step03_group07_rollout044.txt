    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP full smooth max, increasing beta
    for beta in [200, 500, 1000, 2000, 4000]:
        if tl() < 40:
            break
        h_new = slsqp_opt(h_best, beta, maxiter=300, use_topk=False)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Subgradient targeting argmax lag
    if tl() > 100:
        h_sg, c_sg = subgradient_lag1(h_best, 8000, lr0=0.015, decay=0.9997)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
        # Follow up with SLSQP
        if tl() > 40:
            h_new = slsqp_opt(h_best, 2000, maxiter=200, use_topk=False)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 3: Top-k focused SLSQP (focus on worst lags)
    for topk in [3, 5, 8]:
        if tl() < 40:
            break
        for beta in [500, 1500]:
            if tl() < 30:
                break
            h_new = slsqp_opt(h_best, beta, maxiter=300, use_topk=True, topk=topk)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Resolution increase to 600
    if tl() > 250:
        h600 = interp_to_n(h_best, 600)
        n600_best = h600.copy()
        n600_c5 = compute_c5(h600)
        for beta in [300, 800, 2000, 4000]:
            if tl() < 100:
                break
            h_new = slsqp_opt(n600_best, beta, maxiter=250, use_topk=False)
            c = compute_c5(h_new)
            if c < n600_c5:
                n600_best, n600_c5 = h_new, c
        # If 600 is better, keep it; otherwise map back
        if n600_c5 < best_c5:
            h_best = n600_best
            best_c5 = n600_c5
        else:
            # Try to improve 400 by mixing
            h400_from600 = interp_to_n(n600_best, 400)
            h_new = slsqp_opt(h400_from600, 2000, maxiter=300, use_topk=False)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 5: Multi-restart with structured perturbations
    for trial in range(12):
        if tl() < 45:
            break
        n_curr = len(h_best)
        if trial < 4:
            pert = np.random.randn(n_curr) * 0.01
        elif trial < 8:
            # Structured: sinusoidal perturbation
            x = np.linspace(0, 2, n_curr)
            freq = 3 + trial - 4
            pert = 0.02 * np.sin(2 * np.pi * freq * x / 2)
        else:
            # Block perturbation
            block = n_curr // 8
            pert = np.zeros(n_curr)
            for b in range(8):
                pert[b*block:(b+1)*block] = np.random.randn() * 0.008
        h_pert = project_feasible(h_best + pert, n_curr / 2.0)
        h_new = slsqp_opt(h_pert, 1500, maxiter=300, use_topk=False)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 6: Second round of subgradient + SLSQP
    if tl() > 60:
        h_sg2, c_sg2 = subgradient_lag1(h_best, 6000, lr0=0.008, decay=0.9998)
        if c_sg2 < best_c5:
            h_best, best_c5 = h_sg2, c_sg2
        if tl() > 30:
            h_new = slsqp_opt(h_best, 3000, maxiter=400, use_topk=False)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 7: Top-k polish
    if tl() > 40:
        h_new = slsqp_opt(h_best, 3000, maxiter=300, use_topk=True, topk=3)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 8: Final subgradient micro-steps
    if tl() > 30:
        h_sg3, c_sg3 = subgradient_lag1(h_best, 4000, lr0=0.005, decay=0.9999)
        if c_sg3 < best_c5:
            h_best, best_c5 = h_sg3, c_sg3
        if tl() > 20:
            h_new = slsqp_opt(h_best, 5000, maxiter=300, use_topk=False)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Ensure final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
