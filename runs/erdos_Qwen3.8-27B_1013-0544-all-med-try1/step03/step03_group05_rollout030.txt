def project_h(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)

        bh = best[1].copy()
        bc = best[0]
        nf = len(bh)
        tgt = nf / 2.0
        dxf = 2.0 / nf
        step = 0.004
        for it in range(100000):
            if tl() < 1.5:
                break
            corr = np.correlate(bh, 1.0 - bh, mode="full") * dxf
            top = np.argsort(corr)[::-1][:2]
            v = corr[top]
            w = np.exp((v - v[0]) * 250.0)
            w /= w.sum()
            grad = np.zeros(nf)
            center = nf - 1
            for ti, wi in zip(top, w):
                lag = int(ti) - center
                grad += wi * grad_for_lag(bh, lag, nf, dxf)
            gn = np.linalg.norm(grad)
            if gn < 1e-16:
                break
            grad /= gn
            done = False
            for s in [step, step * 0.5, step * 0.25, step * 0.1]:
                ht = project_h(bh - s * grad, tgt)
                ct = compute_c5(ht)
                if ct < bc:
                    bh = ht
                    bc = ct
                    step = min(s * 1.08, 0.01)
                    done = True
                    break
            if not done:
                step *= 0.93
                if step < 1e-6:
                    break

    return bh, bc, len(bh)
