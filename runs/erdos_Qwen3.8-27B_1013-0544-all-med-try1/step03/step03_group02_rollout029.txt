def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            free = np.ones(len(h), dtype=bool)
            nfree = len(h)
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_subgradient(h, top_k, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    weights = np.exp(-np.arange(top_k) / tau)
    weights /= weights.sum()
    grad = np.zeros(n)

        lag = int(idx) - (n - 1)

            grad[lag:] -= w * h[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] -= w * h[la:]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            free = np.ones(len(h), dtype=bool)
            nfree = len(h)
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def weighted_subgradient(h, top_k=10, tau=4.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    weights = np.exp(-np.arange(top_k) / tau)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n-la])
            grad[:n-la] -= w * h[la:]
    return grad

def subgrad_optimize(h0, target, t0, budget, n_iter=None, step0=0.015, top_k=10, tau=4.0):
    n = len(h0)
    if n_iter is None:
        n_iter = int((budget - (time.time() - t0)) * 8)
    cur = h0.copy()
    best = compute_c5(cur)
    mom = np.zeros(n)
    step = step0
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        g = weighted_subgradient(cur, top_k, tau)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.9 * mom - step * g
        trial = project_h(cur + mom, target)
        c = compute_c5(trial)
        if c < best - 1e-14:
            cur = trial
            best = c
            stall = 0
        else:
            stall += 1
            step *= 0.92
            mom *= 0.5
            trial = project_h(cur + mom, target)
            c = compute_c5(trial)
            if c < best - 1e-14:
                cur = trial
                best = c
                stall = 0
        if stall > 500:
            break
    return cur, best

def fourier_perturb(h, target, freqs, scales, t0, budget, n_trials=500):
    """Try perturbations in Fourier basis."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = np.linspace(0, 2*np.pi, n, endpoint=False)
    for trial in range(n_trials):
        if budget - (time.time() - t0) < 2:
            break
        f = np.random.choice(freqs)
        s = np.random.choice(scales)
        phase = np.random.uniform(0, 2*np.pi)
        noise = s * np.cos(f * t + phase)
        # Also add random component
        noise += np.random.randn(n) * s * 0.3
        trial_h = project_h(h + noise, target)
        c = compute_c5(trial_h)
        if c < best_c5 - 1e-14:
            best_h = trial_h
            best_c5 = c
    return best_h, best_c5

def slsqp_pnorm(h0, n, target, p, maxiter, t0, budget):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.sum(corr**p) ** (1.0/p) * (2.0/n))
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except:
        return h0
