import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def build_h_from_v(v):
    n2 = len(v)
    h = np.empty(2 * n2)
    h[:n2] = v
    h[n2:] = 1.0 - v
    return np.clip(h, 0.0, 1.0)

def sml_obj_v(v, beta, dx):
    h = build_h_from_v(v)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sml_obj_full(h_in, beta, dx):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(corr, beta) * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def sum_constr(h, target):
    return h.sum() - target

def sum_constr_jac(h, target):
    return np.ones(len(h))

def optimize_v(v0, n2, beta_list, dx, v_bounds, t0, budget):
    v = v0.copy()
    for beta in beta_list:
        remaining = budget - (time.time() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(sml_obj_v, v, args=(beta, dx), method='L-BFGS-B',
                          bounds=v_bounds, options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-12})
            v = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
    return v

def coord_descent(h, target, deltas, order, t0, budget, max_sweeps=50):
    best_c5 = compute_c5(h)
    nf = len(h)
    sweeps = 0
    while sweeps < max_sweeps and (budget - (time.time() - t0)) > 3:
        improved = False
        for j in order:
            if (budget - (time.time() - t0)) < 2:
                break
            for d in deltas:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-16:
                    h = trial
                    best_c5 = c_try
                    improved = True
                    break
        sweeps += 1
        if not improved and sweeps > 2:
            break
    return h, best_c5
