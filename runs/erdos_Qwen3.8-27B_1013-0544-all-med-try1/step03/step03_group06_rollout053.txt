    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Softmax weights
    m = np.max(corr)

    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]

    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -100, 100)
    w = np.exp(sh)
    w /= w.sum()
    return w

def compute_subgrad(h, weights, n):
    """Compute weighted subgradient of correlation values."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            p = -lag
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad

                         step_scale=0.5, start_time=None, max_time=None):

    target = n / 2.0
    dx = 2.0 / n
    v = np.zeros(n)

    max_t = max_time if max_time else float('inf')

        if time.time() - t0 > max_t:

    n_best = len(best_h)
    if tleft() > 60:
        h_s, c5_s = nesterov_subgrad(best_h, n_best, min(60, tleft() * 0.4), 10, 500)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    if tleft() > 30:
        h_s, c5_s = pairwise_refine(best_h, n_best, min(30, tleft() * 0.3), rng)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    if tleft() > 15:
        best_h, best_c5 = slsqp_polish(best_h, n_best, tleft() * 0.6)

    # Final subgradient sweep
    if tleft() > 5:
        h_s, c5_s = nesterov_subgrad(best_h, len(best_h), tleft() * 0.7, 50, 1000,
                                     momentum=0.95, step0=0.15)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Validate
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = n_out / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        free = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if free.sum() > 0:
            best_h[free] += diff / free.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_out
