def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = dx * corr
    max_x = np.max(x)
    exps = np.exp(beta * (x - max_x))
    weights = exps / np.sum(exps)
    obj = max_x + np.log(np.sum(exps)) / beta
    
    grad = np.zeros(n)
    # Only iterate over lags with significant weight
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for idx in active:
        w = weights[idx]
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            v = n - lag
            grad[:v] += w * (1.0 - h[lag:lag+v])
            grad[lag:] -= w * h[:v]
        elif -n < lag < 0:
            m = -lag
            v = n - m
            grad[m:] += w * (1.0 - h[:v])
            grad[:v] -= w * h[m:m+v]
    
    return obj, grad * dx

        cnt = free.sum()
        if cnt == 0:
            break
        h[free] += diff / cnt

            if stagnant > 100 and it % 150 == 149:
                scale = 0.005 * max(1.0, stagnant / 500.0)
                p = np.random.randn(n_f) * scale
                h_t = project_h(best_h + p, target_f)
                c = compute_c5(h_t)
                if c < best_c5:
                    best_h = h_t
                    best_c5 = c
                    stagnant = 0
                elif stagnant > 2000:
                    break

    # Final high-precision SLSQP
    if time_left() > 12:
        n_p = len(best_h)
        target_p = n_p / 2.0
        bounds_p = Bounds(np.zeros(n_p), np.ones(n_p))
        for beta_p in [5000, 20000]:
            if time_left() < 10:
                break
            so_p = SmoothObj(beta_p)
            con_p = SumCon(target_p, n_p)
            constraints_p = [{'type': 'eq', 'fun': con_p.fun, 'jac': con_p.jac}]
            try:
                res = minimize(so_p.value, best_h, jac=so_p.gradient, method='SLSQP',
                               bounds=bounds_p, constraints=constraints_p,
                               options={'maxiter': 15000, 'ftol': 1e-18})
                h_new = project_h(res.x, target_p)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Last subgradient sweep
    for it in range(20000):
        if time_left() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        grad = np.zeros(n_f)
        for idx in top_idx:
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif 0 < lag < n_f:
                v = n_f - lag
                grad[:v] += (1.0 - best_h[lag:lag + v])
                grad[lag:] -= best_h[:v]
            elif -n_f < lag < 0:
                m = -lag
                v = n_f - m
                grad[m:] += (1.0 - best_h[:v])
                grad[:v] -= best_h[m:m + v]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        for step in [0.01, 0.005, 0.002, 0.001]:
            h_t = project_h(best_h - step * grad, target_f)
            c = compute_c5(h_t)
            if c < best_c5 - 1e-15:
                best_h = h_t
                best_c5 = c
                break

    return best_h, best_c5, len(best_h)
