    dx = 2.0 / n_points
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        sm = smooth_max(corr, beta) * dx
        pen = penalty * (np.sum(x) - target_sum) ** 2
        return sm + pen
    return obj


def build_grad(n_points, beta, target_sum, penalty):
    dx = 2.0 / n_points
    def grad(x):
        g = compute_grad_smooth(x, beta)
        g += 2.0 * penalty * (np.sum(x) - target_sum) * np.ones(n_points)
        return g
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(120, 0.5)

    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-resolution L-BFGS-B with analytical gradient
    resolutions = [100, 150, 200, 300, 400]

    for n_pts in resolutions:
        if time_left() < 80:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_pts)
        h_cur = project_h(np.interp(x_new, x_old, best_h), target)
        cur_c5 = compute_c5(h_cur)
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

        for beta in [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]:
            if time_left() < 20:
                break
            penalty = 5000.0
            obj = build_obj(n_pts, beta, target, penalty)
            grad = build_grad(n_pts, beta, target, penalty)
            try:
                res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-14,
                                        'gtol': 1e-10, 'maxfun': 8000})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h_cur = h_new.copy()
                    cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = cur_c5
            except Exception:
                pass

    # Phase 2: Targeted subgradient on active lags
    n_f = len(best_h)
    target_f = n_f / 2.0
    no_improve = 0

    for it in range(30000):
        if time_left() < 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_active = min(9, max(4, n_f // 25))
        top_idx = np.argsort(corr)[::-1][:n_active]
        top_lags = [int(i) - (n_f - 1) for i in top_idx]

        grad = np.zeros(n_f)
        for lag in top_lags:
            if lag >= 0:
                grad[:n_f - lag] += (1.0 - best_h[lag:])
                grad[lag:] -= best_h[:n_f - lag]
            else:
                ml = -lag
                grad[ml:] += (1.0 - best_h[:n_f - ml])
                grad[:n_f - ml] -= best_h[ml:]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm

        found = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best_h - step * grad, target_f)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_h = h_trial
                best_c5 = c5_trial
                found = True
                no_improve = 0
                break

        if not found:
            no_improve += 1
            if no_improve > 150:
                if it % 200 == 199:
                    scale = 0.005 * (1.0 + 50.0 / (no_improve - 149))
                    perturb = np.random.randn(n_f) * scale
                    h_trial = project_h(best_h + perturb, target_f)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial
                        best_c5 = c5_trial
                        no_improve = 0
                if no_improve > 3000:
                    break

    # Phase 3: Pairwise moves on critical lags
    n_f = len(best_h)
    target_f = n_f / 2.0

    for it in range(50000):
        if time_left() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        improved = False

        for idx in top_idx:
            if time_left() < 1:
                break
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                # Move a single point toward 0 or 1
                j = int(np.argmin(np.minimum(best_h, 1.0 - best_h)))
                for d in [0.03, -0.03, 0.015, -0.015, 0.007, -0.007]:
                    h_trial = best_h.copy()
                    h_trial[j] += d
                    if 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target_f)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-16:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True
                            break
            else:
                abs_lag = abs(lag)
                if abs_lag >= n_f:
                    continue
                n_valid = n_f - abs_lag
                for _ in range(8):
                    i = np.random.randint(n_valid)
                    j = i + abs_lag if lag > 0 else i - lag
                    # Move apart: increase max, decrease min
                    if best_h[i] > best_h[j]:
                        d = 0.03
                    else:
                        d = -0.03
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target_f)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-16:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True
                            break
                    if improved:
                        break
                # Also try "move together" sometimes
                if not improved and np.random.random() < 0.3:
                    i = np.random.randint(n_valid)
                    j = i + abs_lag if lag > 0 else i - lag
                    if best_h[i] > best_h[j]:
                        d = -0.02
                    else:
                        d = 0.02
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target_f)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-16:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True

        if not improved and it % 400 == 399:
            perturb = np.random.randn(n_f) * 0.004
            h_trial = project_h(best_h + perturb, target_f)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial

    # Phase 4: Final L-BFGS-B polish at high beta
    if time_left() > 10:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        bounds_pol = Bounds(np.zeros(n_pol), np.ones(n_pol))
        for beta in [3000, 8000, 15000]:
            if time_left() < 5:
                break
            penalty = 20000.0
            obj = build_obj(n_pol, beta, target_pol, penalty)
            grad = build_grad(n_pol, beta, target_pol, penalty)
            try:
                res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                               bounds=bounds_pol,
                               options={'maxiter': 2000, 'ftol': 1e-15,
                                        'gtol': 1e-12, 'maxfun': 10000})
                h_new = project_h(res.x, target_pol)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Final subgradient sweep
    n_f = len(best_h)
    target_f = n_f / 2.0
    for it in range(5000):
        if time_left() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:5]
        top_lags = [int(i) - (n_f - 1) for i in top_idx]
        grad = np.zeros(n_f)
        for lag in top_lags:
            if lag >= 0:
                grad[:n_f - lag] += (1.0 - best_h[lag:])
                grad[lag:] -= best_h[:n_f - lag]
            else:
                ml = -lag
                grad[ml:] += (1.0 - best_h[:n_f - ml])
                grad[:n_f - ml] -= best_h[ml:]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        for step in [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best_h - step * grad, target_f)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_h = h_trial
                best_c5 = c5_trial
                break

    return best_h, best_c5, len(best_h)
