import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize, Bounds
import time

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} using bisection on lambda."""
    n = len(h)
    # Bisection on lambda: h[i] = clip(x[i] - lambda, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_grad_fft(h, beta):
    """Compute smooth max value and gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw  # length 2n-1
    val = (m + np.log(sw) / beta) * dx
    
    # FFT-based gradient
    # W = w (length 2n-1)
    W = w
    first_sum = fftconvolve(W[::-1], (1.0 - h), mode='full')[2*n-2 : 3*n-2]
    second_sum = fftconvolve(W, h[::-1], mode='full')[n-1 : 2*n-1]
    grad = dx * (first_sum - second_sum)
    return val, grad

def projected_gradient(h, beta, max_iters, lr0, decay, use_acceleration=True):
    """Projected gradient with optional Nesterov acceleration."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    if use_acceleration:
        x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        
        if use_acceleration:
            if t == 0:
                y = x.copy()
            else:
                mu = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / (2.0 * (t+1))
                # Actually standard Nesterov: mu_t = (t-1)/(t+2)
                mu = (t - 1.0) / (t + 2.0)
                y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        else:
            y = x.copy()
        
        val, grad = smooth_grad_fft(y, beta)
        
        # Backtracking line search
        step = lr_t
        x_new = project_feasible(y - step * grad, target)
        v_new = compute_c5(x_new)
        
        # Simple: just take the step
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        
        if use_acceleration:
            x_prev = x
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smooth_grad_fft(x, beta)
        return v

        _, g = smooth_grad_fft(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]

import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize, Bounds
import time

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_grad_fft(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    # Use log-sum-exp for numerical stability
    log_sw = np.log(np.sum(np.exp(beta * shifted)))
    w = np.exp(beta * (corr - m - log_sw / beta))
    # Actually let me redo: w[k] = exp(beta*(corr[k]-m)) / sum(exp(beta*(corr-m)))
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    # FFT-based gradient
    V = w  # length 2n-1
    # A[j] = convolve((1-h)[::-1], V, 'full')[2n-2-j]
    conv1 = fftconvolve((1.0 - h)[::-1], V, mode='full')
    A = conv1[n-1:2*n-1][::-1]
    # B[j] = convolve(h, V, 'full')[j+n-1]
    conv2 = fftconvolve(h, V, mode='full')
    B = conv2[n-1:2*n-1]
    grad = dx * (A - B)
    return val, grad

def projected_gradient(h, beta, max_iters, lr0, decay, accelerate=True):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    if accelerate:
        x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if accelerate:
            if t == 0:
                y = x.copy()
            else:
                mu = (t - 1.0) / (t + 2.0)
                y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        else:
            y = x.copy()
        
        _, grad = smooth_grad_fft(y, beta)
        x_new = project_feasible(y - lr_t * grad, target)
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        if accelerate:
            x_prev = x
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smooth_grad_fft(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smooth_grad_fft(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

        return budget_s - (time.time() - t0)

import numpy as np

_beta_curr = 100.0

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_grad_fft(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    V = w
    conv1 = fftconvolve((1.0 - h)[::-1], V, mode='full')
    A = conv1[n-1:2*n-1][::-1]
    conv2 = fftconvolve(h, V, mode='full')
    B = conv2[n-1:2*n-1]
    grad = dx * (A - B)
    return val, grad

def _obj_func(x):
    v, _ = smooth_grad_fft(x, _beta_curr)
    return v

def _obj_grad(x):
    _, g = smooth_grad_fft(x, _beta_curr)
    return g

    return x.sum() - _target_sum

    return np.ones(len(x))

def slsqp_refine(h, maxiter, beta):

    n = len(h)
    _target_sum = n / 2.0
    _beta_curr = beta

    cons = [{'type': 'eq', 'fun': _con_func, 'jac': _con_jac}]
    try:
        res = minimize(_obj_func, h, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h.copy(), n / 2.0)

def projected_gradient(h, beta, max_iters, lr0, decay, accelerate=True):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    if accelerate:
        x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if accelerate:
            if t == 0:
                y = x.copy()
            else:
                mu = (t - 1.0) / (t + 2.0)
                y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        else:
            y = x.copy()
        _, grad = smooth_grad_fft(y, beta)
        x_new = project_feasible(y - lr_t * grad, target)
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        if accelerate:
            x_prev = x
        x = x_new
    return best_x, best_v

    h_new = np.clip(h_new, 0.0, 1.0)
