    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()

def smooth_max_and_weights(c, beta):
    m = np.max(c)
    x = np.clip(beta * (c - m), -500, 500)
    exp_x = np.exp(x)
    s = np.sum(exp_x)
    sm = m + np.log(s) / beta
    w = exp_x / s
    return sm, w

def correlations(h):

def smooth_max_obj(h, beta):

    dx = 2.0 / n
    c = correlations(h)
    sm, _ = smooth_max_and_weights(c, beta)
    return sm * dx

def smooth_max_grad(h, beta):

    c = correlations(h)

            grad[:n-m] -= wk * h[m:]

                                best_h = ht.copy()
                                best_c5 = ct
                                improved = True
                                break
                    if improved:
                        break
            else:
                al = abs(lag)
                if al >= n:
                    continue
                for _ in range(6):
                    i = np.random.randint(n - al)
                    j = i + al if lag > 0 else i - lag
                    for d in [0.012, -0.012, 0.006, -0.006, 0.003, -0.003]:
                        ht = best_h.copy()
                        ht[i] += d
                        ht[j] -= d
                        if 0 <= ht[i] <= 1 and 0 <= ht[j] <= 1:
                            ht = project_h(ht, target)
                            ct = compute_c5(ht)
                            if ct < best_c5 - 1e-15:
                                best_h = ht.copy()
                                best_c5 = ct
                                improved = True
                                break
                    if improved:
                        break
        if not improved and it % 400 == 399:
            p = np.random.randn(n) * 0.0008
            ht = project_h(best_h + p, target)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_h = ht.copy()
                best_c5 = ct

    # Phase 5: Final SLSQP polish
    if tl() > 20:
        h_new = slsqp_opt(best_h, target, 50000, 5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new

    return best_h, best_c5, len(best_h)
