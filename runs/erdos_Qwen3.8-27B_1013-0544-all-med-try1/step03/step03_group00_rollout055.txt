def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)  # 2n-1
    
    if top_k is not None and top_k < nc:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(nc)
    
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    for idx in top_idx:
        wk = w_sel[idx]
        if wk < 1e-12:
            continue
        shift = idx - (n - 1)
        # d/dh[j] corr[idx] = (1-h[j-shift])*I[0<=j-shift<n] - h[j+shift]*I[0<=j+shift<n]
        # First term: j-shift in [0,n) => j in [shift, shift+n), intersect [0,n) => [max(0,shift), min(n,shift+n))
        j_lo1 = max(0, shift)
        j_hi1 = min(n, shift + n)
        if j_hi1 > j_lo1:
            # (1 - h[j-shift]) for j in [j_lo1, j_hi1)
            grad[j_lo1:j_hi1] += wk * (1.0 - h[j_lo1 - shift : j_hi1 - shift])
        
        # Second term: j+shift in [0,n) => j in [-shift, n-shift), intersect [0,n) => [max(0,-shift), min(n,n-shift))
        j_lo2 = max(0, -shift)
        j_hi2 = min(n, n - shift)
        if j_hi2 > j_lo2:
            # -h[j+shift] for j in [j_lo2, j_hi2)
            grad[j_lo2:j_hi2] += wk * (-h[j_lo2 + shift : j_hi2 + shift])
    
    grad *= dx
    return val, grad

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

            if diff > 0:

                order = np.argsort(h)
                for idx in order:
                    room = 1.0 - h[idx]
                    add = min(room, diff)
                    h[idx] += add
                    diff -= add
                    if diff <= 1e-14:

            else:

                    rem = min(room, -diff)

        if j_hi2 > j_lo2:

    grad *= dx

def lbfgsb_step(h, target, beta, maxiter=200, pen_mult=100.0):
    n = len(h)
    
    def obj(x):
        sm_val, _ = smoothmax_val_grad(x, beta)
        pen = pen_mult * (x.sum() - target) ** 2
        return sm_val + pen
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        g = g + 2.0 * pen_mult * (x.sum() - target)
        return g
    
    res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    return project_feasible(res.x, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    except Exception:
        return project_feasible(h, target)


def lbfgsb_step(h, target, beta, maxiter=300, pen_mult=500.0, top_k=None):
    n = len(h)

    def obj(x):
        sm_val, _ = smoothmax_val_grad(x, beta, top_k)
        pen = pen_mult * (x.sum() - target) ** 2
        return sm_val + pen

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        g = g + 2.0 * pen_mult * (x.sum() - target)
        return g

    res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    return project_feasible(res.x, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def block_flip_perturb(h, target, block_size, rng):
    n = len(h)
    h_new = h.copy()
    if block_size >= n:
        return h_new
    start = rng.randint(0, n - block_size)
    h_new[start:start + block_size] = 1.0 - h_new[start:start + block_size]
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    n = len(h_cur)
    target = n / 2.0
    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: L-BFGS-B warmup (fast)
    for beta in [30, 100, 300]:
        if remaining() < 20:
            break
        h_cur = lbfgsb_step(h_cur, target, beta, maxiter=250)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            h_best = h_cur.copy()

    # Phase 2: SLSQP continuation with increasing beta
    for beta in [200, 500, 1000, 2000]:
        if remaining() < 50:
            break
        h_cur = slsqp_solve(h_cur, beta, maxiter=600, top_k=None)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            h_best = h_cur.copy()

    # Phase 3: Basin hopping with mixed perturbations
    rng = np.random.RandomState(seed)
    n_hops = 50
    for hop in range(n_hops):
        if remaining() < 45:
            break
        # Mix of Gaussian noise and block flips
        if hop % 7 == 3:
            bs = rng.randint(5, 20)
            h_pert = block_flip_perturb(h_best, target, bs, rng)
        elif hop % 7 == 6:
            noise = 0.05
            h_pert = h_best + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
        else:
            noise = 0.02 * (0.88 ** (hop % 15))
            h_pert = h_best + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
        # Local optimization: L-BFGS-B then SLSQP polish
        h_local = lbfgsb_step(h_pert, target, 200, maxiter=150)
        h_local = slsqp_solve(h_local, 500, maxiter=350, top_k=40)
        c = compute_c5(h_local)
        if c < best_c5:
            best_c5 = c
            h_best = h_local.copy()
            n = len(h_best)
            target = n / 2.0

    # Phase 4: Multi-resolution exploration
    for n_try in [300, 400, 500]:
        if remaining() < 130:
            break
        h_interp = interp_to_n(h_best, n_try)
        t_try = n_try / 2.0
        h_interp = lbfgsb_step(h_interp, t_try, 100, maxiter=200)
        h_interp = slsqp_solve(h_interp, 400, maxiter=400, top_k=None)
        h_interp = slsqp_solve(h_interp, 1500, maxiter=500, top_k=None)
        c = compute_c5(h_interp)
        if c < best_c5:
            best_c5 = c
            h_best = h_interp.copy()

    # Phase 5: Basin hopping at best resolution (finer)
    n = len(h_best)
    target = n / 2.0
    rng2 = np.random.RandomState(seed + 500)
    for hop in range(25):
        if remaining() < 40:
            break
        if hop % 5 == 4:
            bs = rng2.randint(3, 12)
            h_pert = block_flip_perturb(h_best, target, bs, rng2)
        else:
            noise = 0.01 * (0.85 ** (hop % 10))
            h_pert = h_best + rng2.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
        h_local = lbfgsb_step(h_pert, target, 300, maxiter=200)
        h_local = slsqp_solve(h_local, 800, maxiter=400, top_k=None)
        c = compute_c5(h_local)
        if c < best_c5:
            best_c5 = c
            h_best = h_local.copy()

    # Phase 6: Final high-beta SLSQP squeeze (very precise)
    if remaining() > 40:
        h_best = slsqp_solve(h_best, 5000, maxiter=1200, top_k=None)
        c = compute_c5(h_best)
        if c < best_c5:
            best_c5 = c

    # Phase 7: One more squeeze if time allows
    if remaining() > 25:
        h_best = slsqp_solve(h_best, 8000, maxiter=800, top_k=None)
        c = compute_c5(h_best)
        if c < best_c5:
            best_c5 = c

    # Final validation and projection
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
