import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(corr)) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(xs)
    return e / np.sum(e)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_obj(h, beta, n_active_max=35):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, beta)
    if np.sum(w > 1e-5) > n_active_max:
        top = np.argsort(w)[::-1][:n_active_max]
        w2 = np.zeros(len(w))
        w2[top] = w[top]
        w = w2
    active = np.where(w > 1e-6)[0]
    grad = np.zeros(n)
    for idx in active:
        wi = w[idx]
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            ml = -lag
            grad[:n - ml] += wi * (1.0 - h[ml:])
            grad[ml:] -= wi * h[:n - ml]
    return grad * dx

    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, beta) * (2.0 / n)

def slsqp_grad(x, beta, n):
    return grad_obj(x, beta)

    return np.sum(x) - n / 2.0

def slsqp_eq_jac(x, n):
    return np.ones(len(x))
