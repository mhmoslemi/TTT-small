import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def compute_subgrad(h, lags):
    n = len(h)
    g = np.zeros(n)
    for lag in lags:
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n-lag] -= h[lag:]
        else:
            m = -lag
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    return g

def slsqp(h, n, beta, target, mx=5000):
    try:
        b = Bounds(np.zeros(n), np.ones(n))
        def f(x):
            return smooth_max(np.correlate(x, 1-x, mode="full"), beta) * (2.0/n)
        def cf(x): return np.sum(x) - target
        def cj(x): return np.ones(n)
        r = minimize(f, h.copy(), method='SLSQP', bounds=b,
                     constraints=[{'type':'eq','fun':cf,'jac':cj}],
                     options={'maxiter':mx,'ftol':1e-15})
        return project_h(r.x, target)
    except: return h

def subgrad(h, target, max_it, t_limit, t_start):
    n = len(h)
    best = compute_c5(h)
    ni = 0
    for it in range(max_it):
        if time.time() - t_start > t_limit: break
        corr = np.correlate(h, 1-h, mode="full")
        na = min(10, max(5, n//15))
        top = np.argsort(corr)[::-1][:na]
        lags = [int(i)-(n-1) for i in top]
        g = compute_subgrad(h, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-12: break
        g /= gn
        ok = False
        for s in [0.05,0.03,0.015,0.008,0.004,0.002,0.001,0.0005]:
            ht = project_h(h - s*g, target)
            c = compute_c5(ht)
            if c < best - 1e-14:
                h, best = ht, c
                ok = True; ni = 0; break
        if not ok:
            ni += 1
            if ni > 200 and it % 500 == 499:
                ht = project_h(h + np.random.randn(n)*0.012, target)
                if compute_c5(ht) < best:
                    h = ht; best = compute_c5(h); ni = 0
            if ni > 4000: break
    return h, best
