import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        sm = shifted.max()
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lbfgs_pen_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max(c, alpha) + lam * (x.sum() - n / 2.0) ** 2

def lbfgs_step(h, n, alpha, lam, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_pen_obj, h, args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def correlation_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def pair_local_search(h, n, deadline, deltas=[0.01, 0.005, 0.001, 0.0005]):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = best_h[i] + s * d
                        hj = best_h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            old_i, old_j = best_h[i], best_h[j]
                            best_h[i] = hi
                            best_h[j] = hj
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] = old_i
                                best_h[j] = old_j
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter=2000):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max(c, alpha)

    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        if shifted.max() > 500:
            sm = shifted.max()
            w = np.exp(shifted - sm)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
        grad = np.zeros(n)
        nfull = len(c)
        for k in range(nfull):
            offset = k - (n - 1)
            wk = w[k]
            valid_first = np.zeros(n, dtype=bool)
            io = np.arange(n) + offset
            valid_first = (io >= 0) & (io < n)
            grad[valid_first] += wk * (1.0 - np.where(valid_first, 1.0 - x[np.clip(io, 0, n-1)], 0.0))
            valid_second = np.zeros(n, dtype=bool)
            io2 = np.arange(n) - offset
            valid_second = (io2 >= 0) & (io2 < n)
            grad[valid_second] -= wk * np.where(valid_second, x[np.clip(io2, 0, n-1)], 0.0)
        grad *= dx
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def harmonic_to_h(params, n, n_harm):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, n_harm + 1):
        h += params[2*(k-1)] * np.cos(2*np.pi*k*i/n)
        h += params[2*(k-1)+1] * np.sin(2*np.pi*k*i/n)
    return project_feasible(h, n)

def harmonic_optimize(n, rng, deadline, max_rounds=30):
    n_harm = min(20, n // 3)
    dim = 2 * n_harm
    best_h = None
    best_c = float('inf')

            c_res = compute_c5(h_res, n)

                best_c = c_res
                best_h = h_res.copy()
                params = extract_harmonics(h_res, n, n_harm)
    return best_h, best_c

def extract_harmonics(h, n, n_harm):
    i = np.arange(n)
    params = np.zeros(2 * n_harm)
    for k in range(1, n_harm + 1):
        params[2*(k-1)] = 2.0/n * np.sum((h - 0.5) * np.cos(2*np.pi*k*i/n))
        params[2*(k-1)+1] = 2.0/n * np.sum((h - 0.5) * np.sin(2*np.pi*k*i/n))
    return params

def make_diverse_starts(n, rng, h_ref):
    starts = [h_ref.copy()]
    x = np.linspace(0, 2, n, endpoint=False)
    for k in range(1, 8):

        starts.append(project_feasible(h_s, n))

    for _ in range(5):
