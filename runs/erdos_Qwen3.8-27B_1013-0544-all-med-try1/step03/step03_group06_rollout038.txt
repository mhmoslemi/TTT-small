    dx = 2.0 / len(h)

    return float(np.max(corr) * dx)

def smooth_max_val(c, beta):
    m = np.max(c)
    s = np.clip(beta * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(c)) / beta

def smooth_max_weights(c, beta):

    s = np.clip(beta * (c - m), -500, 500)
    e = np.exp(s)
    return e / np.sum(e)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
