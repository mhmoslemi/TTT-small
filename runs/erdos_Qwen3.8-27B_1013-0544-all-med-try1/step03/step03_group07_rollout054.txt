def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad_fast(x, beta)
        penalty = mu * (x.sum() - target) ** 2
        return v + penalty

        _, g = smoothmax_val_grad_fast(x, beta)
        g = g + 2 * mu * (x.sum() - target)
        return g

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad_fast(h, beta):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 4 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    OMH = np.fft.rfft(1.0 - h, N)
    H_rev = np.fft.rfft(h[::-1], N)
    corr = np.fft.irfft(OMH * H_rev, N)[:2 * n - 1]
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    W = np.fft.rfft(w, N)
    U_flip = np.fft.rfft((1.0 - h)[::-1], N)
    conv_uw = np.fft.irfft(U_flip * W, N)
    S1 = conv_uw[n - 1:2 * n - 1][::-1]
    conv_wh = np.fft.irfft(W * H, N)
    S2 = conv_wh[n - 1:2 * n - 1]
    grad = dx * (S1 - S2)
    return val, grad

def lbfgs_opt(h, beta, mu, maxiter):
    n = len(h)
    target = n / 2.0
    def obj(x):
        v, _ = smoothmax_val_grad_fast(x, beta)
        return v + mu * (x.sum() - target) ** 2
    def grad(x):
        _, g = smoothmax_val_grad_fast(x, beta)
        return g + 2.0 * mu * (x.sum() - target)
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def nesterov_opt(h, beta, max_iters, lr0, decay):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    y = h.copy()
    best_x = h.copy()
    best_c5 = compute_c5(h)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        _, grad = smoothmax_val_grad_fast(y, beta)
        x_new = project_feasible(y - lr_t * grad, target)
        mu_t = (t) / (t + 2)
        y = project_feasible(x_new + mu_t * (x_new - x), target)
        x = x_new
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
    return best_x, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS-B at starting resolution with increasing beta
    for beta in [40, 80, 160, 320, 640]:
        if tl() < 25:
            break
        for mu in [0.5, 5.0, 50.0]:
            if tl() < 12:
                break
            h_new = lbfgs_opt(h_best, beta, mu, maxiter=300)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new.copy(), c

    # Phase 2: Nesterov with fast gradient
    if tl() > 100:
        h_sg, c_sg = nesterov_opt(h_best, 200, 3000, 0.05, 0.9995)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg.copy(), c_sg
        # Polish with L-BFGS-B
        if tl() > 30:
            h_new = lbfgs_opt(h_best, 400, 10.0, maxiter=300)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new.copy(), c

    # Phase 3: Higher resolution
    for n_try in [600, 800]:
        if tl() < 250:
            break
        h_interp = interp_to_n(h_best, n_try)
        n_cur = n_try
        for beta in [80, 200, 500, 1000]:
            if tl() < 50:
                break
            for mu in [1.0, 10.0, 100.0]:
                if tl() < 20:
                    break
                h_new = lbfgs_opt(h_interp, beta, mu, maxiter=250)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new.copy(), c
                    h_interp = h_best.copy()
        n = len(h_best)

    # Phase 4: Nesterov at higher resolution
    if tl() > 150:
        h_sg2, c_sg2 = nesterov_opt(h_best, 400, 5000, 0.03, 0.9998)
        if c_sg2 < best_c5:
            h_best, best_c5 = h_sg2.copy(), c_sg2
        if tl() > 40:
            h_new = lbfgs_opt(h_best, 800, 50.0, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new.copy(), c

    # Phase 5: Random restarts for diversification
    for trial in range(8):
        if tl() < 35:
            break
        scale = 0.008 * (1.0 + 0.4 * trial)
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new = lbfgs_opt(h_pert, 500, 20.0, maxiter=300)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new.copy(), c

    # Phase 6: Very high beta polish
    for beta in [1500, 3000, 5000]:
        if tl() < 25:
            break
        h_new = lbfgs_opt(h_best, beta, 50.0, maxiter=400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best = h_new.copy()
            best_c5 = c

    # Phase 7: Final Nesterov + polish
    if tl() > 60:
        h_sg3, c_sg3 = nesterov_opt(h_best, 1000, 4000, 0.02, 0.9999)
        if c_sg3 < best_c5:
            h_best, best_c5 = h_sg3.copy(), c_sg3
        if tl() > 25:
            h_new = lbfgs_opt(h_best, 2000, 100.0, maxiter=500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new.copy(), c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
