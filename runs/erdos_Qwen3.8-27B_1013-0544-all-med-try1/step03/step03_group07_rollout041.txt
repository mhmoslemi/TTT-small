    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

    if abs(diff) > 1e-12:

    """Gradient of corr at lag s (s can be negative)"""
    g = np.zeros(n)
    if s >= 0:
        # corr = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])

        g[s:] = (1.0 - h[:n-s])

        g[:n-t] = (1.0 - h[t:])
        g[t:] -= h[:n-t]
    return g

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        if abs(diff2) > 1e-12:

                h = np.clip(h, 0.0, 1.0)

def grad_single_lag(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[s:] = (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    else:
        t = -s
        g[:n-t] = (1.0 - h[t:])
        g[t:] -= h[:n-t]

    dx = 2.0 / n
    corr = compute_corr(h)

The problem is equivalent to: find a set (or weighted set) of points in [0,2] of measure 1, such that for every shift k, the measure of the intersection of the set with a shifted copy of its complement is minimized.

In the binary case, this is: find a subset S of [0,2] with measure 1, minimizing max_k |S ∩ (S+k)^c| (where (S+k)^c is the complement of S shifted by k).

|S ∩ (S+k)^c| = |S| - |S ∩ (S+k)| = 1 - |S ∩ (S+k)|

So we want to maximize min_k |S ∩ (S+k)|, i.e., maximize the minimum autocorrelation.

This is related to the "cross-correlation" or "autocorrelation" of the indicator function of S.

In the continuous case, if h is the indicator of a set S with measure 1 in [0,2], then:
∫ h(x)h(x+k) dx = |S ∩ (S-k)| (the autocorrelation at lag k)

And ∫ h(x)(1-h(x+k)) dx = 1 - |S ∩ (S-k)| (for the appropriate range of k)

So minimizing C₅ is equivalent to maximizing the minimum autocorrelation (over the range of k where the integral is non-trivial).

This is related to the theory of "difference sets" and "bent functions" in coding theory!

A "perfect" autocorrelation would have |S ∩ (S-k)| = 1/2 for all k ≠ 0 (since |S| = 1 and the domain has measure 2, the "random" expectation is 1/2).

If we could achieve |S ∩ (S-k)| = 1/2 for all k, then C₅ = 1 - 1/2 = 1/2. But we know C₅ can be below 1/2 (the uniform h gives C₅ = 1/2).

    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.any():
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[s:] = 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] = 1.0 - h[t:]
        g[t:] -= h[:n - t]
    return g

def smax_active(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    tc = corr[idx]
    m = tc.max()
    sh = tc - m
    ex = np.exp(beta * sh)
    w = ex / ex.sum()
    val = (m + np.log(ex.sum()) / beta) * dx
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-12:
            g += w[i] * grad_lag(h, idx[i] - (n - 1), n)
    g *= dx
    return val, g

def _obj(x):
    return smax_active(x, _g_beta, _g_topk)[0]

def _objg(x):
    return smax_active(x, _g_beta, _g_topk)[1]

def _con(x):
    return x.sum() - _g_tgt

def _conj(x):
    return np.ones(len(x))

    global _g_beta, _g_topk, _g_tgt
    _g_beta, _g_topk = beta, topk
    _g_tgt = len(h) / 2.0
    bnd = Bounds(np.zeros(len(h)), np.ones(len(h)))
    cs = [{'type': 'eq', 'fun': _con, 'jac': _conj}]
    try:
        r = minimize(_obj, h.copy(), jac=_objg, method='SLSQP',
                    bounds=bnd, constraints=cs,
                    options={'maxiter': maxit, 'ftol': 1e-14})
        return project_feasible(r.x, _g_tgt)
    except:
        return project_feasible(h.copy(), _g_tgt)

from scipy.optimize import minimize, Bounds
import time

_g_beta = 100.0
_g_topk = 10
_g_tgt = 200.0
_g_t0 = 0.0
_g_budget = 1100.0

def _tl():
    return _g_budget - (time.time() - _g_t0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.any():
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[s:] = 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] = 1.0 - h[t:]
        g[t:] -= h[:n - t]
    return g

def smax_active(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    tc = corr[idx]
    m = tc.max()
    sh = tc - m
    ex = np.exp(beta * sh)
    w = ex / ex.sum()
    val = (m + np.log(ex.sum()) / beta) * dx
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-12:
            g += w[i] * grad_lag(h, idx[i] - (n - 1), n)
    g *= dx
    return val, g

def _objg(x):
    return smax_active(x, _g_beta, _g_topk)[1]

def _con(x):
    return x.sum() - _g_tgt

def _conj(x):

def _slsqp(h, beta, topk, maxit=400):

    _g_beta, _g_topk = beta, topk
    _g_tgt = len(h) / 2.0
    bnd = Bounds(np.zeros(len(h)), np.ones(len(h)))
    cs = [{'type': 'eq', 'fun': _con, 'jac': _conj}]
    try:
        r = minimize(_obj, h.copy(), jac=_objg, method='SLSQP',
                    bounds=bnd, constraints=cs,
                    options={'maxiter': maxit, 'ftol': 1e-14})
        return project_feasible(r.x, _g_tgt)
    except:
        return project_feasible(h.copy(), _g_tgt)

def _subgrad_nest(h, maxit, lr0):
    n = len(h)
    dx = 2.0 / n
    tgt = n / 2.0
    x = h.copy()
    y = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    lr = lr0

            break
