def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def compute_grad_smoothmax(h, beta, dx, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr * dx
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_x = np.exp(x_shift)
    w = exp_x / np.sum(exp_x)
    
    top_idx = np.argsort(x)[::-1][:top_k]
    w_use = np.zeros_like(x)
    w_use[top_idx] = w[top_idx]
    w_use /= w_use.sum()
    
    grad = np.zeros(n)
    for idx in top_idx:
        wi = w_use[idx]
        if wi < 1e-12:
            continue
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi
            grad[:n - lag] -= wi * h[lag:]
            grad[lag:] -= wi * h[:n - lag]
        else:
            m_lag = -lag
            grad[m_lag:] += wi
            grad[m_lag:] -= wi * h[:n - m_lag]
            grad[:n - m_lag] -= wi * h[m_lag:]
    
    return grad * dx, smooth_max_val(x, beta)


def gradient_descent(h_init, target_sum, max_iters, beta_start, beta_end,
                     top_k, time_func, time_limit, bb_min=0.0005, bb_max=0.08):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    prev_grad = None
    prev_h = None
    bb_step = 0.01
    no_improve = 0
    
    for it in range(max_iters):
        if time_func() < time_limit:
            break
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iters * 0.3))
        grad, obj_val = compute_grad_smoothmax(h, beta, dx, top_k)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad_n = grad / gnorm
        
        if prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            if yy > 1e-16 and sy > 1e-14:
                bb_step = np.clip(np.dot(s, s) / sy, bb_min, bb_max)
        prev_grad = grad.copy()
        prev_h = h.copy()
        
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            h_trial = project_h(h - bb_step * factor * grad_n, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                h = h_trial
                best_h = h_trial.copy()
                best_c5 = c5_trial
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 500 and it % 100 == 99:
                scale = 0.002 + 0.003 * (no_improve % 1000) / 1000.0
                perturb = np.random.randn(n) * scale
                h_trial = project_h(h + perturb, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h = h_trial
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                    no_improve = 0
            if no_improve > 3000:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Gradient descent at initial resolution
    n1 = len(best_h)
    h1, c5_1 = gradient_descent(best_h, n1 / 2.0, 40000, 30, 800,
                                min(20, n1), time_left, 30, bb_min=0.0005, bb_max=0.06)
    if c5_1 < best_c5:
        best_h = h1
        best_c5 = c5_1
    
    # Phase 2: Try a different initial pattern (block structure)
    if time_left() > 150:
        n_blk = 200
        block_sizes = np.array([1, 2, 1, 3, 1, 2, 2, 1, 3, 1])
        pattern = []
        val = 0.0
        for bs in block_sizes:
            pattern.extend([val] * bs)
            val = 1.0 - val
        pattern = np.array(pattern * (n_blk // len(pattern) + 1)[:n_blk])
        h_blk = project_h(pattern, n_blk / 2.0)
        h_blk2, c5_blk = gradient_descent(h_blk, n_blk / 2.0, 20000, 30, 600,
                                          20, time_left, 30, bb_min=0.0005, bb_max=0.06)
        if c5_blk < best_c5:
            best_h = h_blk2
            best_c5 = c5_blk
    
    # Phase 3: Multi-resolution refinement
    target_resolutions = [300, 400, 500, 600, 800]
    for n_r in target_resolutions:
        if time_left() < 60:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_r)
        h_r = np.interp(x_new, x_old, best_h)
        h_r = project_h(h_r, n_r / 2.0)
        
        top_k_r = min(30, n_r)
        max_it_r = min(30000, int(time_left() * 40))
        h_r, c5_r = gradient_descent(h_r, n_r / 2.0, max_it_r, 40, 1200,
                                     top_k_r, time_left, 15, bb_min=0.0003, bb_max=0.05)
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r
    
    # Phase 4: SLSQP polish
    if time_left() > 30:
        n_p = len(best_h)
        target_p = n_p / 2.0
        dx_p = 2.0 / n_p
        bounds_p = Bounds(np.zeros(n_p), np.ones(n_p))
        
        def obj_sl(h_in):
            corr = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max_val(corr, 2000) * dx_p
        
        def con_eq(h_in):
            return np.sum(h_in) - target_p
        
        def jac_eq(h_in):
            return np.ones(n_p)
        
        try:
            res = minimize(obj_sl, best_h, method='SLSQP', bounds=bounds_p,
                           constraints=[{'type': 'eq', 'fun': con_eq, 'jac': jac_eq}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_p)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass
    
    # Phase 5: Final fine gradient sweep
    if time_left() > 10:
        n_f = len(best_h)
        h_final, c5_final = gradient_descent(best_h, n_f / 2.0, 
                                             min(10000, int(time_left() * 20)),
                                             500, 3000, min(25, n_f),
                                             time_left, 3, bb_min=0.0003, bb_max=0.03)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    # Final projection to ensure feasibility
    n_out = len(best_h)
    best_h = project_h(best_h, n_out / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_out
