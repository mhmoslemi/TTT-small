import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m == 0:
        return 0.0
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def smooth_obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    grad = np.zeros(len(h))
    n = len(h)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += (1.0 - h[lag:lag+m]) * w[k_idx]
            grad[lag:lag+m] -= h[:m] * w[k_idx]
        else:
            la = -lag
            m = n - la
            grad[la:la+m] += (1.0 - h[:m]) * w[k_idx]
            grad[:m] -= h[la:la+m] * w[k_idx]
    return grad * dx

            try:
                res = minimize(smooth_obj, x0, args=(dx, beta), method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                c5 = compute_c5(h_new)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_compute(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_compute(corr, beta) * dx

def obj_grad_fn(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += (1.0 - h[lag:lag+m]) * wk
            grad[lag:lag+m] -= h[:m] * wk
        else:
            la = -lag
            m = n - la
            grad[la:la+m] += (1.0 - h[:m]) * wk
            grad[:m] -= h[la:la+m] * wk
    return grad * dx

    return x.sum() - target

    return np.ones(len(x))

    else:

        m = n - la
        grad[la:la+m] += (1.0 - h[:m])
        grad[:m] -= h[la:la+m]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_compute(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_compute(corr, beta) * dx

def obj_grad_fn(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += (1.0 - x[lag:lag+m]) * wk
            grad[lag:lag+m] -= x[:m] * wk
        else:
            la = -lag
            m = n - la
            grad[la:la+m] += (1.0 - x[:m]) * wk
            grad[:m] -= x[la:la+m] * wk
    return grad * dx

def con_eq(x, target):
    return x.sum() - target

def con_eq_jac(x, target):
    return np.ones(len(x))

def subgrad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[:m] += (1.0 - h[lag:lag+m])
        grad[lag:lag+m] -= h[:m]
    else:
        la = -lag
        m = n - la
        grad[la:la+m] += (1.0 - h[:m])
        grad[:m] -= h[la:la+m]
    return grad
