```python
from functools import partial

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def momentum_descent(h, n, target, dx, max_iters, time_fn, min_time, best_state, momentum=0.7):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    v = np.zeros(n)
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, 2 * n // 80 + 2)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 200.0
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        v = momentum * v - 0.05 * grad
        step = 1.0 / (1.0 + 0.005 * it)
        improved = False

                if cur_c5 < best_state[0]:

                    best_state[1] = cur.copy()
                    best_state[2] = n
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def momentum_descent(h, n, target, dx, max_iters, time_fn, min_time, best_state, mu=0.8):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    v = np.zeros(n)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(5, 2 * n // 60 + 2)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 180.0
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        v = mu * v - 0.04 * grad
        step = 1.0 / (1.0 + 0.004 * it)
        improved = False
        for _ in range(25):
            h_trial = project_h(cur + step * v, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.4
        if not improved:
            T = 0.004 * np.exp(-it / 600.0)
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T * 4:
                cur = h_trial
                cur_c5 = c5_trial
                v = np.zeros(n)
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
    return cur, cur_c5


def slsqp_smooth_max(h, n, target, dx, betas, time_fn, min_time):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    for beta in betas:
        if time_fn() < min_time:
            break

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(c, beta) * dx

        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            m = np.max(c)
            shifted = np.clip(beta * (c - m), -500, 500)
            w = np.exp(shifted)
            w /= w.sum()
            g = np.zeros(n)
            for idx in range(len(c)):
                if w[idx] < 1e-10:
                    continue
                lag = idx - center
                g += w[idx] * grad_for_lag(x, lag, n, dx)
            return g

        def con_fun(x):
            return np.sum(x) - target

        def con_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, cur, jac=obj_grad, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return cur, cur_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    # Phase 1: Refine at parent resolution
    n = n_parent
    target = n / 2.0
    dx = 2.0 / n
    h_cur = project_h(h_parent, target)

    h_cur, _ = momentum_descent(h_cur, n, target, dx, 6000, time_left, 80, best_state, mu=0.8)
    h_cur, _ = slsqp_smooth_max(h_cur, n, target, dx, [60, 120, 250, 500], time_left, 25)
    h_cur, _ = momentum_descent(h_cur, n, target, dx, 6000, time_left, 40, best_state, mu=0.85)

    # Phase 2: Try perturbed restarts at parent resolution
    for restart in range(3):
        if time_left() < 60:
            break
        h_pert = project_h(best_state[1] + 0.02 * np.random.normal(0, 1, n), target)
        h_pert, _ = momentum_descent(h_pert, n, target, dx, 3000, time_left, 20, best_state, mu=0.8)
        h_pert, _ = slsqp_smooth_max(h_pert, n, target, dx, [200, 500], time_left, 8)

    # Phase 3: Explore other resolutions
    for n_try in [200, 300, 350]:
        if time_left() < 70:
            break
        target_t = n_try / 2.0
        dx_t = 2.0 / n_try
        h_init = project_h(resample_h(best_state[1], n_try), target_t)
        h_t, _ = momentum_descent(h_init, n_try, target_t, dx_t, 4000, time_left, 30, best_state, mu=0.8)
        h_t, _ = slsqp_smooth_max(h_t, n_try, target_t, dx_t, [80, 200, 400], time_left, 15)
        h_t, _ = momentum_descent(h_t, n_try, target_t, dx_t, 3000, time_left, 5, best_state, mu=0.85)

    # Phase 4: Final polish at best resolution
    if time_left() > 10:
        bh = best_state[1]
        bn = best_state[2]
        bt = bn / 2.0
        bdx = 2.0 / bn
        bh, _ = slsqp_smooth_max(bh, bn, bt, bdx, [400, 800, 1500], time_left, 5)
        if compute_c5(bh) < best_state[0]:
            best_state[0] = compute_c5(bh)
            best_state[1] = bh
        best_state[1], _ = momentum_descent(best_state[1], bn, bt, bdx, 15000, time_left, 2, best_state, mu=0.9)

    final_h = project_h(best_state[1], len(best_state[1]) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_state[0]:
        best_state[0] = final_c5
        best_state[1] = final_h
    return best_state[1], best_state[0], len(best_state[1])
```
