        zeros = np.where(h == 0)[0]

            candidates = [(io, iz) for io in ones for iz in zeros]
        else:
            rng = np.random.default_rng()
            candidates = [(ones[rng.integers(len(ones))], zeros[rng.integers(len(zeros))]) for _ in range(10000)]

            h[io] = 0.0
            h[iz] = 1.0
            c5 = compute_c5(h)
            h[io] = 1.0
            h[iz] = 0.0
            d = c5 - cur_c5
            if d < best_delta:
                best_delta = d
                best_swap = (io, iz)
        
        if best_swap is None or best_delta > -1e-15:
            break
        
        io, iz = best_swap
        h[io] = 0.0
        h[iz] = 1.0
        cur_c5 += best_delta
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta
