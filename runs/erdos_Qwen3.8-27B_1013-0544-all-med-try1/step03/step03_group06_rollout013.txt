    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / len(corr) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(shifted)
    return e / np.sum(e)

def corr_grad(h, weights):
    """Compute gradient of corr (all lags) weighted by weights."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(-n+1, n):
        w = weights[k + n - 1]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            idx1 = np.arange(k, n)
            grad[idx1] += w * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= w * h[idx2 + k]
        else:
            k2 = -k
            idx1 = np.arange(0, n - k2)
            grad[idx1] += w * (1.0 - h[idx1 + k2])
            idx2 = np.arange(k2, n)
            grad[idx2] -= w * h[idx2 - k2]
    return grad

def obj_func(h, beta, lam, rho, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_val(corr, beta) * dx
    constraint_viol = h.sum() - target
    return sm + lam * constraint_viol + 0.5 * rho * constraint_viol**2

def obj_grad(h, beta, lam, rho, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, beta)
    grad = corr_grad(h, w) * dx
    cv = h.sum() - target
    grad += lam + rho * cv
    return grad

            lam += rho * cv
            if abs(cv) > 1e-7:
                rho *= 2.0
                lam = float(np.clip(lam, -500, 500))

        if c5_interp < best_c5:
            best_h = h_interp
            best_c5 = c5_interp
        n = len(best_h)
        target = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 3: Subgradient with smooth max weights
    no_improve = 0
    for iteration in range(30000):
        if budget_s - (time.time() - start_time) < 5:
            break
        corr = np.correlate(best_h, 1.0 - h, mode="full") if False else np.correlate(best_h, 1.0 - best_h, mode="full")
        for beta_sg in [100, 300, 1000]:
            w = smooth_max_weights(corr, beta_sg)
            grad = corr_grad(best_h, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            found = False
            for step in [0.06, 0.03, 0.015, 0.007, 0.003, 0.001]:
                h_trial = project_h(best_h - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_h = h_trial
                    best_c5 = c5_trial
                    found = True
                    break
            if found:
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 300:
                break
            if no_improve % 80 == 0:
                scale = max(0.001, 0.01 * (1.0 - no_improve / 500.0))
                perturb = np.random.randn(n) * scale
                h_trial = project_h(best_h + perturb, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    no_improve = 0

    # Phase 4: Targeted pairwise coordinate moves on active lags
    for iteration in range(50000):
        if budget_s - (time.time() - start_time) < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        improved = False
        for idx in top_idx:
            if budget_s - (time.time() - start_time) < 1:
                break
            lag = int(idx) - (n - 1)
            abs_lag = abs(lag)
            if abs_lag == 0 or abs_lag >= n:
                continue
            n_valid = n - abs_lag
            for _ in range(8):
                i = np.random.randint(n_valid)
                j = i + abs_lag
                for d in [0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004]:
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-15:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved and iteration % 300 == 299:
            perturb = np.random.randn(n) * 0.004
            h_trial = project_h(best_h + perturb, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial

    # Phase 5: Random restarts + local polish
    n_best = len(best_h)
    target_best = n_best / 2.0
    n_restarts = 8
    if budget_s - (time.time() - start_time) > 60:
        for r in range(n_restarts):
            if budget_s - (time.time() - start_time) < 15:
                break
            h_start = best_h + np.random.randn(n_best) * 0.02
            h_start = project_h(h_start, target_best)
            c5_start = compute_c5(h_start)
            if c5_start < best_c5:
                best_h = h_start
                best_c5 = c5_start
            bounds_r = Bounds(np.zeros(n_best), np.ones(n_best))
            lam_r = 0.0
            rho_r = 20.0
            for beta_r in [200, 600, 1500, 4000]:
                if budget_s - (time.time() - start_time) < 5:
                    break
                try:
                    res = minimize(obj_func, h_start, args=(beta_r, lam_r, rho_r, target_best),
                                   jac=obj_grad, method='L-BFGS-B', bounds=bounds_r,
                                   options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 4000})
                    h_new = project_h(res.x, target_best)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_start:
                        h_start = h_new
                        c5_start = c5_new
                    if c5_new < best_c5:
                        best_h = h_start.copy()
                        best_c5 = c5_new
                except Exception:
                    pass
                cv_r = h_start.sum() - target_best
                lam_r += rho_r * cv_r
                if abs(cv_r) > 1e-7:
                    rho_r *= 2.0
                    lam_r = float(np.clip(lam_r, -500, 500))

    # Phase 6: Final high-beta polish
    if budget_s - (time.time() - start_time) > 10:
        n = len(best_h)
        target = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        lam = 0.0
        rho = 100.0
        for beta in [5000, 10000, 20000]:
            if budget_s - (time.time() - start_time) < 5:
                break
            try:
                res = minimize(obj_func, best_h, args=(beta, lam, rho, target),
                               jac=obj_grad, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 50000})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
            cv = best_h.sum() - target
            lam += rho * cv

    return best_h, best_c5, len(best_h)
