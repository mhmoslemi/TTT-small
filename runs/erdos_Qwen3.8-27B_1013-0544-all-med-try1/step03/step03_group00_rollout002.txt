        v = momentum * v - lr * grad

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    from scipy.optimize import minimize, Bounds
    
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

        return smoothmax_val_grad(x, beta, top_k)[1]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        return project_feasible(h, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = len(corr)
    if top_k is not None and top_k < m:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(m)
    c_sel = corr[top_idx]
    cmax = c_sel.max()
    exps = np.exp(beta * (c_sel - cmax))
    sw = exps.sum()
    w_sel = exps / sw
    val = (cmax + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = top_idx[i]
        wk = w_sel[i]
        if wk < 1e-13:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def pg_solve(h_init, beta, n_iter, lr_init, top_k, t0, budget_s):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    val, _ = smoothmax_val_grad(h, beta, top_k)
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 15:
            break
        if it % 30 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        _, grad = smoothmax_val_grad(h, beta, top_k)
        for _ in range(25):
            h_new = project_feasible(h - lr * grad, target)
            val_new, _ = smoothmax_val_grad(h_new, beta, top_k)
            if val_new < val * (1.0 - 1e-12):
                h = h_new
                val = val_new
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-7)
            break
        lr = min(lr * 1.08, lr_init)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()
    n = len(h_cur)

    # Phase 1: PG continuation at native resolution
    for beta in [15, 40, 100, 250, 500]:
        if remaining() < 30:
            break
        n_iter = 3000 if beta <= 100 else 2000
        lr_init = 0.06 if beta <= 100 else 0.03
        h_new, c_new = pg_solve(h_cur, beta, n_iter, lr_init, top_k=35, t0=t0, budget_s=budget_s)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: SLSQP refinement (fewer calls, save time for PG)
    if remaining() > 150:
        for beta in [300, 600]:
            if remaining() < 60:
                break
            h_sls = slsqp_solve(h_cur, beta, 350, top_k=25)
            c_sls = compute_c5(h_sls)
            if c_sls < best_c5:
                best_c5 = c_sls
                h_best = h_sls.copy()
            h_cur = h_sls

    # Phase 3: Basin hopping with PG solver
    rng = np.random.RandomState(seed + 7)
    target = n / 2.0
    noise = 0.04
    for hop in range(25):
        if remaining() < 50:
            break
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_hop, c_hop = pg_solve(h_pert, 200, 1000, 0.05, top_k=25, t0=t0, budget_s=budget_s)
        if c_hop < best_c5:
            best_c5 = c_hop
            h_best = h_hop.copy()
        noise *= 0.88

    # Phase 4: Multi-resolution exploration
    for n_try in [120, 180, 250, 350]:
        if remaining() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        c_interp = compute_c5(h_interp)
        if c_interp < best_c5:
            best_c5 = c_interp
            h_best = h_interp.copy()
        for beta in [30, 100, 300]:
            if remaining() < 40:
                break
            h_t, c_t = pg_solve(h_interp, beta, 2000, 0.05, top_k=30, t0=t0, budget_s=budget_s)
            if c_t < best_c5:
                best_c5 = c_t
                h_best = h_t.copy()
            h_interp = h_t

    # Phase 5: Cold starts - random solutions optimized
    for cs in range(5):
        if remaining() < 80:
            break
        h_rand = project_feasible(np.random.rand(n) * 1.2, n / 2.0)
        h_cs, c_cs = pg_solve(h_rand, 100, 2000, 0.06, top_k=30, t0=t0, budget_s=budget_s)
        if c_cs < best_c5:
            best_c5 = c_cs
            h_best = h_cs.copy()

    # Phase 6: Final PG polish at best resolution
    if remaining() > 40:
        n_best = len(h_best)
        h_fin, c_fin = pg_solve(h_best, 800, 4000, 0.02, top_k=50, t0=t0, budget_s=budget_s)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 7: SLSQP final squeeze
    if remaining() > 35:
        h_fin2 = slsqp_solve(h_best, 1000, 400, top_k=None)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    # Phase 8: Very fine basin hop
    if remaining() > 50:
        nb = len(h_best)
        tb = nb / 2.0
        noise2 = 0.01
        for hop in range(8):
            if remaining() < 40:
                break
            h_p = h_best + rng.randn(nb) * noise2
            h_p = project_feasible(h_p, tb)
            h_h2, c_h2 = pg_solve(h_p, 500, 1500, 0.02, top_k=40, t0=t0, budget_s=budget_s)
            if c_h2 < best_c5:
                best_c5 = c_h2
                h_best = h_h2.copy()
            noise2 *= 0.8

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
