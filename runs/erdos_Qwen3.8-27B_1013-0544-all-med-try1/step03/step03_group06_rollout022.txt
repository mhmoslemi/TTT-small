def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(c, beta):
    m = np.max(c)
    x = np.clip(beta * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(x))) / beta

def smooth_max_weights(c, beta):
    m = np.max(c)
    x = np.clip(beta * (c - m), -500, 500)
    w = np.exp(x)
    return w / np.sum(w)

def corr_gradient(h, d):
    """Compute ∂c_k/∂h where d = k - (n-1) is the lag offset."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        # c = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # ∂c/∂h[p] = (1-h[p-d]) for d<=p<=n-1, minus h[p+d] for 0<=p<=n-1-d
        if d < n:
            grad[d:] += (1.0 - h[:n-d])
            grad[:n-d] -= h[d:]
    else:
        j = -d
        if j < n:
            # c = sum_{i=j}^{n-1} h[i-j]*(1-h[i])
            # ∂c/∂h[p] = (1-h[p+j]) for 0<=p<=n-1-j, minus h[p-j] for j<=p<=n-1
            grad[:n-j] += (1.0 - h[j:])
            grad[j:] -= h[:n-j]
    return grad

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:

        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        w = smooth_max_weights(corr, beta)
        
        # Compute gradient: sum_k w_k * ∂c_k/∂h * dx
        grad = np.zeros(n)
        # Only compute for lags with significant weight
        top_idx = np.argsort(w)[::-1][:15]

                break
            d = int(idx) - (n - 1)
            grad += w[idx] * corr_gradient(best_h, d)
        grad *= dx
        
        # Projected gradient step with line search
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best_h - step * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_val - 1e-14:
                best_h = h_trial
                best_val = c5_trial
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(c, beta):
    m = np.max(c)
    x = np.clip(beta * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(x))) / beta


def smooth_max_weights(c, beta):
    m = np.max(c)
    x = np.clip(beta * (c - m), -500, 500)
    w = np.exp(x)
    return w / np.sum(w)


def corr_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
    else:
        j = -d
        if j < n:
            grad[:n - j] += (1.0 - h[j:])

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")

    grad = np.zeros(n)
    for idx in top_idx:
        if w[idx] < 1e-12:
            break
        d = int(idx) - (n - 1)
        grad += w[idx] * corr_gradient(h, d)

    return val, grad

                best_h = h_trial.copy()
                best_val = c5_trial

            break
    return best_h, best_val
