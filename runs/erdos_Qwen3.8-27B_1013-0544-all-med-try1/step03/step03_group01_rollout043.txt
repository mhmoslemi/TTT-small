   def compute_c5(h, n):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return float(np.max(c) * dx)

   def project_feasible(h, n):
       target_sum = n / 2.0
       h = np.clip(h.copy(), 0.0, 1.0)
       for _ in range(200):
           diff = target_sum - h.sum()
           if abs(diff) < 1e-10:
               break
           if diff > 0:
               room = 1.0 - h
               total = room.sum()
               if total < 1e-14:
                   break
               h += (diff / total) * room
           else:
               total = h.sum()
               if total < 1e-14:
                   break
               h += (diff / total) * h
       return np.clip(h, 0.0, 1.0)

   def smooth_max(c, dx, alpha):
       """Compute smooth max of c*dx using log-sum-exp."""
       vals = c * dx
       m = vals.max()
       if alpha * m > 500:
           # shift for numerical stability
           shifted = alpha * (vals - vals.max())
           return float(vals.max() + (1.0/alpha) * np.log(np.sum(np.exp(shifted))))
       else:
           return float((1.0/alpha) * np.log(np.sum(np.exp(alpha * vals))))

   def obj_smooth(h, n, alpha):
       dx = 2.0 / n
       c = np.correlate(h, 1 - h, mode='full')
       return smooth_max(c, dx, alpha)
