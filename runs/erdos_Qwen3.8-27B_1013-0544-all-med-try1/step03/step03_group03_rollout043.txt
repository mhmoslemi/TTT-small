    """Find optimal h[i] given other coordinates fixed. Returns optimal value."""
    # f_0 = sum h[j]*(1-h[j]) = h[i]*(1-h[i]) + c0 where c0 = sum_{j≠i} h[j]*(1-h[j])
    c0 = np.sum(h * (1 - h)) - h[i] * (1 - h[i])
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_len = 2 * n - 1
    
    # For each lag index m (corresponding to ell = m - (n-1)):
    # f_m = a_m * h[i] + b_m (linear in h[i] for ell ≠ 0, quadratic for ell = 0)
    # a_m = (1-h[i-ell]) [if valid] + (-h[i+ell]) [if valid]
    
    # Compute a for all m
    # ell ranges from -(n-1) to (n-1)
    ells = np.arange(-(n-1), n)
    # For element i:
    # j1 = i - ell (valid if 0 <= i-ell < n)
    # j2 = i (valid if 0 <= i+ell < n, i.e., -i <= ell < n-i)
    
    valid_j1 = (i - ells >= 0) & (i - ells < n)
    valid_j2 = (i + ells >= 0) & (i + ells < n)
    
    a = np.zeros(m_len)
    a[valid_j1] += (1.0 - h[i - ells[valid_j1]])
    a[valid_j2] += (-h[i + ells[valid_j2]])
    
    b = corr - a * h[i]
    
    # For m = n-1 (ell=0): f = h[i] - h[i]^2 + c0 (quadratic)
    # For others: f = a[m]*t + b[m] (linear)
    
    # Find minimum of max(g(t), max_{m≠n-1} (a[m]*t + b[m])) on [0,1]
    # where g(t) = t - t^2 + c0
    
    # Candidates: 0, 1, 0.5, and intersections of g with each linear
    candidates = [0.0, 1.0, 0.5]
    
    # For m ≠ n-1: solve t - t^2 + c0 = a[m]*t + b[m]
    # → t^2 + (a[m]-1)*t + (b[m]-c0) = 0
    for m_idx in range(m_len):
        if m_idx == n - 1:
            continue
        A_coef = 1.0
        B_coef = a[m_idx] - 1.0
        C_coef = b[m_idx] - c0
        disc = B_coef * B_coef - 4.0 * A_coef * C_coef
        if disc >= 0:
            sd = np.sqrt(disc)
            t1 = (-B_coef + sd) / 2.0
            t2 = (-B_coef - sd) / 2.0
            if 0.0 <= t1 <= 1.0:
                candidates.append(t1)
            if 0.0 <= t2 <= 1.0:
                candidates.append(t2)
    
    # Evaluate
    best_val = np.inf
    best_t = h[i]  # fallback
    for t in candidates:
        t = float(np.clip(t, 0.0, 1.0))
        g_val = t - t * t + c0
        lin_vals = a * t + b
        max_val = max(g_val, float(np.max(lin_vals)))
        if max_val < best_val:
            best_val = max_val
            best_t = t
    
    return best_t

def coord_descent(h, n, max_passes=100, tol=1e-10):
    """Cyclic coordinate descent."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        improved = False
        for i in range(n):
            t_new = coord_optimal(h, i, n)
            if abs(t_new - h[i]) > 1e-12:
                h[i] = t_new
                improved = True
        if not improved:
            break
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx + lag] += (1.0 - h[idx])
            grad[idx] -= h[idx + lag]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]

        lag = k - (n - 1)
        grad += w[j] * subgrad_for_lag(h, lag, n)
    return grad
