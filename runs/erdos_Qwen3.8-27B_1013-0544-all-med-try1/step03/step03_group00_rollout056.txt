from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = m + np.log(sw) / beta
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        j_lo1 = max(0, -s)
        j_hi1 = min(n, n - s)

    target = n / 2.0

        return val + mu * (x.sum() - target) ** 2
    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        return g + 2.0 * mu * (x.sum() - target)
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def swap_perturb(h, rng, n_swaps):
    """Perturb by swapping values between pairs of positions."""
    n = len(h)
    h_new = h.copy()
    for _ in range(n_swaps):
        i, j = rng.choice(n, size=2, replace=False)
        delta = h[i] - h[j]
        if abs(delta) > 1e-6:
            h_new[i] = h[j]
            h_new[j] = h[i]
    return h_new

def single_objective_grad(h, k_idx):
    """Gradient of the correlation at a specific lag index."""
    n = len(h)
    dx = 2.0 / n
    s = k_idx - n + 1
    grad = np.zeros(n)

    if j_hi2 > j_lo2:
        grad[j_lo2:j_hi2] += (-h[j_lo2 + s:j_hi2 + s])
    return grad * dx

    n = len(h_init_binary)
    h = h_init_binary.astype(float).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = t_start
    for it in range(n_iters):
        if it % 500 == 0:
            T *= t0_power
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    n = len(h_cur)
    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: L-BFGS-B continuation from parent
    for beta, tk in [(10, 40), (30, 30), (80, 25), (150, 20), (300, 20), (600, 15), (1200, 15)]:
        if remaining() < 30:
            break
        h_new = lbfgsb_solve(h_cur, beta, maxiter=500, mu=500.0, top_k=tk)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            h_best = h_new.copy()
        h_cur = h_new

    # Phase 2: Basin hopping with adaptive noise
    rng = np.random.RandomState(seed + 7)
    noise = 0.035
    stall = 0
    for hop in range(100):
        if remaining() < 40:
            break
        if stall > 25:
            break
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, n / 2.0)
        h_local = h_pert
        for beta, tk in [(60, 15), (200, 15), (500, 10)]:
            if remaining() < 15:
                break
            h_local = lbfgsb_solve(h_local, beta, maxiter=300, mu=500.0, top_k=tk)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
            noise = max(0.004, noise * 0.92)
            stall = 0
        else:
            stall += 1
            if stall % 4 == 0:
                noise = min(0.05, noise * 1.15)

    # Phase 3: Binary structured local search
    if remaining() > 90:
        rng_b = np.random.RandomState(seed + 50)
        h_bin = (h_best > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n / 2.0)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5 + 0.01:
            h_bin_sa, c_bin_sa = binary_sa(h_bin, 50000, 0.9995, seed + 60, 0.005)
            if c_bin_sa < best_c5:
                best_c5 = c_bin_sa
                h_best = h_bin_sa.copy()
            h_relaxed = lbfgsb_solve(h_bin_sa, 200, maxiter=400, mu=500.0, top_k=20)
            h_relaxed = lbfgsb_solve(h_relaxed, 600, maxiter=400, mu=1000.0, top_k=15)
            c_rel = compute_c5(h_relaxed)
            if c_rel < best_c5:
                best_c5 = c_rel
                h_best = h_relaxed.copy()

    # Phase 4: Multi-resolution exploration
    for n_try in [300, 400, 500]:
        if remaining() < 100:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_c = h_interp
        for beta, tk in [(40, 25), (120, 20), (300, 20), (800, 15)]:
            if remaining() < 30:
                break
            h_c = lbfgsb_solve(h_c, beta, maxiter=350, mu=500.0, top_k=tk)
        c_try = compute_c5(h_c)
        if c_try < best_c5:
            best_c5 = c_try
            h_best = h_c.copy()
            n = n_try

    # Phase 5: Basin hopping at best resolution (finer)
    rng2 = np.random.RandomState(seed + 200)
    noise2 = 0.018
    n_cur = len(h_best)
    stall2 = 0
    for hop in range(60):
        if remaining() < 30:
            break
        if stall2 > 18:
            break
        h_pert = h_best + rng2.randn(n_cur) * noise2
        h_pert = project_feasible(h_pert, n_cur / 2.0)
        h_local = h_pert
        for beta in [150, 400, 1000]:
            if remaining() < 15:
                break
            h_local = lbfgsb_solve(h_local, beta, maxiter=250, mu=1000.0, top_k=12)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
            noise2 = max(0.003, noise2 * 0.9)
            stall2 = 0
        else:
            stall2 += 1
            if stall2 % 5 == 0:
                noise2 = min(0.03, noise2 * 1.2)

    # Phase 6: Diverse restarts
    rng3 = np.random.RandomState(seed + 400)
    for restart in range(5):
        if remaining() < 60:
            break
        h_start = h_best + rng3.randn(n_cur) * 0.06
        h_start = project_feasible(h_start, n_cur / 2.0)
        h_local = h_start
        for beta, tk in [(50, 20), (200, 15), (500, 10), (1000, 10)]:
            if remaining() < 20:
                break
            h_local = lbfgsb_solve(h_local, beta, maxiter=300, mu=500.0, top_k=tk)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 7: Very fine basin hopping
    if remaining() > 50:
        rng4 = np.random.RandomState(seed + 600)
        for hop in range(30):
            if remaining() < 20:
                break
            h_pert = h_best + rng4.randn(n_cur) * 0.008
            h_pert = project_feasible(h_pert, n_cur / 2.0)
            h_local = lbfgsb_solve(h_pert, 800, maxiter=350, mu=1000.0, top_k=10)
            h_local = lbfgsb_solve(h_local, 2000, maxiter=250, mu=2000.0, top_k=12)
            c_local = compute_c5(h_local)
            if c_local < best_c5:
                best_c5 = c_local
                h_best = h_local.copy()

    # Phase 8: Final high-beta polish
    if remaining() > 30:
        h_fin = lbfgsb_solve(h_best, 3000, maxiter=600, mu=2000.0, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 9: Ultra-fine final
    if remaining() > 20:
        h_fin2 = lbfgsb_solve(h_best, 5000, maxiter=400, mu=5000.0, top_k=8)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    return h_best, best_c5, len(h_best)
