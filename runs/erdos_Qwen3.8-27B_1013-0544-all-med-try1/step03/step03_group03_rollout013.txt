import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_fn(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx

def sum_con_fun(x, target_sum):
    return x.sum() - target_sum

def sum_con_jac(x, target_sum):
    return np.ones(len(x))

def subgrad_lag(h, lag, n):
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return np.zeros(n)
        idx = np.arange(m)
        grad = np.zeros(n)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m <= 0:
            return np.zeros(n)
        idx = np.arange(m)
        grad = np.zeros(n)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def freq_perturb(n, scale):
    """Generate a frequency-weighted random perturbation"""
    freqs = np.fft.fftfreq(n)
    amp = np.exp(-2.0 * np.abs(freqs) * 2.0 / n)
    return np.random.randn(n) * amp * scale

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient with rotating top-lag focus
    if time.time() - t0 < budget_s - 180:
        h = best_h.copy()
        velocity = np.zeros(n)
        momentum = 0.82
        for it in range(25000):
            if time.time() - t0 > budget_s - 160:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            top_n = min(5, len(corr))
            top_idx = np.argsort(corr)[-top_n:]
            lag = top_idx[it % top_n] - (n - 1)
            grad = subgrad_lag(h, lag, n)
            lr = 0.0025 / (1.0 + 0.0004 * it)
            velocity = momentum * velocity + (1.0 - momentum) * (-lr * grad)
            h_new = project_feasible(h + velocity, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5 + 0.0008:
                h = h_new
                if c_new < best_c5 - 1e-7:
                    best_h = h_new.copy()
                    best_c5 = c_new
            else:
                velocity *= 0.4
                lr *= 0.8

    # Phase 2: SLSQP continuation at base resolution
    if time.time() - t0 < budget_s - 100:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target_sum,)}]
        beta_list = [25, 60, 150, 400, 1000, 2500, 6000, 15000, 40000]
        for beta in beta_list:
            if time.time() - t0 > budget_s - 35:
                break
            n_trials = 5 if beta < 1000 else 4
            for trial in range(n_trials):
                if time.time() - t0 > budget_s - 15:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 = x0 + freq_perturb(n, 0.003 / (trial + 0.5))
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_fn, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 12000, 'ftol': 1e-16, 'maxfun': 20000})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 3: Resolution exploration
    if time.time() - t0 < budget_s - 60:
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        base_n = len(best_h)
        res_list = sorted(set([
            max(50, base_n - 60), max(50, base_n - 40), max(50, base_n - 20),
            base_n + 20, base_n + 40, base_n + 60, base_n + 100, base_n + 150,
            base_n + 200, base_n + 300
        ]))
        for n_try in res_list:
            if time.time() - t0 > budget_s - 25:
                break
            if n_try == base_n:
                continue
            if n_try < 40 or n_try > 800:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (ts_t,)}]
            for beta in [50, 300, 1500, 8000]:
                if time.time() - t0 > budget_s - 15:
                    break
                try:
                    res = minimize(obj_fn, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 8000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        x_old = np.linspace(0, 2, n_try, endpoint=False)
                        base_n = n_try
                except Exception:
                    pass

    # Phase 4: Go back to best resolution, final high-beta polish
    if time.time() - t0 < budget_s - 10:
        fn = len(best_h)
        fdx = 2.0 / fn
        fts = fn / 2.0
        fbnds = Bounds(np.zeros(fn), np.ones(fn))
        fcons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (fts,)}]
        for beta in [20000, 60000, 150000, 400000]:
            if time.time() - t0 > budget_s - 5:
                break
            for trial in range(5):
                if time.time() - t0 > budget_s - 3:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 = x0 + np.random.randn(fn) * (0.0015 / (trial + 0.5))
                    x0 = project_feasible(x0, fts)
                try:
                    res = minimize(obj_fn, x0, args=(fdx, beta), method='SLSQP',
                                   bounds=fbnds, constraints=fcons,
                                   options={'maxiter': 20000, 'ftol': 1e-16, 'maxfun': 30000})
                    h_new = project_feasible(res.x, fts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 5: Tiny subgradient micro-polish on worst lag only
    if time.time() - t0 < budget_s - 3:
        h = best_h.copy()
        fn2 = len(h)
        fts2 = fn2 / 2.0
        for it in range(3000):
            if time.time() - t0 > budget_s - 2:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (fn2 - 1)
            grad = subgrad_lag(h, lag, fn2)
            lr = 0.0008 / (1.0 + 0.005 * it)
            h_new = project_feasible(h - lr * grad, fts2)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
            elif it > 500:
                break

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
