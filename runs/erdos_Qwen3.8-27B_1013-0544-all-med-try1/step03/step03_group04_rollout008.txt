    Returns array of shape (2n-1, n) where grad[k, m] = d(c_k)/d(h[m]).

    # But wait, if m is the "first" index (j=m), then j+l = m+l must be valid
    # If m is the "second" index (j+l=m, so j=m-l), then j=m-l must be valid
    # Note: if l=0, both conditions give the same term, so we get (1-h[m]) - h[m] = 1-2h[m]

    grads = np.zeros((n_full, n))

        for m in range(n):
            g = 0.0
            # m is first index: need 0 <= m+l < n
            if 0 <= m + l < n:
                g += (1.0 - h[m + l])
            # m is second index: m = j+l, so j = m-l, need 0 <= m-l < n
            if 0 <= m - l < n:
                g += -h[m - l]
            grads[ki, m] = g * dx

For l = 0:

def solve_lin_relaxation(h_star, active_lags, n):
    """Solve the LP relaxation at h_star using active lags."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for ki in active_lags:
        l = ki - (n - 1)
        c_k = np.correlate(h_star, 1-h_star, mode="full")[ki] * dx
        g_k = gradient_c5_for_lag(h_star, l, dx)
        # c_k + g_k^T (h - h_star) <= t
        constraints.append(c_k + g_k @ (h_var - h_star) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.GLOP)  # or ECOS, or CBC
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1), t_var.value
    return None, None

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

_g_target = 0.0
_g_dx = 0.0
_g_beta = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(xs)) / len(x))) / beta

def smax_probs(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    e = np.exp(xs)
    return e / e.sum()

def g_lag(h, l, dx):
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        v = n - l
        if v > 0:
            g[:v] += 1.0 - h[l:]
            g[l:] += -h[:v]
    else:
        p = -l
        v = n - p
        if v > 0:
            g[p:] += 1.0 - h[:v]
            g[:v] += -h[p:]
    return g * dx

def slsqp_obj(x):
    c = np.correlate(x, 1.0 - x, mode="full") * _g_dx
    return smax_val(c, _g_beta)

def slsqp_cf(x):
    return np.sum(x) - _g_target

def slsqp_cj(x):
    return np.ones(x.shape)

def subgrad_smooth(h, dx, beta):
    n = len(h)
    corr = get_corr(h)
    probs = smax_probs(corr, beta)
    val = smax_val(corr, beta)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if probs[ki] > 1e-12:
            l = ki - (n - 1)
            grad += probs[ki] * g_lag(h, l, dx)
    return val, grad

def nesterov_phase(h, best_c5, budget_s, t0, target, max_iters=30000):
    n = len(h)
    dx = 2.0 / n
    h_prev = h.copy()
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best = c5_cur
    beta = 150.0
    alpha0 = 0.02
    patience = 0
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 30:
            break
        vt = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * it * it))
        beta_t = (it + 1.0) / vt
        y = h_cur + beta_t * (h_cur - h_prev)
        y = np.clip(y, 0.0, 1.0)
        val, grad = subgrad_smooth(y, dx, beta)
        alpha = alpha0 / np.sqrt(1.0 + 0.5 * it)
        h_new = project_h(y - alpha * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h_prev = h_cur.copy()
            h_cur = h_new
            c5_cur = c5_new
            patience = 0
            if c5_cur < best:
                best = c5_cur
        else:
            h_new2 = project_h(y - alpha * 0.3 * grad, target)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < c5_cur - 1e-15:
                h_prev = h_cur.copy()
                h_cur = h_new2
                c5_cur = c5_new2
                patience = 0
                if c5_cur < best:
                    best = c5_cur
            else:
                h_prev = h_cur.copy()
                patience += 1
                if patience > 2000:
                    beta = min(beta * 1.5, 500.0)
                    alpha0 = max(alpha0 * 0.8, 0.003)
                    patience = 0
    return h_cur, best

def lp_direction(h, target, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    c_max = corr[top_idx[0]]
    A_rows = []
    b_rows = []
    for ki in top_idx:
        l = ki - (n - 1)
        gk = g_lag(h, l, dx)
        c_k = corr[ki]
        A_row = np.append(gk, -1.0)
        b_row = c_k - gk @ h
        A_rows.append(A_row)
        b_rows.append(b_row)
    for m in range(n):
        row = np.zeros(n + 1)
        row[m] = 1.0
        A_rows.append(row)
        b_rows.append(1.0)
        row2 = np.zeros(n+1)
        row2[m] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, 1)] * n + [(None, None)], method='highs')
    if not res.success:
        return None
    h_lp = res.x[:n]
    return project_h(h_lp, target)

def lp_phase(h, best_c5, budget_s, t0, target, max_iters=200):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best = c5_cur
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 30:
            break
        h_lp = lp_direction(h_cur, target, top_k=25)
        if h_lp is None:
            break
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            h_t = project_h(h_cur + alpha * (h_lp - h_cur), target)
            c5_t = compute_c5(h_t)
            if c5_t < c5_cur - 1e-14:
                h_cur = h_t
                c5_cur = c5_t
                improved = True
                if c5_cur < best:
                    best = c5_cur
                break
        if not improved:
            break
    return h_cur, best
