    return val, grad

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_vp(h, beta, top_k=None, penalty=1e6):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        cf = corr[top_idx]
        lags = top_idx
    else:
        cf = corr
        lags = np.arange(len(corr))
    m = np.max(cf)
    shifted = cf - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    if sw < 1e-30:
        w = np.zeros_like(cf)
        w[np.argmax(cf)] = 1.0
    else:
        w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    s = h.sum() - n / 2.0
    val += penalty * s * s
    grad = np.zeros(n)
    for i in range(len(lags)):
        wk = w[i]
        if wk < 1e-12:
            continue
        s_lag = lags[i] - (n - 1)
        if s_lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s_lag > 0:
            grad[s_lag:] += wk * (1.0 - h[:n - s_lag])
            grad[:n - s_lag] += -wk * h[s_lag:]
        else:
            t = -s_lag
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += -wk * h[:n - t]
    grad *= dx
    grad += 2 * penalty * s
    return val, grad

def lbbg_optimize(h, beta, top_k, maxiter, penalty):
    n = len(h)
    x0 = project_feasible(h.copy(), n / 2.0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def f(x):
        return smoothmax_vp(x, beta, top_k, penalty)[0]
    def g(x):
        return smoothmax_vp(x, beta, top_k, penalty)[1]
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-15})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(x0, n / 2.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def gradient_restart(h, n_steps, lr, beta, top_k):
    n = len(h)
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(n_steps):
        lr_t = lr * (0.99 ** t)
        v, g = smoothmax_vp(x, beta, top_k, penalty=1e6)
        x = project_feasible(x - lr_t * g, n / 2.0)
        cv = compute_c5(x)
        if cv < best_v:
            best_v = cv
            best_x = x.copy()
    return best_x, best_v
