Now, for the 1D optimization: given corr (current correlation array) and b (change vector), find δ that minimizes max_k (corr[k] + δ*b[k]) over δ ∈ [δ_lo, δ_hi].

Since this is a convex piecewise linear function, the minimum is at a breakpoint or at a boundary.

A simple approach: since the function is convex, we can use the fact that the "slope" (subgradient) is non-decreasing. At the optimum, 0 is in the subgradient.

The subgradient at δ is the set {b[k] : corr[k] + δ*b[k] = max_j (corr[j] + δ*b[j])}.

So at the optimum, there exist k1, k2 such that:
- corr[k1] + δ*b[k1] = corr[k2] + δ*b[k2] = max
- b[k1] ≤ 0 ≤ b[k2]

(If b[k1] = b[k2] = 0, any δ works.)

To find this, we can:
1. Evaluate at δ_lo and δ_hi
2. If the max at δ_lo is achieved only by terms with b[k] ≥ 0, the function is non-decreasing, so minimum is at δ_lo.
3. If the max at δ_hi is achieved only by terms with b[k] ≤ 0, the function is non-increasing, so minimum is at δ_hi.
4. Otherwise, binary search for the crossing point.

Actually, a simpler approach that works well in practice: golden section search or ternary search on the convex function. Since we only need ~30-50 evaluations to get high precision, and each evaluation is O(n), this is fast.

1. Computing all the "corners" (intersection points of pairs of lines that are on the upper envelope)
2. Finding the corner with the smallest value

- If subgradient at mid is entirely negative → lo = mid (minimum is to the right)

```python
    dx = 2.0 / n

    best_h = h.copy()
    
    ones = np.where(h > 1e-10)[0]  # indices with h > 0
    zeros = np.where(h < 1 - 1e-10)[0]  # indices with h < 1

            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random pair
        i = rng.choice(zeros)  # h[i] < 1, will increase
        j = rng.choice(ones)   # h[j] > 0, will decrease
        
        delta_lo = max(-h[i], h[j] - 1.0)

        # Compute b
        b = compute_b(i, j, h, n)
        
        # Find optimal delta
        delta = find_optimal_delta(corr, b, delta_lo, delta_hi)
        
        # Check improvement
        new_corr = corr + delta * b
        new_c = new_corr.max() * dx
        
        if new_c < best_c - 1e-14:
            h[i] += delta
            h[j] -= delta
            corr = new_corr
            best_c = new_c
```

```python
def swap_delta(i, j, h, n):
    nk = n - 1
    b = np.zeros(2 * n - 1)
    
    # Term 1: (1-h[i-k]) for k where 0<=i-k<n → n-1-i <= k <= i
    # Array index m = k + nk: m from max(0, n-1-i) to min(2n-2, i+nk)
    lo = max(0, n - 1 - i)
    hi = min(2*n-2, i + nk)
    if lo <= hi:
        idx = i - (np.arange(lo, hi+1) - nk)  # i - k = i - (m-nk) = i-m+nk
        b[lo:hi+1] += (1.0 - h[idx])
    
    # Term 2: -h[i+k] for k where 0<=i+k<n → -i <= k <= n-1-i
    # Array index m: m from max(0, nk-i) to min(2n-2, 2n-2-i)
    lo = max(0, nk - i)
```

```python
def build_tables(h, n):
    """Precompute lookup tables for fast swap evaluation."""
    nk = n - 1
    # Table shape: (n, 2n-1)
    # table1[i, m] = (1-h[i-(m-nk)]) if 0 <= i-(m-nk) < n else 0
    ks = np.arange(-(n-1), n)  # length 2n-1
    # For table1: index is i - k, valid when 0 <= i-k < n
    # table1[i, m] = 1 - h[i - ks[m]] if 0 <= i - ks[m] < n else 0
    # This is a difference table: for each (i, m), the index is i - ks[m]
    idx1 = np.arange(n)[:, None] - ks[None, :]  # shape (n, 2n-1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    table1 = np.zeros((n, 2*n-1))
    table1[mask1] = 1.0 - h[np.where(mask1, idx1, 0)]
    
    # table2[i, m] = h[i + ks[m]] if 0 <= i + ks[m] < n else 0
    idx2 = np.arange(n)[:, None] + ks[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    table2 = np.zeros((n, 2*n-1))
    table2[mask2] = h[np.where(mask2, idx2, 0)]
    
    # table3[j, m] = (1-h[j - ks[m]]) if 0 <= j - ks[m] < n else 0
    # Same as table1!
    table3 = table1
    
    # table4[j, m] = h[j + ks[m]] if 0 <= j + ks[m] < n else 0
    # Same as table2!
    table4 = table2
    
    return table1, table2  # table3=table1, table4=table2
```

So after each swap (two positions change), O(n) entries in each table change. That's O(n) per update, which is fine.

```python
def update_tables(table1, table2, p, q, h_old_p, h_new_p, h_old_q, h_new_q, n):
    """Update tables after h[p]: h_old_p→h_new_p and h[q]: h_old_q→h_new_q."""
    nk = n - 1
    ks = np.arange(-(n-1), n)
    
    # h[p] changes from h_old_p to h_new_p
    # table1[i,m] affected where i-ks[m]=p → i=p+ks[m], valid when 0≤p+ks[m]<n
    valid_m = (p + ks >= 0) & (p + ks < n)
    if valid_m.any():
        idx_i = p + ks[valid_m]
        table1[idx_i, valid_m] += (h_old_p - h_new_p)  # (1-h_new) - (1-h_old) = h_old - h_new
    
    # table2[i,m] affected where i+ks[m]=p → i=p-ks[m], valid when 0≤p-ks[m]<n
    valid_m = (p - ks >= 0) & (p - ks < n)
    if valid_m.any():
        idx_i = p - ks[valid_m]
        table2[idx_i, valid_m] += (h_new_p - h_old_p)  # h_new - h_old
    
    # Same for q
    valid_m = (q + ks >= 0) & (q + ks < n)
    if valid_m.any():
        idx_i = q + ks[valid_m]
        table1[idx_i, valid_m] += (h_old_q - h_new_q)
    
    valid_m = (q - ks >= 0) & (q - ks < n)
    if valid_m.any():
        idx_i = q - ks[valid_m]
        table2[idx_i, valid_m] += (h_new_q - h_old_q)
```

Wait, I need to be more careful. table1[i, m] = (1-h[i-ks[m]]) if 0 ≤ i-ks[m] < n else 0.

When h[p] changes by Δh (from h_old to h_new, so Δh = h_new - h_old):
table1[i, m] changes by -Δh when i-ks[m] = p AND 0 ≤ p < n (always true).
So: for all (i, m) with i = p + ks[m] and 0 ≤ i < n: table1[i, m] -= Δh.

Similarly, table2[i, m] = h[i+ks[m]] if 0 ≤ i+ks[m] < n else 0.
When h[p] changes by Δh:
For all (i, m) with i = p - ks[m] and 0 ≤ i < n: table2[i, m] += Δh.

So:
```python
def update_tables(t1, t2, p, delta_h, n):
    nk = n - 1
    ks = np.arange(-(n-1), n)
    valid = (p + ks >= 0) & (p + ks < n)
    if valid.any():
        t1[p + ks[valid], valid] -= delta_h
    valid = (p - ks >= 0) & (p - ks < n)
    if valid.any():
        t2[p - ks[valid], valid] += delta_h
```

This is clean and O(n) per position change.

```python
def simulated_annealing(h, n, deadline, batch_size=500, T0=0.01, T1=0.0001, seed=42):
    rng = np.random.RandomState(seed)
    nk = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, 'full')
    
    # Build initial tables
    t1, t2 = build_tables(h, n)
    
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h < 0.5)[0]
    
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    T = T0
    n_total = 0
    n_accepted = 0
    
    while time.time() < deadline:
        # Generate batch
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        n_batch = min(batch_size, len(ones_list) * len(zeros_list))
        i_batch = rng.choice(zeros_list, size=n_batch)
        j_batch = rng.choice(ones_list, size=n_batch)
        
        # Compute B matrix: (n_batch, 2n-1)
        B = t1[i_batch, :] - t2[i_batch, :] - t1[j_batch, :] + t2[j_batch, :]
        
        # Compute new max for each pair
        new_corrs = corr[None, :] + B  # (n_batch, 2n-1)
        new_maxes = new_corrs.max(axis=1)
        
        # Find best improvement
        best_idx = new_maxes.argmin()
        best_new_max = new_maxes[best_idx]
        
        if best_new_max < cur_max - 1e-15:
            # Accept
            i = i_batch[best_idx]
            j = j_batch[best_idx]
            h[i] = 1.0
            h[j] = 0.0
            corr += B[best_idx]
            update_tables(t1, t2, i, 1.0, n)
            update_tables(t1, t2, j, -1.0, n)
            cur_max = best_new_max
            n_accepted += 1
            # Update lists
            zeros_list = zeros_list[zeros_list != i]
            ones_list = np.append(ones_list, i)
            ones_list = ones_list[ones_list != j]
            zeros_list = np.append(zeros_list, j)
        elif rng.random() < np.exp((cur_max - best_new_max) / T):
            # Accept worse (simulated annealing)
            i = i_batch[best_idx]
            j = j_batch[best_idx]
            h[i] = 1.0
            h[j] = 0.0
            corr += B[best_idx]
            update_tables(t1, t2, i, 1.0, n)
            update_tables(t1, t2, j, -1.0, n)
            cur_max = best_new_max
            n_accepted += 1
            zeros_list = zeros_list[zeros_list != i]
            ones_list = np.append(ones_list, i)
            ones_list = ones_list[ones_list != j]
            zeros_list = np.append(zeros_list, j)
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
        
        n_total += n_batch
        # Cool down
        T = max(T1, T0 * (1.0 - n_total / 5e7))
    
    return best_h, best_max * dx
```
