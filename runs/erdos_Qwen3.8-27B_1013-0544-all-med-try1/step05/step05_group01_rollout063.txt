```python
from scipy.optimize import minimize
import time

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_array(h, n)))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    best_c = float(c.max())
    best_h = h.copy()
    
    for d in deltas:

                return best_h, best_c
            improved = False
            for i in range(n):

                    idx1 = i - s
                    if 0 <= idx1 < n:
                        dc_i[m] += (1.0 - h[idx1])
                    idx2 = i + s
                    if 0 <= idx2 < n:
                        dc_i[m] -= h[idx2]
                
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        return best_h, best_c
                    # Compute dc/dh[j] vector
                    dc_j = np.zeros(2 * n - 1)
                    for s in range(-(n-1), n):
                        m = s + n - 1
                        idx1 = j - s
                        if 0 <= idx1 < n:
                            dc_j[m] += (1.0 - h[idx1])
                        idx2 = j + s
                        if 0 <= idx2 < n:
                            dc_j[m] -= h[idx2]
```

```python
def _compute_dc_fast(h, n, i):
    """Compute dc/dh[i] using vectorized operations."""
    m = np.arange(2*n-1)
    s = m - (n-1)  # shift values
    
    # First term: (1-h[i-s]) for valid i-s
    idx1 = i - s  # shape (2n-1,)
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[idx1[valid1]], 0.0)
    
    # Second term: -h[i+s] for valid i+s
    idx2 = i + s
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, -h[idx2[valid2]], 0.0)
    
    return term1 + term2
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_array(h, n)))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _dc_fast(h, n, i):
    m = np.arange(2 * n - 1)
    s = m - (n - 1)
    idx1 = i - s
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i + s
    valid2 = (idx2 >= 0) & (idx2 < n)
    dc = np.zeros(2 * n - 1)
    dc[valid1] = 1.0 - h[idx1[valid1]]
    dc[valid2] += -h[idx2[valid2]]
    return dc

def compute_all_dc(h, n):
    dc_all = np.array([_dc_fast(h, n, i) for i in range(n)])
    return dc_all
