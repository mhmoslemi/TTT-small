Actually, let me reconsider the approach. The key issue is speed. For n=250:
- One correlation compute: ~15μs
- One Python loop iteration with a few operations: ~1μs

So the bottleneck is the correlation compute. We want to minimize the number of correlation computes.

Strategy: **Batch evaluation with numpy**

Instead of evaluating one swap at a time, we can:
1. Generate a batch of candidate swaps (e.g., 1000)
2. For each, compute the new h and the max correlation
3. Vectorize this somehow

But the max of the correlation makes vectorization hard. We'd need to compute the full correlation (2n-1 values) for each candidate.

Alternative: **Approximate evaluation first, exact evaluation for top candidates**

1. For each candidate swap, compute the delta at the current max lag (cheap, O(1))
2. Select the top K candidates based on this approximation
3. Fully evaluate only those K candidates

This could give a 10-100x speedup.

Let me implement this:

For a swap (i: 0→1, j: 1→0), the delta at lag k* is:
If k* > 0:
  - Position i: was contributing 0*(1-h[i+k*]) = 0, now contributes 1*(1-h[i+k*])
  - Position i-k*: was contributing h[i-k*]*(1-0) = h[i-k*], now contributes h[i-k*]*(1-1) = 0
  - Position j: was contributing 1*(1-h[j+k*]) = 1-h[j+k*], now contributes 0
  - Position j-k*: was contributing h[j-k*]*(1-1) = 0, now contributes h[j-k*]*(1-0) = h[j-k*]
  
  delta = (1-h[i+k*]) - h[i-k*] - (1-h[j+k*]) + h[j-k*]
        = 1 - h[i+k*] - h[i-k*] - 1 + h[j+k*] + h[j-k*]
        = h[j+k*] + h[j-k*] - h[i+k*] - h[i-k*]

(where out-of-range indices contribute 0)

So the change in the max lag correlation is approximately:
delta_max ≈ h[j+k*] + h[j-k*] - h[i+k*] - h[i-k*]

For a good swap, we want delta_max < 0, i.e., h[j+k*] + h[j-k*] < h[i+k*] + h[i-k*].

This gives us a cheap criterion to rank candidates!

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_alpha = 1000.0
_slsqp_n = 250

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (d / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (d / s) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def directed_binary_search(h, n, deadline, rng, max_iters=300000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    corr = compute_corr(best_h, n)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ik = zeros + k_star
        ikv = (ik >= 0) & (ik < n)
        ikr = zeros - k_star
        ikrv = (ikr >= 0) & (ikr < n)
        sz = np.zeros(len(zeros))
        sz[ikv] += best_h[ik[ikv]]
        sz[ikrv] += best_h[ikr[ikrv]]
        jk = ones + k_star
        jkv = (jk >= 0) & (jk < n)
        jkr = ones - k_star
        jkrv = (jkr >= 0) & (jkr < n)
        so = np.zeros(len(ones))
        so[jkv] += best_h[jk[jkv]]
        so[jkrv] += best_h[jkr[jkrv]]
        n_top = min(50, len(zeros), len(ones))
        top_z = zeros[np.argsort(sz)[-n_top:]]
        top_o = ones[np.argsort(so)[:n_top]]
        all_pairs = [(top_z[a], top_o[b]) for a in range(len(top_z)) for b in range(len(top_o))]

            if best_h[i] == 0.0:

                best_h[j] = 0.0

def genetic_search(n, deadline, rng, pop_size=14, max_gens=2000):
    target = n // 2
    pop = []
    for _ in range(pop_size):
        p = np.zeros(n)
        idx = rng.permutation(n)[:target]
        p[idx] = 1.0
        pop.append(p)
    pops_c = [compute_c5(p, n) for p in pop]
    bi = int(np.argmin(pops_c))
    best_h = pop[bi].copy()
    best_c = pops_c[bi]
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        order = np.argsort(pops_c)
        elite = max(2, pop_size // 4)
        survivors = [pop[order[i]].copy() for i in range(elite)]
        children = []
        n_child = pop_size - elite
        for _ in range(n_child):
            p1 = survivors[int(rng.randint(elite))]
            p2 = survivors[int(rng.randint(elite))]
            mask = rng.random(n) > 0.5
            child = np.where(mask, p1, p2)
            n_mut = rng.randint(1, 4)
            ones_c = np.where(child == 1)[0]
            zeros_c = np.where(child == 0)[0]
            for _ in range(n_mut):
                if len(ones_c) < 1 or len(zeros_c) < 1:
                    break
                a = int(rng.choice(ones_c))
                b = int(rng.choice(zeros_c))
                child[a], child[b] = 0.0, 1.0
                ones_c = np.where(child == 1)[0]
                zeros_c = np.where(child == 0)[0]
            children.append(child)
        pop = survivors + children
        pops_c = [compute_c5(p, n) for p in pop]
        if min(pops_c) < best_c:
            bi = int(np.argmin(pops_c))
            best_c = pops_c[bi]
            best_h = pop[bi].copy()
    return best_h, best_c

def subgrad_polish(h0, n, deadline, max_iter=80000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += dx * (1.0 - best_h[:n - k])
            grad[:n - k] -= dx * best_h[k:]
        else:
            to = -k
            grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.001, 0.0005]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-14:
                best_h = ht
                best_c = ct
                improved = True
                break
        if not improved:
            lr *= 0.85
            if lr < 1e-12:
                break
    return best_h, best_c

def _slsqp_obj(x):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _slsqp_n)
    m = g.max()
    s = g - m
    return m + np.log(np.exp(s - s.max()).sum()) / _slsqp_alpha

def _slsqp_eq(x):
    return x.sum() - _slsqp_n / 2.0

def _slsqp_eq_jac(x):
    return np.ones(len(x))

def slsqp_refine(h0, n, deadline, alphas):
    global _slsqp_alpha, _slsqp_n
    _slsqp_n = n
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        _slsqp_alpha = float(alpha)
        try:
            res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-15})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def multi_n_search(n_target, h_source, n_source, deadline, rng):
    xo = np.linspace(0, 2, n_source)
    xn = np.linspace(0, 2, n_target)
