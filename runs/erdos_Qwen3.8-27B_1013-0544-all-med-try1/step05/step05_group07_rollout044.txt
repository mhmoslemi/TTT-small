def subgrad_step(h, n):
    """Compute subgradient at the worst lag"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)

        g[:n - k] += -h[k:]

        g[-k:] += -h[:n + k]
    return g * dx, c * dx

def run_lbfgs(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] += -h[:n + k]
    return g * dx

def lbfgs_optimize(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        v = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        v += 1e6 * (x.sum() - n / 2.0) ** 2
        return v
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_method(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5 / n
    mom = 0.85
    v = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        g = subgrad_at_max(h, n)
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 2.0)
        else:
            h = project_feasible(h - 0.5 * v, n)
            v *= 0.4
            lr *= 0.85
            stall += 1
            if stall > 300:
                h = project_feasible(best_h + 0.04 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.3 / n
                stall = 0
        if lr < 1e-7:
            break
    return best_h, best_c

def binary_sa(h_start, n, deadline, rng, n_iterations=None):

    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = set(idx[:target].tolist())
    zeros = set(range(n)) - ones
    temp = 0.05
    count = 0
    while time.time() < deadline and temp > 1e-7:
        count += 1
        if n_iterations and count >= n_iterations:
            break
        zl = list(zeros)
        ol = list(ones)
        if not zl or not ol:
            break
        i = zl[rng.randint(len(zl))]
        j = ol[rng.randint(len(ol))]
        h[i], h[j] = 1.0, 0.0
        c_new = compute_c5(h, n)
        delta = c_new - best_c if c_new > best_c else c_new - compute_c5(h, n)
        # Use actual current cost comparison
        c_curr = c_new
        if c_curr < best_c or rng.random() < np.exp(-(c_curr - best_c) / max(temp, 1e-9)):
            ones.add(i)
            ones.discard(j)
            zeros.add(j)
            zeros.discard(i)
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        temp *= 0.9997
    return best_h, best_c

def pair_swap_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        improved = True
        epoch = 0
        while improved and time.time() < deadline and epoch < 50:
            improved = False
            epoch += 1
            for _ in range(2000):
                if time.time() > deadline:
                    break
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                for s in (1, -1):
                    hi, hj = h[i] + s * d, h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - s * d, hj + s * d
    return best_h, best_c

def binary_to_continuous(h_bin, n, rng, deadline):
    """L-BFGS from binary start with high alpha"""
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline:
            break
        h2 = lbfgs_optimize(h, n, alpha, 400)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            h = h2
    return h, best_c
