import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_g(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(800):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter=250):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, max_sweeps=250, K=14):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n2 = 2 * n - 1
    dm = np.arange(n2) - (n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if h[i] <= 1e-14 or h[j] <= 1e-14:
                    continue
                ip = i + dm
                im = i - dm
                jp = j + dm
                jm = j - dm
                dgi = np.zeros(n2)
                v = (ip >= 0) & (ip < n)
                dgi[v] = 1.0 - h[ip[v]]
                v = (im >= 0) & (im < n)
                dgi[v] -= h[im[v]]
                dgj = np.zeros(n2)
                v = (jp >= 0) & (jp < n)
                dgj[v] = 1.0 - h[jp[v]]
                v = (jm >= 0) & (jm < n)
                dgj[v] -= h[jm[v]]
                s = dgi - dgj
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(K):
                        if a == b:
                            continue
                        ia, ib = top_idx[a], top_idx[b]
                        ds = s[ib] - s[ia]
                        if abs(ds) > 1e-15:
                            dc = (g[ia] - g[ib]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max

                    val = float(np.max(g + delta * s))

                        best_val = val
                        best_delta = delta

                    h[j] -= best_delta

                    top_idx = np.argsort(g)[::-1][:K]

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
