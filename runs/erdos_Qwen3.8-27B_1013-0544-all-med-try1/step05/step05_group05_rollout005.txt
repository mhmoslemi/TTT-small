from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_of_gk(h, n, k):
    """Gradient of g_k(h) = correlate(h, 1-h, 'full')[k] * dx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    lag = k - (n - 1)
    if lag >= 0:
        # g_k = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        for i in range(n - lag):
            grad[i] += dx * (1 - h[i + lag])
            if i + lag < n:
                grad[i + lag] += dx * (-h[i])
    else:
        to = -lag
        for i in range(to, n):
            grad[i] += dx * (1 - h[i - to])  # h[i] * (1-h[i-to])
            if i - to >= 0:
                grad[i - to] += dx * (-h[i])
    return grad

def subgradient_descent(h0, n, deadline, max_iter=500000):
    """Subgradient method with Polyak-like step sizes."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    # Lower bound: C5 >= 0 (trivial) or we can compute a better one
    lb = 0.0
    
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        gmax = g.max()
        # Find all active indices
        active = np.where(g > gmax - 1e-10)[0]
        
        # Average subgradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += gradient_of_gk(h, n, k)
        grad /= len(active)
        
        # Step size: use gap to lower bound
        gap = gmax - lb
        if gap < 1e-10:
            break
        step = min(lr, gap / max(np.linalg.norm(grad)**2, 1e-20))
        step = min(step, 1.0)  # cap step
        
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller steps
            improved = False
            for f in [0.5, 0.25, 0.1, 0.05, 0.01]:
                h_try = project_feasible(h - step * f * grad, n)
                c_try = compute_c5(h_try, n)
                if c_try < gmax - 1e-12:
                    h = h_try
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                lr *= 0.95
                if lr < 1e-15:
                    break
    
    return best_h, best_c

    c = g.max()

    zeros = np.where(h == 0)[0]
