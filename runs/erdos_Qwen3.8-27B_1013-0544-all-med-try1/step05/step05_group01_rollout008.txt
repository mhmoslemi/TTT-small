```python
def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    M = c.max()
    shifted = alpha * (c - M)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    # c has length 2n-1, indexed 0 to 2n-2
    # lag for index m is: m - (n-1)
    
    grad = np.zeros(2 * n - 1)  # Actually we want grad of length n for h
    grad_h = np.zeros(n)
    
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        lag = m - (n - 1)
        # First term: (1-h[j-lag]) for j where 0 <= j-lag <= n-1
        # Second term: -h[j+lag] for j where 0 <= j+lag <= n-1
        
        # First term: j ranges from max(0,lag) to min(n-1, n-1+lag)
        j_start = max(0, lag)
        j_end = min(n, n + lag)  # exclusive
        if j_end > j_start:
            idx = np.arange(j_start, j_end)
            grad_h[idx] += dx * w[m] * (1 - h[idx - lag])
        
        # Second term: j ranges from max(0,-lag) to min(n-1, n-1-lag)
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)  # exclusive
        if j_end2 > j_start2:
            idx = np.arange(j_start2, j_end2)
            grad_h[idx] += dx * w[m] * (-h[idx + lag])
    
    return grad_h
```

Now let me think about the local search. The key idea is:
1. Identify the critical lags (those with correlation near the maximum)
2. For those lags, find the best pair move

For a pair move (i → j) with amount d:
- Change in h: h[i] -= d, h[j] += d
- Change in c[m] for each m:
  Δc[m] = d * [∂c[m]/∂x[i] * (-1) + ∂c[m]/∂x[j] * (+1)]
        = d * [∂c[m]/∂x[j] - ∂c[m]/∂x[i]]

So the change in the max is determined by which m values are critical and how Δc[m] affects them.

For a small d, the new max is approximately:
max_m (c[m] + d * (g_j[m] - g_i[m]))

where g_j[m] = ∂c[m]/∂x[j] and g_i[m] = ∂c[m]/∂x[i].

To minimize the new max, I want to choose (i, j, d) such that the critical lags are reduced the most.

This is equivalent to: minimize max_m (c[m] + d * (g_j[m] - g_i[m]))

For the critical lags (those where c[m] ≈ M), I want d * (g_j[m] - g_i[m]) < 0, i.e., g_j[m] < g_i[m] for d > 0.

So the best move is: for the critical lags, find (i, j) where g_i[m] >> g_j[m], then move mass from i to j.

This is essentially what the gradient tells us: the gradient at position i is sum_m w[m] * g_i[m]. If grad[i] > 0, increasing x[i] increases the objective. So we want to decrease x[i] where grad[i] > 0 and increase x[j] where grad[j] < 0.

So a simple gradient-based local search would be:
1. Compute grad
2. Find i with max grad[i] (most "bad" to increase)
3. Find j with min grad[j] (most "good" to increase)
4. Move small amount from i to j

```python
            c_arr = np.correlate(h, 1 - h, mode='full')
            top_idx = np.argsort(c_arr)[::-1][:topk_lags]
            crit_lags = set(int(top_idx[k] - (n - 1)) for k in range(len(top_idx)) if top_idx[k] < 2*n-1)
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i], h[j] = hi2, hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi2 + d, hj2 - d
            if not improved:
                break
    return best_h, best_c

def gradient_pair_search(h, n, deadline, alpha=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    stagnation = 0
    for step in range(200000):
        if time.time() > deadline:
            break
        grad = _smooth_grad(h, n, alpha)
        i = np.argmax(grad)
        j = np.argmin(grad)
        if grad[i] - grad[j] < 1e-14:
            break
        best_d = 0.0
        best_c_trial = best_c
        for d in [0.002, 0.005, 0.01, 0.02, 0.04]:
            if h[i] - d < -1e-12 or h[j] + d > 1 + 1e-12:
                continue
            h[i] -= d
            h[j] += d
            c = compute_c5(h, n)
            h[i] += d
            h[j] -= d
            if c < best_c_trial:
                best_c_trial = c
                best_d = d
        if best_d > 0:
            h[i] -= best_d
            h[j] += best_d
            if best_c_trial < best_c:
                best_c = best_c_trial
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
            i2 = int(np.argmax(grad))
            j2 = rng.randint(n)
            if i2 != j2 and h[i2] > 0.01:
                d = 0.005
                h[i2] -= d
                h[j2] += d
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    stagnation = 0
        if stagnation > 50:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 250
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with analytical gradient, increasing alpha
    alphas = [2000, 10000, 50000, 200000, 1000000, 5000000]
    time_per_alpha = min(60, (deadline - t0 - 180) / len(alphas))
    for a in alphas:
        if time.time() > deadline - 120:
            break
        alloc = max(15, time_per_alpha)
        ht = slsqp_opt(best_h, n, a, maxiter=2500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = ht.copy()

    # Phase 2: Projected gradient descent
    if time.time() < deadline - 80:
        gd_deadline = deadline - 80
        ht, ct = proj_grad_descend(best_h, n, 200000, gd_deadline, lr_init=0.01, max_steps=3000)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    # Phase 3: Gradient-guided pair search
    if time.time() < deadline - 60:
        ht, ct = gradient_pair_search(best_h, n, deadline - 60, alpha=500000)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 30:
        ht = slsqp_opt(best_h, n, 10000000, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()

    # Phase 5: Very fine pair search
    if time.time() < deadline - 5:
        if n <= 300:
            ht, ct = pair_search_focused(best_h, n, deadline - 2,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005], topk_lags=3)
        else:
            ht, ct = pair_search_focused(best_h, n, deadline - 2,
                                         [0.001, 0.0005, 0.0002], topk_lags=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
