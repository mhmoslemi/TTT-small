    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * (2.0 / n)
    c_max = c.max()
    # Active set: shifts where c is within active_tol of max
    active = np.where(c >= c_max - active_tol)[0]
    
    if len(active) == 0:
        return h.copy()
    
    n_active = len(active)
    dx = 2.0 / n
    g = np.zeros(n)
    
    for k in active:
        l = k - n + 1  # lag
        # g[j] += (1-h[j-l]) * [0<=j-l<n] - h[j+l] * [0<=j+l<n]
        # First term: j-l in [0,n) => j in [l, l+n) intersect [0,n)
        # Second term: j+l in [0,n) => j in [-l, n-l) intersect [0,n)

            g[l:n] += (1 - h[:n-l])
            # Second term: j in [0, n-l)
            g[0:n-l] -= h[l:n]

            g[m:n] -= h[0:n-m]

def grad_of_active(h, n, active_tols=None):
    """Compute gradient of active shifts of the correlation."""
    c_raw = np.correlate(h, 1 - h, mode='full')
    c_max = c_raw.max()
    if active_tols is None:
        active_tols = [0.0]
    
    # Find active shifts
    active = np.where(c_raw >= c_max - 0.01)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_raw)])
    
    n_act = len(active)
    g = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        if l >= 0:
            g[l:] += (1 - h[:n-l])
            g[:n-l] -= h[l:]
        else:
            m = -l
            g[:n-m] += (1 - h[m:])
            g[m:] -= h[:n-m]
    g /= n_act
    return g, c_max, c_raw

def active_gradient_descent(h, n, deadline, tol=1e-10):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.05
    no_improve = 0
    dx = 2.0 / n
    
    while time.time() < deadline:
        g, c_max, _ = grad_of_active(h, n)
        grad_norm = np.linalg.norm(g)
        if grad_norm < tol:
            break
        
        # Normalized gradient direction
        direction = -g / grad_norm
        
        # Backtracking line search
        improved = False
        for alpha in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = h + alpha * direction
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)

                    best_h = h.copy()

                else:
                    no_improve += 1
                improved = True
                break
        
        if not improved:
            step *= 0.7
            no_improve += 1
            if no_improve > 50:

def random_pair_move(h, n, rng, step_size, deadline):
    """Guided random pair moves."""
    best_c = compute_c5(h, n)
    n_tries = 0
    max_tries = 5000
    while time.time() < deadline and n_tries < max_tries:
        n_tries += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-step_size, step_size)
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
            else:
                h[i], h[j] = hi - d, hj + d
    h = project_feasible(h, n)
    return h, best_c

def slsqp_refine(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_active_gradient(h, n):
    """Compute gradient of the max of the correlation (active set method)."""
    c_raw = np.correlate(h, 1 - h, mode='full')
    c_max = c_raw.max()
    # Active set: top K shifts
    K = min(10, len(c_raw))
    top_idx = np.argsort(c_raw)[-K:]
    n_act = len(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        l = k - (n - 1)
        if l >= 0:
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
        else:
            m = -l
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    g /= n_act
    return g, c_max

def gradient_descent(h, n, deadline, base_step=0.02, K=10):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = base_step
    stall = 0
    
    while time.time() < deadline:
        c_raw = np.correlate(h, 1 - h, mode='full')
        c_max = float(c_raw.max())
        
        # Active gradient
        top_idx = np.argsort(c_raw)[-K:]
        n_act = len(top_idx)
        g = np.zeros(n)
        for k in top_idx:
            l = k - (n - 1)
            if l >= 0:
                g[l:] += (1.0 - h[:n-l])
                g[:n-l] -= h[l:]
            else:
                m = -l
                g[:n-m] += (1.0 - h[m:])
                g[m:] -= h[:n-m]
        g /= n_act
        
        # Line search
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_try = project_feasible(h - s * g, n)
            c_try = float(np.max(np.correlate(h_try, 1-h_try, mode='full')) * (2.0/n))
            if c_try < c_max - 1e-15:
                h = h_try
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                improved = True
                break
        if not improved:
            step *= 0.8
            stall += 1
            if stall > 200 or step < 1e-10:
                break
    
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def slsqp_sum(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_sum_jac(x, n):
    return np.ones(len(x))

def slsqp_refine(h, n, alpha=50000, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local_search(h, n, deadline, d=0.005, max_iter=20000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()

        i = rng.randint(n)
        j = rng.randint(n)

            if c < best_c - 1e-15:
                best_c = c
            else:
                h[i], h[j] = hi - delta, hj + delta
    return h, best_c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break

        best_h, best_c = gradient_descent(best_h, n, deadline - 3, base_step=0.005, K=5)
        if time.time() < deadline - 10:
            ht = slsqp_refine(best_h, n, alpha=300000, maxiter=5000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c, best_h = c, ht
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
