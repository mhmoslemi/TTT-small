    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    k_star = np.argmax(c)
    # Gradient of c[k_star] w.r.t. h
    # c[k_star] = sum_i h[i] * (1-h[i+k_star]) * dx
    # where k_star is the offset in the full correlation
    # In full mode, position k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    # h[j] appears as h[j] in term (1-h[j+lag]) and as h[j-lag] in term h[j-lag]*(1-h[j])
    # So: grad[j] = dx * [(1-h[j+lag]) - h[j-lag]] (with boundary handling)
    for j in range(n):
        j_plus = j + lag
        j_minus = j - lag
        term1 = 1.0
        if 0 <= j_plus < n:
            term1 = 1.0 - h[j_plus]
        term2 = 0.0
        if 0 <= j_minus < n:
            term2 = h[j_minus]
        grad[j] = dx * (term1 - term2)
    return h - lr * grad

        for _pass in range(max_passes):

                return h, best_c
            improved = False
            for i in range(n):

                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    # Try h[i]+d, h[j]-d
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    # Try h[i]-d, h[j]+d
                    hi2 = h[i] - d
                    hj2 = h[j] + d

                            best_c = c
                        else:

            if not improved:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    a = alpha * c
    amax = a.max()
    return float((amax + np.log(np.sum(np.exp(a - amax)))) / alpha)

def lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    a = alpha * c
    amax = a.max()
    exp_a = np.exp(a - amax)
    w = exp_a / exp_a.sum()
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    for p in active:
        lag = p - (n - 1)
        wl = w[p]
        lo1, hi1 = max(0, -lag), min(n, n - lag)
        lo2, hi2 = max(0, lag), min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += wl * dx * (1 - x[lo1+lag:hi1+lag])
        if hi2 > lo2:
            grad[lo2:hi2] -= wl * dx * x[lo2-lag:hi2-lag]
