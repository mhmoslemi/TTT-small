        # For i in [0, n-1-s]: contributes (1-h[i+s]) to g[i] and -h[i] to g[i+s]
        g[:n-s] += (1 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        s = -s
        # For i in [s, n-1]: h[i]*(1-h[i-s])
        # derivative wrt h[i] is (1-h[i-s]), wrt h[i-s] is -h[i]
        g[s:] += (1 - h[:n-s])
        g[:n-s] += (-h[s:])
    return g * (2.0 / n)

Now for the line search: given current h, source index i, destination index j, and direction (move from i to j), find the optimal amount d > 0 to move.

The objective after the move:
h_new = h; h_new[i] -= d; h_new[j] += d

We want to minimize max_k c_k(h_new).

A simpler approach: minimize the sum of the top-m worst c_k values, or just the worst one.

For the worst lag k*, the function f(d) = c_{k*}(h_new) is quadratic in d:
c_{k*}(h_new) = c_{k*}(h) + d * (∂c/∂h[i] - ∂c/∂h[j]) + (contribution from the interaction term)

Actually, since c is quadratic in h, c(h + d*e_i - d*e_j) is quadratic in d. We can find the exact minimum.

    cons = [{'type': 'eq', 'fun': sum_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)

    # Phase 1: Quick SLSQP improvement
    if time.time() < deadline - 200:
        ht = slsqp_round(best_h, n0, 2000, maxiter=500)
        ct = compute_c5(ht, n0)
        if ct < best_c:
            best_c = ct
            best_h = ht
        if time.time() < deadline - 200:
            ht = slsqp_round(best_h, n0, 10000, maxiter=500)
            ct = compute_c5(ht, n0)
            if ct < best_c:
                best_c = ct
                best_h = ht

    # Phase 2: Main local search
    if time.time() < deadline - 20:
        best_h, best_c = local_search(best_h, n0, deadline - 20, rng)

    # Phase 3: Try different n values
    if time.time() < deadline - 80:
        for n_try in [200, 300]:
            if n_try == n0 or time.time() > deadline - 40:
                continue
            x0 = np.linspace(0, 2, n0)
            x1 = np.linspace(0, 2, n_try)
            h_interp = np.interp(x1, x0, best_h)
            h_interp = project_feasible(h_interp, n_try)
            ct = compute_c5(h_interp, n_try)
            if time.time() < deadline - 30:
                ht = slsqp_round(h_interp, n_try, 3000, maxiter=400)
                ct2 = compute_c5(ht, n_try)
                if ct2 < ct:
                    h_interp = ht
                    ct = ct2
            if time.time() < deadline - 15:
                ht, ct = local_search(h_interp, n_try, deadline - 5, rng)
            if ct < best_c:
                best_c = ct
                best_h = ht
                n0 = n_try

    # Phase 4: Final polish from best
    if time.time() < deadline - 5:
        n_final = len(best_h)
        best_h, best_c = local_search(best_h, n_final, deadline - 1, rng)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
