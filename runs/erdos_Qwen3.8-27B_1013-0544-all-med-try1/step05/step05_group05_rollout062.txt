```python
def project_simplex_box(v, n):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(v, 0.0, 1.0)
    # If sum is already target, done
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to increase sum by diff
        # Add to variables that are below 1
        # Optimal: add uniformly to all non-saturated (h<1) variables
        # But if that would push some above 1, need to iterate
        mask = h < 1.0
        count = mask.sum()
        if count == 0:
            return h  # can't satisfy
        add = min(diff / count, 1.0 - h[mask].min())
        h[mask] += add
        # Repeat if not converged
        for _ in range(100):
            diff = target - h.sum()
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-15
                count = mask.sum()
                if count == 0:
                    break
                h[mask] += min(diff / count, 1.0)
            else:
                mask = h > 1e-15
                count = mask.sum()
                if count == 0:
                    break
                h[mask] += min(diff / count, -1.0)
        return np.clip(h, 0.0, 1.0)
    else:
        # Need to decrease
        mask = h > 0.0
        count = mask.sum()
        if count == 0:
            return h
        sub = min(-diff / count, h[mask].max())
        h[mask] -= sub
        for _ in range(100):
            diff = target - h.sum()
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-15
                count = mask.sum()
                if count == 0:
                    break
                h[mask] += min(diff / count, 1.0)
            else:
                mask = h > 1e-15
                count = mask.sum()
                if count == 0:
                    break
                h[mask] += min(diff / count, -1.0)
        return np.clip(h, 0.0, 1.0)
```

```python
def project_to_set(v, target):

    # Find tau such that sum(clip(v_i - tau, 0, 1)) = target

            lo = tau
        else:
            hi = tau
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_to_set(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) / 2, 0.0, 1.0)

def compute_grad_for_lag(h, n, k):
    """Compute gradient of g(k) with respect to h. k is the actual lag (can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        # g(k) = dx * sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        # grad[m] += dx*(1-h[m-k]) for m in [k, n-1]
        # grad[m] -= dx*h[m+k] for m in [0, n-1-k]
        if n - k > 0:
            grad[k:] += dx * (1.0 - h[:n-k])
            grad[:n-k] -= dx * h[k:]
    else:
        l = -k
        # g(-l) = dx * sum_{j=0}^{n-1-l} h[j] * (1-h[j+l])
        # grad[m] += dx*(1-h[m+l]) for m in [0, n-1-l]
        # grad[m] -= dx*h[m-l] for m in [l, n-1]
        if n - l > 0:
            grad[:n-l] += dx * (1.0 - h[l:])
            grad[l:] -= dx * h[:n-l]
    return grad

def subgradient_method(h, n, deadline, max_iter=500000, top_b=3):
    """Subgradient method using top-b lags."""
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    step = 0.1
    cooldown = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        
        # Find top b lags
        sort_idx = np.argsort(g)[::-1]
        top_lags = sort_idx[:top_b]
        
        # Weight by (g_k - second_max) or just use uniform weights on top b
        # Use weights proportional to (g_k - min_g) for the top b
        g_top = g[top_lags]
        weights = g_top - g_top.min() + 1e-10
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx in range(top_b):
            k = top_lags[idx] - (n - 1)  # convert from full correlate index to lag
            grad += weights[idx] * compute_grad_for_lag(best_h, n, k)
        
        # Try step
        h_new = project_to_set(best_h - step * grad, target)
        c_new = compute_c5(h_new, n)
        
        if c_new < m - 1e-15:
            best_h = h_new
            if c_new < best_c:
                best_c = c_new
            step = min(step * 1.05, 1.0)
            cooldown = 0
        else:
            cooldown += 1
            if cooldown > 10:
                step *= 0.8
                cooldown = 0
            # Try smaller step
            for f in [0.5, 0.25, 0.1]:
                h_try = project_to_set(best_h - step * f * grad, target)
                c_try = compute_c5(h_try, n)
                if c_try < m - 1e-15:
                    best_h = h_try
                    if c_try < best_c:
                        best_c = c_try
                    break
    
    return best_h, best_c
```

Now, for the binary local search, let me make it more efficient by precomputing the correlation and updating it incrementally:

Actually, for binary sequences, when we swap two elements (i: 0→1, j: 1→0), we can compute the exact change in each lag efficiently. But for simplicity and since n=250 is small, let's just recompute.

The key improvement for binary search is to use a **priority-based** approach:
1. Compute all lags
2. For the worst lag, identify which positions are "causing" the high overlap
3. Try swaps that would reduce those specific overlaps

For lag k > 0: overlap = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
This is high when there are many positions j where h[j+k]=1 and h[j]=0.
To reduce it, we want to either:
- Set h[j]=1 for some j where h[j+k]=1 and h[j]=0 (moving a 1 to position j)
- Set h[j+k]=0 for some j where h[j+k]=1 and h[j]=0 (moving a 1 away from j+k)

Let me implement a more targeted binary search:

```python
def targeted_binary_search(h, n, deadline, max_iters=100000):
    """Targeted binary local search using lag information."""
    target = n // 2
    # Ensure binary
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    # Precompute: for each position, which lags it contributes to
    # For a 1 at position p, it contributes to lag k for positions p-k (as h[j+k]=h[p])
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_all_g(h_bin, n)
        m = g.max()
        k_star = int(np.argmax(g)) - (n - 1)  # the worst lag
        
        # For this lag, find positions contributing to high overlap
        if k_star >= 0:
            # overlap = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
            # Contributing positions: j where h[j+k]=1 and h[j]=0
            contrib = np.where((h_bin[k_star:] == 1) & (h_bin[:n-k_star] == 0))[0]
            # These are j values; the ones are at j+k_star
            one_positions = contrib + k_star  # where ones are
            zero_positions = contrib  # where zeros are (that we want to fill)
        else:
            l = -k_star
            # overlap = sum_{j=0}^{n-1-l} h[j]*(1-h[j+l])
            contrib = np.where((h_bin[:n-l] == 1) & (h_bin[l:] == 0))[0]
            one_positions = contrib  # where ones are
            zero_positions = contrib + l  # where zeros are (that we want to clear)
        
        if len(contrib) == 0:
            break
        
        # Try to swap: move a 1 from one_positions to a zero position that
        # doesn't hurt other lags too much
        ones_in = np.where(h_bin == 1)[0]
        zeros_in = np.where(h_bin == 0)[0]
        
        improved = False
        # Try targeted swaps
        for z in zero_positions[:20]:  # try up to 20 zero positions
            for o in one_positions[:20]:  # try up to 20 one positions
                if z == o:
                    continue
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h_bin.copy()
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
            if improved:
                break
        
        if not improved:
            # Try random swap as fallback
            rng_swap = np.random.randint(n)
            rng_swap2 = np.random.randint(n)
            if h_bin[rng_swap] == 0 and h_bin[rng_swap2] == 1:
                h_bin[rng_swap], h_bin[rng_swap2] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h_bin.copy()
                else:
                    h_bin[rng_swap], h_bin[rng_swap2] = 0.0, 1.0
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_to_set(v, target):
    """Project v onto {h: 0<=h<=1, sum(h)=target} via bisection."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, n, k):
    """Gradient of g at lag k (actual lag, can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[k:] += dx * (1.0 - h[:n-k])
            grad[:n-k] -= dx * h[k:]
    else:
        l = -k
        if n - l > 0:
            grad[:n-l] += dx * (1.0 - h[l:])
            grad[l:] -= dx * h[:n-l]
    return grad

def subgrad_opt(h, n, deadline, top_b=5, max_iter=200000):
    """Subgradient method with top-b lags and adaptive step size."""
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    step = 0.5
    mom = np.zeros(n)
    mu = 0.9
    counter = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        
        # Top b lags
        top_idx = np.argsort(g)[::-1][:top_b]
        g_top = g[top_idx]
        w = np.exp((g_top - g_top[0]) * 10.0)  # exponential weighting
        w /= w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for i in range(top_b):
            k = int(top_idx[i]) - (n - 1)
            grad += w[i] * grad_lag(best_h, n, k)
        
        # Momentum
        mom = mu * mom - step * grad
        h_new = project_to_set(best_h + mom, target)
        c_new = compute_c5(h_new, n)
        
        counter += 1
        if c_new < m - 1e-15:
            best_h = h_new
            if c_new < best_c:
                best_c = c_new
            step = min(step * 1.01, 2.0)
        else:
            # Backtrack
            for f in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_to_set(best_h - step * f * grad, target)
                c_try = compute_c5(h_try, n)
                if c_try < m - 1e-15:
                    best_h = h_try
                    mom = np.zeros(n)
                    if c_try < best_c:
                        best_c = c_try
                    break
            else:
                step *= 0.7
                mom = np.zeros(n)
    
    return best_h, best_c

def binary_snap_local(h, n, deadline, max_sweeps=5000):
    """Snap to binary and do targeted local search."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
```

```python
    """Euclidean projection onto {0<=h<=1, sum(h)=target}."""
    lo, hi = v.min() - 1.0, v.max() + 1.0

    g = np.zeros(n)
    if k >= 0 and n - k > 0:
        g[k:] += dx * (1.0 - h[:n - k])
        g[:n - k] -= dx * h[k:]
    elif k < 0 and n + k > 0:

        g[:n - l] += dx * (1.0 - h[l:])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_to_set(v, target):
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        if np.clip(v - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += dx * (1.0 - h[:m])
            g[:m] -= dx * h[k:]
    else:
        l = -k
        m = n - l
        if m > 0:
            g[:m] += dx * (1.0 - h[l:])
            g[l:] -= dx * h[:m]
    return g

def smooth_max_obj(x, n, alpha):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    s = g - m
    return m + np.log(np.exp(s - s.max()).sum()) / alpha

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def subgrad_opt(h0, n, deadline, top_b=5, max_iter=400000):
    tgt = n / 2.0
    h = h0.copy()
    bc = compute_c5(h, n)
    step, stale, best_step = 0.3, 0, 0.3
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        m = g.max()
        ti = np.argsort(g)[::-1][:top_b]
        wt = np.exp(20.0 * (g[ti] - g[ti][0]))
        wt /= wt.sum()
        gr = np.zeros(n)
        for i in range(top_b):
            gr += wt[i] * grad_lag(h, n, int(ti[i]) - (n - 1))
        h2 = project_to_set(h - step * gr, tgt)
        c2 = compute_c5(h2, n)
        if c2 < m - 1e-14:
            h = h2
            if c2 < bc:
                bc = c2
            stale = 0
            best_step = step
            step = min(step * 1.003, 1.2)
        else:
            stale += 1
            acc = False
            for f in (0.5, 0.25, 0.1, 0.05):
                ht = project_to_set(h - step * f * gr, tgt)

                if ct < m - 1e-14:
                    h = ht
                    if ct < bc:
                        bc = ct
                    acc = True
                    break
            if not acc:
                step = max(step * 0.75, 1e-5)
            if stale > 200:
                step = min(best_step * 0.5, 0.3)
                stale = 0
    return h, bc

def bin_local(h0, n, deadline):
    tgt = n // 2
    idx = np.argsort(h0)[::-1]
    hb = np.zeros(n)
    hb[idx[:tgt]] = 1.0
    bc = compute_c5(hb, n)
    stall = 0
    while time.time() < deadline:
        g = np.correlate(hb, 1.0 - hb, mode='full') * (2.0 / n)
        ks = int(np.argmax(g)) - (n - 1)

        if ks >= 0:

        else:
```
