```python
def smooth_max_opt(h, n, alpha, deadline):
    dx = 2.0 / n
    def objective(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        # Smooth max approximation
        shifted = alpha * (c - m)
        max_shifted = shifted.max()
        logsumexp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        val = m + logsumexp / alpha
        # Constraint penalty
        penalty = 1e4 * (x.sum() - n/2)**2
        return val + penalty
    
    def gradient(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n-k])
            g[:n-k] += -x[k:]
        else:
            g[:n+k] += (1.0 - x[-k:])
            g[-k:] += -x[:n+k]
        g *= dx
        g += 2e4 * (x.sum() - n/2)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, method='L-BFGS-B', jac=gradient,
                      bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
        return res.x
    except:
        return h
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, deadline, rng, max_iter=500000, temp0=0.005, cool=0.99995):
    target = n // 2
    h = np.where(h_init >= 0.5, 1.0, 0.0).copy()
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        h[ones[target:]] = 0
        ones = np.where(h == 1)[0]
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - len(ones)]] = 1
        ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    temp = temp0
    n_lags = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if temp < 1e-9 or len(ones) == 0 or len(zeros) == 0:
            break
        
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        
        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h, n)
        
        accept = (new_c <= curr_c) or (rng.random() < np.exp(-(new_c - curr_c) / temp))
        
        if accept:
            curr_c = new_c
            ones = np.append(np.delete(ones, jj), i)
            zeros = np.append(np.delete(zeros, ii), j)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
        
        temp *= cool
    
    return best_h, best_c

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    weights = exp_sh / exp_sh.sum()
    logsumexp = max_sh + np.log(exp_sh.sum())
    val = m + logsumexp / alpha
    
    # Gradient: weighted sum of individual lag gradients
    grad = np.zeros(n)
    # Only process lags with significant weight
    sig = np.where(weights > 1e-8)[0]
    for m_idx in sig:
        k = m_idx - (n - 1)
        w = weights[m_idx]
        if k >= 0:
            grad[k:] += w * (1.0 - x[:n-k])
            grad[:n-k] += -w * x[k:]
        else:
            grad[:n+k] += w * (1.0 - x[-k:])
            grad[-k:] += -w * x[:n+k]
    grad *= dx
    
    # Add constraint gradient
    grad += 2e4 * (x.sum() - n/2)
    
    return val, grad

def continuous_opt(h, n, deadline):
    """Multi-alpha continuous optimization."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > deadline - 5:
            break
        
        def obj(x):
            val, _ = smooth_obj_grad(x, n, alpha)
            return val
        
        def grad(x):
            _, g = smooth_obj_grad(x, n, alpha)
            return g
        
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', jac=grad,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except:
            pass
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, deadline, rng, max_iter=500000, temp0=0.005, cool=0.99995):
    target = n // 2
    h = np.where(h_init >= 0.5, 1.0, 0.0).copy()
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        h[ones[target:]] = 0
        ones = np.where(h == 1)[0]
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - len(ones)]] = 1
        ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_c = best_c
    temp = temp0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if temp < 1e-9 or len(ones) == 0 or len(zeros) == 0:
            break
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h, n)
        if new_c <= curr_c or (temp > 1e-9 and rng.random() < np.exp(-(new_c - curr_c) / temp)):
            curr_c = new_c
            ones = np.append(np.delete(ones, jj), i)
            zeros = np.append(np.delete(zeros, ii), j)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
        temp *= cool
    return best_h, best_c

def smooth_obj_and_grad(x, n, alpha, pen_coeff):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    weights = exp_sh / exp_sh.sum()
    logsumexp = max_sh + np.log(exp_sh.sum())
    val = m + logsumexp / alpha
    val += pen_coeff * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    sig = np.where(weights > 1e-10)[0]
    for m_idx in sig:
        k = m_idx - (n - 1)
        w = weights[m_idx]
        if k >= 0:
            grad[k:] += w * (1.0 - x[:n - k])
            grad[:n - k] += -w * x[k:]
        else:
            grad[:n + k] += w * (1.0 - x[-k:])
            grad[-k:] += -w * x[:n + k]
    grad *= dx
    grad += 2.0 * pen_coeff * (x.sum() - n / 2.0)
    return val, grad

def power_obj_and_grad(x, n, p, pen_coeff):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_pos = np.maximum(c, 1e-15)
    S = (np.sum(c_pos ** p)) ** (1.0 / p)
    val = S + pen_coeff * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    c_pow = c_pos ** (p - 1) / (np.sum(c_pos ** p) / S ** p)
    sig = np.where(c_pow > 1e-10)[0]
    for m_idx in sig:
        k = m_idx - (n - 1)
        w = c_pow[m_idx]
        if k >= 0:
            grad[k:] += w * (1.0 - x[:n - k])
            grad[:n - k] += -w * x[k:]
        else:
            grad[:n + k] += w * (1.0 - x[-k:])
            grad[-k:] += -w * x[:n + k]
    grad *= dx
    grad += 2.0 * pen_coeff * (x.sum() - n / 2.0)
    return val, grad

def continuous_opt(h, n, deadline, use_power=False):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    pen = 1e5
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    if use_power:
        ps = [5, 10, 20, 50, 100, 200]
    for i, a in enumerate(alphas):
        if time.time() > deadline - 5:
            break
        if use_power:
            p = ps[i]
            def obj(x):
                return power_obj_and_grad(x, n, p, pen)[0]
            def grad(x):
                return power_obj_and_grad(x, n, p, pen)[1]
        else:
            alpha = a
            def obj(x):
                return smooth_obj_and_grad(x, n, alpha, pen)[0]
            def grad(x):
                return smooth_obj_and_grad(x, n, alpha, pen)[1]
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', jac=grad,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def targeted_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    stall = 0
    step = 0.01
    while time.time() < deadline and stall < 50 and step > 1e-5:
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        thresh = m - step * 0.5
        critical = np.where(c > thresh)[0]
        improved = False
        for m_idx in critical:
            k = m_idx - (n - 1)
            if k == 0:
                continue
            if k > 0:
                positions = [(i, i + k) for i in range(min(30, n - k))
                            if h[i] > 0.05 and h[i + k] < 0.95]
            else:
                ka = -k
                positions = [(i, i - ka) for i in range(ka, min(ka + 30, n))
                            if h[i] > 0.05 and h[i - ka] < 0.95]
            for (i, j) in positions:
                for s in [step, step * 0.5, step * 0.1]:
                    h[i] -= s
                    h[j] += s
                    if 0 <= h[i] and h[j] <= 1:
                        nc = compute_c5(h, n)
                        if nc < best_c - 1e-13:
                            best_c = nc
                            best_h = h.copy()
                            improved = True
                            break
                    h[i] += s
                    h[j] -= s
                if improved:
                    break
            if improved:
                break
        if not improved:
            step *= 0.5
            stall += 1
            for _ in range(5):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = step * rng.choice([-1, 1])
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    nc = compute_c5(h, n)
                    if nc < best_c - 1e-13:
                        best_c = nc
                        best_h = h.copy()
                        improved = True
                        break
                h[i] -= d
                h[j] += d
            if improved:
                stall = 0
    return best_h, best_c
```
