c[k] = sum_i h[i+lag] * (1-h[i]) for the relevant range.

If h is binary, this is just the count of positions where h[i+lag]=1 and h[i]=0.

If I flip h[i] from 1 to 0 and h[j] from 0 to 1:

- For each position j where h[j]=0, compute how much c[k] would increase if h[j] became 1
- The net benefit of moving from i to j is benefit_remove[i] - benefit_add[j]
- Try the top few candidates

This makes each sweep O(n*M) instead of O(n³).

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha=1000.0, top_m=30):
    g = compute_g(h, n)
    m = g.max()
    exponents = alpha * (g - m)
    exponents = np.clip(exponents, -50, 50)
    weights = np.exp(exponents)
    sw = weights.sum()
    if sw < 1e-30: return np.zeros(n)
    weights /= sw
    # Focus on top m
    if len(weights) > top_m:
        top_idx = np.argsort(weights)[-top_m:]
        tw = np.zeros_like(weights)
        tw[top_idx] = weights[top_idx]
        tw /= tw.sum()
        weights = tw
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12: continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            to = -lag
            grad[to:] += w * (1 - h[:n - to])
            grad[:n - to] -= w * h[to:]
    return grad * dx

Actually, here's another idea: **Alternating minimization on lags**. 

1. Pick the worst lag k*
2. Optimize h to minimize c[k*] (this is a linear optimization in h!)
3. Pick the new worst lag
4. Repeat

The key insight is that minimizing c[k*] for a fixed k* is a linear problem in h:
c[k*] = sum_i h[i+lag] * (1-h[i])

This is NOT linear in h (it's bilinear). But if we fix some of the h values, it becomes linear in the others.

Actually, for binary h, c[k] is the count of "1s followed by 0s" (or "0s followed by 1s") at distance lag. Minimizing this for a single lag is a combinatorial optimization problem.

For non-binary h, c[k] = sum_i h[i+lag]*(1-h[i]) is a quadratic (bilinear) function. Minimizing a bilinear function is generally hard, but for a single lag it might be easier.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, top_m=40):
    g = compute_g(h, n)
    m = g.max()
    exponents = alpha * (g - m)
    exponents = np.clip(exponents, -50, 50)
    weights = np.exp(exponents)
    sw = weights.sum()
    if sw < 1e-30:
        return np.zeros(n)
    weights /= sw
    if len(weights) > top_m:
        top_idx = np.argsort(weights)[-top_m:]
        tw = np.zeros_like(weights)
        tw[top_idx] = weights[top_idx]
        tw /= tw.sum()
        weights = tw
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            to = -lag
            grad[to:] += w * (1.0 - h[:n - to])
            grad[:n - to] -= w * h[to:]
    return grad * dx

def proj_grad_momentum(h0, n, deadline, alphas, iters_per_alpha=800):
    h = h0.copy()
    vel = np.zeros(n)
    mu = 0.9
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 0.3
        vel = np.zeros(n)
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            grad = grad_smooth_max(h, n, alpha)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn
            vel = mu * vel - lr * grad
            improved = False
            for _ in range(25):
                h_try = project_feasible(h + vel, n)
                c_try = compute_c5(h_try, n)
                if c_try < compute_c5(h, n) - 1e-14:
                    h = h_try
                    if c_try < best_c:
                        best_c = c_try
                        best_h = h.copy()
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                lr = max(lr * 1.5, 0.01)
                vel *= 0.5
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_sweeps=100000, top_k=12):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        n_lags = min(20, 2 * n - 1)
        top_lags = np.argsort(g)[-n_lags:]
        ones_mask = h == 1
        zeros_mask = 1 - h
        delta = np.zeros(n)
        for k in top_lags:
            lag = k - (n - 1)
            d = np.zeros(n)
            if lag >= 0:
                d[:n - lag] += h[lag:]
                d[lag:] -= (1.0 - h[:n - lag])
            else:
                to = -lag
                d[to:] += h[:n - to]
                d[:n - to] -= (1.0 - h[to:])
            delta += g[k] * d
        remove_scores = np.where(ones_mask, delta, -np.inf)
        add_scores = np.where(zeros_mask, -delta, np.inf)
        top_remove = np.argsort(remove_scores)[-top_k:][::-1]
        top_add = np.argsort(add_scores)[:top_k]
        best_move_c = best_c
        best_i, best_j = -1, -1
        for i in top_remove:
            if remove_scores[i] <= 0:
                continue
            for j in top_add:
                if add_scores[j] <= 0:
                    continue
                h[i] = 0
                h[j] = 1
                c = compute_c5(h, n)
                h[i] = 1
                h[j] = 0
                if c < best_move_c - 1e-15:
                    best_move_c = c
                    best_i, best_j = int(i), int(j)
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            best_c = best_move_c
            best_h = h.copy()
        else:
            break
    return best_h, best_c

def to_other_n(h, n, n_new):
    xo = np.linspace(0, 2, n)
    xn = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(xn, xo, h, left=0, right=0), n_new)

def cyclic_shifts(h, n, num_shifts=4):
    shifts = []
    for s in range(num_shifts):
        sh = np.roll(h, s * n // num_shifts)
        shifts.append(project_feasible(sh, n))
    return shifts
