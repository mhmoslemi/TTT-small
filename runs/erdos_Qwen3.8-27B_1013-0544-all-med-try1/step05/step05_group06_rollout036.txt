def init_F(h, n):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    # F[i, m] = dx * [(1-h[i+dm[m]]) * 1_{valid} - h[i-dm[m]] * 1_{valid}]
    i_idx, m_idx = np.meshgrid(np.arange(n), np.arange(2*n-1), indexing='ij')
    j_plus = i_idx + dm[m_idx]
    j_minus = i_idx - dm[m_idx]
    F = np.zeros((n, 2*n-1))
    mask_p = (j_plus >= 0) & (j_plus < n)
    mask_m = (j_minus >= 0) & (j_minus < n)
    F[mask_p] += (1.0 - h[j_plus[mask_p]])
    F[mask_m] -= h[j_minus[mask_m]]
    F *= dx
    return F

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def init_F(h, n):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    n_m = 2 * n - 1
    # Vectorized: F[i, m] = dx * [(1-h[i+dm[m]]) * 1_{valid} - h[i-dm[m]] * 1_{valid}]
    i_mat = np.arange(n)[:, None]  # (n, 1)
    d_mat = dm[None, :]  # (1, 2n-1)
    jp = i_mat + d_mat  # (n, 2n-1)
    jm = i_mat - d_mat
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)
    F = np.zeros((n, n_m))
    F[mp] += (1.0 - h[jp[mp]])
    F[mm] -= h[jm[mm]]
    F *= dx
    return F

    k4 = j + dm
    m4 = (k4 >= 0) & (k4 < n)
