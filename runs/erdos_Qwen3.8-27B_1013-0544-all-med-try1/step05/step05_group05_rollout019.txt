    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h

def smooth_obj_and_grad(x, n, beta):
    x = np.clip(x, 0.0, 1.0)
    c = compute_c(x, n)
    m = c.max()
    t = beta * (c - m)
    t_max = t.max()
    exp_t = np.exp(t - t_max)
    Z = exp_t.sum()
    p = exp_t / Z
    obj = m + (t_max + np.log(Z)) / beta
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-16:
            continue
        # Term 1: h[j] as first factor in c[k]
        # c[k] = sum_i h[n-1-k+i] * (1-h[i]) * dx
        # h[j] = h[n-1-k+i] => i = j - n + 1 + k
        # Valid: 0 <= i < n => n-1-k <= j < 2n-1-k, and 0 <= j < n
        lo = max(0, n-1-k)
        hi = min(n, 2*n-1-k)
        if lo < hi:
            i_vals = np.arange(lo, hi) - n + 1 + k
            grad[lo:hi] += pk * dx * (1.0 - x[i_vals])
        # Term 2: h[j] in (1-h[i]) where i = j
        # h[n-1-k+j] * (1-h[j]) * dx
        # Valid: 0 <= n-1-k+j < n => k-n+1 <= j <= k, and 0 <= j < n
        lo2 = max(0, k-n+1)
        hi2 = min(n, k+1)
        if lo2 < hi2:
            j_vals = np.arange(lo2, hi2)
            h_idx = n-1-k+j_vals
            grad[lo2:hi2] -= pk * dx * x[h_idx]
    return obj, grad

    def obj(x):
        return smooth_obj_and_grad(x, n, beta)[0]
    def obj_grad(x):
        return smooth_obj_and_grad(x, n, beta)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_grad, h0, method='L-BFGS-B', jac=True,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                               'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h0

                        cur_val = best_c
                        improved = True
                        break
        if not improved:

    return h, best_c

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, beta):
    c = compute_c(x, n)
    m = c.max()
    t = beta * (c - m)
    t_max = t.max()
    et = np.exp(t - t_max)
    Z = et.sum()
    p = et / Z
    obj = m + (t_max + np.log(Z)) / beta
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n, 2 * n - 1 - k)
        if lo < hi:
