            tm[to_flip] = 0
        else:
            to_flip = zeros_idx[:n // 2 - ones_count]
            tm[to_flip] = 1

        to_flip = rng.choice(zeros_idx, n//2 - ones_count, replace=False)
        h[to_flip] = 1

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def thue_morse_binary(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    ones_count = int(tm.sum())
    if ones_count != n // 2:
        ones_idx = np.where(tm == 1)[0]
        zeros_idx = np.where(tm == 0)[0]
        if ones_count > n // 2:
            tm[ones_idx[-(ones_count - n // 2):]] = 0
        else:
            tm[zeros_idx[:n // 2 - ones_count]] = 1
    return tm

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def greedy_local_search(hb, n, deadline, max_sweeps=500, rng=None):
    best_c = compute_c5(hb, n)
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        if rng is not None:
            rng.shuffle(zeros)
            rng.shuffle(ones)
        else:
            np.random.shuffle(zeros)
            np.random.shuffle(ones)
        for i in zeros:
            found = False
            for j in ones:
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    oi = ones.index(j)
                    zi = zeros.index(i)
                    ones[oi] = i
                    zeros[zi] = j
                    found = True
                    improved = True
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if found:
                break
        if not improved:
            break
    return hb, best_c
