import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (g - g.max())
    p = np.exp(shifted)
    s = p.sum()
    p /= s
    q = 1.0 - 2.0 * h
    grad = np.correlate(p, q, mode='valid') * dx
    smooth_val = g.max() + np.log(s) / alpha
    return smooth_val, grad, g

def proj_step(h, grad, lr, n):
    target = n / 2.0
    h_new = h - lr * grad
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        hp = np.clip(h_new - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)

def grad_optimize(h, n, deadline, alphas, max_iter_per_alpha=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_cur = h.copy()
    c_cur = best_c
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr0 = min(0.5, 3.0 / alpha)
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                break
            lr = lr0

            ok = False

                c_t = compute_c5(h_t, n)

                    ok = True

                        best_c = c_t
                        best_h = h_cur.copy()
                    break
                lr *= 0.5
            if not ok:
                break
    return best_h, best_c

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], m_idx[v4]] += delta

def pair_descent(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        current_max = g.max()
        K = min(8, 2 * n - 1)
        top_idx = np.argsort(g)[::-1][:K]
        n_imp = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for a in range(K):
                    for b in range(K):
                        if top_idx[a] == top_idx[b]:
                            continue
                        ds = s[top_idx[b]] - s[top_idx[a]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[a]] - g[top_idx[b]]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                bd = 0.0
                bv = current_max
                for d in cands:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < bv - 1e-13:
                        bv = val
                        bd = d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    g += bd * s
                    update_F(F, i, j, bd, n, dm)
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    n_imp += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if n_imp == 0:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c

    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()

        n = 250
        h0 = np.full(n, 0.5)

    best_c = compute_c5(best_h, n)

            sub_dl2 = dl - 30 - attempt * 5

            if c < best_c:

        if time.time() < dl - 3:
            h, c = pair_descent(best_h.copy(), n, dl - 3, max_sweeps=100)
            if c < best_c:
                best_c, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
