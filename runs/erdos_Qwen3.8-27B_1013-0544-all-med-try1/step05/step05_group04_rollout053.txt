```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]

                       bounds=bounds, constraints=cons,

                if time.time() > deadline:
                    break

                    d = direction * delta
                    if h[i] + d < 0 or h[i] + d > 1:
                        continue
                    # Find best compensating coordinate
                    # Simple: distribute to all others proportionally to their "room"
                    h_test = h.copy()
                    h_test[i] += d
                    # Distribute -d to others
                    others = np.arange(n)
                    others = others[others != i]
                    if d > 0:
                        # Need to decrease others by total d
                        room = h_test[others].copy()  # can decrease by h[j]
                        tot = room.sum()
                        if tot < 1e-15:
                            continue
                        h_test[others] -= (d / tot) * room
                    else:
                        # Need to increase others by total -d
                        room = 1.0 - h_test[others]
                        tot = room.sum()
                        if tot < 1e-15:
                            continue
                        h_test[others] += (-d / tot) * room
                    
                    h_test = np.clip(h_test, 0, 1)
                    # Re-project to fix sum
                    h_test = project_feasible(h_test, n)
                    c = compute_c5(h_test, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        h = h_test
                        improved = True
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
```

Now let me think about the local search more carefully.

For the local search, I'll use a **randomized pair move** approach:
- Each step: pick two random indices (i, j), try to move mass delta from i to j (and vice versa)
- This is O(1) per trial
- With enough trials, it should explore the neighborhood well
- Use simulated annealing: accept worse moves with probability exp(-Δc/T), decreasing T over time

This is much faster than exhaustive pair search and can escape local optima.

```python
def randomized_search(h, n, deadline, n_trials_per_delta=None):
    """Randomized pair-move local search with simulated annealing."""
    best_c = compute_c5(h, n)
    current_c = best_c
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for delta in deltas:
        if time.time() > deadline:
            break
        T = 0.001  # temperature
        T_min = 1e-6
        n_trials = 2000
        for _ in range(n_trials):
            if time.time() > deadline:
                return h, best_c
            # Pick random pair
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            
            # Try moving mass from i to j
            hi = h[i]
            hj = h[j]
            if hi - delta >= 0 and hj + delta <= 1:
                h[i] -= delta
                h[j] += delta
                c = compute_c5(h, n)
                if c < current_c:
                    current_c = c
                    if c < best_c:
                        best_c = c
                else:
                    # Accept with probability exp(-(c-current_c)/T)
                    if np.random.random() < np.exp(-(c - current_c) / T):
                        current_c = c
                    else:
                        h[i] += delta
                        h[j] -= delta
            
            # Try moving mass from j to i
            hi = h[i]
            hj = h[j]
            if hj - delta >= 0 and hi + delta <= 1:
                h[j] -= delta
                h[i] += delta
                c = compute_c5(h, n)
                if c < current_c:
                    current_c = c
                    if c < best_c:
                        best_c = c
                else:
                    if np.random.random() < np.exp(-(c - current_c) / T):
                        current_c = c
                    else:
                        h[j] += delta
                        h[i] -= delta
        
        T *= 0.5
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_c5_corr(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

                    return h, best_c
```
