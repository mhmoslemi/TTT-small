3. If yes, check that it doesn't increase any lag in C_max1 by more than 1 (since those would then be at M).
4. If both conditions met, accept the swap.

            if s < 1e-13:

def softmax_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    return m + (np.log(np.exp(sh).sum())) / alpha

def softmax_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += wk * (1.0 - x[k:n])
            grad[k:n] -= wk * x[0:n-k]
        else:
            kk = -k
            grad[kk:n] += wk * (1.0 - x[0:n-kk])
            grad[0:n-kk] -= wk * x[kk:n]
    return grad

    def obj(x):
        return softmax_obj(x, n, alpha)
    def grad(x):
        return softmax_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

                j = ones[rng.randint(len(ones))]

                    jk = j + k
                    if 0 <= jk < n and hb[jk] == 0: delta -= 1
                    ikb = i - k
                    if 0 <= ikb < n and hb[ikb] == 1: delta -= 1
                    jkb = j - k

            j = ones[rng.randint(len(ones))]
