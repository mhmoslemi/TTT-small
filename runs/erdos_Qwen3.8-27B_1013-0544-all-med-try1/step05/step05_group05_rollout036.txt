import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(candidates, alpha):
    m = candidates.max()
    s = candidates - m
    return m + np.log(np.exp(s - s.max()).sum()) / alpha

def slsqp_step(h0, n, alpha, maxiter, deadline):
    def obj(x):
        g = compute_all_g(x, n)
        return smooth_max(g, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def subgrad_constraint(h, n, top_k=3):
    """Compute subgradient of max_k g(k) using top-k active constraints."""
    g = compute_all_g(h, n)
    m = g.max()
    top = np.argsort(g)[::-1][:top_k]
    grad = np.zeros(n)
    dx = 2.0 / n
    w = 1.0 / top_k
    for kl in top:
        lag = kl - (n - 1)
        if lag > 0:
            grad[:n-lag] += w * dx * (1.0 - h[lag:])
            grad[lag:] -= w * dx * h[:n-lag]
        elif lag < 0:
            mv = -lag
            grad[mv:] += w * dx * (1.0 - h[:n-mv])
            grad[:n-mv] -= w * dx * h[mv:]
        else:
            grad += w * dx * (1.0 - 2.0 * h)
    return grad, m

def subgrad_loop(h0, n, deadline, max_iter=100000, top_k=3):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, m = subgrad_constraint(best_h, n, top_k)
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        accepted = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                best_h = ht
                accepted = True
                break
        if not accepted:
            lr *= 0.7
            if lr < 1e-15:
                break
        else:
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
    return best_h, best_c

def binary_optimize(h0, n, deadline, max_iters=200000, seed=42):

            stagnant = 0
    return best_h, best_c

            if s < 1e-14:

    s = g - m

import numpy as np
from scipy.optimize import minimize
import time

_p = {'n': 0, 'alpha': 1}

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    n, a = _p['n'], _p['alpha']
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max(); s = g - m
    return m + np.log(np.exp(s - s.max()).sum()) / a

def _con_fun(x):
    return x.sum() - _p['n'] / 2.0

def slsqp_step(h0, n, alpha, maxiter, dl):
    _p['n'], _p['alpha'] = n, alpha
    try:
        r = minimize(_obj, h0, method='SLSQP', bounds=[(0.0,1.0)]*n,
                     constraints=[{'type':'eq','fun':_con_fun}],
                     options={'maxiter':maxiter,'ftol':1e-16})
        if time.time() > dl: return h0
        return project_feasible(r.x, n)
    except:
        return h0

def to_bin(h, n):
    t = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n); hb[idx[:t]] = 1.0
    return hb

def subgrad_g(h, n, top_k):
    g = compute_all_g(h, n); m = g.max()
    tp = np.argsort(g)[::-1][:top_k]
    gr = np.zeros(n); dx = 2.0/n; w = 1.0/top_k
    for kl in tp:
        lag = kl - (n - 1)
        if lag > 0:
            gr[:n-lag] += w*dx*(1.0-h[lag:]); gr[lag:] -= w*dx*h[:n-lag]
        elif lag < 0:
            mv = -lag
            gr[mv:] += w*dx*(1.0-h[:n-mv]); gr[:n-mv] -= w*dx*h[mv:]
        else:
            gr += w*dx*(1.0 - 2.0*h)
    return gr, m

def subgrad_run(h0, n, dl, top_k=3):
    bh = h0.copy(); bc = compute_c5(bh, n); lr = 0.5
    while time.time() < dl:
        gr, m = subgrad_g(bh, n, top_k)
        ng = np.linalg.norm(gr)
        if ng < 1e-15: break
        gr /= ng; ok = False
        for f in [1.0,0.5,0.25,0.1,0.05,0.01,0.001]:
            ht = project_feasible(bh - lr*f*gr, n)
            if compute_c5(ht, n) < m - 1e-12:
                bh = ht; ok = True; break
        if not ok:
            lr *= 0.7
            if lr < 1e-15: break
        else:
            c = compute_c5(bh, n)
            if c < bc: bc = c
    return bh, bc

def greedy_best_swap(h, n, ntrials, dl, rng):
    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
    cur_c = compute_c5(h, n)
    best_d = 0.0; best_ij = None
    for _ in range(ntrials):
        if time.time() > dl: break
        i = zeros[rng.randint(len(zeros))]; j = ones[rng.randint(len(ones))]
        h[i] = 1.0; h[j] = 0.0
        nc = compute_c5(h, n)
        h[i] = 0.0; h[j] = 1.0
        if nc < cur_c - best_d:
            best_d = cur_c - nc; best_ij = (i, j)
    return best_ij

def sa_loop(h0, n, dl, t_init, t_fin, seed):
    hb = to_bin(h0, n); cc = compute_c5(hb, n)
    bh, bc = hb.copy(), cc
    rng = np.random.RandomState(seed); total = 500000
    for it in range(total):
        if time.time() > dl: break
        T = t_init * (t_fin/t_init)**(it/total)
        ones = np.where(hb==1)[0]; zeros = np.where(hb==0)[0]
        i = zeros[rng.randint(len(zeros))]; j = ones[rng.randint(len(ones))]
        hb[i]=1.0; hb[j]=0.0; nc = compute_c5(hb,n)
        if nc <= cc or rng.random() < np.exp(-(nc-cc)/max(T,1e-18)):
            cc = nc
            if nc < bc: bc = nc; bh = hb.copy()
        else:
            hb[i]=0.0; hb[j]=1.0
    return bh, bc
