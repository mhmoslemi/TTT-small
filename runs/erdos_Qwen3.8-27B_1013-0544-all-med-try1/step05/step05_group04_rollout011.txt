        h, best_c = subgradient_descent(h, n, deadline - 5, top_m=3)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = None
    best_c = float('inf')

    # Phase 1: Main optimization at initial resolution
    alloc1 = min(400, (deadline - t0) * 0.45)
    h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try different resolutions
    remaining = deadline - time.time()
    if remaining > 80:
        x_init = np.linspace(0, 2, n_init)
        for n_try in [200, 300, 400]:
            if time.time() > deadline - 40:
                break
            if n_try == n_init:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(80, remaining * 0.3)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 3: SA for global exploration
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        h_sa, c_sa = sa_phase(best_h, n, deadline - 30, rng, n_iter_max=200000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
            if time.time() < deadline - 10:
                ht = slsqp(best_h, n, 200000, maxiter=2000)
                c = compute_c5(ht, n)
                if c < best_c:
                    best_c = c
                    best_h = ht
                h2, c2 = subgradient_descent(ht, n, deadline - 3, top_m=3)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2

    # Phase 4: Final intensive local optimization
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        h_opt, c_opt = subgradient_descent(best_h, n, deadline - 3, top_m=2, max_iter=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 2:
            ht = slsqp(best_h, n, 5000000, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
            best_h = project_feasible(ht, n)
            best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
