import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    smax = shifted.max()
    w = np.exp(shifted - smax)
    wsum = w.sum()
    w /= wsum
    val = m + (smax + np.log(wsum)) / alpha
    
    one_minus_h = 1 - h
    grad = np.zeros(n)
    for j in range(n):
        k_lo = -j
        k_hi = n - 1 - j
        w_start = k_lo + n - 1
        n1 = k_hi - k_lo + 1
        grad[j] += (w[w_start:w_start+n1] * one_minus_h[:n-j]).sum()
        
        k_lo2 = max(-(n-1), j - n + 1)
        k_hi2 = min(n-1, j)
        w_start2 = k_lo2 + n - 1
        n2 = k_hi2 - k_lo2 + 1
        h_start = j - k_hi2
        grad[j] -= (w[w_start2:w_start2+n2] * h[h_start:h_start+n2]).sum()
    
    grad *= dx
    return val, grad

def proj_grad_descent(h, n, deadline, alpha_init=500, alpha_end=200000):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    h = h.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.05 / alpha_init * n  # scale with n and alpha
    
    n_total = int(3000)
    for it in range(n_total):
        if time.time() > deadline:
            break
        
        # Nesterov step
        y = h + momentum * v
        y = np.clip(y, 0, 1)
        
        # Compute gradient at y (or h for simplicity)
        val, grad = smooth_max_and_grad(y, n, alpha)
        
        # Adaptive step size
        # Simple: use a fixed lr that decreases
        progress = it / n_total
        alpha_curr = alpha_init * (alpha_end / alpha_init) ** progress
        lr_curr = 2.0 / (alpha_curr + 1)
        
        v = momentum * v - lr_curr * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return m + np.log(np.sum(np.exp(sh))) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        smax = sh.max()
        w = np.exp(sh - smax)
        w /= w.sum()
        val = m + (smax + np.log(w.sum())) / alpha
        
        # Gradient computation (same as smooth_max_and_grad)
        one_minus_x = 1 - x
        g = np.zeros(n)
        for j in range(n):
            k_lo = -j
            k_hi = n - 1 - j
            w_start = k_lo + n - 1
            n1 = k_hi - k_lo + 1
            g[j] += (w[w_start:w_start+n1] * one_minus_x[:n-j]).sum()
            
            k_lo2 = max(-(n-1), j - n + 1)
            k_hi2 = min(n-1, j)
            w_start2 = k_lo2 + n - 1
            n2 = k_hi2 - k_lo2 + 1
            h_start = j - k_hi2
            g[j] -= (w[w_start2:w_start2+n2] * x[h_start:h_start+n2]).sum()
        g *= dx
        return val, g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj_grad, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
