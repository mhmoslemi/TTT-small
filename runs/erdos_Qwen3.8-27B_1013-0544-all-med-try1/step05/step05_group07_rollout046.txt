And I can vectorize this. Now for the 1D optimization, given c (current correlation, length 2p+1) and D (length 2p+1), find δ ∈ [0, δ_max] minimizing max(c + δ*D).

Since this is a convex piecewise-linear function, I'll use the following approach:
1. If max(D) ≤ 0: f is non-increasing, minimum at δ_max
2. If min(D) ≥ 0: f is non-decreasing, minimum at δ=0
3. Otherwise: binary search

For the binary search:
- f(δ) = max(c + δ*D)
- The subgradient of f at δ is the convex hull of {D[k] : k is argmax of c + δ*D}
- If all D[k] for active k are < 0, increase δ
- If all D[k] for active k are > 0, decrease δ
- Otherwise (some positive, some negative), we're at the minimum (or very close)

Actually, a simpler approach: since f is convex piecewise linear with at most O(n) pieces, I can just evaluate it at the "breakpoints." The breakpoints occur where two pieces cross: (c[a] + δ*D[a]) = (c[b] + δ*D[b]), i.e., δ = (c[b] - c[a]) / (D[a] - D[b]).

But there are O(n^2) breakpoints. Instead, let me just do a coarse grid + refinement:

```python
def find_optimal_delta(c, D, delta_max):
    if delta_max <= 0:
        return 0.0
    # Coarse grid
    n_grid = 50
    deltas = np.linspace(0, delta_max, n_grid)
    # Evaluate f at all grid points: f(d) = max(c + d*D)
    # c has shape (2n-1,), D has shape (2n-1,)
    # We want max over last axis
    vals = c[None, :] + deltas[:, None] * D[None, :]  # shape (n_grid, 2n-1)
    f_vals = vals.max(axis=1)
    best_idx = np.argmin(f_vals)
    # Refine around best_idx
    if best_idx == 0:
        return 0.0
    if best_idx == n_grid - 1:
        return delta_max
    lo = deltas[best_idx - 1]
    hi = deltas[best_idx + 1]
    # Binary search / ternary search on [lo, hi]
    for _ in range(30):
        mid = (lo + hi) / 2
        f_lo = np.max(c + lo * D)
        f_mid = np.max(c + mid * D)
        f_hi = np.max(c + hi * D)
        if f_mid < f_lo and f_mid < f_hi:
            # Narrow
            if f_mid <= f_lo:
                hi = mid
            else:
                lo = mid
        elif f_lo <= f_mid:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2
```

Actually, for a convex function, ternary search works:
```python
def find_optimal_delta(c, D, delta_max):
    if delta_max <= 0:
        return 0.0
    lo, hi = 0.0, delta_max
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c + m1 * D)
        f2 = np.max(c + m2 * D)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_h_pad(h, n):
    p = n - 1
    return np.concatenate([np.zeros(p), h, np.zeros(p)])

def get_D_padded(h_pad, i, j, p):
    """D[m] = h_pad[j+m] + h_pad[j-m+2p] - h_pad[i+m] - h_pad[i-m+2p] for m in [0, 2p]"""
    m = np.arange(2 * p + 1)
    return h_pad[j + m] + h_pad[j - m + 2 * p] - h_pad[i + m] - h_pad[i - m + 2 * p]

def find_optimal_delta(c, D, delta_max):
    """Find delta in [0, delta_max] minimizing max(c + delta*D)"""
    if delta_max <= 1e-15:
        return 0.0
    # Quick check
    if np.all(D <= 0):
        return delta_max
    if np.all(D >= 0):
        return 0.0
    lo, hi = 0.0, delta_max
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + m1 * D)
        f2 = np.max(c + m2 * D)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def swap_descent(h, n, deadline, rng, mode='random', max_iter=500000):
    """
    Swap-based coordinate descent.
    mode: 'random' or 'targeted' or 'full'
    """
    p = n - 1
    h_pad = build_h_pad(h, n)
    c = compute_c5_vec(h, n)
    best_c = float(np.max(c))
    best_h = h.copy()
    
    n_pairs = n * (n - 1)  # directed pairs
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        if mode == 'full':
            # Go through all pairs in order
            pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
            for i, j in pairs:
                if time.time() > deadline:
                    break
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                D = get_D_padded(h_pad, i, j, p)
                delta = find_optimal_delta(c, D, delta_max)
                if delta > 1e-10:
                    c_new = c + delta * D
                    if np.max(c_new) < best_c - 1e-12:
                        h[i] -= delta
                        h[j] += delta
                        c = c_new
                        best_c = float(np.max(c))
                        best_h = h.copy()
                        h_pad = build_h_pad(h, n)
        elif mode == 'targeted':
            # Focus on reducing the current max lag
            k_star = int(np.argmax(c))
            k = k_star - p
            # s[i] = h[i+k] + h[i-k] (with out-of-bounds = 0)
            s = np.zeros(n)
            for idx in range(n):
                val = 0.0
                idx2 = idx + k
                if 0 <= idx2 < n:
                    val += h[idx2]
                idx3 = idx - k
                if 0 <= idx3 < n:
                    val += h[idx3]
                s[idx] = val
            # Pick i from top s values, j from bottom s values
            top = np.argsort(s)[::-1][:20]
            bot = np.argsort(s)[:20]
            i = top[rng.randint(len(top))]
            j = bot[rng.randint(len(bot))]
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max < 1e-12:
                continue
            D = get_D_padded(h_pad, i, j, p)
            delta = find_optimal_delta(c, D, delta_max)
            if delta > 1e-10:
                c_new = c + delta * D
                if np.max(c_new) < best_c - 1e-12:
                    h[i] -= delta
                    h[j] += delta
                    c = c_new
                    best_c = float(np.max(c))
                    best_h = h.copy()
                    h_pad = build_h_pad(h, n)
        else:  # random
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max < 1e-12:
                continue
            D = get_D_padded(h_pad, i, j, p)
            delta = find_optimal_delta(c, D, delta_max)
            if delta > 1e-10:
                c_new = c + delta * D
                if np.max(c_new) < best_c - 1e-12:
                    h[i] -= delta
                    h[j] += delta
                    c = c_new
                    best_c = float(np.max(c))
                    best_h = h.copy()
                    h_pad = build_h_pad(h, n)
    
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 1e6 * (x.sum() - n / 2.0) ** 2
    return val

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    m = int(np.argmax(c))
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] += -x[k:]
    else:
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] += -x[:n+k]
    g *= dx
    # Add log-sum-exp gradient (approximate as subgradient at max)
    g += 2e6 * (x.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, alpha, deadline):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    def grad(x):
        return smooth_max_grad(x, n, alpha)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n, lr, deadline, rng, max_iter=10000):
    dx = 2.0 / n
    p = n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    mom = 0.8
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - p
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] += -h[k:]
        else:
            grad[:n+k] += (1.0 - h[-k:])
            grad[-k:] += -h[:n+k]
        grad *= dx
        v = mom * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.01, 5.0)
        else:
            lr *= 0.95
            v *= 0.5
            if lr < 1e-7:
                break
    return best_h, best_c

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial

    p1 = t0 + 0.10 * (budget_s - 5)
    h1, c1 = subgradient_step(best_h.copy(), n, 0.5, p1, rng, max_iter=5000)
    if c1 < best_c:

    p2 = t0 + 0.50 * (budget_s - 5)
    h2, c2 = swap_descent(best_h.copy(), n, p2, rng, mode='targeted', max_iter=200000)
    if c2 < best_c:
        best_c, best_h = c2, h2
    
    # Phase 3: Random swap descent (20% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    h3, c3 = swap_descent(best_h.copy(), n, p3, rng, mode='random', max_iter=500000)
    if c3 < best_c:
        best_c, best_h = c3, h3
    
    # Phase 4: More targeted (15% time)
    p4 = t0 + 0.85 * (budget_s - 5)
    h4, c4 = swap_descent(best_h.copy(), n, p4, rng, mode='targeted', max_iter=200000)
    if c4 < best_c:
        best_c, best_h = c4, h4
    
    # Phase 5: L-BFGS polish (10% time)
    if time.time() < deadline - 30:
        for alpha in [50000, 200000]:
            if time.time() > deadline - 20:
                break
            h5 = lbfgs_refine(best_h.copy(), n, alpha, deadline)
            c5 = compute_c5(h5, n)

            # Quick swap polish after L-BFGS

                if c6 < best_c:
                    best_c, best_h = c6, h6
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
