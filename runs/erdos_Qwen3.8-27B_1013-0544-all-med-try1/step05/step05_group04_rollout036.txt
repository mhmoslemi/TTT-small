def obj_smooth_max(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _cf(x, n):
    return x.sum() - n / 2.0

def _cj(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def adaptive_pair_search(h, n, deadline, d0=0.01, dmin=1e-6):
    best_c = compute_c5(h, n)
    d = d0
    while d >= dmin and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            for j in range(i + 1, n):
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; improved = True
                    else:
                        h[i] -= d; h[j] += d
                if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; improved = True
                    else:
                        h[i] += d; h[j] -= d
        if not improved:
            d *= 0.5
    return h, best_c

def near_pair_search(h, n, deadline, d0=0.01, dmin=1e-5, radius=60):
    best_c = compute_c5(h, n)
    d = d0
    while d >= dmin and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if j <= i:
                    continue
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h[i] += d; h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; improved = True
                    else:
                        h[i] -= d; h[j] += d
        if not improved:
            d *= 0.5
    return h, best_c

def make_structured_starts(n, x_init, h_parent):
    """Generate diverse structured starting points."""
    starts = []
    x = np.linspace(0, 2, n)
    # Interpolated parent
    h0 = np.interp(x, x_init, h_parent)
    starts.append(project_feasible(h0, n))
    # Sine-based
    for freq, amp in [(1, 0.3), (2, 0.25), (3, 0.2), (1, 0.4), (2, 0.35)]:
        hs = 0.5 + amp * np.sin(np.pi * freq * x)
        starts.append(project_feasible(hs, n))
    # Ramp
    hr = np.linspace(0.5, 0.5, n) + 0.3 * (x / 2.0 - 0.5)
    starts.append(project_feasible(hr, n))
    return starts

def optimize_at_n(h_start, n, deadline, rng):
    """Full optimization pipeline at a given n."""
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    # SLSQP with increasing smoothness
    for alpha in [2000, 20000, 100000, 500000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp_opt(h, n, alpha, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    # Adaptive pair search
    if time.time() < deadline - 10:
        if n <= 150:

        else:

        ht = slsqp_opt(h, n, 2000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    return h, best_c
