from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    w_flip = w[::-1]
    conv1 = np.convolve(h, w_flip, mode='full')
    term1 = conv1[n-1:2*n-1]
    conv2 = np.convolve(h, w, mode='full')
    term2 = conv2[n-1:2*n-1]
    return dx * (1.0 - term1 - term2)

def smoothed_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh -= sh.max()
    return cmax + float(np.log(np.sum(np.exp(sh)))) / alpha

                continue

            best_c_try = best_c
            for j in candidates:

                    h[j] = min(h[j], 1.0)

                    if c < best_c_try:
                        best_c_try = c
                        best_j = j
                h[j] -= delta

                h[best_j] = min(h[best_j] + delta, 1.0)
                best_c = best_c_try
            else:
                h[i] = target - delta  # undo
        else:
            # Similar for delta < 0
            ...
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cmax = c.max()
    sh = alpha * (c - cmax)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    wf = w[::-1]
    t1 = np.convolve(h, wf, 'full')[n-1:2*n-1]
    t2 = np.convolve(h, w, 'full')[n-1:2*n-1]
    return dx * (1.0 - t1 - t2)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(100):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            return np.clip(h, 0.0, 1.0)
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def gd(h, n, alpha, max_it, deadline):
    h = h.copy()
    f = smooth_obj(h, n, alpha)
    best_c = compute_c5(h, n)

            break
        g = compute_grad(h, n, alpha)
        g -= g.mean()
        step = 1.0
        ok = False
        for _ in range(60):
            hn = project(h - step * g, n)
            fn = smooth_obj(hn, n, alpha)
            if fn < f - 1e-18 * max(1.0, abs(f)):
                h, f = hn, fn
                ok = True
                break
            step *= 0.5
        if not ok:
            break
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    return h, best_c

def gd_multi(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    h = h.copy()
    best_c = compute_c5(h, n)
    for a in alphas:
        if time.time() > deadline - 2:
            break
        h, c = gd(h, n, a, 20000, deadline - 2)
        if c < best_c:
            best_c = c
    return h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i]+d, h[j]-d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # Try h[i]-d, h[j]+d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def slsqp_polish(h, n, alpha, deadline):
    """Use SLSQP as a final polish."""
    def obj(x):
        return smooth_obj(x, n, alpha)
    def grad(x):
        return compute_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-16})
        if res.fun < smooth_obj(h, n, alpha):
            return project(res.x, n)
    except:
        pass
    return h
