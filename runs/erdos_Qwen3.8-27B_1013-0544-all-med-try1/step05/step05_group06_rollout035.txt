def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def pair_descent(h, n, deadline, max_sweeps=200, K=6):
    """Pair descent with F-matrix for efficient candidate generation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = np.zeros((n, 2 * n - 1))
        for m_idx in range(2 * n - 1):
            d = dm[m_idx]
            i_lo = max(0, d)
            i_hi = min(n, n + d)
            if i_lo < i_hi:
                F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
            j_lo = max(0, -d)
            j_hi = min(n, n - d)
            if j_lo < j_hi:
                F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
        
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = False
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    # Update F
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i, m_idx[v1]] = F[i, m_idx[v1]]  # Actually need to update properly
                    # Rebuild F is expensive, let's just recompute g
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved = True
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for mi in range(2 * n - 1):
        d = dm[mi]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, mi] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, mi] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    mi = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], mi[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], mi[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], mi[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], mi[v4]] += delta

def pair_descent(h, n, deadline, max_sweeps=500, K=8):

    dm = np.arange(2 * n - 1) - (n - 1)

        current_max = g.max()

        improved = False

            if time.time() > deadline:
                break

                ds_vec = dx * s

                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(K):
                        if top_idx[a] == top_idx[b]:
                            continue

                        if abs(ds_val) > 1e-15:
                            dc = (g[top_idx[a]] - g[top_idx[b]]) / ds_val
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))

                        continue

                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:

                    h[j] -= best_delta
                    g += best_delta * ds_vec
                    update_F(F, i, j, best_delta, n, dm)

def softmax_obj(x, n, alpha):
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def softmax_grad(x, n, alpha):
    dx = 2.0 / n
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    weights = exps / exps.sum()
    F_mat = compute_F(x, n)
    return dx * (F_mat.T @ weights)

def slsqp_polish(h, n, deadline, alpha=5000):
    def wrapper_obj(x):
        return softmax_obj(x, n, alpha)
    def wrapper_grad(x):
        return softmax_grad(x, n, alpha)
    def wrapper_cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': wrapper_cons}]
    try:
        remaining = deadline - time.time()
        maxiter = int(min(300, max(50, remaining / 0.5)))
        res = minimize(wrapper_obj, h, jac=wrapper_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
