Actually, let me reconsider. The key bottleneck is: given a binary h and the current g vector, find the swap (i,j) that maximally reduces max(g). This is hard to do exactly, but we can approximate.

**Approach: For each top shift d*, find the best swap that reduces g[d*]**

For shift d* (at index idx in the g array), the change in g[idx] from swap (i: 1→0, j: 0→1) is:
Δg[idx] = (h[j+d*] + h[j-d*] - h[i+d*] - h[i-d*]) * dx

where d* = dm[idx].

To minimize this (make it most negative), we want:
- h[j+d*] + h[j-d*] to be 0 (j such that j±d* are both 0 or OOB)
- h[i+d*] + h[i-d*] to be 2 (i such that i±d* are both 1)

So the algorithm for a given shift d*:
1. Find all i with h[i]=1 and h[i+d*]=1 and h[i-d*]=1 → these give Δg = -2*dx (best)
2. Find all j with h[j]=0 and h[j+d*]=0 and h[j-d*]=0 → these give the best pairing
3. Any pair (i from set 1, j from set 2) gives Δg[idx] = -2*dx

If no such perfect pair exists, we look for the next best:

```python
def compute_grad_matrix(h, n):
    """Return G of shape (2n-1, n) where G[d,j] = dg_d/dh[j]"""
    dm = get_dm(n)  # shape (2n-1,)
    j = np.arange(n)  # shape (n,)
    # For each d and j:
    # G[d,j] = (1-h[j+d])*[0<=j+d<n] - h[j-d]*[0<=j-d<n]
    # j+d has shape (2n-1, n)
    jd = j[None, :] + dm[:, None]  # (2n-1, n)
    jm = j[None, :] - dm[:, None]  # (2n-1, n)
    
    G = np.zeros((2*n-1, n))
    mask_p = (jd >= 0) & (jd < n)
    G[mask_p] = 1.0 - h[jd[mask_p]]
    mask_m = (jm >= 0) & (jm < n)
    G[mask_m] -= h[jm[mask_m]]
    
    return G * (2.0/n)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def get_dm(n):
    return np.arange(2 * n - 1) - (n - 1)

def swap_delta_g(h, dm, n, i, j, dx):
    size = 2 * n - 1
    delta = np.zeros(size)
    jd = j + dm
    jd2 = j - dm
    id_ = i + dm
    id2 = i - dm
    m = (jd >= 0) & (jd < n)
    delta[m] += h[jd[m]]
    m = (jd2 >= 0) & (jd2 < n)
    delta[m] += h[jd2[m]]
    m = (id_ >= 0) & (id_ < n)
    delta[m] -= h[id_[m]]
    m = (id2 >= 0) & (id2 < n)
    delta[m] -= h[id2[m]]
    return delta * dx

def slsqp_optimize(h, n, alpha, maxiter=200, ftol=1e-12):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_binary(h_init, n, deadline, seed=0):
    """Simulated annealing on binary solutions with smart moves"""
    rng = np.random.RandomState(seed)
    dm = get_dm(n)
    dx = 2.0 / n
    
    h = h_init.copy()
    g = compute_g(h, n)
    c = g.max()
    best_h = h.copy()
    best_c = c
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Temperature schedule
    T_start = 0.03
    T_end = 0.0003
    max_iters = 50000
    cool = (T_end / T_start) ** (1.0 / max_iters)
    
    T = T_start
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose swap: biased towards positions that affect worst shifts
        top_k = 4
        top_shifts = np.argsort(g)[::-1][:top_k]
        
        # With probability 0.7, use biased move; else random
        if rng.random() < 0.7:
            # Pick a random top shift
            si = rng.randint(top_k)
            d = dm[top_shifts[si]]
            
            # Find i (h[i]=1) that maximizes h[i+d]+h[i-d] (removing it helps most)
            id_p = ones + d
            id_m = ones - d
            mask_p = (id_p >= 0) & (id_p < n)
            mask_m = (id_m >= 0) & (id_m < n)
            score_i = np.zeros(len(ones))
            score_i[mask_p] += h[id_p[mask_p]]
            score_i[mask_m] += h[id_m[mask_m]]
            
            # Find j (h[j]=0) that minimizes h[j+d]+h[j-d] (adding it costs least)
            jd_p = zeros + d
            jd_m = zeros - d
            mask_jp = (jd_p >= 0) & (jd_p < n)
            mask_jm = (jd_m >= 0) & (jd_m < n)
            score_j = np.zeros(len(zeros))
            score_j[mask_jp] += h[jd_p[mask_jp]]
            score_j[mask_jm] += h[jd_m[mask_jm]]
            
            # Sample from top candidates
            # i: prefer high score
            i_scores = score_i / max(score_i.max(), 1e-10)
            i_probs = i_scores / i_scores.sum()
            i_idx = rng.choice(len(ones), p=i_probs)
            i = ones[i_idx]
            
            # j: prefer low score
            j_scores = 2.0 - score_j  # invert
            j_probs = j_scores / j_scores.sum()
            j_idx = rng.choice(len(zeros), p=j_probs)
            j = zeros[j_idx]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        dg = swap_delta_g(h, dm, n, i, j, dx)
        new_g = g + dg
        new_c = float(new_g.max())
        delta_c = new_c - c
        
        if delta_c < -1e-15 or rng.random() < np.exp(-delta_c / T):
            # Accept
            h[i], h[j] = h[j], h[i]
            # Update ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            g = new_g
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stall = 0
            T *= cool
        else:
            T *= cool
    
    return best_h, best_c

def pair_descent(h, n, deadline, max_sweeps=300):
    """Fast pair descent for continuous h"""
    dx = 2.0 / n
    dm = get_dm(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        K = 6
        top_idx = np.argsort(g)[::-1][:K]
        
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-14:
                    continue
                # Determine direction
                if h[i] > h[j]:
                    # Move from i to j
                    d_lo = 0.0
                    d_hi = h[i] - h[j]
                else:
                    d_lo = h[i] - h[j]
                    d_hi = 0.0
                
                if abs(d_hi - d_lo) < 1e-14:
                    continue
                
                # Effect on g: delta_g[k] = (h[j]-h[i]) direction... 
                # Actually: if we change h[i] by +δ and h[j] by -δ:
                # Δg_d = δ * ((1-h[i+d]) - h[i-d] + h[j+d] - (1-h[j+d]) ... )
                # Wait, let me be more careful.
                # g_d = Σ_l h[l](1-h[l+d])
                # ∂g_d/∂h[i] = (1-h[i+d])*[valid] - h[i-d]*[valid]
                # ∂g_d/∂h[j] = (1-h[j+d])*[valid] - h[j-d]*[valid]
                # Δg_d = δ * (∂g_d/∂h[i] - ∂g_d/∂h[j])
                
                # Precompute these
                iid_p = i + dm
                iid_m = i - dm
                jjd_p = j + dm
                jjd_m = j - dm
                
                mil = (iid_p >= 0) & (iid_p < n)
                mij = (iid_m >= 0) & (iid_m < n)
                mjil = (jjd_p >= 0) & (jjd_p < n)
                mjj = (jjd_m >= 0) & (jjd_m < n)
                
                s = np.zeros(2*n-1)
                s[mil] = 1.0 - h[iid_p[mil]]
                s[mij] -= h[iid_m[mij]]
                s[mjil] -= (1.0 - h[jjd_p[mjil]])
                s[mjj] += h[jjd_m[mjj]]
                
                # Find best delta
                candidates = [d_lo, d_hi]
                for k in range(K):
                    for l in range(K):
                        if top_idx[k] == top_idx[l]:
                            continue
                        ds = s[top_idx[l]] - s[top_idx[k]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[k]] - g[top_idx[l]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = d
                
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved = True
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    
    return best_h, best_c
