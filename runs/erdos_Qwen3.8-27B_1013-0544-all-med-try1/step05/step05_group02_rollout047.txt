Actually, here's another idea: **AVM (Away Variable Method)** for the subgradient of the max function. This is a well-known method for minimizing the max of smooth functions:

1. Maintain a "centroid" point x_avg
2. At each step, compute the subgradient of f at x_avg (which is the gradient of the function achieving the max)
3. Take a step in the negative subgradient direction
4. Project onto the feasible set
5. Update the centroid

This can be much more effective than simple gradient descent for max problems.

def avm_optimize(h_start, n, deadline, max_iter=100000):
    """Away Variable Method for min max problem."""
    x = project_feasible(h_start.copy(), n)
    x_avg = x.copy()  # Running average (centroid)
    best_h = x.copy()
    best_c = compute_c5(x, n)
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        
        # Evaluate f at x_avg
        c_avg = get_corr(x_avg, n)
        k_star = int(np.argmax(c_avg))
        
        # Subgradient at x_avg for the active constraint
        s = k_star - (n - 1)
        g = np.zeros(n)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        idx = np.arange(lo, hi + 1)
        g[idx] += (1.0 - x_avg[idx + s])
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= x_avg[idx2 - s]
        
        # AVM step
        eta = 2.0 / (t + 1)  # Learning rate
        
        # Away step
        x_away = x - eta * g
        x_away = project_feasible(x_away, n)
        
        # Regular step
        x_reg = x - eta * g
        x_reg = project_feasible(x_reg, n)
        
        # Choose the better of away and regular
        c_away = compute_c5(x_away, n)
        c_reg = compute_c5(x_reg, n)
        
        if c_away <= c_reg:
            x = x_away
        else:
            x = x_reg
        
        # Update running average
        x_avg = (t * x_avg + x) / (t + 1)
        
        # Update best
        c_best_avg = compute_c5(x_avg, n)
        if c_best_avg < best_c:
            best_c = c_best_avg
            best_h = project_feasible(x_avg, n)
        
        # Also check current x
        c_x = compute_c5(x, n)
        if c_x < best_c:
            best_c = c_x
            best_h = x.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def worst_lag_grad(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

def directed_polish(h, n, deadline, max_iter=500000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(h, n)
    step = 0.02
    min_step = 5e-7
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        k_star = int(np.argmax(c))
        g = worst_lag_grad(h, n, k_star)
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = min(step, h[i_src] * 0.95, (1.0 - h[i_dst]) * 0.95)
        if d < min_step:
            step *= 0.5
            if step < min_step:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step = min(step * 1.4, 0.05)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 40:
                step *= 0.5
                stall = 0
                if step < min_step:
                    break
    return best_h, best_c

def multi_lag_polish(h, n, deadline, num_lags=4, max_iter=200000):

def subgrad_descent(h, n, deadline, max_iter=50000):
    x = project_feasible(h.copy(), n)
    x_avg = x.copy()
    best_h = x.copy()
    best_c = compute_c5(x, n)
    for t in range(1, max_iter + 1):

            break
        c_avg = get_corr(x_avg, n)
        k_star = int(np.argmax(c_avg))
        g = worst_lag_grad(x_avg, n, k_star)
        eta = 1.0 / (t ** 0.5 + 1.0) * 0.1
        x_new = x - eta * g
        x_new = project_feasible(x_new, n)
        x = x_new
        x_avg = (t * x_avg + x) / (t + 1)
        c_avg_now = compute_c5(x_avg, n)
        if c_avg_now < best_c:
            best_c = c_avg_now
            best_h = project_feasible(x_avg, n)
        c_x = compute_c5(x, n)
        if c_x < best_c:
            best_c = c_x
            best_h = x.copy()
    return best_h, best_c
