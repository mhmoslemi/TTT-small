def subgradient_step(h, n, step_size):
    """One subgradient step: move mass from high-cost to low-cost positions."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    m = n - 1  # offset for 'full' mode
    
    # Subgradient of c[k_star] w.r.t. h[i]:
    # dc/dh[i] = (1 - h[i+k_star-m]) - h[i-k_star+m] (with bounds checking)
    g = np.zeros(n)
    for i in range(n):
        # First term: (1 - h[i+k_star-m]) if i+k_star-m in [0,n-1]
        ip = i + k_star - m
        if 0 <= ip < n:
            g[i] += (1.0 - h[ip])
        # Second term: -h[i-k_star+m] if i-k_star+m in [0,n-1]
        im = i - k_star + m
        if 0 <= im < n:
            g[i] -= h[im]
    
    # We want to decrease c[k_star], so move in -g direction
    # But we need to maintain sum(h) = n/2

    c_old = compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_direction(h, n, k_star, m):
    g = np.zeros(n)
    ip = np.arange(n) + k_star - m
    valid_p = (ip >= 0) & (ip < n)
    g[valid_p] = (1.0 - h[ip[valid_p]])
    im = np.arange(n) - k_star + m
    valid_m = (im >= 0) & (im < n)
    g[valid_m] -= h[im[valid_m]]
    return g

def guided_local_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    m = n - 1
    # Subgradient descent phase
    for ss in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        for _ in range(200):
            if time.time() > deadline:
                return h, best_c
            c_raw = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            k_star = int(np.argmax(c_raw))
            g = subgradient_direction(h, n, k_star, m)
            # Find best single move: increase where g<0, decrease where g>0
            pos_mask = g < -0.01
            neg_mask = g > 0.01
            best_delta = 0.0
            if pos_mask.any() and neg_mask.any():
                # Pick best i (most negative g) and best j (most positive g)
                i_candidates = np.where(pos_mask)[0]
                j_candidates = np.where(neg_mask)[0]
                i_best = i_candidates[np.argmin(g[i_candidates])]
                j_best = j_candidates[np.argmax(g[j_candidates])]
                # Try a few step sizes
                for d in [ss, ss * 2, ss * 0.5]:
                    if h[i_best] + d <= 1.0 and h[j_best] - d >= 0.0:
                        save_i, save_j = h[i_best], h[j_best]
                        h[i_best] += d
                        h[j_best] -= d
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_delta = d
                        else:
                            h[i_best], h[j_best] = save_i, save_j
                    elif h[i_best] - d >= 0.0 and h[j_best] + d <= 1.0:
                        save_i, save_j = h[i_best], h[j_best]
                        h[i_best] -= d
                        h[j_best] += d
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_delta = d
                        else:
                            h[i_best], h[j_best] = save_i, save_j
            if best_delta == 0.0:
                break
    return h, best_c

def full_pair_search(h, n, deadline, eps_list):

    for eps in eps_list:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + eps <= 1.0 and h[j] - eps >= 0.0:
                        h[i] += eps; h[j] -= eps
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= eps; h[j] += eps
                    if h[i] - eps >= 0.0 and h[j] + eps <= 1.0:

                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += eps; h[j] -= eps
