import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, deadline, rng, n_restarts=10):
    """Simulated annealing with single-swap moves on binary sequences."""
    target = n // 2
    best_h = np.zeros(n)
    best_c = np.inf

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            idx = np.argsort(-h_init)
            h_cur = np.zeros(n)
            h_cur[idx[:target]] = 1.0
        else:
            # Structured random init: spiral-like or random
            if restart % 3 == 1:
                # Block-structured: alternating blocks
                h_cur = np.zeros(n)
                block = max(1, n // (2 * (restart + 1)))
                for i in range(0, n, 2 * block):
                    h_cur[i:i+block] = 1.0
                # Fix count
                cur_count = h_cur.sum()
                if cur_count > target:
                    excess = int(cur_count - target)
                    one_pos = np.where(h_cur == 1)[0]
                    h_cur[one_pos[:excess]] = 0
                elif cur_count < target:
                    deficit = int(target - cur_count)
                    zero_pos = np.where(h_cur == 0)[0]
                    h_cur[zero_pos[:deficit]] = 1
            else:
                idx = rng.permutation(n)
                h_cur = np.zeros(n)
                h_cur[idx[:target]] = 1.0

        cur_c = compute_c5(h_cur, n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_cur.copy()

        temp = 0.05
        cool_rate = 0.99995
        no_improve = 0
        max_no_improve = 5000

        while time.time() < deadline and temp > 1e-7:
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h_cur[i], h_cur[j] = 1.0, 0.0
            c_new = compute_c5(h_cur, n)
            delta = c_new - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_c = c_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h_cur[i], h_cur[j] = 0.0, 1.0
                no_improve += 1
            temp *= cool_rate
            if no_improve > max_no_improve:
                break
        # Restart with cooling reset if time remains
    return best_h, best_c

def block_swap_sa(h_init, n, deadline, rng, n_restarts=5):

            if no_imp > max_no_imp:

    return best_h, best_c

def continuous_optimize(h_init, n, alpha, maxiter, deadline):
    """L-BFGS-B with smooth max approximation."""
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        val = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
        val += 1e6 * (x.sum() - n / 2.0) ** 2
        return val

    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(corr))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e6 * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def slsqp_optimize(h_init, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_init
