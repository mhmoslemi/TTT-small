            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-12:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 4: Alternating subgradient + SLSQP with small step
    for outer in range(8):
        if eligible() < 35:
            break
        step = 0.008
        v = np.zeros(n)
        for it in range(500):
            if eligible() < 10:
                break
            grad, c_arr = average_top_gradient(best_h, n, top_k=4)
            v = 0.8 * v + grad
            h_new = best_h - step * v
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c5 - 1e-11:
                best_h = h_new
                best_c5 = c_new
                step = min(step * 1.03, 0.04)
            else:
                step = max(step * 0.97, 0.001)
        if eligible() > 15:
            try:
                res = minimize(obj_lse, best_h, args=(n, 10000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-16})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-12:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 5: Restart with perturbation
    if eligible() > 50:
        scales = [0.02, 0.05, 0.1, 0.03, 0.08, 0.15]
        for s in scales:
            if eligible() < 35:
                break
            h_pert = best_h + rng.normal(0, s, size=n)
            h_pert = project_feasible(h_pert, n)
            try:
                res = minimize(obj_lse, h_pert, args=(n, 2000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-14})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-12:
                    best_h, best_c5 = h_try, c
                    if eligible() > 20:
                        res2 = minimize(obj_lse, best_h, args=(n, 20000), method='SLSQP',
                                       bounds=bounds, constraints=cons,
                                       options={'maxiter': 3000, 'ftol': 1e-16})
                        h_try2 = project_feasible(res2.x, n)
                        c2 = compute_c5(h_try2, n)
                        if c2 < best_c5 - 1e-12:
                            best_h, best_c5 = h_try2, c2
            except Exception:
                pass

    # Phase 6: Fine coordinate descent
    if eligible() > 20:
        deltas = [0.001, 0.002, 0.005, 0.003, -0.001, -0.002, -0.005, -0.003]
        count = 0
        while eligible() > 10 and count < 50000:
            count += 1
            i = rng.randint(n)
            for d in deltas:
                val = best_h[i] + d
                if 0.0 <= val <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = val
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c5 - 1e-13:
                        best_h = h_try
                        best_c5 = c_try
                        break

    # Final high-alpha SLSQP
    if eligible() > 5:
        try:
            res = minimize(obj_lse, best_h, args=(n, 100000), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
