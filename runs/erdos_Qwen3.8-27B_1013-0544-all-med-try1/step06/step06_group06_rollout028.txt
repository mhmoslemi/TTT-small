            break
        h_cur = upscale_func(best_h, best_n, n_try)
        h_cur = project_feasible(h_cur, n_try)
        c_cur = compute_c5(h_cur, n_try)

        # Scale maxit down for larger n
        if n_try <= 100:
            maxit_sched = [5000, 5000, 5000, 4000, 4000]
        elif n_try <= 120:
            maxit_sched = [4000, 4000, 3000, 3000, 2000]
        elif n_try <= 160:
            maxit_sched = [3000, 3000, 2000, 2000, 1000]
        else:
            maxit_sched = [2000, 2000, 1500, 1000, 1000]

        alphas2 = [50, 200, 1000, 5000, 20000]
        for idx, alpha in enumerate(alphas2):
            if budget_s - (time.time() - t0) < 40:
                break
            h_new = slsqp_optimize(h_cur, n_try, alpha, maxit_sched[idx])
            c_new = compute_c5(h_new, n_try)
            h_cur = h_new
            if c_new < c_cur:
                c_cur = c_new

        # Local search at this resolution
        if budget_s - (time.time() - t0) > 50:
            h_ls, c_ls = local_search(h_cur, n_try, rng, min(70, budget_s - (time.time() - t0) - 40), c_cur)
            if c_ls < c_cur:
                h_cur, c_cur = h_ls, c_ls

        if c_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c_cur
            best_n = n_try

    # Phase 3: Final polish at best resolution
    if budget_s - (time.time() - t0) > 25:
        h_cur = project_feasible(best_h, best_n)
        maxit_final = 5000 if best_n <= 100 else 3000
        for alpha in [20000, 100000]:
            if budget_s - (time.time() - t0) < 12:
                break
            h_new = slsqp_optimize(h_cur, best_n, alpha, maxit_final)
            c = compute_c5(h_new, best_n)
            h_cur = h_new
            if c < best_c5:
                best_h, best_c5 = h_new, c

    # Final local search
    if budget_s - (time.time() - t0) > 15:
        h_ls, c_ls = local_search(best_h, best_n, rng, min(budget_s - (time.time() - t0) - 5, 80), best_c5)
        if c_ls < best_c5:
            best_h, best_c5 = h_ls, c_ls

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
