def projection_to_feasible(y, n):
    """Project y onto [0,1]^n ∩ {sum = n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo = y.min() - target  # ensures sum(clip(y-lo, 0, 1)) >= target
    hi = y.max() - (n - target)  # ensures sum(clip(y-hi, 0, 1)) <= target
    # Actually, let's think more carefully
    # We want sum(clip(y_i - lambda, 0, 1)) = target
    # As lambda decreases, clip(y_i - lambda) increases
    # So we want to find lambda where the sum equals target

    for _ in range(100):

        if s > target:
            lo = mid
        else:
            hi = mid

    n = len(h)
    target = n // 2
    # Find the 'target' smallest g values
    threshold = np.partition(g, target)[target]
    s = (g <= threshold).astype(float)
    # Handle ties at the boundary
    if s.sum() > target:
        # Need to reduce some
        equal_mask = (g == threshold)
        n_equal = equal_mask.sum()
        n_excess = s.sum() - target
        # Set 'n_excess' of the equal ones to 0
        equal_indices = np.where(equal_mask)[0]
        to_zero = equal_indices[:n_excess]
        s[to_zero] = 0
    elif s.sum() < target:
        equal_mask = (g == threshold)
        equal_indices = np.where(equal_mask)[0]
        n_needed = target - s.sum()
        s[equal_indices[:n_needed]] = 1

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            return h
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    else:
        s2 = -s
        if s2 < n:
            g[s2:] += (1.0 - h[:n-s2])
            g[:n-s2] -= h[s2:]
    return g

def weighted_subgrad(h, top_k=10, sharpness=5.0):
    n = len(h)
    corr = compute_corr(h)
    K = min(top_k, 2*n-1)
    idx = np.argpartition(-corr, K-1)[:K]
    w = np.exp(sharpness * (corr[idx] - corr.max()))

        g += w[i] * grad_for_shift(h, s)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            return h
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, temp):
    """Compute smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(temp * (corr - m))
    w_sum = w.sum()
    w /= w_sum
    val = (m + np.log(w_sum) / temp) * dx
    
    # Fast gradient
    A = np.correlate(w, 1.0 - h, mode='valid')[::-1]
    B = np.correlate(w, h[::-1], mode='valid')
    grad = (A - B) * dx
    return val, grad

def proximal_project(h, grad, step):
    """Take a proximal gradient step and project onto feasible set."""
    n = len(h)
    target = n / 2.0
    y = h - step * grad
    # Project y onto {x: 0<=x_i<=1, sum(x)=target}
    # Using bisection on lambda
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def nesterov_subgrad(h, n_steps=2000, temp=30.0, lr=0.1, mu=0.9):
    """Nesterov accelerated subgradient method."""
    n = len(h)
    h_curr = h.copy()

    y = h.copy()  # Nesterov lookahead
    v = np.zeros(n)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()

        # Nesterov step
        y = h_curr + mu * (h_curr - h_prev)
        val, grad = smooth_max_and_grad(y, temp)
        
        # Step
        h_new = proximal_project(y, grad, lr)
        
        c5 = compute_c5(h_new)

            best_c5 = c5
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
    
    return best_h, best_c5

def coord_descent(h, n_levels=10, max_sweeps=5):

            else:

        if not improved:

z(t) = exp(-π t(t+1)/N) for t = 0, ..., N-1

Then take h[i] = 0.5 + 0.5 * Re(z(t)) for the real part, clipped to [0,1].

Actually, for a real-valued sequence, a good low-autocorrelation sequence is the Barker sequence (for N=11) or the m-sequence (for N=2^m-1).

For our problem (n=384), we can use a periodic extension of a short low-autocorrelation sequence.

- Map -1 → 0, 1 → 1 (or a smoothed version)
- Tile it to get length n
- Adjust to satisfy the sum constraint

import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            return h
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.exp(temp * (corr - m)).sum()
    return (m + np.log(s) / temp) * dx

def smooth_max_matrix(h, temp):
    """Return value and gradient of smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(temp * (corr - m))
    ws = w.sum()
    w /= ws
    val = (m + np.log(ws) / temp) * dx
    # Gradient via correlate
    A = np.correlate(w, 1.0 - h, mode='valid')[::-1]
    B = np.correlate(w, h[::-1], mode='valid')
    grad = (A - B) * dx
    return val, grad

def proximal_project(h, grad, step):
    n = len(h)
    target = n / 2.0
    y = h - step * grad
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def nesterov_opt(h, n_steps, temp, lr, mu=0.9):
    """Nesterov accelerated projected gradient."""
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    
    for it in range(n_steps):
        y = h_curr + mu * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)  # keep y in bounds for gradient computation
        val, grad = smooth_max_matrix(y, temp)
        h_new = proximal_project(h_curr, grad, lr)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h_curr
        h_curr = h_new
        # Adaptive lr: reduce if oscillating
        if it > 100 and it % 200 == 0:
            lr *= 0.85
    return best_h, best_c5

def slsqp_opt(h, temp, maxiter=300):
    n = len(h)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, temp)
    
    def obj_grad(x):
        _, g = smooth_max_matrix(x, temp)
        return g
    
    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                   bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8, 'maxls': 50})
    return project_feasible(res.x, ts)

def coord_descent(h, n_levels=10, max_sweeps=5):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_c < c5 - 1e-14:
                c5 = best_c
                improved = True
            else:
                h[j] = orig
                h = project_feasible(h, ts)
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: L-BFGS-B with smooth max (multiple temperatures)
    for temp in [40.0, 15.0, 7.0, 3.0]:
        if rem() < 120:
            break
        h_opt = slsqp_opt(best_h, temp, maxiter=200)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt
    
    # Phase 2: Nesterov subgradient
    for temp in [20.0, 8.0]:
        if rem() < 100:
            break
        h_opt, c = nesterov_opt(best_h, n_steps=1000, temp=temp, lr=0.02, mu=0.85)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt
    
    # Phase 3: Coordinate descent
    if rem() > 60:
        h_cd, c = coord_descent(best_h, n_levels=12, max_sweeps=8)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd
    
    # Phase 4: Random restarts + optimize
    n_restarts = 0
    while rem() > 150 and n_restarts < 15:
        n_restarts += 1
        scale = 0.12 * max(0.2, 1.0 - n_restarts / 20.0)
        h_r = best_h + np.random.randn(n0) * scale
        h_r = project_feasible(h_r, n0 / 2.0)
        # Quick L-BFGS
        h_opt = slsqp_opt(h_r, temp=10.0, maxiter=150)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt
            if rem() > 40:
                h_cd, c2 = coord_descent(best_h, n_levels=10, max_sweeps=5)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cd
    
    # Phase 5: Multi-resolution
    for n_try in [256, 512, 192]:
        if rem() < 100:
            break
        x_old = np.arange(len(best_h)) / len(best_h)
        x_new = np.arange(n_try) / n_try
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try / 2.0)
        for temp in [15.0, 5.0]:
            if rem() < 60:
                break
            h_new = slsqp_opt(h_new, temp, maxiter=200)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
        if rem() > 30:
            h_cd, c = coord_descent(best_h, n_levels=10, max_sweeps=5)
            if c < best_c5:
                best_c5 = c
                best_h = h_cd
    
    # Phase 6: Final polish
    if rem() > 15:
        h_cd, c = coord_descent(best_h, n_levels=20, max_sweeps=10)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd
    
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

Wait, I'm using L-BFGS-B with the smooth max objective. This should work well because:
1. L-BFGS-B handles bound constraints efficiently
2. The smooth max provides a differentiable surrogate
3. The constraint sum(h) = n/2 is handled by the projection after each call

But wait, L-BFGS-B doesn't handle equality constraints! I need to handle the sum constraint differently.

Options:
1. Use SLSQP (handles both inequality and equality constraints) - slower
2. Use L-BFGS-B and project after each call - might not work well because the optimizer doesn't know about the constraint
3. Reparameterize: use n-1 free variables and determine the last from the constraint
4. Use a penalty method with L-BFGS-B

Option 3 is interesting: set h[n-1] = n/2 - sum(h[0:n-1]), and optimize over h[0:n-1] with bounds [0,1]. The issue is that h[n-1] might go out of [0,1].

Option 4: Add a large penalty (sum(h) - n/2)² to the objective. This works well with L-BFGS-B.

Let me use option 4:
