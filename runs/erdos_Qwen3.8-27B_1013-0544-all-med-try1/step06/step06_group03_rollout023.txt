import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    return t * np.log(np.sum(np.exp((conv - m) / t))) * dx

def softmax_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    s = (conv - m) / t
    mask = s > -12
    if not np.any(mask):
        mask = np.zeros(len(conv), dtype=bool)
        mask[int(np.argmax(conv))] = True
    e = np.exp(s[mask])
    w = e / e.sum()
    grad = np.zeros(n)
    idxs = np.where(mask)[0]
    for i in range(len(idxs)):
        shift = idxs[i] - (n - 1)
        grad += w[i] * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def lgb(h, n, t, iters=2000):
    def f(x):
        return softmax_obj(x, t) + 1000.0 * (np.sum(x) - n / 2.0) ** 2
    def g(x):
        return softmax_grad(x, t) + 2000.0 * (np.sum(x) - n / 2.0)
    try:
        r = minimize(f, h, jac=g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def pgd_subgrad(h, iters, lr=0.001):
    n = len(h)
    h = h.copy()
    bc5 = compute_c5(h)
    bh = h.copy()
    for it in range(iters):
        c5, grad = c5_subgrad(h)
        gn = np.dot(grad, grad)
        if gn < 1e-24:
            break
        a = lr
        ct = c5
        ht = h
        for _ in range(15):
            ht = project_feasible(h - a * grad, n)
            ct = compute_c5(ht)
            if ct <= c5 - 1e-14 * a * gn:
                break
            a *= 0.5
            if a < 1e-16:
                break
        h = ht
        if ct < bc5:
            bc5, bh = ct, h.copy()
        lr = min(a * 1.2, 0.01)
        if a < 1e-16:
            break
    return bh, bc5

def pgd_smooth(h, iters, lr=0.003, t=0.01):
    n = len(h)
    h = h.copy()
    bc5 = compute_c5(h)
    bh = h.copy()
    for it in range(iters):

        gn = np.dot(grad, grad)
        if gn < 1e-24:
            break

        ht = h

            ht = project_feasible(h - a * grad, n)
            ct = compute_c5(ht)
            if ct <= c5 - 1e-14 * a * gn:

                break
        h = ht
        if ct < bc5:
            bc5, bh = ct, h.copy()
        lr = min(a * 1.15, 0.01)
        if a < 1e-16:
            break
    return bh, bc5
