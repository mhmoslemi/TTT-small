def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

            else:

        else:

            grad[p:n] += w * dx * (1.0 - h[0:n-p])

                v = np.zeros(n)

        alpha *= decay

The proximal operator of C₅ at point h with parameter μ is:
prox(h) = argmin_{h'} C₅(h') + (1/2μ)||h' - h||²

This is a regularized version that's easier to solve (the objective is now the max of bilinear functions plus a quadratic). But it's still non-convex.

Alternatively, I can use a "minorize-maximize" (MM) approach:
- At each iteration, construct a quadratic lower bound (minorant) of C₅ around the current point
- Maximize the minorant (which is easier)

This is like expectation-maximization but for optimization.

For the max function, a natural minorant is:
L(h, h_old) = max_k [g_k(h_old) + ∇g_k(h_old)^T (h - h_old)]

where the max is over active k's. This is the linearization of the max at the current point.

Then the MM step is:
h_new = argmin_{h'} L(h', h) + (1/2μ)||h' - h||²
= argmin_{h'} max_k [g_k(h) + ∇g_k(h)^T (h' - h)] + (1/2μ)||h' - h||²

The inner max makes this hard. But if we fix the active set, it becomes:
h_new = argmin_{h'} p^T h' + (1/2μ)||h' - h||²
where p = (1/|A|) Σ_{k∈A} ∇g_k(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr > 1e-12:
                h += room * (diff / tr)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

    n = len(h)

        top_indices = np.argsort(corr)[-top_k:]
        # Weight by closeness to max
        max_c = np.max(corr)
        min_top = np.min(corr[top_indices])
        weights = np.zeros(len(corr))
        for idx in top_indices:
            w = 1.0 if corr[idx] >= max_c - 1e-6 else (corr[idx] - min_top) / (max_c - min_top + 1e-12)
            weights[idx] = w
        weights /= np.sum(weights)
    else:

        shifted = np.clip(beta * (corr - max_c), -300, 0)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        # Zero out negligible weights
        weights[weights < 1e-4 / n] = 0
        weights /= np.sum(weights)

    grad = np.zeros(n)
