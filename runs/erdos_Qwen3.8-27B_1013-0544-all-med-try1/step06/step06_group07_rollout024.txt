    s = int(seed)

    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    if init is not None and init.size >= 10:
        n_base = int(init.size)
    else:
        n_base = 70

    n_list = [n_base]
    for extra in [n_base - 10, n_base + 10, n_base - 20, n_base + 20, 50, 100]:
        if 30 <= extra <= 130 and extra not in n_list:
            n_list.append(extra)
    n_list = n_list[:6]

    best_h = None
    best_c = 1e9

    for idx, n in enumerate(n_list):
        remaining = deadline - time.time()
        if remaining < 40:
            break

        if init is not None:
            base = resample(init, n)
        else:
            base = repair(np.full(n, 0.5))

        cb = exact_c(base)
        if cb < best_c:
            best_c = cb
            best_h = base.copy()

        time_per = min(remaining * 0.35, 180.0)
        n_starts = 1 + (1 if idx == 0 else 0)

        starts = [base]
        if n_starts > 1:
            starts.append(jitter(base, 0.03, s + n))

        for si, h0 in enumerate(starts):
            if time.time() > deadline - 30:
                break
            alloc = time_per / len(starts)
            if alloc < 5:
                break
            hr, cr = cutting_plane_optimize(h0, alloc, s + 100 * n + si)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()

        if time.time() < deadline - 40:
            h_base = best_h if best_h.size == n else base
            alloc2 = min(deadline - time.time() - 30, 120.0)
            if alloc2 > 10:
                h2, c2 = smooth_optimize(h_base, alloc2, s + 5000 + n)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()

    if best_h is not None and time.time() < deadline - 60:
        nb = best_h.size
        for trial in range(6):
            if time.time() > deadline - 15:
                break
            sc = 0.015 + 0.02 * trial
            ht = jitter(best_h, sc, s + 8000 + trial * 13)
            alloc = min((deadline - time.time() - 15) * 0.25, 80.0)
            if alloc < 5:
                break
            hr, cr = cutting_plane_optimize(ht, alloc, s + 12000 + trial)
            if cr < best_c:
                best_c = cr
                best_h = hr.copy()

    if best_h is None:
        best_h = repair(np.full(n_base, 0.5))
        best_c = exact_c(best_h)

    best_h = repair(best_h)
    best_c = exact_c(best_h)
    return best_h, best_c, int(best_h.size)
