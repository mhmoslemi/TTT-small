from scipy.ndimage import zoom

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def soft_max(values, temp):
    m = np.max(values)
    if temp < 1e-10:
        return m
    shifted = (values - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)))

def obj_slsqp(x, temp, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return soft_max(corr, temp)

def obj_lbfgsb(x, temp, w, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx

    return smax + w * (np.sum(x) - n / 2.0) ** 2 / n
