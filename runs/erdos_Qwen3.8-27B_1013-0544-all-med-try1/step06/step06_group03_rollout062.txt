        if s > target:

        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
    obj = (m + t * np.log(s)) * dx
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w > 1e-12:

                    grad[shift:] -= w * h[:mm]
            else:

                    grad[:mm] -= w * h[ss:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_obj_grad_fast(h, t, thresh=1e-7):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp(np.maximum((conv - m) / t, -50.0))
    s = np.sum(exps)
    obj = (m + t * np.log(s)) * dx
    weights = exps / s
    active_idx = np.where(weights > thresh)[0]
    grad = np.zeros(n)
    for idx in active_idx:
        w = weights[idx]
        shift = idx - (n - 1)
        if shift >= 0:
            mm = n - shift
            if mm > 0:
                grad[:mm] += w * (1.0 - h[shift:])
                grad[shift:] -= w * h[:mm]
        else:
            ss = -shift
            mm = n - ss
            if mm > 0:
                grad[ss:] += w * (1.0 - h[:mm])
                grad[:mm] -= w * h[ss:]
    return obj, grad * dx

def subgrad_obj(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    obj = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        mm = n - shift
        if mm > 0:
            grad[:mm] += (1.0 - h[shift:])
            grad[shift:] -= h[:mm]
    else:
        ss = -shift
        mm = n - ss
        if mm > 0:
            grad[ss:] += (1.0 - h[:mm])
            grad[:mm] -= h[ss:]
    return obj, grad * dx

def momentum_pgd(h0, max_iter, lr, mu, t_init, t_final, use_subgrad=False, binarize_penalty=0.0):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t = t_init
    for it in range(max_iter):
        frac = it / max(max_iter - 1, 1)
        t = t_init * (max(t_final, 1e-9) / t_init) ** frac if t_init > 0 else t_final
        if t < 1e-8:
            t = 1e-8
        if use_subgrad:
            obj, grad = subgrad_obj(h)
        else:
            obj, grad = soft_obj_grad_fast(h, t)
        if binarize_penalty > 0.0:
            grad = grad + binarize_penalty * (1.0 - 2.0 * h)
        h_new = h + mu * (h - h_prev) - lr * grad
        h_prev = h.copy()
        h = project_feasible(h_new, n)
        if it % 50 == 0 or it == max_iter - 1:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def firm_iter(h, t_val, n, max_lbfgs=3000):
    def _obj(hv):
        c, _ = soft_obj_grad_fast(hv, t_val)
        return c + 100.0 * (np.sum(hv) - n / 2.0) ** 2
    def _grad(hv):
        _, g = soft_obj_grad_fast(hv, t_val)
        return g + 200.0 * (np.sum(hv) - n / 2.0)
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_lbfgs, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def upsample(h, n_new):
    nc = len(h)
    xo = np.linspace(0, 2.0, nc, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    hi = np.interp(xn, xo, h)
    return project_feasible(hi, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(80, 0.5)

    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: Momentum PGD with geometric temperature decay
    schedules = [
        (0.04, 0.002, 0.008, 0.90),
        (0.015, 0.0005, 0.004, 0.92),
        (0.005, 0.0001, 0.002, 0.94),
        (0.001, 1e-5, 0.001, 0.96),
    ]
    for (ti, tf, lr, mu) in schedules:
        if tl() < 40:
            break
        ht, c5t = momentum_pgd(best_h, 12000, lr, mu, ti, tf)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        n = len(best_h)

    # Phase 2: L-BFGS-B with progressively tighter soft-max
    for t_val in [0.003, 0.001, 0.0003]:
        if tl() < 35:
            break
        n = len(best_h)
        ht = firm_iter(best_h, t_val, n, max_lbfgs=4000)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Subgradient polish (chase the exact max)
    if tl() > 40:
        ht, c5t = momentum_pgd(best_h, 8000, 0.0015, 0.85, 0.001, 1e-8, use_subgrad=True)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Multi-resolution upsample and re-optimize
    current_n = len(best_h)
    target_sizes = []
    for mult in [1.3, 1.6, 2.0]:
        ts = int(current_n * mult)
        if ts > 1200:
            ts = 1200
        if ts not in target_sizes and ts > current_n:
            target_sizes.append(ts)

    for n_new in target_sizes:
        if tl() < 120:
            break
        hi = upsample(best_h, n_new)

            if tl() < 60:

                best_h, best_c5 = ht, c5t
                hi = ht.copy()
        if tl() > 50:
            ht2 = firm_iter(best_h, 0.0008, len(best_h), max_lbfgs=2000)
            c5t2 = compute_c5(ht2)
            if c5t2 < best_c5:
                best_h, best_c5 = ht2, c5t2

    # Phase 5: Structured random restarts
    n = len(best_h)
    for restart in range(25):
        if tl() < 35:
            break
        x = np.linspace(0, 2, n, endpoint=False)
        choice = rng.randint(4)
        if choice == 0:
            freq = rng.randint(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            hr = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        elif choice == 1:
            hr = np.zeros(n)
            n_br = rng.randint(2, 6)
            brpts = np.sort(rng.choice(range(5, n - 5), size=n_br, replace=False))

                hr[s:e] = rng.uniform(0.15, 0.85)
        elif choice == 2:
            steps = rng.randint(3, 8)
            hr = np.tile(np.linspace(0.1, 0.9, steps), (n // steps + 1))[:n]
        else:
            hr = rng.uniform(0, 1, n)
            k = max(3, n // 30)
            kern = np.ones(k) / k
            hr = np.convolve(hr, kern, mode='same')
        hr = project_feasible(hr, n)
        ht, c5t = momentum_pgd(hr, 6000, 0.01, 0.9, 0.04, 0.001)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final aggressive polish

            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
