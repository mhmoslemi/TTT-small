Let me try a different angle: **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)** applied to the smooth max.

FISTA is an accelerated proximal gradient method. For our problem:
- Proximal operator: project onto {0≤h≤1, sum(h)=n/2}
- Gradient: gradient of the smooth max

The projection onto {0≤h≤1, sum(h)=n/2} is:
1. Solve for λ such that sum(clip(h - λ, 0, 1)) = n/2
2. This is a 1D root-finding problem (bisection)

This is much faster than the iterative projection I was using!

```python
def project_efficient(h, target_sum):
    """Efficient projection onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.asarray(h, dtype=float)
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

This is O(n * 100) per projection, which is much faster than the iterative approach.

When we change h[j] from old_val to new_val, how does the correlation change?

corr[k] = sum_i h[i+k](1-h[i])

The change in corr[k] due to changing h[j]:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_single(h, full_idx):
    n = len(h)
    k = full_idx - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[:n-k] -= h[k:]
        grad[k:] += 1.0 - h[:n-k]
    else:
        l = -k
        grad[l:] += 1.0 - h[:n-l]
        grad[:n-l] -= h[l:]
    return grad

def smooth_obj(h, T, active):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ac = corr[active]
    m = np.max(ac)
    exps = np.exp(T * (ac - m))
    return (m + np.log(exps.sum()) / T) * dx

def smooth_grad(h, T, active):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ac = corr[active]
    m = np.max(ac)
    exps = np.exp(T * (ac - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        grad += wi * grad_corr_single(h, idx)
    return grad * dx

def slsqp_active(h, T, K, maxiter=250):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = np.argsort(corr)[-K:].astype(int)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h, args=(T, active),
                   jac=lambda x: smooth_grad(x, T, active),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_feasible(res.x, target)

def fista_smooth(h_init, T, K, max_iter=500, target_sum=None):
    n = len(h_init)
    if target_sum is None:
        target_sum = n / 2.0
    h = h_init.copy()
    h_prev = h.copy()
    t = 1.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = np.argsort(corr)[-K:].astype(int)
    step = 1.0 / (T * n)
    best_obj = smooth_obj(h, T, active)
    for it in range(max_iter):
        if it % 80 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            active = np.argsort(corr)[-K:].astype(int)
        obj = smooth_obj(h, T, active)
        grad = smooth_grad(h, T, active)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y_grad = smooth_grad(y, T, active)
        step_try = step
        h_new = project_feasible(y - step_try * y_grad, target_sum)
        for _ in range(15):
            obj_new = smooth_obj(h_new, T, active)
            if obj_new <= obj + np.dot(grad, h_new - h) + (1.5 * step_try / (2.0 * t_new)) * np.dot(y_grad, h_new - y):
                break
            step_try *= 0.5
            h_new = project_feasible(y - step_try * y_grad, target_sum)
        step = step_try
        h_prev = h.copy()
        h = h_new
        t = t_new
    return h

def coord_descent(h, n_levels=7, max_sweeps=8):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    grid = np.linspace(0.0, 1.0, n_levels)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            saved = h[j]
            best_c = c5
            best_v = saved
            for v in grid:
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[j] = best_v
            h = project_feasible(h, ts)
            if best_c < c5 - 1e-15:
                c5 = best_c
                improved = True
            else:
                h[j] = saved
                h = project_feasible(h, ts)
        if not improved:
            break
    return h, c5
```
