import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, m):
    N = len(h)
    g = np.zeros(N)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[:N - m] += (1.0 - h[m:N])
        g[m:] -= h[:N - m]
    else:
        m = -m
        g[m:] += (1.0 - h[:N - m])
        g[:N - m] -= h[m:N]
    return g

def soft_subgradient(h, corr, N, alpha=20.0, top_k=10):
    top_idx = np.argsort(corr)[::-1][:top_k]
    c_top = corr[top_idx]
    w = np.exp(alpha * (c_top - c_top.max()))
    w /= w.sum()
    g = np.zeros(N)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-8:
            continue
        m = idx - (N - 1)
        g += wi * grad_lag(h, m)
    return g

def exact_subgradient(h, corr, N, tol=1e-6):
    active = np.where(corr >= corr.max() - tol)[0]
    g = np.zeros(N)
    for idx in active:
        m = idx - (N - 1)
        g += grad_lag(h, m)
    g /= len(active)
    return g

def interpolate_to_n(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def optimize_momentum(h_start, N, rng, t_end, state, alpha=20.0, max_iter=100000):
    h = h_start.copy()
    val, corr, dx = obj_and_corr(h)
    if val < state['best_val']:
        state['best_val'] = val
        state['best_h'] = h.copy()

    velocity = np.zeros(N)
    beta = 0.85
    lr = 0.025
    stagnant = 0
    best_local = val
    best_local_h = h.copy()

    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it % 3 == 0:
            g = soft_subgradient(h, corr, N, alpha=alpha, top_k=8) * dx
        else:
            g = exact_subgradient(h, corr, N) * dx

        gap = val - state['best_val']
        gnorm_sq = g @ g
        if gnorm_sq > 1e-12:
            polyak = max(gap / (gnorm_sq + 1e-12), 0.002)
            lr_eff = min(lr * 2.0, polyak * 0.8)
        else:
            lr_eff = lr

        velocity = beta * velocity - lr_eff * g
        h_new = project_feasible(h + velocity, N)
        val_new, corr_new, _ = obj_and_corr(h_new)

        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
            if val < state['best_val']:
                state['best_val'] = val
                state['best_h'] = h.copy()
        else:
            stagnant += 1
            h = h_new
            val = val_new
            corr = corr_new
            if stagnant > 35:
                scale = 0.04 + 0.08 * min(stagnant / 120.0, 1.0)
                h = project_feasible(state['best_h'] + rng.randn(N) * scale, N)
                val, corr, _ = obj_and_corr(h)
                velocity = np.zeros(N)
                stagnant = 0
                lr = 0.025

    if best_local < state['best_val']:
        state['best_val'] = best_local
        state['best_h'] = best_local_h.copy()
    return best_local_h, best_local

def snap_and_refine(h, N, rng, t_end, state):
    h_work = h.copy()
    h_work[h_work < 0.08] = 0.0
    h_work[h_work > 0.92] = 1.0
    h_work = project_feasible(h_work, N)
    val = obj_and_corr(h_work)[0]
    if val < state['best_val']:
        state['best_val'] = val
        state['best_h'] = h_work.copy()
    t_sub = min(t_end, time.time() + 40)
    return optimize_momentum(h_work, N, rng, t_sub, state, alpha=30.0, max_iter=30000)

def binary_start(N, rng):
    h = np.zeros(N)
    h[rng.choice(N, N // 2, replace=False)] = 1.0
    return h
