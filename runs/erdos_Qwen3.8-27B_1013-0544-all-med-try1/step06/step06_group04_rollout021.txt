import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def smooth_obj(x, n, beta=80.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / beta

def smooth_obj_grad(x, n, beta=80.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    w = np.exp(corr - m)
    w /= np.sum(w)
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[j] = dx * (np.dot(w[mask1], 1.0 - x[idx1[mask1]]) - np.dot(w[mask2], x[idx2[mask2]]))
    return grad

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    # Distribute the correction
    if abs(diff) < 1e-10:
        return h
    # Add/subtract uniformly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still off, fix iteratively
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase some values
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def upsample(h, factor=2):
    return np.repeat(h, factor)

def downsample(h, factor=2):
    n = len(h)
    new_n = n // factor
    return h.reshape(new_n, factor).mean(axis=1)

def optimize_h(h, n, time_limit, beta=80.0):
    t_start = time.time()
    target_sum = n / 2.0
    
    def obj(x):
        if time.time() - t_start > time_limit:
            return compute_c5(x)
        return smooth_obj(x, n, beta)
    
    def obj_grad(x):
        if time.time() - t_start > time_limit:
            return np.ones(n) / n
        return smooth_obj_grad(x, n, beta)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    try:
        result = minimize(obj, h, method='SLSQP', jac=obj_grad,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-12})
        h_out = result.x
    except Exception:
        h_out = h
    
    return fix_sum(h_out, n)

    # Round to binary

                    s += 1

            else:

        if not improved:

def local_search_fast(h_in, n, max_rounds=5000, trials_per_round=100, seed=42):
    rng_local = np.random.RandomState(seed)
    h = h_in.copy()
    # Binarize
    h = (h >= 0.5).astype(float)
    # Fix sum
    target = n // 2
    s = int(np.sum(h))
    # ... fix sum ...
    
    C = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    dx = 2.0 / n
    current_max = np.max(C) * dx
    lags = np.arange(-(n-1), n)
    
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(trials_per_round):
            i = ones[rng_local.randint(len(ones))]
            j = zeros[rng_local.randint(len(zeros))]
            # i: 1→0, j: 0→1, δ_i = -1, δ_j = +1
            # ΔC[lag] = -1*[(1-h[i-lag])*I - h[i+lag]*I] + 1*[(1-h[j-lag])*I - h[j+lag]*I]
            
            # First term (from h[i], δ=-1):
            # -(1-h[i-lag])*I(0≤i-lag<n) + h[i+lag]*I(0≤i+lag<n)
            # Second term (from h[j], δ=+1):
            # +(1-h[j-lag])*I(0≤j-lag<n) - h[j+lag]*I(0≤j+lag<n)
            
            # Compute in numpy:
            idx_i1 = i - lags  # i - lag
            mask_i1 = (idx_i1 >= 0) & (idx_i1 < n)
            idx_i2 = i + lags  # i + lag
            mask_i2 = (idx_i2 >= 0) & (idx_i2 < n)
            
            idx_j1 = j - lags
            mask_j1 = (idx_j1 >= 0) & (idx_j1 < n)
            idx_j2 = j + lags
            mask_j2 = (idx_j2 >= 0) & (idx_j2 < n)
            
            delta = np.zeros(2*n-1)
            delta[mask_i1] -= (1.0 - h[idx_i1[mask_i1]])
            delta[mask_i2] += h[idx_i2[mask_i2]]
            delta[mask_j1] += (1.0 - h[idx_j1[mask_j1]])
            delta[mask_j2] -= h[idx_j2[mask_j2]]
            
            new_C = C + delta
            new_max = np.max(new_C) * dx
            
            if new_max < current_max - 1e-12:
                # Accept
                h[i] = 0; h[j] = 1
                C = new_C
                current_max = new_max
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            # else: reject (h unchanged)
        
        if not improved:
            break
    
    return h, current_max

        tl = min(remaining * 0.30, 200)
        deadline_ls = start_time + budget_s * 0.85
        h_opt = optimize_slsqp(h_up, n_up, tl, beta=60.0)
        c5 = compute_c5(h_opt)
        elapsed2 = time.time() - start_time
        if c5 < best_c5 and budget_s - elapsed2 > 60:
            max_rounds = int(min((budget_s * 0.85 - time.time()) * 3, 4000))
            h_ls, c5_ls = local_search(h_opt, n_up, max_rounds=max_rounds,
                                       trials_per_round=150, seed=seed + 1,
                                       deadline=deadline_ls)
            if c5_ls < c5:
                c5 = c5_ls
                h_opt = h_ls
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_up
        print(f"Upsampled n={n_up}: C5={best_c5:.6f} ({time.time()-start_time:.0f}s)")

    # Phase 4: Final refinement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 40:
        tl = min(remaining * 0.4, 120)
        h_opt = optimize_slsqp(best_h, best_n, tl, beta=200.0)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        elapsed2 = time.time() - start_time
        remaining2 = budget_s - elapsed2
        if remaining2 > 30:
            deadline = start_time + budget_s - 5
            max_rounds = int(min(remaining2 * 3, 4000))
            h_ls, c5_ls = local_search(best_h, best_n, max_rounds=max_rounds,
                                       trials_per_round=200, seed=seed + 7,
                                       deadline=deadline)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Ensure constraints
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    print(f"FINAL: n={best_n}, C5={best_c5:.6f} ({time.time()-start_time:.0f}s)")
    return best_h, best_c5, best_n
