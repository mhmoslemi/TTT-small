import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(h.shape[0])

def slsqp_optimize(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

    t_start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            for j in range(i + 1, n):
                orig_i = h[i]
                orig_j = h[j]
                best_local_c5 = best_c5
                best_di = 0.0

                    for s in [1, -1]:

                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                            h[j] = hj
                            c = compute_c5(h, n)

                                best_di = di
                if best_di != 0.0:
                    h[i] = orig_i + best_di
                    h[j] = orig_j - best_di
                    best_c5 = best_local_c5
                    improved = True
                else:

                    h[j] = orig_j
