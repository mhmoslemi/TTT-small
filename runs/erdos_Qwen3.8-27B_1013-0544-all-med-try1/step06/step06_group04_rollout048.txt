import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr_vals, beta):
    """Approximate max using log-sum-exp"""
    m = np.max(corr_vals)
    if m < 1e-300:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (corr_vals - m)))) / beta

    sm = smooth_max(corr, beta)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(x, beta, n, penalty_weight):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    pen = penalty_weight * (np.sum(x) - n / 2.0) ** 2
    return sm + pen

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0

            idx = np.argsort(1.0 - h)[::-1]

            if step <= 0:
                continue
            h[i] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-12:

    return h
