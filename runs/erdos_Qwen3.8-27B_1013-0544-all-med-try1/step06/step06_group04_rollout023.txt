def compute_v(h):
    """Compute all overlap values v[k]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max_value_and_weights(v, t):
    """Compute smooth max and softmax weights."""
    v_max = v.max()
    exp_v = np.exp((v - v_max) / t)
    weights = exp_v / exp_v.sum()
    return v_max + t * np.log(exp_v.sum()), weights

def gradient_smooth_max(h, t=0.05):
    """Compute gradient of smooth max of v values."""
    n = len(h)
    dx = 2.0 / n
    v = compute_v(h)  # length 2n-1
    _, weights = smooth_max_value_and_weights(v, t)
    
    # v[k] corresponds to lag L = k - (n-1)
    # grad[m] = sum_k weights[k] * (2/n) * [(1-h[m+L]) * 1_{valid} - h[m-L] * 1_{valid}]
    
    grad = np.zeros(n)
    for m in range(n):
        for k in range(2*n-1):
            L = k - (n-1)
            w = weights[k]
            # First term: (1-h[m+L]) if 0 <= m+L < n
            if 0 <= m + L < n:
                grad[m] += w * dx * (1.0 - h[m + L])
            # Second term: -h[m-L] if 0 <= m-L < n
            if 0 <= m - L < n:
                grad[m] -= w * dx * h[m - L]
    
    return grad

r[k] = sum of h in a window related to position k.

Specifically, r[k] = sum_i h[i] * 1[i + k - (n-1)] = sum of h in a valid window.
r[k] = sum_{i = max(0, k-n+1)}^{min(n-1, k)} h[i]

And autocorrelation: a[k] = np.correlate(h, h, mode='full')[k] = sum_{i} h[i] * h[i + k - (n-1)] (for valid i)

So c[k] = r[k] - a[k]

This is helpful. The gradient of r[k] with respect to h[m] is 1 if m is in the window for k, 0 otherwise.
The gradient of a[k] with respect to h[m] is h[m + k - (n-1)] if valid + h[m - k + (n-1)] if valid.

    h = x  # already in [0,1] due to bounds
    n = len(h)

    v_max = v.max()

    # Penalty for sum constraint
    sm += penalty * (np.sum(h) - n/2.0)**2
    return sm
