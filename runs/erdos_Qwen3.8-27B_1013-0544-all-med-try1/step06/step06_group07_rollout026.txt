import numpy as np
import time

def project_simplex(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = -np.max(y) - 2.0
    hi = 1.0 - np.min(y) + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y + lam, 0.0, 1.0)

def compute_c(h):
    """Compute all c values (overlap integrals)."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lags(h, lags, dx):
    """Compute gradient of sum of c[lag] for given lags."""
    n = h.size
    grad = np.zeros(n)
    for lag in lags:
        # c[lag] = dx * sum_i h[i]*(1-h[i+lag]) for valid i
        # d/dh[i] = dx * [(1-h[i+lag]) if 0<=i+lag<n else 0] - dx * h[i-lag] if 0<=i-lag<n else 0
        # First term: i ranges over [0, n-1-lag] if lag>=0, or [0, n-1+lag] if lag<0
        if lag >= 0:
            # (1-h[i+lag]) term: i from 0 to n-1-lag
            i_idx = np.arange(0, n - lag)
            grad[i_idx] += (1.0 - h[i_idx + lag])
            # -h[i-lag] term: i from lag to n-1
            i_idx2 = np.arange(lag, n)
            grad[i_idx2] -= h[i_idx2 - lag]
        else:
            l = -lag
            # (1-h[i+lag]) = (1-h[i-l]): i from l to n-1
            i_idx = np.arange(l, n)
            grad[i_idx] += (1.0 - h[i_idx - l])
            # -h[i-lag] = -h[i+l]: i from 0 to n-1-l
            i_idx2 = np.arange(0, n - l)
            grad[i_idx2] -= h[i_idx2 + l]
    return grad * dx

def optimize_subgrad(h0, time_limit, seed, topk=3):
    """Subgradient optimization with backtracking and momentum."""
    rng = np.random.RandomState(seed)
    h = project_simplex(h0.copy(), h0.size)
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(np.max(compute_c(h)))
    
    deadline = time.time() + time_limit
    step = 0.1
    momentum = np.zeros(n)
    beta = 0.7
    no_improve = 0
    max_no_improve = 30
    
    while time.time() < deadline:
        c = compute_c(h)
        C = float(np.max(c))
        
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
            no_improve = 0
        
        # Identify active lags (center is n-1 in full correlation)
        center = n - 1
        # Convert to lag indices: output index = center + lag
        # c[lag] = c_array[center + lag]
        # Active set: top 'topk' values
        top_indices = np.argsort(c)[::-1][:topk]
        active_lags = top_indices - center
        
        # Compute gradient
        g = grad_for_lags(h, active_lags, dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * g
        direction = -momentum
        
        # Backtracking line search
        accepted = False
        alpha = step
        cur_f = C
        for _ in range(20):
            h_new = project_simplex(h + alpha * direction, n)
            c_new = compute_c(h_new)
            f_new = float(np.max(c_new))
            if f_new < cur_f - 1e-12:
                h = h_new
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if accepted:
            no_improve = 0
            step = min(alpha * 1.5, 0.5)
        else:
            no_improve += 1
        
        # Perturbation if stuck
        if no_improve >= max_no_improve:
            pert_scale = step * 0.3
            h = project_simplex(h + rng.normal(0, pert_scale, n), n)
            # Reset momentum
            momentum *= 0.5
            no_improve = 0
            step = max(0.01, step * 0.5)
        
        if time.time() < deadline and no_improve > max_no_improve * 3:
            # Try larger perturbation
            h = project_simplex(h + rng.normal(0, 0.05, n), n)
            momentum *= 0.3
            no_improve = 0

        if tl < 5:
            continue
        h, c = _subgrad(hr, tl, sv + nn, topk=3)
        _update(h, c)
        if time.time() < dl - 30:
            tl2 = min(25, (dl - time.time()) * 0.15)
            if tl2 > 4:
                h2, c2 = _slsqp(bh, tl2, sv + nn + 7, tau=0.002, topk=3,
                                maxiter=200)
                _update(h2, c2)

    # Final polish
    if time.time() < dl - 15:
        tl = min(60, (dl - time.time()) * 0.35)
        h, c = _subgrad(bh, tl, sv + 7777, topk=2)
        _update(h, c)
        if time.time() < dl - 8:
            tl2 = min(25, (dl - time.time()) * 0.3)
            if tl2 > 4:
                h2, c2 = _slsqp(bh, tl2, sv + 8888, tau=0.001, topk=3,
                                maxiter=200)
                _update(h2, c2)

    bh = _project(bh, bh.size)
    bc = float(np.max(_c(bh)))
    return bh, bc, int(bh.size)
