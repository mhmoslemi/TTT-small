    dx = 2.0 / n

    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    for _ in range(5):

        d = 0.5 * n - np.sum(h)
        h = np.clip(h, 0.0, 1.0)

            rem = d

                add = min(1.0 - h[i], rem)
                h[i] += add
                rem -= add

                    break

            rem = -d

def _grad_single(h, s):
    """Gradient of C(s) = sum_i h[i+s]*(1-h[i])*dx w.r.t. h. O(n)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        s2 = -s
        g[s2:] += (1.0 - h[:n-s2])
        g[:n-s2] -= h[s2:]
    return g * dx

                    moved = True

    return h

    C = _c5_array(h)
    worst = np.argsort(C)[-topk:]
    for s_idx in worst:
        s = s_idx - (n - 1)
        h = _minmax_cd(h, n, s, iters=cd_iters, lr=lr)
    return h

def _softmax_obj_grad(h, n, alpha, beta):
    C = _c5_array(h)
    m = np.max(C)
