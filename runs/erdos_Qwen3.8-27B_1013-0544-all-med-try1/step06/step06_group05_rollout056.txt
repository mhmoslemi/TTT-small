import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-12:
                h += room * (diff / total)
            else:
                h += diff / n
        else:
            total = h.sum()
            if total > 1e-12:
                h *= (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize(h_new, n_new)

def smooth_c5_and_grad(h, temp=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_v = np.max(vals)
    shifted = (vals - max_v) / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w_sum = np.sum(exp_s)
    weights = exp_s / w_sum
    smooth_val = max_v + temp * np.log(w_sum + 1e-30)
    grad = np.zeros(n)
    active = np.where(weights > 1e-5)[0]
    for idx in active:
        w = weights[idx]
        k = int(idx) - (n - 1)
        if k >= 0:
            m = n - k
            grad[:m] += w * dx * (-h[k:k+m])
            grad[k:k+m] += w * dx * (1.0 - h[:m])
        else:
            k2 = -k
            m = n - k2
            grad[:m] += w * dx * (1.0 - h[k2:k2+m])
            grad[k2:k2+m] += w * dx * (-h[:m])
    return smooth_val, grad
