def compute_delta(h, i, j, n):
    """Compute delta[k] for all k when swapping i (1->0) and j (0->1)."""
    # delta[k] = h[i-k] + h[i+k] - h[j-k] - h[j+k] (with 0 for out of bounds)
    # Valid k range:
    k_min = max(-i, i - n + 1, -j, j - n + 1)
    k_max = min(i, n - 1 - i, j, n - 1 - j)
    
    if k_min > k_max:
        return None, 0, 0
    
    # k values in numpy output are at index k + (n-1)
    idx_min = k_min + (n - 1)
    idx_max = k_max + (n - 1)
    
    ks = np.arange(k_min, k_max + 1)
    
    # h[i-k]: valid when 0 <= i-k < n
    a = i - ks
    a_val = np.zeros(len(ks))
    mask = (a >= 0) & (a < n)
    a_val[mask] = h[a[mask]]
    
    # h[i+k]: valid when 0 <= i+k < n
    b = i + ks
    b_val = np.zeros(len(ks))
    mask = (b >= 0) & (b < n)
    b_val[mask] = h[b[mask]]
    
    # h[j-k]
    c = j - ks
    c_val = np.zeros(len(ks))
    mask = (c >= 0) & (c < n)
    c_val[mask] = h[c[mask]]
    
    # h[j+k]
    d = j + ks
    d_val = np.zeros(len(ks))
    mask = (d >= 0) & (d < n)
    d_val[mask] = h[d[mask]]
    
    delta = a_val + b_val - c_val - d_val
