import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def fast_local_search(h0, n, max_time, seed):
    h = make_binary(h0, n).astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    start = time.time()
    no_improve = 0

    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-12
        best_i = -1
        best_j = -1
        best_delta = None
        best_new_max = 0.0
        k_arr = np.arange(2 * n - 1)
        L_arr = k_arr - (n - 1)

        for i in zeros:
            iL = i - L_arr
            iR = i + L_arr
            v1 = (iL >= 0) & (iL < n)
            v2 = (iR >= 0) & (iR < n)
            h_iL = h[np.clip(iL, 0, n - 1)]
            h_iR = h[np.clip(iR, 0, n - 1)]
            delta_i = (1.0 - h_iL) * v1.astype(float) - h_iR * v2.astype(float)

            for j in ones:
                jL = j - L_arr
                jR = j + L_arr
                v3 = (jL >= 0) & (jL < n)
                v4 = (jR >= 0) & (jR < n)
                h_jL = h[np.clip(jL, 0, n - 1)]
                h_jR = h[np.clip(jR, 0, n - 1)]
                delta_j = -(1.0 - h_jL) * v3.astype(float) + h_jR * v4.astype(float)

                delta = delta_i + delta_j
                new_max = float(np.max(corr + delta))
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
                    best_delta = delta
                    best_new_max = new_max

        if best_gain > 1e-10 and best_delta is not None:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = corr + best_delta
            current_max = best_new_max
            no_improve = 0
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 4:
                n_pert = rng.randint(3, min(12, n // 4 + 1))
                for _ in range(n_pert):
                    ii = zeros[rng.randint(len(zeros))]
                    jj = ones[rng.randint(len(ones))]
                    if ii != jj:
                        h[ii] = 1.0
                        h[jj] = 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                current_max = float(np.max(corr))
                no_improve = 0

    return best_h, best_c5

def simulated_annealing(h0, n, max_time, seed):
    h = make_binary(h0, n).astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    start = time.time()

    while (time.time() - start) < max_time:
        elapsed = time.time() - start
        T = 0.02 * (0.01) ** (elapsed / max_time)
        if T < 1e-10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        if i == j:
            continue
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_max = float(np.max(new_corr))
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T * n, 1e-12)):
            corr = new_corr
            current_max = new_max
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0

    return best_h, best_c5

def pair_swap_local_search(h0, n, max_time, seed):
    h = make_binary(h0, n).astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    start = time.time()
    iter_count = 0

    while (time.time() - start) < max_time:
        iter_count += 1
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        # Sample a subset of double swaps
        n_trials = min(500, len(ones) * len(ones) // 2 * len(zeros) * len(zeros) // 2)
        found = False
        for _ in range(n_trials):
            idx1 = rng.randint(len(ones))
            idx2 = rng.randint(len(ones))
            idx3 = rng.randint(len(zeros))
            idx4 = rng.randint(len(zeros))
            if idx1 == idx2 or idx3 == idx4:
                continue
            i1, i2 = ones[idx1], ones[idx2]
            j1, j2 = zeros[idx3], zeros[idx4]
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            new_max = float(np.max(new_corr))
            if new_max < current_max - 1e-10:
                corr = new_corr
                current_max = new_max
                found = True
                if current_max * dx < best_c5:
                    best_c5 = current_max * dx
                    best_h = h.copy()
                break
            else:
                h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
        if not found and iter_count % 50 == 0:
            n_pert = rng.randint(4, 14)
            for _ in range(n_pert):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                if ii != jj:
                    h[ii] = 1.0; h[jj] = 0.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            current_max = float(np.max(corr))

    return best_h, best_c5
