def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h
