import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 + n_lag]
    return g

def focused_grad(h, corr, dx, alpha=20.0, top_m=6):
    N = len(h)
    top_idx = np.argsort(corr)[-top_m:]
    c_vals = corr[top_idx]
    c_max = c_vals.max()
    w = np.exp(alpha * (c_vals - c_max))
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = int(np.argmax(corr))
        return grad_lag(h, k_star - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    for m in range(top_m):
        if w[m] < 1e-8:
            continue
        lag = int(top_idx[m]) - (N - 1)
        g += w[m] * grad_lag(h, lag)
    return g * dx

def interpolate_to_n(h_src, n_dst):

    h = np.interp(x1, x0, h_src)
