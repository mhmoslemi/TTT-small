import numpy as np

def project_capped_simplex(v, t):
    """Project v onto {x: 0<=x<=1, sum(x)=t}"""
    n = len(v)
    # Binary search on tau such that sum(clamp(v - tau, 0, 1)) = t
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > t:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)

def smooth_max(x, alpha):
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(alpha * (x - x_max)))) / alpha

def get_weights(x, alpha):

    e = np.exp(alpha * (x - x_max))
    return e / np.sum(e)

def compute_grad(h, weights):
