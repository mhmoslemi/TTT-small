            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_move = (i, j, sgn * delta)
            h[i] -= sgn * delta
            h[j] += sgn * delta
        if best_move is not None and best_gain > 1e-12:
            i, j, d = best_move
            h[i] = np.clip(h[i] + d, 0, 1)
            h[j] = np.clip(h[j] - d, 0, 1)
            h = normalize_continuous(h, n)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            if rng.random() < 0.4:
                idx = rng.choice(n, size=rng.randint(2, 6), replace=False)
                h[idx] += rng.uniform(-0.08, 0.08, len(idx))
                h = normalize_continuous(h, n)
                current_c5 = compute_c5(h)
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    h_init = normalize_continuous(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init

    # Phase 1: Subgradient descent
    t1 = min(180.0, rem() * 0.2)
    h1, c5_1 = subgradient_descent(h_init, max_time=t1, seed=seed, top_k=5)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Local search polish
    if rem() > 80:
        t2 = min(120.0, rem() * 0.15)
        h2, c5_2 = local_search(best_h, max_time=t2, seed=seed + 200)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Higher resolutions
    for n_cand in [168, 224, 280]:
        if rem() < 180:
            break
        if n_cand % 2 != 0:
            continue
        h_interp = interpolate_h(best_h, n_cand)
        t3a = min(130.0, rem() * 0.2)
        h3, c5_3 = subgradient_descent(h_interp, max_time=t3a, seed=seed + n_cand, top_k=5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_cand
        if rem() > 80:
            t3b = min(80.0, rem() * 0.12)
            h3b, c5_3b = local_search(best_h, max_time=t3b, seed=seed + 2000 + n_cand)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()
                best_n = n_cand

    # Phase 4: Multi-start at best resolution
    if rem() > 80:
        n_best = best_n
        n_starts = min(4, max(1, int(rem() / 80)))
        for s in range(n_starts):
            if rem() < 70:
                break
            rng_s = np.random.RandomState(seed + 5000 + s * 137)
            pert = rng_s.normal(0, 0.04, n_best)
            h_start = best_h + pert
            h_start = normalize_continuous(h_start, n_best)
            t4a = min(rem() * 0.3, 100.0)
            h4, c5_4 = subgradient_descent(h_start, max_time=t4a, seed=seed + 6000 + s, top_k=3)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
                best_n = n_best
            if rem() > 50:
                t4b = min(rem() * 0.2, 50.0)
                h4b, c5_4b = local_search(best_h, max_time=t4b, seed=seed + 7000 + s)
                if c5_4b < best_c5:
                    best_c5 = c5_4b
                    best_h = h4b.copy()
                    best_n = n_best

    # Phase 5: Final local search
    if rem() > 20:
        t5 = rem() * 0.85
        h5, c5_5 = local_search(best_h, max_time=t5, seed=seed + 9000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    best_h = normalize_continuous(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
