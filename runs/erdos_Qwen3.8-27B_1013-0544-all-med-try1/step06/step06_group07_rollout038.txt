    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        d = target - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1.0 - x
            m = s > 1e-14
            if not m.any(): break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-14
            if not m.any(): break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0.0, 1.0, out=x)
    d = target - x.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - x)[::-1] if d > 0 else np.argsort(x)[::-1]
        for idx in order:
            if abs(d) < 1e-15: break
            room = (1.0 - x[idx]) if d > 0 else x[idx]
            if room < 1e-15: continue
            add = min(d, room) if d > 0 else max(d, -room)
            x[idx] += add
            d -= add
    return x

def c_full(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5(h):
    return float(np.max(c_full(h)))

            m = -k
            grad[:n-m] += dx * (1.0 - h[m:])

    y = h + step * direction
    return repair(y)

def backtracking(h, norm_val, direction, alpha0=0.1, c1=1e-4, max_iter=30):
    """Backtracking line search"""
    alpha = alpha0
    for _ in range(max_iter):
        y = repair(h + alpha * direction)
        cy = c5(y)
        # We use a simplified condition: just check if we improved
        if cy < norm_val - 1e-12 or alpha < 1e-10:
            return y, cy, alpha
        alpha *= 0.5
    return h, norm_val, alpha

def polish_subgradient(h, time_limit, topk=5, alpha0=0.05, seed=42):

    x = repair(h)
    best_x = x.copy()
    best_c = c5(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + time_limit
    alpha = alpha0
    stall = 0
    
    while time.time() < deadline:
        c = c_full(x)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 30:

            improved = False
            for sg in (0.01, 0.005, 0.002):
                y = repair(x + rng.normal(0, sg, x.size))
                cy = c5(y)
                if cy < best_c:
                    best_c = cy
                    best_x = y.copy()
                    x = y
                    stall = 0
                    improved = True
                    break
            if not improved:
                break
            continue

        active = np.where(c >= C - tol)[0]

            active = np.argsort(c)[::-1][:topk]

            alpha = min(alpha * 1.1, 0.5)
        else:

    """Refine using SLSQP"""
    n = h.size
    dx = 2.0 / n
    def fun(x):
        return c5(x)
    def jac(x):
        c = c_full(x)
        C = np.max(c)
        tol = max(1e-8, 1e-5 * C)
        active = np.where(c >= C - tol)[0]
        if len(active) > 8:
            active = np.argsort(c)[::-1][:8]
        return grad_c_active(x, active)
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    try:

                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or c5(repair(res.x)) < c5(h):
            return repair(res.x)
    except:
        pass
    return h

    x0 = (np.arange(n0) + 0.5) * 2.0 / n0

    y = np.interp(x1, x0, h0)

            grad[m:] += -h[:n-m]
    return grad * dx / len(active)

def _slsqp_cons(x):
    n = x.size
    return x.sum() - 0.5 * n

def _slsqp_jac_cons(x):
    return np.ones(x.size)

def slsqp_opt(h0, f, j, maxiter=200):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac_cons}]
    try:
        res = so.minimize(f, h0, method='SLSQP', jac=j, bounds=bounds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(res.x)
    except Exception:
        return h0

def smooth_obj(h):
    # Soft-max smoothed objective with small tau
    c = c_full(h)
    tau = 0.003
    C = np.max(c)
    z = (c - C) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return C
    return C + tau * np.log(s)

def smooth_grad(h):
    c = c_full(h)
    tau = 0.003
    C = np.max(c)
    z = (c - C) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        active = np.array([np.argmax(c)])
        return grad_c_active(h, active)
    w = e / s
    # Weighted gradient
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w[idx] * (1.0 - 2.0 * h)
            else:
                grad[k:] += w[idx] * (1.0 - h[:n-k])
                grad[:n-k] += -w[idx] * h[k:]
        else:
            m = -k
            grad[:n-m] += w[idx] * (1.0 - h[m:])
            grad[m:] += -w[idx] * h[:n-m]
    return grad * dx

def exact_obj(h):
    return c5(h)

def exact_grad(h):
    c = c_full(h)
    C = np.max(c)
    tol = max(1e-8, 1e-5 * C)
    active = np.where(c >= C - tol)[0]
    if len(active) > 8:
        active = np.argsort(c)[::-1][:8]
    return grad_c_active(h, active)

def subgradient_polish(h0, t_limit, topk=6, alpha0=0.03, seed=0):
    x = repair(h0)
    best_x = x.copy()
    best_c = c5(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + t_limit
    alpha = alpha0
    stall = 0
    iterations = 0
    
    while time.time() < deadline:
        iterations += 1
        c = c_full(x)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 40:
            # Perturbation
            found = False
            for sg in (0.005, 0.002, 0.001):
                y = repair(x + rng.normal(0, sg, x.size))
                cy = c5(y)
                if cy < best_c:
                    best_c, best_x = cy, y.copy()
                    x = y
                    stall = 0
                    found = True
                    break
            if not found:
                break
            continue
        
        # Active set
        tol = max(1e-7, 3e-5 * C)
        active = np.where(c >= C - tol)[0]
        if len(active) > topk:
            active = np.argsort(c)[::-1][:topk]
        
        g = grad_c_active(x, active)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        
        # Line search
        found_step = False
        a = alpha
        for _ in range(20):
            y = repair(x - a * g)
            cy = c5(y)
            if cy < C - 1e-14:
                x = y
                found_step = True
                alpha = min(a * 1.05, 0.2)
                break
            a *= 0.5
            if a < 1e-12:
                break
        if not found_step:
            alpha *= 0.5
            if alpha < 1e-10:
                break
        elif iterations % 50 == 0 and rng.rand() < 0.2:
            y = repair(x + rng.normal(0, 0.003, x.size))
            if c5(y) < best_c:
                best_c = c5(y)
                best_x = y.copy()
                x = y
    
    return best_x, best_c

def resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * 2.0 / n0
    x1 = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return repair(np.interp(x1, x0, h0))

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def time_left(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    target = 0.5 * n
    if n == 0:
        return x
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        d = target - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1.0 - x
            m = s > 1e-14
            if not m.any():
                break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-14
            if not m.any():
                break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0.0, 1.0, out=x)
    d = target - x.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - x)[::-1] if d > 0 else np.argsort(x)[::-1]
        for idx in order:
            if abs(d) < 1e-15:
                break
            room = (1.0 - x[idx]) if d > 0 else x[idx]
            if room < 1e-15:
                continue
            add = min(d, room) if d > 0 else max(d, -room)
            x[idx] += add
            d -= add
    return x
