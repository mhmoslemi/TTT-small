import numpy as np
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def swap_delta(h, i, j):
    n = len(h)
    nc = 2 * n - 1
    delta = np.zeros(nc)
    for k in range(1, n):
        d = 0.0
        if i + k < n:
            d += h[i + k]
        if i >= k:
            d -= 1.0 - h[i - k]
        if j + k < n:
            d -= h[j + k]
        if j >= k:
            d += 1.0 - h[j - k]
        delta[n - 1 + k] = d
        d = 0.0
        if i + k < n:
            d -= 1.0 - h[i + k]
        if i >= k:
            d += h[i - k]
        if j + k < n:
            d += 1.0 - h[j + k]
        if j >= k:
            d -= h[j - k]
        delta[n - 1 - k] = d
    return delta

def make_binary(h, rng):
    n = len(h)
    hb = (h >= 0.5).astype(float)
    s = int(hb.sum())
    target = n // 2
    if s > target:
        idx = np.where(hb == 1)[0]
        rng.shuffle(idx)
        hb[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(hb == 0)[0]
        rng.shuffle(idx)
        hb[idx[:target - s]] = 1
    return hb

    zeros_l = np.where(h < 0.5)[0].tolist()
    T = 0.02
    T_min = 3e-5
    cool = (T_min / T) ** (1.0 / max(1, max_iters))
    stuck = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ii = rng.randint(len(ones_l))

            h[j] = 1.0
            corr += delta
            val = new_val

                best_h = h.copy()
                stuck = 0

            stuck += 1
        T = max(T * cool, T_min)
        if stuck > 4000:
            T = max(T, 0.01)
            stuck = 0
    return best_h, best_val

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    N = len(h)
    g = np.zeros(N)
    s1, e1 = max(0, k), min(N, N + k)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - k:e1 - k]
    s2, e2 = max(0, -k), min(N, N - k)
    if s2 < e2:
        g[s2:e2] -= h[s2 + k:e2 + k]
    return g

def run_refine(h, rng, t_end, max_iter=100000):

    best_h = h.copy()
    best_val = val

        c = corr - corr.max()

        if ws > 1e-300:

            g = np.zeros(n)
            for jj in range(2 * n - 1):

            g = grad_lag(h, int(np.argmax(corr)) - (n - 1))

        h_new = project_feasible(h - lr * g, n)

        if vn < val - 1e-14:

            lr = min(lr * 1.02, 0.1)
            if val < best_val:

            if lr < 1e-5:
                h = project_feasible(h + rng.randn(n) * 0.05, n)
                val, corr, _ = evaluate(h)
                lr = 0.03
    return best_h, best_val

    val, corr, _ = evaluate(h)
    best_h = h.copy()
    best_val = val
    ones_l = np.where(h >= 0.5)[0].tolist()
    zeros_l = np.where(h < 0.5)[0].tolist()
    T = 0.02
    T_min = 3e-5
    cool = (T_min / T) ** (1.0 / max(1, max_iters))
    stuck = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ii = rng.randint(len(ones_l))
        jj = rng.randint(len(zeros_l))
        i = ones_l[ii]
        j = zeros_l[jj]
        delta = swap_delta(h, i, j)
        new_val = (corr + delta).max() * dx
        dv = new_val - val
        if dv < -1e-15 or (T > T_min and rng.rand() < np.exp(-dv / T)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            val = new_val
            ones_l[ii] = j
            zeros_l[jj] = i
            if val < best_val - 1e-15:
                best_val = val
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        T = max(T * cool, T_min)
        if stuck > 4000:
            T = max(T, 0.01)
            stuck = 0
    return best_h, best_val

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    N = len(h)
    g = np.zeros(N)
    s1, e1 = max(0, k), min(N, N + k)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - k:e1 - k]
    s2, e2 = max(0, -k), min(N, N - k)
    if s2 < e2:
        g[s2:e2] -= h[s2 + k:e2 + k]
    return g

def run_refine(h, rng, t_end, max_iter=100000):
    n = len(h)
    val, corr, dx = evaluate(h)
    best_h = h.copy()
    best_val = val
    lr = 0.03
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = corr - corr.max()
        w = np.exp(25.0 * c)
        ws = w.sum()
        if ws > 1e-300:
            w /= ws
            g = np.zeros(n)
            for jj in range(2 * n - 1):
                if w[jj] < 1e-5:
                    continue
                g += w[jj] * grad_lag(h, jj - (n - 1))
        else:
            g = grad_lag(h, int(np.argmax(corr)) - (n - 1))
        g *= dx
        h_new = project_feasible(h - lr * g, n)
        vn, cn, _ = evaluate(h_new)
        if vn < val - 1e-14:
            h, val, corr = h_new, vn, cn
            lr = min(lr * 1.02, 0.1)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            lr *= 0.7
            if lr < 1e-5:
                h = project_feasible(h + rng.randn(n) * 0.05, n)
                val, corr, _ = evaluate(h)
                lr = 0.03
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 3
    best_h = None
    best_val = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        v = evaluate(h_init)[0]
        best_val, best_h = v, h_init.copy()

    # Phase 1: SA on binarized initial
    if h_init is not None:
        hb = make_binary(h_init, rng)
        vb = evaluate(hb)[0]
        if vb < best_val:
            best_val, best_h = vb, hb.copy()
        t_pe = min(t_end - 180, t0 + 150)
        h1, v1 = run_sa(hb, rng, t_pe, 8000000)
        if v1 < best_val:
            best_val, best_h = v1, h1.copy()

    # Phase 2: Continuous refinement
    if best_h is not None and time.time() < t_end - 100:
        t_pe = min(t_end - 100, time.time() + 80)
        h2, v2 = run_refine(best_h.copy(), rng, t_pe)
        if v2 < best_val:
            best_val, best_h = v2, h2.copy()

    # Phase 3: SA on refined binarized
    if best_h is not None and time.time() < t_end - 80:
        hb2 = make_binary(best_h, rng)
        t_pe = min(t_end - 80, time.time() + 80)
        h3, v3 = run_sa(hb2, rng, t_pe, 6000000)
        if v3 < best_val:
            best_val, best_h = v3, h3.copy()

    # Phase 4: Try other n values
    if best_h is not None and time.time() < t_end - 40:
        for nt in [100, 140, 160, 200]:
            if time.time() > t_end - 20:
                break
            hs = np.interp(np.arange(nt) * len(best_h) / nt, np.arange(len(best_h)), best_h)
            hb = make_binary(hs, rng)
            vb = evaluate(hb)[0]
            if vb < best_val + 0.005:
                t_pe = min(t_end - 20, time.time() + 50)
                h5, v5 = run_sa(hb, rng, t_pe, 4000000)
                if v5 < best_val:
                    best_val, best_h = v5, h5.copy()

    # Phase 5: ILS restarts
    if best_h is not None and time.time() < t_end - 20:
        hb4 = make_binary(best_h, rng)
        for _ in range(4):
            if time.time() > t_end - 10:
                break
            h_p = hb4.copy()
            ones = np.where(h_p == 1)[0]
            zeros = np.where(h_p == 0)[0]
            for _ in range(rng.randint(5, 12)):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h_p[a], h_p[b] = h_p[b], h_p[a]
            t_pe = min(time.time() + 35, t_end - 10)
            hr, vr = run_sa(h_p, rng, t_pe, 3000000)
            if vr < best_val:
                best_val, best_h = vr, hr.copy()
                hb4 = make_binary(best_h, rng)

    # Phase 6: Final continuous refinement
    if best_h is not None and time.time() < t_end:
        hf, vf = run_refine(best_h.copy(), rng, t_end, 50000)
        if vf < best_val:
            best_val, best_h = vf, hf.copy()

    if best_h is None:
        best_h = 0.5 * np.ones(47)
        best_val = evaluate(best_h)[0]
    return best_h, best_val, len(best_h)
