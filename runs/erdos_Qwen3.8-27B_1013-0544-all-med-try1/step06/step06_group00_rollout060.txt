```python
def compute_grad_c(h, p, n):
    """Compute sum_k p[k] * dc[k]/dh, where c = correlate(h, 1-h, 'full')."""
    omh = 1.0 - h
    grad = np.zeros(n)
    
    for shift in range(-(n-1), n):
        k = shift + n - 1
        w = p[k]
        if abs(w) < 1e-15:
            continue
        
        # Contribution: w * ((1-h)[j-shift] - h[j+shift]) for j=0..n-1
        # (1-h)[j-shift]: valid when 0 <= j-shift < n, i.e., shift <= j < shift+n
        #   For shift >= 0: j from shift to n-1, values omh[0:n-shift]
        #   For shift < 0: j from 0 to n-1+shift, values omh[-shift:n]
        # h[j+shift]: valid when 0 <= j+shift < n, i.e., -shift <= j < n-shift
        #   For shift >= 0: j from 0 to n-1-shift, values h[shift:n]
        #   For shift < 0: j from -shift to n-1, values h[0:n+shift]
        
        if shift >= 0:
            # (1-h)[j-shift] for j=shift..n-1: omh[0:n-shift]
            # h[j+shift] for j=0..n-1-shift: h[shift:n]
            omh_part = np.zeros(n)
            omh_part[shift:] = omh[:n-shift]
            h_part = np.zeros(n)
            h_part[:n-shift] = h[shift:]
        else:
            ss = -shift  # positive
            # (1-h)[j+ss] for j=0..n-1-ss: omh[ss:n]
            # h[j-ss] for j=ss..n-1: h[0:n-ss]
            omh_part = np.zeros(n)
            omh_part[:n-ss] = omh[ss:]
            h_part = np.zeros(n)
            h_part[ss:] = h[:n-ss]
        
        grad += w * (omh_part - h_part)
    
    return grad
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_simplex_box(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def symmetrize(h):
    return 0.5 * (h + h[::-1])

def compute_grad_c(h, p, n):
    omh = 1.0 - h
    grad = np.zeros(n)
    for shift in range(n - 1, -1, -1):
        w_pos = p[n - 1 + shift]
        w_neg = p[n - 1 - shift]
        if shift == 0:
            if abs(w_pos) > 1e-14:
                grad += w_pos * (1.0 - 2.0 * h)
        else:
            if abs(w_pos) > 1e-14:
                omh_part = np.zeros(n)
                omh_part[shift:] = omh[:n - shift]
                h_part = np.zeros(n)
                h_part[:n - shift] = h[shift:]
                grad += w_pos * (omh_part - h_part)
            if abs(w_neg) > 1e-14:
                omh_part = np.zeros(n)
                omh_part[:n - shift] = omh[shift:]
                h_part = np.zeros(n)
                h_part[shift:] = h[:n - shift]
                grad += w_neg * (omh_part - h_part)
    return grad

def fista_opt(h_start, n, temps, iters_per_temp, deadline):
    dx = 2.0 / n
    target = n / 2.0
    best_h = project_simplex_box(h_start, target)
    best_val = compute_c5(best_h)

    for temp in temps:
        h_curr = best_h.copy()
        y = h_curr.copy()
        z = h_curr.copy()
        t_k = 1.0
        lr = 2.0 / (temp * dx + 1.0)

        for it in range(iters_per_temp):
            if time.time() >= deadline:
                break

            c_y = np.correlate(y, 1.0 - y, mode="full")
            m_y = np.max(c_y)
            exp_v = np.exp(temp * (c_y - m_y))
            p = exp_v / exp_v.sum()

            grad = compute_grad_c(y, p, n) * dx

            y_new = y - lr * grad
            y_new = project_simplex_box(y_new, target)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z_new = y_new + ((t_k - 1.0) / t_new) * (y_new - z)
            z_new = project_simplex_box(z_new, target)

            z = z_new
            y = y_new
            t_k = t_new

            if it % 10 == 0:
                c_val = compute_c5(y_new)
                if c_val < best_val - 1e-13:
                    best_val = c_val
                    best_h = y_new.copy()

        c_val = compute_c5(best_h)
        if c_val > best_val:
            best_val = compute_c5(best_h)

    return best_h, best_val

def coord_descent(h, n_levels=12, max_sweeps=20, deadline=None):
    n = len(h)
    target = n / 2.0
    h = project_simplex_box(h, target)
    c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if deadline and time.time() >= deadline:
            break
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_h = orig.copy()
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                h_trial = project_simplex_box(h, target)
                c = compute_c5(h_trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h_trial.copy()
            h = orig
            if best_c < c5 - 1e-12:
                h = best_h
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def interpolate_h(h_src, n_dst):
    x_old = np.arange(len(h_src), dtype=float)
    x_new = np.arange(n_dst, dtype=float)
    return np.interp(x_new, x_old, h_src)
