    corr = np.correlate(h, 1.0 - h, mode="full")

from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_step(h_start, n, temp=50.0, maxiter=200):
    dx = 2.0 / n
    ts = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        # Log-sum-exp for smooth max
        shifted = temp * (corr - m)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        s = np.exp(shifted).sum()
        return (m + np.log(s) / temp) * dx
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None:
        return project_feasible(res.x, ts)
    return h_start

def coord_descent_fast(h, n_levels=20, max_sweeps=50):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    levels = np.linspace(0.0, 1.0, n_levels)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in levels:
                if v == orig:
                    continue
                h[j] = v
                # Quick feasibility fix
                d = ts - h.sum()
                if abs(d) > 1e-15:

                        h = np.clip(h, 0.0, 1.0)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_c < c5 - 1e-14:

                        h = np.clip(h, 0.0, 1.0)
                c5 = best_c

        if not improved:

    top_idx = np.argsort(corr)[-4:]
    h_new = h.copy()
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            h_new[:n-s] += strength * (1.0 - h[s:]) * (np.random.rand(n-s) > 0.5).astype(float)
            h_new[s:] -= strength * h[:n-s] * (np.random.rand(n-s) > 0.5).astype(float)
        else:
            s2 = -s
            h_new[s2:] += strength * (1.0 - h[:n-s2]) * (np.random.rand(n-s2) > 0.5).astype(float)
            h_new[:n-s2] -= strength * h[s2:] * (np.random.rand(n-s2) > 0.5).astype(float)
    return project_feasible(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP annealing at n0
    for temp in [300.0, 100.0, 40.0, 15.0, 5.0, 3.0]:
        if remaining() < 50:
            break
        h_cur = slsqp_opt(best_h, n0, temp=temp, maxiter=200)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            best_h = h_cur.copy()

    # Phase 2: Subgradient descent
    if remaining() > 80:
        n_sg = int(min(30000, remaining() * 40))
        h_sg, c_sg = subgradient_descent(best_h, n_steps=n_sg, top_k=5, step_init=0.05)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Phase 3: Heavy coordinate descent
    if remaining() > 20:
        h_cd, c_cd = coord_descent(best_h, n_levels=30, max_sweeps=60)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 4: Targeted shake + re-optimize loop
    for it in range(40):
        if remaining() < 80:
            break
        h_pert = targeted_shake(best_h, strength=0.04 + 0.06 * np.random.rand())
        h_pert = slsqp_opt(h_pert, n0, temp=20.0, maxiter=80)
        h_pert, c_pert = coord_descent(h_pert, n_levels=15, max_sweeps=8)
        if c_pert < best_c5 - 1e-10:
            best_c5 = c_pert
            best_h = h_pert.copy()

    # Phase 5: Multi-scale optimization
    for n_try in [64, 96, 128, 160, 192, 256, 320]:
        if remaining() < 100:
            break
        x_old = np.linspace(0, 1, len(best_h), endpoint=False)
        x_new = np.linspace(0, 1, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try / 2.0)

        for temp in [60.0, 20.0]:
            if remaining() < 50:
                break
            h_new = slsqp_opt(h_new, n_try, temp=temp, maxiter=120)

        if remaining() > 30:
            h_new, c_new = subgradient_descent(h_new, n_steps=2000, top_k=3, step_init=0.03)

        if remaining() > 20:
            h_new, c_new = coord_descent(h_new, n_levels=20, max_sweeps=25)

        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 6: Diverse random restarts
    for it in range(40):
        if remaining() < 70:
            break
        rtype = it % 4
        n_cur = len(best_h)
        if rtype == 0:
            h_r = best_h + np.random.randn(n_cur) * 0.08
        elif rtype == 1:
            h_r = (np.random.rand(n_cur) > 0.5).astype(float)
        elif rtype == 2:
            freq = np.random.choice([1, 2, 3, 4, 5])
            phase = np.random.randn(n_cur) * 0.5
            h_r = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
        else:
            h_r = 0.5 + 0.3 * np.random.randn(n_cur)
            h_r = np.cumsum(h_r - h_r.mean())
            h_r = h_r - h_r.min()
            if h_r.max() > 0:
                h_r = h_r / h_r.max()

        h_r = project_feasible(h_r, n_cur / 2.0)
        h_r = slsqp_opt(h_r, n_cur, temp=40.0, maxiter=100)
        h_r, c_r = coord_descent(h_r, n_levels=12, max_sweeps=8)
        if c_r < best_c5 - 1e-10:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 7: Final intensive polish
    if remaining() > 15:
        h_cd, c_cd = coord_descent(best_h, n_levels=50, max_sweeps=40)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    if remaining() > 10:
        h_sg, c_sg = subgradient_descent(best_h, n_steps=5000, top_k=3, step_init=0.02)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()
        h_cd, c_cd = coord_descent(best_h, n_levels=40, max_sweeps=20)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
