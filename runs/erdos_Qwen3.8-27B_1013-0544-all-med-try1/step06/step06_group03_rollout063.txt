def sens_column(h, i):
    n = len(h)
    c = np.zeros(2 * n - 1)
    # For shift s >= 0 (idx = (n-1) + s, s = 0 to n-1):
    # c[idx] += (1-h[i-s]) if i >= s
    # c[idx] -= h[i+s] if i < n-s
    # For shift s < 0 (idx = (n-1) + s, s = -(n-1) to -1, let m = -s):
    # c[idx] += (1-h[i+m]) if i < n-m
    # c[idx] -= h[i-m] if i >= m
    
    # Positive shifts: s = 0, 1, ..., n-1; idx = n-1+s
    # Condition i >= s: s <= i, so s ranges from 0 to min(i, n-1)
    # Condition i < n-s: s < n-i, so s ranges from 0 to min(n-i-1, n-1)
    
    # (1-h[i-s]) for s = 0 to min(i, n-1): these go to idx = n-1+s
    s_pos = np.arange(min(i+1, n))
    c[n-1+s_pos] += 1.0 - h[i - s_pos]
    
    # -h[i+s] for s = 0 to min(n-i-1, n-1): these go to idx = n-1+s
    s_pos2 = np.arange(min(n-i, n))
    c[n-1+s_pos2] -= h[i + s_pos2]
    
    # Negative shifts: m = 1 to n-1; idx = n-1-m
    # (1-h[i+m]) for m where i < n-m, i.e., m < n-i: m = 1 to min(n-i-1, n-1)
    # -h[i-m] for m where i >= m: m = 1 to min(i, n-1)
    
    m_pos = np.arange(1, min(n-i, n))
    if len(m_pos) > 0:

    m_pos2 = np.arange(1, min(i+1, n))
    if len(m_pos2) > 0:
        c[n-1-m_pos2] -= h[i - m_pos2]
    
    return c

    s1 = np.arange(min(i + 1, n))
    c[n - 1 + s1] += 1.0 - h[i - s1]
    s2 = np.arange(min(n - i, n))

    h = h.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def sens_col(h, i, n):
    c = np.zeros(2 * n - 1)
    s1 = np.arange(min(i + 1, n))
    c[n - 1 + s1] += 1.0 - h[i - s1]
    s2 = np.arange(min(n - i, n))
    c[n - 1 + s2] -= h[i + s2]
    m1 = np.arange(1, min(n - i, n))
    if m1.size:
        c[n - 1 - m1] += 1.0 - h[i + m1]
    m2 = np.arange(1, min(i + 1, n))
    if m2.size:
        c[n - 1 - m2] -= h[i - m2]
    return c

def pair_cd(h, n, n_iter, grid=60, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    conv = np.correlate(h, 1.0 - h, mode="full")
    best_val = np.max(conv)
    best_h = h.copy()
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1) % n
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hiv = min(1.0, s)
        if hiv - lo < 1e-10:
            continue
        ci = sens_col(h, i, n)
        cj = sens_col(h, j, n)
        a = ci - cj
        b = conv - a * h[i]
        xg = np.linspace(lo, hiv, grid)
        fv = np.max(a[None, :] * xg[:, None] + b[None, :], axis=1)
        bn = np.argmin(fv)
        xn = xg[bn]
        d = xn - h[i]
        if abs(d) < 1e-12:
            continue
        h[i] = xn
        h[j] = s - xn
        conv += d * a
        mv = conv.max()
        if mv <= best_val + 1e-11:
            if mv < best_val:
                best_val = mv
                best_h = h.copy()
        else:
            h[i] += -d
            h[j] += d
            conv -= d * a
    return best_h, best_val * 2.0 / n

def sm_grad_desc(h, n, t_init=0.05, t_min=0.001, max_iter=3000, lr_init=0.008):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    t = t_init
    lr = lr_init
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        m = conv.max()
        exps = np.exp((conv - m) / t)
        w = exps / exps.sum()
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            if w[idx] < 1e-10:
                continue
            s = idx - (n - 1)
            ii = np.arange(n)
            if s >= 0:
                g = np.zeros(n)
                m1 = ii >= s
                g[m1] = 1.0 - h[ii[m1] - s]
                m2 = ii < (n - s)
                g[m2] -= h[ii[m2] + s]
            else:
                ma = -s
                g = np.zeros(n)
                m1 = ii < (n - ma)
                g[m1] = 1.0 - h[ii[m1] + ma]
                m2 = ii >= ma
                g[m2] -= h[ii[m2] - ma]
            grad += w[idx] * g
        grad *= 2.0 / n
        h_new = project_feasible(h - lr * grad, n)
        c5n = compute_c5(h_new)
        if c5n < best - 1e-14:
            best = c5n
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.03)
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
        t = max(t * 0.997, t_min)
    return best_h, best
