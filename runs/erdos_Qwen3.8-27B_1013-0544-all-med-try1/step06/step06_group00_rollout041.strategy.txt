   def change_resolution(h, n_from, n_to):  
        # use interpolation preserving integral  
        x_old = np.linspace(0,1,n_from)  
        x_new = np.linspace(0,1,n_to)  
        h_interp = np.interp(x_new, x_old, h)  
        h_adjusted = h_interp * (n_from/2 / sum(h_old) )  # preserve integral  
        return np.clip(h_adjusted, 0,1) * (n_to/n_from)
