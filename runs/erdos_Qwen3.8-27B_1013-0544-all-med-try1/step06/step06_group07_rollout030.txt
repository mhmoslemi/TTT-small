import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(30):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - x
            m = slack > 1e-13
            if not m.any():
                break
            x[m] += diff * slack[m] / slack[m].sum()
        else:
            m = x > 1e-13
            if not m.any():
                break
            x[m] += diff * x[m] / x[m].sum()
        np.clip(x, 0.0, 1.0, out=x)
    # Final exact fix
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        x[:k] = 1.0
        rem = target - k
        if rem > 0 and k < n:
            x[k] = rem
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def lse_value(h, tau, topk=None):
    c = c_values(h)
    cm = c.max()
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z = z - z.max()
    e = np.exp(z)
    if topk is not None and 0 < topk < len(c):
        idx = np.argsort(e)[::-1][:topk]
        e2 = np.zeros(len(c))
        e2[idx] = e[idx]
        e = e2
    s = e.sum()
    if s <= 0:
        return cm
    return cm + tau * np.log(s)

def lse_weights(h, tau, topk=None):
    c = c_values(h)
    cm = c.max()
    if tau <= 0:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    z = (c - cm) / tau
    z = z - z.max()
    e = np.exp(z)
    if topk is not None and 0 < topk < len(c):
        idx = np.argsort(e)[::-1][:topk]
        e2 = np.zeros(len(c))
        e2[idx] = e[idx]
        e = e2
    s = e.sum()
    if s <= 0:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return e / s

def add_lag_grad(x, grad, k, scale):
    n = x.size
    s = int(k) - (n - 1)
    lo = max(s, 0)
    hi = min(s + n - 1, n - 1)
    if lo <= hi:
        grad[lo:hi+1] += scale * (1.0 - x[lo-s:hi+1-s])
    lo = max(-s, 0)
    hi = min(n - 1 - s, n - 1)
    if lo <= hi:
        grad[lo:hi+1] -= scale * x[lo+s:hi+s]

def weighted_grad(x, w):
    n = x.size
    grad = np.zeros(n)
    idxs = np.nonzero(w > 1e-12)[0]
    for k in idxs:
        add_lag_grad(x, grad, int(k), float(w[k]))
    return grad * (2.0 / n)

def subgradient_opt(x, time_limit, seed):

    no_imp = 0
    
    while time.time() < deadline:

        C = float(cvals.max())
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 20:
            break
        
        tol = max(1e-8, 1e-4 * C)
        active = np.where(cvals >= C - tol)[0]
        if active.size > 8:
            active = np.argsort(cvals)[::-1][:8]

        for k in active:
