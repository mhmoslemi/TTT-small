def repair(h):

    n = x.size
    if n == 0: return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)

        else:

        g[t:] -= x[:n-t]
    return g * (2.0/n)

import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

def tl(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0: return x
    target = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(60):
        d = target - x.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            sl = 1.0 - x; m = sl > 1e-12
            if not m.any(): break
            x[m] += d * sl[m] / sl[m].sum()
        else:
            m = x > 1e-12
            if not m.any(): break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = target - x.sum()
    if abs(d) > 1e-10:
        k = int(target); x[:] = 0; x[:k] = 1
        if k < n: x[k] = target - k
    return x

def cv(h):
    n = h.size
    return np.correlate(h, 1-h, mode="full") * (2.0/n)

def ec(h):
    return float(cv(h).max())

def sg1(x, k):
    n = x.size; s = k - (n-1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += 1.0 - x[:n-s]
        g[:n-s] -= x[s:]
    else:
        t = -s
        g[:n-t] += 1.0 - x[t:]
        g[t:] -= x[:n-t]
    return g * (2.0/n)

def cv_smooth(x, tau):
    c = cv(x); C = c.max()
    z = (c - C)/tau; z -= z.max()
    e = np.exp(z); s = e.sum()
    if s <= 0: return C
    return C + tau*np.log(s)

def sv_smooth(x, tau):
    c = cv(x); C = c.max()
    z = (c - C)/tau; z -= z.max()
    e = np.exp(z); w = e/e.sum()
    n = x.size; g = np.zeros(n)
    for k in np.where(w > 1e-10)[0]:
        g += w[k] * sg1(x, int(k))
    return g

def lp_solve(x0, active, cvals_at_x0):
    """Solve cutting plane LP at x0 with active set."""
    n = x0.size
    K = len(active)
    # c_k(x0) + g_k^T (x - x0) <= t
    # => g_k^T x - t <= c_k(x0) - g_k^T x0
    A = np.zeros((K, n+1))
    b = np.zeros(K)
    for i, k in enumerate(active):
        g = sg1(x0, int(k))
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = cvals_at_x0[int(k)] - g @ x0
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5*n])
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    # Bounds: 0<=x<=1, t free
    bounds = [(0,1)]*n + [(-np.inf, np.inf)]
    try:
        r = so.linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if r.success:
            return r.x[:n]
    except:
        pass
    return None

def cutting_plane(x0, budget, seed):
    """Cutting plane / bundle method for the max problem."""
    x = repair(x0)
    best_x = x.copy()
    best_c = ec(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + budget
    no_imp = 0
    
    while tl(deadline) > 0.2:
        c = cv(x)
        C = c.max()
        if C < best_c - 1e-14:
            best_c = C; best_x = x.copy(); no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                y = repair(x + rng.normal(0, 0.015, x.size))
                if ec(y) < best_c:
                    best_c = ec(y); best_x = y.copy(); x = y; no_imp = 0
                else:
                    break
        
        # Active set
        tol = max(1e-9, 1e-5*C)
        active = np.where(c >= C - tol)[0]
        if active.size > 12:
            active = np.argsort(c)[::-1][:12]
        
        # Solve LP
        h_new = lp_solve(x, active, c)
        if h_new is None:
            break
        
        h_new = repair(h_new)
        
        # Line search
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.075]:
            y = (1-alpha)*x + alpha*h_new
            y = np.clip(y, 0, 1)
            y = repair(y)
            cy = ec(y)
            if cy < C - 1e-13:
                x = y; accepted = True; break
        
        if not accepted:
            # Try subgradient step as fallback
            g = np.mean([sg1(x, int(k)) for k in active], axis=0)
            gn = np.linalg.norm(g)
            if gn > 1e-14:
                for alpha in [0.1, 0.05, 0.02, 0.01]:
                    y = repair(x - alpha * g / gn)
                    if ec(y) < C - 1e-13:
                        x = y; accepted = True; break
            if not accepted:
                no_imp += 1
    
    return best_x, best_c

def slsqp_smooth(x0, tau, maxiter):
    n = x0.size
    def fun(x): return cv_smooth(x, tau)
    def jac(x): return sv_smooth(x, tau)
    try:
        r = so.minimize(fun, x0, method='SLSQP', jac=jac,
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-0.5*n,'jac':lambda x: np.ones(n)}],
                       options={'maxiter':maxiter,'ftol':1e-14})
        return r
    except:
        return None

def full_opt(x0, budget, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = ec(x)
    t0 = time.time()
    
    # Phase 1: Smooth SLSQP
    for tau in [0.04, 0.02, 0.01, 0.005, 0.002]:
        if time.time()-t0 > budget*0.3: break
        r = slsqp_smooth(x, tau, 60)
        if r is not None and np.isfinite(r.fun):
            xr = repair(r.x); c = ec(xr)
            if c < best_c: best_c = c; best_x = xr.copy()
            x = xr if c <= best_c + 1e-3 else best_x
    
    # Phase 2: Cutting plane
    rem = budget - (time.time()-t0)
    if rem > 1.0:
        xp, cp = cutting_plane(best_x, rem*0.9, seed)
        if cp < best_c: best_c = cp; best_x = xp
    
    return best_x, best_c

def resample(h0, nn):
    h0 = np.asarray(h0, float); n0 = h0.size
    if n0 == nn: return repair(h0)
    if n0 == 0: return repair(np.full(nn, 0.5))
    x0 = (np.arange(n0)+0.5)*(2.0/n0)
    x1 = (np.arange(nn)+0.5)*(2.0/nn)
    return repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, budget_s)
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, float)
    
    best_x = None; best_c = 1e9
    
    if init is not None and init.size >= 10:
        bn = min(init.size, 80)
    else:
        bn = 70
    
    n_list = []
    for n in [bn, bn-8, bn+8, 64, 72, 80, 88]:
        n = int(n)
        if 40 <= n <= 120 and n not in n_list: n_list.append(n)
    n_list = n_list[:5]
    
    for n in n_list:
        if tl(deadline) < 40: break
        base = resample(init, n) if (init is not None and init.size>0) else repair(np.full(n,0.5))
        rng = np.random.RandomState(sv+n)
        
        starts = [base]
        for s in [0.03, 0.08, 0.12]:
            starts.append(repair(base + rng.normal(0, s, n)))
        t = (np.arange(n)+0.5)*(2.0/n)
        for f in [2, 3, 5]:
            starts.append(repair(0.5+0.35*np.sin(2*np.pi*f*t/2+rng.uniform(0,6.28))))
        
        rem = tl(deadline)
        per = min(50.0, max(6.0, rem/(len(n_list)*5)))
        
        for i, h0 in enumerate(starts):
            if tl(deadline) < 10: break
            x, c = full_opt(h0, per, sv+1000*n+i)
            if c < best_c: best_c = c; best_x = x
    
    if best_x is None:
        best_x = repair(np.full(70, 0.5)); best_c = ec(best_x)
    best_x = repair(best_x); best_c = ec(best_x)
    return best_x, best_c, int(best_x.size)
