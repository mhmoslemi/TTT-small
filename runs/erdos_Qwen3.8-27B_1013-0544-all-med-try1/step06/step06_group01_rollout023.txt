import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-10:
        # Distribute adjustment to elements closest to 0.5
        dist = np.abs(h - 0.5)
        order = np.argsort(dist)
        remaining = d
        for idx in order:
            if remaining > 0:
                adj = min(remaining, 1.0 - h[idx])
                h[idx] += adj
                remaining -= adj
            else:
                adj = max(remaining, h[idx] - 1.0)  # negative
                h[idx] += adj
                remaining -= adj
            if abs(remaining) < 1e-10:
                break
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            h[np.argmax(1.0 - h)] += d
        else:
            h[np.argmin(h)] += d  # Wait, argmax(h) for negative d
        h = np.clip(h, 0.0, 1.0)
    return h

def _softmax_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n)**2
    w = es / se
    
    grad = np.zeros(n)
    for i in range(n):
        j1 = i + np.arange(nc) - (n - 1)
        j2 = i - np.arange(nc) + (n - 1)
        h1 = np.where((j1 >= 0) & (j1 < n), h[np.clip(j1, 0, n - 1)], 0.0)
        h2 = np.where((j2 >= 0) & (j2 < n), h[np.clip(j2, 0, n - 1)], 0.0)
        dC = 1.0 - h1 - h2
        grad[i] = dx * np.dot(w, dC)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)

        grad[i] = dx * np.dot(hot.astype(float), dC)

    # Projection: clip to [0,1], then adjust sum

    # Then clip
    grad_centered = grad - np.mean(grad)
    h_new = h - step_size * grad_centered
    h_new = _fix(h_new, n)

def _grad_optimize(h, n, alpha_sweep, beta, maxiter_per_alpha=300, time_limit=None, t0=None):

    best = h.copy()
    best_c5 = _c5_max(best)

            cx = _c5_max(hx)
            if cx < best_c5:

                best_c5 = cx

def _local_refine(h, n, max_passes=100, time_limit=None, t0=None):

    best = h.copy()

import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        # Simple: add to all elements proportionally to available space
        if d > 0:
            space = 1.0 - h
            total_space = np.sum(space)
            if total_space > 1e-15:
                h += d * space / total_space
            else:
                h += d / n
        else:
            space = h
            total_space = np.sum(space)
            if total_space > 1e-15:
                h += d * space / total_space
            else:
                h += d / n
    h = np.clip(h, 0.0, 1.0)
    # Exact final fix
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + d)
    return h

def _obj(h, alpha, beta):
    n = len(h)
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = (m + np.log(se)) / alpha * alpha  # = m + log(se)/alpha... wait
    # Let me just use: f = m + log(sum(exp(alpha*(C-m)))) / alpha
    # = (alpha*m + log(se)) / alpha
    f = m + np.log(se) / alpha
    s = np.sum(h) - 0.5 * n
    f += beta * s * s
    return f, es / se

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = m + np.log(se) / alpha
    s = np.sum(h) - 0.5 * n
    f += beta * s * s
    w = es / se
    
    grad = np.zeros(n)
    # Vectorized gradient
    for i in range(n):
        j1 = i + np.arange(nc) - (n - 1)
        j2 = i - np.arange(nc) + (n - 1)
        h1 = np.where((j1 >= 0) & (j1 < n), h[np.clip(j1, 0, n - 1)], 0.0)
        h2 = np.where((j2 >= 0) & (j2 < n), h[np.clip(j2, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, 1.0 - h1 - h2)
    grad += 2.0 * beta * s
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=500):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'ftol': 1e-16})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _hot_set_descent(h, n, n_steps=200, base_step=0.01, time_limit=None, t0=None):
    """Subgradient descent on the true max using hot set."""
    best = h.copy()
    best_c5 = _c5_max(best)
    for step_i in range(n_steps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        C = _c5_array(best)
        cmax = np.max(C)
        # Adaptive threshold
        threshold = max(0.005, 0.05 * (cmax - np.min(C)))
        hot = (C >= cmax - threshold).astype(float)
        nhot = np.sum(hot)
        
        # Compute gradient of the average of hot C's
        dx = 2.0 / n
        nc = len(C)
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + np.arange(nc) - (n - 1)
            j2 = i - np.arange(nc) + (n - 1)
            h1 = np.where((j1 >= 0) & (j1 < n), best[np.clip(j1, 0, n - 1)], 0.0)
            h2 = np.where((j2 >= 0) & (j2 < n), best[np.clip(j2, 0, n - 1)], 0.0)
            grad[i] = dx * np.dot(hot, 1.0 - h1 - h2)
        
        # Center the gradient to maintain sum
        grad -= np.mean(grad)
        
        # Adaptive step size
        step = base_step / (1.0 + 0.1 * step_i)
        h_new = _fix(best - step * grad, n)
        c_new = _c5_max(h_new)
        
        if c_new < best_c5 - 1e-12:
            best = h_new
            best_c5 = c_new
        else:
            # Try smaller step
            step = step * 0.5
            h_new = _fix(best - step * grad, n)
            c_new = _c5_max(h_new)
            if c_new < best_c5 - 1e-12:
                best = h_new
                best_c5 = c_new
            else:
                base_step *= 0.95
                if base_step < 1e-6:
                    break
    return best, best_c5

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            space = 1.0 - h
            ts = np.sum(space)
            if ts > 1e-15:
                h += d * space / ts
            else:
                h += d / n
        else:
            space = h.copy()
            ts = np.sum(space)
            if ts > 1e-15:
                h += d * space / ts
            else:
                h += d / n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        if d > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
    return h

    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)

    f = m + np.log(se) / alpha
    s = np.sum(h) - 0.5 * n
    f += beta * s * s
    w = es / se
    grad = np.zeros(n)

        j1 = i + np.arange(nc) - (n - 1)
        j2 = i - np.arange(nc) + (n - 1)

        h2 = np.where((j2 >= 0) & (j2 < n), h[np.clip(j2, 0, n - 1)], 0.0)

                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'ftol': 1e-16})
