import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 + n_lag]
    return g

def soft_grad(h, corr, dx, alpha=20.0):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_lag(h, k_star - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-8:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g * dx

def sa_optimize(h, N, rng, t_end, try_update, n_iter=500000):
    """Simulated annealing on continuous space"""
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    best_h = h.copy()
    best_val = val
    T = 0.1
    T_min = 0.001
    cooling = 0.9999
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 10000 == 0:
            T = max(T_min, T * cooling)
        
        # Propose move: perturb 2-5 random positions
        n_perturb = rng.randint(2, 6)
        idx = rng.choice(N, size=n_perturb, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.randn(n_perturb) * 0.1 * T + 0.1
        h_new = project_feasible(h_new, N)
        
        val_new, corr_new, _ = obj_and_corr(h_new)
        delta = val_new - val
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            val = val_new
            corr = corr_new
            if val < best_val:
                best_val = val
                best_h = h.copy()
                try_update(h, val)
    
    return best_h, best_val

def compute_corr(h):
    N = len(h)
    corr = np.zeros(2*N - 1)
    one_minus_h = 1.0 - h
    for k in range(2*N - 1):
        # k corresponds to lag k - (N-1)
        lag = k - (N - 1)
        if lag >= 0:
            s = min(N, N - lag)
            corr[k] = np.dot(h[:s], one_minus_h[lag:lag+s])
        else:
            s = min(N, N + lag)
            corr[k] = np.dot(h[-lag:], one_minus_h[:s])
    return corr

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 + n_lag]
    return g

def soft_grad(h, corr, dx, alpha=20.0):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_lag(h, k_star - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-8:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g * dx

            val = val_new

                best_h = h.copy()
                try_update(h, val)
