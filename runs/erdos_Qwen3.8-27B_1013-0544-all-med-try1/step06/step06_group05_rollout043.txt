```python
    dx = 2.0 / n
    # Prefix sums
    ps = np.cumsum(h)

    c5 = max(c_pos.max(), c_neg.max()) * dx
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    n = len(h)
    # Bisection on lambda: sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = h.max() - target_sum - 1, h.min() + 1
    # Actually: f(lambda) = sum(clamp(h[i] - lambda, 0, 1))
    # f is decreasing in lambda
    # f(-inf) = n (all clamped to 1), f(inf) = 0
    # We want f(lambda) = target_sum
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def softmax_objective_and_grad(h, tau):
    """Compute soft-max C5 and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    max_val = scaled.max()
    exp_val = np.exp(tau * (scaled - max_val))
    p = exp_val / exp_val.sum()
    obj = max_val + np.log(exp_val.sum()) / tau
    
    # Gradient
    grad = np.zeros(n)
    # d=0 term
    grad += p[n-1] * (1 - 2*h)
    # Positive and negative shifts
    for d in range(1, n):
        pd = p[n-1+d]
        if pd > 1e-15:
            grad[:n-d] += pd * (1 - h[d:n])
            grad[d:] -= pd * h[:n-d]
        pd = p[n-1-d]
        if pd > 1e-15:
            grad[d:] += pd * (1 - h[:n-d])
            grad[:n-d] -= pd * h[d:n]
    grad *= dx
    return obj, grad

def smooth_optimize(h0, n, tau_start=5.0, tau_end=50.0, max_iter=500, step_start=0.1):
    """Smooth optimization with projected gradient"""
    h = h0.copy()
    target_sum = n / 2.0
    h = project_box_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        # Anneal tau
        frac = it / max_iter
        tau = tau_start * (tau_end / tau_start) ** frac
        
        obj, grad = softmax_objective_and_grad(h, tau)
        
        # Adaptive step size (backtracking)
        step = step_start * (1.0 + 0.5 / (1.0 + 0.1 * it))
        for _ in range(20):
            h_new = project_box_sum(h - step * grad, target_sum)
            obj_new, _ = softmax_objective_and_grad(h_new, tau)
            if obj_new < obj or step < 1e-10:
                break
            step *= 0.5
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if np.max(np.abs(grad)) < 1e-8:
            break
    
    return best_h, best_c5
```

1. Given p, find h that minimizes f_p(h) (subject to constraints)
2. Given h, find p that maximizes sum_k p_k * c_k(h) (= the max, so p concentrates on argmax)

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_c5_softmax(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    mv = scaled.max()
    exp_v = np.exp(tau * (scaled - mv))
    p = exp_v / exp_v.sum()
    grad = np.zeros(n)
    grad += p[n - 1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        pd = p[n - 1 + d]
        if pd > 1e-14:
            grad[:n - d] += pd * (1.0 - h[d:])
            grad[d:] -= pd * h[:n - d]
        pd = p[n - 1 - d]
        if pd > 1e-14:
            grad[d:] += pd * (1.0 - h[:n - d])
            grad[:n - d] -= pd * h[d:]
    return grad * dx, mv + np.log(exp_v.sum()) / tau

def smooth_optimize(h0, max_time, tau_start=3.0, tau_end=80.0, n_iter=800):
    h = project_box_sum(h0.copy(), len(h0) / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    n = len(h)
    step = 0.3
    for it in range(n_iter):
        if time.time() - t0 > max_time:
            break
        frac = min(it / max(n_iter - 1, 1), 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        grad, obj = grad_c5_softmax(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        lr = step / (1.0 + 0.02 * it)
        for _ in range(15):
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            obj_new, _ = grad_c5_softmax(h_new, tau)
            if obj_new <= obj + 1e-12:
                break
            lr *= 0.5
        else:
            h_new = project_box_sum(h - lr * grad, n / 2.0)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step = np.clip(step * 0.999, 0.01, 0.5)
    return best_h, best_c5

def local_search_cont(h0, max_time, seed=42):
    n = len(h0)
    h = project_box_sum(h0.copy(), n / 2.0)
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.RandomState(seed)
    t0 = time.time()
    stall = 0
    eps_base = 0.1
    while time.time() - t0 < max_time:
        n_trials = max(200, int(50000 / max(n, 1)))
        best_gain = 1e-12
        best_move = None
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            eps = rng.uniform(0.02, eps_base)
            if h[i] >= eps and h[j] <= 1.0 - eps:
                h[i] -= eps
                h[j] += eps
                new_c5 = compute_c5(h)
                gain = cur - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j, eps)
                h[i] += eps
                h[j] -= eps
        if best_move is not None:
            i, j, eps = best_move
            h[i] -= eps
            h[j] += eps
            h = project_box_sum(h, n / 2.0)
            cur = compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            eps_base = max(0.01, eps_base * 0.95)
        else:
            stall += 1
            eps_base = max(0.005, eps_base * 0.7)
            if stall > 8:
                break
    return best_h, best_c5

def binary_local_search(h0, max_time, seed=42):
    n = len(h0)
    h = np.round(h0).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        rng = np.random.RandomState(seed)
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        rng = np.random.RandomState(seed)
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h[idx] = 1
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.RandomState(seed + 1)
    t0 = time.time()
    while time.time() - t0 < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_swaps = min(2000, len(ones) * len(zeros))
        best_gain = 1e-12
        best_i, best_j = -1, -1
        for _ in range(n_swaps):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            new_c5 = compute_c5(h)
            gain = cur - new_c5
            if gain > best_gain:
                best_gain = gain
                best_i, best_j = i, j
            h[i] = 1
            h[j] = 0
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            cur = compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def make_thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n / 2.0
    t = project_box_sum(t, target)
    return t

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except NameError:
            h_init = None
    else:
        h_init = None

    n_list = [128, 160, 200, 256, 320, 400]

    for n in n_list:
        if remaining() < 60:
            break
        t_init = min(15.0, remaining() * 0.05)
        t_smooth = min(remaining() * 0.45, 200.0)
        t_local = min(remaining() * 0.25, 120.0)
        t_binary = min(remaining() * 0.1, 60.0)
        if t_smooth < 10:
            break

        if h_init is not None:
            h0 = interpolate_to_n(h_init, n)
        else:
            h0 = make_thue_morse(n)

        # Smooth optimization
        h_s, c5_s = smooth_optimize(h0, max_time=t_smooth,
                                    tau_start=3.0 + 0.5 * (n // 100),
                                    tau_end=60.0 + 2.0 * (n // 100),
                                    n_iter=600)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n

        # Local continuous refinement
        if remaining() > 30:
            h_l, c5_l = local_search_cont(h_s, max_time=min(t_local, remaining() - 20),
                                          seed=seed + n)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
                best_n = n

        # Try binary rounding + binary local search
        if remaining() > 20 and n >= 200:
            h_b, c5_b = binary_local_search(h_s, max_time=min(t_binary, remaining() - 10),
                                            seed=seed + 100 + n)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n

    # Final polish on best solution
    if remaining() > 10 and best_h is not None:
        is_binary = np.allclose(best_h, np.round(best_h), atol=0.01)
        if is_binary:

                                            seed=seed + 9999)
        else:

                                          seed=seed + 9999)

            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(h_f)

    # Ensure valid
    if best_h is None:
        best_h = np.full(128, 0.5)
        best_n = 128
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
```
