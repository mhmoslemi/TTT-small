def project_feasible(v, N):
    """Project v onto {h | 0<=h[i]<=1, sum(h)=N/2}"""
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Fix rounding
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_value(h):
    N = len(h)
    dx = 2.0 / N
    return compute_corr(h).max() * dx

def grad_single_lag(h, lag):
    """Gradient of corr[lag] with respect to h"""
    N = len(h)
    g = np.zeros(N)
    # corr[lag] = sum_i h[i] * (1-h[i+lag]) for valid i
    # d/dh[j] of this:
    # If j appears as h[i]: contributes (1-h[j+lag]) for valid j+lag
    # If j appears as h[i+lag]: contributes -h[j-lag] for valid j-lag
    s1 = max(0, lag)
    e1 = min(N, N + lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - lag:e1 - lag]
    s2 = max(0, -lag)
    e2 = min(N, N - lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + lag:e2 - lag]
    return g

def soft_gradient(h, corr, dx, alpha):
    """Weighted gradient using softmax over correlation values"""
    N = len(h)
    c = corr - corr.max()
    # Shift to avoid underflow
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_single_lag(h, k_star - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-6:
            continue
        g += w[j] * grad_single_lag(h, j - (N - 1))
    return g * dx

Simulated annealing (SA) works well for combinatorial optimization. The idea:

- Accept: if new value < old value, accept; otherwise accept with probability exp(-delta/T)
- Cool: decrease T over time

- Actually, this doesn't preserve the constraint sum(h) = N/2 unless we're careful

- If I change h[i] by +d and h[j] by -d (where both stay in [0,1]), the sum is preserved

4. Accept with Metropolis criterion

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if m.sum() > 0:
            h[m] += diff / m.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_obj(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx, c, dx

def grad_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    s1, e1 = max(0, lag), min(N, N + lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - lag:e1 - lag]
    s2, e2 = max(0, -lag), min(N, N - lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + lag:e2 - lag]
    return g

def smooth_grad(h, c, dx, alpha):
    N = len(h)
    w = np.exp(alpha * (c - c.max()))
    ws = w.sum()
    if ws < 1e-300:

    for j in range(len(w)):
        if w[j] > 1e-7:
            g += w[j] * grad_lag(h, j - (N - 1))
    return g * dx

def interp(hs, nd):
    x0 = np.linspace(0, 1, len(hs), endpoint=False) + 0.5 / len(hs)
    x1 = np.linspace(0, 1, nd, endpoint=False) + 0.5 / nd
    return project_feasible(np.interp(x1, x0, hs), nd)
