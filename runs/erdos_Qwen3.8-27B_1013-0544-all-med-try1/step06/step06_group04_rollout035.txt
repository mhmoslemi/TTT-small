def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    # g has length 2n-1, indexed 0 to 2n-2
    # g[k] corresponds to c_{k-(n-1)} in the correlation
    # i.e., g[k] = c_{k-(n-1)} * dx where c_k = sum h[n+k](1-h[n])
    
    # Softmax weights
    k_array = np.arange(2*n-1) - (n-1)  # actual lag k for each entry
    g_max = np.max(g)
    weights = np.exp(T * (g - g_max))
    weights /= np.sum(weights)
    
    grad_h = np.zeros(n)
    for j in range(n):
        for idx in range(2*n-1):
            k = k_array[idx]  # lag
            # First term: (1-h[j-k]) if 0 <= j-k < n
            if 0 <= j-k < n:
                grad_h[j] += weights[idx] * (1.0 - h[j-k])
            # Second term: -h[j+k] if 0 <= j+k < n
            if 0 <= j+k < n:
                grad_h[j] -= weights[idx] * h[j+k]
        grad_h[j] *= dx
    
    return grad_h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0-indexed)."""
    tm = np.zeros(n)
    for i in range(n):
        # Count bits in binary representation of i
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm

            end = start + block_size if b < n_blocks - 1 else n
            h[start:end] = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr * dx)

def project_onto_constraints(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    n_corr = len(g)
    g_max = np.max(g)
    exp_g = np.exp(T * (g - g_max))
    sum_exp = np.sum(exp_g)
    value = g_max + np.log(sum_exp) / T
    weights = exp_g / sum_exp
    W = weights
    grad_h = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        k_lo = max(-(n - 1), j - (n - 1))
        k_hi = min(n - 1, j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            idx = ks + (n - 1)
            m = j - ks
            grad_h[j] += np.sum(W[idx] * one_minus_h[m])
        k_lo2 = max(-(n - 1), -j)
        k_hi2 = min(n - 1, n - 1 - j)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            idx2 = ks2 + (n - 1)
            m2 = j + ks2
            grad_h[j] -= np.sum(W[idx2] * h[m2])
    grad_h *= dx
    return value, grad_h

def pgd_optimize(h0, T_start, max_iters, deadline):
    n = len(h0)
    target_sum = float(n) / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_c5 = best_c5
    step = 0.01
    T = T_start
    for it in range(max_iters):
        if time.time() > deadline:
            break
        val, grad = compute_gradient(h, T)
        T = max(T * 0.999, 3.0)
        h = project_onto_constraints(h - step * grad, target_sum)
        c5_curr = compute_c5(h)
        if c5_curr < best_c5:
            best_h = h.copy()
            best_c5 = c5_curr
        if c5_curr <= prev_c5 + 1e-12:
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.75
        prev_c5 = c5_curr
    return best_h, best_c5

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm

def make_block_pattern(n, n_blocks):
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            h[start:end] = 1.0
    return h

def make_sinusoid(n, freq, amp):
    x = np.arange(n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x / n)
    return np.clip(h, 0.0, 1.0)

def refine_solution(h, deadline, max_iters=20000):
    n = len(h)
    target_sum = float(n) / 2.0
    best_c5 = compute_c5(h)
    T = 10.0
    step = 0.003
    prev_c5 = best_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        val, grad = compute_gradient(h, T)
        T = max(T * 0.998, 2.0)
        h = project_onto_constraints(h - step * grad, target_sum)
        c5_curr = compute_c5(h)
        if c5_curr < best_c5:
            best_c5 = c5_curr
        if c5_curr <= prev_c5 + 1e-13:
            step = min(step * 1.02, 0.02)
        else:
            step *= 0.7
        prev_c5 = c5_curr
    return h, best_c5
