def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    """Compute all correlation values (unmultiplied by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining discrepancy
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        # Distribute to elements with room
        if diff > 0:
            idx = np.argsort(1.0 - h)[-int(abs(diff)*n)//1 + 1:]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            idx = np.argsort(h)[-int(abs(diff)*n)//1 + 1:]
            for i in idx:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

def subgradient_step(h, step_size, top_k=5):
    """Take a subgradient step to reduce the max correlation"""
    n = len(h)
    corr = compute_corr(h)
    # Find top_k correlation indices
    top_indices = np.argsort(corr)[-top_k:]
    
    # Compute subgradient for each coordinate
    subgrad = np.zeros(n)
    for m in top_indices:
        s = m - (n - 1)  # actual shift
        for i in range(max(0, -(n-1-s)), min(n, n-1-s+1)):
            # h[i] contributes to corr at shift s
            # First factor: h[i] * (1-h[i+s]) if 0<=i+s<n
            if 0 <= i + s < n:
                subgrad[i] += (1.0 - h[i+s])
            # Second factor: h[i-s] * (1-h[i]) if 0<=i-s<n
            if 0 <= i - s < n:
                subgrad[i] -= h[i-s]
    
    # Step in negative subgradient direction
    h_new = h - step_size * subgrad
    
    # Project onto feasible set
    h_new = normalize_continuous(h_new, n)
    return h_new

    corr = compute_corr_array(h)
    top_idx = np.argsort(corr)[-top_k:]
    pressure = np.zeros(n)

        for i in range(n):
            # Contribution to pressure from shift s
            # a_s for coordinate i: (1-h[i+s])*1{valid} - h[i-s]*1{valid}
            if 0 <= i + s < n:

            if 0 <= i - s < n:
                pressure[i] -= h[i-s]
    
    return pressure

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(1.0 - h[idx], diff)
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(h[idx], -diff)
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

def compute_pressure(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    pressure = np.zeros(n)
    for m in top_idx:
        s = int(m) - (n - 1)
        for i in range(n):
            if 0 <= i + s < n:
                pressure[i] += (1.0 - h[i + s])
            if 0 <= i - s < n:
                pressure[i] -= h[i - s]
    return pressure, corr

def greedy_pressure_search(h0, max_time, seed=42, top_k=5):
    n = len(h0)
    h = normalize_continuous(h0, n)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    stall = 0
    n_cands = max(4, n // 12)

    while (time.time() - start) < max_time:
        pressure, corr = compute_pressure(h, n, top_k)
        max_corr = float(np.max(corr))
        best_gain = 1e-12
        best_move = None
        deltas = [0.03, 0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]

        i_cands = np.argsort(pressure)[-n_cands:]
        j_cands = np.argsort(pressure)[:n_cands]

        for i in i_cands:
            for d in deltas:
                if h[i] < d:
                    continue
                for j in j_cands:
                    if i == j or h[j] + d > 1.0:
                        continue
                    h[i] -= d
                    h[j] += d
                    new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    h[i] += d
                    h[j] -= d
                    gain = max_corr - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j, d)

        for _ in range(15):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.02, 0.3)
            if h[i] < d or h[j] + d > 1.0:
                continue
            h[i] -= d
            h[j] += d
            new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            h[i] += d

        if best_move is not None:

            h[i] -= d
