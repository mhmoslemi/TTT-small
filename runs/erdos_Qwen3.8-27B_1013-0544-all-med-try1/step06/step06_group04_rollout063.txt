```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def c5_objective_and_gradient(h, n, T=50.0, penalty_weight=100.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Smooth max
    tc = T * corr
    max_tc = np.max(tc)
    shifted = tc - max_tc
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    weights = exp_shifted / Z  # softmax weights
    obj = (max_tc + np.log(Z)) / T
    
    # Gradient of smooth max
    s = weights  # length 2n-1
    s_rev = s[::-1]
    conv1 = np.convolve(h, s, 'full')
    conv2 = np.convolve(h, s_rev, 'full')
    grad_sm = T * (1.0 - conv1[n-1:2*n-1] - conv2[n-1:2*n-1])
    
    # Penalty for sum constraint
    sum_h = np.sum(h)
    target = n / 2.0
    penalty = penalty_weight * (sum_h - target) ** 2
    grad_pen = 2.0 * penalty_weight * (sum_h - target) * np.ones(n)
    
    total_obj = obj + penalty
    total_grad = grad_sm + grad_pen
    
    return total_obj, total_grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 0:
                h += h * (diff / total)
    h = np.clip(h, 0.0, 1.0)
    # Final projection to ensure exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly among non-bound variables
        free = (h > 1e-8) & (h < 1 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 70
        h0 = np.zeros(n0)
        h0[:n0//2] = 1.0
    
    n0 = len(h0)
    h0 = np.clip(h0, 0, 1)
    h0 = project_feasible(h0, n0)
    
    c5_0, _ = compute_c5(h0, n0)
    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n0
    
    # Optimization at different resolutions
    for n in [140, 280, 560, 1120]:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 60:
            break
        
        # Upsample from best
        h = np.repeat(best_h, n // len(best_h))
        # Adjust length if needed
        if len(h) != n:
            h = h[:n]
        h = project_feasible(h, n)
        
        c5_start, _ = compute_c5(h, n)
        
        # Optimize with L-BFGS-B
        T = 30.0
        penalty_w = 50.0
        
        def obj_wrapper(x):
            x = np.clip(x, 0, 1)
            o, g = c5_objective_and_gradient(x, n, T, penalty_w)
            return o
        
        def grad_wrapper(x):
            x = np.clip(x, 0, 1)
            o, g = c5_objective_and_gradient(x, n, T, penalty_w)
            return g
        
        bounds = [(0.0, 1.0)] * n
        max_iter = int(min(remaining * 0.3, 200))
        
        try:
            result = minimize(obj_wrapper, h, jac=grad_wrapper,
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': max_iter, 'ftol': 1e-14,
                                      'gtol': 1e-8, 'maxfun': max_iter * n})
            h_opt = np.clip(result.x, 0, 1)
            h_opt = project_feasible(h_opt, n)
            c5_opt, _ = compute_c5(h_opt, n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
        except Exception:
            pass
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def c5_obj_grad(h, n, T, pw):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    tc = T * corr
    m = np.max(tc)
    zs = np.exp(tc - m)
    Z = np.sum(zs)
    w = zs / Z
    obj = (m + np.log(Z)) / T
    
    s = w
    srev = s[::-1]
    c1 = np.convolve(h, s, mode='full')
    c2 = np.convolve(h, srev, mode='full')
    idx = np.arange(n-1, 2*n-1)
    grad = T * (1.0 - c1[idx] - c2[idx])
    
    sh = np.sum(h)
    tgt = n / 2.0
    obj += pw * (sh - tgt)**2
    grad += 2.0 * pw * (sh - tgt)
    
    return obj, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    d = tgt - np.sum(h)
    if abs(d) > 1e-14:
        if d > 0:
            r = 1.0 - h
            tr = np.sum(r)
            if tr > 0:
                h = h + r * (d / tr)
        else:
            th = np.sum(h)
            if th > 0:
                h = h + h * (d / th)
    h = np.clip(h, 0.0, 1.0)
    return h
```
