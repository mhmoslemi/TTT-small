import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Need to increase; add to elements with most room
            order = np.argsort(1.0 - h)  # most room first
            for i in range(n):
                room = 1.0 - h[order[i]]
                add = min(room, diff)
                if add <= 0:
                    break
                h[order[i]] += add
                diff -= add
        else:
            # Need to decrease
            order = np.argsort(h)  # least h first (easiest to remove)
            # Actually, we want to remove from elements with most h
            order = np.argsort(-h)
            for i in range(n):
                rem = h[order[i]]
                sub = min(rem, -diff)
                if sub <= 0:
                    break
                h[order[i]] -= sub
                diff += sub
    return h

def lse_objective(h, tau):
    corr = compute_corr(h)
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(tau * (corr - max_c)))) / tau

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # raw, not scaled by dx
    # Softmax weights
    tc = tau * corr * dx
    max_tc = np.max(tc)
    w = np.exp(tc - max_tc)
    w /= np.sum(w)
    
    # Compute gradient: sum_m w[m] * d(output[m])/dh[j]
    # output[m] = sum_i h[i+lag_m] * (1-h[i]) where lag_m = m - (n-1)
    # d(output[m])/dh[j] = (1-h[j-lag_m])*I[0<=j-lag_m<n] + (-h[j+lag_m])*I[0<=j+lag_m<n]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-15:
            continue
        lag = m - (n-1)
        for j in range(n):
            # (1-h[j-lag]) if 0 <= j-lag < n
            jp = j - lag
            if 0 <= jp < n:
                grad[j] += wm * (1.0 - h[jp])
            # -h[j+lag] if 0 <= j+lag < n
            jm = j + lag
            if 0 <= jm < n:
                grad[j] += wm * (-h[jm])
    
    return dx * grad

    grad_lse = np.zeros(n)

        if je2 > js2:

    penalty = (np.sum(h) - n/2.0)
    grad_pen = 2.0 * mu * penalty * np.ones(n)

    return obj, grad_lse + grad_pen

    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
