    n = len(h)

    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _subgrad_single(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        grad[0:n-k] += (1.0 - h[k:n])
        if k > 0:
            grad[k:n] -= h[0:n-k]
    else:
        m = -k
        grad[m:n] += (1.0 - h[0:n-m])
        if m > 0:
            grad[0:n-m] -= h[m:n]
    return grad

def soft_max_obj(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    return t * (m + t * np.log(np.sum(np.exp((conv - m) / t)))) * dx

def soft_max_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        grad += w * _subgrad_single(h, k, n)
    return grad * dx

def obj_fn(h, t, n):
    return soft_max_obj(h, t, n) + 50.0 * (h.sum() - n * 0.5) ** 2

def grad_fn(h, t, n):
    return soft_max_grad(h, t, n) + 100.0 * (h.sum() - n * 0.5)

def project_feasible(h, n):
    target = n * 0.5
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def pgd_smooth(h_init, lr, momentum, max_iter, t):
    n = len(h_init)
    h = h_init.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        _, grad = soft_max_grad(h, t, n)
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_curr = compute_c5(h)
        if c5_new > c5_curr + 1e-13:
            lr = max(lr * 0.7, 1e-9)
        else:
            lr = min(lr * 1.03, 0.04)
        h_prev = h
        h = h_new
    return best_h, best_c5

def pgd_true(h_init, lr, momentum, max_iter):
    n = len(h_init)
    h = h_init.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(conv)
        # Weight shifts by proximity to max (soft assignment)
        thresh = max_val - max(0.3, 0.05 * max_val)
        active = np.where(conv >= thresh)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += _subgrad_single(h, k, n)
        grad /= max(len(active), 1)
        grad *= (2.0 / n)
        
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_curr = compute_c5(h)
        if c5_new > c5_curr + 1e-14:
            lr = max(lr * 0.6, 1e-10)
        else:
            lr = min(lr * 1.02, 0.02)
        h_prev = h
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def tl():
        return max(0, budget_s - (time.time() - t0))
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(64, 0.5)
    
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B with decreasing temperature
    for t_val in [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001]:
        if tl() < 30:
            break
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, args=(t_val, n0),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n0,
                          options={'maxiter': 4000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 50})
            ht = project_feasible(res.x, n0)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht.copy(), c5t
        except Exception:
            pass
    
    # Phase 2: PGD smooth with decreasing t
    for t_val in [0.01, 0.005, 0.003, 0.001, 0.0005]:
        if tl() < 25:
            break
        ht, c5t = pgd_smooth(best_h, lr=0.002, momentum=0.85, max_iter=15000, t=t_val)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 3: True subgradient with decreasing steps
    for lr_val in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if tl() < 25:
            break
        ht, c5t = pgd_true(best_h, lr=lr_val, momentum=0.85, max_iter=25000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 4: Multi-resolution
    for n_try in [400, 480, 560]:
        if tl() < 120:
            break
        nc = len(best_h)
        if nc >= n_try:
            continue
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        for t_val in [0.005, 0.002]:
            if tl() < 50:
                break
            try:
                res = minimize(obj_fn, hi, jac=grad_fn, args=(t_val, n_try),
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_try,
                              options={'maxiter': 3000, 'ftol': 1e-16, 'maxcor': 40})
                ht = project_feasible(res.x, n_try)
            except Exception:
                ht = hi.copy()
            ht, c5t = pgd_smooth(ht, lr=0.001, momentum=0.85, max_iter=10000, t=0.003)
            ht, c5t2 = pgd_true(ht, lr=0.0003, momentum=0.85, max_iter=10000)
            if c5t2 < c5t:
                ht, c5t = ht, c5t2
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
    
    # Phase 5: Local perturbations of best solution
    for _ in range(50):
        if tl() < 15:
            break
        noise_scale = max(0.01, 0.1 * (1.0 - (time.time() - t0) / budget_s))
        hp = best_h + rng.uniform(-noise_scale, noise_scale, n0)
        hp = project_feasible(hp, n0)
        ht, c5t = pgd_smooth(hp, lr=0.003, momentum=0.85, max_iter=5000, t=0.005)
        ht, c5t2 = pgd_true(ht, lr=0.0003, momentum=0.85, max_iter=5000)
        if c5t2 < c5t:
            ht, c5t = ht, c5t2
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 6: Random restarts
    for _ in range(20):
        if tl() < 15:
            break
        rtype = _ % 4
        if rtype == 0:
            hr = rng.uniform(0.3, 0.7, n0)
        elif rtype == 1:
            freq = rng.randint(2, 15)
            hr = 0.5 + 0.25 * np.cos(2 * np.pi * freq * np.arange(n0) / n0 + rng.uniform(0, 6.28))
        elif rtype == 2:
            hr = np.zeros(n0)
            block = n0 // 8
            for b in range(8):
                hr[b*block:(b+1)*block] = rng.uniform(0.2, 0.8)
        else:
            hr = rng.dirichlet(np.ones(n0)) * n0 * 0.5 / (rng.dirichlet(np.ones(n0)) * n0 * 0.5).sum()
            hr = np.clip(hr, 0, 1)
        hr = project_feasible(hr, n0)
        ht, c5t = pgd_smooth(hr, lr=0.005, momentum=0.85, max_iter=4000, t=0.03)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Final ultra-fine polish
    if tl() > 20:
        ht, c5t = pgd_true(best_h, lr=0.0001, momentum=0.92, max_iter=40000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    return best_h, best_c5, len(best_h)
