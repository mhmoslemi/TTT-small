import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            # Need to move some values
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
            continue
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_of_corr_lag(h, k):
    """Compute gradient of c(k) = sum_i h[i]*(1-h[i+k]) with respect to h.
    k is the actual lag (can be negative)."""

    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:

        # = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]

        # From first sum: 1 if 0 <= j <= n-1-k
        # From second sum: -h[j+k] if 0 <= j <= n-1-k, -h[j-k] if k <= j <= n-1

        valid2 = (np.arange(n) >= k) & (np.arange(n) <= n - 1)
        grad[valid1] += 1.0
        grad[valid1] -= h[valid1 + k]  # h[j+k] for j in valid1
        grad[valid2] -= h[valid2 - k]  # h[j-k] for j in valid2
    else:

        # = sum_{i=k'}^{n-1} h[i]*(1-h[i-k'])

        grad[valid1] -= h[valid1 - kp]  # h[i-k'] for i in valid1
        grad[valid2] -= h[valid2 + kp]  # h[i+k'] for i in valid2
    return grad

def soft_max_indices(corr, alpha):
    """Get weights for soft-max over correlation values."""
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()
    return weights

def subgradient_descent(h, n_steps, lr_init, lr_decay, alpha_smoothing=10.0):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = compute_corr(h)
    dx = 2.0 / n
    
    for step in range(n_steps):
        lr = lr_init * (lr_decay ** step)
        
        # Soft-max weights over lags
        weights = soft_max_indices(corr, alpha_smoothing)
        
        # Weighted subgradient
        max_idx = np.argmax(corr)
        k_max = max_idx - (n - 1)
        grad = subgradient_of_corr_lag(h, k_max)
        
        # Also add contributions from other active lags (top few)

def coord_descent(h, n_levels=20, max_sweeps=20):

    ts = n / 2.0

            if best_c < c5 - 1e-14:
                h = best_h
                c5 = best_c
                improved = True
            else:
                h = orig
        if not improved:

    return h, c5

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(0.01 * temp * (corr - m))
    s[s < 1e-300] = 1e-300
    return (m + np.log(s.sum()) / (0.01 * temp)) * dx

def optimize_slsqp(h_start, n, temp, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h_start, args=(temp,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
            continue
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n)

    elif k > 0:

        m = np.max(corr)

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = 0.01 * temp
    shifted = corr - m
    shifted[shifted < -50.0/t] = -50.0/t
    s = np.exp(t * shifted)
    return (m + np.log(s.sum()) / t) * dx

def optimize_slsqp(h_start, n, temp, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h_start, args=(temp,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})

            h_bp = np.zeros(n0)
            h_bp[rng.choice(n0, n0 // 2, replace=False)] = 1.0
            h_bp += rng.randn(n0) * 0.08
            h_r = project_feasible(np.clip(h_bp, 0, 1), n0 / 2.0)
        h_r = optimize_slsqp(h_r, n0, 15.0, maxiter=250)
        c = compute_c5(h_r)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_h = h_r.copy()
            if remaining() > 50:
                h_cd, c2 = coord_descent(best_h, n_levels=15, max_sweeps=8)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cd

    # Phase 5: Resolution sweep
    for n_try in [256, 320, 448, 512, 640]:
        if remaining() < 70:
            break
        x_old = np.linspace(0, 1, len(best_h), endpoint=False)
        x_new = np.linspace(0, 1, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try / 2.0)
        for temp in [30.0, 8.0]:
            if remaining() < 50:
                break
            h_new = optimize_slsqp(h_new, n_try, temp, maxiter=300)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
        if remaining() > 30 and len(best_h) == n_try:
            h_cd, c2 = coord_descent(best_h, n_levels=15, max_sweeps=8)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h_cd

    # Phase 6: Final high-precision coordinate descent
    if remaining() > 25:
        h_cd, c = coord_descent(best_h, n_levels=40, max_sweeps=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd

    # Phase 7: Final SLSQP at very low temp
    if remaining() > 15:
        h_fin = optimize_slsqp(best_h, len(best_h), 0.5, maxiter=300)
        c = compute_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            best_h = h_fin

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
