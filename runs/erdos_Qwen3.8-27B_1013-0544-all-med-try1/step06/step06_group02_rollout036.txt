    # Bisection on subgradient
    for _ in range(n_iter):
        mid = (lo + hi) / 2
        new_corr = corr + mid * effect
        max_idx = np.argmax(new_corr)  # or argmax with tolerance
        # Check subgradient
        max_val = new_corr[max_idx]
        # Find all indices at the max (within tolerance)
        at_max = np.where(new_corr >= max_val - 1e-10)[0]
        slopes = effect[at_max]
        if slopes.min() <= 0 <= slopes.max():
            return mid  # 0 is in subgradient
        elif slopes.max() < 0:
            lo = mid  # all slopes negative, need to increase delta
        else:
            hi = mid  # all slopes positive, need to decrease delta
    return (lo + hi) / 2

    """Project v onto {h: 0≤h[i]≤1, sum(h)=N/2}"""
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-14:

            h = np.clip(h, 0.0, 1.0)

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)

    if abs(diff) > 1e-14:

        if free.sum() > 0:
            h[free] += diff / free.sum()

    return effect

def coord_descent(h, N, max_sweeps=100, t_end=None):
    dx = 2.0 / N
    corr = compute_corr(h)
    best_val = corr.max() * dx
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        corr = compute_corr(h)
        current_max = corr.max()
        improved = False
        for j in range(N):
            eff = effect_at_j(h, N, j)
            d = find_best_delta(corr, eff, -h[j], 1.0 - h[j])
            if abs(d) > 1e-14:
                new_max = (corr + d * eff).max()
                if new_max < current_max - 1e-14:
                    h[j] += d
                    corr += d * eff
                    current_max = new_max
                    improved = True
        if current_max * dx < best_val:
            best_val = current_max * dx
        if not improved:
            break
    corr = compute_corr(h)
    best_val = min(best_val, corr.max() * dx)
    return h, best_val

def interpolate_h(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def targeted_perturb(h, N, rng):
    corr = compute_corr(h)
    k_star = np.argmax(corr) - (N - 1)
    h_try = h.copy()
    if k_star >= 0:
        candidates = [i for i in range(N - k_star) if h[i] > 0.5 and h[i + k_star] < 0.5]
    else:
        candidates = [i for i in range(-k_star, N) if h[i] > 0.5 and h[i + k_star] < 0.5]
    if len(candidates) > 0:
        n_mod = min(rng.randint(1, 4), len(candidates))
        for _ in range(n_mod):
            idx = candidates[rng.randint(len(candidates))]
            h_try[idx] = h[idx] * 0.2
    h_try = project_feasible(h_try, N)
    return h_try

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_val = float('inf')

    if h_init is not None:
        best_val = compute_obj(h_init)
        best_h = h_init.copy()

    # Phase 1: Coordinate descent on initial solution
    if best_h is not None:
        N = len(best_h)
        t_phase = min(t0 + 180, t_end - 250)
        h, val = coord_descent(best_h.copy(), N, max_sweeps=300, t_end=t_phase)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # Phase 2: Multi-resolution exploration
    n_list = [47, 64, 80, 96, 120, 144]
    for n_t in n_list:
        if time.time() > t_end - 350:
            break
        if best_h is not None:
            h_start = interpolate_h(best_h, n_t)
        else:
            h_start = 0.5 * np.ones(n_t)
        t_phase = min(time.time() + 80, t_end - 200)
        h, val = coord_descent(h_start, n_t, max_sweeps=150, t_end=t_phase)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # Phase 3: Perturbed local search
    if best_h is not None and time.time() < t_end - 60:
        N = len(best_h)
        for restart in range(400):
            if time.time() > t_end - 20:
                break
            r = rng.rand()
            if r < 0.30:
                scale = 0.03 + 0.20 * min(restart / 100.0, 1.0)
                h_try = project_feasible(best_h + rng.randn(N) * scale, N)
            elif r < 0.55:
                h_try = best_h.copy()
                n_swaps = rng.randint(1, 5)
                for _ in range(n_swaps):
                    i, j = rng.choice(N, 2, replace=False)
                    h_try[i], h_try[j] = h_try[j], h_try[i]
            elif r < 0.75:
                h_try = targeted_perturb(best_h, N, rng)
            elif r < 0.90:
                h_try = best_h.copy()
                n_change = rng.randint(1, 4)
                for _ in range(n_change):
                    i = rng.randint(N)
                    h_try[i] = rng.rand()
                h_try = project_feasible(h_try, N)
            else:
                # Random binary start
                h_try = np.zeros(N)
                h_try[rng.choice(N, N // 2, replace=False)] = 1.0

            t_phase = min(time.time() + 25, t_end - 20)
            h, val = coord_descent(h_try, N, max_sweeps=50, t_end=t_phase)
            if val < best_val:
                best_val = val
                best_h = h.copy()

    # Phase 4: Final refinement with extended coordinate descent
    if best_h is not None and time.time() < t_end - 5:
        N = len(best_h)
        h, val = coord_descent(best_h.copy(), N, max_sweeps=1000, t_end=t_end - 2)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # Try binary rounding as a final check
    if best_h is not None:
        N = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        if abs(h_bin.sum() - N / 2.0) > 0.01:
            # Adjust to meet constraint
            diff = N / 2.0 - h_bin.sum()
            if diff > 0:
                zeros = np.where(h_bin == 0)[0]
                if len(zeros) >= int(diff):
                    h_bin[zeros[:int(diff)]] = 1.0
            elif diff < 0:
                ones = np.where(h_bin == 1)[0]
                if len(ones) >= int(-diff):
                    h_bin[ones[:int(-diff)]] = 0.0
        val_bin = compute_obj(h_bin)
        if val_bin < best_val:
            best_val = val_bin
            best_h = h_bin

    if best_h is None:
        best_h = 0.5 * np.ones(47)
        best_val = compute_obj(best_h)

    return best_h, best_val, len(best_h)
