        for attempt in range(3):
            if time.time() > t_end - 40:
                break
            h_start = random_binary(n_try, rng)
            t_phase = min(time.time() + 20, t_end - 40)
            h, v = optimize(h_start, n_try, rng, t_phase, try_update)
            try_update(h, v)
    
    # Phase 6: Final fine-tuning with very small steps
    if best_h is not None and time.time() < t_end:
        N_best = len(best_h)
        h = best_h.copy()
        val, corr, dx = obj_and_corr(h)
        lr = 0.004
        v = np.zeros(N_best)
        for it in range(200000):
            if time.time() > t_end:
                break
            g = adaptive_grad(h, corr, dx, 40.0, 2)
            v = 0.75 * v + g
            h_new = project_feasible(h - lr * v, N_best)
            val_new, corr_new, _ = obj_and_corr(h_new)
            if val_new < val - 1e-16:
                h = h_new
                val = val_new
                corr = corr_new
                try_update(h, val)
            else:
                lr *= 0.93
                if lr < 0.00005:
                    break
    
    if best_h is None:
        best_h = 0.5 * np.ones(80)
        best_val = obj_and_corr(best_h)[0]
    
    return best_h, best_val, len(best_h)
