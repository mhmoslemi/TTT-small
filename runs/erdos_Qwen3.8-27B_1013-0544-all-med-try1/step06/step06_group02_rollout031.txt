import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(h):
    return get_corr(h).max()

def subgrad_topk(h, corr, k=3):
    N = len(h)
    top = np.argsort(corr)[-k:]
    g = np.zeros(N)
    for idx in top:
        lag = int(idx) - (N - 1)
        if lag >= 0:
            g[lag:] += 1.0 - h[:N - lag]
            g[:N - lag] -= h[lag:]
        else:
            l = -lag
            g[l:] += 1.0 - h[:N - l]
            g[:N - l] -= h[l:]
    return g / k

                lr = 0.05

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_val(corr):
    return corr.max()

def subgrad_topk(h, corr, k=3):
    N = len(h)
    top = np.argsort(corr)[-k:]
    g = np.zeros(N)
    for idx in top:
        lag = int(idx) - (N - 1)
        if lag >= 0:
            g[lag:] += 1.0 - h[:N - lag]
            g[:N - lag] -= h[lag:]
        else:
            l = -lag
            g[l:] += 1.0 - h[:N - l]
            g[:N - l] -= h[l:]
    return g / k

def gradient_descent(h, rng, t_end, k=3, max_iter=200000):
    N = len(h)
    h = h.copy()
    corr = get_corr(h)
    val = corr.max()
    lr = 0.04
    vel = np.zeros(N)
    mu = 0.9
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = subgrad_topk(h, corr, k)
        vel = mu * vel - lr * g
        h_new = project_feasible(h - vel, N)
        corr_new = get_corr(h_new)
        val_new = corr_new.max()
        if val_new < val - 1e-15:
            h = h_new
            corr = corr_new
            val = val_new
            stagnant = 0
            lr = min(lr * 1.005, 0.12)
        else:
            stagnant += 1
            lr *= 0.97
            vel *= 0.9
            if stagnant > 100:
                h = project_feasible(h + rng.randn(N) * 0.03, N)
                corr = get_corr(h)
                val = corr.max()
                lr = 0.04
                vel = np.zeros(N)
                stagnant = 0
    
    return h, val

def binarize(h, N):
    h_bin = (h > 0.5).astype(float)
    target = N // 2
    count = int(h_bin.sum())
    if count != target:
        if count > target:
            excess = count - target
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])[:excess]
            h_bin[ones[order]] = 0
        else:
            deficit = target - count
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])[:deficit]
            h_bin[zeros[order]] = 1
    return h_bin
