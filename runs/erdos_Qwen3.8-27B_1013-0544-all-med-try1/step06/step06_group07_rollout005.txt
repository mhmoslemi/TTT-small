import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def project_box_sum(y, target):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def c_values(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def exact_c(h):
    return float(np.max(c_values(h)))

def lag_grad(h, k):
    n = h.size
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def softmax_obj(h, tau):
    c = c_values(h)
    cm = c.max()
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= z.max()
    e = np.exp(z)
    return cm + tau * np.log(e.sum())

def softmax_grad(h, tau):
    c = c_values(h)
    cm = c.max()
    n = h.size
    dx = 2.0 / n
    if tau <= 0:
        k = int(np.argmax(c))
        return lag_grad(h, k) * dx
    z = (c - cm) / tau
    z -= z.max()
    e = np.exp(z)
    w = e / e.sum()
    g = np.zeros(n)
    idxs = np.where(w > 1e-10)[0]
    for k in idxs:
        g += w[k] * lag_grad(h, k)
    return g * dx

def cons_fun(x, target):
    return x.sum() - target

def cons_jac(x):
    return np.ones(x.size)

def run_slsqp(h0, tau, maxiter, target):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: cons_fun(x, target), 'jac': lambda x: cons_jac(x)}]
    try:
        res = so.minimize(softmax_obj, h0, jac=lambda x: softmax_grad(x, tau),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h0

def subgradient_phase(h0, time_limit, seed, topk=3):
    rng = np.random.RandomState(seed)
    n = h0.size
    target = 0.5 * n
    x = project_box_sum(h0.copy(), target)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    it = 0
    no_improve = 0
    avg_c = best_c + 0.01

    while time.time() < deadline:
        it += 1
        c = c_values(x)
        C = float(c.max())
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        active = np.argsort(c)[::-1][:topk]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in active:
            g += lag_grad(x, int(k))
        g *= dx / topk

        step = (avg_c - best_c) / max(1e-15, float(g @ g))
        step = min(step, 0.5)
        step = max(step, 0.001)

        y = x - step * g
        x = project_box_sum(y, target)

        avg_c = 0.95 * avg_c + 0.05 * C

        if no_improve % 2000 == 0 and no_improve > 2000:
            x = project_box_sum(x + rng.normal(0, 0.003, n), target)
            no_improve = 0

        if no_improve > 20000:
            break

    return best_x, best_c

def sa_phase(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    n = h0.size
    target = 0.5 * n
    x = project_box_sum(h0.copy(), target)
    best_x = x.copy()
    best_c = exact_c(x)
    T = 0.015
    alpha = 0.9997
    deadline = time.time() + time_limit
    it = 0

    while time.time() < deadline:
        it += 1
        n_changes = rng.randint(1, 4)
        idxs = rng.choice(n, n_changes, replace=False)
        deltas = rng.uniform(-0.08, 0.08, n_changes)
        y = x.copy()
        y[idxs] += deltas
        y = project_box_sum(y, target)
        cy = exact_c(y)
        dy = cy - exact_c(x)
        if dy < 0 or rng.rand() < np.exp(-dy / max(T, 1e-10)):
            x = y
            if cy < best_c:
                best_c = cy
                best_x = x.copy()
        T *= alpha

    return best_x, best_c

def resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return project_box_sum(h0.copy(), 0.5 * n_new)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x1, x0, h0)
    return project_box_sum(y, 0.5 * n_new)

def optimize_at_n(h0, time_limit, seed):
    n = h0.size
    target = 0.5 * n
    x = project_box_sum(h0.copy(), target)
    best_c = exact_c(x)
    best_x = x.copy()

    stages = [(0.04, 150), (0.015, 200), (0.006, 250), (0.002, 300), (0.0008, 250)]
    start = time.time()

    for i, (tau, mi) in enumerate(stages):
        if time.time() - start > time_limit * 0.75:
            break
        res = run_slsqp(x, tau, mi, target)
        cr = exact_c(res)
        if cr < best_c:
            best_c = cr
            best_x = res.copy()
        x = project_box_sum(res, target) if cr <= best_c + 1e-4 else best_x

    remaining = time_limit - (time.time() - start)
    if remaining > 3:
        xp, cp = subgradient_phase(best_x, min(remaining * 0.6, time_limit * 0.45), seed)
        if cp < best_c:
            best_c = cp
            best_x = xp

    remaining = time_limit - (time.time() - start)
    if remaining > 3:
        xp, cp = sa_phase(best_x, min(remaining * 0.7, time_limit * 0.3), seed + 1)
        if cp < best_c:
            best_c = cp
            best_x = xp

    return project_box_sum(best_x, target), best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    init = np.asarray(raw, dtype=float) if raw is not None else None

    if init is not None and init.size >= 10:
        base_n = min(int(init.size), 120)
    else:
        base_n = 70
        init = np.full(base_n, 0.5)

    best_x = project_box_sum(init.copy(), 0.5 * init.size)
    best_c = exact_c(best_x)

    ns = []
    for nn in (base_n, base_n + 10, base_n - 10, base_n + 20, base_n - 20, 90, 110, 130):
        nn = int(nn)
        if 40 <= nn <= 160 and nn not in ns:
            ns.append(nn)
    if 70 not in ns:
        ns.insert(0, 70)
    ns = ns[:5]

    for n in ns:
        if time.time() > deadline - 60:
            break
        target = 0.5 * n

        if init is not None and init.size >= 10:
            h_base = resample(init, n)
        else:
            h_base = np.full(n, 0.5)

        per_time = min(60.0, max(15.0, (deadline - time.time()) * 0.18))
        starts = [h_base]
        rng = np.random.RandomState(seed_val + n)
        for sc in (0.01, 0.04, 0.08):
            starts.append(project_box_sum(h_base + rng.normal(0, sc, n), target))

        for ii, h0 in enumerate(starts):
            if time.time() > deadline - 50:
                break
            t_left = deadline - time.time()
            if t_left < 10:
                break
            alloc = min(per_time, t_left * 0.35)
            hx, cx = optimize_at_n(h0, alloc, seed_val + 1000 * n + ii)
            if cx < best_c:
                best_c = cx
                best_x = hx

    # Final polish at best resolution
    if time.time() < deadline - 15:
        n_best = best_x.size
        target = 0.5 * n_best
        t_fill = deadline - time.time() - 5
        hx, cx = subgradient_phase(best_x, min(t_fill * 0.5, 40), seed_val + 9999, topk=4)
        if cx < best_c:
            best_c = cx
            best_x = hx
        if time.time() < deadline - 10:
            res = run_slsqp(best_x, 0.001, 200, target)
            cr = exact_c(res)
            if cr < best_c:
                best_c = cr
                best_x = project_box_sum(res, target)

    best_x = project_box_sum(best_x, 0.5 * best_x.size)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
