def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx, conv * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, -5.0, 5.0)

def subgrad_max_vec(h):
    n = len(h)

        grad[shift:] -= h[:n - shift]
