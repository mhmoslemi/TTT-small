def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        if d2 > 0:
            idx = np.argsort(1.0 - h)
            h[idx[0]] += d2
        else:
            idx = np.argsort(h)
            h[idx[0]] += d2
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = np.max(corr)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):

            k_end = max(k_end1, k_end2)

            else:

            Rj = stack[-1][1]

    # The envelope is: for x >= stack[j][2], line j is the max
    # The envelope is convex, so minimum is where slope changes from negative to positive

    best_val = np.inf
    
    for j in range(len(stack)):
        ci, Ri, x_start = stack[j]
        # This line is active for x in [x_start, x_end] where x_end = next line's x_start or +inf
        if j < len(stack) - 1:
            x_end = stack[j+1][2]
        else:
            x_end = np.inf

            continue

import time

    n = len(h)
    dx = 2.0 / n

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(20):
        d = target - np.sum(h)
        if abs(d) < 1e-12:

        h += d / n

        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
            d = -d

            room = 1.0 - h[idx] if d > 0 else h[idx]
            adj = min(d, room)
            if d > 0:
                h[idx] += adj
            else:

def _smooth_opt(h, n, alpha, beta, maxiter=300):
    """L-BFGS with smooth max approximation."""
    dx = 2.0 / n
    target = 0.5 * n
    
    def obj_and_grad(x):
        x = np.clip(x, 0, 1)
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        nc = len(C)
        sC = alpha * C
        m = np.max(sC)
        e = np.exp(sC - m)
        se = np.sum(e)
        f = (m + np.log(se)) / alpha + beta * (np.sum(x) - target) ** 2
        w = e / se
        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            k1s = max(0, n - 1 - i)
            k1e = min(2*n - 2, 2*n - 2 - i)
            k2s = max(0, i)
            k2e = min(2*n - 2, i + n - 1)
            klo = min(k1s, k2s)
            khi = max(k1e, k2e)
            # d C[k] / d h[i]
            dcd = np.zeros(khi - klo + 1)
            ks = np.arange(klo, khi + 1)
            m1 = (ks >= k1s) & (ks <= k1e)
            if np.any(m1):
                dcd[m1] += (1.0 - x[ks[m1] - (n - 1 - i)]) * dx
            m2 = (ks >= k2s) & (ks <= k2e)
            if np.any(m2):
                dcd[m2] += (-x[i + n - 1 - ks[m2]]) * dx
            grad[i] = np.dot(w[klo:khi+1], dcd)
        grad += 2.0 * beta * (np.sum(x) - target)
        return f, grad
    
    res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 5, 'ftol': 1e-15})
    if res.x is not None:
        h_new = _fix_sum(res.x, n)
        if _c5_max(h_new) < _c5_max(h):
            return h_new
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
            d = -d
        for idx in order:
            room = 1.0 - h[idx] if d > 0 else h[idx]
            adj = min(d, room)
            if d > 0:
                h[idx] += adj
            else:
                h[idx] -= adj
            d -= adj
            if d < 1e-12:
                break
    return h

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _coord_descent(h, n, max_sweeps=300, tol=1e-11):
    dx = 2.0 / n
    corr = _corr_full(h).copy()
    h = h.copy()
    best_c5 = np.max(corr)
    prev_max = best_c5

    for sweep in range(max_sweeps):
        for i in range(n):
            a_cur = h[i]
            k1s = max(0, n - 1 - i)
            k1e = min(2 * n - 2, 2 * n - 2 - i)
            k2s = max(0, i)
            k2e = min(2 * n - 2, i + n - 1)
            klo = min(k1s, k2s)
            khi = max(k1e, k2e)

            ks = np.arange(klo, khi + 1)
            coeff = np.zeros(len(ks))
            m1 = (ks >= k1s) & (ks <= k1e)
            if np.any(m1):
                coeff[m1] += (1.0 - h[ks[m1] - (n - 1 - i)]) * dx
            m2 = (ks >= k2s) & (ks <= k2e)
            if np.any(m2):
                coeff[m2] += (-h[i + n - 1 - ks[m2]]) * dx

            cur = corr[klo:khi + 1]

            cands = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])

def _smooth_opt(h, n, alpha, beta, maxiter=200):

    target = 0.5 * n
    nc = 2 * n - 1

    def obj_and_grad(x):
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        sC = alpha * C
        m = np.max(sC)
        e = np.exp(sC - m)
        se = np.sum(e)
        f = (m + np.log(se)) / alpha + beta * (np.sum(x) - target) ** 2
        w = e / se
        grad = np.zeros(n)
        for i in range(n):
            k1s = max(0, n - 1 - i)
            k1e = min(2 * n - 2, 2 * n - 2 - i)
            k2s = max(0, i)
            k2e = min(2 * n - 2, i + n - 1)
            klo = min(k1s, k2s)
            khi = max(k1e, k2e)
            ks = np.arange(klo, khi + 1)
            dcd = np.zeros(len(ks))
            m1 = (ks >= k1s) & (ks <= k1e)

    if res.x is not None:
        h_new = _fix_sum(res.x, n)
        if _c5_max(h_new) < _c5_max(h):

                    h_best, c_best = h_cd, c_cd

    # Phase 4: Random perturbations + coord descent
    for trial in range(60):
        if elapsed() > budget_s * 0.93:
            break
        scale = 0.015 + 0.04 * rng.rand()
        h_try = _fix_sum(h_best + rng.randn(n_best) * scale, n_best)
        h_cd, c_cd = _coord_descent(h_try, n_best, max_sweeps=120)
        if c_cd < c_best:
            h_best, c_best = h_cd, c_cd

    # Phase 5: Binary search (only for even n)
    if n_best % 2 == 0 and elapsed() < budget_s * 0.96:
        h_bin, c_bin = _try_binary(h_best, n_best, max_iters=200)
        if c_bin < c_best:
            h_best, c_best = h_bin, c_bin

    # Final polish
    if elapsed() < budget_s * 0.99:
        h_cd, c_cd = _coord_descent(h_best, n_best, max_sweeps=300, tol=1e-12)
        if c_cd < c_best:
            h_best, c_best = h_cd, c_cd

    h_best = _fix_sum(h_best, n_best)
    c_best = _c5_max(h_best)
    return (h_best, c_best, n_best)
