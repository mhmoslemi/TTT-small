import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_1d(a, b):
    """Find t in [0,1] minimizing max_k (a[k] + b[k]*t)."""
    m = len(a)
    f0 = float(np.max(a))
    f1 = float(np.max(a + b))
    best_t = 0.0 if f0 <= f1 else 1.0
    best_val = min(f0, f1)
    
    # Compute all pairwise breakpoints
    b_diff = b[:, None] - b[None, :]  # (m,m): b_diff[i,j] = b[i] - b[j]
    a_diff = a[None, :] - a[:, None]  # (m,m): a_diff[i,j] = a[j] - a[i]
    
    mask = np.abs(b_diff) > 1e-14
    b_diff_safe = np.where(mask, b_diff, 1.0)
    t_mat = a_diff / b_diff_safe
    
    valid = mask & (t_mat > 1e-10) & (t_mat < 1.0 - 1e-10)
    np.fill_diagonal(valid, False)
    
    if np.any(valid):
        candidates = t_mat[valid]
        # Vectorized evaluation: f(t) = max_k(a_k + b_k * t)
        f_vals = (a[None, :] + b[None, :] * candidates[:, None]).max(axis=1)
        idx = np.argmin(f_vals)
        if f_vals[idx] < best_val:
            best_val = float(f_vals[idx])
            best_t = float(candidates[idx])
    
    return best_t

    dx = 2.0 / n
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full')
            h[i] = old_val
            
            t_opt = find_optimal_1d(c0, c1 - c0)
            h[i] = t_opt
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c5 - tol:
                best_c5 = c
                improved = True
            else:
                # Keep the new value anyway (for diversity)
                # h stays as is
                pass
        if not improved:
            break
    return h, best_c5

    m = c.max()
    shifted = alpha * (c - m)

    return np.sum(h) - n / 2.0

    return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def find_optimal_1d(a, b):
    f0 = float(np.max(a))
    f1 = float(np.max(a + b))
    best_t = 0.0 if f0 <= f1 else 1.0
    best_val = min(f0, f1)
    grid = np.linspace(0, 1, 500)
    vals = (a[None, :] + b[None, :] * grid[:, None]).max(axis=1)
    idx = np.argmin(vals)
    if vals[idx] < best_val:
        best_val = float(vals[idx])
        best_t = float(grid[idx])
    return best_t

def obj_slsqp(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def do_coord_descent(h, n, max_sweeps, t0, budget):
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if budget - (time.time() - t0) < 25:
            break
        improved = False
        order = range(n)
        for i in order:
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full')
            h[i] = old
            t = find_optimal_1d(c0, c1 - c0)
            h[i] = t
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c5 - 1e-14:
                best_c5 = c
                improved = True
        if not improved:
            break
    return h, best_c5

def gen_initial(n, rng, typ):
    if typ == 0:

    elif typ == 1:

    else:

    return np.clip(h, 0, 1)
