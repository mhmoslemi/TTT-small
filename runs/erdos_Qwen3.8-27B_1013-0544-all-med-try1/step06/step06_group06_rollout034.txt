import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo_b, hi_b = -2.0, 2.0
    for _ in range(200):
        mid = (lo_b + hi_b) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo_b = mid
        else:
            hi_b = mid
    return np.clip(h - (lo_b + hi_b) / 2.0, 0.0, 1.0)

def smooth_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
        w = w / w.sum()
    else:
        w = np.exp(shifted)
        w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        coeff = alpha * w[k] * dx
        if coeff < 1e-18:
            continue
        offset = k - (n - 1)
        j1_lo = max(0, offset)
        j1_hi = min(n - 1, offset + n - 1)
        if j1_lo <= j1_hi:
            idx1 = np.arange(j1_lo, j1_hi + 1) - offset
            grad[j1_lo:j1_hi + 1] += coeff * (1.0 - x[idx1])
        j2_lo = max(0, -offset)
        j2_hi = min(n - 1, n - 1 - offset)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi + 1) + offset
            grad[j2_lo:j2_hi + 1] -= coeff * x[idx2]
    return grad

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def pair_obj_1d(t, h_buf, i, j, target, n):
    h_buf[i] = t
    h_buf[j] = target - t
    return compute_c5(h_buf, n)

def pair_optimize(h, n, i, j):
    target = h[i] + h[j]
    lo = max(0.0, target - 1.0)
    hi = min(1.0, target)
    if hi - lo < 1e-12:
        return h, compute_c5(h, n)
    h_buf = h.copy()
    try:
        res = minimize_scalar(pair_obj_1d, bounds=(lo, hi), method='bounded',
                              args=(h_buf, i, j, target, n),
                              options={'xatol': 1e-9, 'maxiter': 150})
        h_buf[i] = np.clip(res.x, 0.0, 1.0)
        h_buf[j] = np.clip(target - res.x, 0.0, 1.0)
        c = compute_c5(h_buf, n)
        return h_buf, c
    except Exception:
        return h, compute_c5(h, n)

    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
