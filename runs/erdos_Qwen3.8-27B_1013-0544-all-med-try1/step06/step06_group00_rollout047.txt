def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(np.clip(temp * (corr - m), -500, 500))
    val = (m + np.log(s.sum()) / temp) * dx
    if penalty > 0:
        val += penalty * (h.sum() - n / 2.0) ** 2
    return val

def subgradient_step(h, top_k=5, sharpness=60.0, lr_scale=0.02):
    n = len(h)
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    top_idx = np.argsort(corr)[-top_k:]
    w = np.exp(sharpness * (corr[top_idx] - m))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, li in zip(w, top_idx):
        lag = li - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            j = i + lag
            grad[j] += wi * (1.0 - h[i])
            if j != i:
                grad[i] -= wi * h[j]
            else:
                grad[i] -= wi * h[i]
    gnorm = np.max(np.abs(grad)) + 1e-12
    lr = lr_scale / gnorm
    h_new = h - lr * grad
    return project_feasible(h_new, ts)
