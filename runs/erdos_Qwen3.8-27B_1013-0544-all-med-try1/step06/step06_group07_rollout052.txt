def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(20):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:

            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())

def slsqp_run(x0, maxiter=100):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": cons_fun, "jac": cons_jac}]
    try:
        res = so.minimize(
            max_obj, x0, method="SLSQP", jac=jac_max,
            bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12}
        )
        if res is not None:
            return repair(res.x)
    except Exception:
        pass
    return None

def max_obj(x):
    return exact_c(x)

def jac_max(x):
    return max_jac(x)

def resample(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == int(n_new):
        return repair(h0.copy())
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(int(n_new)) + 0.5) * (2.0 / float(n_new))
    return repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)
    
    if init is not None and init.size >= 10:
        base = repair(init.copy())
        best_c = exact_c(base)
        best_x = base.copy()
    else:
        base = repair(np.full(70, 0.5))
        best_c = exact_c(base)
        best_x = base.copy()
    
    # Try multiple approaches and sizes
    sizes = [70, 90, 100, 120, 50]
    
    for n_try in sizes:
        if time.time() > deadline - 30:
            break
        if init is not None:
            h0 = resample(init, n_try)
        else:
            h0 = repair(np.full(n_try, 0.5))
        
        # Coordinate descent with multiple restarts
        n_passes = max(50, int(120.0 * (budget_s / 1100.0)))
        
        # Main run
        if time.time() < deadline - 60:
            alloc = min(250.0, max(30.0, (deadline - time.time()) * 0.25))
            h1, c1 = coord_desc(h0, n_passes, seed_val + n_try, time.time() + alloc, c_only=False)
            if c1 < best_c:
                best_c = c1
                best_x = h1
        
        # Try a jittered restart
        if time.time() < deadline - 60:
            rng = np.random.RandomState(seed_val + 999 + n_try)
            h_jit = repair(h0 + rng.normal(0, 0.03, n_try))
            alloc2 = min(150.0, max(20.0, (deadline - time.time()) * 0.2))
            h2, c2 = coord_desc(h_jit, int(n_passes * 0.7), seed_val + 5000 + n_try, time.time() + alloc2, c_only=False)
            if c2 < best_c:
                best_c = c2
                best_x = h2
        
        # SLSQP polish on best
        if time.time() < deadline - 30:
            xr = slsqp_run(best_x if best_x.size == n_try else resample(best_x, n_try), maxiter=80)
            if xr is not None:
                c_r = exact_c(xr)
                if c_r < best_c:
                    best_c = c_r
                    best_x = xr
    
    # Final SLSQP polish at the best size
    if time.time() < deadline - 10:
        xr = slsqp_run(best_x, maxiter=120)
        if xr is not None:
            c_r = exact_c(xr)
            if c_r < best_c:
                best_c = c_r
                best_x = xr
    
    # Final coord descent polish
    if time.time() < deadline - 5:
        h_f, c_f = coord_desc(best_x, 80, seed_val + 7777, deadline, c_only=False)
        if c_f < best_c:
            best_c = c_f
            best_x = h_f
    
    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
