def _subgradient_step(h, n, steps=100, top_k=5):
    """Subgradient method on the max function."""
    dx = 2.0 / n
    h = h.copy()
    m_range = np.arange(-(n-1), n)
    nc = 2*n - 1
    
    # Initial step size
    step = 0.05
    
    for s in range(steps):
        C = _c5_array(h)
        k_active = np.argsort(C)[-top_k:]  # indices of top-C values
        weights = C[k_active] - C[k_active[0]] + 1.0  # positive weights
        weights /= weights.sum()

        grad = np.zeros(n)
        for k_idx in k_active:
            m = k_idx - (n - 1)
            for i in range(n):
                im = i + m
                im2 = i - m
                contrib = 0.0
                if 0 <= im < n:
                    contrib += (1.0 - h[im])
                if 0 <= im2 < n:
                    contrib -= h[im2]
                grad[i] += weights[list(k_active).index(k_idx)] * dx * contrib

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            h -= step * grad / gnorm  # or h -= step * grad
            h = _fix_sum(h, n)

import numpy as np
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h_adj = h + d / n
        if np.all(h_adj >= -1e-12) and np.all(h_adj <= 1.0 + 1e-12):
            h = np.clip(h_adj, 0.0, 1.0)
        else:
            if d > 0:
                order = np.argsort(-(1.0 - h))
                for idx in order:
                    add = min(d, 1.0 - h[idx])
                    h[idx] += add
                    d -= add
                    if d <= 1e-14:
                        break
            else:
                d = -d
                order = np.argsort(h)
                for idx in order:
                    sub = min(d, h[idx])
                    h[idx] -= sub
                    d -= sub
                    if d <= 1e-14:
                        break
    return np.clip(h, 0.0, 1.0)

def _coord_descent(h, n, max_sweeps=3000, seed=0):
    dx = 2.0 / n
    h = h.astype(float).copy()
    h = _fix_sum(h, n)
    rng = np.random.RandomState(seed)
    C = _c5_array(h)
    best_c5 = np.max(C)
    no_improve = 0
    m_range = np.arange(-(n - 1), n)
    ts_coarse = np.linspace(0.0, 1.0, 64)
    
    for sweep in range(max_sweeps):

                      np.where(im2_v, h[np.clip(im2, 0, n - 1)], 0.0))
            old = h[i]
            vals = np.max(C[None, :] + (ts_coarse[:, None] - old) * g[None, :], axis=1)

            best_val = vals[best_idx]

def _subgrad_descent(h, n, max_iter=2000, seed=0):
    """Subgradient method with top-k averaging to escape coord-descent local minima."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    nc = 2 * n - 1
    best_h = h.copy()
    best_c5 = _c5_max(h)
    
    step0 = 0.04
    for it in range(max_iter):
        C = _c5_array(h)
        top_k = min(8, nc)
        k_active = np.argsort(C)[-top_k:]
        c_top = C[k_active]
        w = c_top - c_top[0] + 0.1
        w = w / w.sum()
        
        grad = np.zeros(n)
        ar = np.arange(n)
        for ki, kk in enumerate(k_active):
            m = kk - (n - 1)
            im = ar + m
            t1 = np.where((im >= 0) & (im < n), 1.0 - h[np.clip(im, 0, n - 1)], 0.0)
            im2 = ar - m
            t2 = np.where((im2 >= 0) & (im2 < n), h[np.clip(im2, 0, n - 1)], 0.0)
            grad += w[ki] * dx * (t1 - t2)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = step0 / (1.0 + 0.01 * it)
        h_new = h - step * grad
        h_new = _fix_sum(h_new, n)
        c_new = _c5_max(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_try = h - step * 0.3 * grad
            h_try = _fix_sum(h_try, n)
            c_try = _c5_max(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5

def _pair_swap(h, n, n_pairs=2000, seed=0):
    """Random pairwise mass transfer to escape local minima."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    best_c5 = _c5_max(h)
    
    for _ in range(n_pairs):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        for delta in [0.01, 0.02, 0.05]:
            for sign in [1, -1]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] - sign * delta, 0, 1)
                h_try[j] = np.clip(h[j] + sign * delta, 0, 1)
                c_try = _c5_max(h_try)
                if c_try < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c_try
                    break
    return h, best_c5

def _interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

        hb, cb = _pair_swap(hb, nb, n_pairs=1000, seed=seed + 11111)
        hb = _fix_sum(hb, nb)
        cb = _c5_max(hb)
    
    hb = _fix_sum(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
