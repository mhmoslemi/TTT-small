The problem:

subject to: corr_k(h) * dx ≤ t for k in active set
             0 ≤ h ≤ 1
             sum(h) = n/2

This is a QP (minimize t subject to quadratic constraints). We can solve it with cvxpy or scipy.

But wait, the corr_k(h) is quadratic in h, so the constraint corr_k(h) * dx ≤ t is a quadratic constraint. The QP would be:
minimize t
subject to: sum_i h[i] * (1-h[i+k]) * dx ≤ t  (for each active k)
            0 ≤ h ≤ 1
            sum(h) = n/2

This is a non-convex QP (the constraint is not convex in h since it involves h[i]*h[i+k]).

So we can't use standard QP solvers. Back to heuristic methods.

Let me try one more idea: **Alternating minimization / Block coordinate descent with larger blocks.**

Instead of changing one coordinate at a time, change a block of m coordinates at a time. For a block of size m, the subproblem is:
minimize max_k corr_k(h)
subject to: h_block ∈ [0,1]^m, sum(h) = n/2

where h_block varies and the rest is fixed. This is still non-convex but has only m variables (m ≪ n). We can solve it with L-BFGS-B in m dimensions, which is much faster.

For m=8 or m=16, this should be quite fast and might escape local optima better than single-coordinate updates.

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0.001) & (h < 0.999)
        if mask.sum() == 0:
            mask = np.ones(len(h), dtype=bool)
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, target_sum, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if temp < 1e-8:
        sm = m
    else:
        exp_arr = np.exp((corr - m) / temp)
        sm = m + temp * np.log(exp_arr.sum())
    pen = penalty * (h.sum() - target_sum) ** 2
    return (sm + pen) * dx

def topk_sum_obj(h, target_sum, k, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    topk = np.sort(corr)[-k:]
    return np.sum(topk) + penalty * (h.sum() - target_sum) ** 2

def optimize_l_bfgs(h_start, ts, temp, penalty, maxiter=500):
    n = len(h_start)
    bnds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_obj, h_start, args=(ts, temp, penalty),
        method='L-BFGS-B', bounds=bnds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10}
    )
    return project_feasible(res.x, ts)

def coord_descent(h, n_sweeps=10):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            oj = h[j]
            best_v = oj
            best_c = c5
            candidates = [0.0, 1.0, max(0, oj - 0.1), min(1, oj + 0.1),
                          max(0, oj - 0.05), min(1, oj + 0.05)]
            k = n - 1 - j
            for v in candidates:
                if abs(v - oj) < 1e-12:
                    continue
                h[j] = v
                h[k] -= (v - oj)
                if h[k] < 0 or h[k] > 1:
                    h = project_feasible(h, ts)
                else:
                    h[k] = max(0.0, min(1.0, h[k]))
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
                    saved = h.copy()
            if best_v != oj:
                h = saved
                c5 = best_c
                improved = True
            else:
                h[j] = oj
                h[k] += (v - oj) if False else 0
                h = project_feasible(h, ts)
        if not improved:
            break
    return h, c5

def paired_cd(h, n_sweeps=15):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            k = n - 1 - j
            if k == j:
                continue
            oj, ok = h[j], h[k]
            for d in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2, 0.5, -0.5]:
                nj, nk = oj + d, ok - d
                if 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
                    h[j], h[k] = nj, nk
                    c = compute_c5(h)
                    if c < c5 - 1e-13:
                        c5 = c
                        improved = True
                    else:
                        h[j], h[k] = oj, ok
        if not improved:
            break
    return h, c5

def binary_threshold(cd_h, c5):
    n = len(cd_h)
    ts = n / 2.0
    improved = True
    while improved:
        improved = False
        for j in range(n):
            k = n - 1 - j
            if k == j:

            oj, ok = cd_h[j], cd_h[k]

            for vj in [0.0, 1.0]:
                for vk in [0.0, 1.0]:
                    if abs(vj - oj) < 1e-10 and abs(vk - ok) < 1e-10:

                    cd_h[j], cd_h[k] = vj, vk

            if not improved:
                cd_h[j], cd_h[k] = oj, ok
                cd_h = project_feasible(cd_h, ts)
        c5 = best_c if improved else c5
    return cd_h, c5

def make_structured_start(n, rng, n_harmonics=5):

        h += amp * np.sin(np.pi * freq * x + phase)
    return h
