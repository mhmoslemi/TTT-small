def swap_delta_corr(h, i, j, n):
    """Compute the change in correlation when h[i]:1->0 and h[j]:0->1"""
    # d ranges where i+d is in [0,n-1]: d in [-i, n-1-i]
    d1_lo, d1_hi = -i, n - 1 - i
    # d ranges where j+d is in [0,n-1]: d in [-j, n-1-j]
    d2_lo, d2_hi = -j, n - 1 - j
    
    # For each affected d, compute delta
    # We need to check the full range [d1_lo, d1_hi] ∪ [d2_lo, d2_hi]
    lo = min(d1_lo, d2_lo)
    hi = max(d1_hi, d2_hi)
    
    deltas = {}
    for d in range(lo, hi + 1):
        delta = 0
        if d1_lo <= d <= d1_hi:
            # i+d is in [0,n-1]
            ip = i + d  # index of h[i+d] and s[i+d]
            delta += (2*h[ip] - 1)  # = +1 if h=1, -1 if h=0
        if d2_lo <= d <= d2_hi:
            # j+d is in [0,n-1]
            jp = j + d
            delta += (1 - 2*h[jp])  # = +1 if h=0, -1 if h=1
        if delta != 0:
            deltas[d] = delta
    return deltas

    start = time.time()
    rng = np.random.RandomState(seed)
    T_init = 0.06
    T_min = 0.0003

    while (time.time() - start) < max_time:
        elapsed = (time.time() - start) / max(1e-9, max_time)
        T = max(T_min, T_init * (T_min / T_init) ** min(1.0, elapsed))

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5

        if delta < 0 or (delta < 0.1 and rng.random() < np.exp(-delta / max(T, 1e-12))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


def interpolate_to_n(h_old, n_new, seed=0):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return make_binary(h_new, n_new, seed=seed)


def perturb_binary(h, n_perturb, rng):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_perturb):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Local search from initial (best improvement)
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        t_alloc = min(80.0, rem * 0.12)
        h1, c5_1 = local_search(best_h, t_alloc, seed=seed)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Multiple SA restarts
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        n_sa_runs = 5
        t_per_sa = rem * 0.45 / n_sa_runs
        for k in range(n_sa_runs):
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            t_sa = min(t_per_sa, rem * 0.4)
            if t_sa < 5:
                break
            # Slightly perturb before each SA to explore different basins
            rng_p = np.random.RandomState(seed + 1000 + k * 7)
            h_pert = perturb_binary(best_h, rng_p.randint(1, 4), rng_p)
            h_sa, c5_sa = simulated_annealing(h_pert, t_sa, seed=seed + 2000 + k * 13)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 3: Local search refinement
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        t_alloc = min(60.0, rem * 0.15)
        h2, c5_2 = local_search(best_h, t_alloc, seed=seed + 5000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 4: ILS with varying perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        t_ils = rem * 0.35
        il_start = time.time()
        rng_ils = np.random.RandomState(seed + 6000)
        h_cur = best_h.copy()
        while (time.time() - il_start) < t_ils:
            rem_inner = budget_s - (time.time() - t0)
            if rem_inner < 20:
                break
            n_pert = int(rng_ils.randint(2, max(3, best_n // 8)))
            h_cur = perturb_binary(h_cur, n_pert, rng_ils)
            t_ls = min(15.0, t_ils - (time.time() - il_start))
            if t_ls < 3:
                break
            h_ls, c5_ls = local_search(h_cur, t_ls, seed=int(rng_ils.randint(99999)), first_improve=True)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
            h_cur = h_ls.copy()

    # Phase 5: Explore other n values
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        n_candidates = [n for n in range(60, 220, 4) if n != best_n]
        n_candidates.sort(key=lambda x: abs(x - best_n))
        for nc in n_candidates[:20]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            h_interp = interpolate_to_n(best_h, nc, seed=seed + nc)
            t_n = min(15.0, rem - 15)
            if t_n < 3:
                break
            h_opt, c5_opt = local_search(h_interp, t_n, seed=seed + nc, first_improve=True)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = nc
            rem = budget_s - (time.time() - t0)
            if rem > 20 and c5_opt < best_c5 + 0.005:
                h_sa3, c5_sa3 = simulated_annealing(h_opt, min(10.0, rem - 10), seed=seed + nc + 8000)
                if c5_sa3 < best_c5:
                    best_c5 = c5_sa3
                    best_h = h_sa3.copy()
                    best_n = nc

    # Phase 6: Final SA + local search polish
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        t_final_sa = rem * 0.4
        h_f, c5_f = simulated_annealing(best_h, t_final_sa, seed=seed + 9000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        rem = budget_s - (time.time() - t0)
        if rem > 10:
            h_f2, c5_f2 = local_search(best_h, rem * 0.6, seed=seed + 9500)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
