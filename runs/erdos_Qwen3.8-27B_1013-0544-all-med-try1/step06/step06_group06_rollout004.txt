import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_sm(h, n, alpha):
    c = get_correlations(h, n)
    return smooth_max_val(c, alpha)

def obj_pen(h, n, alpha, mu):
    c = get_correlations(h, n)
    sm = smooth_max_val(c, alpha)
    return sm + mu * (np.sum(h) - n / 2.0) ** 2

def sum_eq(h, n):
    return np.sum(h) - n / 2.0

def sum_eq_jac(h, n):
    return np.ones(n)

def downsample(h, n_from, n_to):
    if n_to == n_from:
        return h.copy()
    if n_to < n_from:
        result = np.zeros(n_to)
        for i in range(n_to):
            lo = int(i * n_from / n_to)
            hi = int((i + 1) * n_from / n_to)
            result[i] = h[lo:hi].mean()
    else:
        result = np.zeros(n_to)
        for i in range(n_to):
            pos = (i + 0.5) * n_from / n_to
            idx = int(pos)
            idx = min(idx, n_from - 1)
            result[i] = h[idx]
    return result
