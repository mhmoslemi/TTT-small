    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h = np.interp(x_new, x_old, h_old)
    return project_box_sum(h, n_new / 2.0)

def sg_optimize(h0, max_time, seed=42, top_k=3, tol=0.004):
    n = len(h0)
    ts = n / 2.0
    h = project_box_sum(np.clip(h0, 0, 1), ts)
    cur = compute_c5(h)
    bh = h.copy()
    bc = cur
    rng = np.random.RandomState(seed)
    t0 = time.time()
    eta = 0.4
    best_eta = eta
    stall = 0
    while time.time() - t0 < max_time:
        c_all = compute_c5_all(h)
        c_max = c_all.max()
        active = np.where(c_all >= c_max - tol)[0]
        if len(active) > top_k:
            order = np.argsort(c_all[active])[::-1][:top_k]
            active = active[order]
        grad = get_subgrad(h, active)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn
        improved = False
        step = eta
        for _ in range(14):
            h_new = project_box_sum(h - step * grad, ts)
            c_new = compute_c5(h_new)
            if c_new < cur - 1e-12:
                h = h_new
                cur = c_new
                best_eta = step
                if cur < bc:
                    bc = cur
                    bh = h.copy()
                stall = 0
                improved = True
                break
            step *= 0.55
        if not improved:
            eta = best_eta * 1.08
            stall += 1
            if stall % 6 == 0:
                n_p = max(3, n // 12)
                idxs = rng.choice(n, size=n_p, replace=False)
                h[idxs] += rng.normal(0, 0.2, n_p)
                h = project_box_sum(h, ts)
                cur = compute_c5(h)
            if stall > 100:
                break
    return bh, bc

def sa_h(h0, max_time, seed=42):
    n = len(h0)
    ts = n / 2.0
    h = project_box_sum(np.clip(h0, 0, 1), ts)
    cur = compute_c5(h)
    bh = h.copy()
    bc = cur
    rng = np.random.RandomState(seed)
    t0 = time.time()
    while time.time() - t0 < max_time:
        frac = (time.time() - t0) / max_time
        T = 0.06 * (0.003) ** frac
        if T < 1e-10:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        d = rng.uniform(0.03, 0.5) * (1 if rng.random() < 0.5 else -1)
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c_new = compute_c5(h)
        dc = c_new - cur
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-12)):
            cur = c_new
            if cur < bc:
                bc = cur
                bh = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_start = project_box_sum(h_init, n_init / 2.0)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    if rem() > 30:
        h1, c1 = sg_optimize(h_start, min(130, rem() * 0.22), seed, top_k=3)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
    if rem() > 50:
        h1b, c1b = sg_optimize(best_h, min(80, rem() * 0.15), seed + 100, top_k=1)
        if c1b < best_c5:
            best_c5, best_h = c1b, h1b.copy()
    if rem() > 50:
        h2, c2 = sa_h(best_h, min(90, rem() * 0.15), seed + 500)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        if rem() > 30:
            h2b, c2b = sg_optimize(best_h, min(80, rem() * 0.15), seed + 600, top_k=3)
            if c2b < best_c5:
                best_c5, best_h = c2b, h2b.copy()
    for nr in [98, 140, 168, 196, 224, 280, 126, 160]:
        if rem() < 50 or nr % 2 != 0 or nr == n_init:
            continue
        hi = interp_h(best_h, nr)
        t_r = min(55, rem() * 0.1)
        if t_r < 10:
            break
        hr, cr = sg_optimize(hi, t_r, seed + nr, top_k=3)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
    for trial in range(6):
        if rem() < 90:
            break
        h_tr = best_h + 0.1 * np.random.randn(len(best_h))
        h_tr = project_box_sum(h_tr, len(best_h) / 2.0)
        ht, ct = sg_optimize(h_tr, min(70, rem() * 0.11), seed + 2000 + trial * 100, top_k=3)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()
    if rem() > 40:
        hf, cf = sg_optimize(best_h, rem() * 0.88, seed + 9999, top_k=2, tol=0.002)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
