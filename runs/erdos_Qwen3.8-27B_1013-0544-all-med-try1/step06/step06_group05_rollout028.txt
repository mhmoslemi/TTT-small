import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float)
    idx = np.argsort(h)[::-1][:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def fast_binary_ls(h, max_time, seed=42):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    current_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        best_gain = 1e-15
        best_p = -1
        best_q = -1
        
        for p in ones:
            h[p] = 0.0
            for q in zeros:
                h[q] = 1.0
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_p = p
                    best_q = q
                h[q] = 0.0
            h[p] = 1.0

            h[best_p] = 0.0
            h[best_q] = 1.0
            current_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        else:
            break
    
    return h, current_max * dx

def fourier_h(params, n, K):
    """Compute h from Fourier parameters"""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = params[2*(k-1)]
        b = params[2*(k-1)+1]
        h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    return np.clip(h, 0.0, 1.0)

def fourier_objective(params, n, K):
    h = fourier_h(params, n, K)
    # Penalty for not satisfying sum constraint (should be automatic)
    c5 = compute_c5(h)
    # Small penalty for clipping
    h_unclipped = np.full(n, 0.5)
    x = (np.arange(n) + 0.5) * 2.0 / n
    for k in range(1, K + 1):
        a = params[2*(k-1)]
        b = params[2*(k-1)+1]
        h_unclipped += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    clip_penalty = np.sum(np.maximum(0, h_unclipped - 1)) + np.sum(np.maximum(0, -h_unclipped))
    return c5 + 0.01 * clip_penalty

def fourier_optimize(n, K, max_time, seed=42, h_init=None):
    """Optimize Fourier parameters to minimize C5"""
    rng = np.random.RandomState(seed)
    n_params = 2 * K
    best_c5 = float('inf')
    best_params = None
    start = time.time()
    
    n_restarts = 5
    time_per_restart = max_time / n_restarts
    
    for r in range(n_restarts):
        if h_init is not None and r == 0:
            # Initialize from h_init: project onto Fourier basis
            x = (np.arange(n) + 0.5) * 2.0 / n
            h0 = h_init - 0.5  # Remove DC component
            params = np.zeros(n_params)
            for k in range(1, K + 1):
                params[2*(k-1)] = 2.0 / n * np.sum(h0 * np.cos(k * np.pi * x))
                params[2*(k-1)+1] = 2.0 / n * np.sum(h0 * np.sin(k * np.pi * x))
        else:
            params = rng.uniform(-0.1, 0.1, n_params)
        
        res = minimize(
            fourier_objective, params,
            args=(n, K),
            method='Nelder-Mead',
            options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-8,
                     'adaptive': True}
        )
        c5 = compute_c5(fourier_h(res.x, n, K))
        if c5 < best_c5:
            best_c5 = c5
            best_params = res.x.copy()
    
    return fourier_h(best_params, n, K), best_c5

def continuous_local_search(h0, max_time, seed=42):
    """Local search with small perturbations"""
    n = len(h0)
    h = np.array(h0, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    # Normalize
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-10:
            h += room * (diff / total_room)
    h = np.clip(h, 0.0, 1.0)
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    eps_schedule = [0.1, 0.05, 0.02, 0.01, 0.005]
    eps_scale = 0
    
    while (time.time() - start) < max_time:
        eps = eps_schedule[min(eps_scale, len(eps_schedule)-1)]
        improved = False
        
        n_trials = min(500, n * n // 2)
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s = rng.choice([-1, 1])
            h[i] += s * eps
            h[j] -= s * eps
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= s * eps
                h[j] += s * eps
                continue
            new_c5 = compute_c5(h)
            if new_c5 < current_c5 - 1e-12:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                improved = True
            else:
                h[i] -= s * eps
                h[j] += s * eps
        
        if not improved:
            eps_scale += 1
    
    return best_h, best_c5

**Alternative: Use the subgradient of C₅**

C₅(h) = max_k corr_k(h) * dx

The subgradient of C₅ with respect to h at a point where the max is achieved at lag k* is:
∂C₅/∂h = dx * ∂corr_{k*}/∂h

And corr_k = Σ_i h[i+k](1-h[i]), so:
∂corr_k/∂h[j] = (1-h[j-k])_valid + (-h[j+k])_valid

Wait, let me be more careful. corr_k = Σ_i h[i+k] * (1-h[i]) where both indices are valid.

∂corr_k/∂h[j] = Σ_i [∂h[i+k]/∂h[j]] * (1-h[i]) + h[i+k] * ∂(1-h[i])/∂h[j]
= (1-h[j-k]) * I(valid j-k) + (-h[j+k]) * I(valid j+k)

where I(valid) means the index is in range.

So the subgradient is:
sg[j] = dx * ((1-h[j-k*]) * I(0≤j-k*<n) - h[j+k*] * I(0≤j+k*<n))

This gives us a direction to move in. We can do:
h_new = h - α * sg (projected onto the feasible set)

This is a subgradient projection method!

def subgradient_descent(h0, max_time, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = np.array(h0, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    # Ensure sum constraint
    target = n / 2.0
    while abs(np.sum(h) - target) > 1e-10:
        diff = target - np.sum(h)
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) == 0: break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if np.sum(mask) == 0: break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    rng = np.random.RandomState(seed)
    
    # Subgradient method with decreasing step size
    alpha = 0.01
    for step in range(10000):
        if (time.time() - start) > max_time:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        
        # Compute subgradient
        sg = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j - offset < n:
                val += (1.0 - h[j - offset])
            if 0 <= j + offset < n:
                val += (-h[j + offset])
            sg[j] = dx * val
        
        # Move
        h_new = h - alpha * sg
        # Project onto feasible set
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum constraint
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h_new < 1.0
                cnt = np.sum(mask)
                if cnt > 0:
                    h_new[mask] += diff / cnt
            else:
                mask = h_new > 0.0
                cnt = np.sum(mask)
                if cnt > 0:
                    h_new[mask] += diff / cnt
        h_new = np.clip(h_new, 0.0, 1.0)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        alpha *= 0.999  # Slowly decrease step size
    
    return best_h, best_c5

def fast_binary_ls(h, max_time, seed=42):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    current_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        
        # Try all single swaps
        best_gain = 1e-15
        best_p = -1
        best_q = -1
        
        # For speed, shuffle and potentially limit
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for p in ones:
            h[p] = 0.0
            for q in zeros:
                h[q] = 1.0
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_p = p
                    best_q = q
                h[q] = 0.0
            h[p] = 1.0
        
        if best_p >= 0:
            h[best_p] = 0.0
            h[best_q] = 1.0
            current_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                # Kick: random double swap
                if len(ones) >= 2 and len(zeros) >= 2:
                    idx_o = rng.choice(len(ones), 2, replace=False)
                    idx_z = rng.choice(len(zeros), 2, replace=False)
                    h[ones[idx_o[0]]] = 0.0
                    h[ones[idx_o[1]]] = 0.0
                    h[zeros[idx_z[0]]] = 1.0
                    h[zeros[idx_z[1]]] = 1.0
                    current_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    stall = 0
                else:
                    break
    
    return h, current_max * dx
