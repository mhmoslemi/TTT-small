```python
def optimize_coordinate(h, i, n):
    """Optimize h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute c_k for each lag
    # full[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # For coordinate i0=i, shift s = k-(n-1), k = s+(n-1)
    # Terms: h[i]*(1-h[i+s]) and h[i-s]*(1-h[i])
    
    def f(t):
        h_new = h.copy()
        h_new[i] = t
        c = np.correlate(h_new, 1 - h_new, mode='full')
        return np.max(c) * dx
    
    # Ternary search
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    return (a+b)/2
```

```python
    grid = np.linspace(0, 1, 60)
    best_t = h[i]
    best_v = np.inf
    for t in grid:
        h[i] = t
        v = np.max(np.correlate(h, 1 - h, mode='full'))
        if v < best_v:
            best_v = v
            best_t = t
    h[i] = best_t
    idx = int(round(best_t * 60))
    lo = max(0.0, grid[max(0, idx - 2)])
    hi = min(1.0, grid[min(59, idx + 2)])
    for _ in range(35):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        h[i] = m1
        v1 = np.max(np.correlate(h, 1 - h, mode='full'))
        h[i] = m2
        v2 = np.max(np.correlate(h, 1 - h, mode='full'))
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    h[i] = (lo + hi) / 2

def coord_descent(h, n, time_limit=5.0):
    t0 = time.time()
    for sweep in range(500):
        if time.time() - t0 > time_limit:
            break
        any_imp = False
        for i in range(n):
            old = h[i]
            optimize_coord(h, i, n)
            if abs(h[i] - old) > 1e-10:
                any_imp = True
        if not any_imp:
            break
    return h

def subgradient_descend(h, n, steps=200000, time_limit=20.0):
    t0 = time.time()
    dx = 2.0 / n
    alpha = 0.1
    best = h.copy()
    best_c5 = compute_c5(h, n)
    for step in range(steps):
        if time.time() - t0 > time_limit:
            break
        c = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            if s == 0:
                grad[i] = 1.0 - 2.0 * h[i]
            else:
                g = 1.0
                if 0 <= i + s < n:
                    g -= h[i + s]
                if 0 <= i - s < n:
                    g -= h[i - s]
                grad[i] = g
        grad *= dx
        h = np.clip(h - alpha * grad, 0.0, 1.0)
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        alpha *= 0.9999
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: Coordinate descent from parent
    best_h = coord_descent(best_h.copy(), n, time_limit=30)
    best_h = project_feasible(best_h, n)
    c = compute_c5(best_h, n)
    if c < best_c5:
        best_c5 = c

    # Phase 2: SLSQP continuation
    for alpha in [50, 200, 1000, 5000, 20000]:
        if budget_s - (time.time() - t0) < 20:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-18})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-12:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    # Phase 3: Coordinate descent again after SLSQP
    best_h = coord_descent(best_h.copy(), n, time_limit=20)
    best_h = project_feasible(best_h, n)
    c = compute_c5(best_h, n)
    if c < best_c5:
        best_c5 = c

    # Phase 4: Subgradient descent for exploration
    if budget_s - (time.time() - t0) > 100:
        h_sub, c_sub = subgradient_descend(best_h.copy(), n, steps=500000, time_limit=60)
        if c_sub < best_c5:
            best_h, best_c5 = h_sub, c_sub
            # Polish
            best_h = coord_descent(best_h.copy(), n, time_limit=15)
            best_h = project_feasible(best_h, n)
            c = compute_c5(best_h, n)
            if c < best_c5:
                best_c5 = c

    # Phase 5: Multiple restarts with perturbations
    for trial in range(10):
        if budget_s - (time.time() - t0) < 80:
            break
        scale = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        h_pert = coord_descent(h_pert.copy(), n, time_limit=8)
        h_pert = project_feasible(h_pert, n)
        c = compute_c5(h_pert, n)
        if c < best_c5 - 1e-10:
            best_h, best_c5 = h_pert.copy(), c
            # SLSQP polish
            try:
                res = minimize(obj_smooth, best_h, args=(n, 5000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-16})
                ht = project_feasible(res.x, n)
                c2 = compute_c5(ht, n)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = ht
            except Exception:
                pass

    # Phase 6: Thue-Morse inspired restart
    if budget_s - (time.time() - t0) > 60:
        tm = np.zeros(n, dtype=int)
        for i in range(1, n):
            tm[i] = tm[i >> 1] ^ (i & 1)
        h_tm = (1 - tm).astype(float)
        h_tm = project_feasible(h_tm, n)
        h_tm = coord_descent(h_tm.copy(), n, time_limit=10)
        h_tm = project_feasible(h_tm, n)
        c = compute_c5(h_tm, n)
        if c < best_c5 - 1e-10:
            best_h, best_c5 = h_tm.copy(), c
            try:
                res = minimize(obj_smooth, best_h, args=(n, 10000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-18})
                ht = project_feasible(res.x, n)
                c2 = compute_c5(ht, n)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = ht
            except Exception:
                pass

    # Phase 7: Try larger n if time permits
    if budget_s - (time.time() - t0) > 200:
        n2 = 120
        x_fine = np.linspace(0, 2, n2, endpoint=False)
        x_coarse = np.linspace(0, 2, n, endpoint=False)
        h2 = np.interp(x_fine, x_coarse, best_h)
        h2 = project_feasible(h2, n2)
        h2 = coord_descent(h2.copy(), n2, time_limit=20)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)
        if c2 < best_c5 - 1e-6:
            bounds2 = [(0.0, 1.0)] * n2
            cons2 = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n2,)}]
            try:
                res = minimize(obj_smooth, h2, args=(n2, 5000), method='SLSQP',
                              bounds=bounds2, constraints=cons2,
                              options={'maxiter': 5000, 'ftol': 1e-16})
                ht2 = project_feasible(res.x, n2)
                c3 = compute_c5(ht2, n2)
                if c3 < best_c5:
                    best_h, best_c5, n = ht2, c3, n2
            except Exception:
                pass
            if c2 < best_c5:
                best_h, best_c5, n = h2, c2, n2

    # Final: coordinate descent polish
    if budget_s - (time.time() - t0) > 10:
        best_h = coord_descent(best_h.copy(), n, time_limit=15)
        best_h = project_feasible(best_h, n)
        c = compute_c5(best_h, n)
        if c < best_c5:
            best_c5 = c

    # Final SLSQP
    if budget_s - (time.time() - t0) > 10:
        try:
            bounds_f = [(0.0, 1.0)] * n
            cons_f = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
            res = minimize(obj_smooth, best_h, args=(n, 50000), method='SLSQP',
                          bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 10000, 'ftol': 1e-18})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c5:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
