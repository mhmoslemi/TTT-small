    """
    h: length n
    p: length 2n-1, indexed by k = -(n-1), ..., n-1
       p[m] corresponds to k = m - (n-1)
    Returns gradient of sum_k p[k]*c[k] w.r.t. h
    """
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n-1):
        k = m - (n-1)
        if abs(p[m]) < 1e-15:
            continue
