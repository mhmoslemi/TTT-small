    def __init__(self, n_dim, obj_func, bounds,

        return 1.0

            self.trials += self.evaluate()
            if np.min(self.scores) < self.best_fitness:
                self.best_fitness = np.min(self.scores)
                self.best_vector = self.population[np.argmin(self.scores)]
            self.sigma *= 0.99
            if self.sigma < 0.01:
                self.sigma = 0.01
            self.evolve()
        return self.best_vector, self.best_fitness

    def evaluate(self):
        evaluated = []
        scores = []
        for i in range(len(self.population)):
            v = self.population[i].copy()
            f = self.obj(v)
            evaluated.append(v)
            scores.append(f)
        self.scores += scores
        return evaluated, scores
