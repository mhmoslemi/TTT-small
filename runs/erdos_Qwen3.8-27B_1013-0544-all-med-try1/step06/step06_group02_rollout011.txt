    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, dx

def compute_obj(h):
    corr, dx = compute_corr(h)
    return corr.max() * dx

def grad_for_shift(h, k_shift, N):
    offset = N - 1 - k_shift
    g = np.zeros(N)
    s1 = max(0, -offset)
    e1 = min(N, N - offset)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 + offset:e1 + offset]
    s2 = max(0, offset)
    e2 = min(N, N + offset)
    if s2 < e2:
        g[s2:e2] -= h[s2 - offset:e2 - offset]
    return g

def topk_grad(h, corr, dx, k_top, temp):
    N = len(h)
    idx = np.argsort(corr)[::-1][:k_top]
    vals = corr[idx]
    w = np.exp(temp * (vals - vals.max()))
    w /= w.sum()
    g = np.zeros(N)
    for i, wi in zip(idx, w):
        if wi > 1e-10:
            g += wi * grad_for_shift(h, i, N)
    return g * dx

def optimize(h0, N, rng, t_end, try_update, max_iter, k_top=5, temp=5.0, lr0=0.05):
    h = h0.copy()
    val = compute_obj(h)
    try_update(h, val)
    corr, dx = compute_corr(h)
    lr = lr0
    best = val
    best_h = h.copy()
    stag = 0
    
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = topk_grad(h, corr, dx, k_top, temp)
        h_new = project_feasible(h - lr * g, N)
        val_new = compute_obj(h_new)
        corr_new, _ = compute_corr(h_new)
        if val_new < val - 1e-15:
            h, val, corr = h_new, val_new, corr_new
            stag = 0
            lr = min(lr * 1.02, 0.2)
            if val < best:
                best, best_h = val, h.copy()
        else:
            stag += 1
            lr *= 0.8
            if stag > 25:
                sc = 0.02 + 0.06 * min(stag / 80.0, 1.0)
                h = project_feasible(h + rng.randn(N) * sc, N)
                val = compute_obj(h)
                corr, dx = compute_corr(h)
                lr = lr0
                stag = 0
                if val < best:
                    best, best_h = val, h.copy()
    
    try_update(best_h, best)
    return best_h, best

def interp(h, n_dst):
    x0 = np.arange(len(h)) / len(h)
    x1 = np.arange(n_dst) / n_dst
    return project_feasible(np.interp(x1, x0, h), n_dst)

Let me think about what would be a more fundamentally different approach...

**Alternative: Subgradient method with momentum**

Since the objective is f(h) = max_k g_k(h) where g_k(h) = corr_k(h) * dx, and each g_k is a quadratic function of h (bilinear actually), we can use:
- Subgradient at h is: grad of the active (maximizing) g_k
- With momentum: v = beta*v - lr*grad; h_new = project(h + v)

This could be more effective because momentum helps navigate through the "plateau" regions where the active set changes.

**Alternative: Penalty method**

Introduce a penalty parameter t and solve:
minimize sum_k max(0, g_k(h) - t)^2 + lambda * (constraint violation)^2

This converts the max problem into a smooth problem.

Actually, I think the most promising improvement is to combine:
1. **Momentum-based gradient descent** (Nesterov or Adam-like)
2. **Adactive k_top** (start small, increase as we converge)
3. **Binary-like initialization** (start from a good binary sequence)

A better construction: use a "Thue-Morse" like sequence or a "Golay" sequence.

Wait, but Adam might not work well here because the constraint projection changes the effective step. Also, the "rejection" of steps (when val_new >= val) means we're doing a line search within Adam, which is unusual.

Let me think about this differently. The fundamental issue is that we're minimizing a non-smooth function (max of smooth functions). The active set (which lag achieves the max) changes as we optimize. This makes the problem tricky.

**Key insight**: The optimal solution likely has multiple lags achieving (or nearly achieving) the maximum simultaneously. At the optimum, the gradients of all active constraints should be linearly dependent (KKT condition for the continuous relaxation).

So a good approach is:
1. Identify all "active" lags (those within epsilon of the max)
2. Find a direction that reduces all of them simultaneously
3. This is the negative projection of -1 onto the cone generated by the active gradients

This is essentially a **bundle method** or **cutting-plane method** idea.

Let me implement this:

```python
def active_set_gradient(h, corr, dx, margin=0.001):
    """Find direction that reduces all near-maximum correlations."""
    N = len(h)
    max_val = corr.max()
    active = np.where(corr >= max_val - margin)[0]
    
    # Sum of gradients of active constraints
    g_sum = np.zeros(N)
    for k in active:
        g_sum += grad_for_shift(h, k, N)
    
    # Normalize
    norm = np.linalg.norm(g_sum)
    if norm > 1e-10:
        g_sum /= norm
    
    return g_sum * dx
```

This is simpler than the top-k approach and directly targets the KKT condition. But it might be too aggressive (all active constraints weighted equally).

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
