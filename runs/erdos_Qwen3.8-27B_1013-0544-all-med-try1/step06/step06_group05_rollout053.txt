def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def normalize_h(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)

        h += diff / n

            h[idx] = 0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_h(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_h(h_new, n_new)

def make_binary(h0, n):
    h = np.round(np.asarray(h0, dtype=float)).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

    T0 = 0.04
    T_min = 1e-8
    no_improve = 0

    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / max_time)
        eff_frac = min(1.0, frac + 0.3 * no_improve / 8000)
        T = max(T_min, T0 * (1.0 - eff_frac) ** 2)

        for it in range(300):
            if time.time() - start >= max_time:
                return best_h, best_c5

                i = rng.randint(n)

                    h = h_prev

                scale = 0.04 + 0.3 * (T / T0)
                d = rng.uniform(-scale, scale)
                h[i] += d
                h[j] -= d
            elif r < 0.60:
                idx = rng.choice(n, 2, replace=False)
                h[idx[0]], h[idx[1]] = h[idx[1]], h[idx[0]]
            elif r < 0.80:

                noise -= noise.mean()
                h[s:s + blen] += noise
            else:
                i1, j1, i2, j2 = rng.choice(n, 4, replace=False)
