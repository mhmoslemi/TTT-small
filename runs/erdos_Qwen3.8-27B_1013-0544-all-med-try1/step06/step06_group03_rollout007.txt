    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def hard_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    return _conv_grad_single(h, shift) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def momentum_optimize(h0, n_iter, lr, t_start, t_end, hard=False, momentum=0.9, stall_threshold=500):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_lr = lr
    stall = 0

    for it in range(n_iter):
        progress = (it + 1) / n_iter
        t_cur = t_start * (t_end / t_start) ** progress

        y = np.clip(h + momentum * v, 0.0, 1.0)
        if hard:
            grad = hard_subgrad(y)
        else:
            grad = soft_max_grad(y, t_cur)

        h_new = project_feasible(h - cur_lr * grad, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            # Perturb and reset
            h_pert = project_feasible(h_new + 0.05 * (np.random.rand(n) - 0.5), n)
            v = 0.0
            cur_lr = lr * 0.5
            stall = 0
            h = h_pert
        else:
            v = h_new - h
            h = h_new
            if c5_new <= compute_c5(h) + 1e-14:
                cur_lr = min(cur_lr * 1.03, 0.1)
            else:
                cur_lr *= 0.6
                if cur_lr < 1e-15:
                    break

    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    xn = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    hi = np.interp(xn, xo, h)
    return project_feasible(hi, n_new)

def lbfgs_smooth(h0, t_val, max_iter=2000):
    n = len(h0)
    def obj(h):
        conv = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(conv)
        dx = 2.0 / n
        exps = np.exp((conv - m) / t_val)
        return t_val * (m / t_val + np.log(np.sum(exps))) * dx + 500.0 * (np.sum(h) - n / 2.0) ** 2
    def grad(h):
        conv = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(conv)
        dx = 2.0 / n
        exps = np.exp((conv - m) / t_val)
        weights = exps / np.sum(exps)
        g = np.zeros(n)
        for idx in range(len(conv)):
            w = weights[idx]
            if w < 1e-10:
                continue
            shift = idx - (n - 1)
            g += w * _conv_grad_single(h, shift)
        return g * dx + 1000.0 * (np.sum(h) - n / 2.0)
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(50, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n0 = len(h0)

    # Phase 1: Quick momentum optimization at original resolution
    if tl() > 120:
        ht, c5t = momentum_optimize(best_h, 15000, 0.01, 0.08, 0.002, hard=False, momentum=0.85)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Hard subgradient refinement
    if tl() > 100:
        ht, c5t = momentum_optimize(best_h, 20000, 0.003, 0.01, 0.001, hard=True, momentum=0.8)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: L-BFGS-B with smooth objective at original resolution
    if tl() > 80:
        for t_val in [0.02, 0.005]:
            if tl() < 30:
                break
            ht = lbfgs_smooth(best_h, t_val, max_iter=1500)
            ht, c5t = momentum_optimize(ht, 10000, 0.005, 0.05, 0.002, hard=False)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 4: Multi-scale - go lower, optimize, come back
    for n_low in [30, 40, 50, 60]:
        if tl() < 150:
            break
        n_cur = len(best_h)
        if n_low >= n_cur:
            continue
        h_low = interp_h(best_h, n_low)
        # Optimize aggressively at low resolution
        h_low, c5_low = momentum_optimize(h_low, 15000, 0.02, 0.1, 0.005, hard=False, momentum=0.9, stall_threshold=300)
        h_low, c5_low2 = momentum_optimize(h_low, 10000, 0.005, 0.01, 0.001, hard=True, momentum=0.85)
        if c5_low2 < c5_low:
            h_low = h_low  # keep best
        # Upsample back
        h_up = interp_h(h_low, n_cur)
        h_up, c5_up = momentum_optimize(h_up, 15000, 0.003, 0.05, 0.001, hard=False, momentum=0.85)
        h_up, c5_up2 = momentum_optimize(h_up, 15000, 0.001, 0.005, 0.001, hard=True, momentum=0.8)
        c5_final = min(c5_up, c5_up2)
        if c5_final < best_c5:
            best_h = h_up if c5_up <= c5_up2 else h_up
            best_c5 = c5_final
            # Re-identify best
            best_h, best_c5 = (h_up, c5_up) if c5_up <= c5_up2 else (h_up, c5_up2)

    # Phase 5: Try higher resolution
    for n_high in [400, 480, 560, 640]:
        if tl() < 180:
            break
        n_cur = len(best_h)
        if n_high <= n_cur:
            continue
        h_hi = interp_h(best_h, n_high)
        h_hi, c5_hi = momentum_optimize(h_hi, 12000, 0.004, 0.03, 0.002, hard=False, momentum=0.85)
        h_hi, c5_hi2 = momentum_optimize(h_hi, 15000, 0.001, 0.005, 0.001, hard=True, momentum=0.8)
        if c5_hi2 < best_c5:
            best_h, best_c5 = h_hi, c5_hi2
        elif c5_hi < best_c5:
            best_h, best_c5 = h_hi, c5_hi

    # Phase 6: Structured random restarts at current best resolution
    n_cur = len(best_h)
    for restart in range(25):
        if tl() < 60:
            break
        # Structured perturbation: mix of low-freq and random
        x = np.linspace(0, 2.0, n_cur)
        n_comps = rng.randint(2, 5)
        pert = np.zeros(n_cur)
        for c in range(n_comps):
            f = rng.uniform(0.5, 4.0)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.05, 0.25)
            pert += a * np.sin(2 * np.pi * f * x / 2.0 + p)
        hp = project_feasible(best_h + pert, n_cur)
        ht, c5t = momentum_optimize(hp, 8000, 0.008, 0.05, 0.003, hard=False, momentum=0.9, stall_threshold=200)
        ht, c5t2 = momentum_optimize(ht, 10000, 0.002, 0.01, 0.001, hard=True, momentum=0.85)
        if c5t2 < best_c5:
            best_h, best_c5 = ht, c5t2

    # Phase 7: Final hard refinement (long, small steps)
    if tl() > 40:
        ht, c5t = momentum_optimize(best_h, 40000, 0.0008, 0.003, 0.001, hard=True, momentum=0.7, stall_threshold=2000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 8: Very final touch - single active shift optimization
    if tl() > 20:
        h = best_h.copy()
        n = len(h)
        for _ in range(5000):
            if tl() < 10:
                break
            conv = np.correlate(h, 1.0 - h, mode="full")
            m_idx = np.argmax(conv)
            shift = m_idx - (n - 1)
            grad = _conv_grad_single(h, shift) * (2.0 / n)
            for step in [0.002, 0.001, 0.0005, 0.0002]:
                h_try = project_feasible(h - step * grad, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    break

    return best_h, best_c5, len(best_h)
