        # Add ones
        deficit = target_sum - len(ones_idx)
        for i in range(deficit):
            h[zeros_idx[i]] = 1.0

    best_c5 = _compute_c5(h, dx)

        improved = False
        # Try flipping each pair

                c5 = _compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def _compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _compute_c5(h, dx):
    return np.max(_compute_corr(h, dx))

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_smooth_max(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exp_c = np.exp(alpha * (c - m))
    w_sum = np.sum(exp_c)
    weights = exp_c / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * dx * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * dx * h[lag:]
        else:
            ll = -lag
            grad[ll:] += w * dx * (1.0 - h[:n - ll])
            grad[:n - ll] -= w * dx * h[ll:]
    return grad, m

def _optimize_gradient(h_init, dx, time_limit, alpha_init=8.0):
    n = len(h_init)
    target_sum = n / 2.0
    h = _project(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    lr = 0.003
    alpha = alpha_init
    stale = 0
    while time.time() < time_limit:
        grad, c5_val = _grad_smooth_max(h, dx, alpha)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        h_new = _project(h - lr * grad, target_sum)
        c5_new = _compute_c5(h_new, dx)
        if c5_new < c5_val:
            h = h_new
            lr = min(lr * 1.04, 0.03)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
        alpha = min(alpha * 1.003, 60.0)
        if stale > 500:
            break
    return best_h, best_c5

def _optimize_binary(h_init, dx, time_limit, rng):
    n = len(h_init)
    target_ones = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    if len(ones_idx) > target_ones:
        remove = rng.choice(ones_idx, len(ones_idx) - target_ones, replace=False)
        h[remove] = 0.0
    elif len(ones_idx) < target_ones:
        add = rng.choice(zeros_idx, target_ones - len(ones_idx), replace=False)
        h[add] = 1.0
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    while time.time() < time_limit:
        improved = False
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]

            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-14:

                best_h = h.copy()
                improved = True
            else:
                h[oi] = 1.0
                h[zi] = 0.0
        if not improved:
            break
    return best_h, best_c5
