                          bounds=bounds, constraints=constraints,

        h_new = project_to_feasible(h_new, target_sum)

        h_new = h
    
    c5_new, _ = compute_c5(h_new)
    c5_old, _ = compute_c5(h)
    if c5_new < c5_old:
        return h_new, c5_new
    return h, c5_old

    n = len(h)

    return np.max(corr), corr

def project_to_feasible(h, target_sum):
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    lo, hi = h.min() - 1.0, h.max() + 1.0
    if f(lo) <= 0 or f(hi) >= 0:
        return np.clip(h, 0.0, 1.0)
    lam = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h - lam, 0.0, 1.0)

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = alpha * corr
    mc = np.max(c)
    if mc > 500:
        exps = np.zeros_like(corr)
        exps[np.argmax(corr)] = 1.0
    else:
        exps = np.exp(c - mc)
        exps /= np.sum(exps)
    grad = np.zeros(n)
