import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += diff * room / tot
        else:
            room = h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += diff * room / tot
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return np.clip(h, 0.0, 1.0)

def nm_optimize(h, target, max_iter=500, t_limit=None, t0=None):
    n = len(h)
    def obj(x):
        xp = project(x, target)
        return evaluate_c5(xp)
    best_x = h.copy()
    best_f = obj(h)
    for attempt in range(3):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        res = minimize(obj, best_x, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-12,
                               'adaptive': True})
        h_new = project(res.x, target)
        f_new = evaluate_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_x = h_new
    return best_x, best_f

def coord_descent(h, target, n_steps=2000, t_limit=None, t0=None):
    n = len(h)
    best_c5 = evaluate_c5(h)
    h_cur = h.copy()
    for step in range(n_steps):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.1, 0.1)
        h_trial = h_cur.copy()
        h_trial[idx] += delta
        h_trial = project(h_trial, target)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            h_cur = h_trial
            best_c5 = c5_trial
    return h_cur, best_c5
