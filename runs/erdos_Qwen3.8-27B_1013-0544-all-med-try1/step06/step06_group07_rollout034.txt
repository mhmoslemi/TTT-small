import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

_SIGMA = {'tau': 0.01}

def _set_tau(t):
    _SIGMA['tau'] = float(t)

def _obj(x):
    return smooth_obj(x, _SIGMA['tau'])

def _grad(x):
    return smooth_grad(x, _SIGMA['tau'])

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(30):
        d = target - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = 1 - x > 1e-14
            if not m.any():
                break
            x[m] += d * (1 - x[m]) / (1 - x)[m].sum()
        else:
            m = x > 1e-14
            if not m.any():
                break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = target - x.sum()
    if abs(d) > 1e-10:
        k = int(target)
        x[:] = 0
        if k > 0:
            x[:k] = 1
        r = target - k
        if 0 < r < 1 and k < n:
            x[k] = r
    return x

def cvals(h):

    return np.correlate(h, 1 - h, mode="full") * (2.0 / n)

def exact_c(h):
    return float(np.max(cvals(h)))

def add_lag_grad(x, grad, k, scale):
    n = x.size
    s = k - (n - 1)
    lo = max(s, 0)
    hi = min(n - 1, (n - 1) + s)
    if lo <= hi:
        grad[lo:hi + 1] += scale * (1 - x[lo - s:hi + 1 - s])
    lo = max(-s, 0)
    hi = min(n - 1, (n - 1) - s)
    if lo <= hi:
        grad[lo:hi + 1] -= scale * x[lo + s:hi + 1 + s]

def smooth_obj(x, tau):
    c = cvals(x)
    cm = np.max(c)
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    return cm + tau * np.log(e.sum())

def smooth_grad(x, tau):
    n = x.size
    c = cvals(x)
    cm = np.max(c)
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    w = e / e.sum()
    g = np.zeros(n)
    for k in np.nonzero(w > 1e-12)[0]:
        add_lag_grad(x, g, int(k), float(w[k]))
    return g * (2.0 / n)

def run_lbbgfb(h0, tau, tl):
    n = h0.size
    _set_tau(tau)
    try:
        r = so.minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * n,
                        options={'maxiter': 300, 'ftol': 1e-15, 'maxfun': 10000, 'maxls': 50})
        return repair(r.x)
    except Exception:
        return repair(h0)

def pair_descent(x, ti_limit, seed=42):
    x = repair(x).copy()
    n = x.size
    dx = 2.0 / n
    best_c = exact_c(x)
    best_x = x.copy()
    rng = np.random.RandomState(seed)
    dl = time.time() + ti_limit
    stall = 0
    while time.time() < dl:
        c = cvals(x)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        tol = max(1e-9, 5e-5 * C)
        act = np.where(c >= C - tol)[0]
        if len(act) > 12:
            act = np.argsort(c)[::-1][:12]
        ntr = min(400, n * n // 2)
        imp = 0
        for _ in range(ntr):
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            ca = c[act]
            co = np.zeros(len(act))
            for idx in range(len(act)):
                s = int(act[idx]) - (n - 1)
                cv = 0.0
                if 0 <= i + s < n:
                    cv += 1.0 - x[i + s]
                if 0 <= i - s < n:
                    cv -= x[i - s]
                if 0 <= j + s < n:
                    cv -= (1.0 - x[j + s])
                if 0 <= j - s < n:
                    cv += x[j - s]
                co[idx] = cv * dx
            dlo = max(-x[i], x[j] - 1.0)
            dhi = min(1.0 - x[i], x[j])
            if dlo >= dhi - 1e-15:
                continue
            cands = [dlo, dhi]
            for a in range(len(co)):
                for b in range(a + 1, len(co)):
                    dc = co[a] - co[b]
                    if abs(dc) < 1e-15:
                        continue
                    d = (ca[b] - ca[a]) / dc
                    if dlo <= d <= dhi:
                        cands.append(d)
            bd = 0.0
            bv = float(np.max(ca))
            for d in cands:
                v = float(np.max(ca + d * co))
                if v < bv - 1e-14:
                    bv = v
                    bd = d
            if abs(bd) > 1e-10:
                x[i] = np.clip(x[i] + bd, 0.0, 1.0)
                x[j] = np.clip(x[j] - bd, 0.0, 1.0)
                imp += 1
        if imp == 0 and stall > 4:
            idx = rng.permutation(n)[:6]
            for a in range(0, 6, 2):
                d = rng.uniform(-0.03, 0.03)
                x[idx[a]] = np.clip(x[idx[a]] + d, 0.0, 1.0)
                x[idx[a + 1]] = np.clip(x[idx[a + 1]] - d, 0.0, 1.0)
            x = repair(x)
            stall = 0
    xc = exact_c(x)
    if xc < best_c:
        best_c = xc
        best_x = x.copy()
    return repair(best_x), best_c

def grad_polish(x, ti_limit, seed=42):
    x = repair(x).copy()
    n = x.size
    best_c = exact_c(x)
    best_x = x.copy()
    rng = np.random.RandomState(seed)
    dl = time.time() + ti_limit
    step = 0.4
    no_imp = 0
    while time.time() < dl:
        c = cvals(x)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 12:
            break
        tol = max(1e-9, 5e-5 * C)
        act = np.where(c >= C - tol)[0]
        if len(act) > 6:
            act = np.argsort(c)[::-1][:6]
        g = np.zeros(n)
        for k in act:
            add_lag_grad(x, g, int(k), 1.0)
        g *= (2.0 / n / len(act))
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        accepted = False
        for _ in range(14):
            y = repair(x - step * g)
            cy = exact_c(y)
            if cy < C - 1e-13:
                x = y
                accepted = True
                break
            step *= 0.5
            if step < 1e-13:
                break
        if not accepted:
            if rng.rand() < 0.25:
                y = repair(x + rng.normal(0, 0.003, n))
                cy = exact_c(y)
                if cy < best_c:
                    best_c = cy
                    best_x = y
                    x = y
                    no_imp = 0
            else:
                break
    return repair(best_x), best_c

def resample(h0, nn):
    h0 = np.asarray(h0, dtype=float)
    n0 = len(h0)
    nn = int(nn)
    if n0 == nn:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(nn) + 0.5) * (2.0 / nn)
    return repair(np.interp(x1, x0, h0))

def make_starts(n, seed, init):
    starts = []
    if init is not None and init.size > 0:
        starts.append(resample(init, n))
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    for f, a in [(2, 0.35), (3, 0.30), (5, 0.25)]:
        h = 0.5 + a * np.sin(2 * np.pi * f * t / 2.0 + rng.uniform(0, 6.28))
        h += 0.03 * rng.normal(0, 1, n)
        starts.append(repair(h))
    for sc in [0.02, 0.06, 0.12]:
        if init is not None and init.size > 0:
            base = resample(init, n)
            starts.append(repair(base + rng.normal(0, sc, n)))
        else:
            starts.append(repair(np.full(n, 0.5) + rng.normal(0, sc * 0.5, n)))
    return starts

def run(seed=42, budget_s=1100, **kwargs):

    sv = int(seed)
    raw = globals().get("initial_h_values", None)

    if init is None or init.size < 10:

    base_n = init.size

    resolutions = []
    for r in [base_n, base_n + 20, base_n - 20, base_n + 40, 90, 120, 140, 100]:
        r = int(r)
        if 40 <= r <= 180 and r not in resolutions:
            resolutions.append(r)
    if len(resolutions) > 6:
        resolutions = resolutions[:6]

    best_x = None
    best_c = 1e9

    for ri, n in enumerate(resolutions):
        if time.time() > dl - 80:
            break
        remaining = dl - time.time()
        alloc = min(remaining * 0.35, 250.0)
        if alloc < 20:

        starts = make_starts(n, sv + ri * 37, init)
        n_starts = min(len(starts), 4 if ri == 0 else 3)

        for si in range(n_starts):

            h0 = starts[si]

            per = alloc / n_starts

            # Phase 2: Pairwise descent

    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
