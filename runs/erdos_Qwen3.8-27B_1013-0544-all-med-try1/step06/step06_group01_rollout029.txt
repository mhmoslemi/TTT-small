```python
                # delta in [-h[k], min(1-h[l]... wait

                # h[l] new = h[l] - delta, must be in [0,1] → delta in [h[l]-1, h[l]]

                b = lo + gr*(hi-lo)

                fa = _c5_max(h)
                h[k] -= a; h[l] += a
                h[k] += b; h[l] -= b
                fb = _c5_max(h)
                h[k] -= b; h[l] += b
                # Iterate
                for _ in range(15):
                    if fa < fb:
                        hi = b; b = a; fb = fa
                        a = lo + (1-gr)*(hi-lo)
                        h[k] += a; h[l] -= a
                        fa = _c5_max(h)
                        h[k] -= a; h[l] += a
                    else:
                        lo = a; a = b; fa = fb
                        b = lo + gr*(hi-lo)
                        h[k] += b; h[l] -= b
                        fb = _c5_max(h)
                        h[k] -= b; h[l] += b
                # Best point
                best_delta = (a + b) / 2
                h[k] += best_delta; h[l] -= best_delta
                fc = _c5_max(h)
                h[k] -= best_delta; h[l] += best_delta
                if fc < best_c5 - 1e-12:
                    h[k] += best_delta; h[l] -= best_delta
                    best_c5 = fc
                    improved = True
        if not improved:
            break
    return h, best_c5
```

Actually, let me try a different approach: **direct min-max coordinate descent**.

For coordinate k (with sum constraint handled by moving mass between k and a "reservoir"):
- The max C5 as a function of h[k] (with all else fixed except the sum constraint) is a convex piecewise-linear function
- We can find its minimum in O(n) by finding the relevant breakpoints

But the sum constraint makes this tricky. If I change h[k], I need to change something else to maintain the sum. The "something else" could be:
- A specific other coordinate (pair swap)
- All other coordinates proportionally (projection)

Let me use the projection approach:
1. For coordinate k, find h*[k] that minimizes max C5 (ignoring sum constraint)
2. Compute the sum violation: Δ = h*[k] - h[k]
3. Distribute -Δ proportionally to all other coordinates (weighted by their "room")
4. Clip and re-project

This is approximate but should work well in practice.

For step 1: max C5 as a function of v = h[k] (with all other h fixed):
C[m](v) = C[m]_0 + (v - h[k]) * slope[m]
where slope[m] = dx * [(1-h[m-k+n-1]) * I(valid) - h[m-n+1-k] * I(valid)]

The max is minimized at the point where the "knee" of the upper envelope occurs. For a set of lines, this is where two lines cross at the maximum level.

Simplified approach: evaluate max C5 at v = 0, v = 1, and v = the point that equalizes the two largest C[m] values. This is O(n) per coordinate.

```python
def _coord_descent(h, n, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_c5 = _c5_max(h)
    h = h.copy()
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for k in order:
            # Compute slopes for each m
            s = np.arange(-(n-1), n)
            a_idx = k - s
            b_idx = k + s
            mask_a = (a_idx >= 0) & (a_idx < n)
            mask_b = (b_idx >= 0) & (b_idx < n)
            slopes = dx * (np.where(mask_a, 1.0 - h[a_idx[mask_a]], 0.0) 
                         - np.where(mask_b, h[b_idx[mask_b]], 0.0))
            
            # Current C values
            C = np.correlate(h, 1.0 - h, mode='full') * dx
            
            # C[m] as function of h[k]=v: C[m] + (v - h[k]) * slopes[m]
            # Find v that minimizes max_m(C[m] + (v-h[k])*slopes[m])
            # This is a 1D convex optimization
            
            # Try v = 0, v = 1, and v where top-2 correlations are equal
            candidates = [0.0, 1.0, h[k]]
            
            # Find the two largest C[m] and their slopes
            top2 = np.argsort(-C)[:2]
            m1, m2 = top2[0], top2[1]
            if slopes[m1] != slopes[m2]:
                v_eq = h[k] + (C[m2] - C[m1]) / (slopes[m1] - slopes[m2])
                if 0 <= v_eq <= 1:
                    candidates.append(v_eq)
            
            best_v = h[k]
            best_val = best_c5
            for v in candidates:
                h[k] = v
                # Project sum
                h = _fix_sum(h, n)
                val = _c5_max(h)
                if val < best_val:
                    best_val = val
                    best_v = v
            h[k] = best_v
            h = _fix_sum(h, n)
            val = _c5_max(h)
            if val < best_c5 - 1e-12:
                best_c5 = val
                improved = True
        if not improved:
            break
    return h, best_c5
```

```python
def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff

                step = min(remaining, 1.0 - h[idx])
            else:
                step = max(remaining, -h[idx])
            h[idx] += step
            remaining -= step

    n = len(h)

    m = np.max(C)
    es = np.exp(alpha * (C - m))
    return m + np.log(np.sum(es)) / alpha + beta * (np.sum(h) - 0.5*n)**2

    m = np.max(C)

    g = np.zeros(n)

    if res.x is not None:

                    best = c5
                    improved = True
                    break
                h[k] -= delta
                h[l] += delta
        if not improved:
            break
```
