        elif c5_new < best_c5:

    dx = 2.0 / n

One more idea: **alternating minimization**. For a fixed shift k, the optimal h that minimizes ∫h(x)(1-h(x+k))dx (subject to the other constraints) can be found. If I cycle through the active shifts and optimize for each one, I might converge faster.

For a fixed shift k, the objective is:
f_k(h) = sum_i h[i] * (1 - h[i+k]) (over valid i)

This is a quadratic in h: f_k = sum_i h[i] - sum_i h[i]*h[i+k]

The gradient with respect to h[j] is:
- From the first term: 1 (for each i where j appears)
- From the second term: -h[j+k] (when j is the first factor) - h[j-k] (when j is the second factor)

This is exactly what _conv_grad_single computes.

So the "alternating" approach would be: for each active shift k, take a gradient step reducing f_k, then project. Cycle through all active shifts.

This is essentially equivalent to what PGD does with the subgradient (which picks one active shift at a time), but cycling through all active shifts might be more effective.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            if shift == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[shift:] += (1.0 - h[:n - shift])
                g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    obj = t * np.log(np.sum(exps)) * dx
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return obj, grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h_init, max_iter=20000, lr_init=0.005, smooth_t=0.01):
    n = len(h_init)
    h = h_init.copy()
    y = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    t_mom = 1.0
    stalled = 0
    for it in range(max_iter):
        _, grad = soft_max_obj_and_grad(y, smooth_t)
        h_new = project_feasible(y - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 6000:
                y = h_new.copy()
                t_mom = 1.0
                stalled = 0
                if lr < 1e-6:
                    break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        mu_new = (t_mom - 1.0) / t_new
        y = h_new + mu_new * (h_new - h)
        h = h_new
        t_mom = t_new
        if stalled > 400:
            lr *= 0.97
        elif stalled == 0:
            lr = min(lr * 1.04, 0.08)
    return best_h, best_c5

def topk_pgd(h_init, max_iter=40000, lr_init=0.002, topk=5):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    stalled = 0
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(conv)
    active = np.where(conv > max_val - 0.08 * max_val)[0]
    for it in range(max_iter):
        if it % 300 == 0:
            conv = np.correlate(h, 1.0 - h, mode="full")
            max_val = np.max(conv)
            active = np.where(conv > max_val - 0.08 * max_val)[0]
            k = min(topk, len(active))
            top_idx = active[np.argsort(conv[active])[-k:]]
            grad = np.zeros(n)
            for idx in top_idx:
                shift = idx - (n - 1)
                grad += _conv_grad_single(h, shift)
            grad = grad / k * dx
        else:
            shift = top_idx[idx - (n - 1)] if False else (np.argmax(conv) - (n - 1))
            conv = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(conv)
            shift = idx - (n - 1)
            grad = _conv_grad_single(h, shift) * dx
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stalled = 0
            lr = min(lr * 1.02, 0.04)
        else:
            stalled += 1
            lr *= 0.995
            if stalled > 20000 and lr < 1e-7:
                break
        h = h_new
    return best_h, best_c5

def pgd_refine(h_init, max_iter=50000, lr_init=0.001):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    stalled = 0
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = _conv_grad_single(h, shift) * (2.0 / n)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stalled = 0
            lr = min(lr * 1.01, 0.03)
        else:
            stalled += 1
            lr *= 0.996
            if stalled > 30000 and lr < 1e-8:
                break
        h = h_new
    return best_h, best_c5

def local_search(h0, n, rng, max_evals=4000, delta=0.015):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    temp = 0.004
    for it in range(max_evals):
        i, j = rng.choice(n, size=2, replace=False)
        amt = rng.uniform(0, delta)
        if rng.rand() < 0.5:
            di, dj = amt, -amt
        else:
            di, dj = -amt, amt
        hi, hj = h[i] + di, h[j] + dj
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
        elif rng.rand() > np.exp((best_c5 - c5_new) / max(temp, 1e-10)):
            h[i], h[j] = hi - di, hj - dj
        temp *= 0.9997
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(48, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: FISTA with decreasing smoothing
    for st in [0.05, 0.02, 0.01, 0.005, 0.003]:
        if tl() < 15:
            break
        ht, c5t = fista_optimize(best_h, max_iter=12000, lr_init=0.005, smooth_t=st)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Top-k PGD
    if tl() > 30:
        ht, c5t = topk_pgd(best_h, max_iter=30000, lr_init=0.002, topk=5)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: PGD subgradient refinement
    if tl() > 20:
        ht, c5t = pgd_refine(best_h, max_iter=40000, lr_init=0.0008)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Local search + re-optimize cycles
    for _ in range(6):
        if tl() < 25:
            break
        ht, c5t = local_search(best_h, len(best_h), rng, max_evals=3000, delta=0.015)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        if tl() < 30:
            break
        ht, c5t = fista_optimize(best_h, max_iter=8000, lr_init=0.003, smooth_t=0.005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        if tl() < 20:
            break
        ht, c5t = pgd_refine(best_h, max_iter=15000, lr_init=0.0004)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Try different resolutions
    current_n = len(best_h)
    n_candidates = [48, 64, 80, 96, 120, 160, 200, 240, 256, 320, 384, 480]
    n_candidates = [x for x in n_candidates if x != current_n]
    n_candidates.sort(key=lambda x: abs(x - current_n))

    for n_try in n_candidates:
        if tl() < 80:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False) + 1.0 / nc
        xn = np.linspace(0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        for st in [0.02, 0.005]:
            if tl() < 30:
                break
            ht, c5t = fista_optimize(hi, max_iter=8000, lr_init=0.004, smooth_t=st)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
                hi = best_h.copy()
        if tl() < 15:
            continue
        ht, c5t = pgd_refine(best_h, max_iter=10000, lr_init=0.0004)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: L-BFGS-B with soft-max
    for t_val in [0.008, 0.003]:
        if tl() < 40:
            break
        n = len(best_h)
        target_sum = n / 2.0
        def obj_fn(h):
            hp = project_feasible(h, n)
            c, _ = soft_max_obj_and_grad(hp, t_val)
            return c + 500.0 * (np.sum(h) - target_sum) ** 2
        def grad_fn(h):
            hp = project_feasible(h, n)
            _, g = soft_max_obj_and_grad(hp, t_val)
            return g + 1000.0 * (np.sum(h) - target_sum)
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12})
            ht = project_feasible(res.x, n)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 7: Fine local search
    for _ in range(10):
        if tl() < 12:
            break
        ht, c5t = local_search(best_h, len(best_h), rng, max_evals=2500, delta=0.008)
        if c5t < best_c5:

        if tl() < 15:

        ht, c5t = pgd_refine(best_h, max_iter=10000, lr_init=0.0002)
        if c5t < best_c5:

    # Final polish
    if tl() > 25:
        ht, c5t = pgd_refine(best_h, max_iter=100000, lr_init=0.00015)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
