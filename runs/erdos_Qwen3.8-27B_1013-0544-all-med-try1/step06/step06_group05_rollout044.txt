import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def normalize_h(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr > 1e-12:
                h += room * (diff / tr)
            else:
                h += diff / n
        else:
            tot = np.sum(h)
            if tot > 1e-12:
                h = h * (target / tot)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = diff
        for idx in order:
            if rem > 0:
                a = min(1.0 - h[idx], rem)
                h[idx] += a; rem -= a
            elif rem < 0:
                a = min(h[idx], -rem)
                h[idx] -= a; rem += a
            if abs(rem) < 1e-12:
                break
    return h

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        for idx in ones[:n_ones - target]:
            h[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1.0
    return h

def gradient_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = beta * (corr - mc)
    weights = np.exp(shifted)
    s = np.sum(weights)
    if s < 1e-30:
        return np.zeros(n)
    weights /= s
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        k = n - 1 + d
        w = weights[k] * dx
        if abs(w) < 1e-14:
            continue
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        idx1 = np.arange(i_lo, i_hi + 1)
        idx2 = idx1 + d
        grad[idx1] += w * (1.0 - h[idx2])
        grad[idx2] += -w * h[idx1]
    return grad

def gradient_descent(h0, max_time, seed=42):

    h = normalize_h(h0.copy(), n)
    current_c5 = compute_c5(h)
    best_h = h.copy()

    lr = 0.02

        frac = elapsed / max_time

        lr_cur = lr * (1.0 + 2.0 * (1.0 - frac))
        
        grad = gradient_smooth_max(h, beta)
        h_new = h - lr_cur * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = normalize_h(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5 - 1e-10:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Try random perturbation

                i = rng.randint(n)
                j = rng.randint(n)

                    eps = rng.uniform(0.01, 0.15)

                    h[j] = np.clip(h[j] - eps, 0, 1)
                    h = normalize_h(h, n)
                stall = 0
                lr = max(0.005, lr * 0.7)

                lr = max(0.002, lr * 0.8)
    
    return best_h, best_c5

            i = ones[rng.randint(len(ones))]

            gain = current_c5 - new_c5

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_h(h_new, n_new)

def harmonic_init(n, k, phase=0.0):
    """Create a harmonic initial condition."""
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + 0.3 * np.sin(2 * np.pi * k * x + phase)
    return normalize_h(h, n)

def block_init(n, n_blocks):
    """Create a block initial condition."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(0, n, block_size * 2):
        h[i:i+block_size] = 1.0
    return normalize_h(h, n)
