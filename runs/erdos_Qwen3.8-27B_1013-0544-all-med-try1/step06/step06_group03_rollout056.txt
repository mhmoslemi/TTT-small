    n = len(h)

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:n - s]
    return g

def soft_max_grad(h, t):
    n = len(h)

    ms = np.max(shifted)
    exps = np.exp(shifted - ms)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:n - s]
    return g

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    total = np.sum(exps)
    weights = exps / total
    grad = np.zeros(n)
    mask = weights > 1e-7
    for idx in np.where(mask)[0]:
        w = weights[idx]
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _lbfgs_obj(h, n, t):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    ms = np.max(shifted)
    lse = m / t + ms + np.log(np.sum(np.exp(shifted - ms)))
    return t * lse * dx + 200.0 * (np.sum(h) - n / 2.0) ** 2

def _lbfgs_grad(h, n, t):
    g = soft_max_grad(h, t) + 400.0 * (np.sum(h) - n / 2.0)
    return g

def nesterov_pgd(h_init, max_iter, t_init, t_end, lr_init, lr_end):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        frac = it / max(1, max_iter - 1)
        t = t_init * (t_end / t_init) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac
        mu = 0.9 - 0.35 * frac
        h_n = project_feasible(h + mu * v, n)
        grad = soft_max_grad(h_n, t)
        h_new = project_feasible(h_n - lr * grad, n)
        v = h_new - h_n
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def multi_active_pgd(h_init, max_iter, lr_init, n_active=5):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * (2.0 / n)
        top_count = min(n_active, len(conv))
        top_idx = np.argpartition(conv, -top_count)[-top_count:]
        grad = np.zeros(n)
        for idx in top_idx:
            shift = idx - (n - 1)
            grad += _conv_grad_single(h, shift)
        grad *= (2.0 / n) / top_count
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-13:
            lr *= 0.5
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.04, 0.015)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(200, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    if tl() > 80:
        for t_val in [0.02, 0.005]:
            if tl() < 30:
                break
            try:
                res = minimize(_lbfgs_obj, best_h, args=(n0, t_val),
                              jac=_lbfgs_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n0,
                              options={'maxiter': 500, 'ftol': 1e-15})
                ht = project_feasible(res.x, n0)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    if tl() > 200:
        ht, c5t = nesterov_pgd(best_h, 60000, 0.04, 0.0008, 0.004, 0.00008)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    if tl() > 200:
        for na in [10, 5, 3, 1]:
            if tl() < 50:
                break
            ht, c5t = multi_active_pgd(best_h, 30000, 0.001, n_active=na)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    if tl() > 150:
        ht, c5t = multi_active_pgd(best_h, 60000, 0.0004, n_active=1)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    if tl() > 120:
        candidates = []
        for s in [1, 2, 3, -1, -2]:
            candidates.append(np.roll(best_h, s))
        candidates.append(best_h[::-1].copy())
        candidates.append((best_h > 0.5).astype(float))
        for hp in candidates:
            if tl() < 35:
                break
            hp = project_feasible(hp, n0)
            ht, c5t = nesterov_pgd(hp, 8000, 0.03, 0.001, 0.003, 0.0001)
            ht, c5t2 = multi_active_pgd(ht, 12000, 0.0004, n_active=2)
            if c5t2 < best_c5:
                best_h, best_c5 = ht, c5t2

    for _ in range(15):
        if tl() < 40:
            break
        scale = 0.02 + 0.05 * rng.random()
        hp = project_feasible(best_h + rng.normal(0, scale, n0), n0)
        ht, c5t = nesterov_pgd(hp, 7000, 0.02, 0.001, 0.002, 0.0001)
        ht, c5t2 = multi_active_pgd(ht, 10000, 0.0003, n_active=2)
        if c5t2 < best_c5:
            best_h, best_c5 = ht, c5t2

    for n_try in [480, 640]:
        if tl() < 100:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        ht, c5t = nesterov_pgd(hi, 15000, 0.03, 0.001, 0.002, 0.00008)
        ht, c5t2 = multi_active_pgd(ht, 20000, 0.0002, n_active=1)
        if c5t2 < best_c5:
            best_h, best_c5 = ht, c5t2

    if tl() > 30:
        ht, c5t = multi_active_pgd(best_h, 120000, 0.00008, n_active=1)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
