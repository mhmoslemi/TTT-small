def project_box_sum(h, total):

        return np.zeros(n)

def soft_max_obj_grad(h, tau):
    n = len(h)

                print(f"  PG n={best_n} tau={tau} C5={best_c5:.6f}")
            cur = h_t
        if tl() > 20:
            h_t, c5_t = adam_solve(cur, 100.0, tl() * 0.15, lr=0.02)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = len(h_t)
                print(f"  Adam n={best_n} tau=100 C5={best_c5:.6f}")

    # Phase 3: upsample x2 again
    if tl() > 40:
        h3 = upsample_linear(best_h, 2)
        n3 = len(h3)
        print(f"Phase 3: n={n3}, C5_up={compute_c5(h3):.6f}")
        cur = h3
        for tau in [60.0, 150.0]:
            if tl() < 10:
                break
            h_t, c5_t = pg_solve(cur, tau, tl() * 0.3)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = len(h_t)
                print(f"  PG n={best_n} tau={tau} C5={best_c5:.6f}")
            cur = h_t

    # Phase 4: final polish at best resolution
    if tl() > 20:
        for tau in [300.0, 500.0]:
            if tl() < 10:
                break
            h_t, c5_t = pg_solve(best_h, tau, tl() * 0.4)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = len(h_t)
                print(f"  Polish n={best_n} tau={tau} C5={best_c5:.6f}")

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"Final: n={best_n}, C5={best_c5:.6f}, time={time.time()-t_start:.1f}s")
    return best_h, best_c5, best_n
