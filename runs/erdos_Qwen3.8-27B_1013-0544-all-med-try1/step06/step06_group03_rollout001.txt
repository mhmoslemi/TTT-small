import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_shifts(h, shifts, weights):
    """Compute weighted sum of gradients at specified shifts."""
    n = len(h)
    grad = np.zeros(n)
    for shift, w in zip(shifts, weights):
        if w < 1e-12:
            continue
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n - s]
    return grad

def soft_max_grad_topk(h, t, top_k=50):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = np.clip((conv - m) / t, -40, 40)
    exps = np.exp(shifted)
    
    k = min(top_k, len(conv))
    top_idx = np.argpartition(exps, -k)[-k:]
    exps_top = exps[top_idx]
    total = exps_top.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = exps_top / total
    
    shifts = top_idx - (n - 1)
    grad = grad_at_shifts(h, shifts, weights)
    return grad * dx

def fishta_step(h_init, n, max_iter, t_init, t_final, top_k, L_init):
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = t_init
    beta = 1.0
    L = L_init
    
    for it in range(max_iter):
        progress = min(it / max_iter, 1.0)
        t = t_init * (t_final / t_init) ** progress
        
        grad = soft_max_grad_topk(h, t, top_k)
        # Clip gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 10.0:
            grad = grad * (10.0 / gnorm)
        
        h_new = h_prev - grad / L
        h_new = project_feasible(h_new, n)
        
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        h_prev = h_new + ((beta - 1.0) / beta_new) * (h_new - h)
        h = h_new
        beta = beta_new
        
        # Periodically check actual c5
        if it % 20 == 0 or it == max_iter - 1:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                L = min(L * 1.1, 50.0)
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _grad_at_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

        if budget_s - (time.time() - t0) < 55:
            break
        h_pert = h.copy()
        pt = b % 5
        if pt == 0:
            bl = rng.randint(max(4, n // 12), max(5, n // 3))
            st = rng.randint(0, n - bl)
            h_pert[st:st + bl] = 1.0 - h_pert[st:st + bl]
        elif pt == 1:
            sig = rng.uniform(0.05, 0.2)
            h_pert += sig * rng.randn(n)
        elif pt == 2:
            sh = rng.randint(2, max(3, n // 8))
            h_pert = np.roll(h_pert, sh)
        elif pt == 3:
            bl1 = rng.randint(3, max(4, n // 5))
            bl2 = rng.randint(3, max(4, n // 5))
            s1 = rng.randint(0, n - bl1)
            s2 = rng.randint(0, n - bl2)
            tmp = h_pert[s1:s1+bl1].copy()
            h_pert[s1:s1+bl1] = h_pert[s2:s2+bl2]
            h_pert[s2:s2+bl2] = tmp
        else:
            bl = rng.randint(max(3, n // 20), max(4, n // 5))
            st = rng.randint(0, n - bl)
            h_pert[st:st+bl] = rng.uniform(0.0, 1.0, bl)
        h_pert = project_feasible(h_pert, n)
        h_opt, c5_opt = fishta_opt(h_pert, n, inner_iter, 0.08, 0.002, 50, 20, 5.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(64, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: FISTA from parent
    if budget_s - (time.time() - t0) > 100:
        ht, c5t = fishta_opt(best_h, n0, 30000, 0.1, 0.001, 80, 15, 5.0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Basin hopping with diverse perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        n_basins = min(50, max(15, int(rem / 20)))
        ht, c5t = basin_hopping(best_h, n0, n_basins, 7000, seed + 100, t0, budget_s)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Try other resolutions
    for n_try in [256, 384, 512, 640]:
        rem = budget_s - (time.time() - t0)
        if rem < 140:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        ht, c5t = fishta_opt(hi, n_try, 18000, 0.06, 0.001, 60, 20, 5.0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Greedy improvement on best solution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        ht, c5t = greedy_improve(best_h, len(best_h), 8000, seed + 500)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Final cold FISTA polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        ht, c5t = fishta_opt(best_h, len(best_h), 25000, 0.005, 0.0001, 25, 10, 10.0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Another greedy pass
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        ht, c5t = greedy_improve(best_h, len(best_h), 5000, seed + 900)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
