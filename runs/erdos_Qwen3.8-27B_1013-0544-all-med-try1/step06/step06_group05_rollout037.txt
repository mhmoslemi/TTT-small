import time

def compute_corr_full(h):
    """Compute full correlation of h and 1-h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def compute_c5_from_corr(corr, n):
    dx = 2.0 / n
    return float(np.max(corr)) * dx

def repair_binary(h, n):
    """Ensure h is binary with exactly n//2 ones."""
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        for idx in ones[:n_ones - target]:
            h[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1.0
    return h

def corr_flip_0to1(h, i, corr):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n, seed=0):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:n_ones - target]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1
    return h

def local_search(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = h0.copy()
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 0
        best_move = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            current_c5 = compute_c5(h)
        else:
            break
    
    return h, current_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n, seed=0):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        perm = rng.permutation(len(ones))
        h[ones[perm[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros))
        h[zeros[perm[:target - n_ones]]] = 1
    return h

def local_search(h0, max_time=30.0, seed=0):
    n = len(h0)
    h = h0.copy()
    current_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    start = time.time()
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-15
        best_move = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(new_corr))
                gain = float(np.max(corr)) - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                    best_corr = new_corr.copy()
                h[i] = 1; h[j] = 0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0; h[j] = 1
            corr = best_corr
            current_c5 = compute_c5(h)
        else:
            break
    
    return h, current_c5
