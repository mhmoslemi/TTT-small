def repair(h):
    x = np.array(h, float, copy=True)
    n = x.size
    if n == 0:
        return x
    t = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(50):
        d = t - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1 - x
            m = s > 1e-14
            if not m.any():
                break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-14
            if not m.any():
                break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = t - x.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1 - x)[::-1] if d > 0 else np.argsort(x)[::-1]
        for i in o:
            if abs(d) < 1e-15:
                break

def grad_lag(h, k):

    else:
        l = -lag
        idx = np.arange(l, n)
        g[idx] += (1 - h[idx - l])
        g[idx - l] -= h[idx]
    return g * dx
