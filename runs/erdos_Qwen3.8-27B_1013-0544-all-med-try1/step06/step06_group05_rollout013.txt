Instead, I'll do a limited version: for each pair (i1, i2) of 1s, try the best pair (j1, j2) of 0s. But that's still O(n^4) in the worst case.

A better approach: for each 1 at position i, find the best 0 at position j to swap with (as in single swap). Then try a second swap from the resulting state. This is a "2-opt" style move.

Actually, let me just do single swaps but with a "kick" mechanism: if stuck for X iterations, randomly perturb the solution (swap a random pair) to escape the local minimum.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def thue_morse(n):
    seq = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v > 0:
            c += v & 1
            v >>= 1
        seq[i] = c % 2
    return seq

def smooth_obj(h, T, target_sum, pw):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    val = (cmax + T * np.log(np.sum(np.exp(shifted)))) * dx
    penalty = pw * (np.sum(h) - target_sum) ** 2
    return val + penalty

def continuous_opt(h0, T, target_sum, pw, max_iter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T, target_sum, pw),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h = np.clip(res.x, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

        best_gain = 1e-14

            h[i] = 0.0

                new_c5 = compute_c5(h)
                gain = c5_cur - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                # Kick: random swap to escape
                if len(ones) > 0 and len(zeros) > 0:
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
                    h[i] = 0.0
                    h[j] = 1.0
                    c5_cur = compute_c5(h)
                stall = 0
