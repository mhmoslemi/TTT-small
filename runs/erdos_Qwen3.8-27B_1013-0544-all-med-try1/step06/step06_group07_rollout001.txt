import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

_OPT = {}

def time_left(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            slack = 1.0 - x
            m = slack > 1e-14
            if not np.any(m):
                break
            x[m] += diff * slack[m] / float(slack[m].sum())
        else:
            m = x > 1e-14
            if not np.any(m):
                break
            x[m] += diff * x[m] / float(x[m].sum())
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for i in idx:
            if abs(diff) < 1e-15:
                break
            room = (1.0 - x[i]) if diff > 0 else x[i]
            step = min(abs(diff), room)
            x[i] += step if diff > 0 else -step
            diff -= step if diff > 0 else -step
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def _lag_contrib(x, grad, k, scale):
    n = x.size
    s = int(k) - (n - 1)
    if s >= 0:
        lo, hi = 0, n - 1 - s
        if lo <= hi:
            grad[lo:hi+1] += scale * (1.0 - x[lo+s:hi+1+s])
            grad[s:hi+1+s] -= scale * x[lo:hi+1]
    else:
        d = -s
        lo, hi = 0, n - 1 - d
        if lo <= hi:
            grad[lo+d:hi+1+d] += scale * (1.0 - x[lo:hi+1])
            grad[lo:hi+1] -= scale * x[lo+d:hi+1+d]

def weighted_grad(x, w):
    n = x.size
    nx = x.size
    dx = 2.0 / float(nx)
    grad = np.zeros(nx, dtype=float)
    idxs = np.nonzero(w > 1e-13)[0]
    for k in idxs:
        _lag_contrib(x, grad, int(k), float(w[k]))
    return grad * dx

def smooth_eval(x, tau):
    n = x.size
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = float(np.max(c))
    m = c.size
    if tau <= 0.0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return cm, c, w
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return cm, c, w
    w = e / s
    return cm + tau * float(np.log(s)), c, w

def smooth_obj(x, tau, mu=0.0):
    val, c, w = smooth_eval(x, tau)
    if mu > 0:
        val += mu * (x.sum() - 0.5 * x.size)**2
    return val

def smooth_grad(x, tau, mu=0.0):
    val, c, w = smooth_eval(x, tau)
    g = weighted_grad(x, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - 0.5 * x.size)
    return g

def cons_eq(x):
    return np.array([x.sum() - 0.5 * x.size])

def cons_eq_jac(x):
    return np.ones((1, x.size))

def run_slsqp(x0, tau, maxiter, mu=0.0):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": cons_eq, "jac": cons_eq_jac}]
    def obj(x):
        return smooth_obj(x, tau, mu)
    def jac(x):
        return smooth_grad(x, tau, mu)
    try:
        res = so.minimize(obj, x0, method="SLSQP", jac=jac,
                          bounds=bounds, constraints=constraints,
                          options={"maxiter": int(maxiter), "ftol": 1e-14})
        return res.x if res.success or res.status == 0 else None
    except Exception:
        return None

def run_lbfgs(x0, tau, maxiter, mu=1000.0):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, tau, mu)
    def jac(x):
        return smooth_grad(x, tau, mu)
    try:
        res = so.minimize(obj, x0, method="L-BFGS-B", jac=jac,
                          bounds=bounds,
                          options={"maxiter": int(maxiter), "ftol": 1e-15,
                                   "gtol": 1e-8, "maxfun": 5000})
        return repair(res.x)
    except Exception:
        return None

def active_polish(x0, time_limit, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(int(seed))
    deadline = time.time() + max(0.0, float(time_limit))
    step = 0.15
    no_imp = 0
    while time.time() < deadline:
        cvals = c_values(x)
        C = float(np.max(cvals))
        if C < best_c - 1e-15:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 20:
            improved = False
            for sg in (0.015, 0.008, 0.004, 0.002):
                y = repair(x + rng.normal(0.0, sg, x.size))
                cy = exact_c(y)
                if cy < best_c:
                    best_c = cy
                    best_x = y.copy()
                    x = y
                    no_imp = 0
                    improved = True
                    break
            if not improved:
                break
            continue
        tol = max(1e-9, 1e-6 * C)
        active = np.where(cvals >= C - tol)[0]
        if active.size > 8:
            active = np.argsort(cvals)[::-1][:8]
        g = weighted_grad(x, np.ones_like(active, dtype=float) / float(active.size))
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        accepted = False
        for _ in range(20):
            y = repair(x - step * g)
            cy = exact_c(y)
            if cy < C - 1e-13:
                x = y
                accepted = True
                break
            step *= 0.5
            if step < 1e-11:
                break
        if accepted:
            no_imp = 0
            step = min(step * 1.1, 0.3)
        else:
            no_imp += 1
        if rng.rand() < 0.1:
            y = repair(x + rng.normal(0.0, 0.003, x.size))
            cy = exact_c(y)
            if cy < best_c:
                best_c = cy
                best_x = y.copy()
                x = y
                no_imp = 0
    return best_x, best_c

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    n_new = int(n_new)
    if n0 == n_new:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x1, x0, h0)
    return repair(y)

def fourier_perturb(h0, seed, n_modes=3, amp=0.06):
    rng = np.random.RandomState(int(seed))
    n = h0.size
    t = (np.arange(n) + 0.5) * (2.0 / float(n))
    perturb = np.zeros(n)
    for m in range(1, n_modes + 1):
        f = rng.randint(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        a = amp / float(m + 1)
        perturb += a * np.sin(2 * np.pi * f * t / 2.0 + phase)
    return repair(h0 + perturb)

def optimize_multi_stage(x0, time_limit, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    start = time.time()
    elapsed = lambda: time.time() - start

    tau_stages = [0.04, 0.015, 0.007, 0.003, 0.0015, 0.0007]
    total = len(tau_stages)
    for i, tau in enumerate(tau_stages):
        if elapsed() > time_limit * 0.85:
            break
        frac = 1.0 - i / total
        mi = max(30, int(150 * frac))
        xr = run_slsqp(x, tau, mi)
        if xr is not None:
            xr = repair(xr)
            c = exact_c(xr)
            if c < best_c:
                best_c = c
                best_x = xr.copy()
            x = xr if c <= best_c + 1e-5 else best_x
        if elapsed() > time_limit * 0.85:
            break
        xr2 = run_lbfgs(best_x, tau, 80, mu=500.0)
        if xr2 is not None:
            c2 = exact_c(xr2)
            if c2 < best_c:
                best_c = c2
                best_x = xr2.copy()
                x = xr2

    el = elapsed()
    if time_limit - el > 3.0:
        xp, cp = active_polish(best_x, min(15.0, (time_limit - el) * 0.8), seed)
        if cp < best_c:
            best_c = cp
            best_x = xp
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    if init is not None and init.size >= 10:
        base_n = int(init.size)
        if base_n > 150:
            base_n = 150
        ordered = []
        for n in (base_n, base_n - 5, base_n + 5, base_n + 10, base_n - 10,
                  base_n + 15, base_n - 15, 64, 80, 100):
            n = int(n)
            if 30 <= n <= 160 and n not in ordered:
                ordered.append(n)
        if len(ordered) > 7:
            ordered = ordered[:7]
    else:
        ordered = [70, 80, 64, 90, 100, 60]

    best_x = None
    best_c = 1.0

    for ni, n in enumerate(ordered):
        if time_left(deadline) < 45.0:
            break
        if init is not None and init.size > 0:
            base = resample_to(init, n)
        else:
            base = repair(np.full(n, 0.5))

        c_base = exact_c(base)
        if c_base < best_c:
            best_c = c_base
            best_x = base.copy()

        if ni == 0:
            per_time = min(200.0, time_left(deadline) * 0.45)
            n_starts = 4
        elif ni <= 2:
            per_time = min(100.0, time_left(deadline) * 0.25)
            n_starts = 3
        else:
            per_time = min(50.0, time_left(deadline) * 0.15)
            n_starts = 2

        starts = [base]
        for sk in range(1, n_starts):
            starts.append(fourier_perturb(base, seed_val + sk * 31 + n, n_modes=2 + sk, amp=0.04 + 0.02 * sk))

        for ii, h0 in enumerate(starts):
            if time_left(deadline) < 25.0:
                break
            tm = per_time / float(n_starts - ii + 1)
            x, c = optimize_multi_stage(h0, tm, seed_val + 10000 * n + ii)
            if c < best_c:
                best_c = c
                best_x = x.copy()

    if best_x is None:
        best_x = repair(np.full(50, 0.5))
        best_c = exact_c(best_x)

    # Final polish on best
    if time_left(deadline) > 5.0:
        xp, cp = active_polish(best_x, min(30.0, time_left(deadline) * 0.7), seed_val + 99999)
        if cp < best_c:
            best_c = cp
            best_x = xp

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
