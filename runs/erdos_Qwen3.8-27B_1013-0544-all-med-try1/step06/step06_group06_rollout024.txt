            try:
                h_try = slsqp_solve(h_pert, n, alpha, maxiter=3000)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try.copy(), c
                    h_pert = h_try.copy()
            except Exception:
                pass

    # Phase 3: Systematic coordinate descent with projection
    if remaining() > 30:
        for sweep in range(20):
            if remaining() < 10:
                break
            improved = False
            for i in range(n):
                if remaining() < 5:
                    break
                base = best_h[i]
                for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03]:
                    nv = base + delta
                    if 0.0 <= nv <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = nv
                        h_try = project_feasible(h_try, n)
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            base = nv
            if not improved:
                break

    # Phase 4: 2-point swap optimization (more systematic)
    if remaining() > 20:
        count = 0
        deltas_list = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
        while remaining() > 10 and count < 80000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas_list:
                improved = False
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    break

    # Phase 5: Final high-alpha SLSQP polish
    if remaining() > 10:
        for alpha in [50000, 200000]:
            if remaining() < 5:
                break
            try:
                h_try = slsqp_solve(best_h, n, alpha, maxiter=5000)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
