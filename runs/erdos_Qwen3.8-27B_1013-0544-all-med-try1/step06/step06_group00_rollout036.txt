    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(temp * (corr - max_c), -700, 700)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    w = exp_s / s
    obj = (max_c + np.log(s) / temp) * dx
    grad = np.zeros(n)
    for j in range(n):
        ks = max(0, j)
        ke = min(2 * n - 1, j + n)
        if ks < ke:
            idx = np.arange(ks, ke) - j
            grad[j] = dx * np.dot(w[ks:ke], 1.0 - 2.0 * h[idx])
    return obj, grad

            h = np.clip(h, 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)

    # Phase 3: Structured restarts (iterated local search)
    n_restarts = 0
    while budget_s - (time.time() - t0) > 130 and n_restarts < 60:
        n_restarts += 1
        strat = n_restarts % 4
        if strat == 0:
            scale = 0.06 + 0.10 * np.random.rand()
            h_r = best_h + np.random.randn(n0) * scale
        elif strat == 1:
            h_r = best_h.copy()
            bs = int(np.random.choice([6, 12, 24, 48]))
            start = int(np.random.randint(0, n0 - bs))
            t = np.linspace(0, 1, bs)
            amp = np.random.uniform(-0.12, 0.12)
            h_r[start:start + bs] += amp * np.sin(np.pi * t)
        elif strat == 2:
            h_r = best_h.copy()
            freq = int(np.random.randint(1, n0 // 6 + 1))
            phase = np.random.uniform(0, 2.0 * np.pi)
            t = np.arange(n0, dtype=float) / n0
            amp = 0.05 + 0.07 * np.random.rand()
            h_r += amp * np.sin(2.0 * np.pi * freq * t + phase)
        else:
            h_r = best_h.copy()
            bs = int(np.random.choice([8, 16, 32]))
            s1 = int(np.random.randint(0, max(1, n0 - bs)))
            s2 = int(np.random.randint(0, max(1, n0 - bs)))
            if s1 != s2:
                tmp = h_r[s1:s1 + bs].copy()
                h_r[s1:s1 + bs] = h_r[s2:s2 + bs]
                h_r[s2:s2 + bs] = tmp

        h_r = project_feasible(h_r, ts0)
        h_r = slsqp_solve(h_r, 12.0, maxiter=350)
        c = compute_c5(h_r)
        if c < best_c5 - 1e-10:
            best_c5 = c
            best_h = h_r.copy()
            if budget_s - (time.time() - t0) > 50:
                h_cd, c2 = coord_descent(best_h, n_levels=10, max_sweeps=6)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cd

    # Phase 4: Try other resolutions
    for n_try in [192, 256, 512]:
        if budget_s - (time.time() - t0) < 100:
            break
        x_old = np.arange(len(best_h), dtype=float) / len(best_h)
        x_new = np.arange(n_try, dtype=float) / n_try
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try / 2.0)
        for temp in [60.0, 20.0, 8.0, 3.0]:
            if budget_s - (time.time() - t0) < 40:
                break
            h_new = slsqp_solve(h_new, temp, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
        if budget_s - (time.time() - t0) > 25 and len(best_h) == n_try:
            h_cd, c = coord_descent(best_h, n_levels=10, max_sweeps=8)
            if c < best_c5:
                best_c5 = c
                best_h = h_cd

    # Phase 5: Final fine coordinate descent
    if budget_s - (time.time() - t0) > 15:
        h_cd, c = coord_descent(best_h, n_levels=20, max_sweeps=10)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
