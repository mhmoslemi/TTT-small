def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def topk_subgrad(h, k=5):
    """Gradient using top-k shifts (smoothed max approximation)"""
    n = len(h)

    for idx in top_idx:
        shift = idx - (n - 1)
        grad += _conv_grad_single(h, shift)
    grad /= k
    return c5, grad * dx

**Approach: Bisection on C₅ + Semidefinite Programming**

For a given target C, we want to find h such that:
- 0 ≤ h[i] ≤ 1 for all i
- sum(h[i]) = n/2
- For all shifts k: Σⱼ h[j]*(1-h[j+k]) ≤ C/dx

The constraint Σⱼ h[j]*(1-h[j+k]) ≤ C/dx can be rewritten as:
Σⱼ h[j] - Σⱼ h[j]*h[j+k] ≤ C/dx
n/2 - Σⱼ h[j]*h[j+k] ≤ C/dx  (using the sum constraint)
Σⱼ h[j]*h[j+k] ≥ n/2 - C/dx

So we need the autocorrelation at every lag to be at least n/2 - C/dx.

This is a quadratic constraint! The problem becomes:
minimize 0 (feasibility)
subject to:
  hᵀ R_k h ≥ n/2 - C/dx for all k
  0 ≤ h ≤ 1
  sum(h) = n/2

where R_k is a matrix with (R_k)_{j,l} = 1 if l = j+k, 0 otherwise.

This is a non-convex quadratic constraint (since we're requiring hᵀR_k h ≥ constant, which is non-convex for indefinite R_k).

However, we can relax: if we require h to be binary (0 or 1), then hᵀR_k h = number of positions where both h[j]=1 and h[j+k]=1. This is a combinatorial problem.

For the relaxed continuous version, we could try SOCP or SDP relaxations, but they might be too expensive for n=320.

**Approach: Alternating Minimization**

Fix the "active" set of shifts (those near the max), and optimize h to reduce the overlap at those shifts while not increasing others too much.

**Key insight**: The problem is essentially a signal design problem. We want a signal h of unit energy (well, unit area) whose "cross-correlation" with 1-h is as flat as possible (minimax). This is similar to Zadoff-Chu sequences or CAZAC sequences in communication theory.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)

    return grad * dx

def _lbfgs_grad(h, tv, nc):
    gx = soft_max_grad(h, tv)
    gx += 1000.0 * (np.sum(h) - nc / 2.0)
    return gx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(50, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Adaptive soft-max PGD with momentum on parent
    for t_val in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002]:
        if tl() < 80:
            break
        ht, c5t = pgd_optimize(best_h, max_iter=12000, lr_init=0.003,
                                smooth_t=t_val, use_smooth=True, momentum=0.85,
                                time_limit=budget_s - 150, t0_ref=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Hard subgradient with high momentum
    if tl() > 80:
        ht, c5t = pgd_optimize(best_h, max_iter=25000, lr_init=0.001,
                                use_smooth=False, momentum=0.93,
                                time_limit=budget_s - 130, t0_ref=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: L-BFGS-B refinement
    if tl() > 80:
        nc = len(best_h)
        for tv in [0.008, 0.003]:
            if tl() < 40:
                break
            try:
                res = minimize(lambda h: _lbfgs_obj(h, tv, nc), best_h,
                              jac=lambda h: _lbfgs_grad(h, tv, nc),
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * nc,
                              options={'maxiter': 800, 'ftol': 1e-15})
                ht = project_feasible(res.x, nc)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 4: Multi-resolution exploration
    for n_try in [64, 100, 160, 256, 400, 512, 640]:
        if tl() < 90:
            break
        if n_try == n0:
            continue
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        for t_val in [0.01, 0.004]:
            if tl() < 40:
                break
            ht, c5t = pgd_optimize(hi, max_iter=8000, lr_init=0.002,
                                    smooth_t=t_val, use_smooth=True, momentum=0.85,
                                    time_limit=budget_s - 70, t0_ref=t0)
            if tl() > 25:
                ht2, c5t2 = pgd_optimize(ht, max_iter=8000, lr_init=0.0005,
                                          use_smooth=False, momentum=0.92,
                                          time_limit=budget_s - 50, t0_ref=t0)
                if c5t2 < c5t:
                    ht, c5t = ht2, c5t2
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 5: Structured restarts
    for mode in range(5):
        for _ in range(3):
            if tl() < 40:
                break
            hr = make_init(n0, rng, mode)
            ht, c5t = pgd_optimize(hr, max_iter=7000, lr_init=0.007,
                                    smooth_t=0.03, use_smooth=True, momentum=0.8,
                                    time_limit=budget_s - 60, t0_ref=t0)
            if tl() > 20:
                ht2, c5t2 = pgd_optimize(ht, max_iter=4000, lr_init=0.001,
                                          use_smooth=False, momentum=0.9,
                                          time_limit=budget_s - 40, t0_ref=t0)
                if c5t2 < c5t:
                    c5t = c5t2
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 6: Perturbation restarts from best solution
    for _ in range(25):
        if tl() < 25:
            break
        frac = rng.uniform(0.03, 0.2)
        n_pert = max(1, int(n0 * frac))
        idxs = rng.choice(n0, n_pert, replace=False)
        h_pert = best_h.copy()
        h_pert[idxs] = rng.uniform(0, 1, n_pert)
        h_pert = project_feasible(h_pert, n0)
        ht, c5t = pgd_optimize(h_pert, max_iter=4000, lr_init=0.004,
                                smooth_t=0.015, use_smooth=True, momentum=0.8,
                                time_limit=budget_s - 20, t0_ref=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Final polish: very fine hard subgradient
    if tl() > 25:
        ht, c5t = pgd_optimize(best_h, max_iter=40000, lr_init=0.0003,
                                use_smooth=False, momentum=0.95,
                                time_limit=budget_s - 3, t0_ref=t0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
