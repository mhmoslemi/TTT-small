import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_F(h):
    """Compute F[m] in O(n) using prefix sums."""
    n = len(h)
    A = np.zeros(n + 1)
    A[1:] = np.cumsum(h)
    F = np.zeros(2 * n - 1)
    
    # For m in [0, n-1]: F[m] = (n-m) - A[n-m] - A[n] + A[n-1-m]
    m_range = np.arange(0, n)
    F[m_range] = (n - m_range) - A[n - m_range] - A[n] + A[n - 1 - m_range]
    
    # For m in [n, 2n-2]: F[m] = (2n-1-m) - A[n] + A[m-n+1] - A[2n-1-m]
    m_range2 = np.arange(n, 2 * n - 1)
    F[m_range2] = (2 * n - 1 - m_range2) - A[n] + A[m_range2 - n + 1] - A[2 * n - 1 - m_range2]
    
    return F

def compute_g_j(h, j):
    """Compute g_j[m] for coordinate j."""
    n = len(h)
    g = np.zeros(2 * n - 1)
    
    # First term: I(0 <= j+m-n+1 < n) * (1-h[j+m-n+1])
    # j+m-n+1 in [0,n) => m in [n-1-j, 2n-2-j]
    m_start1 = max(0, n - 1 - j)
    m_end1 = min(2 * n - 2, 2 * n - 2 - j)
    if m_start1 <= m_end1:
        idx = j + np.arange(m_start1, m_end1 + 1) - n + 1
        g[m_start1:m_end1 + 1] += (1.0 - h[idx])
    
    # Second term: -I(0 <= j+n-1-m < n) * h[j+n-1-m]
    # j+n-1-m in [0,n) => m in [j, j+n-1]
    m_start2 = max(0, j)
    m_end2 = min(2 * n - 2, j + n - 1)
    if m_start2 <= m_end2:
        idx = j + n - 1 - np.arange(m_start2, m_end2 + 1)
        g[m_start2:m_end2 + 1] -= h[idx]
    
    return g

def fast_coord_descent(h, max_sweeps=100, n_levels=21, time_limit=None, t0=None, verbose=False):
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    c = compute_corr(h)
    F = compute_F(h)
    best_c5 = np.max(c) * dx
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        improved = False
        for j in range(n):
            orig_val = h[j]
            g_j = compute_g_j(h, j)
            # effective gradient: eff[m] = g_j[m] * n/(n-1) - F[m]/(n-1)
            eff = g_j * n / (n - 1) - F / (n - 1)
            
            # Find best v in [0,1]
            best_max = np.max(c)
            best_v = orig_val
            
            candidates = np.linspace(0.0, 1.0, n_levels)
            for v in candidates:
                d = v - orig_val
                if abs(d) < 1e-15:
                    continue
                new_max = np.max(c + d * eff)
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_v = v
            
            if best_v != orig_val:
                d = best_v - orig_val
                c += d * eff
                # Update h with spread
                h[j] = best_v
                h[np.arange(n) != j] += -d / (n - 1)
                h = np.clip(h, 0.0, 1.0)
                
                # Fix sum if clipping occurred
                sum_err = ts - h.sum()
                if abs(sum_err) > 1e-10:
                    h += sum_err / n
                    h = np.clip(h, 0.0, 1.0)
                    c = compute_corr(h)  # Recompute if correction was needed
                
                F = compute_F(h)
                
                cur_c5 = np.max(c) * dx
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    improved = True
        
        if not improved:
            break
    
    # Final projection
    h = project_feasible(h, ts)
    best_c5 = compute_c5(h)
    return h, best_c5

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_F(h):
    n = len(h)
    A = np.zeros(n + 1)
    A[1:] = np.cumsum(h)
    F = np.zeros(2 * n - 1)
    m1 = np.arange(0, n)
    F[m1] = (n - m1).astype(float) - A[n - m1] - A[n] + A[n - 1 - m1]
    if n > 1:
        m2 = np.arange(n, 2 * n - 1)
        F[m2] = (2 * n - 1 - m2).astype(float) - A[n] + A[m2 - n + 1] - A[2 * n - 1 - m2]
    return F

def compute_g_j(h, j):
    n = len(h)
    g = np.zeros(2 * n - 1)
    # First term: (1-h[j+m-n+1]) for valid m
    m_s = max(0, n - 1 - j)
    m_e = min(2 * n - 2, 2 * n - 2 - j)
    if m_s <= m_e:
        idx = j + np.arange(m_s, m_e + 1) - n + 1
        g[m_s:m_e + 1] += (1.0 - h[idx])
    # Second term: -h[j+n-1-m] for valid m
    m_s2 = max(0, j)
    m_e2 = min(2 * n - 2, j + n - 1)
    if m_s2 <= m_e2:
        idx2 = j + n - 1 - np.arange(m_s2, m_e2 + 1)
        g[m_s2:m_e2 + 1] -= h[idx2]
    return g

def fast_cd(h, max_sweeps=100, n_levels=21, time_limit=None, t0=None):
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    h = h.astype(float).copy()
    c = compute_corr(h)
    F = compute_F(h)
    best_c5 = np.max(c) * dx
    last_improve = 0
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and t0 is not None and (j % 64 == 0):
                if time.time() - t0 > time_limit:
                    break
            orig_val = h[j]
            g_j = compute_g_j(h, j)
            eff = (g_j * n - F) / (n - 1)
            
            best_max = np.max(c)
            best_v = orig_val
            candidates = np.linspace(0.0, 1.0, n_levels)
            for v in candidates:
                d = v - orig_val
                if abs(d) < 1e-15:
                    continue
                new_max = np.max(c + d * eff)
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_v = v
            
            if best_v != orig_val:
                d = best_v - orig_val
                c += d * eff
                h[j] = best_v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                h[mask] += -d / (n - 1)
                h = np.clip(h, 0.0, 1.0)
                sum_err = ts - h.sum()
                if abs(sum_err) > 1e-10:
                    h += sum_err / n
                    h = np.clip(h, 0.0, 1.0)
                    c = compute_corr(h)
                F = compute_F(h)
                cur = np.max(c) * dx
                if cur < best_c5:
                    best_c5 = cur
                    improved = True
                    last_improve = sweep
        if not improved:
            break
    
    h = project_feasible(h, ts)
    best_c5 = compute_c5(h)
    return h, best_c5
