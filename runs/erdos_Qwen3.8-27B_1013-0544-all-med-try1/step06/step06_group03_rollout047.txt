import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def soft_max_obj_val(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    return t * (m / t + np.log(np.sum(np.exp((conv - m) / t)))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-14:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return (dx / (t * t)) * grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    h = h_init.copy()
    y = h.copy()
    mu = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 1.0
    for t in t_seq:

        for it in range(steps_per_temp):
            if time.time() > time_limit:
                return best_h, best_c5
            mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            beta = (1.0 + mu) / (1.0 + mu_new)
            y = h + beta * (h - y)
            y = np.clip(y, 0.0, 1.0)
            grad = soft_max_grad(y, t)
            # Backtracking line search
            step_new = min(step * 1.5, t * t * n * 0.5)
            for _ in range(20):
                h_cand = project_feasible(y - step_new * grad, n)
                f_new = soft_max_obj_val(h_cand, t)
                f_y = soft_max_obj_val(y, t)
                grad_y = soft_max_grad(y, t)
                linear = f_y + np.dot(grad_y, h_cand - y) + (1.0 / (2.0 * step_new)) * np.dot(h_cand - y, h_cand - y)
                if f_new <= linear + 1e-12:
                    break
                step_new *= 0.5
            h = h_cand
            step = step_new
            mu = mu_new
            c5 = compute_c5(h)

                best_h = h.copy()

def subgrad_pgd(h_init, n, max_iter, time_limit, lr_init=0.001):

    best_h = h.copy()
