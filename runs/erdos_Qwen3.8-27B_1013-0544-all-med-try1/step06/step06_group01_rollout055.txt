def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        # Distribute evenly, then fix any clipping issues
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        # Iterative fix
        for _ in range(10):
            d = target - np.sum(h)
            if abs(d) < 1e-12:
                break
            if d > 0:
                # Need to increase some values
                mask = h < 1.0
                h[mask] += d / mask.sum()
            else:
                mask = h > 0.0
                h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(20):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += d / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_descent(h, n, max_passes=200, grid_n=21):
    dx = 2.0 / n
    nc = 2 * n - 1
    lags = np.arange(nc) - (n - 1)
    grid = np.linspace(0.0, 1.0, grid_n) if grid_n > 2 else np.array([0.0, 1.0])
    best_c5 = _c5_max(h)
    
    for _pass in range(max_passes):
        improved = False
        C = _c5_array(h)
        
        for i in range(n):
            old = h[i]
            if old <= 1e-12 or old >= 1.0 - 1e-12:
                # Check if moving away from boundary helps
                src_idx = i + lags
            else:
                src_idx = i + lags
            
            tgt_idx = i - lags
            src_val = np.where((src_idx >= 0) & (src_idx < n), h[np.clip(src_idx, 0, n - 1)], 0.0)
            tgt_val = np.where((tgt_idx >= 0) & (tgt_idx < n), h[np.clip(tgt_idx, 0, n - 1)], 0.0)
            B = (1.0 - src_val - tgt_val) * dx
            C0 = C - old * B
            
            best_t = old
            best_v = np.max(C0 + old * B)
            
            v0 = np.max(C0)
            if v0 < best_v:
                best_v, best_t = v0, 0.0
            v1 = np.max(C0 + B)

                    continue
                v_t = np.max(C0 + t * B)

                    best_v, best_t = v_t, t
            
            if abs(best_t - old) > 1e-12:
                h[i] = best_t
                C = C0 + best_t * B
                if best_v < best_c5 - 1e-14:
                    best_c5 = best_v
                    improved = True
        
        h = _fix_sum(h, n)
        new_c5 = _c5_max(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        
        if not improved:
            break
    
    return h, best_c5

def _swap_perturb(h, n, rng, n_swaps=5):
    """Swap-based perturbation that preserves sum exactly."""
    h_try = h.copy()
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Partial swap: move some mass from higher to lower
        hi, hj = h_try[i], h_try[j]
        delta = (hi - hj) * (0.3 + 0.7 * rng.rand())
        if delta > 0:
            h_try[i] = np.clip(hi - delta, 0, 1)
            h_try[j] = np.clip(hj + delta, 0, 1)
        else:
            delta = -delta
            h_try[j] = np.clip(hj - delta, 0, 1)
            h_try[i] = np.clip(hi + delta, 0, 1)
    h_try = _fix_sum(h_try, n)
    return h_try

def _noise_perturb(h, n, rng, scale=0.5):
    """Gaussian noise perturbation."""
    n_pert = max(3, n // 4)
    idx = rng.choice(n, size=n_pert, replace=False)
    h_try = h.copy()
    h_try[idx] = np.clip(h_try[idx] + rng.randn(n_pert) * scale, 0, 1)
    h_try = _fix_sum(h_try, n)
    return h_try

def _flip_perturb(h, n, rng, n_flips=3):
    """Flip near-binary coordinates."""
    h_try = h.copy()
    for _ in range(n_flips):
        i = rng.randint(n)

        else:
            h_try[i] = 1.0 - h_try[i]
        h_try = _fix_sum(h_try, n)
    return h_try
