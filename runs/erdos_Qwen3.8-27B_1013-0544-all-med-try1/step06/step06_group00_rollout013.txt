        # np.correlate(h, 1-h, "full")[k_idx] where k_idx corresponds to lag

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(temp * (corr - m))
    return (m + np.log(s.sum()) / temp) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, temp, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    ones = np.ones(n)
    res = minimize(smooth_obj, h, args=(temp,), method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: ones}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, ts)

def sa_swap(h, n_iter, t_scale=0.005, rng=None):

    ts = n / 2.0

        rng = np.random

    best_c5 = c5
    best_h = h.copy()

        t = t_scale * (1.0 - it / n_iter)
        if t < 1e-8:
            t = 1e-8
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.uniform(-0.15, 0.15)
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c_new = compute_c5(h)
        delta = c_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / t):
            c5 = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(temp * (corr - m))
    return (m + np.log(s.sum()) / temp) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, temp, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    ones = np.ones(n)
    def con_fun(x):
        return x.sum() - ts
    def con_jac(x):
        return ones
    res = minimize(smooth_obj, h, args=(temp,), method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, ts)

def sa_swap(h, n_iter, t_scale=0.005, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    for it in range(n_iter):
        t = t_scale * (1.0 - float(it) / n_iter) + 1e-9
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(-0.12, 0.12)
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi
        h[j] = hj
        c_new = compute_c5(h)
        delta = c_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / t):
            c5 = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i] = hi - d
            h[j] = hj + d
    return best_h, best_c5

def coord_descent(h, n_levels=20, max_sweeps=10):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_h = orig.copy()
            for v in np.linspace(0, 1, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
            if best_c < c5 - 1e-14:
                h = best_h
                c5 = best_c
                improved = True
            else:
                h = orig
        if not improved:
            break
    return h, c5

def focused_lag_opt(h, n_top=5, maxiter=200):
    """Optimize a weighted sum of the top-k lags."""
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-n_top:]
    m = np.max(corr)
    weights = np.exp(50.0 * (corr[top_idx] - m))
    weights /= weights.sum()
    ones = np.ones(n)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return sum(w * c[k] for w, k in zip(weights, top_idx)) * dx
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lag = k - (n - 1)
            if lag >= 0:
                g[0:n-lag] += w * (1.0 - x[lag:n])
                g[lag:n] -= w * x[0:n-lag] * 0 + w * (-x[0:n-lag])
                # Fix: d/dx[j] of sum_i x[i]*(1-x[i+lag])
                # For x[j] appearing as x[i] (i=j): contribution (1-x[j+lag])
                # For x[j] appearing as x[i+lag] (i=j-lag): contribution -x[j-lag]
                g[:] = 0  # reset, recompute
                for w2, k2 in zip(weights, top_idx):
                    lag2 = k2 - (n - 1)
                    if lag2 >= 0:
                        g[0:n-lag2] += w2 * (1.0 - x[lag2:n])
                        g[lag2:n] -= w2 * x[0:n-lag2]
                    else:
                        g[-lag2:n] -= w2 * x[0:n+lag2]
                        g[0:n+lag2] += w2 * (1.0 - x[-lag2:n])
        return g * dx
    
    # Just use finite-difference gradient-free SLSQP for simplicity
    def con_fun(x):
        return x.sum() - ts
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': lambda x: ones}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)
