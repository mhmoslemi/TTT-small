import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def upsample(h, factor=2):
    return np.repeat(h, factor)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, target_sum):
    n = len(h)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:int(target_sum)]] = 1.0
    return h_bin

def hill_climb(h, deadline, max_sweeps=2000, n_samples=40):
    n = len(h)
    c5_cur = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        n_one = len(one_idx)
        n_zero = len(zero_idx)
        if n_one == 0 or n_zero == 0:
            break
        improved = False
        n_trials = min(n_samples, n_one * n_zero)
        for _ in range(n_trials):
            i = one_idx[np.random.randint(n_one)]
            j = zero_idx[np.random.randint(n_zero)]
            h[i] = 0
            h[j] = 1
            c5_new = evaluate_c5(h)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                n_one = len(one_idx)
                n_zero = len(zero_idx)
            else:
                h[i] = 1
                h[j] = 0
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return best_h, best_c5

def soft_max_obj(x, n, alpha=50):
    target = n / 2.0
    hx = project(x, target)
    corr = np.correlate(hx, 1 - hx, mode='full') * (2.0 / n)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
