import numpy as np
from scipy.optimize import minimize
import time

def _c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_vals(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-10:
            if d2 > 0:
                order = np.argsort(-(1.0 - h))
            else:
                order = np.argsort(h)
            rem = d2
            for i in order:
                if rem > 0:
                    add = min(rem, 1.0 - h[i])
                    h[i] += add
                    rem -= add
                else:
                    sub = min(-rem, h[i])
                    h[i] -= sub
                    rem += sub
                if abs(rem) < 1e-12:
                    break
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_vals(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})
    if res.x is None:
        return h
    return _fix_sum(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_vals(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-10:
            if d2 > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            rem = d2
            for i in order:
                if rem > 0:
                    add = min(rem, 1.0 - h[i])
                    h[i] += add
                    rem -= add
                else:
                    sub = min(-rem, h[i])
                    h[i] -= sub
                    rem += sub
                if abs(rem) < 1e-12:
                    break
    return h

def _subgrad(h, active_mask):
    n = len(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    j_arr = np.arange(nc)
    w = active_mask / np.sum(active_mask)
    sg = np.zeros(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        sg[i] = dx * np.dot(w, ta - tb)

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_vals(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgsb(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})
    if res.x is None:
        return h
    return _fix_sum(res.x, n)

def _progressive_opt(h, n, alphas, beta_scale, maxiter, time_fn, time_limit):
    for a in alphas:
        if time_fn() > time_limit:
            break
        b = max(1.0, beta_scale * a)
        h = _lbfgsb(h, a, b, maxiter)
    return h

    h2[mask_lo] = h2[mask_lo] - strength * h2[mask_lo]
    return _fix_sum(h2, n)
