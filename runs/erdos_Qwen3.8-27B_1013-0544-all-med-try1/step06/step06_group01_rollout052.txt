    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(1.0 - h)
            else:
                idx = np.argsort(h)
            for i in idx:
                if d2 > 0:
                    add = min(d2, 1.0 - h[i])
                    h[i] += add
                    d2 -= add
                else:
                    rem = min(-d2, h[i])
                    h[i] -= rem
                    d2 += rem
                if abs(d2) < 1e-12:

    return h

    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * C - alpha * m)
    f = (m + np.log(np.sum(es))) / alpha + beta * (np.sum(h) - 0.5*n)**2
    return f

def _softmax_grad(h, alpha, beta):

    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * C - alpha * m)
    w = es / np.sum(es)

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argsort(1.0 - h) if d2 > 0 else np.argsort(h)
            for i in idx:
                if d2 > 0:
                    a = min(d2, 1.0 - h[i])
                    h[i] += a
                    d2 -= a
                else:
                    a = min(-d2, h[i])
                    h[i] -= a
                    d2 += a
                if abs(d2) < 1e-12:
                    break
    return h

def _coord_optimize(h, n, sweeps=10, grid_size=60):
    h = h.astype(float).copy()
    dx = 2.0 / n
    C = _c5_array(h)
    nc = 2 * n - 1
    t_grid = np.linspace(0.0, 1.0, grid_size)
    for sweep in range(sweeps):
        improved = False
        for j in range(n):
            h_j = h[j]
            if h_j < 1e-12 or h_j > 1.0 - 1e-12:
                continue

            idx1 = j + m_vals - n + 1
            idx2 = j - m_vals + n - 1

            v2 = (idx2 >= 0) & (idx2 < n)
            h_i1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
            h_i2 = np.where(v2, h[np.clip(idx2, 0, n-1)], 0.0)
            coef = (h_i1 - h_i2) * dx
            delta = (t_grid - h_j)[:, None] * coef[None, :]
            C_all = C[None, :] + delta
            C_all[:, n-1] = C[n-1] + dx * (t_grid*(1.0-t_grid) - h_j*(1.0-h_j))

                h[j] = best_t
                improved = True
        h = _fix(h, n)
        C = _c5_array(h)
        if not improved:

    m = np.max(C)
    f = (m + np.log(np.sum(np.exp(alpha*C - alpha*m)))) / alpha + beta*(np.sum(h)-0.5*n)**2
    return f

def _sm_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha*C - alpha*m)
    w = es / np.sum(es)
    nc = 2*n-1
    grad = np.zeros(n)
    m_vals = np.arange(nc)
    for i in range(n):
        idx1 = i + m_vals - n + 1
        idx2 = i - m_vals + n - 1
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        h_i1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)

        dC = (h_i1 - h_i2) * dx
        grad[i] = np.dot(w, dC)
    grad += 2.0*beta*(np.sum(h)-0.5*n)
    return grad

def _sm_optimize(h, n, alpha, beta, maxiter=300):
    res = minimize(_sm_obj, h, args=(alpha, beta), jac=_sm_grad,
                  method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                  options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _binary_snap(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    target = int(0.5 * n)
    s = int(np.sum(h_bin))
    while s > target:
        ones = np.where(h_bin == 1.0)[0]
        idx = ones[np.argmin(h[ones])]
        h_bin[idx] = 0.0
        s -= 1
    while s < target:
        zeros = np.where(h_bin == 0.0)[0]
        idx = zeros[np.argmax(h[zeros])]
        h_bin[idx] = 1.0
        s += 1
    remaining = 0.5*n - target
    if abs(remaining) > 1e-12:
        h_bin = h_bin.copy()

            zeros = np.where(h_bin == 0.0)[0]
            idx = zeros[np.argmax(h[zeros])]
            h_bin[idx] = remaining
        else:
            ones = np.where(h_bin == 1.0)[0]
            idx = ones[np.argmin(h[ones])]
            h_bin[idx] = h_bin[idx] + remaining
    return h_bin
