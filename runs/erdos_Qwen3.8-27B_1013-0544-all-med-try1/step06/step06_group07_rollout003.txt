            slack = 1.0 - x
            mask = slack > 1e-15
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:
            mask = x > 1e-15
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)[::-1]
            for idx in order:
                if diff <= 1e-15:
                    break
                r = 1.0 - x[idx]
                if r > 0:
                    a = min(diff, r)
                    x[idx] += a
                    diff -= a
        else:
            order = np.argsort(x)[::-1]
            for idx in order:
                if diff >= -1e-15:
                    break
                r = x[idx]
                if r > 0:
                    a = max(diff, -r)
                    x[idx] += a
                    diff -= a
    return x

def _cvals(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def _exact_c(h):
    return float(np.max(_cvals(h)))

def _grad_lag(x, k):
    n = x.size
    s = int(k) - (n - 1)
    g = np.zeros(n, dtype=float)
    if s >= 0:
        g[s:n] += (1.0 - x[0:n - s])
        g[0:n - s] -= x[s:n]
    else:
        t = -s
        g[0:n - t] += (1.0 - x[t:n])
        g[t:n] -= x[0:n - t]
    return g * (2.0 / float(n))

def _smooth_max_and_grad(x, tau, topk=None):
    c = _cvals(x)
    cm = float(np.max(c))
    m = c.size
    if tau <= 0:
        idx = int(np.argmax(c))
        w = np.zeros(m)
        w[idx] = 1.0
    else:
        z = (c - cm) / tau
        z = z - np.max(z)
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(m)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        if topk is not None and 0 < topk < m:
            idx = np.argsort(w)[::-1][:topk]
            w2 = np.zeros(m)
            w2[idx] = w[idx]
            ss = float(w2.sum())
            if ss > 0:
                w = w2 / ss
    val = cm + tau * float(np.log(float(np.sum(np.exp((c - cm) / tau - np.max((c - cm) / tau))))))
    grad = np.zeros(x.size, dtype=float)
    for k in np.nonzero(w > 1e-14)[0]:
        grad += _grad_lag(x, int(k)) * float(w[k])
    return val, grad, c

def _projected_descent(x0, tau, iters, momentum=0.85, time_limit=30.0, topk=None):
    x = np.array(x0, dtype=float, copy=True)
    v = np.zeros_like(x)
    best_x = x.copy()
    best_c = _exact_c(x)
    step = 0.5
    t0 = time.time()
    for it in range(iters):
        if time.time() - t0 > time_limit:
            break
        y = x + momentum * v
        y = np.clip(y, 0.0, 1.0)
        _, g, _ = _smooth_max_and_grad(y, tau, topk)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        f_cur = _exact_c(x)
        accepted = False
        s = step
        for _ in range(12):
            xn = _repair(x - s * g)
            fn = _exact_c(xn)
            if fn < f_cur - 1e-14:
                v = xn - x
                x = xn
                step = min(3.0, s * 1.4)
                accepted = True
                break
            s *= 0.4
            if s < 1e-12:
                break
        if not accepted:
            step = s * 0.5
            if step < 1e-12:
                break
        c_now = _exact_c(x)
        if c_now < best_c - 1e-15:
            best_c = c_now
            best_x = x.copy()
    return best_x, best_c

def _active_set_descent(x0, iters, time_limit=20.0, tol=1e-7):
    x = np.array(x0, dtype=float, copy=True)
    best_x = x.copy()
    best_c = _exact_c(x)
    step = 0.3
    t0 = time.time()
    stale = 0
    for it in range(iters):
        if time.time() - t0 > time_limit:
            break
        c = _cvals(x)
        C = float(np.max(c))
        active = np.where(c >= C - max(tol, 1e-6 * C))[0]
        if active.size > 8:
            active = np.argsort(c)[::-1][:8]
        g = np.zeros(x.size, dtype=float)
        for k in active:
            g += _grad_lag(x, int(k))
        g /= float(active.size)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        f_cur = float(np.max(c))
        accepted = False
        s = step
        for _ in range(10):
            xn = _repair(x - s * g)
            fn = _exact_c(xn)
            if fn < f_cur - 1e-14:
                x = xn
                accepted = True
                break
            s *= 0.4
            if s < 1e-12:
                break
        if accepted:
            step = min(2.0, s * 1.3)
            stale = 0
        else:
            step = s * 0.5
            stale += 1
            if stale > 20:
                break
        if _exact_c(x) < best_c - 1e-15:
            best_c = _exact_c(x)
            best_x = x.copy()
    return best_x, best_c

def _swap_polish(x0, n_trials, time_limit=20.0):
    x = np.array(x0, dtype=float, copy=True)
    best_x = x.copy()
    best_c = _exact_c(x)
    n = x.size
    rng = np.random.RandomState(12345)
    t0 = time.time()
    c = _cvals(x)
    C = float(np.max(c))
    for _ in range(n_trials):
        if time.time() - t0 > time_limit:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        di = 1.0 - x[i]
        dj = 1.0 - x[j]
        for a in (0.02, 0.05, 0.1, 0.2, 0.5):
            x[i] += a * di
            x[j] -= a * di
            x[i] = min(1.0, max(0.0, x[i]))
            x[j] = min(1.0, max(0.0, x[j]))
            xn = _repair(x)
            cn = _exact_c(xn)
            if cn < best_c - 1e-14:
                best_c = cn
                best_x = xn.copy()
                x = xn.copy()
                break
        x = best_x.copy()
    return best_x, best_c

def _resample(h, n_new):
    n0 = h.size
    if n0 == n_new:
        return _repair(h)
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return _repair(np.interp(x1, x0, h))

def _perturb(h, scale, seed):
    rng = np.random.RandomState(int(seed))
    return _repair(h + rng.normal(0.0, float(scale), h.size))

def _freq_perturb(h, seed, n_freqs=3, amp=0.08):
    rng = np.random.RandomState(int(seed))
    n = h.size
    t = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = h.copy()
    for f in range(1, n_freqs + 1):
        freq = int(rng.randint(1, max(2, n // 8)))
        y += float(amp) / float(f) * np.sin(2.0 * np.pi * freq * t / 2.0 + rng.uniform(0, 2 * np.pi))
    return _repair(y)

def _run_worker(x0, time_limit, seed, n_tau_stages=None):
    x = _repair(x0)
    best_x = x.copy()
    best_c = _exact_c(x)
    t0 = time.time()
    if n_tau_stages is None:
        n_tau_stages = [0.05, 0.02, 0.008, 0.003, 0.001]
    budget_per = time_limit / (len(n_tau_stages) + 2)
    for tau in n_tau_stages:
        if time.time() - t0 > time_limit * 0.85:
            break
        tl = max(1.0, min(budget_per, time_limit - (time.time() - t0)))
        iters = max(20, int(120 * (n_tau_stages[0] / tau) ** 0.3))
        xr, cr = _projected_descent(x, tau, iters, momentum=0.9, time_limit=tl, topk=15)
        if cr < best_c:
            best_c = cr
            best_x = xr.copy()
        x = xr
    if time.time() - t0 < time_limit * 0.9:
        tl = max(1.0, time_limit - (time.time() - t0))
        xr, cr = _active_set_descent(best_x, 200, time_limit=tl, tol=1e-8)
        if cr < best_c:
            best_c = cr
            best_x = xr.copy()
    if time.time() - t0 < time_limit * 0.95:
        tl = max(0.5, time_limit - (time.time() - t0))
        xr, cr = _swap_polish(best_x, 2000, time_limit=tl)
        if cr < best_c:
            best_c = cr
            best_x = xr.copy()
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        parent = _repair(np.asarray(raw, dtype=float))
    else:
        parent = _repair(np.full(70, 0.5))

    n = parent.size
    best_x = parent.copy()
    best_c = _exact_c(parent)

    # Phase 1: Direct polish of parent (most time)
    t1 = min(250.0, max(60.0, (deadline - time.time()) * 0.4))
    x1, c1 = _run_worker(parent, t1, seed_val)
    if c1 < best_c:
        best_c = c1
        best_x = x1.copy()

    # Phase 2: Structured perturbations and optimize
    if time.time() < deadline - 80:
        perturbations = []
        for s in (0.01, 0.025, 0.05):
            perturbations.append(_perturb(best_x, s, seed_val + 100 * s + 1))
        for fq in range(1, 4):
            perturbations.append(_freq_perturb(best_x, seed_val + 200 + fq, n_freqs=fq, amp=0.05 + 0.02 * fq))
        # Antipodal-type perturbation
        t = (np.arange(n) + 0.5) * (2.0 / float(n))
        perturbations.append(_repair(1.0 - best_x + 0.0 * t))

        per = min(45.0, max(10.0, (deadline - time.time()) / max(1, len(perturbations))))
        for i, p in enumerate(perturbations):
            if time.time() > deadline - 60:
                break
            tl = min(per, max(5.0, deadline - time.time() - 60))
            xr, cr = _run_worker(p, tl, seed_val + 500 + i)
            if cr < best_c:
                best_c = cr
                best_x = xr.copy()

    # Phase 3: Resolution jump to n*2
    if time.time() < deadline - 100:
        n2 = n * 2
        x2 = _resample(best_x, n2)
        c2 = _exact_c(x2)
        tl = min(120.0, max(20.0, deadline - time.time() - 80))
        x2o, c2o = _run_worker(x2, tl, seed_val + 900, n_tau_stages=[0.02, 0.008, 0.003, 0.001])
        if c2o < best_c:
            # Resample back
            x_back = _resample(x2o, n)
            c_back = _exact_c(x_back)
            if c_back < best_c:
                best_c = c_back
                best_x = x_back
            else:
                # Keep the high-res version
                best_c = c2o
                best_x = x2o
        # Also try a perturbed high-res
        if time.time() < deadline - 60:
            x2p = _perturb(x2, 0.03, seed_val + 950)
            tl2 = min(60.0, max(10.0, deadline - time.time() - 50))
            x2o2, c2o2 = _run_worker(x2p, tl2, seed_val + 960, n_tau_stages=[0.02, 0.005, 0.002])
            if c2o2 < best_c:
                best_c = c2o2
                best_x = x2o2

    # Phase 4: Final intensive polish on best
    if time.time() < deadline - 10:
        nfin = best_x.size
        tl_final = max(5.0, deadline - time.time() - 2)
        # Active set + swaps at final resolution
        xr, cr = _active_set_descent(best_x, 300, time_limit=tl_final * 0.5, tol=1e-9)
        if cr < best_c:
            best_c = cr
            best_x = xr.copy()
        xr, cr = _swap_polish(best_x, 3000, time_limit=tl_final * 0.3)
        if cr < best_c:
            best_c = cr
            best_x = xr.copy()
        # One more gradient pass
        if time.time() < deadline - 5:
            xr, cr = _projected_descent(best_x, 0.001, 100, momentum=0.95,
                                         time_limit=max(1.0, deadline - time.time() - 2), topk=10)
            if cr < best_c:
                best_c = cr
                best_x = xr.copy()

    best_x = _repair(best_x)
    best_c = _exact_c(best_x)
    return best_x, best_c, int(best_x.size)
