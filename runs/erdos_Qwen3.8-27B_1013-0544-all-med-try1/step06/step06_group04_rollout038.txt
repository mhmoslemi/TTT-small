import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_max(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def binary_local_search(h, rng, max_rounds=200):
    n = len(h)
    h = h.copy()
    best = corr_max(h)
    target = n // 2
    
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Shuffle pairs to avoid bias
        perm = rng.permutation(len(ones) * len(zeros))
        for idx in perm:
            i = ones[idx // len(zeros)]
            j = zeros[idx % len(zeros)]
            h[i] = 0; h[j] = 1
            v = corr_max(h)
            if v < best:
                best = v
                improved = True
                break
            else:
                h[i] = 1; h[j] = 0
        if not improved:
            break
    return h, best

def continuous_local_search(h, rng, max_iters=10000):
    n = len(h)
    h = h.copy()
    best = corr_max(h)
    target = n / 2.0
    
    for _ in range(max_iters):
        # Pick two random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Random perturbation that preserves sum
        delta = rng.uniform(0.01, 0.3) * rng.choice([-1, 1])
        new_hi = h[i] + delta
        new_hj = h[j] - delta
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
            continue
        h[i] = new_hi
        h[j] = new_hj
        new_c = corr_max(h)
        if new_c <= best:
            best = new_c
        else:
            h[i] -= delta
            h[j] += delta
    return h, best

def simulated_annealing(h, n, rng, start_temp=0.5, end_temp=0.01, max_iters=100000):
    """SA for continuous solutions."""
    h = h.copy()
    best_h = h.copy()
    best_c = corr_max(h)
    curr_c = best_c
    target = n / 2.0
    
    for it in range(max_iters):
        temp = start_temp * (end_temp / start_temp) ** (it / max_iters)
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(0.05, 0.5) * rng.choice([-1, 1])
        new_hi = h[i] + delta
        new_hj = h[j] - delta
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
            continue
        h[i] = new_hi
        h[j] = new_hj
        new_c = corr_max(h)
        d = new_c - curr_c
        if d < 0 or rng.random() < np.exp(-d / temp):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_max_val(h):
    return np.max(corr_vec(h))

def fix_sum(h, n):
    """Adjust h to have sum exactly n/2 while staying in [0,1]."""
    h = h.copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if np.sum(mask) == 0:
            return h
        adj = diff / np.sum(mask)
        h[mask] = np.minimum(1.0, h[mask] + adj)
    else:
        mask = h > 0.0
        if np.sum(mask) == 0:
            return h
        adj = diff / np.sum(mask)
        h[mask] = np.maximum(0.0, h[mask] + adj)
    # Might need multiple passes
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) == 0: break
            h[mask] = np.minimum(1.0, h[mask] + diff / np.sum(mask))
        else:
            mask = h > 0.0
            if np.sum(mask) == 0: break
            h[mask] = np.maximum(0.0, h[mask] + diff / np.sum(mask))
    return h

def to_binary(h, n, rng):
    """Convert to binary with exactly n/2 ones."""
    h = np.where(h > 0.5, 1.0, 0.0)
    count = np.sum(h)
    target = n // 2
    if count > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:int(count) - target]] = 0
    elif count < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - int(count)]] = 1
    return h

def binary_ls(h, rng, max_rounds=200):
    """First-improvement local search for binary."""
    n = len(h)
    h = h.copy()
    best = corr_max_val(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_pairs = len(ones) * len(zeros)
        order = rng.permutation(n_pairs)
        for k in order:
            i = ones[k // len(zeros)]
            j = zeros[k % len(zeros)]
            h[i], h[j] = 0, 1
            v = corr_max_val(h)
            if v < best:
                best = v
                improved = True
                break
            h[i], h[j] = 1, 0
        if not improved:
            break
    return h, best

def cont_ls(h, rng, max_iters=100000, step_range=(0.01, 0.4)):
    """Local search for continuous solutions."""
    n = len(h)
    h = h.copy()
    best = corr_max_val(h)
    for _ in range(max_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(*step_range) * rng.choice([-1.0, 1.0])
        hi2, hj2 = h[i] + delta, h[j] - delta
        if hi2 < 0 or hi2 > 1 or hj2 < 0 or hj2 > 1:
            continue
        h[i], h[j] = hi2, hj2
        v = corr_max_val(h)
        if v <= best:
            best = v
        else:
            h[i], h[j] = hi2 - delta, hj2 + delta
    return h, best

def sa_cont(h, n, rng, t0_temp=0.5, t1_temp=0.005, max_iters=200000):
    """Simulated annealing for continuous."""
    h = h.copy()
    best_h = h.copy()
    best_c = corr_max_val(h)
    curr_c = best_c
    for it in range(max_iters):
        frac = it / max(max_iters - 1, 1)
        temp = t0_temp * (t1_temp / t0_temp) ** frac
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:

        delta = rng.uniform(0.02, 0.6) * rng.choice([-1.0, 1.0])

        if hi2 < 0 or hi2 > 1 or hj2 < 0 or hj2 > 1:

        h[i], h[j] = hi2, hj2
        v = corr_max_val(h)
        d = v - curr_c

            if v < best_c:
                best_c = v
                best_h = h.copy()
        else:
            h[i], h[j] = hi2 - delta, hj2 + delta

import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def cmv(h):
    """Correlation max value (without dx factor)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            if m.sum() == 0: break
            h[m] = np.minimum(1.0, h[m] + d / m.sum())
        else:
            m = h > 0.0
            if m.sum() == 0: break
            h[m] = np.maximum(0.0, h[m] + d / m.sum())
    return h

def to_bin(h, n, rng):
    h = (h > 0.5).astype(float)
    c = int(h.sum())
    t = n // 2
    if c > t:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:c-t]] = 0
    elif c < t:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:t-c]] = 1
    return h

def bin_ls(h, rng, max_rounds=300):
    n = len(h)
    h = h.copy()
    best = cmv(h)
    for _ in range(max_rounds):
        imp = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        npairs = len(ones) * len(zeros)
        for k in rng.permutation(npairs):
            i = ones[k // len(zeros)]
            j = zeros[k % len(zeros)]
            h[i] = 0; h[j] = 1
            v = cmv(h)
            if v < best:
                best = v; imp = True; break
            h[i] = 1; h[j] = 0
        if not imp:
            break
    return h, best

def cont_ls(h, rng, iters=200000):
    n = len(h)
    h = h.copy()
    best = cmv(h)
    for _ in range(iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        d = rng.uniform(0.01, 0.5) * rng.choice([-1.0, 1.0])
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1: continue
        h[i], h[j] = a, b
        v = cmv(h)
        if v <= best:
            best = v
        else:
            h[i], h[j] = a - d, b + d
    return h, best

    bh = h.copy()

    cc = bc

        T = t_hi * (t_lo / t_hi) ** f
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j: continue
        d = rng.uniform(0.02, 0.7) * rng.choice([-1.0, 1.0])
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1: continue
        h[i], h[j] = a, b
        v = cmv(h)
        if v - cc < 0 or rng.random() < np.exp(-(v - cc) / max(T, 1e-15)):
            cc = v
            if v < bc:
                bc = v; bh = h.copy()
        else:
            h[i], h[j] = a - d, b + d
    return bh, bc

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def cmv(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
            if m.sum() == 0:
                break
            h[m] = np.minimum(1.0, h[m] + d / m.sum())
        else:
            m = h > 0.0
            if m.sum() == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / m.sum())
    return h

def to_bin(h, n, rng):
    h = (h > 0.5).astype(float)
    c = int(h.sum())
    t = n // 2
    if c > t:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:c - t]] = 0
    elif c < t:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:t - c]] = 1
    return h

def bin_ls_first(h, rng, max_rounds=300):
    n = len(h)
    h = h.copy()
    best = cmv(h)
    for _ in range(max_rounds):
        imp = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nz = len(zeros)
        npairs = len(ones) * nz
        for k in rng.permutation(npairs):
            i = ones[k // nz]
            j = zeros[k % nz]
            h[i] = 0
            h[j] = 1
            v = cmv(h)
            if v < best:
                best = v
                imp = True
                break
            h[i] = 1
            h[j] = 0
        if not imp:
            break
    return h, best

def bin_ls_best(h, rng, max_rounds=500):
    n = len(h)
    h = h.copy()
    best = cmv(h)
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bv, bi, bj = best, -1, -1
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                v = cmv(h)
                if v < bv:
                    bv, bi, bj = v, i, j
                h[j] = 0
            h[i] = 1
        if bi >= 0:
            h[bi] = 0
            h[bj] = 1
            best = bv
        else:
            break
    return h, best

def cont_ls(h, rng, iters=200000):
    n = len(h)
    h = h.copy()
    best = cmv(h)
    for _ in range(iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(0.01, 0.5) * rng.choice([-1.0, 1.0])
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1:
            continue
        h[i], h[j] = a, b
        v = cmv(h)
        if v <= best:
            best = v
        else:
            h[i], h[j] = a - d, b + d
    return h, best

def sa_cont(h, n, rng, t_hi=0.5, t_lo=0.003, iters=300000):
    h = h.copy()
    bh = h.copy()
    bc = cmv(h)
    cc = bc
    for it in range(iters):
        f = it / max(iters - 1, 1)
        T = t_hi * (t_lo / t_hi) ** f
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(0.02, 0.7) * rng.choice([-1.0, 1.0])
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1:
            continue
        h[i], h[j] = a, b
        v = cmv(h)
        delta = v - cc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cc = v
            if v < bc:
                bc = v
                bh = h.copy()
        else:
            h[i], h[j] = a - d, b + d
    return bh, bc
