The fundamental issue is that we're trying to minimize the maximum of many (2N-1) piecewise linear functions, subject to a simplex-like constraint. This is a convex optimization problem!

Wait, is it convex? Let's check:
- Each f_k(h) = sum_i h[i+k]*(1-h[i]) is... not linear or convex in general because of the h[i+k]*h[i] product terms.

Actually, f_k(h) = sum_i h[i+k] - sum_i h[i+k]*h[i]. The second term is bilinear (product of two variables), so f_k is not convex or concave in general. The max of non-convex functions is also non-convex.

So it's a non-convex problem. That's why we need heuristics.

However, if we think of it differently: for fixed h, f_k is linear in each individual h[i] (since h[i+k] is treated as a constant when differentiating w.r.t. h[i]). So the problem is DC (difference of convex) in some sense.

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

            g[0:N-m] += (1.0 - h[m:N])
            g[m:N] -= h[0:N-m]
    return g

def soft_max_grad(h, corr, dx, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = int(np.argmax(corr)) - (N - 1)
        return grad_for_lag(h, k_star) * dx
    w /= wsum
    # Only compute for significant lags
    thresh = 1e-6
    significant = np.where(w > thresh)[0]
    g = np.zeros(N)
    for j in significant:
        k = int(j) - (N - 1)
        g += w[j] * grad_for_lag(h, k)
    return g * dx

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_for_lag(h, k):
    N = len(h)
    g = np.zeros(N)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        m = N - k
        if m > 0:
            g[k:N] += (1.0 - h[0:m])
            g[0:m] -= h[k:N]
    else:
        m = -k
        if m > 0:
            g[0:N-m] += (1.0 - h[m:N])
            g[m:N] -= h[0:N-m]
    return g

def soft_max_grad(h, corr, dx, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = int(np.argmax(corr)) - (N - 1)
        return grad_for_lag(h, k_star) * dx
    w /= wsum
    thresh = 1e-8
    sig = np.where(w > thresh)[0]
    g = np.zeros(N)
    for j in sig:
        k = int(j) - (N - 1)
        g += w[j] * grad_for_lag(h, k)
    return g * dx
