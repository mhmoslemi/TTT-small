import cvxpy as cp

def linearized_step(h, n_constraints=None):
    n = len(h)
    corr = compute_corr(h)
    # Use top constraints
    if n_constraints is None:
        n_constraints = 20
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:n_constraints]
    
    # Compute gradients for active constraints
    grads = []
    for idx in active:
        m = idx - (n - 1)
        g = np.zeros(n)
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1-m:hi1-m])
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2+m:hi2+m]
        grads.append(g)
    grads = np.array(grads)  # shape (n_constraints, n)
    
    # LP: minimize t
    # s.t. corr[idx] + grads[i] · dh <= t for all i
    #      0 <= h + dh <= 1
    #      sum(dh) = 0
    dh = cp.Variable(n)
    t = cp.Parameter(nonneg=True)  # actually, let's make it a variable
    t = cp.Variable()
    
    constraints = [
        corr[active] + grads @ dh <= t,
        dh >= -h,  # h + dh >= 0
        dh <= 1 - h,  # h + dh <= 1
        cp.sum(dh) == 0
    ]
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or another solver
    
    if dh.value is not None:
        h_new = h + dh.value
        h_new = project_feasible(h_new, n/2.0)
        return h_new
    return h
