import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.default_rng(0)
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5/n_old
    x_new = np.linspace(0, 1, n_new, endpoint=False) + 0.5/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n, seed=0):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    h = np.clip(h, 0, 1)
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for idx in ones[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5/n_old
    x_new = np.linspace(0, 1, n_new, endpoint=False) + 0.5/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)

        best_gain = 1e-15
        best_move = None

                h[j] = 1.0
                new_vals = np.correlate(h, 1-h, 'full')
                new_max = float(np.max(new_vals))
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j, None, None)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_move is None:

            if use_double:

                        i1, i2 = ones[a_idx], ones[b_idx]
