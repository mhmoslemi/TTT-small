    val += 50.0 * s * s
    return float(val)

def compute_correlate(h, n):
    return np.correlate(h, 1 - h, mode='full')

def pair_c5_eval(h, n, i, j, d):
    h2 = h.copy()
    h2[i] += d
    h2[j] -= d
    c = np.correlate(h2, 1 - h2, mode='full')
    return float(np.max(c) * (2.0 / n))

def do_pair_sweep(h, n, best_c5, rng, n_pairs=3000, maxiter_ms=40, time_limit=None, t0=None):
    c = compute_correlate(h, n)
    c_dx = c * (2.0 / n)
    for _ in range(n_pairs):
        if time_limit is not None:
            if time.time() - t0 > budget_s - time_limit:
                break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        d_max = min(hi, 1.0 - hj) - 1e-12
        d_min = max(-hi, hj - 1.0) + 1e-12
        if d_max <= d_min:
            continue
        def f(d):
            return pair_c5_eval(h, n, i, j, d)
        try:
            res = minimize_scalar(f, bounds=(d_min, d_max), method='bounded',
                                 options={'maxiter': maxiter_ms})
            if res.success and res.fun < best_c5 - 1e-14:
                h[i] += res.x
                h[j] -= res.x
                best_c5 = res.fun
                c = compute_correlate(h, n)
        except Exception:
            pass
    return h, best_c5

def do_structured_sweep(h, n, best_c5, t0, time_limit=80):
    """Cycle through structured pairs for thorough local optimization."""
    order = np.random.default_rng(0).permutation(n * (n - 1) // 2)
    idx = 0
    for round_num in range(50):
        if time.time() - t0 > budget_s - time_limit:
            break
        improved = False
        steps_per_round = min(500, n * (n - 1) // 2)
        for _ in range(steps_per_round):
            if time.time() - t0 > budget_s - time_limit:
                break
            i = idx % n
            j = (idx // n) % n
            if i == j:
                idx += 1
                continue
            idx += 1
            hi = h[i]
            hj = h[j]
            d_max = min(hi, 1.0 - hj) - 1e-12
            d_min = max(-hi, hj - 1.0) + 1e-12
            if d_max <= d_min:
                continue
            def f(d):
                return pair_c5_eval(h, n, i, j, d)
            try:
                res = minimize_scalar(f, bounds=(d_min, d_max), method='bounded',
                                     options={'maxiter': 30})
                if res.success and res.fun < best_c5 - 1e-15:
                    h[i] += res.x
                    h[j] -= res.x
                    best_c5 = res.fun
                    improved = True
            except Exception:
                pass
        if not improved and round_num > 2:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with log-sum-exp continuation
    for alpha in [20, 50, 100, 300, 1000, 3000, 10000, 30000]:
        if budget_s - (time.time() - t0) < 40:
            break
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 15000, 'ftol': 1e-22, 'gtol': 1e-12})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass
        # Quick pair sweep after each L-BFGS-B
        best_h, best_c5 = do_pair_sweep(best_h, n, best_c5, rng, n_pairs=500, maxiter_ms=30,
                                        time_limit=budget_s - (time.time() - t0) - 50, t0=t0)

    # Phase 2: Structured pair sweep for thorough local optimization
    best_h, best_c5 = do_structured_sweep(best_h, n, best_c5, t0, time_limit=120)

    # Phase 3: Multi-start perturbation with re-optimization
    for attempt in range(30):
        if budget_s - (time.time() - t0) < 100:
            break
        scale = rng.choice([0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        alpha = rng.choice([500, 1000, 3000, 10000, 30000])
        try:
            res = minimize(lse_obj, h_pert, args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-20, 'gtol': 1e-12})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h = h_try
                best_c5 = c
            # Pair descent polish
            best_h, best_c5 = do_pair_sweep(best_h, n, best_c5, rng, n_pairs=1500, maxiter_ms=40,
                                            time_limit=budget_s - (time.time() - t0) - 80, t0=t0)
        except Exception:
            pass

    # Phase 4: Final intensive pair sweep
    if budget_s - (time.time() - t0) > 20:
        rem = budget_s - (time.time() - t0) - 10
        if rem > 5:
            best_h, best_c5 = do_structured_sweep(best_h, n, best_c5, t0, time_limit=rem)

    # Phase 5: Very fine final polish
    if budget_s - (time.time() - t0) > 10:
        for _ in range(5000):
            if budget_s - (time.time() - t0) < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi = best_h[i]
            hj = best_h[j]
            d_max = min(hi, 1.0 - hj) - 1e-12
            d_min = max(-hi, hj - 1.0) + 1e-12
            if d_max <= d_min:
                continue
            def f_final(d):
                return pair_c5_eval(best_h, n, i, j, d)
            try:
                res = minimize_scalar(f_final, bounds=(d_min, d_max), method='bounded',
                                     options={'maxiter': 100})
                if res.success and res.fun < best_c5 - 1e-16:
                    best_h[i] += res.x
                    best_h[j] -= res.x
                    best_c5 = res.fun
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
