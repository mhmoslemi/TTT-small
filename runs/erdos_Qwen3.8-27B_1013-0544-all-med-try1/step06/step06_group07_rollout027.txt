        # Second factor: -h[j] * h[j+k], contributes to grad at position j+k
        s2_lo = max(0, lo + k)
        s2_hi = min(n, hi + k)
        if s2_lo < s2_hi:
            grad[s2_lo:s2_hi] -= wv * h[s2_lo - k:s2_hi - k]
    return grad * dx

def softmax_weights(c, tau):
    cm = c.max()
    z = (c - cm) / tau
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / s

def project_feasible(h, target):
    return _project_sum(h, target)

def proj_grad_descent(h0, tau, step, n_iter, deadline):

def run_slsqp_phase(h0, tau, maxiter):
    n = h0.size
    target = 0.5 * n
    
    def fun(x):
        c = c_values(x)
        cm = c.max()
        w = softmax_weights(c, tau)
        return cm + tau * np.log(np.exp((c - cm) / tau).sum())
    
    def jac(x):
        c = c_values(x)
        cm = c.max()
        w = softmax_weights(c, tau)
        return compute_grad(x, w)
    
    def con_fun(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(x.size)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = so.minimize(fun, h0, method='SLSQP', jac=jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if res is not None and res.success:
        return repair(res.x), eval_c(repair(res.x))
    elif res is not None:
        xr = repair(res.x)
        return xr, eval_c(xr)
    return h0, eval_c(h0)

def resample(h, n_new):

    if n == n_new:
        return repair(h)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h))

def make_starts(h0, n, seed, n_starts):

        if len(starts) >= n_starts:

        starts.append(repair(h))

Actually, for coordinate descent:
c_k = sum_j h[j] * (1-h[j+k]) * dx

The contribution of h[i] to c_k is:
- From first factor: h[i] * (1-h[i+k]) * dx (when i and i+k are valid)
- From second factor: -h[i-k] * h[i] * dx (when i-k and i are valid)

So c_k is linear in h[i] (for fixed other coordinates). The max over k is the max of linear functions, which is convex in h[i]. So we can find the minimum of this convex piecewise linear function in 1D.

The breakpoints occur where the maximum switches from one k to another. We can find them by:
1. Evaluating the piecewise linear function at h[i] = 0 and h[i] = 1
2. Finding the slopes at these points
3. The minimum is either at 0, at 1, or at a breakpoint

This is O(n^2) per coordinate update (evaluate all c_k at the new h[i]), and O(n) coordinates, so O(n^3) per full sweep. For n=140, that's about 2.7M operations per sweep, which is fast.

            # c_k(t) = c_k(0) + t * (slope_k) for each k

                if 0 <= i + k < n:

    n = x.size
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)

def _proj(x, target):
    n = x.size
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    return _proj(x, target)

def _proj(x, target):
    n = x.size
    lo, hi = -1.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def eval_c(h):
    return float(np.max(c_values(h)))

def compute_grad(h, w):
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wv = w[idx]
        if wv < 1e-15:
            continue
        k = idx - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += wv * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wv * h[lo2 - k:hi2 - k]
    return grad * dx

def softmax_w(c, tau):
    cm = c.max()
    z = (c - cm) / tau
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    w = e / s
    idx = np.argsort(w)[::-1][:20]
    w2 = np.zeros_like(c)
    w2[idx] = w[idx]
    return w2 / w2.sum()

def coord_descent(h, deadline, max_sweeps=2000):
    h = repair(h)
    n = h.size
    dx = 2.0 / n
    c = c_values(h)
    best_h = h.copy()
    best_c = float(c.max())
    ks = np.arange(-(n - 1), n)
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            valid1 = (i + ks >= 0) & (i + ks < n)
            valid2 = (i - ks >= 0) & (i - ks < n)
            slopes = np.zeros(2 * n - 1)
            slopes[valid1] = (1.0 - h[i + ks[valid1]]) * dx
            slopes[valid2] -= h[i - ks[valid2]] * dx
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(c + (m1 - old) * slopes)
                f2 = np.max(c + (m2 - old) * slopes)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1
            t_new = 0.5 * (lo + hi)
            delta = t_new - old
            if abs(delta) > 1e-13:
                h[i] = t_new
                c += delta * slopes
                cm = float(c.max())
                if cm < best_c:
                    best_c = cm
                    best_h = h.copy()
                improved = True
        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0
    return best_h, best_c

def proj_grad(h0, tau, deadline):
    h = repair(h0)
    n = h.size
    target = 0.5 * n
    best_h = h.copy()
    best_c = eval_c(h)
    step = 0.5
    for it in range(500):
        if time.time() >= deadline:
            break
        c = c_values(h)
        cm = float(c.max())
        w = softmax_w(c, tau)
        grad = compute_grad(h, w)
        alpha = step
        accepted = False
        for _ in range(20):
            h_new = _proj(h - alpha * grad, target)
            cn = eval_c(h_new)
            if cn < cm - 1e-13:
                h = h_new
                accepted = True
                step = alpha * 1.5
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        if accepted:
            cn = eval_c(h)
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            if it > 50:
                break
    return best_h, best_c

def resample(h, n_new):
    n = h.size
    if n == n_new:
        return repair(h)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h))

def final_polish(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = repair(h0)
    n = h.size
    best_h = h.copy()
    best_c = eval_c(h)
    deadline = time.time() + max(0.5, time_limit)
    tau_vals = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for tau in tau_vals:
        if time.time() >= deadline:
            break
        rem = deadline - time.time()
        if rem < 2.0:
            break
        h, c = proj_grad(h, tau, deadline - 0.1)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Coordinate descent polish
    if time.time() < deadline:
        h2, c2 = coord_descent(best_h, deadline, max_sweeps=500)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    best_c = 1e9
