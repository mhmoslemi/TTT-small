import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_gradient(h, T=50.0):
    """Compute gradient of smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for numerical stability
    corr_max = np.max(corr)
    weights = np.exp(T * (corr - corr_max))
    Z = weights.sum()
    # Gradient is weighted sum of individual gradients
    # Each corr_k has gradient: (1-h) shifted by lag - h shifted by -lag
    # For efficiency, compute contribution from top lags only
    top_k = np.argsort(weights)[-min(100, len(weights)):]
    grad = np.zeros(n)
    for k in top_k:
        lag = k - (n - 1)
        w = weights[k] / Z
        if w < 1e-10:
            continue
        if lag >= 0:
            # (1-h[j+lag]) for j in [0, n-1-lag]
            grad[:n-lag] += w * (1.0 - h[lag:n])
            # -h[j-lag] for j in [lag, n-1]
            if lag > 0:
                grad[lag:] -= w * h[:n-lag]
        else:
            lagp = -lag
            # (1-h[j-lagp]) for j in [lagp, n-1]
            grad[lagp:] += w * (1.0 - h[:n-lagp])
            # -h[j+lagp] for j in [0, n-1-lagp]
            grad[:n-lagp] -= w * h[lagp:n]
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient(h, T, lr, n_steps=100):
    """Perform projected gradient descent on smooth C5."""
    n = len(h)
    target = n / 2.0
    for step in range(n_steps):
        grad = c5_gradient(h, T)
        h_new = h - lr * grad
        h = project_feasible(h_new, target)
    return h

        grad[:n-lagp] -= h[lagp:n]

    log_w = T * (corr - cmax)
    # Clip for numerical stability
    log_w = np.clip(log_w, -500, 0)
    w = np.exp(log_w)
    Z = w.sum()
    if Z < 1e-300:
        return c5_subgrad(h)
    w = w / Z
    # Only process significant weights
    thresh = 1e-6
    active = np.where(w > thresh)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= wk * h[:n-lag]
        else:
            lagp = -lag
            grad[lagp:] += wk * (1.0 - h[:n-lagp])
            grad[:n-lagp] -= wk * h[lagp:n]
    return grad * dx

    """Projected gradient descent."""
    n = len(h)
    target = n / 2.0
    for _ in range(n_iter):
        grad = c5_smooth_grad(h, T)
        h = project_feasible(h - lr * grad, target)

                    continue

                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < best_c - tol:
                    best_c = c
                    best_v = v
                    h[j] = v  # keep the change
                else:
                    h[j] = orig
            h = project_feasible(h, target)
            if best_c < c5 - tol:

                improved = True
        if not improved:
            break

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        step = d / mask.sum()
        # Limited step to avoid overshooting
        if step > 0:
            max_step = (1.0 - h[mask]).min() if mask.sum() > 0 else 0
            step = min(step, max_step * 1.001)
        else:
            min_step = h[mask].min() if mask.sum() > 0 else 0
            step = max(step, min_step * 1.001)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    log_w = T * (corr - cmax)
    log_w = np.clip(log_w, -400, 0)
    w = np.exp(log_w)
    Z = w.sum()
    if Z < 1e-300:
        # Fallback to subgradient
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += (1.0 - h[:n-lp])
            grad[:n-lp] -= h[lp:n]
        return grad * dx
    w /= Z
    active = np.where(w > 1e-5)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= wk * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += wk * (1.0 - h[:n-lp])
            grad[:n-lp] -= wk * h[lp:n]
    return grad * dx

    target = n / 2.0
    cur = h.copy()
    for T in T_schedule:
        lr = base_lr
        for i in range(n_iters_per_T):
            grad = smooth_grad(cur, T)
            cur = project_feasible(cur - lr * grad, target)
            lr *= 0.998
    return cur

    target = n / 2.0
    c5 = compute_c5(h)
    levels = np.linspace(0.0, 1.0, n_levels)

                c5 = best_c
                improved = True

        c = 0

            j >>= 1

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-10:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            h = np.clip(target_sum / len(h) * np.ones_like(h), 0, 1)
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    log_w = T * (corr - cmax)
    log_w = np.clip(log_w, -400, 0)
    w = np.exp(log_w)
    Z = w.sum()
    if Z < 1e-300:
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += (1.0 - h[:n-lp])
            grad[:n-lp] -= h[lp:n]
        return grad * dx
    w /= Z
    active = np.where(w > 1e-5)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:n])
            if lag > 0:
                grad[lag:] -= wk * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += wk * (1.0 - h[:n-lp])
            grad[:n-lp] -= wk * h[lp:n]
    return grad * dx

def pgd_optimize(h, n, T_schedule, base_lr, n_iters_per_T):

            grad = smooth_grad(cur, T)
            cur = project_feasible(cur - lr * grad, target)
            lr *= 0.998
    return cur

    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    levels = np.linspace(0.0, 1.0, n_levels)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig_h = h.copy()
            orig_v = orig_h[j]
            best_c = c5
            best_h = orig_h
            for v in levels:
                if abs(v - orig_v) < 1e-6:
                    continue
                h_try = orig_h.copy()
                h_try[j] = v
                h_try = project_feasible(h_try, target)
                c = compute_c5(h_try)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h_try
            if best_c < c5 - 1e-12:
                h = best_h
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def make_thue_morse(n):
    seq = np.zeros(n)
    for i in range(n):
        j = i
        c = 0
        while j:
            c += j & 1
            j >>= 1
        seq[i] = 1.0 if c % 2 == 1 else 0.0
    return seq
