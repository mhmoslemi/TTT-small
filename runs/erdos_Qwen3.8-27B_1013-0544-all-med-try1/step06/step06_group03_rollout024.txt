        if s > target:

        else:

    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    target = n / 2.0

        t = t_init  # We'll handle schedule externally
        conv = np.correlate(h, 1.0-h, mode="full")
        m = np.max(conv)
        exps = np.exp((conv - m) / t)
        val = t * (m/t + np.log(np.sum(exps))) * (2.0/n)
        val += penalty * (np.sum(h) - target)**2
        return val

        g = soft_max_grad_fft(h, t)
        g += 2.0 * penalty * (np.sum(h) - target)

            mm = np.max(conv)

            val = tv * (mm/tv + np.log(np.sum(exps))) * (2.0/n)
            val += penalty * (np.sum(h) - target)**2

            g += 2.0 * penalty * (np.sum(h) - target)
            return g

                          bounds=bounds, options={'maxiter': max_iter//4, 'ftol': 1e-14, 'gtol': 1e-10})

            if c5t < best_c5:

        else:

import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def soft_max_grad_fft(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
    W = exps / s
    P = 1
    while P < 3 * n:
        P *= 2
    H = fft(h, P)
    Wr = fft(W, P)
    full_conv = np.real(ifft(H * Wr))
    term2 = full_conv[n-1:2*n-1]
    W_rev = W[::-1]
    Wrev = fft(W_rev, P)
    full_corr = np.real(ifft(H * Wrev))
    term3 = full_corr[n-1:2*n-1]
    prefix = np.zeros(2 * n)
    prefix[1:] = np.cumsum(W)
    term1 = prefix[n:2*n] - prefix[0:n]
    grad = (term1 - term2 - term3) * dx
    return grad

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def lbfgs_optimize(h_init, n, t_seq, max_iter_per_t=2000, penalty=300.0):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for t in t_seq:
        def obj_local(hv, tv=t):
            conv = np.correlate(hv, 1.0 - hv, mode="full")
            mm = np.max(conv)
            exps = np.exp((conv - mm) / tv)
            val = tv * (mm / tv + np.log(np.sum(exps))) * (2.0 / n)
            val += penalty * (np.sum(hv) - target) ** 2
            return val
        def grad_local(hv, tv=t):
            g = soft_max_grad_fft(hv, tv)
            g += 2.0 * penalty * (np.sum(hv) - target)
            return g
        try:
            res = minimize(obj_local, h, jac=grad_local, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iter_per_t, 'ftol': 1e-14, 'gtol': 1e-10})
            h = project_feasible(res.x, n)
            c5t = compute_c5(h)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5

def pgd_optimize(h_init, max_iter=50000, lr_init=0.005, t=0.005):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        grad = soft_max_grad_fft(h, t)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-16:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.05, 0.03)
    return best_h, best_c5

def resample(h_old, n_new):
    nc = len(h_old)
    if nc == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2.0, nc, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def structured_random(rng, n, kind=0):
    if kind == 0:
        return rng.uniform(0, 1, n)
    elif kind == 1:
        period = rng.randint(3, max(4, n // 4))
        phase = rng.uniform(0, period)
        x = np.arange(n)
        return 0.5 + 0.4 * np.sin(2 * np.pi * (x + phase) / period)
    elif kind == 2:
        return np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    else:
        k = rng.randint(2, max(3, n // 5))
        return np.tile(rng.uniform(0.2, 0.8, k), n // k + 1)[:n]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS with cooling on initial n
    if tl() > 80:
        t_seq = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        ht, c5t = lbfgs_optimize(best_h, n0, t_seq, max_iter_per_t=1500, penalty=300.0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: PGD refinement
    if tl() > 60:
        ht, c5t = pgd_optimize(best_h, max_iter=60000, lr_init=0.003, t=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        ht, c5t = pgd_optimize(best_h, max_iter=40000, lr_init=0.001, t=0.001)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Try different n (coarse to fine)
    n_list = sorted(set([45, 60, 90, 120, 150, 200, 240, 320, 400, 512, 640, 800, 1024]))
    n_list = [x for x in n_list if x != n0]
    for n_try in n_list:
        if tl() < 100:
            break
        hi = resample(best_h, n_try)
        hi = project_feasible(hi, n_try)
        t_seq = [0.05, 0.02, 0.008, 0.003, 0.001]
        ht, c5t = lbfgs_optimize(hi, n_try, t_seq, max_iter_per_t=1200, penalty=300.0)
        ht, c5t2 = pgd_optimize(ht, max_iter=40000, lr_init=0.002, t=0.002)
        if c5t2 < c5t:
            c5t = c5t2
        ht, c5t3 = pgd_optimize(ht, max_iter=30000, lr_init=0.0008, t=0.001)
        if c5t3 < c5t:
            c5t = c5t3
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Structured restarts at best n
    nb = len(best_h)
    for i in range(80):
        if tl() < 40:
            break
        kind = i % 4
        hr = structured_random(rng, nb, kind)
        hr = project_feasible(hr, nb)
        ht, c5t = lbfgs_optimize(hr, nb, [0.05, 0.01, 0.003], max_iter_per_t=800, penalty=200.0)
        ht, c5t2 = pgd_optimize(ht, max_iter=15000, lr_init=0.005, t=0.01)
        if c5t2 < c5t:
            c5t = c5t2
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Final high-precision polish
    if tl() > 80:
        t_seq = [0.003, 0.001, 0.0005]
        ht, c5t = lbfgs_optimize(best_h, len(best_h), t_seq, max_iter_per_t=3000, penalty=500.0)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    if tl() > 40:
        ht, c5t = pgd_optimize(best_h, max_iter=100000, lr_init=0.0005, t=0.0005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
