def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

    # But for small n, direct is fine

        it += 1
        current_time = time.time()

        zeros = np.where(h == 0)[0]
        
        # Generate candidate moves

        n_candidates = min(len(ones) * len(zeros), 2000)

        best_move = None

        if n_candidates <= 10000:
            # Try all
            for i in ones:
                for j in zeros:

                    h[i] = 1

        else:

                h[j] = 0

                    best_move = (i, j, new_c5)

            h[i] = 0

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return make_binary(h_new, n_new)

            diffs = np.diff(ones)

Actually, the gradient computation for the correlation is non-trivial. Let me use a simpler approach: coordinate descent with the soft-max objective.

For coordinate descent:
- Fix all coordinates except h[i]
- The objective becomes a function of h[i] alone (since each corr_k depends on at most 2 values of h involving index i)
- Wait, that's not quite right. corr_k = sum_m h[m]*(1-h[m+k]). If we change h[i], then:
  - For lag k: the term h[i]*(1-h[i+k]) changes (if i+k < n)
  - For lag -k: the term h[i+k]*(1-h[i]) changes (if i+k < n)
  - For lag k: the term h[i-k]*(1-h[i]) changes (if i-k >= 0)
  - For lag -k: the term h[i]*(1-h[i-k]) changes (if i-k >= 0)

So changing h[i] affects O(n) correlation values. We can compute the new objective in O(n) time for each coordinate.

                if new_val < 0 or new_val > 1:

                j = (i + 1) % n
                new_j = h[j] - delta
                if new_j < 0 or new_j > 1:
                    continue
                h[i] += delta
                h[j] -= delta

                if obj < current_obj:
                    current_obj = obj
                    improved = True
                    if obj < best_obj:
                        best_obj = obj
                        best_h = h.copy()

                    h[i] -= delta
                    h[j] += delta
        if not improved:
            # Try with smaller steps or break
            break

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h).astype(int)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[target:]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1
    return h

def interp_binary(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old.astype(float))
    return make_binary(h_new, n_new)

def sa_binary(h_init, max_time, seed, T0=0.008, T1=0.00005):

    h = h_init.copy()

    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    best_iter = 0
    stall = 0

    while (time.time() - start) < max_time:
        elapsed = (time.time() - start) / max_time
        T = T0 * (T1 / T0) ** elapsed

        if r < 0.55:

            zeros = np.where(h == 0)[0]

            j = zeros[rng.randint(len(zeros))]
            h[i] = 0; h[j] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5; best_h = h.copy(); best_iter = int(time.time() - start); stall = 0

            else:

        elif r < 0.80:
            ones = np.where(h == 1)[0]

            i1, i2 = rng.choice(len(ones), 2, replace=False)

        else:
