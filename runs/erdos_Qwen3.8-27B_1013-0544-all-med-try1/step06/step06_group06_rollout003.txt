import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_jac(h, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def deb_obj(x, n):
    return compute_c5(project_feasible(x, n), n)

def time_left(t0, budget_s):
    return budget_s - (time.time() - t0)

def continue_optimize(h, n, t0, budget_s, alphas, maxiter=3000):
    for alpha in alphas:
        if time_left(t0, budget_s) < 20:
            break
        h_new = slsqp_opt(h, n, alpha, maxiter)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old - 1e-15:
            h = h_new
    return h

def make_perturbation(rng, n, base, scale):
    noise = rng.normal(0, 1, size=n)
    r = rng.random()
    if r < 0.4:
        freq = rng.randint(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        noise += 0.8 * np.sin(2 * np.pi * freq * t / n + phase)
    elif r < 0.6:
        freq = rng.randint(1, max(2, n // 6))
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        noise += 1.2 * np.cos(2 * np.pi * freq * t / n + phase)
    h_pert = base + scale * noise
    return project_feasible(h_pert, n)
