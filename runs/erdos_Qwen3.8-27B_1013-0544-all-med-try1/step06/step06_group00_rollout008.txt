from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = temp * (corr - m)
    s = np.exp(shifted).sum()
    return (m + np.log(s) / temp) * dx

    p = weights / weights.sum()  # shape (2n-1,)
    
    lag = np.arange(2*n - 1) - (n - 1)

        pk = p[k]
        if pk < 1e-12:
            continue
        lg = lag[k]
        lo = max(0, lg)
        hi = min(n, lg + n)
        if lo < hi:
            g[lo:hi] += pk * omt[lo - lg:hi - lg]
        lo2 = max(0, -lg)
        hi2 = min(n, n - lg)
        if lo2 < hi2:
            g[lo2:hi2] -= pk * h[lo2 + lg:hi2 + lg]
    return g * dx

def proj_grad_descent(h, temp, n_iter=2000, alpha_init=0.02):
    n = len(h)
    ts = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    stall = 0
    for i in range(n_iter):
        g = smooth_grad(h, temp)
        h_new = project_feasible(h - alpha * g, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                alpha *= 0.5
                stall = 0
        h = h_new
    return best_h, best_c5

def coord_descent(h, n_levels=20, max_sweeps=30):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    vals = np.linspace(0.0, 1.0, n_levels)
    for sw in range(max_sweeps):
        improved = False
        for j in range(n):
            orig_val = h[j]
            best_c = c5
            best_v = orig_val
            for v in vals:
                if abs(v - orig_val) < 1e-12:
                    continue
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != orig_val:
                h[j] = best_v
                h = project_feasible(h, ts)
                c5 = best_c
                improved = True
            else:
                h[j] = orig_val
                h = project_feasible(h, ts)
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP with decreasing temperature
    temps = [300.0, 150.0, 80.0, 40.0, 20.0, 10.0, 5.0, 3.0, 2.0]
    for temp in temps:
        if remaining() < 50:
            break
        cur = slsqp_opt(best_h, temp, maxiter=600)
        c = compute_c5(cur)
        if c < best_c5:
            best_c5 = c
            best_h = cur.copy()

    # Phase 2: Projected gradient with smooth max
    if remaining() > 40:
        for temp in [20.0, 10.0, 5.0]:
            if remaining() < 30:
                break
            h_pg, c_pg = proj_grad_descent(best_h, temp, n_iter=2000, alpha_init=0.015)
            if c_pg < best_c5:
                best_c5 = c_pg
                best_h = h_pg.copy()

    # Phase 3: SLSQP again after gradient steps
    for temp in [10.0, 5.0, 3.0]:
        if remaining() < 50:
            break
        cur = slsqp_opt(best_h, temp, maxiter=500)
        c = compute_c5(cur)
        if c < best_c5:
            best_c5 = c
            best_h = cur.copy()

    # Phase 4: Heavy coordinate descent
    if remaining() > 20:
        h_cd, c = coord_descent(best_h, n_levels=25, max_sweeps=40)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd

    # Phase 5: Random restarts with structured perturbations
    n_restarts = 0
    while remaining() > 80 and n_restarts < 30:
        n_restarts += 1
        rtype = n_restarts % 3
        if rtype == 0:
            scale = 0.1 + 0.15 * np.random.random()
            h_r = best_h + np.random.randn(n0) * scale
        elif rtype == 1:
            shift = np.random.randint(1, n0 // 4)
            h_r = np.roll(best_h, shift)
            h_r += np.random.randn(n0) * 0.05
        else:
            freq = np.random.randint(2, 10)
            phase = np.random.random() * 2 * np.pi
            hue = 0.1 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 + phase)
            h_r = best_h + hue
        h_r = project_feasible(h_r, n0 / 2.0)
        h_r = slsqp_opt(h_r, 15.0, maxiter=300)
        c = compute_c5(h_r)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_h = h_r.copy()
            if remaining() > 50:
                h_cd, c2 = coord_descent(best_h, n_levels=20, max_sweeps=25)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_cd

    # Phase 6: Final fine coordinate descent
    if remaining() > 10:
        h_cd, c = coord_descent(best_h, n_levels=30, max_sweeps=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd

    # Phase 7: One more SLSQP polish at very low temp
    if remaining() > 30:
        cur = slsqp_opt(best_h, 5.0, maxiter=400)
        c = compute_c5(cur)
        if c < best_c5:
            best_c5 = c
            best_h = cur.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
