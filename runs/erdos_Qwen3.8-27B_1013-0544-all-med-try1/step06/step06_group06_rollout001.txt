Hmm, that's interesting. For a strictly binary periodic-2 pattern, c[k] = 0 for all even k. What about odd k?

c[1] = dx * sum h[i]*(1-h[i+1])

    # Phase 5: Final active set polish
    if time.time() < t0 + budget_s - 15:
        dl = time.time() + min(40, budget_s * 0.04)
        h5, c5 = active_set_optimize(best_h, n, dl, max_active=10)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Fine 2-point swaps
    if time.time() < t0 + budget_s - 5:
        deltas = [0.003, 0.007, 0.012, 0.02, 0.035]
        steps = 0
        while time.time() < t0 + budget_s - 3 and steps < 30000:
            steps += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                improved = False
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    break

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
