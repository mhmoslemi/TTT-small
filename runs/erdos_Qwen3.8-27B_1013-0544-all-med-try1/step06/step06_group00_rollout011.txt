import numpy as np
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    C = corr_raw * dx
    max_C = np.max(C)
    exp_vals = np.exp(beta * (C - max_C))
    Z = exp_vals.sum()
    weights = exp_vals / Z
    obj = max_C + np.log(Z) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-16:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
        else:
            p = -m
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return obj, grad

        if time.time() >= deadline:
            break

            if time.time() >= deadline:

            _, grad = softmax_obj_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:

            else:

            c5_new = compute_c5_value(h_new)
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

def coord_descent_polish(h, target_sum, deadline):
    n = len(h)
    c5 = compute_c5_value(h)
    vals = np.linspace(0, 1, 13)
    for sweep in range(100):
