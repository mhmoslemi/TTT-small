import scipy.optimize as opt

def run_packing():

    # and some small clusters near the corners and edges

    n = 26
    
    # Initial positions (hexagonal packing plus corner/edge clusters)
    positions = []
    radii = []
    
    # Hexagonal packing in the center
    for row in range(5):
        for col in range(6):
            x = (col + 0.5 * row) / 6
            y = (row + 0.5 * col) / 6
            positions.append([x, y])
    
    # Corner clusters
    for corner in [(0, 0), (0, 1), (1, 0), (1, 1)]:
        for i in range(3):
            x = corner[0] + (i / 4)
            y = corner[1] + (i / 4)
            positions.append([x, y])
    
    # Edge clusters
    for i in range(5):
        positions.append([i / 5, 0.1])
        positions.append([i / 5, 0.9])
    for i in range(5):
        positions.append([0.1, i / 5])
        positions.append([0.9, i / 5])
    
    # Initial radii: start with small uniform values
    initial_radii = np.ones(n) * 0.05
    
    # Objective function: maximize the sum of radii
    def objective(x):
        # x is a vector of radii
        return -np.sum(x)  # Negative because we minimize
    
    # Constraint functions
    def constraint_non_overlap(i, j, x):
        def cons(x):
            x_i = x[i]
            x_j = x[j]
            dx = positions[i][0] - positions[j][0]
            dy = positions[i][1] - positions[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (x_i + x_j)
        return cons
    
    def constraint_boundary(i, x):
        def cons(x):
            x_i = x[i]
            x_center = positions[i]
            return np.array([
                x_center[0] - x_i - 1e-6,  # Left boundary
                x_center[1] - x_i - 1e-6,  # Bottom boundary
                x_i - (1 - x_center[0]) - 1e-6,  # Right boundary
                x_i - (1 - x_center[1]) - 1e-6   # Top boundary
            ])
        return cons
    
    # Define constraints
    constraints = []
    
    # Non-overlap constraints
    for i in range(n):

    )
