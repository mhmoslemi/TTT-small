import numpy as np
import scipy.optimize as optimize

def run_packing():

    # We start by placing them in a hexagonal grid
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Use a uniform radius to start, based on the square's packing density
    # Estimate total area: 26 * π * r^2 <= 1
    # So r^2 <= 1 / (26 * π) => r = sqrt(1 / (26 * π)) ≈ 0.116
    r0 = np.sqrt(1 / (26 * np.pi))
    radii[:] = r0

    # Define bounds for centers [0, 1]
    bounds = [(0, 1) for _ in range(2 * n)]

    def objective_function(x):
        # x is a flattened array of centers and radii
        # Reshape into centers and radii
        x_centers = x[:2 * n].reshape((n, 2))
        x_radii = x[2 * n:]

        # Objective: Minimize the negative of total sum of radii (maximize sum)
        total_sum = np.sum(x_radii)
        return -total_sum

    def constraint_overlap(x):
        x_centers = x[:2 * n].reshape((n, 2))
        x_radii = x[2 * n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < x_radii[i] + x_radii[j] - 1e-6:
                    return dist - (x_radii[i] + x_radii[j] - 1e-6)
        return 0

    def constraint_boundary(x):
        x_centers = x[:2 * n].reshape((n, 2))

            if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                return 1e6  # Large penalty for boundary violation
        return 0

    # Initial guess: centers (hexagonal grid), radii (uniform)
    x0 = np.concatenate([np.random.rand(n, 2), np.full(n, r0)])

    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    result = optimize.minimize(
        objective_function,
        x0,
        method='SLSQP',

    radii = result.x[2 * n:]

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
