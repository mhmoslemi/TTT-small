import scipy.optimize as opt

        return x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1

        return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) >= r1 + r2

    def generate_hexagonal_grid(num_circles):
        # Coordinates for hexagonal packing of num_circles
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i
                if x >= 1 or y >= 1 or x < 0 or y < 0:
                    continue
                grid.append((x, y))
        return np.array(grid)

    def hexagonal_packing_initial_guess(num_circles):
        # Generate initial positions for hexagonal packing
        positions = generate_hexagonal_grid(num_circles)
        # Scale to fit in [0, 1]^2
        positions /= np.max(positions)
        return positions

    def optimize_packing(positions, num_circles):

        radii = np.ones(num_circles) * 0.05  # Small radius to start

        bounds = [(0, 1) for _ in range(2 * num_circles)]

                    x + r <= 1,
                    y - r >= 0,
                    y + r <= 1

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
