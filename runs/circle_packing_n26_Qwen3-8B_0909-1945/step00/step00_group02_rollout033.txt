import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Helper function to calculate the maximum possible radius for a circle
    def max_radius(x, y):
        # Find the distance to the closest boundary
        min_dist = min(x, 1 - x, y, 1 - y)
        return min_dist

    # Helper function to calculate the distance between two centers
    def distance(c1, c2):
        return np.linalg.norm(c1 - c2)

    # Helper function to compute the objective function
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative sum to maximize the sum

    # Helper function to compute the constraints
    def constraints_func(radii, centers):
        n = len(radii)
        constraints = []
        # Constraint: radii must be non-negative
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        # Constraint: each circle must fit in the square
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            min_dist = min(x, 1 - x, y, 1 - y)
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, x=x, y=y, min_dist=min_dist: min_dist - r})
        # Constraint: circles must not overlap
        for i in range(n):
            for j in range(i + 1, n):
                dist = distance(centers[i], centers[j])
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, dist=dist: dist - (r[i] + r[j])})
        return constraints

    # Generate initial centers using a hexagonal grid
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    # Normalize the centers to fit within the unit square
    max_x = np.max(centers[:, 0])

    centers /= np.max([max_x, max_y])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
