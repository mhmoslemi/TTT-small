import scipy.optimize as opt

    radii = np.zeros(n_circles)

    # Use a hexagonal grid to distribute circles
    num_rows = int(np.ceil(np.sqrt(n_circles)))
    num_cols = int(np.ceil(n_circles / num_rows))

    spacing = 1.0 / (num_cols - 1)

                break
            centers[idx] = [x, y]

        return -np.sum(radii)  # Maximize sum of radii

    def constraint_non_overlap(radii, centers):
