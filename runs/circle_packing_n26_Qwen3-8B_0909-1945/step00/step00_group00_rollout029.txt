import scipy.optimize as optimize

                x = col + 0.5 * row
                y = row * (np.sqrt(3) / 2)
                centers.append([x, y])
        return np.array(centers)

    # Function to compute the radius of the largest circle that can fit in a given space
    def max_radius(centers, radii, indices, square_size=1.0):
        if not indices:
            return 0.0

        max_x = max(centers[i][0] for i in indices)
        min_y = min(centers[i][1] for i in indices)
        max_y = max(centers[i][1] for i in indices)

        # Compute the maximum distance to the square boundary
        dist_to_boundary = min(min_x, max_x - square_size, min_y, max_y - square_size)

        # Compute the minimum distance to other circles
        min_dist_to_circle = np.inf
        for i in indices:
            for j in indices:
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_circle:
                        min_dist_to_circle = dist

        return min(dist_to_boundary, min_dist_to_circle / 2)

    def objective_function(radii, centers):
        return -np.sum(radii)  # Negative for minimization

    def constraint_overlap(centers, radii):

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
