import scipy.optimize as opt

def run_packing():

    num_circles = 26
    # Hexagon grid: rows = 5, col = 5 + 1 (row0: 5, row1: 4, row2: 4, row3: 4, row4: 5)
    rows = 5
    cols = 5
    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if row % 2 == 0:
                x = col * (1 / (cols - 1)) * 2 / 3
            else:
                x = (col + 0.5) * (1 / (cols - 1)) * 2 / 3
            y = row * (1 / (rows - 1)) * 2 / 3
            centers[idx] = [x, y]
            idx += 1

    # Initial guess for radii
    # Start with equal radii
    radii = np.full(num_circles, 0.05)

    # Define objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_boundary(x, centers, radii):
        for i in range(len(x)):
            x_i, y_i = centers[i]
            r = x[i]
            if (x_i - r < -1e-12 or x_i + r > 1 + 1e-12
                or y_i - r < -1e-12 or y_i + r > 1 + 1e-12):
                return -1
        return 0

    def constraint_overlap(x, centers):
        for i in range(len(x)):
            for j in range(i + 1, len(x)):

                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-12:

        return 0

        x0=x0,
