import numpy as np
import scipy.optimize as opt
import random

def run_packing():

    def generate_hexagonal_packing(n_circles):
        centers = []
        radii = []
        for i in range(n_circles):
            row = i // 6
            col = i % 6
            x = col * (1 + np.sqrt(3)/2) / 6 + 0.5 * (1 + np.sqrt(3)/2) / 6
            y = row * (np.sqrt(3)/2) / 6
            centers.append([x, y])
            radii.append(1 / 6)
        return np.array(centers), np.array(radii)

    # Define the objective function to maximize the sum of radii
    def objective_function(params, centers, radii):
        # params is a flat array of parameters
        # We will restructure it into centers and radii
        n = len(centers)
        flat_centers = params[:2 * n].reshape(n, 2)
        flat_radii = params[2 * n:]
        return -np.sum(flat_radii)  # Minimize the negative sum to maximize the sum

    # Define the constraints
    def constraint_boundary(params, centers, radii):
        n = len(centers)
        flat_centers = params[:2 * n].reshape(n, 2)
        flat_radii = params[2 * n:]
        for i in range(n):
            x, y = flat_centers[i]
            r = flat_radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def constraint_overlap(params, centers, radii):
        n = len(centers)
        flat_centers = params[:2 * n].reshape(n, 2)
        flat_radii = params[2 * n:]

        return 0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
