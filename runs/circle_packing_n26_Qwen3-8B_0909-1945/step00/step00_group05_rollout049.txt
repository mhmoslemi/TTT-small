import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial setup: hexagonal close packing of 26 circles in a unit square

    # Number of circles
    n = 26

    # HELPER FUNCTIONS
    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraint_non_overlap(centers, radii):
        # Constraint: circles cannot overlap
        def cons(x):
            centers_flat = x[:2 * n].reshape(n, 2)
            radii_flat = x[2 * n:]
            # Check all pairs of circles
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers_flat[i, 0] - centers_flat[j, 0]
                    dy = centers_flat[i, 1] - centers_flat[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                        return dist - (radii_flat[i] + radii_flat[j] - 1e-8)
            return 0.0
        return cons

    def constraint_boundary(centers, radii):
        # Constraint: circles must stay within the unit square
        def cons(x):
            centers_flat = x[:2 * n].reshape(n, 2)
            radii_flat = x[2 * n:]
            for i in range(n):
                x_center, y_center = centers_flat[i]
                r = radii_flat[i]
                if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                    return x_center - r - 1e-8
            return 0.0
        return cons

    def constraint_nonnegative(radii):
        # Constraint: radii must be non-negative
        def cons(x):
            return np.min(x[2 * n:])  # Ensure minimum radius is non-negative
        return cons

    # ---------- Initial guess: hexagonal packing ----------

    # Hexagonal close packing of 26 circles in a unit square
    # Start with approximate hexagonal grid

    num_rows = 4  # 4 rows in hexagonal grid (starting from top)

    # Approximate positions

    centers = np.zeros((n, 2))
    r = 0.035  # Initial radius estimate

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
