                x += offset
            centers[index] = [x, y]
            index += 1

    # Objective function: maximize sum of radii
    def objective(x):
        return -x[0]  # Minimize negative sum to maximize the sum

    # Constraint: non-overlapping circles
    def constraint_overlap(x, centers, radii):

        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:].reshape(n)

                dx = centers_flat[i, 0] - centers_flat[j, 0]
