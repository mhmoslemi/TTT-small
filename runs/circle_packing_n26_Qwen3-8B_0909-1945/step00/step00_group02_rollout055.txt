import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: Hexagonal grid
    num_circles = 26
    # Create a hexagonal grid
    rows = int(np.sqrt(num_circles)) + 1
    cols = int(np.ceil(num_circles / rows))
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place circles in a hexagonal grid
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * row
        y = row
        centers[i, 0] = x / 10  # scale to fit in unit square
        centers[i, 1] = y / 10
        # Initialize radii to a small value
        radii[i] = 0.01

    # Optimization function
    def objective(radii):
        return -np.sum(radii)  # we minimize the negative sum to maximize sum

    def constraint_overlap(i, j, centers, radii):
        dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        return dist - radii[i] - radii[j]  # >= 0 for non-overlapping

    # Build constraints for all pairs of circles
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j, c=centers: constraint_overlap(i, j, c, r)
            })

    # Boundary constraint: each circle must be inside the unit square
    for i in range(num_circles):
        def boundary_constraint(r, i=i):
            x, y = centers[i]
            r_i = r[i]
            return x - r_i  # >= -1e-12 (left boundary)
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint
        })

            x, y = centers[i]
            r_i = r[i]
            return 1 - x - r_i  # >= -1e-12 (right boundary)

            'type': 'ineq',
            'fun': boundary_constraint_right

            x, y = centers[i]
            r_i = r[i]
            return y - r_i  # >= -1e-12 (bottom boundary)

    # Run the optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000}

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
