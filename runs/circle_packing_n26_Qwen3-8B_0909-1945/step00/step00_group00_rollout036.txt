import scipy.optimize as opt

def run_packing():
    # Initial grid-based arrangement
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing = 1.0 / (grid_rows + 1)
    grid_x = np.linspace(spacing, 1 - spacing, grid_cols)
    grid_y = np.linspace(spacing, 1 - spacing, grid_rows)
    centers = np.column_stack((np.repeat(grid_x, grid_rows), np.tile(grid_y, grid_cols)))
    radii = np.full(num_circles, 0.1)  # Initial guess for radii

    # Optimization variables: radii and positions
    x0 = np.concatenate([radii, centers.reshape(-1)])

    # Bounds for radii (non-negative) and positions (within [0, 1]^2)
    bounds = []

        bounds.extend([(-1e-6, 1e6), (0, 1), (0, 1)])

    # Function to compute the total sum of radii
    def objective(x):
        radii_opt = x[:num_circles]
        centers_opt = x[num_circles:].reshape(-1, 2)
        return -np.sum(radii_opt)  # Minimize negative sum to maximize sum

    # Constraints

        centers_opt = x[num_circles:].reshape(-1, 2)

                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_opt[i] + radii_opt[j] - 1e-6:
                    return dist - (radii_opt[i] + radii_opt[j] - 1e-6)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
