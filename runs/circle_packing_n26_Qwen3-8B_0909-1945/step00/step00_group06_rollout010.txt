import scipy.optimize as opt

def run_packing():

    centers = np.zeros((26, 2))

        for j in range(5):
            centers[i * 5 + j] = [j / 4, i / 4]

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):

            x, y = centers[i]
            if (x - radii[i] < 0 or x + radii[i] > 1 or

        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]

                dist = np.sqrt(dx * dx + dy * dy)
