                x = j * spacing
            else:
                x = j * spacing + spacing / 2
            y = i * spacing

            if idx < num_circles:
                centers[idx] = np.array([x, y])

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Maximize sum of radii
