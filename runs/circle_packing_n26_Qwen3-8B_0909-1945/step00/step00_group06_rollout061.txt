import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define initial hexagonal grid layout for 26 circles in a unit square
    num_circles = 26
    radii = np.zeros(num_circles)
    centers = np.zeros((num_circles, 2))

    # Hexagonal packing with variable row spacing
    row_count = 5
    col_count = 6
    spacing = 1.0 / (col_count + 1)
    col_spacing = 1.0 / (row_count + 1)

    # Create an initial hexagonal grid
    for i in range(row_count):
        for j in range(col_count):
            x = j * spacing
            y = i * col_spacing
            if i % 2 == 1:
                x += spacing / 2.0
            centers[i * col_count + j] = [x, y]

    # Optimization: find the best radii configuration
    def objective_function(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - (radii[i] + radii[j])

    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j, c=centers: constraint_overlap(i, j, c, r),
                'args': (centers,)
            })

    # Add boundary constraints (circles must lie within the unit square)
    for i in range(num_circles):
        def boundary_constraint(r, i=i):
            x, y = centers[i]
            return np.array([x - r[0] >= -1e-12, x + r[0] <= 1 + 1e-12,

        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint,
            'args': (centers,)
        })

    # Optimize
    result = opt.minimize(
        objective_function,
        x0=radii,
        bounds=bounds,
        constraints=constraints
    )

    # Validate the result
    if not result.success:
        raise ValueError("Optimization failed")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
