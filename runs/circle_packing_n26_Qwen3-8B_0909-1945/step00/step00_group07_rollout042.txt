def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    radii = np.zeros(num_circles)

            x = j * grid_spacing + grid_spacing / 2
            y = i * grid_spacing + grid_spacing / 2
            centers[idx] = [x, y]
            radii[idx] = min(grid_spacing / 2, 0.05)

    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = sum(best_radii)

    # Define optimization variables as a flat array
    def flatten_params(params):
        return params.reshape(-1)

    def unflatten_params(flattened, shape):
        return flattened.reshape(shape)

    # Objective function to maximize total radius sum
    def objective(params, centers_shape, radii_shape):
        flat_centers, flat_radii = unflatten_params(params, centers_shape, radii_shape)
        return -np.sum(flat_radii)  # Negative for minimization

    # Constraint: no circle outside the square
    def constraint_outside(params, centers_shape, radii_shape):
        flat_centers, flat_radii = unflatten_params(params, centers_shape, radii_shape)
        for i in range(num_circles):
            x, y = flat_centers[i]
            r = flat_radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1000  # Penalize to discourage
        return 0

    # Constraint: no two circles overlapping
    def constraint_overlap(params, centers_shape, radii_shape):
        flat_centers, flat_radii = unflatten_params(params, centers_shape, radii_shape)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < flat_radii[i] + flat_radii[j] - 1e-12:
                    return -1000  # Penalize to discourage
        return 0

    # Initial parameter vector
    flat_centers = centers.reshape(-1)
    flat_radii = radii.reshape(-1)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
