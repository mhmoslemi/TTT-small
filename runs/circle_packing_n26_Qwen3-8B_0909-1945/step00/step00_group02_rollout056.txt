import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    grid_size = int(np.sqrt(num_circles))
    spacing = 1.0 / (grid_size + 1)
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        x = col * spacing + spacing / 2
        y = row * spacing + spacing / 2
        centers[i] = [x, y]
    
    # Initial radii for all circles
    radii = np.zeros(num_circles)
    # Set initial radius to be uniform based on spacing
    initial_radius = spacing / 3
    radii[:] = initial_radius

        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        centers_flat = x[:num_circles * 2]
        radii_flat = x[num_circles * 2:]
        centers = centers_flat.reshape((num_circles, 2))
        radii = radii_flat

        # Objective: maximize sum of radii
        return -np.sum(radii)

    # Define constraints
    def constraints(x):
        centers_flat = x[:num_circles * 2]
        radii_flat = x[num_circles * 2:]
        centers = centers_flat.reshape((num_circles, 2))
        radii = radii_flat

        # Constraints: center[i] must be within [0,1]^2
        center_constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            center_constraints.append((x >= 0) & (x <= 1) & (y >= 0) & (y <= 1))
        center_constraints = np.all(center_constraints, axis=0)

        # Constraints: radius[i] must be non-negative

        radius_constraints = np.all(radius_constraints)

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                overlap = dist < radii[i] + radii[j] - 1e-12
                overlap_constraints.append(not overlap)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
