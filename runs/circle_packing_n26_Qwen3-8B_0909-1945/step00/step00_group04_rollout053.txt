import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing of 26 circles
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles) * 0.05  # Initial small radii

    # Hexagonal packing
    rows = int(np.sqrt(num_circles))
    cols = num_circles // rows
    remaining = num_circles % rows

    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[i * cols + j] = [x, y]
        if i < remaining:
            centers[i * cols + cols] = [cols + 0.5, i * np.sqrt(3) / 2]

    # Scale to unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers = centers / np.array([max_x, max_y])

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    def constraint_center_boundaries(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return -np.inf
        return 0

    def constraint_overlap(radii):

                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx*dx + dy*dy)
                if distance < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return 0

    # Optimization
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_center_boundaries},
            {'type': 'ineq', 'fun': constraint_overlap}
        ],
        tol=1e-4
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
