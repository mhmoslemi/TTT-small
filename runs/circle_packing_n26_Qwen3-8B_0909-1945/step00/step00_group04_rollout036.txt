import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: define a hexagonal grid with a small number of rows and columns
    # We'll use a partial hex grid and manually add circles near the corners
    
    # Generate a hexagonal grid with 5 rows
    num_rows = 5
    num_cols = 5
    grid_radius = 0.15
    
    centers = np.zeros((num_rows * num_cols, 2))
    idx = 0
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * grid_radius * np.sqrt(3)
            y = i * grid_radius * 1.5
            if i % 2 == 1:
                x += grid_radius * np.sqrt(3) / 2
            centers[idx] = [x, y]
            idx += 1

    # Add circles at the corners
    corner_radii = [0.12, 0.12, 0.12, 0.12]
    corner_centers = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.9, 0.9],
        [0.1, 0.9]
    ])
    centers = np.vstack([centers, corner_centers])
    radii = np.zeros(centers.shape[0])
    
    # Assign initial radii to the grid
    for i in range(len(centers)):
        if i < len(centers) - 4:
            radii[i] = 0.15
        else:
            radii[i] = 0.12

    # Define variables
    n = len(centers)
    centers_var = cp.Variable((n, 2))
    radii_var = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii_var))

    # Constraints: circles stay within the unit square
    constraints = []
    for i in range(n):
        constraints.append(centers_var[i, 0] >= radii_var[i])
        constraints.append(centers_var[i, 0] <= 1 - radii_var[i])
        constraints.append(centers_var[i, 1] >= radii_var[i])
        constraints.append(centers_var[i, 1] <= 1 - radii_var[i])

    # Constraints: circles do not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(centers_var[i] - centers_var[j])
            constraints.append(dist >= radii_var[i] + radii_var[j] - 1e-6)

    # Initial guess for centers and radii
    init_centers = centers.copy()
    init_radii = radii.copy()
    for i in range(n):
        init_centers[i, 0] = max(0.0, min(1.0, init_centers[i, 0]))
        init_centers[i, 1] = max(0.0, min(1.0, init_centers[i, 1]))
        init_radii[i] = max(0.0, min(0.2, init_radii[i]))

    # Use a solver to find the optimal packing
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False, init=cp.reshape(np.hstack([init_centers.flatten(), init_radii]), (n * 3,)))

    # Extract the results
    centers_opt = centers_var.value
    radii_opt = radii_var.value
    sum_radii_opt = cp.sum(radii_var).value

    # Validate the solution
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, sum_radii_opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
