import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n_circles = 26

    # Initial guess: hexagonal packing in a square
    # Distribute circles in a hexagonal grid in the unit square
    # We sample points in a grid to approximate a hexagonal packing
    # and place the circles such that they do not overlap
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # Approximate hexagonal grid spacing
    spacing = 0.4
    row_count = int(np.ceil(np.sqrt(n_circles)))
    col_count = int(np.ceil(n_circles / row_count))

    for i in range(n_circles):
        row = i // col_count
        col = i % col_count
        x = col * spacing
        y = row * (spacing * np.sqrt(3) / 2)
        centers[i] = [x, y]
        radii[i] = 0.1

    # Define objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints
    def constraint_center_bounds(radii):
        # Ensure each circle is within the unit square
        for i in range(n_circles):
            x, y = centers[i]
            if not (0 - radii[i] <= x <= 1 + radii[i] and 0 - radii[i] <= y <= 1 + radii[i]):
                return False
        return True

    def constraint_no_overlap(radii):
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                distance = np.sqrt(dx*dx + dy*dy)
                if distance < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Define constraints for optimization
    constraints = [
        {
            'type': 'ineq',
            'fun': lambda r: np.array([r[i] for i in range(n_circles)])  # Non-negative radii
        },
        {
            'type': 'ineq',
            'fun': constraint_center_bounds
        },
        {
            'type': 'ineq',
            'fun': constraint_no_overlap
        }
    ]

    # Bounds for radii: [0, 0.5] since radius can't exceed half the side length
    bounds = [(0, 0.5) for _ in range(n_circles)]

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Extract results
    optimal_radii = result.x
    optimal_centers = centers
    sum_radii = np.sum(optimal_radii)

    # Validate the packing
    if not validate_packing(optimal_centers, optimal_radii)[0]:
        raise ValueError("Invalid packing configuration detected")

    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
