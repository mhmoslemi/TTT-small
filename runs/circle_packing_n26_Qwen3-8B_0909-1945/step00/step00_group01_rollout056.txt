import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define initial hexagonal packing in a unit square
    # Total number of circles
    n = 26

    # Define initial grid positions using hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    # Start by placing 2 rows with 13 circles each
    for y in range(2):
        for x in range(13):
            x_pos = x / 12.0
            y_pos = y / 2.0
            centers[idx, 0] = x_pos
            centers[idx, 1] = y_pos
            idx += 1

    # Add some circles in a hexagonal pattern
    for y in range(3, 6):
        for x in range(12):
            x_pos = x / 11.0
            y_pos = y / 5.0
            centers[idx, 0] = x_pos
            centers[idx, 1] = y_pos
            idx += 1

    # Add more circles in a hexagonal pattern
    for y in range(6, 9):
        for x in range(11):
            x_pos = x / 10.0
            y_pos = y / 8.0
            centers[idx, 0] = x_pos
            centers[idx, 1] = y_pos
            idx += 1

    # Define the radii as a starting guess (all the same)
    radii = np.full(n, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint_overlap(radii, centers):
        n = len(radii)
        dist_between = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dist_between[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        dist_between = np.triu(dist_between)
        return np.minimum(dist_between - (radii[:, np.newaxis] + radii[np.newaxis, :]), 0)

    def constraint_boundary(radii, centers):
        n = len(radii)
        dist_boundary = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            dist_boundary[i] = min(x - r, 1 - x - r, y - r, 1 - y - r)
        return np.minimum(dist_boundary, 0)

    # Optimization problem setup
    bounds = [(0, 0.5) for _ in range(n)]

    # Uses scipy.optimize to perform optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda r: constraint_overlap(r, centers)
            },

                'fun': lambda r: constraint_boundary(r, centers)
            }
        ],
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}

    radii_opt = result.x

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
