import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial placement: hexagonal grid with some circles along edges and corners
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Create a hexagonal grid with some circles near corners
    base_r = 0.08  # Initial radius for central circles
    spacing = 2 * base_r * np.sqrt(3)  # Hexagonal grid spacing

    # Place circles near the center with hexagonal arrangement
    for i in range(5):
        for j in range(5):
            x = 0.5 + j * spacing - 2 * spacing / 3
            y = 0.5 + i * spacing * np.sqrt(3) / 2 - spacing / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[i * 5 + j] = [x, y]
                radii[i * 5 + j] = base_r

    # Place circles near the corners
    corners = [(0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9)]
    for idx, (x, y) in enumerate(corners):
        centers[25 - idx] = [x, y]
        radii[25 - idx] = 0.05

    # Optimization bounds: only adjust radii, not centers
    bounds = [(0, 0.5) for _ in range(26)]

    # Define objective to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Maximize by minimizing negative sum

    # Define constraints: non-overlapping and inside the square
    def constraint_overlap(radii, centers):
        n = len(radii)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dists[i][j] = np.sqrt(dx*dx + dy*dy)
        return np.array([dists[i][j] - (radii[i] + radii[j]) for i in range(n) for j in range(i + 1, n)])

    def constraint_boundary(radii, centers):
        return np.array([centers[i][0] - radii[i] for i in range(n)] + [centers[i][1] - radii[i] for i in range(n)] +
                        [1 - centers[i][0] - radii[i] for i in range(n)] + [1 - centers[i][1] - radii[i] for i in range(n)])

    # Initial guess
    x0 = radii.copy()

    # Run optimization
    cons = [
        {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)},
        {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)}
    ]

    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )

    # Extract final solution
    radii = result.x
    centers = centers  # Positions are fixed during optimization

    # Calculate total sum of radii
    sum_radii = np.sum(radii)

    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
