import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing of 26 circles
    # 5 rows with 5, 5, 5, 5, 6 circles respectively
    num_rows = 5
    num_cols = [5, 5, 5, 5, 6]
    rows = [i for i in range(num_rows)]
    cols = [j for j in range(len(num_cols))]

    # Hexagonal grid parameters
    h = np.sqrt(3) / 2  # Height of hexagon
    w = 1.0 / (max(num_cols) + 1)  # Width of each column

    centers = []
    radii = []

    for row in rows:
        for col in range(num_cols[row]):
            x = col * w + w / 2
            y = row * h + h / 2
            centers.append([x, y])
            radii.append(1.0 / (max(num_cols) + 1))  # Initial radius

    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Define the constraints
    def constraint_center_boundary(x):
        return np.array([
            x[0] - x[26] < 1e-12,
            x[1] - x[27] < 1e-12,
            x[0] + x[26] < 1 - 1e-12,
            x[1] + x[27] < 1 - 1e-12
        ])

    def constraint_overlap(x):
        r = x[26:]
        c = x[:26].reshape(-1, 2)
        dist = np.sqrt(np.sum((c[:, np.newaxis] - c[np.newaxis, :]) ** 2, axis=2))
        constraints = np.zeros((26, 26))
        for i in range(26):
            for j in range(i+1, 26):
                constraints[i, j] = dist[i, j] - r[i] - r[j]
        return constraints.flatten()

    # Combine all constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))

    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(26 * 2)] + [(0, 1) for _ in range(26)],
        tol=1e-6
    )

    # Extract results

    # Validate the packing

    if not validity:
        raise ValueError(f"Packing validation failed: {message}")

    sum_radii = np.sum(optimized_radii)

    return optimized_centers, optimized_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
