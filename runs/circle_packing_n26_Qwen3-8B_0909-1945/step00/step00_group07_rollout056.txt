import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Define the number of circles
    n = 26

    # Generate an initial hexagonal grid layout
    def hexagonal_grid(num_circles):
        grid = []
        # Generate a hexagonal grid with a certain number of circles
        for i in range(int(np.sqrt(num_circles))):
            for j in range(int(np.sqrt(num_circles))):
                # Determine the spacing based on the number of rows and columns
                cols = int(np.sqrt(num_circles))
                rows = int(np.sqrt(num_circles))
                spacing = 1.0 / (rows + 1)
                x = j * spacing
                y = i * spacing * math.sqrt(3)
                grid.append((x, y))
        # If there are more circles than a perfect square, add extra balls
        while len(grid) < num_circles:
            grid.append((0.5, 1.0))
        return np.array(grid)

    centers = hexagonal_grid(n)

    # Initial guess for radii (based on spacing)
    spacing = 1.0 / (np.sqrt(n) + 1)
    radii = np.full(n, spacing / 2)

    # Ensure the radii do not exceed half the square size
    radii = np.minimum(radii, 0.5)

    def objective(radii):
        return -np.sum(radii)  # We minimize the negative sum to maximize the sum

    # Define constraints
    def constraint_circle_in_square(radii, centers):
        return np.sum((centers - radii[:, np.newaxis] < 0) | (centers + radii[:, np.newaxis] > 1))

    def constraint_no_overlap(radii, centers):
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                dists[i, j] = dist - radii[i] - radii[j]
        return np.sum(dists < -1e-12)

    # Optimization problem
    bounds = [(0, 0.5) for _ in range(n)]
    cons = [
        {'type': 'ineq', 'fun': lambda r: constraint_circle_in_square(r, centers)},
        {'type': 'ineq', 'fun': lambda r: constraint_no_overlap(r, centers)}
    ]

    # Solve the optimization problem
    result = optimize.minimize(
        objective, radii, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6
    )

    # Extract the optimized centers and radii
    centers_opt = centers
    radii_opt = result.x

    if not is_valid:
        raise ValueError(f"Pack failed validation: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
