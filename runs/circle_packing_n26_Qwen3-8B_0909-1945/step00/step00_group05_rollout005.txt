import numpy as np
import scipy.optimize as opt
from typing import Tuple

def compute_initial_grid(n: int) -> np.ndarray:
    """
    Generate a basic grid-based initial placement for n circles in a unit square.
    """
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = []
    for i in range(n):
        x = i % cols / cols
        y = i // cols / rows
        centers.append([x, y])
    return np.array(centers)

def objective_function(x: np.ndarray, n: int, radius: float) -> float:
    """
    Objective function to maximize the sum of radii.
    """
    return -radius  # Minimize negative of radius (maximize radius)

def constraint_boundary(x: np.ndarray, radius: float) -> float:
    """
    Constraint to ensure all circles are within the unit square.
    """
    centers = x.reshape(-1, 2)
    for i in range(len(centers)):
        x_i, y_i = centers[i]
        if x_i - radius < -1e-12 or x_i + radius > 1 + 1e-12 or y_i - radius < -1e-12 or y_i + radius > 1 + 1e-12:
            return -1.0  # Violation
    return 0.0

def constraint_non_overlap(x: np.ndarray, radii: np.ndarray) -> float:
    """
    Constraint to ensure non-overlapping circles.
    """
    centers = x.reshape(-1, 2)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return -1.0  # Violation
    return 0.0

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = compute_initial_grid(n)
    initial_radii = np.ones(n) * 0.04  # Initial low radius to avoid overlap

    # Optimize radii and positions
    bounds = [(0.0, 1.0) for _ in range(n * 2)]  # x and y for each circle
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary, 'args': (initial_radii[0],)},

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
