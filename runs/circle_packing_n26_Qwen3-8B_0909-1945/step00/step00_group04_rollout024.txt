import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal close packing (HCP) pattern
    # Parameters for the HCP pattern
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    spacing = 1.0 / (cols + 1)  # horizontal spacing
    vertical_spacing = np.sqrt(3) / (2 * (rows + 1))  # vertical spacing
    centers = np.empty((num_circles, 2))
    index = 0
    for row in range(rows):
        for col in range(cols):
            x = spacing * (col + 1)
            y = vertical_spacing * (row + 1)

                x += spacing / 2
            centers[index] = [x, y]
            index += 1

    # Initial radii
    radii = np.ones(num_circles) * 0.05

    # Objective: maximize the sum of radii
    def objective(x):
        # x contains all radii (first num_circles elements)
        # Make it a minimization problem by negating
        return -np.sum(x)

    # Constraints
    def constraint_center_boundary(x):
        # Ensure all circles are within [0,1] x [0,1]
        for i in range(num_circles):
            if centers[i, 0] - x[i] < -1e-12 or centers[i, 0] + x[i] > 1 + 1e-12:
                return -1
            if centers[i, 1] - x[i] < -1e-12 or centers[i, 1] + x[i] > 1 + 1e-12:
                return -1
        return 0

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
