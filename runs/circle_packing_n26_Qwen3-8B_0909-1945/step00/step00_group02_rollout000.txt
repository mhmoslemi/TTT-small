import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Initial grid-based layout
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing = 1.0 / (grid_rows + 1)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate initial centers using a grid pattern
    for i in range(grid_rows):
        for j in range(grid_cols):
            idx = i * grid_cols + j
            if idx >= num_circles:
                break
            centers[idx, 0] = (j + 1) * spacing
            centers[idx, 1] = (i + 1) * spacing
            radii[idx] = spacing * 0.35  # Initial radius

    # Step 2: Define constraints and objective function
    def objective(radii):
        return -np.sum(radii)  # Negative for maximization

    def constraint_center_bounds(x, i):
        x_i, y_i = x[2 * i], x[2 * i + 1]
        r_i = radii[i]
        return np.array([x_i - r_i + 1e-12, x_i + r_i - 1 + 1e-12,
                         y_i - r_i + 1e-12, y_i + r_i - 1 + 1e-12])
    
    def constraint_overlap(x, i, j):
        x_i, y_i = x[2 * i], x[2 * i + 1]
        x_j, y_j = x[2 * j], x[2 * j + 1]
        r_i = radii[i]
        r_j = radii[j]
        dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
        return dist_sq - (r_i + r_j)**2 + 1e-12

    # Flatten centers to a 1D vector for optimization
    x0 = np.concatenate((centers.reshape(-1), radii))
    bounds = []
    for i in range(num_circles):
        x_low = 0.0
        x_high = 1.0
        y_low = 0.0
        y_high = 1.0
        r_low = 0.0
        r_high = 1.0
        bounds.extend([(x_low, x_high), (y_low, y_high), (r_low, r_high)])
    
    # Set up the optimization problem
    cons = []
    for i in range(num_circles):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_center_bounds(x, i)})
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)})
    
    # Solve the optimization problem
    result = opt.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-6
    )

    # Extract results
    centers = result.x[:2 * num_circles].reshape(-1, 2)
    radii = result.x[2 * num_circles:]
    sum_radii = np.sum(radii)

    # Final check (optional)

    if not valid:

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
