import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Define variable for circle positions (x_i, y_i) and radii r_i
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, nonneg=True)

    # Objective: maximize the total radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be within unit square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= -1e-12)
        constraints.append(x <= 1 + 1e-12)
        constraints.append(y >= -1e-12)
        constraints.append(y <= 1 + 1e-12)

    # Each pair of circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    result = prob.solve()

    if result is None:
        raise RuntimeError("Optimization failed.")

    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)

    # Return centers (numpy array), radii (numpy array), and sum_radii
