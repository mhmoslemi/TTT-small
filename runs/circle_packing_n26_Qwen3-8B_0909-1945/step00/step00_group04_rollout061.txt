import numpy as np
import scipy.optimize as opt
import random
import math

def run_packing():
    # Initial configuration: hexagonal grid near the bottom-left corner
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    circle_radius_initial = 0.042

    # Create initial positions in a hexagonal grid
    centers = np.zeros((num_circles, 2))
    for i in range(grid_rows):
        for j in range(grid_cols):
            # Hexagonal grid base
            x = j * (1 + math.sqrt(3) * circle_radius_initial)
            y = i * (2 * circle_radius_initial) + circle_radius_initial
            if i % 2 == 1:
                x += (1 + math.sqrt(3) * circle_radius_initial) / 2
            centers[i * grid_cols + j] = [x, y]

    # Scale to unit square
    centers[:, 0] /= max(centers[:, 0]) + 1e-8
    centers[:, 1] /= max(centers[:, 1]) + 1e-8

    # Initial radii
    radii = np.full(num_circles, circle_radius_initial)

    # Define objective function to minimize
    def objective(x):
        # x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
        # Reshape
        x_flat = np.array(x).reshape(-1, 4)
        centers_flat = x_flat[:, :2]
        radii_flat = x_flat[:, 2]

        # Check boundaries
        for i in range(num_circles):
            x, y, r = x_flat[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf

        # Check overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf

        # Objective: maximize sum of radii => minimize negative of sum
        return -np.sum(radii_flat)

    # Define bounds for each circle: x, y ∈ [0, 1], r ∈ [0, 0.5]
    bounds = []
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 0.5))  # r

    # Run simulated annealing to optimize
    result = opt.minimize(
        objective,
        x0=np.concatenate([centers.reshape(-1), radii]),
        bounds=bounds,
        method='SLSQP',
        options={
            'ftol': 1e-6,
            'eps': 1e-6,
            'maxiter': 1000
        }
    )

    # Extract optimal centers and radii
    optimized = result.x.reshape(-1, 4)
    centers_opt = optimized[:, :2]
    radii_opt = optimized[:, 2]

    # Validate the result
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    # Return the result
    return centers_opt, radii_opt, -result.fun

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
