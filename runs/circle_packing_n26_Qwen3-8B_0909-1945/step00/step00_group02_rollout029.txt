import numpy as np
from scipy.optimize import Bounds, minimize
import warnings

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:

def objective_function(x, centers, radii, n):

    return -np.sum(radii)

def constraint_overlap(x, centers, radii, n):

    for i in range(n):

            dx = x[2*i] - x[2*j]

def constraint_bounds(x, centers, radii, n):
    # Constraint: circles must be within the unit square
    for i in range(n):
        x_i = x[2*i]

    radii = np.full(n, 0.1)  # Start with uniform radii
