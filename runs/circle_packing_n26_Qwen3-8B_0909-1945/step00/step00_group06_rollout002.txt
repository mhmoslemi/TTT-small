import numpy as np
import scipy.optimize as opt

def generate_initial_centers(n):
    """Generate initial positions for circle centers in a unit square."""
    # Determine number of rows and columns for a grid-like arrangement
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Create a grid of centers spaced evenly
    x_coords = np.linspace(0.1, 0.9, cols)
    y_coords = np.linspace(0.1, 0.9, rows)
    centers = np.array([x_coords[i % cols] for i in range(n)])
    centers = np.vstack([centers, y_coords[i // cols] for i in range(n)]).T

    return centers

def run_packing():
    n = 26
    initial_centers = generate_initial_centers(n)
    initial_radii = np.ones(n) * 0.05  # Start with small radii
    
    # Objective function to minimize (negative sum of radii for maximization)
    def objective(radii):
        return -np.sum(radii)

    # Constraints: ensure no circle is outside the unit square and no two circles overlap
    def constraint_square(radii):
        centers = np.array([initial_centers[i] for i in range(n)])
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf
        return 0

    def constraint_overlap(radii):
        centers = np.array([initial_centers[i] for i in range(n)])
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_square},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Optimization
    result = opt.minimize(
        objective,
        initial_radii,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1.0) for _ in range(n)]
    )

    # Extract results
    centers = np.array([initial_centers[i] for i in range(n)])
    radii = result.x
    total_sum = np.sum(radii)

    # Validate the packing
    success, msg = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Packing validation failed: {msg}")

    return centers, radii, total_sum
