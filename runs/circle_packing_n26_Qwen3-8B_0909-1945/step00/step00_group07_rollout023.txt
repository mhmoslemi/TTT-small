import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Define the number of circles
    n = 26

    # Step 1: Generate initial layout in a hexagonal grid
    # We'll create a grid of (x, y) coordinates representing circle centers
    # and assume equal radii initially to form a hexagonal packing
    radius = 0.05  # Initial guess for radius
    centers = np.zeros((n, 2))
    idx = 0

    # Generate hexagonal grid within the unit square
    for row in range(int(math.sqrt(n))):
        for col in range(int(math.sqrt(n))):
            x = col * (1 + radius * 2)
            y = row * (1 + radius * 2)
            if x < 1 and y < 1:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1

    # Ensure we have exactly 26 circles
    if idx < n:
        # Add remaining circles near the boundary
        for i in range(idx, n):
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            centers[i, 0] = x
            centers[i, 1] = y

    # Optimize the configuration to maximize sum of radii
    initial_radii = np.full(n, radius)

    # Objective function
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    # Constraints
    def constraint_overlap(radii):
        # Non-overlapping constraint
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1  # Violated
        return 0

    def constraint_boundary(radii):
        # Boundary constraint
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1  # Violated
        return 0

    # Define constraints in the format expected by scipy.optimize.minimize
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Optimization
    result = minimize(objective, initial_radii, method='SLSQP', constraints=cons, bounds=[(0, 1)] * n)

    if not result.success:
        raise Exception(f"Optimization failed with message: {result.message}")

    final_radii = result.x
    sum_radii = np.sum(final_radii)

    valid, message = validate_packing(centers, final_radii)
    if not valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
