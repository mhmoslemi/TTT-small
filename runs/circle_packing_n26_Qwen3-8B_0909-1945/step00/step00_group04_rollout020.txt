import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Constants
    N = 26  # number of circles
    TOLERANCE = 1e-6
    MAX_ITERATIONS = 100
    MIN_RADIUS = 1e-6

    # Initialize centers with hexagonal packing
    centers = np.zeros((N, 2))
    row_count = int(np.sqrt(N)) + 1
    col_count = int(np.sqrt(N)) + 1
    spacing = 1.0 / (row_count - 1) * 0.9

    idx = 0
    for i in range(row_count):
        for j in range(col_count):
            if idx >= N:
                break
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            centers[idx] = [x, y]
            idx += 1

    # Initial guess for radii
    radii = np.zeros(N)
    for i in range(N):
        radii[i] = 0.035  # Initial estimate based on hexagonal packing density

    # Define objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_overlap(x, centers):
        r = x
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - TOLERANCE:
                    return -1
        return 0

    def constraint_boundary(x, centers):
        r = x
        for i in range(N):
            x_center, y_center = centers[i]
            if (x_center - r[i] < -TOLERANCE or x_center + r[i] > 1 + TOLERANCE or
                y_center - r[i] < -TOLERANCE or y_center + r[i] > 1 + TOLERANCE):
                return -1
        return 0

    # Optimize
    initial_guess = radii.copy()
    res = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=[(MIN_RADIUS, 1 - TOLERANCE) for _ in range(N)],
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)},
            {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)}
        ],
        options={'maxiter': MAX_ITERATIONS, 'ftol': 1e-12, 'disp': False}
    )

    # Extract results
    radii = res.x
    centers = np.clip(centers, [TOLERANCE, TOLERANCE], [1 - TOLERANCE, 1 - TOLERANCE])
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
