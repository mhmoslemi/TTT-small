import numpy as np
import scipy.optimize as optimize
import sys

def distance(x1, y1, x2, y2):
    return np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

def compute_total_radius(centers, radii):
    """Compute the sum of all radii."""
    return np.sum(radii)

def objective_function(params, n_circles):
    """Objective function to maximize the sum of radii."""
    centers = params[:2*n_circles].reshape(n_circles, 2)
    radii = params[2*n_circles:]
    return -compute_total_radius(centers, radii)  # Negative for minimization

def constraints_function(params, n_circles):
    """Compute constraints for optimization."""
    centers = params[:2*n_circles].reshape(n_circles, 2)
    radii = params[2*n_circles:]
    
    # Boundary constraints
    boundary_constraints = []
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        boundary_constraints.append(x - r + 1e-12)  # x - r >= -1e-12
        boundary_constraints.append(x + r - 1 - 1e-12)  # x + r <= 1 + 1e-12
        boundary_constraints.append(y - r + 1e-12)  # y - r >= -1e-12
        boundary_constraints.append(y + r - 1 - 1e-12)  # y + r <= 1 + 1e-12

    # Non-overlapping constraints
    non_overlapping_constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = distance(centers[i][0], centers[i][1], centers[j][0], centers[j][1])
            non_overlapping_constraints.append(dist - radii[i] - radii[j] + 1e-12)  # dist >= radii[i] + radii[j] - 1e-12

    return np.concatenate((boundary_constraints, non_overlapping_constraints))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
