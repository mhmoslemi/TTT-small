def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26

    rows = 5  # Number of rows (adjusted to fit 26 circles in a hexagonal grid)
    spacing = 0.2  # Initial spacing for hexagonal arrangement

    # Generate grid of centers in a hexagonal pattern
    centers = np.zeros((num_circles, 2))
    index = 0
    for row in range(rows):

                x = col * spacing
            else:
                x = (col + 0.5) * spacing
            y = row * spacing * np.sqrt(3) / 2

                index += 1

    radii = np.ones(num_circles) * 0.05

    def objective(x):
        return -np.sum(x)  # Minimize negative sum of radii (equivalent to maximizing)

    # Bounds for radii (non-negative)
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # Constraints

        return np.array([x[idx]]).reshape(-1, 1)

    # Boundary constraint for center (x, y) must be within [0,1]²
    def boundary_constraints(x, idx):
        x_center = centers[idx, 0]
        y_center = centers[idx, 1]
        r = x[idx]
        return np.array([x_center - r, x_center + r, y_center - r, y_center + r])

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(x):

        constraints = []

                r_i = x[i]
                r_j = x[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                constraints.append(dist - r_i - r_j)
        return np.array(constraints)

    # Define nonlinear constraints
    nonlinear_constraints = NonlinearConstraint(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
