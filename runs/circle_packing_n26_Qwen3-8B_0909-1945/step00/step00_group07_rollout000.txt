    N = 26

            x = i / 4
            y = j / 4
            centers.append([x, y])

    y = np.zeros(N * 2)

    y[::2] = centers[:, 1]
    y[1::2] = centers[:, 1]
    r[:] = radii
    
    # Objective function: maximize the sum of radii
    def objective(xyz):
        return -np.sum(xyz[N*2:])  # Negative for minimization
    
    # Constraints
    # 1. Non-overlapping
    def non_overlap_constraints(xyz):
        x = xyz[:N*2].reshape(N, 2)
        r = xyz[N*2:]
        constraints = []
        for i in range(N):
            for j in range(i+1, N):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons = dist - (r[i] + r[j]) + 1e-12
                constraints.append(cons)
        return constraints
    
    # 2. Boundary constraints: x - r >= 0, x + r <= 1, same for y
    def boundary_constraints(xyz):
        x = xyz[:N*2].reshape(N, 2)
