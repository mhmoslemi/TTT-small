import scipy.optimize as optimize

def run_packing():

    def generate_hex_grid(n):

        # Using a hexagonal tiling pattern

        centers = []
        for i in range(n):

            col = i % 2
            angle = 2 * np.pi / 6 * (row % 3)
            x = 0.5 + max_radius * np.cos(angle)
            y = 0.5 + max_radius * np.sin(angle)
            centers.append([x, y])
        return np.array(centers)

    # Initial configuration: 26 circles
    centers = generate_hex_grid(26)

    def objective(x):
        # x is a flattened array of (centers, radii)
        centers_flat = x[:2 * 26].reshape(26, 2)

        return -np.sum(radii_flat)  # Minimize negative of sum to maximize sum

    def constraint_centers(x):
        # Check that all centers are within [0, 1]x[0, 1]
        centers_flat = x[:2 * 26].reshape(26, 2)
        return np.clip(centers_flat, 0, 1).flatten() - x[:2 * 26]

        radii_flat = x[2 * 26:].reshape(26)

                    return np.inf
