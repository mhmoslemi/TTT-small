from scipy.optimize import minimize

def create_hexagonal_grid(n_circles, square_side=1.0):
    """
    Creates an initial hexagonal grid of circles with spacing that fits in the square.
    """
    # Estimate spacing for a hexagonal grid
    spacing = 1.0 / np.sqrt(3) * (np.sqrt(3) / (np.sqrt(3) + 1))  # spacing between centers

    # Initialize centers
    centers = np.zeros((n_circles, 2))
    count = 0
    row = 0
    col = 0
    while count < n_circles:
        x = col * spacing
        y = row * spacing * np.sqrt(3) / 2
        if x + spacing < square_side and y + spacing < square_side:
            centers[count] = [x, y]
            count += 1
        col += 1
        if col % 2 == 0:
            row += 1
    return centers

def validate_packing(centers, radii):
