import scipy.optimize as optimize

                centers.append([x, y])

    initial_radii = np.random.uniform(0.05, 0.1, 26)

    def objective(radii):
        return -np.sum(radii)

    # Define constraints for optimization
    def constraints(centers, radii):
        # Constraint 1: All circles must be inside the unit square
        # Constraint 2: No two circles can overlap

        # Constraint 1: Each circle must be inside the unit square
        def in_square(c, r):
            return (c[0] - r >= 0) and (c[0] + r <= 1) and (c[1] - r >= 0) and (c[1] + r <= 1)

        # Constraint 2: No two circles can overlap
        def no_overlap(c1, r1, c2, r2):
            return np.sqrt(np.sum((c1 - c2) ** 2)) >= r1 + r2

        # Combine all constraints
        constraints = []
        for i in range(len(centers)):
