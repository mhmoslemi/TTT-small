import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based arrangement: 5x5 grid of centers
    # Slightly offset to allow space for circles to fit
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))

    # Generate grid coordinates
    x_coords = np.linspace(0.15, 0.85, grid_size)
    y_coords = np.linspace(0.15, 0.85, grid_size)
    centers = np.array([[x, y] for x in x_coords for y in y_coords])
    centers = centers[:num_circles]  # Only take up to 26 circles

    # Initial guess for radii: uniform estimate based on grid spacing
    grid_spacing = 0.7 / (grid_size - 1)
    initial_radius = grid_spacing / 2
    radii = np.full(num_circles, initial_radius)

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraints: boundary and non-overlap
    def constraint_boundary(radii, centers):
        constraints = []
        for i in range(num_circles):
            r = radii[i]
            x, y = centers[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                constraints.append({"type": "ineq", "fun": lambda r, i=i: x - r + 1e-12})
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                constraints.append({"type": "ineq", "fun": lambda r, i=i: y - r + 1e-12})
        return constraints

    def constraint_non_overlap(radii, centers):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                def dist_func(radii):
                    r_i = radii[i]
                    r_j = radii[j]
                    x_i, y_i = centers[i]
                    x_j, y_j = centers[j]
                    dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                    return dist - r_i - r_j + 1e-12
                constraints.append({"type": "ineq", "fun": dist_func})
        return constraints

    # Optimize
    bounds = [(0, 1.0) for _ in range(num_circles)]
    constraints = constraint_boundary(radii, centers) + constraint_non_overlap(radii, centers)
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method="SLSQP"
    )

    # Extract result
    optimized_radii = result.x
    optimized_centers = centers[:num_circles]
    total_sum = np.sum(optimized_radii)

    # Validate result (for safety)
    is_valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not is_valid:
        raise ValueError(f"Packing validation failed: {msg}")

    return optimized_centers, optimized_radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
