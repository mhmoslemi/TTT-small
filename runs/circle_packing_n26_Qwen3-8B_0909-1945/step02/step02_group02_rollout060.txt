import scipy.optimize as optimize
import math
import random

def run_packing():

            row = i // 5
            col = i % 5
            x = col * spacing + offset_x + random.uniform(-0.01, 0.01)
            y = row * spacing * math.sqrt(3) / 2 + offset_y + random.uniform(-0.01, 0.01)
            centers.append((x, y))
        return np.array(centers)

    # Generate multiple initial center configurations for multi-start optimization

    def objective(x):
        radii = x[:n]
        centers = x[n:].reshape(n, 2)
        if np.any(radii < 0):
            return np.inf
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]

                return np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return -np.sum(radii)

    # Bounds: radii [0, 0.5], centers [0, 1]
    bounds = []
    for _ in range(n):
        bounds.append((0, 0.5))  # radii bounds
    for _ in range(2 * n):
        bounds.append((0, 1))  # centers bounds

    # Generate multiple initial guesses for multi-start optimization
    initial_guesses = generate_initial_guesses(num_guesses=5)

    best_result = None
    best_sum = -float('inf')

    for guess in initial_guesses:
        result = optimize.minimize(

            options={'ftol': 1e-9, 'maxiter': 1500, 'eps': 1e-8}

                best_result = (optimized_centers, optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
