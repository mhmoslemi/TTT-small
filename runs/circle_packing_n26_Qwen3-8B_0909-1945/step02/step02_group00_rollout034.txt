import scipy.optimize as optimize

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal-like grid (more efficient for circle packing)
    # Scale to fit in [0,1]²
    grid_radius = 0.25
    grid_angle = 2 * np.pi / 6  # 60-degree angle for hexagonal packing
    centers[0, 0] = 0.5
    centers[0, 1] = 0.5
    radii[0] = 0.1  # Central circle

    for i in range(1, n):
        angle = i * grid_angle
        radius = grid_radius * (i + 1)
        x = 0.5 + radius * np.cos(angle)
        y = 0.5 + radius * np.sin(angle)
        if x < 1e-6 or x > 1 - 1e-6 or y < 1e-6 or y > 1 - 1e-6:
            # If outside, adjust to the edge
            x = min(max(x, 1e-6), 1 - 1e-6)
            y = min(max(y, 1e-6), 1 - 1e-6)
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.1

    # Flatten initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    # Define objective function with constraint handling
    def objective(x):
        centers_flat = x[:2 * n].reshape((n, 2))
        radii_flat = x[2 * n:]

        # Objective: maximize sum of radii
        sum_radii = np.sum(radii_flat)

        # Check all constraints
        # --- Boundary constraints
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf

        # --- Circle-circle constraints (non-overlapping)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf

        return -sum_radii  # Minimize negative of sum_radii for optimization

    # Define bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 1))  # radius

    # Run optimization with SLSQP method
    result = optimize.minimize(

        jac=None,

    centers_final = result.x[:2 * n].reshape((n, 2))
