import numpy as np
import scipy.optimize as optimize
import math
from scipy.optimize import Bounds

def run_packing():
    # Define number of circles
    n = 26

    # Initial heuristic for placing centers
    # Define initial positions based on a hexagonal packing but optimized for square
    centers = np.zeros((n, 2))
    for i in range(n):
        # Use hexagonal grid with some random jitter to avoid symmetries
        row = i // 5
        col = i % 5
        # Adjust spacing based on number of circles
        spacing = 0.25 * (1 - 0.05 * i)  # Decrease spacing for more dense packing
        # Offset for hexagonal grid
        offset = 0.5 * spacing
        x = col * spacing + offset
        y = row * (spacing * math.sqrt(3)/2) + offset
        # Add small random jitter to avoid symmetry
        jitter = np.random.uniform(-0.05, 0.05, size=2)
        centers[i, :] = [x + jitter[0], y + jitter[1]]
    
    # Define initial radii (based on spacing and square size)
    # Start with smaller radii to allow for optimization
    radii = np.full(n, 0.04)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Constraint function: enforce non-overlap and boundary constraints
    def constraint(x):
        # x is a flat array of radii and centers (unpacked)
        # Convert to centers and radii
        radii = x[:n]
        centers = x[n:].reshape(n, 2)

        # Ensure radii are non-negative
        if np.any(radii < 0):
            return np.inf

        # Check if all circles are within the square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf

        # Check for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf

        return -np.sum(radii)  # Negative for minimization

    # Flatten centers and radii into a single array
    x0 = np.concatenate([radii, centers.flatten()])

    # Optimization settings with better bounds and tolerance
    bounds = []
    for _ in range(n):
        bounds.append((0, 1))  # For radii
    for _ in range(2 * n):
        bounds.append((0, 1))  # For centers

    # Run optimization using SLSQP for better constraint handling
    result = optimize.minimize(
        fun=constraint,
        x0=x0,
        method='SLSQP',
        bounds=Bounds(bounds),
        options={'ftol': 1e-10, 'maxiter': 5000, 'disp': False}
    )

    # Extract results
    optimized_radii = result.x[:n]
    optimized_centers = result.x[n:].reshape(n, 2)

    # Validate
    valid, _ = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError("Packing is invalid")

    sum_radii = np.sum(optimized_radii)
    return optimized_centers, optimized_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
