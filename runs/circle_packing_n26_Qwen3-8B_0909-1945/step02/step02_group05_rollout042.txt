import scipy.optimize as optimize
import math

def run_packing():

    # Initial positions using a hexagonal packing in the center
    # Adapted from: https://en.wikipedia.org/wiki/Circle_packing_in_a_square
    # This places circles in a hexagonal structure, with hexagonal closepacking
    # places in the center and circles near the edges

    # Parameters
    initial_radius = 0.1  # Initial estimate for circle radius
    spacing = 0.15  # Spacing between centers
    padding = 0.05  # Padding from edges

    # Generate center positions for a hexagonal packing inside the square [0,1]^2
    # This is an adaptive hexagonal packing centered in the square
    # This implements a hexagonal lattice with some adjustment to the corners

    # First, compute the number of rows and columns
    # Hexagonal grid: rows are different between even and odd columns
    # For simplicity, create a modified hexagonal grid

    # Create a hexagonal grid
    centers = []

        for col in range(14):  # 14 columns for wide packing
            x = col * spacing + padding
            y = row * spacing * math.sqrt(3) / 2 + padding
            if row % 2 == 1:
                x += spacing / 2
            # Ensure position is within bounds
            if x < 0 or y < 0 or x > 1 or y > 1:
                continue
            centers.append([x, y])

    centers = np.array(centers)[:26]

    while len(centers) < 26:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
