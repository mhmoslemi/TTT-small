import scipy.optimize as optimize

def run_packing():

    radius = 0.05

    for i in range(26):
        row = i // 5
        col = i % 5
        x = col * (2 * radius) + radius
        y = row * (np.sqrt(3) * radius) + radius
        centers.append([x, y])

    radii = np.full(26, radius)

    def objective(x):
        return -np.sum(x)  # Minimize -sum to maximize sum

    # Bounds for each circle's center coordinates and radius
    bounds = []

    def square_boundary(x):
