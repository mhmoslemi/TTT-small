from scipy.optimize import minimize
from scipy.optimize import Bounds

    num_circles = 26

    # With some padding for boundary constraints

    spacing_y = square_size / (6 + 1)
    vertical_spacing = spacing_y * np.sqrt(3) / 2  # For hexagonal grid

                x = col * spacing_x + spacing_x / 2 + spacing_x / 2
            # Ensure no placement out of bounds

            y = np.clip(y, 0.0, 1.0)
            centers[idx] = [x, y]
            idx += 1

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of total radius
