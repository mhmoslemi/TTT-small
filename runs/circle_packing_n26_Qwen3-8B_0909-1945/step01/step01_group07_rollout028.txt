import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26
    r_initial = 0.06  # Initial radius that fits roughly
    centers_initial = np.zeros((n, 2))
    
    # Start with a hexagonal grid pattern
    positions = []
    for i in range(n):
        row = i // 4
        col = i % 4
        x = col * (1.0 + 0.5) / 4.0  # Adjust this to fit better
        y = row * np.sqrt(3) / 4.0
        positions.append([x, y])
    
    centers_initial = np.array(positions)
    radii_initial = np.full(n, r_initial)

    # Define the objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints
    def constraint_center(x, y):
        return np.array([x - 1e-12, 1 - x - 1e-12, y - 1e-12, 1 - y - 1e-12])

    def constraint_radius(r):
        return r - 1e-12

    def constraint_dist(c1, r1, c2, r2):
        return np.sqrt(np.sum((c1 - c2) ** 2)) - r1 - r2 - 1e-12

    # Define the bounds
    bounds = [(0.0, 1.0) for _ in range(2 * n)]  # x and y for each circle
    bounds += [(0.0, 1.0) for _ in range(n)]     # radii for each circle

    # Initial guess: centers and radii

                 [centers_initial[i][1] for i in range(n)] +
                 [radii_initial[i] for i in range(n)])

    # Constraints for all circles
    cons = []
    for i in range(n):
        # Boundary constraints
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_center(x[2*i], x[2*i+1])})
        # Radius constraint
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_radius(x[2*n + i])})

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlapping constraint
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_dist(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
