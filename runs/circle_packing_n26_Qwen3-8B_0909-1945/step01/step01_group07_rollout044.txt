import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define grid of centers for initial circle positions
    grid_size = 5
    grid_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            grid_centers.append([x, y])
    
    # Convert grid centers to a numpy array
    centers = np.array(grid_centers)
    radii = np.full(len(centers), 1.0 / (grid_size - 1))  # Initial radius assuming spacing

    # Objective: Maximize the sum of radii
    # Variables
    x = cp.Variable(2 * len(radii))
    r = cp.Variable(len(radii))

    # Constraints
    # Boundary constraint
    for i in range(len(radii)):
        x_i, y_i = x[2*i], x[2*i+1]
        r_i = r[i]
        constraints = [
            x_i - r_i >= 0,
            x_i + r_i <= 1,
            y_i - r_i >= 0,
            y_i + r_i <= 1
        ]
        constraints.extend([
            x_i - r_i >= -1e-12,
            x_i + r_i <= 1 + 1e-12,
            y_i - r_i >= -1e-12,
            y_i + r_i <= 1 + 1e-12
        ])

    # Non-overlapping constraint
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            dx = x[2*i] - x[2*j]
            dy = x[2*i+1] - x[2*j+1]
            dist = cp.sqrt(dx*dx + dy*dy)
            constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Objective function
    objective = cp.Maximize(cp.sum(r))

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    # Extract final centers and radii
    final_centers = np.zeros((len(radii), 2))
    for i in range(len(radii)):
        final_centers[i, 0] = x[2*i].value
        final_centers[i, 1] = x[2*i+1].value
    final_radii = r.value

    # Calculate total sum of radii
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
