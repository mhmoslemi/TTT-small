import numpy as np
from scipy.optimize import minimize
import math

def initial_hex_grid(n):
    """
    Generate an initial hexagonal grid of n circles in a unit square.
    """
    if n == 0:
        return np.array([]), np.array([])
    
    min_radius = 0.01
    max_radius = 0.15
    radius = (max_radius + min_radius) / 2  # Initial radius
    row_count = int(np.sqrt(n)) + 1
    col_count = int(np.sqrt(n)) + 1

    centers = []
    for i in range(row_count):
        for j in range(col_count):
            if i * col_count + j < n:
                x = j + 0.5 * (i % 2)
                y = i * np.sqrt(3) / 2
                # Adjust to fit in [0, 1] x [0, 1]
                x = x / 10.0 + 0.05
                y = y / 10.0 + 0.05
                centers.append([x, y])
    
    centers = np.array(centers)
    # Adjust radius to ensure all circles fit
    radius = min(0.05, 1.0 / (1 + np.sqrt(3)) * (1 - 0.1))  # Safety margin
    radii = np.full(n, radius)
    return centers, radii

def objective_function(params, n, centers, radii, n_circles):
    """
    Objective function to maximize the sum of radii.
    """
    if n_circles == 0:
        return 0.0

    param_shape = params.shape
    if param_shape[0] != n_circles * 2:
        raise ValueError("Parameter dimensions do not match expected number of circles")

    # Unpack params into centers and radii
    new_centers = params[:n_circles * 2].reshape((n_circles, 2))
    new_radii = params[n_circles * 2:].reshape((n_circles,))

    # We'll search for the maximum sum of radii, so the objective is the negative of the sum
    # to convert it to a minimization problem
    sum_radii = -np.sum(new_radii)

    # Apply penalties for violating constraints
    # (1) Boundary violation
    boundary_penalty = 0.0
    for i in range(n_circles):
        x, y = new_centers[i]
        r = new_radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            boundary_penalty += 1e5 * (max(0, x - r - 0.01) + max(0, x + r - 1.01) + max(0, y - r - 0.01) + max(0, y + r - 1.01))

    # (2) Overlap between circles
    overlap_penalty = 0.0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = new_centers[i, 0] - new_centers[j, 0]
            dy = new_centers[i, 1] - new_centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < new_radii[i] + new_radii[j] - 1e-12:
                overlap_penalty += 1e5 * (new_radii[i] + new_radii[j] - dist)

    # Total objective is negative sum + penalties
    return sum_radii + boundary_penalty + overlap_penalty

def run_packing():

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
