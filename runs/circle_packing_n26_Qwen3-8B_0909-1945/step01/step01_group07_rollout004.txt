import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial guess: lay out the circles in a hexagonal pattern within the square
    # We'll assume an approximate hexagonal grid with 5 rows
    row_spacing = 0.5  # vertical spacing between rows
    col_spacing = 0.5  # horizontal spacing between adjacent circles in a row

    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_spacing + 0.5 * (col + 1) * col_spacing
        y = row * row_spacing + 0.5 * (row + 1) * row_spacing
        centers[i] = [x, y]

    # Optimization variables
    radii = cp.Variable(n, nonneg=True)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: pack circles in unit square
    # 1. Each circle lies within [0,1]x[0,1]
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= 0)
        constraints.append(x + r <= 1)
        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)

    # 2. No overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1 = radii[i]
            r2 = radii[j]
            dist = cp.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            constraints.append(dist >= r1 + r2 - 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)

        prob.solve(solver=cp.ECOS, eps=1e-8)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
