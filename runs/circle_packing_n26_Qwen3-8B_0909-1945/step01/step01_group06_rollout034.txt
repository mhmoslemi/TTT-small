import numpy as np
import cvxpy as cp
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Step 1: Generate an initial hexagonal grid
    num_circles = 26
    radius_initial = 0.05  # Initial guess for radius
    centers = np.zeros((num_circles, 2))
    radii = np.full(num_circles, radius_initial)

    # Step 2: Generate a hexagonal grid layout
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * (2 * radius_initial)
            else:
                x = (j + 0.5) * (2 * radius_initial)
            y = i * (np.sqrt(3) * radius_initial)
            idx = i * cols + j
            if idx < num_circles:
                centers[idx] = [x, y]
    
    # Adjust positions to fit within the unit square
    margin = 0.05
    centers[:, 0] = (centers[:, 0] - np.min(centers[:, 0])) / (np.max(centers[:, 0]) - np.min(centers[:, 0])) * (1 - 2 * margin) + margin
    centers[:, 1] = (centers[:, 1] - np.min(centers[:, 1])) / (np.max(centers[:, 1]) - np.min(centers[:, 1])) * (1 - 2 * margin) + margin
    
    # Step 3: Optimization: Maximize sum of radii with constraints
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles)
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints:
    # 1. All circles are within the square [0,1]x[0,1]
    constraints = [
        cp.SOC(1 - r[i], cp.norm(x[i] - np.array([0.5, 0.5])) - 0.5 - r[i]) for i in range(num_circles)

    for i in range(num_circles):

            dist = cp.norm(x[i] - x[j])

    # 3. Initialize variables with initial values
    x.value = centers
    r.value = radii
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-6)

        return np.zeros((num_circles, 2)), np.zeros(num_circles), 0.0

    # Extract the optimal solution
    centers_opt = x.value

                return np.zeros((num_circles, 2)), np.zeros(num_circles), 0.0

    return centers_opt, radii_opt, sum_radii
