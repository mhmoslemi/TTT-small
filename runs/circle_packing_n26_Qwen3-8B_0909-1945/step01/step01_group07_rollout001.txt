import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based packing
    num_circles = 26
    grid_size = int(np.sqrt(num_circles))
    grid_spacing = 1.0 / grid_size
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Grid-based initial positions
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        x = col * grid_spacing + 0.5 * grid_spacing
        y = row * grid_spacing + 0.5 * grid_spacing
        centers[i] = [x, y]
        radii[i] = grid_spacing

    # Objective function to maximize sum of radii
    def objective(params):
        # params = [r0, r1, ..., r25]
        r = params
        sum_radii = np.sum(r)
        return -sum_radii  # Minimize negative sum to maximize

    # Constraints: boundary and non-overlapping
    def constraint_boundary(params):
        r = params
        for i in range(num_circles):
            x, y = centers[i]
            ri = r[i]
            if x - ri < -1e-8 or x + ri > 1 + 1e-8 or y - ri < -1e-8 or y + ri > 1 + 1e-8:
                return -1
        return 0

    def constraint_overlap(params):
        r = params
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess
    initial_guess = radii.copy()

    # Optimization
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    result = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1.0) for _ in range(num_circles)]
    )

    # Extract final centers and radii
    final_centers = centers.copy()
    final_radii = result.x
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
