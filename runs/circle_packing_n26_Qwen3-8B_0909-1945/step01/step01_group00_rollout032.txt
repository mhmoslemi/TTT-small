import scipy.optimize as optimize

    row_height = 1.0 / (n ** 0.5)  # Suggest a reasonable row height

    offset = 0.1  # Allow for some flexibility in positioning

    num_rows = 5
    num_cols = 5

            idx = i * num_cols + j
            if idx >= n:
                break

            if i % 2 == 0:

                x = j * col_width + offset
            else:
                y = i * row_height + offset

            radii[idx] = 0.15  # Larger initial radius for better optimization

    def objective(x):
