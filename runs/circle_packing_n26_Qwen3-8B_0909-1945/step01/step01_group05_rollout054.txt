import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Number of circles
    n = 26
    
    # Initial packing: simple grid-based with spacing
    # Arrange in 5 rows and 5 columns with some padding
    # Use a hexagonal-like grid with offset rows
    # Each circle has a small radius initially
    centers = []
    radii = []
    
    # Initial radius (will be optimized)
    initial_radius = 0.05
    
    # Number of rows and columns
    rows = 5
    cols = 5
    
    # Initial spacing (will be optimized)
    h = 0.25
    w = 0.25
    
    # Start with a grid of centers
    for i in range(rows):
        for j in range(cols):
            x = j * w + (i % 2) * (w / 2)
            y = i * h
            centers.append([x, y])
            radii.append(initial_radius)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization: maximize the sum of radii with constraints
    # Define objective function
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Define constraints
    def constraint(x):
        # x is radii vector
        # Ensure radii are non-negative
        if np.any(x < 0):
            return np.inf  # Penalize negative radii
        # Enforce boundary constraints
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or \
               y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf  # Penalize boundary violation
        # Enforce non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < x[i] + x[j] - 1e-12:
                    return np.inf  # Penalize overlapping circles
        return 0
    
    # Setup optimization
    # We use the L-BFGS-B method for bounded optimization
    bounds = [(0, 1.0) for _ in range(n)]
    
    # Initial guess
    initial_guess = np.ones(n) * initial_radius
    
    # Perform optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint},
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    # Extract the solution
    optimal_radii = result.x
    optimal_centers = centers
    
    # Validate the packing
    valid, msg = validate_packing(optimal_centers, optimal_radii)
    
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")
    
    # Return the result
    return optimal_centers, optimal_radii, np.sum(optimal_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
