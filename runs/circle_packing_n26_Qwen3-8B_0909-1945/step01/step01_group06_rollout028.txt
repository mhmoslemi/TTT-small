import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    rows, cols = 5, 5
    grid_spacing = 0.25
    
    # Create a grid of 5x5 circles
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(rows):
        for j in range(cols):
            x = j * grid_spacing + 0.05
            y = i * grid_spacing + 0.05
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.15

    centers[-1] = [0.5, 0.5]
    radii[-1] = 0.15

    def objective(radii_params, centers):
        return -np.sum(radii_params)  # Maximize sum of radii by minimizing negative sum

    # Define constraints
    def constraint_center_inside(radii_params, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii_params[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1000000
        return 0

    def constraint_no_overlap(radii_params, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_params[i] + radii_params[j] - 1e-8:
                    return -1000000
        return 0

    bounds = [(0.0, 0.5) for _ in range(num_circles)]  # Radii can't exceed half the square size

        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_center_inside, 'args': (centers,)},

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
