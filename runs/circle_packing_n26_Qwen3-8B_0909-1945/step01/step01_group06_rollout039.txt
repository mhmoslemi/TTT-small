import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    for i in range(n):

        r = radii[i]
        constraints.append(x - r >= -1e-6)
        constraints.append(x + r <= 1 + 1e-6)
        constraints.append(y - r >= -1e-6)
        constraints.append(y + r <= 1 + 1e-6)

    # Circles must not overlap

            dx = centers[i][0] - centers[j][0]

            dist = cp.sqrt(dx ** 2 + dy ** 2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-6)

    # Non-negativity constraints
    constraints.append(cp.Minimize(0))
    for i in range(n):
        constraints.append(radii[i] >= 0)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
