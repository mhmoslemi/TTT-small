import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: Hexagonal packing with 5 rows (each row has 5, 6, 5, 6, 5 circles)
    # Let's assume initial radii to be equal and then optimize

    # Initial layout parameters
    initial_radius = 0.1
    rows = 5
    cols_in_row = [5, 6, 5, 6, 5]
    spacing = 2 * initial_radius  # diameter of initial guess
    num_circles = sum(cols_in_row)

    # Create initial centers in a hexagonal packing
    centers = np.zeros((num_circles, 2))
    col_idx = 0
    for row in range(rows):
        for col in range(cols_in_row[row]):
            x = col * spacing + (spacing / 2) * (row % 2)
            y = row * spacing * np.sqrt(3) / 2
            centers[col_idx] = [x, y]
            col_idx += 1

    # Adjust centers to fit into [0, 1]×[0, 1]
    centers /= np.max(centers)
    centers -= np.min(centers)
    centers /= np.max(centers)

    # Define variables: radii and centers
    variables = np.concatenate((centers.reshape(-1), np.full(num_circles, initial_radius)))

    def objective_function(x):
        # Unpack variables
        centers = x[:2 * num_circles].reshape(-1, 2)
        radii = x[2 * num_circles:]

        # Objective: maximize sum of radii => minimize negative sum
        return -np.sum(radii)

    def constraints(x):
        # Unpack variables
        centers = x[:2 * num_circles].reshape(-1, 2)
        radii = x[2 * num_circles:]

        # Constraint 1: All circles must be inside the square
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf, "Boundary constraint violated"

        # Constraint 2: All circles must not overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf, "Overlap constraint violated"

        # Constraint 3: Radii must be non-negative
        for r in radii:
            if r < 0:
                return np.inf, "Negative radius"

        return 0, "All constraints satisfied"

    # Run optimization
    result = opt.minimize(
        objective_function,
        variables,
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraints}],
        tol=1e-6
    )

    if result.success:
        final_centers = result.x[:2 * num_circles].reshape(-1, 2)
        final_radii = result.x[2 * num_circles:]
        sum_radii = np.sum(final_radii)
        return (final_centers, final_radii, sum_radii)
    else:
        raise RuntimeError(f"Optimization failed: {result.message}")
