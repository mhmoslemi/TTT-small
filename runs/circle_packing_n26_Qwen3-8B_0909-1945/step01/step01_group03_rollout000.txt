def run_packing():

    # Initial hexagonal grid with fixed spacing
    spacing = 0.3  # A smaller spacing allows for more compact arrangement
    vertical_spacing = np.sqrt(3) / 2 * spacing

    # Create a grid that starts from the bottom left
    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(4):
        for col in range(6):
            # Alternate row offset for hexagonal packing
            if row % 2 == 0:
                x = col * spacing + spacing / 2
            else:
                x = col * spacing + spacing / 2 + spacing / 2
            y = row * vertical_spacing + spacing / 2
            if x + spacing / 2 <= 1 and y + spacing / 2 <= 1:
                centers[idx] = [x, y]
                idx += 1

    # Trim centers if we go out of bounds
    centers = centers[:num_circles]

    # Initialize radii by placing circles touching the closest side
    radii = np.zeros(num_circles)
    for i in range(num_circles):
        min_dist_to_boundary = min(centers[i][0], 1 - centers[i][0],
                                   centers[i][1], 1 - centers[i][1])
        radii[i] = min_dist_to_boundary / 2

    # Define objective function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    # Define constraints:
    # 1. Circles must be inside the unit square
    # 2. Circles must not overlap with each other
    # 3. Radii must be non-negative

    # Lower bound for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    def constraint_boundary(radii, centers):
        return np.array([min(centers[i][0], 1 - centers[i][0],
                            centers[i][1], 1 - centers[i][1]) - 2 * radii[i] for i in range(num_circles)])

                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraints.append(dist - radii[i] - radii[j])
        return np.array(constraints)

    # Optimization with constraints
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        bounds=bounds,
        constraints=[

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
