import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial grid parameters
    num_circles = 26
    grid_size = 5  # number of rows/columns
    spacing = 0.2  # initial spacing between circle centers
    shift = spacing * np.sqrt(3) / 2  # vertical offset for hex packing

    # Generate initial grid positions in a hexagonal lattice
    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(grid_size):
        for col in range(grid_size):
            x = col * spacing
            y = row * spacing + (col % 2) * shift
            if idx < num_circles:
                centers[idx] = [x, y]
                idx += 1

    # Initial guess for radii: uniform
    radii = np.ones(num_circles) * 0.05

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: all circles within the unit square
    def bounds_constraints(radii):
        # Bounding box constraints
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return np.inf
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return np.inf
        return 0

    # Non-overlapping constraint: distance between centers >= sum of radii
    def non_overlap_constraints(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    # Combine all constraints
    def constraints(radii):
        return np.array([bounds_constraints(radii), non_overlap_constraints(radii)])

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=[(0, 1) for _ in range(num_circles)],
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract final solution
    final_radii = result.x
    final_centers = centers
    final_sum = np.sum(final_radii)

    return final_centers, final_radii, final_sum
