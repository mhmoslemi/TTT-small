import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different methods
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with random perturbations
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a guided local search with adaptive perturbation strength
    def guided_local_search(h_current, perturbation_strength=0.1, max_iter=50):
        h = np.copy(h_current)
        for _ in range(max_iter):
            h += np.random.normal(0, perturbation_strength, size=len(h))
            h = np.clip(h, 0, 1)
            h = optimize_h(h)[0]
        return h
    
    # Function to perform a global search with multiple random starts
    def global_search(starts=10, max_iter=50):
        best_h = initial_h
        best_c5 = compute_c5(best_h)
        for _ in range(starts):
            h = np.random.uniform(0, 1, n_points)
            h = optimize_h(h)[0]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform a global search
    best_h, best_c5 = global_search(starts=10)
    
    # Perform a guided local search
    best_h = guided_local_search(best_h, perturbation_strength=0.05, max_iter=50)
    best_c5 = compute_c5(best_h)

    for _ in range(10):
        perturbed_h = np.copy(best_h)

        perturbed_h, _ = optimize_h(perturbed_h)
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Final check to ensure we stay within budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
