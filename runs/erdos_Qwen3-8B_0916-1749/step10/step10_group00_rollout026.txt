import time
from scipy.optimize import Bounds
from scipy.optimize import minimize

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization objective with constraints
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate new starting points
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate perturbed solutions
    def perturb_solution(h_current, strength=0.05):
        h = np.copy(h_current)
        h += np.random.normal(0, strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(100):
        # Generate new starting points
        for _ in range(5):
            new_start = generate_new_start(best_h)
            optimized_h, current_c5 = optimize_h(new_start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Perturb current best solution
        for _ in range(5):
            perturbed_h = perturb_solution(best_h, strength=0.05)
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
