import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Objective function for optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with structured perturbations
    def generate_new_start(h_current):
        h = np.copy(h_current)
        perturbation = np.random.normal(0, 0.05, size=len(h))
        h += perturbation
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 100
    exploration_rate = 0.9
    learning_rate = 0.05
    
    for iter in range(max_iterations):
        # Generate new start points with structured perturbations
        new_starts = [generate_new_start(best_h) for _ in range(5)]
        
        # Evaluate all new starts
        for start in new_starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Adaptive perturbation with learning
        if iter % 10 == 0:
            perturbation_strength = 0.1 / (1 + iter)
            perturbed_h = np.copy(best_h)
            perturbed_h += np.random.normal(0, perturbation_strength, size=len(perturbed_h))
            perturbed_h = np.clip(perturbed_h, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                pass
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
