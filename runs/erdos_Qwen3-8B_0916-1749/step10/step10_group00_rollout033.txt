    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a custom optimization method with guided perturbations and adaptive search
    def optimize_with_guided_perturbations(h_initial, max_iter=100, perturbation_strength=0.1, seed=42):
        h = np.copy(h_initial)
        np.random.seed(seed)
        best_h = h
        best_c5 = compute_c5(h)

            # Generate several perturbations
            perturbations = []

                perturbed = np.copy(h)
                rand_idx = np.random.choice(n_points, size=10, replace=False)
                perturbed[rand_idx] += np.random.normal(0, perturbation_strength, size=10)
                perturbed = np.clip(perturbed, 0, 1)
                perturbations.append(perturbed)
            
            # Evaluate each perturbation
            for pert in perturbations:
                try:
                    result = optimize.minimize(
                        fun=objective,
                        x0=pert,
                        bounds=[(0, 1) for _ in range(n_points)],
                        constraints=[{'type': 'eq', 'fun': constraint}],
                        method='SLSQP',
                        options={'ftol': 1e-10, 'maxiter': 50, 'eps': 1e-8}
                    )
                    if result.success:
                        current_c5 = result.fun
                        if current_c5 < best_c5:

                except:
                    continue
            h = best_h

    for i in range(5):
