import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a grid search over perturbation strengths
    def grid_search_perturb(h_current, strengths):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        for strength in strengths:
            perturbed_h = structured_perturb(h_current, perturbation_strength=strength)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        return best_h, best_c5
    
    # Function to perform a grid search over different starting points
    def grid_search_starts(h_current, num_starts):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        for _ in range(num_starts):
            start = generate_new_start(h_current)
            try:
                optimized_start, current_c5_start = optimize_h(start)
                if current_c5_start < best_c5:
                    best_c5 = current_c5_start
                    best_h = optimized_start
            except:
                continue
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(10):
        # Perform a grid search over perturbation strengths
        strengths = np.linspace(0.01, 0.2, 10)
        best_h, best_c5 = grid_search_perturb(best_h, strengths)
        
        # Perform a grid search over different starting points
        best_h, best_c5 = grid_search_starts(best_h, 5)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
