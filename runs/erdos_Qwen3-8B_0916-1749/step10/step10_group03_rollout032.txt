import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Access the initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem
    def optimize_with_bounds(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with controlled perturbation
    def generate_new_start(h_current, perturbation_strength=0.1):
        h = np.copy(h_current)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with diverse exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(50):
        # Generate starting points with varied perturbations
        starts = [generate_new_start(best_h, perturbation_strength=0.1 / (1 + i)) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_with_bounds(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Try further perturbations with smaller strength
        for j in range(5):
            perturbation_strength = 0.05 / (1 + i)
            perturbed_h = np.copy(best_h)
            perturbed_h += np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_with_bounds(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check for time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
