import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a stochastic search with adaptive step sizes
    def stochastic_search(h_initial, max_iterations=100, step_size=0.1):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        for _ in range(max_iterations):
            perturbation = np.random.normal(0, step_size, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
            step_size *= 0.995
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(10):

        optimized_h, current_c5 = stochastic_search(start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h

    perturbed_h = structured_perturb(best_h, perturbation_strength=0.05)
    optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
    if current_c5_perturbed < best_c5:
        best_c5 = current_c5_perturbed
        best_h = optimized_perturbed
