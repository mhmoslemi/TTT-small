import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to evolve a solution with guided perturbations
    def evolve_solution(h_current, seed):
        np.random.seed(seed)
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to compute the best solution with more exploration
    def explore(h_initial, max_iter=100, perturbation_strength=0.1):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        for _ in range(max_iter):
            # Perturb the current solution
            perturbed = structured_perturb(h, perturbation_strength)
            optimized, current_c5 = optimize_h(perturbed)
            if current_c5 < best_c5:
                best_h = optimized
                best_c5 = current_c5
            # Generate a new starting point
            new_start = generate_new_start(best_h)
            optimized_new, current_c5_new = optimize_h(new_start)
            if current_c5_new < best_c5:
                best_h = optimized_new
                best_c5 = current_c5_new
            # Reduce perturbation strength
            perturbation_strength *= 0.95
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform an exploratory search with multiple iterations
    for _ in range(30):
        # Explore and refine solution
        refined_h, refined_c5 = explore(best_h, max_iter=50, perturbation_strength=0.1)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
