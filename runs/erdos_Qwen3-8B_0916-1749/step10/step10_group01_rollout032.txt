import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with structured perturbations
    def generate_new_start(h_current):
        h = np.copy(h_current)
        perturbation = np.random.normal(0, 0.05, size=len(h))
        h += perturbation
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform guided local search with adaptive perturbations
    def guided_local_search(h_current, iterations=50, perturbation_strength=0.1):
        h = np.copy(h_current)
        for _ in range(iterations):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h))
            h += perturbation
            h = np.clip(h, 0, 1)
            h = optimize_h(h)[0]
        return h
    
    # Function to perform ensemble optimization with multiple starting points
    def ensemble_optimization(starting_points, iterations=30):
        best_h = starting_points[0]
        best_c5 = compute_c5(best_h)
        for i in range(iterations):
            new_starts = [generate_new_start(best_h) for _ in range(5)]
            for start in new_starts:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            best_h = guided_local_search(best_h)
        return best_h, best_c5
    
    # Start optimization
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(100):

        for start in new_starts:

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        best_h = guided_local_search(best_h)
    
    return best_h, best_c5, len(best_h)
