import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define optimization bounds
    h_bounds = [(0, 1) for _ in range(n_points)]
    
    # Define warm start from initial_h
    def generate_initial_points(n_points, seed):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        return h
    
    # Optimization function with adaptive strategy
    def optimize_h(h_initial, max_iter=100, tolerance=1e-8):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tolerance, 'maxiter': max_iter, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Generate multiple initial points
    initial_points = [generate_initial_points(n_points, seed + i) for i in range(10)]
    
    for start in initial_points:
        optimized_h, current_c5 = optimize_h(start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Use active set method for refinement
    final_result = optimize.minimize(
        fun=objective,
        x0=best_h,
        bounds=h_bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
    )
    
    best_h = final_result.x
    best_c5 = final_result.fun
    
    # Check time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
