import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=300):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate guided perturbations based on sensitivity analysis
    def guided_perturbation(h_current, seed):
        np.random.seed(seed)
        h = np.copy(h_current)
        # Compute the sensitivity of each point
        sensitivities = np.zeros_like(h)
        for i in range(n_points):
            h_test = np.copy(h)
            h_test[i] += 0.1
            h_test[i] = np.clip(h_test[i], 0, 1)
            delta = compute_c5(h_test) - compute_c5(h)
            sensitivities[i] = delta
        # Perturb points with high sensitivity
        for i in range(n_points):
            if np.random.rand() < 0.3:
                h[i] += np.random.normal(0, 0.05 * np.abs(sensitivities[i]) / np.max(np.abs(sensitivities)))
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    # Function to generate diverse initial points
    def generate_diverse_starts(h, num_starts=5):
        starts = []
        for i in range(num_starts):
            h_new = np.copy(h)
            for j in range(n_points):
                if np.random.rand() < 0.3:
                    h_new[j] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(20):
        # Generate diverse starting points
        starts = generate_diverse_starts(best_h)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform guided perturbation and optimization
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbed_h = guided_perturbation(best_h, seed_perturb)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
