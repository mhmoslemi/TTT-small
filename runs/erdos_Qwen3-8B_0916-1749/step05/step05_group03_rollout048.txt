import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with tighter tolerances
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to compute the overlap integral for all shifts
    def compute_all_overlaps(h):
        h = np.array(h)
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k == 0:
                continue
            # Compute the overlap for shift k
            overlap = np.sum(h * (1 - np.roll(h, k)))
            overlaps.append(overlap * dx)
        return np.max(overlaps)
    
    # Adaptive optimization strategy with multiple starting points
    best_h = initial_h
    best_c5 = compute_all_overlaps(initial_h)
    best_time = time.time()
    
    start_time = time.time()
    
    for i in range(10):
        # Generate a new starting point
        h_initial = evolve_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
                print(f"Improved bound: {best_c5:.8f} (iteration {i})")
        except:
            continue
        
        # Perform perturbation and optimization
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbed_h = perturb(optimized_h, seed_perturb)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
                    print(f"Improved bound: {best_c5:.8f} (iteration {i}, perturb {j})")
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
