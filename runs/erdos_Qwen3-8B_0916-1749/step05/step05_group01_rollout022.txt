import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the maximum overlap for a given h
    def compute_max_overlap(h):
        h = np.array(h)
        max_overlap = 0
        for k in range(1, n_points):
            if k >= n_points:
                break
            overlap = np.sum(h[:n_points - k] * (1 - h[k:n_points]))
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Define the objective function
    def objective(h):
        return compute_max_overlap(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a more efficient search strategy
    best_h = initial_h
    best_c5 = compute_max_overlap(initial_h)
    start_time = time.time()
    
    # First, perform a few rounds of optimization starting from the initial solution
    for i in range(3):
        h_initial = evolve_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Next, perform a local search around the best solution
    for i in range(5):
        # Generate a new starting point
        h_initial = structured_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Finally, perform a few random restarts
    for i in range(3):
        h_initial = random_start()
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
