import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a local search around a given point
    def local_search(h_current):
        h = np.copy(h_current)
        for _ in range(10):
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            try:
                optimized_h, _ = optimize_h(h_perturbed)
                if compute_c5(optimized_h) < compute_c5(h):
                    h = optimized_h
            except:
                continue
        return h
    
    # Function to generate a new solution using a different starting point
    def generate_new_solution():
        h_initial = random_start()
        optimized_h, _ = optimize_h(h_initial)
        return optimized_h
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Keep track of the elapsed time
    start_time = time.time()
    
    # Perform multiple iterations of optimization and local search
    for i in range(15):
        # Generate a new solution
        try:
            h_initial = generate_new_solution()
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform local search
        try:
            optimized_local = local_search(optimized_h)
            current_c5_local = compute_c5(optimized_local)
            if current_c5_local < best_c5:
                best_c5 = current_c5_local
                best_h = optimized_local
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
