import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with adaptive settings
    def optimize_h(h_initial, max_iter=100, tol=1e-7):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tol, 'maxiter': max_iter, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Perform a guided search with multiple starting points
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    # Initial optimization
    optimized_h, current_c5 = optimize_h(best_h, max_iter=200, tol=1e-8)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h
    
    # Generate and optimize from random starts
    for i in range(10):
        h_initial = random_start()
        optimized_h, current_c5 = optimize_h(h_initial, max_iter=150, tol=1e-8)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Generate and optimize from perturbed best
    for i in range(8):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        optimized_h, current_c5 = optimize_h(perturbed_h, max_iter=150, tol=1e-8)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Evolve from best and optimize
    for i in range(5):
        evolved_h = evolve_start(best_h)
        optimized_h, current_c5 = optimize_h(evolved_h, max_iter=150, tol=1e-8)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
