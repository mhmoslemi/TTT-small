import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a diverse set of starting points
    def generate_starting_points():
        starts = [
            initial_h,
            np.random.uniform(0, 1, n_points),
            np.zeros(n_points),
            np.ones(n_points),
            np.linspace(0, 1, n_points),
            np.sin(np.linspace(0, np.pi, n_points)),
            np.cos(np.linspace(0, np.pi, n_points)),
            np.random.exponential(1, n_points),
            np.random.gamma(2, 1, n_points),
            np.random.binomial(1, 0.5, n_points)
        ]
        return starts
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Start time
    start_time = time.time()
    
    # Generate a diverse set of starting points
    starts = generate_starting_points()
    
    # Perform optimization on each starting point
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Perturb and optimize the best solution multiple times with different seeds
    for i in range(5):
        seed_perturb = seed + i
        np.random.seed(seed_perturb)
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = np.clip(best_h + perturbation, 0, 1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Use a more strategic local search on the best solution
    for _ in range(10):
        # Generate a small random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(best_h + perturbation, 0, 1)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Return the best solution found
    return best_h, best_c5, n_points
