import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a more sophisticated optimization function with multiple strategies
    def optimize_h(h_initial, method='SLSQP', maxiter=100):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-10, 'maxiter': maxiter, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a perturbation function with adaptive parameters
    def perturb(h, strength):
        return np.clip(h + np.random.normal(0, strength, size=n_points), 0, 1)
    
    # Define a function to perform an aggressive local search
    def aggressive_local_search(h, perturbation_strengths):
        best_h = np.copy(h)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            perturbed_h = perturb(best_h, strength)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Define a function to perform a grid search over perturbation strengths
    def grid_search(h, param_grid):
        best_h = np.copy(h)
        best_c5 = compute_c5(best_h)
        for p in param_grid:
            perturbed_h = perturb(best_h, p)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    optimized_h, current_c5 = optimize_h(best_h, method='SLSQP', maxiter=100)
    if current_c5 < best_c5:
        best_h = optimized_h
        best_c5 = current_c5

        best_h = best_h_ags
        best_c5 = best_c5_ags

    param_grid = np.linspace(0.01, 0.1, 5)
    best_h_grid, best_c5_grid = grid_search(best_h, param_grid)

        best_h = best_h_grid
        best_c5 = best_c5_grid

            best_c5 = current_c5
