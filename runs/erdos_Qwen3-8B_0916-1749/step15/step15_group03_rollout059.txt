import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define an optimization method with advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a better starting point via random perturbation
    def generate_perturbed_start(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=n_points), 0, 1)
        return perturbed_h
    
    # Define a more robust local search strategy
    def local_search(h_current, perturbation_strength, iterations):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(iterations):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with advanced strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Start with a more refined local search
    best_h, best_c5 = local_search(best_h, perturbation_strength=0.1, iterations=50)
    
    # Explore alternative starting points through perturbation
    for _ in range(20):
        # Generate a new perturbed start point
        new_start = generate_perturbed_start(best_h, strength=0.2)
        optimized_h, current_c5 = optimize_h(new_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Perform optimization with different strategies
    for _ in range(10):
        # Generate a new perturbed start point
        new_start = generate_perturbed_start(best_h, strength=0.15)
        optimized_h, current_c5 = optimize_h(new_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Final local search with smaller perturbation
    best_h, best_c5 = local_search(best_h, perturbation_strength=0.05, iterations=30)
    
    # Check time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
