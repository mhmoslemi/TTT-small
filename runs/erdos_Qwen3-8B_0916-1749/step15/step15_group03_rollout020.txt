import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize, Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 140
    dx = 2.0 / n_points
    
    # Define initial construction
    initial_h = initial_h_values  # This variable is globally defined
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for the variables
    bounds = Bounds(0, 1)
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Initial optimization with SLSQP
    result = minimize(
        fun=objective,
        x0=best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Perform local search
    best_h, best_c5 = local_search(best_h)
    
    # Adaptive perturbation and optimization
    for i in range(50):
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                result = minimize(
                    fun=objective,
                    x0=start,
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    method='SLSQP',
                    options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
                )
                if result.fun < best_c5:
                    best_h = result.x
                    best_c5 = result.fun
            except:
                continue
        
        # Perform perturbation and optimization with adaptive strength
        perturbation_strength = 0.15 / (1 + i)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        try:
            result = minimize(
                fun=objective,
                x0=perturbed_h,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
            )
            if result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        except:
            continue
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
