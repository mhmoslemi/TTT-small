import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05, num_steps=5):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(num_steps):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate randomized initial solutions
    def generate_initial_solution():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def multi_start_optimize(n_starts=5, perturbation_strength=0.1, num_steps=10):
        best_h = generate_initial_solution()
        best_c5 = compute_c5(best_h)
        for _ in range(n_starts):
            start = generate_initial_solution()
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
            # Perform local search
            best_h, best_c5 = local_search(best_h, perturbation_strength, num_steps)
        return best_h, best_c5
