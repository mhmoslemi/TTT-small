import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300):

    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05, num_steps=5):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(num_steps):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform particle swarm optimization
    def pso_optimize(h_initial, n_particles=20, n_iter=50, w=0.7, c1=1.5, c2=1.5):

        particles = np.random.uniform(0, 1, (n_particles, n_points))
        velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
        best_positions = np.copy(particles)
        best_fitness = np.array([objective(p) for p in particles])
        global_best_position = np.copy(best_positions[np.argmin(best_fitness)])
        global_best_fitness = np.min(best_fitness)
        
        for _ in range(n_iter):
            r1 = np.random.rand(n_particles, n_points)

            global_best_fitness = np.min(best_fitness)

            best_h = optimized_h
