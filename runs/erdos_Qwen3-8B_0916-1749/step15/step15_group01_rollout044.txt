import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300):

    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the maximum k overlap
    def compute_max_k_overlap(h):
        h = np.array(h)
        max_k_overlap = 0
        for k in range(1, int(2 * dx) + 1):
            corr = np.correlate(h, 1 - h, mode='full')
            overlap = np.max(corr)
            if overlap > max_k_overlap:
                max_k_overlap = overlap
        return max_k_overlap * dx
    
    # Define the objective function
    def objective(h):
        return compute_max_k_overlap(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05, num_steps=5):
        best_h = h_current
        best_c5 = compute_max_k_overlap(best_h)
        for _ in range(num_steps):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_max_k_overlap(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform multi-start optimization with adaptive settings
    def multi_start_optimize(h_initial, num_starts=5, perturbation_strength=0.1, num_perturbations=5):
        best_h = h_initial
        best_c5 = compute_max_k_overlap(best_h)
        for _ in range(num_starts):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_initial))
            perturbed_h = np.clip(h_initial + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5

    def adaptive_perturbation_search(h_initial, max_iterations=100, perturbation_decay=0.95):

        best_h = h
        best_c5 = compute_max_k_overlap(h)
        for iteration in range(max_iterations):
            perturbation_strength = 0.1 * (perturbation_decay ** iteration)
            perturbation = np.random.normal(0, perturbation_strength, size=len(h))
            perturbed_h = np.clip(h + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5

    # Perform local search on the best found solution
    best_h, best_c5 = local_search(best_h)
    
    # Check time budget and return result
