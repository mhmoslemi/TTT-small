import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Objective function for optimization
    def objective_with_penalty(h):
        c5 = compute_c5(h)
        penalty = np.max(np.abs(np.sum(h) - n_points / 2))
        return c5 + 1000 * penalty
    
    # Function to perform weighted least squares optimization with advanced solver
    def optimize_h(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective_with_penalty,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform perturbation and optimization with adaptive strength
    def perturb_and_optimize(h_current, strength, method='SLSQP'):
        perturbation = np.random.normal(0, strength, size=n_points)
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h, method=method)
            return optimized_h, current_c5
        except:
            return h_current, compute_c5(h_current)
    
    # Function to perform adaptive local search with multiple perturbations
    def adaptive_local_search(h_current, max_perturbations=5):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        perturbations = np.linspace(0.01, 0.1, max_perturbations)
        for strength in perturbations:
            perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=n_points), 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate random perturbations and optimize
    def random_search(h_current, num_trials=10):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(num_trials):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate a more structured perturbation
    def structured_perturbation(h_current):
        perturbations = np.random.choice([0.05, 0.1, 0.15], size=n_points)
        perturbed_h = np.clip(h_current + perturbations, 0, 1)
        return perturbed_h
    
    # Main optimization loop with adaptive perturbation and structured search
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Random search with multiple trials
    for _ in range(5):
        best_h, best_c5 = random_search(best_h)
    
    # Adaptive local search with multiple perturbations
    for _ in range(5):
        best_h, best_c5 = adaptive_local_search(best_h)
    
    # Optimization with structured perturbation
    best_h = structured_perturbation(best_h)
    best_h, best_c5 = optimize_h(best_h)
    
    # Final optimization with more aggressive strength and multiple perturbations
    best_h, best_c5 = optimize_h(best_h, method='Nelder-Mead')
    
    # Check if time is exhausted
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
