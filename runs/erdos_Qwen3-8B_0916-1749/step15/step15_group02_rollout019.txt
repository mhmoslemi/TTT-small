import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Define a more sophisticated search strategy
    def advanced_search(h_initial, max_iter=100, perturbation_strength=0.15):
        h_current = np.copy(h_initial)
        best_h = np.copy(h_current)
        best_c5 = compute_c5(h_current)
        time_start = time.time()
        
        for iter in range(max_iter):
            # Local search
            h_current, current_c5 = local_search(h_current, perturbation_strength)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current
            
            # Random perturbation
            h_perturbed = random_perturbation(h_current, perturbation_strength / (1 + iter))
            h_opt, current_c5 = optimize_h(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
            
            # Check time budget
            if time.time() - time_start > budget_s:
                break
        
        return best_h, best_c5
    
    # Perform the search
    best_h, best_c5 = advanced_search(initial_h)
