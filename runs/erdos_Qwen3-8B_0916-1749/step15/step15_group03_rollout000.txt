import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Import the initial construction
    initial_h = initial_h_values
    
    # Determine number of points and dx
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define an optimization function with bounds and constraints
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 300, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random perturbations
    def random_perturbation(h, strength):
        perturbed_h = np.clip(h + np.random.normal(0, strength, size=len(h)), 0, 1)
        return perturbed_h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    start_time = time.time()
    
    # Adaptive perturbation and refinement
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Try optimizing from the current best
        try:
            optimized_h, current_c5 = optimize_h(np.array(best_h))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Try random perturbations with adaptive strength
        perturbation_strength = 0.15 / (1 + i)
        for _ in range(5):
            try:
                perturbed_h = random_perturbation(best_h, perturbation_strength)
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Try perturbing the initial construction with small steps
        initial_perturbation = np.random.normal(0, 0.05, size=n_points)
        initial_perturbed_h = np.clip(initial_h + initial_perturbation, 0, 1)
        try:
            optimized_initial, current_c5_initial = optimize_h(initial_perturbed_h)
            if current_c5_initial < best_c5:
                best_c5 = current_c5_initial
                best_h = optimized_initial
        except:
            pass
        
        # Try a smaller grid with optimized values
        small_n_points = max(20, int(n_points * 0.7))
        small_dx = 2.0 / small_n_points

        small_h_values = np.array([0.5 for _ in range(small_n_points)])
        
        # Initialize with the best solution so far
        small_h_values = np.array([best_h[i * n_points // small_n_points] for i in range(small_n_points)])
        small_h_values = np.clip(small_h_values, 0, 1)
        
        # Try optimizing on a smaller grid
        try:
            optimized_small, current_c5_small = optimize_h(small_h_values)
            if current_c5_small < best_c5:
                best_c5 = current_c5_small
                best_h = optimized_small
                best_n_points = small_n_points
        except:
            pass
    
    return best_h, best_c5, best_n_points
