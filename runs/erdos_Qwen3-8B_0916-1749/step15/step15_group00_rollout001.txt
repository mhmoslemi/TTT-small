import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization function with advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=compute_c5,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8, 'disp': False}
        )
        return result.x, result.fun
    
    # Perturbation function for local search
    def perturb(h, strength):
        return np.clip(h + np.random.normal(0, strength, len(h)), 0, 1)
    
    # Function for local search
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbed = perturb(best_h, perturbation_strength)
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function for gradient descent with momentum
    def gradient_descent(h, learning_rate=0.01, momentum=0.9, steps=100):
        h = np.array(h)
        v = np.zeros_like(h)
        best_h = np.copy(h)
        best_c5 = compute_c5(h)
        for _ in range(steps):
            grad = np.zeros_like(h)
            for k in range(1, n_points):
                corr = np.correlate(h, 1 - h, mode='full')
                grad_k = np.gradient(corr, dx)[k]
                grad[k] = -grad_k
            v = momentum * v - learning_rate * grad
            h = np.clip(h + v, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive optimization with local search and momentum
    for iteration in range(100):
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Perform gradient descent
        optimized_h, current_c5 = gradient_descent(best_h, learning_rate=0.01, momentum=0.9, steps=50)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Try perturbation with different strengths
        for strength in [0.01, 0.05, 0.1]:
            perturbed = perturb(best_h, strength)
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
