        )

        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))

            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5

    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)

    def structured_perturbation(h_current, strength):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h

    def multi_start_optimize(initial_h, num_starts=5):
        best_h = initial_h

                optimized_h, current_c5 = optimize_h(np.copy(initial_h))

                    best_h = optimized_h
