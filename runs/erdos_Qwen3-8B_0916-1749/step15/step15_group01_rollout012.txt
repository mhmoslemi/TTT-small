import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Main optimization loop with advanced exploration and adaptive perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define parameters for adaptive perturbation and exploration
    perturbation_strengths = [0.05, 0.02, 0.01, 0.005, 0.002]
    num_perturbations = 5
    num_steps = 100
    num_suggestions = 5
    
    # Main optimization loop
    for step in range(100):
        # Generate multiple perturbations
        perturbations = []
        for _ in range(num_perturbations):
            perturb = np.random.normal(0, perturbation_strengths[step % len(perturbation_strengths)], size=n_points)
            perturbed_h = np.clip(best_h + perturb, 0, 1)
            perturbations.append(perturbed_h)
        
        # Optimize all perturbed vectors
        for perturbed_h in perturbations:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Perform local search with adaptive step sizes
        for _ in range(num_steps):
            perturbation = np.random.normal(0, 0.01, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
