import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization with advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate diverse starting points
    def generate_variants(h_current, num_variants=5):
        variants = []
        for _ in range(num_variants):
            perturbation = np.random.normal(0, 0.15, size=n_points)
            variant = np.clip(h_current + perturbation, 0, 1)
            variants.append(variant)
        return variants
    
    # Function to perform multi-step local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05, max_iterations=10):
        h = np.copy(h_current)
        current_c5 = compute_c5(h)
        for _ in range(max_iterations):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(perturbed_h)
            if new_c5 < current_c5:
                h = perturbed_h
                current_c5 = new_c5
        return h, current_c5
    
    # Main optimization loop with diverse exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform multiple rounds of refinement
    for _ in range(10):
        # Generate diverse variants
        variants = generate_variants(best_h)
        
        # Evaluate variants and optimize
        for variant in variants:
            try:
                optimized_h, current_c5 = optimize_h(variant)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform local search on best_h
        best_h, best_c5 = local_search(best_h)
        
        # Check for time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
