import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a more robust optimization method with advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define a function to generate random perturbations
    def random_perturbation(h, strength):
        perturbed_h = np.clip(h + np.random.normal(0, strength, size=len(h)), 0, 1)
        return perturbed_h
    
    # Adaptive perturbation and local search
    for iteration in range(100):
        # Generate alternative starting points with random perturbations
        for _ in range(5):
            perturbation_strength = 0.15 / (1 + iteration)
