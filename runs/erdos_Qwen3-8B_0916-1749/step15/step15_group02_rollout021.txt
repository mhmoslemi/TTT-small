import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    def compute_c5(h):
        h = np.array(h)

        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]

            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Main optimization loop with adaptive perturbation and exploration
    start_time = time.time()
    best_h = initial_h
    best_c5 = compute_c5(best_h)

    perturbation_strength = 0.15
