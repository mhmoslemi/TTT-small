import scipy.optimize as optimize

from scipy.optimize import minimize

    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform constrained optimization with advanced solver
    def optimize_h(h_initial):
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=n_points), 0, 1)
        return perturbed_h
    
    # Function to generate more structured perturbations (e.g., segment-based)
    def structured_perturbation(h_current, strength):
        perturbed_h = np.copy(h_current)
        for i in range(n_points):
            if np.random.rand() < strength:
                perturbed_h[i] += np.random.choice([-0.1, 0.1])

        return perturbed_h

    best_h = initial_h

    start_time = time.time()

    for i in range(100):

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
