import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Construct initial guess with piecewise constant values
    for i in range(n_points):
        x = i * dx
        if 0.5 <= x < 1.5:
            initial_h[i] = 0.5
        elif 1.5 <= x < 2.0:
            initial_h[i] = 0.75
        elif 0.0 <= x < 0.5:
            initial_h[i] = 0.25
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05, num_steps=5):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(num_steps):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    start_time = time.time()
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Use a combination of gradient-based optimization, adaptive perturbation, and local search
    for i in range(150):
        # Perform optimization from the current best
        try:
            optimized_h, current_c5 = optimize_h(np.copy(best_h))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Generate random perturbations and perform optimization
        for _ in range(5):
            perturbation = np.random.normal(0, 0.1, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
