import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(10):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a more structured search with adaptive parameters
    def structured_search(h_current, strength, n_steps):
        h = np.copy(h_current)
        for _ in range(n_steps):
            perturbation = np.random.normal(0, strength, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            h = optimize_h(h)[0]
        return h, compute_c5(h)
    
    # Main optimization loop with structured search and perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive structured search
    for i in range(20):
        # Perform structured search with decreasing perturbation strength
        current_h = best_h
        for strength in [0.15, 0.1, 0.05, 0.025, 0.01]:
            current_h = structured_search(current_h, strength, 5)[0]
        
        # Perform random perturbation and optimization
        for _ in range(5):
            perturbed_h = random_perturbation(best_h, 0.15 / (1 + i))
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
