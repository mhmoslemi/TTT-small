import numpy as np
import scipy.optimize as optimize
import time
import random

# Global variable for initial h values (n=70 samples)
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float64)

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial h with 70 samples
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a more exploratory search
    def explore_h(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform genetic algorithm search
    def genetic_search(h_initial, generations=20, pop_size=20, mutation_rate=0.1):
        def fitness(h):
            return compute_c5(h)
        
        pop = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        pop = [h for h in pop if np.sum(h) * dx * 2 == 1]
        
        for gen in range(generations):
            fitness_scores = [fitness(h) for h in pop]
            best_idx = np.argmin(fitness_scores)
            best_h = pop[best_idx]
            
            # Select parents
            parents = np.random.choice(pop, size=pop_size, replace=False)
            
            # Crossover
            new_pop = []
            for i in range(0, pop_size, 2):
                if i + 1 < pop_size:
                    h1 = parents[i]
                    h2 = parents[i + 1]
                    crossover_point = np.random.randint(1, n_points - 1)
                    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                    new_pop.extend([child1, child2])
            
            # Mutation
            for h in new_pop:
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Replace population
            pop = new_pop
        
        best_idx = np.argmin([fitness(h) for h in pop])
        return pop[best_idx], fitness(pop[best_idx])
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try multiple starting points
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        start = np.random.uniform(0, 1, n_points)
        start = np.clip(start, 0, 1)
        start = np.array(start)
        if np.sum(start) * dx * 2 != 1:
            continue
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        except:
            continue
    
    # Try local search
    best_h, best_c5 = local_search(best_h)
    
    # Try genetic search
    genetic_h, genetic_c5 = genetic_search(best_h)
    if genetic_c5 < best_c5:
        best_h = genetic_h
        best_c5 = genetic_c5
    
    # Try a more aggressive exploration
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        if np.sum(perturbed_h) * dx * 2 != 1:
            continue
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    return best_h, best_c5, len(best_h)
