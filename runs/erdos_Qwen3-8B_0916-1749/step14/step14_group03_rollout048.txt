import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h_values = [0.5 for _ in range(70)]
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform random restart with adaptive perturbation
    def random_restart_search(h_initial):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        for i in range(50):
            perturbation = np.random.normal(0, 0.1, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            h = optimize_h(h)[0]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        return best_h, best_c5
    
    # Function to perform ensemble optimization with multiple perturbations
    def ensemble_optimization(h_initial):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        for i in range(20):
            perturbations = [np.random.normal(0, 0.1, size=len(h)) for _ in range(5)]
            candidates = [np.clip(h + p, 0, 1) for p in perturbations]
            results = []
            for c in candidates:
                try:
                    optimized, c5 = optimize_h(c)
                    results.append((optimized, c5))
                except:
                    continue
            if results:
                best_result = min(results, key=lambda x: x[1])
                best_h, best_c5 = best_result
        return best_h, best_c5
    
    # Main optimization loop with ensemble optimization, random restarts, and adaptive perturbation
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Initial optimization with ensemble approach
    best_h, best_c5 = ensemble_optimization(best_h)
    
    # Iterative refinement with random restarts
    for i in range(50):
        perturbation = np.random.normal(0, 0.1, size=len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        best_h, best_c5 = optimize_h(perturbed_h)
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
