import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    # Define initial construction with 70 samples

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Function to generate random samples in [0,1] with given length
    def random_initial(n):
        return np.random.rand(n)
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define hyperparameters
    num_iterations = 50
    num_perturbations = 5
    num_random_starts = 3
    perturbation_strength = 0.15
    
    # Adaptive optimization with multiple random starts, perturbations, and local search
    for iteration in range(num_iterations):
        # Generate multiple random starts
        for _ in range(num_random_starts):

            # Enforce integral constraint
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            try:
                optimized_h, current_c5 = optimize_h(h_start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Generate perturbations around best_h
        for _ in range(num_perturbations):
