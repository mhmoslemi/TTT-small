import numpy as np
import scipy.optimize as optimize
import time

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a global search with adaptive perturbation and population-based optimization
    def population_search(h_population, n_iter=10):
        best_h = h_population[0]
        best_c5 = compute_c5(best_h)
        for _ in range(n_iter):
            new_population = []
            for h in h_population:
                # Generate perturbed versions with different strengths
                strengths = np.random.uniform(0.01, 0.1, size=len(h))
                perturbed_h = np.clip(h + np.random.normal(0, strengths, size=len(h)), 0, 1)
                new_population.append(perturbed_h)
            # Evaluate all new population members
            for h in new_population:
                try:
                    h_evaluated, c5_val = optimize_h(h)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_evaluated
                except:
                    continue
            h_population = new_population
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    population = [np.random.uniform(0, 1, n_points) for _ in range(10)]

    for _ in range(50):
        # Add the best solution to the population
        population.append(best_h)
        # Keep only the top 20 members
        population = sorted(population, key=lambda h: compute_c5(h))[:20]
