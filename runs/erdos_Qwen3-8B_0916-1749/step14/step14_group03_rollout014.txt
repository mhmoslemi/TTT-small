import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with a more structured approach
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate perturbations with adaptive strength
    def generate_perturbations(h_current, n_perturbations=5):
        perturbations = []
        for _ in range(n_perturbations):
            strength = 0.1 / (1 + np.random.rand())
            perturb = np.random.normal(0, strength, size=len(h_current))
            perturbed = np.clip(h_current + perturb, 0, 1)
            perturbations.append(perturbed)
        return perturbations
    
    # Function to perform a simple local search
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and multi-start approach
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive perturbation and multi-start optimization
    for i in range(100):
        # Generate multiple perturbations and optimize
        perturbations = generate_perturbations(best_h)
        for pert in perturbations:
            optimized_h, current_c5 = optimize_h(pert)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
