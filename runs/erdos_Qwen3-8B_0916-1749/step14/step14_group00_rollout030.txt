import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Function to generate perturbations based on gradient direction
    def gradient_perturbation(h_current, strength):
        h = np.array(h_current)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        shift = max_corr_index - len(h) + 1
        if shift < 0:
            shift = 0
        elif shift >= len(h):
            shift = len(h) - 1
        grad = np.zeros_like(h)
        for i in range(len(h)):
            shift_i = i - shift
            if 0 <= shift_i < len(h):
                grad[i] = np.correlate(h, 1 - h, mode='full')[shift_i] / (np.max(np.correlate(h, 1 - h, mode='full')) + 1e-10)
        perturbed_h = np.clip(h_current + strength * grad, 0, 1)
        return perturbed_h
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive perturbation and optimization
    for i in range(100):
        # Generate alternative starting points with different perturbations
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform perturbation and optimization with adaptive strength
        for j in range(5):
            perturbation_strength = 0.1 / (1 + i)
            perturbed_h = random_perturbation(best_h, perturbation_strength)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Perform perturbation based on gradient direction
        perturbation_strength = 0.1 / (1 + i)
        perturbed_h = gradient_perturbation(best_h, perturbation_strength)
        try:
            optimized_gradient, current_c5_gradient = optimize_h(perturbed_h)
            if current_c5_gradient < best_c5:
                best_c5 = current_c5_gradient
                best_h = optimized_gradient
        except:
            continue
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
