import numpy as np
import scipy.optimize as optimize
import time
import sys

def run(seed=42, budget_s=300):

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def perturb(h, strength=0.05):
        perturbation = np.random.normal(0, strength, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate a random solution
    def random_solution():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h - np.mean(h) / np.std(h) * 0.5, 0, 1)
        return h
    
    # Function to perform a grid search over different parameter configurations
    def grid_search(h_current, params):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for p in params:
            perturbed = perturb(h_current, p)
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a local search with decreasing perturbation strength
    def local_search(h_current, max_perturb=0.1, steps=10):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for step in range(steps):
            perturb_strength = max_perturb / (1 + step)
            perturbed = perturb(h_current, perturb_strength)
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
        return best_h, best_c5

                if current_c5 < best_c5:

                    best_h = optimized_h
            except:

                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed

        best_h, best_c5 = local_search(best_h)

            best_h = random_h
    
    return best_h, best_c5, n_points
