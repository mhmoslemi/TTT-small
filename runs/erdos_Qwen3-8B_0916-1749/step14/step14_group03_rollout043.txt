import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform gradient-based optimization
    def gradient_based_optimization(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        return result.x, result.fun
    
    # Define a new initial construction with 100 samples
    def construct_new_initial_h(n_points):
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            if 0.5 <= x <= 1.5:
                h[i] = 0.5
            elif 0.2 <= x <= 0.8:
                h[i] = 0.75
            else:
                h[i] = 0.25
        return h
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = construct_new_initial_h(n_points)
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(50):
        # Generate alternative starting points using different perturbations
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform gradient-based optimization
        try:
            optimized_h, current_c5 = gradient_based_optimization(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
