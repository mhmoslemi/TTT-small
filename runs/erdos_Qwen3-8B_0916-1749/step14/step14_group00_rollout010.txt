import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Objective function to minimize
    def objective(h):
        return compute_c5(h)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations with adaptive strength
    def perturb(h, strength):
        return np.clip(h + np.random.normal(0, strength, size=len(h)), 0, 1)
    
    # Main optimization loop with adaptive perturbation and optimization
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    n_iterations = 100
    perturbation_strength = 0.15
    
    for i in range(n_iterations):
        # Generate perturbations with adaptive strength
        for _ in range(5):
            perturbed_h = perturb(best_h, perturbation_strength / (1 + i))
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
