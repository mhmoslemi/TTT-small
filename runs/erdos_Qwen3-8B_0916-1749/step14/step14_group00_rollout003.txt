import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 140  # Double the initial samples for better resolution

    # Resample the initial_h to n_points
    def resample_h(initial_h, n_points):
        initial_h_resampled = np.zeros(n_points)
        for i in range(len(initial_h)):
            x = initial_h[i]
            idx = int(x * n_points / 2)
            if idx >= n_points:
                idx = n_points - 1
            initial_h_resampled[idx] += 1 / n_points
        return initial_h_resampled
    
    h_initial = resample_h(initial_h, n_points)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Objective function to minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbations
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(10):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate random perturbations
    def random_perturbation(h_current, strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=len(h_current)), 0, 1)
        return perturbed_h
    
    # Main optimization loop with adaptive perturbations and exploration
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    start_time = time.time()

        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
