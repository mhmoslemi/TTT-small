import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import BoundsHandler

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 140
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform local search with adaptive perturbation
    def adaptive_local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for i in range(10):
            perturbation_strength = 0.05 / (1 + i)
            perturbed_h = np.clip(h_current + np.random.normal(0, perturbation_strength, size=len(h_current)), 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate a random starting point
    def random_start():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Adjust to satisfy the constraint
        return h
    
    # Main optimization loop with multiple starting points and advanced features
    best_h = np.random.uniform(0, 1, n_points)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Set of starting points
    starting_points = [best_h]
    for i in range(5):
        starting_points.append(random_start())
    
    # Run optimization with multiple starting points
    for h_start in starting_points:
        optimized_h, current_c5 = optimize_h(h_start)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
    
    # Additional local search
    best_h, best_c5 = adaptive_local_search(best_h)
    
    # Time-based perturbation and optimization
    for i in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        perturbation_strength = 0.1 / (1 + i)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed
    
    # Final local search
    best_h, best_c5 = adaptive_local_search(best_h)
    
    return best_h, best_c5, n_points
