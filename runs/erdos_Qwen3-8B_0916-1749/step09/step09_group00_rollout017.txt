import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbations and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define a set of perturbation strategies
    def perturb_1(h, strength):
        return np.clip(h + np.random.normal(0, strength, len(h)), 0, 1)
    
    def perturb_2(h, strength):
        return np.clip(h + np.random.uniform(-strength, strength, len(h)), 0, 1)
    
    def perturb_3(h, strength):
        return np.clip(h * (1 - np.random.exponential(strength, len(h))), 0, 1)
    
    def perturb_4(h, strength):
        return np.clip(h + np.random.laplace(0, strength, len(h)), 0, 1)
    
    # Define perturbation strategy selection
    def select_perturbation(h, strength, strategy):
        if strategy == 1:
            return perturb_1(h, strength)
        elif strategy == 2:
            return perturb_2(h, strength)
        elif strategy == 3:
            return perturb_3(h, strength)
        elif strategy == 4:
            return perturb_4(h, strength)
        else:
            return h
    
    # Main optimization loop with adaptive perturbations
    for i in range(15):
        # Generate multiple starting points with different perturbation strategies
        strategies = [1, 2, 3, 4]
        for strategy in strategies:
            perturbation_strength = 0.1 / (1 + i)
            perturbed_h = select_perturbation(best_h, perturbation_strength, strategy)
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Generate a new starting point based on the current solution
        new_start = generate_new_start(best_h)
        try:
            optimized_new_start, current_c5_new_start = optimize_h(new_start)
            if current_c5_new_start < best_c5:
                best_c5 = current_c5_new_start
                best_h = optimized_new_start
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
