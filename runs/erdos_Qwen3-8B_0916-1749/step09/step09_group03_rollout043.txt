import scipy.optimize as optimize
import time

        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(

            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun

    def generate_alternative_starts(h, num_starts=5):
        starts = []
        for i in range(num_starts):
            h_new = np.copy(h)
            for j in range(len(h_new)):
                if np.random.rand() < 0.3:
                    h_new[j] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts

    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    def guided_perturb(h, gradient, strength=0.1):
        h = np.copy(h)
        for i in range(len(h)):

                h[i] = np.clip(h[i] - strength, 0, 1)
            elif gradient[i] < 0:
                h[i] = np.clip(h[i] + strength, 0, 1)
        return h

    def deterministic_perturb(h, gradient, strength=0.1):
        h = np.copy(h)
        for i in range(len(h)):
            if gradient[i] > 0:
                h[i] = np.clip(h[i] - strength, 0, 1)
            elif gradient[i] < 0:
                h[i] = np.clip(h[i] + strength, 0, 1)
