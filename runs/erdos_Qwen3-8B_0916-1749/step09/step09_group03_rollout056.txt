import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization bounds
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate alternative starting points with structure
    def generate_alternative_starts(h, num_starts=5, perturbation_scale=0.2):
        starts = []
        for _ in range(num_starts):
            h_new = np.copy(h)
            for i in range(len(h_new)):
                if np.random.rand() < perturbation_scale:
                    h_new[i] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts
    
    # Define a function to create structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        for i in range(len(h)):
            if np.random.rand() < 0.3:
                h[i] += np.random.normal(0, perturbation_strength)
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    # Define a function to perform guided perturbations
    def guided_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] += np.random.normal(0, perturbation_strength)
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbations and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(20):
        # Generate alternative starting points with different structures
        starts = generate_alternative_starts(best_h, num_starts=5, perturbation_scale=0.2)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform guided perturbations and optimization
        perturbed_h = guided_perturb(best_h, perturbation_strength=0.1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
