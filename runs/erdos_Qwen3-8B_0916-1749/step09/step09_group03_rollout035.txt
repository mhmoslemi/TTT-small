import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points

    # Function to compute the overlap integral
    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to evolve a solution with guided perturbations
    def evolve_solution(h_current, seed):
        np.random.seed(seed)
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate alternative starting points with different structures
    def generate_alternative_starts(h, num_starts=5):
        starts = []

            h_new = np.copy(h)

                if np.random.rand() < 0.3:
                    h_new[j] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts

    best_h = h_initial
    best_c5 = compute_c5(best_h)
    start_time = time.time()

        perturb_strength = 0.1 / (1 + i)

        perturbed_h = structured_perturb(best_h, perturbation_strength=perturb_strength)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Check if time budget is exceeded
