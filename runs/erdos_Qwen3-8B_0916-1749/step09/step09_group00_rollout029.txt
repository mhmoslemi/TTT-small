import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a grid search on perturbations
    def grid_search(h_initial, perturbation_range):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        for shift in perturbation_range:
            h = np.copy(h_initial)
            h += np.random.normal(0, shift, size=n_points)
            h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Use a grid search with adaptive perturbation ranges
    for i in range(10):
        perturbation_ranges = [0.01, 0.05, 0.1, 0.15, 0.2]
        for shift in perturbation_ranges:
            h = np.copy(best_h)
            h += np.random.normal(0, shift, size=n_points)
            h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Perform a few local optimizations
    for _ in range(5):
        optimized_h, current_c5 = optimize_h(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
