import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with random structure
    def generate_new_start():
        h = np.random.uniform(0, 1, size=n_points)
        return h
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, max_perturb=0.1, max_iter=10):
        h = np.copy(h_current)
        best_h = h
        best_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            perturbation = np.random.normal(0, max_perturb, size=n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
            h = best_h
        
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(30):
        # Generate alternative starting points
        starts = []
        for _ in range(5):
            starts.append(generate_new_start())
        starts.append(best_h)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform a local search from best_h
        try:
            optimized_h, current_c5 = local_search(best_h, max_perturb=0.05, max_iter=20)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
