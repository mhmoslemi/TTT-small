import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

    dx = 2.0 / n_points
    
    # Define the objective function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=compute_c5,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate new starting points based on the current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a local search from a given point
    def local_search(h_current):
        h = np.copy(h_current)
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h += perturbation
            h = np.clip(h, 0, 1)
            h = optimize_h(h)[0]
        return h
    
    # Function to perform a global search using random perturbations
    def global_search(h_initial):
        h = np.copy(h_initial)
        for _ in range(10):
            perturbation = np.random.normal(0, 0.1, size=len(h))
            h += perturbation
            h = np.clip(h, 0, 1)
            h = optimize_h(h)[0]
        return h
    
    # Main optimization loop with adaptive strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(20):
        # Perform a local search from the best solution
        local_h = local_search(best_h)
        local_c5 = compute_c5(local_h)
        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = local_h

        perturbed_h = np.copy(best_h)
        perturbation = np.random.normal(0, 0.1, size=len(perturbed_h))

        perturbed_h = np.clip(perturbed_h, 0, 1)
        global_h = global_search(perturbed_h)
        global_c5 = compute_c5(global_h)
        if global_c5 < best_c5:
            best_c5 = global_c5
            best_h = global_h
        
        # If time budget is exceeded, break the loop
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
