import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a flexible optimization function with adaptive strategies
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with random structure
    def generate_new_start():
        h = np.random.uniform(0, 1, size=n_points)
        return h
    
    # Function to generate a structured perturbation
    def structured_perturb(h, target_overlap):
        h = np.array(h)
        h = h / np.max(h) if np.max(h) > 0 else h
        perturbation = np.random.normal(0, 0.1, size=len(h))
        h = np.clip(h + perturbation, 0, 1)
        return h
    
    # Main optimization loop with exploration and exploitation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(30):
        # Generate alternative starting points with different structures
        starts = []
        starts.append(best_h)
        for j in range(5):
            starts.append(structured_perturb(best_h, target_overlap=0.5))
        starts.append(generate_new_start())
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
