import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values  # This variable is globally defined
    h = np.array(initial_h)
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the constraint
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Objective function to minimize C5
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new random starting point
    def generate_new_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to create a perturbed version of a solution
    def perturb_solution(h, strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to create a random structure solution
    def generate_random_structure():
        h = np.zeros(n_points)
        num_ones = n_points // 2
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1
        return h
    
    # Main optimization loop with exploration and perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(10):
        # Try a few random starts
        for _ in range(5):
            start = generate_new_start()
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Try a few structured starts
        for _ in range(5):
            start = generate_random_structure()
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Try perturbed versions of the best solution
        for _ in range(5):
            perturbed_h = perturb_solution(best_h, strength=0.05)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
