import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Import initial construction
    from initial_h_values import initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with random perturbation
    def generate_new_start(h_current):
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h_current))
        h = np.copy(h_current)
        h += perturbation
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform simulated annealing
    def simulated_annealing(h_initial, T_start=1.0, T_end=0.1, n_steps=1000):
        h = np.copy(h_initial)
        current_c5 = compute_c5(h)
        best_h = h
        best_c5 = current_c5
        T = T_start
        step_size = 0.1
        
        for step in range(n_steps):
            T *= 0.995
            if T < T_end:
                break
            perturbation = np.random.normal(0, step_size, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(h_perturbed)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_perturbed
                h = h_perturbed
            else:
                delta = new_c5 - best_c5
                prob = np.exp(-delta / T)
                if np.random.rand() < prob:
                    h = h_perturbed
        
        return best_h, best_c5
    
    # Main optimization loop with adaptive strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(10):
        # Perform simulated annealing with different temperatures
        for T_scale in [1.0, 0.5, 0.25]:
            T = T_scale * 1.0
            h_start = generate_new_start(best_h)
            h_opt, c5_opt = simulated_annealing(h_start, T_start=T, n_steps=500)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        
        # Perform local optimization with SLSQP
        h_opt, c5_opt = optimize_h(best_h)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
