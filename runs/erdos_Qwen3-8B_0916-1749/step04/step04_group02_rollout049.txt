import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Use a more refined grid with 150 samples for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Define the initial construction with a slightly more sophisticated pattern
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        # Create a smooth transition from 0 to 1 in the first half
        if i < n_points // 2:
            initial_h_values[i] = 1.0 * (i / (n_points // 2))
        else:
            initial_h_values[i] = 1.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function with more advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Start with the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try a few perturbations to explore the search space
    for i in range(10):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Perform a more refined optimization with tighter tolerances
    refined_h, refined_c5 = optimize_h(best_h)
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
