import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h_with_bounds(h_initial, bounds):
        h_bounds = bounds
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to generate a structured starting point
    def structured_start():
        structured = np.linspace(0, 1, n_points)
        structured = np.concatenate([structured, structured[::-1]])
        return structured
    
    # Function to generate a periodic starting point
    def periodic_start():
        periodic = np.sin(np.linspace(0, np.pi, n_points))
        periodic = np.concatenate([periodic, periodic[::-1]])
        return periodic
    
    # Function to generate a custom starting point based on existing structure
    def custom_start():
        base = np.linspace(0, 1, n_points)
        base = np.concatenate([base, base[::-1]])
        return base
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Start time
    start_time = time.time()
    
    # Define a set of starting points to explore
    starts = [
        initial_h,
        random_start(),
        structured_start(),
        periodic_start(),
        custom_start(),
    ]
    
    # Evaluate all starting points
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h_with_bounds(start, [(0, 1) for _ in range(n_points)])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue

    for i in range(5):

            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
