import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to create a sinusoidal-based starting point
    def sinusoidal_start():
        freq = np.random.uniform(1, 3)
        phase = np.random.uniform(0, 2*np.pi)
        return np.sin(np.linspace(0, 2*np.pi, n_points) * freq + phase)
    
    # Function to create a piecewise constant starting point
    def piecewise_start():
        x = np.linspace(0, 2, n_points)
        thresholds = np.random.uniform(0, 2, size=3)
        thresholds.sort()
        thresholds = np.concatenate(([0], thresholds, [2]))
        h = np.zeros(n_points)
        for i in range(len(thresholds) - 1):
            mask = (x >= thresholds[i]) & (x <= thresholds[i+1])
            h[mask] = np.random.uniform(0, 1)
        return h
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Start time
    start_time = time.time()
    
    # Perform a more systematic search with diverse starting points
    starts = [
        initial_h,
        random_start(),
        np.zeros(n_points),
        np.ones(n_points),
        np.linspace(0, 1, n_points),
        sinusoidal_start(),
        piecewise_start(),
        np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    ]
    
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Perform perturbation and optimization on the best solution
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Return the best solution found
    return best_h, best_c5, n_points
