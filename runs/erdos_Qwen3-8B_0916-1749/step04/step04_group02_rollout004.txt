import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a higher resolution to explore the space better
    n_points = 100
    dx = 2.0 / n_points
    
    # Prepare initial guess based on a sinusoidal pattern with discontinuities
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        angle = 2 * np.pi * i / n_points
        if angle < np.pi / 2 or angle > 3 * np.pi / 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to explore multiple starting points
    def explore_starting_points(starting_points, seed):
        np.random.seed(seed)
        best_h = starting_points[0]
        best_c5 = compute_c5(best_h)
        
        for i in range(1, len(starting_points)):
            h = starting_points[i]
            if np.random.rand() < 0.5:
                h = perturb(h, seed + i)
            try:
                optimized_h, current_c5 = optimize_h(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5
    
    # Generate a variety of starting points
    starting_points = [initial_h_values]
    for i in range(5):
        perturbed = perturb(initial_h_values, seed + i)
        starting_points.append(perturbed)
    
    # Try multiple starting points to avoid local minima
    best_h, best_c5 = explore_starting_points(starting_points, seed)
    
    # Refine the solution with a more precise optimization
    refined_h, refined_c5 = optimize_h(best_h)
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
