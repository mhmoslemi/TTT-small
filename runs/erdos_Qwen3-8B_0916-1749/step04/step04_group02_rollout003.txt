import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define a range of sample sizes to explore
    n_candidates = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Function to perform optimization with bounds and constraints
    def optimize_h(h_initial, n_points):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Explore different sample sizes
    for n_points in n_candidates:
        dx = 2.0 / n_points
        
        # Start with a uniform distribution
        h_initial = np.full(n_points, 0.5)
        
        # Try multiple perturbations to avoid local minima
        best_c5_for_n = float('inf')
        best_h_for_n = h_initial
        
        for i in range(5):
            seed_perturb = seed + i
            perturbed_h = perturb(h_initial, seed_perturb)
            
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h, n_points)
                if current_c5 < best_c5_for_n:
                    best_c5_for_n = current_c5
                    best_h_for_n = optimized_h
            except:
                continue
        
        # Try a more refined optimization with a smaller step size
        refined_h, refined_c5 = optimize_h(best_h_for_n, n_points)
        if refined_c5 < best_c5_for_n:
            best_c5_for_n = refined_c5
        
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = refined_h
            best_n = n_points
    
    return best_h, best_c5, len(best_h)
