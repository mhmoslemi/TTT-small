import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, fmin_l_bfgs_b

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform optimization with tighter settings
    def optimize_h(h_initial):
        result = fmin_l_bfgs_b(
            func=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=optimize.box_ineq(
                np.array([constraint(h_initial)]),
                np.array([0.0])
            ),
            maxiter=100,
            iprint=-1
        )
        return result[0], result[1]
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Start time
    start_time = time.time()
    
    # Generate a diverse set of starting points
    starts = [
        initial_h,
        np.random.uniform(0, 1, n_points),
        np.zeros(n_points),
        np.ones(n_points),
        np.linspace(0, 1, n_points),
        np.sin(np.linspace(0, np.pi, n_points)),
        np.cos(np.linspace(0, np.pi, n_points)),
        np.random.choice([0, 1], size=n_points),
        np.random.exponential(1, n_points),
        np.random.gamma(2, 1, n_points),
    ]
    
    # Perform optimization with multiple starting points
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Perform perturbation and optimization on the best solution
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Return the best solution found
    return best_h, best_c5, n_points
