import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Perform a grid search with multiple starting points
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    # Generate a set of starting points based on perturbations and random samples
    start_points = []
    for i in range(5):
        # Evolve from the current best
        h = np.copy(best_h)
        h += np.random.normal(0, 0.05, size=n_points)
        h = np.clip(h, 0, 1)
        start_points.append(h)
    
    # Add random start points
    for _ in range(5):
        start_points.append(np.random.uniform(0, 1, size=n_points))
    
    # Optimize each start point
    for start_point in start_points:
        try:
            optimized_h, current_c5 = optimize_h(start_point)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # If time is running out, break early
        if time.time() - start_time > budget_s - 10:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
