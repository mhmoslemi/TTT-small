import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the number of samples and the step size
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial construction using a sine wave modulated by a step function
    def initial_h_values():
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * x / 2))
        return h
    
    initial_h_values = initial_h_values()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem
    def objective(h):
        return compute_c5(h)
    
    # Define the bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to optimize the function
    def optimize_h(h_initial):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 200}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Check if the best solution meets the target
    if best_c5 < 0.380876:
        return best_h, best_c5, n_points
    else:
        return best_h, best_c5, n_points
