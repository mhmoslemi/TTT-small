```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Define the initial construction
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 35:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate random starting points
    def generate_random_start():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        scaling_factor = (n_points / 2) / np.sum(h)
        h = np.clip(h * scaling_factor, 0, 1)
        return h
    
    # Try multiple starting points
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        h_initial = generate_random_start()
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    return best_h, best_c5, len(best_h)
```
