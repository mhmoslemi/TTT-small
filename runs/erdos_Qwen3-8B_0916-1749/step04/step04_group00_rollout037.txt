import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform grid search on a small subset of points
    def grid_search(h_initial, n_samples=5):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        
        for i in range(n_samples):
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Function to perform a local search with adaptive step sizes
    def local_search(h_initial, max_iter=50, step_size=0.05):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            perturbation = np.random.normal(0, step_size, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Perform a combination of grid search and local search
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(3):
        # Local search
        optimized_h, current_c5 = local_search(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Grid search
        grid_h, grid_c5 = grid_search(best_h)
        if grid_c5 < best_c5:
            best_c5 = grid_c5
            best_h = grid_h
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
