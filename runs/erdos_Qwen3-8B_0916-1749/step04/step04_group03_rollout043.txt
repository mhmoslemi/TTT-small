import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a base of 100 samples
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial construction
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 50:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Try multiple starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Generate multiple perturbations of the initial solution
    for i in range(10):
        seed_perturb = seed + i
        np.random.seed(seed_perturb)
        perturbed_h = np.clip(initial_h_values + np.random.uniform(-0.1, 0.1, size=n_points), 0, 1)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
