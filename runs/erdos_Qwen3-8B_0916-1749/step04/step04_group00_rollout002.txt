import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Precompute initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to perform a local search from a starting point
    def local_search(h_initial):
        h = np.copy(h_initial)
        for _ in range(50):
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            if compute_c5(h_perturbed) < compute_c5(h):
                h = h_perturbed
        return h
    
    # Define a function to perform a grid search with multiple starting points
    def grid_search():
        start_time = time.time()
        best_h = initial_h
        best_c5 = compute_c5(initial_h)
        
        # Try multiple starting points and perform a grid search
        for i in range(5):
            # Generate a new starting point
            h_initial = evolve_start(best_h)
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            # Perform local search
            try:
                local_h = local_search(optimized_h)
                current_c5 = compute_c5(local_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = local_h
            except:
                continue
            
            # Generate a random starting point
            try:
                h_initial = random_start()
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            # Perform local search
            try:
                local_h = local_search(optimized_h)
                current_c5 = compute_c5(local_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = local_h
            except:
                continue
            
            # Check if time budget is exceeded
            if time.time() - start_time > budget_s:
                break
        
        return best_h, best_c5
    
    # Perform the grid search
    best_h, best_c5 = grid_search()
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
