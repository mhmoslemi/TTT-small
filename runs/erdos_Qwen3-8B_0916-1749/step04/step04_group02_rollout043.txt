import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Try different n_points values to find the optimal balance between accuracy and computation time
    n_point_options = [50, 60, 70, 80, 90, 100]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 70
    
    # Define the initial construction with 70 samples
    initial_h_values = np.zeros(70)
    for i in range(70):
        if i < 35:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Function to perform optimization
    def optimize_h(h_initial, n_points):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to refine the solution
    def refine_solution(h_initial, n_points):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Evaluate the initial solution
    initial_c5 = compute_c5(initial_h_values)
    best_h = initial_h_values
    best_c5 = initial_c5
    
    # Try different n_points values
    for n_points in n_point_options:
        # Create a new initial construction for this n_points
        dx = 2.0 / n_points
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h_initial[i] = 1.0
            else:
                h_initial[i] = 0.0
        
        # Try multiple starting points to avoid local minima
        best_h_for_n = h_initial
        best_c5_for_n = compute_c5(h_initial)
        
        # Perturb the initial solution
        for i in range(5):
            seed_perturb = seed + i
            perturbed_h = perturb(h_initial, seed_perturb)
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h, n_points)
                if current_c5 < best_c5_for_n:
                    best_c5_for_n = current_c5
                    best_h_for_n = optimized_h
            except:
                continue
        
        # Refine the solution
        refined_h, refined_c5 = refine_solution(best_h_for_n, n_points)
        if refined_c5 < best_c5_for_n:
            best_c5_for_n = refined_c5
            best_h_for_n = refined_h
        
        # Update the global best
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n
            best_n_points = n_points
    
    return best_h, best_c5, len(best_h)
