import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with more structure
    def structured_start(h_base):
        h = np.copy(h_base)
        h[::2] = 0.5  # Set even indices to 0.5
        h[1::2] = np.random.uniform(0, 1, len(h[1::2]))  # Random values for odd indices
        return h
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with a fixed pattern
    def pattern_start():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 0.5
            elif i % 5 == 1:
                h[i] = 0.75
            elif i % 5 == 2:
                h[i] = 0.25
            else:
                h[i] = 0.0
        return h
    
    # Initialize with a structured start
    best_h = structured_start(initial_h)
    best_c5 = compute_c5(best_h)
    
    # Track the best solution found
    start_time = time.time()
    
    # Try multiple strategies: structured start, random start, and evolved start
    for _ in range(5):
        # Try structured start
        h_initial = structured_start(best_h)
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Try evolved start
        h_initial = evolve_start(best_h)
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Try pattern start
        h_initial = pattern_start()
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
