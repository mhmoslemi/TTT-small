import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a base number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial construction with alternating blocks
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0 or i % 5 == 1:
            initial_h_values[i] = 1.0
        elif i % 5 == 2 or i % 5 == 3:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to evaluate multiple perturbations in parallel
    def evaluate_parallel(start_points):
        results = []
        for i in range(len(start_points)):
            try:
                optimized_h, current_c5 = optimize_h(start_points[i])
                results.append((current_c5, optimized_h))
            except:
                results.append((float('inf'), np.zeros(n_points)))
        return results
    
    # Try multiple starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Generate multiple perturbations of the initial solution
    perturbations = [perturb(initial_h_values, seed + i) for i in range(10)]
    
    # Evaluate all perturbations in parallel
    start_time = time.time()
    results = evaluate_parallel(perturbations)
    elapsed = time.time() - start_time
    
    # Find the best solution among all results
    for c5, h in results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # If time allows, try a larger number of points
    if elapsed < budget_s - 10:
        n_points = 150
        dx = 2.0 / n_points
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0 or i % 5 == 1:
                initial_h_values[i] = 1.0
            elif i % 5 == 2 or i % 5 == 3:
                initial_h_values[i] = 0.5
            else:
                initial_h_values[i] = 0.0
        
        # Adjust the initial construction for larger n_points
        def compute_c5_large(h):
            h = np.array(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def objective_large(h):
            return compute_c5_large(h)
        
        def constraint_large(h):
            return np.sum(h) - n_points / 2
        
        def optimize_h_large(h_initial):
            h_bounds = [(0, 1) for _ in range(n_points)]
            
            # Use a robust optimization method with tighter tolerances
            result = optimize.minimize(
                fun=objective_large,
                x0=h_initial,
                bounds=h_bounds,
                constraints=[{'type': 'eq', 'fun': constraint_large}],
                method='SLSQP',
                options={'ftol': 1e-12, 'maxiter': 1000}
            )
            return result.x, result.fun
        
        def perturb_large(h, seed):
            np.random.seed(seed)
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            perturbed = np.clip(h + perturbation, 0, 1)
            return perturbed
        
        perturbations_large = [perturb_large(initial_h_values, seed + i) for i in range(10)]
        
        # Evaluate all perturbations in parallel for larger n_points
        results_large = evaluate_parallel(perturbations_large)
        
        # Find the best solution among all results
        for c5, h in results_large:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    return best_h, best_c5, len(best_h)
