import scipy.optimize as optimize

        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
        )
        return result.x, result.fun

    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h))

        return perturbed

        return np.random.uniform(0, 1, n_points)
