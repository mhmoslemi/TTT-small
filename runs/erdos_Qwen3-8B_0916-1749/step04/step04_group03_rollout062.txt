import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Function to perform optimization with different starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(len(h_initial))]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Start with a more refined initial construction
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Create a more refined initial step function with alternating high and low regions
    for i in range(n_points):
        if i % 5 == 0 or i % 5 == 1:
            initial_h_values[i] = 0.8
        elif i % 5 == 2 or i % 5 == 3:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.2
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Try multiple starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try a few perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    return best_h, best_c5, len(best_h)
