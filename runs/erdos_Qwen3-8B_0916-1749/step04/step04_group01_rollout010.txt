import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to create a piecewise constant function from a pattern
    def create_pattern(pattern, n_points):
        step = n_points // len(pattern)
        h = np.zeros(n_points)
        for i, val in enumerate(pattern):
            start = i * step
            end = start + step
            h[start:end] = val
        return h
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Start time
    start_time = time.time()
    
    # Define a set of patterns to try
    patterns = [
        [0, 1],  # Alternating 0-1
        [0.2, 0.5, 0.8],  # Increasing steps
        [0.5, 0.75, 0.25],  # Custom step pattern
        [0.3, 0.6, 0.4],  # Another step pattern
        [0.7, 0.4, 0.3],  # More varied
        [0.5],  # Constant
    ]
    
    # Try each pattern
    for pattern in patterns:
        h = create_pattern(pattern, n_points)
        optimized_h, current_c5 = optimize_h(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Perturb and optimize the best solution
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    # Return the best solution found
    return best_h, best_c5, n_points
