import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with 200 samples
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the initial construction with a more refined pattern
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 80:
            initial_h_values[i] = 1.0
        elif i < 160:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Custom optimization function using Nelder-Mead with adaptive bounds
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_wrapper(x):
            return objective(x)
        
        def constraint_wrapper(x):
            return constraint(x)
        
        result = optimize.minimize(
            fun=objective_wrapper,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint_wrapper}],
            method='L-BFGS-B',
            options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Evaluate initial solution
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to explore the search space
    for i in range(10):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Final optimization with the best solution found
    optimized_h, final_c5 = optimize_h(best_h)
    
    # Return the best solution found
    return optimized_h, final_c5, len(optimized_h)
