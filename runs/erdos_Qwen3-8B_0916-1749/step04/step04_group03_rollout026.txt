import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Try different n_points values to find the best trade-off between accuracy and speed
    n_points_list = [100, 140, 180]
    best_c5 = float('inf')
    best_h = np.array([])
    
    # Precompute initial construction for n=70
    initial_n = 70
    initial_h_values = np.zeros(initial_n)
    for i in range(initial_n):
        if i < initial_n // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Define the function to compute C5
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(len(h_initial))]
        
        # Use SLSQP for optimization with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution with some randomness
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Use the initial construction as a starting point
    initial_h = initial_h_values
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Try multiple perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Evaluate the best solution found so far
    best_h = np.array(best_h)
    best_c5 = compute_c5(best_h)
    
    # Try different n_points values to see which one gives the best result
    for n_points in n_points_list:
        # Start from the best solution found so far
        h_start = np.zeros(n_points)
        h_start[:int(n_points / 2)] = 1.0
        
        # Use the same initial construction but with different n_points
        h_start = np.zeros(n_points)
        h_start[:int(n_points / 2)] = 1.0
        h_start = np.array(h_start)
        
        # Optimize the new n_points solution
        optimized_h, current_c5 = optimize_h(h_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
