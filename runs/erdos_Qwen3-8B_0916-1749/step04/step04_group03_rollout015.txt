```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the number of points and dx
    n_points = 120
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 200}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        h = np.random.uniform(0, 1, n_points)
        # Ensure the constraint is approximately satisfied
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        return h
    
    # Generate multiple starting points
    num_starts = 10
    best_h = None
    best_c5 = float('inf')
    
    for start_idx in range(num_starts):
        start_time = time.time()
        h_initial = random_start()
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s / 2:
            break
    
    return best_h, best_c5, len(best_h)
```
