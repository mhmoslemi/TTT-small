import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Define the initial construction with 70 samples
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 35:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to create a more diverse initial population
    def generate_initial_population(size):
        population = []
        for i in range(size):
            h = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    h[j] = 1.0
            population.append(h)
        return population
    
    # Function to perform genetic algorithm optimization
    def genetic_algorithm(population_size=20, generations=10, mutation_rate=0.1):
        population = generate_initial_population(population_size)
        
        for generation in range(generations):
            # Evaluate fitness
            fitness = [compute_c5(h) for h in population]
            
            # Select the top 10% of the population
            indices = np.argsort(fitness)
            selected = population[indices[:int(len(population) * 0.1)]]
            
            # Create new population through crossover and mutation
            new_population = []
            for _ in range(population_size):
                # Crossover
                parent1 = np.random.choice(selected)
                parent2 = np.random.choice(selected)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                
                new_population.append(child)
            
            population = new_population
        
        # Find the best solution in the final population
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        return population[best_idx], fitness[best_idx]
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try genetic algorithm optimization
    best_h, best_c5 = genetic_algorithm(population_size=20, generations=10, mutation_rate=0.1)
    
    # Try a more refined optimization with a smaller step size
    def refine_optimize(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Refine the best solution
    refined_h, refined_c5 = refine_optimize(best_h)
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
