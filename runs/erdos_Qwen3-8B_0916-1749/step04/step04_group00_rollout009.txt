        )

    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with random perturbations
    def random_start():
        return np.random.uniform(0, 1, size=n_points)

    best_h = initial_h
    best_c5 = compute_c5(best_h)

    for _ in range(10):

        h_initial = evolve_start(best_h)

        try:
