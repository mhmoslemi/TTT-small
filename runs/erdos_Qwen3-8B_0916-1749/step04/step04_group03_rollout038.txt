import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Define the initial construction
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 35:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate random starting points
    def generate_random_start():
        h = np.random.uniform(0, 1, n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        return h
    
    # Define a function to perform local search
    def local_search(h):
        h = np.array(h)
        for _ in range(100):
            perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
            new_h = np.clip(h + perturbation, 0, 1)
            if np.sum(new_h) * dx >= 1 - 1e-8 and np.sum(new_h) * dx <= 1 + 1e-8:
                h = new_h
        return h
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Use a more aggressive optimization strategy
    for _ in range(5):
        # Generate a random starting point
        start_h = generate_random_start()
        # Perform local search
        start_h = local_search(start_h)
        # Optimize
        optimized_h, current_c5 = optimize_h(start_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Try a few perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
