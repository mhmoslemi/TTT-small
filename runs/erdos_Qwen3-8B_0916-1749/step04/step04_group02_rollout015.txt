import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 100 samples
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial construction based on a piecewise function
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 50:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to refine the solution with a smaller step size
    def refine_optimize(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate multiple starting points
    def generate_starting_points(seed, num_points):
        np.random.seed(seed)
        starting_points = []
        for i in range(num_points):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed = np.clip(initial_h_values + perturbation, 0, 1)
            starting_points.append(perturbed)
        return starting_points
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to avoid local minima
    starting_points = generate_starting_points(seed, 10)
    for i in range(len(starting_points)):
        try:
            optimized_h, current_c5 = optimize_h(starting_points[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Refine the best solution
    refined_h, refined_c5 = refine_optimize(best_h)
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
