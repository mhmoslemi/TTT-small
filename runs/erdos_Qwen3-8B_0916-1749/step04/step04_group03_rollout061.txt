import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Starting with a moderate number of samples
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction based on a sinusoidal pattern
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        initial_h_values[i] = 0.5 * (1 + np.cos(2 * np.pi * x / 2))
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Function to generate perturbations for multiple starting points
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Initialize best solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try several starting points
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Additional refinement with a finer grid
    n_points_refine = 200
    dx_refine = 2.0 / n_points_refine
    initial_h_refine = np.zeros(n_points_refine)

        x = i * dx_refine

        initial_h_refine[i] = 0.5 * (1 + np.cos(2 * np.pi * x / 2))
    
    # Function to compute the overlap integral for refined grid
    def compute_c5_refined(h):

        corr = np.correlate(h, 1 - h, mode='full')

        return max_corr * dx_refine
    
    # Define the objective function for refinement
    def objective_refined(h):
        return compute_c5_refined(h)
    
    # Define the constraint function for refinement
    def constraint_refined(h):
        return np.sum(h) - n_points_refine / 2
    
    # Function to perform optimization for refinement
    def optimize_h_refined(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points_refine)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective_refined,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint_refined}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 500}

        return result.x, result.fun

    best_c5_refined = best_c5

        seed_perturb = seed + 10 + i
        perturbed_h_refine = perturb(initial_h_refine, seed_perturb)

                best_c5_refined = current_c5_refined

            continue

    for i in range(5):
        seed_perturb = seed + 20 + i

        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5_refined:
                best_c5_refined = current_c5
                best_h = optimized_h
        except:
            continue

    return best_h, best_c5_refined, len(best_h)
