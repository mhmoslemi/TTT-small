import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with random perturbations
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perform a grid search with multiple starting points
    def grid_search(start_points, max_iters=5):
        best_h = start_points[0]
        best_c5 = compute_c5(best_h)
        
        for i in range(max_iters):
            print(f"Iteration {i+1}/{max_iters}")
            start_time = time.time()
            for j in range(len(start_points)):
                h_initial = start_points[j]
                try:
                    optimized_h, current_c5 = optimize_h(h_initial)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                except:
                    continue
            print(f"Time elapsed: {time.time() - start_time:.2f} seconds")
            if time.time() - start_time > budget_s:
                break
        return best_h, best_c5
    
    # Generate multiple starting points
    start_points = [initial_h]
    for _ in range(5):
        start_points.append(random_start())
    
    # Perform the grid search
    best_h, best_c5 = grid_search(start_points)
    
    # Perturbation and optimization
    for _ in range(5):
        h_initial = evolve_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    return best_h, best_c5, len(best_h)
