import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a base resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial construction as a simple step function
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 50:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with multiple starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Try multiple starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try a few perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Further refinement with higher resolution
    n_points_refine = 200
    dx_refine = 2.0 / n_points_refine
    
    # Interpolate the best_h to higher resolution
    def interpolate_h(h_low, n_points_high):
        h_high = np.zeros(n_points_high)
        for i in range(n_points_high):
            pos = i * dx_refine
            low_idx = int(pos / dx)
            low_idx = min(max(low_idx, 0), len(h_low) - 1)
            h_high[i] = h_low[low_idx]
        return h_high
    
    h_high = interpolate_h(best_h, n_points_refine)
    
    # Function to perform optimization with multiple starting points
    def optimize_h_high(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points_refine)]
        
        # Use a robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Try a few perturbations to explore the search space
    best_h_high = h_high
    best_c5_high = compute_c5(h_high)
    
    for i in range(5):
        seed_perturb = seed + 10 + i
        perturbed_h_high = perturb(best_h_high, seed_perturb)
        try:
            optimized_h_high, current_c5_high = optimize_h_high(perturbed_h_high)
            if current_c5_high < best_c5_high:
                best_c5_high = current_c5_high
                best_h_high = optimized_h_high
        except:
            continue
    
    # Return the best solution found
    return best_h_high, best_c5_high, n_points_refine
