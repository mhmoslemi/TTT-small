import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Define the initial construction with 70 samples
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 35:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perform a multi-start optimization
    def multi_start_optimize(initial_guesses, n_starts=5):
        best_h = initial_guesses[0]
        best_c5 = compute_c5(best_h)
        
        for guess in initial_guesses[1:]:
            try:
                optimized_h, current_c5 = optimize_h(guess)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5
    
    # Define a function to generate random initial guesses
    def generate_random_guesses(n_guesses):
        guesses = []

            guess = np.random.rand(n_points)
            guess = np.clip(guess, 0, 1)
            # Enforce the integral constraint
            scale = (n_points / 2) / np.sum(guess)
            guess *= scale
            guesses.append(guess)
        return guesses
    
    # Try multiple starting points to avoid local minima
    initial_guesses = [initial_h_values]
    initial_guesses += [generate_random_guesses(5)]

    def refine_optimize(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,

            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}

        return result.x, result.fun
    
    refined_h, refined_c5 = refine_optimize(best_h)
