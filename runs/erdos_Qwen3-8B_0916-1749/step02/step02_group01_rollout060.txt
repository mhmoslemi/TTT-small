import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Function to perform optimization with adaptive parameters
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def generate_random_perturbation(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to create a grid-based starting point
    def create_grid_starting_point(n_points):
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if grid[i] < 1:
                h[i] = 1
            else:
                h[i] = 0
        return h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_n_points = n_points
    
    # Try multiple starting points to avoid local minima
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = generate_random_perturbation(initial_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
                best_n_points = n_points
        except:
            continue
    
    # Try grid-based starting point
    grid_h = create_grid_starting_point(n_points)
    try:
        optimized_h, current_c5 = optimize_h(grid_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
            best_n_points = n_points
