import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate random samples for exploration
    def generate_random_h():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h - np.min(h)
        h = h / np.max(h)
        return h
    
    # Function to perform a local search
    def local_search(h_initial):
        h = h_initial.copy()
        for _ in range(100):
            perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            h_perturbed = h_perturbed - np.min(h_perturbed)
            h_perturbed = h_perturbed / np.max(h_perturbed)
            obj_perturbed = compute_c5(h_perturbed)
            if obj_perturbed < compute_c5(h):
                h = h_perturbed
        return h
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Perform multiple local searches with different perturbations
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(initial_h, seed_perturb)
        optimized_h, current_c5 = optimize_h(perturbed_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Perform a local search on the best solution found
    best_h = local_search(best_h)
    best_c5 = compute_c5(best_h)
    
    # Generate random samples to explore the search space
    for i in range(5):
        random_h = generate_random_h()
        optimized_h, current_c5 = optimize_h(random_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
