    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    def evaluate(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Create a copy to avoid modifying the input
        h_initial = np.array(h_initial, copy=True)
        
        # Regularize the initial guess to avoid discontinuities
        h_initial = np.clip(h_initial, 0, 1)
        
        result = optimize.minimize(
            fun=evaluate,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
        )
        return result.x, result.fun
    
    def generate_perturbed_h(h, perturbation_std=0.05):
        perturbation = np.random.normal(0, perturbation_std, h.shape)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(initial_h)
    
    # Generate a few perturbations and optimize
    for _ in range(5):
        perturbed_h = generate_perturbed_h(best_h)
        new_h, new_c5 = optimize_h(perturbed_h)

        if new_c5 < best_c5:

        if new_c5 < best_c5:
