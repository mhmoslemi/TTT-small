import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = np.concatenate([initial_h[:n_points//2], initial_h[n_points//2+1:]])
    initial_h = np.random.permutation(initial_h)
    
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds and optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    
    # Try optimization with different starting points
    def optimize_with_starting_point(starting_h):
        h0 = starting_h.copy()
        start_time = time.time()
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
        else:
            return (starting_h, compute_C5(starting_h), n_points)
    
    best_c5 = float('inf')
    best_h = initial_h
    
    # Try initial_h
    result = optimize_with_starting_point(initial_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try random starting point
    random_h = np.random.uniform(0, 1, n_points)
    result = optimize_with_starting_point(random_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try perturbed starting point
    perturbed_h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
    perturbed_h = np.clip(perturbed_h, 0, 1)
    result = optimize_with_starting_point(perturbed_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try another perturbed starting point
    perturbed_h = initial_h + np.random.normal(0, 0.05, n_points)
    perturbed_h = np.clip(perturbed_h, 0, 1)
    result = optimize_with_starting_point(perturbed_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try a more structured perturbation
    for i in range(n_points):
        for j in range(n_points):
            if i != j:
                perturbed_h = initial_h.copy()
                perturbed_h[i] += 0.1
                perturbed_h[j] -= 0.1
                perturbed_h = np.clip(perturbed_h, 0, 1)
                result = optimize_with_starting_point(perturbed_h)
                if result[1] < best_c5:
                    best_c5 = result[1]
                    best_h = result[0]
    
    return (best_h, best_c5, n_points)
