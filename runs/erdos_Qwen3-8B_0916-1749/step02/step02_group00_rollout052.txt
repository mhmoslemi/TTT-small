    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx

        return compute_c5(h)

    def optimize_h(h_initial):

        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )

    best_h, best_c5 = optimize_h(initial_h)
