import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different methods
    def optimize_h(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_n_points = n_points
    
    # Try different starting points and methods to avoid local minima
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(initial_h, seed_perturb)
        
        # Try different optimization methods
        for method in ['SLSQP', 'Nelder-Mead', 'L-BFGS-B']:
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h, method=method)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
                    best_n_points = len(best_h)
            except:
                continue
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
