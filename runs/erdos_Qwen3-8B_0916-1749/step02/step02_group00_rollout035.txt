    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')

        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):

        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(initial_h)

    for _ in range(10):

        perturbed_h = np.clip(best_h + perturbation, 0, 1)

        new_h, new_c5 = optimize_h(perturbed_h)
