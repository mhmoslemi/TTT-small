import scipy.optimize as optimize

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(initial_h)
    
    # Define a set of candidate starting points for diversification
    candidate_starting_points = [
        initial_h,
        np.random.uniform(0, 1, n_points),
        np.clip(initial_h * 0.5 + np.random.uniform(0, 1, n_points) * 0.5, 0, 1),
        np.clip(initial_h * 0.8 + np.random.uniform(0, 0.2, n_points), 0, 1),
        np.clip(initial_h * 0.2 + np.random.uniform(0.8, 1, n_points), 0, 1)
    ]
    
    # Evaluate each candidate starting point
    for start in candidate_starting_points:
        # Clip values to [0, 1] and ensure sum constraint
        start = np.clip(start, 0, 1)
        start = np.array(start, dtype=np.float64)
        # Adjust to satisfy the sum constraint
        total_sum = np.sum(start)
        if total_sum == 0:
            start = np.random.uniform(0, 1, n_points)
        else:
            scale_factor = (n_points / 2) / total_sum
            start = np.clip(start * scale_factor, 0, 1)
        
        # Optimize from the perturbed point
        new_h, new_c5 = optimize_h(start)
        
        # Update best solution if improvement
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
