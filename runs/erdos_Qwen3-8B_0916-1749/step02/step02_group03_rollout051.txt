import scipy.optimize as optimize
import time

    def compute_C5(h):
        h = np.array(h)

        shift_values = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shift_values:
            if k == 0:
                continue
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    def objective(h):
        return compute_C5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization function with bounds and constraints
    def optimize_with_starting_point(starting_h):
        bounds = [(0, 1) for _ in range(n_points)]
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            starting_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 200, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
