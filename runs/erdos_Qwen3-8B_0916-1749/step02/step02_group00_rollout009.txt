import scipy.optimize as optimize

    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun

    best_h, best_c5 = optimize_h(initial_h)
    best_c5_value = best_c5

    for _ in range(5):

        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        # Optimize from the perturbed point
        new_h, new_c5 = optimize_h(perturbed_h)
