import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined

    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_n_points = n_points
    
    # Try multiple starting points with different perturbations
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = np.random.uniform(0, 1, n_points)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
                best_n_points = n_points
        except:
            continue
    
    # Try with fewer samples to improve performance
    for n in [50, 40, 30, 20]:
