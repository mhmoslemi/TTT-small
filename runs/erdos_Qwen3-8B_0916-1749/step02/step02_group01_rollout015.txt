import numpy as np
import scipy.optimize as optimize

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 2000}
        )
        return result.x, result.fun
    
    # Define a function to generate random perturbations
    def generate_perturbations(h, num_perturbations=5):
        perturbations = []
        for i in range(num_perturbations):
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            perturbed = np.clip(h + perturbation, 0, 1)
            perturbations.append(perturbed)
        return perturbations
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Try multiple perturbations to explore the search space
    for perturbation in generate_perturbations(initial_h, num_perturbations=10):
        try:
            optimized_h, current_c5 = optimize_h(perturbation)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
