import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define n_points for the function h
    n_points = 100
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to optimize h using SLSQP
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Start with a better initial guess inspired by known optimal patterns
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        # Distribute 1s in a periodic and spread-out pattern
        if i % 10 == 0:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    best_h, best_c5 = optimize_h(initial_h)
    
    # Try multiple random perturbations of the best solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        new_h, new_c5 = optimize_h(perturbed_h)
        
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    
    # Try another starting point based on a different initial pattern
    initial_h2 = np.zeros(n_points)
    for i in range(n_points):
        if i % 15 == 0:
            initial_h2[i] = 1.0
        else:
            initial_h2[i] = 0.0
    
    new_h, new_c5 = optimize_h(initial_h2)
    if new_c5 < best_c5:
        best_h, best_c5 = new_h, new_c5
    
    return best_h, best_c5, n_points
