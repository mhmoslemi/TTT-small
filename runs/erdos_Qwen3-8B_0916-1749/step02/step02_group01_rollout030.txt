import numpy as np
import scipy.optimize as optimize

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to generate a random starting point
    def generate_random_start():
        h = np.random.uniform(0, 1, size=n_points)
        return h
    
    # Define a function to generate a grid-based starting point
    def generate_grid_start():
        h = np.zeros(n_points)
        step = 2.0 / n_points
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            elif i % 4 == 0:
                h[i] = 0.5
            else:
                h[i] = 0.0
        return h
    
    # Try a few different starting points to avoid local minima
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Try multiple starting points
    num_starts = 5

            start_h = generate_grid_start()
