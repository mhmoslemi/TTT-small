import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Load initial construction
    n_points = 100  # Reduced from 70 for faster optimization
    
    # Define the initial construction
    def initial_h_values(n):
        r = np.linspace(0, 2, n)
        h = np.zeros(n)
        for i in range(n):
            x = r[i]
            if x < 1:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    initial_h = initial_h_values(n_points)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(initial_h)
    
    # Introduce a more structured exploration of the parameter space
    for _ in range(10):
        # Generate random perturbation with adaptive scaling
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        new_h, new_c5 = optimize_h(perturbed_h)
        
        # Update best solution if improvement
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    
    return best_h, best_c5, len(best_h)
