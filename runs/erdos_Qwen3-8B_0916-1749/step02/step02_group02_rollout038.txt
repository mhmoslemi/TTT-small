        return compute_C5(h)

    def optimize_h(h0):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = minimize(
            fun=objective,
            x0=h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 200, "ftol": 1e-8, "eps": 1e-8, "disp": False}
        )
        return result.x, result.fun

    best_h, best_c5 = initial_h, compute_C5(initial_h)

    for _ in range(5):

        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)

        try:
