import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Function to optimize with a given starting point
    def optimize_with_starting_point(starting_h):
        h0 = starting_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 150, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
        else:
            # Fallback to starting point if time runs out
            best_c5 = compute_C5(starting_h)
            return (starting_h, best_c5, n_points)
    
    best_c5 = float('inf')
    best_h = initial_h
    
    # Try initial_h_values
    result = optimize_with_starting_point(initial_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try perturbed initial_h_values
    perturbed_h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
    perturbed_h = np.clip(perturbed_h, 0, 1)
    result = optimize_with_starting_point(perturbed_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try random starting point
    random_h = np.random.uniform(0, 1, n_points)
    result = optimize_with_starting_point(random_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]

    h_freq = np.fft.fft(initial_h)

    h_freq = np.clip(h_freq, -1, 1)
    h_perturbed = np.fft.ifft(h_freq).real
    h_perturbed = np.clip(h_perturbed, 0, 1)
    
    result = optimize_with_starting_point(h_perturbed)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    # Try another perturbation based on local maxima
    max_indices = np.argmax(np.correlate(initial_h, 1 - initial_h, mode="full"))
    perturbed_h = initial_h.copy()
    perturbed_h[max_indices % n_points] += 0.1
    perturbed_h[max_indices % n_points] = np.clip(perturbed_h[max_indices % n_points], 0, 1)
    result = optimize_with_starting_point(perturbed_h)
    if result[1] < best_c5:
        best_c5 = result[1]
        best_h = result[0]
    
    return (best_h, best_c5, n_points)
