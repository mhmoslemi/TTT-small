import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):

    n_points = 45  # Use the same n_points as the initial construction

    initial_h = initial_h_values
    initial_sum = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        # Shift h to compute overlaps with different shifts
        overlap_values = []
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted))
            overlap_values.append(overlap)
        c5 = np.max(overlap_values) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with a given starting point
    def optimize_with_starting_point(starting_h):
        h0 = starting_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
        else:
            # Fallback to starting point if time runs out
            best_c5 = compute_C5(starting_h)
            return (starting_h, best_c5, n_points)
