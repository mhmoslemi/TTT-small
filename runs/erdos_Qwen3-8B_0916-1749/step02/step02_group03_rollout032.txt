        c5 = np.max(overlap) * dx
        return c5

    def objective(h):
        return compute_C5(h)

        return np.sum(h) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]

            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 150, "ftol": 1e-10, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time

            best_h = result["x"]
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
