import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate a random sample
    def generate_random_solution():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        return h
    
    # Function to generate a structured sample
    def generate_structured_sample():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h
    
    # Evaluate the initial solution
    initial_c5 = compute_c5(initial_h)
    best_h = initial_h
    best_c5 = initial_c5
    
    # Try a few perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(initial_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try random samples
    for i in range(5):
        seed_rand = seed + i + 5
        random_h = generate_random_solution()

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try structured samples
    for i in range(5):
        seed_struct = seed + i + 10
        structured_h = generate_structured_sample()
        try:
            optimized_h, current_c5 = optimize_h(structured_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
