def refine_solution(h, n_points, dx=2.0 / n_points):
    bounds = [(0, 1) for _ in range(n_points)]
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        constraints={"type": "eq", "fun": lambda h: np.sum(h) - n_points / 2},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-9, "eps": 1e-9}
    )
    return result["x"]
