    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)

    def constraint(h):

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Define a function to generate a random initial guess
    def random_h():
        return np.random.uniform(0, 1, size=n_points)
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Try multiple random initial guesses to explore the search space
    for i in range(5):
        random_h_initial = random_h()

            optimized_h, current_c5 = optimize_h(random_h_initial)
