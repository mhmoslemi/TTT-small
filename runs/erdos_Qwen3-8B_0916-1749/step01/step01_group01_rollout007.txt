    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with a different starting point and
    # switching to a global optimization approach to avoid local optima
    def optimize_h(initial_h):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=initial_h,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B'
        )
        return result.x, result.fun
    
    # Try multiple starting points to avoid local optima
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    # Try starting with the initial construction
    best_h, best_c5 = optimize_h(initial_h)
