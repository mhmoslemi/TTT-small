    def compute_c5(h):
        h = np.array(h)

        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        return result.x, result.fun

    best_h = initial_h
