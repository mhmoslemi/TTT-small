import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from previous work (n=70)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this file is available
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Precompute 1 - h for initial values
    one_minus_h = 1 - initial_h_values
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = opt.minimize(
        objective,
        initial_h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Check if we can improve by slightly perturbing the solution
    def perturb_solution(h, perturbation_range=0.01):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-perturbation_range, perturbation_range)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        return h_perturbed
    
    # Try a few perturbations to escape local minima
    best_c5 = c5_best
    best_h = h_best
    
    for _ in range(5):
        perturbed_h = perturb_solution(h_best)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h
    
    return best_h, best_c5, len(best_h)
