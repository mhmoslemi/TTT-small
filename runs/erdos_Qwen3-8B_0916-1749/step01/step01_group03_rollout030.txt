        return compute_C5(h)

    def optimize_with_adaptive_bounds(h0, bounds, maxiter=100):
        result = minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": maxiter, "ftol": 1e-8, "eps": 1e-8}
        )
        return result
    
    # Adaptive refinement strategy
    def refine_solution(h, n_points, max_refinement=5):
        for _ in range(max_refinement):

            new_c5 = compute_C5(refined_h)

            if new_c5 < compute_C5(h):
