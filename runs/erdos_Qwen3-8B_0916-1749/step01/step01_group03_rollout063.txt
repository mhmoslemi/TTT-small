import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    n_points = 300
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    def compute_C5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    h0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective_with_gradient(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def grad_objective(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_index = np.argmax(overlap)
        grad = np.zeros_like(h)
        if max_overlap > 0:
            for i in range(n_points):
                grad[i] = -overlap[n_points - 1 + i] + overlap[i + max_index + 1]
            grad = grad * dx
        return grad
    
    start_time = time.time()
    
    result = minimize(
        objective_with_gradient,
        h0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        jac=grad_objective,
        options={"maxiter": 1000, "ftol": 1e-8, "eps": 1e-8}
    )
    
    elapsed_time = time.time() - start_time
    
    if elapsed_time < budget_s:
        best_h = result["x"]
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
    else:
        best_h = initial_h
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
