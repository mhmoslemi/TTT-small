import scipy.optimize as opt
import scipy.signal as signal

from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=300):

    n_points = len(initial_h)

    starting_points = [initial_h]

        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)

        starting_points.append(perturbed_h)

        def compute_c5(h):
            h = np.array(h)

            corr = signal.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            return c5

        def objective(h):
            return compute_c5(h)
        
        # Define constraint: sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization settings
        max_iter = 100
        tolerance = 1e-6
        time_left = budget_s
        
        # Initialize timer
        start_time = time.time()
        
        # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
        result = opt.minimize(
            objective,
            start_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': tolerance, 'maxiter': max_iter}
        )
        
        # Check if the optimization completed within time
        elapsed = time.time() - start_time
        if elapsed >= time_left:
            continue
