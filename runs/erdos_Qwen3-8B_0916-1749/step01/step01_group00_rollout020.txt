    # First optimization step to get a better initial guess
    result = opt.minimize(

        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 50}
    )
    
    h_best = result.x
    
    # Second optimization step using SLSQP for refinement
    result = opt.minimize(
        objective,
        h_best,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],

    )
