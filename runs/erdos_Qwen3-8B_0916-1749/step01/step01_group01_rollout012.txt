def run(seed=42, budget_s=300):
    # Start timing
    start_time = time.time()
    
    # Check if initial_h_values is provided, otherwise use a default
    if initial_h_values is None:
        n_points = 100
        initial_h_values = np.ones(n_points) * 0.5
    else:
        n_points = len(initial_h_values)
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initialize h with initial_h_values
    h = initial_h_values.copy()
    
    # Let's run a few iterations of simulated annealing with a local optimizer
    def optimize_step(h):
        # Define bounds
        bounds = Bounds(0, 1)
        # Define constraints: sum(h) * dx = 1 => sum(h) = n_points / 2
        constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
        
        # Objective function
        def objective(x):
            return compute_c5(x)
        
        # Optimization
        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Simulated annealing loop
    best_h = h
    best_c5 = compute_c5(h)
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    
    while time.time() - start_time < budget_s - 1:
        # Generate a perturbation
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbation = np.clip(perturbation, -0.5, 0.5)
        new_h = np.clip(h + perturbation, 0, 1)
        
        # Compute new c5
        new_c5 = compute_c5(new_h)
        
        # Acceptance probability
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_h = new_h
                best_c5 = new_c5
        
        # Cool down
        temp *= cooling_rate
        iterations += 1
        
        # If we've run enough iterations, break
        if iterations > 1000:
            break
    
    # Final optimization with SLSQP
    final_h, final_c5 = optimize_step(best_h)
    
    # Return best solution
    return final_h, final_c5, len(final_h)
