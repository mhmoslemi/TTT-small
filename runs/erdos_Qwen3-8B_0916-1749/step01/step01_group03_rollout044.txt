from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300, **kwargs):

    n_points = len(initial_h)

    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using a more sophisticated approach with adaptive bounds and multiple restarts
    def optimize_h():
        best_h = initial_h.copy()
        best_c5 = compute_C5(best_h)
        
        # Try multiple restarts with different initializations
        for restart in range(3):
            # Perturb the initial guess
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h0 = np.clip(initial_h + perturbation, 0, 1)
            
            bounds = Bounds([0]*n_points, [1]*n_points)
            
            result = minimize(
                objective,
                h0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method="SLSQP",
                options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
            )
            
            current_h = result["x"]
            current_c5 = compute_C5(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h

    start_time = time.time()
    
    best_h, best_c5 = optimize_h()
    
    elapsed_time = time.time() - start_time
    
    # If within budget, return the best found solution
    if elapsed_time < budget_s:
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
