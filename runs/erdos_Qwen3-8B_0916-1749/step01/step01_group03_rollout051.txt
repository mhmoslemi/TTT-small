import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_function(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        start_time = time.time()
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        elapsed_time = time.time() - start_time
        return result, elapsed_time
    
    # Generate a few alternative starting points to avoid local optima
    def generate_alternative_points(initial_h, num_alternatives=5):
        alternatives = []
        for i in range(num_alternatives):
            perturbed = initial_h.copy()
            perturbed += np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            alternatives.append(perturbed)
        return alternatives
    
    # Main optimization loop
    alternative_points = generate_alternative_points(initial_h)
    best_c5 = np.inf
    best_h = initial_h
    best_n_points = n_points
    
    for point in alternative_points:
        result, elapsed_time = optimize_function(point)
        if elapsed_time < budget_s:
            current_c5 = result["fun"]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result["x"]
                best_n_points = n_points
    
    # If no better solution found, fallback to initial
    if best_c5 == np.inf:
        best_c5 = compute_C5(initial_h)
        best_h = initial_h
        best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
