    n_points = len(initial_h)
    dx = 2.0 / n_points
    total_mass = n_points / 2
    
    # Objective function: maximize the correlation (we minimize it)
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Constraint: sum(h) = total_mass
    def constraint(h_vals):
        return np.sum(h_vals) - total_mass
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    h_initial[np.random.choice(n_points, size=5, replace=False)] = np.random.rand(5)
    
    # Optimization with simulated annealing for global exploration
    def simulated_annealing():
        best_h = h_initial.copy()
        best_c5 = objective(best_h)
        
        T = 1.0
        T_min = 1e-6
        cooling_rate = 0.999
        
        for step in range(1000):
            T *= cooling_rate
            if T < T_min:
                break
            
            # Perturb the current solution
            perturbation = np.random.normal(0, 0.1, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            
            # Evaluate the perturbed solution
            current_c5 = objective(perturbed_h)
            
            # Acceptance probability
            delta = current_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = perturbed_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Run simulated annealing
    best_h, best_c5 = simulated_annealing()
    
    # Try local optimization with SLSQP
    def local_optimization(h_vals):
        result = minimize(objective, h_vals, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 200})
        return result.x, result.fun
    
    best_h, best_c5 = local_optimization(best_h)
    
    # Ensure the sum is exactly total_mass
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - total_mass) > 1e-6:
        best_h[-1] = total_mass - sum_h + best_h[-1]
    
    # Final evaluation
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
    
    return (best_h, best_c5, n_points)
