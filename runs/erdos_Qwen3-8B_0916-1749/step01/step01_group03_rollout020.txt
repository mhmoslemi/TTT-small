```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    n_points = 120
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points // 3 else 0.5 if i < 2 * n_points // 3 else 1
    initial_h = initial_h + np.random.normal(0, 0.05, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    dx = 2.0 / n_points
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different settings
    def objective(h):
        return compute_C5(h)
    
    h0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Use a more robust optimization method with improved settings
    result = minimize(
        objective,
        h0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        options={"maxiter": 200, "ftol": 1e-8, "gtol": 1e-8, "eps": 1e-8}
    )
    
    elapsed_time = time.time() - start_time
    
    # If within budget, return the best found solution
    if elapsed_time < budget_s:
        best_h = result["x"]
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
```
