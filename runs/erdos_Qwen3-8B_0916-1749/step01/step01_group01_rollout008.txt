import numpy as np
import scipy.optimize as optimize
import time

    np.random.seed(seed)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to optimize with different starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            tol=1e-8
        )
        return result.x, result.fun
    
    # Try multiple starting points to avoid local optima
    best_h = None
    best_c5 = float('inf')
    n_points_list = [100, 150, 200]  # Try different numbers of points
    
    for n in n_points_list:
        # Generate a random starting point
        h_initial = np.random.uniform(0, 1, n)
        h_initial /= np.sum(h_initial) * 2  # Adjust to meet the constraint
        
        # Run the optimization
        start_time = time.time()
        h_opt, c5 = optimize_h(h_initial)
        elapsed_time = time.time() - start_time
        
        # Check if we are within the time budget
        if elapsed_time < budget_s:
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        else:
            print(f"Time exceeded for n={n}, skipping")
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
