import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Try multiple starting points to avoid local optima
    candidates = [initial_h]
    for _ in range(3):
        # Perturb the initial guess
        perturbed = initial_h.copy()
        for i in range(n_points):
            perturbed[i] += np.random.uniform(-0.1, 0.1)
            perturbed[i] = np.clip(perturbed[i], 0, 1)
        candidates.append(perturbed)
    
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    start_time = time.time()
    for h_initial in candidates:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            h_opt, c5 = optimize_h(h_initial)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
        except:
            pass
    
    return best_h, best_c5, len(best_h)
