    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    def optimize_h(initial_guess):
        result = opt.minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else initial_guess
    
    # Initial optimization
    h_best = optimize_h(initial_h)
    c5_best = compute_c5(h_best)
    
    # Try multiple perturbations to escape local optima
    perturbations = [0.01, 0.05, 0.1]
    for p in perturbations:
        perturbed_h = h_best + np.random.uniform(-p, p, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        c5_perturbed = compute_c5(perturbed_h)
        if c5_perturbed < c5_best:
            h_best = perturbed_h
