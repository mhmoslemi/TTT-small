    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        # The relevant part of the correlation corresponds to the overlap integral
        # Extract the relevant part of the correlation (centered around the peak)
        start_idx = max(0, (len(corr) - n_points) // 2)
        end_idx = min(len(corr), start_idx + n_points)
        corr_window = corr[start_idx:end_idx]
        max_corr = np.max(corr_window)
        return max_corr * dx

    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Optimization using scipy.optimize with different starting points
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Evaluate the initial solution
    best_h, best_c5 = optimize_h(initial_h)
    best_c5_val = best_c5
    best_h_values = best_h

    for _ in range(5):
        # Generate a random perturbation of the initial solution
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        new_h = np.clip(initial_h + perturbation, 0, 1)

        h_initial = new_h

        result = optimize.minimize(
