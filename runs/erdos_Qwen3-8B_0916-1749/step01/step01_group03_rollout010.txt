import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)

    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)

    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization using scipy.optimize with bounds and constraints
    h0 = initial_h.copy()
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    start_time = time.time()

    result = optimize.minimize(
        objective,
        h0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"maxiter": 150, "ftol": 1e-8, "eps": 1e-8}
    )

    elapsed_time = time.time() - start_time

    # Try a few local perturbations to escape local optima
    if elapsed_time < budget_s:
        best_h = result["x"]
        best_c5 = compute_C5(best_h)
        
        # Try small random perturbations
        for _ in range(5):
            perturb_h = best_h.copy()
            perturb_h += np.random.normal(0, 0.05, n_points)
            perturb_h = np.clip(perturb_h, 0, 1)

            perturb_c5 = compute_C5(perturb_h)
            if perturb_c5 < best_c5:
                best_c5 = perturb_c5
                best_h = perturb_h

        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
