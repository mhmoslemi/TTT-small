import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different settings
    def optimize_with_adaptive_steps(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        return result["x"], result["fun"]
    
    # Adaptive optimization approach
    def adaptive_optimization():
        current_h = initial_h.copy()
        best_h = current_h
        best_c5 = compute_C5(current_h)
        elapsed = 0
        
        for _ in range(10):
            if elapsed >= budget_s:
                break
            start_time = time.time()
            
            # Optimize from current_h
            optimized_h, c5 = optimize_with_adaptive_steps(current_h)
            
            # Check if the new solution is better
            if c5 < best_c5:
                best_h = optimized_h
                best_c5 = c5
                current_h = best_h
            
            # Add some random perturbations to escape local optima
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = np.clip(current_h + perturb, 0, 1)
            
            # Evaluate the perturbed solution
            perturbed_c5 = compute_C5(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
                current_h = best_h
            
            elapsed = time.time() - start_time
            print(f"Iteration {_ + 1}: Best C5 = {best_c5}, Time elapsed: {elapsed:.2f}s")
        
        return best_h, best_c5, n_points
    
    # Run the adaptive optimization
    best_h, best_c5, n_points = adaptive_optimization()
    
    return (best_h, best_c5, n_points)
