import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import BoundsError
from sklearn.cluster import KMeans

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different starting points
    def optimize_solution(initial_point):
        h0 = initial_point.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        try:
            result = optimize.minimize(
                objective,
                h0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method="SLSQP",
                options={"maxiter": 100, "ftol": 1e-7, "eps": 1e-8}
            )
            elapsed_time = time.time() - start_time
            if elapsed_time < budget_s:
                return result["x"], compute_C5(result["x"])
            else:
                return initial_point, compute_C5(initial_point)
        except BoundsError:
            return initial_point, compute_C5(initial_point)
    
    # Generate diverse starting points using clustering
    def generate_diverse_points(initial_point, n_points, n_clusters=3):
        h_initial = initial_point.copy()
        h_initial = h_initial.reshape(1, -1)
        kmeans = KMeans(n_clusters=n_clusters, n_init=1).fit(h_initial)
        centroids = kmeans.cluster_centers_.flatten()
        return centroids
    
    # Generate multiple diverse starting points
    start_points = []
    for _ in range(3):
        start_points.append(generate_diverse_points(initial_h, n_points))
    
    # Evaluate all starting points and select the best one
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    for start_point in start_points:
        if len(start_point) != n_points:
            continue
        h, c5 = optimize_solution(start_point)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return (best_h, best_c5, n_points)
