    def compute_C5(h):

        return c5

    def objective(h):
        return compute_C5(h)

        return np.sum(h) - n_points / 2

        best_solution = initial_guess.copy()
        best_value = objective(best_solution)

        current_value = best_value

                # Generate a perturbation
                perturbation = np.random.normal(0, 0.05, size=len(current_solution))
                perturbed = np.clip(current_solution + perturbation, 0, 1)
                
                # Check constraint
                if np.abs(constraint(perturbed)) > 1e-6:
                    continue
                
                # Compute new value
                new_value = objective(perturbed)
                
                # Acceptance probability
                if new_value < current_value:
                    current_solution = perturbed
                    current_value = new_value
                    if new_value < best_value:
                        best_solution = perturbed
                        best_value = new_value
                else:
                    if np.random.rand() < np.exp((current_value - new_value) / temp):

                        current_value = new_value
                
            temp *= cooling_rate
            iter_count += 1
        
        return best_solution, best_value
