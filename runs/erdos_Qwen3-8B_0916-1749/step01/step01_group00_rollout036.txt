```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial point with n_points = 128 samples
    n_points = 128
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    
    # Create an initial guess with alternating values to spread out the function
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            h_initial[i] = 0.5
        elif i % 4 == 2:
            h_initial[i] = 0.75
        else:
            h_initial[i] = 0.25
    
    # Ensure the initial sum is correct
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = opt.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, len(h_best)
```
