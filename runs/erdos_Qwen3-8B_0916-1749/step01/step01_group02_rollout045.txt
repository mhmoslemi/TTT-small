import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):

    start_time = time.time()

    initial_h = initial_h_values
    n_points = len(initial_h)

    # Objective function to minimize max correlation
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    h_initial = initial_h.copy()

    # Optimization with some randomness
    def optimize_with_random_perturbations():
        result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
        
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Try several random perturbations
        for _ in range(5):
            perturbed_h = best_h.copy()
            indices = np.random.choice(n_points, size=5, replace=False)
            perturbed_h[indices] = np.clip(perturbed_h[indices] + np.random.uniform(-0.1, 0.1), 0, 1)
            perturbed_c5 = objective(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
        
        # Ensure the sum is exactly n_points / 2
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if abs(sum_h - n_points / 2) > 1e-6:
            best_h[-1] = n_points / 2 - sum_h + best_h[-1]
        
        return best_h, best_c5
