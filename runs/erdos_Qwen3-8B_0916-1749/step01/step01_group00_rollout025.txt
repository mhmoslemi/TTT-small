import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction (n=70)
    n_points = 70
    dx = 2.0 / n_points
    
    # Create initial h with a simple sinusoidal profile
    x = np.linspace(0, 2, n_points)
    initial_h = np.sin(2 * np.pi * x / 2) ** 2
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = (initial_h - np.min(initial_h)) / (np.max(initial_h) - np.min(initial_h))
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a more aggressive optimization strategy with different starting points
    n_trials = 5
    best_c5 = float('inf')
    best_h = initial_h
    
    for _ in range(n_trials):
        # Perturb the initial construction slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = initial_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Use Nelder-Mead simplex algorithm for optimization
        result = opt.minimize(
            objective,
            h_perturbed,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_best = result.x
        c5_best = compute_c5(h_best)
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
    
    return best_h, best_c5, len(best_h)
