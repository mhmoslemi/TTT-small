```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    n_points = 100  # Using 100 points for a balance between accuracy and speed
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        h_bounds = Bounds([0.0] * n_points, [1.0] * n_points)
        
        # Define the optimization problem
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        return result.x, result.fun
    
    # Run the optimization
    best_h, best_c5 = optimize_h(initial_h)
    
    # Return the best solution
    return best_h, best_c5, len(best_h)
```
