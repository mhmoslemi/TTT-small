    def compute_c5(h):

        # Perform optimization with SLSQP
        result = opt.minimize(
            objective,

            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )

        current_c5 = result.fun
