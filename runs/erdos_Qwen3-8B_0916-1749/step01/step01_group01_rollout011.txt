import scipy.optimize as optimize

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Helper function to evaluate C5 for a given h
    def evaluate_h(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the initial guess
    h_initial = initial_h
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint_fun = lambda h: constraint(h)
    
    # Function to perform the optimization
    def optimize_h(h_initial, max_iterations=100):
        best_h = h_initial.copy()
        best_c5 = evaluate_h(best_h)
        
        for _ in range(max_iterations):
            # Perform a local optimization
            result = optimize.minimize(
                fun=evaluate_h,
                x0=h_initial,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_fun}],
                method='L-BFGS-B',
                options={'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            # Update best solution
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            
            # Perturb the solution to escape local minima
            perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            h_initial = perturbed_h
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h_initial)
    
    # Return the best solution
    return best_h, best_c5, len(best_h)
