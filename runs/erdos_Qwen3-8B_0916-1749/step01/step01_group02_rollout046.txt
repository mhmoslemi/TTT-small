import time

import random

def run(seed=42, budget_s=300, **kwargs):

    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    h_initial = initial_h.copy()

    # Optimization with different strategies
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    
    # Best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Try with a different initial guess
    h_initial_2 = np.random.rand(n_points)
    h_initial_2 = np.clip(h_initial_2, 0, 1)
    h_initial_2 = h_initial_2 - np.sum(h_initial_2) / n_points + 0.5
    result_2 = minimize(objective, h_initial_2, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    best_h_2 = result_2.x
    best_c5_2 = objective(best_h_2)

        best_h = best_h_2

    h_initial_3 = np.zeros(n_points)

    h_initial_3 = h_initial_3 - np.sum(h_initial_3) / n_points + 0.5
    result_3 = minimize(objective, h_initial_3, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})

    best_c5_3 = objective(best_h_3)

        best_h = best_h_3

    if elapsed_time > budget_s - 10:
