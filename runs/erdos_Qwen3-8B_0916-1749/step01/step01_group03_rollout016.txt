import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize.minimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define initial_h_values with n_points = 45 (as a base for search)
    initial_h_values = np.array([0.5 if i % 3 == 0 else 0.0 for i in range(45)])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def bounds_func(h):
        return [(0, 1) for _ in range(n_points)]
    
    # Start with the initial solution
    best_h = initial_h_values
    best_c5 = compute_C5(initial_h_values)
    best_n_points = n_points
    
    # Try a few different n_points values (to find the optimal one)
    for n in [45, 60, 75, 90, 105, 120]:
        n_points = n
        dx = 2.0 / n_points
        initial_h = np.array([0.5 if i % 3 == 0 else 0.0 for i in range(n_points)])
        sum_h = np.sum(initial_h)
        
        # Define bounds and constraints
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        # Run optimization
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                best_n_points = n_points
        
        # Early exit if time runs out
        elapsed_time = time.time() - time.time()
        if elapsed_time > budget_s - 10:
            break
    
    return (best_h, best_c5, best_n_points)
