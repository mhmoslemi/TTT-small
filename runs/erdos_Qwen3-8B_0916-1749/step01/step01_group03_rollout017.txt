    def objective(h):

            # Use the initial guess
            current_h = initial_h.copy()

            # Random perturbation
            current_h = initial_h.copy()

            current_h = np.random.uniform(0, 1, n_points)

        current_sum = np.sum(current_h)
        if current_sum != 0:
            current_h = current_h * (n_points / 2) / current_sum
        
        # Run optimization
        result = optimize.minimize(
            objective,
            current_h,
