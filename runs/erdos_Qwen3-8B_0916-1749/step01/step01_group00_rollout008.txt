import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a coarse grid
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess: piecewise constant function with 5 segments
    segments = 5
    segment_size = 2.0 / segments
    for i in range(segments):
        start = i * segment_size
        end = (i + 1) * segment_size
        idx_start = int(start / dx)
        idx_end = int(end / dx)
        for j in range(idx_start, idx_end):
            h[j] = (i + 1) / segments
    
    # Refine the grid iteratively using a gradient-based approach
    for _ in range(5):
        n_points *= 2
        dx = 2.0 / n_points
        h_new = np.zeros(n_points)
        h_new[::2] = h[:len(h) // 2]
        h = h_new
    
    # Use the current h as initial guess
    initial_h = h
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = np.array(h_values)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = opt.minimize(
        objective,
        initial_h,
        method='SLS</span>QP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, len(h_best)
