    n_points = len(initial_h)

    def objective(h_vals):
        h = h_vals
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Constraint: sum(h) = n_points / 2
    def constraint_func(h_vals):
        return np.sum(h_vals) - n_points / 2

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    # Global variables
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Define the function to evaluate a candidate solution
    def evaluate_candidate(h_vals):
        nonlocal best_c5
        nonlocal best_h
        h_vals = np.clip(h_vals, 0, 1)
        sum_h = np.sum(h_vals)
        if abs(sum_h - n_points / 2) > 1e-6:
            h_vals[-1] = n_points / 2 - sum_h + h_vals[-1]
        current_c5 = objective(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        return current_c5

    # Optimization strategy: Use a combination of local optimization and population-based search
    for _ in range(3):
        # Local optimization starting from a perturbed version of current best
        perturbed_h = best_h.copy()
        indices = np.random.choice(n_points, size=5, replace=False)
        perturbed_h[indices] = np.clip(perturbed_h[indices] + np.random.uniform(-0.1, 0.1), 0, 1)
        result = minimize(evaluate_candidate, perturbed_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_func}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})

            current_c5 = objective(current_h)
            if current_c5 < best_c5:

                best_c5 = current_c5

        population_size = 10
        population = [np.clip(best_h + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(population_size)]

            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                h[-1] = n_points / 2 - sum_h + h[-1]
            current_c5 = objective(h)
