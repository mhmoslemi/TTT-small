    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Generate initial h_values with reasonable distribution
        if n_points == 100:
            h = np.zeros(n_points)
            h[20:80] = 1
            h[80:100] = 0.5
            h[:20] = 0.5
            h = np.clip(h, 0, 1)
        elif n_points == 150:
            h = np.zeros(n_points)
            h[30:120] = 1
            h[120:150] = 0.5

        elif n_points == 200:
            h = np.zeros(n_points)
            h[40:160] = 1
            h[160:200] = 0.5
            h[:40] = 0.5
            h = np.clip(h, 0, 1)
        
        # Define the overlap integral function
        def compute_c5(h_values):
            h = np.array(h_values)
            one_minus_h = 1 - h
            corr = signal.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define the objective function for optimization
        def objective(h_values):
            return compute_c5(h_values)
        
        # Define constraints
        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_best = result.x
        
        # Compute the best C5 value
        c5_best = compute_c5(h_best)
        
        # Update best solution
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
            best_n = n_points
        
        # Print progress
        print(f"n_points={n_points}, c5={c5_best:.6f}")
    
    # Return the best solution found
    return best_h, best_c5, best_n
