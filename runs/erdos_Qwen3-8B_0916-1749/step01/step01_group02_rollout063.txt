import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function to minimize max correlation
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization with a mixture of methods
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B", options={"ftol": 1e-8, "maxiter": 100})
    
    # Try a few different starting points for exploration
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Try a few different starting points with small perturbations
    for _ in range(5):
        h_perturb = best_h.copy()
        indices = np.random.choice(n_points, size=5, replace=False)
        h_perturb[indices] += np.random.uniform(-0.1, 0.1)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb[-1] = n_points / 2 - np.sum(h_perturb[:-1])
        perturbed_c5 = objective(h_perturb)
        if perturbed_c5 < best_c5:
            best_h = h_perturb
            best_c5 = perturbed_c5
    
    # Ensure the sum is exactly n_points / 2
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h[-1] = n_points / 2 - sum_h + best_h[-1]
    
    # Check if we have enough time for a more refined optimization
    if time.time() - start_time < budget_s - 10:
        result = minimize(objective, best_h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B", options={"ftol": 1e-9, "maxiter": 50})
        best_h = result.x
        best_c5 = objective(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
    
    return (best_h, best_c5, n_points)
