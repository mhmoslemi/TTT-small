    def adam_minimize(objective, x0, bounds, max_iter=100, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = np.zeros_like(x0)
        v = np.zeros_like(x0)
        for _ in range(max_iter):
            grad = np.gradient(objective, x0, method='central', axis=0)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2

            v_hat = v / (1 - beta2 ** (_ + 1))
            x0 = np.clip(x0 - lr * m_hat / (np.sqrt(v_hat) + epsilon), 0, 1)
            if np.abs(np.sum(x0) - n_points / 2) < 1e-6:

        return x0
