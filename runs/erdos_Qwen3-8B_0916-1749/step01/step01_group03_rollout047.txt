import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):

    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with a different approach
    def optimize_h(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="L-BFGS-B",
            options={"maxiter": 500, "ftol": 1e-10, "gtol": 1e-10}
        )
        if result.success:
            return result.x
        else:
            return h0
    
    # Try multiple starting points and pick the best
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    # Try random perturbations around the initial solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h0 = initial_h + perturbation
        h0 = np.clip(h0, 0, 1)
        h0 = optimize_h(h0)
        current_c5 = compute_C5(h0)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h0
    
    # Try a different starting point with more uniform distribution
    uniform_h = np.repeat(0.5, n_points)
    h0 = optimize_h(uniform_h)
    current_c5 = compute_C5(h0)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h0
    
    # Return the best solution found within the time budget
    return (best_h, best_c5, n_points)
