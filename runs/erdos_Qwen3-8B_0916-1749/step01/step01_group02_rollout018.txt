import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function to minimize max correlation
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    # Initial guess
    h_initial = initial_h.copy()
    h_initial[np.random.choice(n_points, size=5, replace=False)] = np.random.rand(5)
    
    # Optimization
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B", options={"ftol": 1e-8, "maxiter": 200})
    
    # Best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Try different starting points via random perturbations
    for _ in range(10):
        perturbed_h = best_h.copy()
        indices = np.random.choice(n_points, size=5, replace=False)
        perturbed_h[indices] = np.clip(perturbed_h[indices] + np.random.uniform(-0.1, 0.1), 0, 1)
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Try another approach with random initial points
    for _ in range(5):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        perturbed_c5 = objective(h_random)
        if perturbed_c5 < best_c5:
            best_h = h_random
            best_c5 = perturbed_c5
    
    # Ensure the sum is exactly n_points / 2
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h[-1] = n_points / 2 - sum_h + best_h[-1]
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
    
    return (best_h, best_c5, n_points)
