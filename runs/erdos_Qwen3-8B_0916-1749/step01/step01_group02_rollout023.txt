import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):

    start_time = time.time()

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Initial guess (perturbed to avoid local minima)
    h_initial = initial_h.copy()
    h_initial[np.random.choice(n_points, size=5, replace=False)] = np.random.rand(5)
    
    # Optimization using SLSQP
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    n_points_used = n_points
    
    # Explore perturbations and different n_points
    for n_points in [50, 100, 150, 200, 250, 300, 350, 400, 450, 500]:
        # Initialize a new h with n_points samples
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h + np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            n_points_used = n_points
    
    # Ensure the sum is exactly n_points / 2 for the best_h
    best_h = np.clip(best_h, 0, 1)

    if abs(sum_h - n_points_used / 2) > 1e-6:
        best_h[-1] = n_points_used / 2 - sum_h + best_h[-1]
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
    
    return (best_h, best_c5, n_points_used)
