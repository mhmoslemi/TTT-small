import scipy.signal as signal

    # Try different n_points values: 50, 70, 100 to explore better configurations
    n_points_list = [50, 70, 100]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Create initial guess based on initial_h_values scaled to n_points
        if n_points == 70:
            h_initial = initial_h.copy()
        else:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        
        # Define the overlap integral function
        def compute_c5(h):
            h = np.array(h)
            one_minus_h = 1 - h
            corr = signal.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define the objective function for optimization
        def objective(h):
            return compute_c5(h)
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h_best = result.x
        c5_best = compute_c5(h_best)
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
            best_n = n_points
    
    return best_h, best_c5, best_n
