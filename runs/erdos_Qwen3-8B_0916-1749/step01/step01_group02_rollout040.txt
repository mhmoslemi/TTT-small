import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):

    start_time = time.time()

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the objective function to minimize max correlation
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess with some random perturbations
    h_initial = initial_h.copy()
    h_initial[np.random.choice(n_points, size=5, replace=False)] = np.random.rand(5)
    
    # Optimization using SLSQP
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Try a few random perturbations to escape local optima
    for _ in range(5):
        perturbed_h = best_h.copy()
        indices = np.random.choice(n_points, size=5, replace=False)
        perturbed_h[indices] = np.clip(perturbed_h[indices] + np.random.uniform(-0.1, 0.1), 0, 1)
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Optionally try a different starting point with random construction
    if np.random.rand() < 0.3:
        h_new = np.random.uniform(0, 1, n_points)

        if perturbed_c5 < best_c5:
            best_h = h_new

    # Ensure the sum is exactly n_points / 2

    if abs(sum_h - n_points / 2) > 1e-6:
