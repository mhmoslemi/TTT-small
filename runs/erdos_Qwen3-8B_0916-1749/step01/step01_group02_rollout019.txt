        result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})

        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:

    for _ in range(5):
