import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

    n_points = len(initial_h)

    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization settings
    options = {"ftol": 1e-8, "gtol": 1e-8, "maxiter": 200}
    
    # First optimization with SLSQP
    result = optimize.minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options=options)
    
    # Extract best solution from first run
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Optional refinement using differential evolution
    if time.time() - start_time < budget_s - 10:
        def objective_de(h_vals):
            h = h_vals.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Bounds for differential evolution
        de_bounds = [(0, 1) for _ in range(n_points)]
        
        # Run differential evolution
        result_de = optimize.differential_evolution(objective_de, de_bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options=options)
        
        # Update best solution
        best_h = result_de.x

    best_h = np.clip(best_h, 0, 1)

    if abs(sum_h - n_points / 2) > 1e-6:

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
