    def objective(h):

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def sample_grid(n_samples):

    def grid_search(n_samples, max_iter=10):
        best_c5 = float('inf')
        best_h = None
        for _ in range(max_iter):
            h = sample_grid(n_samples)
            # Ensure the sum constraint is approximately satisfied
            h /= np.sum(h)
            c5 = compute_C5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        return best_h, best_c5
    
    # Function to perform a local optimization from a starting point
    def local_optimize(start_h, max_iter=100):
        h = start_h.copy()
        bounds = [(0, 1) for _ in range(len(h))]
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="L-BFGS-B",
            options={"maxiter": max_iter, "ftol": 1e-8, "eps": 1e-8}
        )
        return result["x"], result["fun"]
