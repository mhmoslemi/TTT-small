import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with adjusted parameters
    def optimize_solution(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 500, "ftol": 1e-9, "eps": 1e-9, "disp": False}
        )
        return result["x"]
    
    # Try multiple starting points to avoid local optima
    def generate_starting_points(n_start):
        start_points = []
        for _ in range(n_start):
            perturbed = initial_h.copy()
            for i in range(n_points):
                perturbed[i] += np.random.uniform(-0.1, 0.1)
                perturbed[i] = np.clip(perturbed[i], 0, 1)
            start_points.append(perturbed)
        return start_points
    
    start_points = generate_starting_points(5)
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    for h0 in start_points:
        if time.time() - time.time() > budget_s:
            break
        try:
            optimized_h = optimize_solution(h0)
            current_c5 = compute_C5(optimized_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
    
    return (best_h, best_c5, n_points)
