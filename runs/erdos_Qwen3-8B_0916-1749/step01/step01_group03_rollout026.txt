import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h():
        h0 = initial_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
        else:
            # Fallback to initial solution if time runs out
            best_h = initial_h
            best_c5 = compute_C5(best_h)
            return (best_h, best_c5, n_points)
    
    # Try multiple starting points to avoid local optima
    best_c5 = float('inf')
    best_h = initial_h
    best_n = n_points
    
    for _ in range(5):
        # Perturb the initial point slightly to explore different regions
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(initial_h + perturb, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        h0 = perturbed_h
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_h = result["x"]
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n = n_points
        else:
            # Fallback to initial solution if time runs out
            current_h = initial_h
            current_c5 = compute_C5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n = n_points
    
    return (best_h, best_c5, best_n)
