import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define a new initial point with more diversity
    def generate_diverse_initial(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            elif i % 3 == 0:
                h[i] = 0.5
            else:
                h[i] = np.random.uniform(0, 1)
        return h
    
    # Try multiple initial points to avoid local optima
    initial_points = [initial_h, generate_diverse_initial(n_points)]
    
    best_c5 = float('inf')
    best_h = initial_h
    
    for initial in initial_points:
        h0 = initial.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = time.time()
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time < budget_s:
            current_c5 = compute_C5(result["x"])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result["x"]
    
    return (best_h, best_c5, n_points)
