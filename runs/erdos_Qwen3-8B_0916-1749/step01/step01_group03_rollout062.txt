import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with multiple starting points
    def optimize_with_multiple_starts(n_starts=5):
        best_c5 = float('inf')
        best_h = None
        bounds = [(0, 1) for _ in range(n_points)]
        
        for start_idx in range(n_starts):
            # Generate a perturbed starting point
            perturbation = np.random.normal(0, 0.05, n_points)
            h0 = initial_h + perturbation
            h0 = np.clip(h0, 0, 1)
            
            start_time = time.time()
            
            result = optimize.minimize(
                objective,
                h0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method="SLSQP",
                options={"maxiter": 150, "ftol": 1e-8, "eps": 1e-8}
            )
            
            elapsed_time = time.time() - start_time
            
            if elapsed_time < budget_s:
                current_c5 = result["fun"]
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result["x"]
        
        return best_h, best_c5, n_points
    
    # Run optimization with multiple starts
    best_h, best_c5, n_points = optimize_with_multiple_starts()
    return (best_h, best_c5, n_points)
