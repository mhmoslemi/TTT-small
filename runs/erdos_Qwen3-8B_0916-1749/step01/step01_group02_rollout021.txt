import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Optimization parameters
    max_attempts = 10
    max_iter = 50
    temp = 0.1
    cooling_rate = 0.995
    
    # Objective function to minimize max correlation
    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    best_h = h_initial
    best_c5 = objective(best_h)
    
    # Simulated annealing
    for attempt in range(max_attempts):
        current_h = best_h.copy()
        current_temp = temp * (cooling_rate ** attempt)
        
        for iteration in range(max_iter):
            # Generate a perturbed solution
            perturbed_h = current_h.copy()
            indices = np.random.choice(n_points, size=5, replace=False)
            perturbed_h[indices] = np.clip(perturbed_h[indices] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Ensure the sum is exactly n_points / 2
            sum_perturbed = np.sum(perturbed_h)
            if abs(sum_perturbed - n_points / 2) > 1e-6:
                perturbed_h[-1] = n_points / 2 - sum_perturbed + perturbed_h[-1]
            
            # Compute objective
            perturbed_c5 = objective(perturbed_h)
            
            # Acceptance probability
            delta_c5 = perturbed_c5 - best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / current_temp):
                current_h = perturbed_h
                current_c5 = perturbed_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        
        # After each attempt, refine with local optimization
        result = minimize(objective, best_h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 10})
        best_h = result.x
        best_c5 = objective(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")
    
    return (best_h, best_c5, n_points)
