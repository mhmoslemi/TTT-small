    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Precompute 1 - h
    one_minus_h = 1 - initial_h
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a hybrid approach: first a simulated annealing step to explore globally, then SLSQP for fine-tuning
    def simulated_annealing_step(initial_guess, temp=1.0, cooling_rate=0.995, iterations=1000):
        h = initial_guess.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for _ in range(iterations):
            # Generate a small random perturbation
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            
            # Compute new cost
            new_c5 = compute_c5(perturbed_h)
            
            # Acceptance probability
            if new_c5 < best_c5:
                h = perturbed_h.copy()
                best_h = h.copy()
                best_c5 = new_c5
            else:
                # Metropolis-Hastings acceptance
                prob = np.exp((best_c5 - new_c5) / temp)
                if np.random.random() < prob:
                    h = perturbed_h.copy()
            
            temp *= cooling_rate
        
        return best_h, best_c5
    
    # Perform a global search using simulated annealing
    best_h, best_c5 = simulated_annealing_step(initial_h)
    
    # Now perform a local optimization using SLSQP to refine the solution
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, len(h_best)
