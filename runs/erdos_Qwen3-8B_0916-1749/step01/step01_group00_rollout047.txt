```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Define initial guess as a piecewise constant function with alternating values
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            initial_h[i] = 1.0
        elif i % 5 == 1:
            initial_h[i] = 0.5
        elif i % 5 == 2:
            initial_h[i] = 0.25
        elif i % 5 == 3:
            initial_h[i] = 0.1
        elif i % 5 == 4:
            initial_h[i] = 0.0
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 200
    tol = 1e-6
    
    # Use a more robust optimization algorithm with bounds and constraints
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': tol, 'maxiter': max_iter}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, len(h_best)
```
