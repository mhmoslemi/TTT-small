import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP'
        )
        return result.x, result.fun
    
    # Define a simple grid-based starting point
    def generate_grid_start(n_points):
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if grid[i] < 1:
                h[i] = 0.5
            else:
                h[i] = 0.5
        return h
    
    # Try multiple starting points to avoid local optima
    start_times = []
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    h_start = generate_grid_start(n_points)
    h_opt, c5 = optimize_h(h_start)
    best_h = h_opt
    best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Try initial_h_values
    start_time = time.time()
    h_start = initial_h
    h_opt, c5 = optimize_h(h_start)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Try a random starting point
    start_time = time.time()
    h_start = np.random.rand(n_points)
    h_opt, c5 = optimize_h(h_start)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)

    start_time = time.time()

    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Try a more complex pattern with spikes
    start_time = time.time()

        if i % 5 == 0:
            h_start[i] = 1.0
        elif i % 5 == 1:
            h_start[i] = 0.75
        elif i % 5 == 2:
            h_start[i] = 0.5
        elif i % 5 == 3:
            h_start[i] = 0.25
        else:
            h_start[i] = 0.0
    h_opt, c5 = optimize_h(h_start)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Try a pattern with clusters
    start_time = time.time()
    h_start = np.zeros(n_points)
    for i in range(n_points):
        if i % 10 < 5:
            h_start[i] = 1.0
        else:
            h_start[i] = 0.0
    h_opt, c5 = optimize_h(h_start)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Try a pattern with alternating high and low values
    start_time = time.time()
    h_start = np.array([0.5 if i % 2 == 0 else 0.0 for i in range(n_points)])
    h_opt, c5 = optimize_h(h_start)
    if c5 < best_c5:
        best_h = h_opt
        best_c5 = c5
    start_times.append(time.time() - start_time)
    
    # Verify that we are within the time budget
    total_time = sum(start_times)
    if total_time > budget_s:
        print(f"Time exceeded: {total_time} seconds")
    
    return best_h, best_c5, len(best_h)
