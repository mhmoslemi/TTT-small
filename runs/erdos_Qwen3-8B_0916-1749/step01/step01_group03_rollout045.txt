import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)

    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)

    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization using scipy.optimize
    def optimize_solution(h0, n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        start_time = time.time()
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 150, "ftol": 1e-9, "eps": 1e-9, "disp": False}
        )
        elapsed_time = time.time() - start_time
        return result["x"], compute_C5(result["x"]), elapsed_time

    # Start with initial solution
    h0_initial = initial_h
    best_h, best_c5, elapsed = optimize_solution(h0_initial, n_points)
    best_result = (best_h, best_c5, n_points)

    # Try a few perturbations of the initial solution
    for perturb in range(3):
        # Create a perturbed version of the initial solution
        perturbed_h = initial_h.copy()
        for i in range(n_points):

                perturbed_h[i] = np.clip(perturbed_h[i] + np.random.normal(0, 0.1), 0, 1)
