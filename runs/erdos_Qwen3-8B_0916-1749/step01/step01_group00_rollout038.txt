import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with n=70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    # Function to compute the C5 value
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization function with bounds and constraints
    def objective(h):
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_best = result.x
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, n_points
