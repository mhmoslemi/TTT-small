import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with adaptive bounds
    def optimize_h(h_initial, max_iter=100, tol=1e-5):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': tol}
        )
        return result.x, result.fun
    
    # Adaptive search with multiple initial points
    def adaptive_search(initial_guesses, max_initial_guesses=5, max_iter=50):
        best_h = initial_guesses[0]
        best_c5 = compute_c5(initial_guesses[0])
        elapsed = 0
        
        for i in range(max_initial_guesses):
            if elapsed > budget_s:
                break
            guess = initial_guesses[i] if i < len(initial_guesses) else np.random.uniform(0, 1, n_points)
            h_initial = guess
            start_time = time.time()
            optimized_h, c5 = optimize_h(h_initial)
            elapsed = time.time() - start_time
            if c5 < best_c5:
                best_h = optimized_h
                best_c5 = c5
            print(f"Initial guess {i+1} done, best C5: {best_c5}, elapsed: {elapsed}")
        
        return best_h, best_c5
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h]
    for _ in range(3):
        initial_guesses.append(np.random.uniform(0, 1, n_points))
    
    # Run adaptive search
    best_h, best_c5 = adaptive_search(initial_guesses, max_initial_guesses=5, max_iter=100)
    
    return best_h, best_c5, len(best_h)
