import scipy.optimize as optimize
import time

    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with a more refined approach
    def optimize_with_refinement():
        # Start with the initial guess
        h0 = initial_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        best_h = result["x"]
        best_c5 = compute_C5(best_h)
        
        # Refinement step: perturb and improve with a local search
        for _ in range(5):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_sum = np.sum(perturbed_h)
            if np.abs(perturbed_sum - n_points / 2) > 1e-5:
                perturbed_h += (n_points / 2 - perturbed_sum) / n_points
            result = optimize.minimize(
                objective,
                perturbed_h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method="L-BFGS-B",
                options={"maxiter": 50, "ftol": 1e-8, "eps": 1e-8}
            )
            best_h = result["x"]
            best_c5 = compute_C5(best_h)
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_with_refinement()
    elapsed_time = time.time() - start_time
    
    # If within budget, return the best found solution
    if elapsed_time < budget_s:
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
        best_c5 = compute_C5(best_h)
