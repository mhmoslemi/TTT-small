```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial number of points and step size
    n_points = 100  # Adjust based on time budget
    dx = 2.0 / n_points
    
    # Generate an initial guess with a periodic-like pattern
    def initial_guess(n):
        h = np.zeros(n)
        for i in range(n):
            angle = 2 * np.pi * i / n
            h[i] = 0.5 + 0.5 * np.sin(2 * angle)
        return h
    
    initial_h = initial_guess(n_points)
    
    # Precompute 1 - h
    one_minus_h = 1 - initial_h
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = opt.minimize(
        objective,
        initial_h,
        method='SLS_CONTROLLER',  # Using a more robust controller for optimization
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100, 'disp': False}
    )
    
    h_best = result.x
    
    # Compute the best C5 value
    c5_best = compute_c5(h_best)
    
    # Return the best solution found
    return h_best, c5_best, len(h_best)
```
