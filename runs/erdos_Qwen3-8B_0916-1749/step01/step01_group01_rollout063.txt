    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Compute the correlation (overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max over k of (integral of h(x)(1-h(x+k)) dx)
        # This corresponds to the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Run the optimization with random restarts to avoid local optima
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_n_points = n_points
    
    # Try a few random restarts
    for _ in range(5):

        perturbed_h = np.array(initial_h)

        start_time = time.time()
