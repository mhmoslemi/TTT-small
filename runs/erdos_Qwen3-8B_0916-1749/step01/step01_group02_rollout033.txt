import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h_vals):
        h = h_vals.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    # Run initial optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    best_time = time.time()

    # Try different starting points with perturbations for exploration
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        sum_perturbed = np.sum(perturbed_h)
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            perturbed_h[-1] = n_points / 2 - sum_perturbed + perturbed_h[-1]
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    # Evaluate the best solution
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h[-1] = n_points / 2 - sum_h + best_h[-1]

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded. Returning best solution found so far.")

    return (best_h, best_c5, n_points)
