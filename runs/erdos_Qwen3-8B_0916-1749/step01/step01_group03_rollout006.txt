import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Function to generate a random starting point
    def generate_random_initial(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        return h
    
    # Optimization using scipy.optimize with multiple starting points
    def optimize_with_multiple_starts(n_starts=5):
        best_h = initial_h
        best_c5 = compute_C5(initial_h)
        
        for start_idx in range(n_starts):
            print(f"Starting optimization {start_idx + 1}/{n_starts}")
            h0 = generate_random_initial(n_points)
            bounds = [(0, 1) for _ in range(n_points)]
            
            start_time = time.time()
            
            result = optimize.minimize(
                objective,
                h0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method="SLSQP",
                options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
            )
            
            elapsed_time = time.time() - start_time
            
            if elapsed_time < budget_s:
                current_h = result["x"]
                current_c5 = compute_C5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        
        return best_h, best_c5, n_points
    
    # Run the optimization with multiple starting points
    best_h, best_c5, n_points = optimize_with_multiple_starts()
