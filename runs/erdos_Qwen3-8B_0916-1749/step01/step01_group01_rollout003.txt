    def compute_c5(h):
        h = np.array(h)

        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part of the correlation
        # The overlap integral is the max value in the correlation result
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        best_result = None
        best_c5 = float('inf')
        
        # Perform multiple restarts with perturbations
        for _ in range(5):
            perturbed_h = h_initial + np.random.normal(0, 0.05, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_h,
                bounds=h_bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_result = result.x
