import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with n=70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Create initial h_values with a simple construction
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points // 2 else 0
    
    # Function to compute C5 value
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization algorithm with multiple iterations
    def optimize_h(h_initial, n_iter=10, n_points=70):
        best_h = h_initial.copy()
        best_c5 = compute_c5(best_h)
        for _ in range(n_iter):
            # Create a perturbed version of the initial h
            perturbation = np.random.normal(0, 0.1, n_points)
            h_perturbed = np.clip(h_initial + perturbation, 0, 1)
            
            # Ensure the sum constraint is satisfied
            total = np.sum(h_perturbed)
            scaling_factor = (n_points / 2) / total if total != 0 else 1
            h_perturbed = np.clip(h_perturbed * scaling_factor, 0, 1)
            
            # Compute C5 for the perturbed h
            current_c5 = compute_c5(h_perturbed)
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = current_c5
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(initial_h)
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
