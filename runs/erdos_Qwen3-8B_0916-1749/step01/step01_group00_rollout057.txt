import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Precompute 1 - h

    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define a more aggressive optimization strategy
    def optimize_with_different_starting_points(n_start_points=5):
        best_c5 = float('inf')
        best_h = None
        
        for start_idx in range(n_start_points):
            # Random perturbation around the initial solution
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            start_h = np.clip(initial_h + perturbation, 0, 1)
            
            # Optimize with SLSQP
            result = opt.minimize(
                objective,
                start_h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        
        return best_h, best_c5
    
    # Run the optimization with multiple starting points
    start_time = time.time()
    h_best, c5_best = optimize_with_different_starting_points(n_start_points=5)

    elapsed_time = time.time() - start_time
