import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with a simple initial construction
    initial_h = np.zeros(100)
    initial_h[0:50] = 1.0
    initial_h = initial_h / (2.0 / 100)
    
    # Try different starting points for exploration
    starting_points = [
        initial_h,
        np.zeros(100),
        np.ones(100),
        np.sin(np.linspace(0, np.pi, 100)) / 2 + 0.5,
        np.random.rand(100),
    ]
    
    best_c5 = float('inf')
    best_h = initial_h
    best_n = 100
    
    for start in starting_points:
        n_points = len(start)
        dx = 2.0 / n_points
        
        # Precompute 1 - h
        one_minus_h = 1 - start
        
        # Define the overlap integral function
        def compute_c5(h):
            h = np.array(h)
            one_minus_h = 1 - h
            corr = signal.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define the objective function for optimization
        def objective(h):
            return compute_c5(h)
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
        result = opt.minimize(
            objective,
            start,
            method='SLS.hs',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            h_best = result.x
            c5_best = compute_c5(h_best)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best
                best_n = n_points
    
    return best_h, best_c5, best_n
