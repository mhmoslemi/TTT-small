        points = np.concatenate([points, np.random.uniform(0, 1, n_points)])

        starting_points.append(generate_alternative_starting_points(n_points, seed + s))
    
    best_c5 = float('inf')
    best_h = initial_h
    
    start_time = time.time()
    
    for start in starting_points:
        # Use dual annealing for global optimization
        result = dual_annealing(
            objective_with_h,
