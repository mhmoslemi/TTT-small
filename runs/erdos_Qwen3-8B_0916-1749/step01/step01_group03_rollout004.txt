import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different parameters
    def optimize_with_params(h0, bounds, method="SLSQP", maxiter=100, ftol=1e-8, eps=1e-8):
        start_time = time.time()
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method=method,
            options={"maxiter": maxiter, "ftol": ftol, "eps": eps}
        )
        elapsed_time = time.time() - start_time
        return result, elapsed_time
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with default parameters
    result, elapsed_time = optimize_with_params(initial_h, bounds)
    
    if elapsed_time < budget_s:
        best_h = result["x"]
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
