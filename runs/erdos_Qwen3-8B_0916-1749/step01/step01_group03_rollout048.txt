import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Helper function to generate random perturbations
    def perturb(h, perturb_strength=0.05):
        h = h.copy()
        delta = np.random.normal(0, perturb_strength, h.shape)
        h += delta
        h = np.clip(h, 0, 1)
        return h
    
    # Optimization using scipy.optimize with multiple restarts and perturbations
    def optimize_with_perturbations(initial_h, max_iterations=20, perturb_strength=0.05):
        best_h = initial_h.copy()
        best_c5 = compute_C5(best_h)
        
        for _ in range(max_iterations):
            h = perturb(best_h, perturb_strength)
            result = minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={"type": "eq", "fun": constraint_sum},
                method="L-BFGS-B",
                options={"ftol": 1e-8, "eps": 1e-8, "maxiter": 50}
            )
            current_h = result["x"]
            current_c5 = compute_C5(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    start_time = time.time()
    
    # Run the optimization with multiple restarts
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    elapsed_time = time.time() - start_time
    
    if elapsed_time < budget_s:
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        best_h = initial_h
        best_c5 = compute_C5(best_h)
        return (best_h, best_c5, n_points)
