import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial starting point (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the C_5 value for a given h array
    def compute_C5(h):
        h = np.array(h)
        shifted = np.concatenate([h[-n_points+1:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize with different settings
    h0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Try multiple starting points with perturbations
    best_h = initial_h
    best_c5 = compute_C5(initial_h)
    
    for _ in range(5):
        # Perturb the initial guess
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h0 + perturbation, 0, 1)
        
        result = optimize.minimize(
            objective,
            h_perturbed,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="L-BFGS-B",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result["x"]
        
        # Check elapsed time
        if time.time() - start_time > budget_s - 10:
            break
    
    # If within budget, return the best found solution
    if time.time() - start_time < budget_s:
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution if time runs out
        return (initial_h, compute_C5(initial_h), n_points)
