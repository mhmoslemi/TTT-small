import scipy.optimize as optimize
import time

        return max_corr * dx

    def objective(h):
        return compute_c5(h)

            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )

        h = np.copy(h_base)
