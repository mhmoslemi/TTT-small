import scipy.optimize as optimize
import time

    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method with advanced settings
    def optimize_h(h_initial, method='SLSQP', maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with more complex variation
    def complex_start(h_base):
        h = np.copy(h_base)
        h += np.random.normal(0, 0.03, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    # Perform optimization with multiple starting points and methods
    for i in range(10):
        # Generate a new starting point
        h_initial = evolve_start(best_h)
        
        # Optimize with SLSQP
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:

        try:

                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue

        try:
            optimized_h, current_c5 = optimize_h(h_initial, method='L-BFGS-B', maxiter=500, ftol=1e-8, eps=1e-6)

                best_c5 = current_c5

        except:
            continue
        
        # Perturb and optimize
        for j in range(5):

            perturbed_h = perturb(best_h, seed_perturb)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
