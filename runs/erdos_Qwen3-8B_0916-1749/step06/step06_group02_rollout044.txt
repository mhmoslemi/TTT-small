import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 100 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a random starting point
    def random_start():
        h = np.random.uniform(0, 1, size=n_points)
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a greedy starting point
    def greedy_start():
        h = np.zeros(n_points)
        remaining = n_points / 2
        for i in range(n_points):
            if remaining <= 0:
                break
            if i == n_points - 1:
                h[i] = remaining
            else:
                h[i] = min(1, remaining)
                remaining -= h[i]
        return h
    
    # Function to generate a balanced starting point
    def balanced_start():
        h = np.full(n_points, 0.5)
        return h
    
    # Function to generate a sparse starting point
    def sparse_start():
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1
        return h
    
    # Define a function to perform a grid search with multiple starting points
    def grid_search():
        best_h = initial_h
        best_c5 = compute_c5(initial_h)
        start_time = time.time()
        
        # Define a list of starting points
        start_points = [
            initial_h,
            structured_start(initial_h),
            random_start(),
            greedy_start(),
            balanced_start(),
            sparse_start()
        ]
        
        # Perform optimization for each starting point
        for start_point in start_points:
            try:
                optimized_h, current_c5 = optimize_h(start_point)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform perturbation and optimization
        for i in range(10):
            h_initial = evolve_start(best_h)
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            return best_h, best_c5, len(best_h)
        
        # If not, return the best solution found
        return best_h, best_c5, len(best_h)
    
    # Perform the grid search
    best_h, best_c5, n_points = grid_search()
    
    # Return the best solution found
    return best_h, best_c5, n_points
