import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 100 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with periodic variation
    def periodic_start(h_base):
        h = np.copy(h_base)
        period = len(h) // 2
        h[::period] += np.random.uniform(-0.1, 0.1, size=len(h) // period)
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with block variation
    def block_start(h_base):
        h = np.copy(h_base)
        block_size = len(h) // 4
        h[:block_size] += np.random.uniform(-0.1, 0.1, size=block_size)
        h[block_size:2*block_size] += np.random.uniform(-0.1, 0.1, size=block_size)
        h[2*block_size:3*block_size] += np.random.uniform(-0.1, 0.1, size=block_size)
        h[3*block_size:] += np.random.uniform(-0.1, 0.1, size=block_size)
        h = np.clip(h, 0, 1)
        return h
    
    # Perform a grid search with multiple starting points
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    for i in range(20):
        # Generate a new starting point
        h_initial = structured_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Generate a new starting point with periodic variation
        h_initial = periodic_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Generate a new starting point with block variation
        h_initial = block_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
