import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function with more robust settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to explore multiple starting points with different perturbations
    def explore_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to perform a local search from a given solution
    def local_search(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100}
        )
        return result.x, result.fun
    
    # Define a function to perform a global search over multiple perturbations
    def global_search(h_initial):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        for i in range(10):
            perturbed_h = np.clip(h_initial + np.random.normal(0, 0.1, size=len(h_initial)), 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Define a function to perform a multi-start optimization
    def multi_start_optimization(h_initial, n_starts=10):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        for _ in range(n_starts):
            perturbed_h = np.clip(h_initial + np.random.normal(0, 0.1, size=len(h_initial)), 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Define the stepwise refinement process
    def refine_solution(h_current, n_refine=5):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_refine):
            # Generate a new starting point
            h_initial = structured_start(best_h)
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
            
            # Perform local search from the current solution
            optimized_h_local, _ = local_search(best_h)
            if compute_c5(optimized_h_local) < best_c5:
                best_h = optimized_h_local
        
        return best_h, best_c5
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform refinement process
    start_time = time.time()
    for refine_iter in range(5):
        best_h, best_c5 = refine_solution(best_h)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Perform global search from the current best solution
    best_h, best_c5 = global_search(best_h)
    
    # Perform multi-start optimization from the current best solution
    best_h, best_c5 = multi_start_optimization(best_h)
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
