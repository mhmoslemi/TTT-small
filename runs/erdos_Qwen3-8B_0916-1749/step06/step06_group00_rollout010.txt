import time

    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a new starting point
    def generate_starting_point():
        return np.random.uniform(0, 1, size=n_points)

    def coordinate_descent(h, max_iter=100):
        h = np.array(h)
        for _ in range(max_iter):
            for i in range(n_points):
                h[i] = np.clip(h[i] - 0.05 * (compute_c5(h) - compute_c5(np.roll(h, 1))) / (np.correlate(h, np.ones_like(h), mode='full')[i] - np.correlate(h, np.ones_like(h), mode='full')[i - 1]), 0, 1)
        return h

    def simulated_annealing(h_initial, temp=1.0, cooling_rate=0.995, max_iter=100):
        h = np.array(h_initial)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, 0.05, size=n_points)

            delta = new_c5 - best_c5

                h = perturbed_h

                    best_h = h.copy()
                    best_c5 = new_c5
            temp *= cooling_rate
        return best_h, best_c5

        return h, compute_c5(h)

            best_c5 = current_c5
