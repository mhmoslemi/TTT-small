import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on the current best
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with some structured variation
    def smart_start(h_base, perturbation):
        h = np.copy(h_base)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        h += perturbation
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate multiple perturbations
    def generate_perturbations(h_base, num_perturbations=10, std_dev=0.05):
        perturbations = np.random.normal(0, std_dev, (num_perturbations, len(h_base)))
        return [smart_start(h_base, p) for p in perturbations]

    best_h = initial_h
