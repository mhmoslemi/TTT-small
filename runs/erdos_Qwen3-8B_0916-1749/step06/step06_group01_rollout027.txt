    def compute_c5(h):

        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with random variation
    def random_start():

    def local_search(h_current, n_search=10):

                best_h = perturbed
                best_c5 = current_c5

    def refine_solution(h_current, n_refine=5):

            h_initial = structured_start(best_h)

                best_c5 = current_c5
