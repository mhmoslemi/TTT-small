import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial, max_iter=200, tol=1e-10):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tol, 'maxiter': max_iter, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    def evolve_start(h):
        h = np.copy(h)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    def structured_start(h_base):
        h = np.copy(h_base)
        offset = np.random.uniform(-0.1, 0.1, size=len(h))
        h += offset
        h = np.clip(h, 0, 1)
        return h
    
    best_h = h_initial
    best_c5 = compute_c5(h_initial)
    start_time = time.time()
    
    for i in range(10):
        h_initial = structured_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial, max_iter=200, tol=1e-10)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        for j in range(5):
            perturbed_h = evolve_start(optimized_h)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h, max_iter=200, tol=1e-10)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
