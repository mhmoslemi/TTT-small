import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 100 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a new starting point with some structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with some structured variation
    def perturbation_start(h_base, perturb_scale):
        h = np.copy(h_base)
        h += np.random.normal(0, perturb_scale, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to check if the solution is within the target range
    def is_target_reached(current_c5):
        return current_c5 <= 0.380876 + 1e-7
    
    # Define a function to perform a grid search with multiple starting points
    def grid_search(start_h, perturb_scale_range, num_perturbations, num_iterations):
        best_h = start_h
        best_c5 = compute_c5(start_h)
        for _ in range(num_iterations):
            for scale in perturb_scale_range:
                for _ in range(num_perturbations):
                    perturbed_h = perturbation_start(best_h, scale)
                    optimized_h, current_c5 = optimize_h(perturbed_h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                        if is_target_reached(best_c5):
                            return best_h, best_c5
        return best_h, best_c5
    
    # Define the main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform a grid search with multiple starting points
    perturb_scale_range = np.logspace(-3, 0, 10)
    num_perturbations = 10
    num_iterations = 50
    
    best_h, best_c5 = grid_search(
        start_h=best_h,
        perturb_scale_range=perturb_scale_range,
        num_perturbations=num_perturbations,
        num_iterations=num_iterations
    )
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
