import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    # Import initial construction
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with tighter tolerances
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with structured perturbations
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with random perturbations
    def random_start():
        h = np.random.uniform(0, 1, size=n_points)
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate a new starting point with initial construction
    def initial_start():
        return np.copy(initial_h_values)
    
    # Perform a series of optimizations with different starting points
    best_h = initial_h_values
    best_c5 = compute_c5(initial_h_values)
    start_time = time.time()
    
    for i in range(15):
        # Try initial construction
        try:
            optimized_h, current_c5 = optimize_h(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Try random starting point
        try:
            h_initial = random_start()
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Try perturbed starting point
        try:
            h_initial = evolve_start(best_h)
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
