import scipy.optimize as optimize

        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):

        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    def structured_start(h_base):
        h = np.copy(h_base)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    def perturb_start(h_base, perturbation_strength):
        h = np.copy(h_base)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
