    def objective(h):

    bounds = [(0, 1) for _ in range(n_points)]

        )

    def structured_perturbation(h):

        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    def random_start():
        h = np.random.uniform(0, 1, size=n_points)

    best_h = initial_h
