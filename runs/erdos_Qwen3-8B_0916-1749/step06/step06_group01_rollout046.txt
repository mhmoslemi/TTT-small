import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to generate structured starting points
    def structured_start():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 0.5
            elif i % 3 == 1:
                h[i] = 0.75
            else:
                h[i] = 0.25
        return h
    
    # Define a function to generate a new starting point with some structured variation
    def evolve_start(h_current):
        h = np.copy(h_current)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define the stepwise refinement process
    def refine_solution(h_current, n_refine=5):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_refine):
            # Generate a new starting point
            h_initial = evolve_start(best_h)
            
            # Perform optimization with tighter tolerances
            result = minimize(
                fun=objective,
                x0=h_initial,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
            )
            
            optimized_h = result.x
            current_c5 = result.fun
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
            
            # Perform perturbation and optimization
            for j in range(5):
                seed_perturb = seed + _ * 10 + j
                perturbed_h = perturb(optimized_h, seed_perturb)
                
                result = minimize(
                    fun=objective,
                    x0=perturbed_h,
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    method='SLSQP',
                    options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
                )
                
                optimized_perturbed = result.x
                current_c5_perturbed = result.fun
                
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed

    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform refinement process
    start_time = time.time()
    for refine_iter in range(5):
        best_h, best_c5 = refine_solution(best_h)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
