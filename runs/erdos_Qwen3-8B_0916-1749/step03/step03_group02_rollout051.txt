import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initialize with the provided initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Placeholder for actual initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more robust and efficient approach
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a random perturbation
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Start with the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to avoid local minima
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try to refine the solution with a local search
    def local_search(h):
        h = np.copy(h)
        for _ in range(100):
            perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
            new_h = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(new_h)
            if new_c5 < compute_c5(h):
                h = new_h
        return h
    
    refined_h = local_search(best_h)
    refined_c5 = compute_c5(refined_h)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
