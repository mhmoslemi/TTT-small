import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate a random perturbation for the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a grid-based initial guess
    def grid_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    # Function to generate a random initial guess
    def random_initial_guess(n_points):
        return np.random.uniform(0, 1, n_points)
    
    # Function to generate a grid-based initial guess with flat regions
    def flat_grid_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 0.25
            else:
                h[i] = 0.75
        return h
    
    # Function to generate a sigmoid-based initial guess
    def sigmoid_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = 1 / (1 + np.exp(-10 * (x - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    # Try multiple different starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    
    # Define a list of initial guess strategies
    initial_guess_strategies = [
        lambda n: grid_initial_guess(n),
        lambda n: random_initial_guess(n),
        lambda n: flat_grid_initial_guess(n),
        lambda n: sigmoid_initial_guess(n),
        lambda n: initial_h_values
    ]
    
    # Try multiple starting points to explore the search space
    for strategy in initial_guess_strategies:
        n_points = len(initial_h_values)
        h_initial = strategy(n_points)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
                best_n_points = n_points
        except:
            continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
