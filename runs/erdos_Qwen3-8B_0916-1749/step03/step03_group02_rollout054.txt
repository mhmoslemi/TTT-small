import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define a more efficient and robust initial construction with 200 samples
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate a simple initial construction: sinusoidal-like function with 0.5 baseline
    initial_h_values = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate a random perturbation
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Try a range of starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution
    best_c5 = compute_c5(best_h)
    
    # Try a few perturbations to explore the search space
    for i in range(10):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try a few additional random starting points
    for i in range(5):
        seed_start = seed + 10 + i
        np.random.seed(seed_start)
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        # Enforce the constraint: sum(h) = n_points / 2
        h_initial = h_initial - (np.sum(h_initial) - n_points / 2) / n_points
        h_initial = np.clip(h_initial, 0, 1)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Final optimization with the best solution found
    try:
        optimized_h, final_c5 = optimize_h(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = optimized_h
    except:
        pass
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
