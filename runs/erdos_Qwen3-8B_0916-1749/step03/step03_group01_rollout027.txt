import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Try multiple starting points and perform a grid search
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_h_values = best_h
    best_n_points = n_points
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a greedy starting point
    def greedy_start():
        h = np.random.uniform(0, 1, size=n_points)
        h[h < 0.5] = 0
        h[h >= 0.5] = 1
        return h
    
    # Define a function to generate a periodic starting point
    def periodic_start():
        h = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        return h
    
    # Perform a grid search with multiple starting points
    start_time = time.time()
    for i in range(5):
        # Generate a new starting point
        h_initial = random_start()
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_values = optimized_h
                best_n_points = n_points
        except:
            continue
        
        # Perform optimization on the greedy start
        h_initial_greedy = greedy_start()
        try:
            optimized_h_greedy, current_c5_greedy = optimize_h(h_initial_greedy)
            if current_c5_greedy < best_c5:
                best_c5 = current_c5_greedy
                best_h_values = optimized_h_greedy
                best_n_points = n_points
        except:
            continue
        
        # Perform optimization on the periodic start
        h_initial_periodic = periodic_start()
        try:
            optimized_h_periodic, current_c5_periodic = optimize_h(h_initial_periodic)
            if current_c5_periodic < best_c5:
                best_c5 = current_c5_periodic
                best_h_values = optimized_h_periodic
                best_n_points = n_points
        except:
            continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h_values, best_c5, best_n_points
