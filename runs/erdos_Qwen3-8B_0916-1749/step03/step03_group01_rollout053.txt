import scipy.optimize as optimize
import time

    # Function to compute the overlap integral
    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to perform simulated annealing
    def simulated_annealing(initial_h, T_initial=1.0, T_final=1e-6, n_iter=1000):
        h = initial_h.copy()
        current_c5 = compute_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        T = T_initial
        for i in range(n_iter):
            if T < T_final:
                break

            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            perturbed = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(perturbed)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = perturbed
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            T *= 0.995
        return best_h, best_c5
    
    # Initialize best solution
    best_h = initial_h
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for i in range(5):

        h_initial = random_start()

        optimized_h, current_c5 = simulated_annealing(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

        for j in range(5):

            perturbed_h = perturb(optimized_h, seed_perturb)

            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
