    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')

        max_corr = np.max(corr)

    def objective(h):
        return compute_c5(h)

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]

        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate diverse starting points
    def generate_diverse_points(initial_h, num_points=5, perturbation_range=0.3):
        points = [initial_h]
        for _ in range(num_points):
            perturbation = np.random.uniform(-perturbation_range, perturbation_range, n_points)
            perturbed_h = np.clip(initial_h + perturbation, 0, 1)
            points.append(perturbed_h)
        return points
    
    # Generate diverse starting points
    diverse_points = generate_diverse_points(initial_h, num_points=10, perturbation_range=0.4)
    
    # Optimize from each starting point
    best_h = None
    best_c5 = float('inf')

        # Optimize from the starting point
        new_h, new_c5 = optimize_h(h_initial)

        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
