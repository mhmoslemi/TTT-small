import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings using CVXOPT
    def optimize_h(h_initial):
        from cvxopt import matrix, solvers
        solvers.options['show_progress'] = False
        
        n = n_points
        h = np.array(h_initial)
        A_ub = np.zeros((n, n))
        for i in range(n):
            A_ub[i, i] = 1
        b_ub = np.array([1.0])
        
        c = np.zeros(n)
        for i in range(n):
            c[i] = -objective(h_initial)
        
        A_eq = np.zeros((1, n))
        A_eq[0, :] = 1.0
        b_eq = np.array([n_points / 2])
        
        lb = np.zeros(n)
        ub = np.ones(n)
        
        sol = solvers.lp(c, A_ub, b_ub, A_eq=A_eq, b_eq=b_eq, lb=lb, ub=ub)
        if 'status' in sol and sol['status'] != 'optimal':
            return h_initial, compute_c5(h_initial)
        else:
            h_opt = np.array(sol['x'])
            return h_opt, compute_c5(h_opt)
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Initialize the best solution
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    # Perform optimization with multiple starting points and perturbations
    start_time = time.time()
    
    # Run multiple optimizations from different initial points
    for i in range(10):
        # Generate a new starting point
        h_initial = random_start()
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform perturbation and optimization
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbed_h = perturb(optimized_h, seed_perturb)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
