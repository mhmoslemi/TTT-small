import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different algorithmic strategies
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to generate a more structured starting point
    def structured_start():
        # Create a piecewise constant function with 5 intervals
        intervals = np.linspace(0, 2, 6)
        values = np.zeros(n_points)
        for i in range(5):
            mask = (intervals[i] <= intervals) & (intervals <= intervals[i+1])
            values[mask] = 0.5
        return values
    
    # Try multiple starting points and perform a grid search
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    # Try a structured start
    h_initial = structured_start()
    optimized_h, current_c5 = optimize_h(h_initial)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h
    
    # Try random start
    h_initial = random_start()
    optimized_h, current_c5 = optimize_h(h_initial)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h
    
    # Try perturbing the initial construction
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(initial_h, seed_perturb)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed
    
    # Try optimization with multiple restarts
    for i in range(5):
        h_initial = random_start()
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Try restarting with the best solution so far
    for i in range(3):
        h_initial = best_h
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        pass
    
    return best_h, best_c5, len(best_h)
