import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    dx = 2.0 / n_points
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different methods
    def optimize_h(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-12, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Try different starting points and optimization methods
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    best_method = 'SLSQP'
    
    # Evaluate the initial solution
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution with different methods
    for method in ['SLSQP', 'L-BFGS-B']:
        try:
            optimized_h, current_c5 = optimize_h(best_h, method=method)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try perturbations and evaluate with different methods
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        for method in ['SLSQP', 'L-BFGS-B']:
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h, method=method)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
