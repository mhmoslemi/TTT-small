import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a new start based on improved heuristic
    def improved_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 0.7
            elif i % 5 == 1:
                h[i] = 0.5
            elif i % 5 == 2:
                h[i] = 0.3
            elif i % 5 == 3:
                h[i] = 0.2
            else:
                h[i] = 0.1
        return h
    
    # Try multiple starting points to avoid local minima
    best_h = improved_initial_guess()
    best_c5 = compute_c5(best_h)
    
    # Evaluate the initial solution
    best_c5 = compute_c5(best_h)
    
    # Try a few perturbations to explore the search space
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Additional annealing to refine the solution
    def simulated_annealing(h_initial, temp=1.0, cooling_rate=0.995, iterations=100):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for _ in range(iterations):
            temp *= cooling_rate
            if temp < 1e-10:
                break
            perturbation = np.random.normal(0, 0.05, size=len(h))
            new_h = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(new_h)
            if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                h = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
        return best_h, best_c5
    
    try:
        optimized_h, current_c5 = optimize_h(best_h)
        best_h = optimized_h
        best_c5 = current_c5
    except:
        pass
    
    try:
        best_h, best_c5 = simulated_annealing(best_h)
    except:
        pass
    
    return best_h, best_c5, len(best_h)
