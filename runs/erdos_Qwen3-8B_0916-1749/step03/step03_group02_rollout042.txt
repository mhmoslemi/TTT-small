import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a more effective initial construction
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            initial_h_values[i] = 1.0
        elif i % 3 == 0:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.2
    
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to avoid local minima
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try an alternative initial construction
    alt_initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i % 7 == 0:
            alt_initial_h_values[i] = 1.0
        elif i % 5 == 0:
            alt_initial_h_values[i] = 0.5
        else:
            alt_initial_h_values[i] = 0.3
    
    best_alt_h = alt_initial_h_values
    best_alt_c5 = compute_c5(best_alt_h)
    
    # Try multiple starting points for the alternative construction
    for i in range(5):
        seed_perturb = seed + i + 5
        perturbed_h = perturb(best_alt_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
