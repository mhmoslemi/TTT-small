import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to create a perturbed version of the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to create a piecewise constant function with given breakpoints
    def create_piecewise_constant(breakpoints, n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < breakpoints[0]:
                h[i] = 0.0
            elif i < breakpoints[1]:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    # Function to evaluate a step function with given breakpoints
    def evaluate_step_function(breakpoints, n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < breakpoints[0]:
                h[i] = 0.0
            elif i < breakpoints[1]:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    # Function to find optimal breakpoints using binary search and gradient descent
    def find_optimal_breakpoints(n_points, max_iter=100, learning_rate=0.01):
        # Initial guess
        breakpoints = [n_points // 2]
        for _ in range(max_iter):
            # Evaluate the current function
            h = evaluate_step_function(breakpoints, n_points)
            c5 = compute_c5(h)
            
            # Compute the gradient
            grad = np.zeros(len(breakpoints))
            for i in range(len(breakpoints)):
                # Perturb the breakpoints
                perturbed_breakpoints = breakpoints.copy()
                perturbed_breakpoints[i] += 1
                h_perturbed = evaluate_step_function(perturbed_breakpoints, n_points)
                c5_perturbed = compute_c5(h_perturbed)
                grad[i] = c5_perturbed - c5
            
            # Update the breakpoints
            for i in range(len(breakpoints)):
                breakpoints[i] += learning_rate * grad[i]
        
        # Ensure the breakpoints are within the valid range
        breakpoints = np.clip(breakpoints, 1, n_points - 1)
        return breakpoints
    
    # Main optimization loop
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try different starting points and find the optimal breakpoints
    for start in range(5):
        np.random.seed(seed + start)
        try:
            # Create a perturbed version of the initial solution
            perturbed_h = perturb(initial_h_values, seed + start)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
            
            # Try finding optimal breakpoints using gradient descent
            breakpoints = find_optimal_breakpoints(n_points)
            h = evaluate_step_function(breakpoints, n_points)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        except:
            continue
    
    return best_h, best_c5, n_points
