import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial samples with 70 points (as provided)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    # Perturb initial solution to avoid local minima
    initial_h_values = np.random.uniform(0, 1, n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = np.sort(initial_h_values)
    initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), initial_h_values)
    initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), initial_h_values)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the optimization problem
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-10
    num_perturbations = 5
    
    # Function to perform optimization with adaptive step size
    def optimize_h(h_initial, perturb_factor=0.1):
        h_bounds = [(0, 1) for _ in range(n_points)]
        initial_step = 0.1 / np.sqrt(n_points)
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': tolerance, 'gtol': tolerance, 'maxiter': max_iter, 'disp': False}
        )
        
        # Apply perturbations to escape local minima
        perturbed_h = np.clip(h_initial + np.random.uniform(-perturb_factor, perturb_factor, size=n_points), 0, 1)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': tolerance, 'gtol': tolerance, 'maxiter': max_iter, 'disp': False}
        )
        
        return result_perturbed.x, result_perturbed.fun
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to avoid local minima
    for i in range(num_perturbations):
        perturbed_h = np.clip(best_h + np.random.uniform(-0.1, 0.1, n_points), 0, 1)
        optimized_h, current_c5 = optimize_h(perturbed_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Return the best solution found
    return best_h, best_c5, n_points
