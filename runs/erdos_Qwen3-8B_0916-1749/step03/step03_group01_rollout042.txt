import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to perform simulated annealing on the solution
    def simulated_annealing(h_initial, T, cooling_rate, max_iter):
        h = h_initial.copy()
        current_c5 = compute_c5(h)
        best_h = h
        best_c5 = current_c5
        
        for _ in range(max_iter):
            T *= cooling_rate
            if T < 1e-10:
                break
            # Generate a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            new_h = np.clip(h + perturbation, 0, 1)
            new_c5 = compute_c5(new_h)
            
            # Accept the new solution with probability depending on temperature
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h
        
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Optimization steps
    for _ in range(3):
        # Generate a random starting point
        h_initial = random_start()
        
        # Perform a local optimization
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Perform simulated annealing on the optimized solution
        sa_h, sa_c5 = simulated_annealing(optimized_h, T=1.0, cooling_rate=0.995, max_iter=100)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_h = sa_h
    
    # Final perturbation and optimization
    for _ in range(5):
        seed_perturb = seed + 100 * _
        perturbed_h = perturb(best_h, seed_perturb)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
