import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define a function to perform a local search
    def local_search(h, seed, max_iter=50):
        np.random.seed(seed)
        h_current = h.copy()
        best_h = h_current
        best_c5 = compute_c5(h_current)
        for _ in range(max_iter):
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h_current))
            h_perturbed = np.clip(h_current + perturbation, 0, 1)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
            h_current = best_h
        return best_h, best_c5
    
    # Define a function to perform a grid search
    def grid_search(h_initial, num_points=5):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        for i in range(num_points):
            h_initial = random_start()
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
            for j in range(num_points):
                seed_perturb = seed + i * 10 + j
                perturbed_h = perturb(optimized_h, seed_perturb)
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_h = optimized_perturbed
                    best_c5 = current_c5_perturbed
        return best_h, best_c5
    
    # Define a function to perform a large-scale grid search with more points
    def large_grid_search(h_initial, num_points=10, max_iter=5):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        for _ in range(max_iter):
            h_initial = random_start()
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
            for j in range(num_points):
                seed_perturb = seed + _ * 10 + j
                perturbed_h = perturb(optimized_h, seed_perturb)
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_h = optimized_perturbed
                    best_c5 = current_c5_perturbed
        return best_h, best_c5

    # Then, perform a grid search with more points
    best_h, best_c5 = grid_search(best_h, num_points=5)
    
    # Finally, perform a large-scale grid search with more points
    best_h, best_c5 = large_grid_search(best_h, num_points=10, max_iter=5)
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
