import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with tighter tolerances
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to create a random initial guess
    def create_random_initial():
        return np.random.uniform(0, 1, n_points)
    
    # Evaluate the initial solution
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    best_c5_val = best_c5
    
    # Try multiple starting points to avoid local minima
    for i in range(5):
        h_initial = create_random_initial()
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5_val:
                best_c5_val = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try to refine the best solution
    refined_h = best_h
    for i in range(3):
        refined_h = optimize_h(refined_h)[0]
    
    # Evaluate the final solution
    final_c5 = compute_c5(refined_h)
    
    return refined_h, final_c5, len(refined_h)
