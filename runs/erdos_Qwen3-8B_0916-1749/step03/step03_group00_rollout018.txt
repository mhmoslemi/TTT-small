import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial h_values with n_points = 200 and a sine-like profile
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    initial_h = np.sin(2 * np.pi * x / 2) ** 2
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = np.minimum(initial_h, 1)
    initial_h = np.maximum(initial_h, 0)
    initial_h = (initial_h - np.min(initial_h)) / (np.max(initial_h) - np.min(initial_h))
    initial_h = np.minimum(initial_h, 1)
    initial_h = np.maximum(initial_h, 0)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        return result.x, result.fun
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(initial_h)
    
    # Perturb and optimize multiple times with different starting points
    for _ in range(5):
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.2, 0.2, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        # Optimize from the perturbed point
        new_h, new_c5 = optimize_h(perturbed_h)
        
        # Update best solution if improvement
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    
    # Fine-tuning with a lower learning rate
    for _ in range(3):
        # Generate small perturbations
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        # Optimize from the perturbed point
        new_h, new_c5 = optimize_h(perturbed_h)
        
        # Update best solution if improvement
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
    
    # Return the best solution
    return best_h, best_c5, len(best_h)
