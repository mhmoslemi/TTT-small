import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with adaptive bounds
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Define a function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Define an initial guess based on a dynamically adaptive construction
    def adaptive_initial_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
        return h
    
    # Initialize with adaptive construction
    initial_h = adaptive_initial_guess(n_points)
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Try multiple starting points to avoid local minima
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Final optimization with a more refined initial guess
    refined_initial = best_h
    refined_h, refined_c5 = optimize_h(refined_initial)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return best_h, best_c5, len(best_h)
