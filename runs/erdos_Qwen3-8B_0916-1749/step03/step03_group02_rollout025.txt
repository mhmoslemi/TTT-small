import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 100 samples
    n_points = 100
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate a random initial guess
    def generate_initial_guess():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the initial solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Run multiple optimizations with different starting points
    best_c5 = 1.0
    best_h = np.zeros(n_points)
    
    # Try multiple starting points
    for i in range(10):
        seed_perturb = seed + i
        initial_guess = generate_initial_guess()
        
        # Optimize
        optimized_h, current_c5 = optimize_h(initial_guess)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Further refine the solution with a more precise optimization
    optimized_h_final, final_c5 = optimize_h(best_h)
    
    return optimized_h_final, final_c5, len(optimized_h_final)
