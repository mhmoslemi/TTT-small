```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        return result.x, result.fun
    
    # Function to generate a random perturbation
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate a smooth piecewise constant function
    def generate_smooth_piecewise_const(n_points, seed):
        np.random.seed(seed)
        h = np.array([0.5] * n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = np.random.uniform(0, 1)
        return h
    
    # Try multiple starting points to avoid local minima
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    
    # Try random perturbations
    for i in range(5):
        seed_perturb = seed + i
        perturbed_h = perturb(best_h, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try smooth piecewise constant functions
    for i in range(5):
        seed_perturb = seed + i + 5
        smooth_h = generate_smooth_piecewise_const(n_points, seed_perturb)
        try:
            optimized_h, current_c5 = optimize_h(smooth_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    return best_h, best_c5, len(best_h)
```
