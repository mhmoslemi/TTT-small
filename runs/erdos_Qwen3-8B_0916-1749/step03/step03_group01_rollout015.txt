import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to perturb the solution
    def perturb(h, seed):
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    # Function to generate a new starting point with a specific pattern
    def pattern_start():
        pattern = np.concatenate([
            np.full(n_points // 4, 0.5),
            np.full(n_points // 4, 1.0),
            np.full(n_points // 4, 0.5),
            np.full(n_points // 4, 0.0)
        ])
        return pattern[:n_points]
    
    # Function to perform a more systematic search
    def systematic_search():
        best_h = initial_h
        best_c5 = compute_c5(initial_h)
        
        # Split the time budget between random and pattern searches
        time_budget = budget_s * 0.5
        
        # Perform systematic search with pattern-based starting points
        start_time = time.time()
        for i in range(5):
            # Generate a new pattern-based starting point
            h_initial = pattern_start()
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            # Check if time budget is exceeded
            if time.time() - start_time > time_budget:
                break
        
        # Perform random search with the remaining time budget
        start_time = time.time()
        for i in range(5):
            # Generate a new random starting point
            h_initial = random_start()
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            # Check if time budget is exceeded
            if time.time() - start_time > time_budget:
                break
        
        return best_h, best_c5
    
    # Perform a more systematic search
    best_h, best_c5 = systematic_search()
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
