import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with bounds and constraints
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbation
    def perturb(h, strength):
        perturbed = np.copy(h)
        noise = np.random.normal(0, strength, size=len(h))
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        return perturbed
    
    # Function to perform simulated annealing
    def simulated_annealing(h_initial, temperature=1.0, cooling_rate=0.995, max_iter=1000):
        current_h = np.copy(h_initial)
        current_c5 = compute_c5(current_h)
        best_h = np.copy(current_h)
        best_c5 = current_c5
        T = temperature
        
        for _ in range(max_iter):
            T *= cooling_rate
            if T < 1e-6:
                break
            perturbation_strength = T * 0.5
            perturbed_h = perturb(current_h, perturbation_strength)
            perturbed_c5 = compute_c5(perturbed_h)
            
            # Acceptance probability
            delta = perturbed_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = perturbed_h
                current_c5 = perturbed_c5
                if current_c5 < best_c5:
                    best_h = np.copy(current_h)
                    best_c5 = current_c5
        
        return best_h, best_c5
    
    # Function to perform a tabu search with adaptive neighborhood
    def tabu_search(h_initial, tabu_size=5, max_iter=100):
        h = np.copy(h_initial)
        best_h = np.copy(h)
        best_c5 = compute_c5(h)
        tabu_list = []
        
        for _ in range(max_iter):
            # Generate neighbors by altering values in the neighborhood
            neighbors = []
            for i in range(len(h)):
                for delta in [-0.1, 0.1]:
                    neighbor = np.copy(h)
                    neighbor[i] += delta
                    neighbor = np.clip(neighbor, 0, 1)
                    neighbors.append(neighbor)
            
            # Select best neighbor not in tabu list
            best_neighbor = None
            best_neighbor_c5 = float('inf')
            
            for neighbor in neighbors:
                if np.any(np.abs(neighbor - h) < 1e-6):
                    continue  # Skip identical to current
                if not np.any(np.abs(neighbor - np.array(tabu_list)) < 1e-6):
                    neighbor_c5 = compute_c5(neighbor)
                    if neighbor_c5 < best_neighbor_c5:
                        best_neighbor = neighbor
                        best_neighbor_c5 = neighbor_c5
            
            if best_neighbor is not None:
                h = np.copy(best_neighbor)
                current_c5 = best_neighbor_c5
                if current_c5 < best_c5:
                    best_h = np.copy(h)
                    best_c5 = current_c5
                tabu_list.append(h)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        
        return best_h, best_c5
    
    # Main optimization loop combining multiple strategies
    best_h = np.copy(initial_h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try initial optimization
    optimized_h, optimized_c5 = optimize_h(best_h)
    best_h, best_c5 = optimized_h, optimized_c5
    
    # Perform simulated annealing
    sa_h, sa_c5 = simulated_annealing(best_h)
    best_h, best_c5 = sa_h, sa_c5
    
    # Perform tabu search
    ts_h, ts_c5 = tabu_search(best_h)
    best_h, best_c5 = ts_h, ts_c5
    
    # Additional local search and exploration
    for _ in range(10):
        perturbation_strength = 0.1
        perturbed_h = perturb(best_h, perturbation_strength)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_h, best_c5 = optimized_perturbed, current_c5_perturbed
    
    # Check if time is up
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
