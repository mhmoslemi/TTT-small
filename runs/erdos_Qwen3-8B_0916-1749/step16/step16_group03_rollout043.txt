import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function for minimization
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solvers
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=Bounds(h_bounds),
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform random perturbations with adaptive strength
    def perturb_h(h_current, strength):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h
    
    # Function to perform a more aggressive local search
    def aggressive_local_search(h_current, perturbation_strengths):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            perturbed_h = perturb_h(best_h, strength)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and multiple starting points
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Initial optimization with local search
    optimized_h, current_c5 = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = current_c5
    
    # Use a grid of perturbation strengths for local search
    perturbation_strengths = np.linspace(0.01, 0.1, 10)
    best_h, best_c5 = aggressive_local_search(best_h, perturbation_strengths)
    
    # Generate multiple random starting points for optimization
    for _ in range(10):
        random_start = np.random.uniform(0, 1, size=n_points)
        optimized_h, current_c5 = optimize_h(random_start)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5

    for _ in range(5):
        perturbation_strength = 0.05
        perturbed_h = perturb_h(best_h, perturbation_strength)

        if current_c5_perturbed < best_c5:
            best_h = optimized_perturbed
            best_c5 = current_c5_perturbed
    
    # Final aggressive search with fine-grained perturbations
    perturbation_strengths = np.linspace(0.001, 0.05, 20)
    best_h, best_c5 = aggressive_local_search(best_h, perturbation_strengths)
    
    # Check for time budget
    if time.time() - start_time > budget_s:
        pass
    
    return best_h, best_c5, len(best_h)
