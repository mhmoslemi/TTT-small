import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to perform optimization with bounds
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        h_initial = np.array(h_initial)
        result = optimize.minimize(
            fun=compute_c5,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a greedy descent
    def greedy_descent(h_current, step_size=0.01, max_iter=100):
        h_current = np.array(h_current)
        for _ in range(max_iter):
            grad = np.zeros_like(h_current)
            for i in range(n_points):
                # Compute gradient with respect to h[i]
                # We approximate gradient with finite differences
                h_plus = np.array(h_current)
                h_plus[i] += step_size
                h_minus = np.array(h_current)
                h_minus[i] -= step_size
                grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * step_size)
            # Update h_current in direction of negative gradient
            h_current -= 0.1 * grad
            # Clip to [0, 1]
            h_current = np.clip(h_current, 0, 1)
        return h_current, compute_c5(h_current)
    
    # Function to generate random perturbations
    def perturb(h, strength=0.05):
        return np.clip(h + np.random.normal(0, strength, size=len(h)), 0, 1)
    
    # Main optimization loop with multiple strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for iteration in range(100):
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Perform greedy descent
        best_h, best_c5 = greedy_descent(best_h)
        
        # Generate random perturbations and optimize
        for _ in range(5):
            perturbed_h = perturb(best_h)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
