import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    def perturb(h, strength):
        perturbation = np.random.normal(0, strength, size=len(h))
        return np.clip(h + perturbation, 0, 1)

    def local_search(h_current, perturbation_strength=0.05, iters=5):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(iters):
            perturbed_h = perturb(h_current, perturbation_strength)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5

    def run_optimization(start_h, perturbation_strengths, iters=10):
        best_h = np.copy(start_h)
        best_c5 = compute_c5(best_h)
        for _ in range(iters):
            for strength in perturbation_strengths:
                perturbed_h = perturb(best_h, strength)
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
        return best_h, best_c5

    # Initial optimization
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # One round of mutation and optimization
    perturbation_strengths = np.linspace(0.01, 0.1, 10)
    best_h, best_c5 = run_optimization(best_h, perturbation_strengths, iters=10)

    # Local search with perturbation
    best_h, best_c5 = local_search(best_h, perturbation_strength=0.05, iters=5)

    # Check for time
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)

    return best_h, best_c5, len(best_h)
