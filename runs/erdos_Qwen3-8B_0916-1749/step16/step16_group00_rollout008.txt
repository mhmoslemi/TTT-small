import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values  # This variable is globally defined

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    def perturb_h(h, strength):
        perturbation = np.random.normal(0, strength, size=len(h))
        return np.clip(h + perturbation, 0, 1)

    def adaptive_search(h_initial, max_iter=100, strength_decay=0.99):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)
        for _ in range(max_iter):
            strength = 0.1 * (0.99 ** _)
            h_perturbed = perturb_h(h, strength)
            h_opt, c5 = optimize_h(h_perturbed)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
            h = best_h
        return best_h, best_c5

    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(5):
        perturbed_h = perturb_h(best_h, 0.1)
        optimized_h, c5 = optimize_h(perturbed_h)
        if c5 < best_c5:
            best_h = optimized_h
            best_c5 = c5

    best_h, best_c5 = adaptive_search(best_h, max_iter=50, strength_decay=0.995)

    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)

    return best_h, best_c5, len(best_h)
