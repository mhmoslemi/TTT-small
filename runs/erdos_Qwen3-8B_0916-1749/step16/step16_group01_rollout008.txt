import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a function to perform a more structured perturbation
    def structured_perturbation(h, strength, perturb_type='gaussian'):
        if perturb_type == 'gaussian':
            perturbation = np.random.normal(0, strength, size=len(h))
        elif perturb_type == 'uniform':
            perturbation = np.random.uniform(-strength, strength, size=len(h))
        elif perturb_type == 'block':
            block_size = len(h) // 5
            perturbation = np.zeros(len(h))
            for i in range(5):
                start = i * block_size
                end = start + block_size
                perturbation[start:end] = np.random.normal(0, strength, size=block_size)
        else:
            perturbation = np.zeros(len(h))
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h
    
    # Function to perform optimization with structured perturbations
    def optimize_with_perturbations(h_initial, perturbation_strengths, perturb_types):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            for pt in perturb_types:
                perturbed_h = structured_perturbation(best_h, strength, pt)
                try:
                    result = optimize.minimize(
                        fun=objective,
                        x0=perturbed_h,
                        bounds=[(0, 1) for _ in range(n_points)],
                        constraints=[{'type': 'eq', 'fun': constraint}],
                        method='SLSQP',
                        options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
                    )
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
                except:
                    continue
        return best_h, best_c5
    
    # Function to evaluate the current best solution
    def evaluate_best(best_h):
        best_c5 = compute_c5(best_h)
        return best_h, best_c5

    perturbation_strengths = np.linspace(0.01, 0.1, 10)
    perturb_types = ['gaussian', 'uniform', 'block']

    best_h = initial_h

    for strength in perturbation_strengths:

            perturbed_h = structured_perturbation(best_h, strength, pt)
            try:
                result = optimize.minimize(
                    fun=objective,
                    x0=perturbed_h,
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    method='SLSQP',
                    options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
                )
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            except:
                continue
    
    # Now perform a more aggressive search with multiple optimization trials
    for i in range(10):

        starts = []
        for _ in range(5):
            perturbed_h = structured_perturbation(best_h, 0.05, 'gaussian')
            starts.append(perturbed_h)
        
        for start in starts:
            try:
                result = optimize.minimize(
                    fun=objective,
                    x0=start,
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    method='SLSQP',
                    options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
                )
                current_c5 = result.fun

                    best_c5 = current_c5
                    best_h = result.x
            except:
                continue
        
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate the best solution found
    best_h, best_c5 = evaluate_best(best_h)
    
    return best_h, best_c5, len(best_h)
