import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a random walk with adaptive step size
    def random_walk(h, step_size=0.05, max_steps=100):
        h_current = np.copy(h)
        best_h = np.copy(h)
        best_c5 = compute_c5(h_current)
        for _ in range(max_steps):
            perturbation = np.random.normal(0, step_size, size=len(h_current))
            h_perturbed = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
            h_current = h_perturbed
        return best_h, best_c5
    
    # Function to perform a directional search with mutation
    def directional_search(h, direction, step_size=0.05, max_steps=50):
        h_current = np.copy(h)
        best_h = np.copy(h)
        best_c5 = compute_c5(h_current)
        for _ in range(max_steps):
            perturbation = np.array(direction) * step_size
            h_perturbed = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
            h_current = h_perturbed
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform perturbation and optimization with adaptive strength
        for j in range(5):
            perturbation_strength = 0.1 / (1 + i)
            perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Perform directional search with gradient-like perturbations
        for j in range(5):
            direction = np.random.normal(0, 0.1, size=n_points)
            perturbed_h = np.clip(best_h + direction, 0, 1)
            try:
                optimized_direction, current_c5_direction = optimize_h(perturbed_h)
                if current_c5_direction < best_c5:
                    best_c5 = current_c5_direction
                    best_h = optimized_direction
            except:
                continue
        
        # Perform random walk exploration
        best_h, best_c5 = random_walk(best_h, step_size=0.05, max_steps=50)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
