import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate perturbations with different strategies
    def generate_perturbations(h_current, strengths, num_perturbations=5):
        perturbations = []
        for strength in strengths:
            for _ in range(num_perturbations):
                perturbation = np.random.normal(0, strength, size=n_points)
                perturbed_h = np.clip(h_current + perturbation, 0, 1)
                perturbations.append(perturbed_h)
        return perturbations
    
    # Function to perform a coordinated exploration of the search space
    def explore_search_space(h_current, perturbation_strengths, num_perturbations=5):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        perturbations = generate_perturbations(h_current, perturbation_strengths, num_perturbations)
        for perturbed_h in perturbations:
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        return best_h, best_c5
    
    # Function to perform a more aggressive search with mixed strategies
    def aggressive_search(h_current, strengths):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for strength in strengths:
            for _ in range(5):
                perturbation = np.random.normal(0, strength, size=n_points)
                perturbed_h = np.clip(h_current + perturbation, 0, 1)
                try:
                    optimized_h, current_c5 = optimize_h(perturbed_h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                except:
                    continue
        return best_h, best_c5
    
    # Main optimization loop with coordinated exploration and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define a range of perturbation strengths for exploration
    perturbation_strengths = np.linspace(0.01, 0.1, 10)
    
    # Main loop for iterative refinement
    for iteration in range(10):
        # Explore the search space with different perturbation strengths
        best_h, best_c5 = explore_search_space(best_h, perturbation_strengths)
        
        # Perform a more aggressive search with mixed perturbation strategies
        best_h, best_c5 = aggressive_search(best_h, perturbation_strengths)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
