import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform constrained optimization
    def optimize_h(h_initial):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a more aggressive local search
    def aggressive_local_search(h_current, perturbation_strength=0.1):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(50):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a multi-start optimization with adaptive perturbation
    def multi_start_optimization(h_initial, num_starts=5, perturbation_strengths=np.linspace(0.01, 0.1, 10)):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            for _ in range(num_starts):
                perturbed_h = np.clip(h_initial + np.random.normal(0, strength, size=n_points), 0, 1)
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
