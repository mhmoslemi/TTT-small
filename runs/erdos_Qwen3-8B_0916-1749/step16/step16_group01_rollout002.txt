import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples (initial_h_values)
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values  # Assume this variable is globally defined
    h = np.array(initial_h)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function (sum(h) = n_points / 2)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization method with advanced parameters
    def optimize_h(h_initial, method='SLSQP', **kwargs):
        bounds = Bounds([0, 1], [0, 1])
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8, **kwargs}
        )
        return result.x, result.fun
    
    # Function to generate random perturbations with different strengths
    def perturb_h(h, strength):
        perturbation = np.random.normal(0, strength, size=len(h))
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h
    
    # Function to generate a new starting point with random changes
    def random_start():
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        # Adjust to satisfy the sum constraint
        total = np.sum(h)
        h = h * (n_points / 2) / total
        return h
    
    # Define a more diverse set of perturbation strengths
    perturbation_strengths = np.linspace(0.01, 0.1, 20)
    
    # Main optimization loop with multiple strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try multiple starting points
    for _ in range(5):
        h_start = random_start()
        h_opt, c5_opt = optimize_h(h_start, method='SLSQP')
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Try multiple perturbations and optimizations
    for strength in perturbation_strengths:
        h_perturbed = perturb_h(best_h, strength)
        try:
            h_opt, c5_opt = optimize_h(h_perturbed, method='SLSQP')
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except:
            continue
    
    # Try a grid search over perturbation strengths
    for strength in perturbation_strengths:
        h_perturbed = perturb_h(best_h, strength)
        try:
            h_opt, c5_opt = optimize_h(h_perturbed, method='SLSQP')
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except:
            continue
    
    # Try more aggressive perturbations
    for strength in np.linspace(0.1, 0.2, 10):
        h_perturbed = perturb_h(best_h, strength)
        try:
            h_opt, c5_opt = optimize_h(h_perturbed, method='SLSQP')
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        except:
            continue
    
    # Try multiple random starts with different methods
    for _ in range(5):
        h_start = random_start()
        h_opt, c5_opt = optimize_h(h_start, method='Nelder-Mead')
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    for strength in np.linspace(0.05, 0.2, 20):

                best_h = h_opt
                best_c5 = c5_opt
        except:
            continue
    
    # Ensure we don't exceed the time budget
