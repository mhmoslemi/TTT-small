import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a grid-based search over perturbation strengths
    def grid_search(h_initial, strength_range, num_steps):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for strength in np.linspace(*strength_range, num_steps):
            perturbed_h = np.clip(h_initial + np.random.normal(0, strength, size=n_points), 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h
        return best_h, best_c5
    
    # Define adaptive perturbation strength range
    strength_range = (0.01, 0.1)
    num_steps = 10
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive search with multiple perturbations and local search
    for i in range(100):
        # Adaptive grid search with perturbation strength
        perturbed_h, current_c5 = grid_search(best_h, strength_range, num_steps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = perturbed_h
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    # Final refinement with L-BFGS-B
    result = optimize.minimize(
        fun=objective,
        x0=best_h,
        bounds=([(0, 1) for _ in range(n_points)]),
        method='L-BFGS-B',
        options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
    )
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, len(best_h)
