import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a gradient-based descent with momentum
    def gradient_descent(h_initial, learning_rate=0.05, momentum=0.9, n_iter=100):
        h = np.copy(h_initial)
        h_prev = np.copy(h_initial)
        for _ in range(n_iter):
            grad = np.zeros_like(h)
            for shift in range(1, 10):
                h_plus = np.clip(h + 1e-4 * np.array([1 if i % shift == 0 else 0 for i in range(n_points)]), 0, 1)
                h_minus = np.clip(h - 1e-4 * np.array([1 if i % shift == 0 else 0 for i in range(n_points)]), 0, 1)
                grad += (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * 1e-4)
            grad /= n_points
            h = h - learning_rate * grad + momentum * (h - h_prev)
            h_prev = h
        return h, compute_c5(h)
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Run multiple local optimizations
    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform gradient descent from the current best
        try:
            optimized_h, current_c5 = gradient_descent(best_h, learning_rate=0.01, momentum=0.9, n_iter=50)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Adaptive perturbation with decreasing strength
        perturbation_strength = 0.1 / (1 + i)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            pass
    
    return best_h, best_c5, len(best_h)
