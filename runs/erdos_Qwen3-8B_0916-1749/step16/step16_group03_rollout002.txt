    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a function to generate perturbed h with different structures
    def generate_perturbations(h):
        perturbations = []
        # Flip some values to 0 or 1
        for i in range(5):
            h_flip = np.copy(h)
            indices = np.random.choice(n_points, size=5, replace=False)
            h_flip[indices] = np.where(h_flip[indices] < 0.5, 1, 0)
            perturbations.append(h_flip)
        # Add random noise
        for i in range(5):
            h_noise = np.copy(h)
            noise = np.random.normal(0, 0.1, n_points)
            h_noise = np.clip(h_noise + noise, 0, 1)
            perturbations.append(h_noise)
        return perturbations
    
    # Function to perform optimization with accelerated convergence using more structured perturbations
    def optimize_with_perturbations(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Evaluation function
    def evaluate(h):
        return compute_c5(h)
    
    # Main optimization loop with enhanced exploration and adaptive perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(10):
        # Generate perturbations
        perturbations = generate_perturbations(best_h)
        
        # Run optimization on each perturbation
        for p in perturbations:
            try:
                optimized_h, current_c5 = optimize_with_perturbations(p)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:

        # Check for convergence and time budget
