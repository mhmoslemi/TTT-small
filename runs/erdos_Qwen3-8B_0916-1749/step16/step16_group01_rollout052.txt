import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial, method='SLSQP'):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    def perturb(h, strength=0.05):
        perturbation = np.random.normal(0, strength, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    def local_search(h, steps=10):
        best_h = np.copy(h)
        best_c5 = compute_c5(best_h)
        for _ in range(steps):
            perturbation = np.random.normal(0, 0.05, size=len(h))
            perturbed_h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    def aggressive_search(h, strengths):
        best_h = np.copy(h)
        best_c5 = compute_c5(best_h)
        for strength in strengths:
            perturbation = np.random.normal(0, strength, size=len(h))
            perturbed_h = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Optimize from initial point
        optimized_h, current_c5 = optimize_h(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Local search
        best_h, best_c5 = local_search(best_h)
        
        # Aggressive search with different strengths
        perturbation_strengths = np.linspace(0.01, 0.1, 10)
        best_h_ags, best_c5_ags = aggressive_search(best_h, perturbation_strengths)
        if best_c5_ags < best_c5:
            best_c5 = best_c5_ags
            best_h = best_h_ags
        
        # Random perturbation
        best_h = perturb(best_h, strength=0.05)
        
        # Check if we are converging
        if np.abs(best_c5 - 0.380876) < 1e-5:
            break
    
    return best_h, best_c5, len(best_h)
