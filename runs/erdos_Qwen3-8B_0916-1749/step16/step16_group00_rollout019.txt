import scipy.optimize as optimize
import time

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5

    def guided_optimization(h_initial, max_iterations=50, perturbation_strength=0.1):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for _ in range(max_iterations):
            # Generate perturbations
            perturbations = [np.random.normal(0, perturbation_strength, size=n_points) for _ in range(5)]

            # Optimize each candidate
            for candidate in candidates:
                optimized_h, current_c5 = optimize_h(candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h

            best_h, best_c5 = local_search(best_h, perturbation_strength / 2)
