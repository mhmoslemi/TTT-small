import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=Bounds(h_bounds),
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform simulated annealing optimization
    def simulated_annealing(h_initial, temp_initial=1.0, temp_final=0.1, steps=1000):
        h_current = np.copy(h_initial)
        current_c5 = compute_c5(h_current)
        best_h = h_current
        best_c5 = current_c5
        temp = temp_initial
        for _ in range(steps):
            if temp < temp_final:
                break
            perturbation = np.random.normal(0, temp, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            perturbed_c5 = compute_c5(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
            h_current = perturbed_h
            temp *= 0.9995
        return best_h, best_c5
    
    # Function to perform a coordinate-wise optimization of the function shape
    def coordinatewise_optimization(h_initial):
        h = np.copy(h_initial)
        best_h = h
        best_c5 = compute_c5(h)

                    h_frozen = np.copy(h)
                    h_frozen[j] = np.random.uniform(0, 1)
                    # Optimize using SLSQP with frozen coordinates
                    h_opt, c5_opt = optimize_h(h_frozen)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        h = best_h
        return best_h, best_c5
