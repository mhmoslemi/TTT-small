import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function with bounds and constraints
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform adaptive perturbation and optimization
    def perturb_and_optimize(h_current, perturbation_strengths, n_perturbations=5):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            perturbations = np.random.normal(0, strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbations, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform adaptive local search
    def local_search(h_current, perturbation_strength=0.05, n_steps=10):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(n_steps):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Initial optimization
    optimized_h, current_c5 = optimize_h(best_h)
    best_h, best_c5 = optimized_h, current_c5
    
    # Adaptive perturbation and optimization
    for i in range(10):
        if time.time() - start_time > budget_s:

        best_h, best_c5 = perturb_and_optimize(best_h, perturbation_strengths)
        best_h, best_c5 = local_search(best_h)
    
    return best_h, best_c5, len(best_h)
