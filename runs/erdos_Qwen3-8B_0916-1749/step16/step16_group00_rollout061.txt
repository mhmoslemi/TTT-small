import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Define the objective function
    def objective(h):
        return compute_c5(h)

    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5

    # Function to generate a random perturbation with adaptive strength
    def generate_perturbation(h, strength):
        perturbation = np.random.normal(0, strength, size=len(h))
        return np.clip(h + perturbation, 0, 1)

    # Function to perform a gradient-based local search
    def gradient_search(h_current):
        h = np.array(h_current)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = np.copy(h)
            h_plus[i] += 1e-5
            h_minus = np.copy(h)
            h_minus[i] -= 1e-5
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-5)
        step_size = 0.1
        h_new = np.clip(h - step_size * grad, 0, 1)
        return h_new, objective(h_new)

    # Function to perform a coordinate-wise descent
    def coordinate_descent(h_current):
        h = np.array(h_current)
        best_h = np.copy(h)
        best_c5 = compute_c5(h)
        for i in range(len(h)):
            for val in [0, 1]:
                h_test = np.copy(h)
                h_test[i] = val
                c5 = compute_c5(h_test)
                if c5 < best_c5:
                    best_h = h_test
                    best_c5 = c5
        return best_h, best_c5

    # Function to evaluate the function values at multiple points
    def evaluate_points(x, h):
        if x < 0 or x > 2:
            return 0
        if len(h) == 0:
            return 0
        idx = int(x * n_points / 2)
        if idx < 0 or idx >= len(h):
            return 0
        return h[idx]

    # Function to compute the maximum overlap integral
    def compute_max_overlap(h):
        c5 = 0
        for k in range(1, 10):
            integral = 0
            for x in np.linspace(0, 2, 1000):
                integral += evaluate_points(x, h) * (1 - evaluate_points(x + k, h)) * (2 / 1000)
            c5 = max(c5, integral)
        return c5

    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    best_c5_with_overlap = compute_max_overlap(best_h)
    start_time = time.time()

    # Adaptive search with multiple perturbations and hybrid methods
    for i in range(100):
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
                if compute_max_overlap(optimized_h) < best_c5_with_overlap:
                    best_c5_with_overlap = compute_max_overlap(optimized_h)
            except:
                continue

        # Perform local search
        best_h, best_c5 = local_search(best_h)
        if compute_max_overlap(best_h) < best_c5_with_overlap:
            best_c5_with_overlap = compute_max_overlap(best_h)

        # Perform gradient-based local search
        best_h, best_c5 = gradient_search(best_h)
        if compute_max_overlap(best_h) < best_c5_with_overlap:
            best_c5_with_overlap = compute_max_overlap(best_h)

        # Perform coordinate descent
        best_h, best_c5 = coordinate_descent(best_h)
        if compute_max_overlap(best_h) < best_c5_with_overlap:
            best_c5_with_overlap = compute_max_overlap(best_h)

        # Generate perturbations with adaptive strength
        for j in range(5):
            perturbation_strength = 0.1 / (1 + i)
            perturbed_h = generate_perturbation(best_h, perturbation_strength)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
                if compute_max_overlap(optimized_perturbed) < best_c5_with_overlap:
                    best_c5_with_overlap = compute_max_overlap(optimized_perturbed)
            except:
                continue

        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5_with_overlap, len(best_h)
