import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate random perturbation
    def perturb(h, strength):
        return np.clip(h + np.random.normal(0, strength, size=len(h)), 0, 1)
    
    # Main optimization loop with adaptive search and perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive parameter sweep
    for iteration in range(100):
        # Generate multiple starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perturb and optimize
        for _ in range(5):
            strength = 0.1 / (1 + iteration)
            perturbed_h = perturb(best_h, strength)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check for time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
