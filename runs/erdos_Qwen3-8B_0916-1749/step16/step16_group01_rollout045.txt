import scipy.optimize as optimize
import time

        return max_corr * dx

    def compute_shifted_integral(h, shift):
        h = np.array(h)
        # Compute the integral over the shifted function
        # The shift is normalized by dx
        shift_normalized = shift / dx
        # Create a shifted version of h
        shifted_h = np.roll(h, int(shift_normalized))
        # Compute the integral of h(x)(1 - h(x + shift))
        integral = np.sum(h * (1 - shifted_h))
        return integral
    
    # Function to compute all shifted integrals
    def compute_all_shifted_integrals(h):
        h = np.array(h)
        n = len(h)
        max_shift = 2.0
        dx = 2.0 / n
        all_integrals = []
        for k in range(n):
            shift = k * dx
            shifted_h = np.roll(h, k)
            integral = np.sum(h * (1 - shifted_h))
            all_integrals.append(integral)
        # Also include negative shifts
        for k in range(1, n):
            shift = -k * dx
            shifted_h = np.roll(h, -k)
            integral = np.sum(h * (1 - shifted_h))
            all_integrals.append(integral)
        return all_integrals
    
    # Function to compute the maximum shifted integral
    def compute_max_shifted_integral(h):
        integrals = compute_all_shifted_integrals(h)
        return max(integrals)
    
    # Objective function to minimize the max shifted integral
    def objective(h):
        return compute_max_shifted_integral(h)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-8, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun

    def local_search(h_current, steps=10, strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_max_shifted_integral(best_h)
        for _ in range(steps):
            perturbation = np.random.normal(0, strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_max_shifted_integral(perturbed_h)

    best_h = optimized_h
    best_c5 = optimized_c5
    
    # Final local search
    best_h, best_c5 = local_search(best_h)
    
    # Check for time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
