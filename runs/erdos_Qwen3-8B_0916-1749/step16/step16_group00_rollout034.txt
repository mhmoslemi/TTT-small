import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate random perturbations
    def generate_random_perturbation(h_current, strength=0.05):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Additional parameters
    num_iterations = 100
    max_perturbations = 5
    perturbation_strength = 0.1
    
    for iteration in range(num_iterations):
        # Generate multiple perturbations
        for _ in range(max_perturbations):
            perturbed_h = generate_random_perturbation(best_h, perturbation_strength)
            try:
                optimized_perturbed, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_perturbed
            except:
                continue
        
        # Perform local search
        best_h, best_c5 = local_search(best_h, perturbation_strength / 2)
        
        # Reduce perturbation strength
        perturbation_strength *= 0.95
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
