import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral (C5)
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Objective function to minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds and variables
    bounds = Bounds(lb=0, ub=1)
    
    # Function to perform constrained optimization
    def optimize_h(h_initial):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a random perturbation and optimization
    def perturb_and_optimize(h_current, perturbation_strength):
        perturbed_h = np.clip(h_current + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        return optimize_h(perturbed_h)
    
    # Function for local search with adaptive perturbations
    def local_search(h_current, perturbation_strengths, n_perturbations=10):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for strength in perturbation_strengths:
            perturbations = np.random.normal(0, strength, size=n_perturbations)
            perturbed_h = np.clip(h_current + perturbations, 0, 1)
            for h in perturbed_h:
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbations and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive perturbation search
    for i in range(50):
        # Generate multiple perturbations
        for _ in range(5):
            h_perturbed = np.clip(best_h + np.random.normal(0, 0.05, size=n_points), 0, 1)
            h_optimized, c5 = optimize_h(h_perturbed)
            if c5 < best_c5:
                best_h = h_optimized
                best_c5 = c5
        
        # Local search with different perturbation strengths
        perturbation_strengths = np.linspace(0.01, 0.1, 10)

        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
