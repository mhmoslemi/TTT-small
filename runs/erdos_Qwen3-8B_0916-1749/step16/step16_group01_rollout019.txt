import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem with bounds and constraints
    def optimize_h(h_initial):
        h_bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate perturbations based on a base configuration
    def generate_perturbations(h_base):
        perturbations = []
        for i in range(5):
            noise = np.random.normal(0, 0.05, n_points)
            perturbation = np.clip(h_base + noise, 0, 1)
            perturbations.append(perturbation)
        return perturbations
    
    # Function to perform global search with multiple starting points
    def global_search(h_initial):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        starts = generate_perturbations(best_h)
        for start in starts:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        return best_h, best_c5
    
    # Function to refine solution with multi-scale perturbations
    def refine_solution(h_current):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        
        # Multi-scale perturbations
        for scale in [0.01, 0.05, 0.1]:
            noise = np.random.normal(0, scale, n_points)
            perturbed_h = np.clip(h_current + noise, 0, 1)
            optimized_perturbed, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_perturbed
        
        return best_h, best_c5
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)

                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Refine the solution with multi-scale perturbations
        best_h, best_c5 = refine_solution(best_h)
        
        # Check for time budget
        if time.time() - start_time > budget_s:
            break
