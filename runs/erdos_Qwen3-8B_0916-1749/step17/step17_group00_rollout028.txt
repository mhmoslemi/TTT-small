import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more aggressive exploration
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to generate a perturbed version of the current solution
    def perturb_h(h_current, strength=0.05):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h
    
    # Main optimization loop with global search and adaptive learning
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform a combination of local search, random restarts, and gradient-based optimization
    for i in range(50):
        # Random restart
        random_h = random_start()
        optimized_h, current_c5 = optimize_h(random_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Local search around the current best
        best_h, best_c5 = local_search(best_h)
        
        # Adaptive perturbation and optimization
        strength = 0.1 / (1 + i)
        perturbed_h = perturb_h(best_h, strength)
        optimized_perturbed_h, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed_h
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
