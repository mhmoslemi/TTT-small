import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values  # Access global initial construction
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform constrained optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform perturbation-based exploration
    def perturbate(h_current, strength=0.1):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h
    
    # Function to perform greedy search with adaptive step size
    def greedy_search(h_current, step_size=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for i in range(100):
            for j in range(len(h_current)):
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] + step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] - step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform multi-start optimization
    def multi_start_search(h_initial, num_starts=5):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for _ in range(num_starts):
            start_h = np.copy(h_initial)
            np.random.shuffle(start_h)
            optimized_h, current_c5 = optimize_h(start_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = np.copy(initial_h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Iterative refinement with perturbation, multi-start, and greedy search
    for iteration in range(100):
        # Perturbation and optimization
        for _ in range(5):
            perturbed_h = perturbate(best_h, strength=0.05)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        
        # Multi-start optimization
        best_h, best_c5 = multi_start_search(best_h, num_starts=5)
        
        # Greedy search
        best_h, best_c5 = greedy_search(best_h, step_size=0.02)
        
        # Check for convergence or time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
