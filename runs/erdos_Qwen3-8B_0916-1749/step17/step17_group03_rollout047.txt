import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a greedy search with adaptive step size
    def greedy_search(h_current, step_size=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for i in range(100):
            for j in range(len(h_current)):
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] + step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] - step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a simulated annealing search
    def simulated_annealing(h_current, max_iter=100, temp=1.0, temp_decay=0.995):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        current_h = np.copy(h_current)
        current_c5 = best_c5
        for _ in range(max_iter):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(current_h + perturbation, 0, 1)
            new_c5 = compute_c5(perturbed_h)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h = perturbed_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            temp *= temp_decay
        return best_h, best_c5
    
    # Main optimization loop with adaptive search
    best_h = np.copy(initial_h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform multiple initial optimizations with different starting points
    for _ in range(5):
        start = np.copy(best_h)
        start += np.random.normal(0, 0.05, size=n_points)
        start = np.clip(start, 0, 1)
        optimized_h, current_c5 = optimize_h(start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Perform simulated annealing
    best_h, best_c5 = simulated_annealing(best_h)
    
    # Perform local search
    best_h, best_c5 = local_search(best_h)
    
    # Perform greedy search
    best_h, best_c5 = greedy_search(best_h)
    
    # Perform final optimization
    optimized_h, _ = optimize_h(best_h)
    best_h = optimized_h
    
    # Check time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
