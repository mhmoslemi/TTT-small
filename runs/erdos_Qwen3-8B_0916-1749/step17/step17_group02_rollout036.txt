import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 100, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a random perturbation and optimization
    def perturb_and_optimize(h_current, perturbation_strength=0.05):
        perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            return optimized_perturbed, current_c5_perturbed
        except:
            return h_current, compute_c5(h_current)
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a grid search on a subset of parameters
    def grid_search(h_current, param_name, param_values, num_params_to_search):

        best_c5 = compute_c5(best_h)
        for _ in range(num_params_to_search):
            idx = np.random.randint(len(h_current))
            for val in param_values:
                h_new = np.copy(h_current)
                h_new[idx] = val
                current_c5 = compute_c5(h_new)
                if current_c5 < best_c5:
                    best_h = h_new
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    best_h, best_c5 = local_search(best_h)

    for _ in range(20):

        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Grid search on a subset of parameters
    best_h, best_c5 = grid_search(best_h, 'value', np.linspace(0, 1, 10), 5)
    
    # Final check for convergence and time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
