import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a random starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Function to generate a random perturbation with adaptive strength
    def perturb_h(h_current, strength=0.05):
        perturbation = np.random.normal(0, strength, size=len(h_current))
        perturbed_h = np.clip(h_current + perturbation, 0, 1)
        return perturbed_h
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a grid search on perturbation strength
    def grid_search(h_current, strengths):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for strength in strengths:
            perturbed_h = np.clip(h_current + np.random.normal(0, strength, size=n_points), 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a simulated annealing search
    def simulated_annealing(h_initial, max_iter=100, temp=1.0, temp_decay=0.99):
        h_current = np.copy(h_initial)
        current_c5 = compute_c5(h_current)
        best_h = np.copy(h_current)
        best_c5 = current_c5
        for _ in range(max_iter):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = np.clip(h_current + perturbation, 0, 1)
            new_c5 = compute_c5(h_perturbed)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_perturbed
                current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h_current
                best_c5 = current_c5
            temp *= temp_decay
        return best_h, best_c5
    
    # Main optimization loop with multiple strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try multiple strategies in sequence
    # 1. Local optimization starting from initial_h
    optimized_h, _ = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = compute_c5(best_h)
    
    # 2. Perform grid search on perturbation strengths
    strengths = np.linspace(0.01, 0.1, 10)
    best_h, best_c5 = grid_search(best_h, strengths)
    
    # 3. Perform local search with adaptive perturbation
    best_h, best_c5 = local_search(best_h)
    
    # 4. Perform simulated annealing with adaptive cooling
    best_h, best_c5 = simulated_annealing(best_h, max_iter=200, temp=1.0, temp_decay=0.995)
    
    # 5. Additional random restarts with local search
    for _ in range(5):
        start = random_start()
        optimized_h, _ = optimize_h(start)
        optimized_h, _ = local_search(optimized_h)
        current_c5 = compute_c5(optimized_h)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
    
    # Check remaining time and proceed with final optimizations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Perform additional simulated annealing with smaller temperature
        best_h, best_c5 = simulated_annealing(best_h, max_iter=100, temp=0.1, temp_decay=0.995)
    
    return best_h, best_c5, len(best_h)
