import numpy as np
import scipy.optimize as optimize
import time

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more aggressive parameters
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with adaptive perturbation
    def new_start():
        h = np.random.uniform(0, 1, n_points)
        # Apply a small adjustment to maintain constraint
        adjustment = (n_points / 2 - np.sum(h)) / n_points
        h = np.clip(h + adjustment, 0, 1)
        return h
    
    # Main optimization loop with more aggressive exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try multiple starting points
    for _ in range(20):
        h_start = new_start()
        optimized_h, current_c5 = optimize_h(h_start)

            best_c5 = current_c5
            best_h = optimized_h
    
    # Perform additional optimization with different perturbation strategies
    for _ in range(20):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)

        if current_c5 < best_c5:

            best_h = optimized_perturbed

    def local_search(h_current):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
