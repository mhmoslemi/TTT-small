import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform optimization with more advanced solver settings
    def optimize_h(h_initial):
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8, 'disp': False}
        )
        return result.x, result.fun
    
    # Define a more sophisticated local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.01, num_perturbations=5):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(num_perturbations):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with global search and local refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Run initial optimization with a more aggressive global search
    for _ in range(15):
        try:
            optimized_h, current_c5 = optimize_h(np.copy(best_h))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Run a series of local searches with decreasing perturbation strength
    for i in range(10):
        perturbation_strength = 0.1 / (1 + i)
        best_h, best_c5 = local_search(best_h, perturbation_strength)
    
    # Run a global search on a subset of the domain
    subset_size = int(n_points * 0.5)

    # Final local search with smaller perturbation
    best_h, best_c5 = local_search(best_h, perturbation_strength=0.005)
    
    # Ensure we stay within the time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
