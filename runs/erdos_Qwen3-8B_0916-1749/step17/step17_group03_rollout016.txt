import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to perform optimization with adaptive bounds and constraints
    def optimize_h(h_initial, max_iter=200, tol=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=compute_c5,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            method='SLSQP',
            options={'ftol': tol, 'maxiter': max_iter, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a greedy search with adaptive step size
    def greedy_search(h_current, step_size=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for i in range(100):
            for j in range(n_points):
                # Try increasing the value
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] + step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
                # Try decreasing the value
                try_h = np.copy(h_current)
                try_h[j] = np.clip(try_h[j] - step_size, 0, 1)
                current_c5 = compute_c5(try_h)
                if current_c5 < best_c5:
                    best_h = try_h
                    best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop
    best_h = np.copy(initial_h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform initial optimization
    optimized_h, current_c5 = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = current_c5
    
    # Main optimization loop
    for i in range(100):
        # Perturb and optimize
        perturbation_strength = 0.15 / (1 + i)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
        
        # Local search
        best_h, best_c5 = local_search(best_h)
        
        # Greedy search
        best_h, best_c5 = greedy_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
