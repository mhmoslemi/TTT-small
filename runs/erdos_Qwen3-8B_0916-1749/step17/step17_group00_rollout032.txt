import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform gradient descent with momentum
    def gradient_descent(h_initial, learning_rate=0.01, momentum=0.9, max_iter=1000):
        h = np.copy(h_initial)
        h_prev = np.copy(h)
        for i in range(max_iter):
            grad = np.zeros_like(h)
            for k in range(n_points):
                for j in range(n_points):
                    overlap = 0
                    for l in range(n_points):
                        overlap += h[l] * (1 - h[(l + k) % n_points])
                    grad[k] = -2 * overlap / n_points
            h_new = h - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            h_prev = momentum * h_prev + (1 - momentum) * h_new
            h = h_prev
        return h, compute_c5(h)
    
    # Main optimization loop with alternating optimization and gradient descent
    best_h = np.copy(initial_h)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive search with gradient descent
    for i in range(100):
        # Perform gradient descent with varying learning rate
        learning_rate = 0.01 / (1 + i)
        optimized_h, current_c5 = gradient_descent(best_h, learning_rate=learning_rate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Perform local search with random perturbation
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        current_c5_perturbed = compute_c5(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = perturbed_h
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
