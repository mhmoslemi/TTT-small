import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):

    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        noise = np.random.normal(0, 0.1, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform local search with directional perturbations
    def directional_search(h_current, direction):
        h = np.copy(h_current)
        h += direction * 0.1
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive perturbation and local search
    for i in range(100):
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]

                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue

        for j in range(5):
            # Generate random directions
            direction = np.random.normal(0, 1, size=n_points)
            direction /= np.linalg.norm(direction)

                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
