import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Global initial construction (n=70)

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Function to perform optimization
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(len(h_initial))]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a multi-start optimization with adaptive perturbations
    def multi_start_optimization(initial_guess, num_starts=5, num_perturbations=5, perturb_strength=0.1):
        best_solution = initial_guess
        best_value = compute_c5(best_solution)
        
        for _ in range(num_starts):
            perturbation = np.random.normal(0, perturb_strength, size=len(best_solution))
            perturbed = np.clip(best_solution + perturbation, 0, 1)
            optimized, value = optimize_h(perturbed)
            if value < best_value:
                best_solution = optimized
                best_value = value
        
        for _ in range(num_perturbations):
            perturbation = np.random.normal(0, perturb_strength / 2, size=len(best_solution))
            perturbed = np.clip(best_solution + perturbation, 0, 1)
            optimized, value = optimize_h(perturbed)
            if value < best_value:
                best_solution = optimized
                best_value = value
        
        return best_solution, best_value
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Run multiple rounds of multi-start optimization with decreasing perturbation strength
    for round_num in range(3):
        print(f"Round {round_num + 1} started")
        best_h, best_c5 = multi_start_optimization(best_h, num_starts=10, num_perturbations=10, perturb_strength=0.1 / (round_num + 1))
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
