        max_corr = np.max(corr)
        return max_corr * dx

    def compute_overlap(h, shift):
        h = np.array(h)
        if shift > 0:
            shifted_h = np.roll(h, shift)
        else:
            shifted_h = np.roll(h, shift)
        return np.sum(h * (1 - shifted_h)) * dx

    def objective(h):
        max_overlap = 0

            current_overlap = compute_overlap(h, k)
            if current_overlap > max_overlap:
                max_overlap = current_overlap
        return max_overlap

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    def smart_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)

        h = np.clip(h, 0, 1)
        return h
