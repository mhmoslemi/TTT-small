import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a guided random search
    def guided_random_search(start_h, iterations=50):
        best_h = np.copy(start_h)
        best_c5 = compute_c5(best_h)
        for _ in range(iterations):
            h = np.copy(best_h)
            h += np.random.normal(0, 0.05, size=len(h))
            h = np.clip(h, 0, 1)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        return best_h, best_c5
    
    # Main optimization loop with diverse strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # First, perform guided random search with multiple starts
    starts = [np.copy(best_h) for _ in range(10)]
    for start in starts:
        optimized_h, current_c5 = optimize_h(start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Second, perform guided random search with the current best
    optimized_h, current_c5 = guided_random_search(best_h, iterations=100)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h
    
    # Third, perform local optimization with perturbations
    for _ in range(50):
        h = np.copy(best_h)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        optimized_h, current_c5 = optimize_h(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
