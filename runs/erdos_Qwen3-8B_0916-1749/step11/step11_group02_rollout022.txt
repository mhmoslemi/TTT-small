import time

    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with informed sampling
    def generate_new_start(h_current):
        h = np.copy(h_current)
        perturbation = np.random.normal(0, 0.05, size=len(h))
        h += perturbation
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization with global search and adaptive perturbation
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(10):
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]

                optimized_h, current_c5 = optimize_h(start)

                    best_c5 = current_c5
                    best_h = optimized_h
