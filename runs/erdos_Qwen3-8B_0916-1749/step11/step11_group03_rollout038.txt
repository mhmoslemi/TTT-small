import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = np.array([0.5] * n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a more structured perturbation strategy
    def generate_new_start(h_current):
        h = np.copy(h_current)
        for i in range(n_points):
            if np.random.rand() < 0.2:
                h[i] = 0.5
        return h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h_values
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform a more systematic search
    for i in range(100):
        # Generate a new starting point
        new_start = generate_new_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(new_start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform a local search with varying perturbation strengths
        for j in range(5):
            strength = 0.1 / (1 + i)
            perturbed_h = np.copy(best_h)
            perturbed_h += np.random.normal(0, strength, size=n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
