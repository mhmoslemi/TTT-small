import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different methods
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate new starting points via random perturbations
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform gradient descent with line search
    def gradient_descent(h_initial, max_iter=100, step=0.1):
        h = np.copy(h_initial)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Computation of gradient using finite differences
                h_plus = np.copy(h)
                h_plus[i] += 1e-6
                h_minus = np.copy(h)
                h_minus[i] -= 1e-6
                grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * 1e-6)
            h -= step * grad
            h = np.clip(h, 0, 1)
        return h, compute_c5(h)
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of adaptive optimization with different methods
    for i in range(50):
        # Generate multiple starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform gradient descent on current best solution
        try:
            optimized_h, current_c5 = gradient_descent(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform perturbation and optimization
        for j in range(5):
            perturbed_h = np.copy(best_h)
            perturbed_h += np.random.normal(0, 0.05, size=len(perturbed_h))
            perturbed_h = np.clip(perturbed_h, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
