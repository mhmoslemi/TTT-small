import scipy.optimize as optimize
import time

    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with informed sampling
    def generate_new_start(h_current):
        h = np.copy(h_current)
        for i in range(len(h)):
            if np.random.rand() < 0.3:
                h[i] = np.random.uniform(0, 1)
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform local search with guided perturbations
    def local_search(h_current, perturbation_strength=0.1):
        h = np.copy(h_current)
        h += np.random.normal(0, perturbation_strength, size=len(h))

    for i in range(100):

        new_start = generate_new_start(best_h)

            optimized_h, current_c5 = optimize_h(new_start)

                best_c5 = current_c5
