import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Precompute the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with different methods and parameter sweeps
    def optimize_h(h_initial, method='SLSQP', maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations with different patterns
    def structured_perturb(h, pattern, strength=0.1):
        h = np.copy(h)
        h += strength * pattern
        h = np.clip(h, 0, 1)
        return h
    
    # Generate structured patterns for perturbation
    def generate_patterns(n_points):
        patterns = []
        # Step patterns
        for step in range(1, n_points // 2):
            pattern = np.zeros(n_points)
            for i in range(n_points):
                pattern[i] = ((i % step) < step // 2) * 1.0 / step
            patterns.append(pattern)
        # Sine wave patterns
        for freq in range(1, 6):
            pattern = np.sin(2 * np.pi * freq * np.linspace(0, 1, n_points))
            pattern = (pattern + 1) / 2
            patterns.append(pattern)
        # Random patterns
        for _ in range(5):
            pattern = np.random.normal(0, 0.1, n_points)
            patterns.append(pattern)
        return patterns
    
    # Main optimization loop with multiple strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Define multiple optimization strategies
    strategies = [
        ('SLSQP', 1000, 1e-10, 1e-8),
        ('Nelder-Mead', 500, 1e-8, 1e-8),
        ('COBYLA', 500, 1e-8, 1e-8)
    ]
    
    # Generate structured patterns for perturbation
    patterns = generate_patterns(n_points)
    
    # Initial optimization phase
    for strategy in strategies:
        method, maxiter, ftol, eps = strategy
        optimized_h, current_c5 = optimize_h(np.copy(best_h), method=method, maxiter=maxiter, ftol=ftol, eps=eps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Iterative optimization with structured perturbations
    for i in range(10):
        # Generate multiple perturbations
        for pattern in patterns:
            perturbed_h = structured_perturb(best_h, pattern, strength=0.1 / (i + 1))
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h, method='SLSQP', maxiter=500, ftol=1e-10, eps=1e-8)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
