import scipy.optimize as optimize
import time

    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization function with improved parameters
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with structured perturbations
    def generate_new_start(h_current, perturbation_strength=0.1):
        h = np.copy(h_current)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a guided local search
    def guided_local_search(h_current, seed, perturbation_strength=0.1):
        np.random.seed(seed)
        h = np.copy(h_current)

        h = np.clip(h, 0, 1)
        return h

    best_h = initial_h
    best_c5 = compute_c5(best_h)
