import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define a more efficient optimization method with adaptive perturbations
    def optimize_h(h_initial, maxiter=1000, eps=1e-8, ftol=1e-10):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate perturbations
    def generate_perturbation(h, strength=0.1):
        perturb = np.random.normal(0, strength, size=len(h))
        h_perturbed = np.clip(h + perturb, 0, 1)
        return h_perturbed
    
    # Define a function to evolve solutions with guided mutation
    def evolve_solution(h, seed, strength=0.1):
        np.random.seed(seed)
        h_evolved = np.copy(h)
        h_evolved += np.random.normal(0, strength, size=len(h_evolved))
        h_evolved = np.clip(h_evolved, 0, 1)
        return h_evolved
    
    # Main optimization loop with adaptive exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    iteration = 0
    
    # Perform a series of local optimizations with adaptive perturbations
    while time.time() - start_time < budget_s:
        # Local optimization with adaptive constraints
        optimized_h, current_c5 = optimize_h(np.copy(best_h), maxiter=500, eps=1e-10, ftol=1e-10)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Generate and optimize perturbed versions
        for _ in range(3):
            perturbation_strength = 0.1 / (1 + iteration)
            perturbed_h = generate_perturbation(best_h, strength=perturbation_strength)
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h, maxiter=200, eps=1e-10, ftol=1e-10)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        
        # Adaptive mutation with increasing randomness
        if iteration % 5 == 0:
            seed_mutation = seed + iteration
            best_h = evolve_solution(best_h, seed_mutation, strength=0.2)
        
        iteration += 1
    
    return best_h, best_c5, len(best_h)
