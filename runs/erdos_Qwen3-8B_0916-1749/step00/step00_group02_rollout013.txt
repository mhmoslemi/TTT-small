import numpy as np
import scipy.optimize as optimize
import time
import sys

# Pre-defined initial h_values for starting point
initial_h_values = np.load('initial_h_values.npy')  # assuming this file exists

def run(seed=42, budget_s=300, **kwargs):
    # Step 1: Set up parameters
    n_points = 100  # number of samples
    dx = 2.0 / n_points
    max_shift = 5  # maximum shift to consider for overlaps

    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        max_c5 = 0
        for k in range(1, max_shift + 1):
            # Compute shifted h
            shifted_h = np.zeros_like(h)
            for i in range(n_points):
                shifted_i = (i - k) % n_points
                shifted_h[i] = h[shifted_i]
            # Compute overlap integral
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_c5:
                max_c5 = overlap
        return max_c5

    # Function to evaluate the overlap integral for a given h
    def objective(h):
        return compute_c5(h)

    # Function to evaluate the overlap integral and return the maximum value
    def evaluate(h):
        return compute_c5(h)

    # Function to compute the integral for a given h
    def compute_integral(h):
        return np.sum(h) * dx

    # Constraint: integral of h must be 1
    def constraint(h):
        return compute_integral(h)

    # Initialize h with the initial values, scaled to match the integral constraint
    h_initial = initial_h_values[:n_points]
    h_initial = np.interp(h_initial, (h_initial.min(), h_initial.max()), (0, 1))
    h_initial = h_initial / np.sum(h_initial) * n_points / 2

    # Optimization: minimize the maximum overlap integral using simulated annealing
    def objective_function(h):
        return evaluate(h)

    # Simulated Annealing Parameters
    T = 100.0
    cooling_rate = 0.995
    max_iterations = 1000
    current_h = h_initial.copy()
    best_h = current_h.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Perturb h
        perturbation = np.random.normal(0, 0.1, size=n_points)
        perturbation = np.clip(perturbation, -0.5, 0.5)
        new_h = np.clip(current_h + perturbation, 0, 1)

        # Rescale to ensure integral is 1
        new_h = new_h / np.sum(new_h) * n_points / 2

        # Evaluate the objective function
        new_c5 = evaluate(new_h)

        # Acceptance probability
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

        T *= cooling_rate

    # Return the best solution found
    return best_h, best_c5, n_points
