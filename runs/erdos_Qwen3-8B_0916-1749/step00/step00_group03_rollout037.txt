```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from numpy import linalg as la
from scipy.integrate import trapz
from scipy.optimize import minimize, Bounds

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Number of points
    n_points = 100  # Optimal for balance between speed and accuracy
    
    # Define initial guess based on preloaded initial_h_values or random
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess
    
    # Ensure the sum of h is n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    
    # Define the objective function to minimize the maximum overlap
    def objective(x):
        h_current = x
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Optimization bounds
    bounds = Bounds(0, 1)
    
    # Define the objective as a minimization problem
    def objective_min(x):
        return objective(x)
    
    # Use the L-BFGS-B algorithm for optimization
    result = minimize(
        objective_min,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    
    # Compute the corresponding C5 value
    c5_bound = evaluate_erdos_solution(best_h)
    
    # Ensure we stay within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return (best_h, c5_bound, n_points)
    
    return (best_h, c5_bound, n_points)
```
