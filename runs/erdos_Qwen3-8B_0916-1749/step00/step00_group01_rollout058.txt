```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define initial h_values
    h_initial = initial_h_values
    
    # Best solution found so far
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    
    # Track time
    start_time = time.time()
    
    # Function to evaluate C5
    def evaluate_c5(h):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization parameters
    n_points_options = [100, 200, 300, 400, 500]
    for n_points in n_points_options:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate initial guess with n_points samples
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        total_sum = n_points / 2
        
        # Generate a rough initial guess with alternating values
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        
        # Convert to float and scale to meet the integral constraint
        h = h.astype(np.float64)
        current_sum = np.sum(h)
        scaling_factor = total_sum / current_sum if current_sum != 0 else 1.0
        h *= scaling_factor
        
        # Define the optimization problem
        def objective(x):
            h_new = x.copy()
            dx = 2.0 / len(h_new)
            corr = np.correlate(h_new, 1 - h_new, mode="full")
            c5_val = np.max(corr) * dx
            return c5_val
        
        def constraints(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use a local optimization method
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result.success:
            best_h = result.x
            best_c5 = result.fun
            
            # Integrate to check if constraint is satisfied
            integral = np.sum(best_h) * (2.0 / len(best_h))
            if abs(integral - 1.0) > 1e-6:
                print("Integral constraint not satisfied")
            else:
                print(f"Found solution with n_points={n_points}, C5={best_c5}")
        
    return best_h, best_c5, len(best_h)
```
