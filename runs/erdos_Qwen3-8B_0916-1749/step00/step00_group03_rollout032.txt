import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial sample from the given initial_h_values
    initial_h_values = np.array([0.85, 0.9, 0.95, 0.9, 0.85, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 
                                 0.1, 0.05, 0.0, 0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 
                                 0.45, 0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0, 
                                 0.95, 0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55, 0.5, 0.45, 0.4, 
                                 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.0, 0.05, 0.1, 0.15, 0.2, 
                                 0.25, 0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 
                                 0.85, 0.9, 0.95, 1.0])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Compute initial overlap constant
    def compute_c5(h):
        h_reshaped = np.concatenate((h, h))  # To handle circular shifts
        correlation = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(correlation * dx)
        return overlap
    
    initial_c5 = compute_c5(initial_h_values)
    best_c5 = initial_c5
    best_h = initial_h_values.copy()
    
    # Function to compute the overlap integral for a given h
    def compute_overlap(h):
        nonlocal best_c5, best_h
        h_reshaped = np.concatenate((h, h))  # To handle circular shifts
        # Compute all shifts and their overlaps
        max_overlap = 0
        for k in range(1, len(h) + 1):
            if k > len(h) // 2:
                break
            shifted = h_reshaped[k: k + len(h)]
            overlap = np.sum(h * (1 - shifted)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Optimization function
    def objective(x):
        nonlocal best_h, best_c5
        h = x.copy()
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Optimization bounds
    bounds = Bounds([0.0] * n_points, [1.0] * n_points)
    
    # Perform optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=initial_h_values,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'disp': False},
        tol=1e-6
    )
    
    # If we run out of time, use the best solution found so far
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
