    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        # Compute (1 - h[x + k]) for all k (with wrap-around)
        c5_values = []
        for k in range(n_points):
            # Ensure x + k is in [0, 2]
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            c5_values.append(overlap)
        return max(c5_values)

    # Function to compute the objective and constraints
    def objective(x):
        return compute_c5(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Bounds for each sample
    bounds = Bounds(0, 1)

    # Initial guess
    initial_guess = initial_h

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    time_start = time.time()

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=tol,
        max_iter=max_iter,
        options={'gtol': tol, 'disp': False}
    )

    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    n_points = len(best_h)

    # Ensure we stay within the time budget
    elapsed_time = time.time() - time_start
    if elapsed_time > budget_s:
        print(f"Time exceeded budget. Returning best found so far.")
        return (initial_h, compute_c5(initial_h), n_points)
