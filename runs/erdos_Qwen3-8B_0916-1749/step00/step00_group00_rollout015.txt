import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import warnings

# Pre-imported functions (simulated for the purpose of this code)
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the overlap integral for all k
    corr = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Simulated initial construction (n=45 samples)
    return np.random.rand(45) * 0.5 + 0.25

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define initial sampling points
    n_points = 100
    x = np.linspace(0, 2, n_points)
    
    # Initial guess: uniform distribution
    h_initial = np.ones(n_points) * 0.5
    h_initial = np.clip(h_initial, 0, 1)
    
    # Objective function: minimize the maximum overlap integral
    def objective(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        # Compute correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraints: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using bounded optimization
    result = optimize.minimize(
        objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check elapsed time and return the best result
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
