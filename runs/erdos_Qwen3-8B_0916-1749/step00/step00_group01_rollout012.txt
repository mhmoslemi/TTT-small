```python
import numpy as np
import scipy.optimize as optimize
from scipy.integrate import simps
from scipy.optimize import minimize
import time
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Precompute initial h_values
    h_initial = initial_h_values
    n_points_initial = len(h_initial)
    
    # Convert initial h to numpy array
    h = np.array(h_initial)
    
    # Define the objective function to minimize
    def objective(h):
        # Normalize the h to ensure the integral constraint
        h = h / np.sum(h)
        # Compute the correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        # Compute the maximum value
        c5 = np.max(correlation)
        return c5
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - (len(h) / 2)
    
    # Define the bounds for each h[i]
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Optimization settings
    max_iter = 200
    tol = 1e-5
    
    # Initial guess
    initial_guess = h_initial
    
    # Optimization loop
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_guess
    
    for _ in range(5):
        # Initial guess for this iteration
        if _ == 0:
            x0 = initial_guess
        else:
            x0 = np.random.uniform(0, 1, len(h))
        
        # Perform constrained optimization
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            method='SLSQP',
            options={'maxiter': max_iter, 'ftol': tol, 'eps': 1e-6}
        )
        
        # Extract the result
        h_opt = result.x
        c5 = objective(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        # Check if we've used up the budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution
    return (best_h, best_c5, len(best_h))
```
