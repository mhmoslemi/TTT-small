import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    # Initial h_values is provided
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_C5(h_values):
        h = np.array(h_values)
        # Compute all shifted 1 - h(x + k)
        # Use numpy's correlate for efficiency
        corr = np.correlate(h, 1 - h, mode="full")
        # The maximum value across all correlations
        max_overlap = np.max(corr)
        # Normalize by dx
        c5 = max_overlap * dx
        return c5

    def objective(h_values):
        return compute_C5(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Bounds for h_values: [0, 1]
    bounds = Bounds(0, 1)

    # Initial guess
    h_initial = initial_h_values.copy()

    # Optimization settings
    max_iter = 100
    tol = 1e-6

    # Define the constraints
    cons = LinearConstraint(np.ones(n_points), lb=0, ub= n_points / 2)

    # Setup the optimization problem
    result = minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': max_iter, 'ftol': tol, 'disp': False},
        tol=tol,
        callback=lambda x: None
    )

    best_h_values = result.x
    best_c5 = compute_C5(best_h_values)
    return best_h_values, best_c5, len(best_h_values)
