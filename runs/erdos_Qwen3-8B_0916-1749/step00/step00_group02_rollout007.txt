```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def initial_h_values(n_points):
    return np.array([0.0 for _ in range(n_points)])

def evaluate_erdos_solution(h):
    h = np.array(h)
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.concatenate([h, h])
    overlap = np.correlate(h, 1 - h, mode='full')
    overlap = np.abs(overlap)
    max_overlap = np.max(overlap)
    c5 = max_overlap * dx
    return c5

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial guess based on the current construction
    n_points = 500
    h = initial_h_values(n_points)
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the objective function (max overlap)
    def objective(x):
        h = x.copy()
        # Ensure the integral constraint is satisfied
        if np.abs(np.sum(h) - n_points / 2) > 1e-8:
            return float('inf')
        # Compute the overlap integral
        h = np.concatenate([h, h])
        overlap = np.correlate(h, 1 - h, mode='full')
        overlap = np.abs(overlap)
        return np.max(overlap) * (2.0 / n_points)

    # Define the constraints
    constraints = [
        {
            'type': 'eq',
            'fun': lambda x: np.sum(x) - n_points / 2
        }
    ]
    
    # Set up the optimization
    result = minimize(
        objective,
        x0=h,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # If we ran out of time, return the best solution found so far
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    # If optimization completed, return the solution
    return best_h, best_c5, n_points
```
