import numpy as np

import time

from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefine some constants
    n_points = 100  # Start with 100 points, can be increased if needed
    
    # Define initial h_values using the initial construction provided
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.array(initial_h)
    
    def objective(h):
        """
        Objective function to minimize the max overlap integral (C5)
        """
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * (2.0 / n_points))
        return c5
    
    def constraint_sum(h):
        """Constraint: sum(h) == n_points / 2"""
        return np.sum(h) - n_points / 2
    
    # Initial guess from the provided initial_h_values
    initial_guess = initial_h[:n_points]
    
    # Set up bounds for each point
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up the optimization problem
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # If the result is not improving, try a random perturbation
    if best_c5 > 0.38:
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            c5 = objective(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h
                best_c5 = c5
    
    # Final check for convergence
    if time.time() - start_time > budget_s - 5:
        print("Time budget exceeded, returning best found solution")
        return best_h, best_c5, n_points
    
    # Return the best solution
    return best_h, best_c5, n_points
