import time
from scipy.optimize import Bounds, LinearConstraint, minimize

    np.random.seed(seed)
    
    # Initial construction from initial_h_values

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        initial_h = (initial_h / sum_h) * (n_points / 2)
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        return compute_c5(h)
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a weighted average between initial guess and a uniform distribution
    uniform_h = np.full(n_points, 0.5)
    h0 = (initial_h + uniform_h) / 2
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    time_limit = budget_s
    
    start_time = time.time()
    
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        options={
            'maxiter': max_iter,
            'ftol': tol,
            'gtol': tol,
            'eps': 1e-8
        }
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we can reduce the number of points
    n_points = len(best_h)
    if n_points < 100:
        # Try with fewer points by down-sampling
        downsampled_h = []
        for i in range(n_points):
            downsampled_h.append(best_h[i])
        best_h = downsampled_h
        best_c5 = compute_c5(best_h)
        n_points = len(best_h)
    
    # Ensure the integral condition is met
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        best_h = (best_h / sum_h) * (n_points / 2)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > time_limit:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
