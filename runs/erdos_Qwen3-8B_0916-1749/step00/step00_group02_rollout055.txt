import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def initial_h_values(n_points=45):
    # Initial construction from known literature
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1.0
        elif i < 30:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def evaluate_erdos_solution(h_values, n_points):
    # Compute the overlap integral for all shifts
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_extended = np.concatenate([h, h, h])[n_points:-n_points]
    c5_values = []
    for k in range(1, n_points):
        shifted = np.roll(h_extended, k)
        overlap = np.sum(h * (1 - shifted)) * dx
        c5_values.append(overlap)
    return max(c5_values)

def cost_function(h_values, n_points):
    return evaluate_erdos_solution(h_values, n_points)

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 500
    n_iter = 100
    best_c5 = float('inf')
    best_h = initial_h_values(n_points)
    best_c5 = cost_function(best_h, n_points)
    
    start_time = time.time()
    
    for _ in range(n_iter):
        # Perturb the current solution
        h = np.array(best_h)
        perturbation = np.random.normal(0, 0.1, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral condition is satisfied
        total = np.sum(h)
        scaling_factor = (n_points / 2) / total
        h = h * scaling_factor
        # Evaluate the solution
        current_c5 = cost_function(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
