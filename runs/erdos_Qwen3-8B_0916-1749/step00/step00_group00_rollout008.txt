        overlap = np.max(corr) * (2.0 / n_points)
        return overlap

    def compute_gradient(h):
        h = np.array(h)
        h = h / np.mean(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(corr)
        overlap = corr[max_index] * (2.0 / n_points)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                if j + i < len(corr):
                    corr_val = corr[j + i]
                    if corr_val == max_index:
                        grad[i] += (1 - h[j]) * (2.0 / n_points)
                        grad[j] -= h[i] * (2.0 / n_points)
        return grad

    def compute_loss(h):
        h = np.array(h)
        h = h / np.mean(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)

    def loss(h):
        h = np.array(h)
        h = h / np.mean(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)

    def constraints(h):
        h = np.array(h)
        h = h / np.mean(h) * (n_points / 2)
        sum_h = np.sum(h)
        return sum_h - (n_points / 2)
