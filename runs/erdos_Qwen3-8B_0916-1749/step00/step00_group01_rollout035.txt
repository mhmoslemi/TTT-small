import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    # Predefined initial h_values with n=47 samples
    initial_h_values = np.load('initial_h_values.npy')

    def evaluate_erdos_solution(h_values, n_points):
        dx = 2.0 / n_points
        h = np.array(h_values).flatten()
        shifted_h = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def objective_function(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted_h = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_sum(h_values):
        return np.sum(h_values) - len(h_values) / 2

    # Start from the initial_h_values
    initial_n = len(initial_h_values)
    initial_h = initial_h_values
    current_c5 = evaluate_erdos_solution(initial_h, initial_n)
    best_h = initial_h
    best_c5 = current_c5
    best_n = initial_n

    # Define a function to optimize a given n
    def optimize_for_n(n_points):
        # Define bounds: 0 <= h[i] <= 1
        bounds = Bounds(0, 1)
        # Initial guess
        initial_guess = np.ones(n_points) * 0.5
        # Objective function
        def objective(x):
            return objective_function(x)
        # Use L-BFGS-B for optimization
        result = opt.minimize(
            fun=objective,
            x0=initial_guess,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x, result.fun
        else:
            return None, float('inf')

    # Try various n_points starting from the initial n and search in both directions
    for n_point in [initial_n - 10, initial_n, initial_n + 10]:
        if n_point < 10:
            continue
        # Optimize for n_point
        start_time = time.time()
        optimized_h, c5 = optimize_for_n(n_point)
        if optimized_h is not None:
            if c5 < best_c5:
                best_h = optimized_h
                best_c5 = c5
                best_n = n_point
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s * 0.9:
                break

    return best_h, best_c5, best_n
