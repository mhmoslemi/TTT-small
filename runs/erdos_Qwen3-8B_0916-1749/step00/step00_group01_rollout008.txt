import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy import sparse
from scipy.sparse import csr_matrix
from scipy.sparse.csgraph import reverse_cuthill_mckee
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Precompute initial h_values
    initial_h_values = np.load('initial_h_values.npy')
    n_points_initial = len(initial_h_values)
    
    # Define the function to compute overlap integral for a given h
    def compute_overlap(h, dx):
        h = np.array(h)
        n = len(h)
        overlap = []
        for k in range(1, n):
            shifted = np.roll(h, k)
            integral = np.sum(h * (1 - shifted)) * dx
            overlap.append(integral)
        return max(overlap)
    
    # Define the objective function for optimization
    def objective(h, dx):
        return compute_overlap(h, dx)
    
    # Try different n_points from the initial one and smaller
    n_points_candidates = [n_points_initial // 2, n_points_initial // 3, n_points_initial // 4, n_points_initial // 5]
    n_points_candidates = list(set(n_points_candidates))
    n_points_candidates.sort(reverse=True)
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_points_initial
    
    for n in n_points_candidates:
        if n < 2:
            continue
        
        dx = 2.0 / n
        h = np.zeros(n)
        
        # Initial guess: uniform distribution
        h = np.random.uniform(0, 1, n)
        sum_h = np.sum(h)
        h = h / (sum_h / (n / 2))
        
        # Bounds for h: [0, 1]
        bounds = [(0, 1) for _ in range(n)]
        
        # Define the constraints: sum(h) * dx = 1
        # sum(h) = n / 2
        def constraint_sum(h):
            return np.sum(h) - n / 2
        
        # Optimization problem: minimize overlap integral
        def objective_func(x):
            return compute_overlap(x, dx)
        
        # Perform optimization using L-BFGS-B
        result = optimize.minimize(objective_func, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8})
        h_opt = result.x
        c5 = compute_overlap(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n
    
    # If no candidate found, fall back to initial n_points
    if best_h is None:
        dx = 2.0 / n_points_initial
        best_h = initial_h_values
        best_c5 = compute_overlap(best_h, dx)
        best_n = n_points_initial
    
    return best_h, best_c5, best_n
