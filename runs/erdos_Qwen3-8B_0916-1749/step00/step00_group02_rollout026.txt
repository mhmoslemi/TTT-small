import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions and variables
evaluate_erdos_solution = lambda h_values: max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(45) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    elapsed = 0
    
    # Parameters
    n_points = 100  # Optimal number of points for precision and speed
    dx = 2.0 / n_points
    max_iterations = 200
    convergence_threshold = 1e-5
    step_size = 0.01
    
    # Initialize h_values with a base distribution
    h_values = np.random.rand(n_points) * 2 - 1
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.sum(h_values) * n_points / 2
    
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    for iteration in range(max_iterations):
        if elapsed > budget_s:
            break
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Compute overlap integrals
        overlaps = np.correlate(h_values, 1 - h_values, mode="full")
        overlaps = overlaps * dx
        
        # Find the maximum overlap
        max_overlap = np.max(overlaps)
        
        # Compute gradients for each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Impact on the overlap integral for this point
            impact = 0
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    impact += (1 - h_values[i + k]) if k > 0 else h_values[i + k]
            grad[i] = -impact
        
        # Update h_values using gradient descent
        h_values = np.clip(h_values + step_size * grad, 0, 1)
        h_values = h_values / np.sum(h_values) * n_points / 2
        
        # Check for convergence
        if np.abs(current_c5 - best_c5) < convergence_threshold:
            break
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5, n_points
