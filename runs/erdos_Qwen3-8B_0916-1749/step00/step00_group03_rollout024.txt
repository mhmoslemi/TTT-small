```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial number of samples
    n_points = 100
    dx = 2.0 / n_points
    max_shifts = 5  # We evaluate shifts from -1 to 1 (inclusive)

    # Initial guess
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    h = np.roll(h, n_points // 4)
    h = h / np.sum(h) * n_points / 2
    
    def evaluate(h):
        nonlocal n_points, dx, max_shifts
        h = h.copy()
        shifts = np.linspace(-1, 1, max_shifts + 1)
        current_c5 = 0.0
        for shift in shifts:
            shifted = np.roll(h, int(round(shift * n_points / 2)))
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > current_c5:
                current_c5 = overlap
        return current_c5
    
    def objective(h):
        return evaluate(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def bounds(h):
        return np.array([0.0, 1.0])
    
    # Optimization using a local solver
    result = optimize.minimize(
        objective,
        h,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
```
