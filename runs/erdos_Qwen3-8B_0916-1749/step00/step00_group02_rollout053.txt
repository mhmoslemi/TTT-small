    best_h = h.copy()

        nonlocal best_c5
        # Compute the overlap integral
        h_reshaped = h / n_points

    for _ in range(30):
        # Evaluate current solution
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Update rule
        h = h * np.clip(1 - np.random.uniform(0.01, 0.1), 0, 1)

        if c5 < best_c5:

    # Return best solution
