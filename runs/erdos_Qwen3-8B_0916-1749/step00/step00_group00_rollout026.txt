import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 if i % 3 == 0 else 0.0
    h = np.concatenate([h, h])
    h = h[:n_points]
    h = np.random.permutation(h)
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0.0, 1.0)

    def compute_C5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = np.array(h_values)
        result = []
        for k in range(1, n):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted))
            result.append(overlap * dx)
        if len(result) == 0:
            return 0.0
        return np.max(result)

    def objective(x):
        nonlocal h
        h = x
        return compute_C5(h)

    def constraint(x):
        nonlocal h
        h = x
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = result.x
    best_C5 = compute_C5(best_h)
    return best_h, best_C5, n_points
