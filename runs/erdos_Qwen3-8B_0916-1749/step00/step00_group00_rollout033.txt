```python
import numpy as np
import scipy.optimize as optimize
import time
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a moderate number of points
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 2 - 1  # Initial guess

    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h))  # Duplicate to handle wrap-around
        overlap = np.correlate(h, 1 - h[:n_points], mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx

    def objective(x):
        return evaluate_c5(x)

    def gradient(x):
        h = np.array(x)
        h = np.concatenate((h, h))
        h_shifted = np.roll(h, 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_i = h[i]
            h_shifted_i = h_shifted[i]
            grad[i] = (h_shifted_i - h_i) * (1 - h[i]) * dx
        return grad[:n_points]

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h[initial_h > 1] = 1
    initial_h[initial_h < 0] = 0

    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2

    # Setup optimization with constraints
    bounds = Bounds(0, 1)
    x0 = initial_h.copy()

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=[
            {'type': 'eq', 'fun': constraint}
        ]
    )

    if result.success:
        best_h = result.x
    else:
        best_h = initial_h

    # Evaluate the best solution
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n_points
```
