```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
from scipy.integrate import trapz
from scipy.signal import correlate
import time
import sys

# Pre-imported functions
def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute C5
    c5_values = []
    for k in range(1, n_points - 1):
        # Compute the overlap integral for shift k
        shifted_h = np.roll(h, k)
        integral = trapz(h * (1 - shifted_h), dx=dx)
        c5_values.append(integral)
    return max(c5_values)

def initial_h_values():
    return np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess with 70 samples
    n_points = 70
    h = initial_h_values()
    if len(h) != n_points:
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = (h - h.min()) / (h.max() - h.min())  # Normalize to [0, 1]
        h = (h * n_points / 2)  # Ensure integral over [0, 2] is 1

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Try reducing n_points to 50 for faster evaluation
    n_points = 50
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = (h - h.min()) / (h.max() - h.min())
    h = (h * n_points / 2)  # Ensure integral over [0, 2] is 1
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Try reducing n_points to 30 for even faster evaluation
    n_points = 30
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = (h - h.min()) / (h.max() - h.min())
    h = (h * n_points / 2)  # Ensure integral over [0, 2] is 1
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Try reducing n_points to 20 for even faster evaluation
    n_points = 20
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = (h - h.min()) / (h.max() - h.min())
    h = (h * n_points / 2)  # Ensure integral over [0, 2] is 1
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Final evaluation with best_h
    best_c5 = evaluate_erdos_solution(best_h)
    current_time = time.time()
    remaining_time = budget_s - (current_time - start_time)
    if remaining_time > 1:
        # Try a local optimization
        def objective(x):
            x = x.reshape(n_points)
            x = np.clip(x, 0, 1)
            x = (x - x.min()) / (x.max() - x.min())
            x = (x * n_points / 2)
            return evaluate_erdos_solution(x)

        result = optimize.minimize(
            objective,
            x0=best_h,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, len(best_h)
```
