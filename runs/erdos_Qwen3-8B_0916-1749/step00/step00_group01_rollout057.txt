```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 100  # Starting with 100 points for balance between accuracy and speed
    h = np.copy(initial_h_values[:n_points])  # Start with initial guess
    
    # Define the overlap integral for a given h
    def compute_C5(h):
        # Compute the overlap integral using cross-correlation
        dx = 2.0 / n_points
        h_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_corr) * dx
        return c5
    
    # Objective function to minimize
    def objective(h):
        return compute_C5(h)
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    result = minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # Check if we need to increase n_points for better accuracy
    if best_c5 > 0.380876:
        n_points = 200
        h = np.zeros(n_points)
        # Initialize with a more balanced approach
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Perform another optimization with increased n_points
        result = minimize(
            objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'disp': False}
        )
        best_h = result.x
        best_c5 = compute_C5(best_h)
    
    # Final check for very tight bounds
    if best_c5 > 0.380876:
        n_points = 300
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        result = minimize(
            objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 30, 'disp': False}
        )
        best_h = result.x
        best_c5 = compute_C5(best_h)
    
    return (best_h, best_c5, n_points)
```
