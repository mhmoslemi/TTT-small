import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, DifferentialEvolution
from functools import lru_cache
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 1000  # Starting with 1000 points for better resolution
    dx = 2.0 / n_points
    max_shift = 5  # Max shift to consider for overlap
    max_k = int(2.0 / dx)  # Maximum index to consider due to periodicity

    # Initial solution
    h_initial = initial_h_values[:n_points]
    if len(h_initial) != n_points:
        h_initial = np.repeat(h_initial, 2)  # Adjust for n_points
        h_initial = h_initial[:n_points]
    
    # Initialize bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        # Ensure integrals are satisfied
        if np.abs(np.sum(h) * dx - 1.0) > 1e-10:
            return np.inf
        # Compute overlap integrals
        overlap = []
        for k in range(1, max_k + 1):
            shifted = np.roll(h, k)
            integral = np.sum(h * (1 - shifted)) * dx
            overlap.append(integral)
        return np.max(overlap)

    # Use differential evolution for optimization
    result = DifferentialEvolution(objective, bounds, strategy='best1bin', popsize=20, maxiter=500, tol=1e-10, mutation=0.9, recombination=0.5)

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we need to refine further
    if time.time() - start_time < budget_s - 5:
        # Try a smaller n_points for refinement
        n_points_refine = 500
        dx_refine = 2.0 / n_points_refine
        max_k_refine = int(2.0 / dx_refine)
        bounds_refine = [(0.0, 1.0) for _ in range(n_points_refine)]
        
        # Interpolate best_h to refine
        best_h_refined = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), best_h)
        
        def objective_refine(h):
            if np.abs(np.sum(h) * dx_refine - 1.0) > 1e-10:
                return np.inf
            overlap = []
            for k in range(1, max_k_refine + 1):
                shifted = np.roll(h, k)
                integral = np.sum(h * (1 - shifted)) * dx_refine
                overlap.append(integral)
            return np.max(overlap)

        result_refine = DifferentialEvolution(objective_refine, bounds_refine, strategy='best1bin', popsize=20, maxiter=200, tol=1e-10, mutation=0.9, recombination=0.5)
        best_h_refined = result_refine.x
        best_c5 = evaluate_erdos_solution(best_h_refined)
    
    # Ensure n_points is reasonable
    if len(best_h_refined) > 2000:
        best_h_refined = best_h_refined[:2000]
    
    return best_h_refined, best_c5, len(best_h_refined)
