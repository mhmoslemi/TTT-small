import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 200  # Start with 200 points (adjust as needed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess from provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    # Function to compute the C5 value
    def compute_C5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_values = h
        shifted = np.zeros_like(h_values)
        for k in range(1, len(h_values) - 1):
            shifted[k:] = h_values[:len(h_values) - k]
        overlap = np.sum(h_values * (1 - shifted))
        return overlap * dx
    
    # Objective function for optimization (to minimize C5)
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = optimize.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Get best solution
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return (best_h, best_C5, n_points)
