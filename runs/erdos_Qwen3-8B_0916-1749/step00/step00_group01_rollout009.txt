import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Predefined starting point with n_points = 47
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    dx_initial = 2.0 / n_points_initial
    
    # Define the overlap integral function
    def compute_C5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points // 2 + 1):
            shifted_h[k:] = h[:n_points - k]
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return np.max(overlap)
    
    # Objective function to minimize
    def objective(h_values):
        return compute_C5(h_values)
    
    # Define bounds and constraints
    def constraints(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        sum_h = np.sum(h_values)
        return [sum_h - n_points / 2]
    
    # Evaluation function
    def evaluate(h_values):
        return compute_C5(h_values)
    
    # Function to perform optimization
    def optimize_solution(initial_h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraints}
        result = optimize.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x, result.fun
    
    # Function to refine solution with larger n_points
    def refine_solution(h_initial, n_initial, n_final):
        n_points = n_final
        dx = 2.0 / n_points
        h_refined = np.zeros(n_points)
        # Use interpolation or nearest neighbor to assign values
        for i in range(n_points):
            x = i * dx
            # Find the closest point in the initial h_values
            idx = np.argmin(np.abs(np.linspace(0, 2, n_initial) - x))
            h_refined[i] = h_initial[idx]
        return h_refined
    
    # Initial optimization with initial_h
    start_time = time.time()
    
    # First, optimize with the initial_h
    h_initial, c5_initial = optimize_solution(initial_h, n_points_initial)
    print(f"Initial C5: {c5_initial} with n_points={n_points_initial}")
    
    # Refine with larger n_points
    n_points_refined = 100
    h_refined = refine_solution(h_initial, n_points_initial, n_points_refined)
    h_refined, c5_refined = optimize_solution(h_refined, n_points_refined)
    print(f"Refined C5: {c5_refined} with n_points={n_points_refined}")
    
    # Final optimization with even larger n_points (if time allows)
    n_points_final = 200
    h_final = refine_solution(h_refined, n_points_refined, n_points_final)
    h_final, c5_final = optimize_solution(h_final, n_points_final)
    print(f"Final C5: {c5_final} with n_points={n_points_final}")
    
    # Ensure we stay within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning the best solution found")
        return h_refined, c5_refined, n_points_refined
    
    return h_final, c5_final, n_points_final
