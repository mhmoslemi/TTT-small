import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, linprog
from scipy.sparse import csc_matrix
from scipy.sparse.linalg import lsqr
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and precision
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    # Use initial_h_values provided, if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Ensure initial_h has correct length
        if len(initial_h) != n_points:
            initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    # Normalize to satisfy the integral constraint
    total_integral = np.sum(initial_h) * dx
    scaling_factor = (1 / total_integral) if total_integral != 0 else 1
    initial_h = np.clip(initial_h * scaling_factor, 0, 1)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        total_integral = np.sum(h) * dx
        if total_integral == 0:
            return float('inf')
        scaled_h = h / total_integral
        c5_values = []
        for k in range(1, n_points - 1):  # k from 1 to n_points-2
            shifted_h = np.roll(h, k)
            overlap = np.sum(scaled_h * (1 - shifted_h))
            c5_values.append(overlap)
        return max(c5_values)
    
    # Function to evaluate a candidate h
    def evaluate(h):
        return compute_c5(h)
    
    # We start with the initial guess
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    # Define a simplex optimization with bounds [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(
        evaluate,
        x0=best_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_h.copy()
        best_c5 = evaluate(best_h)
    
    # Try using the Nelder-Mead method if L-BFGS-B failed
    result_nm = minimize(
        evaluate,
        x0=best_h,
        bounds=bounds,
        method='Nelder-Mead',
        options={'maxiter': 100}
    )
    if result_nm.fun < best_c5:
        best_h = result_nm.x
        best_c5 = result_nm.fun
    
    # Try using a local descent from the best so far
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        candidate_h = best_h + perturbation
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (np.sum(candidate_h) * dx) / (np.sum(candidate_h) * dx)
        c5 = evaluate(candidate_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate_h
    
    # Check if we can improve by increasing the number of points
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.zeros(n_points)
        if 'initial_h_values' in globals():
            initial_h = np.array(initial_h_values)
            initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        else:
            initial_h = np.sin(np.pi * np.linspace(0, 2, n_points)) / 2 + 0.5
        initial_h = np.clip(initial_h, 0, 1)
        total_integral = np.sum(initial_h) * dx
        scaling_factor = (1 / total_integral) if total_integral != 0 else 1
        initial_h = np.clip(initial_h * scaling_factor, 0, 1)
        
        def compute_c5_higher_res(h):
            h = np.clip(h, 0, 1)
            total_integral = np.sum(h) * dx
            if total_integral == 0:
                return float('inf')
            scaled_h = h / total_integral
            c5_values = []
            for k in range(1, n_points - 1):  # k from 1 to n_points-2
                shifted_h = np.roll(h, k)
                overlap = np.sum(scaled_h * (1 - shifted_h))
                c5_values.append(overlap)
            return max(c5_values)
        
        best_h_higher = initial_h.copy()
        best_c5_higher = compute_c5_higher_res(initial_h)
        result_higher = minimize(
            compute_c5_higher_res,
            x0=initial_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        if result_higher.success:
            best_h_higher = result_higher.x
            best_c5_higher = result_higher.fun
        else:
            best_h_higher = initial_h.copy()
            best_c5_higher = compute_c5_higher_res(initial_h)
        
        # Try Nelder-Mead
        result_nm_higher = minimize(
            compute_c5_higher_res,
            x0=best_h_higher,
            bounds=bounds,
            method='Nelder-Mead',
            options={'maxiter': 50}
        )
        if result_nm_higher.fun < best_c5_higher:
            best_c5_higher = result_nm_higher.fun
            best_h_higher = result_nm_higher.x
        
        if best_c5_higher < best_c5:
            best_c5 = best_c5_higher
            best_h = best_h_higher
            n_points = 200
    
    # Final check for best solution
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
