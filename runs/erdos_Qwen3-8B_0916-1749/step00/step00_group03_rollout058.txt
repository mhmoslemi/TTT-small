import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def compute_overlap(h):
        h = h.reshape(-1)
        n = len(h)
        result = np.zeros(n)
        for k in range(1, n):
            result[k] = np.sum(h[:n - k] * (1 - h[k:]))
        return np.max(result) * dx
    
    def objective(x):
        h = x.reshape(-1)
        return compute_overlap(h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
            tol=1e-8
        )
        best_h = result.x
        best_c5 = compute_overlap(best_h)
        return best_h, best_c5, len(best_h)
    except:
        return initial_h, compute_overlap(initial_h), n_points
