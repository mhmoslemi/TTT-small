import numpy as np

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps * dx)
        return c5
    
    def cost_func(h):
        return objective(h)
    
    def gradient_func(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        h_vec = h.reshape(-1)
        overlaps = np.correlate(h_vec, 1 - h_vec, mode="full")
        
        # Compute derivative of overlap with respect to h
        dh = np.zeros_like(h_vec)
        for i in range(n_points):
            for j in range(n_points):
                if i + j < len(overlaps):
                    dh[i] += overlaps[i + j] * (1 - h_vec[j]) * dx
                    dh[i] -= overlaps[i + j] * h_vec[j] * dx
        return dh
    
    def evaluate_solution(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps * dx)
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    def optimize_h():
        initial_guess = initial_h.copy()
        result = optimize.minimize(
            fun=cost_func,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=gradient_func,
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if not result.success:
            result = optimize.minimize(
                fun=cost_func,
                x0=initial_guess,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[
                    {'type': 'eq', 'fun': constraint_sum},
                    {'type': 'ineq', 'fun': constraint_bounds}
                ],
                options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
            )
        return result
    
    start_time = time.time()
    result = optimize_h()
    elapsed_time = time.time() - start_time
    
    if elapsed_time > budget_s:
        warnings.warn(f"Optimization exceeded budget time of {budget_s} seconds.")
        return initial_h, evaluate_solution(initial_h), n_points
    
    best_h = result.x
    best_c5 = evaluate_solution(best_h)
    return best_h, best_c5, n_points
