    def compute_c5(h):
        h = np.array(h)

        overlap = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)

    def objective(x):
        h = x.copy()
        # Normalize to meet the integral constraint
        total = np.sum(h) * dx
        h = h * (1 / total) * (n_points / 2)
        c5 = compute_c5(h)

        # Constraint: integral of h must be 1

    # Setup for optimization
    x0 = h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints_func, 'jac': lambda x: np.array([dx for _ in range(n_points)])}

    # Optimization
    start_time = time.time()
    result = optimize.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5, options={'maxiter': 200})
