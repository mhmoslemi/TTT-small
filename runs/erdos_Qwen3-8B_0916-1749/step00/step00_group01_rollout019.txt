import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 1000
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 0.0001
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial guess
    h = np.copy(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h
    
    def compute_c5(h):
        h = h.reshape(-1)
        n = len(h)
        result = np.correlate(h, 1 - h, mode="full")
        result = result[n - 1:-1] * dx
        return np.max(result)
    
    # Optimization
    for _ in range(max_iterations):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        
        # Try random perturbations
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h * (n_points / 2) / sum_h
        
        # Compute the current C5
        current_c5 = compute_c5(h)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Print progress
        print(f"Iteration {_ + 1}: C5 = {best_c5}")
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
