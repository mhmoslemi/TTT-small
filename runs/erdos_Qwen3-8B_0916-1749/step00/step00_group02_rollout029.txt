# Pre-imported helper functions
def evaluate_erdos_solution(h):
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def initial_h_values():
    return np.load('initial_h.npy')

def objective_func(h_values, dx, k_values):
    max_overlap = 0.0
    for k in k_values:
        shift = np.roll(h_values, k)
        overlap = np.sum(h_values * (1 - shift))
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def constraints_func(h_values, dx):
    return np.sum(h_values) - (len(h_values) / 2)

def create_shifts(n_points):
    return np.arange(1, n_points)
