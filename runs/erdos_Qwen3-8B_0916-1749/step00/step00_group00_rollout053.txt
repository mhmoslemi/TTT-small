```python
import numpy as np
import scipy.optimize as opt
import scipy.interpolate as interp
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function
    def evaluate_candidate(h):
        h = np.array(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute integral constraint
        if np.abs(np.sum(h) * dx - 1) > 1e-8:
            return np.inf
        # Compute the convolution (overlap integral)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[np.abs(corr) > 1e-8]
        corr = corr * dx
        return np.max(corr)
    
    # Optimization function
    def optimize_candidate(h):
        result = minimize(evaluate_candidate, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'ftol': 1e-8, 'maxiter': 100})
        return result.x
    
    # Evaluate initial candidate
    initial_h = initial_h_values
    initial_c5 = evaluate_candidate(initial_h)
    
    # Best solution so far
    best_h = initial_h.copy()
    best_c5 = initial_c5
    
    # Time budget
    start_time = time.time()
    
    # Try optimization with different starting points
    for _ in range(5):
        # Perturb the initial solution
        perturbed = initial_h.copy()
        for i in range(n_points):
            perturbed[i] += np.random.normal(0, 0.05)
            perturbed[i] = np.clip(perturbed[i], 0, 1)
        
        # Optimize
        optimized = optimize_candidate(perturbed)
        
        # Evaluate
        current_c5 = evaluate_candidate(optimized)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = evaluate_candidate(best_h)
    
    return best_h, final_c5, n_points
```
