import numpy as np
import scipy.optimize as optimize
import time
import random

initial_h_values = np.load("initial_h_values.npy")  # Preloaded initial construction

def run(seed=42, budget_s=300, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Precompute initial h_values
    h_initial = initial_h_values
    
    # Try different n_points starting from the initial size and working down
    n_points_list = [len(h_initial), len(h_initial)-1, len(h_initial)-2, len(h_initial)-3, len(h_initial)-4]
    n_points_list = [n for n in n_points_list if n > 10]
    
    best_c5 = float('inf')
    best_h = None
    best_n = len(h_initial)
    
    for n in n_points_list:
        # Create a grid over [0, 2]
        x = np.linspace(0, 2, n, endpoint=False)
        dx = 2.0 / n
        
        # Try different starting points
        for start in range(5):
            # Initialize h with some randomness
            h = np.random.uniform(0, 1, n)
            
            # Normalize to satisfy the integral constraint
            h = h * (n / 2) / np.sum(h)
            
            # Optimize h using gradient descent
            for _ in range(100):
                # Compute current C5
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * dx
                
                # Check if we can improve
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n
                
                # Compute gradients for each point
                grad = np.zeros(n)
                for i in range(n):
                    # For each x_i, compute the contribution to all k
                    for k in range(-n+1, n):
                        if 0 <= i + k < n:
                            # Contribution to the integral at k
                            contrib = 1 - h[i + k]
                            grad[i] += contrib
                    grad[i] = -grad[i] * dx
                
                # Update h with gradient descent
                h += grad * 0.01
                
                # Keep h in [0, 1]
                h = np.clip(h, 0, 1)
            
            # Also try starting from the initial construction
            h = np.random.permutation(h_initial[:n])
            h = h * (n / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
            # Optimize h using gradient descent
            for _ in range(100):
                # Compute current C5
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * dx
                
                # Check if we can improve
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

                grad = np.zeros(n)

                # Update h with gradient descent
                h += grad * 0.01
                
                # Keep h in [0, 1]
                h = np.clip(h, 0, 1)
    
    # Return best solution
    return best_h, best_c5, len(best_h)
