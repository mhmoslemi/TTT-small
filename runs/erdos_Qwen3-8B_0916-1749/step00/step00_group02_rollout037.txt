```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import warnings
from scipy.optimize import Bounds
from scipy.optimize import BoundsError

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 200  # Maximum number of points to try
    n_points = 100  # Starting point
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    def evaluate(h):
        """Compute the max overlap integral."""
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have exactly {} points".format(n_points))
        
        # Compute all shifted 1 - h(x + k)
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k < 0:
                shifted_h = np.roll(h, -k)
            elif k > 0:
                shifted_h = np.roll(h, k)
            else:
                shifted_h = h
            
            shifted_1_minus_h = 1.0 - shifted_h
            overlap = np.sum(h * shifted_1_minus_h) / (n_points / 2)
            overlaps.append(overlap)
        
        # Return the max overlap
        return max(overlaps)
    
    def objective(h):
        """Objective function to minimize."""
        return evaluate(h)
    
    def constraint(h):
        """Constraint on the total integral."""
        return np.sum(h) - n_points / 2
    
    # Start from initial_h_values
    initial_h = initial_h_values[:n_points]
    if np.sum(initial_h) * 2 / n_points < 1 - 1e-8:
        initial_h = np.clip(initial_h, 0, 1)
    if np.sum(initial_h) * 2 / n_points > 1 + 1e-8:
        initial_h = np.clip(initial_h, 0, 1)
    
    # Optimize
    try:
        # Use a local optimizer to minimize the overlap
        result = opt.minimize(
            fun=objective,
            x0=initial_h,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        best_h = result.x
        best_c5 = evaluate(best_h)
    except:
        best_c5 = float('inf')
    
    # Try smaller n_points if necessary
    for n_points in range(50, max_n_points, 10):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate random initial guess
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        try:
            result = opt.minimize(
                fun=objective,
                x0=initial_h,
                bounds=[(0, 1) for _ in range(n_points)],
                method='L-BFGS-B',
                options={'ftol': 1e-6, 'maxiter': 100}
            )
            current_c5 = evaluate(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    return best_h, final_c5, len(best_h)
```
