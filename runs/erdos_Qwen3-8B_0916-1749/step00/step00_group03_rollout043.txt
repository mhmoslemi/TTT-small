```python
import numpy as np
import scipy.optimize as optimize
import time
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points as a balance between speed and resolution

    # Helper function to compute overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Initial guess with initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = initial_h[:n_points] + initial_h[-(n_points - len(initial_h)):] if len(initial_h) > n_points else initial_h + [0] * (n_points - len(initial_h))

    # Convert to numpy array
    initial_h = np.array(initial_h, dtype=float)

    def objective(x):
        nonlocal best_c5
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Define bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]

    # Define the optimizer
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # Check if we have a better solution
    if best_h is not None:
        # Try reducing n_points for faster evaluation
        n_points = 100
        while time.time() - start_time < budget_s - 1:
            # Try with a smaller n_points
            n_points = max(10, n_points - 20)
            # Resample the best_h to n_points
            resampled_h = np.repeat(best_h, len(best_h) // n_points + 1)[:n_points]
            resampled_h = np.clip(resampled_h, 0, 1)
            # Use this as the new initial guess
            initial_h = resampled_h
            # Compute the new c5
            current_c5 = compute_c5(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h.copy()

    # Final solution
    final_h = best_h
    final_c5 = best_c5
    final_n_points = len(final_h)
    return final_h, final_c5, final_n_points
```
