import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interpolate
import time

def initial_h_values(n_points):
    return np.load('initial_h.npy')

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.concatenate([h_values, h_values])
    shifted = np.roll(h, 1)
    overlap = np.correlate(h, shifted, mode='full')
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Start with the initial construction
    n_initial = len(initial_h_values(47))
    h_initial = initial_h_values(47)
    
    # Try with different n_points
    n_points_list = [47, 53, 65, 89, 121, 161]
    
    best_c5 = float('inf')
    best_h = h_initial
    
    for n_points in n_points_list:
        if n_points < 47:
            continue
        
        # Ensure the initial points are resampled to n_points
        t = np.linspace(0, 2, n_points)
        t_initial = np.linspace(0, 2, n_initial)
        h_initial_resampled = np.interp(t, t_initial, h_initial)
        
        # Define the function to minimize
        def objective(x):
            h = np.concatenate([x, x])
            dx = 2.0 / n_points
            shifted = np.roll(h, 1)
            overlap = np.correlate(h, shifted, mode='full')
            return np.max(overlap) * dx
        
        # Constraints: sum(h) * dx = 1 (equivalently sum(h) = n_points / 2)
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        # Bounds: each h[i] ∈ [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a local optimizer
        result = optimize.minimize(objective, x0=h_initial_resampled, bounds=bounds, constraints={"type": "eq", "fun": constraint})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Print progress
        print(f"n_points={n_points}, c5={best_c5}")
    
    return best_h, best_c5, len(best_h)
