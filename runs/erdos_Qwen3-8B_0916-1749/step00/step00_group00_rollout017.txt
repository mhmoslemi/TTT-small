import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    
    # Try a few starting points to avoid local optima
    start_points = [initial_h]
    for i in range(2):
        # Perturb the initial solution slightly
        perturbed = initial_h.copy()
        perturbed[np.random.choice(len(perturbed), size=10, replace=False)] = np.random.uniform(0, 1)
        start_points.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_points_initial
    
    for start in start_points:
        n_points = len(start)
        if n_points > 1000:
            continue
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Ensure the function is in [0, 1]
        h = np.clip(start, 0, 1)
        # Enforce the integral constraint
        integral = np.sum(h) * dx
        if integral == 0:
            continue
        
        # Define the variables
        h_var = cp.Variable(n_points, lb=0, ub=1)
        h_var.value = h
        
        # Set up the objective: minimize the maximum overlap
        overlap = []
        for k in range(1, n_points):
            # Compute (1 - h[x + k]) where x + k is modulo 2
            shifted = np.roll(h, k)
            shifted = np.clip(shifted, 0, 1)
            overlap.append(1 - shifted)
        
        # For each k, compute the integral of h(x) * (1 - h(x + k))
        # We need to find the maximum of these integrals
        max_overlap = cp.Variable()
        constraints = []
        
        for k in range(1, n_points):
            # Compute the integral
            integral_k = cp.sum(h_var * (1 - np.roll(h_var, k))) * dx
            constraints.append(max_overlap >= integral_k)
        
        # Objective: minimize the max_overlap
        obj = cp.Minimize(max_overlap)
        
        # Define the problem
        problem = cp.Problem(obj, constraints)
        
        # Solve the problem
        try:
            start_time = time.time()
            problem.solve(solver=cp.OSQP, eps_abs=1e-6, eps_rel=1e-6, max_iter=1000)
            elapsed = time.time() - start_time
            
            if elapsed > budget_s:
                continue
            
            if problem.status == 'optimal':
                current_h = h_var.value
                # Compute the overlap integral
                max_overlap_val = max_overlap.value
                if max_overlap_val < best_c5:
                    best_c5 = max_overlap_val
                    best_h = current_h
                    best_n = n_points
        except:
            pass
    
    # If no solution found, return initial_h with its value
    if best_h is None:
        best_h = initial_h
        best_c5 = np.max([np.sum(initial_h[i] * (1 - initial_h[(i + k) % n_points_initial])) * (2.0 / n_points_initial) for k in range(1, n_points_initial)])
    
    return best_h, best_c5, best_n
