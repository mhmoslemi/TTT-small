import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.sparse import eye
from scipy.sparse.linalg import lsqr

import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Start with initial h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    
    # Evaluate the initial C5
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        overlap = corr[n_points - 1 : 2 * n_points - 1]
        return np.max(overlap * dx)
    
    c5_initial = compute_c5(h)
    best_c5 = c5_initial
    best_h = h.copy()
    
    # Define the optimization function
    def objective(x):
        nonlocal best_c5, best_h
        h = x
        dx = 2.0 / len(h)
        h = np.concatenate((h, h[:len(h)//2]))
        corr = np.correlate(h, 1 - h[:len(h)], mode='full')
        overlap = corr[len(h) - 1 : 2 * len(h) - 1]
        current_c5 = np.max(overlap * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        return current_c5
    
    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - len(x) / 2
    
    # Optimization settings
    n = len(h)
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use L-BFGS-B for faster convergence
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'disp': False}
    )
    
    # If the optimization didn't converge, try a different starting point
    if result.success:
        h_opt = result.x
        dx = 2.0 / len(h_opt)
        h_opt = np.concatenate((h_opt, h_opt[:len(h_opt)//2]))
        corr = np.correlate(h_opt, 1 - h_opt[:len(h_opt)], mode='full')
        overlap = corr[len(h_opt) - 1 : 2 * len(h_opt) - 1]
        c5_opt = np.max(overlap * dx)
    else:
        # Fallback to initial solution
        h_opt = h
        dx = 2.0 / len(h_opt)
        h_opt = np.concatenate((h_opt, h_opt[:len(h_opt)//2]))
        corr = np.correllete(h_opt, 1 - h_opt[:len(h_opt)], mode='full')
        overlap = corr[len(h_opt) - 1 : 2 * len(h_opt) - 1]
        c5_opt = np.max(overlap * dx)
    
    return (h_opt[:len(h_opt)//2].tolist(), c5_opt, len(h_opt)//2)
