    n_points = len(initial_h)
    
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        # We are interested in the maximum overlap
        max_overlap = np.max(corr)
        return max_overlap
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h[i] to be in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Set up the optimization problem
    result = opt.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 200}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    n_points = len(h_values)
    
    return h_values, c5_bound, n_points
