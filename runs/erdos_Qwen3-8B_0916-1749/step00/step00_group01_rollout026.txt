```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.integrate import trapz
from scipy.signal import correlate
from functools import partial

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the objective function to minimize
    def objective(h):
        h = np.array(h, dtype=np.float64)
        # Compute the overlap integral for all k
        c5_values = []
        for k in range(1, len(h) - 1):
            shift = np.roll(h, k)
            integral = trapz(h * (1 - shift), dx=2.0 / len(h))
            c5_values.append(integral)
        return max(c5_values)
    
    # Define the constraints
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    # Initial guess
    n_points = len(initial_h_values)
    if n_points < 100:
        n_points = 100
    
    # Start from the initial solution
    initial_h = initial_h_values[:n_points].copy()
    initial_h[initial_h < 0] = 0
    initial_h[initial_h > 1] = 1
    initial_h = np.clip(initial_h, 0, 1)
    
    # Perform local optimization
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=None,
        options={'maxiter': 100, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we can reduce the number of points further
    if n_points > 50:
        n_points = 50
        initial_h = best_h[:n_points].copy()
        initial_h[initial_h < 0] = 0
        initial_h[initial_h > 1] = 1
        initial_h = np.clip(initial_h, 0, 1)
        
        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 50, 'disp': False}
        )
        
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Ensure we stay within the time budget
    if time.time() - start_time > budget_s - 10:
        return (best_h, best_c5, len(best_h))
    
    return (best_h, best_c5, len(best_h))
```
