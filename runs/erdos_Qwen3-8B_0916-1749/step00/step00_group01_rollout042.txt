```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = np.tile(h, 2)  # Start with a simple alternating pattern
    h = h[:n_points]
    
    def compute_c5(h):
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant')
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = x
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant')
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
    obj = cp.Minimize(cp.Max(signal.correlate(x.value, 1 - x.value, mode='full') * dx))
    
    start_time = time.time()
    
    # Optimization loop with adaptive steps
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess
        x0 = np.clip(h, 0, 1)
        
        # Optimization using CVXPY
        try:
            prob = cp.Problem(cp.Minimize(cp.Max(signal.correlate(x, 1 - x, mode='full') * dx)), constraints)
            result = prob.solve(solver='MOSEK', eps=1e-4)
            if result is not None and result.success:
                h = x.value
        except:
            pass
    
    c5 = compute_c5(h)
    return h, c5, n_points
```
