```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
from scipy.optimize import Bounds, minimize
import sys
import os

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Initial construction with n=70 samples
    n = 70
    h = np.zeros(n)
    for i in range(n):
        x = i * 2.0 / n
        if x < 0.5:
            h[i] = 0.5
        elif x < 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.5
    return h.tolist()

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Start with the initial solution
    best_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h
    
    # Define a function to evaluate the cost for a given h
    def objective(h):
        nonlocal best_c5, best_h
        start_time = time.time()
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        dx = 2.0 / n_points
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        elapsed = time.time() - start_time
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B optimizer with a budget of 100 iterations
    result = minimize(
        objective,
        x0=initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check if we have a better solution found during optimization
    if best_c5 < result.fun:
        best_h = best_h
    
    # Ensure we don't exceed the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, len(best_h))
    
    return (best_h, best_c5, len(best_h))
```
