import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a reasonable number of points

    # Initial construction from initial_h_values
    global initial_h_values
    if len(initial_h_values) == 0:
        initial_h_values = np.random.uniform(0, 1, n_points)
    else:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    # Ensure the initial_h_values has the correct sum
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / initial_sum
    initial_h_values = np.clip(initial_h_values * scale_factor, 0, 1)

    # Define the function to compute C5
    def compute_C5(h):
        h = np.clip(h, 0, 1)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Define the optimization problem
    def optimize_h_with_cvxpy(h_guess):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap
        # We can use a convex optimization approach to minimize the maximum value
        # by introducing a variable t and constraints t >= overlap for all k
        # Since the overlap depends on the correlation with shifted versions of h,
        # we approximate by considering a few shifts to make the problem tractable

        # Select a subset of shifts to consider
        shifts = np.linspace(-1, 1, 20)
        # Create a list of constraints t >= overlap for each shift
        constraints.extend([cp.sum(h[i] * (1 - h[(i + int(shifts[k])) % n_points])) <= cp.Variable() for k in range(len(shifts))])

        # The total number of constraints grows with the number of shifts, but we'll limit it to 20
        # for computational efficiency

        # Create a variable to represent the maximum overlap
        t = cp.Variable()

        # Define the objective: minimize t
        objective = cp.Minimize(t)

        # Add constraints t >= overlap for each shift
        for k in range(len(shifts)):
            shift = shifts[k]
            idxs = np.floor((np.arange(n_points) + shift) % n_points).astype(int)
            overlap = np.sum(h * (1 - h[idxs]))
            constraints.append(t >= overlap)

        # Solve the optimization problem
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.OSQP, eps=1e-4, max_iter=1000)

        if problem.status != 'optimal':
            return h_guess

        return h.value

    # Evaluate initial guess
    initial_h = np.copy(initial_h_values)
    current_c5 = compute_C5(initial_h)
    best_h = initial_h
    best_c5 = current_c5

    # Try to optimize
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run a local optimization
        h_opt = optimize_h_with_cvxpy(best_h)

        # Check if the optimization succeeded
        if h_opt is not None:
            # Ensure the sum is correct
            h_opt = np.clip(h_opt, 0, 1)
            sum_h_opt = np.sum(h_opt)
            scale_factor = (n_points / 2) / sum_h_opt
            h_opt = np.clip(h_opt * scale_factor, 0, 1)

            # Evaluate the new solution
            c5 = compute_C5(h_opt)

            # Update best solution
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5

        # Try to increase the number of points for better resolution
        n_points *= 2
        n_points = np.clip(n_points, 100, 1000)

        # Reconstruct the initial guess with the new n_points
        initial_h_values = np.random.uniform(0, 1, n_points)
        initial_sum = np.sum(initial_h_values)
        scale_factor = (n_points / 2) / initial_sum
        initial_h_values = np.clip(initial_h_values * scale_factor, 0, 1)

    # Return the best solution found
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale_factor = (n_points / 2) / sum_h
    best_h = np.clip(best_h * scale_factor, 0, 1)

    return best_h, best_c5, n_points
