```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Objective function and constraints
    def compute_c5(h):
        h = np.array(h)
        conv = np.convolve(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Optimization strategy: Greedy local search with random perturbations
    def greedy_local_search(initial_h, max_iter=100, perturbation_size=0.1, n_points=70):
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            # Random perturbation
            indices = np.random.choice(n_points, size=10, replace=False)
            for i in indices:
                h[i] += np.random.uniform(-perturbation_size, perturbation_size)
                h[i] = np.clip(h[i], 0.0, 1.0)
            
            # Check if the integral constraint is satisfied
            total_sum = np.sum(h)
            if total_sum < n_points / 2:
                h += (n_points / 2 - total_sum) / n_points
            elif total_sum > n_points / 2:
                h -= (total_sum - n_points / 2) / n_points
            
            # Compute current C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Start from the initial construction
    best_h, best_c5 = greedy_local_search(initial_h_values, max_iter=50, perturbation_size=0.1, n_points=len(initial_h_values))
    
    # Evaluate the best solution found
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
