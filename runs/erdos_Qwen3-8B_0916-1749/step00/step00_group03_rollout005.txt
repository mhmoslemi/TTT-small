import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
from cvxpy import Problem, Variable, Max, Min, norm, sum_entries, Constant, QuadForm, quad_form, Minimize, Maximize
from scipy.optimize import Bounds, minimize, LinearConstraint, minimize_slsqp
from functools import lru_cache
import itertools

# Predefined initial h_values
initial_h_values = np.load('initial_h_values.npy')  # Assume this is a numpy array of shape (n_points,)

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Start with initial h_values
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    # Normalize to satisfy ∫h dx = 1
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_C5(h):
        h = h / np.sum(h) * (n_points / 2)
        # Correlate h with (1 - h)
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum value is the C5
        max_overlap = np.max(corr)
        # Account for dx
        return max_overlap * dx
    
    # Function to evaluate C5
    def evaluate_candidate(h):
        return compute_C5(h)
    
    # Function to create a shifted version of the array
    def shift_array(arr, k):
        shifted = np.zeros_like(arr)
        for i in range(arr.size):
            shifted[i] = arr[(i - k) % arr.size]
        return shifted
    
    # Function to compute the maximum overlap integral over all shifts k
    def compute_C5_max(h):
        n = len(h)
        max_overlap = 0.0
        for k in range(1, n):
            shifted = shift_array(h, k)
            overlap = np.sum(h * (1 - shifted))
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Define the function to optimize
    def objective(x):
        h = x.reshape((n_points,))
        return compute_C5_max(h)
    
    # Define bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a local optimization method with bounds
    result = minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6},
        jac=None
    )
    
    best_h = result.x
    best_C5 = evaluate_candidate(best_h)
    
    # Return the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return (best_h, best_C5, n_points)
    
    # If not enough time, try a finer grid
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h = best_h[:n_points]
    h = h / np.sum(h) * (n_points / 2)
    
    result = minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6},
        jac=None
    )
    
    best_h = result.x
    best_C5 = evaluate_candidate(best_h)
    
    return (best_h, best_C5, n_points)
