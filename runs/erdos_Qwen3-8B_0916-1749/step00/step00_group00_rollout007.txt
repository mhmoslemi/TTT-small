    def compute_overlap(h):
        h_ten = np.repeat(h, 10)
        h_ten = h_ten[:n_points * 10]
        shifted_h = np.zeros_like(h_ten)
        for k in range(1, 10):
            shifted_h[k:] = h_ten[:-k]
        overlap = np.zeros(10)
        for k in range(1, 10):
            overlap[k] = np.sum(h_ten * shifted_h[k:]) * dx
        return np.max(overlap)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_overlap(h)

    def constraint_func(h_flat):
        return np.sum(h_flat) - n_points / 2

    def constraint_func_jac(h_flat):
        return np.ones_like(h_flat)

    bounds = [(0, 1) for _ in range(n_points)]
    h_flat = h.copy()
    result = optimize.minimize(
        objective,
        x0=h_flat,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_func_func, 'jac': constraint_func_jac}],
        options={'ftol': tolerance, 'maxiter': max_iterations}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > tolerance:
        best_h += (n_points / 2 - sum_h) / n_points
