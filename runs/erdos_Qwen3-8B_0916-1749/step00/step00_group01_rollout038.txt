def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all shifts from k = -n_points + 1 to k = n_points - 1
    max_overlap = 0.0
    for k in range(-n_points + 1, n_points):
        if k == 0:
            continue
        # Compute the overlap integral
        overlap = 0.0
        for i in range(n_points):
            x = i * dx
            x_shift = x + k
            if 0 <= x_shift < 2:
                overlap += h_values[i] * (1 - h_values[np.argmin(np.abs(x_shift - np.arange(n_points) * dx))])
        max_overlap = max(max_overlap, overlap * dx)
    return max_overlap
