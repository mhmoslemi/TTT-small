import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial guess based on provided initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points_initial
    sum_h_initial = np.sum(h_initial)
    
    # Convert to a list of samples
    h_values = h_initial.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.array(h, dtype=np.float64)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to minimize C5 with constraints
    def objective(h):
        return compute_c5(h)
    
    # Constraints: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_sum(h):
        n_points = len(h)
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(np.zeros(len(h_initial)), np.ones(len(h_initial)))
    
    # Initial guess
    h0 = h_initial.copy()
    
    # Optimization settings
    maxiter = 100
    tol = 1e-6
    
    # Run the optimization
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=tol, options={'maxiter': maxiter})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, len(best_h))
