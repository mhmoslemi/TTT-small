```python
import numpy as np
import scipy.optimize
import time
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def initial_h_values(n_points):
    # Initial guess: evenly spaced samples with alternating 0.5 and 0.5
    # This is a placeholder; in practice, we'd load a better initial guess
    return np.full(n_points, 0.5)

def run(seed=42, budget_s=300, **kwargs):
    random.seed(seed)
    start_time = time.time()

    n_points = 100  # Start with a moderate number of points
    h = initial_h_values(n_points)
    h = h * (2.0 / n_points)  # Normalize to satisfy ∫h(x)dx = 1

    # Objective function: maximize the overlap integral
    def objective(x):
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / len(h)))

    # Bounds for each sample
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization: minimize the maximum overlap integral
    result = scipy.optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 200}
    )

    # Extract the best solution
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = evaluate_erdos_solution(best_h)

    # Track time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print(f"Time budget exceeded. Returning current best solution.")
        return best_h, best_c5, n_points

    # Try increasing n_points if time allows
    n_points = min(300, n_points * 2)
    h = initial_h_values(n_points)
    h = h * (2.0 / n_points)

    # Re-optimization with higher n_points
    result = scipy.optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 200}
    )

    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
```
