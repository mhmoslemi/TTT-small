import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=300, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define the overlap integral function
    def compute_c5(h, dx):
        h_array = np.array(h)
        result = signal.correlate(h_array, 1 - h_array, mode="full")
        return np.max(result) * dx

    # Define the objective function with bounds
    def objective(params):
        h = params.reshape(n_points)
        return compute_c5(h, dx)

    # Define bounds for the parameters
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use scipy.optimize to minimize the maximum overlap
    # Use a reduced budget for optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8},
        callback=lambda x: print(f"Time used: {time.time() - start_time:.2f}s")
    )

    h_values = result.x
    c5_bound = compute_c5(h_values, dx)
    n_points = len(h_values)

    return h_values, c5_bound, n_points
