```python
import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    # Compute all shifted versions of (1 - h)
    shifts = np.arange(-n_points + 1, n_points)
    integrals = []
    for shift in shifts:
        shifted_h = np.roll(h, shift)
        product = h * (1 - shifted_h)
        integral = np.sum(product) * dx
        integrals.append(integral)
    return max(integrals)

def initial_h_values():
    return [0.5] * 70

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Reduced from 700 for efficiency
    dx = 2.0 / n_points

    # Initial guess based on the initial_h_values
    initial_h = np.array(initial_h_values(), dtype=np.float64)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    # Define objective function to minimize the maximum overlap
    def objective(x):
        h = np.array(x, dtype=np.float64)
        integrals = []
        for shift in range(-n_points + 1, n_points):
            shifted_h = np.roll(h, shift)
            product = h * (1 - shifted_h)
            integral = np.sum(product) * dx
            integrals.append(integral)
        return max(integrals)

    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization
    start_time = time.time()

    result = optimize.minimize(
        objective,
        x0=initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    elapsed_time = time.time() - start_time

    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning initial solution")
        return initial_h.tolist(), evaluate_erdos_solution(initial_h), len(initial_h)

    h_final = result.x
    c5_bound = result.fun
    n_points_final = len(h_final)
    return h_final.tolist(), c5_bound, n_points_final
```
