```python
import numpy as np
import scipy.optimize as optimize
import time
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction: 47 samples
    h_initial = initial_h_values
    n_points_initial = len(h_initial)

    # Propose a smaller n_points for faster optimization
    n_points = 100  # Try with 100 samples for better performance

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1 : 2 * n_points - 1]  # Take full overlap
        c5 = np.max(corr) * dx
        return c5

    # Objective function for optimization
    def obj(h):
        return compute_c5(h)

    # Constraints: sum(h) == n_points / 2
    def cons(h):
        return np.sum(h) - n_points / 2

    # Initial guess: sample from initial_h_values and interpolate
    def sample_initial_guess():
        indices = np.random.choice(n_points_initial, size=n_points, replace=True)
        h_guess = np.zeros(n_points)
        for i in range(n_points):
            h_guess[i] = initial_h_values[indices[i]]
        return h_guess

    h_guess = sample_initial_guess()

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization using SLSQP
    result = optimize.minimize(obj, h_guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons, 'jac': lambda h: np.zeros(n_points)})

    # Extract the best solution
    h_values = result.x
    c5_bound = compute_c5(h_values)
    elapsed_time = time.time() - start_time

    # Ensure we do not exceed time budget
    if elapsed_time > budget_s:
        print(f"Time exceeded budget: {elapsed_time} > {budget_s}")
        return None

    return h_values, c5_bound, n_points
```
