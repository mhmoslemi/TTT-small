    initial_h = initial_h_values.copy()
    dx = 2.0 / n_initial
    
    # Define the function to compute C_5 for a given h
    def compute_C5(h, dx):
        n = len(h)
        h_shifts = [np.roll(h, shift) for shift in range(1, n)]
        integrals = []
        for shift in h_shifts:
            integral = np.sum(h * (1 - shift)) * dx
            integrals.append(integral)
        return max(integrals)
    
    # Define the objective function to minimize max integral
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_C5(h, dx)
    
    # Define the bounds for the variables
    bounds = [(0, 1) for _ in range(n_initial)]
    
    # Use L-BFGS-B to minimize the maximum integral
    result = optimize.minimize(
        objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    
    # Check if we can reduce the number of points
    n_points = len(best_h)
    n_points_candidates = [n_points // 2, n_points // 3, n_points // 4]
    n_points_candidates = [n for n in n_points_candidates if n >= 10]
    
    best_c5 = compute_C5(best_h, dx)
    best_h_values = best_h
    
    for n in n_points_candidates:
        # Interpolate best_h to n points
        x = np.linspace(0, 2, n_points)
        h_interp = np.interp(x, np.linspace(0, 2, n_points), best_h)
        
        # Clip to [0, 1] and ensure sum(h) * dx = 1
        h = np.clip(h_interp, 0, 1)
        total = np.sum(h) * (2.0 / n)
        if total != 1:
            scaling = 1.0 / total
            h = h * scaling
        
        c5 = compute_C5(h, 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h
            n_points = n
    
    # Final evaluation
    final_h = best_h_values
    final_dx = 2.0 / len(final_h)
    final_c5 = compute_C5(final_h, final_dx)
    
    return (final_h, final_c5, len(final_h))
