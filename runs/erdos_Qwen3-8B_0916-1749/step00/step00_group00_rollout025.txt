import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

# Preloaded initial_h_values (n=45 samples)
initial_h_values = np.loadtxt("initial_h_values.txt")

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points and parameters
    n_points = 100  # Reduced from initial 45 for faster optimization
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-4
    
    # Define the initial guess based on the initial_h_values (resampled)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Objective function: max over all k of the integral of h(x)(1 - h(x + k))
    def objective(h):
        h_values = h.reshape(-1)
        h_array = np.zeros(2 * n_points)
        h_array[:n_points] = h_values
        h_array[n_points:] = h_values
        integrals = []
        for k in range(1, n_points):
            shifted_h = h_array[k:n_points + k]
            integral = np.sum(h_values * (1 - shifted_h)) * dx
            integrals.append(integral)
        return np.max(integrals)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0, 1)
    
    # Linear constraint: sum(h) == n_points / 2
    cons = LinearConstraint(np.ones(n_points), n_points / 2, n_points / 2)
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance, options={'maxiter': max_iterations})
    
    # Extract the best solution
    best_h = result.x
    c5_bound = objective(best_h)
    
    # Format the output
    h_values = best_h
    n_points_used = n_points
    
    return h_values, c5_bound, n_points_used
