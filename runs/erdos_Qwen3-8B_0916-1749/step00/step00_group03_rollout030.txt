import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Example initial construction (n=70)

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for balance between speed and resolution

    def objective_func(h, n_points, dx):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def evaluate(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def generate_initial_h(n_points):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scaling = (n_points / 2) / h_sum if h_sum != 0 else 1
        return h * scaling

    # Main optimization loop
    for n in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 10:
            break

        print(f"Trying n_points = {n}")

        h = generate_initial_h(n)
        current_c5 = evaluate(h, n)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Local optimization with gradient descent
        for _ in range(50):
            dx = 2.0 / n
            h = best_h.copy()
            h += np.random.normal(0, 0.01, n)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            scaling = (n / 2) / h_sum
            h = h * scaling

            current_c5 = evaluate(h, n)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        # Try using the initial construction if it's better
        initial_h = initial_h_values
        if len(initial_h) == n:
            current_c5 = evaluate(initial_h, n)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h.copy()

    # For fun, try a GA approach with a small population
    population_size = 20
    max_gen = 30
    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 10:
            break

        population = [np.random.uniform(0, 1, n) for _ in range(population_size)]
        for i in range(population_size):
            h = population[i]
            h_sum = np.sum(h)
            scaling = (n / 2) / h_sum
            population[i] = h * scaling

        fitness = [evaluate(h, n) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]

        # Crossover and mutation
        for i in range(population_size):
            j = np.random.randint(0, population_size)
            crossover_point = np.random.randint(0, n)
            h1 = population[i].copy()
            h2 = population[j].copy()
            h = h1[:crossover_point] + h2[crossover_point:]
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            scaling = (n / 2) / h_sum
            h = h * scaling
            population[i] = h

        for i in range(population_size):
            mutation_rate = 0.1
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n)
                population[i][idx] = np.random.uniform(0, 1)

    # Final result
    best_c5 = evaluate(best_h, n)
    return best_h, best_c5, n
