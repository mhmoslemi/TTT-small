import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = initial_h_values + np.random.normal(0, 0.1, n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values[:n_points]
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.arange(len(corr))
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    initial = initial_h_values
    result = opt.minimize(
        objective,
        initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000},
        callback=lambda x: None
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = best_h[:n_points]
    
    c5 = objective(best_h)
    return best_h, c5, n_points
