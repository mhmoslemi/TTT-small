```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import time

# Pre-imported functions
initial_h_values = np.load("initial_h_values.npy")  # Load initial construction
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster evaluations
    h = initial_h_values[:n_points]  # Use initial construction
    h = h / np.sum(h) * (n_points / 2)  # Rescale to meet integral constraint
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / len(h)))
    
    # Function to compute loss for optimization
    def loss_func(h_values):
        h = np.array(h_values, dtype=np.float64)
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        if np.sum(h) != n_points / 2:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / len(h)))
    
    # Optimization using gradient descent
    def optimize():
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(loss_func, x0=h, method='L-BFGS-B', bounds=bounds, tol=1e-8, maxiter=1000)
        return result.x
    
    start = time.time()
    best_h = h
    best_c5 = compute_c5(h)
    
    # Attempt optimization
    optimized_h = optimize()
    optimized_c5 = compute_c5(optimized_h)
    
    if optimized_c5 < best_c5:
        best_h = optimized_h
        best_c5 = optimized_c5
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 100:
        n_points = 200
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        h[n_points // 2:] = 0.0
        h = h / np.sum(h) * (n_points / 2)
        
        def loss_func_large(h_values):
            h = np.array(h_values, dtype=np.float64)
            if np.any(h < 0) or np.any(h > 1):
                return float('inf')
            if np.sum(h) != n_points / 2:
                return float('inf')
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / len(h)))
        
        # Optimization with more points
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(loss_func_large, x0=h, method='L-BFGS-B', bounds=bounds, tol=1e-8, maxiter=500)
        optimized_h_large = result.x
        optimized_c5_large = compute_c5(optimized_h_large)
        
        if optimized_c5_large < best_c5:
            best_h = optimized_h_large
            best_c5 = optimized_c5_large
    
    return best_h, best_c5, len(best_h)
```
