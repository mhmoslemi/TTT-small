import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    # Initial guess based on a simple repeating pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 if i % 3 == 0 else 0.25
    h = h * (n_points / 2) / np.sum(h)
    
    def objective_func(h_values):
        h = h_values.reshape(n_points)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        shifted_h = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return overlap
    
    def constraint_func(h_values):
        h = h_values.reshape(n_points)
        # Ensure integral of h is 1
        return np.sum(h) * dx - 1
    
    # Define bounds for h
    bounds = Bounds(0, 1)
    
    # Use L-BFGS-B optimizer for faster convergence
    result = optimize.minimize(
        fun=objective_func,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func}
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = result.fun
    
    return best_h, c5_bound, n_points
