```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.integrate import simps
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points):
    # Initial construction with n_points samples
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_initial = len(initial_h_values(70))
    h_initial = initial_h_values(n_initial)
    c5_initial = evaluate_erdos_solution(h_initial)
    best_c5 = c5_initial
    best_h = h_initial
    best_n = n_initial
    
    # Try different n_points starting from lower values
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate initial guess
        h = np.random.rand(n_points)
        h = np.minimum(np.maximum(h, 0.0), 1.0)
        total = np.sum(h)
        h = h * (n_points / 2) / total

        # Define the objective function and constraints
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * (2.0 / n_points)

        # Optimization setup
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            h_result = result.x
            c5 = evaluate_erdos_solution(h_result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_result
                best_n = n_points

    # Final check with the initial construction
    if evaluate_erdos_solution(h_initial) < best_c5:
        best_c5 = evaluate_erdos_solution(h_initial)
        best_h = h_initial
        best_n = n_initial

    return best_h, best_c5, best_n
```
