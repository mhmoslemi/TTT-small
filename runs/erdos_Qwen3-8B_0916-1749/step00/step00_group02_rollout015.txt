```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.sparse import diags
from scipy.linalg import norm
from numpy import linalg as la
from numpy.random import choice, seed
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=300, **kwargs):
    # Set seed for reproducibility
    seed(seed)
    
    # Initial guess: use the initial_h_values provided
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Ensure the sum of h_values is correct
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h
    
    # Objective function to minimize: the maximum overlap C5
    def objective(h):
        # Convert to numpy array
        h = np.array(h, dtype=np.float64)
        # Normalization
        dx = 2.0 / len(h)
        # Compute all shifts (k from -1 to 1)
        shifts = np.unique(np.arange(-1, 2))
        overlaps = []
        for k in shifts:
            if k == 0:
                continue
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 500,
        'ftol': 1e-8,
        'gtol': 1e-8
    }
    
    # Perform optimization
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='L-BFGS-B',
        options=options
    )
    
    # Check if optimization was successful
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        # Fallback: use initial guess if optimization fails
        h_values = initial_h
        c5_bound = objective(h_values)
    
    # Ensure the solution is within bounds and sum is correct
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        h_values = h_values * (n_points / 2) / sum_h
    
    # Compute C5 using the precomputed function
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Return the result
    return h_values, c5_bound, n_points
```
