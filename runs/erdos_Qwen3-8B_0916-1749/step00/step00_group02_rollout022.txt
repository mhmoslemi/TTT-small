import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import time
import logging
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 1000  # Initial number of points for the step function
    max_iter = 100  # Maximum number of iterations
    tol = 1e-5  # Tolerance for convergence
    
    # Sample initial function
    h = initial_h_values.copy()
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Objective function to minimize the maximum overlap integral
    def objective(h):
        # Reshape to array
        h_vals = h.reshape(-1)
        
        # Compute all shifted versions of 1 - h
        shifted = []
        for k in range(1, n_points // 2):
            shifted_k = np.roll(h_vals, k)
            shifted.append(1 - shifted_k)
        
        # Compute the overlap integrals
        overlap = []
        for shift in shifted:
            corr = np.correlate(h_vals, shift, mode='full')
            overlap.append(np.max(corr))
        
        # Return the maximum overlap
        return np.max(overlap)
    
    # Optimization
    def optimize_h(h_init, max_iter, tol):
        best_h = h_init.copy()
        best_c5 = objective(best_h)
        
        for _ in range(max_iter):
            # Initialize perturbation
            perturbation = np.random.rand(n_points) * 0.05
            perturbation = np.clip(perturbation, -0.05, 0.05)
            
            # Apply perturbation
            h_candidate = np.clip(h_init + perturbation, 0, 1)
            
            # Compute the overlap
            c5 = objective(h_candidate)
            
            # Accept if it improves
            if c5 < best_c5:
                best_h = h_candidate
                best_c5 = c5
            
            # Log progress
            if _ % 10 == 0:
                print(f"Round {_} / {max_iter}: Best C5 = {best_c5}")
        
        return best_h, best_c5
    
    # Optimize the initial guess
    best_h, best_c5 = optimize_h(h, max_iter, tol)
    
    # Convert to a reduced representation for the final answer
    reduced_h = np.round(best_h, 5)
    reduced_h = np.clip(reduced_h, 0, 1)
    reduced_h = np.round(reduced_h, 5)
    
    # Return the result as specified
    return reduced_h, best_c5, n_points
