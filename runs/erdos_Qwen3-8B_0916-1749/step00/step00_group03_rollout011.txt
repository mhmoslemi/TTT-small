import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction
n_points_initial = len(initial_h_values)

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial guess with a small number of points
    n_points = 200
    h = np.random.rand(n_points) * 2 - 1  # Random initial guess scaled to [0, 1]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint
    
    # Define the overlap integral function
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Evaluate the best solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    
    return best_h, c5_bound, n_points
