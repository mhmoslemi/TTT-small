def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initialize with the given starting point
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    total_mass = sum(h) * dx
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Objective function to minimize the maximum overlap
    def objective(x):
        return compute_c5(x)
    
    # Constraints: sum(h) * dx = 1 --> sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-6
    time_limit = budget_s
    start_time = time.time()
    
    # Optimization loop
    best_h = x0
    best_c5 = compute_c5(x0)
    
    for _ in range(max_iterations):
        if time.time() - start_time > time_limit:
            break
        
        # Use L-BFGS-B for local optimization with bounds
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', tol=tolerance)
        x0 = result.x
        current_c5 = compute_c5(x0)
        
        if current_c5 < best_c5:
            best_h = x0
            best_c5 = current_c5
        
        # Early stopping if improvement is minimal
        if abs(current_c5 - best_c5) < tolerance:
            break
    
    return best_h, best_c5, n_points
