```python
import numpy as np
import scipy.optimize as optimize
import time
import os
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial guess
    n_points = 128
    h_initial = initial_h_values[:n_points]
    h = np.array(h_initial, dtype=np.float64)
    h = np.repeat(h, 2)  # Double the number of points to increase resolution
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h, dtype=np.float64)
        h = np.pad(h, (0, len(h) - len(h_initial)), 'constant')
        h = h[:len(h_initial)]
        
        # Compute all shifted versions of 1 - h
        shifts = np.arange(0, 2, 0.01)
        all_1_minus_h = []
        for shift in shifts:
            shifted_h = np.roll(h, int(shift / (2.0 / len(h))))
            all_1_minus_h.append(1.0 - shifted_h)
        
        # Compute overlaps
        C5_values = []
        for idx in range(len(all_1_minus_h)):
            overlap = np.sum(h * all_1_minus_h[idx])
            C5_values.append(overlap)
        
        return max(C5_values)
    
    # Optimization function
    def objective(x):
        x = np.array(x, dtype=np.float64)
        x = np.clip(x, 0.0, 1.0)
        x = np.repeat(x, 2)  # Double the number of points to increase resolution
        return compute_C5(x)
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(len(h_initial))]
    
    # Optimizer settings
    options = {
        'disp': False,
        'maxiter': 100,
        'ftol': 1e-6,
        'eps': 1e-6
    }
    
    # Run optimization
    result = optimize.minimize(
        objective,
        x0=h_initial,
        bounds=bounds,
        method='L-BFGS-B',
        options=options
    )
    
    # Get the best solution
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # Final check to ensure we stay within time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 5:
        return (best_h, best_c5, len(best_h))
    
    # Return the best solution
    return (best_h, best_c5, len(best_h))
```
