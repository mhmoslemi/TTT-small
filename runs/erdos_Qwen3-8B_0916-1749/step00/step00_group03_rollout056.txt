import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

def run(seed=42, budget_s=300):
    # Initial setup
    n_points = 1000
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    
    # Integrate the initial guess to ensure it satisfies the constraint
    initial_h_values = np.array(initial_h_values)
    sum_h = np.sum(initial_h_values)
    normalization_factor = n_points / 2.0
    initial_h_values = initial_h_values / (sum_h / normalization_factor)
    
    # Convert initial_h_values to a numpy array
    h = initial_h_values
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iterations = 100
    tol = 1e-4
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Define the function to compute C5
    def compute_C5(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        n = len(h)
        C5_values = []
        for k in range(1, n // 2 + 1):
            overlap = np.correlate(h, 1 - h, mode='full')
            C5 = np.max(overlap * dx)
            C5_values.append(C5)
        return np.max(C5_values)
    
    # Optimization strategy
    def optimize_h():
        nonlocal best_C5, best_h
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        
        # Define variables
        x = cp.Variable(n_points, nonneg=True)
        
        # Constraints
        constraints = [
            cp.sum(x) == n_points / 2.0,
            cp.sum(x) == n_points / 2.0,
            x >= 0,
            x <= 1,
        ]
        
        # Objective function: minimize the maximum C5 over all k
        # We use a robust approach to approximate the maximum C5
        # over k by trying a few k values and optimizing for each
        # to avoid a non-convex problem
        
        # Try a few k values
        k_values = np.linspace(1, n_points // 2, 5, endpoint=False).astype(int)
        obj = cp.Minimize(0)
        for k in k_values:
            shift = np.roll(x.value, k)
            overlap = np.multiply(x.value, 1 - shift)
            obj = cp.Minimize(cp.max(overlap))
        
        # Solve the optimization problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='ECOS', abstol=1e-8, reltol=1e-8, maxiter=100, verbose=False)
        except:
            pass
        
        if prob.status == 'optimal':
            h = x.value
            h = np.clip(h, 0, 1)
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
    
    # Main optimization loop
    start_time = time.time()
    for _ in range(max_iterations):
        optimize_h()
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_C5, n_points
