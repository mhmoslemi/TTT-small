```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, NonlinearConstraint

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 0.5 + 0.5

    def compute_overlap(h):
        h = h.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)

    def objective(h):
        return compute_overlap(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = NonlinearConstraint(constraint_sum, 0, 0)

    h0 = initial_h_values
    result = optimize.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})

    best_h = result.x
    best_c5 = compute_overlap(best_h)

    return best_h, best_c5, n_points
```
