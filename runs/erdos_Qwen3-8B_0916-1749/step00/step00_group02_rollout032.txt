```python
import numpy as np
import scipy.optimize as opt
import time
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    def evaluate_erdos_solution(h_values, dx):
        n_points = len(h_values)
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        overlap = overlap * dx
        return np.max(overlap)
    
    def cost_function(x, dx):
        n_points = len(x)
        if n_points < 2:
            return float('inf')
        dx_val = dx
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx_val
        return np.max(overlap)
    
    def objective(x):
        return cost_function(x, 2.0 / len(x))
    
    def constraint_sum(x):
        return np.sum(x) - len(x) / 2
    
    def initial_h_values(n_points):
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        return h
    
    def construct_initial_solution(n_points):
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        return h
    
    def optimize_with_bounds(h_initial, n_points, max_iter=100, tol=1e-6):
        x0 = h_initial
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=0, ub=len(x0)/2)
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=tol, options={'maxiter': max_iter})
        return result.x
    
    def evaluate_solution(h, n_points):
        dx = 2.0 / n_points
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        return np.max(overlap)
    
    def find_best_solution(n_points):
        initial_h = construct_initial_solution(n_points)
        optimized_h = optimize_with_bounds(initial_h, n_points)
        best_c5 = evaluate_solution(optimized_h, n_points)
        return optimized_h, best_c5
    
    def try_n_points(n_points):
        if n_points < 2:
            return float('inf'), None
        try:
            h, c5 = find_best_solution(n_points)
            return c5, h
        except:
            return float('inf'), None
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Try with n_points from 100 to 200
    for n in range(100, 200, 10):
        if n < 2:
            continue
        start_time = time.time()
        c5, h = try_n_points(n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
    
    return best_h, best_c5, best_n
```
