import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
from scipy.optimize import Bounds, minimize
from typing import Tuple, List, Dict, Any
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=300, **kwargs) -> Tuple[np.ndarray, float, int]:
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values[:n_points_initial]
    h_initial = h_initial[:n_points] if n_points < n_points_initial else h_initial
    h_initial = np.tile(h_initial, n_points // n_points_initial + 1)[:n_points]
    h_initial = np.random.permutation(h_initial)
    h_initial = h_initial[:n_points]
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    def objective_func(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h[:n_points]
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2 / n_points)
        return c5
    
    def constraint_func(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h[:n_points]
        total = np.sum(h) * (2 / n_points)
        return total - 1
    
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = h_initial.copy()
    
    # Optimization
    result = minimize(objective_func, x0, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 1000})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h[:n_points]
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Final evaluation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_corr = np.correlate(final_h, 1 - final_h, mode="full")
    c5 = np.max(final_corr) * (2 / n_points)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time exceeded budget. Returning best found solution.")
        return (best_h, c5, n_points)
    
    return (final_h, c5, n_points)
