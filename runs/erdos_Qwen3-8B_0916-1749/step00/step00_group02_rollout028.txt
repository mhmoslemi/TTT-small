import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.integrate import simps

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h_initial[i] = 1.0
        else:
            h_initial[i] = 0.0
    h_initial = np.concatenate([h_initial, h_initial[:n_points // 2]])[:n_points]
    h_initial /= np.sum(h_initial) * dx
    h_initial = np.clip(h_initial, 0, 1)

    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points // 2]])[:n_points]
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap

    def optimize_h(h, max_iterations=100):
        for _ in range(max_iterations):
            h = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h[i] = np.random.rand()
            h = np.clip(h, 0, 1)
            h = h * (np.sum(h) * dx) / np.sum(h * dx)
            c5 = compute_c5(h)
            if c5 < compute_c5(h_initial):
                h_initial = h
        return h_initial

    start_time = time.time()
    h = optimize_h(h_initial)
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (h_initial, compute_c5(h_initial), n_points)

    for _ in range(10):
        h = np.random.rand(n_points)
        h = h * (np.sum(h) * dx) / np.sum(h * dx)
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < compute_c5(h_initial):
            h_initial = h

    return (h_initial, compute_c5(h_initial), n_points)
