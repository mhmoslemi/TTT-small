    def objective(h):
        # Reshape h to be a vector
        h_vec = h.copy()
        # Compute the overlap integral for all k in [-1, -0.5, 0, 0.5, 1]
        # We only need the maximum of these
        max_overlap = 0
        for k in [-1, -0.5, 0, 0.5, 1]:
            if k <= -dx or k >= 2:
                continue
            # Compute the integral
            overlap = 0
            for i in range(n_points):
                x = i * dx
                if x + k >= 2:
                    continue
                j = int(np.floor((x + k) * (n_points / 2)))
                if 0 <= j < n_points:
                    overlap += h_vec[i] * (1 - h_vec[j])
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    # Set bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1}

    # Optimization parameters
    maxiter = 500
    tol = 1e-3

    # Define initial guess
    initial_guess = initial_h

    # Use L-BFGS-B for optimization
    result = opt.minimize(

        initial_guess,
        bounds=bounds,
        constraints=constraint,
