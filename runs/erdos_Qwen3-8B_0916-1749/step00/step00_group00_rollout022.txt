```python
import numpy as np
import scipy.optimize as optimize
import scipy.integrate as integrate
import time
import random
from scipy.sparse import diags
from scipy.sparse.linalg import eigs
from scipy.optimize import minimize
import matplotlib.pyplot as plt
from functools import lru_cache

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial samples
    initial_h_values = np.load("initial_h_values.npy")
    n_points_initial = len(initial_h_values)
    
    # Try different starting points
    start_points = [initial_h_values]
    
    # We'll try different point numbers for efficiency
    n_points_list = [100, 200, 300]
    
    best_c5 = float('inf')
    best_h = None
    
    for n_points in n_points_list:
        if n_points < n_points_initial:
            continue
        
        # Create initial guess with uniform distribution
        h0 = np.random.rand(n_points) * 2 - 1
        h0 = np.clip(h0, 0, 1)
        h0 = (h0 * 2).astype(float)
        
        # Define the objective function
        def objective(h):
            nonlocal n_points
            h = h.astype(float)
            h = np.clip(h, 0, 1)
            # Compute the overlap integral for all shifts
            max_overlap = 0.0
            for k in range(-n_points + 1, n_points):
                if k == 0:
                    continue
                # Shift the h function by k samples
                shift = np.roll(h, k)
                overlap = integrate.simps(h * (1 - shift), dx=2.0 / n_points)
                max_overlap = max(max_overlap, overlap)
            return max_overlap
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Optimization settings
        results = []
        for _ in range(5):
            try:
                # Initial guess
                guess = h0.copy()
                guess = guess + np.random.normal(0, 0.05, guess.shape)
                guess = np.clip(guess, 0, 1)
                
                # Solve the optimization problem
                result = minimize(
                    objective,
                    guess,
                    method='L-BFGS-B',
                    jac=None,
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    options={'maxiter': 100, 'disp': False}
                )
                
                if result.success:
                    current_c5 = result.fun
                    results.append((current_c5, result.x))
            except:
                pass
        
        # Select the best result
        if results:
            best_result = min(results, key=lambda x: x[0])
            best_c5 = best_result[0]
            best_h = best_result[1]
            print(f"Found C5 = {best_c5} with {n_points} points")
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
```
