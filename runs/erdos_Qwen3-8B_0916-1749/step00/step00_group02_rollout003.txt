def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 256  # Tune this based on what fits in budget
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-5

    # Initial guess (we use the initial_h_values if available, otherwise a random one)
    if initial_h_values is not None:
        initial_h = initial_h_values[:n_points]
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
    
    # Ensure the integral constraint is satisfied
    integral = np.sum(initial_h) * dx
    correction = (1 - integral) / dx
    initial_h = np.clip(initial_h + correction, 0, 1)

    # Define the cost function to minimize: max overlap
    def cost_func(h_values):
        h = h_values
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)

    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) * dx - 1.0

    # Optimization with constraints
    def optimize_h(n_points, initial_guess):
        h = initial_guess.copy()
        for _ in range(max_iterations):
            # Evaluate current cost
            current_cost = cost_func(h)
            if current_cost < tolerance:
                break

            # Update h by minimizing the cost while satisfying the integral constraint
            # We use a simple gradient descent approach here as a heuristic
            h_shifted = np.roll(h, 1)
            penalty = 1000.0 * (np.sum(h) * dx - 1.0)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                for k in range(1, n_points):
                    overlap = np.correlate(h, 1 - h, mode="full")[k]
                    gradient[i] += (overlap * dx) * (2 * np.roll(h, k) - 1) * dx
            h -= 0.01 * gradient
            h = np.clip(h, 0, 1)
            h = np.clip(h - (np.sum(h) * dx - 1.0) / dx, 0, 1)

        return h

    # Optimize
    best_h = optimize_h(n_points, initial_h)
    best_c5 = evaluate_erdos_solution(best_h)

    # Ensure the best solution is found within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded. Returning best found so far.")
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
