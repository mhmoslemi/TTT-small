```python
import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.optimize.slsqp import minimize_SLSQP
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    # Start with initial_h_values from the global variable
    global initial_h_values
    if initial_h_values is not None:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # If no initial_h_values provided, use a grid-based approach
        initial_h = np.linspace(0, 1, n_points)
    
    # Rescale to satisfy the integral constraint
    total = np.sum(initial_h) * dx
    initial_h = initial_h * (1 / total) * 2

    # Define the objective function to minimize the max overlap
    def objective(x):
        h = x.reshape(n_points)
        # Compute the overlap integral for all k
        overlap = []
        for k in range(1, n_points):
            if k > n_points - k:
                continue
            shifted = np.roll(h, k)
            integral = np.sum(h * (1 - shifted)) * dx
            overlap.append(integral)
        return np.max(overlap)
    
    # Define bounds for the variables
    bounds = Bounds(0, 1)
    
    # Start the optimization
    start_time = time.time()
    result = minimize_SLSQP(
        objective,
        initial_h,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
        tol=1e-8
    )
    
    # Extract the best solution found
    h_values = result.x
    c5_bound = result.fun
    elapsed = time.time() - start_time
    
    # Ensure we don't exceed the budget
    if elapsed > budget_s:
        print("Optimization exceeded the budget. Returning the best solution so far.")
    
    return h_values, c5_bound, n_points
```
