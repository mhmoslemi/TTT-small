def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    h = np.copy(initial_h_values[:n_points])
    if len(h) < n_points:
        h = np.concatenate([h, np.zeros(n_points - len(h))])
    
    # Function to evaluate C5
    def objective(x):
        nonlocal h
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * 2.0 / len(h)
        return c5
    
    # Constraints
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = np.copy(h)
    
    # Setup optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
    
    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we can improve with more points
    if best_c5 > 0.38092:
        n_points = 200
        h = np.zeros(n_points)
        # Random initial distribution
        h = np.random.uniform(0, 1, n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))  # Normalize to [0,1]
        h = h[:n_points]
        if np.sum(h) < n_points / 2:
            h = np.random.uniform(0, 1, n_points)
            h = (h - np.min(h)) / (np.max(h) - np.min(h))
            h = h[:n_points]
        
        # New objective
        def objective2(x):
            nonlocal h
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")

        # New constraint
        def constraint2(x):

        # New bounds
        bounds2 = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess
        x0 = np.copy(h)
        
        # Second optimization
        result2 = minimize(objective2, x0, method='L-BFGS-B', bounds=bounds2, constraints=[{'type': 'eq', 'fun': constraint2}])
        best_h = result2.x
        best_c5 = objective2(best_h)
