def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    best_c5 = np.inf
    best_h = h.copy()
    
    # Define the overlap integral for a given h
    def compute_c5(h):
        h = h.reshape(-1)
        n = len(h)
        dx = 2.0 / n
        # Compute all shifted versions of h
        overlaps = []
        for k in range(1, n):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    # Define the objective function to minimize
    def objective(x):
        nonlocal best_c5, best_h
        x = x.reshape(-1)
        dx = 2.0 / len(x)
        # Ensure the solution is within [0, 1]
        x = np.clip(x, 0, 1)
        # Compute the overlap integral
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5
    
    # Run optimization
    t0 = time.time()
    while time.time() - t0 < budget_s:
        # Start from the current best solution
        x0 = best_h.copy()
        # Define the optimization problem
        result = optimize.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(x0))],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        # Update the best solution
        if result.success:
            best_h = result.x
        else:
            break
    
    return best_h, best_c5, len(best_h)
