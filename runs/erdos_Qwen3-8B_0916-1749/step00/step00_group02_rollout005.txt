import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate([h[-2:], h])
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5] * 45

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from the known construction
    n_points = 45
    h = initial_h_values()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Define the cost function to minimize
    def cost(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        shifted = np.concatenate([h[-2:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Use simulated annealing for optimization
    def simulated_annealing():
        nonlocal best_c5, best_h
        current_h = copy.copy(h)
        current_c5 = cost(current_h)
        best_c5 = current_c5
        best_h = current_h
        
        temp = 1.0
        cooling_rate = 0.995
        iterations = 10000
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s:
                return best_h, best_c5
            
            # Perturb the current solution
            perturbation = np.random.normal(0, 0.1, len(current_h))
            new_h = np.clip(current_h + perturbation, 0.0, 1.0)
            new_c5 = cost(new_h)
            
            # Acceptance criterion
            if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            
            temp *= cooling_rate
    
    simulated_annealing()
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
