```python
import numpy as np
import scipy.optimize as optimize
import scipy.linalg as la
import time
import sys
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 1000  # Start with 1000 points, can be reduced later if needed
    dx = 2.0 / n_points

    # Helper: Compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        result = np.max(corr) * dx
        return result

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    initial_h = np.array(initial_h)
    initial_h = initial_h[:n_points]  # Trim to n_points

    # Optimization: use a grid search with some randomness
    # Start with a grid search over a reduced number of samples
    # We will use a greedy approach to refine the solution

    # First, try with the initial guess
    initial_c5 = compute_c5(initial_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h.copy()

    # Now reduce the number of points and perform optimization
    for n_points in [500, 300, 200, 150, 100, 80, 60, 50, 40, 30, 20]:
        # Create a new h with n_points samples
        h = np.zeros(n_points)
        # Randomly select positions where h[i] = 1 or 0
        # We will use a greedy approach to place high values
        for i in range(n_points):
            # Randomly decide to place a high or low value
            h[i] = np.random.choice([0, 1], p=[0.5, 0.5])

        # Ensure the integral constraint is met
        total = np.sum(h) * dx
        needed = 1.0 - total
        # Adjust the last value to meet the constraint
        if needed > 0:
            h[-1] += needed / dx
            h[-1] = np.clip(h[-1], 0, 1)
        elif needed < 0:
            h[-1] -= (-needed) / dx
            h[-1] = np.clip(h[-1], 0, 1)

        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Print progress
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

    # If best_h is not found, use the initial guess
    if best_h is None:
        best_h = initial_h.copy()

    # Compute final C5
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), final_c5, len(best_h)
```
