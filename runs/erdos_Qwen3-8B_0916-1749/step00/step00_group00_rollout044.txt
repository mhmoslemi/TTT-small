import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h_values = np.array([0.5]*45)
    
    # Best solution found
    best_c5 = float('inf')
    best_h = initial_h_values
    
    # Function to compute overlap integral
    def compute_overlap(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_shifted = np.roll(h, shift=1)
        h_shifted[:n_points-1] = 0
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        overlap = overlap * dx
        return np.max(overlap)
    
    # Optimization parameters
    n_points = 100
    max_iter = 100
    step_size = 0.1
    
    # Function to generate a random perturbation
    def generate_perturbation(h, max_change=0.2):
        n = len(h)
        pert = np.random.uniform(-max_change, max_change, n)
        h_new = np.clip(h + pert, 0, 1)
        # Normalize to ensure ∫h dx = 1
        total = np.sum(h_new) * (2.0 / n)
        if total == 0:
            return h_new
        scaling = 1.0 / total
        return h_new * scaling
    
    # Main optimization loop
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a new perturbation
        h_candidate = generate_perturbation(best_h)
        
        # Compute overlap for candidate
        c5_candidate = compute_overlap(h_candidate)
        
        # Update best solution
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            print(f"Iteration {iteration+1}: C5 = {best_c5:.6f}")
    
    # Return best solution
    return best_h, best_c5, len(best_h)
