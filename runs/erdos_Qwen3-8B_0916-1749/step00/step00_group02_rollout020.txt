import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction with n_points = 45
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5])
    
    # Function to compute the overlap integral
    def compute_C5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h = h.astype(float)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)
    
    # Function to optimize h with constraints
    def optimize_h(h):
        n = len(h)
        dx = 2.0 / n
        idx = np.arange(n)
        
        # Objective: minimize the maximum overlap
        def objective(x):
            h = x
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap * dx)
        
        # Constraints:
        # 1. Sum of h must be equal to n / 2
        def constraint_sum(x):
            return np.sum(x) - n / 2
        
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n)]
        
        # Optimization
        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        
        return result.x
    
    # Start with the initial h_values
    h = initial_h_values.copy()
    best_c5 = compute_C5(h)
    best_h = h
    
    # Try different n_points values to find the best solution
    n_values = [45, 60, 75, 90, 105, 120, 135, 150]
    
    for n in n_values:
        # Resample initial_h_values to n points
        h_resampled = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        
        # Optimize the resampled h
        start_time = time.time()
        h_opt = optimize_h(h_resampled)
        elapsed_time = time.time() - start_time
        
        # Compute the C5 for the optimized h
        c5 = compute_C5(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n
    
    return best_h, best_c5, len(best_h)
