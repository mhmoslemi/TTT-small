import numpy as np
import scipy.optimize as opt
import time
import copy

initial_h_values = np.loadtxt('initial_h_values.txt')

def run(seed=42, budget_s=300, **kwargs):
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-3
    seed = seed or np.random.SeedSequence(42).generate_state(1)[0]
    np.random.seed(seed)
    
    # Initial guess from given samples
    h = np.array(initial_h_values[:n_points])
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate C5
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > best_c5:
                best_c5 = overlap
                best_h = h.copy()
        
        # Optimization step: adjust h to reduce maximum overlap
        def objective(x):
            h = np.clip(x, 0, 1)
            shifted_h = np.zeros_like(h)
            for k in range(1, n_points):
                shifted_h = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted_h)) * dx
            return np.max(overlap)
        
        # Use a simple gradient descent with momentum
        h_new = h.copy()
        for _ in range(50):
            shifted_h = np.zeros_like(h)
            for k in range(1, n_points):
                shifted_h = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted_h)) * dx
            grad = np.zeros_like(h)
            for k in range(1, n_points):
                shifted_h = np.roll(h, k)
                grad += np.sum((1 - shifted_h) * dx * np.gradient(h))
            h_new -= 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h = h_new
        
        h = h_new
        print(f"Iteration {iteration + 1}, Current C5: {best_c5:.6f}")
    
    return best_h, best_c5, n_points
