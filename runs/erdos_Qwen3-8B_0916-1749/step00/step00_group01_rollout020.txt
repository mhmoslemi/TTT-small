```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    h = initial_h_values[:n_points]  # Use initial construction as starting point

    # Ensure the sum of h is exactly n_points / 2
    sum_h = np.sum(h)
    scaling = (n_points / 2) / sum_h
    h = h * scaling
    h = np.clip(h, 0, 1)

    # Function to compute C5
    def compute_C5(h):
        # Compute the overlap integrals
        kernel = 1 - h
        result = signal.correlate(h, kernel, mode="full")
        result = result[len(result) // 2 : ]  # Keep the symmetric part
        result = result * dx
        return np.max(result)

    # Objective function to minimize C5
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h
        h = h * scaling
        return compute_C5(h)

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization settings
    max_iterations = 100
    tolerance = 1e-5

    # Use BFGS to minimize C5
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iterations, 'ftol': tolerance}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scaling = (n_points / 2) / sum_h
    best_h = best_h * scaling
    best_C5 = compute_C5(best_h)

    return best_h, best_C5, n_points
```
