def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=300, **kwargs):
    n_points = 100  # Start with 100 points for efficient optimization
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points]  # Take first n_points samples
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)  # Rescale to meet the constraint

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - (n_points / 2)

    # Bounds
    bounds = Bounds(0, 1)

    # Initial guess
    x0 = initial_h.copy()

    # Optimization
    result = minimize(objective, x0, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}],
                     method='SLSQP', options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})

    if result.success:
        best_h = result.x
    else:
        best_h = initial_h

    # Evaluate and return
    c5_bound = evaluate_erdos_solution(best_h)
    return best_h, c5_bound, n_points
