import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction from initial_h_values (n=47)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points

    # Transform h to be in [0, 1]
    h = np.clip(h, 0, 1)

    # Check the integral constraint
    if np.abs(np.sum(h) * dx - 1.0) > 1e-8:
        raise ValueError("Initial h does not satisfy the integral constraint")

    # Function to compute the overlap integral
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        convolution = np.fft.ifft(h_fft * one_minus_h_fft).real
        # Take the full convolution and extract the relevant part
        full_conv = convolution[:n]
        # Compute the maximum integral over all k
        max_overlap = np.max(full_conv * dx)
        return max_overlap

    # Objective function: minimize the maximum overlap
    def objective(h):
        return compute_c5(h)

    # Bounds for h values
    bounds = Bounds(0, 1)

    # Optimization settings
    max_iter = 100
    tol = 1e-5

    # Initial optimization with L-BFGS-B
    result = optimize.minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'maxiter': max_iter, 'ftol': tol}
    )

    # Evaluate the result
    best_h = result['x']
    best_c5 = compute_c5(best_h)

    # Check if we can refine the solution using a small perturbation
    for _ in range(5):
        perturbation = np.random.normal(0, 0.01, size=len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    # Final result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Warning: Time budget exceeded ({budget_s} seconds), returning best found so far.")
    return best_h, best_c5, len(best_h)
