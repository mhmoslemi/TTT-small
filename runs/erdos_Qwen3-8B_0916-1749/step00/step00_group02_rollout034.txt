```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time
from collections import deque

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initial guess: constant
    initial_h = np.full(n_points, 0.5)
    
    # Objective function: maximize the overlap integral for each shift
    def objective(h):
        H = h.reshape(-1, 1)
        H_shifted = np.roll(H, shift=range(n_points), axis=0)
        overlap = np.sum(H * (1 - H_shifted), axis=0)
        return np.max(overlap) * dx

    # Constraints: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Optimization using Sequential Least Squares Programming (SLSQP)
    def optimize_h():
        result = optimize.minimize(
            lambda h: objective(h),
            x0=initial_h,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'fun': constraint, 'type': 'eq'}],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x

    # Try multiple initial guesses
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    for _ in range(5):
        h = optimize_h()
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
```
