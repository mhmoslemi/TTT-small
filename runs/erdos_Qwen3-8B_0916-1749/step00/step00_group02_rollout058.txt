```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import linear_sum_assignment

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points for balance between speed and accuracy
    
    # Initial guess: evenly spaced samples with a piecewise constant function
    def initial_h_values(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h

    h_initial = initial_h_values(n_points)
    
    # Evaluate the current C5
    def evaluate_C5(h):
        dx = 2.0 / n_points
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap
    
    c5_initial = evaluate_C5(h_initial)
    print(f"Initial C5: {c5_initial}")
    
    # Objective function for optimization
    def objective(h):
        return evaluate_C5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Bounds
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = evaluate_C5(best_h)
    
    print(f"Optimized C5: {best_c5}")
    
    # If the current n_points is too low, increase n_points and repeat
    if n_points < 200:
        n_points = 200
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h_initial[i] = 1.0
            else:
                h_initial[i] = 0.0
        
        def evaluate_C5_higher(h):
            dx = 2.0 / n_points
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = np.max(corr * dx)
            return overlap
        
        def objective_higher(h):
            return evaluate_C5_higher(h)
        
        def constraint_sum_higher(h):
            return np.sum(h) - n_points / 2
        
        cons_higher = ({'type': 'eq', 'fun': constraint_sum_higher})
        
        bounds_higher = Bounds([0.0]*n_points, [1.0]*n_points)
        
        result_higher = minimize(objective_higher, h_initial, method='SLSQP', bounds=bounds_higher, constraints=cons_higher, tol=1e-10, options={'maxiter': 100})
        
        best_h_higher = result_higher.x
        best_c5_higher = evaluate_C5_higher(best_h_higher)
        
        best_h = best_h_higher
        best_c5 = best_c5_higher
        
        print(f"Optimized C5 with higher points: {best_c5}")
    
    return best_h, best_c5, n_points
```
