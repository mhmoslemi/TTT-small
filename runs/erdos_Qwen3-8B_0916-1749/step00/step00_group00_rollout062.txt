import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-5

    # Initial guess: evenly spaced with some variation
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        h_initial[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraints(h):
        return [constraint_sum(h)]

    # Use a simple gradient descent with adaptive step size
    h = h_initial.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_iter = 0
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad = np.gradient(h, dx)
        grad = grad / np.max(np.abs(grad))
        step = 0.1
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_iter = _
        h = h_new

    print(f"Best C5: {best_c5}")
    return best_h, best_c5, n_points
