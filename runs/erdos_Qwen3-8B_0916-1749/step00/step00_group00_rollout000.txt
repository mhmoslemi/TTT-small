```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    result = np.correlate(h_values, 1 - h_values, mode="full")
    c5 = np.max(result) * dx
    return c5

def initial_h_values(n_points):
    # Custom initial construction for n=45 samples
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial n_points
    n_points = 300  # A reasonable compromise between speed and resolution
    dx = 2.0 / n_points
    h = initial_h_values(n_points)
    
    # Define the objective function to minimize the max overlap
    def objective(x):
        h_values = x
        result = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(result) * dx
        return c5
    
    # Define constraints: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Optimization bounds: h in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use SLSQP with good initial guess
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )
    
    # Check if optimization was successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h
        best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Ensure the solution satisfies the integral constraint
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        best_h = initial_h_values(n_points)
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    time_elapsed = time.time() - start_time
    if time_elapsed > budget_s - 1:
        print("Time exceeded, returning initial solution")
        return (initial_h_values(n_points), evaluate_erdos_solution(initial_h_values(n_points), n_points), n_points)
    
    return (best_h, best_c5, n_points)
```
