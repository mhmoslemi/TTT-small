import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

# Initial construction is provided as initial_h_values
initial_h_values = np.load('initial_h_values.npy')  # Assuming this file is available

def run(seed=42, budget_s=300, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial number of points -- start with the same as the initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Convert initial_h_values to a numpy array if not already
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate objective and constraints
    def objective(x):
        return compute_c5(x)
    
    # Bounds for h: each element must be between 0 and 1
    bounds = Bounds(0, 1)
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_func(x):
        return np.sum(x) - n_points / 2
    
    constraint = LinearConstraint(np.ones(n_points), 0, 0, name='sum_constraint')
    
    # Initial guess
    initial_guess = h.copy()
    
    # Optimization settings
    max_iter = 1000
    tol = 1e-6
    
    # Start timing
    start_time = time.time()
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[constraint],
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'eps': 1e-8}
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we need to refine further (optional)
    # Could add more iterations or use different solvers
    
    # Format output
    return best_h, best_c5, len(best_h)
