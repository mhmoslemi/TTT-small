```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, fmin_l_bfgs_b
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 256
    max_attempts = 10
    best_c5 = float('inf')
    best_h = initial_h_values[:n_points]
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum != n_points / 2:
        best_h = np.clip(best_h, 0, 1)
        best_h = (best_h * n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    for _ in range(max_attempts):
        start_time = time.time()
        h = np.random.uniform(0, 1, n_points)
        h = (h * n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        def objective(x):
            nonlocal h
            h = x
            h = np.clip(h, 0, 1)
            h = (h * n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h)
        
        result = fmin_l_bfgs_b(
            objective,
            x0=h,
            bounds=Bounds(0, 1),
            maxiter=100,
            iprint=-1
        )
        
        if time.time() - start_time > budget_s:
            break
        
        if result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0]
    
    return best_h, best_c5, n_points
```
