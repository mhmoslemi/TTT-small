import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    h = np.random.rand(n_points) * 2 - 1  # Random starting point
    h = (h + 1) / 2  # Scale to [0, 1]
    h = h * (n_points / 2) / np.sum(h)  # Normalize to integral 1

    def evaluate_erdos_solution(h_values):
        n = len(h_values)
        dx = 2.0 / n
        shifts = np.arange(-n + 1, n)
        overlaps = []
        for k in shifts:
            if k < 0:
                overlap = np.sum(h_values[:n + k] * (1 - h_values[k:]))
            elif k > 0:
                overlap = np.sum(h_values[:n - k] * (1 - h_values[k:]))
            else:
                overlap = np.sum(h_values * (1 - h_values))
            overlaps.append(overlap * dx)
        return max(overlaps)

    def objective(h_values):
        return evaluate_erdos_solution(h_values)

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Optimizer configuration
    bounds = [(0, 1) for _ in range(n_points)]
    h_opt = np.copy(h)
    best_c5 = objective(h_opt)

    # First iteration: local search around initial guess
    for _ in range(10):
        step = 0.01
        h_try = h_opt + np.random.randn(n_points) * step
        h_try = np.clip(h_try, 0, 1)
        h_try = h_try * (n_points / 2) / np.sum(h_try)
        c5 = objective(h_try)
        if c5 < best_c5:
            best_c5 = c5
            h_opt = h_try

    # Second iteration: use CVXPY to find better solution
    h_var = cp.Variable(n_points)
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    objective_func = cp.Maximize(cp.sum(h_var * (1 - h_var)))
    prob = cp.Problem(objective_func, constraints)
    try:
        prob.solve()
        h_opt = h_var.value
        best_c5 = evaluate_erdos_solution(h_opt)
    except:
        pass  # Fallback to previous best if optimization fails

    # Final check with small perturbation to avoid local optima
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    best_c5 = evaluate_erdos_solution(h_opt)

    return h_opt, best_c5, n_points
