import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values with n=47 samples
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Current best solution
    best_c5 = np.inf
    best_h = h_initial
    best_n_points = n_points

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        h_shifted = np.zeros(2 * n_points)
        h_shifted[n_points//2 : 2*n_points//2] = h
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(overlap)
        return c5

    # Function to compute the loss for optimization
    def objective(h):
        nonlocal best_c5, best_h, best_n_points
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        if np.sum(h) * dx != 1:
            return np.inf
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
        return c5

    # Optimization loop
    start_time = time.time()
    elapsed = 0
    n_points_to_check = [100, 200, 300, 400, 500]  # Try different n_points

    for n in n_points_to_check:
        if elapsed > budget_s - 10:
            break
        # Generate initial samples
        h = np.random.uniform(0, 1, n)
        h = h / np.sum(h) * n / 2  # Normalize to satisfy the integral constraint

        # Run optimization
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n)],
            options={'max_iter': 100, 'disp': False}
        )

        # Update best solution
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
            best_n_points = n

        elapsed = time.time() - start_time

    return (best_h, best_c5, best_n_points)
