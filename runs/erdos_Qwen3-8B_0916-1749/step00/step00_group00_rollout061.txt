import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # We'll use 100 points for better resolution
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial guess
    h = h * 2.0 / np.sum(h)  # Normalize to ensure ∫h = 1

    # For optimization, we'll use a simpler approach to minimize the maximum overlap
    # Define the function to compute C5
    def compute_C5(h):
        h = h.reshape(-1)
        h_shifted = np.concatenate([h[-1:], h[:-1]])  # Shift h(x) by 1
        overlap = np.sum(h * (1 - h_shifted)) * dx
        return overlap

    # Define the objective function to minimize the maximum overlap
    def objective(h):
        return np.max([compute_C5(h), compute_C5(h[1:] + [h[0]]), compute_C5(h[2:] + h[:2])])

    # Setup bounds and constraints
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), 1.0 / 2, 1.0 / 2)  # sum(h) == n_points / 2

    # Run optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    elapsed = time.time() - start_time

    # Return the best solution found
    best_h = result.x
    c5_val = result.fun
    return (best_h, c5_val, n_points)
