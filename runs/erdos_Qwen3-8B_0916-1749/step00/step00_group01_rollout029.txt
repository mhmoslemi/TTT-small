    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to evaluate the C5 value given a function h
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        shifts = np.arange(-int(dx * 2), int(dx * 2) + 1)
        overlap = 0
        for k in shifts:
            if k == 0:
                continue
            h_k = np.roll(h, int(k / dx))
            overlap = max(overlap, np.dot(h, 1 - h_k) * dx)
        return overlap
    
    # Optimization function
    def objective_func(h):
        return evaluate_c5(h)
    
    # Gradient descent with momentum
    def gradient_descent(h0, step_size=0.1, momentum=0.9, max_iter=1000, tol=1e-6):
        h = h0.copy()
        v = np.zeros_like(h)
        for _ in range(max_iter):
            c5 = evaluate_c5(h)
            # Compute gradient numerically
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                c5_plus = evaluate_c5(h_plus)
                grad[i] = (c5_plus - c5) / 1e-6
            # Update momentum
            v = momentum * v - step_size * grad
            h += v
            # Clamp values to [0, 1]
            h = np.clip(h, 0, 1)
            # Enforce integral constraint
            integral = np.sum(h) * (2.0 / len(h))
            if integral < 1.0 - 1e-6:
                h = np.clip(h, 0, 1)
                scale = (1.0 - 1e-6) / integral
                h = np.clip(h * scale, 0, 1)
            elif integral > 1.0 + 1e-6:
                h = np.clip(h, 0, 1)
                scale = (1.0 + 1e-6) / integral
                h = np.clip(h * scale, 0, 1)
            # Check for convergence
            if np.linalg.norm(grad) < tol:
                break
        return h
