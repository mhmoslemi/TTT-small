    for _ in range(50):

        if time.time() - t0 > budget_s - 5:

        h_pert = h.copy()
        idx = np.random.choice(n_points, size=10, replace=False)
        for i in idx:
            h_pert[i] = np.clip(h_pert[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Compute overlap
        C5 = compute_C5(h_pert)
        if C5 < current_C5:
            h = h_pert
            current_C5 = C5
    
    # Try gradient descent
    for _ in range(100):
        t0 = time.time()
        if time.time() - t0 > budget_s - 5:
            break
        
        # Define variables
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 1)) * dx))
        constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
        
        # Solve
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.SCS, max_iter=100)

                h = np.array(h_var.value).flatten()
                current_C5 = compute_C5(h)

            pass
