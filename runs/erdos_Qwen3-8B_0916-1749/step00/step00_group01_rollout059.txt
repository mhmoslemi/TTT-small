import numpy as np

def run(seed=42, budget_s=300):

    start_time = time.time()

    # Initial guess from the provided initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    # Function to compute the C5 value
    def compute_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        n = len(h)
        dx = 2.0 / n
        shifted_h = np.zeros_like(h)
        for k in range(1, n):
            shifted_h[k:] = h[:n - k]
        overlaps = np.zeros(n)
        for k in range(n):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps[k] = overlap
        return np.max(overlaps)

    # Objective function for optimization
    def objective(x):
        return compute_c5(x)

    # Optimization using simulated annealing
    n_points = 500
    h = np.random.rand(n_points)
    h = h * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (2.0 / n_points)  # Ensure sum(h) = 1
    h = h / np.sum(h)  # Normalize to ensure sum(h) = 1

    def constraint(x):
        return np.sum(x) - n_points / 2

    def objective(x):
        return compute_c5(x)

    # Use scipy.optimize with a smart initial guess
    def optimize_h():
        h = np.random.rand(n_points)

        h = h / np.sum(h)  # Normalize to ensure sum(h) = 1

        return best_h, best_c5
