```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.sparse import csr_matrix
from scipy.sparse.csgraph import minimum_spanning_tree
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points = 45
    n_points_initial = 45
    x_initial = np.linspace(0, 2, n_points_initial)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming it's provided
    best_c5 = np.inf
    best_h = initial_h_values
    
    # Try with different n_points values
    n_points_list = [20, 30, 40, 50, 60, 70, 80, 90, 100]
    
    for n_points in n_points_list:
        start_time = time.time()
        
        # Create grid
        x = np.linspace(0, 2, n_points)
        dx = 2.0 / n_points
        
        # Objective function
        def objective(h):
            # Compute overlap integrals for all shifts k in [0, 2]
            c5_vals = []
            for k in np.linspace(0, 2, n_points):
                # Shift h by k
                h_shifted = np.roll(h, int(k / dx))
                # Evaluate (1 - h_shifted) on the same grid
                one_minus_h_shifted = 1 - h_shifted
                # Compute overlap integral
                overlap = np.sum(h * one_minus_h_shifted) * dx
                c5_vals.append(overlap)
            return np.max(c5_vals)
        
        # Constraints
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Initial guess
        h0 = np.copy(initial_h_values)
        h0 = np.interp(x, x_initial, h0)
        
        # Optimization
        res = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], constraints=[{'type': 'eq', 'fun': constraint}])
        
        if res.success:
            h_candidate = res.x
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
