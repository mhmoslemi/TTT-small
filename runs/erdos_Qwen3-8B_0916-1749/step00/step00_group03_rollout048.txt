def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    max_n_points = 500  # Balance between speed and accuracy
    n_points = max_n_points
    dx = 2.0 / n_points
    max_iterations = 200
    tolerance = 1e-5

    h = np.copy(initial_h_values)
    h = h[:n_points]  # Ensure it's of correct length
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-5:
        h = np.minimum(1, np.maximum(0, h + (n_points / 2 - sum_h) / n_points))

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        cor = np.correlate(h, 1 - h, mode="full")
        cor = cor[len(cor) // 2 : len(cor) // 2 + 1]
        return np.max(cor) * dx

    def objective(x):
        x = np.array(x, dtype=np.float64)
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - n_points / 2) > 1e-5:
            x = np.minimum(1, np.maximum(0, x + (n_points / 2 - sum_x) / n_points))
        return compute_overlap(x)

    def constraints(x):
        x = np.array(x, dtype=np.float64)
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - n_points / 2) > 1e->5:
            x = np.minimum(1, np.maximum(0, x + (n_points / 2 - sum_x) / n_points))
        return []

    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        options={'gtol': 1e-5, 'maxiter': max_iterations}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-5:
        best_h = np.minimum(1, np.maximum(0, best_h + (n_points / 2 - sum_h) / n_points))

    c5_bound = compute_overlap(best_h)
    return best_h, c5_bound, n_points
