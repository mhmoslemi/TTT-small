import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')[:n_points]
    initial_h = initial_h_values
    initial_sum = np.sum(initial_h)
    if initial_sum != n_points / 2:
        initial_h = initial_h * (n_points / 2) / initial_sum

    def compute_overlap(h):
        h = h.reshape(-1)
        overlap = np.zeros_like(h)
        for k in range(1, n_points - 1):
            shift = k
            if shift > n_points - 1:
                shift = n_points - 1
            h_shifted = np.roll(h, shift)
            overlap += h * (1 - h_shifted)
        return np.max(overlap) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)

    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})
    h_values = result.x
    c5_bound = compute_overlap(h_values)
    return h_values, c5_bound, n_points
