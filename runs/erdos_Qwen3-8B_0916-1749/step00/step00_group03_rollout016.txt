```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp
import pandas as pd

def run(seed=42, budget_s=300, **kwargs):
    # Constants
    n_points = 100  # Number of points for the step function
    dx = 2.0 / n_points
    max_iterations = 50  # Max iterations for optimization
    initial_h_values = np.array([0.0 for _ in range(n_points)])
    
    # Initial guess based on current record construction
    initial_h_values = np.load("initial_h_values.npy")
    initial_h_values = initial_h_values[:n_points]
    
    # Define the cost function to minimize
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h), axis=0)  # Double the array to handle wrap-around
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        overlap = corr[n_points - 1 : n_points + 1]  # Only consider relevant shifts
        return np.max(overlap) * dx
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = Bounds(0, 1)
    A = np.ones((1, n_points))
    constraints = LinearConstraint(A, [n_points / 2], [n_points / 2])
    
    # Start optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values
    
    for _ in range(max_iterations):
        # Perform optimization using SLSQP
        try:
            result = optimize.minimize(
                compute_c5,
                best_h,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                best_h = result.x
                c5 = compute_c5(best_h)
                best_c5 = c5
                print(f"Epoch: {_,} | C5: {best_c5}")
            else:
                print("Optimization failed. Using best so far.")
                break
        except:
            print("Optimization error. Using best so far.")
            break
        
        # Check if budget is exceeded
        if time.time() - start_time > budget_s:
            print(f"Time budget exceeded. Returning best so far.")
            break
    
    # Final check with CVXPY for better solution
    if time.time() - start_time < budget_s:
        try:
            h = cp.Variable(n_points)
            objective = cp.Maximize(cp.multiply(h, 1 - h[1:]))  # Placeholder for correlation
            constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='MOSEK')
            
            if prob.status == 'optimal':
                best_h = h.value
                best_c5 = compute_c5(best_h)
                print(f"CVXPY: C5 = {best_c5}")
            else:
                print("CVXPY failed. Using best so far.")
        except:
            print("CVXPY error. Using best so far.")
    
    return (best_h, best_c5, n_points)
```
