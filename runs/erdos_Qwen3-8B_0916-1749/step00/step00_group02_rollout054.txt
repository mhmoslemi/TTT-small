import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Initial construction from initial_h_values
initial_h = initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h_correlation = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(h_correlation * dx)
        return overlap

    def constraint_sum(h):
        return np.sum(h) - target_sum

    # Initial guess
    h0 = np.copy(initial_h)
    h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    h0 = np.clip(h0, 0, 1)

    # Optimization
    result = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': constraint_sum}])

    # Check if we need to increase n_points
    if result.success:
        best_c5 = result.fun
        best_h = result.x
        best_n_points = n_points
    else:
        best_c5 = np.inf
        best_h = np.zeros(n_points)
        best_n_points = n_points

    # Try increasing n_points if needed
    for n_points in [200, 300, 400]:
        dx = 2.0 / n_points
        target_sum = n_points / 2

        def objective_new(h):
            h = np.clip(h, 0, 1)
            h_correlation = np.correlate(h, 1 - h, mode="full")
            overlap = np.max(h_correlation * dx)
            return overlap

        def constraint_sum_new(h):
            return np.sum(h) - target_sum

        # Initial guess based on previous best
        h0_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
        h0_new = np.clip(h0_new, 0, 1)

        # Optimization
        result = minimize(objective_new, h0_new, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': constraint_sum_new}])

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points

    return best_h, best_c5, best_n_points
