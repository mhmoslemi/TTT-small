        h = np.array(h, dtype=np.float64)
        overlap = np.correlate(h, 1 - h, mode='full')
        return np.max(overlap * dx)
    
    def objective(x):
        return compute_overlap(x)
    
    def bounds(x):
        return [(0, 1) for _ in x]
    
    start_time = time.time()
    result = minimize(
        objective,
        x0=h,
        bounds=bounds(h),
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8},
        jac=None
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    if time.time() - start_time < budget_s:
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbation = np.clip(perturbation, -0.1, 0.1)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            c5 = compute_overlap(new_h)
            if c5 < best_c5:
                best_h = new_h
                best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
