```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import scipy.sparse.linalg as spla
from scipy.optimize import Bounds, minimize
import time

# Precomputed initial solution
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points, can be adjusted
    dx = 2.0 / n_points
    
    # Initial guess
    if initial_h_values.size == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Ensure the initial guess satisfies the integral constraint
    total = np.sum(h) * dx
    if abs(total - 1) > 1e-5:
        h = np.random.uniform(0, 1, n_points)
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Function to compute C_5
    def compute_C5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        h_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_corr) * dx
        return c5
    
    # Optimization objective: minimize C_5
    def objective(h_flat):
        h = h_flat.reshape(-1)
        dx = 2.0 / len(h)
        h_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_corr) * dx
        return c5
    
    # Constraints: sum(h) * dx = 1
    def constraint(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) * 2.0 / len(h) - 1
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="L-BFGS-B",
        options={"max_iter": max_iter, "ftol": tol, "gtol": tol}
    )
    
    if time.time() - start_time > budget_s - 1:
        return result.x, compute_C5(result.x), n_points
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # Try local search with random perturbations
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_h = np.clip(best_h + perturb, 0, 1)
        total = np.sum(new_h) * 2.0 / n_points
        if abs(total - 1) > 1e-5:
            continue
        new_c5 = compute_C5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return best_h, best_c5, n_points
```
