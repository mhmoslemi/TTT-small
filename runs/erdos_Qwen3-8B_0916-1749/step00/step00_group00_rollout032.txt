import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

# Pre-imported functions
# initial_h_values = ... (n=45 samples)

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial step function
    def initial_h_values_func(n_points):
        if n_points == 45:
            return initial_h_values
        else:
            return np.random.uniform(0, 1, n_points)
    
    # Parametrize the step function with n_points samples
    def evaluate_erdos_solution(h_values, n_points):
        dx = 2.0 / n_points
        h = np.array(h_values).reshape(-1)
        h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=0)
        h = h[:n_points]
        
        # Compute correlations
        correlate = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(correlate)
        c5 = max_overlap * dx
        return c5
    
    def fitness_function(params, n_points, dx):
        h = params.reshape(n_points)
        h = np.clip(h, 0, 1)
        # Pad to match full mode correlate
        h_full = np.zeros(2 * n_points)
        h_full[:n_points] = h
        correlate = np.correlate(h_full, 1 - h_full[:n_points], mode="full")
        max_overlap = np.max(correlate)
        c5 = max_overlap * dx
        return c5
    
    def optimize_h(n_points):
        if n_points == 45:
            initial_guess = initial_h_values
        else:
            initial_guess = np.random.uniform(0, 1, n_points)
        
        bounds = Bounds(0, 1)
        result = minimize(
            lambda x: fitness_function(x, n_points, 2.0 / n_points),
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_h = result.x
        best_c5 = fitness_function(best_h, n_points, 2.0 / n_points)
        return best_h, best_c5
    
    # Try different n_points values
    n_points_options = [45, 60, 80, 100, 120, 150, 180, 200]
    
    best_c5 = float('inf')
    best_h_values = None
    best_n_points = 45
    
    for n_points in n_points_options:
        start_time = time.time()
        h_values, c5 = optimize_h(n_points)
        elapsed = time.time() - start_time
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h_values
            best_n_points = n_points
        print(f"n_points: {n_points}, c5: {c5:.6f}, time: {elapsed:.2f}s")
        if elapsed > budget_s / 2:
            break
    
    return best_h_values, best_c5, best_n_points
