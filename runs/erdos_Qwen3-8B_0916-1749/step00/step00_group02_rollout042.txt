import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial construction from given variable
    initial_h_values = np.load("initial_h_values.npy")  # Preloaded with n=45 samples
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Function to compute overlap integral for a given h
    def compute_C5(h):
        h = np.concatenate((h, h))
        shifted = np.zeros_like(h)
        for k in range(1, n_points + 1):
            shifted[k:] = h[:n_points - k + 1]
            overlap = np.sum(h[:n_points] * (1 - shifted[:n_points])) * dx
            if overlap > C5:
                C5 = overlap
        return C5
    
    # Function to compute all overlaps
    def compute_all_overlaps(h):
        h = np.concatenate((h, h))
        C5 = 0.0
        for k in range(1, n_points + 1):
            shifted = h[k:]
            overlap = np.sum(h[:n_points] * (1 - shifted[:n_points])) * dx
            if overlap > C5:
                C5 = overlap
        return C5
    
    # Optimization objective: minimize max overlap
    def objective(h):
        return compute_all_overlaps(h)
    
    # Bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h0 = initial_h_values.copy()
    h0 = h0 / np.sum(h0) * (n_points / 2.0)
    
    # Use COBYLA for constrained optimization
    result = optimize.minimize(
        objective,
        h0,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
        tol=1e-8
    )
    
    # Final solution
    h_values = result.x
    c5_bound = compute_all_overlaps(h_values)
    n_points = len(h_values)
    
    return h_values, c5_bound, n_points
