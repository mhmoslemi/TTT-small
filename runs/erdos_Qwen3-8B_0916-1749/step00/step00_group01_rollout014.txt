import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
import time
import os
import sys

# Pre-imported functions and variables
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 1000
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=True)
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = float('inf')
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h.copy()

    max_evals = 100
    for _ in range(max_evals):
        # Generate a perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)

        # Evaluate the overlap integral
        c5 = evaluate_erdos_solution(h_perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()

        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

    return (best_h, best_c5, n_points)
