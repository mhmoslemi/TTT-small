```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    max_points = 250
    n_points = 250
    h_values = initial_h_values[:n_points]
    best_c5 = evaluate_erdos_solution(h_values)
    start_time = time.time()

    def objective(h):
        return evaluate_erdos_solution(h)

    # Local search
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the solution
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            h_values[i] = np.clip(h_values[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        # Ensure integral constraint
        total = np.sum(h_values)
        scale = (n_points / 2) / total
        h_values = np.clip(h_values * scale, 0, 1)
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            print(f"Improved C5: {best_c5:.6f}")
    
    return h_values, best_c5, n_points
```
