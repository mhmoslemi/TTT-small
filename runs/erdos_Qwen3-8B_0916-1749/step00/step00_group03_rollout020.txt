import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)

    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        initial_h = initial_h * (n_points / 2) / sum_h

    # Function to compute C_5 for a given h_values
    def compute_C5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = np.array(h_values)
        h_shifted = np.concatenate([h[-2:], h[:-2]])
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        C5 = np.max(overlap * dx)
        return C5

    # Function to compute relative change in C5
    def objective(x, current_C5):
        h = np.array(x)
        new_C5 = compute_C5(h)
        return np.abs(new_C5 - current_C5)

    # Function to ensure the sum of h is exactly n/2
    def constraint(x):
        return np.sum(x) - n_points / 2

    # Optimization parameters
    tol = 1e-6
    max_iter = 100
    time_limit = budget_s

    start_time = time.time()

    # Initial optimization
    x0 = initial_h.copy()
    result = minimize(
        lambda x: objective(x, compute_C5(x0)),
        x0,
        method='L-BFGS-B',
        bounds=(0, 1),
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=tol,
        options={'maxiter': max_iter, 'disp': False}
    )

    best_h = result.x
    best_C5 = compute_C5(best_h)

    # Fine-tuning with a smaller step size
    for _ in range(5):
        if time.time() - start_time > time_limit - 1:
            break
        x0 = best_h.copy()
        result = minimize(
            lambda x: objective(x, best_C5),
            x0,
            method='L-BFGS-B',
            bounds=(0, 1),
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=tol,
            options={'maxiter': 20, 'disp': False}
        )
        best_h = result.x
        best_C5 = compute_C5(best_h)

    # Return the best solution
    return (best_h, best_C5, len(best_h))
