import numpy as np
import scipy.optimize
import scipy.interpolate
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)

    # Initial samples (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Objective function to minimize C_5
    def objective(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points//2]])  # Mirror for full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i] (0 <= h[i] <= 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h_initial = initial_h.tolist()

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    time_limit = budget_s

    # Start the optimization
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': max_iter, 'ftol': tol}
    )

    end_time = time occasionally, but this loop tries to find better solutions if time remains
    remaining_time = time_limit - (end_time - start_time)

    if remaining_time > 0:
        # Try with fewer points to avoid local minima
        n_points_new = max(10, n_points // 2)
        dx_new = 2.0 / n_points_new
        h_new = np.linspace(0, 2, n_points_new)
        h_new = np.random.rand(n_points_new)
        h_new = np.minimum(np.maximum(h_new, 0), 1)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)

        # Try optimizing with this new guess
        def objective_new(h):
            h = np.array(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx_new)
            return c5

        def constraint_new(h):
            return np.sum(h) - n_points_new / 2

        new_bounds = [(0.0, 1.0) for _ in range(n_points_new)]
        result_new = minimize(
            fun=objective_new,
            x0=h_new,
            method='L-BFGS-B',
            bounds=new_bounds,
            constraints={'type': 'eq', 'fun': constraint_new},
            options={'maxiter': max_iter, 'ftol': tol}
        )

        if result_new.fun < result.fun:
            result = result_new

    # Return the best solution found
    best_h = result.x
    c5_value = result.fun
    n_points_result = len(best_h)
