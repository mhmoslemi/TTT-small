```python
import numpy as np
import scipy.optimize as optimize
import time
import warnings
from scipy.optimize import LinearConstraint, minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for initial construction

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    C5 = evaluate_erdos_solution(h)
    best_h = h
    best_C5 = C5
    
    # Try a few initial perturbations to escape local optima
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + perturb, 0, 1)
        C5 = evaluate_erdos_solution(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
    
    # Define a function to compute the C5 value for a given h
    def compute_C5(h):
        return evaluate_erdos_solution(h)
    
    # Optimization settings
    max_iter = 100
    time_remaining = budget_s
    start_time = time.time()
    
    # Define the optimization problem
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create constraints
    cons = [
        LinearConstraint(lambda h: np.sum(h), 0, n_points / 2)
    ]
    
    # Initial guess
    x0 = h
    
    # Run optimization with L-BFGS-B
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    if result.success:
        h = result.x
        C5 = result.fun
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
    
    # Try a few more perturbations to escape local optima
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.clip(best_h + perturb, 0, 1)
        C5 = evaluate_erdos_solution(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    return best_h, final_C5, n_points
```
