```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

# Pre-imported functions
initial_h_values = np.load('initial_h.npy')  # Example initial construction

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 100  # Adjust as needed
    dx = 2.0 / n_points
    n_bins = n_points
    max_iter = 100
    tolerance = 1e-6

    # Initial guess: use initial_h_values or a random one
    if n_points != len(initial_h_values):
        initial_h_values = np.random.rand(n_points) * 2 - 1  # Random initial guess for different n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.mean(h) * (n_points / 2)  # Normalize to ensure ∫h(x)dx = 1

    # Define the overlap integral function
    def compute_c5(h):
        h_float = h.astype(np.float64)
        shifts = np.arange(-n_points + 1, n_points)
        max_overlap = 0.0
        for k in shifts:
            if k < 0:
                shift = np.roll(h_float, -k)
            else:
                shift = np.roll(h_float, k)
            overlap = np.sum(h_float * (1 - shift)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    # Optimization function
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.mean(h) * (n_points / 2)  # Ensure ∫h(x)dx = 1
        return compute_c5(h)

    # Constraints: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2

    # Bounds for h
    bounds = Bounds(0, 1)

    # Optimization
    result = minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': max_iter, 'ftol': tolerance}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        print("Time budget exceeded")
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
```
