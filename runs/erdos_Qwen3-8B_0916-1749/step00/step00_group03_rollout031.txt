import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assume initial_h.npy is available

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 1000  # A reasonable starting point for balance between speed and precision
    h = np.random.uniform(0, 1, n_points)
    total = np.sum(h) * (2.0 / n_points)
    h = h * (1 / (total / (2.0 / n_points)))

    # Optimization loop
    start_time = time.time()

    def objective(x):
        h = x.copy()
        total = np.sum(h) * (2.0 / n_points)
        h = h * (1 / (total / (2.0 / n_points)))
        return evaluate_erdos_solution(h)

    # Constraints: sum(h) must be n_points / 2
    def constraint(x):
        return np.sum(x) * (2.0 / n_points) - n_points / 2

    def bounds(x):
        return np.array([0.0, 1.0])

    def gradient(x):
        # Placeholder for gradient, but we'll use a numerical approximation
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        jac=gradient,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points

    # Check if we can reduce n_points further
    if time.time() - start_time > budget_s - 10:
        return (best_h, best_c5, best_n_points)

    # Try with a lower n_points
    n_points = 500
    h = np.random.uniform(0, 1, n_points)
    total = np.sum(h) * (2.0 / n_points)
    h = h * (1 / (total / (2.0 / n_points)))

    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        jac=gradient,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points

    return (best_h, best_c5, best_n_points)
