```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # initial n_points
    initial_h_values = np.array([0.5] * n_points)
    
    def evaluate_erdos_solution(h):
        half = h.shape[0] // 2
        cross_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(cross_corr) * (2.0 / n_points)
    
    # Define the initial point
    if initial_h_values is not None:
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.concatenate([initial_h_values[:n_points // 2], initial_h_values[n_points // 2:]])
        initial_h_values = initial_h_values / np.sum(initial_h_values) * n_points / 2
    
    def _fun(x):
        nonlocal n_points
        n_points = len(x)
        dx = 2.0 / n_points
        h = x
        cross_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(cross_corr) * dx
    
    def _grad(x):
        return np.zeros_like(x)
    
    def _constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def _constraint_bounds(x):
        return np.all((x >= 0) & (x <= 1))
    
    def _minimize_erdos():
        n = 100  # number of points
        dx = 2.0 / n
        variables = np.random.rand(n)
        variables = variables / np.sum(variables) * n / 2
        
        # Create bounds and constraints
        bounds = Bounds(0, 1)
        constraint = LinearConstraint(np.ones(n), lb=n / 2, ub=n / 2)
        
        # Initialize the optimization problem
        result = optimize.minimize(
            _fun,
            variables,
            bounds=bounds,
            constraints=constraint,
            method='SLSQP',
            jac=_grad,
            options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            return result.x
        else:
            return variables
    
    start_time = time.time()
    
    # Try with different n_points to find the best fit
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    
    for n in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 10:
            break
        variables = _minimize_erdos()
        c5 = evaluate_erdos_solution(variables)
        if c5 < best_c5:
            best_h = variables
            best_c5 = c5
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
```
