import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from pre-defined points
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this is loaded from a file
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = initial_h / np.mean(initial_h) * (n_points / 2)  # Normalize to match constraint
    
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.pad(h, (0, 1), 'constant')  # To handle full mode in correlate
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - (n_points / 2)
    
    # Initial guess based on provided initial_h_values
    initial_guess = initial_h
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Reduce n_points for faster evaluation
    n_points = 100
    dx = 2.0 / n_points
    
    # Define function to generate a step function with integer values
    def generate_step(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5
        return h
    
    # Optimization with reduced n_points
    def objective_reduced(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.pad(h, (0, 1), 'constant')  # To handle full mode in correlate
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum_reduced(h):
        return np.sum(h) - (n_points / 2)
    
    initial_guess_reduced = generate_step(n_points)
    
    result_reduced = minimize(
        objective_reduced,
        initial_guess_reduced,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum_reduced}],
        options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e': 1e-8, 'disp': False}
    )
    
    best_h_reduced = result_reduced.x
    best_c5_reduced = compute_c5(best_h_reduced)
    
    return (best_h_reduced, best_c5_reduced, n_points)
