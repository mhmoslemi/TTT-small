```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h_values = np.array([0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                               0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                               0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                               0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                               0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75])

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2

    def compute_c5(h):
        h = np.array(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h):
        return compute_c5(h)

    def bounds(h):
        return [(0.0, 1.0) for _ in h]

    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def evaluate_solution(h):
        nonlocal best_c5, best_h
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Use a grid search with random perturbations as a starting point
    for _ in range(50):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        h = initial_h_values + perturb
        h = np.clip(h, 0.0, 1.0)
        h = h * (sum_h / np.sum(h)) if np.sum(h) != 0 else h

        evaluate_solution(h)

    # Use a local optimizer to refine the solution
    for _ in range(20):
        start_time = time.time()
        result = optimize.minimize(
            objective,
            x0=best_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'gtol': 1e-5, 'ftol': 1e-5, 'maxiter': 100}
        )
        if time.time() - start_time > budget_s:
            break
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (sum_h / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
        evaluate_solution(best_h)

    return best_h, best_c5, n_points
```
