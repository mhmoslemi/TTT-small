import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial construction (from initial_h_values)
    initial_h = initial_h_values
    
    # Discretize the interval [0, 2] into n_points samples
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        # Compute the overlap integral
        h_array = np.array(h)
        overlap = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Define bounds for h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define initial guess
    x0 = initial_h
    
    # Perform optimization with bounds and constraints
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxfun': 1000, 'disp': False}
    )
    
    # Check if the result is within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (x0, objective(x0), n_points)
    
    # Return the best solution found
    return (result.x, result.fun, n_points)

# Initialize start time
start_time = time.time()
