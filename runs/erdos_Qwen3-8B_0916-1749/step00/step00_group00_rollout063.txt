import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported evaluation functions and initial construction
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    # Start timing
    start_time = time.time()
    
    # Initial n_points from the provided initial_h_values
    initial_n_points = len(initial_h_values)
    initial_h = initial_h_values
    
    # We'll try a few different starting points, including the initial one
    starting_points = [initial_h]
    for _ in range(3):
        # Randomly perturb the starting point to escape local optima
        perturbed_h = initial_h.copy()
        perturbed_h[np.random.choice(len(perturbed_h), size=10, replace=False)] = np.random.rand(10)
        starting_points.append(perturbed_h)
    
    best_c5 = float('inf')
    best_h = initial_h
    
    for start_h in starting_points:
        # Ensure the starting point is valid
        if np.sum(start_h) * 2 / len(start_h) != 1:
            start_h = np.array([1.0 if i % 3 == 0 else 0.0 for i in range(len(start_h))])
        
        # Define the optimization variables
        n_points = len(start_h)
        dx = 2.0 / n_points
        
        def objective(x, dx, n_points):
            h = x
            # Compute the overlap integral for each k
            # Since the maximum is over k, we need to check all k such that x+k is in [0,2]
            k_values = np.arange(-n_points // 2, n_points // 2 + 1)
            max_overlap = 0.0
            for k in k_values:
                if k < 0:
                    shifted_h = np.roll(h, k)
                else:
                    shifted_h = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted_h)) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
            return max_overlap
        
        # Define the bounds for the variables
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess
        x0 = start_h
        
        # Optimization using L-BFGS-B
        result = optimize.minimize(
            fun=lambda x: objective(x, dx, n_points),
            x0=x0,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        # Extract the best solution found
        best_h_current = result.x
        current_c5 = objective(best_h_current, dx, n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h_current
        
        # Check the time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, len(best_h)
