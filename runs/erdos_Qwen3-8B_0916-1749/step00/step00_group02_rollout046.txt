import numpy as np
import scipy.optimize as optimize
import time
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Set up for optimization
def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster evaluation
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-5
    best_c5 = np.inf
    best_h = None
    
    # Initial guess from provided starting point
    initial_h = np.array(initial_h_values[:n_points])
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Normalize to satisfy constraint
    
    def objective(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy constraint
        
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Use differential evolution for global optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a reduced population and lower precision for faster evaluation
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = optimize.differential_evolution(
            objective,
            bounds,
            maxiter=max_iter,
            popsize=20,
            strategy='best1bin',
            tol=tol,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            polish=True,
            disp=False,
            callback=lambda x: None
        )
    
    # Extract the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Evaluate the best solution
    c5 = np.max(np.abs(np.correlate(best_h, 1 - best_h, mode="full"))) * dx
    
    return (best_h, c5, n_points)
