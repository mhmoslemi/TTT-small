def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values as starting point
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h_flat = h.flatten()
        corr = correlate(h_flat, 1 - h_flat, mode="full")
        corr = corr[n_points - 1: 2 * n_points - 1]
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = minimize(
        objective,
        initial_h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8},
        tol=1e-6
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # If no improvement, return the initial solution
    if best_c5 < compute_c5(initial_h_values):
        return (best_h, best_c5, n_points)
    else:
        return (initial_h_values, compute_c5(initial_h_values), n_points)
