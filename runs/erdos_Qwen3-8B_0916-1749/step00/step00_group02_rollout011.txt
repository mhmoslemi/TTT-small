```python
import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=300, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial configuration
    n_points = 200
    initial_h_values = np.array([0.5] * n_points)
    initial_h_values = np.roll(initial_h_values, 50)
    
    def objective(x):
        h = np.array(x)
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * (2.0 / n_points)
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Initial guess
    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}]
    )
    
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    # Check if we can reduce n_points for faster evaluation
    for n in [100, 80, 60, 40, 20]:
        h = np.zeros(n)
        # Simple heuristic: alternate 1s and 0s
        h[::2] = 1
        h[1::2] = 0
        h = np.clip(h, 0, 1)
        # Ensure sum is n / 2
        h = h / np.sum(h) * n / 2
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)
        if c5 < c5_bound:
            c5_bound = c5
            h_opt = h
            n_points = n
    
    return (h_opt, c5_bound, n_points)
```
