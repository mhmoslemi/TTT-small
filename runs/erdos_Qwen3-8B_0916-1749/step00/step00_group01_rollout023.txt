import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import eigs
from scipy.optimize import minimize, LinearConstraint, Bounds
from sklearn.decomposition import PCA

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)

    # Define initial construction from the initial_h_values
    initial_h = np.load("initial_h_values.npy")
    n_points = len(initial_h)
    
    # If initial_h is not provided, default to a basic construction
    if not hasattr(initial_h, '__len__') or len(initial_h) == 0:
        n_points = 500
        initial_h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        initial_h[x < 1] = 1
        initial_h[x > 1.5] = 0
        initial_h[ (x > 1) & (x < 1.5) ] = 0.5
        initial_h = initial_h.tolist()

    # Function to compute C5 from the step function
    def compute_C5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute all shifted versions of h
        h_shifted = [np.roll(h, k) for k in range(n)]
        # Compute overlap integrals
        C5_values = []
        for k in range(n):
            overlap = np.sum(h * (1 - h_shifted[k]))
            C5_values.append(overlap * dx)
        return max(C5_values)

    # Define objective function to minimize C5
    def objective(x):
        return compute_C5(x)

    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Initial guess
    x0 = initial_h

    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization start time
    start_time = time.time()

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 500, 'ftol': 1e-8})

    # Extract the best solution
    best_h = result.x
    best_C5 = compute_C5(best_h)
    n_points_used = len(best_h)

    # Return the result
    return (best_h, best_C5, n_points_used)
