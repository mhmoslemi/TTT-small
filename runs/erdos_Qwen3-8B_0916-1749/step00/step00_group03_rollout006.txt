import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial guess based on the provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Assuming 'initial_h_values.npy' is provided
    n_initial = len(initial_h_values)
    
    # Start with a smaller number of points to speed up the search
    n_points = 100
    dx = 2.0 / n_points
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        h = np.array(h)
        h = np.concatenate((h, h))[:n_points]  # Ensure the length is correct
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess based on the initial_h_values scaled to the new n_points
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_initial), initial_h_values)
    
    # Initialize best solution
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    best_time = 0
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Perturb the current guess
        perturbation = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturbation, 0.0, 1.0)
        
        # Compute objective for the new guess
        new_c5 = objective(new_h)
        
        # Acceptance probability
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                best_h = new_h
                best_c5 = new_c5
        
        # Cool down
        temp *= cooling_rate
        
        # Check if time budget is reached
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
    
    return (best_h, best_c5, n_points)
