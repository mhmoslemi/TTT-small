import scipy.optimize as optimize

from scipy.optimize import LinearConstraint
from scipy.optimize import minimize

def run(seed=42, budget_s=300, **kwargs):
    n_points = 100  # Start with 100 points for optimization
    dx = 2.0 / n_points
    max_attempts = 50
    best_c5 = float('inf')
    best_h = None

    # Define the function to compute C5
    def compute_C5(h_values):
        h = h_values
        h = np.concatenate([h, h])  # Double the array for full convolution
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Optimization function
    def objective_func(h_values):
        nonlocal best_c5
        c5 = compute_C5(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
        return c5

    # Constrains: sum(h) = n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - (n_points / 2)

    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0, 1)

    # Initial guess
    initial_guess = np.array(initial_h)

    # Rescale initial guess to match n_points
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    # Create the constraint
    cons = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)

    # Perform optimization
    start_time = time.time()
    result = minimize(
        objective_func,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'eps': 1e-6, 'ftol': 1e-9, 'disp': False}
    )

    # Best solution found
    best_h = result.x
    best_c5 = compute_C5(best_h)

    # Optionally refine with more points
    if best_c5 > 0.380876:
        n_points = 200

        initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)

        cons = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)

        result = minimize(
            objective_func,

            method='SLSQP',
            bounds=bounds,
            constraints=cons,
