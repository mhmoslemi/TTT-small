import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

# Preloaded functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = initial_h_values[:n_points]  # Initial guess
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize C5
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        shifts = np.arange(-n_points + 1, n_points)
        integrals = []
        for k in shifts:
            if k == 0:
                continue
            shifted = np.roll(h, k)
            integral = np.sum(h * (1 - shifted)) * dx
            integrals.append(integral)
        return max(integrals)
    
    # Gradient function (approximate using finite differences)
    def gradient(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        shifts = np.arange(-n_points + 1, n_points)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h[i] += 1e-6
            integrals = []
            for k in shifts:
                if k == 0:
                    continue
                shifted = np.roll(h, k)
                integral = np.sum(h * (1 - shifted)) * dx
                integrals.append(integral)
            max_integral = max(integrals)
            grad[i] = (max_integral - objective(x)) / 1e-6
            h[i] -= 1e-6
        return grad
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-4
    start_time = time.time()
    
    # Perform optimization
    result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=Bounds([0, 1]), options={'maxiter': max_iterations, 'ftol': tolerance})
    final_h = result.x
    final_h = np.clip(final_h, 0, 1)
    
    # Compute C5
    shifts = np.arange(-n_points + 1, n_points)
    integrals = []
    for k in shifts:
        if k == 0:
            continue
        shifted = np.roll(final_h, k)
        integral = np.sum(final_h * (1 - shifted)) * dx
        integrals.append(integral)
    c5_bound = max(integrals)
    
    # If time is left, try increasing n_points
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = final_h[:n_points]
        h = np.clip(h, 0, 1)
        
        # Redefine objective and gradient
        def objective_h(x):
            h = x.copy()

            shifts = np.arange(-n_points + 1, n_points)
            integrals = []
            for k in shifts:
                if k == 0:
                    continue
                shifted = np.roll(h, k)
                integral = np.sum(h * (1 - shifted)) * dx
                integrals.append(integral)
            return max(integrals)
        
        def gradient_h(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            shifts = np.arange(-n_points + 1, n_points)
            grad = np.zeros_like(h)
            for i in range(n_points):
                h[i] += 1e-6
                integrals = []
                for k in shifts:
                    if k == 0:
                        continue
                    shifted = np.roll(h, k)
                    integral = np.sum(h * (1 - shifted)) * dx
                    integrals.append(integral)
                max_integral = max(integrals)
                grad[i] = (max_integral - objective_h(x)) / 1e-6
                h[i] -= 1e-6
            return grad
        
        result = minimize(objective_h, h, method='L-BFGS-B', jac=gradient_h, bounds=Bounds([0, 1]), options={'maxiter': 50, 'ftol': tolerance})
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        
        shifts = np.arange(-n_points + 1, n_points)
        integrals = []
        for k in shifts:
            if k == 0:
                continue
            shifted = np.roll(final_h, k)
            integral = np.sum(final_h * (1 - shifted)) * dx
            integrals.append(integral)
        c5_bound = max(integrals)
    
    return (final_h, c5_bound, n_points)
