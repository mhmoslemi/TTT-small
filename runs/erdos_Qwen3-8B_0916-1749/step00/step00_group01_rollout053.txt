import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction
    n_points = 47
    h = initial_h_values.copy()
    h = h[:n_points]
    
    # Function to evaluate C5
    def evaluate_c5(h):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Function to compute the overlap integral for a given shift
    def compute_overlap(h, shift):
        n = len(h)
        dx = 2.0 / n
        h_shifted = np.roll(h, shift)
        integral = np.sum(h * (1 - h_shifted)) * dx
        return integral
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        n = len(h)
        max_overlap = 0.0
        for shift in range(-n + 1, n):
            if shift == 0:
                continue
            overlap = compute_overlap(h, shift)
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = h
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    
    # Perform optimization
    result = opt.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="L-BFGS-B",
        options={"maxiter": max_iter, "ftol": tol, "gtol": tol}
    )
    
    # Check if optimization succeeded
    if result.success:
        h_opt = result.x
        c5 = evaluate_c5(h_opt)
    else:
        h_opt = h_initial
        c5 = evaluate_c5(h_opt)
    
    # Return the best solution found
    return (h_opt, c5, n_points)
