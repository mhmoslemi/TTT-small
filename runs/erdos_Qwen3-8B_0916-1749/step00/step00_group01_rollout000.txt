```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = np.array([0.5] * n_points)
    dx = 2.0 / n_points
    evaluations = []

    def compute_overlap(h):
        h = np.array(h)
        shifted_h = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return overlap

    def compute_c5(h):
        max_overlap = 0
        for k in range(1, n_points):
            shifted_h = np.roll(h, shift=k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    def objective(x):
        return compute_c5(x)

    def bounds_func(x):
        return np.array([0.0, 1.0])

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    def optimize_h():
        x0 = np.array([0.5] * n_points)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: constraint_sum(x)})
        result = optimize.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x

    best_h = initial_h_values
    best_c5 = compute_c5(best_h)

    for _ in range(10):
        start_time = time.time()
        h = optimize_h()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
```
