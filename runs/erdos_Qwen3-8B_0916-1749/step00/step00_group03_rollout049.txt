```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h_values = np.load("initial_h_values.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    def compute_C5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # For full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1 : 2 * n_points - 1]
        C5 = np.max(corr * dx)
        return C5
    
    def objective(x):
        x = np.array(x)
        x = np.clip(x, 0, 1)
        return compute_C5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial guess
    x0 = initial_h_values.copy()
    
    # Optimization
    start_time = time.time()
    result = minimize(objective, x0, method='L-BFGS-B', constraints=constraints, options={'ftol': 1e-8, 'maxiter': 100})
    
    # Extract best solution
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # Ensure within bounds
    best_h = np.clip(best_h, 0, 1)
    best_C5 = compute_C5(best_h)
    
    # Return result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Warning: Optimization exceeded budget")
    
    return (best_h, best_C5, len(best_h))
```
