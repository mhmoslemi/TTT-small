import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Initial construction from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    def compute_c5(h):
        h = np.array(h)
        h_pad = np.concatenate([h, h])  # Pad for full convolution
        result = np.correlate(h_pad, 1 - h, mode="full")
        result = result[n_points - 1 : n_points + 1]
        c5 = np.max(result * dx)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h[i] in [0, 1]
    bounds = Bounds(lb=0.0, ub=1.0)
    
    # Initial guess
    h_initial = initial_h
    
    # Optimizer settings
    method = 'SLSQP'  # Sequential Least Squares Quadratic Programming
    maxiter = 1000
    tol = 1e-6
    
    start_time = time.time()
    
    # Optimization loop with time budget
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Run optimization
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method=method,
            options={'maxiter': maxiter, 'ftol': tol}
        )
        
        if result.success:
            h_initial = result.x
            current_c5 = compute_c5(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_initial
        else:
            break
    
    return best_h, best_c5, len(best_h)
