```python
import numpy as np
import scipy.optimize as optimize
import sys
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess from pre-imported starting point
    h0 = initial_h_values
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Objective function: minimize the maximum overlap
    def objective(x):
        h = np.copy(x)
        h[h < 0] = 0
        h[h > 1] = 1
        return compute_c5(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Initial guess
    x0 = np.copy(h0)
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    
    # Start timing
    start_time = time.time()
    
    # Perform optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iter, 'ftol': tolerance}
    )
    
    # Check if we are within budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        print("Warning: Optimization time exceeded budget, returning initial solution")
        return (h0, compute_c5(h0), n_points)
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
```
