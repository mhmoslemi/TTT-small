            h >= 0,

        ]
        obj = cp.Minimize(cp.Max([cp.sum(cp.multiply(h[i], 1 - h[j])) * dx for i in range(n_points) for j in range(n_points)]))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='ECOS', abstol=1e-8, reltol=1e-8, maxiters=1000)
        except:
            pass
        return h.value
    
    # Perform a series of local optimizations starting from initial_h
    h_values = initial_h.copy()
    best_C5 = compute_C5(h_values)
    best_h = h_values
    
    # Local search
    for _ in range(5):
        h = optimize_h()
        if h is not None:
            h_values = h
            current_C5 = compute_C5(h_values)
            if current_C5 < best_C5:
