import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse

from scipy.sparse import csr_matrix

from numpy import linalg as la
from numba import njit, prange
import math

# Pre-imports
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=300, **kwargs):
    # Fix seed
    np.random.seed(seed)

    # Define the number of points
    n_points = 100  # Start with 100 points for speed and accuracy balance

    # Precompute dx
    dx = 2.0 / n_points

    # Define initial guess from provided initial_h_values
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        # Create a simple initial guess: step function
        for i in range(n_points):
            if i < n_points / 2:
                h[i] = 0.5
            else:
                h[i] = 1.0

    # Ensure the sum of h is correct
    total_sum = np.sum(h) * dx
    if not np.isclose(total_sum, 1.0):
        # Adjust h to meet the constraint
        scale_factor = 1.0 / total_sum
        h *= scale_factor

    # Function to compute overlap integral for a given h
    def compute_c5(h):
        h = h.copy()
        # Compute 1 - h
        one_minus_h = 1.0 - h
        # Compute correlations
        corrs = np.correlate(h, one_minus_h, mode='full')
        # Multiply by dx
        return np.max(corrs * dx)

    # Objective function to minimize
    def objective(x):
        x = x.copy()
        # Ensure x is in [0, 1]
        x = np.clip(x, 0.0, 1.0)
        # Compute the overlap integral
        return compute_c5(x)

    # Function to generate shifted versions of h
    def shifted_h(x, k):
        # Shift x by k
        shifted = np.roll(x, k)
        return shifted

    # Define the constraints
    # Constraint: sum(h) * dx == 1.0
    # This is already satisfied by scaling

    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use SLSQP to minimize C5
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the best solution
    best_h = result.x

    # Compute the best C5 value
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
