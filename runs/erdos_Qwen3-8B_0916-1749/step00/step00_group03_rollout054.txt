import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
import matplotlib.pyplot as plt
from scipy.integrate import simps

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points

    # Initial guess based on a uniform distribution with peaks
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4 or i > 3 * n_points // 4:
            h_initial[i] = 0.5
        else:
            h_initial[i] = 0.25

    def objective(h):
        # Compute the overlap integral for all lags
        h_values = np.array(h)
        correlates = np.correlate(h_values, 1 - h_values, mode="full")
        # Only consider lags where x + k is within [0, 2]
        valid_lags = np.where((np.arange(len(correlates)) >= -n_points + 1) &
                             (np.arange(len(correlates)) <= n_points - 1))
        valid_correlates = correlates[valid_lags]
        # Compute the maximum overlap integral
        max_overlap = np.max(valid_correlates * dx)
        return max_overlap

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        objective,
        x0=h_initial,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
