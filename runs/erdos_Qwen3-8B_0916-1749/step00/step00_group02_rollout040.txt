import numpy as np
import scipy.optimize
import scipy.sparse
import scipy.sparse.linalg
import time
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 200
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess from the provided initial_h_values
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Set up the variables
    h_values = h
    
    # Define the function to minimize
    def objective(h):
        nonlocal n_points, dx
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define the constraint: sum(h) * dx = 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds for h: each entry must be in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use scipy.optimize to minimize the objective
    result = scipy.optimize.minimize(
        objective,
        h_values,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Ensure the final h is clamped to [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Check if the final solution meets the constraints
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-5:
        best_h = np.random.rand(n_points)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
