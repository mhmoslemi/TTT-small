import numpy as np
import time
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize_slsqp
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import sys
import os
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100
    total_samples = n_points
    dx = 2.0 / total_samples

    # Initial guess from pre-defined h_values
    initial_guess = initial_h_values
    if len(initial_guess) != total_samples:
        initial_guess = [0.5] * total_samples

    def objective(h):
        nonlocal best_c5
        h = np.array(h, dtype=np.float64)
        conv = np.convolve(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    def constraint_sum(h):
        return np.sum(h) - total_samples / 2

    def bounds_func(x):
        return [0.0, 1.0]

    bounds = Bounds(lambda x: 0.0, lambda x: 1.0)

    # Optimization settings
    options = {
        'maxiter': 1000,
        'ftol': 1e-8,
        'disp': False
    }

    # Initial optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=LinearConstraint(lambda h: np.sum(h), 1, 1),
        options=options
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Post-processing and refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        # Try perturbing the solution slightly
        perturbation = np.random.normal(0, 0.05, size=total_samples)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = np.array(perturbed_h, dtype=np.float64)
        conv = np.convolve(perturbed_h, 1 - perturbed_h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h.copy()

    # Final evaluation
    final_c5 = best_c5
    print(f"Final C5 value: {final_c5}")
    return best_h, final_c5, total_samples
