        n_points = len(h_values)

        h = np.array(h_values, dtype=np.float64)
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:-k]
        overlap = np.sum(h * (1 - shifted))
        return overlap * dx

    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        c5_values = []
        n_points = len(h_values)
        for k in range(1, n_points):
            shifted = np.roll(h_values, k)
            c5 = np.sum(h_values * (1 - shifted)) * (2.0 / n_points)
            c5_values.append(c5)
        return np.max(c5_values)

    # Define the constraint: sum(h) = n_points / 2
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Define bounds for each h[i] as [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use Sequential Least Squares Programming (SLSQP)
    result = optimize.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False},
