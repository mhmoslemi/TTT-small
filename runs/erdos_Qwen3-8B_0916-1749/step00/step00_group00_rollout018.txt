```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
import itertools
from functools import lru_cache

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=300, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial configuration
    n_points = 200
    max_iter = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = None

    # Define the cost function for optimization
    def cost_function(h):
        nonlocal best_c5
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        integral = np.sum(h) * (2.0 / n_points)
        if abs(integral - 1.0) > 1e-6:
            return float('inf')  # Penalize constraint violation
        
        # Compute correlation (overlap integral)
        shifted_h = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted_h, mode='full')
        c5 = np.max(overlap * (2.0 / n_points))
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        return c5

    # Initial guess
    h_initial = initial_h_values.copy()[:n_points]
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Optimization using L-BFGS-B
    def objective(h):
        return cost_function(h)

    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iter, 'ftol': tolerance}
    )

    # Extract best solution
    best_h = result.x
    best_c5 = cost_function(best_h)

    # Evaluate and return result
    return best_h, best_c5, n_points
```
