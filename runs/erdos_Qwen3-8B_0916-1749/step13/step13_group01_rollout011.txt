```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.array([0.5 if (i % 7 == 0 or i % 10 == 0) else 0.0 for i in range(n_points)])
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h + np.random.uniform(-0.1, 0.1, size=n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with adaptive perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution with informed sampling
    def generate_new_start(h_current):
        h = np.copy(h_current)
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for iteration in range(100):
        # Generate new start points
        new_starts = [generate_new_start(best_h) for _ in range(5)]
        
        # Evaluate new starts
        for start in new_starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perform local search
        best_h, best_c5 = optimize_h(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
