import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Starting with a base of 70 samples
    initial_h_values = initial_h_values  # Globally defined
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Objective function: minimize max overlap
    def objective(h):
        return compute_c5(h)
    
    # Constraint function: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    def perturb_h(h, strength):
        perturbation = np.random.normal(0, strength, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    def explore_starting_points(h):
        starts = [np.copy(h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                yield optimized_h, current_c5
            except:
                continue
    
    def run_optimization():
        best_h = np.copy(initial_h_values)
        best_c5 = compute_c5(best_h)
        
        time_start = time.time()
        
        # Initial optimization
        for _ in range(20):
            for start in explore_starting_points(best_h):
                optimized_h, current_c5 = start
                if current_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = current_c5
            if time.time() - time_start > budget_s:
                break
        
        # Adaptive perturbation with gradually decreasing strength
        for perturbation_strength in [0.1, 0.05, 0.02, 0.01, 0.005]:
            for _ in range(10):
                perturbed_h = perturb_h(best_h, perturbation_strength)
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = current_c5
            if time.time() - time_start > budget_s:
                break
        
        # Local search with fine-tuned perturbation
        for _ in range(10):
            perturbed_h = perturb_h(best_h, 0.005)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        
        return best_h, best_c5, n_points

    best_h, best_c5, n_points = run_optimization()
    return best_h, best_c5, n_points
