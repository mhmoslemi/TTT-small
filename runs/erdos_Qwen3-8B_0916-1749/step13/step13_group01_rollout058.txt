        )

    def smart_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    def local_search(h_current):
        best_h = h_current

        except:

        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
