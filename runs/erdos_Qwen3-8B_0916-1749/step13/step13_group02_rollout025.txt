import scipy.optimize as optimize
import time

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute all shift overlaps
    def compute_all_overlaps(h):
        h = np.array(h)
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k == 0:
                continue
            if k > 0:
                overlap = np.sum(h[:n_points - k] * (1 - h[k:]))
            else:
                overlap = np.sum(h[-k:] * (1 - h[:n_points + k]))
            overlaps.append(overlap * dx)
        return overlaps
    
    # Function to compute the maximum overlap
    def compute_max_overlap(h):
        return max(compute_all_overlaps(h))
    
    # Function to compute the average overlap
    def compute_average_overlap(h):
        return np.mean(compute_all_overlaps(h))
    
    # Function to compute the variance of overlaps
    def compute_overlap_variance(h):
        overlaps = compute_all_overlaps(h)
        return np.var(overlaps)
    
    # Define the objective function
    def objective(h):
        return compute_max_overlap(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with structured perturbations and adaptive methods
    def optimize_h(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_max_overlap(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_max_overlap(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5

            optimized_h, current_c5 = optimize_h(start_h)
            if current_c5 < best_c5:

                best_c5 = current_c5
