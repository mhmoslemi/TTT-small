import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem with constraints
    def optimize_with_constraints(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function for adaptive perturbation and random restarts
    def perturb_and_restart(h_current, num_restarts=5, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(num_restarts):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_with_constraints(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function for group-wise optimization with blocking
    def block_optimize(h, block_size=10):
        h_blocks = np.array_split(h, block_size)
        optimized_blocks = []
        for block in h_blocks:
            block = np.array(block)
            result = optimize_with_constraints(block)
            optimized_blocks.append(result[0])
        return np.concatenate(optimized_blocks)
    
    # Main optimization loop with multiple stages
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Stage 1: Basic optimization
    for _ in range(5):
        optimized_h, _ = optimize_with_constraints(np.copy(best_h))
        best_h = optimized_h
    
    # Stage 2: Random restarts and perturbations
    for _ in range(3):
        best_h, best_c5 = perturb_and_restart(best_h)
    
    # Stage 3: Block-wise optimization
    for _ in range(3):
        best_h = block_optimize(best_h)
    
    # Stage 4: Final fine-tuning
    for _ in range(10):
        optimized_h, current_c5 = optimize_with_constraints(np.copy(best_h))
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
    
    # Stage 5: Local search with adaptive perturbation
    for _ in range(5):
        perturbation_strength = 0.1 / (1 + _)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
        optimized_perturbed, current_c5_perturbed = optimize_with_constraints(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_h = optimized_perturbed
            best_c5 = current_c5_perturbed
    
    # Check time budget and return the best result
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
