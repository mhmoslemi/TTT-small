import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver and adaptive parameters
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8, 'disp': False}
        )
        return result.x, result.fun
    
    # Function for multi-start local search with diverse perturbations
    def multi_start_local_search(h_initial, num_starts=5, perturbation_strength=0.1):
        best_h = np.copy(h_initial)
        best_c5 = compute_c5(best_h)
        for _ in range(num_starts):
            perturbation = np.random.normal(0, perturbation_strength, size=n_points)
            perturbed_h = np.clip(h_initial + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                best_h = optimized_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function for adaptive perturbation with increasing diversity
    def adaptive_perturbation_search(h_initial, max_iterations=100, strength_decay=0.95):
        h = np.copy(h_initial)
        best_c5 = compute_c5(h)
        for i in range(max_iterations):
            strength = 0.1 * (strength_decay ** i)
            perturbation = np.random.normal(0, strength, size=n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h)
            if current_c5 < best_c5:
                h = optimized_h
                best_c5 = current_c5
        return h, best_c5
    
    # Main optimization loop with diverse strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform initial optimization
    optimized_h, current_c5 = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = current_c5
    
    # Perform multi-start local search
    best_h, best_c5 = multi_start_local_search(best_h)
    
    # Perform adaptive perturbation search
    best_h, best_c5 = adaptive_perturbation_search(best_h)
    
    # Check time budget
    if time.time() - start_time > budget_s - 10:
        return best_h, best_c5, len(best_h)
    
    # Final refinement with more aggressive local search
    best_h, best_c5 = multi_start_local_search(best_h, num_starts=10, perturbation_strength=0.2)
    
    return best_h, best_c5, len(best_h)
