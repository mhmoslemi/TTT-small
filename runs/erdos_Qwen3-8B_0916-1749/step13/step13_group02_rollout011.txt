import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Load initial construction
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Create an initial guess based on a piecewise-constant function
    # We divide the interval [0, 2] into 5 equal parts and set h(x) = 0.5 in the first two and 1.0 in the last three
    h[0:n_points//5] = 0.5
    h[n_points//5:2*n_points//5] = 0.5
    h[2*n_points//5:] = 1.0
    h = np.clip(h, 0, 1)
    
    # Define the function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h_values):
        return compute_c5(h_values)
    
    # Define the constraint function
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current, perturbation_strength=0.05):
        best_h = np.copy(h_current)
        best_c5 = compute_c5(best_h)
        for _ in range(5):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a global search with multiple starting points
    def global_search(start_points):
        best_h = start_points[0]
        best_c5 = compute_c5(best_h)
        for start in start_points[1:]:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Generate multiple starting points
    start_points = [np.copy(best_h) for _ in range(5)]
    for _ in range(10):
        # Perform optimization from each starting point
        for start in start_points:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perturb the current best solution and optimize
        perturbation_strength = 0.1 / (1 + _)  # Decreasing perturbation strength over time
        for _ in range(5):
            perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_strength, size=n_points), 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
