import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with informed sampling
    def generate_new_start(h_current):
        h = np.copy(h_current)
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Adaptive optimization loop
    for i in range(100):
        # Generate new starting points with informed sampling
        for _ in range(5):
            new_start = generate_new_start(best_h)
            optimized_h, current_c5 = optimize_h(new_start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        # Local search with directional perturbations
        best_h, best_c5 = local_search(best_h, best_c5)
        
        # Check for convergence and time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)

def local_search(h_current, best_c5):
    h = np.copy(h_current)
    for _ in range(10):
        # Generate perturbations in the direction of maximum overlap
        perturbation = np.zeros_like(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        perturbation[:(max_index + 1)] = 0.05
        perturbation[(max_index + 1):] = -0.05
        perturbed_h = np.clip(h + perturbation, 0, 1)
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            return perturbed_h, current_c5
    return h, best_c5
