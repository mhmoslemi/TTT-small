import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate perturbed starting points
    def perturb_start(h_initial, perturbation_range=0.05):
        perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
        perturbed_h = np.clip(h_initial + perturbation, 0, 1)
        return perturbed_h
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform multiple optimization trials
    def multi_optimize(h_initial, n_trials=10, n_perturb=5, perturbation_range=0.05):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        
        for _ in range(n_trials):
            # Generate a random starting point
            h_start = random_start()
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_start)
                if current_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = current_c5
            except:
                continue
            
            # Generate perturbations of the current best and optimize
            for _ in range(n_perturb):
                perturbed_h = perturb_start(best_h, perturbation_range)
                try:
                    optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                    if current_c5_perturbed < best_c5:
                        best_h = optimized_perturbed
                        best_c5 = current_c5_perturbed
                except:
                    continue
