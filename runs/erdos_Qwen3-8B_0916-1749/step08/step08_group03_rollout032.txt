import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    initial_h = initial_h_values  # This variable is globally defined
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial, maxiter=100, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    def refine_solution(h_current, n_refine=5):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_refine):
            h_initial = random_start()
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            for j in range(5):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                perturbed_h = np.clip(h_current + perturbation, 0, 1)
                try:
                    optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                    if current_c5_perturbed < best_c5:
                        best_c5 = current_c5_perturbed
                        best_h = optimized_perturbed
                except:
                    continue
        
        return best_h, best_c5
    
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for refine_iter in range(5):
        best_h, best_c5 = refine_solution(best_h)
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
