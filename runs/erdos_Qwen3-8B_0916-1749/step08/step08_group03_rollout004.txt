import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform local search with perturbation
    def perturb(h, perturbation_range=0.05):
        perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Define a function to perform a multi-start optimization with perturbation
    def multi_start_optimize(n_starts=5, perturbation_range=0.05, maxiter=100, ftol=1e-10, eps=1e-8):
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(n_starts):
            h_initial = random_start()
            h_opt, c5 = optimize_h(h_initial, maxiter=maxiter, ftol=ftol, eps=eps)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        
        # Local search with perturbation
        for _ in range(5):
            h_perturbed = perturb(best_h, perturbation_range)
            h_opt, c5 = optimize_h(h_perturbed, maxiter=maxiter, ftol=ftol, eps=eps)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        
        return best_h, best_c5
    
    # Initialize with the initial construction
    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)
    
    # Start the optimization process
    start_time = time.time()
    elapsed_time = 0
    
    # Perform multi-start optimization with adaptive perturbation
    for _ in range(10):
        if elapsed_time > budget_s - 1:
            break
        best_h, best_c5 = multi_start_optimize(n_starts=5, perturbation_range=0.05, maxiter=100, ftol=1e-10, eps=1e-8)
        elapsed_time = time.time() - start_time
    
    # Final refinement
    best_h, best_c5 = multi_start_optimize(n_starts=5, perturbation_range=0.05, maxiter=100, ftol=1e-10, eps=1e-8)
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
