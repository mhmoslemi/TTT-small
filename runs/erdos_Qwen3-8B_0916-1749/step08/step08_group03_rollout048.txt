import scipy.optimize as optimize
import time

    def compute_c5(h):
        h = np.array(h)

        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform coordinate descent optimization
    def coordinate_descent(h_initial, max_iter=100, ftol=1e-6, eps=1e-8):
        h = h_initial.copy()
        for _ in range(max_iter):
            for i in range(n_points):
                # Compute current value
                val = h[i]
                
                # Compute the impact of increasing or decreasing h[i]
                # We'll compute the gradient of the objective function w.r.t h[i]
                # Since the objective function is max_k (correlation), this is tricky
                # To simplify, we'll do a line search in both directions
                
                # Try increasing h[i]
                h_try = h.copy()
                h_try[i] = min(1.0, val + 1e-6)
                obj_inc = compute_c5(h_try)
                
                # Try decreasing h[i]
                h_try = h.copy()
                h_try[i] = max(0.0, val - 1e-6)
                obj_dec = compute_c5(h_try)

                if obj_dec < obj_inc:
                    h[i] = max(0.0, val - 1e-6)

                    h[i] = min(1.0, val + 1e-6)
