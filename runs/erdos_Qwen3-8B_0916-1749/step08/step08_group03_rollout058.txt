import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform local search with perturbation
    def local_search(h_current, n_perturb=10, perturb_strength=0.05):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_perturb):
            perturbation = np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5
    
    # Define a function to perform multi-start optimization
    def multi_start_optimize(n_starts=5, n_local_steps=5):
        best_h = initial_h
        best_c5 = compute_c5(initial_h)
        
        for _ in range(n_starts):
            h_initial = random_start()
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        for _ in range(n_local_steps):
            best_h, best_c5 = local_search(best_h)
        
        return best_h, best_c5
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform multi-start optimization with adaptive perturbation
    start_time = time.time()
    best_h, best_c5 = multi_start_optimize(n_starts=5, n_local_steps=10)
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    # Final refinement
    best_h, best_c5 = local_search(best_h, n_perturb=10, perturb_strength=0.1)
    
    return best_h, best_c5, len(best_h)
