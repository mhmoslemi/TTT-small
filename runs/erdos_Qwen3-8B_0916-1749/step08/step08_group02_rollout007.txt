import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_alternative_starts(h, num_starts=5):
        starts = []
        for i in range(num_starts):
            h_new = np.copy(h)
            # Randomly perturb a fraction of the points
            indices = np.random.choice(len(h_new), size=int(len(h_new)*0.2), replace=False)
            for idx in indices:
                h_new[idx] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Explore alternative starting points
    starts = generate_alternative_starts(best_h, num_starts=5)
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Try with more aggressive perturbations
    for i in range(2):
        seed_perturb = seed + i
        np.random.seed(seed_perturb)
        perturbed_h = np.copy(best_h)
        perturbed_h += np.random.normal(0, 0.1, size=len(perturbed_h))
        perturbed_h = np.clip(perturbed_h, 0, 1)
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            continue
    
    # Evolve the solution with more iterations
    for i in range(5):
        np.random.seed(seed + i)
        h = np.copy(best_h)
        h += np.random.normal(0, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        try:
            optimized_h, current_c5 = optimize_h(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
