import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    n_points = 140
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < 70:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method with Adam-like updates
    def optimize_h(h_initial, maxiter=1000, lr=0.1, tol=1e-10):
        h_bounds = [(0, 1) for _ in range(n_points)]
        h = np.array(h_initial).copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for _ in range(maxiter):
            # Compute gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Compute the contribution to the max correlation
                overlap = 0.0
                for k in range(n_points):
                    if i + k < n_points:
                        overlap += h[i] * (1 - h[i + k])
                    if i - k >= 0:
                        overlap += h[i] * (1 - h[i - k])
                # Gradient with respect to h[i]
                grad[i] = -np.sum(1 - h[i + k] for k in range(n_points) if i + k < n_points) + np.sum(1 - h[i - k] for k in range(n_points) if i - k >= 0)
            
            # Update h
            h = np.clip(h - lr * grad, 0, 1)
            
            # Keep track of the best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Check for convergence
            if np.abs(current_c5 - best_c5) < tol:
                break
        
        return best_h, best_c5
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform multiple restarts with different initializations
    def refine_solution(h_initial, n_refine=5):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        
        for _ in range(n_refine):
            # Generate a new random starting point
            h_initial = random_start()
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform refinement process with adaptive perturbation and optimization
    start_time = time.time()
    
    for refine_iter in range(5):
        best_h, best_c5 = refine_solution(best_h)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
