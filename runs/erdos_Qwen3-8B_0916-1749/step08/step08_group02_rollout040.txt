import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def structured_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate new starting points with different structures
    def generate_new_start(h_current):
        h = np.copy(h_current)
        indices = np.random.choice(len(h), size=10, replace=False)
        for idx in indices:
            h[idx] = np.random.uniform(0, 1)
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with more structured exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(15):
        # Generate new starts with different structures
        new_starts = [generate_new_start(best_h) for _ in range(5)]
        
        # Try each new start
        for start in new_starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Perturb and optimize
        for _ in range(5):
            perturbed_h = structured_perturb(best_h, perturbation_strength=0.05)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
