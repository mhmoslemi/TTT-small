import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate a structured starting point
    def structured_start():
        n = n_points
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        h[:n//2] = 1
        h[n//2:] = 0
        return h
    
    # Define a function to generate a randomized starting point
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to generate a hybrid starting point
    def hybrid_start():
        n = n_points
        grid = np.linspace(0, 2, n)
        h = np.zeros(n)
        h[:n//2] = 1
        h[n//2:] = 0
        h = np.sin(grid * np.pi / 2) * 0.5 + 0.5
        return h
    
    # Define a function to perform multi-start optimization
    def multi_start_optimize(n_starts=5, maxiter=1000, ftol=1e-10, eps=1e-8):
        best_h = structured_start()
        best_c5 = compute_c5(best_h)
        
        for _ in range(n_starts):
            h_initial = np.random.uniform(0, 1, size=n_points)
            optimized_h, current_c5 = optimize_h(h_initial, maxiter=maxiter, ftol=ftol, eps=eps)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        for _ in range(n_starts):
            h_initial = hybrid_start()
            optimized_h, current_c5 = optimize_h(h_initial, maxiter=maxiter, ftol=ftol, eps=eps)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        return best_h, best_c5
    
    # Define a function to perform adaptive perturbation
    def adaptive_perturb(h_current, n_perturb=5, perturbation_range=0.05):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_perturb):
            perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(perturbed_h, maxiter=500, ftol=1e-10, eps=1e-8)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        
        return best_h, best_c5
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform refinement process with multi-start optimization and adaptive perturbation
    start_time = time.time()

    for _ in range(3):
        best_h, best_c5 = adaptive_perturb(best_h, n_perturb=5, perturbation_range=0.05)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
