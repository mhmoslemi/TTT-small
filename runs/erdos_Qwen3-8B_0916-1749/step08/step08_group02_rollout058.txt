import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization function with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Build alternative starting points based on frequency of high/low values
    def generate_frequency_driven_starts(h, num_starts=5):
        starts = []
        for _ in range(num_starts):
            h_new = np.copy(h)
            high_indices = np.where(h_new > 0.5)[0]
            low_indices = np.where(h_new <= 0.5)[0]
            
            # Randomly flip 10% of high values to low and 10% of low values to high
            num_high_flips = max(1, int(len(high_indices) * 0.1))
            num_low_flips = max(1, int(len(low_indices) * 0.1))
            
            high_to_flip = np.random.choice(high_indices, size=num_high_flips, replace=False)
            low_to_flip = np.random.choice(low_indices, size=num_low_flips, replace=False)
            
            h_new[high_to_flip] = np.random.uniform(0, 0.5)
            h_new[low_to_flip] = np.random.uniform(0.5, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts
    
    # Main optimization loop with early stopping
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Use a multi-start approach with adaptive perturbation
    for iteration in range(20):
        # Generate alternative starting points based on frequency
        starts = generate_frequency_driven_starts(best_h)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Create a new perturbation based on the current solution
        perturbation_strength = 0.05
        perturbed_h = np.copy(best_h)
        perturbed_h += np.random.normal(0, perturbation_strength, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        try:
            optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = optimized_perturbed
        except:
            pass
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
