import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with more exploration
    def generate_new_start(h_current):
        h = np.copy(h_current)
        for i in range(len(h)):
            if np.random.rand() < 0.3:
                h[i] = np.random.uniform(0, 1)
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function for evolutionary search with guided mutation
    def evolve_solution(h_current, mutation_strength=0.05, seed=None):
        if seed is not None:
            np.random.seed(seed)
        h = np.copy(h_current)
        for i in range(len(h)):
            if np.random.rand() < 0.1:
                h[i] += np.random.normal(0, mutation_strength)
                h[i] = np.clip(h[i], 0, 1)
        return h
    
    # Main optimization loop with hybrid strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for iteration in range(15):
        # Generate a new starting point through exploration
        new_start = generate_new_start(best_h)
        optimized_h, current_c5 = optimize_h(new_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Perform guided evolution with adaptive mutation strength
        for _ in range(5):
            seed_evolve = seed + iteration * 100
            mutated_h = evolve_solution(best_h, mutation_strength=0.05, seed=seed_evolve)
            optimized_mutated, current_c5_mutated = optimize_h(mutated_h)
            if current_c5_mutated < best_c5:
                best_c5 = current_c5_mutated
                best_h = optimized_mutated
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
