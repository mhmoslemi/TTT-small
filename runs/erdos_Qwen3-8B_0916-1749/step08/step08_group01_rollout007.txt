```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Create initial pattern with alternating high and low values
    for i in range(n_points):
        if i % 5 == 0 or i % 5 == 1:
            initial_h[i] = 1.0
        elif i % 5 == 2 or i % 5 == 3:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0

    initial_h = np.clip(initial_h, 0, 1)
    initial_sum = np.sum(initial_h) * dx
    delta = (1 - initial_sum) / dx
    initial_h[0] += delta
    initial_h = np.clip(initial_h, 0, 1)

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun

    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(10):
        h_initial = np.copy(best_h)
        h_initial += np.random.normal(0, 0.05, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

        h_initial = np.copy(best_h)
        h_initial += np.random.normal(0, 0.02, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
```
