import scipy.optimize as optimize
import time

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)

    def coordinate_descent(h_initial, max_steps=1000, tolerance=1e-6):
        h = h_initial.copy()
        best_c5 = compute_c5(h)
        for step in range(max_steps):
            # Compute the current C5 value
            current_c5 = compute_c5(h)
            if current_c5 < best_c5 - tolerance:
                best_c5 = current_c5
                continue

            for i in range(n_points):

                h_new = h.copy()

                new_c5 = compute_c5(h_new)
                if new_c5 < current_c5:

                    break  # Early exit if we found an improvement

                # No improvement in this step
                break
