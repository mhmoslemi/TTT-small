import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform constrained optimization using SLSQP
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with structured perturbations
    def generate_new_start(h_current, perturbation_strength=0.1):
        h = np.copy(h_current)
        # Add random perturbation and clip
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate alternative starting points with random values
    def generate_alternative_starts(h, num_starts=5):
        starts = []
        for _ in range(num_starts):
            h_new = np.copy(h)
            # Randomly perturb 10% of the points
            indices = np.random.choice(len(h_new), size=int(len(h_new) * 0.1), replace=False)
            for idx in indices:
                h_new[idx] = np.random.uniform(0, 1)
            h_new = np.clip(h_new, 0, 1)
            starts.append(h_new)
        return starts
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(20):
        # Generate alternative starting points based on best_h
        starts = generate_alternative_starts(best_h)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue

            except:
