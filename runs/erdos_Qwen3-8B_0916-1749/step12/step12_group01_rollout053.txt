import scipy.optimize as optimize
import time

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        # Create a new starting point by mixing with a random perturbation
        random_perturbation = np.random.uniform(0, 1, size=len(h))
        h = h * 0.5 + random_perturbation * 0.5
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform random perturbation with adaptive strength
    def perturb(h, strength):
        perturbed = np.copy(h)
        perturbed += np.random.normal(0, strength, size=len(h))
        perturbed = np.clip(perturbed, 0, 1)
        return perturbed
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Start with a few rounds of optimization with different starting points
    for _ in range(10):
        # Generate a new starting point
        start = generate_new_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h

        # Perform perturbation and optimization with adaptive strength
        for strength in np.logspace(-5, -1, num=5):
