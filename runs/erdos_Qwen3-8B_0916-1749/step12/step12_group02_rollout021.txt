        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):

    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate new starting points based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        for i in range(len(h)):
            if random.random() < 0.1:
                h[i] = 0 if h[i] > 0.5 else 1
        return h
    
    # Adaptive optimization strategy
    def adaptive_optimization(h_initial):
        h = np.copy(h_initial)
        for _ in range(10):
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            optimized_h, current_c5 = optimize_h(h_perturbed)
            if current_c5 < compute_c5(h):
                h = optimized_h
        return h, compute_c5(h)

    best_h = initial_h

    for _ in range(20):

            best_c5 = current_c5
            best_h = optimized_h
