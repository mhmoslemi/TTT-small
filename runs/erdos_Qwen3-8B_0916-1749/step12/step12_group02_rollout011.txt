import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Generate a set of candidate perturbations
    def generate_candidates(h_current, num_candidates=5):
        candidates = []
        for i in range(num_candidates):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            candidate = np.clip(h_current + perturbation, 0, 1)
            candidates.append(candidate)
        return candidates
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of optimization with diverse starting points
    for i in range(50):
        # Evaluate current best
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        # Generate diverse starting points
        candidates = generate_candidates(best_h, num_candidates=10)
        
        # Perform optimization on each candidate
        for candidate in candidates:
            try:
                optimized_h, current_c5 = optimize_h(candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
