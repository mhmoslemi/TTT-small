import scipy.optimize as optimize

    def compute_c5(h):

        max_corr = np.max(corr)
        return max_corr * dx
    
    # Objective function for minimization
    def objective(h):
        return compute_c5(h)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Helper function: Create a new sample with perturbation
    def perturb_sample(h, scale):
        h = np.array(h)
        perturbed = np.clip(h + np.random.normal(0, scale, size=len(h)), 0, 1)
        return perturbed
    
    # Optimization function with multiple restarts and adaptive perturbations
    def optimize_with_restarts(h_initial, max_restarts=5, max_iterations=100):
        best_h = h_initial
        best_c5 = compute_c5(best_h)
        for restart in range(max_restarts):
            perturbation_scale = 0.1 / (1 + restart)
            new_h = perturb_sample(h_initial, perturbation_scale)
            result = optimize.minimize(
                fun=objective,
                x0=new_h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': max_iterations}
            )
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Local search with perturbations
    def local_search(h, perturbation_scale=0.05, num_perturbations=5):
        best_h = h
        best_c5 = compute_c5(best_h)
        for _ in range(num_perturbations):
            perturbation = np.random.normal(0, perturbation_scale, size=len(h))
            perturbed = np.clip(h + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed
                best_c5 = current_c5
        return best_h, best_c5

    best_h = initial_h
