import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Try different starting points based on the initial construction
    def perturb_h(h, perturbation):
        h_perturbed = np.copy(h)
        h_perturbed += np.random.uniform(-perturbation, perturbation, size=len(h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of adaptive optimization with perturbation
    for i in range(100):
        # Generate multiple starting points with different perturbations
        perturbations = [0.05, 0.1, 0.15]
        for p in perturbations:
            starts = [perturb_h(best_h, p) for _ in range(5)]
            for start in starts:
                try:
                    optimized_h, current_c5 = optimize_h(start)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                except:
                    continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
