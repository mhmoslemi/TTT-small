import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        # Create a new starting point by mixing with a random perturbation
        random_perturbation = np.random.uniform(0, 1, size=len(h))
        h = h * 0.5 + random_perturbation * 0.5
        h = np.clip(h, 0, 1)
        return h
    
    # Function to create a new construction by dividing the interval
    def divide_interval(h_current, num_intervals):
        h = np.copy(h_current)
        n_points = len(h)
        interval_size = n_points // num_intervals
        if interval_size < 1:
            interval_size = 1
        new_h = np.zeros(n_points)
        for i in range(num_intervals):
            start_idx = i * interval_size
            end_idx = (i + 1) * interval_size
            if end_idx > n_points:
                end_idx = n_points
            for j in range(start_idx, end_idx):
                new_h[j] = h[j]
        return new_h
    
    # Function to create a new construction by blending two existing solutions
    def blend_solutions(h1, h2, alpha=0.5):
        h = np.copy(h1)
        for i in range(len(h)):
            h[i] = alpha * h1[i] + (1 - alpha) * h2[i]
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform multiple rounds of optimization with different starting points
    for _ in range(5):
        # Generate multiple starting points from the current best
        starts = [generate_new_start(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Create a new construction by dividing the interval
        new_h = divide_interval(best_h, 5)
        optimized_new_h, current_c5_new = optimize_h(new_h)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = optimized_new_h
        
        # Blend with initial solution
        blended_h = blend_solutions(best_h, initial_h, alpha=0.7)
        optimized_blended_h, current_c5_blend = optimize_h(blended_h)
        if current_c5_blend < best_c5:
            best_c5 = current_c5_blend
            best_h = optimized_blended_h
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
