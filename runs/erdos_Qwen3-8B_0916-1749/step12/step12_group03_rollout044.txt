import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform global optimization with multiple starting points
    def global_optimization(start_points, max_iterations=50):
        best_h = start_points[0]
        best_c5 = compute_c5(best_h)
        for _ in range(max_iterations):
            new_starts = [generate_new_start(h) for h in start_points]
            for start in new_starts:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            start_points = [best_h] + new_starts
        return best_h, best_c5
    
    # Main optimization loop with exploration and refinement
    start_time = time.time()

    start_points = [best_h]
    for _ in range(5):
        start_points.append(generate_new_start(best_h))
    
    # Perform global optimization
    best_h, best_c5 = global_optimization(start_points)
