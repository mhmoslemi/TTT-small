import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations and advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution with informed sampling
    def generate_new_start(h_current):
        # Generate a new start by perturbing the current solution with informed sampling
        h = np.copy(h_current)
        # Add noise to high-value points and reduce noise to low-value points
        h = np.array(h)
        h = h / np.max(h) if np.max(h) > 0 else h
        h = h * np.random.uniform(0.8, 1.2)
        h = np.clip(h, 0, 1)
        return h

    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)

            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive perturbation, exploration, and convergence checking
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for i in range(50):
        # Generate alternative starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:

                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:

                best_c5 = current_c5_new

        except:
            continue
        
        # Perform local search
        best_h, best_c5 = local_search(best_h)
        
        # Check for convergence and time budget
