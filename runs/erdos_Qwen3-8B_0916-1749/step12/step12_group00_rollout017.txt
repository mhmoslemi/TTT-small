import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to perform local search with greedy perturbations
    def local_search(h_current, perturbation_strength=0.05, n_perturbations=10):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(n_perturbations):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to generate a diverse set of starting points
    def generate_starting_points(h_current, n_points=10, perturbation_strength=0.1):
        start_points = []
        for _ in range(n_points):
            perturbation = np.random.normal(0, perturbation_strength, size=len(h_current))
            start_points.append(np.clip(h_current + perturbation, 0, 1))
        return start_points
    
    # Function to perform multi-start optimization
    def multi_start_optimize(h_initial, n_starts=10):
        best_h = h_initial
        best_c5 = compute_c5(best_h)
        start_points = generate_starting_points(h_initial, n_starts=n_starts)
        for start in start_points:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        return best_h, best_c5
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform initial optimization
    best_h, best_c5 = optimize_h(best_h)
    
    # Perform local search to refine the solution
    best_h, best_c5 = local_search(best_h)
    
    # Perform multi-start optimization to escape local optima
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        best_h, best_c5 = multi_start_optimize(best_h)
        best_h, best_c5 = local_search(best_h)
    
    return best_h, best_c5, len(best_h)
