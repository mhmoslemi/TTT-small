import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a multi-start optimization with adaptive perturbations
    def multi_start_optimize(h_initial, num_starts=5, num_iterations=5):
        best_h = h_initial
        best_c5 = compute_c5(best_h)
        for _ in range(num_iterations):
            starts = [generate_new_start(best_h) for _ in range(num_starts)]
            for start in starts:
                try:
                    optimized_h, current_c5 = optimize_h(start)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                except:
                    continue
        return best_h, best_c5
    
    # Initial optimization with adaptive perturbations
    best_h, best_c5 = multi_start_optimize(initial_h)
    
    # Function to perform a more structured local search
    def local_search(h, num_steps=10, step_size=0.01):
        h = np.copy(h)
        for _ in range(num_steps):
            perturb = np.random.uniform(-step_size, step_size, size=len(h))
            h_perturb = np.clip(h + perturb, 0, 1)
            h_perturb = np.clip(h_perturb, 0, 1)
            current_c5 = compute_c5(h_perturb)
            if current_c5 < compute_c5(h):
                h = h_perturb
        return h

    start_time = time.time()
    
    # Perform additional rounds of optimization
    for _ in range(3):
        best_h, best_c5 = multi_start_optimize(best_h)
    
    # Final check for time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
