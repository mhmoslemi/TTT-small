import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a grid search over perturbation parameters
    def grid_search_perturbation(h_initial, perturbation_range, num_steps):
        best_h = h_initial
        best_c5 = compute_c5(best_h)
        for delta in np.linspace(*perturbation_range, num_steps):
            perturbed_h = h_initial + delta
            perturbed_h = np.clip(perturbed_h, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h
        return best_h, best_c5
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform adaptive optimization with perturbation and grid search
    for i in range(20):
        # Generate multiple starting points
        starts = [generate_new_start(best_h) for _ in range(5)]
        
        # Perform grid search on perturbation for each start
        for start in starts:
            try:
                # Perform grid search on perturbation parameter
                perturbation_range = (-0.1, 0.1)
                num_steps = 10
                optimized_h, current_c5 = grid_search_perturbation(start, perturbation_range, num_steps)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
