import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[0:35] = 1.0
    initial_h[35:70] = 0.0
    initial_h = np.roll(initial_h, 15)
    initial_h = np.append(initial_h[15:], initial_h[:15])
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h_with_mixed_methods(h_initial, method='SLSQP'):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        if method == 'SLSQP':
            result = optimize.minimize(
                fun=objective,
                x0=h_initial,
                bounds=h_bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
            )
        elif method == 'COBYLA':
            result = optimize.minimize(
                fun=objective,
                x0=h_initial,
                bounds=h_bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='COBYLA',
                options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
            )
        elif method == 'L-BFGS-B':
            result = optimize.minimize(
                fun=objective,
                x0=h_initial,
                bounds=h_bounds,
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='L-BFGS-B',
                options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
            )
        else:
            raise ValueError("Unsupported optimization method")
        
        return result.x, result.fun
    
    # Function to generate new starting points based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.05, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbation and exploration
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Use multiple optimization methods to avoid local optima
    methods = ['SLSQP', 'COBYLA', 'L-BFGS-B']
    
    for i in range(30):
        # Generate multiple starting points
        starts = [np.copy(best_h) for _ in range(5)]
        for start in starts:
            for method in methods:
                try:
                    optimized_h, current_c5 = optimize_h_with_mixed_methods(start, method)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = optimized_h
                except:
                    continue
        
        # Perform perturbation and optimization with adaptive strength
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbation_strength = 0.05 / (1 + i)
            perturbed_h = np.copy(best_h)
            perturbed_h += np.random.normal(0, perturbation_strength, size=len(perturbed_h))
            perturbed_h = np.clip(perturbed_h, 0, 1)
            
            for method in methods:
                try:
                    optimized_perturbed, current_c5_perturbed = optimize_h_with_mixed_methods(perturbed_h, method)
                    if current_c5_perturbed < best_c5:
                        best_c5 = current_c5_perturbed
                        best_h = optimized_perturbed
                except:
                    continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
