import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with advanced solver
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate structured perturbations
    def smart_perturb(h, perturbation_strength=0.1):
        h = np.copy(h)
        h += np.random.normal(0, perturbation_strength, size=len(h))
        h = np.clip(h, 0, 1)
        return h

    # Custom gradient descent with momentum
    def gradient_descent(h_initial, learning_rate=0.01, momentum=0.9, max_iter=1000):
        h = np.copy(h_initial)
        h_prev = np.copy(h)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Compute gradient using finite differences
                h_plus = np.copy(h)
                h_plus[i] += 1e-6
                f_plus = compute_c5(h_plus)
                h_minus = np.copy(h)
                h_minus[i] -= 1e-6
                f_minus = compute_c5(h_minus)
                grad[i] = (f_plus - f_minus) / (2e-6)
            
            # Update with momentum
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_prev = h_prev * momentum + h * (1 - momentum)
            h = h_prev
        return h, compute_c5(h)
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h_current):
        best_h = h_current
        best_c5 = compute_c5(best_h)
        for _ in range(10):
            perturbation = np.random.normal(0, 0.05, size=len(h_current))
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            current_c5 = compute_c5(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization loop with adaptive strategies
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform initial optimization
    optimized_h, optimized_c5 = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = optimized_c5
    
    # Perform gradient descent to refine the solution
    for _ in range(5):
        h_grad, c5_grad = gradient_descent(best_h, learning_rate=0.05, momentum=0.8, max_iter=100)
        if c5_grad < best_c5:
            best_h = h_grad
            best_c5 = c5_grad
    
    # Perform local search
    best_h, best_c5 = local_search(best_h)

    for _ in range(3):
        perturbed_h = smart_perturb(best_h, perturbation_strength=0.05)
        optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
        if current_c5_perturbed < best_c5:
            best_c5 = current_c5_perturbed
            best_h = optimized_perturbed
    
    # Ensure we respect the time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
