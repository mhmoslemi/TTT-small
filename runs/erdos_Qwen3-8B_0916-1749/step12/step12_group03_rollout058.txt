import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform particle swarm optimization
    def pso_optimize(h_initial, n_particles=20, max_iter=50):
        particles = np.array([np.random.uniform(0, 1, n_points) for _ in range(n_particles)])
        velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
        personal_best = particles.copy()
        global_best = particles[0].copy()
        personal_best_fitness = np.array([objective(p) for p in particles])
        global_best_fitness = np.min(personal_best_fitness)
        
        for _ in range(max_iter):
            for i in range(n_particles):
                r1, r2 = np.random.rand(2)
                cognitive = r1 * (personal_best[i] - particles[i])
                social = r2 * (global_best - particles[i])
                velocities[i] = velocities[i] * 0.8 + cognitive * 1.5 + social * 1.5
                particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
                
                # Update personal best
                fitness = objective(particles[i])
                if fitness < personal_best_fitness[i]:
                    personal_best_fitness[i] = fitness
                    personal_best[i] = particles[i].copy()
            
            # Update global best
            current_global_fitness = np.min(personal_best_fitness)
            if current_global_fitness < global_best_fitness:
                global_best_fitness = current_global_fitness
                global_best = personal_best[np.argmin(personal_best_fitness)].copy()
        
        return global_best, global_best_fitness
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of adaptive optimization with perturbation
    for i in range(50):
        # Generate multiple starting points
        starts = [generate_new_start(best_h) for _ in range(5)]
        for start in starts:
            try:
                # Use PSO for more effective exploration
                optimized_h, current_c5 = pso_optimize(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
