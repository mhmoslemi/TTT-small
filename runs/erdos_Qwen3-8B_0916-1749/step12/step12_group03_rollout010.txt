import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform coordinate descent with structured updates
    def coordinate_descent(h_initial, max_iter=100):
        h = np.copy(h_initial)
        for _ in range(max_iter):
            for i in range(len(h)):
                # Compute the current value
                current_value = h[i]
                # Compute the effect of reducing/having this value
                # Ignore the constraint for now and compute the effect
                # For simplicity, we can use a gradient-like update
                delta = 0.0
                # Compute impact on the max overlap
                h_test = np.copy(h)
                h_test[i] = 0.0
                c5_test = compute_c5(h_test)
                delta -= c5_test
                h_test[i] = 1.0
                c5_test = compute_c5(h_test)
                delta += c5_test
                delta -= compute_c5(h)
                # Update the value
                h[i] = np.clip(current_value + 0.1 * delta, 0, 1)
        return h
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Start with a few rounds of coordinate descent
    for _ in range(10):
        best_h = coordinate_descent(best_h)
        best_c5 = compute_c5(best_h)
    
    # Perform several rounds of adaptive optimization with perturbation
    for i in range(20):
        # Generate multiple starting points
        starts = [generate_new_start(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
