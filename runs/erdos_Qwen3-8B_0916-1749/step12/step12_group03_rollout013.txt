import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the overlap integral for a given shift
    def compute_c5_shifted(h, shift):
        h = np.array(h)
        h_shifted = np.roll(h, shift)
        corr = np.sum(h * (1 - h_shifted))
        return corr * dx
    
    # Define the objective function for a specific shift
    def objective(h, shift):
        return compute_c5_shifted(h, shift)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform suboptimal optimization with structured perturbations
    def optimize_h(h_initial, shifts):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=lambda h: np.max([objective(h, shift) for shift in shifts]),
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Define shifts to focus on
    shifts = [1, 2, 3, 4, 5, 6]  # Adjust based on previous analysis
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of adaptive optimization with perturbation
    for i in range(50):
        # Generate multiple starting points
        starts = [generate_new_start(best_h) for _ in range(5)]
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start, shifts)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
