import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point based on current solution
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.1, 0.1, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform local search with gradient descent
    def local_search(h_initial, learning_rate=0.01, max_iter=100):
        h = np.copy(h_initial)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Compute gradient approximation
                perturb = np.zeros_like(h)
                perturb[i] = 1e-4
                h_perturb = np.clip(h + perturb, 0, 1)
                obj_perturb = compute_c5(h_perturb)
                h_neg_perturb = np.clip(h - perturb, 0, 1)
                obj_neg_perturb = compute_c5(h_neg_perturb)
                grad[i] = (obj_perturb - obj_neg_perturb) / (2 * 1e-4)
            
            # Project gradient to satisfy the constraint
            grad_scalar = np.dot(grad, h) / (n_points / 2)
            grad = grad - 2 * grad_scalar / n_points * np.ones_like(h)
            
            # Update h
            h = np.clip(h - learning_rate * grad, 0, 1)
        
        return h, compute_c5(h)
    
    # Main optimization loop with exploration and refinement
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Perform several rounds of adaptive optimization with perturbation and local search
    for i in range(50):
        # Generate multiple starting points
        starts = [generate_new_start(best_h) for _ in range(5)]
        for start in starts:
            try:
                # Perform local search
                optimized_h, current_c5 = local_search(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
