import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform local optimization with perturbations
    def perturbate_and_optimize(h_initial, n_perturbations=5, perturbation_range=0.1):
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
        
        for _ in range(n_perturbations):
            # Generate a new random starting point
            h_initial_perturbed = np.clip(h_initial + np.random.uniform(-perturbation_range, perturbation_range, size=n_points), 0, 1)
            try:
                optimized_h, current_c5 = optimize_h(h_initial_perturbed)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5

    best_h = initial_h
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    
    # Initial optimization
    best_h, best_c5 = optimize_h(best_h)
    
    # Refinement with perturbations and local optimization
    for _ in range(5):
        best_h, best_c5 = perturbate_and_optimize(best_h)
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
