    def compute_c5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):

    # Define the optimization problem
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point with guided mutations
    def generate_new_start(h_current):
        h = np.copy(h_current)
        noise = np.random.normal(loc=0, scale=0.1, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Function to evolve a solution with guided perturbations
    def evolve_solution(h_current, seed):
        np.random.seed(seed)
        h = np.copy(h_current)
        noise = np.random.normal(loc=0, scale=0.1, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(50):
        # Generate a new starting point
        h_initial = generate_new_start(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Perform perturbation and optimization with guided steps
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbed_h = evolve_solution(optimized_h, seed_perturb)
