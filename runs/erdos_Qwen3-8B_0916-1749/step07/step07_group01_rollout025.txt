import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Try multiple starting points with random perturbations
    for i in range(50):
        # Generate multiple random starting points
        starts = []
        for _ in range(5):
            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            starts.append(h_rand)
        
        # Try each starting point
        for start in starts:
            try:
                optimized_h, current_c5 = optimize_h(start)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
