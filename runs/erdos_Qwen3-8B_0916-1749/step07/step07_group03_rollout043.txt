import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to adaptively refine the solution
    def refine_solution(h_current, n_refine=5, perturbation_range=0.1):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_refine):
            # Generate a new random starting point
            h_initial = random_start()
            
            # Perform optimization
            try:
                optimized_h, current_c5 = optimize_h(h_initial)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
            
            # Perturb the current solution and optimize
            for j in range(5):
                perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
                perturbed_h = np.clip(h_current + perturbation, 0, 1)
                try:
                    optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                    if current_c5_perturbed < best_c5:
                        best_c5 = current_c5_perturbed
                        best_h = optimized_perturbed
                except:
                    continue
        
        return best_h, best_c5
    
    # Try different starting points and select the best one
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    best_h_values = best_h
    
    # Define a function to generate a different starting point
    def generate_starting_point():
        return np.random.uniform(0, 1, size=n_points)
    
    # Try multiple starting points
    num_starting_points = 10
    start_time = time.time()
    
    for i in range(num_starting_points):
        h_initial = generate_starting_point()
        optimized_h, current_c5 = optimize_h(h_initial)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_values = optimized_h
            print(f"Improved C5 to {best_c5} at iteration {i+1}")
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Perform refinement process
    refined_h, refined_c5 = refine_solution(best_h_values)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h_values = refined_h
    
    return best_h_values, best_c5, len(best_h_values)
