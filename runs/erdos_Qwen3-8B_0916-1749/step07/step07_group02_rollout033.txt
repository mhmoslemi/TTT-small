import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with custom settings
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimization method with tighter tolerances
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a function to generate a new starting point with structured variation
    def structured_start(h_base):
        h = np.copy(h_base)
        shift = np.random.uniform(-0.1, 0.1, size=len(h))
        h += shift
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with low-frequency sinusoidal variation
    def sinusoidal_start(h_base):
        h = np.copy(h_base)
        x = np.linspace(0, 2, len(h))
        freq = np.random.uniform(0.1, 0.5)
        amplitude = np.random.uniform(0.05, 0.15)
        h += amplitude * np.sin(2 * np.pi * freq * x)
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with high-frequency noise
    def noise_start(h_base):
        h = np.copy(h_base)
        noise = np.random.normal(0, 0.05, len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with polynomial variation
    def polynomial_start(h_base):
        h = np.copy(h_base)
        x = np.linspace(0, 2, len(h))
        poly = np.polyfit(x, h, deg=2)
        h_poly = np.polyval(poly, x)
        h = (h + h_poly) / 2
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with low-frequency variation
    def low_freq_start(h_base):
        h = np.copy(h_base)
        x = np.linspace(0, 2, len(h))
        freq = np.random.uniform(0.1, 0.5)
        amplitude = np.random.uniform(0.05, 0.15)
        h += amplitude * np.cos(2 * np.pi * freq * x)
        h = np.clip(h, 0, 1)
        return h
    
    # Define a function to generate a new starting point with high-frequency variation
    def high_freq_start(h_base):
        h = np.copy(h_base)
        x = np.linspace(0, 2, len(h))
        freq = np.random.uniform(1.0, 3.0)
        amplitude = np.random.uniform(0.05, 0.15)
        h += amplitude * np.sin(2 * np.pi * freq * x)
        h = np.clip(h, 0, 1)
        return h
    
    # Multi-start optimization with a combination of starting points
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()
    
    # Use a variety of starting methods
    for _ in range(20):
        # Generate structured start
        h_initial = structured_start(best_h)
        try:

            if current_c5 < best_c5:

        except:

        # Generate polynomial start
        h_initial = polynomial_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Generate sinusoidal start
        h_initial = sinusoidal_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Generate noise start
        h_initial = noise_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Generate low frequency start
        h_initial = low_freq_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Generate high frequency start
        h_initial = high_freq_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            pass
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
