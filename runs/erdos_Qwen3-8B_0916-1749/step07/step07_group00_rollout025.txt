```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import fsolve

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 100
    initial_h = np.zeros(n_points)
    initial_h[:50] = 1.0
    initial_h[50:] = 0.0
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    def evolve_solution(h_current, seed):
        np.random.seed(seed)
        h = np.copy(h_current)
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    def generate_new_start(h_current):
        h = np.copy(h_current)
        h += np.random.uniform(-0.05, 0.05, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(30):
        h_initial = generate_new_start(best_h)
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        for j in range(5):
            seed_perturb = seed + i * 10 + j
            perturbed_h = evolve_solution(optimized_h, seed_perturb)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
```
