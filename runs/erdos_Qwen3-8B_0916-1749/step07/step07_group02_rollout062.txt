    def local_search(h_initial, max_iter=100, eps=1e-6):
        h = np.copy(h_initial)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for k in range(1, len(h) - 1):
                corr = np.correlate(h, 1 - h, mode='full')
                for i in range(len(corr)):
                    if i - k >= 0 and i - k < len(h):
                        grad[i] += 2 * (corr[i - k] - corr[i]) * (1 - 2 * h[i])
                    if i + k < len(corr) and i + k >= 0:
                        grad[i] += 2 * (corr[i + k] - corr[i]) * (1 - 2 * h[i])
            grad = grad * (1 - 2 * h) * (1 - 2 * (1 - h))
            h -= grad * eps
            h = np.clip(h, 0, 1)
        return h
