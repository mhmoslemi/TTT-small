import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define the initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization method
    def optimize_h(h_initial, maxiter=1000, ftol=1e-10, eps=1e-8):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': maxiter, 'eps': eps}
        )
        return result.x, result.fun
    
    # Define a function to generate random starting points
    def random_start():
        return np.random.uniform(0, 1, size=n_points)
    
    # Define a function to perform a local search with perturbation
    def local_search(h_current, perturbation_range=0.1, n_steps=5):
        best_h = h_current
        best_c5 = compute_c5(h_current)
        
        for _ in range(n_steps):
            perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
            perturbed_h = np.clip(h_current + perturbation, 0, 1)
            try:
                optimized_h, current_c5 = optimize_h(perturbed_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
            except:
                continue
        
        return best_h, best_c5
    
    # Define a function to perform a population-based search
    def population_search(pop_size=10, n_generations=5, mutation_rate=0.1, crossover_rate=0.7):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        best_h = population[0]
        best_c5 = compute_c5(best_h)
        
        for _ in range(n_generations):
            # Evaluate population
            fitness = [compute_c5(h) for h in population]
            
            # Select parents
            parents = [population[i] for i in np.argsort(fitness)[:int(pop_size/2)]]
            
            # Crossover
            new_population = []
            for i in range(0, len(parents), 2):
                if i+1 < len(parents):
                    child = parents[i] * (1 - crossover_rate) + parents[i+1] * crossover_rate
                    new_population.append(np.clip(child, 0, 1))
                else:
                    new_population.append(parents[i])
            
            # Mutation
            for h in new_population:
                mask = np.random.rand(n_points) < mutation_rate
                h[mask] += np.random.uniform(-0.1, 0.1, np.sum(mask))
                h = np.clip(h, 0, 1)
            
            # Replace population
            population = new_population
            
            # Update best solution
            for h in population:
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h
        
        return best_h, best_c5
    
    # Start with the initial construction
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Perform initial optimization
    optimized_h, current_c5 = optimize_h(best_h)
    best_h = optimized_h
    best_c5 = current_c5
    
    # Perform local search
    best_h, best_c5 = local_search(best_h)
    
    # Perform population-based search
    best_h, best_c5 = population_search()
    
    # Check time budget
    start_time = time.time()
    
    # Final refinement
    for _ in range(3):
        best_h, best_c5 = local_search(best_h)
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
