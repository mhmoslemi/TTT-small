import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Define initial construction using a piecewise constant function
    # Start with an initial guess that's a simple repeating pattern
    def initial_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 3:
                h[i] = 1.0
            elif i < 2 * n_points // 3:
                h[i] = 0.5
            else:
                h[i] = 0.0
        return h
    
    h_initial = initial_guess(n_points)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem
    def optimize_h(h_initial):
        h_bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate a new starting point
    def perturb_solution(h_current):
        h = np.copy(h_current)
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Function to apply a guided step
    def guided_step(h_current, step_size=0.05):
        h = np.copy(h_current)
        for i in range(n_points):
            if h[i] < 1.0:
                h[i] = np.clip(h[i] + step_size, 0, 1)
            elif h[i] > 0.0:
                h[i] = np.clip(h[i] - step_size, 0, 1)
        return h
    
    # Main optimization loop
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for i in range(50):
        # Generate a new starting point
        h_initial = perturb_solution(best_h)
        
        # Perform optimization
        try:
            optimized_h, current_c5 = optimize_h(h_initial)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
        
        # Apply guided steps
        for j in range(5):
            h_guided = guided_step(optimized_h)
            try:
                optimized_guided, current_c5_guided = optimize_h(h_guided)
                if current_c5_guided < best_c5:
                    best_c5 = current_c5_guided
                    best_h = optimized_guided
            except:
                continue
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
