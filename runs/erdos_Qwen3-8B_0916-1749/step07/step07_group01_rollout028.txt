import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Function to generate points with more even distribution
    def even_distribution(n_points):
        h = np.zeros(n_points)
        h[::2] = 1
        h[1::2] = 0.5
        return h
    
    # Function to generate points with more uniform spacing
    def uniform_spacing(n_points):
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        return h
    
    # Function to generate points with a more complex pattern
    def complex_pattern(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1
            elif i % 3 == 0:
                h[i] = 0.75
            elif i % 4 == 0:
                h[i] = 0.5
            else:
                h[i] = 0.25
        return h
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Generate alternative initial points
    starts = [
        initial_h,
        even_distribution(n_points),
        uniform_spacing(n_points),
        complex_pattern(n_points),
    ]
    
    for start in starts:
        try:
            optimized_h, current_c5 = optimize_h(start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
        except:
            continue
    
    # Evolve the solution with small perturbations
    for i in range(10):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed points
        perturbations = [
            np.random.normal(0, 0.05, n_points),
            np.random.normal(0, 0.02, n_points),
            np.random.normal(0, 0.01, n_points),
        ]
        
        for p in perturbations:
            perturbed_h = np.clip(best_h + p, 0, 1)
            try:
                optimized_perturbed, current_c5_perturbed = optimize_h(perturbed_h)
                if current_c5_perturbed < best_c5:
                    best_c5 = current_c5_perturbed
                    best_h = optimized_perturbed
            except:
                continue
    
    return best_h, best_c5, len(best_h)
