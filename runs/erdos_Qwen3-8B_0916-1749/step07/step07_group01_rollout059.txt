import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=300):
    np.random.seed(seed)
    
    # Define initial construction with 70 samples
    initial_h = initial_h_values  # This variable is globally defined
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to perform optimization with more structured perturbations
    def optimize_h(h_initial):
        h_bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=h_bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 1000, 'eps': 1e-8}
        )
        return result.x, result.fun
    
    # Define a new heuristic based on greedy descent with adaptive perturbations
    def greedy_descent(h, max_iter=100, step_size=0.1):
        h = np.copy(h)
        for _ in range(max_iter):
            # Evaluate current C5
            current_c5 = compute_c5(h)
            
            # Try perturbations in all directions
            best_perturbation = None
            best_c5 = current_c5
            
            for i in range(len(h)):
                # Try increasing h[i]
                h_inc = h.copy()
                h_inc[i] = np.clip(h_inc[i] + step_size, 0, 1)
                c5_inc = compute_c5(h_inc)
                if c5_inc < best_c5:
                    best_perturbation = h_inc
                    best_c5 = c5_inc
                
                # Try decreasing h[i]
                h_dec = h.copy()
                h_dec[i] = np.clip(h_dec[i] - step_size, 0, 1)
                c5_dec = compute_c5(h_dec)
                if c5_dec < best_c5:
                    best_perturbation = h_dec
                    best_c5 = c5_dec
            
            if best_perturbation is not None:
                h = best_perturbation
            else:
                break
        
        return h, best_c5
    
    # Define a function to generate random perturbations with static noise
    def generate_perturbation(h, noise_scale=0.05):
        h = np.copy(h)
        h += np.random.normal(0, noise_scale, size=len(h))
        h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop with adaptive perturbation and greedy descent
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Initial greedy descent with adaptive step size
    for _ in range(5):
        h, current_c5 = greedy_descent(best_h, max_iter=50, step_size=0.1)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try multiple random restarts with adaptive perturbations
    for i in range(5):
        perturbed_h = generate_perturbation(best_h, noise_scale=0.1)
        h, current_c5 = optimize_h(perturbed_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Further local search with small perturbations
    for _ in range(10):
        h, current_c5 = greedy_descent(best_h, max_iter=20, step_size=0.05)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Check if time budget is exceeded
    if time.time() - start_time > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
