def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return corr * dx, dx

def subgradient_step(h, lr, n):
    """One subgradient step for minimizing max overlap."""
    overlaps, dx = compute_overlaps(h)
    k_star = np.argmax(overlaps)
    
    # overlaps[k_star] corresponds to some shift
    # np.correlate full mode: output[k] = sum_i h[i] * (1-h[i - (k-(n-1))])
    # The shift s = k - (n-1)
    s = k_star - (n - 1)
    
    # Compute gradient of overlap at shift s
    # overlap[s] = sum_i h[i] * (1-h[i+s]) (valid i, i+s)
    # d/dh[j] = dx * (1 - (h[j+s] if valid) - (h[j-s] if valid))
    grad = np.ones(n) * dx
    for j in range(n):
        if s > 0:
            if j + s < n:
                grad[j] -= h[j+s]
            if j - s >= 0:
                grad[j] -= h[j-s]
        else:
            # s <= 0, shift is -|s|
            # overlap = sum_i h[i] * (1-h[i-s]) = sum_i h[i] * (1-h[i+|s|])
            # Same formula with |s|
            as_ = -s
            if j + as_ < n:
                grad[j] -= h[j+as_]
            if j - as_ >= 0:
                grad[j] -= h[j-as_]
    
    # Gradient descent step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    # Repair sum constraint
    h_new = repair(h_new, n)
    return h_new, overlaps[k_star]

    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        rem = diff
        for i in order:
            step = min(room[i], rem)
            if step <= 0: break
            h[i] += step
            rem -= step
            if rem <= 1e-14: break
    else:
        room = h.copy()
        order = np.argsort(-room)
        rem = -diff
        for i in order:
            step = min(room[i], rem)
            if step <= 0: break
            h[i] -= step
            rem -= step
            if rem <= 1e-14: break
    return h

def get_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    # Top active constraints
    top = min(4, len(overlaps))
    ks = np.argsort(-overlaps)[:top]
    grad = np.zeros(n)
    for ki in range(top):
        k_star = int(ks[ki])

            if n - s > 0:

            if s > 0:
                g[s:] -= h[:n-s]
        else:
            a = -s
            if n - a > 0:
                g[a:] += 1.0 - h[:n-a]
            if a > 0:
                g[:n-a] -= h[a:]
        grad += g / top
    return grad * dx, float(overlaps[int(ks[0])])

            if compute_c5(h_new) <= compute_c5(h):
                h = h_new
            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        rem = diff
        for i in order:
            step = min(room[i], rem)
            if step <= 0:
                break
            h[i] += step
            rem -= step
            if rem <= 1e-14:
                break
    else:
        room = h.copy()
        order = np.argsort(-room)
        rem = -diff
        for i in order:
            step = min(room[i], rem)
            if step <= 0:
                break
            h[i] -= step
            rem -= step
            if rem <= 1e-14:
                break
    return h

def get_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    top = min(4, len(overlaps))
    ks = np.argsort(-overlaps)[:top]
    grad = np.zeros(n)
    for ki in range(top):
        k_star = int(ks[ki])
        s = k_star - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            if n - s > 0:
                g[:n - s] += 1.0 - h[s:]
            if s > 0:
                g[s:] -= h[:n - s]
        else:
            a = -s
            if n - a > 0:
                g[a:] += 1.0 - h[:n - a]
            if a > 0:
                g[:n - a] -= h[a:]
        grad += g / top
    return grad * dx, float(overlaps[int(ks[0])])
