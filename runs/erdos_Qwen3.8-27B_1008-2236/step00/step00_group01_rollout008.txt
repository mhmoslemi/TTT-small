        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = c.max()
    active = np.where(c >= max_c - 1e-10)[0]
    grad = np.zeros(n)
    for s_idx in active:
        m = s_idx - (n - 1)
        grad += grad_c_m(h, m, dx)
    grad /= len(active)
    return grad

def optimize_pg(h, dx, max_iters=3000, time_limit=None, t_start=0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c = c5_val(h, dx)
    step = dx * 0.3
    v_mom = np.zeros(n)
    mu = 0.8

    for it in range(max_iters):
        if time_limit is not None and time.time() - t_start > time_limit - 30:
            break
        g = subgradient(h, dx)
        v_mom = mu * v_mom + g
        h_new = project_box_sum(h - step * v_mom, target)
        c_new = c5_val(h_new, dx)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            step = min(step * 1.08, 4.0 * dx)
        else:
            step *= 0.6
            if step < 1e-10:
                break
    return h, best_c

def local_search(h, dx, max_rounds=300, time_limit=None, t_start=0, seed=456):
    n = len(h)
    best_c = c5_val(h, dx)
    h = h.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time_limit is not None and time.time() - t_start > time_limit - 20:
            break
        improved = False
        n_tries = 600
        for _ in range(n_tries):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            scale_fwd = min(h[i], 1.0 - h[j])
            scale_bwd = min(1.0 - h[i], h[j])
            if scale_fwd < 1e-10 and scale_bwd < 1e-10:
                continue
            if scale_fwd >= scale_bwd:
                delta = float(rng.uniform(0.005, max(scale_fwd, 0.005)))
            else:
                delta = -float(rng.uniform(0.005, max(scale_bwd, 0.005)))
            h[i] -= delta
            h[j] += delta
            c_new = c5_val(h, dx)
            if c_new < best_c - 1e-13:
                best_c = c_new
                improved = True
            else:
                h[i] += delta
                h[j] -= delta
        if not improved:
            break
    return h, best_c

def interpolate_to_n(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = np.arange(n_old) * dx_old
    x_new = np.linspace(dx_new / 2.0, 2.0 - dx_new / 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old, left=0, right=0)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    remaining = budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Get initial solution
    best_h, best_c, best_n = None, np.inf, 0
    try:
        h_init = np.array(initial_h_values, dtype=float).flatten()
    except Exception:
        h_init = None

    if h_init is not None and len(h_init) > 0:
        n47 = len(h_init)
        dx47 = 2.0 / n47
        h47 = np.clip(h_init, 0.0, 1.0)
        h47 = project_box_sum(h47, n47 / 2.0)
        c47 = c5_val(h47, dx47)
        best_c, best_h, best_n = c47, h47.copy(), n47

        # Phase 1: Projected gradient on n=47
        h47, c47 = optimize_pg(h47, dx47, max_iters=2000, time_limit=remaining, t_start=t_start)
        if c47 < best_c:
            best_c, best_h, best_n = c47, h47.copy(), n47

        # Phase 2: Local search on n=47
        h47, c47 = local_search(best_h, dx47, max_rounds=200, time_limit=remaining, t_start=t_start)
        if c47 < best_c:
            best_c, best_h, best_n = c47, h47.copy(), n47

        # Phase 3: More PG + LS rounds
        for round_idx in range(4):
            if time.time() - t_start > remaining - 60:
                break
            h47, c47 = optimize_pg(best_h, dx47, max_iters=800, time_limit=remaining, t_start=t_start)
            c47 = max(c47, 0.0)
            h47, c47 = local_search(h47, dx47, max_rounds=100, time_limit=remaining, t_start=t_start, seed=seed + 100 + round_idx)
            if c47 < best_c:
                best_c, best_h, best_n = c47, h47.copy(), n47

    # Phase 4: Try n=100 from interpolation
    if h_init is not None:
        if time.time() - t_start < remaining - 120:
            n100 = 100
            dx100 = 2.0 / n100
            h100 = interpolate_to_n(h_init, len(h_init), n100)
            h100, c100 = optimize_pg(h100, dx100, max_iters=1500, time_limit=remaining, t_start=t_start)
            h100, c100 = local_search(h100, dx100, max_rounds=150, time_limit=remaining, t_start=t_start, seed=seed + 200)
            if c100 < best_c and time.time() - t_start < remaining - 5:
                best_c, best_h, best_n = c100, h100.copy(), n100

    # Phase 5: Random restarts at original n
    n_orig = best_n
    dx_orig = 2.0 / n_orig
    if time.time() - t_start < remaining - 90:
        for restart in range(5):
            if time.time() - t_start > remaining - 30:
                break
            pert = rng.normal(0, 0.05, size=n_orig)
            h_r = project_box_sum(best_h + pert, n_orig / 2.0)
            h_r, c_r = optimize_pg(h_r, dx_orig, max_iters=600, time_limit=remaining, t_start=t_start)
            h_r, c_r = local_search(h_r, dx_orig, max_rounds=80, time_limit=remaining, t_start=t_start, seed=seed + 50 + restart)
            if c_r < best_c:
                best_c, best_h, best_n = c_r, h_r.copy(), n_orig

    # Final verification
    final_c = c5_val(best_h, 2.0 / best_n)
    if final_c < best_c:
        best_c = final_c
    # Binarize option: try rounding
    h_bin = project_box_sum(np.round(best_h), best_n / 2.0)
    c_bin = c5_val(h_bin, 2.0 / best_n)
    if c_bin < best_c:
        best_c, best_h = c_bin, h_bin.copy()

    return best_h, best_c, best_n
