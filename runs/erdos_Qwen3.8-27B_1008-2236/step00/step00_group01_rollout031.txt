import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(h) if diff > 0 else np.argsort(-h)
            rem = diff
            for i in idx:
                if rem > 0:
                    a = min(rem, 1.0 - h[i])
                    h[i] += a; rem -= a
                elif rem < 0:
                    a = max(rem, -h[i])
                    h[i] += a; rem -= a
                if abs(rem) < 1e-12:
                    break
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(xn, xo, h_old)
    return project_h(np.clip(h_new, 0, 1), n_new)

def local_search(h0, n, n_iter, rng):
    h = h0.copy()
    best_c5 = compute_C5(h)
    best = h.copy()
    step = 0.2
    for it in range(n_iter):
        k = rng.randint(1, 4)
        idx = rng.choice(n, k, replace=False)
        h_new = best.copy()
        h_new[idx] += rng.uniform(-step, step, k)
        h_new = project_h(h_new, n)
        c5 = compute_C5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new
    return best, best_c5

def swap_search(h0, n, n_iter, rng):
    h = h0.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    # Fix count
    cnt = int(h.sum())
    need = n // 2
    if cnt > need:
        idx1 = np.where(h == 1)[0]
        for i in range(cnt - need):
            h[idx1[-1]] = 0
            idx1 = idx1[:-1]
    elif cnt < need:
        idx0 = np.where(h == 0)[0]
        for i in range(need - cnt):
            h[idx0[-1]] = 1
            idx0 = idx0[:-1]
    best = h.copy()
    best_c5 = compute_C5(best)
    T = 0.005
    for it in range(n_iter):
        T *= 0.9995
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        po = ones[rng.randint(len(ones))]
        pz = zeros[rng.randint(len(zeros))]
        h[po], h[pz] = h[pz], h[po]
        c5 = compute_C5(h)
        d = c5 - best_c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:
            h[po], h[pz] = h[pz], h[po]
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Initial from parent
    best_h, best_c5, best_n = None, float('inf'), 0
    parent = None
    if 'initial_h_values' in dir():
        try:
            parent = np.asarray(initial_h_values, dtype=float)
            best_h = parent.copy()
            best_c5 = compute_C5(parent)
            best_n = len(parent)
        except:
            pass
    
    if parent is None:
        parent = np.zeros(48)
        parent[:24] = 1.0
        best_h = parent.copy()
        best_c5 = compute_C5(parent)
        best_n = 48
    
    # Phase 1: Continuous optimization at various n
    for n_test in [48, 50, 52, 54]:
        if time.time() - t0 > budget_s * 0.5:
            break
        h_start = upsample(parent, n_test)
        for restart in range(3):
            if time.time() - t0 > budget_s * 0.5:
                break
            h_s = h_start.copy()
            if restart > 0:
                h_s += rng.uniform(-0.1, 0.1, n_test)
                h_s = project_h(h_s, n_test)
            h_opt, c5_opt = local_search(h_s, n_test, 8000, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_test
    
    # Phase 2: Binary swap search
    for n_test in [48, 50]:
        if time.time() - t0 > budget_s * 0.8:
            break
        h_bin = np.where(best_h > 0.5, 1.0, 0.0) if len(best_h) == n_test else upsample(best_h, n_test)
        h_bin = np.where(h_bin > 0.5, 1.0, 0.0)
        h_bin, c5_bin = swap_search(h_bin, n_test, 30000, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_test
            # Refine
            h_ref, c5_ref = local_search(h_bin.copy(), n_test, 5000, rng)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_test
    
    # Phase 3: Final refinement
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        n_ref = len(best_h)
        h_ref, c5_ref = local_search(best_h, n_ref, int(min(rem * 8000, 200000)), rng)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_ref
    
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_C5(best_h)
    return best_h, best_c5, len(best_h)

def subgradient_local_search(h, n, n_iter, rng):
    h = h.copy().astype(float)
    best = h.copy()
    best_c5 = compute_C5(h)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_k = int(np.argmax(corr))
        
        # Compute gradient of g_{active_k}
        # g_k = dx * sum_i h[i] * (1 - h[i+s]) where s = (n-1) - k
        s = (n - 1) - active_k
        dx = 2.0 / n
        
        grad = np.zeros(n)
        for i in range(n):
            # h[i] appears as h[i] with coefficient (1-h[i+s]) if i+s valid
            if 0 <= i + s < n:
                grad[i] += (1.0 - h[i + s])
            # h[i] appears as h[j+s] = h[i] where j = i-s, coefficient -h[j] = -h[i-s]
            if 0 <= i - s < n:
                grad[i] -= h[i - s]
        grad *= dx
        
        # Take a step
        lr = 0.05 / (0.5 + it / n_iter)
        h_new = h - lr * grad
        h_new = project_h(h_new, n)
        
        # Add small random perturbation to escape plateaus
        if rng.random() < 0.1:
            h_new += rng.uniform(-0.05, 0.05, n)
            h_new = project_h(h_new, n)
        
        c5 = compute_C5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            h = h_new.copy()
        elif rng.random() < 0.05:
            h = h_new.copy()  # Occasional random walk
    
    return best, best_c5

def compute_active_gradient(h, n, active_k):
    dx = 2.0 / n
    tau = active_k - (n - 1)
    grad = np.zeros(n)
    
    # Part 1: (1-h[m-tau]) for valid m-tau
    if tau >= 0:
        # m-tau >= 0 means m >= tau; m-tau < n means m < n+tau (always true for m < n)
        valid = np.arange(tau, n)
        grad[valid] += (1.0 - h[valid - tau])
    else:
        # m-tau >= 0 always (since -tau > 0); m-tau < n means m < n+tau
        valid = np.arange(0, n + tau)
        grad[valid] += (1.0 - h[valid - tau])
    
    # Part 2: -h[m+tau] for valid m+tau
    if tau >= 0:
        # m+tau < n means m < n-tau; m+tau >= 0 always
        valid = np.arange(0, n - tau)
        grad[valid] -= h[valid + tau]
    else:
        # m+tau >= 0 means m >= -tau; m+tau < n always (since tau < 0, m+tau < m < n)
        valid = np.arange(-tau, n)
        grad[valid] -= h[valid + tau]
    
    return grad * dx
