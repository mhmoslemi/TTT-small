import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h, n):
    if n % 2 == 0:
        num_ones = n // 2
        idx = np.argsort(h)[::-1][:num_ones]
        bh = np.zeros(n)
        bh[idx] = 1.0
    else:
        num_ones = (n - 1) // 2
        idx = np.argsort(h)[::-1]
        bh = np.zeros(n)
        bh[idx[:num_ones]] = 1.0
        bh[idx[num_ones]] = 0.5
    return bh

def get_binary_sets(h):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    return ones, zeros

def local_search(h, n, max_sweeps=80, deadline=None):
    best_c5 = compute_c5(h, n)
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        ones, zeros = get_binary_sets(h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h, n)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def sa_run(h, n, T0, T1, n_iter, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    ones, zeros = get_binary_sets(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        d = new_c5 - cur_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-12)):
            cur_c5 = new_c5
            ones, zeros = get_binary_sets(h)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def random_binary(n, seed_offset=0):
    rng = np.random.RandomState(12345 + seed_offset)
    if n % 2 == 0:
        num_ones = n // 2
        idx = rng.choice(n, num_ones, replace=False)
        bh = np.zeros(n)
        bh[idx] = 1.0
    else:
        num_ones = (n - 1) // 2
        idx = rng.choice(n, num_ones + 1, replace=False)
        bh = np.zeros(n)
        bh[idx[:num_ones]] = 1.0
        bh[idx[num_ones]] = 0.5
    return bh

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    hb = make_binary(h0, n0)
    best_h = hb.copy()
    best_c5 = compute_c5(best_h, n0)

    # Local search on initial
    h_ls, c5_ls = local_search(hb.copy(), n0, max_sweeps=50, deadline=deadline)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls.copy(), c5_ls

    # Try multiple n values and random starts
    n_candidates = sorted(set([n0, n0 - 1, n0 + 1, n0 + 2, n0 - 2, 50, 44, 46, 48]))
    n_candidates = [n for n in n_candidates if n >= 20 and n <= 200]
    n_candidates = sorted(set(n_candidates))

    attempt = 0
    while time.time() < deadline:
        attempt += 1
        # Pick an n to try
        n_try = n_candidates[attempt % len(n_candidates)]
        
        # Choose starting point
        if attempt <= len(n_candidates):
            # First pass: local search from initial (resized)

    return best_h, best_c5, len(best_h)
