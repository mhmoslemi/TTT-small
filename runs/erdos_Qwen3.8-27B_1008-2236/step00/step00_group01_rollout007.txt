def project_constrained(h, target):

        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

        else:
            hi = mid

def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    return cmax + np.log(np.sum(np.exp(beta*(corr - cmax)))) / beta

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n

        else:
            m = -k
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
