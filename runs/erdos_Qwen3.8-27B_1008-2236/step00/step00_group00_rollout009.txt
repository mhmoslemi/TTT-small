    while time.time() < t_end:
        if n < max_n and (time.time() - last_good > stall_s or
                          (time.time() - last_up > up_gap and n < max_n)):
            h = np.repeat(h, 2)
            n = len(h)
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            last_good = time.time()
            last_up = time.time()
            temp = 0.0015
            continue

        # Batch of npass iterations
        for _ in range(npass):
            if time.time() >= t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-dmax, dmax)
            a = h[i]
            b = h[j]
            h[i] = a + d
            h[j] = b - d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                nc = compute_c5(h)
                if nc < cur_c5 or rng.random() < np.exp(-(nc - cur_c5) / temp):
                    cur_c5 = nc
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                        last_good = time.time()
                else:
                    h[i] = a
                    h[j] = b
            else:
                h[i] = a
                h[j] = b
            temp *= cool
            if temp < min_temp:
                h[:] = best_h
                cur_c5 = best_c5
                temp = 0.003

        # Occasional perturbation restart if very stalled
        if time.time() - last_good > stall_s * 2.5 and n < max_n:
            h[:] = best_h
            k = len(best_h)
            for _ in range(k // 3):
                idx = rng.randint(k)
                h[idx] += rng.uniform(-0.02, 0.02)
            np.clip(h, 0.0, 1.0, out=h)
            tgt = k / 2.0
            diff = tgt - h.sum()
            if abs(diff) > 1e-14:
                h += diff / k
                np.clip(h, 0.0, 1.0, out=h)
            cur_c5 = compute_c5(h)
            temp = 0.008
            last_good = time.time()

    # Final feasibility fix on best solution
    bh = best_h.copy()
    tgt = len(bh) / 2.0
    s = bh.sum()
    if abs(s - tgt) > 1e-13:
        bh += (tgt - s) / len(bh)
    np.clip(bh, 0.0, 1.0, out=bh)
    s2 = bh.sum()
    if abs(s2 - tgt) > 1e-13:
        # Nudge one element
        idx = int(np.argmax(bh < 1.0) if np.any(bh < 1.0) else int(np.argmin(bh > 0.0)))
        bh[idx] += (tgt - s2)
        np.clip(bh, 0.0, 1.0, out=bh)

    final_c5 = compute_c5(bh)
    return bh, final_c5, len(bh)
