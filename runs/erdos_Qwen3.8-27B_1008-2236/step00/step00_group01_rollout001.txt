    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h to feasible set {0<=x<=1, sum(x)=target}"""
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        jp = i + lag
        if 0 <= jp < n:
            grad[i] = 1.0 - h[jp]
        jm = i - lag
        if 0 <= jm < n:
            grad[i] -= h[jm]
    return grad

def pdas(h0, target, max_iter, step0):

                best = h_avg.copy()
    c = compute_c5(h_avg)
    if c < best_c5:
        best_c5 = c
        best = h_avg.copy()
    return best, best_c5

def local_search(h0, target, max_rounds=100):
    """Try flipping/adjusting individual coordinates"""
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            # Try a few candidate values
            candidates = [0.0, 1.0, 1.0 - old]
            # Add current ± small
            for dv in (-0.1, -0.05, -0.01, 0.01, 0.05, 0.1):
                candidates.append(old + dv)

                    c5 = c_new
                    improved = True
                    break
                h[i] = old
                h = project_feasible(h, target)
        if not improved:

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:

def subgradient(h):
