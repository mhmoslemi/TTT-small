            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] - (-diff) / cnt, 0.0)
    return h

def upscale_h(h):
    n = len(h)
    out = np.empty(2 * n)
    out[0::2] = h
    out[1::2] = h
    return out

def make_binary_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)[:half]
    h[idx] = 1.0
    if n % 2 == 1:
        h[0] = 0.5
    return h

def anneal(h, n, rng, t_end, T_init):
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_init
    cool = 0.99997

    while time.time() < t_end:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        mt = int(rng.integers(5))
        new_h = h.copy()
        if mt == 0:
            new_h[i], new_h[j] = h[j], h[i]
        elif mt == 1:
            d = float(rng.uniform(0.03, 0.5))
            if rng.random() < 0.5:
                new_h[i] -= d
                new_h[j] += d
            else:
                new_h[i] += d
                new_h[j] -= d
            if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
                continue
        elif mt == 2:
            d = float(rng.uniform(-0.08, 0.08))
            new_h[i] += d
            if new_h[i] < 0 or new_h[i] > 1:
                continue
            comp = -d / (n - 1)
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            new_h[mask] += comp
            if np.any(new_h < -1e-9) or np.any(new_h > 1 + 1e-9):
                continue
            new_h = np.clip(new_h, 0, 1)
        elif mt == 3:
            tv = 1.0 if h[i] < 0.5 else 0.0
            d = tv - h[i]
            if abs(d) < 1e-6:
                continue
            new_h[i] = tv
            comp = -d / (n - 1)
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            new_h[mask] += comp
            if np.any(new_h < -1e-9) or np.any(new_h > 1 + 1e-9):
                continue
            new_h = np.clip(new_h, 0, 1)
        else:
            # Swap with random pair, prefer farther indices for diversity
            if rng.random() < 0.3:
                gap = int(rng.integers(1, n))
                j = (i + gap) % n
                if j == i:
                    continue
            new_h[i], new_h[j] = h[j], h[i]

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        diff = new_c5 - cur_c5
        if diff <= 0 or (T > 1e-10 and rng.random() < np.exp(-diff / T)):
            h = new_h
            cur_c5 = new_c5
        T = max(T * cool, 1e-10)

    return best_h, best_c5

def sweep(h, n, t_end):
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                h[i], h[j] = h[j], h[i]
                nc = compute_c5(h)
                if nc < cur_c5 - 1e-14:
                    cur_c5 = nc
                    improved = True
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # Get starting solution
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        h_init = make_binary_start(45, rng)
    n0 = len(h_init)
    h0 = fix_sum(h_init, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Anneal from initial (30% time)
    t_p1 = t_start + 0.30 * budget_s
    h1, c5_1 = anneal(h0.copy(), n0, rng, t_p1, 0.008)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Sweep at original n (8% time)
    t_p2 = t_start + 0.38 * budget_s
    h2, c5_2 = sweep(h1.copy(), n0, t_p2)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_1 if c5_2 > best_c5 else c5_2
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: Try binary restart + anneal (22% time)
    t_p3 = t_start + 0.60 * budget_s
    h_bin = make_binary_start(n0, rng)
    h_bin = fix_sum(h_bin, n0 / 2.0)
    h3, c5_3 = anneal(h_bin, n0, rng, t_p3, 0.010)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Upscale best to 2n and anneal (25% time)
    n2 = 2 * len(best_h)
    t_p4 = t_start + 0.85 * budget_s
    h4 = upscale_h(best_h)
    h4 = fix_sum(h4, n2 / 2.0)
    h4, c5_4 = anneal(h4, n2, rng, t_p4, 0.004)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    # Phase 5: Final sweep (remaining time)
    t_p5 = t_start + budget_s - 8
    if time.time() < t_p5:
        nf = len(best_h)
        h5, c5_5 = sweep(best_h.copy(), nf, t_p5)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Final validation
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
