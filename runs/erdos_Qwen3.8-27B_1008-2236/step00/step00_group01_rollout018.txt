    """Gradient of O(s*) where s* is the argmax shift."""
    overlaps = compute_overlaps(h)
    s_star = np.argmax(overlaps)
    # Convert to actual shift

        # dO/dh[i] = (1 - h[i+s]) for valid i+s, minus h[i-s] for valid i-s

        if 0 <= j2 < n:
            grad[i] -= h[j2]

        w = ex / ex.sum()
