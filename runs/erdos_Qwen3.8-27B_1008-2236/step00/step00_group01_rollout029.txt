def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def beatty_init(n, rng):
    """Initialize using Beatty-like sequence for good distribution."""
    h = np.zeros(n)
    alpha = rng.uniform(0.4, 0.6)
    h[np.floor(np.arange(n) * alpha + rng.uniform(0, 1, n)).astype(int) % n == np.arange(n)] = 0
    # Simpler: use fractional parts
    h = (np.floor(np.arange(n) * alpha) % 2).astype(float)
    if h.sum() > n // 2:
        idxs = np.where(h == 1)[0]
        for i in idxs[:int(h.sum() - n // 2)]:
            h[i] = 0
    elif h.sum() < n // 2:
        idxs = np.where(h == 0)[0]
        for i in idxs[:int(n // 2 - h.sum())]:
            h[i] = 1
    else:
        pass
    return h

def parent_to_binary(parent, n, rng):
    """Interpolate parent to n points, binarize with exact sum."""
    pn = len(parent)
    xp = np.arange(pn) / pn * 2.0
    xn = np.arange(n) / n * 2.0
    hc = np.interp(xn, xp, parent, period=2.0)
    thresh = np.partition(hc, n - n // 2)[n - n // 2]
    h = (hc >= thresh).astype(float)
    cnt = int(h.sum())
    while cnt > n // 2:
        idxs = np.where(h == 1)[0]
        h[idxs[0]] = 0
        cnt -= 1
    while cnt < n // 2:
        idxs = np.where(h == 0)[0]
        h[idxs[0]] = 1
        cnt += 1
    return h

            ia = ones[rng.integers(len(ones))]
            ib = zeros[rng.integers(len(zeros))]
