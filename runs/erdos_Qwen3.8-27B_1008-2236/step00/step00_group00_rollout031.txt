import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-12:
        return m
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def fix_sum(h, n):
    h = h.copy()
    for _ in range(300):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def srun(h_start, n, bounds, constraints, t_limit):
    """Run SLSQP with decreasing log-sum-exp temperature, return best (h, c5)."""
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda hh: smooth_c5(hh, temp), h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 400, 'ftol': 1e-15})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    # Unsmoothed final SLSQP
    if time.time() < t_limit:
        try:
            res = minimize(evaluate_c5, h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-15})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def pairwise_polish(h_best, n, t_limit):
    """Local pairwise-coordinate exchange to polish the solution."""
    best_c5 = evaluate_c5(h_best)
    h_work = h_best.copy()
    for rnd in range(60):
        if time.time() > t_limit:
            break
        improved = False
        # Try a subset of pairs per round for speed
        idx = np.random.permutation(n * (n - 1) // 2)
        n_try = min(len(idx), 500)
        for k in range(n_try):
            # Map flat index to (i, j) pair
            flat = idx[k]
            # Invert: find i, j such that flat = i*n - i*(i+1)//2 + (j - i - 1)
            # Simpler: use nested mapping
            i = 0
            while flat >= n - i - 1:
                flat -= (n - i - 1)
                i += 1
            j = i + 1 + flat
            # Try moving from i to j
            max_d = min(h_work[i], 1.0 - h_work[j])
            if max_d > 1e-6:
                for frac in [0.5, 0.25, 0.75]:
                    d = frac * max_d
                    h_work[i] -= d
                    h_work[j] += d
                    c5 = evaluate_c5(h_work)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        h_best = h_work.copy()
                        improved = True
                    else:
                        h_work[i] += d
                        h_work[j] -= d
                        break
            # Try reverse
            max_d2 = min(h_work[j], 1.0 - h_work[i])
            if max_d2 > 1e-6:
                for frac in [0.5, 0.25, 0.75]:
                    d = frac * max_d2
                    h_work[j] -= d
                    h_work[i] += d
                    c5 = evaluate_c5(h_work)
                    if c5 < best_c5 - 1e-13:

                        improved = True
                    else:

                        h_work[i] -= d

    return h_best, best_c5
