def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def upsamples(h, n_new):
    """Upsample h from n_old points to n_new points using step function interpolation."""
    n_old = len(h)
    # h represents step function on [0, 2] with n_old steps
    # Each step has width 2/n_old
    # New grid has n_new steps, each of width 2/n_new
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)  # centers of old steps
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)  # centers of new steps
    # Interpolate
    return np.interp(x_new, x_old, h)

def project_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    current = np.sum(h)
    if current == target:
        return h
    diff = target - current
    # Scale
    if current > 0:
        h = h + diff / current * h  # proportional adjustment
        h = np.clip(h, 0, 1)
    # Second pass: add/subtract uniformly
    current = np.sum(h)
    diff = target - current
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still not exact, fine-tune
    current = np.sum(h)
    diff = target - current
    if abs(diff) > 1e-10:
        # Distribute among non-saturated coordinates
        if diff > 0:
            candidates = np.where(h < 1 - 1e-10)[0]
        else:
            candidates = np.where(h > 1e-10)[0]
        if len(candidates) > 0:
            per = diff / len(candidates)
            h[candidates] += per
            h = np.clip(h, 0, 1)
    return h

def local_search(h, n, dx, max_iter=10000, tol=1e-8):

    return m + temperature * np.log(np.mean(np.exp((corr - m) / temperature)))

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    """Upsample step function."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def golden_init(n):
    """Initialize using golden ratio spacing for 1's positions."""
    h = np.zeros(n)
    n_ones = n // 2
    phi = (1 + np.sqrt(5)) / 2
    positions = []
    for i in range(n_ones):
        pos = int((i * (1.0 / (phi - 1.0)) ) % n)
        positions.append(pos)
    for p in positions:
        h[p] = 1.0
    # Adjust for exact count
    total = h.sum()
    needed = n_ones - total
    if needed > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:needed]] = 1.0
    elif needed < 0:
        ones = np.where(h == 1)[0]
        h[ones[-needed:]] = 0.0
    return h

def hilbert_init(n):
    """Initialize using van der Corput sequence."""
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for i in range(1, 2 * n):
        # van der Corput in base 2
        x = i
        vdc = 0.0
        d = 0.5
        while x > 0:
            vdc += d * (x & 1)
            x >>= 1
            d /= 2.0
        pos = int(vdc * n) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for p in positions:
        h[p] = 1.0
    # Ensure exact count
    while h.sum() < n_ones:
        zeros = np.where(h < 1)[0]
        h[zeros[0]] = 1.0
    while h.sum() > n_ones:
        ones = np.where(h > 0)[0]
        h[ones[0]] = 0.0
    return h

    rng = np.random.default_rng(seed)

    best_c5 = compute_c5(h, dx)
    T = step

        r = rng.random()
        new_h = h.copy()
        
        if r < 0.4:

            new_h[idx] += rng.uniform(-step, step, size=np_)

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def golden_init(n):
    h = np.zeros(n)
    n_ones = n // 2
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    seen = set()
    for i in range(n):
        pos = int((i * inv_phi) % 1.0 * n) % n
        if pos not in seen and len(seen) < n_ones:
            h[pos] = 1.0
            seen.add(pos)
    if len(seen) < n_ones:
        for i in range(n):
            if h[i] == 0 and len(seen) < n_ones:
                h[i] = 1.0
                seen.add(i)
    return project_to_constraint(h, n)

def vdc_init(n):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    for i in range(1, 4 * n):
        x = i
        vdc = 0.0
        d = 0.5
        while x > 0:
            vdc += d * (x & 1)
            x >>= 1
            d /= 2.0

            positions.append(pos)

            break

def grad_step(h, n, lr=0.005, n_steps=50):
    h = h.copy()
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = corr.max()
        active = np.where(corr > mx - 0.5)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            for j in range(n):
                g = 0.0
                ii = j - k
                if 0 <= ii < n:
                    g += (1.0 - h[ii])
                ii2 = j + k
                if 0 <= ii2 < n:
                    g += -h[ii2]
                grad[j] += g
        grad /= max(len(active), 1)
        target = n * 0.5
        grad += 1.0 * (h.sum() - target) / n
        h -= lr * grad
        h = project_to_constraint(h, n)
    return h
