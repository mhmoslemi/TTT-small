def compute_C5(h):

        h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, n, alpha, corr):
    """One subgradient step for minimax"""
    dx = 2.0 / n
    # Active set: all k where corr[k] >= max - eps
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 1e-6)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k_idx in active:
        # k_idx corresponds to lag k = k_idx - (n-1) in the full correlation
        k = k_idx - (n - 1)
        # ∂f_k/∂h[j] = (1-h[j-k]) - h[j+k] for valid indices
        if k >= 0:
            # h[j+k] exists for j < n-k
            for j in range(n - k):
                grad[j] += (1.0 - h[j - k]) - h[j + k]
        else:
            # k < 0, let k' = -k
            k_neg = -k
            for j in range(k_neg, n):
                grad[j] += (1.0 - h[j + k_neg]) - h[j - k_neg]
    
    grad /= len(active)
    h_new = h - alpha * grad
    h_new = project_h(h_new, n)
    return h_new

import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_direction(h, corr, n):
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-9)[0]
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= h[k:]
            else:
                grad[:] -= h[:]
        else:
            kp = -k
            grad[:n - kp] += (1.0 - h[kp:])
            if kp > 0:
                grad[kp:] -= h[:n - kp]
    grad /= float(len(active))
    return grad

def local_search(h, n, max_iters, rng):
    h = h.copy().astype(float)
    best_c5, _ = compute_C5(h)
    for _ in range(max_iters):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = float(rng.uniform(-0.06, 0.06))
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i] = hi
        h[j] = hj
        c5, _ = compute_C5(h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            h[i] -= delta
            h[j] += delta
    return h, best_c5

def anneal(h, n, rng, t_start, max_iters):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5, _ = compute_C5(h)
    T = t_start
    for it in range(max_iters):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = float(rng.uniform(-0.10, 0.10))
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i] = hi
        h[j] = hj
        c5, _ = compute_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        elif rng.random() > np.exp(-(c5 - best_c5) / max(T, 1e-9)):
            h[i] -= delta
            h[j] += delta
        T = max(T * 0.9997, 1e-6)
    return best_h, best_c5

def try_binary(n, rng, max_swaps):
    half = n // 2
    S_idx = np.arange(0, n, 2)[:half]
    h = np.zeros(n)
    h[S_idx] = 1.0
    c5, _ = compute_C5(h)
    best_c5 = c5
    best_h = h.copy()
    ones = list(S_idx)
    zeros = [i for i in range(n) if i not in set(S_idx.tolist())]
    for _ in range(max_swaps):
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        oi, oz = ones[i], zeros[j]
        h[oz] = 1.0
        h[oi] = 0.0
        c5, _ = compute_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            ones[i] = oz
            zeros[j] = oi
        else:
            h[oz] = 0.0
            h[oi] = 1.0
    return best_h, best_c5
