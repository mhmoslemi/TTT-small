    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    # Distribute among interior values
    interior = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
    if len(interior) == 0:
        # Pick some positions to adjust
        if diff > 0:
            idx = np.where(h < 1)[0][:max(1, int(diff)+1)]
        else:
            idx = np.where(h > 0)[0][:max(1, int(-diff)+1)]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0, 1)
        # Final fix
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            interior = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
            if len(interior) > 0:

        h = np.clip(h, 0, 1)

Each corr_k is linear in h[i] (since the correlation is bilinear, fixing all but one variable makes it linear). So f(h[i]) is the max of linear functions of h[i], which is a convex piecewise-linear function. The minimum over [0,1] is at a breakpoint or at the boundary.

corr_k = a_k * h[i] + b_k (linear in h[i])

The minimum of max_k (a_k * h[i] + b_k) over h[i] ∈ [0,1] can be found by:
- Evaluating at h[i] = 0, h[i] = 1
- Evaluating at the intersections of each pair of lines
- Taking the minimum

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def c5obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    return float(c.max())

def slsqp(h0, mi=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
    return r.x, float(r.fun)

def bls(h0, rng, tl=5.0):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    h = project_sum(h)
    b = c = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < tl:
        o = np.flatnonzero(h > 0.3)
        z = np.flatnonzero(h < 0.7)
        if len(o) == 0 or len(z) == 0:
            break
        if len(o) >= 2 and len(z) >= 2 and rng.random() < 0.4:
            i1 = int(o[rng.integers(len(o))])
            i2 = int(o[rng.integers(len(o))])
            if i1 == i2:
                i2 = int(o[rng.integers(len(o))])
            if i1 == i2:
                continue
            j1 = int(z[rng.integers(len(z))])
            j2 = int(z[rng.integers(len(z))])
            if j1 == j2:
                j2 = int(z[rng.integers(len(z))])
            if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
                continue
            h[i1], h[i2] = 0.0, 0.0
            h[j1], h[j2] = 1.0, 1.0
            nv = compute_c5(h)
            if nv <= c * 1.00001:
                c = nv
                if nv < b:
                    b = nv
            else:
                h[i1], h[i2] = 1.0, 1.0
                h[j1], h[j2] = 0.0, 0.0
        else:
            i = int(o[rng.integers(len(o))])
            j = int(z[rng.integers(len(z))])
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            nv = compute_c5(h)
            if nv <= c * 1.00001:
                c = nv
                if nv < b:
                    b = nv
            else:
                h[i], h[j] = 1.0, 0.0
    return h, b

def anl(h0, rng, tl=30.0):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    bh = h.copy()
    b = c
    t0 = time.time()
    T = 0.04
    while time.time() - t0 < tl and T > 1e-6:
        k = int(rng.integers(1, 6))
        idx = rng.choice(n, k, replace=False)
        svd = h[idx].copy()
        h[idx] = np.clip(h[idx] + rng.normal(0, 0.25, k), 0, 1)
        h = project_sum(h)
        nv = compute_c5(h)
        if nv <= c or rng.random() < np.exp(-(nv - c) / T):
            c = nv
            if nv < b:
                b = nv
                bh = h.copy()
        else:
            h[idx] = svd
            h = project_sum(h)
            c = compute_c5(h)
        T *= 0.997
    return bh, b

def coord_desc(h0, rng, tl=10.0):
    n = len(h0)
    h = h0.copy()
    b = c = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < tl:
        improved = False
        for i in range(n):
            old = h[i]
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial - old) < 1e-6:
                    continue
                hp = h.copy()
                hp[i] = trial
                hp = project_sum(hp)
                nv = compute_c5(hp)
                if nv < c * 0.9999:
                    h = hp
                    c = nv
                    if nv < b:
                        b = nv
                    improved = True
                    break
        if not improved:
            break
    return h, b

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h0 = np.asarray(globals()['initial_h_values'], dtype=float).copy()
    else:
        h0 = np.zeros(45)
        h0[0::2] = 1.0
        h0[22] = 0.5
        h0 = project_sum(h0)

    n = len(h0)
    h0 = project_sum(h0)
    bh = h0.copy()
    bc5 = compute_c5(bh)

    # Phase 1: SLSQP from initial
    if time.time() - ts < budget_s - 60:
        h, v = slsqp(bh, 800)
        h = project_sum(h)
        v = compute_c5(h)
        if v < bc5:
            bc5, bh = v, h.copy()

    # Phase 2: SLSQP with perturbations (main loop)
    no_improve = 0
    while time.time() - ts < budget_s - 250:
        if no_improve > 30:
            break
        s = 0.02 + 0.08 * rng.random() + 0.02 * no_improve * 0.1
        s = min(s, 0.3)
        hp = project_sum(bh + rng.normal(0, s, n))
        h, v = slsqp(hp, 400)
        h = project_sum(h)
        v = compute_c5(h)
        if v < bc5 - 1e-8:
            bc5, bh = v, h.copy()
            no_improve = 0
        else:
            no_improve += 1

    # Phase 3: Binary local search
    while time.time() - ts < budget_s - 120:
        hb, vb = bls(bh, rng, 4.0)
        if vb < bc5 - 1e-9:
            bc5, bh = vb, hb.copy()

    # Phase 4: Coordinate descent
    remaining = budget_s - (time.time() - ts) - 60
    if remaining > 15:
        hc, vc = coord_desc(bh, rng, min(remaining * 0.3, 40))
        if vc < bc5 - 1e-9:
            bc5, bh = vc, hc.copy()
            h, v = slsqp(bh, 500)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bc5:
                bc5, bh = v, h.copy()

    # Phase 5: Simulated annealing
    remaining = budget_s - (time.time() - ts) - 30
    if remaining > 10:
        ha, va = anl(bh, rng, min(remaining * 0.6, 100))
        if va < bc5 - 1e-9:
            bc5, bh = va, ha.copy()
            if time.time() - ts < budget_s - 15:
                h, v = slsqp(bh, 500)
                h = project_sum(h)
                v = compute_c5(h)
                if v < bc5:
                    bc5, bh = v, h.copy()

    # Phase 6: Final polish
    if time.time() - ts < budget_s - 5:
        h, v = slsqp(bh, 1000)
        h = project_sum(h)
        v = compute_c5(h)
        if v < bc5:
            bc5, bh = v, h.copy()

    bh = project_sum(bh)
    final = compute_c5(bh)
    return bh, min(final, bc5), n
