import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 45
        h0 = np.full(n, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0  # sum(h) must equal this
    
    # Ensure constraints are met
    h0 = np.clip(h0, 0, 1)
    # Adjust to meet sum constraint
    diff = target_sum - np.sum(h0)
    if abs(diff) > 1e-10:
        # Distribute evenly
        h0 += diff / n
        h0 = np.clip(h0, 0, 1)
        # Final adjustment (might not be exact after clip)
        # Use water-filling
        remaining = target_sum - np.sum(h0)
        non_clipped = (h0 > 0) & (h0 < 1)
        if np.any(non_clipped):
            h0[non_clipped] += remaining / np.sum(non_clipped)
    
    best_h = h0.copy()
    best_c5 = evaluate(best_h)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Pairwise coordinate descent
    def sweep(h, pair_order, max_iter=50):
        n = len(h)
        improved = True
        while improved and time_left() > 5:
            improved = False
            for i, j in pair_order:
                if time_left() < 1:
                    break
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_lo >= t_hi:
                    continue
                
                # Ternary search for optimal t = h[i]
                a, b = t_lo, t_hi
                for _ in range(max_iter):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if m1 >= b or m2 <= a:
                        break
                    h[i], h[j] = m1, s - m1
                    f1 = np.max(np.correlate(h, 1-h, mode='full'))
                    h[i], h[j] = m2, s - m2
                    f2 = np.max(np.correlate(h, 1-h, mode='full'))
                    if f1 < f2:

                    else:
                        a = m1

                t_opt = (a + b) / 2.0

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        free = (h >= 0.0) & (h <= 1.0)
        at_lo = (h == 0.0) & (d > 0)
        at_hi = (h == 1.0) & (d < 0)
        free = free | at_lo | at_hi
        cnt = float(np.sum(free))
        if cnt == 0:
            break
        h[free] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def opt_pair(h, i, j, iters=22):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return
    a, b = lo, hi
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = a + 2.0 * (b - a) / 3.0
        h[i] = m1
        h[j] = s - m1
        f1 = np.max(np.correlate(h, 1.0 - h, mode='full'))
        h[i] = m2
        h[j] = s - m2
        f2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if f1 < f2:
            b = m2
        else:
            a = m1
    h[i] = (a + b) * 0.5
    h[j] = s - h[i]
    if h[i] < 0:
        h[i] = 0.0
        h[j] = s
    elif h[i] > 1:
        h[i] = 1.0
        h[j] = s - 1.0

def do_sweeps(h, pairs, rng, n_sweeps):
