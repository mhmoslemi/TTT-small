            h_s = np.clip(h_s, 0, 1)
            c5_s = compute_C5(h_s)
            if c5_s < best_c5:
                best_h, best_c5, best_n = h_s, c5_s, n

    # Phase 5: Final fine annealing
    if elapsed() < 0.96:
        h, c5 = anneal(best_h, best_n, 500000, 0.008, 0.0003, rng, elapsed, 0.98)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, best_n

    return best_h, best_c5, best_n
