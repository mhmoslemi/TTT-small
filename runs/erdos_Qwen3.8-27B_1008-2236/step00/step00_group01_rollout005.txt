    dx = 2.0 / n
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    best_h = h.copy()
    
    n_shifts = 2 * n - 1
    alpha = np.ones(n_shifts) / n_shifts

        corr = np.correlate(h, 1-h, mode='full')

        grad = np.zeros(n)

                    grad[i+k] += alpha[p] * (1 - h[i])
                    grad[i] -= alpha[p] * h[i+k]

                    grad[i+j0] -= alpha[p] * h[i]

        h = project_feasible(h, n)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 1e-15:
        order = np.argsort(h)
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    elif diff < -1e-15:
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    return h

        h_bins = round_binary(best_h, best_n, 0.7)
        h_opt, c5 = opt_smooth(h_bins, best_n, [20.0, 50.0, 100.0, 300.0], max_iter=200)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, best_n
        if budget_s - elapsed() > 20:
            h_opt, c5 = opt_subgrad(best_h, best_n, 30000, lr0=0.04, momentum=0.92)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, best_n

    # Phase 4: Try different resolutions
    for n_try in [n0 + 1, n0 + 3, 60, 80, 100, 120, 160]:
        if n_try % 2 != 0:
            n_try += 1
        if budget_s - elapsed() < 30:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5 = opt_smooth(h_interp, n_try, [10.0, 30.0, 80.0, 200.0], max_iter=150)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, n_try
        if budget_s - elapsed() > 15:
            h_opt, c5 = opt_subgrad(h_opt, n_try, 20000, lr0=0.04, momentum=0.9)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n_try

    # Phase 5: Multi-start
    for trial in range(8):
        if budget_s - elapsed() < 20:
            break
        if trial == 0:
            h_start = round_binary(best_h, best_n, 1.0)
        elif trial == 1:
            h_start = 1.0 - best_h
            h_start = project_feasible(h_start, best_n)
        else:
            h_start = best_h + np.random.randn(best_n) * 0.1
            h_start = project_feasible(h_start, best_n)
        h_opt, c5 = opt_smooth(h_start, best_n, [10.0, 40.0, 150.0], max_iter=150)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, best_n
        if budget_s - elapsed() > 15 and c5 < best_c5 + 0.002:
            h_opt, c5 = opt_subgrad(h_opt, best_n, 20000, lr0=0.03, momentum=0.92)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, best_n

    # Phase 6: Final fine polish
    if budget_s - elapsed() > 15:
        T_final = [100.0, 300.0, 500.0, 800.0]
        h_opt, c5 = opt_smooth(best_h, best_n, T_final, max_iter=300)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, best_n
        if budget_s - elapsed() > 10:
            h_opt, c5 = opt_subgrad(best_h, best_n, 40000, lr0=0.02, momentum=0.95)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, best_n

    # Ensure valid output
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
