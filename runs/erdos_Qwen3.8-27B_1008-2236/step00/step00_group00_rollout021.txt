import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def get_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def make_01_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
        h[idx[n//2]] = 0.5
    return h

def local_search(h, target_sum, rng, deadline, max_iters=200000):
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    stuck = 0
    steps = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        improved = False
        perm = rng.permutation(len(steps))
        for pi in perm:
            delta = steps[pi]
            h_new = h.copy()
            val = h_new[i] + delta
            if val < 0 or val > 1:
                continue
            h_new[i] = val
            h_new = project_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                c5 = c5_new
                improved = True
                break
        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                pert = 0.03 * stuck / 500.0
                h = h + rng.uniform(-pert, pert, n)
                h = project_sum(h, target_sum)
                c5 = compute_c5(h)
                stuck = 0
    return h, c5

def targeted_search(h, target_sum, rng, deadline, max_rounds=5000):
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        corr = get_corr_scaled(h)
        c5 = np.max(corr)
        top_k = np.where(corr > c5 - 1e-4)[0]
        
        improved = False
        for lag in top_k[:5]:
            shift = lag - (n - 1)
            if shift >= 0:
                pairs = [(j, j + shift) for j in range(n - shift)
                         if h[j] > 0.01 and (1 - h[j + shift]) > 0.01]
            else:
                shift_abs = -shift
                pairs = [(j + shift_abs, j) for j in range(n - shift_abs)
                         if h[j + shift_abs] > 0.01 and (1 - h[j]) > 0.01]
            
            if not pairs:
                continue
            j1, j2 = pairs[rng.randint(len(pairs))]
            # Try reducing the product h[j1]*(1-h[j2])
            for dj1, dj2 in [(-0.02, 0.02), (0.02, -0.02), (-0.01, 0.01), (-0.05, 0.05)]:
                h_new = h.copy()
                h_new[j1] = np.clip(h_new[j1] + dj1, 0, 1)
                h_new[j2] = np.clip(h_new[j2] + dj2, 0, 1)
                h_new = project_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-12:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
            if improved:
                break
    return h, c5

def nm_obj(x, target_sum):
    h = project_sum(x, target_sum)
    return compute_c5(h)

def nm_obj_grad(x, target_sum):
    h = project_sum(x, target_sum)
    corr = np.correlate(h, 1 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    c5 = np.max(corr) * dx
    # Subgradient: indicator of max lag
    k_max = np.argmax(corr)
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        for j in range(n - shift):
            grad[j] += (1 - h[j + shift]) * dx
            if j + shift < n:
                grad[j + shift] += h[j] * (-1) * dx
    else:
        sa = -shift
        for j in range(n - sa):
            grad[j + sa] += (1 - h[j]) * dx
            grad[j] += h[j + sa] * (-1) * dx
    return c5, grad

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h)
