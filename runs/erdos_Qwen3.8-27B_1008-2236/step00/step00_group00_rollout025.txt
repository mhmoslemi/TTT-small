        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new <= c5 or rng.random() < np.exp(-(c5_new - c5) / max(T, 1e-12)):
            c5 = c5_new
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= 0.99995
    return h_best, c5_best

def perturb_and_refine(h, rng, n_perturb, start_time, time_budget, Tmax=0.2):
    h2 = h.copy()
    for _ in range(n_perturb):
        ones = np.where(h2 == 1)[0]
        zeros = np.where(h2 == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h2[i], h2[j] = 0.0, 1.0
    return refine_sa(h2, rng, time_budget, start_time, Tmax)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, np.inf, 0

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        best_c5 = compute_c5(h0)
        best_h, best_n = h0.copy(), len(h0)
    except Exception:
        pass

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    n_list = [44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 96, 112, 128, 160, 200]

    for n in n_list:
        if rem() < 10:
            break
        m = n // 2
        n_trials = 4 if rem() > 300 else (2 if rem() > 60 else 1)

        for trial in range(n_trials):
            if rem() < 4:
                break
            # Initialize
            if trial == 0:
                h = np.zeros(n)
                h[:m] = 1.0
            elif trial == 1:
                h = np.zeros(n)
                h[rng.choice(n, m, replace=False)] = 1.0
            elif trial == 2:
                h = (np.arange(n) % 3 <= 0).astype(float)
                h[::2] = 1.0
                s = int(h.sum())
                if s > m:
                    idx = np.where(h == 1)[0]
                    h[idx[rng.choice(s, s - m, replace=False)]] = 0.0
                elif s < m:
                    idx = np.where(h == 0)[0]
                    h[idx[rng.choice(len(idx), m - s, replace=False)]] = 1.0
            else:
                if best_h is not None and rem() > 30:
                    h_interp = np.interp(np.linspace(0, n - 1, n),
                                         np.linspace(0, len(best_h) - 1, len(best_h)), best_h)
                    h = (h_interp >= 0.5).astype(float)
                    s = int(h.sum())
                    if s > m:
                        idx = np.where(h == 1)[0]
                        h[idx[rng.choice(s, s - m, replace=False)]] = 0.0
                    elif s < m:
                        idx = np.where(h == 0)[0]
                        h[idx[rng.choice(len(idx), m - s, replace=False)]] = 1.0
                else:
                    h = np.zeros(n)
                    h[rng.choice(n, m, replace=False)] = 1.0

            per_time = rem() * 0.68 / (n_trials - trial)
            per_time = min(per_time, 50.0)

            if n <= 60 and per_time > 3:
                h, c5 = steepest_descent(h, max_passes=80, start_time=t0, time_budget=per_time * 0.3)
                h, c5 = refine_sa(h, rng, per_time * 0.7 - 1.0, t0)
            else:
                h, c5 = refine_sa(h, rng, per_time * 0.7, t0, Tmax=0.5)

            if c5 < best_c5:
                best_c5 = c5
                best_h, best_n = h.copy(), n

            # Memetic: perturb best and try to improve
            if best_n == n and rem() > 20 and per_time > 8:
                h_p, c5_p = perturb_and_refine(best_h, rng, rng.randint(2, 6), t0, min(15.0, rem() * 0.3))
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Continuous refinement on the best binary solution
    if best_h is not None and rem() > 8:
        n = best_n
        hc = best_h.astype(float).copy()
        c5_before = best_c5

        def obj_fn(x):
            x = np.clip(x, 0.0, 1.0)
            return compute_c5(x) + 200.0 * (np.sum(x) - n / 2.0) ** 2

        try:
            res = minimize(obj_fn, hc, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12, 'adapt': True})
            hc2 = np.clip(res.x, 0.0, 1.0)
            if abs(np.sum(hc2) - n / 2.0) < 0.02:
                c52 = compute_c5(hc2)
                if c52 < best_c5:
                    best_c5, best_h = c52, hc2
        except Exception:
            pass

        # Also try COBYLA
        if rem() > 15:
            try:
                res2 = minimize(obj_fn, hc, method='COBYLA',
                               options={'maxiter': 3000, 'rhobeg': 0.05})
                hc3 = np.clip(res2.x, 0.0, 1.0)
                if abs(np.sum(hc3) - n / 2.0) < 0.02:
                    c53 = compute_c5(hc3)
                    if c53 < best_c5:
                        best_c5, best_h = c53, hc3
            except Exception:
                pass

    if best_h is None:
        best_h = np.zeros(44)
        best_h[:22] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = 44

    return (best_h, best_c5, best_n)
