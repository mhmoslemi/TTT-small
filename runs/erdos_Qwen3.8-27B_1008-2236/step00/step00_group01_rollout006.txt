def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h, top_k=3):
    """Compute subgradient using top-k active constraints."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top-k indices
    top_idx = np.argsort(corr)[::-1][:top_k]
    
    # Average gradient of top-k
    grad = np.zeros(n)
    for k in top_idx:
        m = int(k) - (n - 1)
        for i in range(n):
            g = 0.0
            ip = i + m
            if 0 <= ip < n:
                g += 1.0 - h[ip]
            im = i - m
            if 0 <= im < n:
                g -= h[im]
            grad[i] += g
    grad /= top_k
    return grad * dx, float(np.max(corr) * dx), top_idx

    else:
        mask = h > 0.0
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h, nk=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[::-1][:nk]
    grad = np.zeros(n)
    for ki in top:
        m = int(ki) - (n - 1)
        for i in range(n):
            ip = i + m
            if 0 <= ip < n:
                grad[i] += 1.0 - h[ip]
            im = i - m
            if 0 <= im < n:
                grad[i] -= h[im]
    grad /= nk
    return grad * dx, float(np.max(corr) * dx)

def project_h(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        m = h < 1.0
    else:
        m = h > 0.0
    c = m.sum()
    if c > 0:
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(beta * (corr - corr.max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        m = k - (n - 1)
        for i in range(n):
            ip = i + m
            if 0 <= ip < n:
                grad[i] += weights[k] * (1.0 - h[ip])
            im = i - m
            if 0 <= im < n:
                grad[i] -= weights[k] * h[im]
    return grad * dx, float(np.max(corr) * dx)

def smooth_c5_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)

        wk = w[k]
        if wk < 1e-12:
            continue

            im = i - m
            if 0 <= im < n:
                grad[i] -= wk * h[im]
