import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            m = room > 1e-15
            if not m.any():
                break
            h[m] = np.minimum(1.0, h[m] + d / m.sum())
        else:
            m = h > 1e-15
            if not m.any():
                break
            h[m] = np.maximum(0.0, h[m] + d / m.sum())
    return h
