        ac = np.correlate(best_h, best_h, mode='full')
        s_star = int(np.argmin(ac))  # index of minimum autocorrelation
        # Convert to shift: actual shift = s_star - (n-1)
        shift = s_star - (n - 1)
        
        # Subgradient: direction[j] = best_h[j-shift] + best_h[j+shift] (with bounds)
        direction = np.zeros(n)

            if 0 <= j - shift < n:
                direction[j] += best_h[j - shift]
            if 0 <= j + shift < n:

        direction -= direction.mean()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)
    cnt = mask.sum()
    if cnt == 0:
        mask = np.ones(len(h), dtype=bool)
        cnt = len(h)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def grad_at_max(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(max(0, s), n - s) if n - s > s else np.array([], dtype=int)
        # j-s in [0,n-1] and j+s in [0,n-1]
        lo = max(0, s)
        hi = min(n, n - s)
        if hi > lo:
            js = np.arange(lo, hi)
            grad[js] += (1.0 - h[js + s])
            grad[js] -= h[js - s]
        # Only j+s valid: j in [0, s)
        if s > 0:
            lo2 = 0
            hi2 = min(s, n)
            if hi2 > lo2:
                js2 = np.arange(lo2, hi2)
                grad[js2] += (1.0 - h[js2 + s])
        # Only j-s valid: j in [n-s, n)
        if s > 0:
            lo3 = n - s
            hi3 = n
            if hi3 > lo3:
                js3 = np.arange(lo3, hi3)
                grad[js3] -= h[js3 - s]
    else:
        t = -s
        # j+s in [0,n-1]: j >= t; j-s in [0,n-1]: j < n-t
        lo = t
        hi = n - t
        if hi > lo:
            js = np.arange(lo, hi)
            grad[js] += (1.0 - h[js + s])
            grad[js] -= h[js - s]
        # Only j+s valid (j >= t):
        if t > 0:
            lo2 = t
            hi2 = n
            if hi2 > lo2:
                js2 = np.arange(lo2, hi2)
                # j+s = j-t, valid since j>=t
                # j-s = j+t, valid only if j+t < n, i.e., j < n-t
                # In this range j>=t and j<n, but j-(negative s)=j+t may exceed n-1 for j>=n-t
                # Already handled in main block, so nothing extra
                pass
        # Only j-s valid (j < n-t):
        if t > 0:
            lo3 = 0
            hi3 = n - t
            if hi3 > lo3:
                js3 = np.arange(lo3, hi3)
                pass  # already in main block
    return grad, c, k_star

def local_optimize(h, target_sum, n_steps=5000, seed=42, time_limit=20.0):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    step_size = 0.012
    no_impr = 0

    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        grad, c, _ = grad_at_max(best_h)
        direction = -grad
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-12:
            direction = rng.randn(n)
            direction -= direction.mean()
        new_h = best_h + step_size * direction / max(norm, 1e-6) * 0.5
        new_h = project(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-14:
            best_h = new_h
            best_c5 = new_c5
            no_impr = 0
            step_size = min(step_size * 1.03, 0.03)
        else:
            no_impr += 1
            # Coordinate perturbation
            if no_impr % 3 == 0:
                i, j = rng.choice(n, size=2, replace=False)
                d = rng.uniform(-0.03, 0.03)
                new_h = best_h.copy()
                new_h[i] += d
                new_h[j] -= d
                new_h = project(new_h, target_sum)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    no_impr = 0
            # Random kick
            elif no_impr > 200 and rng.random() < 0.05:
                delta = rng.randn(n) * 0.015
                delta -= delta.mean()
                new_h = project(best_h + delta, target_sum)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    no_impr = 0
            if no_impr % 100 == 0:
                step_size = max(step_size * 0.85, 0.002)
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    # Initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)

        h0 = np.full(45, 0.5)
    n0 = len(h0)

    best_c5 = compute_c5(best_h)
    n_best = n0

    def elapsed():
        return time.time() - t_start

    def remaining_budget():
        return max(0, budget_s - elapsed() - 8)

    # Phase 1: Main optimization at original n
    t1 = min(80, remaining_budget() * 0.2)

        if c5 < best_c5:
            best_h, best_c5, n_best = h_opt, c5, n0

    # Phase 2: Multiple restarts
    n_restarts = int(min(remaining_budget() // 12, 15))
    n_restarts = max(1, n_restarts)

    for r in range(n_restarts):
        if remaining_budget() < 10:

        n = n0

        if typ == 0:

            hi = np.zeros(n)
            freq = 2 + (r % 4)

        elif typ == 2:

            mid = n // 2

        else:

            n = max(20, min(n, 120))
            hi = best_h.copy()
            if n != n0:
                hi = resample(best_h, n)
        hi = project(hi, n / 2.0)
        t_lim = min(12, remaining_budget() * 0.5 / max(1, n_restarts - r))
        h_opt, c5 = local_optimize(hi, n / 2.0, 4000, seed + r * 7919, t_lim)
        if c5 < best_c5:
            best_h, best_c5, n_best = h_opt, c5, n

    # Phase 3: Fine-tune best
    if remaining_budget() > 15:
        h_opt, c5 = local_optimize(best_h, n_best / 2.0, 12000, seed + 4242, remaining_budget() - 5)
        if c5 < best_c5:

    if remaining_budget() > 10:

        target = n / 2.0
        rng2 = np.random.RandomState(seed + 777)
        t_lim = remaining_budget() - 3

            if time.time() - t0 > t_lim:

            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best_h = new_h
                best_c5 = c5_new

    best_h = project(best_h, n_best / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
