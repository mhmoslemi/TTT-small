def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = np.max(corr)
    tol = 1e-8
    active = corr >= max_val - tol
    n_active = float(np.sum(active))
    if n_active == 0:
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
        n_active = 1.0
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if not active[j]:
            continue
        s = j - (n - 1)
        if s >= 0:
            for m in range(s, n):
                grad[m] += (1 - h[m - s])
            for m in range(n - s):
                grad[m] -= h[m + s]
        else:
            sp = -s
            for m in range(n - sp):
                grad[m] += (1 - h[m + sp])
            for m in range(sp, n):
                grad[m] -= h[m - sp]
    grad /= n_active
    return grad * dx

            cnt = np.sum(mask)
            if cnt == 0:

            h[mask] += diff / cnt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    max_val = np.max(corr)
    active = corr >= (max_val - 1e-12)
    n_act = float(np.sum(active))
    if n_act == 0:
        active = np.zeros(2 * n - 1, dtype=bool)
        active[int(np.argmax(corr))] = True
        n_act = 1.0
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if not active[j]:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:n] += (1.0 - h[0:n - s])
            grad[0:n - s] -= h[s:n]
        else:
            p = -s
            grad[0:n - p] += (1.0 - h[p:n])
            grad[p:n] -= h[0:n - p]
    return grad / n_act

def smooth_fun(x, beta, n):
    x_c = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr_raw = np.correlate(x_c, 1.0 - x_c, mode="full")
    corr = corr_raw * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    e = np.exp(shifted)
    se = np.sum(e)
    weights = e / se
    value = max_c + np.log(se) / beta
    target = n / 2.0
    pen_val = 50.0 * (x_c.sum() - target) ** 2
    value += pen_val
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-15:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:n] += w * (1.0 - x_c[0:n - s])
            grad[0:n - s] -= w * x_c[s:n]
        else:
            p = -s
            grad[0:n - p] += w * (1.0 - x_c[p:n])
            grad[p:n] -= w * x_c[0:n - p]
    pen_grad = 100.0 * (x_c.sum() - target) / n * np.ones(n)
    return value, grad + pen_grad

def project_constraint(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            c = mask.sum()
            if c == 0:
                break
            h[mask] += diff / c
        else:
            mask = h > 0.0
            c = mask.sum()
            if c == 0:
                break
            h[mask] += diff / c
    return np.clip(h, 0.0, 1.0)

def nm_obj(x, n):
    x = np.clip(x, 0.0, 1.0)
    c5 = compute_c5(x)
    pen = 200.0 * abs(x.sum() - n / 2.0)
    return c5 + pen

def run_subgrad(h, n, lr, max_iter, t_start, budget_s, margin):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    for it in range(max_iter):
        if time.time() - t_start > budget_s - margin:
            break
        grad = subgradient_c5(h)
        h = h - lr * grad
        h = project_constraint(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 199:
            lr = max(0.005, lr * 0.82)
    return best_h, best_c5
