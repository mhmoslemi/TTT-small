def _compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def _project(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on lambda."""
    lo, hi = -2.0, 2.0
    for _ in range(60):

        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    # corr[m] = sum_i h[i] * (1-h[i-k]) where k = m-(n-1)
    # d corr[m] / d h[p] = (1-h[p-k]) - h[p+k]  (with boundary terms)

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def _proj(h, s):
    """Project onto {x: sum=x=s, 0<=x<=1}"""
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > s:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    d = s - x.sum()
    if abs(d) > 1e-12:
        f = (x > 1e-10) & (x < 1.0 - 1e-10)
        if f.any():
            x[f] += d / f.sum()
            np.clip(x, 0, 1, out=x)
    return x

def _subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = corr.max()
    act = np.where(corr >= mv - 1e-6)[0]
    grad = np.zeros(n)
    for m in act:
        k = int(m - (n - 1))
        g = np.ones(n)
        K = abs(k)
        if K > 0:
            g[K:] -= h[:n-K]
            g[:n-K] -= h[K:]
        else:
            g -= 2.0 * h
        grad += g
    return grad / len(act), mv

def _local_search_binary(h0, n, tl):
    """Binary local search: try all single flips (need pair swaps for fixed sum)."""
    # Round to binary, keeping one fractional if needed
    h = h0.copy()
    s = n / 2.0
    # Make it as binary as possible
    h = np.where(h >= 0.5, 1.0, 0.0)
    # Fix sum
    while h.sum() > s:
        idx = np.argmin(h)  # should find a 1 to flip
        h[idx] = 0.0
    while h.sum() < s:
        idx = np.argmax(h)
        h[idx] = 1.0
    
    cur = _c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    while time.time() < tl:
        improved = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nc = _c5(h)
                if nc < cur - 1e-12:
                    cur = nc
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
        it += 1
    return h, cur

    else:

    best_h = None

            x_old = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n)
            f = interp1d(x_old, h_init, kind='linear', fill_value=0, bounds_error=False)
            candidates.append(np.clip(f(x_new), 0, 1))

        A = np.zeros((n, M))
        for m in range(M):
            k = m - (n - 1)
            for p in range(n):
                val = 0.0
                if 0 <= p - k < n:
                    val += (1.0 - h[p - k])
                if 0 <= p + k < n:
                    val -= h[p + k]

                    break
