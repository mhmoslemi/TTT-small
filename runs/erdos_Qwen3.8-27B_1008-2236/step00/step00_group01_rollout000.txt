    """Project h to satisfy sum(h)*dx = 1 and 0 <= h <= 1."""
    h = h.copy()

    h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    current_sum = np.sum(h)
    if current_sum > 0 and current_sum < n:
        # Distribute the difference
        diff = target - current_sum
        # Only adjust elements that aren't at bounds
        # Simple approach: adjust all proportionally
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

        for _ in range(20):

            if np.any(free):
                h[free] += diff / np.sum(free)
            else:
                break

import numpy as np
from scipy.optimize import minimize
import time

def c5_max(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def c5_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_nelder(h_flat, n, dx):
    h = project(h_flat, n)
    return c5_max(h, dx)

def coord_refine(h, n, dx, best_c5, n_passes=5, n_steps=31):
    deltas = np.linspace(-0.05, 0.05, n_steps)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for d in deltas:
                h[i] += d
                if h[i] < 0 or h[i] > 1:
                    h[i] -= d
                    continue
                h = project(h, n)
                c = c5_max(h, dx)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                    h[i] += 0  # keep updated value
                else:
                    h[i] -= d
                    h[i] = np.clip(h[i], 0, 1)
                    h = project(h, n)
        if not improved:
            break
    return h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old + 1)
    h_ext = np.concatenate([h, [h[-1]]])
    x_new = np.linspace(0, 2.0, n_new + 1)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        idx = np.searchsorted(x_old, x_new[i]) - 1
        idx = np.clip(idx, 0, n_old - 1)
        h_new[i] = h_ext[idx]
    return h_new
