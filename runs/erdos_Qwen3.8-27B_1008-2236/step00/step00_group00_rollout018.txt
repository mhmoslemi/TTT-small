The objective is f(h) = max_l L(h, l) where L(h,l) = dx * sum_i h[i] * (1-h[i+l])

For a fixed lag l, L is a quadratic in h. The gradient of L w.r.t. h[j] is:
∂L/∂h[j] = dx * [∂/∂h[j] sum_i h[i](1-h[i+l])]

For the sum S = sum_i h[i](1-h[i+l]):
- When h[j] appears as h[i] (i.e., i=j): contributes (1-h[j+l])
- When h[j] appears as h[i+l] (i.e., i = j-l): contributes -h[j-l]

So ∂S/∂h[j] = (1-h[j+l]) * 1[j+l ∈ [0,n-1]] - h[j-l] * 1[j-l ∈ [0,n-1]]

And ∂L/∂h[j] = dx * ∂S/∂h[j]

For the max, the subgradient is the gradient of the active (maximizing) lag.

To minimize f, we take a step in the direction of -∂L/∂h for the active lag.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(y, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find best single element to adjust
        idx = int(np.argmin(np.abs(np.where(h > 0.01, np.where(h < 0.99, 0.5, 2.0), 2.0))))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

            best_h = h.copy()

        valid_plus = (np.arange(n) + L >= 0) & (np.arange(n) + L < n)
        valid_minus = (np.arange(n) - L >= 0) & (np.arange(n) - L < n)
        grad[valid_plus] += (1.0 - h[np.arange(n)[valid_plus] + L])
        grad[valid_minus] -= h[np.arange(n)[valid_minus] - L]

        j = rng.randint(n)

        val = compute_c5(h)

            best_h = h.copy()

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(y, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def get_lags(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def subgrad_optimize(h0, t_start, deadline, seed, max_iter=50000):
    rng = np.random.RandomState(seed)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.astype(float).copy()
    best_h = h.copy()

            break

            g[vm] -= h[idx[vm] - L]

            break
        lg = _h_lags(h)
        cv = float(np.max(lg))
        if cv < bv - 1e-11:
            bv, bh, stk = cv, h.copy(), 0
        else:
            stk += 1
        if stk > 40000 and lr < 0.002:
            break
        K = min(4, n)
        ti = np.argsort(lg)[::-1][:K]
        w = lg[ti].copy()
        w /= w.sum()
        gd = np.zeros(n)
        for ww, li in zip(w, ti):
            L = int(li) - (n - 1)
            g = np.zeros(n)
            vm = (idx - L >= 0) & (idx - L < n)
            vp = (idx + L >= 0) & (idx + L < n)
            g[vm] = 1.0 - h[idx[vm] - L]
            g[vp] -= h[idx[vp] + L]
            gd += ww * g
        gd *= dx
        hn = _h_proj(h - lr * gd, ts)
        nv = _h_c5(hn)
        if nv <= cv + 1e-11:
            h = hn
            if stk > 800 and lr < 0.4:
                lr *= 1.04
        else:
            lr *= 0.82
            if lr < 0.003:
                h = _h_proj(h + 0.006 * rng.randn(n), ts)
                lr, stk = 0.05, 0
        if stk > 15000:
            h = _h_proj(bh + 0.01 * rng.randn(n), ts)
            lr, stk = 0.08, 0
    return bh, bv

def _h_polish(h, ts, nx, seed, deadline):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.astype(float).copy()
    bh, bv = h.copy(), _h_c5(h)
    for _ in range(nx):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        d = 0.02 * rng.choice([1.0, 2.0, 3.0, 5.0])
        s = 1.0 if rng.random() < 0.5 else -1.0
        h[i] += d * s
        h[j] -= d * s
        h = np.clip(h, 0.0, 1.0)
        d2 = ts - np.sum(h)
        if abs(d2) > 1e-12:
            h = _h_proj(h, ts)
        v = _h_c5(h)
        if v < bv - 1e-12:
            bv, bh = v, h.copy()
    return bh, bv

def _h_bin(n, rng):
    h = np.zeros(n)
    nk = int(n / 2)
    fr = n / 2.0 - nk
    pos = rng.choice(n, nk + 1, replace=False)
    h[pos[:nk]] = 1.0
    if fr > 0:
        h[pos[nk]] = fr
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    bh, bv, bn = None, 1e10, 0
    try:
        hi = np.array(initial_h_values, dtype=float)
        n0 = len(hi)
        h1, v1 = _h_subgrad(hi, dl, seed, 50000)
        if v1 < bv:
            bv, bh, bn = v1, h1, n0
    except Exception:
        pass
    if bh is None:
        h1, v1 = _h_subgrad(_h_bin(45, rng), dl, seed, 30000)
        bv, bh, bn = v1, h1, 45
    n45 = 45
    ts45 = n45 / 2.0
    for t in range(50):
        if time.time() > dl:
            break
        h0 = _h_bin(n45, rng)
        h0 += 0.015 * rng.randn(n45)
        h0 = _h_proj(h0, ts45)
        ho, vo = _h_subgrad(h0, dl, seed + 100 + t, 10000)
        if vo < bv:
            bv, bh, bn = vo, ho, n45
    if time.time() < dl - 5:
        hp, vp = _h_polish(bh, bn / 2.0, 10000, seed + 500, dl)
        if vp < bv:
            bv, bh = vp, hp
    for nt in [50, 60, 90]:
        if time.time() > dl - 25:
            break
        ts_t = nt / 2.0
        if len(bh) != nt:
            xo = np.linspace(0.0, 2.0, len(bh))
            xn = np.linspace(0.0, 2.0, nt)
            hb = np.interp(xn, xo, bh)
        else:
            hb = bh.copy()
        hb = _h_proj(hb, ts_t)
        for t in range(4):
            if time.time() > dl - 8:
                break
            hs = hb + 0.015 * rng.randn(nt)
            hs = _h_proj(hs, ts_t)
            ho, vo = _h_subgrad(hs, dl, seed + 900 + nt * 10 + t, 10000)
            if vo < bv:
                bv, bh, bn = vo, ho, nt
    if time.time() < dl - 3:
        hp, vp = _h_polish(bh, bn / 2.0, 8000, seed + 777, dl)
        if vp < bv:
            bv, bh = vp, hp
    if bh is None:
        bh = np.full(45, 0.5)
        bv = _h_c5(bh)
        bn = 45
    return (bh, float(bv), int(bn))
