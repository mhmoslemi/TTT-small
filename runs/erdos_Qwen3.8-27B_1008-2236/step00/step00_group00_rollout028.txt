    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full') * dx))

def _clamp(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5
    d = h.sum() - t
    if d > 1e-13:
        w = np.argsort(-h)
        rem = d
        for k in w:
            s = min(h[k], rem)
            h[k] -= s
            rem -= s
