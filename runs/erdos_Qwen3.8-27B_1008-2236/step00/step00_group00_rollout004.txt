    r = np.fft.ifft(f * np.conj(f)).real[:n]

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

    R = compute_auto_corr(h)
    f_vals = dx * (target_sum - R)
    max_f = np.max(f_vals)
    threshold = max_f - 1e-4

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_auto_corr(h):
    n = len(h)
    f = np.fft.fft(h, 2 * n)
    r = np.fft.ifft(f * np.conj(f)).real[:n]
    return r

def project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, dx, target_sum, step_size):
    n = len(h)
    R = compute_auto_corr(h)
