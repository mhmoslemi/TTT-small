def project_onto_constraint(h, n):
    # Project h to satisfy 0<=h<=1 and sum(h)=n/2
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Water-filling: adjust by adding/subtracting uniformly
    if diff > 0:

        # Add to entries below 1
        candidates = np.where(h < 1)[0]
        if len(candidates) == 0:

        lo, hi = 0, 1 - h[candidates].min()
        for _ in range(50):

            adj = np.clip(mid, 0, 1 - h)
            if np.sum(adj) > diff:
                hi = mid
            else:
                lo = mid
        adj = np.clip((lo+hi)/2, 0, 1 - h)
        h = h + adj
    else:

            else:

        adj = np.clip((lo+hi)/2, 0, h)
        h = h - adj

def subgradient_step(h, n, alpha):
    # Compute subgradient of max_s C(s)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    # Active set: lags within epsilon of max
    active = np.where(corr > max_val - 1e-6)[0]
    # Compute gradient for each active lag
    grad = np.zeros(n)

Let me define: k-th element of correlate(h, 1-h, "full") corresponds to shift s = k - n + 1.

For shift s (can be negative, zero, or positive):
C(s) = sum over valid (i, i+s) pairs of h[i] * (1-h[i+s])

The gradient ∂C(s)/∂h[j]:
- For each valid pair (i, i+s):

Where "j valid as i" means 0≤j<n and 0≤j+s<n.

    """Gradient of the k-th correlate value w.r.t. h"""
    s = k - n + 1
    grad = np.zeros(n)
    if s >= 0:
        # Pairs (i, i+s) for i in [0, n-1-s]
        idx1 = np.arange(n - s)  # i values
        idx2 = idx1 + s  # i+s values
        grad[idx1] += (1.0 - h[idx2])  # (1-h[i+s])
        grad[idx2] += (-h[idx1])  # -h[i]
    elif s < 0:
        # Pairs (i, i+s) for i in [-s, n-1], i.e., (i, i+|s|) where i≥|s|
        # Equivalently: pairs (i+s, i) for i in [0, n-1-s-1]... 
        # Let me redo: s < 0, so i+s < i. Valid i: i+s ≥ 0 and i < n, so i ≥ -s = |s|
        # Pairs (i, i+s) for i in [|s|, n-1]
        # h[i]*(1-h[i+s]), i in [|s|, n-1]
        # ∂/∂h[j]: 
        #   j = i (i in [|s|, n-1]): (1-h[j+s]) → h[j+s] = h[j-|s|]
        #   j = i+s (i in [|s|, n-1], so j in [0, n-1-|s|]): -h[i] = -h[j+s] = -h[j+|s|]
        sp = -s  # |s|
        idx1 = np.arange(sp, n)  # i values
        idx2 = idx1 - sp  # i+s values (= i-sp)
        grad[idx1] += (1.0 - h[idx2])  # (1-h[i+s]) where i+s = i-sp

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def corr_vec(h):

    dx = 2.0 / n

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0

        if abs(d) < 1e-11:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0: break
            h[m] += d / c
            np.minimum(h, 1.0, out=h)
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0: break
            h[m] -= (-d) / c
            np.maximum(h, 0.0, out=h)

def grad_lag(h, k, n):

        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] += (-h[sp:])
    return g

    else:
        lo = int(tgt)
        v[order[:lo]] = 1.0
        v[order[lo]] = tgt - lo
    return v

import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(60):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(h[m] + d / c, 1.0)
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(h[m] - (-d) / c, 0.0)
    return h

def grad_lag(h, k, n):
    s = k - n + 1
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = n - s
        g[:m] += (1.0 - h[s:s+m])
        g[s:] += (-h[:m])
    else:
        sp = -s
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] += (-h[sp:])
    return g

def fw_vertex(g, n):
    tgt = n / 2.0
    v = np.zeros(n)
    order = np.argsort(g)
    lo = int(tgt)
    if tgt - lo < 1e-10:
        v[order[:lo]] = 1.0
    else:
        v[order[:lo]] = 1.0
        v[order[lo]] = tgt - lo
    return v

def optimize_fw(h0, n, max_it=8000, deadline=None, n_act=7):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_C5 = compute_C5(h)
    for t in range(1, max_it + 1):
        if deadline is not None and time.time() > deadline:
            break
        cv = corr_vec(h)
        cvm = cv.max()
        active = np.where(cv > cvm - 1e-9)[0]
        if len(active) > n_act:
            active = np.argsort(cv)[-n_act:]
        g = np.zeros(n)
        for kk in active:
            g += grad_lag(h, kk, n)
        g /= len(active)
        v = fw_vertex(g, n)
        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = proj(h_new, n)
        C5_new = compute_C5(h_new)
        if C5_new < best_C5 - 1e-14:
            best_C5 = C5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_C5

def fine_gradient(h0, n, max_it=30000, deadline=None):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_C5 = compute_C5(h)
    cv = corr_vec(h)
    cvm = cv.max()
    alpha = 5e-5
    stuck = 0

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.astype(float).copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, period=2.0)

def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        h[ones[0]] = 0.5
    return proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    best_h = None
    best_C5 = 1e10

    parent = None
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
        np_ = len(parent)
        parent = proj(parent, np_)
        best_C5 = compute_C5(parent)
        best_h = parent.copy()

    n_candidates = [47, 48, 50, 52, 54, 56, 60, 64, 72, 80, 100, 120, 160, 200]
    if parent is not None:
        n_candidates.insert(0, len(parent))
    n_candidates = list(dict.fromkeys(n_candidates))

    for n in n_candidates:
        if time.time() > deadline:
            break
        starts = []

        if parent is not None:
            up = upsample(parent, n)
            starts.append(proj(up, n))

        starts.append(make_binary_start(n, rng))
        starts.append(proj(rng.uniform(0, 1, n), n))

        if best_h is not None:
            hpt = upsample(best_h, n) + rng.normal(0, 0.06, n)
            starts.append(proj(hpt, n))

        for s in starts:
            if time.time() > deadline:
                break
            h_fw, c5_fw = optimize_fw(s, n, max_it=6000, deadline=deadline, n_act=5)
            if c5_fw < best_C5:
                best_C5 = c5_fw
                best_h = h_fw.copy()
            if time.time() > deadline:
                break
            h_fine, c5_fine = fine_gradient(h_fw, n, max_it=20000, deadline=deadline)
            if c5_fine < best_C5:
                best_C5 = c5_fine
                best_h = h_fine.copy()

    # Final fine polish on best
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_final, c5_final = fine_gradient(best_h, n, max_it=50000, deadline=deadline)
        if c5_final < best_C5:
            best_C5 = c5_final
            best_h = h_final

    best_h = proj(best_h, len(best_h))
    best_C5 = compute_C5(best_h)
    return best_h, best_C5, len(best_h)
