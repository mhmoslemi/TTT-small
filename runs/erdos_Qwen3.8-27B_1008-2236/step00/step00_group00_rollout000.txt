import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def upsample(h, factor):
    nh = len(h) * factor
    new_h = np.empty(nh)
    for i in range(factor):
        new_h[i::factor] = h
    return new_h

def anneal(h, rng, time_limit, T_init, delta_range):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    T = T_init
    half_dr = delta_range * 0.5
    while time.time() < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-half_dr, half_dr)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i] = hi
        h[j] = hj
        nc = compute_c5(h)
        if nc < best_c5:
            best_c5 = nc
            best_h = h.copy()
        if nc > cur_c5:
            if rng.random() > np.exp(-(nc - cur_c5) / T):
                h[i] -= delta
                h[j] += delta
        else:
            cur_c5 = nc
        if rng.random() < 0.0003:
            T *= 0.93
            if T < T_init * 0.01:
                T = T_init * 0.5
                h[:] = best_h
                cur_c5 = best_c5
    return best_h, best_c5

        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Second anneal at original resolution with different params
    t2 = t0 + budget_s * 0.55
    h2 = h.copy()
    h2, c5_2 = anneal(h2, rng, t2, 0.006, 0.25)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Upsample best to 2n and optimize
    if time.time() < deadline:
        h3 = upsample(best_h, 2)
        t3 = t0 + budget_s * 0.80
        h3, c5_3 = anneal(h3, rng, t3, 0.002, 0.2)
        if c5_3 < best_c5:

            best_h = h3.copy()

    # Phase 4: Final short polish
