def lp_step(h, n):
    y = 1.0 - h
    # Build constraint matrix for: sum_i h[i+k] * y[i] ≤ t for each lag k
    # Lag k means shift by k, so term is h[i+k]*y[i]
    # For np.correlate(h, y, 'full'), lag k goes from -(n-1) to (n-1)
    # In the output array, index = k + (n-1)
    
    num_lags = 2 * n - 1
    # A matrix: (num_lags, n+1) where last column is -t
    A = np.zeros((num_lags, n + 1))
    for k in range(num_lags):
        actual_shift = k - (n - 1)
        # sum_i h[i + actual_shift] * y[i] for valid i
        for i in range(n):
            j = i + actual_shift
            if 0 <= j < n:
                A[k, j] = y[i]
        A[k, n] = -1.0  # coefficient for t
    
    b = np.zeros(num_lags)
    
    # Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = n / 2.0 - h.sum()
    if abs(d) > 1e-15:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() > 0:
            h[m] += d / m.sum()
        else:
            h += d / n
    h = np.clip(h, 0.0, 1.0)
    d2 = n / 2.0 - h.sum()
    if abs(d2) > 1e-15:
        idx = np.argmax(h) if d2 < 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + d2, 0, 1)
    return h

def solve_lp(fixed, n):
    nl = 2 * n - 1
    A = np.zeros((nl, n + 1))
    for k in range(nl):
        s = k - (n - 1)
        if s >= 0:
            idx = np.arange(s, n)
            A[k, idx] = fixed[idx - s]
        else:
            idx = np.arange(0, n + s)
            A[k, idx] = fixed[idx - s]
        A[k, n] = -1.0
    b = np.zeros(nl)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    r = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[n / 2.0], bounds=bounds, method='highs')
    return r.x[:n] if r.success else np.full(n, 0.5)

def lp_alternate(h, n, max_iters=80, t_limit=0.0):
    t_s = time.time()

    m = np.max(corr)
    if eps <= 0:
        return m
    return m + eps * np.log(np.sum(np.exp((corr - m) / eps)))
