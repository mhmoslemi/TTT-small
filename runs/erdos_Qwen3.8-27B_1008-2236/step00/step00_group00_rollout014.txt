import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def constraint_fun(x):
    return np.sum(x) - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def sa(h, rng, time_limit, T_init, delta_lo, delta_hi):
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best = cur_c5
    best_h = h.copy()
    T = T_init
    start = time.time()
    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.rand() < 0.5:
            md = min(h[i], 1.0 - h[j])
            if md <= 0:
                continue
            d = rng.uniform(delta_lo, min(delta_hi, md)) if md > delta_lo else md * 0.5
        else:
            md = min(h[j], 1.0 - h[i])
            if md <= 0:
                continue
            d = -rng.uniform(delta_lo, min(delta_hi, md)) if md > delta_lo else -md * 0.5
        h[i] += d
        h[j] -= d
        nc5 = compute_c5(h)
        dd = nc5 - cur_c5
        if dd < 0 or (T > 1e-12 and rng.rand() < np.exp(-dd / T)):
            cur_c5 = nc5
            if cur_c5 < best:
                best = cur_c5
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        T = max(T * 0.99997, 1e-12)
    return best_h, best

def directed_search(h, rng, time_limit):
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best = cur_c5
    best_h = h.copy()
    start = time.time()
    while time.time() - start < time_limit:
        c = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(c))
        k_star = m_star - (n - 1)
        if rng.rand() < 0.7:
            i = rng.randint(n)
            ip = i + k_star
            im = i - k_star
            targets = []
            if 0 <= ip < n and h[i] > 0.2 and h[ip] < 0.8:
                targets.append((i, ip))
            if 0 <= im < n and h[im] > 0.2 and h[i] < 0.8:
                targets.append((im, i))
            if targets:
                a, b = targets[rng.randint(len(targets))]
                d = min(rng.uniform(0.01, 0.08), h[a], 1.0 - h[b])
                if d > 0.001:
                    h[a] -= d
                    h[b] += d
                    nc5 = compute_c5(h)
                    if nc5 < cur_c5:
                        cur_c5 = nc5
                        if cur_c5 < best:
                            best = cur_c5
                            best_h = h.copy()
                    else:
                        h[a] += d
                        h[b] -= d
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:

            d = rng.uniform(-0.05, 0.05)

def try_slsqp(h, n):
    try:
        cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        bounds = [(0.0, 1.0)] * n
        res = minimize(compute_c5, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 800, 'ftol': 1e-13, 'iprint': 0})
        if res.fun < compute_c5(h):
            return res.x
    except Exception:
        pass
    return h

def fix_sum(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:
            adj = diff / cnt
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
        else:
            idxs = np.argsort(h) if diff > 0 else np.argsort(-h)
            for idx in idxs:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-10:
                    break
    return h
