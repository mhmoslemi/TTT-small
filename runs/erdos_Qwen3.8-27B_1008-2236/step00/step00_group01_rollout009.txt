    return float(np.max(corr)) * dx

def project(h, S):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = corr.max()
    ew = np.exp(alpha * (corr - mx))
    w = ew / ew.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        js, je = max(0, -lag), min(n, n - lag)
        if je > js:
            grad[js:je] += wk * (1.0 - h[js + lag:je + lag])
        js2, je2 = max(0, lag), min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] -= wk * h[js2 - lag:je2 - lag]
    return grad

def optim(h, S, t_lim, n_it, a0, a1, lr0):
    n = len(h)
    h = project(h.copy(), S)
    bh, bv = h.copy(), compute_c5(h)
    a, lr = a0, lr0
    for it in range(n_it):
        if time.time() > t_lim:
            break
        g = smooth_grad(h, a)
        accepted = False
        for _ in range(14):
            h2 = project(h - lr * g, S)
            v2 = compute_c5(h2)
            if v2 <= bv + 1e-9 or lr < 1e-7:
                accepted = True
                break
            lr *= 0.5
        if accepted and v2 < bv:
            bv = v2
            bh = h.copy()
        h = h2
        if v2 < bv:
            bv = v2
            bh = h.copy()
        if (it + 1) % 600 == 0:
            a = min(a * 1.6, a1)
            lr *= 0.82
    return bh, bv

def rand_init(n, rng):
    h = rng.uniform(0, 1, n)
    return project(h, n * 0.5)

def bin_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    if n % 2:
        zi = np.where(h == 0)[0]
        h[rng.choice(zi)] = 0.5
    return project(h, n * 0.5)

def sin_init(n, rng):
    x = np.linspace(0, 2, n)
    freq = rng.uniform(2.0, 8.0)
    ph = rng.uniform(0, 2 * np.pi)
    h = (np.sin(freq * np.pi * x + ph) > 0.0).astype(float)
    return project(h, n * 0.5)

def interp_h(h, n_new):
    x = np.linspace(0, 2, len(h))
    x2 = np.linspace(0, 2, n_new)
    return project(np.interp(x2, x, h), n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)
    bh, bv, bn = None, 1.0, 0

    parent = None
    if 'initial_h_values' in globals():
        parent = np.asarray(globals()['initial_h_values'], dtype=float)

    # Phase 1: broad sweep over resolutions
    n_list = [32, 48, 64, 80, 96, 128, 160, 192, 256]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_list:
            n_list.append(np_)
        n_list.sort()

    t_p1 = min(t0 + budget_s * 0.18, t_end - 100)
    for n in n_list:
        if time.time() > t_p1:
            break
        for itype in ['r', 'b', 's']:
            if time.time() > t_p1:
                break
            tl = min(time.time() + 12.0, t_p1)
            h0 = rand_init(n, rng) if itype == 'r' else (
                 bin_init(n, rng) if itype == 'b' else sin_init(n, rng))
            hh, vv = optim(h0, n * 0.5, tl, 2500, 4.0, 35.0, 0.06)
            if vv < bv:
                bv, bh, bn = vv, hh.copy(), n

    # Phase 2: parent-based optimization
    if parent is not None and time.time() < t_end - 60:
        n_p = len(parent)
        tl = min(time.time() + budget_s * 0.2, t_end - 60)
        hh, vv = optim(parent.copy(), n_p * 0.5, tl, 20000, 3.0, 80.0, 0.03)
        if vv < bv:
            bv, bh, bn = vv, hh.copy(), n_p
        # Also try finer resolution from parent
        for n_f in [n_p + 16, n_p * 2]:
            if time.time() > t_end - 40:
                break
            if n_f > 256:
                continue
            h_f = interp_h(parent, n_f)
            tl2 = min(time.time() + budget_s * 0.08, t_end - 40)
            hh2, vv2 = optim(h_f, n_f * 0.5, tl2, 8000, 4.0, 50.0, 0.04)
            if vv2 < bv:
                bv, bh, bn = vv2, hh2.copy(), n_f

    # Phase 3: focused refinement on best candidate
    if bh is not None and time.time() < t_end - 5:
        n = len(bh)
        S = n * 0.5
        best_streak = 0
        while time.time() < t_end - 4:
            tl = min(time.time() + min(budget_s * 0.1, 120.0), t_end - 4)
            scale = 0.04 if best_streak < 3 else 0.015
            h0 = bh + scale * rng.standard_normal(n)
            h0 = project(h0, S)
            hh, vv = optim(h0, S, tl, 12000, 5.0, 100.0, 0.025)
            if vv < bv - 1e-8:
                bv, bh = vv, hh.copy()
                best_streak = 0
            else:
                best_streak += 1
            if best_streak > 8:
                break

    # Phase 4: try coarsening then re-optimizing (escape local min)
    if bh is not None and time.time() < t_end - 10:
        n_coarse = max(16, len(bh) // 3)
        h_c = interp_h(bh, n_coarse)
        tl = min(time.time() + 30.0, t_end - 10)
        hh_c, vv_c = optim(h_c, n_coarse * 0.5, tl, 5000, 3.0, 40.0, 0.05)
        if vv_c < bv:
            bv, bh, bn = vv_c, hh_c.copy(), n_coarse

    # Final: verify and ensure constraints
    if bh is not None:
        n = len(bh)
        bh = project(bh, n * 0.5)
        bv = compute_c5(bh)
        bn = n

    if bh is None:
        n = 48
        bh = bin_init(n, rng)
        bv = compute_c5(bh)
        bn = n

    return bh, bv, bn
