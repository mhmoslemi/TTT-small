def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-15:
                break
            h += diff * room / rs
        else:
            room = h.copy()
            rs = room.sum()
            if rs < 1e-15:
                break
            h += diff * room / rs
        h = np.clip(h, 0.0, 1.0)
