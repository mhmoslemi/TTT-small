import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-14:
            break
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        if diff > 0:
            h[idx] = min(1.0, h[idx] + diff)
        else:
            h[idx] = max(0.0, h[idx] + diff)
        diff = target - h.sum()
    return h

def nm_objective(x, n_pts):
    return eval_c5(project_feasible(x, n_pts))

def local_search(h, n, max_iters, rng, sa_temp=0.0):
    best = h.copy()
    best_c5 = eval_c5(best)
    fracs = (1.0, 0.5, 0.25, 0.125, 0.06)
    for it in range(max_iters):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if best[i] <= best[j]:
            max_d = min(1.0 - best[i], best[j])
            di, dj = 1.0, -1.0
        else:
            max_d = min(best[i], 1.0 - best[j])
            di, dj = -1.0, 1.0
        if max_d < 1e-11:
            continue
        for f in fracs:
            d = f * max_d
            test = best.copy()
            test[i] += di * d
            test[j] += dj * d
            c5 = eval_c5(test)
            if c5 < best_c5 - 1e-13:
                best = test
                best_c5 = c5
                break
            if sa_temp > 1e-10 and (c5 - best_c5) < sa_temp * 10.0:
                if rng.random() < np.exp(-(c5 - best_c5) / sa_temp):
                    best = test
                    best_c5 = c5
                    break
        if sa_temp > 1e-10:
            sa_temp *= 0.99995
    return best, best_c5

def interpolate_h(h0, n0, n_pts):
    ox = np.linspace(0.0, 1.0, n0, endpoint=False)
    nx = np.linspace(0.0, 1.0, n_pts, endpoint=False)
    return np.interp(nx, ox, h0)

def random_init(n_pts, rng):
    h = rng.uniform(0.0, 1.0, n_pts)
    return project_feasible(h, n_pts)

def cosine_init(n_pts):
    x = np.linspace(0.0, 2.0, n_pts, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return project_feasible(h, n_pts)

    t0 = time.time()
    rng = np.random.default_rng(seed)

    print(f"Initial: n={n0}, C5={best_c5:.6f}")

    # Build list of (n_pts, initial_h) pairs to try
    trials = [(n0, h0.copy())]
    for s in [48, 54, 60, 64, 72, 80, 96]:
        if s != n0:

    # Add a cosine and random init
    if n0 != 64:

        if c5_ls < best_c5:
            best_h, best_c5, best_n = h.copy(), c5_ls, n_pts
            print(f"  n={n_pts}: C5={best_c5:.6f} (new best)")

    # Final polish on best solution
    time_left = budget_s - 4 - (time.time() - t0)
    if time_left > 5:

        best_h, best_c5 = local_search(best_h, best_n, iters, rng)

    best_c5 = eval_c5(best_h)
    print(f"Final: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, len(best_h)
