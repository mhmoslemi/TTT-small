        h_rr, c5_rr = coord_descent(h_r, n0, dx0, 12, rng)
        if c5_rr < best_c5:
            best_h, best_c5, best_n = h_rr, c5_rr, n0

    # Phase 3: Double resolution
    t_lim4 = budget_s * 0.50
    n2 = n0 * 2
    dx2 = 2.0 / n2
    if time.time() - t_start < t_lim4:
        x_lo = np.linspace(0, 2, best_n, endpoint=False)
        x_hi = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_hi, x_lo, best_h)
        h_up = project_feasible(h_up, n2)
        h2, c5_2 = coord_descent(h_up, n2, dx2, 90, rng)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2
        hb2 = make_binary_random(n2, rng)
        h2b, c5_2b = coord_descent(hb2, n2, dx2, 40, rng)
        if c5_2b < best_c5:
            best_h, best_c5, best_n = h2b, c5_2b, n2

    # Phase 4: Triple resolution
    t_lim5 = budget_s * 0.65
    n3 = n0 * 3
    dx3 = 2.0 / n3
    if time.time() - t_start < t_lim5:
        x_lo = np.linspace(0, 2, best_n, endpoint=False)
        x_hi = np.linspace(0, 2, n3, endpoint=False)
        h_up3 = np.interp(x_hi, x_lo, best_h)
        h_up3 = project_feasible(h_up3, n3)
        h3, c5_3 = coord_descent(h_up3, n3, dx3, 60, rng)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    # Phase 5: More diverse restarts
    t_lim6 = budget_s * 0.80
    sizes = [n0, n2, n3]
    for it in range(30):
        if time.time() - t_start > t_lim6:
            break
        nt = sizes[it % 3]
        dx_t = 2.0 / nt
        style = it % 4
        if style == 0:
            h_r = make_binary_random(nt, rng)
        elif style == 1:
            h_r = project_feasible(rng.uniform(0.2, 0.8, nt), nt)
        elif style == 2:
            h_r = project_feasible(np.clip(0.5 + rng.normal(0, 0.15, nt), 0, 1), nt)
        else:
            h_r = project_feasible(np.abs(rng.normal(0.5, 0.3, nt)), nt)
        t_sub = min(15, (t_lim6 - (time.time() - t_start)) / 2)
        if t_sub < 3:
            break
        h_r, c5_r = coord_descent(h_r, nt, dx_t, t_sub, rng)
        if c5_r < best_c5:
            best_h, best_c5, best_n = h_r, c5_r, nt

    # Phase 6: Final refinement
    t_left = budget_s - (time.time() - t_start) - 5
    if t_left > 10:
        dx_b = 2.0 / best_n
        h_f, c5_f = coord_descent(best_h, best_n, dx_b, t_left * 0.6, rng)
        if c5_f < best_c5:
            best_h, best_c5, best_n = h_f, c5_f, best_n
        t_left2 = budget_s - (time.time() - t_start) - 3
        if t_left2 > 5:
            h_f2, c5_f2 = pair_descent(best_h, best_n, 2.0 / best_n, t_left2, rng)
            if c5_f2 < best_c5:
                best_h, best_c5, best_n = h_f2, c5_f2, best_n

    best_h = project_feasible(best_h, best_n)
    c5_final = compute_c5(best_h, 2.0 / best_n)
    return (best_h, c5_final, best_n)
