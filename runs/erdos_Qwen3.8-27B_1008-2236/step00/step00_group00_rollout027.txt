import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)

def soft_max_obj(h, beta, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(cor)
    f = m + (1.0/beta) * np.log(np.sum(np.exp(beta * (cor - m))))
    penalty = penalty_weight * (np.sum(h) - target) ** 2
    return f + penalty

def soft_max_grad(h, beta, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(cor)
    exp_cor = np.exp(beta * (cor - m))
    W = exp_cor / np.sum(exp_cor)
    
    # grad = dx * correlate(1-2h, W, 'full')[n-1 : 2n-1]
    g = dx * np.correlate(1.0 - 2.0 * h, W, mode="full")[n-1:2*n-1]
    
    # Add constraint penalty gradient
    g += 2.0 * penalty_weight * (np.sum(h) - target)
    
    return g

def fix_sum(h, target):
    """Adjust h to satisfy sum(h) = target while keeping 0<=h<=1."""
    h = h.copy()
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    # Final adjustment: distribute residual among non-boundary elements
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h
