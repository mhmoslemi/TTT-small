import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(c)
    # Find active constraints (within tolerance)
    tol = 1e-9
    active = np.where(c >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        s = k_idx - (n - 1)
        gk = np.zeros(n)
        if s >= 0:
            # (1-h[j+s]) for j in [0, n-1-s]
            gk[0:n-s] += (1.0 - h[s:n])
            # -h[j-s] for j in [s, n-1]
            gk[s:n] -= h[0:n-s]
        else:
            t = -s
            # (1-h[j+s]) = (1-h[j-t]) for j in [t, n-1]
            gk[t:n] += (1.0 - h[0:n-t])
            # -h[j-s] = -h[j+t] for j in [0, n-1-t]
            gk[0:n-t] -= h[t:n]
        grad += gk
    grad /= len(active)
    grad *= dx
    return max_val, grad

def project_constraint(h, target):
    # Project onto {x: 0<=x<=1, sum(x)=target}
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        p = np.clip(h - mid, 0, 1)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def projected_gradient(h0, target, max_iter=5000, init_lr=0.05, time_limit=60):
    start = time.time()
    h = h0.copy()
    h = project_constraint(h, target)
    val = compute_c5(h)
    lr = init_lr
    best_val = val
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        v, g = c5_grad(h)
        # Backtracking
        for _ in range(30):
            h_new = h - lr * g
            h_new = project_constraint(h_new, target)
            new_val = compute_c5(h_new)
            if new_val < v - 1e-12:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        val = new_val
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # Adaptive lr
        if val >= best_val - 1e-10:
            lr = max(lr * 1.1, init_lr * 0.1)

        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(47, 0.5)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    print(f"Init n={n0}: {best_c5:.6f}")

    # Phase 1: optimize initial grid
    tl = min(120.0, budget_s * 0.15)
    h_o, c5 = projected_gradient(h0, n0 / 2.0, 6000, 0.03, tl)
    if c5 < best_c5:
        best_c5, best_h = c5, h_o
    print(f"P1 n={n0}: {best_c5:.6f}")

    # Phase 2: multi-resolution
    for ns in [32, 64, 96, 128, 192, 256]:
        if time.time() - t_start > budget_s - 80:
            break
        hi = resample_to(best_h, ns)
        s = np.sum(hi)
        if s > 1e-12:
            hi *= (ns / 2.0) / s
        hi = project_constraint(hi, ns / 2.0)
        rem = budget_s - (time.time() - t_start) - 40
        tl = min(80.0, max(10.0, rem * 0.3))
        h_o, c5 = projected_gradient(hi, ns / 2.0, 4000, 0.02, tl)
        if c5 < best_c5:
            best_c5, best_h = c5, h_o
            print(f"P2 n={ns}: NEW BEST {best_c5:.6f}")
        else:
            print(f"P2 n={ns}: {c5:.6f}")

    # Phase 3: random restarts on best grid
    nb = len(best_h)
    rem = budget_s - (time.time() - t_start) - 30
    n_rest = min(50, max(3, int(rem / 22)))
    for r in range(n_rest):
        if time.time() - t_start > budget_s - 30:
            break
        h_r = best_h + rng.randn(nb) * 0.03
        h_r = project_constraint(h_r, nb / 2.0)
        tl = min(18.0, max(5.0, (budget_s - (time.time() - t_start)) * 0.4))
        h_o, c5 = projected_gradient(h_r, nb / 2.0, 2000, 0.015, tl)
        if c5 < best_c5:
            best_c5, best_h = c5, h_o
            print(f"P3 r{r}: NEW BEST {best_c5:.6f}")

    # Phase 4: final polish on best grid
    rem = budget_s - (time.time() - t_start) - 10
    if rem > 15:
        h_o, c5 = projected_gradient(best_h, len(best_h) / 2.0, 3000, 0.005, rem)
        if c5 < best_c5:
            best_c5, best_h = c5, h_o
            print(f"P4 polish: {best_c5:.6f}")

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
