    cur_c5 = c5_value(cur)
    best = cur.copy()
    best_c5 = cur_c5
    temp = 0.0008
    cooling = 0.9994
    min_temp = 1e-9
    while temp > min_temp:
        if time.time() > budget_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, 0.05)
        if cur[i] < d or cur[j] > 1.0 - d:
            continue
        oi, oj = cur[i], cur[j]
        cur[i] -= d
        cur[j] += d
        nc5 = c5_value(cur)
        dd = nc5 - cur_c5
        if dd < 0 or rng.random() < np.exp(-dd / temp):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best = cur.copy()
        else:
            cur[i], cur[j] = oi, oj
        temp *= cooling
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    h = initial_h_values.copy().astype(float)
    n = len(h)
    h = fix_sum(h, n)

    best_h = h.copy()
    best_c5 = c5_value(h)

    # Phase 1: Exhaustive swap hill climbing
    t1 = min(t0 + 40, deadline - 700)
    if time.time() < t1:
        bh, bc5 = swap_climb(h, 20)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh

    # Phase 2: SA (swaps + transfers)
    t2 = min(t0 + 300, deadline - 350)
    if time.time() < t2:
        bh, bc5 = sa_search(best_h, t2, seed, True)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh

    # Phase 3: Fine-tuning SA
    t3 = min(t0 + 450, deadline - 200)
    if time.time() < t3:
        bh, bc5 = fine_sa(best_h, t3, seed + 111)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh

    # Phase 4: Random restarts
    t4 = min(t0 + 900, deadline - 100)
    rng = np.random.RandomState(seed + 777)
    attempt = 0
    while time.time() < t4 and attempt < 50:
        hr = random_feasible(n, rng)
        hr = fix_sum(hr, n)
        bh, bc5 = swap_climb(hr, 8)
        if bc5 < best_c5 - 1e-6:
            best_c5, best_h = bc5, bh
            t_sub = min(time.time() + 40, t4 - 5)
            bh2, bc52 = sa_search(bh, t_sub, seed + attempt * 31, True)
            if bc52 < best_c5:
                best_c5, best_h = bc52, bh2
            if time.time() < t4 - 10:
                bh3, bc53 = fine_sa(best_h, min(time.time() + 20, t4 - 5), seed + 5000 + attempt)
                if bc53 < best_c5:
                    best_c5, best_h = bc53, bh3
        attempt += 1

    # Phase 5: Upscale and optimize (try 2x grid)
    t5 = min(t0 + budget_s - 40, deadline)
    if time.time() < t5 - 60 and n < 200:
        h_up = np.repeat(best_h, 2)
        n_up = len(h_up)
        h_up = fix_sum(h_up, n_up)
        bh_up, bc5_up = swap_climb(h_up, 5)
        if bc5_up < best_c5 - 1e-7:
            best_c5, best_h = bc5_up, bh_up
            if time.time() < t5 - 20:
                bh_u2, bc5_u2 = sa_search(best_h, min(time.time() + 50, t5 - 10), seed + 9000, True)
                if bc5_u2 < best_c5:
                    best_c5, best_h = bc5_u2, bh_u2

    # Phase 6: Final polish
    if time.time() < deadline - 2:
        bh, bc5 = swap_climb(best_h, 5)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh
        if time.time() < deadline - 1:
            bh, bc5 = fine_sa(best_h, deadline - 1, seed + 12345)
            if bc5 < best_c5:
                best_c5, best_h = bc5, bh

    return best_h, best_c5, len(best_h)
