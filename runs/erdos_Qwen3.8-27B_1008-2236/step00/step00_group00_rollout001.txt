    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max(corr, T):
    c = np.max(corr)
    if T < 1e-8:
        return c
    return c + T * np.log(np.mean(np.exp((corr - c) / T)))
