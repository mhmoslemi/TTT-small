import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            tr = room.sum()
            if tr < diff * (1 + 1e-15):
                break
            h += diff * room / tr
        else:
            room = np.maximum(h, 0.0)
            tr = room.sum()
            if tr < (-diff) * (1 + 1e-15):
                break
            h += diff * room / tr
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_new = np.arange(new_n) * old_n / new_n
    lo = np.floor(x_new).astype(int) % old_n
    hi = (lo + 1) % old_n
    frac = x_new - np.floor(x_new)
    return h[lo] * (1.0 - frac) + h[hi] * frac

def smooth_obj(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    a = c * dx
    max_a = a.max()
    s = beta * (a - max_a)
    exp_s = np.exp(s)
    obj = max_a + np.log(exp_s.sum()) / beta
    obj += gamma * (np.sum(h) - n / 2.0) ** 2
    return obj

def smooth_grad(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    a = c * dx
    max_a = a.max()
    s = beta * (a - max_a)
    exp_s = np.exp(s)
    p = exp_s / exp_s.sum()
    shifts = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    k_idx = shifts[None, :]
    i1 = j_idx + k_idx
    i2 = j_idx - k_idx
    valid1 = (i1 >= 0) & (i1 < n)
    valid2 = (i2 >= 0) & (i2 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(i1, 0, n - 1)], 0.0)
    term2 = np.where(valid2, h[np.clip(i2, 0, n - 1)], 0.0)
    grad = dx * (term1 @ p - term2 @ p)
    grad += 2.0 * gamma * (np.sum(h) - n / 2.0)
    return grad

def optimize_stage(h0, beta, gamma, max_iter=300):
    n = len(h0)
    h = project_sum(h0.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj(x, beta, gamma)

    def grad(x):
        return smooth_grad(x, beta, gamma)

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-7})
        h_opt = project_sum(res.x, n)
        val = compute_c5(h_opt)
        if val < best_val:
            best_h = h_opt
            best_val = val
    except Exception:
        pass
    return best_h, best_val

def local_search(h0, n_iters=5000, seed=42):
    n = len(h0)
    h = project_sum(h0.copy(), n)
    best_val = compute_c5(h)
    rng = np.random.default_rng(seed)
    for _ in range(n_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-0.08, 0.08)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        val = compute_c5(h_new)
        if val < best_val - 1e-13:
            h = h_new
            best_val = val
    return project_sum(h, n), best_val
