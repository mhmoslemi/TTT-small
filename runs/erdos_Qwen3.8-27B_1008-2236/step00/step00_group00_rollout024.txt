                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r
                    best_n = n
            except Exception:
                pass

    # Phase 2: Binary optimization at various even n
    n_list = [46, 48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72,
              74, 76, 78, 80, 84, 88, 92, 96, 100, 104, 108, 112, 116,
              120, 124, 128, 132, 136, 140, 144, 148, 152, 156, 160,
              164, 168, 172, 176, 180, 184, 188, 192, 196, 200, 208, 216,
              224, 232, 240, 248, 256, 272, 288, 304, 320, 336, 352, 368,
              384, 400, 448, 480, 512, 576, 640, 720, 800, 900]

    for n_t in n_list:
        t_left = budget_s - 15 - (time.time() - t0)
        if t_left < 5:
            break
        target_t = n_t // 2
        # Allocate time: smaller n get more time and more starts
        if n_t <= 60:
            t_alloc = min(20.0, t_left * 0.08)
            n_starts_use = 6
        elif n_t <= 100:
            t_alloc = min(15.0, t_left * 0.06)
            n_starts_use = 5
        elif n_t <= 200:
            t_alloc = min(10.0, t_left * 0.04)
            n_starts_use = 4
        else:
            t_alloc = min(7.0, t_left * 0.025)
            n_starts_use = 3
        t_alloc = max(1.5, t_alloc)
        starts = make_starts(n_t, target_t, seed + n_t)
        t_per_start = t_alloc / min(n_starts_use, len(starts))
        for hb in starts[:n_starts_use]:
            if time.time() - t0 > budget_s - 15:
                break
            t_ps = max(0.5, t_per_start)
            hb_opt, c5_opt = binary_local_search(hb, t_ps)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = hb_opt.copy()
                best_n = n_t

    # Phase 3: Final refinement on best solution
    t_left = budget_s - 3 - (time.time() - t0)
    if t_left > 5:
        if best_n % 2 == 0 and np.allclose(best_h, np.round(best_h)):
            best_h, best_c5 = binary_local_search(best_h, t_left)
        else:
            # Also try rounding to binary at same n if even
            if best_n % 2 == 0:
                h_bin = np.round(best_h).astype(float)
                if abs(h_bin.sum() - best_n / 2) < 0.1:
                    h_bin = enforce_sum(h_bin, best_n / 2)
                    if h_bin.sum() == best_n / 2 and np.allclose(h_bin, np.round(h_bin)):
                        h_bin, c5_b = binary_local_search(h_bin, t_left * 0.7)
                        if c5_b < best_c5:
                            best_c5 = c5_b
                            best_h = h_bin
            # Also try continuous refinement
            if time.time() - t0 < budget_s - 5:
                remaining = budget_s - 5 - (time.time() - t0)
                if remaining > 10:
                    def obj2(hf):
                        h = enforce_sum(hf, len(hf) / 2.0)
                        return compute_c5(h)
                    try:
                        res = minimize(obj2, best_h, method='powell',
                                       options={'maxiter': 5000, 'xtol': 1e-14, 'ftol': 1e-14})
                        h_r = enforce_sum(np.clip(res.x, 0, 1), best_n / 2.0)
                        c5_r = compute_c5(h_r)
                        if c5_r < best_c5:
                            best_c5 = c5_r
                            best_h = h_r
                    except Exception:
                        pass

    # Ensure final constraint satisfaction
    best_h = enforce_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
