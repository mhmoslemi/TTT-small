from scipy.optimize import minimize
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project onto {x in [0,1]^N : sum = target_sum}"""
    h = np.asarray(h, dtype=float)
    # Bisection for the constant shift
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_step(h, s):
    """Compute subgradient at shift s"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        g_i = 0.0
        if 0 <= i - s < n:
            g_i += (1.0 - h[i - s])
        if 0 <= i + s < n:
            g_i -= h[i + s]
        g[i] = g_i
    return g

def subgrad_descent(h0, n_steps=3000, alpha0=0.05, time_limit=100):
    h = h0.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_C5(h)
    best_h = h.copy()
    t0 = time.time()
    
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        
        curr_c5 = corr[k_max] * dx

            best_c5 = curr_c5
            best_h = h.copy()
        
        g = subgrad_step(h, s)
        alpha = alpha0 / (1.0 + 0.005 * step)
        h = h - alpha * g
        h = project_feasible(h, target_sum)
    
    return best_h, best_c5

    """Smooth approximation of max via log-sum-exp"""
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for stability
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(corr - m)))) / len(corr) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_descent(h0, n_steps=3000, alpha0=0.1, time_limit=60):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    best_c5 = compute_C5(h)
    best_h = h.copy()
    t0 = time.time()
    momentum = np.zeros(n)
    mu = 0.9
    
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        
        c5 = corr[k_max] * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient
        g = np.zeros(n)
        lo1, hi1 = max(0, s), min(n, s + n)
        g[lo1:hi1] += 1.0 - h[lo1 - s: hi1 - s]
        lo2, hi2 = max(0, -s), min(n, n - s)
        g[lo2:hi2] -= h[lo2 + s: hi2 + s]
        
        alpha = alpha0 / (1.0 + 0.01 * step)
        momentum = mu * momentum - alpha * g
        h = h + momentum
        h = project_feasible(h, target)
    
    return best_h, best_c5

def coord_descent(h0, n_iters=30, time_limit=60):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    best_c5 = compute_C5(h)
    best_h = h.copy()
    t0 = time.time()
    order = list(range(n))
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        import random
        random.shuffle(order)
        for i in order:
            # Try 0 and 1
            old = h[i]
            h[i] = 0.0
            c0 = compute_C5(h)
            h[i] = 1.0
            c1 = compute_C5(h)
            h[i] = 0.0 if c0 <= c1 else 1.0
        h = project_feasible(h, target)
        c = compute_C5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_opt(h0, beta=20.0, time_limit=120):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        c = corr * dx
        m = np.max(c)
        return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - m))))
    
    from scipy.optimize import minimize as _min
    t0 = time.time()
    res = _min(obj, h, method='L-BFGS-B', options={
        'maxiter': 500,
        'maxls': 50,
        'ftol': 1e-14,
        'gtol': 1e-10
    })
    if time.time() - t0 < time_limit:
        # Try with higher beta
        res2 = _min(obj, res.x, method='L-BFGS-B', options={
            'maxiter': 300,
            'maxls': 30,
            'ftol': 1e-14,
            'gtol': 1e-10
        })
        if time.time() - t0 < time_limit * 0.8:
            res = res2 if res2.fun < res.fun else res
    
    h_opt = project_feasible(res.x, target)
    return h_opt, compute_C5(h_opt)

import numpy as np
from scipy.optimize import minimize
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_descent(h0, n_steps=5000, alpha0=0.08, time_limit=80):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    best_c5 = compute_C5(h)
    best_h = h.copy()
    t0 = time.time()
    mom = np.zeros(n)
    mu = 0.85
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        c5_val = corr[k_max] * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        g = np.zeros(n)
        lo1, hi1 = max(0, s), min(n, s + n)
        if lo1 < hi1:
            g[lo1:hi1] += 1.0 - h[lo1 - s: hi1 - s]
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + s: hi2 + s]
        alpha = alpha0 / (1.0 + 0.008 * step)
        mom = mu * mom - alpha * g
        h = project_feasible(h + mom, target)
    return best_h, best_c5

def coord_descent(h0, n_iters=20, time_limit=60):

    target = n / 2.0

        if time.time() - t0 > time_limit:
            break
        np.random.permutation(order)
        for i in range(n):
            idx = order[i]
            old = h[idx]
            h[idx] = 0.0
            c0 = compute_C5(h)
            h[idx] = 1.0
            c1 = compute_C5(h)
            h[idx] = 0.0 if c0 <= c1 else 1.0
        h = project_feasible(h, target)
        c = compute_C5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def smooth_refine(h0, beta=30.0, time_limit=100):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))
    res = minimize(obj, h, method='L-BFGS-B', options={'maxiter': 300, 'maxls': 40, 'ftol': 1e-15})
    if time.time() - t0 < time_limit * 0.6:
        res = minimize(obj, res.x, method='L-BFGS-B', options={'maxiter': 200, 'maxls': 30, 'ftol': 1e-15})
    h_opt = project_feasible(res.x, target)
    return h_opt, compute_C5(h_opt)

def multi_subgrad(h0, n_steps=4000, time_limit=80):
    """Subgradient with multiple active shifts."""
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    best_c5 = compute_C5(h)
    best_h = h.copy()
    t0 = time.time()
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        threshold = np.mean(corr) + 0.5 * np.std(corr)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for k in active:
            s = int(k) - (n - 1)
            lo1, hi1 = max(0, s), min(n, s + n)
            if lo1 < hi1:
                g[lo1:hi1] += 1.0 - h[lo1 - s: hi1 - s]
            lo2, hi2 = max(0, -s), min(n, n - s)
            if lo2 < hi2:

            best_h = h.copy()
        alpha = 0.05 / (1.0 + 0.005 * step)

    return best_h, best_c5
