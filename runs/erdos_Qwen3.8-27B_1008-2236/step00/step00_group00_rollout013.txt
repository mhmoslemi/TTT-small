import time

def eval_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def sa_optimize(h_start, n, T_init, t_end, rng, best_h, best_c5):
    T = T_init
    h = h_start.copy()
    stall = 0
    while time.time() < t_end:
        if T < 1e-10:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            T *= 0.9999995
            continue
        dmax = 0.01 + 0.05 * (T / T_init)
        delta = rng.uniform(dmax * 0.1, dmax) * (1.0 if rng.random() < 0.5 else -1.0)
        old_i, old_j = h[i], h[j]
        h[i] += delta
        h[j] -= delta
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i], h[j] = old_i, old_j
            T *= 0.9999995
            stall += 1
            continue
        c5_new = eval_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
