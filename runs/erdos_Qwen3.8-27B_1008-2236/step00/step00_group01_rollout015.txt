import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not m.any():
            break
        h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def make_bresenham(n, k):
    h = np.zeros(n)
    err = 0.0
    for i in range(n):
        err += k
        if err >= n:
            h[i] = 1.0
            err -= n
    return h

def make_initial_n(n, init, rng):
    ts = n / 2.0
    if init is not None and len(init) == n:
        return init.copy()
    if init is not None:
        idx = np.round(np.linspace(0, len(init) - 1, n)).astype(int)
        h = init[idx].copy()
    else:
        k = n // 2
        h = make_bresenham(n, k)
    h = project(h, ts)
    return h

def local_search(h, n, rng, deadline):

    h = project(h.copy(), ts)
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stag = 0

        stag += 1
        improved = False
        for i in range(n):
            if time.time() > deadline - 30:

            for j in range(i + 1, n):

                hi0 = h[i]

                for tgt in (0.0, 1.0, hi0 - 0.15, hi0 + 0.15):
                    if abs(tgt - hi0) < 0.005:

                    if tgt < -1e-9 or tgt > 1.0 + 1e-9:

                    htn = s - tgt
