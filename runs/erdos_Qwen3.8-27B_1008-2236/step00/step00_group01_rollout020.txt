        tgt = nt / 2.0
        dx = 2.0 / nt
        alphas = [(1 + np.sqrt(5)) / 2.0, np.sqrt(2), np.sqrt(3), 2.0]
        starts = []
        starts.append(upsample(h_par, nt, tgt))
        starts.append(upsample(best_h, nt, tgt))
        for a in alphas:
            starts.append(beatty_init(nt, tgt, a))
        for hs in starts:
            el = time.time() - t0
            rem = budget_s - el
            if rem < 50:
                break
            ho, co = coord_descent(hs, nt, tgt, dx, 100)
            if co < best_c5:
                best_h, best_c5, best_n = ho.copy(), co, nt
            if time.time() - t0 < budget_s - 30:
                nis = int(min(2500, (budget_s - time.time() + t0) * 25))
                nis = max(nis, 300)
                hsa, csa = sim_anneal(ho, nt, tgt, dx, nis, seed + nt)
                if csa < best_c5:
                    best_h, best_c5, best_n = hsa.copy(), csa, nt
                hf, cf = coord_descent(hsa, nt, tgt, dx, 80)
                if cf < best_c5:
                    best_h, best_c5, best_n = hf.copy(), cf, nt

    el = time.time() - t0
    if el < budget_s * 0.92:
        dx_b = 2.0 / best_n
        hfin, c5fin = coord_descent(best_h, best_n, best_n / 2.0, dx_b, 500)
        if c5fin < best_c5:
            best_h, best_c5 = hfin, c5fin
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
