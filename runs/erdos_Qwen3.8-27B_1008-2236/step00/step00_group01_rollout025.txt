import time

def compute_correlations(h):
    return np.correlate(h, 1-h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Iterative water-filling
        for _ in range(10):
            if diff > 0:
                mask = h < 1.0
                cnt = np.sum(mask)
                if cnt == 0: break
                adj = diff / cnt
                h[mask] = np.minimum(1.0, h[mask] + adj)
            else:
                mask = h > 0.0
                cnt = np.sum(mask)
                if cnt == 0: break
                adj = -diff / cnt
                h[mask] = np.maximum(0.0, h[mask] - adj)
            diff = target - np.sum(h)
            if abs(diff) < 1e-12: break
    return h

def subgradient_step(h, alpha, n, top_k=3):
    """One subgradient step focusing on top-k correlations."""
    corr = compute_correlations(h)
    # Get top-k indices
    top_idx = np.argsort(corr)[-top_k:][::-1]
    weights = np.ones(top_k) / top_k  # uniform weight on active constraints
    
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        for j in range(n):
            val = 0.0
            j1 = j + s
            if 0 <= j1 < n:
                val += (1 - h[j1])
            j2 = j - s
            if 0 <= j2 < n:
                val -= h[j2]
            grad[j] += weights[list(top_idx).index(idx)] * val
    
    h_new = h - alpha * grad
    h_new = project(h_new, n)
    return h_new

import numpy as np
import time

def corr_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    d = t - h.sum()
    for _ in range(50):
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] = np.minimum(1.0, h[m] + d / c)
        else:
            m = h > 0.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / c)
        d = t - h.sum()
    return h

def grad_lag(h, s, n):
    g = np.zeros(n)
    a = np.arange(n) + s
    v = (a >= 0) & (a < n)
    if v.any():
        g[v] += (1.0 - h[a[v]])
    b = np.arange(n) - s
    v2 = (b >= 0) & (b < n)
    if v2.any():
        g[v2] -= h[b[v2]]
    return g

def to_binary(h, n):
    h = np.copy(h)
    if n % 2 == 0:
        idx = np.argsort(h)[::-1]
        h[:] = 0.0
        h[idx[:n // 2]] = 1.0
    else:
        k = (n - 1) // 2
        idx = np.argsort(h)[::-1]
        h[:] = 0.0
        h[idx[:k]] = 1.0
        h[idx[k]] = 0.5
    return h

def swap_search(h, n, max_rounds=5000, rng=None):

    if rng is None:
        rng = np.random.RandomState(42)

        rng.shuffle(zeros)

            for j in zeros:

                h[i], h[j] = h[j], h[i]

    return h, best

def continuous_refine(h, n, max_iter=10000, rng=None):
    if rng is None:
        rng = np.random.RandomState(7)
    h = project(h.copy(), n)
    best = corr_c5(h)
    best_h = h.copy()
    alpha = 0.04

        k1 = int(np.argmax(c))
