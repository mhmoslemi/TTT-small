First sum: ∑_i h[i+s]. h[j] appears when i+s=j, i.e., i=j-s. Coefficient is 1 (if 0≤j-s<n).

- When i = j: term is h[j+s]*h[j], derivative w.r.t. h[j] is h[j+s]

So d(corr[m])/d(h[j]) = 1_{valid(j-s)} - h_{j-s}·1_{valid} - h_{j+s}·1_{valid}

    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < tol:
            break
        if diff > 0:
            # Need to add; add to elements with most room

            else:
                step = max(-h[idx], diff)

            diff -= step

                h0 = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n_pts) / n_pts + phase)
                h0 += 0.1 * rng.randn(n_pts)
                h0 = np.clip(h0, 0, 1)
                h0 = fix_sum(h0, target)
            else:
                if n_pts == n_parent:
                    h0 = h_parent + 0.15 * rng.randn(n_pts)
                else:
                    xo = np.linspace(0, 2, n_parent, endpoint=False)
                    xn = np.linspace(0, 2, n_pts, endpoint=False)
                    h0 = np.interp(xn, xo, h_parent, period=2.0)
                    h0 += 0.15 * rng.randn(n_pts)
                h0 = np.clip(h0, 0, 1)
                h0 = fix_sum(h0, target)

            def obj(x):
                c = compute_corr(x)
                return float(np.sum((c * dx)**P)) ** (1.0 / P)

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n_pts)}]
            bnds = [(0.0, 1.0)] * n_pts
            try:
                res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                               constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
                h_opt = np.clip(res.x, 0, 1)
                h_opt = fix_sum(h_opt, target)
                c5_new = compute_c5(h_opt, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_opt.copy()
            except Exception:
                pass
    # Local coordinate polish on best
    n_pts = len(best_h)
    dx = 2.0 / n_pts
    target = n_pts / 2.0
    h_cur = best_h.copy()
    cur_val = compute_c5(h_cur, dx)
    step_sizes = [0.02, 0.05, 0.1, 0.15, 0.01, -0.01, -0.02, -0.05, -0.1, -0.15]
    for outer in range(3000):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        i = (outer * 7 + 3) % n_pts
        for ds in step_sizes:
            h_test = h_cur.copy()
            new_val_i = h_cur[i] + ds
            if new_val_i < 0 or new_val_i > 1:
                continue
            h_test[i] = new_val_i
            h_test = fix_sum(h_test, target)
            v = compute_c5(h_test, dx)
            if v < cur_val - 1e-12:
                h_cur = h_test
                cur_val = v
                improved = True
                break
        if not improved and outer > 200:
            break
    if cur_val < best_c5:
        best_c5 = cur_val
        best_h = h_cur.copy()
    # 0-1 snap attempt
    h_snap = np.round(best_h)
    n_ones = int(np.sum(h_snap))
    n_target = int(target)
    if n_ones != n_target:
        if n_ones > n_target:
            idx = np.where(h_snap == 1)[0]
            for i in range(n_ones - n_target):
                h_snap[idx[i]] = 0
        else:
            idx = np.where(h_snap == 0)[0]
            for i in range(n_target - n_ones):
                h_snap[idx[i]] = 1
    c5_snap = compute_c5(h_snap, dx)
    if c5_snap < best_c5:
        best_c5 = c5_snap
        best_h = h_snap.copy()
    best_h = fix_sum(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, len(best_h)
