import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def c5obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(c.max() * dx)

def slsqp(h0, maxiter=400):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return r.x, float(r.fun)

    elif len(ones) < target:

    if best_h is None:

        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n

    best_h = project_sum(best_h)
    final = compute_c5(best_h)
    return best_h, min(final, best_c5), best_n
