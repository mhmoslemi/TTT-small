                            best_h = h_new.copy()
                            h_try = h_new.copy()
                except Exception:
                    pass
            # Subgradient polish
            if time.time() < t2:
                for _ in range(30):
                    if time.time() > t2:
                        break
                    h_sg = subgradient_step(h_try, 0.01, n)
                    c5_sg = evaluate_c5(h_sg)
                    if c5_sg < best_c5:
                        best_c5 = c5_sg
                        best_h = h_sg.copy()
                        h_try = h_sg.copy()

    # Phase 3: Coordinate descent (18%)
    t3 = t0 + budget_s * 0.86
    best_h, best_c5 = coord_descend(best_h, n, t3, n_rounds=80)

    # Phase 4: Final SLSQP + CD (12%)
    if time.time() < deadline:
        for temp in (0.0005, 0.0002, 0.0001, 0.00005):
            if time.time() > deadline:
                break
            _current_temp = temp
            try:
                res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 800, 'ftol': 1e-16})
                if res.x is not None:
                    h_new = fix_sum(res.x, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
            except Exception:
                pass

    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descend(best_h, n, t0 + budget_s * 0.995, n_rounds=60)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
