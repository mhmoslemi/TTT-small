    **Approach: Successive Convex Approximation (SCA)**

    Iteratively:
    1. Given current h₀, linearize each g_k(h) around h₀:
       g_k(h) ≈ g_k(h₀) + ∇g_k(h₀)ᵀ (h - h₀)
    2. Solve: minimize t subject to g_k(h₀) + ∇g_k(h₀)ᵀ(h-h₀) ≤ t ∀k, sum h = n/2, 0≤h≤1
       This is a LINEAR program!
    3. Update h₀ = h, repeat.

    The LP in step 2 can be solved with scipy.linprog or cvxpy.

    Let me work out the gradient:
    g_k(h) = dx * sum_i h[i+k]*(1-h[i]) (with valid indices)
    
    ∂g_k/∂h[m] = dx * [sum over valid i where i+k=m of (1-h[i]) - sum over valid i where i=m of h[i+k]]
    
    For h[m] to appear in the sum:
    - As h[i+k] where i+k=m, i.e., i=m-k: contribution is (1-h[m-k]) if 0≤m-k<n
    - As h[i] where i=m: contribution is -h[m+k] if 0≤m+k<n
    
    So ∂g_k/∂h[m] = dx * [(1-h[m-k]) * I(0≤m-k<n) - h[m+k] * I(0≤m+k<n)]
    
    (where I is indicator)

    This is sparse - for each k, only O(n) entries are non-zero.

    The LP has n+1 variables (h[0],...,h[n-1], t) and 2n + (2n-1) + 1 = 4n constraints (approximately). For n=45, that's manageable.

def sca_step(h0, n, dx, max_iter_lp=1000):
    """One SCA iteration: linearize around h0, solve LP."""
    # Compute g_k and gradient for each lag
    # np.correlate(h, 1-h, 'full') index l corresponds to lag = l - (n-1)
    # corr[l] = sum_{i: 0<=i<n, 0<=i+lag<n} h[i+lag] * (1-h[i])
    
    # For the LP: minimize t
    # s.t. g_k(h0) + grad_k · (h - h0) <= t  for all k
    #      sum(h) = n/2
    #      0 <= h[i] <= 1
    #      t free (or t >= 0)
    
    n_lags = 2 * n - 1
    # Build constraint matrix A (inequality: A @ x <= b)
    # x = [h[0], ..., h[n-1], t]
    
    # For each lag k (index l from 0 to 2n-2):
    # g_l(h) = dx * sum_i h[i+lag]*(1-h[i])
    # Linearized: g_l(h0) + dx * sum_i [grad component]
    
    # grad_l[m] = dx * [(1-h0[m-lag]) if valid(m-lag) - h0[m+lag] if valid(m+lag)]
    # where lag = l - (n-1)
    
    A_rows = []
    b_vals = []
    
    for l in range(n_lags):
        lag = l - (n - 1)
        # Compute g_l(h0)
        # sum_i h0[i+lag] * (1-h0[i]) for valid i, i+lag
        val = 0.0
        row = np.zeros(n + 1)  # last element is for -t (we'll do: g_l <= t → g_l - t <= 0)
        for i in range(n):
            i2 = i + lag
            if 0 <= i2 < n:
                # h[i+lag] * (1 - h[i])
                # Linearize: h0[i+lag]*(1-h0[i]) + (1-h0[i])*(h[i+lag]-h0[i+lag]) + h0[i+lag]*(-(h[i]-h0[i]))
                # = h0[i+lag]*(1-h0[i]) + (1-h0[i])*h[i+lag] - h0[i+lag]*h[i]
                # Gradient wrt h[i+lag]: (1-h0[i])
                # Gradient wrt h[i]: -h0[i+lag]
                row[i2] += (1 - h0[i])
                row[i] += -h0[i2]
                val += h0[i2] * (1 - h0[i])
        row[-1] = -1.0  # -t
        A_rows.append(row)
        b_vals.append(dx * val)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    #Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0 (or t >= -something)
    bounds = [(0, 1)] * n + [(-0.1, None)]  # t can be negative in principle
    
    res = linprog(c=np.zeros(n + 1), A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                 bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n].copy()
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        d = n/2.0 - h_new.sum()
        # Simple projection
        for _ in range(100):
            if abs(d) < 1e-12:
                break
            if d > 0:
                mask = h_new < 1.0
            else:
                mask = h_new > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h_new[mask] += d / cnt
            h_new = np.clip(h_new, 0, 1)
            d = n/2.0 - h_new.sum()
        return h_new
    return None
