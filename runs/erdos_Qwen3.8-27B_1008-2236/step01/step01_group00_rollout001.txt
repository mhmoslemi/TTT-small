import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_active(h, k_max):
    """Gradient of g_{k_max}(h) = sum_n h[n+k_max]*(1-h[n])"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each j, grad[j] = dx * [(1-h[j-k_max]) * I(j-k_max valid) - h[j+k_max] * I(j+k_max valid)]
    for j in range(n):
        g = 0.0
        jm = j - k_max
        if 0 <= jm < n:
            g += (1.0 - h[jm])
        jp = j + k_max
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = dx * g
    return grad

def project(h, n, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def subgradient_descent(h0, n, t_limit, alpha0=0.01, decay=0.999):
    """Projected subgradient on the active lag."""

    h = h0.copy()

        if c5 < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, n, t_limit, maxiter=500):
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(evaluate_c5, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h0.copy(), evaluate_c5(h0)

def coord_descent(h0, n, t_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    stucks = 0
    while time.time() < t_limit and stucks < 8:
        improved = False
        order = np.random.permutation(n)
        for j in order:
            old = h[j]
            cands = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0,
                     max(0.0, old - 0.08), min(1.0, old + 0.08)]
            for cv in cands:
                if abs(cv - old) < 1e-9:
                    continue
                h[j] = cv
                h = project_feasible(h, n)
                c5t = evaluate_c5(h)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    best_h = h.copy()
                    h = best_h.copy()
                    improved = True
                    break
                else:
                    h = best_h.copy()
        if not improved:
            stucks += 1
    return best_h, best_c5

def swap_search(h0, n, t_limit, n_pairs=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    deltas = [0.03, 0.06, 0.1, 0.15, 0.2, 0.02, -0.03, -0.06, -0.1, -0.15, -0.2, -0.02]
    while time.time() < t_limit:
        improved = False
        for _ in range(n_pairs):
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in deltas:
                if hi + d < -1e-9 or hi + d > 1 + 1e-9:
                    continue
                if hj - d < -1e-9 or hj - d > 1 + 1e-9:
                    continue
                h[i] = hi + d
                h[j] = hj - d
                c5t = evaluate_c5(h)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    best_h = h.copy()
                    h = best_h.copy()
                    improved = True
                    break
                else:
                    h[i] = hi
                    h[j] = hj
        if not improved:
            n_pairs = max(500, n_pairs // 2)
            if n_pairs < 500:
                break
    return best_h, best_c5

def periodic_optimize(n, p, t_limit, n_starts=20):
    reps = n // p
    if reps * p != n:
        return None, None
    target_pat_sum = n / 2.0 / reps
    best_pat = np.full(p, target_pat_sum / p)
    best_c5 = float('inf')
    bounds = tuple((0.0, 1.0) for _ in range(p))
    cons = [{'type': 'eq', 'fun': lambda pat: pat.sum() - target_pat_sum,
             'jac': lambda pat: np.ones(p)}]
    for s in range(n_starts):
        if time.time() > t_limit:
            break
        if s == 0:
            pat0 = np.full(p, target_pat_sum / p)
        else:
            pat0 = np.random.dirichlet(np.ones(p) * 2) * target_pat_sum
            pat0 = np.clip(pat0, 0, 1)
            if abs(pat0.sum() - target_pat_sum) < 1e-6:
                pass
            else:
                pat0 = pat0 * target_pat_sum / max(pat0.sum(), 1e-12)
                pat0 = np.clip(pat0, 0, 1)
                if pat0.sum() != target_pat_sum:
                    continue
        def obj(pat):
            h = np.tile(pat, reps)
            return evaluate_c5(h)
        try:
            res = minimize(obj, pat0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None:
                pat_new = np.clip(res.x, 0, 1)
                h_new = np.tile(pat_new, reps)
                c5v = evaluate_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_pat = pat_new.copy()
        except Exception:
            pass
    h_best = np.tile(best_pat, reps)
    return h_best, best_c5

def multi_temperature_slsqp(h0, n, t_limit):
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    dx = 2.0 / n
    def smooth_obj(h, temp):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        if temp < 1e-12:
            return m
        return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    for temp in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda hh: smooth_obj(hh, temp), best_h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5v = evaluate_c5(h_new)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = project_feasible(np.clip(h0, 0, 1), n)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Multi-temperature SLSQP from parent (40% time)
    t1 = t0 + budget_s * 0.40
    best_h, best_c5 = multi_temperature_slsqp(best_h, n, t1)

    # Phase 2: Coordinate descent polish (15% time)
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t2)

    # Phase 3: Swap search (15% time)
    t3 = t0 + budget_s * 0.70
    best_h, best_c5 = swap_search(best_h, n, t3, n_pairs=1500)

    # Phase 4: Periodic structure exploration on n=45 (10% time)
    t4 = t0 + budget_s * 0.80
    for p in [3, 5, 9, 15]:
        h_per, c5_per = periodic_optimize(n, p, t4, n_starts=10)
        if c5_per is not None and c5_per < best_c5:
            best_c5 = c5_per
            best_h = project_feasible(h_per, n)

    # Phase 5: Diverse restarts with SLSQP (15% time)
    t5 = t0 + budget_s * 0.93
    for r in range(30):
        if time.time() > t5:
            break
        scale = 0.12 * (1.0 - r / 30.0) + 0.03
        pert = np.random.randn(n) * scale
        h_try = project_feasible(best_h + pert, n)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        elif c5_try < best_c5 + 0.015:
            h_r, c5_r = slsqp_opt(h_try, n, t5, maxiter=200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Try n=90 if time remains (7% time)
    t6 = t0 + budget_s * 0.97
    if time.time() < t6 - 60:
        n2 = 90
        for r in range(6):
            if time.time() > t6:
                break
            if r == 0:
                h2 = np.repeat(best_h, 2)[:n2]
            elif r == 1:
                h2 = np.tile(np.full(5, 0.5), 18)
            else:
                h2 = project_feasible(np.clip(np.random.rand(n2) * 0.5 + 0.25, 0, 1), n2)
            h2, c5_2 = slsqp_opt(h2, n2, t6, maxiter=300)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2
                n = n2

    # Final polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = slsqp_opt(best_h, n, t0 + budget_s * 0.995, maxiter=100)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
