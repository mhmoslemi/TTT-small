import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def make_binary(n, rng):
    """Create a binary vector with n/2 ones (n must be even)."""
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def make_binary_frac(n, rng):
    """Create a vector with (n-1)/2 ones, (n-1)/2 zeros, and one 0.5 (n odd)."""
    h = np.zeros(n)
    remaining = n - 1
    ones = rng.choice(remaining, remaining//2, replace=False)
    h[ones] = 1.0
    # Pick one position for the 0.5
    zero_positions = np.where(h == 0)[0]
    h[zero_positions[rng.integers(len(zero_positions))]] = 0.5
    return h

def swap_local_search(h, rng, time_limit=5.0, restarts=20):
    """Local search using swaps for binary/nearly-binary vectors."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for restart in range(restarts):
        if time.time() - t0 > time_limit:
            break
        h = best_h.copy() if restart > 0 else h.copy()
        c5 = best_c5
        if restart > 0:
            # Perturb: swap 2-5 random pairs
            for _ in range(rng.integers(2, 6)):
                ones = np.where(h >= 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                if len(ones) < 2 or len(zeros) < 2:
                    break
                i = int(ones[rng.integers(len(ones))])
                j = int(zeros[rng.integers(len(zeros))])
                h[i], h[j] = 0.0, 1.0
            h = project_sum(h)
            c5 = compute_c5(h)
        
        # Hill climbing with single swaps
        stuck = 0
        while time.time() - t0 < time_limit and stuck < 200:
            ones = np.where(h >= 0.99)[0]
            zeros = np.where(h <= 0.01)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 <= c5 + 1e-10:
                c5 = min(c5, new_c5)
                stuck = 0
            else:
                h[i], h[j] = h[j], h[i]
                stuck += 1
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def sa_binary(h0, rng, time_limit=10.0):
    """Simulated annealing on binary vectors."""
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    T = 0.5
    
    while time.time() - t0 < time_limit and T > 1e-4:
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= 0.9995
        # Occasionally try double swap
        if rng.random() < 0.2 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = int(ones[rng.integers(len(ones))])
            i2 = int(ones[rng.integers(len(ones))])
            j1 = int(zeros[rng.integers(len(zeros))])
            j2 = int(zeros[rng.integers(len(zeros))])
            if len(set([i1,i2,j1,j2])) < 4:

                h[i1], h[i2], h[j1], h[j2] = 1, 1, 0, 0

    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def make_half(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, (n - 1) // 2, replace=False)] = 1.0
    z = np.where(h == 0)[0]
    h[z[rng.integers(len(z))]] = 0.5
    return h

def slsqp_opt(h0, maxiter=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    r = minimize(lambda x: compute_c5(x), h0, method='SLSQP',
                 bounds=bounds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(r.x), float(r.fun)

def swap_sa(h0, rng, time_limit=5.0):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    bh, bc5 = h.copy(), c5
    t0 = time.time()
    T = 0.015
    while time.time() - t0 < time_limit and T > 1e-7:
        hi = np.where(h > 0.3)[0]
        lo = np.where(h < 0.7)[0]
        if len(hi) == 0 or len(lo) == 0:
            break
        if rng.random() < 0.75:
            i = int(hi[rng.integers(len(hi))])
            j = int(lo[rng.integers(len(lo))])
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            h = project_sum(h)
            nv = compute_c5(h)
            d = nv - c5
            if d <= 0 or rng.random() < np.exp(-d / T):
                c5 = nv
                if c5 < bc5:
                    bc5, bh = c5, h.copy()
            else:
                h[i], h[j] = h[j], h[i]
                h = project_sum(h)
                c5 = compute_c5(h)
        else:
            if len(hi) < 2 or len(lo) < 2:
                T *= 0.999
                continue
            i1 = int(hi[rng.integers(len(hi))])
            i2 = int(hi[rng.integers(len(hi))])
            j1 = int(lo[rng.integers(len(lo))])
j2 = int(lo[rng.integers(len(lo))])
            if len({i1, i2, j1, j2}) < 4:
                continue
            sv = (h[i1], h[i2], h[j1], h[j2])
            h[i1], h[i2], h[j1], h[j2] = sv[2], sv[3], sv[0], sv[1]
            h = project_sum(h)
            nv = compute_c5(h)
            d = nv - c5
            if d <= 0 or rng.random() < np.exp(-d / T):
                c5 = nv
                if c5 < bc5:
                    bc5, bh = c5, h.copy()
            else:
                h[i1], h[i2], h[j1], h[j2] = sv
                h = project_sum(h)
                c5 = compute_c5(h)
        T *= 0.9995
    return bh, bc5

def slsqp_loop(h0, rng, time_limit=60.0):
    n = len(h0)
    bh, bc5 = h0.copy(), compute_c5(h0)
    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limit:
        s = 0.03 + 0.06 * rng.random() + 0.005 * stall
        s = min(s, 0.25)
        hp = project_sum(bh + rng.normal(0, s, n))
        h, v = slsqp_opt(hp, 400)
        v = compute_c5(h)
        if v < bc5 - 1e-10:
            bc5, bh, stall = v, h.copy(), 0
        else:
            stall += 1
            if stall > 40:
                break
    return bh, bc5

def structured_binary(n, rng):
    inits = []
    inits.append(make_binary(n, rng))
    for bs in [3, 4, 5, 6, 7, 8, 10, 11]:
        h = np.zeros(n)
        p = 0
        fill = 1
        while p < n:
            h[p:p + bs] = fill
            p += bs
            fill = 1 - fill
        h = project_sum(h)
        inits.append(h.copy())
    h = np.zeros(n)
    step = max(2, n // (n // 4 + 1))
    h[np.arange(0, n, step)[:n // 2]] = 1.0
    inits.append(project_sum(h))
    return inits

def exhaustive_halves(h_bin, n):
    """Given binary with n-1 entries set, try 0.5 in each zero slot."""
    best_v, best_h = 1e9, None
    zeros = np.where(h_bin < 0.01)[0]
    for z in zeros:
        h = h_bin.copy()
        h[z] = 0.5
        h = project_sum(h)
        v = compute_c5(h)
        if v < best_v:
            best_v, best_h = v, h.copy()
    return best_h, best_v

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    best_h, best_c5 = None, 1.0

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        init = np.asarray(globals()['initial_h_values'], dtype=float).copy()
        init = project_sum(init)
        best_h = init.copy()
        best_c5 = compute_c5(init)
    if best_h is None:
        best_h = make_half(45, rng)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish parent
    if time.time() - ts < budget_s - 200:
        h, v = slsqp_opt(best_h, 1200)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    # Phase 2: SLSQP with perturbations
    rem = budget_s - (time.time() - ts) - 100
    if rem > 30:
        h, v = slsqp_loop(best_h, rng, min(150, rem))
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    # Phase 3: Binary swap SA on parent structure (n=45)
    rem = budget_s - (time.time() - ts) - 80
    if rem > 20:
        h_bin = np.where(best_h >= 0.5, 1.0, 0.0)
        n = len(best_h)
        if n % 2 == 1:
            h_bin = make_half(n, rng)
            h_bin = (best_h >= 0.5).astype(float)
            h_bin = project_sum(h_bin)
        hb, cb = swap_sa(h_bin, rng, min(35, rem))
        if cb < best_c5 + 0.001:
            h, v = slsqp_opt(hb, 800)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()

    # Phase 4: Try n=44 purely binary
    rem = budget_s - (time.time() - ts) - 60
    if rem > 20:
        for _ in range(4):
            if time.time() - ts > budget_s - 60:
                break
            h44 = make_binary(44, rng)
            hb, cb = swap_sa(h44, rng, min(20, budget_s - (time.time() - ts) - 60))
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()
            if time.time() - ts > budget_s - 60:
                break
            h, v = slsqp_opt(hb, 400)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()

    # Phase 5: Try n=46 purely binary
    rem = budget_s - (time.time() - ts) - 40
    if rem > 15:
        for _ in range(3):
            if time.time() - ts > budget_s - 40:
                break
            h46 = make_binary(46, rng)
            hb, cb = swap_sa(h46, rng, min(15, budget_s - (time.time() - ts) - 40))
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()
            if time.time() - ts > budget_s - 40:
                break
            h, v = slsqp_opt(hb, 300)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()

    # Phase 6: Structured init + SA on n=45
    rem = budget_s - (time.time() - ts) - 20
    if rem > 15:
        for hs in structured_binary(44, rng)[:4]:
            if time.time() - ts > budget_s - 20:
                break
            h45 = make_half(45, rng)
            # Use structured 44 as base, add a half
            h45[:44] = hs
            h45[44] = 0.5
            h45 = project_sum(h45)
            hb, cb = swap_sa(h45, rng, min(10, budget_s - (time.time() - ts) - 20))
            if cb < best_c5 + 0.001:
                h, v = slsqp_opt(hb, 500)
                v = compute_c5(h)
                if v < best_c5:
                    best_c5, best_h = v, h.copy()

    # Phase 7: Final SLSQP
    if time.time() - ts < budget_s - 3:
        h, v = slsqp_opt(best_h, 1500)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    best_h = project_sum(best_h)
    final_v = compute_c5(best_h)
    return best_h, min(best_c5, final_v), len(best_h)
