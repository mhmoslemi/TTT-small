    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))

def to_binary(h):
    """Convert to binary (or binary+half for odd n) while preserving sum."""
    n = len(h)
    target = n * 0.5
    # Round to 0 or 1
    h_b = np.where(h >= 0.5, 1.0, 0.0)
    # Adjust to get correct sum
    s = h_b.sum()
    if n % 2 == 0:
        # Need exactly n/2 ones
        excess = s - n//2
        if excess > 0:
            # Remove 'excess' ones (from positions with highest h)
            ones = np.where(h_b == 1.0)[0]
            ones_sorted = ones[np.argsort(-h[ones])]
            for k in range(int(excess)):
                h_b[ones_sorted[k]] = 0.0
        elif excess < 0:
            zeros = np.where(h_b == 0.0)[0]
            zeros_sorted = zeros[np.argsort(-h[zeros])]
            for k in range(int(-excess)):
                h_b[zeros_sorted[k]] = 1.0

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)

        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def sum_fun(x):
    return float(x.sum() - 0.5 * len(x))

def sum_jac(x):
    return np.ones(len(x))

def slsqp_polish(h0, max_iter=600):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    r = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return r.x, float(r.fun)

def swap_ls(h, rng, time_limit=2.0):
    n = len(h)
    t0 = time.time()
    c = compute_c5(h)
    stall = 0
    while time.time() - t0 < time_limit and stall < 80:
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for a in range(n):
            for b in range(a + 1, n):
                i, k = order[a], order[b]
                if abs(h[i] - h[k]) < 1e-8:
                    continue
                sv1, sv2 = h[i], h[k]
                h[i], h[k] = sv2, sv1
                nc = compute_c5(h)
                if nc < c - 1e-14:
                    c = nc
                    improved = True
                    order = list(range(n))
                    rng.shuffle(order)
                    break
                else:
                    h[i], h[k] = sv1, sv2
            if improved:
                break
        if not improved:
            stall += 1
    return h, c

    bc = compute_c5(bh)

        if cc < bc - 1e-12:
            bc = cc
            bh = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

            if i != k and abs(cur[i] - cur[k]) > 1e-8:

    return bh, bc
