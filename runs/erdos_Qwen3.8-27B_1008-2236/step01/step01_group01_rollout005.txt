def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)

    n = len(x)

    return float(x.sum() - 0.5 * len(x))

    return np.ones(len(x))

    """Log-sum-exp approximation of max correlation."""
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = np.max(c)
    if cm <= 0:
        return 0.0
    s = c - cm
    return cm + np.log(np.sum(np.exp(alpha * s))) / alpha

    return compute_c5(x)

    r = minimize(lambda x: lse_objective(x, alpha), h0, method='SLSQP',
                 bounds=bounds, constraints=cons,
