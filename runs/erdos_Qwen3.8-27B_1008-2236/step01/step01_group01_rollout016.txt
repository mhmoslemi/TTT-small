import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    for _ in range(40):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        idx = np.nonzero(h < 1.0 if d > 0 else h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def subgrad_vec(h, nu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cs = c - c.max()
    w = np.exp(nu * cs)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            m = -lag
            grad[m:] -= wk * h[:n - m]
            grad[:n - m] += wk * (1.0 - h[m:])
    return grad * dx, float(c.max())

def obj_func(x):
    return compute_c5(x)

def eq_fun(x):
    return float(x.sum() - 0.5 * len(x))

def eq_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, miter):
    n = len(h0)
    bnds = list(zip(np.zeros(n), np.ones(n)))
    cns = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    r = minimize(obj_func, h0, method='SLSQP', bounds=bnds,
                 constraints=cns, options={'maxiter': miter, 'ftol': 1e-14})
    return r.x, float(r.fun)

def do_subgrad(h0, nsteps, nu0, nu_end, tdeadline):
    h = h0.copy()
    bh = h.copy()
    bc5 = compute_c5(h)
    for s in range(nsteps):
        if time.time() > tdeadline:
            break
        nu = nu0 + (nu_end - nu0) * s / max(nsteps, 1)
        g, _ = subgrad_vec(h, nu)
        lr = 0.004 / (1.0 + 0.0003 * s)
        h = project_sum(h - lr * g)
        c5 = compute_c5(h)
        if c5 < bc5 - 1e-12:
            bc5 = c5
            bh = h.copy()
    return bh, bc5

def do_binary_ls(h0, rng, tdeadline):
    n = len(h0)
    h = np.where(h0 > 0.5, 1.0, 0.0)
    h = project_sum(h)
    c5 = compute_c5(h)
    bh = h.copy()
    bc5 = c5
    for it in range(3000):
        if time.time() > tdeadline:
            break
        ones = np.flatnonzero(h >= 1.0 - 1e-9)
        zeros = np.flatnonzero(h <= 1e-9)
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(ones[rng.integers(len(ones))])
        while j == i:
            j = int(ones[rng.integers(len(ones))])
        k = int(zeros[rng.integers(len(zeros))])
        m = int(zeros[rng.integers(len(zeros))])
        while m == k or m == i or m == j:
            m = int(zeros[rng.integers(len(zeros))])
        h[i], h[j] = 0.0, 0.0
        h[k], h[m] = 1.0, 1.0
        h = project_sum(h)
        nc5 = compute_c5(h)
        if nc5 <= bc5 + 1e-12:
            if nc5 < bc5:
                bc5 = nc5
                bh = h.copy()
        else:
            h = bh.copy()
    return bh, bc5

def make_init(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.5
    h += rng.normal(0, 0.15, n)
    return project_sum(h)

def make_init_even(n, rng):
    h = np.zeros(n)
    np.fill(h[::2], 1.0)
    return project_sum(h)

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = 999.0

    ph = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        ph = np.asarray(globals()['initial_h_values'], dtype=float).copy()
        ph = project_sum(ph)
        c5p = compute_c5(ph)
        best_h = ph.copy()
        best_c5 = c5p

    n_list = [44, 45, 48, 40, 50, 36, 54, 60]

    for n in n_list:
        if budget_s - (time.time() - ts) < 70:
            break
        if ph is not None and len(ph) == n:
            h0 = ph.copy()
        else:
            h0 = make_init(n, rng)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
        rem = budget_s - (time.time() - ts)

        if rem > 120:
            try:
                hs, vs = do_slsqp(h0, 350)
                hs = project_sum(hs)
                vs = compute_c5(hs)
                if vs < best_c5:
                    best_c5 = vs
                    best_h = hs.copy()
                if vs < c5_0:
                    h0, c5_0 = hs, vs
            except Exception:
                pass

        rem = budget_s - (time.time() - ts)
        if rem > 80:
            td = time.time() + min(rem * 0.25, 120)
            hsub, vsub = do_subgrad(h0, 40000, 25, 200, td)
            if vsub < best_c5:
                best_c5 = vsub
                best_h = hsub.copy()
            if vsub < c5_0:
                h0, c5_0 = hsub, vsub

        rem = budget_s - (time.time() - ts)
        if rem > 50 and n % 2 == 0:
            td = time.time() + min(rem * 0.25, 60)
            hb, vb = do_binary_ls(h0, rng, td)
            if vb < best_c5:
                best_c5 = vb
                best_h = hb.copy()
            if vb < c5_0:
                h0, c5_0 = hb, vb
            try:
                hs2, vs2 = do_slsqp(h0, 400)
                hs2 = project_sum(hs2)
                vs2 = compute_c5(hs2)
                if vs2 < best_c5:
                    best_c5 = vs2
                    best_h = hs2.copy()
                if vs2 < c5_0:
                    h0, c5_0 = hs2, vs2
            except Exception:
                pass

        rem = budget_s - (time.time() - ts)
        if rem > 30:
            td = time.time() + min(rem * 0.3, 80)
            hsub2, vsub2 = do_subgrad(h0, 30000, 60, 400, td)
            if vsub2 < best_c5:
                best_c5 = vsub2
                best_h = hsub2.copy()

        rem = budget_s - (time.time() - ts)
        if rem > 20:
            try:
                hf, vf = do_slsqp(h0, 500)
                hf = project_sum(hf)
                vf = compute_c5(hf)
                if vf < best_c5:
                    best_c5 = vf
                    best_h = hf.copy()
            except Exception:
                pass

    if best_h is not None:
        rem = budget_s - (time.time() - ts)
        if rem > 30:
            tf = time.time() + min(rem * 0.4, 200)
            hf2, vf2 = do_subgrad(best_h, 60000, 80, 500, tf)
            if vf2 < best_c5:
                best_c5 = vf2
                best_h = hf2
            rem = budget_s - (time.time() - ts)
        if rem > 10:
            try:
                hf3, vf3 = do_slsqp(best_h, 2000)
                hf3 = project_sum(hf3)
                vf3 = compute_c5(hf3)
                if vf3 < best_c5:
                    best_c5 = vf3
                    best_h = hf3
            except Exception:
                pass

    if best_h is None:
        best_h = make_init(45, rng)
        best_c5 = compute_c5(best_h)

    best_h = project_sum(best_h)
    fc5 = compute_c5(best_h)
    return best_h, min(fc5, best_c5), len(best_h)
