def coord_minimize_1d(b, a):
    """Minimize max_p (b[p] + a[p]*x) over x in [0,1]. Returns (x*, f(x*))."""
    n = len(b)
    val_0 = float(np.max(b))
    val_1 = float(np.max(b + a))

        v1 = (1.0 - h_work[idx1]) if 0 <= idx1 < n else 1.0
        v2 = h_work[idx2] if 0 <= idx2 < n else 0.0
        a[p] = dx * (v1 - v2)
    return a

def coordinate_descent(h, n, max_sweeps, t_limit):
    h_work = h.copy()
    best_c5 = evaluate_c5(h_work)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        corr_val = full_corr_scaled(h_work)
        for i in range(n):
            if time.time() > t_limit:
                return h_work, best_c5
            old = h_work[i]
            a = compute_derivs(h_work, i, n, dx)
            b_arr = corr_val - a * old
            x_opt = coord_min_1d(b_arr, a)
            if abs(x_opt - old) > 1e-13:
                corr_val += a * (x_opt - old)
                h_work[i] = x_opt
                new_c5 = float(np.max(corr_val))
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    improved = True
                else:
                    h_work[i] = old
                    corr_val -= a * (x_opt - old)
        if not improved:
            break
    return h_work, best_c5

def pair_move_descent(h_work, corr_val, n, dx, t_limit, max_pairs=200):
    best_c5 = float(np.max(corr_val))
    improved = False
    for attempt in range(max_pairs):
        if time.time() > t_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        a_i = compute_derivs(h_work, i, n, dx)
        a_j = compute_derivs(h_work, j, n, dx)
        c = a_i - a_j
        b_arr = corr_val - a_i * h_work[i]
        lo_d = -h_work[i]
        hi_d = 1.0 - h_work[j]
        if lo_d >= hi_d:
            continue
        # Minimize max(b_arr + c*delta) over delta in [lo_d, hi_d]
        # Check boundaries
        v_lo = float(np.max(b_arr + c * lo_d))
        v_hi = float(np.max(b_arr + c * hi_d))
        # Subgradient at lo_d
        act_lo = (b_arr + c * lo_d) >= v_lo - 1e-11
        if np.min(c[act_lo]) >= -1e-10:
            continue
        # Subgradient at hi_d
        act_hi = (b_arr + c * hi_d) >= v_hi - 1e-11
        if np.max(c[act_hi]) <= 1e-10:
            continue
        # Bisection
        dlo, dhi = lo_d, hi_d
        for _ in range(40):
            dm = (dlo + dhi) * 0.5
            vals = b_arr + c * dm
            vm = float(np.max(vals))
            act = vals >= vm - 1e-11
            if float(np.min(c[act])) >= -1e-12:
                dhi = dm
            else:
                dlo = dm
        d_opt = (dlo + dhi) * 0.5
        new_val = float(np.max(b_arr + c * d_opt))
        if new_val < best_c5 - 1e-13:
            h_work[i] += d_opt
            h_work[j] -= d_opt
            corr_val += c * d_opt
            best_c5 = new_val
            improved = True
    return h_work, corr_val, best_c5, improved

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    if temp < 1e-12:
        return m
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def smooth_obj_wrapper(h):
    return smooth_obj(h, _sm_temp)

def sum_constraint(h):
    return h.sum() - _sm_n / 2.0

def sum_constraint_jac(h):
    return np.ones(_sm_n)

_sm_temp = 0.005
_sm_n = 45

def slsqp_polish(h, n, t_limit):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    global _sm_temp, _sm_n
    _sm_n = n
    bounds = tuple((0.0, 1.0) for _ in range(n))
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_limit:
            break
        _sm_temp = temp
        try:
            res = minimize(smooth_obj_wrapper, best_h, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': sum_constraint,
                                        'jac': sum_constraint_jac}],
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    _sm_temp = 0.001
    if time.time() < t_limit:
        try:
            res = minimize(smooth_obj_wrapper, best_h, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': sum_constraint,
                                        'jac': sum_constraint_jac}],
                          options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Coordinate descent from initial
    t1 = t0 + budget_s * 0.15
    h_cd, c5_cd = coordinate_descent(h0, n, max_sweeps=500, t_limit=t1)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd

    # Phase 2: Simulated annealing restarts with coordinate descent
    t2 = t0 + budget_s * 0.70
    n_restarts = 80
    for r in range(n_restarts):
        if time.time() > t2:
            break
        progress = r / max(n_restarts, 1)
        scale = 0.4 * (1.0 - progress) * max(0.1, 1.0 - progress * 0.8)

        strat = r % 5
        if strat == 0:
            pert = np.random.randn(n) * scale
        elif strat == 1:
            pert = np.random.randn(n) * scale * 2
        elif strat == 2:
            idx = np.random.choice(n, size=max(1, n // 3), replace=False)
            pert = np.zeros(n)
            pert[idx] = np.random.choice([-1, 1], size=len(idx)) * scale
        elif strat == 3:
            perm = np.random.permutation(n)
            pert = (np.random.permutation(n) / (n - 1) - 0.5) * scale * 3
            pert = [pert[perm[i]] for i in range(n)]
            pert = np.array(pert)
        else:
            pert = np.zeros(n)
            n_swap = max(1, n // 5)
            for _ in range(n_swap):
                ii = np.random.randint(n)
                jj = np.random.randint(n)
                d = np.random.uniform(-scale, scale)
                pert[ii] += d
                pert[jj] -= d

        h_try = fix_sum(best_h + pert, n)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

        h_r, c5_r = coordinate_descent(h_try, n, max_sweeps=150, t_limit=t2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

        # Pair moves
        corr_val = full_corr_scaled(best_h)
        h_p, corr_p, c5_p, _ = pair_move_descent(
            best_h, corr_val, n, 2.0 / n, t2, max_pairs=100)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

        # Another CD round after pair moves
        h_r2, c5_r2 = coordinate_descent(best_h, n, max_sweeps=50, t_limit=t2)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2

    # Phase 3: SLSQP polish
    t3 = t0 + budget_s * 0.88
    h_sl, c5_sl = slsqp_polish(best_h, n, t3)
    if c5_sl < best_c5:
        best_c5 = c5_sl
        best_h = h_sl

    # Phase 4: Final coordinate descent + pair moves
    t4 = t0 + budget_s * 0.97
    h_f, c5_f = coordinate_descent(best_h, n, max_sweeps=1000, t_limit=t4)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f

    corr_val = full_corr_scaled(best_h)
    h_p, corr_p, c5_p, _ = pair_move_descent(
        best_h, corr_val, n, 2.0 / n, t0 + budget_s * 0.99, max_pairs=500)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p

    h_f2, c5_f2 = coordinate_descent(best_h, n, max_sweeps=200,
                                     t_limit=t0 + budget_s * 0.995)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2

    best_h = fix_sum(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n)
