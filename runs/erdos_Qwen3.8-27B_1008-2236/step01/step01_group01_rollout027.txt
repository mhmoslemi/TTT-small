    # Phase 6: Binary swap on best
    if time.time() - ts < budget_s - 20:
        hb, vb = binary_swap_search(bh, rng, min(5.0, budget_s - (time.time() - ts) - 15))
        if vb < bc5 - 1e-10:
            bc5, bh = vb, hb.copy()

    # Phase 7: Final polish
    if time.time() - ts < budget_s - 5:
        rem_iters = max(50, int((budget_s - (time.time() - ts) - 5) * 30))
        h, v = slsqp_smooth(bh, 250, min(rem_iters, 500))
        h = project_sum(h)
        v = compute_c5(h)
        if v < bc5:
            bc5, bh = v, h.copy()

    bh = project_sum(bh)
    final = compute_c5(bh)
    return bh, min(final, bc5), len(bh)
