import cvxpy as cp
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((n, m))
    for j in range(n):
        for k in range(-(n-1), n):
            m_idx = k + n - 1
            g = 0.0
            jp = j + k
            jm = j - k
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            if 0 <= jm < n:
                g -= h[jm]
            G[j, m_idx] = g
    return G

def slp_step(h0, rho, solver='ECOS'):
    n = len(h0)
    m = 2 * n - 1
    c = compute_corr_vec(h0)
    G = compute_grad_matrix(h0)
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Linearized constraints: c[m] + G[:,m] @ (h - h0) <= t
    # => G[:,m] @ h <= t - c[m] + G[:,m] @ h0
    constraints = []
    for mi in range(m):
        rhs = -c[mi] + G[:, mi] @ h0
        constraints.append(G[:, mi] @ h <= t + rhs)
    
    # Trust region: ||h - h0|| <= rho
    constraints.append(cp.norm(h - h0) <= rho)
    
    # Sum constraint
    constraints.append(cp.sum(h) == n * 0.5)
    
    # Box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver)
    
    if h.value is None:
        return h0, rho
    h_new = np.clip(h.value, 0, 1)
    h_new = project_sum(h_new)
    return h_new, rho

        return float(c.max())

        return float(x.sum() - 0.5 * len(x))
    
    def sjac(x):
        return np.ones(len(x))
    
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_sum(r.x), float(r.fun)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def sum_fun(x):
    return float(x.sum() - 0.5 * len(x))

def sum_jac(x):
    return np.ones(len(x))

def lse_obj(x, M):
    c = corr_vec(x)
    Mc = M * c
    Mc = Mc - Mc.max()
    return float((M * c.max() + np.log(np.sum(np.exp(Mc)))) / M)

def true_max_obj(x):
    return compute_c5(x)

def run_slsqp(h0, M, maxiter, bounds, cons):
    obj = true_max_obj if M <= 0 else (lambda x: lse_obj(x, M))
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return r.x, compute_c5(r.x)

def continuation_opt(h0, n, bounds, cons, Ms, iters_per_M, time_left):
    h = h0.copy()
    for M in Ms:
        if time.time() > time_left:
            break
        h_new, _ = run_slsqp(h, M, iters_per_M, bounds, cons)
        h_new = project_sum(h_new)
        if np.all(np.isfinite(h_new)) and compute_c5(h_new) < 1.0:
            h = h_new
    return h

    m = 2 * n - 1
    c_all = corr_vec(h)
    top = np.argsort(c_all)[-top_k:]
    G = np.zeros((n, len(top)))
    for j in range(n):
        for ii, mi in enumerate(top):
            lag = int(mi) - (n - 1)
            g = 0.0
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            if 0 <= jm < n:
                g -= h[jm]
            G[j, ii] = g
    c_top = c_all[top]
    # Gradient descent on the top lags with projection
    grad = G.T @ np.ones(len(top)) / len(top) + G.T @ (c_top - c_top.mean()) / (c_top.max() - c_top.min() + 1e-10)
