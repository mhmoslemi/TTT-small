        constraints.append(cp.sum(h[idx_h] * w) * dx <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.GLOP)
    
    if h.value is not None:
        return np.asarray(h.value).flatten(), t.value
    return h_fixed.copy(), None

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            free = np.where(h < 1.0 - 1e-10)[0]
        else:
            free = np.where(h > 1e-10)[0]
        if len(free) == 0:
            break
        h[free] += d / len(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_linearized(h_fixed):
    n = len(h_fixed)
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == n * 0.5]
    for L in range(-(n - 1), n):
        i_s = max(0, -L)
        i_e = min(n - 1, n - 1 - L)
        if i_s > i_e:
            continue
        idx_h = np.arange(i_s + L, i_e + 1 + L)
        w = 1.0 - h_fixed[i_s:i_e + 1]
        cons.append(cp.sum(h[idx_h] * w) * dx <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                return None, None
    if h.value is None:
        return None, None
    return np.asarray(h.value).flatten(), float(t.value) if t.value is not None else None

def lp_iteration(h, max_iters=300, damp=0.6, thresh=1e-9):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        h_lp, t_lp = solve_lp_linearized(h_cur)
        if h_lp is None:
            break
        h_lp = project_sum(h_lp)
        c_lp = compute_c5(h_lp)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp.copy()
        h_new = (1.0 - damp) * h_cur + damp * h_lp
        h_new = project_sum(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if abs(c_new - compute_c5(h_cur)) < thresh and c_new >= best_c:
            break
        h_cur = h_new
    return best_h, best_c

def pair_swap_search(h, rng, time_limit=15.0):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    stuck = 0
    while time.time() - t0 < time_limit and stuck < 200:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        vi, vj = h[i], h[j]
        if abs(vi - vj) < 1e-8:
            continue
        h[i], h[j] = vj, vi
        c = compute_c5(h)
        if c <= best + 1e-12:
            best = c
            stuck = 0
        else:
            h[i], h[j] = vi, vj
            stuck += 1
    return h, best

def fractional_local(h, rng, time_limit=20.0):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            old = h[i]
            for trial in (0.0, 1.0, 0.25, 0.5, 0.75):
                if abs(trial - old) < 1e-7:
                    continue
                h[i] = trial
                h = project_sum(h)
                c = compute_c5(h)
                if c < best - 1e-10:
                    best = c
                    improved = True
                    break
                else:
                    h[i] = old
                    h = project_sum(h)
        if not improved:
            break
    return h, best

def slsqp_refine(h0, maxiter=400):
    n = len(h0)

    def sfun(x):
        return float(x.sum() - 0.5 * len(x))

    def sjac(x):
        return np.ones(len(x))

    def c5obj(x):
        dx = 2.0 / len(x)
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        return float(c.max())

    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_sum(r.x), compute_c5(project_sum(r.x))

            h = project_sum(h)

    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, len(best_h)
