import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_obj_val(h, temp):
    c = corr_scaled(h)
    m = np.max(c)
    if temp < 1e-12:
        return m
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = corr_scaled(h)
    m = np.max(c)
    g = np.zeros(n)
    if temp < 1e-12:
        idx = int(np.argmax(c))
        lag = idx - (n - 1)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
        else:
            M = -lag
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
        return g * dx
    w = np.exp((c - m) / temp)
    w /= np.sum(w)
    for ki in range(2 * n - 1):
        if w[ki] < 5e-7:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            g[lag:] += w[ki] * (1.0 - h[:n-lag])
            g[:n-lag] -= w[ki] * h[lag:]
        else:
            M = -lag
            g[:n-M] += w[ki] * (1.0 - h[M:])
            g[M:] -= w[ki] * h[:n-M]
    return g * dx

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_from(h0, n, t_limit, temps):
    target = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h_cur = h0.copy()
    bounds = tuple((0.0, 1.0) for _ in range(n))
    for temp in temps:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj_val(x, temp), h_cur,
                          jac=lambda x: smooth_obj_grad(x, temp),
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 500, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_sum(res.x, target)
                c = evaluate_c5(h_new)
                if c < best_c:
                    best_c = c
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def coord_polish(h0, n, t_limit, max_sweeps=200):
    target = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            old = best_h[j]
            # Try values: 0, 1, midpoint, and small perturbations
            candidates = np.concatenate([[0.0, 1.0, old, (old+0)/2, (old+1)/2],
                                         np.clip(old + np.array([-0.3,-0.1,0.1,0.3]), 0, 1)])
            for t in candidates:
                best_h[j] = t
                best_h = project_sum(best_h, target)
                c = evaluate_c5(best_h)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    old = t
                    break
            if not improved:
                best_h[j] = old
                best_h = project_sum(best_h, target)
        if not improved:
            break
    return best_h, best_c

def subgrad_phase(h0, n, t_limit, n_iter=500):
    target = n / 2.0
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        lr = 0.05 / (1.0 + 0.01 * it)
        g = smooth_obj_grad(h_cur, 0.0)
        h_try = project_sum(h_cur - lr * g, target)
        c = evaluate_c5(h_try)
        if c < best_c:
            best_c = c
            best_h = h_try.copy()
        h_cur = h_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)
    
    n = len(h0)
    h0 = project_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Phase 1: SLSQP from initial (30% budget)
    t1 = t0 + budget_s * 0.30
    best_h, best_c = slsqp_from(h0, n, t1, temps)
    
    # Phase 2: Coordinate polish (15% budget)
    t2 = t0 + budget_s * 0.45
    best_h, best_c = coord_polish(best_h, n, t2)
    
    # Phase 3: SLSQP again from polished point (25% budget)
    t3 = t0 + budget_s * 0.70
    best_h, best_c = slsqp_from(best_h, n, t3, temps[2:])
    
    # Phase 4: Subgradient for diversification (10% budget)
    t4 = t0 + budget_s * 0.80
    for s in range(5):
        if time.time() > t4:
            break
        h_try = project_sum(best_h + np.random.randn(n) * 0.1, n / 2.0)
        h_sub, c_sub = subgrad_phase(h_try, n, t4, 200)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub.copy()
    
    # Phase 5: Final SLSQP + polish (15% budget)
    t5 = t0 + budget_s * 0.95
    best_h, best_c = slsqp_from(best_h, n, t5, temps[3:])
    final_t = t_end - 2
    best_h, best_c = coord_polish(best_h, n, final_t)
    
    # Final SLSQP squeeze
    if time.time() < final_t - 5:
        best_h, best_c = slsqp_from(best_h, n, final_t, temps[4:])
    
    # Ensure feasibility
    best_h = project_sum(best_h, n / 2.0)
    final_c = evaluate_c5(best_h)
    return (best_h, final_c, n)

    if t < 1e-12:

            g[:n - lag] -= h[lag:]

        best_h, best_c = do_slsqp(best_h, t_end - 8, [0.0005, 0.0003, 0.0001], maxiter=500)
        if time.time() < t_end - 3:
            best_h, best_c = coord_polish(best_h, t_end - 2, max_sweeps=80)

    best_h = project_sum(best_h, n / 2.0)
    final_c = evaluate_c5(best_h)
    return (best_h, final_c, n)
