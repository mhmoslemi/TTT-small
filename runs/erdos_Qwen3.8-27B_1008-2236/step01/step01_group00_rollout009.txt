    # h[i] -= delta, h[j] += delta
    # 0 <= delta <= h[i]
    # h[j] + delta <= 1, so delta <= 1 - h[j]

    if max_d < 1e-10:
        return 0.0
    
    # Ternary search for optimal delta in [0, max_d]
    lo, hi = 0.0, max_d
    for _ in range(30):

        m2 = hi - (hi - lo) / 3

        f2 = evaluate_c5(h)
        if f1 < f2:
            hi = m2
        else:

    delta = (lo + hi) / 2

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def fix_sum(h, n):
    h = h.copy()
    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_run(h0, n, tlim):
    bh, bc = h0.copy(), evaluate_c5(h0)
    hc = bh.copy()

        if time.time() > tlim:
            break

            r = minimize(lambda h: lse_c5(h, t), hc, method='SLSQP',
                        bounds=bd, constraints=ct, options={'maxiter': 250, 'ftol': 1e-14})

                hn, c5 = fix_sum(r.x, n), evaluate_c5(fix_sum(r.x, n))
                if c5 < bc:
                    bc, bh, hc = c5, hn, hn.copy()

            pass

                c5 = evaluate_c5(hn)
                if c5 < bc:
                    bc, bh, hc = c5, hn, hn.copy()
        except:
            pass
    return bh, bc

    bh, bc = h0.copy(), evaluate_c5(h0)
    h = bh.copy()

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return float(m + temp * np.log(np.sum(np.exp((c - m) / temp))))

def fix_sum(h, n):
    h = h.copy()
    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_run(h0, n, tlim):
    bh, bc = h0.copy(), evaluate_c5(h0)
    hc = bh.copy()
    bd = [(0.0, 1.0)] * n
    ct = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
           'jac': lambda h: np.ones(n)}]
    for t in [0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001, 0.00005]:
        if time.time() > tlim:
            break
        try:
            r = minimize(lambda h: lse_c5(h, t), hc, method='SLSQP',
                         bounds=bd, constraints=ct,
                         options={'maxiter': 250, 'ftol': 1e-14})
            if r.x is not None:
                hn = fix_sum(r.x, n)
                c5 = evaluate_c5(hn)
                if c5 < bc:
                    bc, bh, hc = c5, hn, hn.copy()
        except Exception:
            pass
    if time.time() < tlim:
        try:
            r = minimize(evaluate_c5, hc, method='SLSQP',
                         bounds=bd, constraints=ct,
                         options={'maxiter': 150, 'ftol': 1e-14})
            if r.x is not None:
                hn = fix_sum(r.x, n)
                c5 = evaluate_c5(hn)
                if c5 < bc:
                    bc, bh = c5, hn
        except Exception:
            pass
    return bh, bc

def lbfgsb_run(h0, n, tlim):
    bh, bc = h0.copy(), evaluate_c5(h0)
    hc = bh.copy()
    bd = [(0.0, 1.0)] * n
    for pen, tmp in [(5e3, 0.01), (5e4, 0.003), (5e5, 0.001)]:
        if time.time() > tlim:
            break
        try:
            r = minimize(lambda h: lse_c5(h, tmp) + pen * (h.sum() - n / 2.0) ** 2,
                         hc, method='L-BFGS-B', bounds=bd,
                         options={'maxiter': 500, 'ftol': 1e-15})
            if r.x is not None:
                hn = fix_sum(r.x, n)
                c5 = evaluate_c5(hn)
                if c5 < bc:
                    bc, bh, hc = c5, hn, hn.copy()
        except Exception:
            pass
    return bh, bc

def pair_polish(h0, n, tlim):
    bh, bc = h0.copy(), evaluate_c5(h0)
    h = bh.copy()
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    while time.time() < tlim:
        improved = False
        for i, j in pairs:
            if time.time() > tlim:
                return bh, bc
            md = min(h[i], 1.0 - h[j])
            if md < 1e-9:
                continue
            lo, hi = 0.0, md
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i], h[j] = h[i] - m1, h[j] + m1
                f1 = evaluate_c5(h)
                h[i], h[j] = h[i] - m2, h[j] + m2
                f2 = evaluate_c5(h)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            bd2 = (lo + hi) / 2.0
            h[i], h[j] = h[i] - bd2, h[j] + bd2
            c5 = evaluate_c5(h)
            if c5 < bc - 1e-14:
                bc, bh = c5, h.copy()
                improved = True
            else:
                h[:] = bh
        if not improved:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n)

    bh, bc = h0.copy(), evaluate_c5(h0)

    # Phase 1: SLSQP from parent
    bh, bc = slsqp_run(h0, n, t0 + budget_s * 0.22)

    # Phase 2: L-BFGS-B refinement
    h2, c2 = lbfgsb_run(bh, n, t0 + budget_s * 0.32)
    if c2 < bc:
        bh, bc = h2, c2

    # Phase 3: Pair polish (exact max, no smoothing)
    bh, bc = pair_polish(bh, n, t0 + budget_s * 0.50)

    # Phase 4: SLSQP from polished point
    bh, bc = slsqp_run(bh, n, t0 + budget_s * 0.65)

    # Phase 5: Diverse restarts with mixed perturbation scales
    t5 = t0 + budget_s * 0.88
    for r in range(30):
        if time.time() > t5:
            break
        if r % 3 == 0:
            s = 0.03
        elif r % 3 == 1:
            s = 0.06
        else:
            s = 0.10 * max(0.3, 1.0 - r / 30.0)
        ht = fix_sum(bh + np.random.randn(n) * s, n)
        ct = evaluate_c5(ht)
        if ct < bc:
            bh, bc = ht, ct
        elif ct < bc + 0.003:
            hr, cr = slsqp_run(ht, n, t5)
            if cr < bc:
                bh, bc = hr, cr

    # Phase 6: Final pair polish + SLSQP
    bh, bc = pair_polish(bh, n, t0 + budget_s * 0.95)
    if time.time() < t0 + budget_s * 0.98:
        bh, bc = slsqp_run(bh, n, t0 + budget_s * 0.985)

    final_c5 = evaluate_c5(bh)
    return (bh, final_c5, n)
