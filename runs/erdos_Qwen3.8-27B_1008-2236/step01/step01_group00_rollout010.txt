import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.copy()
    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = d / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)

        cnt = active.sum()
        if cnt == 0:
            active = np.zeros(2*n-1); active[np.argmax(corr)] = 1; cnt = 1
        grad = np.zeros(n)

                if s2 == 0:

                else:
                    grad[:n-s2] += (1.0 - h[s2:])
                    grad[s2:] -= h[:n-s2]
        return grad * dx / cnt
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)

            if s2 > 0:

                grad[s2:] -= wk * h[:n-s2]

def coord_descent(h, n_sweeps=200, embed_tol=1e-10):
    """Exact coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old_t = h[j]
            # C_k(t) = A_k + B_k * t
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = old_t
            
            A = c0.copy()  # C_k at t=0
            B = c1 - c0   # slope: C_k at t=1 minus C_k at t=0
            
            # Minimize max_k (A[k] + B[k]*t) over t in [0,1]
            # Convex piecewise linear → use bisection on subgradient
            def f_max(t):
                return np.max(A + B * t)
            
            f0 = f_max(0.0)
            f1 = f_max(1.0)
            
            # Check if minimum is at boundary
            # Subgradient at 0: max of B[k] for k in argmax(A)
            active0 = np.where(A >= f0 - 1e-12)[0]
            sg0 = B[active0].max()  # right subgradient at 0
            if sg0 >= 0:
                t_new = 0.0
            else:
                active1 = np.where(A + B >= f1 - 1e-12)[0]
                sg1 = B[active1].min()  # left subgradient at 1
                if sg1 <= 0:
                    t_new = 1.0
                else:
                    # Bisection
                    lo, hi = 0.0, 1.0
                    for _ in range(40):
                        mid = (lo + hi) / 2
                        fm = f_max(mid)
                        # Subgradient at mid
                        active_m = np.where(A + B * mid >= fm - 1e-12)[0]
                        sg_min = B[active_m].min()
                        sg_max = B[active_m].max()
                        if sg_min > 0:
                            lo = mid
                        elif sg_max < 0:
                            hi = mid
                        else:
                            break
                    t_new = (lo + hi) / 2
            
            t_new = np.clip(t_new, 0.0, 1.0)
            if abs(t_new - old_t) > embed_tol:
                h[j] = t_new
                # Don't evaluate full C5 every time (expensive); do at end of sweep
                improved = True
        
        # Project and evaluate at end of sweep
        h = fix_sum(h, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    
    h = fix_sum(h, n)
    return h, evaluate_c5(h)

def slsqp_optimize(h, n, t_limit):
    """SLSQP with decreasing LSE temperature."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    
    bounds = tuple((0.0, 1.0) for _ in range(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                    'jac': lambda x: np.ones(n)}]
    
    for temp in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_limit:
            break
        def obj(x):
            return lse_c5(x, temp)
        def grad(x):
            g = lse_grad(x, temp)
            return g if g is not None else None
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.copy()
    for _ in range(100):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    return float(m + temp * np.log(np.max(corr - m) + temp * np.sum(np.exp((corr - m) / temp)) / np.max(corr - m) + 1e-300)))

def lse_c5_safe(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = (corr - m) / temp
    return float(m + temp * (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))))

def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        active = (corr >= m - 1e-12)
        cnt = active.sum()
        if cnt == 0:
            idx = np.argmax(corr)
            active = np.zeros(2*n-1)
            active[idx] = 1
            cnt = 1
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            s = k - n + 1
            if 0 <= s < n:
                if s == 0:
                    grad += (1.0 - 2.0 * h)
                else:
                    grad[s:] += (1.0 - h[:n-s])
                    grad[:n-s] -= h[s:]
            elif -n < s < 0:
                s2 = -s
                grad[:n-s2] += (1.0 - h[s2:])
                grad[s2:] -= h[:n-s2]
        return grad * dx / cnt
    shifted = (corr - m) / temp
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-16:
            continue
        s = k - n + 1
        if 0 <= s < n:
            if s == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[s:] += wk * (1.0 - h[:n-s])
                grad[:n-s] -= wk * h[s:]
        elif -n < s < 0:
            s2 = -s
            grad[:n-s2] += wk * (1.0 - h[s2:])
            grad[s2:] -= wk * h[:n-s2]
    return grad * dx

def coord_descent(h, n_sweeps=500, check_interval=50):
    n = len(h)
    h = h.copy()
    best_c5 = evaluate_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old_t = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = old_t
            A = c0
            B = c1 - c0
            f0 = np.max(A)
            f1 = np.max(A + B)
            active0 = np.where(A >= f0 - 1e-12)[0]
            sg_right = B[active0].max()
            if sg_right >= -1e-14:
                t_new = 0.0
            else:
                active1 = np.where(A + B >= f1 - 1e-12)[0]
                sg_left = B[active1].min()
                if sg_left <= 1e-14:
                    t_new = 1.0
                else:
                    lo, hi = 0.0, 1.0
                    for _ in range(50):
                        mid = (lo + hi) * 0.5
                        fm = np.max(A + B * mid)
                        act = np.where(A + B * mid >= fm - 1e-12)[0]
                        if B[act].max() < 0:
                            hi = mid
                        elif B[act].min() > 0:
                            lo = mid
                        else:
                            break
                    t_new = (lo + hi) * 0.5
            t_new = np.clip(t_new, 0.0, 1.0)
            if abs(t_new - old_t) > 1e-10:
                h[j] = t_new
                improved = True
        if sweep % check_interval == 0 or sweep == n_sweeps - 1:
            h = fix_sum(h, n)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
        if not improved:
            break
    h = fix_sum(h, n)
    return h, evaluate_c5(h)

def slsqp_optimize(h, n, t_limit, temps=None):
    if temps is None:
        temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    bounds = tuple((0.0, 1.0) for _ in range(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    for temp in temps:
        if time.time() > t_limit:
            break
        def obj(x):
            return lse_c5_safe(x, temp)
        def grd(x):
            return lse_grad(x, temp)
        try:
            res = minimize(obj, h_cur, jac=grd, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 400, 'ftol': 1e-14, 'maxls': 50})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = fix_sum(np.clip(res.x, 0, 1), n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    elif 'initial_h_values' in globals() and globals().get('initial_h_values') is not None:

    if h0 is None:
        h0 = np.full(45, 0.5)

    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Heavy coordinate descent
    t_lim1 = t0 + budget_s * 0.25
    while time.time() < t_lim1:

    # Phase 2: SLSQP with LSE smoothing
    t_lim2 = t0 + budget_s * 0.50
    best_h, best_c5 = slsqp_optimize(best_h, n, t_lim2)

    # Phase 3: More coordinate descent
    t_lim3 = t0 + budget_s * 0.70
    while time.time() < t_lim3:
        best_h, best_c5 = coord_descent(best_h, n_sweeps=300)

    # Phase 4: Diverse restarts
    t_lim4 = t0 + budget_s * 0.88
    n_restarts = 30
    for r in range(n_restarts):
        if time.time() > t_lim4:
            break
        if r % 5 == 0:
            pert = np.random.randn(n) * 0.12
        elif r % 5 == 1:
            k = r + 1
            pert = np.sin(2.0 * np.pi * np.arange(n) / n * k) * 0.15
        elif r % 5 == 2:

        elif r % 5 == 3:

        else:
            pert = np.random.randn(n) * 0.08
        h_try = np.clip(best_h + pert, 0.0, 1.0)
        h_try = fix_sum(h_try, n)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5 + 0.01:
            h_r, c5_r = coord_descent(h_try, n_sweeps=200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = np.clip(h_r, 0, 1)
                best_h = fix_sum(best_h, n)
                if c5_r > best_c5 + 1e-14:
                    h_r, c5_r = slsqp_optimize(best_h, n, t0 + budget_s * 0.45)
                    if c5_r < best_c5:

                        best_h = np.clip(h_r, 0, 1)

    # Phase 5: Final SLSQP + CD
    t_lim5 = t0 + budget_s * 0.96
    best_h, best_c5 = slsqp_optimize(best_h, n, t_lim5,
                                     temps=[0.01, 0.005, 0.002, 0.001, 0.0005])
    while time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n_sweeps=200)

    best_h = fix_sum(np.clip(best_h, 0.0, 1.0), n)
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n)
