        # Second term: 0 <= j <= n-1-A
        for j in range(0, n - A):

def fix_sum(h, target):
    h = h.copy()

        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:

        cnt = mask.sum()
        if cnt == 0:

def compute_subgrad(h, max_idx):

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    bounds = tuple((0.0, 1.0) for _ in range(n))
    h_cur = best_h.copy()
    for temp in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > t_limit:
            break
        mu = 1e4
        def obj_pen(x):
            return lse_obj(x, temp) + mu * (x.sum() - target) ** 2
        try:
            res = minimize(obj_pen, h_cur, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = fix_sum(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def pair_polish(h0, t_limit, n_trials=50000):
    n = len(h0)
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    deltas = np.array([0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15])
    count = 0
    no_imp = 0
    while count < n_trials and time.time() < t_limit:
        i = int(np.random.randint(n))
        j = int(np.random.randint(n))
        if i == j:
            continue
        moved = False
        for d in deltas:
            if h[i] >= d and h[j] <= 1.0 - d:
                h[i] -= d
                h[j] += d
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    moved = True
                    no_imp = 0
                else:
                    h[i] += d
                    h[j] -= d
                count += 1
            if h[j] >= d and h[i] <= 1.0 - d:
                h[j] -= d
                h[i] += d
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    moved = True
                    no_imp = 0
                else:
                    h[j] += d
                    h[i] -= d
                count += 1
        if not moved:
            no_imp += 1
        else:
            no_imp = 0
        if no_imp > 5000:
            break
    return best_h, best_c5

def make_start(n, kind, seed_off):
    rng = np.random.RandomState(seed_off)
    if kind == 0:
        h = (rng.rand(n) < 0.5).astype(float)
    elif kind == 1:
        x = np.linspace(0, 2, n)
        f = rng.randint(1, 5)
        p = rng.rand() * 2 * np.pi
        h = 0.5 + 0.35 * np.sin(np.pi * f * x + p)
        h = np.clip(h, 0, 1)
    elif kind == 2:
        h = rng.rand(n)
        h = np.clip(h, 0, 1)
    else:
        k = max(2, n // (rng.randint(2, 5) + 1))
        h = np.zeros(n)
        for s in range(0, n, 2 * k):
            h[s:s+k] = 1.0
        h = np.clip(h, 0, 1)
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    target = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Subgradient (12%)
    t1 = t0 + budget_s * 0.12
    h1, c1 = sg_optimize(h0, t1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: SLSQP+LSE from subgradient result (20%)
    t2 = t0 + budget_s * 0.32
    h2, c2 = slsqp_opt(best_h, t2)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: L-BFGS-B with penalty from current best (13%)
    t3 = t0 + budget_s * 0.45
    h3, c3 = lfbgss_opt(best_h, t3)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: SLSQP from original initial (10%)
    t4 = t0 + budget_s * 0.55
    h4, c4 = slsqp_opt(h0, t4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Diverse restarts (20%)
    t5 = t0 + budget_s * 0.75
    np.random.seed(seed + 123)
    for s in range(20):
        if time.time() > t5:
            break
        kind = s % 4
        h_s = make_start(n, kind, seed + 500 + s * 13)
        c_s = evaluate_c5(h_s)
        if c_s < best_c5 + 0.04:
            if s % 2 == 0:
                h_s2, c_s2 = slsqp_opt(h_s, t5)
            else:
                h_s2, c_s2 = lfbgss_opt(h_s, t5)
            if c_s2 < best_c5:
                best_c5, best_h = c_s2, h_s2

    # Phase 6: Pairwise polish (15%)
    t6 = t0 + budget_s * 0.92
    best_h, best_c5 = pair_polish(best_h, t6)

    # Phase 7: Final SLSQP (6%)
    t7 = t0 + budget_s * 0.98
    h7, c7 = slsqp_opt(best_h, t7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7

    best_h = np.clip(fix_sum(best_h, len(best_h) / 2.0), 0.0, 1.0)
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, len(best_h))
