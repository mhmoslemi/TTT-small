def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50, 0)
    return m + temp * float(np.log(np.sum(np.exp(shifted))))

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return float(m)
    s = np.clip((corr - m) / temp, -50, 0)
    return float(m + temp * np.log(np.sum(np.exp(s))))

def fix_sum(h, n):
    h = h.copy()
    for _ in range(500):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def binary_climb(h0, n, t_end, rounds=200, sample=3000):
