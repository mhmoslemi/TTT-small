import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_corr(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50, 0)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            mask = (h < 1.0) if d > 0 else (h > 0.0)
        if mask.sum() == 0:
            break
        step = d / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, t_limit, temps=None):
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_corr(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = np.clip((corr - m) / temp, -50, 0)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_run(h0, n, t_limit):
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    best = h0.copy()
    bc = evaluate_c5(best)
    temps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0, 0.0]
    for temp in temps:
        if time.time() > t_limit:
            break
        obj = evaluate_c5 if temp == 0 else (lambda x, t=temp: soft_max_corr(x, t))
        try:
            r = minimize(obj, best, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
            if r.x is not None:
                hn = fix_sum(r.x, n)
                c = evaluate_c5(hn)
                if c < bc:
                    bc, best = c, hn
        except Exception:
            pass
    return best, bc

def coord_descent(h, n, t_limit, rounds=50):
    bi = h.copy()
    bc = evaluate_c5(bi)
    for _ in range(rounds):
        if time.time() > t_limit:
            break
        imp = False
        for i in range(n):
            old = bi[i]
            s = n / 2.0 - old
            lo = max(0.0, s - (n - 1))
            hi = min(1.0, s)
            if hi <= lo:
                continue
            for t in np.linspace(lo, hi, 11):
                if abs(t - old) < 1e-10:
                    continue
                bi[i] = t
                bi = fix_sum(bi, n)
                c = evaluate_c5(bi)
                if c < bc - 1e-15:
                    bc = c
                    imp = True
                    break
            else:
                bi[i] = old
                bi = fix_sum(bi, n)
        if not imp:
            break
    return bi, bc

def pair_polish(h, n, t_limit, rounds=20):
    bi = h.copy()
    bc = evaluate_c5(bi)
    for _ in range(rounds):
        if time.time() > t_limit:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):

                if mx > 1e-7:
                    for fr in (0.5, 0.2):

                        bi = fix_sum(bi, n)
                        c = evaluate_c5(bi)
                        if c < bc - 1e-14:
                            bc = c

                        bi[i] += d
                        bi[j] -= d

                mx2 = min(bi[j], 1.0 - bi[i])

                    else:

                        bi[i] -= d

        if not imp:
            break
    return bi, bc

def gen_start(n, mode, rng):
    x = np.linspace(0, 2, n)

        h = rng.uniform(0, 1, n)
    else:

    return fix_sum(h, n)

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    return (bh, evaluate_c5(bh), len(bh))

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def soft_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if _GT < 1e-14:
        return m
    s = np.clip((c - m) / _GT, -50, 0)
    return m + _GT * np.log(np.sum(np.exp(s)))

def cfun(x):
    return x.sum() - _GN / 2.0

def cjac(x):
    return np.ones(len(x))

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = mask.sum()
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GT = 0.0

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def soft_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(c))
    if _GT < 1e-14:
        return m
    s = np.clip((c - m) / _GT, -50.0, 0.0)
    return m + _GT * float(np.log(np.sum(np.exp(s))))

def cfun(x):
    return x.sum() - _GN * 0.5

def cjac(x):
    return np.ones(len(x))

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(400):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        k = mask.sum()
        if k == 0:
            break
        h[mask] += d / k
        h = np.clip(h, 0.0, 1.0)
    d = n * 0.5 - h.sum()
    if abs(d) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
    return h

def slsqp(h0, n, tl):
    global _GN, _GT
    _GN = n
    bd = tuple((0.0, 1.0) for _ in range(n))
    cs = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bh = h0.copy()
    bc = eval_c5(bh)
    for tp in [0.05, 0.02, 0.008, 0.003, 0.001, 0.0003, 0.0, 0.0]:
        if time.time() > tl:
            break
        _GT = tp
        f = eval_c5 if tp == 0 else soft_obj
        try:
            r = minimize(f, bh, method='SLSQP', bounds=bd,
                         constraints=cs,
                         options={'maxiter': 300, 'ftol': 1e-16})
            if r.x is not None:
                hn = fix_sum(r.x, n)
                c = eval_c5(hn)
                if c < bc:
                    bc, bh = c, hn
        except Exception:
            pass
    return bh, bc

    for _ in range(nr):

            break
        imp = False

            rest = n * 0.5 - old
            lo = max(0.0, rest - (n - 1))
            hi = min(1.0, rest)
            if hi - lo < 1e-10:

            for t in np.linspace(lo, hi, 15):

                c = eval_c5(bh)
                if c < best_c:

        if not imp:
