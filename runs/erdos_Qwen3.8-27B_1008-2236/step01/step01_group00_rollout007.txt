def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50, 50)  # numerical stability
    return float(m + temp * np.log(np.sum(np.exp(shifted))))

def slsqp_optimize(h0, n, t_limit, temps=None):

    for temp in temps:

            break
        try:

                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:

                c5 = evaluate_c5(h_new)

    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt == 0:
            break
        h[mask] += d / cnt
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return float(m)

The key structure: we want h such that h(x) and h(x+k) are "anti-correlated" for all k. If h is 0/1, this means the set S and its shifted versions S-k should have minimal overlap.

For a set of measure 1 in [0,2], the minimum possible max overlap... Let's think. If S = [0,1], then S-k for k=0 has overlap 1, for k=0.5 has overlap 0.5, etc. The max is 1 (at k=0).

But wait, at k=0: ∫ h(x)(1-h(x)) dx. If h is 0/1, this is 0! So k=0 always gives 0.

For k≠0: if h is 0/1, ∫ h(x)(1-h(x+k)) dx = |S| - |S ∩ (S-k)| = 1 - |S ∩ (S-k)|.

So we want to MAXIMIZE min_{k≠0} |S ∩ (S-k)|.

For S = [0,1]: |S ∩ (S-k)| = 1-|k| for |k|≤1, 0 for |k|>1. Min over k∈(0,2] is 0 (at k=2). So this gives C5 = 1.

For S to be good, we need S ∩ (S-k) to be large for ALL k. This is like a "cross-intersecting" family.

The best case: if S is "spread out" so that every shift has significant overlap. For example, if S is a union of many small intervals spread across [0,2].

In the limit, if h(x) = 1/2 everywhere, then ∫ h(x)(1-h(x+k)) dx = ∫ 1/2 * 1/2 dx = 1/2 for all k. But ∫ h = 1, so h=1/2 everywhere works! C5 = 1/2 = 0.5.

def multiplicative_updates(h0, n, t_limit, max_iter=3000):

        m = np.max(corr)

        sorted_idx = np.argsort(corr)[-top_k:]

    x0 = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x1 = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    h_dst = np.interp(x1, x0, h_src, period=2.0)
    return fix_sum(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = fix_sum(np.full(45, 0.5), 45)

    n0 = len(h0)
    h0 = fix_sum(h0, n0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    temps_full = [0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    temps_mid = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    temps_short = [0.01, 0.005, 0.002, 0.001, 0.0005]

    # Phase 1: SLSQP main (250s)
    best_h, best_c5 = slsqp_run(best_h, n0, t0 + 200, temps_full)

    # Phase 2: Multiplicative updates (150s)
    best_h, best_c5 = multiplicative_updates(best_h, n0, t0 + 350, max_iter=4000)

    # Phase 3: SLSQP refine (150s)
    best_h, best_c5 = slsqp_run(best_h, n0, t0 + 500, temps_mid)

    # Phase 4: Pairwise polish (150s)
    best_h, best_c5 = pairwise_polish(best_h, n0, t0 + 650)

    # Phase 5: Diverse restarts (150s)
    t5_end = t0 + 800
    rng = np.random.RandomState(seed + 1)
    for trial in range(40):
        if time.time() > t5_end:
            break
        scale = 0.12 * max(0.3, 1.0 - float(trial) / 40.0)
        h_try = fix_sum(best_h + rng.randn(n0) * scale, n0)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        elif c5_try < best_c5 + 0.015:
            tl = min(t5_end, time.time() + 15)
            h_r, c5_r = slsqp_run(h_try, n0, tl, temps_short)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Try different n (150s)
    t6_end = t0 + 950
    for n_try in [60, 80, 100]:
        if time.time() > t6_end:
            break
        h_up = interpolate_h(best_h, n0, n_try)
        tl = min(t6_end, time.time() + 40)
        h_r, c5_r = slsqp_run(h_up, n_try, tl, temps_short)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        # Also try harmonic
        h_harm = make_harmonic(n_try, seed=seed + n_try)
        tl2 = min(t6_end, time.time() + 20)
        h_r2, c5_r2 = slsqp_run(h_harm, n_try, tl2, temps_short)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()

    # Phase 7: Final polish (80s)
    t7_end = t0 + budget_s - 15
    n_best = len(best_h)
    best_h, best_c5 = pairwise_polish(best_h, n_best, t7_end, n_pairs_max=5000)
    if time.time() < t0 + budget_s - 8:
        best_h, best_c5 = slsqp_run(best_h, n_best, t0 + budget_s - 5,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0002])
    if time.time() < t0 + budget_s - 3:
        best_h, best_c5 = pairwise_polish(best_h, n_best, t0 + budget_s - 1, n_pairs_max=2000)

    # Ensure exact sum
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_final)
