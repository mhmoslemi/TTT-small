import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            # Distribute among available positions
            per = d / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def c5obj(x):
    return compute_c5(x)

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def slsqp(h0, mi=300):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
    return r.x, float(r.fun)

def binary_local_search(h0, rng, tl=5.0):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    # Fix sum
    h = project_sum(h)
    b = c = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < tl:
        o = np.flatnonzero(h > 0.4)
        z = np.flatnonzero(h < 0.6)
        if len(o) == 0 or len(z) == 0:
            break
        if len(o) >= 2 and len(z) >= 2 and rng.random() < 0.5:
            # 2-swap
            i1 = int(o[rng.integers(len(o))])
            i2 = int(o[rng.integers(len(o))])
            if i1 == i2:
                continue
            j1 = int(z[rng.integers(len(z))])
            j2 = int(z[rng.integers(len(z))])
            if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
                continue
            h[i1], h[i2] = 0.0, 0.0
            h[j1], h[j2] = 1.0, 1.0
            nv = compute_c5(h)
            if nv <= c + 1e-10:
                c = nv
                if nv < b:
                    b = nv
            else:
                h[i1], h[i2] = 1.0, 1.0
                h[j1], h[j2] = 0.0, 0.0
        else:
            i = int(o[rng.integers(len(o))])
            j = int(z[rng.integers(len(z))])
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            nv = compute_c5(h)
            if nv <= c + 1e-10:
                c = nv
                if nv < b:
                    b = nv
            else:
                h[i], h[j] = 1.0, 0.0
    return h, b

def anneal(h0, rng, tl=30.0):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    bh = h.copy()
    b = c
    t0 = time.time()
    T = 0.15
    while time.time() - t0 < tl and T > 1e-8:
        k = int(rng.integers(1, 8))

        h[idx] = np.clip(h[idx] + rng.normal(0, 0.3, k), 0, 1)

                bh = h.copy()
        else:

            h = project_sum(h)
            c = compute_c5(h)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max()), c

def project_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + d/len(idx), 0.0, 1.0)
    return h

def c5obj(x):
    v, _ = compute_c5(x)
    return v

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def slsqp(h0, mi=300):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
    return r.x, float(r.fun)
