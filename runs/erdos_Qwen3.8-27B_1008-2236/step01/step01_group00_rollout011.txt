    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = d / cnt
        # Clip the step to avoid overshooting
        if d > 0:
            max_step = 1.0 - h[mask].max()
            if step > max_step:
                # Need to only adjust some coordinates
                # This is a water-filling problem
                # For simplicity, just do a smaller step
                step = max_step * 0.5
        else:
            max_step = -h[mask].min()
            if -step > max_step:
                step = -max_step * 0.5
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minmax(betas, alphas, t_lo=0.0, t_hi=1.0):
    """Solve min_{t in [t_lo, t_hi]} max_l (betas[l] + alphas[l]*t)"""
    # Evaluate at endpoints
    val_lo = np.max(betas + alphas * t_lo)
    val_hi = np.max(betas + alphas * t_hi)
    
    # Find upper envelope breakpoints in [t_lo, t_hi]
    # For efficiency, just sample densely and refine
    n_samples = 200
    t_grid = np.linspace(t_lo, t_hi, n_samples)
    vals = np.max(betas[None, :] + alphas[None, :] * t_grid[:, None], axis=1)
    t_best = t_grid[np.argmin(vals)]
    best_val = np.min(vals)
    
    # Refine
    for _ in range(3):
        eps = (t_hi - t_lo) / n_samples
        t_grid2 = np.linspace(max(t_lo, t_best - eps), min(t_hi, t_best + eps), 100)
        vals2 = np.max(betas[None, :] + alphas[None, :] * t_grid2[:, None], axis=1)
        t_new = t_grid2[np.argmin(vals2)]
        if np.max(betas + alphas * t_new) < best_val:
            best_val = np.max(betas + alphas * t_new)
        t_best = t_new
    
    return t_best, best_val

def coord_descent(h, n, target, n_sweeps, t_limit, rng):
    """Coordinate descent with 1D minmax solve for each coordinate."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for sweep in range(n_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            old_val = h[i]
            
            # Compute betas and alphas by evaluating at t=0 and t=1
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            h[i] = old_val
            
            betas = c0
            alphas = c1 - c0
            
            t_opt, _ = solve_1d_minmax(betas, alphas)
            
            h[i] = t_opt
            h = project_sum(h, target)
            
            c5 = evaluate_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, temp, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def slsqp_run(h0, n, target, t_limit, temps=None, maxiter=200):
    if temps is None:
        temps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)},)
    
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h_cur)
    
    for temp in temps:
        if time.time() > t_limit:
            break
        def obj(x):
            return smooth_c5(x, temp, n)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_sum(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except:
            pass
    
    # Unsmoothed
    if time.time() < t_limit:
        try:
            res = minimize(lambda x: evaluate_c5(x), h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-12})
            h_new = project_sum(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except:
            pass
    
    return best_h, best_c5

def coord_descent(h0, n, target, t_limit, n_sweeps=100, rng=None):
    """Coordinate descent: for each coord, find optimal 1D value."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            # Evaluate at t=0 and t=1 to get alpha, beta
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = old
            
            alphas = c1 - c0
            betas = c0
            
            # Find min of max(betas + alphas*t) for t in [0,1]
            # Dense sampling + refinement
            t_grid = np.linspace(0, 1, 100)
            vals = np.maximum.reduce(betas[None,:] + alphas[None,:] * t_grid[:,None])
            t_best = t_grid[np.argmin(vals)]
            # Refine
            lo = max(0, t_best - 0.02)
            hi = min(1, t_best + 0.02)
            t_grid2 = np.linspace(lo, hi, 50)
            vals2 = np.maximum.reduce(betas[None,:] + alphas[None,:] * t_grid2[:,None])
            t_best = t_grid2[np.argmin(vals2)]
            
            h[i] = t_best
            h = project_sum(h, target)
            c5 = evaluate_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def pairwise_polish(h0, n, target, t_limit, rng):
    """Optimize pairs of coordinates (redistribute sum between pairs)."""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    
    for rnd in range(500):
        if time.time() > t_limit:
            break
        improved = False
        n_pairs = min(n*(n-1)//2, 2000)
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-8:
                continue
            # Try optimal split
            # For pair (i,j), corr changes as function of t=h[i], h[j]=s-t
            # This is more complex (quadratic) but we can just try a few values
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                t = lo + frac * (hi - lo)
                h[i] = t
                h[j] = s - t
                c5 = evaluate_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best_h, best_c5
