    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + d, 0.0, 1.0)
    return h

def slsqp_obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    return float(c.max())

def slsqp_con_fun(x):
    return x.sum() - 0.5 * len(x)

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp(h0, max_iter=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
    r = minimize(slsqp_obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h = project_sum(r.x)
    return h, compute_c5(h)

def binarize(h):
    h_b = np.where(h >= 0.5, 1.0, 0.0)
    return project_sum(h_b)

def exhaustive_1swap(h):
    """Try all 1-swap pairs until no improvement."""
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    for sweep in range(100):
        improved = False
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nc = compute_c5(h)
                if nc < c - 1e-14:
                    c = nc
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, c

def random_2swap(h, rng, n_iter=5000):
    """Try random 2-swap pairs."""
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    if len(ones) < 2 or len(zeros) < 2:
        return best_h, best_c
    for _ in range(n_iter):
        i1 = ones[rng.integers(len(ones))]
        i2 = ones[rng.integers(len(ones))]
        if i1 == i2:
            continue
        j1 = zeros[rng.integers(len(zeros))]
        j2 = zeros[rng.integers(len(zeros))]
        if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
            continue
        h[i1], h[i2] = 0.0, 0.0
        h[j1], h[j2] = 1.0, 1.0
        nc = compute_c5(h)
        if nc < c - 1e-14:
            c = nc
            if nc < best_c:

                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i1], h[i2] = 1.0, 1.0
            h[j1], h[j2] = 0.0, 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + d, 0.0, 1.0)
    return h

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    return float(c.max())

def _con_fun(x):
    return x.sum() - 0.5 * len(x)

def _con_jac(x):
    return np.ones(len(x))

def slsqp(h0, max_iter=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    r = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-13})
    h = project_sum(r.x)
    return h, compute_c5(h)

def binarize(h):
    h_b = np.where(h >= 0.5, 1.0, 0.0)
    return project_sum(h_b)

def exhaustive_1swap(h):
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    for _ in range(200):
        improved = False
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nc = compute_c5(h)
                if nc < c - 1e-14:
                    c = nc
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, c

def swap_2(h, rng, n_iter):
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    if len(ones) < 2 or len(zeros) < 2:
        return best_h, best_c
    no_imp = 0
    for _ in range(n_iter):
        i1 = int(ones[rng.integers(len(ones))])
        i2 = int(ones[rng.integers(len(ones))])
        if i1 == i2:
            continue
        j1 = int(zeros[rng.integers(len(zeros))])
        j2 = int(zeros[rng.integers(len(zeros))])
        if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
            continue
        h[i1], h[i2] = 0.0, 0.0
        h[j1], h[j2] = 1.0, 1.0
        nc = compute_c5(h)
        if nc < c - 1e-14:
            c = nc
            no_imp = 0
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
        else:
            h[i1], h[i2] = 1.0, 1.0
            h[j1], h[j2] = 0.0, 0.0
            no_imp += 1
            if no_imp > 5000:
                break
    return best_h, best_c

def reversal_move(h, rng, n_iter):
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for _ in range(n_iter):
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a > b:
            a, b = b, a
        if b - a < 2:
            continue
        h[a:b+1] = h[a:b+1][::-1]
        h = project_sum(h)
        nc = compute_c5(h)
        if nc < c - 1e-14:
            c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
        else:
            h[a:b+1] = h[a:b+1][::-1]
            h = project_sum(h)
    return best_h, best_c

def make_random_binary(n, rng):
    h = np.zeros(n)
    picks = rng.choice(n, size=n // 2, replace=False)
    h[picks] = 1.0
    if n % 2 == 1:
        h[rng.integers(n)] += 0.5
    return h
