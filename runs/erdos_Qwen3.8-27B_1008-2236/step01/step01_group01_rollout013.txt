    if abs(d) > 1e-12 and len(idx) > 0:

def smooth_pnorm(h, p=60.0):
    n = len(h)
    dx = 2.0 / n
    c = np.abs(np.correlate(h, 1.0 - h, mode="full") * dx)

    # Numerical gradient (too slow for large n, use for n<=50)
    n = len(h)
    eps = 1e-6
    g = np.zeros(n)
    f0 = smooth_pnorm(h, p)
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        g[i] = (smooth_pnorm(hp, p) - f0) / eps
    return g

    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    def f(x):
        return pnorm_obj(project_sum(x), p)
    res = minimize(f, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return project_sum(res.x)

def binary_search(h0, rng, tlim=8.0):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0).copy()
    h = project_sum(h)
    c = compute_c5(h)
    bh, bc = h.copy(), c
    t0 = time.time()
    stall = 0
    while time.time() - t0 < tlim:
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        imp = False
        for _ in range(300):
            mode = rng.integers(3)
            if mode < 2:
                i = int(ones[rng.integers(len(ones))])
                j = int(zeros[rng.integers(len(zeros))])
                if i == j:
                    continue
                h[i], h[j] = 0.0, 1.0
                nv = compute_c5(h)
                if nv < c - 1e-10:
                    c = nv
                    imp = True
                    if nv < bc:
                        bc, bh = nv, h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0
            else:
                bl = int(rng.integers(1, min(4, n // 6) + 1))
                if bl >= n - 1:
                    continue
                i = int(rng.integers(n - bl + 1))
                j = int(rng.integers(n - bl + 1))
                if abs(i - j) < bl:
                    continue
                bi = h[i:i+bl].copy()
                bj = h[j:j+bl].copy()
                if abs(bi.sum() - bj.sum()) > 1e-9:
                    continue
                h[i:i+bl], h[j:j+bl] = bj, bi
                nv = compute_c5(h)
                if nv < c - 1e-10:
                    c = nv
                    imp = True
                    if nv < bc:
                        bc, bh = nv, h.copy()
                else:
                    h[i:i+bl], h[j:j+bl] = bi, bj
        if not imp:
            stall += 1
            if stall > 5:
                h = bh.copy()
                c = bc
                stall = 0
        else:
            stall = 0
    return bh, bc

def bin_sa(h0, rng, tlim=20.0):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0).copy()
    h = project_sum(h)
    c = compute_c5(h)
    bh, bc = h.copy(), c
    t0 = time.time()
    T = 0.02
    while time.time() - t0 < tlim and T > 1e-4:
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        if i == j:
            T *= 0.999
            continue
        h[i], h[j] = 0.0, 1.0
        nv = compute_c5(h)
        if nv <= c or rng.random() < np.exp(-(nv - c) / T):
            c = nv
            if nv < bc:
                bc, bh = nv, h.copy()
                T = max(T * 0.99, 0.005)
        else:
            h[i], h[j] = 1.0, 0.0
            T *= 0.999
    return bh, bc

def cont_search(h0, rng, tlim=8.0):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    bh, bc = h.copy(), c
    t0 = time.time()
    while time.time() - t0 < tlim:
        k = int(rng.integers(1, max(2, n // 10) + 1))
        idx = rng.choice(n, k, replace=False)
        sv = h[idx].copy()
        h[idx] = np.clip(h[idx] + rng.normal(0, 0.2, k), 0, 1)
        h = project_sum(h)
        nv = compute_c5(h)
        if nv < c - 1e-10:
            c = nv
            if nv < bc:
                bc, bh = nv, h.copy()
        else:
            h[idx] = sv
            h = project_sum(h)
    return bh, bc

def make_starts(n, rng, n_seed):
    starts = []
    for s in range(5):
        sr = np.random.default_rng(n_seed + s * 7919)
        if n % 2 == 0:
            h = np.zeros(n)
            h[sr.choice(n, n // 2, replace=False)] = 1.0
            starts.append(h)
        else:
            h = np.zeros(n)
            h[sr.choice(n, n // 2, replace=False)] = 1.0
            h = np.append(h, 1.0 - h.sum() + n * 0.5) if False else h
            h = project_sum(h)
            starts.append(h)
    # Structured: quasi-periodic
    alphas = [1.6180339887, 1.414213562, 2.0, 1.7320508, 2.236]
    for a in alphas[:2]:
        h = np.zeros(n)
        pos = sorted(set(((k + 1) * a) % n for k in range(n)))[:n // 2]
        for p in pos:
            if int(p) < n:
                h[int(p)] = 1.0
        if n % 2 == 0:
            h = project_sum(h)
        else:
            h = np.append(h, 0.0)
            h = project_sum(h)
        starts.append(h.copy())
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    h_init = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.asarray(globals()['initial_h_values'], dtype=float).copy()

    n_list = [44, 45, 46, 48]
    if h_init is not None:
        n0 = len(h_init)
        if n0 not in n_list:
            n_list.insert(0, n0)

    time_per = (budget_s - 120) / len(n_list)

    for ni, n in enumerate(n_list):
        t_start = time.time()
        t_alloc = min(time_per, budget_s - 120 - (time.time() - ts))
        if t_alloc < 20:
            break
        starts = make_starts(n, rng, seed + n * 100)
        if h_init is not None and len(h_init) == n:
            starts.insert(0, h_init.copy())

        for si, hs in enumerate(starts):
            if time.time() - ts > budget_s - 100:
                break
            hs = project_sum(hs)
            # Smooth L-BFGS-B
            if time.time() - ts < budget_s - 60:
                ho = lbfgs_opt(hs, p=40, maxiter=200)
                co = compute_c5(ho)
                if co < best_c5:
                    best_c5, best_h, best_n = co, ho.copy(), n
                hs = ho
            # Binary search for even n
            if n % 2 == 0 and time.time() - ts < budget_s - 60:
                hb, cb = binary_search(hs, rng, min(6.0, t_alloc * 0.3))
                if cb < best_c5:
                    best_c5, best_h, best_n = cb, hb.copy(), n
                hs = hb
                # SA from binary
                if time.time() - ts < budget_s - 60:
                    ha, ca = bin_sa(hs, rng, min(8.0, t_alloc * 0.3))
                    if ca < best_c5:
                        best_c5, best_h, best_n = ca, ha.copy(), n
                    hs = ha
            # Continuous refinement
            if time.time() - ts < budget_s - 60:
                hc, cc = cont_search(hs, rng, min(4.0, t_alloc * 0.2))
                if cc < best_c5:
                    best_c5, best_h, best_n = cc, hc.copy(), n

    # Iterated local search on best
    if best_h is not None:
        n_best = len(best_h)
        for it in range(30):
            if time.time() - ts > budget_s - 50:
                break
            hp = project_sum(best_h + rng.normal(0, 0.08 + 0.04 * it * 0.1, n_best))
            ho = lbfgs_opt(hp, p=50, maxiter=150)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho.copy()
            if n_best % 2 == 0:
                hb, cb = binary_search(ho, rng, min(4.0, (budget_s - 50 - (time.time()-ts)) * 0.3))
                if cb < best_c5:
                    best_c5, best_h = cb, hb.copy()

    # Final high-precision polish
    if best_h is not None and time.time() - ts < budget_s - 10:
        nf = len(best_h)
        ho = lbfgs_opt(best_h, p=100, maxiter=500)
        co = compute_c5(ho)
        if co < best_c5:
            best_c5, best_h = co, ho.copy()
        # SLSQP final
        def sfun(x):
            return x.sum() - 0.5 * len(x)
        def sjac(x):
            return np.ones(len(x))
        def sc5(x):
            n = len(x)
            dx = 2.0 / n
            return float(np.correlate(x, 1.0 - x, mode="full").max() * dx)
        bounds = list(zip(np.zeros(nf), np.ones(nf)))
        cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
        try:
            res = minimize(sc5, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            hf = project_sum(res.x)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf.copy()
        except Exception:
            pass

    if best_h is None:
        best_h = project_sum(np.array([1.0 if i % 2 == 0 else 0.0 for i in range(44)]))
        best_c5 = compute_c5(best_h)
        best_n = 44
    else:
        best_h = project_sum(best_h)
        best_c5 = min(best_c5, compute_c5(best_h))
        best_n = len(best_h)

    return best_h, best_c5, best_n
