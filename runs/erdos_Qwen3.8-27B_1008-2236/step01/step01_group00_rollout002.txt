import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-12:
        return m
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50, 50)  # numerical stability
    return float(m + temp * np.log(np.sum(np.exp(shifted))))

def slsqp_run(h_start, n, t_limit, max_temps=None):
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    
    if max_temps is None:
        max_temps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    bounds = tuple((0.0, 1.0) for _ in range(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    
    for temp in max_temps:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda x: smooth_c5(x, temp), h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.success or res.fun < best_c5 + 0.01:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    
    # Final unsmoothed
    if time.time() < t_limit:
        try:
            res = minimize(lambda x: evaluate_c5(x), h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None:

                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def coordinate_descent(h_best, n, t_limit, n_rounds=200):

    h_work = h_best.copy()

def subgradient_step(h, n, lr):
    """One subgradient step for max_k f_k(h)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find active set (k values achieving max)
    m = np.max(corr)
    active = corr > m - 1e-10
    
    # Compute subgradient: average gradient of active f_k
    grad = np.zeros(n)
    n_active = np.sum(active)
    if n_active == 0:
        return h
    
    # For each active k, compute gradient of f_k(h)
    # f_k = sum_i h[i]*(1-h[i+k])*dx (for appropriate i)
    # df_k/dh[i] = (1-h[i+k])*dx - h[i-k]*dx (with boundary handling)
    
    # Use numpy correlate to efficiently compute this
    # Actually, let's just compute it directly
    # For the full correlation array, index k_full corresponds to shift k = k_full - (n-1)
    # corr[k_full] = sum_i h[i] * (1-h[i + k]) where k = k_full - (n-1)
    #   and the sum is over valid i
    
    # Gradient component for h[i]:
    # From terms where i appears as the first argument: (1-h[i+k])*dx for valid k
    # From terms where i appears as the second argument: -h[i-k]*dx for valid k
    
    for k_full in np.where(active)[0]:
        k = k_full - (n - 1)  # shift
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])*dx
            for i in range(n - k):
                grad[i] += (1 - h[i+k]) * dx
                grad[i+k] += -h[i] * dx
        else:
            kk = -k
            for i in range(kk, n):
                grad[i] += -h[i-kk] * dx
                grad[i-kk] += (1 - h[i]) * dx
    
    grad /= n_active
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = fix_sum(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-12:
        return m
    shifted = np.clip((corr - m) / temp, -60, 0)
    return float(m + temp * np.log(np.sum(np.exp(shifted))))

def slsqp_run(h_start, n, t_limit, temps=None):
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    if temps is None:
        temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    bounds = tuple((0.0, 1.0) for _ in range(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    for temp in temps:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda x: smooth_c5(x, temp), h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-15})
            h_new = fix_sum(np.clip(res.x, 0, 1), n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass
    if time.time() < t_limit:
        try:
            res = minimize(lambda x: evaluate_c5(x), h_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-15})
            h_new = fix_sum(np.clip(res.x, 0, 1), n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def subgradient_step(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    tol = 0.02
    active = corr > m - tol
    n_act = max(1, int(np.sum(active)))
    grad = np.zeros(n)
    idxs = np.where(active)[0]
    for k_full in idxs:
        k = k_full - (n - 1)
        if k >= 0:
            ii = np.arange(n - k)
            grad[ii] += (1 - h[ii + k]) * dx
            grad[ii + k] += -h[ii] * dx
        else:
            kk = -k
            ii = np.arange(kk, n)
            grad[ii] += -h[ii - kk] * dx
            grad[ii - kk] += (1 - h[ii]) * dx
    grad /= n_act
    # Line search
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
        h_try = fix_sum(h - lr * shrink * grad, n)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def coord_descent(h_best, n, t_limit, max_rounds=500):
    best_c5 = evaluate_c5(h_best)
    h_work = h_best.copy()
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > t_limit:
                break
            for j in range(i + 1, n):
                md = min(h_work[i], 1.0 - h_work[j])
                if md > 1e-7:
                    for frac in (0.4, 0.6):
                        d = frac * md
                        h_work[i] -= d
                        h_work[j] += d
                        c5 = evaluate_c5(h_work)
                        if c5 < best_c5 - 1e-14:
                            best_c5 = c5
                            h_best = h_work.copy()
                            improved = True
                        else:
                            h_work[i] += d
                            h_work[j] -= d
                            break
                md2 = min(h_work[j], 1.0 - h_work[i])
                if md2 > 1e-7:
                    for frac in (0.4, 0.6):
                        d = frac * md2
                        h_work[j] -= d
                        h_work[i] += d
                        c5 = evaluate_c5(h_work)
                        if c5 < best_c5 - 1e-14:
                            best_c5 = c5
                            h_best = h_work.copy()
                            improved = True
                        else:
                            h_work[j] += d
                            h_work[i] -= d
                            break
            h_work[:] = h_best
        if not improved:
            break
    return h_best, best_c5

    t2 = t0 + budget_s * 0.70

        if c5_try < best_c5 + 0.012:

    if c5_f < best_c5:

        best_h = h_f.copy()
