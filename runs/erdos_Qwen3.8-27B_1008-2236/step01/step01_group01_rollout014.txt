import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    h += d / n
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    d2 = target - h.sum()
    if abs(d2) > 1e-12:
        if d2 > 0:
            idx = np.argsort(h)  # smallest first (most room to increase)
        else:
            idx = np.argsort(-h)  # largest first (most room to decrease)
        h[idx[0]] += d2
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, c_vals, n):
    # c_vals has 2n-1 entries
    # Find top k maximizers
    top_k = 5
    max_val = c_vals.max()
    # Get indices where c_vals is within a small tolerance of max
    thresh = max_val - 0.05
    active = np.where(c_vals >= thresh)[0]
    if len(active) == 0:
        active = np.where(c_vals == max_val)[0]
    
    # Compute subgradient (average over active)
    g = np.zeros(n)
    for k in active:
        s = k - (n - 1)  # lag
        if s >= 0:
            # c(s) = sum_{i=0}^{n-1-s} h[i+s](1-h[i])
            # dc/dh[k] = (1-h[k-s])*1[s<=k<=n-1] - h[k+s]*1[0<=k<=n-1-s]
            mask1 = (k >= s) & (k <= n-1)
            mask2 = (k >= 0) & (k <= n-1-s)
            if mask1:
                g += (1.0 - h[k-s])
            if mask2:
                g -= h[k+s]
        else:
            t = -s
            # c(-t) = sum_{j=0}^{n-1-t} h[j](1-h[j+t])
            # dc/dh[k] = (1-h[k+t])*1[0<=k<=n-1-t] - h[k-t]*1[t<=k<=n-1]
            mask1 = (k >= 0) & (k <= n-1-t)
            mask2 = (k >= t) & (k <= n-1)
            if mask1:
                g += (1.0 - h[k+t])
            if mask2:
                g -= h[k-t]
    
    g /= len(active)
    return g

def subgradient_descent(h0, budget=30.0):

    h = h0.copy()
    t0 = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    eta = 0.01
    for step in range(10000):
        if time.time() - t0 > budget:
            break
        c_vals = compute_c5_all(h)
        g = subgradient_step(h, c_vals, n)
        # Polyak-like step size
        h_new = h - eta * g
        h_new = project_sum(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        eta *= 0.9999  # slow decay
    return best_h, best_c

def slsqp_opt(h0, maxiter=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    
    def obj(x):
        return compute_c5(x)
    
    def con_fun(x):
        return x.sum() - n * 0.5
    
    def con_jac(x):
        return np.ones(n)
    
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                 options={'maxiter': maxiter, 'ftol': 1e-10})
    return r.x, float(r.fun)

def binary_local_search(h0, budget=30.0):
    n = len(h0)
    # Round to binary (or near-binary)
    h = np.where(h0 > 0.5, 1.0, 0.0)
    target = n * 0.5
    # Adjust to get exact sum

            if len(zeros) > 0:

                best_c = c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
    return h

def grad_c(h, n):
    """Compute gradients of all 2n-1 correlation values w.r.t. h. Returns shape (2n-1, n)."""
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            # c(s) = sum_{i=0}^{n-1-s} h[i+s](1-h[i])
            for i in range(n - s):
                ki = i + s
                G[k, ki] += (1.0 - h[i])
                G[k, i] += -h[i + s]
        else:
            t = -s
            # c(-t) = sum_{i=t}^{n-1} h[i-t](1-h[i])
            for i in range(t, n):
                ki = i - t
                G[k, ki] += (1.0 - h[i])
                G[k, i] += -h[i - t]
    return G

def smooth_max_grad(h, mu=30.0):
    """Gradient of smooth max approximation using log-sum-exp."""
    n = len(h)
    c = compute_c5_raw(h)
    cmax = c.max()
    exps = np.exp(mu * (c - cmax))
    weights = exps / exps.sum()
    # Weighted gradient
    g = np.zeros(n)
    active = np.where(weights > 1.0 / (2.0 * n))[0]
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                ki = i + s
                g[ki] += weights[k] * (1.0 - h[i])
                g[i] += weights[k] * (-h[i + s])
        else:
            t = -s
            for i in range(t, n):
                ki = i - t
                g[ki] += weights[k] * (1.0 - h[i])
                g[i] += weights[k] * (-h[i - t])
    return g, cmax

def smooth_max_descent(h0, budget=40.0, mu0=30.0, mu1=80.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    n_steps = 0
    while time.time() - t0 < budget:
        batch = min(200, int((budget - (time.time() - t0)) * 20) + 1)
        for _ in range(batch):
            if time.time() - t0 > budget:
                break
            n_steps += 1
            frac = min(n_steps / 500.0, 1.0)
            mu = mu0 + (mu1 - mu0) * frac
            eta = 0.8 / (1.0 + 0.001 * n_steps)
            g, _ = smooth_max_grad(h, mu)
            h_new = h - eta * g
            h_new = project_sum(h_new)
            c_new = compute_c5(h_new)
            if c_new < best_c * 1.00001:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            else:
                eta *= 0.5
                h_new = project_sum(h - eta * g)
                c_new = compute_c5(h_new)
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
    return best_h, best_c

def slsqp_opt(h0, maxiter=500):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    def obj(x):
        return float(compute_c5(x))
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                 options={'maxiter': maxiter, 'ftol': 1e-12})
    return r.x, float(r.fun)

def binary_search(h0, budget=30.0, rng=None):
    n = len(h0)
    if rng is None:
        rng = np.random.default_rng(0)
    h = np.where(h0 > 0.5, 1.0, 0.0)
    target = n * 0.5
    if target == int(target):
        need = int(target)
        ones = np.sum(h == 1.0)
        if ones > need:
            idx = np.where(h == 1.0)[0]
            for i in range(ones - need):
                h[idx[i]] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            for i in range(need - ones):
                h[idx[i]] = 1.0
    else:
        ti = int(target)
        fr = target - ti
        ones = np.sum(h == 1.0)
        if ones > ti:
            idx = np.where(h == 1.0)[0]
            for i in range(ones - ti):
                h[idx[i]] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            for i in range(ti - ones):
                h[idx[i]] = 1.0
        if fr > 0:
            z = np.where(h == 0.0)[0]
            if len(z) > 0:
                h[z[0]] = fr
    h = project_sum(h)
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    t0 = time.time()
    while time.time() - t0 < budget:
        oi = np.where(h > 0.3)[0]
        zi = np.where(h < 0.7)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = int(oi[rng.integers(len(oi))])
        j = int(zi[rng.integers(len(zi))])
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        nc = compute_c5(h)
        if nc <= c * 1.00005:
            c = nc
            if nc < best_c:
                best_c, best_h = nc, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c

def pair_swap_search(h0, budget=30.0, rng=None):
    n = len(h0)
    if rng is None:
        rng = np.random.default_rng(1)
    h = np.where(h0 > 0.5, 1.0, 0.0)
    target = n * 0.5
    if target == int(target):
        need = int(target)
        ones = np.where(h == 1.0)[0]
        if len(ones) > need:
            for i in range(len(ones) - need):
                h[ones[i]] = 0.0
        elif len(ones) < need:
            zeros = np.where(h == 0.0)[0]
            for i in range(need - len(ones)):
                h[zeros[i]] = 1.0
    else:
        ti = int(target)
        ones = np.where(h == 1.0)[0]
        if len(ones) > ti:
            for i in range(len(ones) - ti):
                h[ones[i]] = 0.0
        elif len(ones) < ti:
            zeros = np.where(h == 0.0)[0]
            for i in range(ti - len(ones)):
                h[zeros[i]] = 1.0
        fr = target - ti
        if fr > 0:
            z = np.where(h == 0.0)[0]
            if len(z) > 0:
                h[z[0]] = fr
    h = project_sum(h)
    c = compute_c5(h)
    best_h, best_c = h.copy(), c
    t0 = time.time()
    while time.time() - t0 < budget:
        oi = np.where(h > 0.4)[0]
        zi = np.where(h < 0.6)[0]
        if len(oi) < 2 or len(zi) < 2:
            break
        i1 = int(oi[rng.integers(len(oi))])
        i2 = int(oi[rng.integers(len(oi))])
        if i1 == i2:
            i2 = int(oi[rng.integers(len(oi))])
            if i1 == i2:
                continue
        j1 = int(zi[rng.integers(len(zi))])
        j2 = int(zi[rng.integers(len(zi))])
        if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
            continue
        h[i1], h[i2], h[j1], h[j2] = 0.0, 0.0, 1.0, 1.0
        nc = compute_c5(h)
        if nc <= c * 1.00005:
            c = nc
            if nc < best_c:
                best_c, best_h = nc, h.copy()
        else:
            h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.asarray(globals()['initial_h_values'], dtype=float).copy()
    else:
        h_init = np.zeros(45)
        h_init[0::2] = 1.0
        h_init[44] = 0.5
        h_init = project_sum(h_init)

    best_h = project_sum(h_init)
    best_c5 = compute_c5(best_h)

    n_candidates = [40, 45, 50]
    n_init = len(h_init)

    # Phase 1: Smooth-max descent + SLSQP for multiple n
    for n in n_candidates:
        if time.time() - ts > budget_s - 350:
            break
        n_starts = 3 if n == n_init else 2
        for trial in range(n_starts):
            if time.time() - ts > budget_s - 350:
                break
            if n == n_init and trial == 0:
                h0 = h_init.copy()
            else:
                h0 = rng.random(n)
                h0 = project_sum(h0)
            h_sm, c_sm = smooth_max_descent(h0, budget=25, mu0=25.0, mu1=80.0)
            if c_sm < best_c5:
                best_c5, best_h = c_sm, h_sm.copy()
            if time.time() - ts < budget_s - 330:
                h_sl, c_sl = slsqp_opt(project_sum(h_sm), 400)
                h_sl = project_sum(h_sl)
                c_sl = compute_c5(h_sl)
                if c_sl < best_c5:
                    best_c5, best_h = c_sl, h_sl.copy()

    # Phase 2: Multi-start SLSQP with perturbations around best
    no_improve = 0
    while time.time() - ts < budget_s - 200 and no_improve < 20:
        sigma = 0.03 + 0.01 * no_improve
        sigma = min(sigma, 0.25)
        h_p = project_sum(best_h + rng.normal(0, sigma, len(best_h)))
        h_sl, c_sl = slsqp_opt(h_p, 350)
        h_sl = project_sum(h_sl)
        c_sl = compute_c5(h_sl)
        if c_sl < best_c5 - 1e-9:
            best_c5, best_h = c_sl, h_sl.copy()
            no_improve = 0
        else:
            no_improve += 1

    # Phase 3: Smooth-max descent from best
    if time.time() - ts < budget_s - 150:
        h_sm, c_sm = smooth_max_descent(best_h, budget=40, mu0=40.0, mu1=100.0)
        if c_sm < best_c5:
            best_c5, best_h = c_sm, h_sm.copy()
            if time.time() - ts < budget_s - 120:
                h_sl, c_sl = slsqp_opt(best_h, 500)
                h_sl = project_sum(h_sl)
                c_sl = compute_c5(h_sl)
                if c_sl < best_c5:
                    best_c5, best_h = c_sl, h_sl.copy()

    # Phase 4: Pair swap + binary search
    if time.time() - ts < budget_s - 80:
        h_ps, c_ps = pair_swap_search(best_h, budget=30, rng=rng)
        if c_ps < best_c5:
            best_c5, best_h = c_ps, h_ps.copy()
        if time.time() - ts < budget_s - 60:
            h_bs, c_bs = binary_search(best_h, budget=20, rng=rng)
            if c_bs < best_c5:
                best_c5, best_h = c_bs, h_bs.copy()

    # Phase 5: Final SLSQP polish
    if time.time() - ts < budget_s - 10:
        h_sl, c_sl = slsqp_opt(best_h, 800)
        h_sl = project_sum(h_sl)
        c_sl = compute_c5(h_sl)
        if c_sl < best_c5:
            best_c5, best_h = c_sl, h_sl.copy()

    best_h = project_sum(best_h)
    final = compute_c5(best_h)
    return best_h, min(final, best_c5), len(best_h)
