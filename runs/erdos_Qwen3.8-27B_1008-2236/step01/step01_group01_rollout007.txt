def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_vec(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(-h)
        for i in idx:
            adj = min(abs(diff), 1.0 - h[i]) if diff > 0 else -min(abs(diff), h[i])
            h[i] += adj
            diff = target - h.sum()
            if abs(diff) < 1e-10:

    return h

    def obj(x):
        return compute_c5(x)
    def gf(x):
        return x.sum() - 0.5 * n
    def gj(x):
        return np.ones(n)
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': gf, 'jac': gj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
