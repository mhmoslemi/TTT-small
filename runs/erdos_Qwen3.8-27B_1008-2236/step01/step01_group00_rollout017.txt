def project_feasible(h, n):
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -max(np.max(h) - 0, 10) - 1, max(np.min(h), 0) + 10 + 1

    # sum(h_proj) = target

    # As lam -> +inf, h_proj -> 0, sum -> 0

        return np.sum(np.clip(h - lam, 0.0, 1.0))

    # Actually lo should give sum > target, hi should give sum < target

    hi = np.min(h) + 1  # h - hi < 0 for all, so clip to 0, sum = 0

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    # Bisection on lambda for projection onto {clip(h-lam, 0, 1), sum = target}
    lo = -1.0 - float(np.max(h)) if n > 0 else -2.0
    hi = 1.0 + float(np.min(h)) if n > 0 else 1.0
    # Ensure: at lo, sum >= target; at hi, sum <= target
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_step(h, n, top_k=10):
    """Compute normalized subgradient of max correlation"""
    corr = compute_corr(h)
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        # d/dh[t] corr[k] = (1-h[t-s]) - h[t+s] with zero padding
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])   # (1-h[t-s]) for t>=s
            grad[:n-s] -= h[s:]           # -h[t+s] for t <= n-1-s
        else:
            a = -s
            grad[:n-a] += (1.0 - h[a:])   # (1-h[t+a]) for t <= n-1-a
            grad[a:] -= h[:n-a]           # -h[t-a] for t >= a
    # Subtract mean to maintain sum constraint approximately (projection handles it)
    return grad

def nesterov_subgrad(h, n, t_limit, lr_init=0.5, decay=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    vel = np.zeros(n)
    lr = lr_init
    stuck = 0
    
    while time.time() < t_limit:
        lr *= decay
        if lr < 0.001:
            lr = 0.001
        
        grad = subgradient_step(h, n, top_k=15)
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            grad /= gn
        
        vel = 0.9 * vel - lr * grad
        h_new = project_feasible(h + vel, n)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            h = project_feasible(h - lr * grad / max(gn, 1e-10), n)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stuck += 1
            if stuck > 200:
                break
    
    return best_h, best_c5

def coordinate_descent(h, n, t_limit, n_passes=None):
    """Pair coordinate descent: for pair (i,j), optimize h[i],h[j] keeping sum fixed"""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_work = best_h.copy()
    
    if n_passes is None:
        n_passes = 50
    
    for rnd in range(n_passes):
        if time.time() > t_limit:
            break
        order = np.random.permutation(n)
        for i in order:
            # Find a partner j to trade with
            if h_work[i] >= 1.0 - 1e-9:
                continue
            # Try swapping with a random j
            j = np.random.randint(n)
            if j == i:
                continue
            # Current values
            hi, hj = h_work[i], h_work[j]
            # Total to maintain: hi + hj
            total = hi + hj
            # Optimize: choose t in [max(0, total-1), min(1, total)] for h[i]=t, h[j]=total-t
            # Minimize C5
            lo = max(0.0, total - 1.0)
            hi_bound = min(1.0, total)
            if hi_bound - lo < 1e-10:
                continue
            
            def obj(t):
                h_work[i] = t
                h_work[j] = total - t
                return evaluate_c5(h_work)
            
            res = minimize_scalar(obj, bounds=(lo, hi_bound), method='bounded',
                                options={'maxiter': 30})
            if res.fun < best_c5 - 1e-12:
                best_c5 = res.fun
                best_h = h_work.copy()
            
            h_work[i] = hi
            h_work[j] = hj
    
    return best_h, best_c5

def slsqp_refine(h, n, t_limit):
    """SLSQP refinement with log-sum-exp"""
    dx = 2.0 / n
    
    def smooth_obj(hh, temp):
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        m = np.max(corr)
        if temp < 1e-14:
            return m
        return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))
    
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda hh: hh.sum() - n/2.0,
             'jac': lambda hh: np.ones(n)}]
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = h.copy()
    
    for temp in [0.05, 0.01, 0.002, 0.0005]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda hh: smooth_obj(hh, temp), h_cur,
                         method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
                    h_cur = h_new
        except:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(60, 0.5)
    
    n = len(h0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: SLSQP from initial (30% of budget)
    t1 = t0 + budget_s * 0.3
    best_h, best_c5 = slsqp_refine(best_h, n, t1)
    
    # Phase 2: Nesterov subgradient (20% of budget)
    t2 = t0 + budget_s * 0.5
    best_h, best_c5 = nesterov_subgrad(best_h, n, t2)
    
    # Phase 3: Coordinate descent (20% of budget)
    t3 = t0 + budget_s * 0.7
    best_h, best_c5 = coordinate_descent(best_h, n, t3)
    
    # Phase 4: Restart with perturbations + SLSQP (20% of budget)
    t4 = t0 + budget_s * 0.9

        scale = 0.1 * (1.0 - r / 30.0) + 0.02

                best_h = h_r

    t5 = t0 + budget_s * 0.98
    best_h, best_c5 = coordinate_descent(best_h, n, t5)
    
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n)

def project_feasible(h, n):

            break
        try:
            res = minimize(lambda hh: smooth_obj(hh, temp), h_cur,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    # Unsmoothed final
    if time.time() < t_limit - 5:
        try:
            res = minimize(evaluate_c5, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def anneal_polish(h, n, t_limit, rng, T0=0.005, T1=0.0001, max_iter=50000):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_w = best_h.copy()
    T = T0
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        T = T0 * (T1 / T0) ** (it / float(max_iter))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi_v, hj_v = h_w[i], h_w[j]
        total = hi_v + hj_v
        lo = max(0.0, total - 1.0)
        hi_b = min(1.0, total)
        if hi_b - lo < 1e-12:
            continue
        t = rng.uniform(lo, hi_b)
        h_w[i] = t
        h_w[j] = total - t
        c5 = evaluate_c5(h_w)
        delta = c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_w.copy()
        else:
            h_w[i] = hi_v
            h_w[j] = hj_v
    return best_h, best_c5

def try_resolution(h_start, n_target, t_limit, rng, seed):
    """Try optimizing at n_target resolution, return best (h, c5, n)"""
    if n_target == len(h_start):
        h = project_feasible(h_start.copy(), n_target)
    elif n_target < len(h_start):
        # Downsample by averaging
        n_orig = len(h_start)
        h = np.zeros(n_target)
        for i in range(n_target):
            lo = int(i * n_orig / n_target)
            hi = int((i + 1) * n_orig / n_target)
            hi = max(hi, lo + 1)
            h[i] = h_start[lo:hi].mean()
        h = project_feasible(h, n_target)
    else:
        # Upsample by linear interpolation
        x_old = np.linspace(0, 2, len(h_start))
        x_new = np.linspace(0, 2, n_target)
        h = np.interp(x_new, x_old, h_start)
        h = project_feasible(h, n_target)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # SLSQP
    t1 = t0 + (t_limit - t0) * 0.4
    best_h, best_c5 = slsqp_refine(best_h, n_target, t1)
    
    # Coord descent
    t2 = t0 + (t_limit - t0) * 0.6
    best_h, best_c5 = coord_descent(best_h, n_target, t2, rng, n_trials=30000)
    
    # SLSQP again
    t3 = t0 + (t_limit - t0) * 0.75
    best_h, best_c5 = slsqp_refine(best_h, n_target, t3, temps=[0.002, 0.0005])
    
    # Anneal
    t4 = t0 + (t_limit - t0) * 0.9
    best_h, best_c5 = anneal_polish(best_h, n_target, t4, rng)
    
    # Final coord descent
    t5 = t0 + (t_limit - t0) * 0.98
    best_h, best_c5 = coord_descent(best_h, n_target, t5, rng, n_trials=20000)
    
    return best_h, evaluate_c5(best_h), n_target
