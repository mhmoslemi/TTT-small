            jm = j + k  # j+k should be in [0,n-1]
            if 0 <= jp < n:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    fval = float(c.max())
    tol = 0.02 * fval + 1e-9
    active = np.where(c >= fval - tol)[0]
    g = np.zeros(n)
    for m in active:

        for j in range(n):
            t = 0.0
            jp = j - k
            jm = j + k
            if 0 <= jp < n:
                t += (1.0 - h[jp])
            if 0 <= jm < n:
                t -= h[jm]
            g[j] += t
    g *= dx / max(1, len(active))
    return g, fval

def subgrad_run(h0, t_lim=20.0, lb=0.30):
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h0)
    beta = 0.4
    t0 = time.time()
    eta_max = 0.5
    for it in range(500000):
        if time.time() - t0 >= t_lim:
            break
        z = project_sum(z)
        g, fv = c5_subgrad(z)
        if fv < best_c - 1e-12:
            best_c = fv
            best_h = z.copy()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        # Polyak step
        eta = (fv - lb) / (gnorm * gnorm)
        eta = min(eta, eta_max)
        eta = max(eta, 1e-6)
        z = z - eta * g
        beta = min(beta + 0.0005, 0.8)
        z = z + beta * (z - h)
        h = project_sum(z)
    return best_h, best_c

    def obj(x):

        g, _ = c5_subgrad(x)
        return g

    def sfun(x):
        return float(x.sum() - 0.5 * len(x))
    def sjac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    fval = float(c.max())
    tol = 0.015 * fval + 1e-9
    active = np.where(c >= fval - tol)[0]
    g = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        for j in range(n):
            t = 0.0
            jp = j - k
            jm = j + k
            if 0 <= jp < n:
                t += (1.0 - h[jp])
            if 0 <= jm < n:
                t -= h[jm]
            g[j] += t
    g *= dx / max(1, len(active))
    return g, fval

def subgrad_run(h0, t_lim, lb):
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h0)
    beta = 0.3
    t0 = time.time()
    for _ in range(800000):
        if time.time() - t0 >= t_lim:
            break
        z = project_sum(z)
        g, fv = c5_subgrad(z)
        if fv < best_c - 1e-13:
            best_c = fv
            best_h = z.copy()
        gnorm2 = float(np.dot(g, g))
        if gnorm2 < 1e-20:
            break
        eta = (fv - lb) / gnorm2
        eta = min(eta, 0.4)
        eta = max(eta, 1e-7)
        z = z - eta * g
        beta = min(beta + 0.0008, 0.85)
        z = z + beta * (z - h)
        h = project_sum(z)
    return best_h, best_c

def slsqp_obj(x):
    return compute_c5(x)

def slsqp_jac(x):
    g, _ = c5_subgrad(x)
    return g

def slsqp_sfun(x):
    return float(x.sum() - 0.5 * len(x))

def slsqp_sjac(x):
    return np.ones(len(x))

def slsqp_run(h0, maxit, t_lim):
    n = len(h0)
    t0 = time.time()
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': slsqp_sfun, 'jac': slsqp_sjac}]
    r = minimize(slsqp_obj, h0, jac=slsqp_jac, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
    return r.x, float(r.fun)
