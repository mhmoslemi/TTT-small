import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)
    return h

def sfun(x):
    return x.sum() - 0.5 * len(x)

def sjac(x):
    return np.ones(len(x))

def slsqp(h0, maxiter=800):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(lambda x: compute_c5(x), h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return r.x, float(r.fun)

def binary_sa(h0, rng, time_limit=10.0):
    n = len(h0)
    target = int(round(0.5 * n))
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    # Ensure exactly target ones
    extra = int(h.sum()) - target
    if extra > 0:
        ones = np.flatnonzero(h == 1)
        for i in ones[:extra]:
            h[i] = 0
    elif extra < 0:
        zeros = np.flatnonzero(h == 0)
        for i in zeros[:abs(extra)]:
            h[i] = 1
    
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    T = 0.015
    t0 = time.time()
    stuck = 0
    
    while time.time() - t0 < time_limit and T > 1e-9:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0, 1
        nc = compute_c5(h)
        if nc <= c or rng.random() < np.exp(-min((nc - c) / T, 500)):
            c = nc
            stuck = 0
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
        else:
            h[j], h[i] = 0, 1
            stuck += 1
        if stuck > 500:
            T *= 0.5
            stuck = 0
        else:
            T *= 0.9997
    
    return best_h, best_c

def make_structured(n, rng):
    """Generate structured initial binary sequences."""
    starts = []
    # Alternating
    h = np.zeros(n)
    h[::2] = 1.0
    if int(h.sum()) != n//2:
        h[-1] = 1.0 - h[-1]
    starts.append(h.copy())
    # Block (half ones, half zeros) - bad but different
    h2 = np.zeros(n)
    h2[:n//2] = 1.0
    starts.append(h2)
    # Various period patterns
    for p in [3, 4, 5, 6, 7, 8, 11, 13]:
        h3 = np.zeros(n)
        fill = p // 2
        for i in range(n):
            if (i % p) < fill:
                h3[i] = 1.0
        # Adjust to get exactly n/2 ones
        while h3.sum() > n // 2:
            idx = np.flatnonzero(h3 == 1)
            h3[idx[rng.integers(len(idx))]] = 0
        while h3.sum() < n // 2:
            idx = np.flatnonzero(h3 == 0)
            h3[idx[rng.integers(len(idx))]] = 1
        starts.append(h3.copy())
    # Random
    for _ in range(5):
        hr = np.zeros(n)
        positions = rng.choice(n, n//2, replace=False)
        hr[positions] = 1.0
        starts.append(hr)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    
    # Get or create initial solution
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.asarray(globals()['initial_h_values'], dtype=float).copy()
    else:
        h_init = np.zeros(45)
        h_init[0::2] = 1.0
        h_init[22] = 0.5
        h_init = project_sum(h_init)
    
    best_h = h_init.copy()
    best_c = compute_c5(best_h)
    
    # Try multiple grid sizes
    n_list = [44, 46, 50, 54, 45, 56, 48]
    
    for n in n_list:
        if time.time() - ts > budget_s - 40:
            break
        
        # Generate starts for this n
        starts = make_structured(n, rng)
        
        # Also interpolate from best found so far
        if len(best_h) != n:
            xs_old = np.linspace(0.5, len(best_h)-0.5, len(best_h))
            xs_new = np.linspace(0.5, n-0.5, n)
            h_interp = np.interp(xs_new, xs_old, best_h)
            h_bin = np.where(h_interp >= 0.5, 1.0, 0.0)
            # Fix sum
            while h_bin.sum() > n//2:
                h_bin[np.argmax(h_bin * rng.random(n))] = 0
            while h_bin.sum() < n//2:
                h_bin[np.argmin((1-h_bin) * rng.random(n))] = 1
            starts.append(h_bin)
        
        # Run binary SA on each start
        for si, start in enumerate(starts):
            if time.time() - ts > budget_s - 30:
                break
            sa_time = min(12.0, (budget_s - (time.time() - ts) - 30) / (len(starts) - si))
            if sa_time < 3:
                break
            h_sa, c_sa = binary_sa(start, rng, sa_time)
            if c_sa < best_c:
                best_c = c_sa
                best_h = h_sa.copy()
        
        # SLSQP polish
        if time.time() - ts > budget_s - 20:
            break
        # Use best found for this n as SLSQP start
        # Also try a few perturbed starts
        for trial in range(3):
            if time.time() - ts > budget_s - 15:
                break
            if trial == 0:
                h0 = best_h.copy() if len(best_h) == n else project_sum(np.interp(
                    np.linspace(0.5, n-0.5, n), np.linspace(0.5, len(best_h)-0.5, len(best_h)), best_h))
            else:
                h0 = project_sum(np.clip(best_h + rng.normal(0, 0.1, n) if len(best_h)==n else np.zeros(n), 0, 1))
            h_sl, c_sl = slsqp(h0, 600)
            h_sl = project_sum(h_sl)
            c_sl = compute_c5(h_sl)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl.copy()
    
    # Final polish on the overall best
    if time.time() - ts < budget_s - 10:
        h_fin, c_fin = slsqp(best_h, 1000)
        h_fin = project_sum(h_fin)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    
    best_h = project_sum(best_h)
    final_c = compute_c5(best_h)
    return best_h, min(final_c, best_c), len(best_h)
