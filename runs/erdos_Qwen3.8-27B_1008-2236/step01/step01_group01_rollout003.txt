import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        mask = (h < 1.0 - 1e-8) if d > 0 else (h > 1e-8)
        idx = np.where(mask)[0]
        for i in idx:
            delta = float(np.clip(d, -h[i], 1.0 - h[i]))
            h[i] += delta
            d -= delta
            if abs(d) < 1e-12:
                break
    return h

def sum_con(x):
    return x.sum() - 0.5 * len(x)

def sum_jac(x):
    return np.ones(len(x))

def soft_max_c5(x, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / beta)

def hard_max_c5(x):
    n = len(x)
    dx = 2.0 / n
    return float(np.correlate(x, 1.0 - x, mode="full").max() * dx)

def do_slsqp(h0, obj_func, maxiter=400, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    r = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(r.x), compute_c5(project_sum(r.x))

def swap_search(h0, rng, tlimit=5.0):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    bh = h.copy()
    bc = c
    t0 = time.time()
    stall = 0
    while time.time() - t0 < tlimit and stall < 500:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = h[i], h[j]
        h[i], h[j] = oj, oi
        h = np.clip(h, 0, 1)
        nv = compute_c5(h)
        if nv < c - 1e-13:
            c = nv
            stall = 0
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i], h[j] = oi, oj
            stall += 1
        # Also try perturbation
        if rng.random() < 0.25:
            i = int(rng.integers(n))
            old = h[i]
            h[i] = np.clip(old + rng.normal(0, 0.08), 0, 1)
            h = project_sum(h)
            nv = compute_c5(h)
            if nv < c - 1e-13:
                c = nv
                stall = 0
                if c < bc:
                    bc, bh = c, h.copy()
            else:
                h[i] = old
                h = project_sum(h)
                stall += 1
    return bh, bc

def anneal_swap(h0, rng, tlimit=30.0, T0=0.02):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    bh = h.copy()
    bc = c
    t0 = time.time()
    T = T0
    while time.time() - t0 < tlimit and T > 1e-6:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = h[i], h[j]
        h[i], h[j] = oj, oi
        h = np.clip(h, 0, 1)
        nv = compute_c5(h)
        if nv <= c or rng.random() < np.exp(-(nv - c) / T):
            c = nv
            if nv < bc:
                bc, bh = nv, h.copy()
        else:
            h[i], h[j] = oi, oj
        T *= 0.999
    return bh, bc
