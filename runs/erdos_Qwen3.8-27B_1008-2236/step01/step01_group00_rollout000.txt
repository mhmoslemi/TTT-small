import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj(h, temp):
    c = get_corr(h)
    m = np.max(c)
    if temp < 1e-14:
        return m
    return float(m + temp * np.log(np.sum(np.exp((c - m) / temp))))

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def sg_step(h, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    L = abs(lag)
    if L == 0:
        grad = 1.0 - 2.0 * h
    else:
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            if i + L < n:
                g += (1.0 - h[i + L])
            if i - L >= 0:
                g -= h[i - L]
            grad[i] = g
    grad -= np.mean(grad)
    dx = 2.0 / n
    h_new = h - lr * grad * dx
    return fix_sum(h_new, n)

def multisg_step(h, lr, n_active=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-n_active:]
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in top_k:
        lag = int(k_idx) - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        else:
            for i in range(n):
                g = 0.0
                if i + L < n:
                    g += (1.0 - h[i + L])
                if i - L >= 0:
                    g -= h[i - L]
                grad[i] += g
    grad -= np.mean(grad)
    h_new = h - lr * grad * dx
    return fix_sum(h_new, n)

def slsqp_optimize(h, n, t_end, temps=None):
    if temps is None:
        temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0]
    h_work = h.copy()
    c5_best = evaluate_c5(h_work)
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for temp in temps:
        if time.time() > t_end:
            break
        obj = (lambda x: smooth_obj(x, temp)) if temp > 0 else evaluate_c5
        try:
            res = minimize(obj, h_work, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 600, 'ftol': 1e-16})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_work = h_new.copy()
        except Exception:
            pass
    return h_work, c5_best

    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, len(best_h))
