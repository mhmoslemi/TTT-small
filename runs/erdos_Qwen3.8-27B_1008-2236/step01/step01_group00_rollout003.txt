import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-14:
        return m
    shifted = np.minimum((corr - m) / temp, 0.0)
    return float(m + temp * np.log(np.sum(np.exp(shifted))))


def slsqp_opt(h0, n, t_limit, temps=None, max_iter=250):
    if temps is None:
        temps = [0.02, 0.005, 0.001, 0.0003, 0.0001]
    target = n / 2.0
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
             'jac': lambda h: np.ones(n)}]
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = h0.copy()
    for temp in temps:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda h: smooth_c5(h, temp), h_cur,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    if time.time() < t_limit:
        try:
            res = minimize(evaluate_c5, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-15})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5


def ternary_1d(f, a, b, n_iter=15):
    """Minimize unimodal f on [a,b] via ternary search."""
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_polish(h0, n, t_limit, max_rounds=400, tol=1e-13):
    best_c5 = evaluate_c5(h0)
    bh = h0.copy()
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        max_improve = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                for src, dst in [(i, j), (j, i)]:
                    max_d = min(bh[src], 1.0 - bh[dst])
                    if max_d < 1e-12:
                        continue
                    # Quick screen: midpoint
                    mid = max_d * 0.5
                    bh[src] -= mid
                    bh[dst] += mid
                    c_mid = evaluate_c5(bh)
                    bh[src] += mid
                    bh[dst] -= mid
                    if c_mid > best_c5 + tol:
                        # Try full transfer as edge case
                        bh[src] -= max_d
                        bh[dst] += max_d
                        c_full = evaluate_c5(bh)
                        if c_full < best_c5 - tol:
                            best_c5 = c_full
                            max_improve = max(max_improve, max_improve + 1)
                            improved = True
                        else:
                            bh[src] += max_d
                            bh[dst] -= max_d
                        continue
                    # Ternary search for optimal transfer
                    def obj_t(t):
                        bh[src] -= t
                        bh[dst] += t
                        c = evaluate_c5(bh)
                        bh[src] += t
                        bh[dst] -= t
                        return c
                    t_best = ternary_1d(obj_t, 0.0, max_d, n_iter=12)
                    c_best = obj_t(t_best)
                    if c_best < best_c5 - tol:
                        bh[src] -= t_best
                        bh[dst] += t_best
                        best_c5 = c_best
                        max_improve += best_c5 - 0  # track
                        improved = True
        if not improved:
            break
    return bh, best_c5


def block_move_polish(h0, n, t_limit, max_rounds=100):
    """Try shifting contiguous blocks of mass."""
    best_c5 = evaluate_c5(h0)
    bh = h0.copy()
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        # Try random block moves
        n_tries = 200
        for _ in range(n_tries):
            a = np.random.randint(0, n)
            b = np.random.randint(0, n)
            length = np.random.randint(1, max(2, n // 3))
            a = a % n
            a_end = min(a + length, n)
            b = b % n
            b_end = min(b + length, n)
            if a_end - a < 1 or b_end - b < 1:
                continue
            # Move mass from [a,a_end) to [b,b_end)
            src_slice = slice(a, a_end)
            dst_slice = slice(b, b_end)
            # Max we can move
            avail = bh[src_slice].sum()
            room = (1.0 - bh[dst_slice]).sum()
            max_move = min(avail, room)
            if max_move < 1e-12:
                continue
            for frac in [0.2, 0.5, 0.8]:
                move = frac * max_move
                per_src = move / (a_end - a)
                per_dst = move / (b_end - b)
                if per_src > bh[src_slice].max() + 1e-12 or per_dst > (1.0 - bh[dst_slice]).max() + 1e-12:
                    continue
                bh[src_slice] -= per_src
                bh[dst_slice] += per_dst
                bh = np.clip(bh, 0.0, 1.0)
                bh = fix_sum(bh, n)
                c = evaluate_c5(bh)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                else:
                    bh[src_slice] += per_src
                    bh[dst_slice] -= per_dst
                    bh = fix_sum(bh, n)
                    break
        if not improved:
            break
    return bh, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
