import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def power_mean_obj(h, p, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    s = np.maximum(corr, 1e-30)
    return float(np.sum(s ** p) ** (1.0 / p))

def power_mean_grad(h, p, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    s = np.maximum(corr, 1e-30)
    F = np.sum(s ** p) ** (1.0 / p)
    if F < 1e-30:
        return np.zeros(n)
    w = s ** (p - 1)
    a = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(w[j:j+n], a)
    return F ** (1 - p) * grad * dx

def lse_obj(h, temp, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def two_coord_descent(h, n, t_limit, max_rounds=200, n_per_round=500):
    best_c5 = evaluate_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(42)
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        pairs = set()
        while len(pairs) < n_per_round:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        for (i, j) in pairs:
            if time.time() > t_limit:
                break
            h_w = h_best.copy()
            # Move from high to low
            max_d = min(h_best[i], 1.0 - h_best[j])
            if max_d > 1e-5:
                for frac in [0.5, 0.25, 0.1, 0.75, 0.15]:
                    d = frac * max_d
                    h_w[i] -= d
                    h_w[j] += d
                    c5 = evaluate_c5(h_w)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        h_best = h_w.copy()
                        improved = True
                    else:
                        h_w = h_best.copy()
            # Move from j to i
            max_d2 = min(h_best[j], 1.0 - h_best[i])
            if max_d2 > 1e-5:
                for frac in [0.5, 0.25, 0.1, 0.75, 0.15]:
                    d = frac * max_d2
                    h_w = h_best.copy()
                    h_w[j] -= d
                    h_w[i] += d
                    c5 = evaluate_c5(h_w)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        h_best = h_w.copy()
                        improved = True
                    else:
                        h_w = h_best.copy()
        if not improved:
            break
    return h_best, best_c5

def single_coord_linsearch(h, n, t_limit, max_rounds=100):
