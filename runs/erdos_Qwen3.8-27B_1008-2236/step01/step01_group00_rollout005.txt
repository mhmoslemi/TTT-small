    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(h, n, iters=300):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(iters):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, temp):
    cv = corr_values(h)
    m = cv.max()
    if temp < 1e-14:
        return float(m)
    return float(m + temp * np.log(np.sum(np.exp((cv - m) / temp))))
