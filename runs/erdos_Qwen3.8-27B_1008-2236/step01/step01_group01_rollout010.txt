        for _ in range(4):
            starts.append(project_sum(np.clip(rng.random(n) * 1.5, 0, 1)))
        for _ in range(3):
            hbin = np.where(rng.random(n) < 0.5, 1.0, 0.0)
            starts.append(project_sum(hbin))
        # Checkerboard variant
        hc = np.zeros(n)
        hc[0::2] = 1.0
        starts.append(project_sum(hc))
        hc2 = np.zeros(n)
        hc2[1::2] = 1.0
        starts.append(project_sum(hc2))

        for st in starts:
            if rem() < 25:
                break
            hc = softmax_slsqp(st, [8, 25, 60, 150, 400], 200)
            vc = compute_c5_full(hc)[0]
            if vc < best_c5:
                best_h, best_c5 = hc.copy(), vc

    # Phase 3: Iterative refinement with subgradient mixing
    if rem() > 120:
        stall = 0
        while rem() > 20 and stall < 80:
            s = min(0.04 + 0.006 * stall, 0.5)
            hp = project_sum(best_h + rng.standard_normal(n) * s)
            # Subgradient pre-steps
            for lr in [0.02, 0.01, 0.005]:
                hp = subgrad_step(hp, lr, top_k=5)
            hc = softmax_slsqp(hp, [15, 50, 120, 350], 250)
            vc = compute_c5_full(hc)[0]
            if vc < best_c5 - 1e-13:
                best_h = hc.copy()
                best_c5 = vc
                stall = 0
            else:
                stall += 1

    # Phase 4: Local search (pair + coordinate + binary)
    if rem() > 50:
        hp, vp = pair_search(best_h, rng, min(rem() * 0.2, 30))
        if vp < best_c5:
            best_h, best_c5 = hp.copy(), vp
        if rem() > 30:
            hc, vc = coord_desc(best_h, rng, min(rem() * 0.25, 25))
            if vc < best_c5:
                best_h, best_c5 = hc.copy(), vc
                if rem() > 20:
                    hf = softmax_slsqp(best_h, [40, 100, 300], 250)
                    vf = compute_c5_full(hf)[0]
                    if vf < best_c5:
                        best_h, best_c5 = hf.copy(), vf
        if rem() > 20:
            hb, vb = binary_local_search(best_h, rng, min(rem() * 0.3, 20))
            if vb < best_c5:
                # Use binary as starting point for continuous refinement
                hb2 = softmax_slsqp(hb, [20, 60, 200], 200)
                vb2 = compute_c5_full(hb2)[0]
                if vb2 < best_c5:
                    best_h, best_c5 = hb2.copy(), vb2

    # Phase 5: Second refinement round
    if rem() > 60:
        stall = 0
        while rem() > 15 and stall < 30:
            s = 0.03 + 0.01 * stall
            hp = project_sum(best_h + rng.standard_normal(n) * min(s, 0.3))
            hc = softmax_slsqp(hp, [30, 100, 300, 800], 250)
            vc = compute_c5_full(hc)[0]
            if vc < best_c5 - 1e-13:
                best_h = hc.copy()
                best_c5 = vc
                stall = 0
            else:
                stall += 1

    # Phase 6: Final polish
    if rem() > 8:
        hf = softmax_slsqp(best_h, [100, 300, 800, 2000], 400)
        vf = compute_c5_full(hf)[0]
        if vf < best_c5:
            best_h, best_c5 = hf.copy(), vf
    if rem() > 3:
        hp, vp = pair_search(best_h, rng, min(rem() * 0.5, 5))
        if vp < best_c5:
            best_h, best_c5 = hp.copy(), vp

    best_h = project_sum(best_h)
    final = compute_c5_full(best_h)[0]
    return best_h, min(final, best_c5), best_n
