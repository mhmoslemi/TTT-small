    n = len(h)

def c5_full(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(h)[-20:]
        else:
            idx = np.argsort(h)[:20]
        for i in idx:
            old = h[i]
            new = np.clip(old + d, 0, 1)
            h[i] = new
            d -= (new - old)
            if abs(d) < 1e-12:
                break
    return h

def c5_obj(x):
    return compute_c5(x)

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def slsqp(h0, maxiter=300, ftol=1e-12):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5_obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return r.x, float(r.fun)

def subgrad_descent(h0, max_iter=2000, alpha0=0.01, alpha_decay=0.999):
    n = len(h0)
    h = h0.copy()
    alpha = alpha0
    best = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        one_minus_h = 1.0 - h
        c = np.correlate(h, one_minus_h, mode="full")
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        # Compute gradient
        grad = np.zeros(n)
        for j in range(n):
            i1 = j - lag
            if 0 <= i1 < n:
                grad[j] += (1.0 - h[i1])
            i2 = j + lag
            if 0 <= i2 < n:
                grad[j] -= h[i2]
        h = h - alpha * grad
        h = project_sum(h)
        cv = compute_c5(h)
        if cv < best:
            best = cv
            best_h = h.copy()
        alpha *= alpha_decay
    return best_h, best

    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)

            new = float(np.clip(old + d, 0.0, 1.0))
            h[i] = new
            d -= (new - old)
            if abs(d) < 1e-13:
                break
    return h

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(c5_obj, h0, method='SLSQP', bounds=bounds,

    """Sub-gradient with softmax weighting over top correlations."""

    h = h0.copy()
    alpha = init_alpha
    best = compute_c5(h)
    best_h = h.copy()
    stuck = 0
    for it in range(max_iter):

        c = np.correlate(h, one_minus_h, mode="full")
        # Softmax weighting
        c_max = c.max()
        weights = np.exp((c - c_max) * beta)
        weights /= weights.sum()
        # Weighted gradient
        grad = np.zeros(n)
        for k_idx in range(len(c)):

            if w < 1e-8:

            stuck = 0

            continue

            # j+lag valid: j from 0 to n-1-lag, h[j+lag] = h[lag:n]

            grad[:n-lag_abs] += w * (1.0 - h[lag_abs:])
            grad[lag_abs:] -= w * h[:n-lag_abs]
    return grad

        h0[22] = 0.5
        h0 = project_sum(h0)
    
    n = len(h0)
    h0 = project_sum(h0)
    bh = h0.copy()
    bc5 = compute_c5(bh)

    if time.time() - ts < budget_s - 250:
        hs, cs = weighted_subgrad(bh, max_iter=4000, init_alpha=0.004, decay=0.9996, beta=10)
        if cs < bc5:
            bc5, bh = cs, hs.copy()

        h, v = slsqp(bh, 300)
        h = project_sum(h)
        v = compute_c5(h)
        if v < bc5:
            bc5, bh = v, h.copy()
