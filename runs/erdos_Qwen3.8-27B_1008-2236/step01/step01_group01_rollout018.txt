def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

    return float(x.sum() - 0.5 * len(x))

    return np.ones(len(x))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (t - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.where(h < 1.0 - 1e-6)[0] if d > 0 else np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
            d2 = t - h.sum()
            if abs(d2) > 1e-12:
                idx2 = np.where(h < 1.0 - 1e-6)[0] if d2 > 0 else np.where(h > 1e-6)[0]
                if len(idx2) > 0:
                    h[idx2[0]] = float(np.clip(h[idx2[0]] + d2, 0.0, 1.0))
    return h

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def hard_obj(x):
    return compute_c5(x)

def sq_obj(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    return float(np.sum(c * c))

def slsqp(h0, obj, mi):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': mi, 'ftol': 1e-14})
    return r.x, float(r.fun)

def make_bin(n, rng, mode):
    h = np.zeros(n)
    m = n // 2
    if mode == 1:
        h[0::2] = 1.0
        if n % 2 == 1:

    elif mode == 2:

        h[np.argsort(t)[:m]] = 1.0
    elif mode == 3:
        h[:m] = 1.0
    else:
        h[rng.choice(n, m, replace=False)] = 1.0
    return project_sum(h)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (t - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.where(h < 1.0 - 1e-6)[0] if d > 0 else np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
            d2 = t - h.sum()
            if abs(d2) > 1e-12:
                idx2 = np.where(h < 1.0 - 1e-6)[0] if d2 > 0 else np.where(h > 1e-6)[0]
                if len(idx2) > 0:
                    h[idx2[0]] = float(np.clip(h[idx2[0]] + d2, 0.0, 1.0))
    return h

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def hard_obj(x):
    return compute_c5(x)

def sq_obj(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    return float(np.sum(c * c))

def slsqp(h0, obj, mi):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': mi, 'ftol': 1e-14})
    return r.x, float(r.fun)

def make_bin(n, rng, mode):
    h = np.zeros(n)
    m = n // 2
    if mode == 1:
        h[0::2] = 1.0
        if n % 2 == 1:
            h[n - 1] = 1.0
    elif mode == 2:
        t = np.zeros(n, dtype=int)
        for i in range(1, n):
            t[i] = t[i >> 1] ^ (i & 1)
        h[np.argsort(t)[:m]] = 1.0
    elif mode == 3:
        h[:m] = 1.0
    elif mode == 4:
        idx = np.arange(n)
        pos = np.argsort(np.sin(2.0 * np.pi * idx / (n * 0.3)) + 0.3 * np.sin(2.0 * np.pi * idx / (n * 0.17)))[:m]
        h[pos] = 1.0
    else:
        h[rng.choice(n, m, replace=False)] = 1.0
    return project_sum(h)

def bls(h0, rng, tl):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    if n % 2 == 1:
        h[rng.integers(n)] = 0.5
    h = project_sum(h)
    c = compute_c5(h)
    b, bh = c, h.copy()
    t0 = time.time()
    while time.time() - t0 < tl:
        imp = False
        o = np.flatnonzero(h >= 0.5)
        z = np.flatnonzero(h < 0.5)
        if len(o) == 0 or len(z) == 0:
            break
        for i in o:
            z = np.flatnonzero(h < 0.5)
            if not len(z):
                break
            for j in z:
                if i == j:
                    continue
                h[i], h[j] = 0.0, 1.0
                h = project_sum(h)
                nv = compute_c5(h)
                if nv < c - 1e-12:
                    c = nv; imp = True
                    if nv < b:
                        b = nv; bh = h.copy()
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
                    h = project_sum(h)
            if imp:
                break
        if not imp:
            o = np.flatnonzero(h >= 0.5)
            z = np.flatnonzero(h < 0.5)
            if len(o) >= 2 and len(z) >= 2:
                for _ in range(20):
                    p = [int(o[rng.integers(len(o))]), int(o[rng.integers(len(o))]),
                         int(z[rng.integers(len(z))]), int(z[rng.integers(len(z)))]]
                    if len(set(p)) != 4:
                        continue
                    sv = h[p].copy()
                    h[p[0]], h[p[1]] = 0.0, 0.0
                    h[p[2]], h[p[3]] = 1.0, 1.0
                    h = project_sum(h)
                    nv = compute_c5(h)
                    if nv < c - 1e-12:
                        c = nv; imp = True
                        if nv < b:
                            b = nv; bh = h.copy()
                        break
                    else:
                        h[p] = sv
                        h = project_sum(h)
        if not imp:
            break
    return bh, b

def coord_desc(h0, rng, tl):
    n = len(h0)
    h, c = h0.copy(), compute_c5(h0)
    b, bh = c, h.copy()
    t0 = time.time()
    while time.time() - t0 < tl:
        imp = False
        for i in range(n):
            old = h[i]
            for tr in (0.0, 0.25, 0.5, 0.75, 1.0):
                if abs(tr - old) < 1e-8:
                    continue
                hp = h.copy(); hp[i] = tr
                hp = project_sum(hp)
                nv = compute_c5(hp)
                if nv < c * (1 - 1e-6):
                    h, c = hp, nv; imp = True
                    if nv < b:
                        b = nv; bh = h.copy()
                    break
        if not imp:
            break
    return bh, c

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        hp = np.asarray(globals()['initial_h_values'], dtype=float).copy()
    else:
        hp = np.zeros(45)
        hp[0::2] = 1.0
        hp[22] = 0.5
        hp = project_sum(hp)
    hp = project_sum(hp)
    bh, bc5 = hp.copy(), compute_c5(hp)
    cands = [(hp.copy(), bc5)]

    for n in (30, 40, 45, 50, 55, 60, 70, 80):
        if time.time() - ts > budget_s - 200:
            break
        for md in (0, 1, 2, 3, 4):
            if time.time() - ts > budget_s - 200:
                break
            hi = make_bin(n, rng, md)
            ho, _ = slsqp(hi, hard_obj, 80)
            ho = project_sum(ho)
            cands.append((ho, compute_c5(ho)))

    cands.sort(key=lambda x: x[1])

    tk = min(5, len(cands))
    rem = budget_s - (time.time() - ts) - 80
    pc = max(rem / tk, 10.0)

    for ci in range(tk):
        if time.time() - ts > budget_s - 60:
            break
        hc, cc = cands[ci][0].copy(), cands[ci][1]
        nc = len(hc)
        tp = time.time()
        st = 0
        while time.time() - tp < pc - 25 and st < 15:
            s = 0.02 + 0.06 * rng.random() + 0.008 * st
            hn, _ = slsqp(project_sum(hc + rng.normal(0, s, nc)), hard_obj, 200)
            hn = project_sum(hn)
            cn = compute_c5(hn)
            if cn < cc - 1e-10:
                cc, hc, st = cn, hn.copy(), 0
            else:
                st += 1
        if time.time() - ts < budget_s - 50:
            hs, _ = slsqp(hc, sq_obj, 150)
            hs = project_sum(hs)
            cs = compute_c5(hs)
            if cs < cc - 1e-10:
                cc, hc = cs, hs.copy()
        tr = budget_s - (time.time() - ts) - 40
        if tr > 3:
            hb, cb = bls(hc, rng, min(8.0, tr * 0.4))
            if cb < cc - 1e-10:
                cc, hc = cb, hb.copy()
        if time.time() - ts < budget_s - 15:
            hf, _ = slsqp(hc, hard_obj, 300)
            hf = project_sum(hf)
            cf = compute_c5(hf)
            if cf < cc:
                cc, hc = cf, hf.copy()
        if time.time() - ts < budget_s - 5:
            hcd, ccd = coord_desc(hc, rng, min(5.0, budget_s - (time.time() - ts) - 5))
            if ccd < cc:
                cc, hc = ccd, hcd.copy()
        if cc < bc5:
            bc5, bh = cc, hc.copy()

    if time.time() - ts < budget_s - 3:
        hf, _ = slsqp(bh, hard_obj, 600)
        hf = project_sum(hf)
        cf = compute_c5(hf)
        if cf < bc5:
            bc5, bh = cf, hf.copy()

    bh = project_sum(bh)
    return bh, min(compute_c5(bh), bc5), len(bh)
