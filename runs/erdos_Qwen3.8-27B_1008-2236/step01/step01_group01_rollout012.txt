def solve_lp_cvxpy(n):

                constraints.append(z[i, j] <= h[j])
                constraints.append(z[i, j] >= h[i] + h[j] - 1)
                constraints.append(z[i, j] >= 0)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = int(np.argmax(h)) if d < 0 else int(np.argmin(h))
        h = h.copy()
        h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
    return h

def slsqp_refine(h0, maxiter=1000):
    n = len(h0)
    def obj(x):
        return compute_c5(x)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    hx = project_sum(r.x)
    return hx, compute_c5(hx)

    elif kind == 'rand':

def solve_lp(n, timeout=60):
    """Solve LP relaxation. Returns (h_lower_bound, lp_solution) or (None, None)."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    z_var = cp.Variable((n, n))
    
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2, t_var >= 0]
    
    # McCormick envelopes
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            cons.append(z_var[i,j] <= h_var[i])
            cons.append(z_var[i,j] <= h_var[j])
            cons.append(z_var[i,j] >= h_var[i] + h_var[j] - 1)
            cons.append(z_var[i,j] >= 0)
    
    # Correlation constraints
    corr_full = np.correlate(np.ones(n), np.ones(n), mode='full')
    for k in range(-(n-1), n):
        if k == 0:
            continue
        idx = n - 1 + k
        if k > 0:
            # sum_{i=0}^{n-1-k} (h[i+k] - z[i+k,i]) * dx <= t
            terms = []
            for i in range(n - k):
                terms.append(h_var[i+k] - z_var[i+k, i])
        else:
            # sum_{i=-k}^{n-1} (h[i+k] - z[i+k,i]) * dx <= t
            terms = []
            for i in range(-k, n):
                terms.append(h_var[i+k] - z_var[i+k, i])
        if terms:
            cons.append(cp.sum(terms) * dx <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)

            prob.solve(solver=cp.CLARABEL, verbose=False)

            try:
                prob.solve()
            except:
                return None, None
    
    if h_var.value is None:
        return None, None
    return t_var.value, h_var.value

import numpy as np
from scipy.optimize import minimize, differential_evolution
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = int(np.argmax(h)) if d < 0 else int(np.argmin(h))
        h = h.copy()
        h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
    return h

def slsqp_refine(h0, maxiter=1000):
    n = len(h0)
    def obj(x):
        return compute_c5(x)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                 constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    hx = project_sum(r.x)
    return hx, compute_c5(hx)

def make_init(n, kind, rng):
    if kind == 'alt':
        h = np.zeros(n); h[::2] = 1.0
    elif kind == 'block':
        h = np.zeros(n); h[:n//2] = 1.0
    elif kind == 'semi':
        h = np.zeros(n); p = max(n // 6, 2)
        for i in range(n):
            h[i] = 1.0 if (i % (2 * p)) < p else 0.0
    elif kind == 'quasi':
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (np.sin(2.3 * i + 0.7) > 0) else 0.0
    else:
        h = rng.random(n)
    return project_sum(h)

def solve_lp_relaxation(n, time_limit=90):
    t0 = time.time()
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    z_var = cp.Variable((n, n))
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2, t_var >= 0]
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            cons.append(z_var[i, j] <= h_var[i])
            cons.append(z_var[i, j] <= h_var[j])
            cons.append(z_var[i, j] >= h_var[i] + h_var[j] - 1)
            cons.append(z_var[i, j] >= 0)
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        terms = []
        if k > 0:
            for i in range(n - k):
                terms.append(h_var[i + k] - z_var[i + k, i])
        else:
            for i in range(-k, n):
                terms.append(h_var[i + k] - z_var[i + k, i])
        if terms:
            cons.append(cp.sum(terms) * dx <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, abs_gap=1e-9)
    except:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            try:
                prob.solve()
            except:
                return None, None
    if h_var.value is None or time.time() - t0 > time_limit * 2:
        return None, None
    return float(t_var.value), np.array(h_var.value)

def de_objective(x, n):
    h = np.clip(x, 0, 1)
    return compute_c5(h) + 50.0 * (h.sum() - 0.5 * n) ** 2

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = 1.0

    def tl():
        return budget_s - (time.time() - ts)

    # Phase 0: Refine parent
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:

        h0 = project_sum(h0)

                best_c5, best_h = v, h

    final_c5 = compute_c5(best_h)
    return best_h, min(final_c5, best_c5), len(best_h)
