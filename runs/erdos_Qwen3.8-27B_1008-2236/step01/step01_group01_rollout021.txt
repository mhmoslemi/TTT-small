def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining deficit
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(h)[::-1]  # most room to grow
        else:
            idx = np.argsort(h)  # most room to shrink
        for i in idx:
            if abs(d) < 1e-14:
                break
            adj = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += np.sign(d) * adj
            d -= np.sign(d) * adj
    return h

def sfun(x):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            adj = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += np.sign(d) * adj
            d -= np.sign(d) * adj
    return h

def smooth_max_obj(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c = np.maximum(c, 1e-15)
    return float(np.sum(c ** p) ** (1.0 / p))

def sfun(x):
    return float(x.sum() - 0.5 * len(x))

def sjac(x):
    return np.ones(len(x))

def slsqp_smooth(h0, p=10, maxiter=400):
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        r = minimize(lambda x: smooth_max_obj(x, p), h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(r.x, 0, 1), compute_c5(r.x)
    except Exception:
        return h0.copy(), compute_c5(h0)

def slsqp_exact(h0, maxiter=500):
    n = len(h0)

    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        r = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(r.x, 0, 1), compute_c5(r.x)
