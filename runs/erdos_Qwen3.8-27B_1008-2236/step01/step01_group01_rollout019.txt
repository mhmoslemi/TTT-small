    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Find top few lags
    top_k = np.argsort(c)[::-1][:5]
    grad = np.zeros(n)
    for k in top_k:
        lag = int(k) - (n - 1)
        cf = c[k] / c.max()  # weight by how close to max
        for j in range(n):
            g = 0.0
            j_plus = j + lag
            if 0 <= j_plus < n:
                g += (1.0 - h[j_plus])
            j_minus = j - lag
            if 0 <= j_minus < n:
                g -= h[j_minus]
            grad[j] += cf * g
    grad *= dx
    h = h - alpha * grad
    return project_sum(h)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode="full") * dx).max())


def project_sum(h):
    n = len(h)
    tgt = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (tgt - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    top = np.argsort(c)[::-1][:5]

            g[:n - p] -= w * h[p:]
    return project_sum(h - a * g)
