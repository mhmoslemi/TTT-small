import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def compute_c5_components(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h

def swr(h0, rng, tl=10.0, T0=0.05, T1=0.001):
    """Simulated annealing with swap moves for binary/near-binary h."""
    n = len(h0)
    h = h0.copy()
    c_max = compute_c5(h)
    bh = h.copy()
    b = c_max
    t0 = time.time()
    T = T0
    n_iter = 0
    while time.time() - t0 < tl:
        # Choose two indices to swap
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        # Swap values
        sv = h.copy()
        h[i], h[j] = h[j], h[i]
        nv = compute_c5(h)
        if nv <= c_max or rng.random() < np.exp(-(nv - c_max) / T):
            c_max = nv
            if nv < b:
                b = nv
                bh = h.copy()
        else:
            h = sv
        n_iter += 1
        if n_iter % 100 == 0:
            T = max(T1, T0 * (T1 / T0) ** (n_iter / 5000))
    return bh, b

def bls(h0, rng, tl=10.0):
    """Binary local search with swap and double-swap moves."""
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    h = project_sum(h)
    c = compute_c5(h)
    b = c
    bh = h.copy()
    t0 = time.time()
    while time.time() - t0 < tl:
        improved = False
        o = np.flatnonzero(h > 0.5)
        z = np.flatnonzero(h < 0.5)
        if len(o) == 0 or len(z) == 0:
            break
        # Try all single swaps
        for i in o:
            for j in z:
                h[i], h[j] = 0.0, 1.0
                nv = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if nv < c - 1e-10:
                    h[i], h[j] = 0.0, 1.0
                    c = nv
                    if nv < b:
                        b = nv
                        bh = h.copy()
                    improved = True
        if not improved:
            # Try random double swaps
            for _ in range(200):
                if time.time() - t0 >= tl:
                    break
                o = np.flatnonzero(h > 0.5)
                z = np.flatnonzero(h < 0.5)
                if len(o) < 2 or len(z) < 2:
                    break
                i1, i2 = rng.choice(o, 2, replace=False)
                j1, j2 = rng.choice(z, 2, replace=False)
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                sv = h.copy()
                h[i1], h[i2] = 0.0, 0.0
                h[j1], h[j2] = 1.0, 1.0
                nv = compute_c5(h)
                if nv < c - 1e-10:
                    c = nv
                    if nv < b:
                        b = nv
                        bh = h.copy()
                else:
                    h = sv
    return bh, b

    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    c = compute_c5(h)
    bh = h.copy()
    b = c
    t0 = time.time()
    improved = True
    passes = 0
    while improved and time.time() - t0 < tl:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nv = compute_c5(h)
                if nv < c - 1e-12:
                    c = nv
                    if nv < b:
                        b = nv
                        bh = h.copy()
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
            else:
                continue
            break
        passes += 1
        if time.time() - t0 >= tl:
            break
    return bh, b

def _binary_sa(h0, rng, tl=5.0, T0=0.12, T_end=0.002):
    """SA with single-swap moves."""
    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    c = compute_c5(h)
    bh = h.copy()
    b = c
    t0 = time.time()
    T = T0
    step = 0
    n_total = int(tl * 3000)
    while time.time() - t0 < tl:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or h[i] == h[j]:
            step += 1
            continue
        h[i], h[j] = 1 - h[i], 1 - h[j]  # swap
        nv = compute_c5(h)
        if nv <= c or rng.random() < np.exp(-(nv - c) / T):
            c = nv
            if nv < b:
                b = nv
                bh = h.copy()
        else:
            h[i], h[j] = 1 - h[i], 1 - h[j]  # revert
        step += 1
        T = max(T_end, T0 * (T_end / T0) ** (step / max(n_total, 1)))
    return bh, b

def _random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _block_binary(n, rng):
    """Binary sequence with random unequal blocks."""
    n_half = n // 2
    n_blocks = int(rng.integers(6, 20))
    # Random positive sizes summing to n
    sizes = rng.dirichlet(np.ones(n_blocks)) * n
    sizes = np.maximum(1, np.round(sizes).astype(int))
    # Adjust to sum to n
    diff = n - sizes.sum()
    while diff > 0:
        sizes[int(rng.integers(n_blocks))] += 1
        diff -= 1
    while diff < 0:
        k = int(rng.integers(n_blocks))
        if sizes[k] > 1:
            sizes[k] -= 1
            diff += 1
    h = np.zeros(n)
    start_val = int(rng.integers(2))
    pos = 0
    for b in range(n_blocks):
        h[pos:pos+sizes[b]] = start_val if b % 2 == 0 else 1 - start_val
        pos += sizes[b]
    # Fix to have exactly n/2 ones
    target = n // 2
    cur = int(h.sum())
    diff = target - cur
    if diff > 0:
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        h[rng.choice(zeros, diff, replace=False)] = 1.0
        h[rng.choice(ones, diff, replace=False)] = 0.0
    elif diff < 0:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(ones, -diff, replace=False)] = 0.0
        h[rng.choice(zeros, -diff, replace=False)] = 1.0
    return h
