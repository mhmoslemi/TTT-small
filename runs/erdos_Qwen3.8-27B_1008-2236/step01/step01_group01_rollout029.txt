            i = int(hi[rng.integers(len(hi))])
            j = int(lo[rng.integers(len(lo))])
            if i == j:
                stall += 1
                continue
            h[i], h[j] = 0.0, 1.0
            h = project_sum(h)
            nv = compute_c5(h)
            if nv <= c + 1e-13:
                c = nv
                stall = 0
                if nv < b:
                    b, bh = nv, h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
                h = project_sum(h)
                stall += 1
    return bh, b

def sweep_search(h0, rng, tlim=5.0):
    """Best-improvement 1-swap sweep with sampled 2-swaps."""
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    b, bh = c, h.copy()
    t0 = time.time()
    sweeps = 0
    while time.time() - t0 < tlim and sweeps < 200:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_gain, best_i, best_j = 1e-15, -1, -1
        if len(ones) > 0 and len(zeros) > 0:
            for i in ones:
                for j in zeros:
                    h[i], h[j] = 0.0, 1.0
                    nv = compute_c5(h)
                    h[i], h[j] = 1.0, 0.0
                    if c - nv > best_gain:
                        best_gain, best_i, best_j = c - nv, i, j
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            h = project_sum(h)
            c = compute_c5(h)
            if c < b:
                b, bh = c, h.copy()
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                improved2 = False
                for _ in range(2000):
                    i1, i2 = ones[rng.integers(len(ones), 2)]
                    j1, j2 = zeros[rng.integers(len(zeros), 2)]
                    if len(set([i1, i2, j1, j2])) < 4:
                        continue
                    sv = [float(h[i1]), float(h[i2]), float(h[j1]), float(h[j2])]
                    h[i1], h[i2] = 0.0, 0.0
                    h[j1], h[j2] = 1.0, 1.0
                    h = project_sum(h)
                    nv = compute_c5(h)
                    if nv < c:
                        c = nv
                        if nv < b:
                            b, bh = c, h.copy()
                        improved2 = True
                        break
                    else:
                        h[i1], h[i2], h[j1], h[j2] = sv
                if not improved2 and not improved2:
                    break
            else:
                break
        sweeps += 1
    return bh, b

def anneal(h0, rng, tlim=12.0):
    n = len(h0)
    h = h0.copy()
    c = compute_c5(h)
    b, bh = c, h.copy()
    t0 = time.time()
    T = 0.04
    while time.time() - t0 < tlim and T > 1e-9:
        hi = np.where(h > 0.3)[0]
        lo = np.where(h < 0.7)[0]
        if len(hi) == 0 or len(lo) == 0:
            break
        if rng.random() < 0.65:
            i = int(hi[rng.integers(len(hi))])
            j = int(lo[rng.integers(len(lo))])
            if i == j:
                T *= 0.999
                continue
            sv = (float(h[i]), float(h[j]))
            h[i], h[j] = 0.0, 1.0
            h = project_sum(h)
            nv = compute_c5(h)
            if nv <= c or rng.random() < np.exp(-(nv - c) / max(T, 1e-12)):
                c = nv
                if nv < b:
                    b, bh = nv, h.copy()
            else:
                h[i], h[j] = sv
                h = project_sum(h)
        else:
            kk = int(rng.integers(2, 5))
            idx = rng.choice(n, kk, replace=False)
            sv = h[idx].copy()
            h[idx] = np.clip(h[idx] + rng.normal(0, 0.25, kk), 0, 1)
            h = project_sum(h)
            nv = compute_c5(h)
            if nv <= c or rng.random() < np.exp(-(nv - c) / max(T, 1e-12)):
                c = nv
                if nv < b:
                    b, bh = nv, h.copy()
            else:
                h[idx] = sv
                h = project_sum(h)
        T *= 0.999
    return bh, b

def gen_starts(n, rng, init=None):
    starts = []
    if init is not None and len(init) == n:
        starts.append(init.copy())
    h = np.zeros(n)
    h[0::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.5
    starts.append(project_sum(h))
    h = np.zeros(n)
    h[1::2] = 1.0
    if n % 2 == 1:
        h[0] = 0.5
    starts.append(project_sum(h))
    for p in [3, 4, 5]:
        h = np.zeros(n)
        h[0::p] = 1.0
        starts.append(project_sum(h))
    for _ in range(2):
        starts.append(project_sum(rng.random(n)))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.default_rng(seed)
    init = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        init = np.asarray(globals()['initial_h_values'], dtype=float).copy()
        init = project_sum(init)
    bh, bc5 = None, 1.0
    n_list = [44, 45, 36, 50, 42]
    if init is not None:
        ni = len(init)
        n_list = [ni] + [x for x in n_list if x != ni]
    for n in n_list:
        if time.time() - ts > budget_s - 50:
            break
        starts = gen_starts(n, rng, init)
        bn_c5, bn_h = 1.0, None
        for h0 in starts:
            if time.time() - ts > budget_s - 50:
                break
            h, v = slsqp(h0, 400)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bn_c5:
                bn_c5, bn_h = v, h.copy()
            hs, vs = local_search(bn_h, rng, 3.5)
            if vs < bn_c5:
                bn_c5, bn_h = vs, hs.copy()
            h, v = slsqp(bn_h, 250)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bn_c5:
                bn_c5, bn_h = v, h.copy()
        if bn_h is None:
            continue
        if time.time() - ts < budget_s - 50:
            hs, vs = sweep_search(bn_h, rng, 5.0)
            if vs < bn_c5:
                bn_c5, bn_h = vs, hs.copy()
            ha, va = anneal(bn_h, rng, 10.0)
            if va < bn_c5:
                bn_c5, bn_h = va, ha.copy()
            h, v = slsqp(bn_h, 500)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bn_c5:
                bn_c5, bn_h = v, h.copy()
            hs, vs = local_search(bn_h, rng, 3.0)
            if vs < bn_c5:
                bn_c5, bn_h = vs, hs.copy()
        if bn_c5 < bc5:
            bc5, bh = bn_c5, bn_h.copy()
    if bh is not None and time.time() - ts < budget_s - 8:
        rem = budget_s - (time.time() - ts) - 8
        if rem > 10:
            h, v = slsqp(bh, 600)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bc5:
                bc5, bh = v, h
            hs, vs = local_search(bh, rng, min(rem * 0.4, 8.0))
            if vs < bc5:
                bc5, bh = vs, hs
            h, v = slsqp(bh, 400)
            h = project_sum(h)
            v = compute_c5(h)
            if v < bc5:
                bc5, bh = v, h
    if bh is None:
        bh = np.zeros(44)
        bh[0::2] = 1.0
        bh = project_sum(bh)
        bc5 = compute_c5(bh)
    bh = project_sum(bh)
    final = compute_c5(bh)
    return bh, min(final, bc5), len(bh)
