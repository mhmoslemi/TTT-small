    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, n):
    """Return (c5_value, subgradient_of_c5)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # d/dh[j]: for j >= lag, j < n: (1-h[j-lag]) [from h[j]*(1-h[j-lag])]
        #          for j < n-lag: -h[j+lag] [from (1-h[j])*h[j+lag]]
        for i in range(n - lag):
            grad[i + lag] += (1.0 - h[i])
            grad[i] -= h[i + lag]
    else:
        m = -lag
        for i in range(n - m):
            grad[i + m] += (1.0 - h[i])
            grad[i] -= h[i + m]
    return c5, grad * dx

def coordinate_descent(h, n, target, t_limit, n_rounds=50):

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

                        h_work[i_dst] -= d

_state = {'n': 45, 'temp': 0.01}

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i + lag] += (1.0 - h[i])
            grad[i] -= h[i + lag]
    else:
        m = -lag
        for i in range(n - m):
            grad[i + m] += (1.0 - h[i])
            grad[i] -= h[i + m]
    return grad * dx

    n = _state['n']
    temp = _state['temp']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
