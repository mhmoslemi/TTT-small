def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        residual = target - h.sum()
        if abs(residual) > 1e-15:
            mask = (h > 1e-10) & (h < 1.0 - 1e-10)
            if mask.sum() > 0:
                h[mask] += residual / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = c.max()
    ec = np.exp(c - cm)
    return cm + np.log(ec.sum()) / tau

def lse_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = c.max()
    ec = np.exp(c - cm)
    w = ec / ec.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for m in range(2 * n - 1):
        lag = lags[m]
        a = max(0, lag)
        b = min(n, lag + n)
        if b > a:
            g[a:b] += w[m] * (1.0 - x[a - lag:b - lag])
        a2 = max(0, -lag)
        b2 = min(n, n - lag)
        if b2 > a2:
            g[a2:b2] -= w[m] * x[a2 + lag:b2 + lag]
    g *= dx
    return g
