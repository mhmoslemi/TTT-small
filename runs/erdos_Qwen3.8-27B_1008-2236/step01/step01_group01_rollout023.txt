                h[i], h[j] = 1.0, 0.0
                if c_new < best_c - 1e-14:

                    best_pair = (i, j)
        if best_pair is not None:

            h[i], h[j] = 0.0, 1.0
            c = compute_c5(h)
            if c < b:
                b = c
                bh = h.copy()
            stuck = 0
        else:

                h[j1], h[j2] = 1.0, 1.0

def compute_c5(h):
    n = len(h)

    return float(c.max())

def project_sum(h):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx = np.where(h > 1e-6)[0]
        if len(idx) > 0:
            h[idx[0]] = float(np.clip(h[idx[0]] + d, 0.0, 1.0))
    return h
