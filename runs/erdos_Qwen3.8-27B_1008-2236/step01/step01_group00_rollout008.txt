def min_max_linear(A, B, t_lo=0.0, t_hi=1.0):
    """Minimize max_k(A[k] + B[k]*t) over t in [t_lo, t_hi]."""
    # Golden section search for 1D convex minimization
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = np.max(A + B * c)
    fd = np.max(A + B * d)

            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = np.max(A + B * c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = np.max(A + B * d)
    t_star = (a + b) / 2.0

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n, tol=1e-13, max_iter=2000):
    h = h.copy()
    target = n / 2.0
    for _ in range(max_iter):
        d = target - h.sum()
        if abs(d) < tol:
            return h
        h = np.clip(h, 0.0, 1.0)
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return h

def min_max_linear(A, B):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(A + B * c)
    fd = np.max(A + B * d)
    for _ in range(55):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = np.max(A + B * c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = np.max(A + B * d)
    return (a + b) * 0.5

def coord_descent(h_init, t_limit, rng):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    stall = 0
    while time.time() < t_limit:
        order = rng.permutation(n)
        for j in order:
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            B = c1 - c0
            h[j] = min_max_linear(c0, B)
        h = fix_sum(h, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            scale = min(0.05, 0.01 * (stall // 40))
            h = fix_sum(np.clip(h + rng.randn(n) * scale, 0.0, 1.0), n)
            stall = 0
    best_h = fix_sum(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5

def slsqp_polish(h_init, t_limit, n):
    from scipy.optimize import minimize
    dx = 2.0 / n
    def smooth_obj(hh, temp):
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        m = np.max(corr)
        if temp < 1e-12:
            return m
        return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda hh: hh.sum() - n / 2.0}]
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    h_cur = best_h.copy()
    for temp in [0.02, 0.005, 0.001, 0.0005]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(lambda hh: smooth_obj(hh, temp), h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to_n(parent, n):
    if n == len(parent):
        return parent.copy()
    pos_old = np.linspace(0.0, 1.0, len(parent))
    pos_new = np.linspace(0.0, 1.0, n)
    return np.interp(pos_new, pos_old, parent)

def make_structured(n, kind, rng):
    if kind == 0:
        return np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        mid = n // 2
        h[:mid] = 1.0
        h[mid:mid + (n - 2 * mid)] = 0.5
        return h
    elif kind == 2:
        h = np.zeros(n)
        h[:(n // 3)] = 1.0
        h[(n // 2):(2 * n // 3)] = 1.0
        return h
    else:
        h = np.zeros(n)
        for i in range(0, n, max(1, n // 5)):
            w = max(1, n // 10)
            h[i:i + w] = 1.0
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    t0 = time.time()

    n_entries = len(n_list)
    total_time = budget_s - 80
    time_per_n = total_time / n_entries
    
    for idx, n in enumerate(n_list):
        t_start = time.time()
        t_limit = min(t0 + (idx + 1) * time_per_n + 20, t0 + budget_s - 80)

            continue
        
        h_init = interpolate_to_n(parent, n)
        h_init = fix_sum(h_init, n)
        c5_init = evaluate_c5(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()
            best_n = n
        
        sub_limit = min(t_limit, time.time() + time_per_n * 0.7)
        h_opt, c5_opt = coord_descent(h_init, sub_limit, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        
        # Try structured starts if time remains
        if time.time() < t_limit - 10:
            for kind in range(4):

                    break
                h_struct = fix_sum(make_structured(n, kind, rng), n)
                h_s2, c5_s2 = coord_descent(h_struct, min(time.time() + 30, t_limit), rng)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = h_s2.copy()
                    best_n = n
    
    # SLSQP polish on best solution
    time_left = t0 + budget_s - 10 - time.time()
    if time_left > 15:
        n = best_n
        h_pol, c5_pol = slsqp_polish(best_h, t0 + budget_s - 10, n)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        # Coord descent from polished point
        if time.time() < t0 + budget_s - 5:
            h_cd, c5_cd = coord_descent(best_h, t0 + budget_s - 5, rng)
            if c5_cd < best_c5:

                best_h = h_cd.copy()

    time_left = t0 + budget_s - 3 - time.time()
    if time_left > 10:
        n = best_n
        per_try = max(time_left / 4, 5)
        for att in range(4):
            if time.time() > t0 + budget_s - 3:

            h_op, c5_op = coord_descent(h_try, time.time() + per_try, rng)

                best_c5 = c5_op
                best_h = h_op.copy()
    
    final_h = fix_sum(best_h, best_n)
    final_c5 = evaluate_c5(final_h)
    return (final_h, final_c5, best_n)
