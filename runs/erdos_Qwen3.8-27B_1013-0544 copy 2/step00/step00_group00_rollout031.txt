from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    s = np.exp(temp * (corr - max_c))
    return (max_c + np.log(np.sum(s)) / temp) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(mask) == 0:
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

def coordinate_descent(h, n_levels=20, max_iter=50):
    """Refine solution using coordinate descent."""
    n = len(h)
    target_sum = n / 2.0
    current_c5 = compute_c5(h)
    
    for it in range(max_iter):
        improved = False
        for j in range(n):
            best_val = h[j]
            best_c5 = current_c5
            for level in np.linspace(0, 1, n_levels):
                h_trial = h.copy()
                h_trial[j] = level
                h_trial = project_feasible(h_trial, target_sum)
                c = compute_c5(h_trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_val = h_trial.copy()
            if not np.array_equal(h, best_val):
                h = best_val
                current_c5 = best_c5
                improved = True
        if not improved:
            break
    return h, current_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(temp * (corr - m))
    return (m + np.log(np.sum(s)) / temp) * dx
