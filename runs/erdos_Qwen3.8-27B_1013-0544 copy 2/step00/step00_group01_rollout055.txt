    return h_bin, compute_c5(h_bin)

import numpy as np
import time


def project_to_set(h, n):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_gradient(h, T):
    """Gradient of smooth C5 = (2/n) * LSE(corr, T)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ex = np.exp((corr - m) / T)
    sx = np.sum(ex)
    w = ex / sx
    grad = np.zeros(n)
    for m_idx in range(len(w)):
        wm = w[m_idx]
        if wm < 1e-10:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            i_vals = np.arange(0, n - k)
        else:
            i_vals = np.arange(-k, n)
        if i_vals.size == 0:
            continue
        j1 = i_vals + k
        j2 = i_vals
        grad[j1] += wm * (1.0 - h[i_vals])
        grad[j2] += wm * (-h[j1])
    return (2.0 / n) * grad


def interpolate_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

    it = 0

    while time.time() - t_start < max_time:
        T = max(T0 * (0.998 ** it), T_min)
        grad = compute_gradient(h, T)
        it += 1

        vel = momentum * vel - step * grad
        h_new = project_to_set(h + vel, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:

            best_c5 = c5_new

                    best_c5 = c5_try
