    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    # Find the active constraint
    max_idx = np.argmax(conv)
    # conv[max_idx] = sum of h[a] * (1-h[b]) where a-b = max_idx - (n-1)
    shift = max_idx - (n - 1)
    # Gradient w.r.t. h
    grad = np.zeros(n)
    if shift >= 0:
        # conv = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
        # d/dh[j] = (1-h[j-shift]) * [shift <= j <= n-1] - h[j+shift] * [0 <= j <= n-1-shift]
        for i in range(n - shift):
            # h[i+shift] contributes: d/dh[i+shift] = (1-h[i])
            grad[i + shift] += (1.0 - h[i])
            # (1-h[i]) contributes: d/dh[i] = -h[i+shift]
            grad[i] -= h[i + shift]
    else:
        s = -shift
        # conv = sum_{i=s}^{n-1} h[i] * (1-h[i-s])
        for i in range(s, n):
            grad[i] += (1.0 - h[i - s])
            grad[i - s] -= h[i]
    return c5, grad * dx

    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0, 1)

    return np.max(conv) * dx

def c5_subgrad(h):
    """Returns (c5, subgradient) where subgradient is for the active constraint."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        for i in range(n - shift):
            grad[i + shift] += (1.0 - h[i])
            grad[i] -= h[i + shift]
    else:
        s = -shift
        for i in range(s, n):
            grad[i] += (1.0 - h[i - s])
            grad[i - s] -= h[i]
    return c5, grad * dx

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on offset."""
    h = h.copy()
    target = n / 2.0
    
    def sum_after_clip(offset):
        return np.clip(h + offset, 0, 1).sum()
    
    # Bisection to find offset such that sum = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_after_clip(mid) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def c5_grad(h, smooth_t=0.0):
    """Returns (c5, gradient). If smooth_t > 0, uses soft-max gradient."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    
    if smooth_t > 0:
        # Soft-max gradient
        shifted = conv / smooth_t
        shifted -= np.max(shifted)  # for numerical stability
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for idx in range(len(conv)):
            if weights[idx] < 1e-10:
                continue
            shift = idx - (n - 1)
            g = np.zeros(n)
            if shift >= 0:
                if shift < n:
                    g[shift:] += (1.0 - h[:n-shift])
                g[:n-shift] -= h[shift:]
            else:
                s = -shift
                g[:n-s] += (1.0 - h[s:])
                g[s:] -= h[:n-s]
            grad += weights[idx] * g
        c5 = (smooth_t * np.log(np.sum(np.exp(conv / smooth_t))) ) * dx
        # Actually c5 should still be the hard max for tracking
        c5 = np.max(conv) * dx
        return c5, grad * dx
    else:

        idx = np.argmax(conv)
        shift = idx - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)

def soft_max_grad(h, t):
