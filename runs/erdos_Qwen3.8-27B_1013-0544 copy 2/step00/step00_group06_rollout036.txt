    target = n / 2.0

    lo, hi = -h.max() - 2, -h.min() + 2

        if s > target:

def _eq_fun(h):
    return np.sum(h) - _g_n / 2.0

def _c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def _c5(h):
    return np.max(_c5_vals(h))

def _smooth_max(vals, eps):
    m = vals.max()
    return m + eps * np.log(np.mean(np.exp((vals - m) / eps)))

def _project(h, n):
    """Project h to {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 2, h.max() + 2
    for _ in range(200):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0, 1)
        s = hp.sum()
        if s > target:
            lo = mid
        else:

        else:
            mask = hp > 0
            if mask.sum() > 0:
                hp[mask] += diff / mask.sum()

    return hp
