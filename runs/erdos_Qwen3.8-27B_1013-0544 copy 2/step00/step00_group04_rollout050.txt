def project_onto_feasible(g, S):
    """Project g onto {h: 0 ≤ h ≤ 1, sum(h) = S}."""
    n = len(g)
    # Bisection on λ
    lo, hi = g.min() - 1, g.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(g - mid, 0, 1)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2, 0, 1)
    # Fix numerical issues
    diff = S - h.sum()
    # Distribute the difference
    order = np.argsort(g)
    for idx in order:
        if diff > 0:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
        elif diff < 0:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
        if abs(diff) < 1e-10:
            break
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def local_search(h, rng, time_limit=60.0):
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    t0 = time.time()
    no_improve = 0
    max_no_improve = 20

    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_imp = 0.0
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                nc5 = compute_c5(h)
                h[j] = 0.0; h[i] = 1.0
                imp = c5 - nc5
                if imp > best_imp:
                    best_imp = imp
                    best_move = (i, j)
        if best_move is None:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            # Perturbation: random swap
            i = rng.choice(ones); j = rng.choice(zeros)
            h[i] = 0.0; h[j] = 1.0
            nc5 = compute_c5(h)
            if nc5 < c5:
                c5 = nc5
                no_improve = 0
            # else: keep the perturbation (walk)
        else:
            no_improve = 0
            i, j = best_move
            h[i] = 0.0; h[j] = 1.0
            c5 = compute_c5(h)
    return h, compute_c5(h)

def double_swap_search(h, rng, time_limit=60.0):

def simulated_anneal(h, c5, rng, time_limit=60.0):

def upsample(h, target_n):
    n = len(h)
    if target_n == n:
        return h.copy()
    x_old = np.arange(n) + 0.5
    x_new = np.arange(target_n) + 0.5
    h_interp = np.interp(x_new * n / target_n, x_old, h)
    h_bin = (h_interp >= 0.5).astype(float)
    n_target = target_n // 2
    n_ones = int(h_bin.sum())
    if n_ones > n_target:
        idx_ones = np.where(h_bin == 1)[0]
        sorted_idx = idx_ones[np.argsort(h_interp[idx_ones])[:n_ones - n_target]]
        h_bin[sorted_idx] = 0
    elif n_ones < n_target:
        idx_zeros = np.where(h_bin == 0)[0]
        sorted_idx = idx_zeros[np.argsort(-h_interp[idx_zeros])[:n_target - n_ones]]
        h_bin[sorted_idx] = 1
    return h_bin

def project_sum(h, S):
    """Adjust h to have sum exactly S while staying in [0,1]."""
    n = len(h)
    diff = S - h.sum()
    if abs(diff) < 1e-10:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(-h)
    per = abs(diff) / n
    for idx in order:
        if diff > 0:
            add = min(per * 1.5, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            rem = min(per * 1.5, h[idx])
            h[idx] -= rem
            diff += rem
        if abs(diff) < 1e-12:
            break
    h = np.clip(h, 0, 1)
    return h

        improved = False
        for _ in range(200):

            idx = rng.randint(n)

                h[idx] = np.clip(h[idx] + delta, 0, 1)
                h = project_sum(h, S)
                nc5 = compute_c5(h)
                if nc5 < c5:
                    c5 = nc5
                    improved = True
                    break
                else:
                    h[idx] = np.clip(h[idx], 0, 1)
                    h = project_sum(h, S)
        if not improved:
            step *= 0.7
    return h, c5
