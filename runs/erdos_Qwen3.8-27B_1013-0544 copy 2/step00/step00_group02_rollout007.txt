    """Project onto {0 <= x <= 1, sum(x) = target_sum} using bisection on Lagrange multipliers."""

        """Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum"""
        # h - lambda, clipped to [0,1]
        # = max(0, min(1, h - lambda))
        # As lambda increases, sum decreases

            else:
