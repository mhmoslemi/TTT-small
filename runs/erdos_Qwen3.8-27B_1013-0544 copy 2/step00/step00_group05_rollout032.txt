```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_eval(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    deficit = target - s
    if deficit > 1e-12:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += deficit * room / total
    elif deficit < -1e-12:
        room = h
        total = room.sum()
        if total > 1e-12:
            h += deficit * room / total
    return np.clip(h, 0.0, 1.0)

def nm_obj(x, n):
    return c5_eval(repair(x, n), n)

def subgrad(corr, h, n, n_pts):
    """Compute subgradient of max(corr) w.r.t. h"""
    max_val = np.max(corr)
    # Active set: where corr is close to max
    active = corr > max_val - 1e-6
    # For each active k (array index), compute gradient
    k_shifts = np.arange(len(corr)) - (n_pts - 1)
    grad = np.zeros(n_pts)
    for idx in np.where(active)[0]:
        k = k_shifts[idx]  # actual shift
        # d corr[k] / d h[m] = (1-h[m-k])*[valid] - h[m+k]*[valid]
        for m in range(n_pts):
            val = 0.0
            if 0 <= m - k < n_pts:
                val += (1.0 - h[m - k])
            if 0 <= m + k < n_pts:
                val -= h[m + k]
            grad[m] += val / np.sum(active)
    return grad

def run(seed=42, budget_s=1100, **kwargs):
```

```python
    return float(np.max(corr) * dx)

        if tot > 1e-12:

        room = h
        tot = room.sum()
        if tot > 1e-12:
            h += d * room / tot
    return np.clip(h, 0.0, 1.0)

    """Compute subgradient of C5 objective at h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    # Soft active set: use softmax weighting for smoothness
    # Scale for numerical stability
    scale = 1.0 / (max(1e-8, max_val * 0.01))
    weights = np.exp((corr - max_val) * scale)
    weights /= weights.sum()

    # d corr[k] / d h[m] = (1-h[m-k])*[0<=m-k<n] - h[m+k]*[0<=m+k<n]
    k_shifts = np.arange(len(corr)) - (n_pts - 1)
    grad = np.zeros(n_pts)

        w = weights[idx]
        if w < 1e-10:
            continue
        k = k_shifts[idx]
        # Contribution from first term: sum_m (1-h[m-k])*[valid]*w
        # m-k valid: 0 <= m-k < n_pts => k <= m < k + n_pts
        lo1 = max(0, k)
        hi1 = min(n_pts, k + n_pts)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[np.arange(lo1, hi1) - k])
```

3. Check if other shifts have become the new max
4. Repeat

For step 2, the gradient of corr[k*] w.r.t. h is:
d corr[k*] / d h[m] = (1-h[m-k*]) * [valid] - h[m+k*] * [valid]

We can do gradient descent on corr[k*] (with projection onto the feasible set).

Actually, let me reconsider the whole approach one more time.

The problem is: minimize max_k corr[k] * dx, subject to sum(h) = n/2, 0 ≤ h ≤ 1.

Key observations:
1. The objective is the max of 2n-1 quadratic functions (each corr[k] is quadratic in h)
2. The feasible region is a polytope (intersection of a hyperplane with a box)
3. The problem is non-convex (max of non-convex quadratics... wait, is corr[k] convex or concave in h?)

corr[k] = sum_n h[n+k](1-h[n]) = sum_n h[n+k] - sum_n h[n+k]h[n]

The first term is linear in h. The second term is a quadratic form with matrix -Q where Q has 1s on the (n+k, n) and (n, n+k) positions. This is a symmetric matrix with -1s off-diagonal, which is not positive semidefinite in general. So corr[k] is not convex (and not concave) in h.

So the problem is genuinely non-convex, and we can't expect polynomial-time algorithms. Local optimization is the way to go.
