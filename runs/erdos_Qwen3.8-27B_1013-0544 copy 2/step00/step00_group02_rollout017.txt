def project_onto_constraint(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target}."""
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to increase sum: increase smallest values
        order = np.argsort(h)
        for i in order:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-15:

    else:
        # Need to decrease sum: decrease largest values
        order = np.argsort(-h)
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-15:
                break
    return h

    dx = 2.0 / n

    return np.sum(x) - target

    for _ in range(iters):

        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0, 1
        nc5 = compute_c5(b.astype(float))
        if nc5 < c5:
            c5 = nc5
        else:
