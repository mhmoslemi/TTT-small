For τ≠0: c[τ] is linear in h[j].

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=300, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(
        lambda x: compute_c5(x, dx),
        h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    return project_feasible(res.x, n)
