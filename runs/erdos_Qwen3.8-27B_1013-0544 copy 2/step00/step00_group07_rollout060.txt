import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def adjust_sum(h, target_sum):
    """Adjust h to have the target sum while maintaining bounds."""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    n = len(h)
    if diff > 0:
        # Need to increase sum; add to elements < 1
        mask = h < 1.0 - 1e-12
        if not np.any(mask):
            return h
        num_add = np.sum(mask)
        h[mask] += diff / num_add
    else:
        # Need to decrease sum; subtract from elements > 0
        mask = h > 1e-12
        if not np.any(mask):
            return h
        num_sub = np.sum(mask)
        h[mask] += diff / num_sub
    return np.clip(h, 0, 1)

    c = np.correlate(h, 1-h, mode="full")
    max_val = np.max(c)
    
    # Find all lags within epsilon of max
    active = np.where(c >= max_val - 0.01)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:
        j = k - (n-1)  # lag
        # For each valid i, grad[m] gets contribution from h[i] and h[i+j]
        # h[i] contributes: (1-h[i+j]) to grad[i]

            for i in valid_i:
                grad[i] += (1-h[i+j])
                grad[i+j] -= h[i]
