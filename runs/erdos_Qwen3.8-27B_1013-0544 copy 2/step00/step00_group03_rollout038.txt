from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 from h values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_and_grad(h, dx):
    """Compute C5 and its subgradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_vals = corr * dx
    idx = np.argmax(c5_vals)
    # Gradient of the active correlation
    # c[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # Let lag = idx - (n-1)
    lag = idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        j = i - lag  # index in (1-h)
        if 0 <= j < n:
            # h[i] * (1-h[j])
            # d/dh[i] = (1-h[j])
            # if i != j: d/dh[j] of h[i]*(1-h[j]) = -h[i]

            if i != j:
                grad[i] += 0  # wait, we need d/dh[j]

    # where lag = k - (n-1), and valid means 0 <= i < n and 0 <= i-lag < n
    
    # d/dh[m] c[k] = sum over valid i where i=m: (1-h[m-lag])
    #                + sum over valid i where i-lag=m: -h[i] = -h[m+lag]
    
    grad = np.zeros(n)
    for i in range(n):

        if 0 <= j < n:
            # term: h[i] * (1-h[j])
            # d/dh[i]: (1-h[j])
            grad[i] += (1.0 - h[j])
            # d/dh[j]: -h[i] (only if j != i)

    hi = float(y.max()) + 1.0
    for _ in range(80):

        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

        return h0
