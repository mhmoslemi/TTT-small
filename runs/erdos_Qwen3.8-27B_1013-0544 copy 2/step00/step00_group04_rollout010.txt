import numpy as np
from time import time

def compute_c5(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def max_corr(h):
    """Just the max correlation value (before multiplying by dx)."""
    corr = np.correlate(h, 1 - h, mode='full')
    return int(np.max(corr))

def hill_climb(h, max_rounds=200, deadline=None):
    """Best-improvement hill climbing with all-pair swaps on binary h."""
    h = h.copy().astype(int)
    n = len(h)
    dx = 2.0 / n
    mc = max_corr(h)  # integer max correlation
    
    for _ in range(max_rounds):
        if deadline is not None and time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_mc = mc
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_mc = max_corr(h)
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_i, best_j = i, j
                h[i] = 1
                h[j] = 0
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            mc = best_mc
        else:
            break
    
    return h, mc * dx

import numpy as np
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def max_corr_int(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return int(np.max(corr))

def hill_climb(h, max_rounds=200, deadline=None):
    h = h.copy().astype(int)
    n = len(h)
    dx = 2.0 / n
    mc = max_corr_int(h)
    for _ in range(max_rounds):
        if deadline is not None and time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_mc = mc
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                nm = max_corr_int(h)
                if nm < best_mc:
                    best_mc = nm
                    best_i, best_j = i, j
                h[i] = 1
                h[j] = 0
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            mc = best_mc
        else:
            break
    return h, mc * dx

def sa_binary(h, n_iter, rng, deadline=None):
    h = h.copy().astype(int)
    n = len(h)
    dx = 2.0 / n
    mc = max_corr_int(h)
    best_h = h.copy()
    best_mc = mc
    t0, t1 = 5.0, 0.01
    for it in range(n_iter):
        if deadline is not None and time() > deadline:
            break
        t = t0 * (t1 / t0) ** (float(it) / max(float(n_iter - 1), 1.0))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        nm = max_corr_int(h)
        if nm <= mc or (t > 1e-10 and rng.random() < np.exp(-(nm - mc) / t)):
            mc = nm
            if mc < best_mc:
                best_mc = mc
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_mc * dx

def resize_binary(h, n_new, rng):
    target = n_new // 2
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) / float(n_old)
    x_new = np.arange(n_new, dtype=float) / float(n_new)
    vals = np.interp(x_new, x_old, h.astype(float))
    h_new = (vals >= 0.5).astype(int)
    s = h_new.sum()
    if s > target:
        idx = np.where(h_new == 1)[0]
        h_new[rng.choice(idx, s - target, replace=False)] = 0
    elif s < target:
        idx = np.where(h_new == 0)[0]
        h_new[rng.choice(idx, target - s, replace=False)] = 1
    return h_new

def random_binary(n, rng):
    h = np.zeros(n, dtype=int)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h
