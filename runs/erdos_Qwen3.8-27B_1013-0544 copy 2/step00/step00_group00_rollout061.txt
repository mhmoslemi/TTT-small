def compute_c5(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)

    return np.max(corr) * dx

def smooth_max(values, temperature):
    """Smoothed maximum using log-sum-exp"""
    m = np.max(values)
    return m + temperature * np.log(np.sum(np.exp((values - m) / temperature)))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    weights = np.exp((corr - max_c) / T)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        w = weights[k_idx] * dx
        if w < 1e-10:
            continue
        for j in range(n):
            if 0 <= j - k < n:
                grad[j] += w * (1.0 - h[j - k])
            if 0 <= j + k < n:
                grad[j] -= w * h[j + k]
    return grad

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    # Bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    return 1.0 - (np.arange(n) + 0.5) / n

    s = (n / 2.0) / h.sum()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    w = np.exp((corr - mc) / T)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx] * dx
        if wk < 1e-12:
            continue
        lo_j = max(0, k)
        hi_j = min(n, n + k)
        for j in range(lo_j, hi_j):
            grad[j] += wk * (1.0 - h[j - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        for j in range(lo2, hi2):
            grad[j] -= wk * h[j + k]
    return grad

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_ramp(n):
    return 1.0 - (np.arange(n) + 0.5) / n

def make_parametric(n, p):
    x = (np.arange(n) + 0.5) / n
    h = np.power(1.0 - x, p)
    s = (n / 2.0) / h.sum()
    return np.clip(h * s, 0.0, 1.0)

def make_perturbed(n, amp, freq, phase):
    h = make_ramp(n).copy()
    x = (np.arange(n) + 0.5) / n
    h = h + amp * np.sin(freq * np.pi * x + phase)
    return np.clip(h, 0.0, 1.0)

def optimize_from(h0, n, target_sum, time_limit, T_init=0.05, lr_init=0.05):
    h = project(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    lr = lr_init
    start = time.time()
    stall = 0
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        grad = compute_grad(h, T)
        h_new = project(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.03, 0.3)
            stall = 0
        else:
            lr *= 0.93
            stall += 1
            if stall > 200:
                lr = lr_init * 0.5
                T = max(0.003, T * 0.8)
                stall = 0
        T = max(0.003, T * 0.9995)
    return best_h, best_c5

    t0 = time.time()
    n = 45
    target_sum = n / 2.0

            best_h = h_opt.copy()
            n = n_try
            target_sum = ts

    # Final polish from best
    remaining = budget_s * 0.95 - (time.time() - t0)
    if remaining > 5 and best_h is not None:

                                       T_init=0.01, lr_init=0.01)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = project(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
