```python
def fw_step(h, grad, target_sum):
    """One Frank-Wolfe step: find s = argmin <grad, v> over feasible set"""
    # Sort grad ascending
    order = np.argsort(grad)
    s = np.zeros(len(h))
    # Set s_i = 1 for the smallest target_sum indices
    n_full = int(target_sum)
    frac = target_sum - n_full
    s[order[:n_full]] = 1.0
    if frac > 0:
        s[order[n_full]] = frac
    return s
```
