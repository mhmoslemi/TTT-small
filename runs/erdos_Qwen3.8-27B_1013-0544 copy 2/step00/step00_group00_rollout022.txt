import numpy as np
from scipy.optimize import minimize
import time

_obj_n = 45
_obj_alpha = 100.0

def _smooth_obj(h):
    dx = 2.0 / _obj_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = _obj_alpha * corr * dx
    xmax = np.max(x)
    return xmax / _obj_alpha + np.log(np.sum(np.exp(x - xmax))) / _obj_alpha

def _topk_obj(h):
    dx = 2.0 / _obj_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr * dx
    idx = np.argsort(x)[::-1][:8]
    return np.sum(x[idx])

def _con_fun(h):
    return np.sum(h) - _obj_n * 0.5

def _con_jac(h):
    return np.ones(_obj_n)

def _project(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        i = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h

def _slsqp(h0, npts, alpha, Nit, ftol, use_topk=False):
    global _obj_n, _obj_alpha
    _obj_n = npts
    _obj_alpha = alpha
    bnds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    func = _topk_obj if use_topk else _smooth_obj
    try:
        r = minimize(func, h0, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': Nit, 'ftol': ftol})
        return _project(r.x)
    except Exception:
        return h0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _coordinatedescent(h, npts, n_passes=10, step=0.02):
    global _obj_n
    _obj_n = npts
    h = h.copy()
    target = npts * 0.5
    for _ in range(n_passes):
        improved = False
        order = np.random.permutation(npts)
        for j in order:
            old = h[j]
            for cand in [old - step, old - step*0.5, old + step*0.5, old + step]:
                if cand < 0 or cand > 1:
                    continue
                h[j] = cand
                d = target - h.sum()
                if abs(d) > 1e-10:
                    h[j] = np.clip(cand + d, 0.0, 1.0)
                c = _c5(h)
                if c < _c5(h) - 1e-14:
                    improved = True
                    break
            h[j] = old
        if not improved:
            break
    return h
