def analytic_gradient(h, temp=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp(corr - m)
    w = w / w.sum()
    
    # w[k] corresponds to shift s = k - n + 1
    s_vals = np.arange(-(n-1), n)  # s from -(n-1) to n-1
    # w indexed by k = s + n - 1, so w[k] = w[s + n - 1]
    
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_s w[s+n-1] * (1-h[i-s]) for s where 0<=i-s<=n-1
        # i-s in [0, n-1] => s in [i-(n-1), i]
        s_lo = max(i - (n-1), -(n-1))
        s_hi = min(i, n-1)
        if s_lo <= s_hi:
            s_range = np.arange(s_lo, s_hi + 1)
            k_range = s_range + n - 1
            vals = 1.0 - h[i - s_range]
            grad[i] += np.dot(w[k_range], vals)
        
        # Second term: -sum_s w[s+n-1] * h[i+s] for s where 0<=i+s<=n-1
        # i+s in [0, n-1] => s in [-i, n-1-i]
        s_lo2 = max(-i, -(n-1))
        s_hi2 = min(n-1-i, n-1)
        if s_lo2 <= s_hi2:
            s_range2 = np.arange(s_lo2, s_hi2 + 1)
            k_range2 = s_range2 + n - 1
            vals2 = h[i + s_range2]
            grad[i] -= np.dot(w[k_range2], vals2)
    
    return grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_val(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / temp

def analytic_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp(corr - m)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: s in [max(i-n+1, -(n-1)), min(i, n-1)]
        s_lo = max(i - n + 1, -(n-1))
        s_hi = min(i, n-1)
        if s_lo <= s_hi:
            idx = np.arange(s_lo, s_hi + 1) + n - 1
            vals = 1.0 - h[i - (np.arange(s_lo, s_hi + 1))]
            grad[i] += np.dot(w[idx], vals)
        # Term 2: s in [max(-i, -(n-1)), min(n-1-i, n-1)]
        s_lo2 = max(-i, -(n-1))
        s_hi2 = min(n-1-i, n-1)
        if s_lo2 <= s_hi2:
            idx2 = np.arange(s_lo2, s_hi2 + 1) + n - 1
            vals2 = h[i + np.arange(s_lo2, s_hi2 + 1)]
            grad[i] -= np.dot(w[idx2], vals2)
    return grad

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_val(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / temp)

def analytic_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp(corr - m)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        s_lo = max(i - n + 1, -(n - 1))
        s_hi = min(i, n - 1)
        if s_lo <= s_hi:
            s_arr = np.arange(s_lo, s_hi + 1)
            k_arr = s_arr + n - 1
            grad[i] += np.dot(w[k_arr], 1.0 - h[i - s_arr])
        s_lo2 = max(-i, -(n - 1))
        s_hi2 = min(n - 1 - i, n - 1)
        if s_lo2 <= s_hi2:
            s_arr2 = np.arange(s_lo2, s_hi2 + 1)
            k_arr2 = s_arr2 + n - 1
            grad[i] -= np.dot(w[k_arr2], h[i + s_arr2])
    return grad

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    step = 0.01
    best = h.copy()
    best_val = compute_c5(h)

        s_hi2 = min(n - 1 - i, n - 1)
        if s_lo2 <= s_hi2:
            s_arr2 = np.arange(s_lo2, s_hi2 + 1)
            k_arr2 = s_arr2 + n - 1
            grad[i] -= np.dot(w[k_arr2], h[i + s_arr2])
    return grad

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_pg(h_start, n, temp, max_iter, deadline):
    h = h_start.copy()
    step = 0.01
    best = h.copy()
    best_val = compute_c5(h)
    stale = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = analytic_grad(h, temp)
        h_new = project_feasible(h - step * g, n)
        v = compute_c5(h_new)
        if v < best_val - 1e-12:
            best_val = v
            best = h_new.copy()
            h = h_new
            step = min(step * 1.03, 0.05)
            stale = 0
        else:
            stale += 1
            step *= 0.92
            if step < 1e-7 or stale > 300:
                break
    return best, best_val

def optimize_slsqp(h_start, n, temp, deadline):
    h0 = h_start.copy()
    def obj(x):
        return smooth_c5_val(x, temp)
    def jac(x):
        return analytic_grad(x, temp)
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': 500, 'ftol': 1e-12})
    if res.success and np.all(res.x >= -1e-9) and np.all(res.x <= 1 + 1e-9):
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with decreasing temperature
    dl1 = t0 + budget_s * 0.30
    for temp in [300, 150, 80, 40, 20, 10, 5]:
        if time.time() > dl1:
            break
        h_new, val = optimize_pg(best_h, n, temp, 10000, dl1)
        if val < best_c5:
            best_c5 = val
            best_h = h_new

    # Phase 2: SLSQP refinement
    dl2 = t0 + budget_s * 0.45
    if time.time() < dl2:
        h_new, val = optimize_slsqp(best_h, n, 20, dl2)
        if val < best_c5:
            best_c5 = val
            best_h = h_new
        h_new, val = optimize_slsqp(best_h, n, 5, dl2)
        if val < best_c5:
            best_c5 = val
            best_h = h_new

    # Phase 3: Multi-start with perturbation + local optimization
    dl3 = t0 + budget_s * 0.75
    for attempt in range(40):
        if time.time() > dl3:
            break
        scale = rng.choice([0.05, 0.08, 0.12, 0.18, 0.25])
        h_try = project_feasible(best_h + scale * rng.randn(n), n)
        dl_sub = t0 + (budget_s * 0.75 - (time.time() - t0)) * 0.08
        h_new, val = optimize_pg(h_try, n, 30, 2000, dl_sub)
        if val < best_c5:
            best_c5 = val
            best_h = h_new

    # Phase 4: Fine perturbation search
    dl4 = t0 + budget_s * 0.97
    for _ in range(1000000):
        if time.time() > dl4:
            break
        k = int(rng.randint(1, 4))
        idx = rng.choice(n, size=k, replace=False)
        h_try = best_h.copy()
        h_try[idx] += 0.03 * rng.randn(k)
        h_try = project_feasible(h_try, n)
        v = compute_c5(h_try)
        if v < best_c5:
            best_c5 = v
            best_h = h_try

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(best_c5), len(best_h))
