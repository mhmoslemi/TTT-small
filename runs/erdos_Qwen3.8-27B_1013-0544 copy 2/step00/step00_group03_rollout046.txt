```python
def smooth_max(corr, alpha=50):
    """Smooth approximation of max using scaled log-sum-exp."""
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return log_sum / alpha

def smooth_max_grad(corr, alpha=50):
    """Gradient of smooth max with respect to corr values."""
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    weights = np.exp(shifted - max_shifted)
    weights = weights / np.sum(weights)
    return weights
```

```python
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def smooth_max_weights(corr, alpha=100):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))

    n = len(h)
    if k >= 0:
        g = np.zeros(n)
        g[k:] += (1 - h[:n-k])
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k_idx in active:
        k = k_idx - (n - 1)
        grad += w[k_idx] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def optimize_h(h0, target_sum, deadline, rng, alpha_init=80):
    n = len(h0)
    best_h = project_onto_simplex_box(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    lr = 0.005
    alpha = alpha_init
    patience = 0
    stall_count = 0

    while time.time() < deadline:
        grad = compute_smooth_grad(best_h, alpha)
        h_new = best_h - lr * grad
        h_new = project_onto_simplex_box(h_new, target_sum)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = new_c5
            patience = 0
            if lr < 0.05:
                lr = min(lr * 1.03, 0.1)
        else:
            patience += 1
            lr *= 0.92
            if patience > 80:
                pert_size = 0.01 + 0.02 * (stall_count * 0.3)
                h_pert = best_h + rng.randn(n) * pert_size
                h_pert = project_onto_simplex_box(h_pert, target_sum)
                pert_c5 = compute_c5(h_pert)
                if pert_c5 < best_c5:
                    best_h = h_pert
                    best_c5 = pert_c5
                alpha = min(alpha * 1.3, 400.0)
                lr = 0.005
                patience = 0
                stall_count += 1
                if stall_count > 8:
                    break

    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h0 = 0.5 + 0.2 * rng.randn(n)
        h0 = project_onto_simplex_box(h0, n / 2.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_simplex_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: Optimize at initial resolution
    deadline1 = t0 + min(budget_s * 0.4, 300.0)
    best_h, best_c5 = optimize_h(h0, target_sum, deadline1, rng, alpha_init=80)
    best_n = len(best_h)

    # Phase 2: Try higher resolutions
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 30
    for n_try in [100, 140, 200]:
        if remaining < 40:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_onto_simplex_box(h_interp, n_try / 2.0)
        phase_deadline = time.time() + min(remaining * 0.35, 150.0)
        h_opt, c5_opt = optimize_h(h_interp, n_try / 2.0, phase_deadline, rng, alpha_init=60)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        remaining = budget_s - (time.time() - t0) - 20

    # Phase 3: Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = optimize_h(best_h, len(best_h) / 2.0,
                                      t0 + budget_s - 5, rng, alpha_init=150)

    best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
