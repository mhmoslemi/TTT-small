```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Use bisection on the Lagrange multiplier
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

def gradient_step(h, lr, alpha):
    """One gradient descent step with projection."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_corr = np.max(corr)
    exp_alpha = np.exp(alpha * (corr - max_corr))
    p = exp_alpha / np.sum(exp_alpha)
    fval = max_corr + np.log(np.sum(exp_alpha)) / alpha
    q = 1 - 2 * h
    grad = dx * np.correlate(p, q, mode='valid')
    # Gradient step
    h_new = h - lr * grad
    # Project
    h_new = project(h_new, n / 2)
    return h_new, fval

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    best_h = None
    best_c5 = np.inf
    
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
        # Verify constraints
        assert n / 2 * 2.0 / n == 1.0
        if np.all(init >= 0) and np.all(init <= 1) and abs(np.sum(init) - n/2) < 1e-6:
            best_h = init.copy()
            best_c5 = compute_c5(best_h)
    
    if best_h is None:
        # Create a simple initial solution
        n = 100
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n = len(best_h)
    target_sum = n / 2
    
    # Main optimization loop
    alpha = 50  # temperature parameter for log-sum-exp
    lr = 0.01  # learning rate
    
    while time.time() - start_time < budget_s * 0.95:
        # Try different alpha values
        alphas = [10, 30, 50, 100, 200, 500]
        
        for a in alphas:
            if time.time() - start_time >= budget_s * 0.95:
                break
            h = best_h.copy()
            lr = 0.005
            for step in range(2000):
                if time.time() - start_time >= budget_s * 0.95:
                    break
                h, fval = gradient_step(h, lr, a)
                # decrease learning rate
                lr = max(0.0001, 0.005 * (0.999 ** step))
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Occasionally try random restarts
        if int((time.time() - start_time) * 2) % 10 == 0:
            h = np.clip(best_h + 0.1 * rng.randn(n), 0, 1)
            h = project(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5, n
```
