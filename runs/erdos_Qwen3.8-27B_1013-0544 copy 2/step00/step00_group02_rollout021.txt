        scale = 0.02 + 0.01 * i
        h_p = h_best + rng.randn(n) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=300, ftol=1e-10)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 5: Increase resolution
    for n_new in [90, 180]:
        if time_left() < 250:
            break
        h_interp = interp_h(h_best, n_new)
        h_cur = h_interp
        c5_cur = compute_c5(h_cur)
        # Optimize at new resolution
        h_new, c5_new = optimize_t_var(h_cur, max_iter=400, ftol=1e-10)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new
            h_cur = h_new
        for temp in [0.01, 0.005, 0.002]:
            if time_left() < 100:
                break
            h_new, c5_new = optimize_smooth(h_cur, temp, max_iter=300, ftol=1e-12)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new
                h_cur = h_new
        # Final t-var pass at this resolution
        if time_left() > 150:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=500, ftol=1e-11)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
