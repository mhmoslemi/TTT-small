def project_onto_feasible(h, n_points):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n_points / 2.0
    # Water filling: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h_proj
