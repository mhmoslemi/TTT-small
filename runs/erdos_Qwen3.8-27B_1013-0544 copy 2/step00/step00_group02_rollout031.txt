```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_simplex_box(v, target_sum):
    """Project v onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(v)
    # Binary search on Lagrange multiplier
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    return h

def smooth_max(x, t):
    """Smooth approximation to max using log-sum-exp"""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(t * (x - x_max)))) / t

def objective_smooth(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return smooth_max(corr * dx, t)
```

out[m] = sum_i a[i] * v[i + m - (n-1)] for m = 0, 1, ..., 2n-2

So out[m] = sum_i h[i] * (1-h[i + m - (n-1)])

Let's define s = m - (n-1), so s ranges from -(n-1) to (n-1), and:
out[m] = sum_i h[i] * (1-h[i+s]) where i+s must be in [0, n-1]

The derivative with respect to h[i]:
d(out[m])/dh[i] = (1-h[i+s]) [if i+s is valid] + sum_j h[i] * (-1) * d(h[j+s])/dh[i]

The second term: d(h[j+s])/dh[i] = 1 if j+s = i, i.e., j = i-s.
So the second term is -h[i-s] if i-s is valid.

Total: d(out[m])/dh[i] = (1-h[i+s]) * [0≤i+s<n] - h[i-s] * [0≤i-s<n]

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_box_sum(v, target_sum):
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_val(x, t):
    x_max = x.max()
    return x_max + np.log(np.sum(np.exp(t * (x - x_max)))) / t

def objective_soft(h, t, n, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    sm = soft_max_val(corr, t)
    pen = penalty * (h.sum() - n / 2.0) ** 2
    return sm + pen

def get_subgradient(h, n, eps=1e-4):
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n
    c = corr * dx
    c_max = c.max()
    active = np.where(c > c_max - eps)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            grad[-s:] += (1.0 - h[:n + s])
            grad[:n + s] -= h[-s:]
    grad /= len(active)
    return grad, c_max

def subgradient_optimize(h, n, n_iter=20000, time_limit=100):
    t0 = time.time()
    h_curr = h.copy()
    h_prev = h.copy()
    best = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    alpha0 = 1.5
    no_improve = 0

    for t in range(1, n_iter + 1):
        if time.time() - t0 > time_limit:
            break
        beta = (t - 1) / (t + 2.0)
        h_lk = h_curr + beta * (h_curr - h_prev)
        h_lk = np.clip(h_lk, 0.0, 1.0)
        grad, c_max = get_subgradient(h_lk, n)
        alpha = alpha0 / np.sqrt(t)
        h_new = h_lk - alpha * grad
        h_new = project_box_sum(h_new, n / 2.0)
        h_prev = h_curr
        h_curr = h_new
        c5_val = compute_c5(h_curr)
        if c5_val < best_c5 - 1e-9:
            best_c5 = c5_val
            best = h_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            h_prev = h_curr.copy()
            no_improve = 0
    return best, best_c5

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    init_vals = globals().get('initial_h_values', None)
    if init_vals is not None:
        init = np.asarray(init_vals, dtype=float).copy()
        best_h = init
        best_c5 = compute_c5(init)

    n_targets = [45, 60, 75, 90, 120, 150, 200]

    for n in n_targets:
        if time.time() - t0 > budget_s - 120:
            break

        if best_h is None:
            h0 = np.clip(0.5 + 0.2 * rng.randn(n), 0, 1)
            h0 = project_box_sum(h0, n / 2.0)
        else:
            h0 = interpolate_to(best_h, n)

        bounds = [(0.0, 1.0)] * n

        for t_val in [15, 40, 100, 200]:
            if time.time() - t0 > budget_s - 90:
                break
            try:
                res = minimize(objective_soft, h0, args=(t_val, n, 50.0),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-12})
                h0 = project_box_sum(res.x, n / 2.0)
                c = compute_c5(h0)
                if c < best_c5:
                    best_c5 = c
                    best_h = h0.copy()
            except Exception:
                pass

        # Subgradient polishing
        remaining = budget_s - 60 - (time.time() - t0)
        if remaining > 15:
            tl = min(remaining * 0.6, 150)
            h_sg, c5_sg = subgradient_optimize(h0, n, n_iter=50000, time_limit=tl)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

        # Random restart if time allows
        remaining = budget_s - 60 - (time.time() - t0)
        if remaining > 30:
            h_rand = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
            h_rand = project_box_sum(h_rand, n / 2.0)
            try:
                res = minimize(objective_soft, h_rand, args=(50, n, 50.0),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-10})
                h_rand = project_box_sum(res.x, n / 2.0)
                c = compute_c5(h_rand)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_rand.copy()
            except Exception:
                pass

    # Final SLSQP polish on best solution
    remaining = budget_s - 10 - (time.time() - t0)
    if remaining > 10 and best_h is not None:
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        try:
            res = minimize(objective_soft, best_h, args=(500, n_best, 100.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_final = project_box_sum(res.x, n_best / 2.0)
            c_final = compute_c5(h_final)
            if c_final < best_c5:
                best_c5 = c_final
                best_h = h_final
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
```
