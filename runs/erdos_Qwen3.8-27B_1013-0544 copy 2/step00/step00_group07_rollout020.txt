from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = np.ones(80) * 0.5
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def objective(x):
        if np.any(x < -0.1) or np.any(x > 1.1):
            return 1.0
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode='full') * dx
        return np.max(corr)
    
    equality_cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                     'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    res = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                  constraints=equality_cons, options={'maxiter': 300, 'ftol': 1e-9})
    if res.x is not None:

        # Fix sum
        diff = target_sum - np.sum(h_try)
        h_try = h_try + diff / n

    # Phase 1: SLSQP from initial
    h_s = slsqp_optimize(best_h, n, dx, target_sum, maxiter=300)
    c5_s = compute_c5(h_s, dx)
    if c5_s < best_c5:
        best_h, best_c5 = h_s.copy(), c5_s
    
    if time.time() - t0 > budget_s * 0.3:
        return best_h, best_c5, n
    
    # Phase 2: Binary projection + local search
    h_bin = binary_project(best_h, n, target_sum)
    h_ls, c5_ls = local_search_binary(h_bin, n, dx, rng, max_iters=300,
                                       time_limit=budget_s * 0.5,
                                       start_time=type('T', (), {'start_time': t0}),
                                       budget_s=budget_s)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls.copy(), c5_ls
    
    # Phase 3: SLSQP from binary
    h_c = slsqp_optimize(h_ls, n, dx, target_sum, maxiter=200)
    c5_c = compute_c5(h_c, dx)
    if c5_c < best_c5:
        best_h, best_c5 = h_c.copy(), c5_c
    
    if time.time() - t0 > budget_s * 0.6:
        return best_h, best_c5, n
    
    # Phase 4: Multiple random restarts
    deadline = t0 + budget_s * 0.92
    phi = 0.618033988749895  # golden ratio conjugate
    
    while time.time() < deadline:
        # Choose starting point
        r = rng.random()
        if r < 0.4:
            # Random binary
            h_start = np.zeros(n)
            h_start[rng.choice(n, size=int(target_sum), replace=False)] = 1.0
        elif r < 0.7:
            # Golden ratio distribution
            positions = sorted(set(int(j * n * phi) % n for j in range(int(target_sum))))
            while len(positions) < int(target_sum):
                p = rng.randint(n)
                if p not in positions:
                    positions.append(p)
            h_start = np.zeros(n)
            h_start[positions[:int(target_sum)]] = 1.0
        else:
            # Perturbed best
            h_start = best_h.copy()
            h_start += rng.normal(0, 0.1, n)
            h_start = np.clip(h_start, 0, 1)
            h_start += (target_sum - np.sum(h_start)) / n
            h_start = np.clip(h_start, 0, 1)
        
        # Local search
        h_ls2, c5_ls2 = local_search_binary(h_start, n, dx, rng, max_iters=150,
                                             time_limit=deadline,
                                             start_time=type('T', (), {'start_time': t0}),
                                             budget_s=budget_s)
        if c5_ls2 < best_c5:
            best_h, best_c5 = h_ls2.copy(), c5_ls2
        
        # SLSQP from promising local search result
        if c5_ls2 <= best_c5 + 0.01 and time.time() < deadline and time.time() < t0 + budget_s * 0.85:
            h_c2 = slsqp_optimize(h_ls2, n, dx, target_sum, maxiter=150)
            c5_c2 = compute_c5(h_c2, dx)
            if c5_c2 < best_c5:
                best_h, best_c5 = h_c2.copy(), c5_c2
    
    return best_h, best_c5, n
