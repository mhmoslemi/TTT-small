import numpy as np

def project_feasible(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_eval(h, dx):
    return float(np.max(corr_full(h)) * dx)

def smooth_grad(h, dx, beta):
    n = len(h)
    c = corr_full(h)
    mc = np.max(c)
    e = np.exp(beta * (c - mc))
    p = e / e.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        pk = p[ki]
        if pk < 1e-14:
            continue
        k = ki - (n - 1)
        ns = max(0, -k)
        ne = min(n - 1, n - 1 - k)
        nn = np.arange(ns, ne + 1)
        pp = nn + k
        g[nn] -= h[pp] * pk
        g[pp] += (1.0 - h[nn]) * pk
    return g * dx
