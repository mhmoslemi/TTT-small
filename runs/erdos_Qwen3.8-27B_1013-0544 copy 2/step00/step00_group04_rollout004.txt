    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

    h = h.copy()
    curr_c5 = evaluate_c5(h, n_target)
    for _ in range(max_passes):
