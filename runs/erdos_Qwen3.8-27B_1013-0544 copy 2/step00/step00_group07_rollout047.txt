def compute_gradient(h, dx, alpha):
    n = len(h)
    # Compute smooth max weights
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    m = np.max(overlaps)
    if m < 1e-12:
        return np.zeros(n)
    exps = np.exp(alpha * (overlaps - m))
    weights = exps / np.sum(exps)
    
    # For each shift s (corresponding to correlation index k = s + n - 1):
    # d(overlap_k)/d(h[j]) = dx * [(1-h[j+s]) - h[j-s]]
    # where terms are included only when indices are valid
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        # For this shift s, the valid range of i is:
        # 0 ≤ i < n and 0 ≤ i+s < n
        # So i ranges from max(0, -s) to min(n-1, n-1-s)
        i_start = max(0, -s)
        i_end = min(n, n - s)  # exclusive
        if i_start >= i_end:
            continue
        # d(overlap)/d(h[i]) gets contribution: dx * (1 - h[i+s])
        for i in range(i_start, i_end):

            # d(overlap)/d(h[i+s]) gets contribution: -dx * h[i]
            grad[i + s] += weights[k] * dx * (-h[i])
    
    return grad

            # Reduce sum h[i]*(1-h[i]) by pushing h toward 0 or 1
            idx = int(np.argmax(h * (1.0 - h)))
            for delta in [0.1, -0.1, 0.05, -0.05]:
                h_try = h.copy()
                h_try[idx] += delta
                h_try = project_onto_constraint(h_try, target_sum)
                c5t = compute_c5(h_try, dx)
                if c5t < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5t
                    break
        else:
            t = abs(s)
            cnt = n - t
            if cnt <= 0:
                continue
            # Pick the index with largest contribution
            if s > 0:
                contrib = h[:cnt] * (1.0 - h[t:])
            else:
                contrib = h[t:] * (1.0 - h[:cnt])
            imax = int(np.argmax(contrib))
            if s > 0:
                i, j = imax, imax + t
            else:
                i, j = imax + t, imax
            for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]:
                h_try = h.copy()
                if s > 0:
                    h_try[i] += delta
                    h_try[j] -= delta
                else:
                    h_try[j] += delta
                    h_try[i] -= delta
                h_try = project_onto_constraint(h_try, target_sum)
                c5t = compute_c5(h_try, dx)
                if c5t < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5t
                    break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = len(parent)
    except Exception:
        parent = None
        n_parent = 0

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_list = sorted(set([40, 60, 80, 100, 120]))
    if parent is not None:
        n_list.insert(0, n_parent)
        n_list = sorted(set(n_list))

    n_trials = len(n_list)
    time_alloc = (budget_s * 0.88) / n_trials

    for n in n_list:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.92:
            break
        rem = budget_s * 0.92 - elapsed
        t_alloc = min(time_alloc, rem)
        if t_alloc < 3.0:
            continue

        dx = 2.0 / n
        target_sum = n / 2.0

        # Initial guess
        if parent is not None and n_parent == n:
            h0 = parent.copy()
        elif parent is not None:
            x_old = np.arange(n_parent) * (2.0 / n_parent)
            x_new = np.arange(n) * (2.0 / n)
            h0 = np.interp(x_new, x_old, parent)
            h0 = np.clip(h0, 0.0, 1.0)
        else:
            h0 = np.full(n, 0.5)

        h0 = project_onto_constraint(h0, target_sum)

        # Main optimization: projected gradient
        pg_time = t_alloc * 0.55
        h_pg, c5_pg = optimize_pg(h0, dx, target_sum, pg_time)

        # Polish
        pol_time = min(t_alloc * 0.35, 30.0)
        h_pol, c5_pol = polish_local(h_pg, dx, target_sum, pol_time)

        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n

        # Random restart with remaining time
        rem = t_alloc * 0.10
        t_r = time.time()
        n_restarts = max(1, int(rem / max(t_alloc * 0.02, 2.0)))
        for r in range(n_restarts):
            if time.time() - t_r > rem:
                break
            noise_scale = 0.15 + 0.1 * r
            h_r = project_onto_constraint(h0 + noise_scale * np.random.randn(n), target_sum)
            ht = min(rem - (time.time() - t_r), t_alloc * 0.04)
            if ht < 1.0:
                break
            h_r2, c5_r = optimize_pg(h_r, dx, target_sum, ht)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r2.copy()
                best_n = n

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / 80)
        best_n = 80

    return best_h, float(best_c5), int(best_n)
