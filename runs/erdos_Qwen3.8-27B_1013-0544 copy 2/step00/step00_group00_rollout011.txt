from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_single_lag(h, k, n, dx):
    """Gradient of c[k]*dx with respect to h"""
    d = k - n + 1
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + d < n:
            grad[j] += (1.0 - h[j + d])
        if 0 <= j - d < n:
            grad[j] -= h[j - d]
    return grad * dx

def make_obj_and_grad(dx, penalty):
    n = int(round(2.0 / dx))
    def obj(h):
        c5 = eval_c5(h, dx)
        viol = np.sum(h) * dx - 1.0
        return c5 + penalty * viol * viol
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        g = compute_grad_single_lag(h, k_max, n, dx)
        viol = np.sum(h) * dx - 1.0
        g += 2.0 * penalty * dx * viol
        return g
    return obj, grad

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while staying in [0,1]"""
    for _ in range(100):
        deficit = target_sum - np.sum(h)
        if abs(deficit) < 1e-12:
            break
        # Distribute to elements not at bounds
        can_increase = h < 1.0 - 1e-10
        can_decrease = h > 1e-10
        if deficit > 0:
            mask = can_increase
        else:
            mask = can_decrease
        if not np.any(mask):
            break
        h[mask] += deficit / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

            hp, cp = optimize_phase(h_pert, best_n, dx_cur, ts_cur, bounds_cur, tl)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Phase 4: Fine-tune with high penalty
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        dx_cur = 2.0 / best_n
        ts_cur = float(best_n) / 2.0
        bounds_cur = [(0.0, 1.0)] * best_n
        h_final = run_lbgfgsb(best_h, dx_cur, best_n, 1e8, bounds_cur, 1000, remaining - 2)
        h_final = project_constraint(h_final, ts_cur)
        c5_final = eval_c5(h_final, dx_cur)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ensure final constraint satisfaction
    dx_final = 2.0 / best_n
    best_h = project_constraint(best_h, float(best_n) / 2.0)
    best_c5 = eval_c5(best_h, dx_final)

    return best_h, best_c5, best_n
