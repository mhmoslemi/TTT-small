import numpy as np
from scipy.optimize import minimize
import time

def compute_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-10
    active = np.where(np.abs(corr - max_val) < tol)[0]
    grad = np.zeros(n)
    for k in active:
        m = k - n + 1
        lo1 = max(0, m)
        hi1 = min(n, m + n)
        if lo1 < hi1:
            grad[lo1:hi1] += 1.0 - h[lo1-m:hi1-m]
        lo2 = max(0, -m)
        hi2 = min(n, n - m)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+m:hi2+m]
    grad /= len(active)
    return max_val, grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_descent(h, target_sum, max_iter=5000, t0=0.1, decay=0.999, 
                        time_limit=None, start_time=None):
    h = h.copy()
    best_h = h.copy()
    best_obj = compute_obj(h)
    
    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        _, grad = compute_subgradient(h)
        # Polyak-like step or decreasing step
        t = t0 * (decay ** it)
        h_new = h - t * grad
        h_new = project_feasible(h_new, target_sum)
        
        obj = compute_obj(h_new)
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_obj

def nelder_mead_refine(h, target_sum, time_limit=None, start_time=None):
    def obj_func(x):
        x_proj = project_feasible(x, target_sum)
        return compute_obj(x_proj)
    
    result = minimize(obj_func, h, method='Nelder-Mead', 
                      options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
    h_refined = project_feasible(result.x, target_sum)
    return h_refined, compute_obj(h_refined)

    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:

    final_c5 = compute_obj(best_h)
