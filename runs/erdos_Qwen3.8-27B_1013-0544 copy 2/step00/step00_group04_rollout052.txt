import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search(h, time_limit=30.0, seed=42):
    n = len(h)
    h = h.copy().astype(float)
    best_score = evaluate_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_score_new = best_score
        best_move = None
        
        # Best improvement: try all swaps
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                s = evaluate_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if s < best_score_new - 1e-12:
                    best_score_new = s
                    best_move = (i, j)
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            best_score = best_score_new
        else:
            # Stuck: try random pair of swaps
            attempts = 0
            found = False
            while attempts < 50:
                i1 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                # Find another 1 and 0 (different from first)
                remaining_ones = [x for x in ones if x != i1]
                remaining_zeros = [x for x in zeros if x != j1]
                if not remaining_ones or not remaining_zeros:
                    break
                i2 = remaining_ones[rng.randint(len(remaining_ones))]
                j2 = remaining_zeros[rng.randint(len(remaining_zeros))]
                
                h[i1] = 0; h[j1] = 1; h[i2] = 0; h[j2] = 1
                s = evaluate_c5(h)
                if s < best_score:
                    best_score = s
                    found = True
                    break
                else:
                    h[i1] = 1; h[j1] = 0; h[i2] = 1; h[j2] = 0
                attempts += 1
            
            if not found:
                break  # Give up
    
    return h, best_score

                h_bin[zeros[i]] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = evaluate_c5(best_h)
    n_best = n_init
    
    # Optimization loop
    # Strategy: try n_init first with multiple seeds, then try other n values
    n_list = [n_init] + [n_init + 2*i for i in range(1, 8) if n_init + 2*i <= 64]
    
    for n_try in n_list:
        if time.time() - t_start > budget_s - 20:
            break
        
        n_attempts = 5 if n_try == n_init else 3
        for att in range(n_attempts):
            if time.time() - t_start > budget_s - 10:
                break
            
            # Budget per attempt
            remaining = budget_s - 10 - (time.time() - t_start)
            # Rough estimate of remaining attempts
            total_left = 0
            for nn in n_list[n_list.index(n_try):]:
                total_left += 5 if nn == n_init else 3
            tl = min(50.0, remaining / max(1, total_left))
            
            # Initialization
            if n_try == n_init and att == 0:
                h_start = best_h.copy()
            elif n_try == n_init:
                h_start = best_h.copy()
                rng = np.random.RandomState(seed + att * 37)
                for _ in range(2 + att):
                    ones = np.where(h_start == 1.0)[0]
                    zeros = np.where(h_start == 0.0)[0]
                    h_start[ones[rng.randint(len(ones))]] = 0.0
                    h_start[zeros[rng.randint(len(zeros))]] = 1.0
            else:
                h_start = make_binary(n_try, seed=seed + att * 37 + n_try * 7)
            
            h, s = local_search(h_start, time_limit=tl, seed=seed + att * 13 + n_try)
            
            if s < best_c5:
                best_h = h
                best_c5 = s
                n_best = n_try
    
    return best_h, best_c5, n_best
