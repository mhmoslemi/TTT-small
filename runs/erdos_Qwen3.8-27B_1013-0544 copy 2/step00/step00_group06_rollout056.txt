    no_improve = 0

    for it in range(50000):
        if time.time() - start_time > budget_s - 80:
            break
        if it % 5000 == 4999:
            step *= 0.75
        corr = compute_corr(h)
        w = get_weights(corr, beta_sg)
        g = weighted_grad(h, w)
        v = mom * v - step * g
        h = project_to_feasible(h + v, target_sum)
        c5 = c5_value(h)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            h_best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                h = h_best + rng.normal(0, 0.05, n)
                h = project_to_feasible(h, target_sum)
                v = np.zeros(n)
                no_improve = 0
                step = 0.03

    # Phase 3: Random restarts with local optimization
    for r in range(300):
        if time.time() - start_time > budget_s - 15:
            break
        scale = 0.05 + 0.05 * (r / 300.0)
        h = h_best + rng.normal(0, scale, n)
        h = project_to_feasible(h, target_sum)
        v = np.zeros(n)
        for _ in range(80):
            corr = compute_corr(h)
            w = get_weights(corr, 40)
            g = weighted_grad(h, w)
            v = 0.9 * v - 0.015 * g
            h = project_to_feasible(h + v, target_sum)
        c5 = c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    # Phase 4: Final polish with tight L-BFGS-B
    if time.time() - start_time < budget_s - 30:
        mu = 500.0
        beta = 200

        def obj_final(z):
            h = np.clip(z, 0, 1)
            corr = compute_corr(h)
            m = np.max(corr)
            val = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
            return val + mu * (np.sum(h) - target_sum) ** 2

        def grad_final(z):
            h = np.clip(z, 0, 1)
            corr = compute_corr(h)
            w = get_weights(corr, beta)
            g = weighted_grad(h, w)
            g += 2.0 * mu * (np.sum(h) - target_sum)
            return g

        try:
            res = minimize(obj_final, h_best, jac=grad_final, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 1000, 'ftol': 1e-15})
            h_c = project_to_feasible(res.x, target_sum)
            c5_c = c5_value(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                h_best = h_c.copy()
        except Exception:
            pass

    return h_best, best_c5, n
