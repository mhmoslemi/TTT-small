    """Generate m-sequence of length 2^m - 1 using LFSR."""

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:

            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)

    val = (m + np.log(sw) / beta) * dx

        jp = j + s
        ms = (js >= 0) & (js < n)

        if mp.any():

    """Nesterov with adaptive smoothmax beta."""
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):

The first term is like a weighted sum of (1-h) shifted by -s: it's a convolution of w with (1-h)
The second term is like a weighted sum of h shifted by s: it's a convolution of w with h

    # For each j, grad[j] = dx * sum_k w[k] * d(corr[k])/d(h[j])

    #   = (1-h[j+s]) if 0<=j+s<n  [h[j] is the "first" factor]
    #   + (-h[j-s]) if 0<=j-s<n  [h[j] is the "second" factor, i.e., h[i+s] where i=j-s]

    # and term2[j] = sum_s w[s+n-1] * h[j-s] for valid j-s
    
    # term1 is like a cross-correlation
    # Let A = 1-h, B = w (indexed by shift)
    # term1[j] = sum_s B[s] * A[j+s] where s = k-(n-1), so B has length 2n-1
    # This is: sum over s of B[s] * A[j+s] = (B_reversed * A)[j + offset]
    
    # Actually, the easiest is to use the fact that:
    # term1[j] = sum_k w[k] * (1-h[j + k - n + 1]) for valid indices
    # Let's define an extended array and use convolution
    
    # Simpler approach: just loop but vectorize the inner operations
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # h[j] appears as first factor at position j, paired with (1-h)[j+s]
        # valid: 0 <= j < n and 0 <= j+s < n
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            # h[j] for j in [lo1, hi1), paired with (1-h)[j+s]
            grad[lo1:hi1] += wk * (1.0 - h[lo1+s:hi1+s])

        # paired with (1-h)[i] = (1-h)[j-s]
        # valid: 0 <= j < n and 0 <= j-s < n
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] += wk * (-h[lo2-s:hi2-s])
    
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] += wk * (-h[lo2 - s:hi2 - s])
    grad *= dx
    return val, grad


def nesterov_adaptive(h, max_iters, lr0, decay, beta0, beta_max, top_k):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    beta = beta0
    stall = 0
    patience = 300

    for t in range(max_iters):
        step = max(1, max_iters // 8)
        if t % step == 0 and t > 0:
            beta = min(beta * 1.8, beta_max)
        lr = lr0 * (decay ** t)

        if t == 0 or stall >= patience:
            y = x.copy()
            x_prev = x.copy()
            stall = 0
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)

        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx if (dx := 2.0 / n) else grad

        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        x_prev = x
        x = x_new

    return best_x, best_v


def local_lag_refine(h, n_iters, lr, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    cur_lr = lr

    for t in range(n_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - x[js[ms]])
            if mp.any():
                grad[mp] += (-x[jp[mp]])
        grad = grad / top_k * dx
        x = project_feasible(x - cur_lr * grad, target)
        v = compute_c5(x)
        if v < best_v:
            best_v = v
            best_x = x.copy()
            cur_lr = min(cur_lr * 1.05, lr * 2)
        else:
            cur_lr *= 0.9

    return best_x, best_v


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def msequence_init(n, m_bits=7):
    L = (1 << m_bits) - 1
    polys = [3, 5, 6, 7, 4, 2, 5, 6]
    rng = np.random.RandomState(42)
    best_h = None
    best_c5 = float('inf')
    for poly in polys:
        state = 1
        seq = np.zeros(L)
        for i in range(L):
            seq[i] = state & 1
            fb = 0
            for b in range(m_bits):
                fb ^= (state >> b) & 1
            state = (state >> 1) | (fb << (m_bits - 1))
        if seq.sum() == n // 2:
            h = seq[:n].copy()
        elif seq.sum() > n // 2:
            h = seq[:n].copy()
            h[h == 1][:int(seq.sum() - n // 2)] = 0
        else:
            h = np.zeros(n)
            h[:] = seq
            extra = n // 2 - int(h.sum())
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, min(extra, len(zeros)), replace=False)] = 1
        h = project_feasible(h, n / 2.0)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    if best_h is None:
        best_h = random_binary_init(n, 123)
    return best_h
