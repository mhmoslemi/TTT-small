def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = m.sum()
        if cnt == 0:

def smooth_max_obj(h, n, beta, lam, ts):
    c = np.correlate(h, 1.0 - h, mode="full")

    s = c * dx
    m = np.max(s)

        elif strat == 3:
            freq = np.random.randint(1, 6)
            phase = np.random.uniform(0, 2 * np.pi)
            x = np.arange(n) * 2.0 / n
            h += 0.012 * np.sin(2 * np.pi * freq * x + phase)
        else:
            # Anti-worst construction: reduce overlap at worst lag
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top3 = np.argsort(corr)[-3:]
            for kk in top3:
                k_shift = int(kk) - (n - 1)
                if k_shift > 0:
                    for i in np.random.choice(n - k_shift, min(6, n - k_shift), replace=False):
                        if best_h[i] > 0.5 and best_h[i + k_shift] < 0.5:
                            h[i] -= 0.01
                            h[i + k_shift] += 0.01
                elif k_shift < 0:
                    ks = -k_shift
                    for i in np.random.choice(n - ks, min(6, n - ks), replace=False):
                        if best_h[i + ks] > 0.5 and best_h[i] < 0.5:
                            h[i + ks] -= 0.01
                            h[i] += 0.01

        h = project_box_sum(h, ts)
        for bt in [250, 500]:
            if remaining() < 25:
                break
            try:
                def o_r(h):
                    return smooth_max_obj(h, n, bt, 800.0, ts)[0]
                def g_r(h):
                    _, W = smooth_max_obj(h, n, bt, 800.0, ts)
                    return gradient_smooth_max(h, n, W, 800.0, ts)
                r = minimize(o_r, h, jac=g_r, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    h = h_new.copy()
            except Exception:
                pass

    # Phase 5: High-precision final polish
    if remaining() > 20:
        for bf in [1500, 3000, 8000]:
            if remaining() < 12:
                break
            try:
                def o_f(h):
                    return smooth_max_obj(h, n, bf, 2000.0, ts)[0]
                def g_f(h):
                    _, W = smooth_max_obj(h, n, bf, 2000.0, ts)
                    return gradient_smooth_max(h, n, W, 2000.0, ts)
                r = minimize(o_f, best_h, jac=g_f, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-13})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
