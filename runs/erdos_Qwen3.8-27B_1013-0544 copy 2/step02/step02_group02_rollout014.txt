import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    if alpha < 10:
        return float(m)
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    # gradient of sum w_k * c_k w.r.t. h
    # c[k_idx] = sum_i h[i] * (1-h[i+k]) * dx where k = k_idx - (n-1)
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - (n - 1)
        # d c[k] / d h[j] = dx * [(1-h[j+k]) * 1(j+k<n) - h[j-k] * 1(j-k>=0)]

            jj = -k
            grad[jj:] += w[k_idx] * dx * (1 - h[:n-jj])
            grad[:n-jj] += w[k_idx] * dx * (-h[jj:])
    return grad

def slsqp(h, n, alpha, maxiter, ftol):

        return softmax_obj(x, n, alpha)
    def jac(x):
        return softmax_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    if alpha < 10:
        return float(m)
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_jac(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for ki in range(len(c)):
        if w[ki] < 1e-12:
            continue
        k = ki - (n - 1)
        if k >= 0:
            grad[:n - k] += w[ki] * dx * (1.0 - x[k:])
            grad[k:] += w[ki] * dx * (-x[:n - k])
        else:
            jj = -k
            grad[jj:] += w[ki] * dx * (1.0 - x[:n - jj])
            grad[:n - jj] += w[ki] * dx * (-x[jj:])
    return grad

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(softmax_obj, h, args=(n, alpha), jac=lambda x: softmax_jac(x, n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

            if ni.min() < 0 or ni.max() > 1 or nj.min() < 0 or nj.max() > 1:
                continue
            h[i:i+bl] = ni
            h[j:j+bl] = nj
            nc = compute_c5(h, n)
            delta = nc - cur_c5
            if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(E, 1e-12)):
                cur_c5 = nc
                if nc < best_c5:
                    best_h, best_c5 = h.copy(), nc
            else:
                h[i:i+bl] = bi
                h[j:j+bl] = bj
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-D * 0.3, D * 0.3)
            hi, hj = h[i] + d, h[j] - d
            if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
                continue
            h[i], h[j] = hi, hj
            nc = compute_c5(h, n)
            delta = nc - cur_c5
            if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(E, 1e-12)):
                cur_c5 = nc
                if nc < best_c5:
                    best_h, best_c5 = h.copy(), nc
            else:
                h[i], h[j] = hi - d, hj + d
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

def de_objective(x, n):
    h = project_feasible(x, n)
    return compute_c5(h, n)

def de_search(h_init, n, t0, budget_s, rng, popsize, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(de_objective, bounds, args=(n,),
                                     seed=rng.randint(2**31), popsize=popsize,
                                     maxiter=maxiter, tol=1e-10,
                                     mutate=(0.5, 1.0), recomb=0.8,
                                     polish=False, update_interval=1)
        h = project_feasible(res.x, n)
        return h
    except Exception:
        return h_init

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation
    for alpha in (500, 5000, 50000, 200000):
        if rem() < 20:
            break
        ht = slsqp_polish(best_h, n, alpha, 4000, 1e-17)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 2: Subgradient descent
    if rem() > 120:
        ht, c = subgrad_descent(best_h, n, 8000, t0, budget_s, lr_init=0.03)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 3: SLSQP after subgrad
    for alpha in (200000, 1000000):
        if rem() < 20:
            break
        ht = slsqp_polish(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 4: Differential evolution (global search)
    if rem() > 250:
        de_result = de_search(best_h, n, t0, budget_s, rng, popsize=30, maxiter=80)
        de_c5 = compute_c5(de_result, n)
        if de_c5 < best_c5:
            best_h, best_c5 = de_result, de_c5
        # Polish DE result
        for alpha in (50000, 500000):
            if rem() < 20:
                break
            ht = slsqp_polish(best_h, n, alpha, 3000, 1e-16)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = ht, c

    # Phase 5: Simulated annealing
    if rem() > 150:
        best_h, best_c5 = sa_optimize(best_h, n, t0, budget_s, rng,
                                      400000, 0.04, 1e-6, 0.25, 0.008)

    # Phase 6: SLSQP polish
    for alpha in (500000, 2000000, 5000000):
        if rem() < 15:
            break
        ht = slsqp_polish(best_h, n, alpha, 6000, 1e-19)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 7: Subgradient again
    if rem() > 80:
        ht, c = subgrad_descent(best_h, n, 5000, t0, budget_s, lr_init=0.02)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 8: Pairwise sweep
    if rem() > 40:
        best_h, best_c5 = pairwise_sweep(best_h, n, t0, budget_s,
                                         steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001])

    # Phase 9: Final SLSQP
    if rem() > 10:
        ht = slsqp_polish(best_h, n, 10000000, 4000, 1e-20)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
