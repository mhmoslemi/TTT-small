    dx = 2.0 / n

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_grad(h, beta):
    """Returns (smooth_max_value, gradient) of smooth max of C5."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr_vals = corr_raw * dx
    
    m = np.max(corr_vals)
    x = np.clip(beta * (corr_vals - m), -500, 500)
    e = np.exp(x)
    s = e.sum()
    w = e / s
    sm = m + np.log(s / len(corr_vals)) / beta
    
    grad = np.zeros(n)
    # Process top lags only (_weights are concentrated)
    top_n = min(100, len(corr_raw))
    top_idx = np.argsort(w)[::-1][:top_n]
    
    for k in top_idx:
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        s_val = k - (n - 1)
        if s_val >= 0:
            grad[:n-s_val] += wk * (1.0 - h[s_val:])
            grad[s_val:] -= wk * h[:n-s_val]
        else:
            t = -s_val
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] -= wk * h[t:]
    
    return sm, grad

def pgd_optimize(h, target_sum, max_iter, beta_start, beta_end, time_check, time_limit):
    """Projected gradient descent with adaptive smooth max."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h = h.copy()
    
    for it in range(max_iter):
        if time_check() > time_limit:
            break
        frac = it / max(iter(max_iter - 1, 1), 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        sm, grad = c5_grad(h, beta)
        
        # Project direction to maintain sum constraint
        c = np.mean(grad)
        direction = -(grad - c)
        
        # Backtracking line search
        alpha = 0.05
        n_iter_ls = 0
        for _ in range(40):
            h_trial = h + alpha * direction
            h_trial = project_h(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < sm - 1e-5 * alpha * np.dot(direction, direction):
                break
            alpha *= 0.5
            n_iter_ls += 1
            if alpha < 1e-8:
                break
        
        h_trial = project_h(h + alpha * direction, target_sum)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < best_c5 - 1e-14:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        
        h = h_trial
    
    return best_h, best_c5

def coordinate_descent(h, target_sum, max_iter, time_check, time_limit):

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h = h.copy()
    
    dx = 2.0 / n

        if time_check() > time_limit:
            break
        improved = False
        # Process coordinates in random order

            # So corr[k] is a quadratic (concave? no, it's -h[j]*h[j+s] + h[j] + ...)
            # f(k) = ... + h[j]*(1-h[j+s]) + ... = ... - h[j]^2 + h[j]*(1-h[j+s]+...)

            #   So the terms involving h[j] are: h[j]*(1-h[j+s]) - h[j-s]*h[j]
            #   = h[j] - h[j]*(h[j+s]+h[j-s])

            # 
            # Wait, that's for a single lag. Summing over all lags:
            # C5 = max_k sum_i h[i](1-h[i+s_k]) 
            # For the max, it's the max of linear functions of h[j] (for fixed other coordinates).
            # So the optimal h[j] is at a boundary (0 or 1) or where two lags are tied.

            # (or where the slope changes sign, which for max of linear functions means at a kink).
            # Actually, for max of concave functions it's concave, but max of linear is convex.
            # So min of convex over [0,1] is at the boundary!
            # 
            # Wait, C5 = max_k f_k(h) where each f_k is linear in h[j].
            # max of linear functions is convex.
            # We're minimizing a convex function over [0,1], so the minimum is at the boundary.
            # 
            # So the optimal h[j] is either 0 or 1!
            # 
            # This means coordinate descent should try h[j]=0 and h[j]=1!
            
            h_j_old = h[j]
            for new_val in [0.0, 1.0]:

                h[j] = new_val

                    free = (h > 1e-12) & (h < 1.0 - 1e-12) & (np.arange(n) != j)

                    best_c5 = c5_trial

                else:

                    # Re-reproject
                    if abs(h.sum() - target_sum) > 1e-12:
                        h = project_h(h, target_sum)
        
        if not improved:
            # Try small perturbations
            break
    
    return best_h, best_c5

    s = e.sum()
    w = e / s
    sm = m + np.log(s / len(corr_vals)) / beta
    grad = np.zeros(n)
    top_idx = np.argsort(w)[::-1][:top_lags]
    for k in top_idx:
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        s_val = k - (n - 1)
        if s_val >= 0:
            grad[:n-s_val] += wk * (1.0 - h[s_val:])
            grad[s_val:] -= wk * h[:n-s_val]
        else:
            t = -s_val
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] -= wk * h[t:]
    return sm, grad

def pgd_step(h, beta, target_sum):
    sm, grad = c5_grad(h, beta)
    c = np.mean(grad)
    direction = -(grad - c)
    alpha = 0.05
    best_alpha = 1e-10
    best_trial = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(30):
        h_trial = project_h(h + alpha * direction, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-8:
            best_alpha = alpha
            best_trial = h_trial
            best_c5 = c5_trial
            break
        if c5_trial < best_c5:
            best_alpha = alpha
            best_trial = h_trial
            best_c5 = c5_trial
        alpha *= 0.5
    return best_trial, best_c5

def coordinate_descent(h, target_sum, max_sweeps, time_check, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time_check() > time_limit:
            break
        order = np.random.permutation(n)
        improved = False
        for j in order:
            old_val = h[j]
            for new_val in [0.0, 1.0]:
                h[j] = new_val
                h = project_h(h, target_sum)
                c5_trial = compute_c5(h)
                if c5_trial < best_c5 - 1e-14:
                    best_c5 = c5_trial
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] = old_val
                    h = project_h(h, target_sum)
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_grad(h, beta, top_lags=100):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr_vals = corr_raw * dx
    m = np.max(corr_vals)
    x = np.clip(beta * (corr_vals - m), -500, 500)
    e = np.exp(x)
    s = e.sum()
    w = e / s
    sm = m + np.log(s / len(corr_vals)) / beta
    grad = np.zeros(n)
    top_idx = np.argsort(w)[::-1][:top_lags]
    for k in top_idx:
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        s_val = k - (n - 1)
        if s_val >= 0:
            grad[:n - s_val] += wk * (1.0 - h[s_val:])
            grad[s_val:] -= wk * h[:n - s_val]
        else:
            t = -s_val
            grad[t:] += wk * (1.0 - h[:n - t])
            grad[:n - t] -= wk * h[t:]
    return sm, grad

def pgd_optimize(h, target_sum, max_iter, beta_start, beta_end, time_limit, start_time):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        frac = it / max(max_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        sm, grad = c5_grad(h, beta)
        c = np.mean(grad)
        direction = -(grad - c)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-14:
            break
        alpha = min(1.0 / dnorm, 0.05)
        improved = False
        for _ in range(35):
            h_trial = project_h(h + alpha * direction, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                best_h = h_trial.copy()
                best_c5 = c5_trial
            if c5_trial < sm - 1e-6 * alpha * dnorm:
                h = h_trial
                improved = True
                break
            alpha *= 0.5
        if not improved:
            h = project_h(h + alpha * direction, target_sum)
            c5_chk = compute_c5(h)
            if c5_chk < best_c5:
                best_h = h.copy()
                best_c5 = c5_chk
    return best_h, best_c5
