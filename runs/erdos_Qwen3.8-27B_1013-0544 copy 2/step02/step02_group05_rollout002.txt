    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on lambda."""
    lo, hi = -max(0.0, h.max() - target_sum/len(h)), max(0.0, h.min() - target_sum/len(h) + 1)
    # Actually, let's use a wider range
    lo, hi = -2.0, 2.0
