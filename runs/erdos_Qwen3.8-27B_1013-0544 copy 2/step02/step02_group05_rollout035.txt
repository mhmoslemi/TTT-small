def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[:n][::-1] * dx

def compute_c5(h):
    return float(np.max(compute_all_overlaps(h)))

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_overlap_s(h, s, n, dx):

        return dx * (1.0 - 2.0 * h)
    g = np.zeros(n)
    g[:n - s] = (1.0 - h[s:])
    g[s:] -= h[:n - s]
    return dx * g

    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

            ys = h - prev_h
            gg = grad - prev_grad
            num = np.dot(ys, ys)
            den = np.dot(ys, gg)
            if den > 1e-14:
                alpha_bb = num / den
                alpha = np.clip(alpha_bb, 0.005, 0.5)

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < curr_c5 - 1e-12:
            prev_h = h.copy()
            prev_grad = grad.copy()
            h = h_new
            curr_c5 = c5_new
            stall = 0
        else:
            alpha *= 0.7
            stall += 1
            if stall > 30:
                break
    return h, curr_c5

def run_slsqp_polish(h, n, target_sum, beta, maxiter, time_limit):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(hv):
        return smooth_max(compute_all_overlaps(hv), beta)
    def con_f(x):
        return np.sum(x) - target_sum
    constraints = [{'type': 'eq', 'fun': con_f, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_feasible(res.x, target_sum)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    n_list = [70, 60, 80, 90]

    for ni, n in enumerate(n_list):
        t_remaining = budget_s - (time.time() - start_time)
        if t_remaining < 60:
            break
        time_limit = start_time + budget_s - 20 - (len(n_list) - 1 - ni) * 40

        target_sum = n / 2.0
        dx = 2.0 / n

        if len(parent) == n:
            h0 = parent.copy()
        else:
            xo = np.linspace(0, 2, len(parent), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(xn, xo, parent, period=2.0)
        h = project_feasible(h0, target_sum)
        curr_c5 = compute_c5(h)

        # Phase 1: Projected gradient
        h, curr_c5 = run_projected_gradient(h, n, target_sum, dx, 250, time_limit)

        # Phase 2: SLSQP with increasing smoothness
        for beta in [60, 120, 250]:
            if time.time() > time_limit:
                break
            h, curr_c5 = run_slsqp_polish(h, n, target_sum, beta, 200, time_limit)

        # Phase 3: Stochastic restarts
        base = h.copy()
        no_improve = 0
        for restart in range(25):
            if time.time() > time_limit:
                break
            if no_improve > 12:
                break
            scale = 0.015 + 0.04 * np.exp(-restart * 0.15)
            h_r = base + np.random.randn(n) * scale
            h_r = project_feasible(h_r, target_sum)
            h_new, c5_new = run_slsqp_polish(h_r, n, target_sum, 150, 120, time_limit)
            if c5_new < curr_c5:
                h = h_new
                curr_c5 = c5_new
                base = h.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Phase 4: Final high-beta polish
        if time.time() < time_limit:
            h, curr_c5 = run_slsqp_polish(h, n, target_sum, 400, 300, time_limit)

        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
            best_n = n

    # Ultimate polish on best solution
    if best_h is not None and time.time() < start_time + budget_s - 5:
        n = best_n
        ts = n / 2.0
        best_h, best_c5 = run_slsqp_polish(best_h, n, ts, 500, 500, start_time + budget_s - 2)

    if best_h is None:
        best_h = parent
        best_c5 = compute_c5(parent)
        best_n = len(parent)

    return best_h, best_c5, best_n
