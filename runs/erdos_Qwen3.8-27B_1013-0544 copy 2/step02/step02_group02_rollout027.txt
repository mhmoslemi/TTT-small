        ht = slsqp(best_h, n, alpha, 2000, 1e-13)
        c = compute_c5_val(ht)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 2: Coordinate descent on n (25% budget)
    d2 = t0 + budget_s * 0.25
    if time.time() < d2 - 10:
        best_h, best_c5 = coord_descent(best_h, n, d2, rng)
        best_n = n

    # Phase 3: Try larger n values (up to 70% budget)
    d3 = t0 + budget_s * 0.70
    n_options = [100, 128, 160]
    for n_try in n_options:
        if time.time() > d3 - 60:
            break
        h_try = upsample_func(best_h, best_n, n_try)
        if time.time() < d3 - 40:
            h_try = slsqp(h_try, n_try, 5000, 1500, 1e-12)
        phase_dl = min(d3, time.time() + budget_s * 0.12)
        if time.time() < phase_dl - 5:
            h_try, c_try = coord_descent(h_try, n_try, phase_dl, rng)
            if c_try < best_c5:
                best_h, best_c5, best_n = h_try, c_try, n_try

    # Phase 4: Random restarts (up to 85% budget)
    d4 = t0 + budget_s * 0.85
    for _ in range(8):
        if time.time() > d4 - 25:
            break
        scale = rng.choice([0.01, 0.02, 0.04])
        h_pert = best_h + rng.normal(0, scale, size=best_n)
        h_pert = project_feasible(h_pert, best_n)
        rd = min(d4, time.time() + 30)
        h_pert, c_pert = coord_descent(h_pert, best_n, rd, rng)
        if c_pert < best_c5:
            best_h, best_c5 = h_pert, c_pert

    # Phase 5: Final SLSQP + coord descent polish
    if time.time() < total_deadline - 30:
        ht = slsqp(best_h, best_n, 100000, 3000, 1e-15)
        c = compute_c5_val(ht)
        if c < best_c5:
            best_h, best_c5 = ht, c
    if time.time() < total_deadline - 5:
        best_h, best_c5 = coord_descent(best_h, best_n, total_deadline - 1, rng)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5_val(best_h)
    return best_h, best_c5, best_n
