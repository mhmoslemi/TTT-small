def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h_i) = target_sum}"""
    h = np.asarray(h, dtype=float)
    n = len(h)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune to exact sum
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        # Adjust non-boundary elements
        interior = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if interior.sum() > 0:
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    n = len(h)

        corr = corr[idx]

def project_to_constraint(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target_sum:

        else:

    if abs(diff) > 1e-15:

        if interior.sum() > 0:

            proj = np.clip(proj, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-15:
        interior = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if interior.sum() > 0:
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    val = m + np.log(e.sum()) / beta
    grad = np.zeros(n)
    for jj in range(2 * n - 1):
        wi = w[jj] * dx
        if wi < 1e-15:
            continue
        l = jj - (n - 1)
        if l == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif l > 0:
            grad[:n - l] += wi * (1.0 - h[l:])
            grad[l:] -= wi * h[:n - l]
        else:
            ml = -l
            grad[ml:] += wi * (1.0 - h[:n - ml])
            grad[:n - ml] -= wi * h[ml:]
    return val, grad

def pgd_optimize(h0, target_sum, betas, max_iter, deadline, bb=True):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_h, prev_g = None, None
    alpha = 0.03
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        phase = it / max(1, max_iter - 1)
        bi = betas[min(int(phase * len(betas)), len(betas) - 1)]
        val, g = obj_and_grad(h, bi)
        if bb and prev_g is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s.dot(y)
            ss = s.dot(s)
            if ss > 1e-16 and sy > 1e-16:
                alpha = max(1e-6, min(sy / ss, 0.4))
        h_new = project_to_constraint(h - alpha * g, target_sum)
        val_new, _ = obj_and_grad(h_new, bi)
        if val_new > val:
            alpha *= 0.5
            h_new = project_to_constraint(h - alpha * g, target_sum)
        prev_h, prev_g = h.copy(), g.copy()
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interp_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src, period=2.0)

def slsqp_polish(h0, n, target_sum, beta, deadline):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        e = np.exp(beta * (corr - m))
        return m + np.log(e.sum()) / beta

    def _objg(x):
        _, g = obj_and_grad(x, beta)
        return g

    def _con(x):
        return x.sum() - target_sum

    def _conj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _con, 'jac': _conj}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds, jac=_objg,
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
        h_new = project_to_constraint(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def generate_starts(n_pts, target_sum, parent, rng):
    starts = []
    h_int = project_to_constraint(interp_to(parent, n_pts), target_sum)
    starts.append(h_int)

    h_bin = np.zeros(n_pts)
    h_bin[rng.choice(n_pts, n_pts // 2, replace=False)] = 1.0
    starts.append(h_bin.copy())

    h_rand = rng.random(n_pts)
    ksz = max(2, n_pts // 25)
    kern = np.ones(ksz) / ksz
    h_filt = np.convolve(h_rand, kern, mode='same')
    starts.append(project_to_constraint(h_filt, target_sum))

    h_2ph = np.zeros(n_pts)
    q = n_pts // 4
    h_2ph[:q] = 1.0
    h_2ph[2 * q:3 * q] = 1.0
    starts.append(project_to_constraint(h_2ph, target_sum))

    h_5ph = np.zeros(n_pts)
    fifth = n_pts // 5
    for i in range(0, 5, 2):
        h_5ph[i * fifth:(i + 1) * fifth] = 1.0
    starts.append(project_to_constraint(h_5ph, target_sum))

    h_sine = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * np.arange(n_pts) / n_pts)
    starts.append(project_to_constraint(h_sine, target_sum))

    return starts

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    abs_deadline = t0 + budget_s - 12
    best_h, best_c5, best_n = None, float('inf'), 0

    parent = None
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
    if parent is None or len(parent) == 0:
        parent = np.full(70, 0.5)

    rng = np.random.RandomState(seed)
    base_n = len(parent)
    if base_n <= 80:
        res_list = [base_n, 90, 100, 120, 140, 160]
    elif base_n <= 120:
        res_list = [base_n, base_n + 20, base_n + 40, base_n + 60]
    else:
        res_list = [base_n, base_n + 20]
    res_list = [r for r in res_list if r <= 200]

    n_res = len(res_list)
    weights = np.linspace(0.7, 1.3, n_res)
    weights /= weights.sum()
    time_alloc = weights * (budget_s - 100)

    curr_parent = parent.copy()

    for r_idx, n_pts in enumerate(res_list):
        if time.time() >= abs_deadline - 30:
            break
        t_start = time.time()
        t_dead = min(abs_deadline - 25, t_start + time_alloc[r_idx])
        target_sum = n_pts / 2.0

        starts = generate_starts(n_pts, target_sum, curr_parent, rng)
        curr_best_h, curr_best_c5 = None, float('inf')

        for si, h0 in enumerate(starts):
            if time.time() >= t_dead:
                break
            rem = t_dead - time.time()
            if rem < 5:
                break

            betas_pgd = [15, 30, 60, 100, 150]
            max_it = max(300, int(rem * 20))
            h_sol, c5_sol = pgd_optimize(h0, target_sum, betas_pgd, max_it,
                                         min(t_dead - 5, time.time() + rem * 0.6))
            if c5_sol < curr_best_c5:
                curr_best_h, curr_best_c5 = h_sol.copy(), c5_sol

            if time.time() < t_dead - 3:
                h_pol, c5_pol = slsqp_polish(h_sol, n_pts, target_sum, 150, t_dead)
                if c5_pol < curr_best_c5:
                    curr_best_h, curr_best_c5 = h_pol.copy(), c5_pol

        if curr_best_h is not None:
            if curr_best_c5 < best_c5:
                best_h, best_c5, best_n = curr_best_h.copy(), curr_best_c5, n_pts
            curr_parent = curr_best_h.copy()

    if best_h is not None and time.time() < abs_deadline:
        n_f = best_n
        target = n_f / 2.0
        h_fin, c5_fin = slsqp_polish(best_h, n_f, target, 300, abs_deadline - 5)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
        if time.time() < abs_deadline - 2:
            h_fin2, c5_fin2 = pgd_optimize(best_h, target, [200, 400, 600], 300, abs_deadline)
            if c5_fin2 < best_c5:
                best_h, best_c5 = h_fin2, c5_fin2

    if best_h is None:
        best_h = parent.astype(float).copy()
        best_c5 = compute_c5(best_h)
        best_n = len(best_h)

    return best_h, best_c5, best_n
