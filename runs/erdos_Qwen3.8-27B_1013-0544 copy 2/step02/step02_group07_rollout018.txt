def compute_gradient(h, beta, dx):
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    # weights = softmax(beta * corr)
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        d = k - (n - 1)
        w = weights[k]
        if w < 1e-10:
            continue
        if d >= 0:
            # (1-h[j+d]) for j in [0, n-1-d]
            if n - 1 - d >= 0:
                js = np.arange(0, n - d)
                grad[js] += w * (1.0 - h[js + d])
            # -h[j-d] for j in [d, n-1]
            if d <= n - 1:
                js = np.arange(d, n)
                grad[js] -= w * h[js - d]
        else:
            e = -d
            # (1-h[j-e]) for j in [e, n-1]
            if e <= n - 1:
                js = np.arange(e, n)
                grad[js] += w * (1.0 - h[js - e])
            # -h[j+e] for j in [0, n-1-e]
            if n - 1 - e >= 0:
                js = np.arange(0, n - e)
                grad[js] -= w * h[js + e]
    
    return grad * dx

            else:
                i, j = np.random.choice(n_points, 2, replace=False)
                h_new[i], h_new[j] = h_new[j], h_new[i]

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(500.0, delta / max(T, 1e-14))):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            T = max(T * cooling, T_min)
            if no_improve > 350:
                T = max(T, 0.003)
                no_improve = 0

    # Phase 3: Multi-start L-BFGS-B from perturbed best
    if tleft() > 80:
        for attempt in range(12):
            if tleft() < 20:
                break
            h0 = best_h.copy()
            np_ = np.random.randint(3, 10)
            idx = np.random.choice(n_points, np_, replace=False)
            h0[idx] += np.random.randn(np_) * 0.025
            h0 = project_h(h0, target_sum)
            beta = np.random.choice([300, 600, 1200, 2500])
            try:
                res = minimize(lse_obj, h0, args=(beta, dx, target_sum, 1e6),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-17, 'gtol': 1e-11})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 4: Try larger n_points
    for n_try in [80, 100, 140]:
        if tleft() < 100:
            break
        x_old = np.linspace(0, 2, n_points, endpoint=False) + 1.0 / n_points
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        h_interp = project_h(h_interp, target_try)
        bounds_try = Bounds(np.zeros(n_try), np.ones(n_try))
        cur_try = h_interp.copy()

        for beta in [100, 300, 800, 2000]:
            if tleft() < 40:
                break
            try:
                res = minimize(lse_obj, cur_try, args=(beta, dx_try, target_try, 1e5),
                               method='L-BFGS-B', bounds=bounds_try,
                               options={'maxiter': 3000, 'ftol': 1e-17, 'gtol': 1e-10})
                h_new = project_h(res.x, target_try)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    n_points = n_try
                    dx = dx_try
                    target_sum = target_try
                    bounds = bounds_try
                    cur_try = h_new.copy()
            except Exception:
                pass

    # Phase 5: Final high-beta polish with SLSQP (constraint-exact)
    if tleft() > 30:
        for beta in [3000, 8000, 15000]:
            if tleft() < 12:
                break
            try:
                res = minimize(lse_obj, best_h, args=(beta, dx, target_sum, 1e8),
                               method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': sum_equal,
                                             'jac': sum_jac, 'args': (target_sum,)}],
                               options={'maxiter': 8000, 'ftol': 1e-17})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
