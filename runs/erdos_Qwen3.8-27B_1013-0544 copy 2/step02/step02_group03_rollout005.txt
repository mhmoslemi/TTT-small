    h[i]*(1-h[i+L]) = h[i] - h[i]*h[i+L]
    ∂/∂h[j]: if j = i, get 1 - h[j+L]; if j = i+L, get -h[j-L]
    For j = i: i = j, valid if 0 <= j <= n-1-L
    For j = i+L: i = j-L, valid if 0 <= j-L <= n-1-L, i.e., L <= j <= n-1
  
  So for L >= 0:
  ∂f_L/∂h[j] = (1-h[j+L]) * [j <= n-1-L] + (-h[j-L]) * [L <= j <= n-1]
  
  For j in [0, L-1]: only first term, contribution = 1-h[j+L]
  For j in [L, n-1-L]: both terms, contribution = 1-h[j+L] - h[j-L]
  For j in [n-L, n-1]: only second term, contribution = -h[j-L]

This matches what the previous code had. Let me keep the same subgradient computation.

Now, for the full gradient of smooth_max:
∇ smooth_max(corr, beta) = sum_k w_k * ∇ corr_k
where w_k = exp(beta*(corr_k - m)) / sum_j exp(beta*(corr_j - m))

And corr_k corresponds to lag L = k - (n-1).

So the algorithm is:
1. Compute corr = np.correlate(h, 1-h, mode='full')
2. Compute weights w_k for each k
3. For each k with significant weight, compute the subgradient for that lag and add w_k * subgrad_k

This is O(n * num_significant_lags) per iteration. With n=70 and beta large, only a few lags will have significant weight, so it's fast.

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def smooth_max_jac(h, dx, beta, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    log_weights = np.clip(beta * (corr - m), -700, 700)
    log_weights -= np.max(log_weights)
    weights = np.exp(log_weights)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += w * (1.0 - h[i + lag])
                grad[i + lag] -= w * h[i]
        else:
            la = -lag
            for i in range(n - la):
                grad[i + la] += w * (1.0 - h[i])
                grad[i] -= w * h[i + la]
    return grad * dx

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def smooth_max_jac(h, dx, beta, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    log_w = np.clip(beta * (corr - m), -700, 700)
    log_w -= np.max(log_w)
    weights = np.exp(log_w)
    weights /= weights.sum()
    grad = np.zeros(n)
    for lag in range(n):
        w = weights[n - 1 + lag]
        if w < 1e-10:
            continue
        idx = np.arange(n - lag)
        grad[idx] += w * (1.0 - h[idx + lag])
        grad[idx + lag] -= w * h[idx]
    for la in range(1, n):
        w = weights[n - 1 - la]
        if w < 1e-10:
            continue
        idx = np.arange(n - la)
        grad[idx + la] += w * (1.0 - h[idx])
        grad[idx] -= w * h[idx + la]
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h[h > 0.5] = np.minimum(1.0, h[h > 0.5] + (target_sum - s) / max(1, (h > 0.5).sum()))
        h[h <= 0.5] = np.maximum(0.0, h[h <= 0.5] - (target_sum - s) / max(1, (h <= 0.5).sum()))
        h = np.clip(h, 0.0, 1.0)
    return h

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def slsqp_optimize(h, dx, target_sum, n, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    try:
        res = minimize(smooth_max_obj, h, args=(dx, beta), method='SLSQP',
                       jac=lambda x: smooth_max_jac(x, dx, beta, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h

def momentum_gradient_descent(h, dx, target_sum, n, beta, n_iters, tl):
    h_cur = h.copy()
    velocity = np.zeros(n)
    mu = 0.7
    lr = 0.05
    avg_h = h_cur.copy()
    best_local_h = h_cur.copy()
    best_local_c5 = compute_c5(h_cur)
    for it in range(n_iters):
        if tl() < 30:
            break
        f, g = smooth_max_obj(h_cur, dx, beta), smooth_max_jac(h_cur, dx, beta, n)
        g = g - g.mean()
        velocity = mu * velocity - lr * g
        h_cur = h_cur + velocity
        h_cur = project_feasible(h_cur, target_sum)
        avg_h = (it * avg_h + h_cur) / (it + 1)
        if it % 2000 == 1999:
            c5_curr = compute_c5(h_cur)
            if c5_curr < best_local_c5:
                best_local_c5 = c5_curr
                best_local_h = h_cur.copy()
            c5_avg = compute_c5(avg_h)
            if c5_avg < best_local_c5:
                best_local_c5 = c5_avg
                best_local_h = avg_h.copy()
        if it % 10000 == 9999:
            lr *= 0.8
    c5_final = compute_c5(h_cur)
    if c5_final < best_local_c5:
        best_local_c5 = c5_final
        best_local_h = h_cur.copy()
    return best_local_h, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with gradient, moderate beta
    for beta in [5, 20, 80]:
        if tl() < 80:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, n, beta, 5000, 1e-13)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 2: SLSQP with gradient, higher beta
    for beta in [300, 1000, 3000]:
        if tl() < 150:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, n, beta, 8000, 1e-14)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 3: Momentum gradient descent on smooth max
    if tl() > 120:
        h_mg, c5_mg = momentum_gradient_descent(best_h, dx, target_sum, n, 500, 100000, tl)
        if c5_mg < best_c5:
            best_h = h_mg
            best_c5 = c5_mg

    # Phase 4: SLSQP polish from gradient descent result
    for beta in [5000, 20000, 100000]:
        if tl() < 60:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, n, beta, 10000, 1e-15)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 5: Try different resolutions
    for n_try in [n - 10, n + 15, n + 30, n - 20, n + 50]:
        if n_try < 40 or tl() < 60:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        c5_t = compute_c5(h_t)
        for beta in [500, 5000]:

                break
            h_new = slsqp_optimize(h_t, dx_t, ts_t, n_try, beta, 6000, 1e-13)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
                n = n_try
                dx = dx_t
                target_sum = ts_t
                h_t = best_h.copy()

    # Phase 6: Final high-beta polish
    if tl() > 30:
        for beta in [50000, 200000]:
            if tl() < 15:
                break
            h_new = slsqp_optimize(best_h, dx, target_sum, n, beta, 12000, 1e-15)
            c5 = compute_c5(h_new)
            if c5 < best_c5:

        if c5_mg < best_c5:
            best_h = h_mg

        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
