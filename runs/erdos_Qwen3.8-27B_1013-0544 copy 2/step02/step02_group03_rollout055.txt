def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient is weighted sum of per-lag gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += weights[k] * (1.0 - h[idx + lag])
            grad[idx + lag] -= weights[k] * h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += weights[k] * (1.0 - h[idx])
            grad[idx] -= weights[k] * h[idx + la]
    return grad * dx * beta

    for beta in [100, 500, 2000, 10000]:
        if remaining() < 30:
            break
        h_new = slsqp_optimize(best_h, n, beta, maxiter=5000, ftol=1e-15)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 3: Subgradient with momentum on true objective
    if remaining() > 60:
        h_sg, c5_sg = subgrad_momentum(best_h, n, target_sum, 60000, t0, budget_s)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    # Phase 4: High-beta SLSQP polish
    for beta in [50000, 200000, 1000000]:
        if remaining() < 20:
            break
        h_new = slsqp_optimize(best_h, n, beta, maxiter=8000, ftol=1e-16)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 5: Multi-start with small perturbations
    if remaining() > 50:
        for trial in range(10):
            if remaining() < 15:
                break
            scale = 0.002 + 0.003 * trial
            x0 = best_h + np.random.randn(n) * scale
            x0 = project_feasible(x0, target_sum)
            h_new = slsqp_optimize(x0, n, 50000, maxiter=4000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 6: Resolution exploration
    if remaining() > 80:
        for n_try in [n + 30, n - 20, 100, 140, n + 70]:
            if n_try < 30 or remaining() < 40:
                continue
            h_t = resample_h(best_h, n_try)
            ts_t = n_try / 2.0
            h_t = project_feasible(h_t, ts_t)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 + 0.002:
                for beta in [200, 2000, 20000]:
                    if remaining() < 20:
                        break
                    h_new = slsqp_optimize(h_t, n_try, beta, maxiter=4000)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        n = n_try
                        target_sum = ts_t

    # Phase 7: Final subgradient + SLSQP
    if remaining() > 40:
        h_sg, c5_sg = subgrad_momentum(best_h, n, target_sum, 40000, t0, budget_s)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg
    if remaining() > 15:
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=10000, ftol=1e-16)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 8: Final projected gradient polish
    if remaining() > 15:
        h_new, c5_new = proj_grad_smooth(best_h, n, 10000, 20000, t0, budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    return best_h, best_c5, len(best_h)
