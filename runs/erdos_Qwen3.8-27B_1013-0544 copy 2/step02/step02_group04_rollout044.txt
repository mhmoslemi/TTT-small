import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_feasible(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def corr_grad(h, k, n):
    grad = np.zeros(n)
    for j in range(n):
        v = 0.0
        if 0 <= j + k < n:
            v += 1.0 - h[j + k]
        if 0 <= j - k < n:
            v -= h[j - k]
        grad[j] = v
    return grad


def active_set_grad(h, n, frac=0.005):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    active = np.where(corr >= mx - frac * mx)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        grad += corr_grad(h, k, n)
    grad /= len(active)
    return grad, mx


def make_slsqp_obj(beta, dx, n):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    def obj_jac(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        grad = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] < 1e-12:
                continue
            k = idx - (n - 1)
            grad += w[idx] * corr_grad(h, k, n)
        return grad * dx
    return obj, obj_jac

    best_c5 = compute_c5(best_h)

                    'jac': lambda x: np.ones(x.shape[0])}]
