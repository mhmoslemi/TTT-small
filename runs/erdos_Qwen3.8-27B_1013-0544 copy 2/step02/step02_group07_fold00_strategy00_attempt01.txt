def gaussian_mutation(individual, sigma=0.1):
   mask = np.random.random(len(individual)) < 0.2
   noise = np.random.normal(0, sigma, np.sum(mask))
   idx = np.random.choice(len(individual), np.sum(mask), replace=False)
   individual[mask] += noise[:len(noise)]
   individual = project_h(individual, target_sum)

def block_swap_mutation(individual):
   start = np.random.randint(0, len(individual)-10)
   length = np.random.randint(3, 15)
   block = individual[start:start+length]
   individual = np.concatenate([individual[:start], individual[start+length:], block])
   individual = project_h(individual, target_sum)

def lag_targeted_mutation(individual):
    # Identify worst lags
   corr = np.correlate(individual, 1-individual, mode="full")
   k_max = np.argmax(corr) - (len(individual)-1)
   d_positive = k_max  # positive lag
   d_negative = -(np.argmax(corr[::-1]) - 1)  # negative lag
   d = max(d_positive, d_negative)
   # Apply adjustment at position (if d_positive was worst)
   if d_positive >= abs(d_negative):
      if individual[len(individual)-d_positive] > 0.5 and individual[d_positive] < 0.5:
          individual[d_positive] -= 0.05
      else:
          individual[d_positive] += 0.05
   else:
      # Similar logic for negative lag
   individual = np.clip(individual, 0, 1)
   individual = project_h(individual, target_sum)

   for ind in new_population:
      if random.random() < mutation_rate:
         mutate(ind)

   # Local optimization on best candidates
   elite_fitness = np.sort(fitnesses)[0]  # Best fitness
   while True:
      new_candidates = []
      for _ in range(num_local_optimizations):
         if time_remaining low: break
         individual = new_population.pop(random.randint(0, len(new_population)-1))
         # Local optimization
         optimized_ind = local_optimization(individual)
         new_candidates.append(optimized_ind)

      # Choose best from new_candidates
      best_new = min(new_candidates, key=fitness_function)
      if fitness(best_new) < elite_fitness:
         # Merge with old population
         new_population.append(best_new)
