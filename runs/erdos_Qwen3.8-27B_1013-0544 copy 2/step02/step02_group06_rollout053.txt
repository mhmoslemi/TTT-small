                         options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 2: Exact coordinate descent (key improvement)
    if tl() > 60:
        h_cd = best_h.copy()
        h_cd = coord_descent(h_cd, n, n_sweeps=50, lam=500.0)
        c = compute_c5(h_cd)
        if c < best_c5:
            best_h, best_c5 = h_cd, c

    # Phase 3: Alternating L-BFGS-B and coordinate descent with restarts
    n_restarts = 0
    while tl() > 150 and n_restarts < 50:
        n_restarts += 1
        temp = max(0.003, 0.04 * np.exp(-n_restarts / 12.0))

        h = best_h.copy()
        strat = n_restarts % 7
        if strat == 0:
            h += temp * rng.randn(n)
        elif strat == 1:
            mask = rng.random(n) < 0.12
            h[mask] = np.where(h[mask] > 0.5, 0.85, 0.15)
        elif strat == 2:
            shift = rng.randint(1, n // 3 + 1)
            h = np.roll(h, shift)
        elif strat == 3:
            freq = rng.uniform(1, 5) * n
            h += 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        elif strat == 4:
            b1 = rng.randint(0, max(1, n // 3))
            blen = rng.randint(4, min(12, n // 3))
            h[b1:b1 + blen] = 1.0 - h[b1:b1 + blen]
        elif strat == 5:
            h_rand = rng.uniform(0, 1, n)
            mask = rng.random(n) < 0.25
            h[mask] = h_rand[mask]
        else:
            h = rng.uniform(0, 1, n)

        h = project_box_sum(h, ts)

        # Optimize: L-BFGS-B then coordinate descent
        for beta in [400, 1000]:
            if tl() < 15:
                break
            try:
                r = minimize(obj_lbfgs, h, args=(n, ts, beta, 500.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

        # Coordinate descent polish
        if tl() > 15:
            h_cd = best_h.copy()
            h_cd = coord_descent(h_cd, n, n_sweeps=25, lam=500.0)
            c = compute_c5(h_cd)
            if c < best_c5:
                best_h, best_c5 = h_cd, c

    # Phase 4: Final high-precision polish
    if tl() > 30:
        for beta, lam in [(5000, 5000), (8000, 8000)]:
            if tl() < 15:
                break
            try:
                r = minimize(obj_lbfgs, best_h, args=(n, ts, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-16})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Final coordinate descent with high precision
    if tl() > 20:
        h_cd = best_h.copy()
        h_cd = coord_descent(h_cd, n, n_sweeps=50, lam=1000.0)
        c = compute_c5(h_cd)
        if c < best_c5:
            best_h, best_c5 = h_cd, c

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
