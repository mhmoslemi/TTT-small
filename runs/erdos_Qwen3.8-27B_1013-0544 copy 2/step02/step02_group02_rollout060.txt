```python
def grad_c(idx, h, n):
    """Gradient of np.correlate(h, 1-h, 'full')[idx] * dx with respect to h"""
    dx = 2.0 / n
    n_minus_1 = n - 1
    grad = np.zeros(n)
    
    if idx == n_minus_1:
        # Lag 0: c = sum h[i](1-h[i]), dc/dh[j] = 1 - 2*h[j]
        grad = (1 - 2 * h) * dx
    elif idx < n_minus_1:
        # idx = n-1-lag, so lag = n-1-idx
        lag = n_minus_1 - idx
        # c = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
        # dc/dh[j] = (1-h[j+lag]) * I(0<=j<=n-1-lag) - h[j-lag] * I(lag<=j<=n-1)
        for j in range(n - lag):
            grad[j] += (1 - h[j + lag]) * dx
        for j in range(lag, n):
            grad[j] += (-h[j - lag]) * dx
    else:
        # idx > n-1, lag = idx - (n-1)
        lag = idx - n_minus_1
        # c = sum_{i=0}^{n-1-lag} h[i+lag](1-h[i])
        # dc/dh[j] = (1-h[j-lag]) * I(lag<=j<=n-1) - h[j+lag] * I(0<=j<=n-1-lag)
        for j in range(lag, n):
            grad[j] += (1 - h[j - lag]) * dx
        for j in range(n - lag):
            grad[j] += (-h[j + lag]) * dx
    
    return grad
```

```python
def projected_gradient(h, n, t0, budget_s, alpha=1000, max_iter=10000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    lr = 1e-4
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - 30:
            break
        # Compute correlations
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        
        # Compute weights
        shifted = alpha * (c - m)
        # Clip to avoid underflow issues
        shifted = np.clip(shifted, -700, 0)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] > 1e-10:
                grad += w[k] * grad_c(k, h, n)
        grad *= alpha
        
        # Step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
            h = h_new
        else:
            # Backtracking: reduce lr
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5
```

```python
def momentum_gradient(h, n, t0, budget_s, alpha=1000, max_iter=20000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    lr = 5e-4
    momentum = np.zeros(n)
    beta = 0.9
    lr_history = []
    
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 30:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        
        shifted = np.clip(alpha * (c - m), -700, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Sparse gradient: only compute for lags with significant weight
        grad = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for k in active:
            grad += w[k] * grad_c(k, h, n)
        grad *= alpha
        
        # Momentum step
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
            h = h_new
            # Occasionally increase lr
            if it % 1000 == 0:
                lr = min(lr * 1.5, 1e-2)
        else:
            h = best_h.copy()  # revert
            lr *= 0.7
            momentum *= 0.5
            if lr < 1e-8:
                # Try a different direction
                lr = 1e-3
    
    return best_h, best_c5
```
