Let me redo:
∂corr[m]/∂h[j] = [0≤j+l<n] * (1-h[j+l]) - [0≤j-l<n] * h[j-l]

For the extended array:

And: [0≤j-l<n] * h[j-l] = [0≤n+j-l<2n] * h_ext[n+j-l]

def grad_smooth_max_corr(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Softmax weights
    shifted = beta * (corr - corr.max())
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w /= w.sum()  # shape (2n-1,)
    
    # Lag for output index m: l = (n-1) - m
    m_idx = np.arange(2*n - 1)  # (2n-1,)
    lags = (n - 1) - m_idx  # (2n-1,)
    
    j = np.arange(n)  # (n,)
    
    # idx1[j, k] = j + lags[k] (position for 1-h term)
    # valid when 0 <= idx1 < n
    offset1 = j[:, None] + lags[None, :]  # (n, 2n-1)
    valid1 = (offset1 >= 0) & (offset1 < n)
    
    # idx2[j, k] = j - lags[k] (position for h term)
    # valid when 0 <= idx2 < n
    offset2 = j[:, None] - lags[None, :]  # (n, 2n-1)
    valid2 = (offset2 >= 0) & (offset2 < n)
    
    # term1[j] = Σ_k w[k] * valid1[j,k] * (1 - h[offset1[j,k]])
    # term2[j] = Σ_k w[k] * valid2[j,k] * h[offset2[j,k]]
    
    # Use np.take with clipping for safe indexing
    idx1_clip = np.clip(offset1, 0, n-1)
    idx2_clip = np.clip(offset2, 0, n-1)
    
    h1 = h[idx1_clip]  # (n, 2n-1)
    h2 = h[idx2_clip]  # (n, 2n-1)
    
    term1 = (w[None, :] * valid1 * (1.0 - h1)).sum(axis=1)
    term2 = (w[None, :] * valid2 * h2).sum(axis=1)
    
    return dx * (term1 - term2)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(sh))) / beta

def grad_smooth_max(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sh = np.clip(beta * (corr - corr.max()), -50, 50)
    w = np.exp(sh)
    w /= w.sum()
    lags = (n - 1) - np.arange(2 * n - 1)
    j = np.arange(n)
    off1 = j[:, None] + lags[None, :]
    off2 = j[:, None] - lags[None, :]
    v1 = (off1 >= 0) & (off1 < n)
    v2 = (off2 >= 0) & (off2 < n)
    c1 = np.clip(off1, 0, n - 1)
    c2 = np.clip(off2, 0, n - 1)
    t1 = (w[None, :] * v1 * (1.0 - h[c1])).sum(axis=1)
    t2 = (w[None, :] * v2 * h[c2]).sum(axis=1)
    return dx * (t1 - t2)

def obj_pen(h, beta, dx, tgt, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx + lam * (h.sum() - tgt) ** 2

def obj_pen_grad(h, beta, dx, tgt, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sh = np.clip(beta * (corr - corr.max()), -50, 50)
    w = np.exp(sh)
    w /= w.sum()

    off2 = j[:, None] - lags[None, :]
    v1 = (off1 >= 0) & (off1 < n_global)
    v2 = (off2 >= 0) & (off2 < n_global)
    c1 = np.clip(off1, 0, n_global - 1)
    c2 = np.clip(off2, 0, n_global - 1)
    t1 = (w[None, :] * v1 * (1.0 - h[c1])).sum(axis=1)
    t2 = (w[None, :] * v2 * h[c2]).sum(axis=1)
    return dx * (t1 - t2) + 2.0 * lam * (h.sum() - tgt)

n_global = 0
