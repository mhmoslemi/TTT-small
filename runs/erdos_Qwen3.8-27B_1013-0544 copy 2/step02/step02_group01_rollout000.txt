def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons_sum = lambda x: x.sum() - target
    jac_sum = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_sum, 'jac': jac_sum}]
    try:

                       bounds=bounds, constraints=cons,

        # Multiple BFPG with different active set sizes
        for na in [5, 10, 15]:
            if remaining() < 30:
                break
            h_r, c_r = bfpg_optimize(h_b, 3000, n_active=na, lr0=0.03, lr_decay=0.9999, update_period=20)
            if c_r < best_c5:
                h_best, best_c5 = h_r, c_r
                h_b = h_best.copy()

    # Phase 5: Final high-beta SLSQP polish
    for beta in [500, 800, 1200]:
        if remaining() < 20:
            break
        h_f = slsqp_refine(h_best, maxiter=600, beta=beta)
        cf = compute_c5(h_f)
        if cf < best_c5:
            h_best, best_c5 = h_f, cf
        # Short BFPG after SLSQP to escape potential bad direction
        if remaining() > 30:
            h_p, cp = bfpg_optimize(h_f, 500, n_active=5, lr0=0.02, lr_decay=0.9999, update_period=10)
            if cp < best_c5:
                h_best, best_c5 = h_p, cp

    # Final verification and projection
    n_final = len(h_best)
    h_best = project_feasible(h_best, n_final / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
