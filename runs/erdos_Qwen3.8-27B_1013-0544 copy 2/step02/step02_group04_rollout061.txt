    max_shifted = np.max(shifted)

    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):

    for _ in range(100):

        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            break

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    ms = np.max(shifted)
    exp_s = np.exp(shifted - ms)
    s = exp_s.sum()
    weights = exp_s / s
    value = m + (ms + np.log(s)) / beta
    return value, weights

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max_and_weights(corr, beta)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k in range(-(n-1), n):
            wk = w[k + n - 1]
            if wk < 1e-14:

                val += wk * (1.0 - h[j - k])
            if 0 <= j + k < n:
                val -= wk * h[j + k]
        grad[j] = val
    return dx * grad

    # Create a Toeplitz-like matrix M where M[j, i] = w[j-i+n-1] if valid

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    ms = np.max(shifted)
    exp_s = np.exp(shifted - ms)
    s = exp_s.sum()
    weights = exp_s / s
    value = m + (ms + np.log(s)) / beta
    return value, weights

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def _obj_h(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max_and_weights(corr, beta)
    return val * dx

def _obj_h_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max_and_weights(corr, beta)
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in range(n):
        i_min = max(0, j - (n - 1))
        i_max = min(j, n - 1)
        if i_min <= i_max:
            idx = j - np.arange(i_min, i_max + 1) + (n - 1)
            grad[j] += w[idx] @ omh[i_min:i_max + 1]
        i_min2 = j
        i_max2 = min(j + (n - 1), n - 1)
        if i_min2 <= i_max2:
            idx2 = np.arange(i_min2, i_max2 + 1) - j + (n - 1)
            grad[j] -= w[idx2] @ h[i_min2:i_max2 + 1]
    return dx * grad

def _con_fun(h, ts):
    return np.sum(h) - ts

def _con_jac(h, ts):
    return np.ones(len(h))

def _subgrad_single(h, k_lag, n):
    grad = np.zeros(n)
    if k_lag >= 0:
        grad[k_lag:] += (1.0 - h[:n - k_lag])
        grad[:n - k_lag] -= h[k_lag:]
    else:
        kk = -k_lag
        grad[:n - kk] += (1.0 - h[kk:])
        grad[kk:] -= h[:n - kk]
    grad -= np.mean(grad)
    return grad
