import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_block(h, n, t0, budget_s, rng, n_iter, e_start, e_min, d_start, d_min, block_size):

    D = d_start
    e_ratio = (e_min / e_start) ** (1.0 / n_iter)
    d_ratio = (d_min / d_start) ** (1.0 / n_iter)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5

    polish_interval = 30.0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 30:

            h = best_h.copy()
            current_c5 = best_c5

            if c_pol < best_c5:

                best_c5 = c_pol
                h = h_pol.copy()
                current_c5 = c_pol
            if budget_s - (time.time() - t0) < 30:
                break

        start = rng.randint(0, n - block_size + 1)
        idx = np.arange(start, start + block_size)
        deltas = rng.uniform(-D, D, size=block_size)
        h_new = h.copy()
        h_new[idx] += deltas
        # Project to feasible

        # Check if projection changed values outside block significantly
        # (project_feasible might change other values to maintain sum)
        new_c5 = compute_c5(h_new, n)
        delta = new_c5 - current_c5

            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5

            h[idx] -= deltas  # This isn't quite right after projection
            # Actually, let's just keep the old h
        E *= e_ratio
        D *= d_ratio
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_lag_driven_step(h, n, t0, budget_s, rng, n_steps):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_steps):
        if budget_s - (time.time() - t0) < 20:
            break
        c = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_active = np.argmax(c)
        lag = k_active - (n - 1)
        # Try to reduce this lag by moving h values
        if lag > 0:
            pairs = [(i, i + lag) for i in range(n - lag)]
        else:
            pairs = [(i - lag, i) for i in range(n + lag)]
        # Randomly pick a pair and try to adjust
        pi, pj = pairs[rng.randint(len(pairs))]
        for d in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
            hi = best_h[pi] + d
            hj = best_h[pj] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                best_h[pi] = hi
                best_h[pj] = hj
                c_new = compute_c5(best_h, n)
                if c_new < best_c5:
                    best_c5 = c_new
                    break
                else:
                    best_h[pi] -= d
                    best_h[pj] += d
        # Also try generic 2-point moves
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
