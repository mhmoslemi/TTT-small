import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def sum_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def sum_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def slsqp_solve(h0, n, maxiter=5000, pen_weight=1e6):
    c0 = compute_c5(h0, n)
    z0 = np.concatenate([h0.copy(), [c0 + 0.01]])
    
    def obj(z):
        h = z[:n]
        M = z[n]
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        viol = np.maximum(0.0, c - M)
        return M + pen_weight * np.sum(viol * viol)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'eq', 'fun': lambda z: sum_fun(z, n),
             'jac': lambda z: sum_jac(z, n)}]
    
    try:
        res = minimize(obj, z0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x[:n], n)
    except Exception:
        return h0.copy()

def nm_solve(h0, n, maxiter=20000):
