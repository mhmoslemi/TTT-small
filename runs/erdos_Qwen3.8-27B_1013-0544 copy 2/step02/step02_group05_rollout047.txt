import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def lag_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            grad[j] += (-h[j + k])
    else:
        m = -k
        for j in range(0, n - m):
            grad[j] += (1.0 - h[j + m])
        for j in range(m, n):
            grad[j] += (-h[j - m])
    return grad

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        grad += weights[idx] * lag_gradient(h, k)
    return grad * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def cons_fun(x, target_sum):
    return x.sum() - target_sum

def cons_jac(x):
    return np.ones(len(x))

def active_subgrad(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        grad += lag_gradient(h, k)
    return grad / len(top_idx)

    start_time = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        parent = np.full(70, 0.5)

    def get_h0(n_points):
        if len(parent) == n_points:
            return parent.copy()
        elif len(parent) > n_points:
            idx = np.linspace(0, len(parent) - 1, n_points).astype(int)
            return parent[idx]
        else:
            x_old = np.linspace(0, 2, len(parent), endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            return np.interp(x_new, x_old, parent, period=2.0)

    n_candidates = [70, 80, 100, 120]
    time_per_n = (budget_s - 60) / len(n_candidates)

    for n_points in n_candidates:
        t0 = time.time()
        if time.time() - start_time > budget_s - 35:
            break

        target_sum = n_points / 2.0
        dx = 2.0 / n_points
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))

        h0 = project_feasible(get_h0(n_points), target_sum)
        curr_h = h0.copy()
        curr_c5 = compute_c5(curr_h)
        if curr_c5 < best_c5:

        # Phase 1: SLSQP with increasing beta, analytical gradient
        for beta in [25, 50, 100, 200, 400]:
            if time.time() - start_time > budget_s - 40:

                return smooth_max_val(np.correlate(h, 1.0 - h, mode="full"), float(beta)) * dx

                return smooth_max_grad(h, float(beta))

                     'jac': lambda x: np.ones(len(x))}]

                              constraints=cons, jac=obj_jac,
