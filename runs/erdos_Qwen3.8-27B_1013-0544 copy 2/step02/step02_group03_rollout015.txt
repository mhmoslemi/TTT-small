    # Autocorrelation via FFT
    N = 2 * n
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    # ac[m] = sum_{i=0}^{n-1-m} h[i]*h[i+m]
    
    # Prefix sums
    ps = np.concatenate([[0], np.cumsum(h)])
    # ps[k] = sum of h[0..k-1]
    
    F = np.zeros(n - 1)
    B = np.zeros(n - 1)
    for m in range(1, n):
        F[m-1] = ps[n-m] - ac[m]  # sum of h[0..n-1-m] minus overlap
        B[m-1] = (ps[n] - ps[m]) - ac[m]  # sum of h[m..n-1] minus overlap
    return F, B
