        top_idx = np.argsort(c_vals)[-top_k:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
        for idx in top_idx:
            lag = idx - (n-1)
            cv = c_vals[idx]
            g = _lag_grad(h_curr, n, lag)
            cons.append(cv + g @ (h_var - h_curr) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = project_feasible(h_var.value, n) if h_var.value is not None else h_curr
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        h_curr = h_new
        if step < 1e-10:
            break
    return best_h, best_c

def smooth_max_grad(h, n, alpha):
    """Compute gradient of smooth max of f_k(h)."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Softmax weights
    shifted = alpha * (c - c.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    n_lags = len(c)
    for k in range(n_lags):
        if weights[k] > 1e-10:
            lag = k - (n-1)
            grad += weights[k] * _lag_grad(h, n, lag)
    return grad

def projected_gradient(h, n, deadline, alpha=10000, lr_init=0.001):

    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    lr = lr_init
    
    for it in range(5000):
        if time.time() > deadline - 1:
            break
        g = smooth_max_grad(h_curr, n, alpha)
        # Projected gradient step
        h_new = h_curr - lr * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            lr = min(lr * 1.1, 0.01)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break
        h_curr = h_new
    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000):
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def obj_grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            e = np.exp(s - mx)
            weights = e / e.sum()
        else:
            e = np.exp(s)
            weights = e / e.sum()
        grad = np.zeros(n)
        for k in range(len(c)):
            if weights[k] > 1e-10:
                lag = k - (n-1)
                grad += weights[k] * _lag_grad(x, n, lag)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n: return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += 1.0 - h[:n-m]
    else:
        grad[:n-m] += 1.0 - h[m:]
        grad[m:] -= h[:n-m]
    return grad

def smooth_max_val(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    e = np.exp(s - mx)
    return m + (mx + np.log(e.sum())) / alpha

def smooth_max_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    e = np.exp(s - mx)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-10:
            grad += w[k] * _lag_grad(x, n, k - (n - 1))
    return grad

def cp_optimize(h, n, deadline, top_k=10):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for _ in range(150):
        if time.time() > deadline - 2:
            break
        c = compute_all_lags(h_curr, n)
        top = np.argsort(c)[-top_k:]
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1]
        for idx in top:
            lag = idx - (n - 1)
            g = _lag_grad(h_curr, n, lag)
            cons.append(c[idx] + g @ (hv - h_curr) <= tv)
        prob = cp.Problem(cp.Minimize(tv), cons)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = project_feasible(hv.value, n) if hv.value is not None else h_curr
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
    return best_h, best_c

def slsqp_opt(h, n, alpha, maxiter=2000):
    def obj(x):
        return smooth_max_val(x, n, alpha)
    def jac(x):
        return smooth_max_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.004, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            ks = int(np.argmax(c))
            lag = ks - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            stride = max(1, (n - m) // 250)
            for i in range(0, n - m, stride):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-6:
                    dt = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if dt < 1e-8:
                        continue
                    h[i] -= dt
                    h[j] += dt
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                    else:
                        h[i] += dt
                        h[j] -= dt
                elif h[j] > h[i] + 1e-6:
                    dt = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if dt < 1e-8:
                        continue
                    h[i] += dt
                    h[j] -= dt
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                    else:
                        h[i] -= dt
                        h[j] += dt
    return h, best_c

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def pipeline(h0, n, deadline, rng):
    h = project_feasible(h0.copy(), n)
    bh = h.copy()
    bc = compute_c5(h, n)
    if time.time() < deadline - 20:
        cp_dl = time.time() + min(50, (deadline - time.time()) * 0.35)
        h2, c2 = cp_optimize(bh, n, cp_dl)
        if c2 < bc:
            bc, bh = c2, h2
    if time.time() < deadline - 12:
        for a in [5000, 50000, 200000]:
            if time.time() > deadline - 10:
                break
            h2 = slsqp_opt(bh, n, a, 2500)
            c2 = compute_c5(h2, n)
            if c2 < bc:
                bc, bh = c2, h2
    if time.time() < deadline - 3:
        bh, bc = local_search(bh, n, deadline - 1)
    return bh, bc

def binary_restart(h, n, rng):
    """Round to binary and randomize slightly."""
    h_bin = np.round(h)
    n_target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > n_target:
        ones = np.where(h_bin == 1)[0]
        to_remove = rng.choice(ones, n_ones - n_target, replace=False)
        h_bin[to_remove] = 0
    elif n_ones < n_target:
        zeros = np.where(h_bin == 0)[0]
        to_add = rng.choice(zeros, n_target - n_ones, replace=False)
        h_bin[to_add] = 1
    return project_feasible(h_bin + rng.normal(0, 0.05, n), n)
