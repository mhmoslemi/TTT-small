import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def greedy_best_improve(h, n, max_rounds=80, deadline=None):
    dx = 2.0 / n
    cur_val = float(np.correlate(h, 1 - h, mode='full').max())
    for _ in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_d = 1e-14
        best_i, best_j = -1, -1
        for i in zeros:
            for j in ones:
                h[int(i)], h[int(j)] = 1.0, 0.0
                v = float(np.correlate(h, 1 - h, mode='full').max())
                h[int(i)], h[int(j)] = 0.0, 1.0
                d = cur_val - v
                if d > best_d:
                    best_d = d
                    best_i, best_j = int(i), int(j)
        if best_i < 0:
            break
        h[best_i], h[best_j] = 1.0, 0.0
        cur_val -= best_d
    return h, cur_val * dx

def sa_search(h, n, deadline, seed=42, t0=0.01, t1=0.00005, max_iters=None):

            else:

def slsqp_smooth(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        lse = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return m + lse
    def cons_f(x):
        return x.sum() - n / 2.0
    def cons_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

            best_h = hp.copy()
    return best_h, best_c
