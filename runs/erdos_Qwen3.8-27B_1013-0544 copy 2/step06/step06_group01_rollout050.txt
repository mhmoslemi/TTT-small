def project_onto_set(h, target_sum):
    """Project h onto {x ∈ [0,1]^n : sum(x) = target_sum}"""
    n = len(h)
    lo, hi = -1.0, 1.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        if diff > 0:
            mask = (h_proj < 1) & (h_proj > 0)
            if mask.sum() > 0:
                h_proj[mask] += diff / mask.sum()
                h_proj = np.clip(h_proj, 0, 1)
        else:
            mask = (h_proj > 0) & (h_proj < 1)
            if mask.sum() > 0:
                h_proj[mask] += diff / mask.sum()
                h_proj = np.clip(h_proj, 0, 1)
    return h_proj

    best_c = compute_c5(h, n)
    lo = -h[p]
    hi = 1.0 - h[p]
    if hi - lo < 1e-14:
        return h, best_c
    c_full = np.correlate(h, 1.0 - h, mode='full')
    um = 0.0
    k_lo = max(0, min(p, n - 1 - p))
    k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
    if k_lo > 0:
        um = float(np.max(c_full[:k_lo]))
    if k_hi < 2 * n - 2:
        um = max(um, float(np.max(c_full[k_hi + 1:])))
    for _ in range(40):
        mid = (lo + hi) / 2.0
        c_mid = _eval_c_at_delta(h, n, p, mid, k_lo, k_hi, c_full)
        q = mid / 4.0
        c_q = _eval_c_at_delta(h, n, p, q, k_lo, k_hi, c_full)
        if c_mid < c_q:
            hi = mid
        else:
            lo = max(lo, q * 2.0)
    deltas = np.linspace(lo, hi, 30)
    best_val = um
    for d in deltas:
        if abs(d) < 1e-14:
            val = c_full[n - 1]
        else:
            val = _eval_c_at_delta(h, n, p, d, k_lo, k_hi, c_full)
        val = max(um, val)
        if val < best_val:
            best_val = val
            best_d = d
    if abs(best_d) > 1e-13:
        h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
    return h, best_val * (2.0 / n)

def _eval_c_at_delta(h, n, p, d, k_lo, k_hi, c_full):
    if abs(d) < 1e-15:
        return c_full[n - 1]
    ks = np.arange(k_lo, k_hi + 1)
    s = np.zeros(len(ks))
    fwd = (ks >= p) & (ks <= p + n - 1)
    bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
    if np.any(fwd):
        s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
    if np.any(bwd):
        s[bwd] -= h[p + ks[bwd] - (n - 1)]
    new_c = c_full[ks] + d * s
    ci = (ks == (n - 1))
    if np.any(ci):
        new_c[ci] = c_full[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
    return float(np.max(new_c))

def cd_pass(h, n, deadline, rng, max_passes=200):
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = rng.permutation(n)
        for p in indices:
            h2, c2 = coords_1d(h, n, p, deadline)
            if c2 < best_c - 1e-15:
                h = h2
                best_c = c2
                improved = True
            else:
                h = h2
        if not improved:
            break
    return h, best_c

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _obj_slsqp(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def slsqp_polish(h0, n, alpha=100000, maxiter=2000):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h = project_onto_set(res.x, n / 2.0)
    except Exception:
        pass
    return h

def optimize_from(h0, n, deadline, rng):
    h = project_onto_set(h0, n / 2.0)
    best_c = compute_c5(h, n)
    t1 = min(time.time() + 60, deadline - 80)
    if time.time() < t1:
        h, c = fista(h, n, t1, alpha=300)
        if c < best_c:
            h, best_c = h, c
    if time.time() < deadline - 50:
        h = slsqp_polish(h, n, alpha=100000, maxiter=2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    t2 = min(time.time() + 50, deadline - 30)
    if time.time() < t2:
        h, c = cd_pass(h, n, t2, rng, max_passes=100)
        if c < best_c:
            h, best_c = h, c
    if time.time() < deadline - 20:
        h = slsqp_polish(h, n, alpha=300000, maxiter=2000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
    if time.time() < deadline - 10:
        h, c = cd_pass(h, n, deadline - 8, rng, max_passes=80)
        if c < best_c:
            h, best_c = h, c
    h = project_onto_set(h, n / 2.0)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    h_best = project_onto_set(h_init, n / 2.0)
    c_best = compute_c5(h_best, n)
    alloc1 = min(300, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_from(h_init, n, t0 + alloc1, rng)
    if c_opt < c_best:
        h_best, c_best = h_opt, c_opt
    # Multi-restart
    remaining = deadline - time.time() - 50
    if remaining > 80:
        scales = [0.02, 0.04, 0.06, 0.08, 0.10, 0.15]
        for r, scale in enumerate(scales):
            if time.time() > deadline - 45:
                break
            h_pert = h_best + rng.normal(0, scale, len(h_best))
            h_pert = project_onto_set(h_pert, len(h_best) / 2.0)
            alloc = min(50, (deadline - time.time() - 40) / max(1, len(scales) - r))
            if alloc < 20:
                break
            h_opt, c_opt = optimize_from(h_pert, n, time.time() + alloc, rng)
            if c_opt < c_best:
                h_best, c_best = h_opt, c_opt
    # Try different n
    if time.time() < deadline - 100:
        for n_try in [250, 400, 200]:
            if time.time() > deadline - 90:
                break
            x_old = np.linspace(0, 2, len(h_best))
            x_new = np.linspace(0, 2, n_try)
            h_new = project_onto_set(np.interp(x_new, x_old, h_best), n_try / 2.0)
            alloc = min(40, deadline - time.time() - 80)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_from(h_new, n_try, time.time() + alloc, rng)
            if c_opt < c_best:
                h_best, c_best = h_opt, c_opt
    # Final polish
    if time.time() < deadline - 15:
        h = slsqp_polish(h_best, len(h_best), alpha=500000, maxiter=3000)
        c = compute_c5(h, len(h))
        if c < c_best:
            h_best, c_best = h, c
    if time.time() < deadline - 5:
        h, c = cd_pass(h_best, len(h_best), deadline - 3, rng, max_passes=50)
        if c < c_best:
            h_best, c_best = h, c
    h_best = project_onto_set(h_best, len(h_best) / 2.0)
    c_best = compute_c5(h_best, len(h_best))
    return h_best, c_best, len(h_best)
