    lags = np.arange(-(n-1), n)

def simulated_annealing(h, n, deadline, seed=42):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = c.max()
    best_h = h.copy()
    best_c = cur_max
    best_val = cur_max * dx
    
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = 3.0  # Initial temperature
    T_min = 0.01
    cooling = 0.9995
    
    while time.time() < deadline and T > T_min:
        # Propose swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        lags = np.arange(-(n-1), n)
        il = i + lags
        jl = j + lags
        il_n = i - lags
        jl_n = j - lags
        
        delta = np.zeros(2*n-1)
        m1 = (il >= 0) & (il < n)
        m2 = (il_n >= 0) & (il_n < n)
        m3 = (jl >= 0) & (jl < n)
        m4 = (jl_n >= 0) & (jl_n < n)
        delta[m1] += h[il[m1]]
        delta[m2] += h[il_n[m2]]
        delta[m3] -= h[jl[m3]]
        delta[m4] -= h[jl_n[m4]]
        
        new_c = c + delta
        new_max = new_c.max()
        
        if new_max <= cur_max:
            # Accept
            h[i], h[j] = h[j], h[i]
            c = new_c
            cur_max = new_max
            # Update ones/zeros
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            ones = np.append(ones, j)
            if cur_max < best_c:
                best_c = cur_max
                best_val = cur_max * dx
                best_h = h.copy()
        else:
            # Maybe accept
            delta_log = -(new_max - cur_max) / T
            if np.log(rng.random()) < delta_log:
                h[i], h[j] = h[j], h[i]
                c = new_c
                cur_max = new_max
                ones = ones[ones != i]
                zeros = zeros[zeros != j]
                zeros = np.append(zeros, i)
                ones = np.append(ones, j)
        
        T *= cooling
    
    return best_h, best_val

    rng = np.random.RandomState(seed)
    h = binarize(h, n)
    lags = np.arange(-(n-1), n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()
    dx = 2.0 / n
    best_val = cur_max * dx
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        found = False
        # Try up to 100 random swaps per round
        for _ in range(100):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = make_swap_delta(h, i, j, n, lags)
            new_max = (c + delta).max()
            if new_max < cur_max:
                h[i], h[j] = h[j], h[i]
                c += delta
                cur_max = new_max
                found = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = h.copy()
                    best_val = cur_max * dx
                break
        if not found:
            break
    return best_h, best_val

def iterated_local_search(h, n, deadline, seed=42):
    """ILS: local search + perturbation + local search."""
    rng = np.random.RandomState(seed)
    h = binarize(h, n)
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    
    while time.time() < deadline:
        # Local search
        t_sub = min(time.time() + 30, deadline - 10)
        hb, cb = greedy_local_search(h, n, t_sub, seed=seed)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        h = hb.copy()

        for _ in range(n_pert):

            zi = np.where(h == 0)[0]
            if len(oi) > 0 and len(zi) > 0:
                h[rng.choice(oi)] = 0.0
                h[rng.choice(zi)] = 1.0
        seed += 1
    
    return best_h, best_c

def continuous_optimize(h, n, deadline, alpha=5000):
    """SLSQP with soft-max objective."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        am = alpha * c
        am -= am.max()
        return c.max() + np.log(np.exp(am).sum()) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        am = alpha * c
        am -= am.max()
        w = np.exp(am)
        w /= w.sum()
        # Gradient: sum_k w_k * grad c_k
        # grad c_k[i] = (1-x[i-s]) - x[i+s] where s = k-(n-1)
        # This is O(n * 2n) = O(n^2)
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-10:
                continue
            s = k - (n-1)
            # x[i+s] term: contributes -x[i+s] to grad[i]
            # (1-x[i-s]) term: contributes (1-x[i-s]) to grad[i]
            # For valid i: 0 <= i < n, 0 <= i+s < n, 0 <= i-s < n
            i_vals = np.arange(n)
            # -x[i+s] part
            valid_plus = (i_vals + s >= 0) & (i_vals + s < n)
            grad[valid_plus] += w[k] * (-x[i_vals[valid_plus] + s])
            # (1-x[i-s]) part
            valid_minus = (i_vals - s >= 0) & (i_vals - s < n)
            grad[valid_minus] += w[k] * (1 - x[i_vals[valid_minus] - s])
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_grad(x, w, n):
    """Compute gradient of soft-max objective."""
    grad = np.zeros(n)
    # w[k] is the weight for lag k, s_k = k - (n-1)
    # grad[i] = sum_k w[k] * [(1-x[i-s_k])_valid - x[i+s_k]_valid]

    # For each i, the valid s_k values are those where 0<=i-s_k<n and 0<=i+s_k<n
    # i.e., s_k in [max(i-n+1, -(n-1)), min(i, n-1)] for the first term
    # and s_k in [max(-(i), -(n-1)), min(n-1-i, n-1)] for the second term

        s = k - (n-1)
        # First term: (1-x[i-s]) for valid i-s

    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def make_delta(h, i, j, n, lags):
    il = i + lags
    jl = j + lags
    iln = i - lags
    jln = j - lags
    d = np.zeros(2 * n - 1)
    m1 = (il >= 0) & (il < n)
    m2 = (iln >= 0) & (iln < n)
    m3 = (jl >= 0) & (jl < n)
    m4 = (jln >= 0) & (jln < n)
    d[m1] += h[il[m1]]
    d[m2] += h[iln[m2]]
    d[m3] -= h[jl[m3]]
    d[m4] -= h[jln[m4]]
    return d

def greedy_ls(h, n, deadline, seed=42):

    h = binarize(h, n)
    lags = np.arange(-(n-1), n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = c.max()
    best = cur_max
    best_h = h.copy()
    dx = 2.0 / n
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        found = False
        for _ in range(200):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            d = make_delta(h, i, j, n, lags)
            nm = (c + d).max()
            if nm < cur_max:
                h[i], h[j] = h[j], h[i]
                c += d
                cur_max = nm
                found = True
                if cur_max < best:
                    best = cur_max
                    best_h = h.copy()
                break
        if not found:

    return best_h, best * dx

    best = cur_max

    dx = 2.0 / n
    max_iter = 200000
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (it / max_iter)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        d = make_delta(h, i, j, n, lags)
        nc = c + d
        nm = nc.max()
        diff = nm - cur_max
        if diff <= 0 or np.log(rng.random()) < -diff / T:
            h[i], h[j] = h[j], h[i]
            c = nc
            cur_max = nm
            if cur_max < best:
                best = cur_max
                best_h = h.copy()
    return best_h, best * dx

def slsqp_smooth(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        am = alpha * c
        am -= am.max()
        return c.max() + np.log(np.exp(am).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        am = alpha * c
        am -= am.max()
        w = np.exp(am)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            a0 = max(0, s)
            a1 = min(n, n + s)
            if a0 < a1:
                g[a0:a1] += wk * (1.0 - x[a0 - s:a1 - s])
            b0 = max(0, -s)
            b1 = min(n, n - s)
            if b0 < b1:
                g[b0:b1] += wk * (-x[b0 + s:b1 + s])
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
