import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        w[idx] = w_top
    else:
        c_shift = alpha * (c - c.max())
        mx = c_shift.max()
        w = np.exp(c_shift - mx)
        w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m]
        if k >= 0:
            nv = n - k
            grad[k:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[k:n])
        else:
            kk = -k
            nv = n - kk
            grad[0:nv] += wm * (1.0 - h[kk:n])
            grad[kk:n] += wm * (-h[0:nv])
    return grad

def smooth_max_obj(x, n, alpha, top_k=None):
    c = compute_c5_vec(x, n)
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(-c, top_k)[:top_k]
        ct = c[idx]
        sh = alpha * (ct - ct.max())
        mx = sh.max()
        return float(ct.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    else:
        sh = alpha * (c - c.max())
        mx = sh.max()
        return float(c.max() + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
