    return np.ones(n)

def slsqp_opt(h, n, lags, alpha, maxiter, deadline):
    if time.time() > deadline - 2:
        return h.copy()
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _sj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, lags, alpha), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_descent(h, n, deadline, K_start=3, K_max=15):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    for t in range(5000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        K = min(K_start + t // 100, K_max)
        top_idx = np.argsort(c)[-K:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += lag_grad(h_curr, n, k - (n - 1))
        grad /= (K * (2.0 / n))
        if t < 80:
            step = 0.04
        elif t < 300:
            step = 0.015
        else:
            step = 0.008 * (150.0 / (t + 1.0))
        h_new = project_feasible(h_curr - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
    return best_h, best_c

def sa_local_search(h, n, deadline, T0=0.003, T_end=0.00005, max_iters=80000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    rng = np.random.RandomState(12345)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max_iters
        T = T0 * (T_end / T0) ** frac
        c = compute_all_lags(h_curr, n)
        c_max = c.max()
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m >= n or m == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j: continue
        else:
            grad = lag_grad(h_curr, n, lag)
            if rng.random() < 0.7:
                pos_idx = np.argsort(-grad)[:15]
                neg_idx = np.argsort(grad)[:15]
                i = pos_idx[rng.randint(len(pos_idx))]
                j = neg_idx[rng.randint(len(neg_idx))]
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            if i == j: continue
        d = rng.choice([0.001, 0.003, 0.007, 0.015, 0.04])
        direction = 1 if h_curr[i] >= h_curr[j] else -1
        if direction == 1:
            d = min(d, h_curr[i], 1.0 - h_curr[j])
        else:
            d = min(d, h_curr[j], 1.0 - h_curr[i])
        if d < 1e-9: continue
        h_curr[i] -= direction * d
        h_curr[j] += direction * d
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = project_feasible(h_curr, n)
        c_new = compute_c5(h_curr, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_curr.copy()
        elif rng.random() > np.exp(-(c_new - best_c) / T):
            h_curr[i] += direction * d
            h_curr[j] -= direction * d
            h_curr = np.clip(h_curr, 0.0, 1.0)
            h_curr = project_feasible(h_curr, n)
    return best_h, best_c

def greedy_local_search(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        c = compute_all_lags(h_curr, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m >= n or m == 0:
            break
        grad = lag_grad(h_curr, n, lag)
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        for d in deltas:
            if time.time() > deadline: break
            pos_i = np.argmax(grad)
            neg_i = np.argmin(grad)
            if grad[pos_i] < 0 or grad[neg_i] > 0:
                break
            h_curr[pos_i] -= d
            h_curr[neg_i] += d
            h_curr = np.clip(h_curr, 0.0, 1.0)
            h_curr = project_feasible(h_curr, n)
            c_new = compute_c5(h_curr, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                improved = True
            else:
                h_curr[pos_i] += d
                h_curr[neg_i] -= d
                h_curr = np.clip(h_curr, 0.0, 1.0)
                h_curr = project_feasible(h_curr, n)
    return h_curr, best_c

def optimize_one(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 40:
        h_sg, c_sg = subgrad_descent(best_h, n, time.time() + min(50, (deadline - time.time()) * 0.2))
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
    if time.time() < deadline - 30:
        c = compute_all_lags(best_h, n)
        top5 = np.argsort(c)[-5:]
        for alpha in [5000, 50000, 200000, 1000000]:
            if time.time() > deadline - 30: break
            h_new = slsqp_opt(best_h, n, top5, alpha, 4000, deadline)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                c = compute_all_lags(best_h, n)
                top5 = np.argsort(c)[-5:]
    if time.time() < deadline - 15:
        h_new, c_new = greedy_local_search(best_h, n, deadline - 10)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 5:
        h_new, c_new = sa_local_search(best_h, n, deadline - 3, T0=0.002, T_end=0.0001, max_iters=40000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 3:
        c = compute_all_lags(best_h, n)
        top5 = np.argsort(c)[-5:]
        h_new = slsqp_opt(best_h, n, top5, 2000000, 3000, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    except Exception:
        best_h = None
        best_c = float('inf')
        n_init = 300
    if best_h is not None:
        alloc = min(250, (deadline - t0) * 0.3)
        h_opt, c_opt = optimize_one(best_h, n_init, t0 + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    n_candidates = [200, 250, 350]
    for n_try in n_candidates:
        if time.time() > deadline - 50: break
        remaining = deadline - time.time() - 15
        if remaining < 35: break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_one(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None:
        n = len(best_h)
        scales = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]
        for s_val in scales:
            if time.time() > deadline - 25: break
            hp = best_h + rng.normal(0, s_val, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 15) * 0.3)
            if alloc < 12: break
            h_opt, c_opt = optimize_one(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 8:
        n = len(best_h)
        c = compute_all_lags(best_h, n)
        top5 = np.argsort(c)[-5:]
        h_new = slsqp_opt(best_h, n, top5, 5000000, 4000, deadline - 3)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = sa_local_search(best_h, n, deadline - 2, T0=0.001, T_end=0.00002, max_iters=30000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, float(best_c), len(best_h)
