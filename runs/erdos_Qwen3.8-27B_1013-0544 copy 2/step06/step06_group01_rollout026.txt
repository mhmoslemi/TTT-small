    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = c.max()
    w = np.exp(alpha * (c - max_c))
    w_sum = w.sum()
    w_norm = w / w_sum
    
    w_rev = w_norm[::-1]
    
    corr1 = np.correlate(h, w_norm, mode='full')

    smooth_val = max_c + np.log(w_sum) / alpha
    return smooth_val, G

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
