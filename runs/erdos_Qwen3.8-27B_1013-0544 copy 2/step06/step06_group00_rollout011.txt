    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    m = c.max()

    exp_sh = np.exp(shifted - shifted.max())
    w = exp_sh / exp_sh.sum()
    # w has length 2n-1, w[k] corresponds to lag k-(n-1)
    
    # gradient at position j:
    # sum_k w[k] * [(1-h[j-lag]) * I(0<=j-lag<=n-1) - h[j+lag] * I(0<=j+lag<=n-1)]

        lag_max = j

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_sh = np.exp(shifted - sm)
    Z = exp_sh.sum()
    w = exp_sh / Z
    logsum = (sm + np.log(Z)) / alpha
    obj = m + logsum
    # gradient
    grad = compute_grad(h, w, n) * dx
    return obj, grad

def compute_grad(h, w, n):
    """Compute gradient of smooth max of correlation."""
    # w has length 2n-1
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wl = w[lag + (n-1)]
        if wl < 1e-18:
            continue
        # First term: +wl * (1-h[j-lag]) for valid j
        if lag >= 0:
            grad[lag:] += wl * (1 - h[:n-lag])
        else:
            grad[:n+lag] += wl * (1 - h[-lag:])
        # Second term: -wl * h[j+lag] for valid j
        if lag >= 0:
            grad[:n-lag] -= wl * h[lag:]
        else:
            grad[-lag:] -= wl * h[:n+lag]
    return grad

def adam_optimize(h, n, alpha, n_iter, lr, deadline):
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(1, n_iter+1):
        if time.time() > deadline:
            break
        obj, grad = smooth_objective_and_grad(h, n, alpha)
        # Adam update
        m_state = beta1 * m_state + (1-beta1) * grad
        v_state = beta2 * v_state + (1-beta2) * grad**2
        m_hat = m_state / (1 - beta1**t)
        v_hat = v_state / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Handle the linear constraint by projecting
        h_new = project_feasible(h_new, n)
        # Check improvement
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

            break

            if s < 1e-14:

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)

    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_sh = np.exp(shifted - sm)
    Z = exp_sh.sum()
    w = exp_sh / Z
    obj = m + (sm + np.log(Z)) / alpha
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        wl = w[lag + (n - 1)]
        if wl < 1e-16:
            continue
        if lag >= 0:
            grad[lag:] += wl * (1.0 - h[:n - lag])
            grad[:n - lag] -= wl * h[lag:]
        else:
            grad[:n + lag] += wl * (1.0 - h[-lag:])
            grad[-lag:] -= wl * h[:n + lag]
    grad *= dx
    return obj, grad

def adam_opt(h, n, alpha, n_iter, lr0, lr1, deadline, patience=500):
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c = compute_c5(h, n)
    best_h = h.copy()
    best_t = 0
    for t in range(1, n_iter + 1):
        if time.time() > deadline:
            break
        frac = t / n_iter
        lr = lr0 + (lr1 - lr0) * frac
        obj, grad = smooth_obj_grad(h, n, alpha)
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad * grad
        m_hat = m_s / (1 - b1 ** t)
        v_hat = v_s / (1 - b2 ** t)
        h = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c = compute_c5(h, n)
        if c < best_c - 1e-7:
            best_c = c
            best_h = h.copy()
            best_t = t
        elif t - best_t > patience:
            break
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_greedy(h_init, n, deadline, max_sweeps=80):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            for j in ones_list:
                if i == j:
                    continue
                jm = j + m_range
                msk2 = (jm >= 0) & (jm < n)
                delta[msk2] -= 1.0 - h[jm[msk2]]
                jmn = j - m_range
                msk2 = (jmn >= 0) & (jmn < n)
                delta[msk2] += h[jmn[msk2]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
                delta[n - 1] += 2.0
                delta[n - 1 + (j - i)] -= 1.0
                delta[n - 1 + (i - j)] -= 1.0
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def fine_threshold(h, n):
    """Try thresholding at various levels and local binary search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    percentiles = [45, 48, 50, 52, 55]
    for p in percentiles:
        thresh = np.percentile(h, p)
        h_b = (h >= thresh).astype(float)
        if h_b.sum() != n // 2:
            idx = np.argsort(h)[::-1]
            h_b = np.zeros(n)
            h_b[idx[:n // 2]] = 1.0
        c = compute_c5(h_b, n)
        if c < best_c:
            best_c = c
            best_h = h_b.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 12
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Adam with increasing alpha (main workhorse)
    p1 = t0 + 0.60 * total
    h = best_h.copy()
    alphas = [50, 100, 200, 500, 1000, 2500, 5000, 10000, 25000, 50000, 100000]
    for i, alpha in enumerate(alphas):
        if time.time() > p1 - 30:
            break
        remaining = p1 - time.time()
        per_alpha = max(30, remaining / (len(alphas) - i))
        phase_dl = time.time() + per_alpha
        lr0 = max(0.005, 0.05 * (0.7 ** i))
        h, c = adam_opt(h, n, alpha, 100000, lr0, lr0 * 0.1, phase_dl, patience=400)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Multiple restarts with perturbations
    p2 = t0 + 0.82 * total
    n_restarts = 6
    for trial in range(n_restarts):
        if time.time() > p2 - 20:
            break
        scale = 0.005 + 0.01 * trial / n_restarts
        h_pert = best_h + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        sub_dl = p2 - 15
        for alpha in [500, 2000, 10000]:
            if time.time() > sub_dl - 5:
                break
            h_t, c_t = adam_opt(h_pert, n, alpha, 30000, 0.02, 0.002, sub_dl - 5, patience=300)
            if c_t < best_c:
                best_c, best_h = c_t, h_t.copy()
                h_pert = best_h.copy()

    # Phase 3: Binary polish via greedy
    p3 = deadline - 5
    if time.time() < p3:
        hb, cb = binary_greedy(best_h, n, p3, max_sweeps=60)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 4: Fine thresholding
    if time.time() < deadline - 2:
        ht, ct = fine_threshold(best_h, n)
        if ct < best_c:
            best_c, best_h = ct, ht.copy()
            hb2, cb2 = binary_greedy(best_h, n, deadline - 1, max_sweeps=30)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
