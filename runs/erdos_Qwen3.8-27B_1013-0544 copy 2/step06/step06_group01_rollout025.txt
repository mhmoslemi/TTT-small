```python
def min_1d(um, a, b, d_lo, d_hi, S, n_target, penalty):
    # Find d in [d_lo, d_hi] minimizing max(um, max_j(a_j + b_j*d)) + penalty*(S+d-n_target)²
    def f(d):
        vals = a + b * d
        m = max(um, vals.max())
        return m + penalty * (S + d - n_target)**2
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    x1 = d_hi - gr * (d_hi - d_lo)
    x2 = d_lo + gr * (d_hi - d_lo)
    f1, f2 = f(x1), f(x2)
    for _ in range(60):  # ~60 iterations gives ~10^-12 precision
        if f1 < f2:
            d_hi = x2; x2 = x1; f2 = f1
            x1 = d_hi - gr * (d_hi - d_lo)
            f1 = f(x1)
        else:
            d_lo = x1; x1 = x2; f1 = f2
            x2 = d_lo + gr * (d_hi - d_lo)
            f2 = f(x2)
    return (d_lo + d_hi) / 2
```

```python
def slp_step(h, n, top_lags=100):
    """One step of Successive Linear Programming."""
    c_full = np.correlate(h, 1.0 - h, mode='full')
    
    # Select which lags to include (top ones by value)
    if top_lags < 2*n-1:
        idx = np.argsort(c_full)[::-1][:top_lags]
    else:
        idx = np.arange(2*n-1)
    
    m_act = len(idx)
    A_ub = np.zeros((m_act, n + 1))
    b_ub = np.zeros(m_act)
    
    for row, m in enumerate(idx):
        k = m - (n - 1)
        grad = np.zeros(n)
        # Forward: h[j] appears as h[i] with i=j, need j+k in [0,n-1]
        j_fwd_lo = max(0, -k)
        j_fwd_hi = min(n-1, n-1-k)
        if j_fwd_lo <= j_fwd_hi:
            j_fwd = np.arange(j_fwd_lo, j_fwd_hi+1)
            grad[j_fwd] += (1.0 - h[j_fwd + k])
        # Backward: h[j] appears in (1-h[i+k]) with i+k=j, i=j-k, need j-k in [0,n-1]
        j_bwd_lo = max(0, k)
        j_bwd_hi = min(n-1, n-1+k)
        if j_bwd_lo <= j_bwd_hi:
            j_bwd = np.arange(j_bwd_lo, j_bwd_hi+1)
            grad[j_bwd] -= h[j_bwd - k]
        
        A_ub[row, :n] = grad
        A_ub[row, n] = -1.0
        b_ub[row] = grad @ h - c_full[m]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return h.copy()
```

```python
def golden_min(f, a, b, tol=1e-10, max_iter=50):
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2
```

```python
def compute_gradients(h, n):

    Returns array of shape (2n-1, n) where grad[m, j] = ∂corr[m]/∂h[j]

        if j_lo2 <= j_hi2:

            grads[m, js2] -= h[js2 + ell]
```

```python
def slp_solve(h, n, max_lags=None):
    """Solve the linearized subproblem. Returns proposed h and t."""
    c_full = np.correlate(h, 1.0 - h, mode='full')
    n_full = 2 * n - 1
    
    if max_lags is not None and max_lags < n_full:
        # Only include top lags
        idx = np.argsort(c_full)[::-1][:max_lags]
    else:
        idx = np.arange(n_full)

    b = np.zeros(m_act)

        ell = m - (n - 1)
        # grad[m, j] = (1-h[j-ell]) * 1[0≤j-ell<n] - h[j+ell] * 1[0≤j+ell<n]
        row_vec = np.zeros(n)
```

```python
def cd_fast(h, n, deadline, rng, max_passes=300, penalty=3.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    sumh = float(h.sum())
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for p in indices:
            dlo, dhi = -h[p], 1.0 - h[p]
            if dhi - dlo < 1e-14:
                continue
            mlo = max(0, p)
            mhi = min(2 * n - 2, 2 * n - 2 - p)
            um = 0.0
            if mlo > 0:
                um = max(um, float(np.max(c[:mlo])))
            if mhi < 2 * n - 2:
                um = max(um, float(np.max(c[mhi + 1:])))
            ks = np.arange(mlo, mhi + 1)
            kvals = ks - (n - 1)
            s = np.zeros(len(ks))
            fwd = (p + kvals >= 0) & (p + kvals < n)
            bwd = (p - kvals >= 0) & (p - kvals < n)
            if np.any(fwd):
                s[fwd] += (1.0 - h[p + kvals[fwd]])
            if np.any(bwd):
                s[bwd] -= h[p - kvals[bwd]]
            c_aff = c[ks]
            S = sumh
            tgt = n / 2.0

            def f_of_d(d):
                return max(um, float(np.max(c_aff + d * s))) + penalty * (S + d - tgt) ** 2

            res = minimize_scalar(f_of_d, bounds=(dlo, dhi), method='bounded',

                    c[ks] += d * s
                    c[n - 1] -= d * d
                    h[p] = np.clip(h[p] + d, 0.0, 1.0)
                    sumh = float(h.sum())
                    improved = True

        h_f, c_f = cd_fast(best_h, n_f, min(time.time() + 30, deadline - 4), rng, max_passes=200, penalty=0.1)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 1,
                                     [0.002, 0.001, 0.0005, 0.0002])
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
