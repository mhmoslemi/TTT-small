    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-8:
                    delta = min(d, (h[i] - h[j]) * 0.4)
                    h[i] -= delta
                    h[j] += delta
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-8:
                    delta = min(d, (h[j] - h[i]) * 0.4)
                    h[i] += delta
                    h[j] -= delta
                    cn = compute_c5(h, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() < deadline - 20:
        cd_end = time.time() + min(120, (deadline - time.time()) * 0.5)
        h_cd, c_cd = coord_descent(best_h, n, cd_end)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd
    if time.time() < deadline - 10:
        for al in [20000, 200000]:
            if time.time() > deadline - 8:
                break
            h_s = slsqp_polish(best_h, n, deadline, alpha=al, maxiter=2000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                best_c, best_h = c_s, h_s
    if time.time() < deadline - 4:
        h_cd2, c_cd2 = coord_descent(best_h, n, min(deadline - 2, time.time() + 30))
        if c_cd2 < best_c:
            best_c, best_h = c_cd2, h_cd2
    if time.time() < deadline - 1:
        h_ls, c_ls = local_search(best_h, n, deadline - 0.5)
        if c_ls < best_c:
            best_c, best_h = c_ls, h_ls
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        n0 = len(h_init)
        alloc = min(400, (deadline - t0) * 0.45)
        h_opt, c_opt = optimize_at_n(h_init, n0, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 400, 150, 250]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(100, remaining * 0.3)
        if best_h is not None:
            x0 = np.linspace(0, 2, len(best_h))
            x1 = np.linspace(0, 2, n_try)
            h_start = np.interp(x1, x0, best_h)
        elif h_init is not None:
            x0 = np.linspace(0, 2, len(h_init))
            x1 = np.linspace(0, 2, n_try)
            h_start = np.interp(x1, x0, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        remaining = deadline - time.time() - 20
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.02 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        h_final = slsqp_polish(best_h, n, deadline - 2, alpha=500000, maxiter=5000)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c, best_h = c_final, h_final
        best_h, best_c = local_search(best_h, n, deadline - 0.5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = 300
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
