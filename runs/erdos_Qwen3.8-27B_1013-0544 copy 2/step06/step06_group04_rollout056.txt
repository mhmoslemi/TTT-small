def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 2, n - 1 + i) + 1)
    delta[m2] -= h[n - 1 + i - m2]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 2, n - 1 + j) + 1)
    delta[m2] += h[n - 1 + j - m2]
    delta[n - 1] -= 2.0

        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
