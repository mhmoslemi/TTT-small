def project_sum_box(x, target_sum):
    """Project x onto [0,1]^n ∩ {sum = target_sum}."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

        best_h, best_c = h_opt, c_opt
    # Try other resolutions
    n_list = [200, 250, 400]
    for n_try in n_list:
        if best_h is not None and n_try == len(best_h):
            continue
        if time.time() > deadline - 80:
            break
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(150, remaining * 0.35)
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc, seed + n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Final polishing on best
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        # One more SLSQP with high alpha
        if time.time() < deadline - 25:
            hs = slsqp_refine(best_h, n, 200000, deadline - 20)
            cs = compute_c5(hs, n)
            if cs < best_c:
                best_c = cs
                best_h = hs
        # Coordinate descent
        if time.time() < deadline - 5:
            hc, cc = coord_descent(best_h, n, deadline - 2)
            if cc < best_c:
                best_c = cc
                best_h = hc
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
