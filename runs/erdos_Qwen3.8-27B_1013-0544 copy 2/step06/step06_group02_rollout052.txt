            t = h.sum()
            if t < 1e-15:

def grad_lag_vec(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += dx * (1.0 - h[k:])
        g[k:] += -dx * h[:nv]
    else:
        kk = -k
        nv = n - kk
        g[kk:] += dx * (1.0 - h[:nv])
        g[:nv] += -dx * h[kk:]
    return g

    corr = compute_corr(h, n)
    mx = corr.max()
    # Soft weights
    w = np.exp(beta * (corr - mx))
    # Keep only top m for efficiency
    top_idx = np.argsort(corr)[-m:]

             0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            g = active_set_grad(h, n, m=15, beta=50.0)
            order_inc = np.argsort(g)
            order_dec = np.argsort(-g)
            found = False
            for ii in order_inc[:n_try]:
                if found:
                    break
                for jj in order_dec[:n_try]:
                    if ii == jj:
                        continue
                    if h[ii] + step <= 1.0 and h[jj] - step >= 0.0:
                        old_i, old_j = h[ii], h[jj]
                        h[ii] += step; h[jj] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True; found = True
                            break
                        else:
                            h[ii] = old_i; h[jj] = old_j
                    if h[ii] - step >= 0.0 and h[jj] + step <= 1.0:
                        old_i, old_j = h[ii], h[jj]
                        h[ii] -= step; h[jj] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True; found = True
                            break
                        else:
                            h[ii] = old_i; h[jj] = old_j
    return h, best_c

def broad_pair_search(h, n, deadline, steps=None, radius=40):
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005]
    h = project(h, n)
    best_c = compute_c5(h, n)
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 300
        h = np.full(n, 0.5)

    h = project(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: Momentum gradient descent
    if time.time() < deadline - 400:
        h, c = momentum_grad_descent(h, n, min(t0 + 160, deadline - 400),
                                      m=15, beta=50.0, lr_init=0.005, rng=rng)
        if c < best_c:
            best_c = c; best_h = h

    # Phase 2: Gradient-guided pair search (main workhorse)
    if time.time() < deadline - 250:
        h, c = grad_guided_pairs(h, n, min(t0 + 550, deadline - 250),
                                  base_step=0.004, n_try=20)
        if c < best_c:
            best_c = c; best_h = h

    # Phase 3: Broad pair search with moderate radius
    if time.time() < deadline - 150:
        nf = len(best_h)
        h, c = broad_pair_search(best_h, nf, min(t0 + 750, deadline - 150),
                                  steps=[0.003, 0.001, 0.0005], radius=50)
        if c < best_c:
            best_c = c; best_h = h

    # Phase 4: Try alternative n values
    if time.time() < deadline - 100:
        for n_try in [200, 250, 350]:
            if time.time() > deadline - 80:
                break
            xs = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_try)
            h_interp = np.interp(xn, xs, best_h)
            h_interp = project(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 40)
            if alloc < 20:
                break
            ht, ct = momentum_grad_descent(h_interp, n_try, time.time() + alloc * 0.6,
                                           m=12, beta=40.0, rng=rng)
            if time.time() < deadline - 30:
                ht, ct = grad_guided_pairs(ht, n_try, min(time.time() + 25, deadline - 30),
                                            base_step=0.003, n_try=15)
            if ct < best_c:
                best_c = ct; best_h = ht

    # Phase 5: Final fine refinement
    nf = len(best_h)
    if time.time() < deadline - 20:
        h, c = grad_guided_pairs(best_h, nf, min(t0 + 1030, deadline - 20),
                                  base_step=0.002, n_try=15)
        if c < best_c:
            best_c = c; best_h = h

    # Phase 6: Very fine pair search
    if time.time() < deadline - 3:
        h, c = broad_pair_search(best_h, nf, deadline - 1,
                                  steps=[0.0002, 0.0001, 0.00005], radius=30)
        if c < best_c:
            best_c = c; best_h = h

    best_h = project(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
