def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(200):
        s=h.sum()

        if abs(d)<1e-12: break
        if d>0:
            room=1.0-h
            tot=room.sum()
            if tot<=1e-15: break
            h += (d/tot)*room
        else:
            tot=s
            if tot<=1e-15: break

def compute_corr(h,n):

def project_feasible(h,n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(200):
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (d / tot) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = target - float(h.sum())
        if abs(d) < 1e-12:
            break
        if d > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
            if idx.size == 0:
                break
            j = int(idx[np.argmax(1.0 - h[idx])])

                break
            j = int(idx[np.argmin(h[idx])])

            if sub <= 0.0:

    return h

def threshold_binary_from_floats(xf,n,thr):
    h = np.asarray(xf, dtype=float)
    x = (h > thr).astype(float)
    target = int(n // 2)
    cnt = int(np.sum(x))
    if cnt > target:
        ones = np.where(x == 1.0)[0]
        if ones.size:
            order = ones[np.argsort(h[ones], kind='stable')]
            m = min(cnt-target, ones.size)
            x[order[:m]] = 0.0
    elif cnt < target:
        zeros = np.where(x == 0.0)[0]
        if zeros.size:
            order = zeros[np.argsort(-h[zeros], kind='stable')]
            m = min(target-cnt, zeros.size)
            x[order[:m]] = 1.0
    return x

import numpy as np

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)) * (2.0 / float(n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(200):
        s = float(h.sum())
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (d / tot) * room
        else:
            tot = s
            if tot <= 1e-15:
                break
            h += (d / tot) * h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = target - float(h.sum())
        if abs(d) < 1e-12:

        if d > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]

            j = int(idx[np.argmax(1.0 - h[idx])])

                break
            h[j] += add
        else:

def threshold_binary_from_floats(xf, n, thr):

    x = (xf > thr).astype(float)
    target = int(n // 2)

            x[order[:m]] = 0.0
    elif cnt < target:

def interp_parent(parent, n):
    if parent is None:
        return None
    parent = np.asarray(parent, dtype=float)
    if parent.size < 2:
        return None
    xp = np.linspace(0.0, 2.0, parent.size)
    xq = np.linspace(0.0, 2.0, n)
    return np.interp(xq, xp, parent)


def obj_fun(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    if alpha <= 0.0:
        return m * dx
    e = np.exp(alpha * (c - m))
    z = float(e.sum())
    if z <= 0.0:
        return m * dx
    return dx * (m + math.log(z) / alpha)


def obj_jac(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    if alpha <= 0.0:
        p = np.full(c.size, 1.0 / c.size)
    else:
        e = np.exp(alpha * (c - m))
        z = float(e.sum())
        if z <= 0.0:
            p = np.zeros(c.size)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / z
    ones = np.ones(n, dtype=float)
    A = np.convolve(ones, p, mode="full")
    A = A[n - 1:n + n - 1]
    v = p + p[::-1]
    C = np.convolve(x, v, mode="full")
    C = C[n - 1:n + n - 1]
    return dx * (A - C)


def sum_con_fun(x, n):
    return float(x.sum()) - 0.5 * float(n)


def sum_con_jac(x, n):
    return np.ones_like(x, dtype=float)


def slsqp_alpha(h, n, alpha, maxiter, ftol=1e-11):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_con_fun, "jac": sum_con_jac, "args": (n,)}]
    try:
        res = minimize(
            obj_fun, h0, method="SLSQP", jac=obj_jac, args=(n, alpha),
            bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": ftol, "iprint": 0}
        )
        if res is not None and hasattr(res, "x"):
            x = np.asarray(res.x, dtype=float)
            if x.size == n and np.all(np.isfinite(x)):
                return project_feasible(x, n)
    except Exception:
        pass
    return h0


def grad_weights(h, n, w):
    dx = 2.0 / float(n)
    ones = np.ones(n, dtype=float)
    A = np.convolve(ones, w, mode="full")
    A = A[n - 1:n + n - 1]
    v = w + w[::-1]
    C = np.convolve(h, v, mode="full")
    C = C[n - 1:n + n - 1]
    return dx * (A - C)


def subgrad_polish(h, n, maxiter=200, beta=300.0, start_step=0.04, deadline=None):

        x[j] = 0.0
        accept = False
        if val < cur - 1e-14:
            accept = True
        elif val <= cur + 0.03 * temp:
            accept = (rng.rand() < math.exp((cur - val) / temp))
        if accept:
            x[i] = 0.0
            x[j] = 1.0
            cur = val
            if cur < best_c:
                best_c = cur
                best_h = x.copy()
            ones = np.where(x > 0.5)[0]
            zeros = np.where(x <= 0.5)[0]
            temp = max(1e-6, temp * 0.9995)
        it += 1
    return best_h, best_c


def make_candidate(n, mode, rng, parent):
    target = int(n // 2)
    hp = interp_parent(parent, n)
    if mode == 0:
        if hp is not None:
            return hp + rng.normal(0.0, 0.04, n)
        return rng.uniform(0.0, 1.0, n)
    if mode == 1:
        if hp is not None:
            return threshold_binary_from_floats(hp, n, 0.5)
        return threshold_binary_from_floats(rng.uniform(0.0, 1.0, n), n, 0.5)
    if mode == 2:
        x = np.zeros(n, dtype=float)
        idx = rng.permutation(n)
        x[idx[:target]] = 1.0
        return x
    if mode == 3:
        if hp is not None:
            return 1.0 - threshold_binary_from_floats(hp, n, 0.5)
        return 1.0 - threshold_binary_from_floats(rng.uniform(0.0, 1.0, n), n, 0.5)
    if mode == 4:
        return rng.beta(0.5, 0.5, n)
    if mode == 5:
        f = float(FREQS[int(rng.randint(len(FREQS)))])
        phase = rng.uniform(0.0, 2.0 * math.pi)
        t = np.sin(2.0 * math.pi * f * np.arange(n, dtype=float) / float(n) + phase)
        return threshold_binary_from_floats(t, n, 0.0)
    if mode == 6:
        f1 = float(FREQS[int(rng.randint(len(FREQS)))])
        f2 = float(FREQS[int(rng.randint(len(FREQS)))])
        p1 = rng.uniform(0.0, 2.0 * math.pi)
        p2 = rng.uniform(0.0, 2.0 * math.pi)
        arr = np.arange(n, dtype=float) / float(n)
        h = 0.5 + 0.40 * np.sin(2.0 * math.pi * f1 * arr + p1) + 0.25 * np.sin(2.0 * math.pi * f2 * arr + p2)
        return np.clip(h, 0.0, 1.0)
    if mode == 7:
        invphi = 0.6180339887498949
        t = (np.arange(n, dtype=float) * invphi) % 1.0
        return threshold_binary_from_floats(t, n, 0.5)
    if mode == 8:
        x = ((np.arange(n) % 5) < 2).astype(float)
        return threshold_binary_from_floats(x, n, 0.5)
    if mode == 9:
        x = ((np.arange(n) % 4) < 2).astype(float)
        return threshold_binary_from_floats(x, n, 0.5)
    if mode == 10:
        if hp is not None:
            return hp + rng.normal(0.0, 0.10, n)
        return rng.uniform(0.0, 1.0, n)
    if mode == 11:
        f1 = float(FREQS2[int(rng.randint(len(FREQS2)))])
        f2 = float(FREQS2[int(rng.randint(len(FREQS2)))])
        p1 = rng.uniform(0.0, 2.0 * math.pi)
        p2 = rng.uniform(0.0, 2.0 * math.pi)
        arr = np.arange(n, dtype=float) / float(n)
        h = 0.5 + 0.35 * np.cos(2.0 * math.pi * f1 * arr + p1) + 0.35 * np.cos(2.0 * math.pi * f2 * arr + p2)
        return np.clip(h, 0.0, 1.0)
    if hp is not None:
        return hp + rng.normal(0.0, 0.02, n)
    return rng.uniform(0.0, 1.0, n)


def polish_candidate(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if quick:
        seq = ((20.0, 35), (100.0, 50), (400.0, 65))
    else:
        seq = ((30.0, 45), (100.0, 60), (350.0, 80), (1200.0, 95), (4000.0, 105), (12000.0, 90))
    reserve = 25 if quick else 55
    for alpha, mi in seq:
        if deadline is not None and time.time() > deadline - reserve:
            break
        h2 = slsqp_alpha(best_h, n, alpha, mi, 1e-11)
        c2 = compute_c5(h2, n)
        if c2 + 1e-14 < best_c:
            best_h = h2
            best_c = c2
    if deadline is not None and time.time() < deadline - (20 if quick else 35):
        h2, c2 = subgrad_polish(
            best_h, n,
            maxiter=70 if quick else 220,
            beta=300.0,
            start_step=0.04,
            deadline=deadline - 5
        )
        if c2 + 1e-14 < best_c:
            best_h = h2
            best_c = c2
    if deadline is not None and time.time() < deadline - (12 if quick else 45):
        h2, c2 = fast_cd(
            best_h, n,
            max_passes=2 if quick else 8,
            grid=32 if quick else 55,
            deadline=deadline,
            rng=rng,
            penalty=20.0
        )
        if c2 + 1e-14 < best_c:
            best_h = h2
            best_c = c2
    if (not quick) and deadline is not None and time.time() < deadline - 95:
        xb = threshold_binary_from_floats(best_h, n, 0.5)
        bl = min(deadline - 30, time.time() + 35)
        xb, cb = binary_hill(xb, n, bl, rng, max_iters=25000)
        val_xb = cb
        if cb < best_c:
            best_h = xb.copy()
            best_c = cb
        if deadline is not None and time.time() < deadline - 45:
            start = xb if val_xb < compute_c5(best_h, n) else threshold_binary_from_floats(best_h, n, 0.5)
            h2 = slsqp_alpha(start, n, 100.0, 70, 1e-11)
            c2 = compute_c5(h2, n)
            if c2 + 1e-14 < best_c:
                best_h = h2
                best_c = c2
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if budget > 250.0:
        deadline = t0 + budget - 150.0
    elif budget > 100.0:
        deadline = t0 + budget - 60.0
    else:
        deadline = t0 + max(15.0, budget - 5.0)
    rng = np.random.RandomState(seed)

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h_init = np.asarray(parent, dtype=float).copy()
        n_init = len(h_init)
        if n_init % 2 != 0:
            h_init = h_init[:n_init - 1]
            n_init -= 1
        if n_init < 20:
            n_init = 300
            h_init = np.full(n_init, 0.5)
    else:
        n_init = 300
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    best_n = int(best_h.size)

    remaining = deadline - t0
    alloc1 = min(max(50.0, remaining * 0.35), 350.0)
    if alloc1 > 50.0:
        h1, c1 = polish_candidate(best_h, best_n, t0 + alloc1, rng, quick=False)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, int(h1.size)

    for n_try in ALT_NS:
        if n_try % 2:
            n_try -= 1
        if n_try < 40:
            continue
        if time.time() > deadline - 140.0:
            break
        if int(n_try) == best_n:
            continue
        hp = interp_parent(best_h, n_try)
        if hp is None:
            continue
        probe = min(55.0, deadline - time.time() - 50.0)
        if probe < 25.0:
            break
        hq, cq = polish_candidate(hp, n_try, time.time() + probe, rng, quick=True)
        if cq < best_c:
            hc = hq
            cc = cq
            full_alloc = min(130.0, deadline - time.time() - 80.0)
            if full_alloc > 35.0:
                hf, cf = polish_candidate(hc, n_try, time.time() + full_alloc, rng, quick=False)
                if cf < cc:
                    hc, cc = hf, cf
            if cc < best_c:
                best_h, best_c, best_n = hc, cc, int(hc.size)

    mode = 0
    n_cur = int(best_h.size)
    while mode < 14 and time.time() < deadline - 90.0:
        q_alloc = min(55.0, deadline - time.time() - 40.0)
        if q_alloc < 25.0:
            break
        h_c = make_candidate(n_cur, mode, rng, best_h)
        h_c = project_feasible(h_c, n_cur)
        hq, cq = polish_candidate(h_c, n_cur, time.time() + q_alloc, rng, quick=True)
        if cq < best_c:
            hc = hq
            cc = cq
            full_alloc = min(95.0, deadline - time.time() - 60.0)
            if full_alloc > 30.0:
                hf, cf = polish_candidate(hc, n_cur, time.time() + full_alloc, rng, quick=False)
                if cf < cc:
                    hc, cc = hf, cf
            if cc < best_c:
                best_h, best_c = hc, cc
                n_cur = int(best_h.size)
        mode += 1

    final_alloc = min(140.0, deadline - time.time() - 20.0)
    if final_alloc > 25.0:
        hf, cf = polish_candidate(best_h, best_n, time.time() + final_alloc, rng, quick=(budget < 450.0))
        if cf < best_c:
            best_h, best_c, best_n = hf, cf, int(hf.size)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
