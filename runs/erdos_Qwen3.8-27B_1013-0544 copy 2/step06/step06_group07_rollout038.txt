- If max(a_k for k active) < 0: move left
- If min(a_k for k active) > 0: move right
- Otherwise: we're at the minimum

def min_max_lines(a, b, lo=0.0, hi=1.0, iters=60):
    """Find t in [lo,hi] minimizing max_k (a[k]*t + b[k])"""
    # Quick check: evaluate at boundaries
    f_lo = b  # values at t=lo (which is 0)
    f_hi = a + b  # values at t=hi (which is 1)

    g_hi = f_hi.max()

            t_hi = t_mid
        elif a[active].max() < -1e-15:
            t_lo = t_mid
        else:
            break

    return t_opt, np.max(a * t_opt + b)

def coord_descent_lagrangian(h, n, deadline, max_sweeps=200, lambda_init=0.0, lambda_lr=0.01):
    """Coordinate descent with Lagrange multiplier for sum constraint"""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    lam = lambda_init
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            i_plus = i + lags
            i_minus = i - lags
            valid1 = (i_plus >= 0) & (i_plus < n)
            valid2 = (i_minus >= 0) & (i_minus < n)
            
            # a_k = valid1*(1 - h[i+l_k]) - valid2*h[i-l_k]
            a = np.zeros(2*n-1)
            a[valid1] = 1.0 - h[i_plus[valid1]]
            a[valid2] -= h[i_minus[valid2]]
            
            # b_k = corr[k] - a_k * h[i]  (since corr[k] = a_k*h[i] + b_k)
            b = corr - a * h[i]
            
            # Minimize max_k (a_k*t + b_k)*dx + lam*t
            # = max_k ((a_k*dx + lam)*t + b_k*dx) + lam*0  (constant doesn't matter)
            # Actually: max_k (a_k*t + b_k)*dx + lam*t 
            #         = max_k ((a_k*dx)*t + b_k*dx) + lam*t
            # This is NOT the same as max_k ((a_k*dx + lam)*t + b_k*dx)
            # because lam*t is added AFTER the max, not inside.
            
            # So we need to minimize: max_k (a_k*t + b_k)*dx + lam*t
            # The function is still convex piecewise linear.
            # g(t) = dx*max_k(a_k*t + b_k) + lam*t
            
            # For bisection, we need the subgradient:
            # g'(t) = dx * {a_k : k active} + lam (as a set)
            # = {dx*a_k + lam : k active}
            # So 0 in subgradient iff there exists active k with dx*a_k + lam = 0
            # i.e., a_k = -lam/dx
            
            # Let me modify min_max_lines to handle the extra linear term
            # Actually, let me just incorporate lam*dx into a:
            # g(t) = max_k (a_k*t + b_k)*dx + lam*t
            # Let A_k = a_k * dx, B_k = b_k * dx
            # g(t) = max_k (A_k*t + B_k) + lam*t
            # Subgradient at t: {A_k + lam : k is argmax}
            # So we're minimizing a function with effective slopes A_k + lam
            
            A = a * dx + lam  # effective slopes
            B = corr - a * h[i]  # wait, b_k = corr[k] - a_k*h[i], so B_k = b_k*dx
            B = (corr - a * h[i]) * dx
            
            t_opt, _ = min_max_lines(A, B, 0.0, 1.0)
            
            # Update
            delta = t_opt - h[i]
            if abs(delta) > 1e-15:
                h[i] = t_opt
                corr += a * delta
        
        # Update lambda
        sum_diff = h.sum() - n/2.0
        lam += lambda_lr * sum_diff
        # Clip h to [0,1] and project sum
        h = np.clip(h, 0.0, 1.0)
    
    return h, corr

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def min_max_lines(a, b, iters=35):
    f_lo = b
    f_hi = a + b
    g_lo = f_lo.max()
    g_hi = f_hi.max()
    tol = 1e-12
    active_lo = f_lo >= g_lo - 1e-10
    if a[active_lo].min() >= -tol:
        return 0.0
    active_hi = f_hi >= g_hi - 1e-10
    if a[active_hi].max() <= tol:
        return 1.0
    t_lo, t_hi = 0.0, 1.0
    for _ in range(iters):
        t_mid = 0.5 * (t_lo + t_hi)
        f = a * t_mid + b
        g_mid = f.max()
        active = f >= g_mid - 1e-12
        if a[active].min() > 1e-11:
            t_hi = t_mid
        elif a[active].max() < -1e-11:
            t_lo = t_mid
        else:
            break
    return 0.5 * (t_lo + t_hi)

def coord_descent(h, n, deadline, max_sweeps=300, use_lambda=False, lr=0.005):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    lags = np.arange(-(n - 1), n)
    m = 2 * n - 1
    lam = 0.0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        max_delta = 0.0
        for i in range(n):
            i_plus = i + lags
            i_minus = i - lags
            v1 = (i_plus >= 0) & (i_plus < n)
            v2 = (i_minus >= 0) & (i_minus < n)
            a = np.zeros(m)
            a[v1] = 1.0 - h[i_plus[v1]]
            a[v2] -= h[i_minus[v2]]
            b = corr - a * h[i]
            A = a * dx
            if use_lambda:
                A = A + lam
            t_opt = min_max_lines(A, b * dx)
            delta = t_opt - h[i]
            if abs(delta) > 1e-14:
                h[i] = t_opt
                corr += a * delta
                if abs(delta) > max_delta:
                    max_delta = abs(delta)
        h = np.clip(h, 0.0, 1.0)
        if use_lambda:
            lam += lr * (h.sum() - n / 2.0)
        if max_delta < 1e-10:
            break
    return h, compute_c5(h, n)

def lbfgs_smooth(h, n, alpha, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 800, 'ftol': 1e-15})
        return project_sum(res.x, n)
    except Exception:
        return project_sum(h, n)

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00005, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start
    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        d = np.zeros(2 * n - 1)
        ip = i + m_range
        mk = (ip >= 0) & (ip < n)
        d[mk] += 1.0 - h[ip[mk]]
        im_ = i - m_range
        mk = (im_ >= 0) & (im_ < n)
        d[mk] -= h[im_[mk]]
        jp = j + m_range
        mk = (jp >= 0) & (jp < n)
        d[mk] -= 1.0 - h[jp[mk]]
        jm_ = j - m_range
        mk = (jm_ >= 0) & (jm_ < n)
        d[mk] += h[jm_[mk]]
        d[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        d[k1] += 1.0
        k2 = n - 1 + (i - j)
        d[k2] += 1.0
        nc = (corr + d).max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr += d
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(7)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 4.0)
    return best_h, best_c
