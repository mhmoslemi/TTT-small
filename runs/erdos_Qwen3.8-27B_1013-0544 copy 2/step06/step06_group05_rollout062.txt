import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            sr = room.sum()
            if sr < 1e-14:
                break
            h = np.clip(h + diff / sr * room, 0, 1)
        else:
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def build_S(h, n):
    """S[p, k] = h[p+L_k] + h[p-L_k], L_k = k-(n-1), 0 if out of bounds."""
    ncorr = 2 * n - 1
    S = np.zeros((n, ncorr), dtype=np.float64)
    p = np.arange(n)
    for k in range(ncorr):
        L = k - (n - 1)
        ip = p + L
        im = p - L
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        if vp.any():
            S[vp, k] += h[ip[vp]]
        if vm.any():
            S[vm, k] += h[im[vm]]
    return S

def update_S(S, i, j, n):
    """Update S after swapping h[i]:0->1 and h[j]:1->0."""
    ncorr = 2 * n - 1
    k_range = np.arange(ncorr)
    L = k_range - (n - 1)
    # h[i] goes 0->1: add 1 to S[p,k] where p+L=i or p-L=i
    p1 = i - L  # p such that p + L = i
    p2 = i + L  # p such that p - L = i
    v1 = (p1 >= 0) & (p1 < n)
    v2 = (p2 >= 0) & (p2 < n)
    if v1.any():
        S[p1[v1], k_range[v1]] += 1
    if v2.any():
        S[p2[v2], k_range[v2]] += 1
    # h[j] goes 1->0: subtract 1
    p3 = j - L
    p4 = j + L
    v3 = (p3 >= 0) & (p3 < n)
    v4 = (p4 >= 0) & (p4 < n)
    if v3.any():
        S[p3[v3], k_range[v3]] -= 1
    if v4.any():
        S[p4[v4], k_range[v4]] -= 1

def fast_binary_search(h, n, deadline, rng):
    """Fast vectorized binary local search with annealing kicks."""
    dx = 2.0 / n
    h = binarize(h, n)
    S = build_S(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    ncorr = 2 * n - 1
    
    kick_cooldown = 0
    no_improve = 0
    
    while time.time() < deadline:
        # Determine critical lags
        thresh0 = cur_max - 0.5
        thresh1 = cur_max - 1.5
        crit0 = np.where(c >= thresh0)[0]
        crit1 = np.where((c >= thresh1) & (c < thresh0))[0]
        all_crit = np.concatenate([crit0, crit1])
        
        if len(all_crit) == 0:
            all_crit = np.argsort(c)[-20:]
        if len(all_crit) > 100:
            # Too many critical lags, use top 50
            all_crit = np.argsort(c)[-50:]
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        found = False
        # Random order for diversity
        order = zeros.copy()
        rng.shuffle(order)
        
        for i in order:
            if time.time() > deadline:
                break
            
            constraint = S[i, all_crit].astype(np.float64).copy()
            # For crit0 lags: need S[j,k] <= S[i,k] - 1
            c0_mask = np.isin(all_crit, crit0)
            constraint[c0_mask] -= 1.0
            
            Sj = S[ones, all_crit]
            valid = np.all(Sj <= constraint[None, :], axis=1)
            
            if valid.any():
                candidates = ones[valid]
                # Pick best candidate
                best_j = candidates[0]
                best_nm = cur_max + 1
                for j in candidates[:15]:
                    nm = (c + S[j, :] - S[i, :]).max()
                    if nm < best_nm:
                        best_nm = nm
                        best_j = j
                
                if best_nm < cur_max:
                    j = best_j
                    h[i] = 1.0
                    h[j] = 0.0
                    c = c + S[j, :] - S[i, :]
                    cur_max = c.max()
                    update_S(S, i, j, n)
                    found = True
                    no_improve = 0
                    break
        
        if not found:
            no_improve += 1
            # Annealing kick: accept a random non-worsening swap
            if no_improve > 20 and kick_cooldown <= 0:
                i = rng.choice(zeros)
                j = rng.choice(ones)
                delta = S[j, :] - S[i, :]
                nm = (c + delta).max()
                if nm <= cur_max + 2:
                    h[i] = 1.0
                    h[j] = 0.0
                    c = c + delta
                    cur_max = c.max()
                    update_S(S, i, j, n)
                    no_improve = 0
                    kick_cooldown = 5
            elif no_improve > len(zeros):
                break
        if kick_cooldown > 0:
            kick_cooldown -= 1
    
    return h, cur_max * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

            if time.time() > p2:

            c2 = compute_c5(h2, n)

    best_h = project_feasible(best_h, n)

def build_S(h, n):
    nc = 2 * n - 1
    S = np.zeros((n, nc), dtype=np.float64)
    p = np.arange(n)
    for k in range(nc):
        L = k - (n - 1)
        ip, im = p + L, p - L
        vp, vm = (ip >= 0) & (ip < n), (im >= 0) & (im < n)
        if vp.any(): S[vp, k] += h[ip[vp]]
        if vm.any(): S[vm, k] += h[im[vm]]
    return S
