    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_polish(h, n, deadline, n_steps=50000):
    """Polish using subgradient of exact max."""
    dx = 2.0 / n
    j = np.arange(n)
    best = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    for it in range(n_steps):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(g))
        d = k_star - (n - 1)
        # Gradient of c_k_star w.r.t. h
        grad = np.zeros(n)
        valid1 = (j - d >= 0) & (j - d < n)
        grad[valid1] += 1.0
        valid2 = (j + d >= 0) & (j + d < n)
        grad[valid2] -= h[j[valid2] + d]
        valid3 = (j - d >= 0) & (j - d < n)
        grad[valid3] -= h[j[valid3] - d]
        grad *= dx
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new <= compute_c5(h, n):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best = h.copy()
        else:
            step *= 0.95
        if it % 1000 == 0:
            step = max(step * 0.99, 0.001)
    return best, best_c

                jmn = j - m_range

                delta[n - 1] -= 2.0

                k2 = n - 1 + (i - j)

                nc = (corr + delta).max() * dx
                if nc < cur_c + best_delta_c:

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm_val = shifted.max()
    return m + sm_val + np.log(np.exp(shifted - sm_val).sum()) / alpha

def lbbfgs_opt(h, n, alpha, lam, maxiter=5000):
    target = n / 2.0
    def obj(x):
        return soft_max_obj(x, n, alpha) + lam * (x.sum() - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-13})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h
