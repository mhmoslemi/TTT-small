```python
def build_delta(h, n):
    m = 2 * n - 1
    delta = np.zeros((n, m))
    h = h.astype(float)
    # Source term: delta[p, k] += (1-h[p+k-n+1]) for valid k
    # Target term: delta[p, k] -= h[p-k+n+1] for valid k  (wait, p-k+n-1)
    for p in range(n):
        # Source: k from max(0, n-1-p) to min(2n-2, n-1+p)
        k_start = max(0, n - 1 - p)
        k_end = min(2*n-1, n - p)  # exclusive
        if k_end > k_start:
            idx = p + (np.arange(k_start, k_end) - n + 1)
            delta[p, k_start:k_end] += (1 - h[idx])
        # Target: k from max(0, p-n+1) to min(2n-2, p)
        k_start2 = max(0, p - n + 1)
        k_end2 = min(2*n-1, p + 1)  # exclusive
        if k_end2 > k_start2:
            idx2 = p - (np.arange(k_start2, k_end2) - n + 1)
            delta[p, k_start2:k_end2] -= h[idx2]
    return delta
```

This should be fast since the inner operations are vectorized numpy. The loop is only n=250 iterations, each doing O(n) numpy operations. Total: O(n^2) numpy ops ≈ 62K, which is essentially instant.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def thue_morse(n):
    t = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def build_delta(h, n):
    m = 2 * n - 1
    delta = np.zeros((n, m))
    h = h.astype(float)
    for p in range(n):
        ks = max(0, n - 1 - p)
        ke = min(2 * n - 1, n - p)
        if ke > ks:
            idx = p + np.arange(ks, ke) - n + 1
            delta[p, ks:ke] += (1.0 - h[idx])
        ks2 = max(0, p - n + 1)
        ke2 = min(2 * n - 1, p + 1)
        if ke2 > ks2:
            idx2 = p - np.arange(ks2, ke2) + n - 1
            delta[p, ks2:ke2] -= h[idx2]
    return delta

def local_search(h_init, n, deadline, seed=42, kick_interval=20, max_kicks=50):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = binarize(h_init, n)
    m = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    delta = build_delta(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    kicks = 0
    no_improve = 0
    
    while time.time() < deadline:
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Find best swap
        dz = delta[zeros]
        do = delta[ones]
        D = (corr[None, None, :] + dz[:, None, :] - do[None, :, :]).max(axis=2)
        flat_idx = D.argmin()
        ii = flat_idx // len(ones)
        jj = flat_idx % len(ones)
        i = zeros[ii]
        j = ones[jj]
        new_c = D[ii, jj] * dx
        
        if new_c < cur_c - 1e-12:
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta[i] - delta[j]
            delta = build_delta(h, n)
            cur_c = new_c
            stall = 0
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]

            if stall >= kick_interval and kicks < max_kicks:

                delta = build_delta(h, n)
                cur_c = corr.max() * dx
                zeros = np.where(h == 0)[0]

def slsqp_refine(h, n, maxiter=300, alpha=200000):
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        sm = g.max()
        sh = alpha * (g - sm)
        top = sh.max()
        return sm + (top + np.log(np.exp(sh - top).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
