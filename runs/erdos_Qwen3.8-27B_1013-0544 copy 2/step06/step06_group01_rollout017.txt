Wait, is c[k] concave or convex? c[k] = Σ_i h[i]*(1-h[i+k-(n-1)]). The term h[i]*(1-h[j]) = h[i] - h[i]*h[j]. The Hessian of h[i]*h[j] w.r.t. (h[i], h[j]) is [[0,1],[1,0]], which has eigenvalues 1 and -1. So it's indefinite, meaning c[k] is neither convex nor concave in general.

This confirms the problem is non-convex and our optimization approach needs to handle this.

For the 1D subproblem in coordinate descent, instead of a grid, I can find the exact minimum of max(um, max_k(c[k] + d*s[k] - d²*δ_{k,n-1})) over d ∈ [d_lo, d_hi].

The function g(d) = max(um, max_k(c[k] + d*s[k] - d²*δ_{k,n-1})) is a convex function of d (max of affine functions and one concave parabola... wait, the parabola c[n-1] + d*s[n-1] - d² is concave, not convex. So g(d) is the max of a constant, several affine functions, and one concave parabola. This is NOT convex in general.

However, the minimum of g(d) over a bounded interval can be found by:
1. Finding all breakpoints (where the max changes identity)
2. Evaluating at breakpoints and at the vertex of the parabola (if it's in range)

The breakpoints occur where two of the functions cross:
- c[k] + d*s[k] = c[k'] + d*s[k'] => d = (c[k']-c[k])/(s[k]-s[k'])
- c[k] + d*s[k] = c[n-1] + d*s[n-1] - d² => d² + d*(s[k]-s[n-1]) + c[k]-c[n-1] = 0
- um = c[k] + d*s[k] => d = (um - c[k])/s[k]
- um = c[n-1] + d*s[n-1] - d² => d² - d*s[n-1] + c[n-1]-um = 0

This could give many candidate points. A simpler approach: use a fine grid (say 200 points) plus ternary search refinement. Or just use 200 grid points - for n=150, that's 150*200 = 30000 evaluations per pass, which is fast.

Actually, for the step function case, there's an even simpler approach: just try d = ±(step size) and see if it improves. The "step size" can be adaptive.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        soft = (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        soft = np.log(np.sum(np.exp(s))) / alpha
    return float((m + soft) * dx)

def slsqp_optimize(h, n, alpha=50000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, max_passes=300, deadline=None, rng=None, grid=80):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(c))
    sumh = float(h.sum())
    target = n / 2.0
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for p in order:
            k_lo1 = max(0, n - 1 - p)
            k_hi1 = min(2 * n - 2, 2 * n - 2 - p)
            k_lo2 = max(0, p)
            k_hi2 = min(2 * n - 2, p + n - 1)
            k_lo = min(k_lo1, k_lo2)
            k_hi = max(k_hi1, k_hi2)
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                if k == n - 1:
                    s[idx] = 1.0 - 2.0 * h[p]
                else:
                    j1 = p + k - (n - 1)
                    if 0 <= j1 < n:
                        s[idx] += 1.0 - h[j1]
                    j2 = p - k + (n - 1)
                    if 0 <= j2 < n:
                        s[idx] -= h[j2]
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            is_n1 = (ks == n - 1)
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = cur_max + 50.0 * (sumh - target) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                nc = c[ks] + d * s
                if np.any(is_n1):
                    nc[is_n1] -= d * d
                val = max(um, float(np.max(nc)))
                val += 50.0 * (sumh + d - target) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-10:
                c[ks] += best_d * s
                if np.any(is_n1):
                    c[is_n1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                cur_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, cur_max * (2.0 / n)

def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        nb = rng.randint(3, 8)
        bnd = np.sort(rng.choice(np.arange(1, n), size=nb - 1, replace=False))
        bnd = np.concatenate([[0], bnd, [n]])
        for i in range(nb):
            if rng.random() < 0.5:
                h[bnd[i]:bnd[i + 1]] = 1.0
        return project_feasible(h, n)
    elif mode == 1:
        x = np.linspace(0, 2, n)
        freq = rng.uniform(1.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.15, 0.45)
        h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
        return project_feasible(h, n)
    elif mode == 2:
        h = 0.5 + rng.normal(0, 0.15, n)
        ker = np.ones(7) / 7.0
        h = np.convolve(h, ker, mode='same')
        return project_feasible(h, n)
    else:
        h = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
        return project_feasible(h, n)

def optimize_one(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    if time.time() < deadline - 15:
        t_sub = min(60, (deadline - time.time()) * 0.35)
        h1, c1 = coord_descent(h, n, max_passes=400, deadline=time.time() + t_sub, rng=rng, grid=60)
        h1 = project_feasible(h1, n)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            h, best_c = h1, c1
    if time.time() < deadline - 20:
        a = 30000 if time.time() < deadline - 50 else 100000
        h2 = slsqp_optimize(h, n, alpha=a, maxiter=1500)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h, best_c = h2, c1 if False else c2
            h, best_c = h2, c2
    if time.time() < deadline - 10:
        h3, c3 = coord_descent(h, n, max_passes=300, deadline=time.time() + (deadline - time.time() - 5), rng=rng, grid=100)
        h3 = project_feasible(h3, n)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            h, best_c = h3, c3
    if time.time() < deadline - 5:
        h4 = slsqp_optimize(h, n, alpha=200000, maxiter=2000)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            h, best_c = h4, c4
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def subgradient_step(h, n, alpha, rng):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    thresh = c_max - 0.02 * c_max
    active = np.where(c >= thresh)[0]
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            g = 0.0
            j1 = j + k - (n - 1)
            if 0 <= j1 < n:
                g += 1.0 - h[j1]
            j2 = j - k + (n - 1)
            if 0 <= j2 < n:
                g -= h[j2]
            grad[j] += g
    grad /= max(1, len(active))
    h_new = h - alpha * grad
    h_new = project_feasible(h_new, n)
    return h_new

        if time.time() < t_end - 10 and len(best_h) == n:
