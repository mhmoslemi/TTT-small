`grad_from_weights`:
```
def grad_from_weights(h,w):
    n=len(h)
    if n==0: return 0.0,np.zeros(0)
    dx=2.0/n
    idx=np.arange(n)
    pref=np.zeros(2*n)
    pref[1:]=np.cumsum(w)
    A=pref[2*n-1-idx]-pref[n-1-idx]
    cB=np.convolve(w,h[::-1],mode='full')
    B=cB[2*n-2-idx]
    cD=np.convolve(h,w,mode='full')
    D=cD[n-1:2*n-1]
```

```
def project(v):
    v=np.asarray(v,dtype=float).reshape(-1)
    n=len(v); t=n*0.5
    v=np.clip(v,0,1)
    lo=-1.0; hi=1.0
    for _ in range(50):
        m=0.5*(lo+hi)
        if float(np.sum(np.clip(v+m,0,1))) < t: lo=m
        else: hi=m
    h=np.clip(v+0.5*(lo+hi),0,1)
    for _ in range(20):
        d=t-h.sum()
        if abs(d)<1e-13: break
        if d>0:
            r=1-h; s=r.sum()
            if s>1e-15: h=h+d*r/s
        else:
            s=h.sum()
            if s>1e-15: h=h+d*h/s
    return np.clip(h,0,1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_SLSQP_N = 0
_SLSQP_BETA = 1.0
_CX = None
_CO = None
_CG = None


def current_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)


def compute(h):
    return float(np.max(current_corr(h)) * (2.0 / len(h)))


def project(v):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = len(v)
    t = n * 0.5
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v + m, 0.0, 1.0))) < t:
            lo = m
        else:
            hi = m
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = float(r.sum())
            if s > 1e-15:
                h = h + d * r / s
        else:
            s = float(h.sum())
            if s > 1e-15:
                h = h + d * h / s
    return np.clip(h, 0.0, 1.0)


def grad_w(h, w):
    n = len(h)
    idx = np.arange(n)
    p = np.zeros(2 * n)
    p[1:] = np.cumsum(w)
    A = p[2 * n - 1 - idx] - p[n - 1 - idx]
    B = np.convolve(w, h[::-1], mode='full')[2 * n - 2 - idx]
    D = np.convolve(h, w, mode='full')[n - 1:2 * n - 1]
    return (A - B - D) * (2.0 / n)


def lse_obj_grad(h, beta):
    n = len(h)
    c = current_corr(h) * (2.0 / n)
    mx = float(c.max())
    z = c - mx
    if beta <= 1e-8:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        return mx, grad_w(h, w)
    e = np.exp(beta * z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return float(mx + np.log(s) / beta), grad_w(h, w)


def slsqp_sum(x):
    x = np.asarray(x, dtype=float).reshape(_SLSQP_N)
    return float(x.sum() - 0.5 * _SLSQP_N)


def slsqp_sum_jac(x):
    return np.ones(_SLSQP_N)


def slsqp_obj(x):
    global _CX, _CO, _CG
    x = np.asarray(x, dtype=float).reshape(_SLSQP_N)
    if _CX is not None and _CX.shape == x.shape and np.array_equal(_CX, x):
        return _CO
    o, g = lse_obj_grad(x, _SLSQP_BETA)
    _CX = x.copy()
    _CO = float(o)
    _CG = g
    return _CO


def slsqp_jac(x):
    global _CX, _CO, _CG
    x = np.asarray(x, dtype=float).reshape(_SLSQP_N)
    if _CX is not None and _CX.shape == x.shape and np.array_equal(_CX, x):
        return _CG.copy()
    o, g = lse_obj_grad(x, _SLSQP_BETA)
    _CX = x.copy()
    _CO = float(o)
    _CG = g
    return _CG.copy()


def run_slsqp(h0, beta, it):
    global _SLSQP_N, _SLSQP_BETA, _CX
    h0 = project(h0)
    n = len(h0)
    _SLSQP_N = n
    _SLSQP_BETA = float(beta)
    _CX = None
    try:
        r = minimize(
            slsqp_obj, h0, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac}],
            jac=slsqp_jac,
            options={'maxiter': max(1, min(int(it), 250)), 'ftol': 1e-10, 'maxls': 50}
        )
        if np.isfinite(r.fun) and np.all(np.isfinite(r.x)):
            x = project(np.asarray(r.x, dtype=float).reshape(n))
            return x, compute(x)
    except Exception:
        pass
    return h0, compute(h0)


def slsqp_seq(h, dl, betas, it=100):
    h = project(h)
    bc = compute(h)
    bh = h.copy()
    for b in betas:

def resample(h, n2):
    if len(h) == n2:
        return project(h)
    x0 = np.linspace(0.0, 2.0, len(h))
    x1 = np.linspace(0.0, 2.0, n2)
    return project(np.interp(x1, x0, h))


def rand_bin(n, seed):
    rng = np.random.RandomState(int(seed) % (2**32 - 1))
    out = np.zeros(n)
    idx = np.argsort(rng.rand(n))[::-1][:n // 2]
    out[idx] = 1.0
    return out


def round_bin(h):
    n = len(h)
    out = np.zeros(n)
    idx = np.argsort(h)[::-1][:n // 2]
    out[idx] = 1.0
    return out


def swap_delta(h, n, i, j, mr):
    d = np.zeros(2 * n - 1)
    t = i + mr
    m = (t >= 0) & (t < n)
    d[m] += 1.0 - h[t[m]]
    t = i - mr
    m = (t >= 0) & (t < n)
    d[m] -= h[t[m]]
    t = j + mr
    m = (t >= 0) & (t < n)
    d[m] -= 1.0 - h[t[m]]
    t = j - mr
    m = (t >= 0) & (t < n)
    d[m] += h[t[m]]
    d[n - 1] -= 2.0
    d[n - 1 + (j - i)] += 1.0
    d[n - 1 + (i - j)] += 1.0
    return d


def bin_score(c, K, mx):
    v = c - float(K) - 1e-9
    m = v > 0.0
    if np.any(m):
        vv = v[m]
        return float(np.dot(vv, vv)) + 1e-3 * float(mx)
    return 1e-3 * float(mx)

                        ln = n - abs(s)
                        r = st + rng.randint(ln)
                        if h[r] > 0.5 and h[r + s] < 0.5:
                            i = int(r + s)
                            j = int(r)
                            break
        if i < 0:
            for _ in range(12):
                a = rng.randint(n)
                b = rng.randint(n)
                if h[a] > 0.5 and h[b] < 0.5 and a != b:
                    i = b
                    j = a
                    break
        if i < 0:
            i = int(np.argmin(h))
            j = int(np.argmax(h))
        if i == j:
            continue
        dlt = swap_delta(h, n, i, j, mr)
        nc = corr + dlt
        nm = float(nc.max())
        ns = bin_score(nc, K, nm)
        if ns < cs or (T > 1e-12 and rng.random() < np.exp(-(ns - cs) / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = nc
            cm = nm
            cs = ns
            acs += 1
            if acs % 5000 == 0:
                corr = current_corr(h).astype(float)
                cm = float(corr.max())
                cs = bin_score(corr, K, cm)
            if nm < bm or (nm < bm + 1e-9 and ns < bs):
                bm = nm
                bs = ns
                bh = h.copy()
                bc = nm * 2.0 / n
            ni = 0
        else:
            ni += 1
        T *= cr
        if ni > 30000:
            for _ in range(5):
                ok = False
                for __ in range(20):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if h[a] > 0.5 and h[b] < 0.5 and a != b:
                        h[b] = 1.0
                        h[a] = 0.0
                        ok = True
                        break
                if not ok:
                    break
            corr = current_corr(h).astype(float)
            cm = float(corr.max())
            cs = bin_score(corr, K, cm)
            ni = 0
            T = max(T, 0.15)
        if bm <= K and ni > 8000:
            break
    return bh, bc


def active_desc(h0, dl, it=120):
    h = project(h0)
    c = compute(h)
    bc, bh, fail = c, h.copy(), 0
    for _ in range(it):
        if time.time() > dl:
            break
        n = len(h)
        C = current_corr(h) * (2.0 / n)
        k = max(1, int(0.03 * len(C)))
        if k < len(C):
            idx = np.argpartition(C, -k)[-k:]
        else:
            idx = np.arange(len(C))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / float(len(idx))
        g = grad_w(h, w)
        d = -g
        d = d - d.mean()
        nn = float(np.linalg.norm(d))
        if nn < 1e-15:
            break
        d = d / nn
        ok = False
        a = 0.08
        for __ in range(25):
            cand = project(h + a * d)
            cc = compute(cand)
            if cc < c - 1e-14:
                h = cand
                c = cc
                ok = True
                break
            a *= 0.7
        if ok:
            fail = 0
            if c < bc:
                bc = c
                bh = h.copy()
        else:
            fail += 1
            if fail > 60:
                break
    return bh, bc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    span = max(1.0, float(budget_s) - 4.0)
    dl = t0 + span
    rng = np.random.RandomState(int(seed) % (2**32 - 1))

    try:
        init = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except Exception:
        init = None
    if init is None or len(init) < 16:
        n0 = 250
        init = rand_bin(n0, seed)
    else:
        n0 = len(init)
        if n0 % 2:
            init = resample(init, n0 + 1)
            n0 += 1

    best = project(init)
    bc = compute(best)

    p1 = t0 + 0.22 * span
    if time.time() < p1:
        h1, c1 = slsqp_seq(best, p1, [10.0, 40.0, 160.0, 600.0, 2400.0, 8000.0, 24000.0], 110)
        if c1 < bc:
            best, bc = h1, c1
        if time.time() < p1 - 30.0:
            hw = project(best + 0.04 * rng.randn(len(best)))
            h2, c2 = slsqp_seq(hw, p1, [50.0, 250.0, 1000.0, 4000.0], 70)
            if c2 < bc:
                best, bc = h2, c2
        if time.time() < p1 - 30.0:
            hw = rand_bin(len(best), seed + 7)
            hw = project(hw + 0.12 * rng.randn(len(hw)))
            h3, c3 = slsqp_seq(hw, p1, [250.0, 1000.0, 4000.0], 50)
            if c3 < bc:
                best, bc = h3, c3

    p2 = t0 + 0.62 * span
    n0 = len(best)
    bin_end = min(p2, dl - 20.0)
    if n0 % 2 == 0 and time.time() < bin_end:
        K = target_count(n0, bc)
        if K > 0:
            st = time.time()
            d = bin_end - st
            if d > 20.0:
                hb, cb = binary_search(round_bin(best), n0, st + 0.45 * d, K, seed + 21)
                if cb < bc:
                    best, bc = hb, cb
                if time.time() < bin_end:
                    hst = round_bin(project(best + 0.08 * rng.randn(n0)))
                    hb, cb = binary_search(hst, n0, st + 0.75 * d, K, seed + 31)
                    if cb < bc:
                        best, bc = hb, cb
                if time.time() < bin_end:
                    hb, cb = binary_search(rand_bin(n0, seed + 41), n0, bin_end, K, seed + 51)
                    if cb < bc:
                        best, bc = hb, cb

    nnow = len(best)
    p3 = t0 + 0.85 * span
    if nnow < 320 and time.time() < p3 - 70.0:
        n2 = 256 if nnow < 256 else 320
        h2 = resample(best, n2)
        h2, c2 = slsqp_seq(h2, p3, [100.0, 500.0, 2000.0, 8000.0, 24000.0], 65)
        if c2 < bc:
            best, bc = h2, c2

    if time.time() < dl - 20.0:
        hf, ch = slsqp_seq(best, dl - 8.0, [1000.0, 4000.0, 12000.0, 30000.0], 60)
        if ch < bc:
            best, bc = hf, ch

    nfin = len(best)
    if nfin % 2 == 0 and time.time() < dl - 80.0:
        K = target_count(nfin, bc)
        if K > 0:
            hb, cb = binary_search(round_bin(best), nfin, dl - 5.0, K, seed + 901)
            if cb < bc:
                best, bc = hb, cb

    if time.time() < dl - 5.0:
        ha, ca = active_desc(best, dl - 3.0, 120)
        if ca < bc:
            best, bc = ha, bc

    hfinal = project(best)
    cf = compute(hfinal)
    if len(hfinal) % 2 == 0:
        hb = round_bin(hfinal)
        cb = compute(hb)
        if cb < cf:
            hfinal = project(hb)
            cf = cb
    return hfinal, float(cf), len(hfinal)
```
