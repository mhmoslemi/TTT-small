def coord_descent(h, n, max_passes=2000, deadline=None, rng=None, penalty=50.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    dx = 2.0 / n
    sumh = float(h.sum())
    n2 = 2 * n - 1
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            h_p = h[p]
            if h_p < 1e-14 or h_p > 1.0 - 1e-14:
                continue
            lo_bin = n - 1 - p
            hi_bin = n - 1 + p
            k_vals = np.arange(lo_bin, hi_bin + 1) - (n - 1)
            slopes = np.zeros(len(k_vals))
            fv = (p + k_vals >= 0) & (p + k_vals < n)
            slopes[fv] += (1.0 - h[p + k_vals[fv]])
            sv = (p - k_vals >= 0) & (p - k_vals < n)
            slopes[sv] -= h[p - k_vals[sv]]
            
            unaffected_max = 0.0
            if lo_bin > 0:
                unaffected_max = float(np.max(c[:lo_bin]))
            if hi_bin < n2 - 1:
                unaffected_max = max(unaffected_max, float(np.max(c[hi_bin+1:])))
            
            d_lo = -h_p
            d_hi = 1.0 - h_p
            a = c[lo_bin:hi_bin+1].copy()
            
            # Solve 1D: min max(unaffected_max, max(a + d*slopes)) + penalty*(sumh+d-n/2)^2
            # Use golden section on [d_lo, d_hi]
            def obj_1d(d):
                nc = a + d * slopes
                return max(unaffected_max, float(np.max(nc))) + penalty * (sumh + d - n/2.0)**2
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_gs, b_gs = d_lo, d_hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = obj_1d(c_gs)
            fd = obj_1d(d_gs)
            for _ in range(40):  # ~40 iterations gives precision ~ (b-a)/phi^40 ≈ 1e-8 * (b-a)
                if fc < fd:
                    b_gs = d_gs; d_gs = c_gs; fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs); fc = obj_1d(c_gs)
                else:
                    a_gs = c_gs; c_gs = d_gs; fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs); fd = obj_1d(d_gs)
            best_d = (a_gs + b_gs) / 2.0
            
            if abs(best_d) > 1e-14:
                c[lo_bin:hi_bin+1] += best_d * slopes
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(unaffected_max, float(np.max(c[lo_bin:hi_bin+1])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def pair_descent(h, n, max_passes=500, deadline=None, rng=None, penalty=100.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    dx = 2.0 / n
    sumh = float(h.sum())
    n2 = 2 * n - 1
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Random pairs
        for _ in range(2 * n):
            i, j = rng.randint(0, n - 1), rng.randint(0, n - 1)
            if i == j:
                continue
            s = h[i] + h[j]
            # t = h[i], so h[j] = s - t
            # Constraints: 0 <= t <= 1, 0 <= s-t <= 1
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-14:
                continue
            if abs(h[i] - t_lo) < 1e-14 and abs(h[i] - t_hi) < 1e-14:
                continue
            
            # Compute affected bins
            k_i = np.array([n-1-i, n-1+i])  # bins affected by h[i]
            k_j = np.array([n-1-j, n-1+j])  # bins affected by h[j]
            affected = np.sort(np.unique(np.concatenate([k_i, k_j])))
            affected = affected[(affected >= 0) & (affected < n2)]
            
            # Current values at affected bins
            c_affected = c[affected].copy()
            
            # Slopes with respect to t (change in h[i] by dt changes h[j] by -dt)
            # For bin b = n-1+k:
            # C(k) = sum_m h[m+k](1-h[m])
            # ∂C(k)/∂h[i] = (1-h[i-k])_{valid} - h[i+k]_{valid}  [from our formula]
            # Wait, let me use: ∂C(k)/∂h[j_idx] = (1-h[j_idx-k])_{0≤j_idx-k<n} - h[j_idx+k]_{0≤j_idx+k<n}
            
            slopes_i = np.zeros(len(affected))
            slopes_j = np.zeros(len(affected))
            ks = affected - (n - 1)
            
            for idx, k in enumerate(ks):
                # ∂C(k)/∂h[i]
                if 0 <= i - k < n:

                if 0 <= i + k < n:

                # ∂C(k)/∂h[j]
                if 0 <= j - k < n:
                    slopes_j[idx] += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    slopes_j[idx] -= h[j + k]

            mask = np.zeros(n2, dtype=bool)
            mask[affected] = True
            if np.any(~mask):
                unaffected_max = float(np.max(c[~mask]))
            else:
                unaffected_max = 0.0

            # = c_affected + (t-h[i]) * (slopes_i - slopes_j)
            
            combined_slopes = slopes_i - slopes_j
            t0 = h[i]
            
            def obj_t(t):
                nc = c_affected + (t - t0) * combined_slopes
                return max(unaffected_max, float(np.max(nc)))
            
            # Golden section on [t_lo, t_hi]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = t_lo, t_hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = obj_t(c_g)
            fd = obj_t(d_g)
            for _ in range(50):

                    b_g = d_g; d_g = c_g; fd = fc
                    c_g = b_g - gr * (b_g - a_g); fc = obj_t(c_g)
                else:
                    a_g = c_g; c_g = d_g; fc = fd

            best_t = (a_g + b_g) / 2.0
            
            if abs(best_t - h[i]) > 1e-14:

                h[i] = best_t
                h[j] = s - best_t
                c[affected] += dt * combined_slopes
                current_max = max(unaffected_max, float(np.max(c[affected])))
                improved = True
        
        if not improved:
            break
    return h, current_max * dx

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _golden_max(a_func, lo, hi, iters=45):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g, b_g = lo, hi
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)
    fc = a_func(c_g)
    fd = a_func(d_g)
    for _ in range(iters):
        if fc < fd:
            b_g = d_g; d_g = c_g; fd = fc
            c_g = b_g - gr * (b_g - a_g); fc = a_func(c_g)
        else:
            a_g = c_g; c_g = d_g; fc = fd
            d_g = a_g + gr * (b_g - a_g); fd = a_func(d_g)
    return (a_g + b_g) / 2.0

def cd_step(h, n, c, p, penalty):
    h_p = h[p]
    if h_p < 1e-14 or h_p > 1.0 - 1e-14:
        return None
    lo_bin = n - 1 - p
    hi_bin = n - 1 + p
    a = c[lo_bin:hi_bin + 1].copy()
    k_vals = np.arange(lo_bin, hi_bin + 1) - (n - 1)
    slopes = np.zeros(len(k_vals))
    fv = (p + k_vals >= 0) & (p + k_vals < n)
    slopes[fv] += (1.0 - h[p + k_vals[fv]])
    sv = (p - k_vals >= 0) & (p - k_vals < n)
    slopes[sv] -= h[p - k_vals[sv]]
    n2 = 2 * n - 1
    unaff = 0.0
    if lo_bin > 0:
        unaff = float(np.max(c[:lo_bin]))
    if hi_bin < n2 - 1:
        unaff = max(unaff, float(np.max(c[hi_bin + 1:])))
    sumh = float(h.sum())
    target = n / 2.0
    d_lo = -h_p
    d_hi = 1.0 - h_p
    if d_hi - d_lo < 1e-15:
        return None
    def obj(d):
        nc = a + d * slopes
        return max(unaff, float(np.max(nc))) + penalty * (sumh + d - target) ** 2
    best_d = _golden_max(obj, d_lo, d_hi, 40)
    if abs(best_d) < 1e-14:
        return None
    return best_d

def coord_descent(h, n, max_passes=2000, deadline=None, rng=None, penalty=50.0):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_cmax = float(np.max(c))
    dx = 2.0 / n
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d = cd_step(h, n, c, p, penalty)
            if d is not None:
                lo_bin = n - 1 - p
                hi_bin = n - 1 + p
                k_vals = np.arange(lo_bin, hi_bin + 1) - (n - 1)
                slopes = np.zeros(len(k_vals))
                fv = (p + k_vals >= 0) & (p + k_vals < n)
                slopes[fv] += (1.0 - h[p + k_vals[fv]])
                sv = (p - k_vals >= 0) & (p - k_vals < n)
                slopes[sv] -= h[p - k_vals[sv]]
                c[lo_bin:hi_bin + 1] += d * slopes
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                n2 = 2 * n - 1
                unaff = 0.0
                if lo_bin > 0:
                    unaff = float(np.max(c[:lo_bin]))
                if hi_bin < n2 - 1:
                    unaff = max(unaff, float(np.max(c[hi_bin + 1:])))
                current_cmax = max(unaff, float(np.max(c[lo_bin:hi_bin + 1])))
                improved = True
        if not improved:
            break
    return h, current_cmax * dx

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsum = mx + np.log(np.sum(np.exp(shifted - mx))) if mx > 500 else np.log(np.sum(np.exp(shifted)))
    return float((m + logsum / alpha) * dx)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w = w / w.sum()
    grad = np.zeros(n)
    ks_all = np.arange(-(n - 1), n)
    for j in range(n):
        v0 = (j - ks_all >= 0) & (j - ks_all < n)
        v1 = (j + ks_all >= 0) & (j + ks_all < n)
        grad[j] = float(np.sum(w[v0])) - float(np.sum(w[v0] * x[j - ks_all[v0]])) - float(np.sum(w[v1] * x[j + ks_all[v1]]))
    return grad * dx

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       jac=lambda x, nn, aa: smooth_grad(x, nn, aa),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_descent(h, n, max_passes=200, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_cmax = float(np.max(c))
    dx = 2.0 / n
    n2 = 2 * n - 1
    ks_all = np.arange(-(n - 1), n)
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(n // 2):
            i = rng.randint(0, n - 1)
            j = rng.randint(0, n - 1)
            if i == j:
                continue
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-14:
                continue
            si = np.zeros(n2)
            sj = np.zeros(n2)
            v0i = (i - ks_all >= 0) & (i - ks_all < n)
            v1i = (i + ks_all >= 0) & (i + ks_all < n)
            si[v0i] += (1.0 - h[i - ks_all[v0i]])
            si[v1i] -= h[i + ks_all[v1i]]
            v0j = (j - ks_all >= 0) & (j - ks_all < n)
            v1j = (j + ks_all >= 0) & (j + ks_all < n)
            sj[v0j] += (1.0 - h[j - ks_all[v0j]])
            sj[v1j] -= h[j + ks_all[v1j]]
            combined = si - sj
            active = combined != 0.0
            if not np.any(active):
                continue
            c_act = c[active].copy()
            sl_act = combined[active]
            mask = np.zeros(n2, dtype=bool)
            mask[active] = True
            unaff = float(np.max(c[~mask])) if np.any(~mask) else 0.0
            t0 = h[i]
            def obj_t(t):
                nc = c_act + (t - t0) * sl_act
                return max(unaff, float(np.max(nc)))
            best_t = _golden_max(obj_t, t_lo, t_hi, 40)
            if abs(best_t - h[i]) > 1e-14:
                dt = best_t - h[i]
                h[i] = best_t
                h[j] = s - best_t
                c[active] += dt * sl_act
                current_cmax = max(unaff, float(np.max(c[active])))
                improved = True
        if not improved:
            break
    return h, current_cmax * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n = 300
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: aggressive coordinate descent
    t1 = min(deadline, t0 + 350)
    h1, c1 = coord_descent(best_h, n, max_passes=3000, deadline=t1, rng=rng, penalty=50.0)
    h1 = project_feasible(h1, n)
    c1 = compute_c5(h1, n)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: SLSQP polish
    if time.time() < deadline - 200:
        for alpha in [100000, 300000, 800000]:
            if time.time() > deadline - 150:
                break
            h2 = slsqp_opt(best_h, n, alpha=alpha, maxiter=1500)
            c2 = compute_c5(h2, n)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 3: pair descent
    if time.time() < deadline - 150:
        h3, c3 = pair_descent(best_h, n, max_passes=300, deadline=time.time() + 80, rng=rng)
        h3 = project_feasible(h3, n)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: more coordinate descent (finer)
    if time.time() < deadline - 100:
        t4 = min(deadline - 80, time.time() + 150)
        h4, c4 = coord_descent(best_h, n, max_passes=3000, deadline=t4, rng=rng, penalty=200.0)
        h4 = project_feasible(h4, n)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: final SLSQP
    if time.time() < deadline - 60:
        h5 = slsqp_opt(best_h, n, alpha=1000000, maxiter=2000)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: structured restarts
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 120))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.01 + 0.04 * (r / max(1, n_restarts))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(100, (deadline - time.time() - 30))
            if alloc < 30:
                break
            t_r = time.time() + alloc
            hr, cr = coord_descent(hp, n, max_passes=2000, deadline=t_r, rng=rng, penalty=100.0)
            hr = project_feasible(hr, n)
            cr = compute_c5(hr, n)
            if cr < best_c:
                best_c, best_h = cr, hr
                if time.time() < deadline - 40:
                    hf = slsqp_opt(best_h, n, alpha=500000, maxiter=1000)
                    cf = compute_c5(hf, n)
                    if cf < best_c:
                        best_c, best_h = cf, hf

    # Phase 7: final fine polish
    if time.time() < deadline - 20:
        h_final, c_final = coord_descent(best_h, n, max_passes=500,
                                          deadline=time.time() + 15, rng=rng, penalty=500.0)
        h_final = project_feasible(h_final, n)
        c_final = compute_c5(h_final, n)
        if c_final < best_c:
            best_c, best_h = c_final, h_final

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
