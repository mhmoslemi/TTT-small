```python
            break

                return h, best_c

            # Vectorized computation
            # k ranges from -(n-1) to (n-1)

                best_c = new_c

            break
```

But actually, I can do much better by computing the change analytically. When I swap h[i] (1→0) and h[j] (0→1):

The correlation C[m] = sum_l h[l](1-h[l+k]) where k = m-(n-1).

The change ΔC[m] = C_new[m] - C_old[m]:

Terms that change are those involving index i or index j:
1. l = i: was h[i](1-h[i+k]), now 0. Change: -h[i](1-h[i+k]) = -(1-h[i+k]) [since h[i]=1]
2. l+k = i, i.e., l = i-k: was h[i-k](1-h[i]) = h[i-k]*0 = 0, now h[i-k]*(1-0) = h[i-k]. Change: +h[i-k]
3. l = j: was h[j](1-h[j+k]) = 0, now 1*(1-h[j+k]). Change: +(1-h[j+k])
4. l+k = j, i.e., l = j-k: was h[j-k](1-h[j]) = h[j-k]*1 = h[j-k], now h[j-k]*0. Change: -h[j-k]

(All with appropriate boundary checks: 0 ≤ l < n and 0 ≤ l+k < n)

So ΔC[m] = -(1-h[i+k])*[valid1] + h[i-k]*[valid2] + (1-h[j+k])*[valid3] - h[j-k]*[valid4]

where valid1 = (0≤i<n and 0≤i+k<n), etc.

This is O(1) per lag! So computing ΔC for all lags is O(n) = O(599). And I just need to check if max(C + ΔC) < best_c.

This makes the binary local search much faster: O(n) per swap candidate, O(n²) per pass (since there are n²/4 swap candidates). For n=300: 22,500 * 599 ≈ 13.5M per pass, about 50ms per pass. In 100 seconds: about 2000 passes.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, alpha=1000.0):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    C = compute_c5_vec(h, n)
    best_c = float(C.max())
    order = np.arange(n)
    for sweep in range(1000000):
        if time.time() > deadline:
            break
        np.random.shuffle(order)
        any_change = False
        for i in order:
            if time.time() > deadline:
                break
            h_i = h[i]
            B = np.zeros(n_lags)
            k_lo = max(-i, i - n + 1)
            k_hi = min(n - 1 - i, i)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi + 1)
                ms = (ks + n - 1).astype(int)
                v1 = (i + ks >= 0) & (i + ks < n)
                if v1.any():
                    B[ms[v1]] += (1.0 - h[i + ks[v1]])
                v2 = (i - ks >= 0) & (i - ks < n)
                if v2.any():
                    B[ms[v2]] -= h[i - ks[v2]]
            if np.all(B == 0):
                continue
            z_lo = alpha * (C - B * h_i)
            zm = z_lo.max()
            w_lo = np.exp(z_lo - zm)
            fp_lo = w_lo.dot(B) / w_lo.sum()
            if fp_lo > 1e-15:
                new_t = 0.0
            else:
                z_hi = alpha * (C + B * (1.0 - h_i))
                zm = z_hi.max()
                w_hi = np.exp(z_hi - zm)
                fp_hi = w_hi.dot(B) / w_hi.sum()
                if fp_hi < -1e-15:
                    new_t = 1.0
                else:
                    lo_t, hi_t = 0.0, 1.0
                    for _ in range(45):
                        mid = (lo_t + hi_t) * 0.5
                        z = alpha * (C + B * (mid - h_i))
                        zm2 = z.max()
                        w = np.exp(z - zm2)
                        ws = w.sum()
                        if ws < 1e-300:
                            break
                        fp = w.dot(B) / ws
                        if fp > 1e-15:
                            hi_t = mid
                        elif fp < -1e-15:
                            lo_t = mid
                        else:
                            break
                    new_t = (lo_t + hi_t) * 0.5
            if abs(new_t - h_i) < 1e-14:
                continue
            delta = new_t - h_i
            C += B * delta
            h[i] = new_t
            any_change = True
            c_new = C.max()
            if c_new < best_c:
                best_c = c_new
        if not any_change:
            break
    C_final = compute_c5_vec(h, n)
    final_c = float(C_final.max())
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def binary_round(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_local_search(h, n, deadline):
    h = h.copy()
    n_lags = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    ms = (ks + n - 1).astype(int)
    C = compute_c5_vec(h, n)
    best_c = float(C.max())
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    while time.time() < deadline:
        improved = False
        j_arr = np.array(zeros) if zeros else np.array([], dtype=int)
        if len(j_arr) == 0:
            break
        for i in ones[:]:
            if time.time() > deadline:
                return h, best_c
            delta_i = np.zeros(n_lags)
            idx1 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            delta_i[ms[v1]] -= (1.0 - h[idx1[v1]])
            idx2 = i - ks
            v2 = (idx2 >= 0) & (idx2 < n)
            delta_i[ms[v2]] += h[idx2[v2]]
            if len(j_arr) == 0:
                break
            delta_js = np.zeros((len(j_arr), n_lags))
            for jj, j in enumerate(j_arr):
                idx3 = j + ks
                v3 = (idx3 >= 0) & (idx3 < n)
                delta_js[jj, ms[v3]] += (1.0 - h[idx3[v3]])
                idx4 = j - ks
                v4 = (idx4 >= 0) & (idx4 < n)
                delta_js[jj, ms[v4]] -= h[idx4[v4]]
            new_maxes = (C + delta_i[None, :] + delta_js).max(axis=1)
            best_j_idx = int(new_maxes.argmin())
            if new_maxes[best_j_idx] < best_c - 1e-14:
                j = j_arr[best_j_idx]
                h[i] = 0
                h[j] = 1
                C += delta_i + delta_js[best_j_idx]
                best_c = float(new_maxes[best_j_idx])
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                j_arr = np.array(zeros) if zeros else np.array([], dtype=int)
                improved = True
                break
        if not improved:
            break
    return h, best_c

def pair_search(h, n, deadline, deltas=None, radius=50):
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5
    try:

    best_h = h.copy()
    alphas = [50, 200, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        remaining = deadline - time.time() - 60
        if remaining < 15:
            break
        alloc = min(80, remaining * 0.4)
        if alloc < 10:
            break
        h, c = coord_descent(h, n, time.time() + alloc, alpha=alpha)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < deadline - 120:
        h_bin = binary_round(best_h, n)
        h_bin, c_bin = binary_local_search(h_bin, n, time.time() + min(80, deadline - time.time() - 100))
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        if time.time() < deadline - 60:
            h_cd = h_bin.copy()
            h_cd, c_cd = coord_descent(h_cd, n, time.time() + min(50, deadline - time.time() - 40), alpha=5000)
            h_cd = project_feasible(h_cd, n)
            c_cd = compute_c5(h_cd, n)
            if c_cd < best_c:

    if time.time() < deadline - 40:

    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
