```python
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full') * dx

    w = np.exp(shifted)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    """Compute gradient of LSE(alpha * c_k) objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (corr - corr.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - n1
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        # First term: (1-h[j+k]) for valid j
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_start < j_end:
            grad[j_start:j_end] += wk * (1.0 - h[j_start + k:j_end + k])
        # Second term: -h[j-k] for valid j
        j_start2 = max(0, k)
        j_end2 = min(n, n + k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= wk * h[j_start2 - k:j_end2 - k]
    return grad

def lse_subgradient(h, n, alpha, lr, n_steps, deadline, verbose=False):
    """Run LSE smoothed projected subgradient."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        grad = compute_gradient(h, n, alpha)
        h_new = project_feasible(h - lr * grad, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        # Adapt learning rate
        if c > best_c * 1.01:
            lr *= 0.9
        else:
            lr = min(lr * 1.01, 0.05)
        h = h_new
    return best_h, best_c
```

```python
    h = h.copy().astype(float)
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if lr is None:
        # Estimate Lipschitz constant (rough)
        lr = 0.01
        
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Gradient at y
        grad = compute_gradient(y, n, alpha)
        
        # Gradient step
        x = y - lr * grad
        
        # Project
        x = project_feasible(x, n)
        
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - h)
        # Project y too (important for convergence)
        y = project_feasible(y, n)
        
        h = x
        t = t_new
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_lse_gradient(y, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(y, 1.0 - y, mode='full') * dx
    shifted = alpha * (corr - corr.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w /= w_sum
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(2*n - 1):
        k = k_idx - n1
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        js = max(0, -k)
        je = min(n, n - k)
        if js < je:
            grad[js:je] += wk * (1.0 - y[js+k:je+k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if js2 < je2:
            grad[js2:je2] -= wk * y[js2-k:je2-k]
    return grad
```

```python
    sh = alpha * (corr - corr.max())

    else:

        if a2 < b2:
```

```python
def compute_c5(h, n):

    dx = 2.0 / n
    corr = np.correlate(y, 1.0 - y, mode='full') * dx

                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def make_delta(i, j, h, mr, n, n1):
    delta = np.zeros(2 * n - 1)
    im = i + mr
    m1 = (im >= 0) & (im < n)
    delta[m1] += 1.0 - h[im[m1]]
    im2 = i - mr
    m2 = (im2 >= 0) & (im2 < n)
    delta[m2] -= h[im2[m2]]
    jm = j + mr
    m3 = (jm >= 0) & (jm < n)
    delta[m3] -= 1.0 - h[jm[m3]]
    jm2 = j - mr
    m4 = (jm2 >= 0) & (jm2 < n)
    delta[m4] += h[jm2[m4]]
    delta[n1] -= 2.0
    delta[n1 + j - i] += 1.0
    delta[n1 + i - j] += 1.0
    return delta

def hill_climb(hb, n, deadline, seed=42, T0=0.004, cool=0.999995):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    n1 = n - 1
    mr = np.arange(-n1, n)
    h = hb.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = T0
    cnt = n // 2
    for _ in range(10**9):
        if time.time() > deadline:
            break
        ai = rng.randint(cnt)
        bi = rng.randint(cnt)
        i, j = int(zeros[ai]), int(ones[bi])
        delta = make_delta(i, j, h, mr, n, n1)
        nc = (corr + delta).max() * dx
        if nc < cur_c:
            accept = True
        elif T < 1e-14:
            accept = False
        else:
            accept = rng.random() < np.exp(-(nc - cur_c) / T)
        if accept:
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def greedy_sweep(hb, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    n1 = n - 1
    mr = np.arange(-n1, n)
    h = hb.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c - 1e-16
            for j in ones:
                delta = make_delta(i, j, h, mr, n, n1)
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                delta = make_delta(i, best_j, h,mr, n, n1)
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones.discard(best_j)
                ones.add(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # Phase 1: FISTA with increasing alpha
    p1_end = t0 + 0.15 * (budget_s - 5)
    alphas_p1 = [50, 200, 1000, 5000]
    hh = h.copy()
    for a in alphas_p1:
        if time.time() > p1_end - 1:
            break
        hh, cc = fista(hh, n, a, min(time.time() + 15, p1_end))
        if cc < best_c:
            best_c, best_h = cc, hh.copy()
    # Phase 2: Binary optimization
    p2_end = t0 + 0.45 * (budget_s - 5)
    hb = to_binary(best_h, n)
    _, cb = greedy_sweep(hb, n, min(time.time() + 20, p2_end - 100), max_sweeps=200)
    hb, cb = hill_climb(hb, n, p2_end, seed=seed, T0=0.005, cool=0.999997)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 3: FISTA from binary
    p3_end = t0 + 0.65 * (budget_s - 5)
    hh = best_h.copy()
    for a in [1000, 10000, 50000, 200000]:
        if time.time() > p3_end - 1:
            break
        hh, cc = fista(hh, n, a, min(time.time() + 20, p3_end))
        if cc < best_c:
            best_c, best_h = cc, hh.copy()
    # Phase 4: More binary hill-climb with reheat
    p4_end = t0 + 0.82 * (budget_s - 5)
    hb = to_binary(best_h, n)
    _, cb = greedy_sweep(hb, n, min(time.time() + 15, p4_end - 50), max_sweeps=150)
    hb, cb = hill_climb(hb, n, p4_end, seed=seed + 1, T0=0.003, cool=0.999996)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 5: Multi-resolution
    p5_end = t0 + 0.92 * (budget_s - 5)
    for nh in [300, 500]:
        if time.time() > p5_end - 30:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hb2 = to_binary(hh, nh)
        _, cb2 = greedy_sweep(hb2, nh, min(time.time() + 10, p5_end - 15), max_sweeps=100)
        hb2, cb2 = hill_climb(hb2, nh, min(time.time() + 20, p5_end - 5), seed=seed+2, T0=0.003)
        if cb2 < best_c:
            best_c, best_h, n = cb2, hb2.copy(), nh
    # Phase 6: Final polish
    if time.time() < deadline - 5:
        hb = to_binary(best_h, n)
        _, cb = greedy_sweep(hb, n, min(time.time() + 10, deadline - 3), max_sweeps=200)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
