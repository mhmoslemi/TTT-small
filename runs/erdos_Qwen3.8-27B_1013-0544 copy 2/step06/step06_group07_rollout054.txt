                            candidates.append(t)
            # Also add the current value
            candidates.append(old_hi)
            
            # Evaluate at each candidate
            best_t = old_hi
            best_obj = np.max(A * old_hi + B)
            for t in candidates:
                val = np.max(A * t + B)
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_t = t
            
            if abs(best_t - old_hi) > 1e-10:
                h[i] = best_t
                # Update corr
                corr += A * (best_t - old_hi)
                improved = True
        
        # Project to feasible set
        h = project_feasible(h, n)
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        
        if not improved:
            break
    
    return h, np.max(corr) * dx

Wait, the O(n²) per coordinate (for finding breakpoints) makes this O(n³) per sweep, which for n=250 is about 15.6 million * 250 = 3.9 billion operations per sweep. That's too slow!

I need a faster way to find the minimum of the max of linear functions. Instead of checking all pairs, I can:
1. Evaluate the current max
2. Only check breakpoints of the "active" correlations (those near the max)

Or better: use a bisection-like approach. The function f(t) = max_m(A_m * t + B_m) is convex and piecewise linear. I can find its minimum by:
- If f'(0) > 0, minimum is at t=0 (or the nearest breakpoint where derivative changes sign)
- If f'(1) < 0, minimum is at t=1
- Otherwise, the minimum is at a breakpoint where the derivative changes from negative to positive

The derivative at t is: A[m*] where m* is the active constraint (the m that achieves the max). At a breakpoint between m1 and m2, the derivative changes from A[m1] to A[m2].

So the algorithm is:
1. Start at t=0
2. Find the active m (max A[m]*0 + B[m] = max B[m])
3. If A[m_active] >= 0, the function is non-decreasing, so optimum is at t=0
4. Otherwise, move to the next breakpoint (where another m becomes active)
5. Repeat until we find where the slope changes sign

The number of breakpoints we visit is at most the number of distinct "active" m values, which is typically small (maybe 5-20).

But finding the next breakpoint still requires checking all m values. So it's O(n) per breakpoint, and we might visit O(n) breakpoints in the worst case, giving O(n²) per coordinate.

For n=250: 250² = 62500 per coordinate, times 250 coordinates = 15.6 million per sweep. That's very fast!

Let me optimize the implementation:

```python
def coord_optimize(h, corr, i, n):
    """Find optimal h[i] given all other h values fixed."""
    old_hi = h[i]
    A = np.zeros(2*n - 1)
    
    # Case 1: h[i] appears as h[i] * (1-h[j]) where j = i + m - (n-1)
    m_lo1 = max(n - 1 - i, 0)
    m_hi1 = min(2*n - 2 - i, 2*n - 2)
    for m in range(m_lo1, m_hi1 + 1):
        j = i + m - (n - 1)
        if 0 <= j < n:
            A[m] += (1.0 - h[j])
    
    # Case 2: h[i] appears as (1-h[i]) * h[j] where j = i - m + (n-1)
    m_lo2 = max(i, 0)
    m_hi2 = min(i + n - 1, 2*n - 2)
    for m in range(m_lo2, m_hi2 + 1):
        j = i - m + (n - 1)
        if 0 <= j < n:
            A[m] -= h[j]
    
    B = corr - A * old_hi
    
    # Find min of max_m(A[m]*t + B[m]) for t in [0,1]
    # f(t) is convex piecewise linear
    # Evaluate at t=0 and t=1
    f0 = np.max(B)  # f(0)
    f1 = np.max(A + B)  # f(1)
    
    if f0 <= f1 + 1e-12:
        # Check if t=0 is optimal: is f non-decreasing at 0?
        # f'(0+) = max A[m] for m that achieves f(0)
        active0 = np.argmax(B)
        if A[active0] >= 0:
            return 0.0, f0
        # Otherwise need to find breakpoint
    
    if f1 <= f0 + 1e-12:
        active1 = np.argmax(A + B)
        if A[active1] <= 0:
            return 1.0, f1
        # Otherwise need to find breakpoint
    
    # Find the minimum by walking breakpoints
    # Start from t=0, find where the active constraint changes
    t = 0.0
    f = f0
    active = np.argmax(B)  # m that achieves max at t=0
    
    # Walk forward
    while A[active] < 0 and t < 1.0:
        # Current active m has negative slope, so f is decreasing
        # Find next breakpoint: min over m≠active of (B[m]-B[active])/(A[active]-A[m]) for those where A[active] < A[m]
        # At the breakpoint, A[active]*t + B[active] = A[m]*t + B[m]
        # t = (B[m] - B[active]) / (A[active] - A[m])
        # We need the smallest positive t where a different m becomes active
        
        curr_vals = A * t + B  # values of all lines at current t
        curr_max = curr_vals[active]
        # Find which m will become active next (smallest t where it crosses above)
        # For m with A[m] > A[active]: crosses at t = (B[m] - B[active]) / (A[active] - A[m])
        # Since A[active] < A[m], denominator is negative, so we need B[m] < B[active] for positive t
        # i.e., (B[m] - B[active]) / (A[active] - A[m]) > 0 requires B[m] < B[active] when A[m] > A[active]
        
        # For m with A[m] < A[active]: never crosses above (stays below)
        # For m with A[m] == A[active]: parallel, never crosses (or always same)
        
        # So we only need to consider m with A[m] > A[active] and B[m] < B[active] (at t=0, but actually at current t)
        # At current t: curr_vals[m] < curr_vals[active] for m ≠ active (by definition of active)
        # t_cross = (curr_vals[m] - curr_vals[active]) / (A[active] - A[m]) for A[m] > A[active]
        # Since curr_vals[m] < curr_vals[active] and A[active] < A[m], 
        # numerator < 0, denominator < 0, so t_cross > 0. Good.
        
        mask = A > A[active]
        if not np.any(mask):
            # No line has higher slope; f decreases forever (but bounded by t=1)
            t = 1.0
            f = np.max(A + B)
            active = np.argmax(A + B)
            break
        
        # Compute crossing times
        curr_vals_other = curr_vals[mask]
        t_cross = (curr_vals_other - curr_max) / (A[active] - A[mask])
        # All should be positive
        t_cross = t_cross[t_cross > t + 1e-15]
        
        if len(t_cross) == 0:
            t = 1.0
            f = np.max(A + B)
            active = np.argmax(A + B)
            break
        
        next_t = max(t, min(t_cross))
        if next_t >= 1.0:
            t = 1.0
            f = np.max(A + B)
            active = np.argmax(A + B)
            break
        
        t = next_t
        f = np.max(A * t + B)
        active = np.argmax(A * t + B)
    
    # Similarly check walking backward from t=1
    # Actually, since f is convex, the minimum is unique (or an interval)
    # The above should find it
    
    return t, f
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, n_grid=200):
    """Coordinate descent: optimize each h[i] while keeping sum fixed via projection."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    target = n / 2.0
    
    for i in range(n):
        old_hi = h[i]
        # Compute A and B such that corr[m] = A[m]*h[i] + B[m]
        A = np.zeros(2 * n - 1)
        
        # h[i] appears in corr[m] as:
        # Term 1: when the first argument index is i: h[i]*(1-h[i+m-(n-1)])
        #   valid when 0 <= i+m-(n-1) < n
        # Term 2: when the second argument index is i: h[i-m+(n-1)]*(1-h[i])
        #   valid when 0 <= i-m+(n-1) < n
        
        m1_lo = max(0, n - 1 - i)
        m1_hi = min(2*n - 2, 2*n - 2 - i)
        for m in range(m1_lo, m1_hi + 1):
            j = i + m - (n - 1)
            A[m] += (1.0 - h[j])
        
        m2_lo = max(0, i)
        m2_hi = min(2*n - 2, i + n - 1)
        for m in range(m2_lo, m2_hi + 1):
            j = i - m + (n - 1)
            A[m] -= h[j]
        
        B = corr - A * old_hi
        
        # Find t in [0,1] that minimizes max_m(A[m]*t + B[m])
        # Use grid search
        t_grid = np.linspace(0, 1, n_grid)
        vals = np.max(A[None, :] * t_grid[:, None] + B[None, :], axis=1)
        best_idx = np.argmin(vals)
        best_t = t_grid[best_idx]
        
        # Refine around best
        if best_idx > 0 and best_idx < n_grid - 1:
            lo = t_grid[max(0, best_idx - 2)]
            hi = t_grid[min(n_grid - 1, best_idx + 2)]
            t_fine = np.linspace(lo, hi, 50)
            vals_fine = np.max(A[None, :] * t_fine[:, None] + B[None, :], axis=1)
            best_idx2 = np.argmin(vals_fine)
            best_t = t_fine[best_idx2]
        
        if abs(best_t - old_hi) > 1e-10:
            h[i] = best_t
            corr += A * (best_t - old_hi)
    
    # Project to feasible
    h = project_feasible(h, n)
    c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return h, c5

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.0001, patience=30000):
    """Simulated annealing on binary h with block moves."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start
    step = 0
    
    while time.time() < deadline:
        # Choose move type: single swap (70%) or block swap (30%)
        if rng.random() < 0.7:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
            delta = compute_swap_delta(h, n, i, j, m_range)
        else:
            # Block move: swap a block of length L
            L = rng.randint(2, 6)
            # Find a zero block and a one block
            i = rng.randint(0, n - L)
            j = rng.randint(0, n - L)
            # Check if i has zeros and j has ones (approximately)
            # For simplicity, just do individual swaps in the block
            delta = np.zeros(2*n - 1)
            for offset in range(L):
                ii = i + offset
                jj = j + offset
                if h[ii] == 0 and h[jj] == 1:
                    delta += compute_swap_delta(h, n, ii, jj, m_range)
                    h[ii] = 1.0  # tentatively
                    h[jj] = 0.0  # tentatively
            # If delta is all zeros, skip
            if np.allclose(delta, 0):
                step += 1
                T *= cool_rate
                continue
            # Revert tentative changes
            for offset in range(L):
                ii = i + offset
                jj = j + offset
                # restore
                pass  # This is getting too complicated, skip block moves for now
        
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-min(delta_c / T, 50))):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        step += 1
        T *= cool_rate
        if no_improve > patience:
            # Restart with random perturbation
            nk = rng.randint(5, 15)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    
    return best_h, best_c

def compute_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

This is still max of linear functions in h[i] (since corr[m] is linear in h[i], and λ*h[i] is linear). So the same approach works!

def coord_sweep_lagrangian(h, n, n_grid=200, lambda_update=True):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    target = n / 2.0
    lam = 0.0
    
    for i in range(n):
        old_hi = h[i]
        A = np.zeros(2 * n - 1)
        
        m1_lo = max(0, n - 1 - i)
        m1_hi = min(2*n - 2, 2*n - 2 - i)
        for m in range(m1_lo, m1_hi + 1):
            j = i + m - (n - 1)
            A[m] += (1.0 - h[j])
        
        m2_lo = max(0, i)
        m2_hi = min(2*n - 2, i + n - 1)
        for m in range(m2_lo, m2_hi + 1):
            j = i - m + (n - 1)
            A[m] -= h[j]
        
        B = corr - A * old_hi
        
        # Minimize max_m(A[m]*t + B[m]) + lam * t
        # = max_m(A[m]*t + B[m]) + lam * t
        # Equivalent to max_m(A[m]*t + B[m] + lam*t) if we absorb lam
        # Actually: max_m(f_m(t)) + lam*t where f_m(t) = A[m]*t + B[m]
        # This is NOT the same as max_m(f_m(t) + lam*t) in general!
        # So I need to evaluate g(t) = max_m(A[m]*t + B[m]) + lam*t
        
        t_grid = np.linspace(0, 1, n_grid)
        max_vals = np.max(A[None, :] * t_grid[:, None] + B[None, :], axis=1)
        g_vals = max_vals + lam * t_grid
        best_idx = np.argmin(g_vals)
        best_t = t_grid[best_idx]
        
        # Refine
        if best_idx > 0 and best_idx < n_grid - 1:
            lo = t_grid[max(0, best_idx - 2)]
            hi = t_grid[min(n_grid - 1, best_idx + 2)]
            t_fine = np.linspace(lo, hi, 50)
            max_fine = np.max(A[None, :] * t_fine[:, None] + B[None, :], axis=1)
            g_fine = max_fine + lam * t_fine
            best_idx2 = np.argmin(g_fine)
            best_t = t_fine[best_idx2]
        
        if abs(best_t - old_hi) > 1e-10:
            h[i] = best_t
            corr += A * (best_t - old_hi)
    
    # Update lambda to satisfy sum constraint
    diff = target - h.sum()
    lam += 0.1 * diff  # subgradient update
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return h, c5, lam

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, n_grid=200):

def compute_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta
