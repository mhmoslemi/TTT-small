```python
def compute_slopes(h, j, n):
    # Returns array s of length 2n-1 where s[k] = dc[k]/dh[j]
    k = np.arange(2*n-1)
    d = k - (n-1)  # shift, from -(n-1) to n-1
    
    s = np.zeros(2*n-1)
    # First term: (1-h[j-d]) for d where 0 <= j-d < n, i.e., j-n+1 <= d <= j
    mask1 = (d >= j-n+1) & (d <= j)
    # j-d must be valid index: 0 <= j-d < n
    jd = j - d
    mask1 = mask1 & (jd >= 0) & (jd < n)
    s[mask1] += (1 - h[jd[mask1]])
    
    # Second term: -h[j+d] for d where 0 <= j+d < n, i.e., -j <= d < n-j
    mask2 = (d >= -j) & (d < n-j)
    jd2 = j + d
    mask2 = mask2 & (jd2 >= 0) & (jd2 < n)
    s[mask2] -= h[jd2[mask2]]
    
    return s
```

        # For all zeros z and all ones o, compute new_max = max_k (corr[k] + slope_k[z] - slope_k[o])
        # where slope_k[i] is the slope of c[k] with respect to h[i]
        
        # Compute all slopes at once:
        j_idx = np.arange(n)[:, None]  # (n, 1)
        d_vals = np.arange(2*n-1)[None, :] - (n-1)  # (1, 2n-1)
        jd1 = j_idx - d_vals  # (n, 2n-1)
        jd2 = j_idx + d_vals  # (n, 2n-1)
        m1 = (jd1 >= 0) & (jd1 < n)
        m2 = (jd2 >= 0) & (jd2 < n)
        jdc1 = np.clip(jd1, 0, n-1)
        jdc2 = np.clip(jd2, 0, n-1)
        slopes = np.zeros((n, 2*n-1))
        slopes[m1] = 1.0 - h[jdc1[m1]]
        slopes[m2] -= h[jdc2[m2]]
        
        # For swap (z, o): new_corr[k] = corr[k] + slopes[z, k] - slopes[o, k]
        # new_max = max_k new_corr[k]
        # We want to minimize new_max over all (z, o) pairs.
        
        # This is: min_{z in zeros, o in ones} max_k (corr[k] + slopes[z,k] - slopes[o,k])
        
        # For a fixed set of active k values:
        active = top_idx
        corr_act = corr[active]  # (top_k,)
        sl_z = slopes[zeros, :][:, active]  # (|zeros|, top_k)
        sl_o = slopes[ones, :][:, active]  # (|ones|, top_k)
        
        # For each (z, o): new_corr_act[k] = corr_act[k] + sl_z[z,k] - sl_o[o,k]
        # new_max >= max_k new_corr_act[k]
        # = max_k (corr_act[k] + sl_z[z,k]) - min... no, it's max_k (corr_act[k] + sl_z[z,k] - sl_o[o,k])
        
        # This is max_k (A[z,k] - B[o,k]) where A[z,k] = corr_act[k] + sl_z[z,k] and B[o,k] = sl_o[o,k]
        # = max_k A[z,k] - min_k B[o,k]? NO! max of (A-B) != max(A) - min(B) in general.
        
        # The correct computation: for each (z,o), new_max_partial = max_k (corr_act[k] + sl_z[z,k] - sl_o[o,k])
        # This is (|zeros| * |ones| * top_k) operations. For n=250, that's 125*125*15 = 234K. Fine.
        
        # Vectorized:
        # D[z, o, k] = corr_act[k] + sl_z[z, k] - sl_o[o, k]
        # new_max[z, o] = max_k D[z, o, k]
        
        # Memory: 125 * 125 * 15 = 234K floats = 2MB. Fine.
        D = corr_act[None, None, :] + sl_z[:, None, :] - sl_o[None, :, :]
        new_maxes = D.max(axis=2)  # (|zeros|, |ones|)
        
        flat = new_maxes.argmin()
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = new_maxes[ii, jj]
        z = zeros[ii]
        o = ones[jj]
        new_c = best_val * dx
        
        if new_c < cur_c - 1e-14:
            # Perform swap
            h[z] = 1.0
            h[o] = 0.0
            # Recompute corr
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            # Update zeros and ones
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, o)
            ones = np.append(ones, z)
            
            stall = 0
            temp = max(0.0005, temp * 0.999)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 50 or (no_improve > 300 and rng.rand() < 0.1):
                # Random kick: swap a random pair
                ri = rng.randint(len(zeros))
                rj = rng.randint(len(ones))
                h[zeros[ri]] = 1.0
                h[ones[rj]] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                zeros = np.delete(zeros, ri)
                ones = np.delete(ones, rj)
                zeros = np.append(zeros, ones[rj])
                ones = np.append(ones, zeros[ri])
                temp = max(0.0005, temp * 0.9)
                stall = 0
    
    return best_h, best_c

def sqp_continuous(h_start, n, deadline):
    """Sequential quadratic programming using linearization + LP"""
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, n)
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    
    lam = 1.0
    no_improve = 0
    
    # Precompute the slope matrix structure
    j_idx = np.arange(n)[:, None]
    d_vals = np.arange(2*n-1)[None, :] - (n-1)
    jd1 = j_idx - d_vals
    jd2 = j_idx + d_vals
    m1 = (jd1 >= 0) & (jd1 < n)
    m2 = (jd2 >= 0) & (jd2 < n)
    jdc1 = np.clip(jd1, 0, n-1)
    jdc2 = np.clip(jd2, 0, n-1)
    # m1, m2, jdc1, jdc2 are fixed (don't depend on h)
    
    while time.time() < deadline:
        # Compute slopes at current h
        slopes = np.zeros((n, 2*n-1))
        slopes[m1] = 1.0 - h[jdc1[m1]]
        slopes[m2] -= h[jdc2[m2]]
        
        # Constants (current correlation)
        C = np.correlate(h, 1-h, mode='full').astype(float)
        
        # Solve LP: min t s.t. C_k + sum_j slopes[j,k]*h'[j] <= t, sum h' = n/2, 0<=h'<=1
        # LP variables: h' (n) + t (1) = n+1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = slopes.T
        A_ub[:, n] = -1.0
        b_ub = -C
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                lam *= 0.5
                if lam < 0.001:
                    break
                continue
        except Exception:
            break
        
        h_lp = res.x[:n]
        t_lp = res.x[n]
        
        # Damped step
        h_new = (1 - lam) * h + lam * h_lp
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < cur_c - 1e-12:
            h = h_new
            cur_c = c_new
            no_improve = 0
            lam = min(1.0, lam * 1.1)  # increase step size
            if cur_c < best_c - 1e-12:
                best_c = cur_c
                best_h = h.copy()
        else:
            lam *= 0.5
            no_improve += 1
            if no_improve > 50:
                # Try a different approach: perturb and restart
                h = project_feasible(h + 0.01 * np.random.randn(n), n)
                cur_c = compute_c5(h, n)
                lam = 0.5
                no_improve = 0
    
    return best_h, best_c
