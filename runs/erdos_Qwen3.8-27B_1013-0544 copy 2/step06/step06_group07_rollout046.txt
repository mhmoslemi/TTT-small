def build_g_matrix(h, n):
    # g_j[shift_index] for each one position j
    # shift_index = shift + (n-1), ranges from 0 to 2n-2
    ones = np.where(h == 1)[0]
    n_ones = len(ones)
    G = np.zeros((n_ones, 2*n-1))
    for idx, j in enumerate(ones):
        # g_j[shift] = h[j+shift]*1_{valid} + (h[j-shift]-1)*1_{valid}
        # shift ranges from -(n-1) to n-1
        # j+shift valid: 0 <= j+shift <= n-1, i.e., -j <= shift <= n-1-j
        # j-shift valid: 0 <= j-shift <= n-1, i.e., j-n+1 <= shift <= j
        shifts = np.arange(-(n-1), n)
        # First term: h[j+shift] where valid
        js = j + shifts
        mask1 = (js >= 0) & (js < n)
        G[idx, mask1] += h[js[mask1]]
        # Second term: (h[j-shift] - 1) where valid
        js2 = j - shifts
        mask2 = (js2 >= 0) & (js2 < n)
        G[idx, mask2] += (h[js2[mask2]] - 1)
    return G, ones

def build_f_vectors(h, n):
    # f_i[shift] for each zero position i
    zeros = np.where(h == 0)[0]
    F = np.zeros((len(zeros), 2*n-1))
    for idx, i in enumerate(zeros):
        shifts = np.arange(-(n-1), n)
        # f_i[shift] = -h[i+shift]*1_{valid} + (1-h[i-shift])*1_{valid}
        is_val = i + shifts
        mask1 = (is_val >= 0) & (is_val < n)
        F[idx, mask1] += -h[is_val[mask1]]
        is_val2 = i - shifts
        mask2 = (is_val2 >= 0) & (is_val2 < n)
        F[idx, mask2] += (1 - h[is_val2[mask2]])
    return F, zeros

def steepest_swap(h, n):
    # Find the best single swap (i from zeros, j from ones)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    
    F, zeros = build_f_vectors(h, n)
    G, ones = build_g_matrix(h, n)
    
    best_val = cur_max
    best_i = -1
    best_j = -1
    
    for ii in range(len(zeros)):
        B = corr + F[ii]
        # For all j: max(B + G[j])
        vals = (B[None, :] + G).max(axis=1)
        min_val = vals.min()
        if min_val < best_val:
            best_val = min_val
            best_i = ii
            best_j = vals.argmin()
    
    if best_i < 0:
        return h, cur_max  # No improving swap found
    
    i = zeros[best_i]
    j = ones[best_j]
    h[i] = 1.0
    h[j] = 0.0
    return h, best_val * dx

        no = len(ones)

            break

        best_ii = -1
        best_jj = -1

        # This is nz × no × (2n-1) operations

                best_ii = ii
                best_jj = jmin

            break  # Local optimum

        j = ones[best_jj]
        h[i] = 1.0
        h[j] = 0.0
        corr = np.correlate(h, 1-h, mode='full').astype(float)
        cur_max = corr.max()
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_h = h.copy()

def perturb_binary(h, n, k, rng):
    """Make k random swaps."""
    h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]

        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        # Update zeros and ones
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
    return h

def ils_binary(h_init, n, deadline, seed=42):
    """Iterated Local Search on binary solutions."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h, best_c = steepest_descent(h, n, deadline=deadline)
    
    while time.time() < deadline:

        k = 3 + rng.randint(8)  # 3 to 10 swaps
        h_pert = perturb_binary(best_h, n, k, rng)
        # Local search
        h_new, c_new = steepest_descent(h_pert, n, deadline=deadline)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        # Occasionally accept non-improving solutions (like in SA)
        elif rng.random() < 0.1:
            best_h = h_new
    
    return best_h, best_c

def slsqp_optimize(h_init, n, alpha, maxiter=2000, ftol=1e-15, deadline=None):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh_max = sh.max()
        return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    def constraint(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': constraint}]
    try:
        if deadline is not None and time.time() > deadline:
            return h_init
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def build_F_G(h, n):
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h >= 0.5)[0]
    nz = len(zeros)
    no = len(ones)
    F = np.zeros((nz, 2 * n - 1))
    G = np.zeros((no, 2 * n - 1))
    shifts = np.arange(-(n - 1), n)
    for idx in range(nz):
        i = zeros[idx]
        is1 = i + shifts
        m1 = (is1 >= 0) & (is1 < n)
        F[idx, m1] += -h[is1[m1]]
        is2 = i - shifts
        m2 = (is2 >= 0) & (is2 < n)
        F[idx, m2] += (1.0 - h[is2[m2]])
    for idx in range(no):
        j = ones[idx]
        js1 = j + shifts
        m1 = (js1 >= 0) & (js1 < n)
        G[idx, m1] += h[js1[m1]]
        js2 = j - shifts
        m2 = (js2 >= 0) & (js2 < n)
        G[idx, m2] += (h[js2[m2]] - 1.0)
    return F, zeros, G, ones

        h[i] = 1.0
        h[j] = 0.0

        swaps_done += 1
