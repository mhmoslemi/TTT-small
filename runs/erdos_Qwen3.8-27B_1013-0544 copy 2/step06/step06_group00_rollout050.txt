import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 0.005) & (h < 0.995)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx
    f_s = f - f.max()
    w = np.exp(alpha * f_s)
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(n):
        ml1 = max(0, i)
        mh1 = min(2*n-2, i+n-1)
        if ml1 <= mh1:
            idx1 = i - np.arange(ml1, mh1+1) + n - 1
            grad[i] += np.dot(w[ml1:mh1+1], omh[idx1])
        ml2 = max(0, n-1-i)
        mh2 = min(2*n-2, 2*n-2-i)
        if ml2 <= mh2:
            idx2 = i + np.arange(ml2, mh2+1) - n + 1
            grad[i] -= np.dot(w[ml2:mh2+1], h[idx2])
    return grad * dx

def nesterov_opt(h, n, alpha, deadline, lr_init=0.1, max_iter=5000):

            f_new = (np.log(np.exp(alpha * (np.correlate(x_new, 1-x_new, mode='full')*dx - np.max(np.correlate(x_new, 1-x_new, mode='full')*dx))).sum()) / alpha if alpha < 1e6 else np.max(np.correlate(x_new, 1-x_new, mode='full'))*dx

def greedy_binary(h_init, n, deadline, max_sweeps=500):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk2 = (imn >= 0) & (imn < n)
                delta[msk2] -= h[imn[msk2]]
                jm = j + m_range
                msk3 = (jm >= 0) & (jm < n)
                delta[msk3] -= 1.0 - h[jm[msk3]]
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                delta[msk4] += h[jmn[msk4]]
                delta[n-1] -= 2.0
                delta[n-1+j-i] += 1.0
                delta[n-1+i-j] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:

        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
