```
def project_sum_box(z, n):
    z = np.asarray(z, dtype=float)
    target = float(n) / 2.0
    if z.size != n:
```

```python
import numpy as np
import time
import math
from scipy.optimize import minimize, linprog

def lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def val5(h, n):
    return float(lags(h, n).max())

def proj(z, n):
    z = np.asarray(z, dtype=float)
    if z.size != n:
        z = z[:n] if z.size > n else np.pad(z, (0, n - z.size))
    z = np.clip(z, 0.0, 1.0)
    t = 0.5 * n
    lo = float(z.min()) - n - 1.0
    hi = float(z.max()) + n + 1.0
    for _ in range(72):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > t:
            lo = mid
        else:
            hi = mid
    z = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(12):
        d = t - float(z.sum())
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - z
            s = float(room.sum())
            if s > 1e-14:
                z += (d / s) * room
        else:
            s = float(z.sum())
            if s > 1e-14:
                z += (d / s) * z
    return np.clip(z, 0.0, 1.0)

def lag_val(h, k):
    n = len(h)
    m = abs(int(k))
    dx = 2.0 / float(n)
    if m >= n:
        return 0.0
    if m == 0:
        return float(dx * np.sum(h * (1.0 - h)))
    if k > 0:
        return float(dx * np.sum(h[m:] * (1.0 - h[:n - m])))
    return float(dx * np.sum(h[:n - m] * (1.0 - h[m:])))

def lag_grad(h, k):
    n = len(h)
    m = abs(int(k))
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if m >= n:
        return g
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g * dx

def sel_act(c, k, band):
    m = float(c.max())
    near = np.where(c >= m - band)[0]
    top = np.argsort(c)[::-1][:k]
    idx = np.unique(np.concatenate((top, near)))
    if idx.size > k:
        idx = idx[np.argsort(c[idx])[::-1]][:k]
    return idx

def act_grad(h, n, c, alpha, top, band):
    m = float(c.max())
    sel = np.where(c >= m - band)[0]
    if sel.size == 0:
        sel = np.array([int(np.argmax(c))])
    elif sel.size > top:
        sel = sel[np.argsort(c[sel])[::-1]][:top]
    ac = c[sel]
    a = float(ac.max())
    e = np.exp(alpha * (ac - a))
    s = float(e.sum())
    w = e / s
    g = np.zeros(n, dtype=float)
    cen = n - 1
    for ix, wi in zip(sel, w):
        if wi > 1e-13:
            g += wi * lag_grad(h, int(ix) - cen)
    F = float(a + math.log(s) / alpha) if s > 0 else a
    return g, F, sel

def eq_sum(x, n):
    return float(np.sum(x) - 0.5 * n)

def eq_jac(x, n):
    return np.ones(x.size, dtype=float)

def sm_obj(x, n, act, alpha):
    c = np.empty(act.size, dtype=float)
    cen = n - 1
    for i, k in enumerate(act):
        c[i] = lag_val(x, int(k) - cen)
    if c.size == 1:
        return float(c[0])
    m = float(c.max())
    s = float(np.exp(alpha * (c - m)).sum())
    return float(m + math.log(s) / alpha)

def sm_jac(x, n, act, alpha):
    c = np.empty(act.size, dtype=float)
    cen = n - 1
    for i, k in enumerate(act):
        c[i] = lag_val(x, int(k) - cen)
    if c.size == 1:
        return lag_grad(x, int(act[0]) - cen)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    w = e / float(e.sum())
    g = np.zeros(n, dtype=float)
    for i, k in enumerate(act):
        if w[i] > 1e-13:
            g += w[i] * lag_grad(x, int(k) - cen)
    return g

def sm_once(h, n, act, alpha, mi):
    try:
        res = minimize(sm_obj, h, args=(n, act, alpha), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_sum, 'jac': eq_jac, 'args': (n,)}],
                       jac=sm_jac, options={'maxiter': mi, 'ftol': 1e-12})
        if res.x is not None:
            return proj(res.x, n)
    except Exception:
        pass
    return h.copy()

def smooth_loop(h, n, deadline, rng):
    h = proj(h, n)
    best = h.copy()
    c = lags(h, n)
    bc = float(c.max())
    r = 0
    stall = 0
    while time.time() < deadline - 3 and r < 60:
        rem = deadline - time.time()
        if rem < 8:
            break
        mk = min(34, 8 + 2 * r)
        band = max(0.0008, 0.005 - 0.00012 * r)
        act = sel_act(c, mk, band)
        mi = 90 if rem > 120 else (50 if rem > 45 else 25)
        if mk > 25:
            mi = max(22, mi // 2)
        base = 2000.0 if r % 3 == 0 else (8000.0 if r % 3 == 1 else 30000.0)
        cur_m = float(c.max())
        imp = False
        for aa in (base, 4.0 * base):
            if time.time() > deadline - 5:
                break
            hn = sm_once(h, n, act, aa, mi)
            cn = lags(hn, n)
            mn = float(cn.max())
            if mn < bc - 1e-14:
                bc = mn

                h = hn

                h = hn

        if not imp:

                    h = best.copy()
                    c = lags(h, n)
                stall = max(0, stall - 5)
            elif stall >= 8:

        r += 1
    return best, bc

def lp_step(h, n, trust, act):
    h = proj(h, n)
    c = lags(h, n)
    cen = n - 1
    idx = np.arange(len(c)) if act is None or len(act) == 0 else np.asarray(act)
    rows = []
    rhs = []
    for k in idx:
        ki = int(k)
        g = lag_grad(h, ki - cen)
        row = np.zeros(n + 1, dtype=float)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        rhs.append(float(g.dot(h) - c[ki]))
    if trust > 0:
        for i in range(n):
            r1 = np.zeros(n + 1, dtype=float)
            r1[i] = 1.0
            rows.append(r1)
            rhs.append(trust + float(h[i]))
            r2 = np.zeros(n + 1, dtype=float)
            r2[i] = -1.0
            rows.append(r2)
            rhs.append(trust - float(h[i]))
    try:
        A = np.vstack(rows)
        b = np.asarray(rhs, dtype=float)
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[:n] = 1.0
        beq = np.array([0.5 * n], dtype=float)
        cv = np.zeros(n + 1, dtype=float)
        cv[-1] = 1.0
        res = linprog(cv, A, b, A_eq=Aeq, b_eq=beq,
                      bounds=[(0.0, 1.0)] * (n + 1), method='highs')
        if res.success and res.x is not None:
            return proj(res.x[:n], n)
    except Exception:
        pass
    return h

def pair_polish(h, n, deadline, rng):
    h = proj(h, n)
    best = h.copy()
    bc = float(lags(h, n).max())
    for d in (0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001):
        if time.time() > deadline - 2:
            break
        st = 0
        while time.time() < deadline - 2 and st < 10:
            c = lags(h, n)
            M = float(c.max())
            G, _, _ = act_grad(h, n, c, 3000.0, 16, 0.003)
            hi = np.where(h > 1e-7)[0]
            lo = np.where(h < 1.0 - 1e-7)[0]
            if hi.size == 0 or lo.size == 0:
                break
            thi = hi[np.argsort(-G[hi])][:14]
            tlo = lo[np.argsort(G[lo])][:14]
            found = False
            for i in thi:
                for j in tlo:
                    if i == j or G[i] <= G[j]:
                        continue
                    delta = min(d, float(h[i]), 1.0 - float(h[j]))
                    if delta < 1e-9:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    cn = lags(h, n)
                    mn = float(cn.max())
                    if mn < M - 1e-14:
                        c = cn
                        M = mn
                        if mn < bc - 1e-14:
                            bc = mn
                            best = h.copy()
                        found = True
                        st = 0
                        break
                    h[i] += delta
                    h[j] -= delta
                if found:
                    break
            if not found:
                rand_hit = False
                for _ in range(6):
                    ii = int(rng.randint(hi.size))
                    jj = int(rng.randint(lo.size))
                    i = hi[ii]
                    j = lo[jj]
                    if i == j:
                        continue
                    if G[i] > G[j]:
                        ip, jp = i, j
                    else:
                        ip, jp = j, i
                    delta = min(d, float(h[ip]), 1.0 - float(h[jp]))
                    if delta < 1e-9:
                        continue
                    h[ip] -= delta
                    h[jp] += delta
                    cn = lags(h, n)
                    mn = float(cn.max())
                    if mn < bc - 1e-14:
                        c = cn
                        M = mn
                        bc = mn
                        best = h.copy()
                        rand_hit = True
                        break
                    h[ip] += delta
                    h[jp] -= delta
                if not rand_hit:
                    st += 1
    return best, bc

def interp(h, old, new):
    if old == new:
        return proj(h, new)
    return proj(np.interp(np.linspace(0.0, 2.0, new), np.linspace(0.0, 2.0, old), h), new)

def optimize_path(h, n, deadline, rng):
    h = proj(h, n)
    best = h.copy()
    bc = float(lags(h, n).max())
    c = lags(h, n)
    rem = deadline - time.time()
    if rem <= 5:
        return best, bc

    end_lp = min(deadline, time.time() + max(8.0, min(60.0, 0.2 * rem)))
    hc = best.copy()
    cc = c.copy()
    for trust, mk, use_all in ((0.35, 12, 1), (0.2, 20, 0), (0.12, 0, 1), (0.07, 30, 0), (0.04, 0, 1)):
        if time.time() > end_lp - 3:
            break
        act = None if use_all else sel_act(cc, mk, 0.004)
        hn = lp_step(hc, n, trust, act)
        cn = lags(hn, n)
        mn = float(cn.max())
        if mn < bc - 1e-14:
            bc = mn
            best = hn.copy()
            hc = hn
            cc = cn
        elif mn <= float(cc.max()) + 1e-9:
            hc = hn
            cc = cn

    end_sm = min(deadline, time.time() + max(20.0, min(240.0, 0.5 * max(0.0, deadline - time.time()))))
    cand, c2 = smooth_loop(best, n, end_sm, rng)
    if c2 < bc - 1e-14:
        bc = c2
        best = cand

    end_pa = min(deadline, time.time() + max(10.0, min(70.0, 0.4 * max(0.0, deadline - time.time()))))
    cand, c2 = pair_polish(best, n, end_pa, rng)
    if c2 < bc - 1e-14:
        bc = c2
        best = cand

    if time.time() < deadline - 15:
        hc = best.copy()
        cc = lags(hc, n)
        for trust, mk, use_all in ((0.1, 16, 0), (0.05, 0, 1), (0.02, 24, 0)):
            if time.time() > deadline - 8:
                break
            act = None if use_all else sel_act(cc, mk, 0.003)
            hn = lp_step(hc, n, trust, act)
            cn = lags(hn, n)
            mn = float(cn.max())
            if mn < bc - 1e-14:
                bc = mn
                best = hn.copy()
                hc = hn
                cc = cn
            elif mn <= float(cc.max()) + 1e-9:
                hc = hn
                cc = cn
        cand, c2 = smooth_loop(best, n, min(deadline - 3, time.time() + 25), rng)
        if c2 < bc - 1e-14:
            bc = c2
            best = cand
        cand, c2 = pair_polish(best, n, max(deadline - 1, time.time() + 8), rng)
        if c2 < bc - 1e-14:
            bc = c2
            best = cand
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + max(1.0, total) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 300
        x = np.linspace(0.0, 2.0, n0)
        h0 = 0.5 + 0.35 * np.cos(np.pi * x) + 0.10 * np.cos(3.0 * np.pi * x)
    n0 = len(h0)
    h0 = proj(h0, n0)

    best = h0.copy()
    bc = float(lags(h0, n0).max())

    if total > 30.0:
        main_alloc = min(0.55 * total, 650.0)
        main_end = min(deadline - 30.0, t0 + max(30.0, main_alloc))
        if main_end > time.time() + 15.0:
            cand, c2 = optimize_path(best, n0, main_end, rng)
            if c2 < bc - 1e-14:
                bc = c2
                best = cand

    ntries = [240, 200, 150, 100]
    if len(best) != 300:
        ntries = [300] + ntries
    for nt in ntries:
        if time.time() > deadline - 80.0:
            break
        if nt == len(best):
            continue
        rem = deadline - time.time() - 70.0
        if rem < 60.0:
            break
        alloc = min(90.0, max(35.0, 0.30 * rem))
        start = interp(best, len(best), nt)
        cand, c2 = optimize_path(start, nt, time.time() + alloc, rng)
        if c2 < bc - 1e-14:
            bc = c2
            best = cand

    nf = len(best)
    if time.time() < deadline - 20.0:
        cand, c2 = smooth_loop(best, nf, min(deadline - 5.0, time.time() + 50.0), rng)
        if c2 < bc - 1e-14:
            bc = c2
            best = cand
        cand, c2 = pair_polish(best, nf, deadline - 1.0, rng)
        if c2 < bc - 1e-14:
            bc = c2
            best = cand

    if time.time() < deadline - 2.0:
        hr = best.copy()
        cr = bc
        scale = 0.02
        for _ in range(2000):
            if time.time() > deadline - 1.0:
                break
            hv = proj(hr + rng.normal(0.0, scale, nf), nf)
            cv = float(lags(hv, nf).max())
            if cv < cr - 1e-14:
                cr = cv
                best = hv.copy()
                hr = hv
                scale = max(0.002, 0.90 * scale)
            else:
                scale = max(0.001, 0.98 * scale)
        if cr < bc:
            bc = cr
            best = hr

    oldh = best.copy()
    oldc = bc
    nf = len(best)
    best = proj(best, nf)
    nc = float(lags(best, nf).max())
    if nc > oldc + 1e-12:
        best = oldh
        nf = len(oldh)
        nc = oldc
    return best, nc, nf
```
