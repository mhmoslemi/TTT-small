    """Compute delta in raw correlation for swap h[i]:1→0, h[j]:0→1. O(n)."""

    # From j: s ∈ [-j, n-1-j], k = s+(n-1) ∈ [n-1-j, 2n-2-j]

For the current worst lag k* (shift s*):
- The "bad pairs" are positions i where h[i]=1 and h[i-s*]=0
- Moving a "1" from i to j (where h[j]=0) will:

  - But adds new pairs at lag k*: if h[j-s*]=0, adds a new bad pair

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    # Phase 1: Start from initial, binarize, binary search
    if h_init is not None:
        n = n_init
        h_bin = binarize(h_init, n)
        bs_end = t0 + min(180, (deadline - t0) * 0.3)
        h_b, c_b = binary_greedy(h_bin, n, bs_end)
        best_h, best_c = h_b, c_b
        # Phase 2: Continuous relaxation from binary
        t_rem = deadline - time.time()
        if t_rem > 60:
            a1 = min(60, t_rem * 0.3)
            h_new = slsqp_optimize(h_b, n, 50000, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        # Phase 3: Binarize again and search
        t_rem = deadline - time.time()
        if t_rem > 40:
            h_bin2 = binarize(best_h, n)
            bs2_end = time.time() + min(120, t_rem * 0.4)
            h_b2, c_b2 = binary_greedy(h_bin2, n, bs2_end)
            if c_b2 < best_c:
                best_c = c_b2
                best_h = h_b2
            # Continuous from new binary
            t_rem = deadline - time.time()
            if t_rem > 30:
                h_new = slsqp_optimize(best_h, n, 100000, maxiter=4000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
        # Phase 4: Final local search
        t_rem = deadline - time.time()
        if t_rem > 15:
            h_ls, c_ls = local_search_cont(best_h, n, time.time() + min(30, t_rem * 0.5))
            if c_ls < best_c:
                best_c = c_ls
                best_h = h_ls
        # Phase 5: One more binarize + search cycle
        t_rem = deadline - time.time()
        if t_rem > 60:
            h_bin3 = binarize(best_h, n)
            bs3_end = time.time() + min(90, t_rem * 0.5)
            h_b3, c_b3 = binary_greedy(h_bin3, n, bs3_end)
            if c_b3 < best_c:
                best_c = c_b3
                best_h = h_b3
            t_rem = deadline - time.time()
            if t_rem > 20:
                h_new = slsqp_optimize(best_h, n, 200000, maxiter=5000)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                h_ls, c_ls = local_search_cont(best_h, n, deadline - 2)
                if c_ls < best_c:
                    best_c = c_ls
                    best_h = h_ls
    # Try alternative n if time permits
    if best_h is not None:
        t_rem = deadline - time.time()
        if t_rem > 120:
            for n_try in [200, 400]:
                if time.time() > deadline - 80:
                    break
                x_best = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_try)
                h_start = np.interp(x_new, x_best, best_h)
                h_start = project_feasible(h_start, n_try)
                h_bin_t = binarize(h_start, n_try)
                bs_end = time.time() + min(60, (deadline - time.time() - 40) * 0.5)
                h_t, c_t = binary_greedy(h_bin_t, n_try, bs_end)
                if time.time() < deadline - 20:
                    h_new = slsqp_optimize(h_t, n_try, 100000, maxiter=3000)
                    c_new = compute_c5(h_new, n_try)
                    if c_new < c_t:
                        h_t, c_t = h_new, c_new
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, best_c = local_search_cont(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
