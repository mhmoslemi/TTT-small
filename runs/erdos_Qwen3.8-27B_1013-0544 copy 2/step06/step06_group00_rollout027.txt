import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_grad_softmax(h, n, alpha):
    """Compute gradient of soft-max approximation to max_k corr[k]*dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0
    else:
        p /= s
    # grad[i] = sum_k p[k] * ((1-h[i+k]) - h[i-k]) for valid indices
    grad = np.zeros(n)
    for k_idx in range(len(p)):
        k = k_idx - (n - 1)
        pk = p[k_idx]
        if pk < 1e-14:
            continue
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += pk * (1.0 - h[k:n])
            if k < n:
                grad[k:] -= pk * h[:n-k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += pk * (1.0 - h[:n-kk])
            if kk < n:
                grad[:n-kk] -= pk * h[kk:]
    return grad

def proj_grad_descent(h, n, alphas, deadline):
    """Projected gradient descent with increasing alpha."""
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # For each alpha, do several steps of PD
        n_steps = 5000
        lr = 2.0 / (alpha * 2 * n)  # initial learning rate
        for step in range(n_steps):
            if time.time() > deadline:
                break
            grad = compute_grad_softmax(h, n, alpha)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            c_old = compute_c5(h, n)
            if c < c_old:
                h = h_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                # Increase lr a bit
                lr = min(lr * 1.05, 1.0)
            else:
                # Decrease lr
                lr *= 0.7
                if lr < 1e-10:
                    break
    return best_h, best_c

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0

def binary_local_search(h, n, deadline, max_sweeps=5000):

    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    m_range = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_multi(h, n, alphas, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_smooth(best_h, n, alpha, 3000, 1e-16)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c, best_h = c, h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: binary local search
    p1 = t0 + 0.10 * total
    hb, cb = binary_local_search(h0, n, p1, max_sweeps=3000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    # Phase 2: SLSQP continuous optimization
    p2 = t0 + 0.50 * total
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 10:
            break
        h = slsqp_smooth(h, n, alpha, 2000, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 3: anneal for better binary
    p3 = t0 + 0.62 * total
    ha, ca = anneal_binary(best_h, n, p3, seed=seed, n_iters=2000000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    # Phase 4: binary local search from annealed
    p4 = t0 + 0.72 * total
    h4, c4 = binary_local_search(best_h, n, p4, max_sweeps=3000)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()
    # Phase 5: re-smooth
    p5 = t0 + 0.85 * total
    h = best_h.copy()
    for alpha in [10000, 100000, 1000000, 100000000]:
        if time.time() > p5 - 5:
            break
        h = slsqp_smooth(h, n, alpha, 3000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # Phase 6: try other resolutions
    p6 = t0 + 0.95 * total
    for nh in [300, 500]:
        if time.time() > p6 - 20:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hq, cq = binary_local_search(hh, nh, p6 - 15, max_sweeps=1000)
        if cq < compute_c5(hh, nh):
            hh = hq
        for alpha in [10000, 100000, 10000000]:
            if time.time() > p6 - 10:
                break
            hh = slsqp_smooth(hh, nh, alpha, 1500, 1e-15)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh
    # Phase 7: final polish
    if time.time() < dl - 3:
        for trial in range(3):
            if time.time() > dl - 4:
                break
            h = best_h + 0.01 * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [100000, 10000000]:
                if time.time() > dl - 4:
                    break
                h = slsqp_smooth(h, n, alpha, 2000, 1e-18)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
