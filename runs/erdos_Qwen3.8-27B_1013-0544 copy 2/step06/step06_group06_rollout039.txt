import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    val = m + np.log(np.exp(sh).sum()) / alpha
    val += penalty * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    p = np.exp(sh)
    p_sum = p.sum()
    p = p / p_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-16:
            continue
        j = k - (n - 1)
        if j >= 0:
            nv = n - j
            grad[:nv] += w * (1.0 - x[j:j+nv])
            grad[j:] -= w * x[:nv]
        else:
            jj = -j
            nv = n - jj
            grad[jj:] += w * (1.0 - x[:nv])
            grad[:nv] -= w * x[jj:jj+nv]
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h_start, n, alphas, penalty, deadline, maxiter=300):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        def obj(x):
            return smooth_obj(x, n, alpha, penalty)
        def grad(x):
            return smooth_grad(x, n, alpha, penalty)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            h = np.clip(res.x, 0.0, 1.0)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(h, n, deadline, seed=42, top_k=15):

    dx = 2.0 / n
    # Ensure binary and feasible
    h = (h >= 0.5).astype(float)
    # Fix sum
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        extra = ones[len(ones)-target:]
        h[extra] = 0.0
        zeros = np.append(zeros, extra)
        ones = ones[:target]
    elif len(ones) < target:
        need = target - len(ones)
        h[zeros[:need]] = 1.0
        ones = np.append(ones, zeros[:need])
        zeros = zeros[need:]
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    stall = 0
    temp = 0.01
    kicks = 0
    
    while time.time() < deadline:
        # Build g matrix for active lags
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute full g for all zeros and ones (needed for best swap)
        # g[z, k] = change in corr[k] if h[zeros[z]] goes from 0 to 1
        # g[o, k] = change in corr[k] if h[ones[o]] goes from 1 to 0
        
        nz = len(zeros)
        no = len(ones)
        nl = len(top_idx)
        
        # For efficiency, compute g[:, top_idx]
        g_z = np.zeros((nz, nl))
        g_o = np.zeros((no, nl))
        
        for k_idx, k in enumerate(top_idx):
            j = k - (n - 1)
            if j >= 0:
                nv = n - j
                # g for zero at position r: (1-h[r+j]) for the h[i] term contribution
                # Actually, g[r, k] = d(corr[k])/d(h[r]) when h[r] goes 0->1
                # = (1-h[r+j])*I[valid] - h[r-j]*I[valid]
                zr = zeros
                # For zero positions:
                valid_plus = (zr + j) >= 0 & (zr + j) < n
                valid_minus = (zr - j) >= 0 & (zr - j) < n

                term_m = np.where(valid_minus, h[zr[valid_minus] - j], 0.0)
                g_z[:, k_idx] = term_p - term_m
                
                or_ = ones

                term_m_o = np.where(valid_minus_o, h[or_[valid_minus_o] - j], 0.0)

            else:
                jj = -j
                nv = n - jj
                zr = zeros

                valid_minus = (zr + jj) < n  # r-j = r+jj < n
                term_p = np.where(valid_plus, 1.0 - h[zr[valid_plus] - jj], 0.0)
                term_m = np.where(valid_minus, h[zr[valid_minus] + jj], 0.0)
                g_z[:, k_idx] = term_p - term_m
                
                or_ = ones
                valid_plus_o = (or_ - jj) >= 0
                valid_minus_o = (or_ + jj) < n
                term_p_o = np.where(valid_plus_o, 1.0 - h[or_[valid_plus_o] - jj], 0.0)
                term_m_o = np.where(valid_minus_o, h[or_[valid_minus_o] + jj], 0.0)
                g_o[:, k_idx] = term_p_o - term_m_o
        
        # New correlation = corr + g_z[zi] - g_o[oi] for each (zi, oi) pair
        # But this is only for the top_k lags. The full new max could be different.
        # For the greedy step, we check the top lags first.
        
        # Compute new max for top lags:
        # new_c_top[zi, oi, ki] = corr[top_idx[ki]] + g_z[zi, ki] - g_o[oi, ki]
        base = corr[top_idx][None, None, :]
        D = base + g_z[:, None, :] - g_o[None, :, :]
        new_maxes = D.max(axis=2)  # shape (nz, no)
        
        flat_idx = np.argmin(new_maxes)
        zi = flat_idx // no
        oi = flat_idx % no
        best_val = new_maxes[zi, oi]
        
        i = zeros[zi]
        j_pos = ones[oi]
        
        # Do the swap and check actual improvement
        h[i] = 1.0
        h[j_pos] = 0.0
        new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
        actual_c = new_corr.max() * dx
        
        if actual_c < cur_c - 1e-14:
            corr = new_corr
            cur_c = actual_c
            zeros = np.delete(zeros, zi)
            ones = np.delete(ones, oi)
            zeros = np.append(zeros, j_pos)
            ones = np.append(ones, i)
            stall = 0
            temp = max(0.002, temp * 0.999)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Revert
            h[i] = 0.0
            h[j_pos] = 1.0
            stall += 1
            if stall > 50 or (kicks < 50 and stall > 20):
                # Try a random kick (2-for-2 swap)
                kicks += 1
                nz_try = min(4, nz)
                no_try = min(4, no)
                rz = rng.choice(nz, nz_try, replace=False)
                ro = rng.choice(no, no_try, replace=False)
                # Try all pairs
                best_kick_c = cur_c + 0.1
                best_kick = None
                for a in range(len(rz)):
                    for b in range(len(ro)):
                        za, ob = zeros[rz[a]], ones[ro[b]]
                        h[za] = 1.0
                        h[ob] = 0.0
                        kc = np.correlate(h, 1-h, mode='full').max() * dx
                        h[za] = 0.0
                        h[ob] = 1.0
                        if kc < best_kick_c:

                            best_kick = (za, ob)

                    h[za] = 1.0
                    h[ob] = 0.0
                    corr = np.correlate(h, 1-h, mode='full').astype(float)
                    cur_c = corr.max() * dx
                    zi2 = np.where(zeros == za)[0][0]
                    oi2 = np.where(ones == ob)[0][0]
                    zeros = np.delete(zeros, zi2)
                    ones = np.delete(ones, oi2)
                    zeros = np.append(zeros, ob)
                    ones = np.append(ones, za)
                temp = max(0.002, temp * 0.95)
                stall = 0
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    val = m + np.log(np.exp(sh).sum()) / alpha
    val += penalty * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    p = np.exp(sh)
    p = p / p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-16:
            continue
        j = k - (n - 1)
        if j >= 0:
            nv = n - j
            grad[:nv] += w * (1.0 - x[j:j + nv])
            grad[j:] -= w * x[:nv]
        else:
            jj = -j
            nv = n - jj
            grad[jj:] += w * (1.0 - x[:nv])
            grad[:nv] -= w * x[jj:jj + nv]
    grad += 2.0 * penalty * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h_start, n, alphas, penalty, deadline, maxiter=200):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_obj(x, n, alpha, penalty)
        def gra(x):
            return smooth_grad(x, n, alpha, penalty)
        try:
            res = minimize(obj, h, jac=gra, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6})
            h = np.clip(res.x, 0.0, 1.0)
            h = project_feasible(h, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def round_to_binary(h, n):
    h = np.clip(h, 0, 1)
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb
