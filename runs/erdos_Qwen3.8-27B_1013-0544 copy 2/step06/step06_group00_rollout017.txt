import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, alpha, n):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    M = C.max()
    shifted = alpha * (C - M)
    mx = shifted.max()
    shifted -= mx
    exp_s = np.exp(shifted)
    s = exp_s / exp_s.sum()
    obj = M + (mx + np.log(exp_s.sum())) / alpha
    A = np.zeros(n)
    B = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        sv = s[k_idx]
        if sv < 1e-14:
            continue
        if k >= 0:
            if k < n:
                A[:n-k] += sv * h[k:]
                B[k:] += sv * h[:n-k]
        else:
            k2 = -k
            if k2 < n:
                A[k2:] += sv * h[:n-k2]
                B[:n-k2] += sv * h[k2:]
    grad = dx * (1.0 - A - B)
    return obj, grad

def opt_lbfgs(h0, n, alpha, maxiter=5000, pen=20000.0):
    def f(x):
        o, _ = obj_grad(x, alpha, n)
        return o + 0.5 * pen * (x.sum() - n / 2.0) ** 2
    def g(x):
        _, gr = obj_grad(x, alpha, n)
        return gr + pen * (x.sum() - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8, 'iprint': -1})
        return project_feasible(r.x, n)
    except Exception:
        return h0

def opt_slsqp(h0, n, alpha, maxiter=2000):
    def f(x):
        return obj_grad(x, alpha, n)[0]
    def g(x):
        return obj_grad(x, alpha, n)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h0

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_anneal(h_init, n, deadline, n_iters=5000000, seed=99):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    mr = np.arange(-(n - 1), n)
    T = 0.008
    cool = 0.99998
    sc = 3000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros))
        bi = rng.randint(len(ones))
        i = zeros[ai]
        j = ones[bi]
        d = np.zeros(2 * n - 1)
        im = i + mr
        ms = (im >= 0) & (im < n)
        d[ms] += 1.0 - h[im[ms]]
        imn = i - mr
        ms = (imn >= 0) & (imn < n)
        d[ms] -= h[imn[ms]]
        jm = j + mr
        ms = (jm >= 0) & (jm < n)
        d[ms] -= 1.0 - h[jm[ms]]
        jmn = j - mr
        ms = (jmn >= 0) & (jmn < n)
        d[ms] += h[jmn[ms]]
        d[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            d[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            d[k2] += 1.0
        nc = (corr + d).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(sc * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c = nc
            zeros[ai] = j
            ones[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def greedy_binary(h_init, n, deadline, max_sweeps=300):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    mr = np.arange(-(n - 1), n)
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                d = np.zeros(2 * n - 1)
                im = i + mr
                ms = (im >= 0) & (im < n)
                d[ms] += 1.0 - h[im[ms]]
                imn = i - mr
                ms = (imn >= 0) & (imn < n)
                d[ms] -= h[imn[ms]]
                jm = j + mr
                ms = (jm >= 0) & (jm < n)
                d[ms] -= 1.0 - h[jm[ms]]
                jmn = j - mr
                ms = (jmn >= 0) & (jmn < n)
                d[ms] += h[jmn[ms]]
                d[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    d[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    d[k2] += 1.0
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c
