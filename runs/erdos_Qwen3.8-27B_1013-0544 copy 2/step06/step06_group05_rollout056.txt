    value = m + np.log(exp_g.sum()) / alpha

        if m_start2 < m_end2:
            grad[m_start2:m_end2] -= w[k_idx] * dx * h[m_start2 - k:m_end2 - k]

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:

def slsqp_smooth(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_gradient_smooth(h, n, alpha, n_steps, step_size, deadline):
    """Projected gradient with smooth max objective."""
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1.0 - h, mode='full') * dx
        m = g.max()
        exp_g = np.exp(alpha * (g - m))
        w = exp_g / exp_g.sum()
        # Compute gradient as weighted sum of individual gradients
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx]
            if wk < 1e-12:
                continue
            k = k_idx - (n - 1)
            # ∂g_k/∂h[m] = dx * [(1-h[m+k])*I(0≤m+k<n) - h[m-k]*I(0≤m-k<n)]
            # First part: for m where 0≤m<n and 0≤m+k<n
            # m from max(0,-k) to min(n-1, n-1-k)
            ms = max(0, -k)
            me = min(n, n - k)
            if ms < me:
                grad[ms:me] += wk * dx * (1.0 - h[ms+k:me+k])
            # Second part: for m where 0≤m<n and 0≤m-k<n

            ms2 = max(0, k)
            me2 = min(n, n + k)
            if ms2 < me2:
                grad[ms2:me2] -= wk * dx * h[ms2-k:me2-k]
        # Gradient step
        h_new = h - step_size * grad
        # Project
        h_new = project_feasible(h_new, n)
        h = h_new
    return h
