def add_lag_gradient(G,h,s,w):
    n=h.size
    if w == 0.0 or abs(s) >= n:
        return
    if s >=0:
        if s ==0:
           G += (1.0 - 2.0*h)*w
        else:
           G[:n-s] -= w*h[s:]
           G[s:] += w*(1.0 - h[:n-s])
    else:
        t=-s
        G[t:] -= w*h[:n-t]
        G[:n-t] += w*(1.0 - h[t:])

def line_search_delta(ca, aa, bb, dmax):
    if dmax <= 1e-14: return 0.0
    a=0.0; b=float(dmax)
    gr=(math.sqrt(5.0)-1.0)/2.0
    c=b-gr*(b-a)
    d=a+gr*(b-a)
    fc=float(np.max(ca + c*aa + c*c*bb))
    fd=float(np.max(ca + d*aa + d*d*bb))
    for _ in range(24):
       if fc < fd:
          b=d; d=c; fd=fc
          c=b-gr*(b-a)
          fc=float(np.max(ca + c*aa + c*c*bb))
       else:
          a=c; c=d; fc=fd
          d=a+gr*(b-a)
          fd=float(np.max(ca + d*aa + d*d*bb))
    if fc < fd:
       return c
    return d

import time
import math
import numpy as np
from scipy.optimize import minimize

_OPT_ALPHA = 1.0
_TARGET_SUM = 0.0
_OPT_DEADLINE = 0.0

def project_feasible(y, n):
    y = np.array(y, dtype=float, copy=True)
    if y.size != n:
        y = np.resize(y, n)
    S = 0.5 * n
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    if hi - lo < 1e-12:
        hi = lo + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-9)[0]
        else:
            idx = np.where(x > 1e-9)[0]
        if idx.size:
            x[idx] += diff / idx.size
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full").astype(np.float64)


def add_lag_gradient(G, h, s, w):
    n = h.size
    if w == 0.0 or abs(s) >= n:
        return
    if s >= 0:
        if s == 0:
            G += w * (1.0 - 2.0 * h)
        else:
            G[:n - s] -= w * h[s:]
            G[s:] += w * (1.0 - h[:n - s])
    else:
        t = -s
        G[t:] -= w * h[:n - t]
        G[:n - t] += w * (1.0 - h[t:])


def weighted_lag_gradient(h, indices, weights):
    G = np.zeros(h.size, dtype=float)
    n = h.size
    for p, w in zip(indices, weights):
        if w > 1e-13:
            add_lag_gradient(G, h, int(p) - (n - 1), float(w))
    return G


def point_grad_over_lags(i, n, h, om_rev):
    L = 2 * n - 1
    g = np.zeros(L, dtype=float)
    g[i:i + n] += om_rev
    start = n - 1 - i
    end = 2 * n - 1 - i
    g[start:end] -= h
    return g


def line_search_delta(ca, aa, bb, dmax):
    if dmax <= 1e-14:
        return 0.0
    a = 0.0
    b = float(dmax)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(ca + c * aa + c * c * bb))
    fd = float(np.max(ca + d * aa + d * d * bb))
    for _ in range(24):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(ca + c * aa + c * c * bb))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(ca + d * aa + d * d * bb))
    if fc < fd:
        return c
    return d

        best_d = 0.0

                corr = compute_corr(h)
                current_m = best_count
            noimp = 0

def smooth_value_grad(x, alpha, need_grad):
    n = x.size
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    u = alpha * (c - m)
    e = np.exp(u)
    s = float(e.sum())
    if s <= 0.0:
        val = m
    else:
        val = m + math.log(s) / alpha
    if not need_grad:
        return val, None
    G = np.zeros(n, dtype=float)
    thresh = 1e-10 * max(s, 1.0)
    for p in range(c.size):
        ep = e[p]
        if ep > thresh:
            add_lag_gradient(G, x, int(p) - (n - 1), ep / s)
    return val, G


def slsqp_obj(x):
    if time.time() > _OPT_DEADLINE:
        raise RuntimeError("budget")
    val, _ = smooth_value_grad(x, _OPT_ALPHA, False)
    return val


def slsqp_grad(x):
    if time.time() > _OPT_DEADLINE:
        raise RuntimeError("budget")
    _, g = smooth_value_grad(x, _OPT_ALPHA, True)
    return g


def slsqp_cons(x):
    return float(x.sum() - _TARGET_SUM)


def slsqp_conjac(x):
    return np.ones((1, x.size), dtype=float)


def slsqp_callback(xk):
    global _SLSQP_LAST_X
    _SLSQP_LAST_X = xk
    if time.time() > _OPT_DEADLINE:
        raise RuntimeError("budget")


def slsqp_run(x, n, alpha, maxiter, deadline):
    global _OPT_ALPHA, _TARGET_SUM, _OPT_DEADLINE, _SLSQP_LAST_X
    _OPT_ALPHA = float(alpha)
    _TARGET_SUM = 0.5 * n
    _OPT_DEADLINE = float(deadline)
    _SLSQP_LAST_X = project_feasible(x, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": slsqp_cons, "jac": slsqp_conjac}]
    try:
        res = minimize(
            slsqp_obj,
            _SLSQP_LAST_X,
            method="SLSQP",
            jac=slsqp_grad,
            bounds=bounds,
            constraints=constraints,
            callback=slsqp_callback,
            options={"maxiter": int(maxiter), "ftol": 1e-11},
        )
        out = project_feasible(res.x, n)
    except Exception:
        out = project_feasible(_SLSQP_LAST_X, n)
    return out
