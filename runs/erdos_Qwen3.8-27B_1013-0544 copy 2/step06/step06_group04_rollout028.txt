def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 2, n - 1 + i) + 1)
    jp = n - 1 + i - m2
    delta[m2] -= h[jp]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 2, n - 1 + j) + 1)
    jp = n - 1 + j - m2
    delta[m2] += h[jp]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

    best_h = h.copy()

        rng.shuffle(zeros)

                    found = True
                    break

def two_swap_search(h, n, deadline, max_iters=5000, seed=42):
    """2-swap local search: swap two pairs simultaneously."""

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = int(rng.choice(zeros))
        i2 = int(rng.choice(zeros))
        if i1 == i2:
            continue
        j1 = int(rng.choice(ones))
        j2 = int(rng.choice(ones))
        if j1 == j2:
            continue
        # Apply 2-swap
        new_h = h.copy()
        new_h[i1], new_h[j1] = 1.0, 0.0
        new_h[i2], new_h[j2] = 1.0, 0.0
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        nc = float(new_corr.max()) * dx
        if nc < cur_c - 1e-15:
            h = new_h
            corr = new_corr
            cur_c = nc
            if cur_c < best_c:
                best_c = nc
                best_h = h.copy()
    return best_h, best_c
