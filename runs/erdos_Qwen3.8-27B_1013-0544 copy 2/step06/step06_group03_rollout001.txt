The directional derivative of f_k in direction d (where d[i] = -δ, d[j] = +δ, others 0) is:

For the max, the directional derivative is max over active k of δ*(g_k[j] - g_k[i]).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _lse_active(x, n, idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    a = c[idx].max()
    shifted = alpha * (c[idx] - a)
    mx = shifted.max()
    if mx > 500:
        return float(a + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(a + np.log(np.sum(np.exp(shifted))) / alpha)

def _penalty_obj(x, n, tau, gamma):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    excess = np.maximum(c * dx - tau, 0.0)
    return float(gamma * np.sum(excess ** 2) + 0.5 * np.sum((x - 0.5) ** 2) * 1e-6)

def slsqp(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_lse_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, idx, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_lse_active, h, args=(n, idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def penalty_solve(h, n, tau, gamma, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_penalty_obj, h, args=(n, tau, gamma), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    if m == 0:
        return (1.0 - 2.0 * h) * dx
    if lag > 0:
        grad = np.zeros(n)
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
        return grad * dx
    else:
        grad = np.zeros(n)
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
        return grad * dx

def grad_local_search(h, n, deadline, top_m=8):
    h = h.copy()
    best_c = compute_c5(h, n)
    scales = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for s in scales:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            c = compute_all_lags(h, n)
            c_max = c.max()
            thresh = c_max - 0.0005
            active = np.where(c > thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            combined_grad = np.zeros(n)
            for ki in active:
                combined_grad += lag_gradient(h, n, ki - (n - 1))
            combined_grad /= len(active)
            n_tried = 0
            for i in range(n):
                if time.time() > deadline - 1:
                    break
                for j in range(i + 1, n):
                    if n_tried > 500:
                        break
                    score = combined_grad[j] - combined_grad[i]
                    if score > -1e-8:
                        continue
                    if h[i] > 0 and h[j] < 1:
                        d = min(s, h[i] * 0.4, 1.0 - h[j])
                        if d < 1e-12:
                            continue
                        h[i] -= d
                        h[j] += d
                        c_new = compute_c5(h, n)
                        n_tried += 1
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                    elif h[j] > 0 and h[i] < 1:
                        d = min(s, h[j] * 0.4, 1.0 - h[i])
                        if d < 1e-12:
                            continue
                        h[j] -= d
                        h[i] += d
                        c_new = compute_c5(h, n)
                        n_tried += 1
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
    return h, best_c

def targeted_local_search(h, n, deadline):

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP phase 1
    if time.time() < deadline - 40:
        t_end = min(deadline - 30, time.time() + min(100, (deadline - time.time()) * 0.35))
        h1 = slsqp(best_h, n, 50000, 4000, t_end)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    # Active set SLSQP
    if time.time() < deadline - 30:
        c_arr = compute_all_lags(best_h, n)
        top_idx = np.argsort(c_arr)[-6:]
        h2 = slsqp_active(best_h, n, top_idx, 100000, 4000, deadline - 20)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    # Penalty approach
    if time.time() < deadline - 30:
        tau_target = best_c - 0.001
        h3 = penalty_solve(best_h, n, tau_target, 1e6, 3000, deadline - 20)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    # Gradient local search
    if time.time() < deadline - 10:
        h4, c4 = grad_local_search(best_h, n, min(deadline - 3, time.time() + 60), top_m=8)

def run(seed=42, budget_s=1100, **kwargs):
