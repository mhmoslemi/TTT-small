def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(h + mid, 0.0, 1.0).sum()
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    c_max = c.max()
    cs = alpha * (c - c_max)
    cs_max = cs.max()
    w = np.exp(cs - cs_max)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1

            continue
        k = m - nm1

        if k >= 0:

            grad[k:] += wm * (-h[:nv])

            grad[kk:] += wm * (1.0 - h[:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    c_max = c.max()
    cs = alpha * (c - c_max)
    cs_max = cs.max()
    w = np.exp(cs - cs_max)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - nm1
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _sm_obj(h, n, alpha, mu):
    c = compute_c5_all(h, n)
    m = c.max()
    cs = alpha * (c - m)
    cs_max = cs.max()
    val = m + (cs_max + np.log(np.sum(np.exp(cs - cs_max)))) / alpha
    val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

    g = grad_smooth_max(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, deadline, max_iters=200):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    mus = [1.0, 5.0, 50.0, 500.0, 5000.0]
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 2:
            break
        mu = mus[min(i, len(mus) - 1)]
        rem = deadline - time.time() - 1
        if rem < 3:
            break
        max_it = min(max_iters, max(10, int(rem * 3)))
        try:
            res = minimize(_sm_obj, h, jac=_sm_grad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_it, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_it * 8, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c

def local_pair_search(h, n, deadline, radius=35, steps=None):
    if steps is None:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for d in steps:

        for _pass in range(50):
            if time.time() > deadline:

            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d; h[j] -= d

                        else:

def grad_guided_pair(h, n, deadline, step=0.003, max_rounds=30):

    alphas = [200, 2000, 20000, 200000]

            if best_move is not None:

                h[i] += mv; h[j] -= mv

            else:
                break
