    corr = np.zeros(2*n-1)  # all zero initially

        best_pos = -1
        best_max = np.inf
        for pos in range(n):
            if h[pos] == 1:
                continue
            # Compute change in corr if we set h[pos] = 1
            # For lag l (indexed 0 to 2n-2, actual lag = l-(n-1)):
            # h[pos] changes from 0 to 1
            # This adds to corr at positions where pos is the "first" index
            # corr[l] += (1 - h[pos+lag]) where lag = l-(n-1), i.e., pos+lag = pos + l - n + 1
            # And corr[l] += h[pos-lag] where pos-lag = pos - l + n - 1
            # = (1 - h[pos + l - n + 1]) + h[pos - l + n - 1] (if valid)
            new_corr = corr.copy()
            for l in range(2*n-1):

                    new_corr[l] += (1 - h[pos + lag])
                if 0 <= pos - lag < n:
                    new_corr[l] += h[pos - lag]
            m = new_corr.max()

                best_pos = pos

            if 0 <= best_pos + lag < n:

            if 0 <= best_pos - lag < n:

    return h

def project_binary(h, n):
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    thresh = h_sorted[target-1] if target > 0 else 1
    b = (h >= thresh).astype(float)
    if b.sum() > target:
        # Too many ones, flip some
        ones = np.where(b == 1)[0]
        rng = np.random.RandomState(0)
