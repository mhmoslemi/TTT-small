import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine correction
    diff = target - h_proj.sum()
    if abs(diff) < 1e-10:
        return h_proj
    # Distribute remaining
    if diff > 0:
        room = 1.0 - h_proj
        s_room = room.sum()
        if s_room > 1e-14:
            h_proj += diff / s_room * room
    else:
        s_h = h_proj.sum()
        if s_h > 1e-14:
            h_proj += diff / s_h * h_proj
    return np.clip(h_proj, 0.0, 1.0)

def compute_gradient(h, n, w):
    """Compute gradient of smoothed objective at h given softmax weights w."""
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx] * dx
        if wk < 1e-16:
            continue
        # A[j] contribution: w[k] * (1-h[j+k]) for valid j
        # B[j] contribution: -w[k] * h[j-k] for valid j
        if k >= 0:
            # j+k < n => j in [0, n-k-1]
            g[:n-k] += wk * (1.0 - h[k:])
            # j-k >= 0 => j in [k, n-1]
            g[k:] -= wk * h[:n-k]
        else:
            kk = -k  # positive
            # j+k >= 0 => j in [kk, n-1], j+k < n => j < n (always)
            g[kk:] += wk * (1.0 - h[:n-kk])
            # j-k >= 0 => j >= 0 (always), j-k < n => j < n+kk (always since j<n)
            g[:] -= wk * h[kk:]
    return g

def projected_gradient(h, n, alpha, lr, max_iter, deadline):
    """Projected gradient with log-sum-exp smoothing."""
    dx = 2.0 / n
    curr = np.correlate(h, 1-h, mode='full') * dx
    best_h = h.copy()
    best_obj = curr.max()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        max_c = curr.max()
        exp_c = np.exp(alpha * (curr - max_c))
        w = exp_c / exp_c.sum()
        g = compute_gradient(h, n, w)
        
        # Backtracking line search
        step = lr
        f_curr = curr.max()
        for _ in range(30):
            h_trial = project_feasible(h - step * g, n)
            curr_trial = np.correlate(h_trial, 1-h_trial, mode='full') * dx
            f_trial = curr_trial.max()
            if f_trial < f_curr - 1e-12:
                break
            step *= 0.5
        else:
            h_trial = h
            f_trial = f_curr
        
        h = h_trial
        curr = curr_trial
        if f_trial < best_obj:
            best_obj = f_trial
            best_h = h.copy()
    
    return best_h, best_obj

    best_h = h.copy()
    curr = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(curr.max())
    lam = 0.0
    lam_lr = 0.1
    
    rng = np.random.RandomState(0)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        # Update Lagrange multiplier
        lam += lam_lr * (h.sum() - n/2.0) * dx
        lam = np.clip(lam, -10.0, 10.0)

        order = rng.permutation(n)
        
        for j in order:
            if time.time() > deadline:
                break

            # g_k[j] = dx * [(1-h[j-k])*I(0≤j-k<n) - h[j+k]*I(0≤j+k<n)]
            gj = np.zeros(2*n-1)

                k = k_idx - (n-1)
                val = 0.0
                if 0 <= j-k < n:
                    val += (1.0 - h[j-k])
                if 0 <= j+k < n:
                    val -= h[j+k]

            best_h = h.copy()

    c = np.correlate(h, 1 - h, mode='full')

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

    if abs(diff) > 1e-12:

            sr = room.sum()
            if sr > 1e-14:

        else:
            sh = h_p.sum()
            if sh > 1e-14:
                h_p += diff / sh * h_p
    return np.clip(h_p, 0.0, 1.0)

def build_jacobian(h, n):
    """Build J[j, k_idx] = partial r_k / partial h[j] for all j, k_idx."""
    dx = 2.0 / n
    J = np.zeros((n, 2*n-1))
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        if k >= 0:
            J[k:, k_idx] = (1.0 - h[:n-k]) * dx
            J[:n-k, k_idx] += -h[k:] * dx
        else:
            kk = -k
            J[:n-kk, k_idx] = (1.0 - h[kk:]) * dx
            J[kk:, k_idx] += -h[:n-kk] * dx
    return J

def coord_desc_pass(h, n, J, curr, alpha, lam, dx, rng):
    """One pass of coordinate descent."""
    order = rng.permutation(n)
    for j in order:
        gj = J[j]  # gradient of r_k w.r.t. h[j], for all k
        gj_lam = gj + lam * dx  # add Lagrange term

            s_exp = exp_s.sum()
            if s_exp < 1e-300:
                deriv = 0.0
            else:
                w = exp_s / s_exp
                deriv = (w * gj_lam).sum()
            if deriv > 0:
                hi_t = mid_t
            else:
                lo_t = mid_t
        h[j] = (lo_t + hi_t) / 2.0
    
    return h

        # Coordinate descent pass
        h = coord_desc_pass(h, n, J, curr, alpha, lam, dx, rng)

        if c_val < best_c:
            best_c = c_val

        for j in order:
            gj = J[j]
            hj = h[j]
            base = curr - hj * gj
            lo_t, hi_t = 0.0, 1.0
            for _ in range(35):
                t = (lo_t + hi_t) * 0.5
                s = alpha * (base + t * gj)
                s -= s.max()
                e = np.exp(s)
                es = e.sum()
                if es < 1e-300:
                    d = 0.0
                else:
                    d = ((e / es) * gj).sum()
                if d > 0:
                    hi_t = t
                else:
                    lo_t = t
            h[j] = (lo_t + hi_t) * 0.5
            curr += (h[j] - hj) * gj
        h = project_feasible(h, n)
        curr = np.correlate(h, 1 - h, mode='full') * dx
        c = curr.max()
        if c < best_c:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break
    return best_h, best_c

def pg_optimize(h0, n, alpha, lr, deadline, max_iter=50000):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    curr = np.correlate(h, 1 - h, mode='full') * dx
    best_c = curr.max()
    v = np.zeros(n)
    stale = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        mc = curr.max()
        ec = np.exp(alpha * (curr - mc))
        es = ec.sum()
        w = ec / es if es > 1e-300 else np.zeros_like(ec)
        active = np.where(w > 1e-12)[0]
        g = np.zeros(n)
        for ki in active:
            k = ki - (n - 1)
            wk = w[ki]
            if k >= 0:
                g[k:] += wk * (1 - h[:n - k])
                g[:n - k] -= wk * h[k:]
            else:
                kk = -k
                g[:n - kk] += wk * (1 - h[kk:])
                g[kk:] -= wk * h[:n - kk]
        g *= dx
        v = 0.85 * v + 0.15 * g
        step = lr
        fc = curr.max()
        ok = False
        for _ in range(12):
            ht = project_feasible(h - step * v, n)
            ct = np.correlate(ht, 1 - ht, mode='full') * dx
            ctmax = ct.max()
            if ctmax < fc - 1e-14:
                h = ht
                curr = ct
                ok = True
                if ctmax < best_c:
                    best_c = ctmax
                    best_h = h.copy()
                    stale = 0
                break
            step *= 0.5
        if not ok:
            stale += 1
            lr *= 0.97
            if stale > 300:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: coordinate descent, moderate alpha (30%)
    t1 = t0 + 0.30 * total
    for alpha in [50, 200, 1000, 5000]:
        if time.time() > t1:
            break
        st = time.time()
        rem = t1 - st
        if rem < 5:
            break
        hh, cc = coord_descent(best_h, n, alpha, st + rem * 0.9, seed=seed, max_passes=200)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 2: coordinate descent, high alpha (30%)
    t2 = t0 + 0.60 * total
    for alpha in [20000, 100000, 500000, 2000000]:
        if time.time() > t2:
            break
        st = time.time()
        rem = t2 - st
        if rem < 5:
            break
        hh, cc = coord_descent(best_h, n, alpha, st + rem * 0.9, seed=seed + 1, max_passes=150)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 3: projected gradient (25%)
    t3 = t0 + 0.85 * total
    st = time.time()
    hh, cc = pg_optimize(best_h, n, 50000, 0.05, st + (t3 - st) * 0.5, max_iter=50000)
    if cc < best_c:
        best_h, best_c = hh, cc
    st = time.time()
    hh, cc = pg_optimize(best_h, n, 500000, 0.02, min(st + 30, t3), max_iter=30000)
    if cc < best_c:
        best_h, best_c = hh, cc

    # Phase 4: coordinate descent on best with very high alpha (10%)
    if time.time() < dl - 10:
        hh, cc = coord_descent(best_h, n, 5000000, dl - 5, seed=seed + 2, max_passes=100)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 5: try different resolution
    for nh in [300, 500]:
        if time.time() > dl - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh, cc = coord_descent(hh, nh, 50000, dl - 3, seed=seed + 3, max_passes=100)
        if cc < best_c:
            best_h, best_c, n = hh, cc, nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
