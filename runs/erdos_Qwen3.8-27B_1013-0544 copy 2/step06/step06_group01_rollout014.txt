    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top_k indices

    g = np.zeros(n)

        # S = {n : 0 <= n < n AND 0 <= n+lag < n}
        if lag >= 0:
            # n in [0, n-lag)
            s_lo, s_hi = 0, n - lag
        else:
            # n in [-lag, n)
            s_lo, s_hi = -lag, n
        
        # ∂c/∂h[j] = (1-h[j+lag])*1{j in S} - h[j-lag]*1{j-lag in S}

        # j+lag must be in [0, n)

            # j in [0, n-lag), j+lag in [lag, n)
            g[0:n-lag] += (1.0 - h[lag:n])
        else:
            # j in [-lag, n), j+lag in [0, n+lag)
            g[-lag:n] += (1.0 - h[0:n+lag])
        
        # Second term: -h[j-lag] for j-lag in [s_lo, s_hi)
        # j in [s_lo+lag, s_hi+lag)

        j_hi = s_hi + lag

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n-lag]
    else:
        L = -lag
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    return g

def subgrad_descent(h, n, max_iters=20000, top_k=15, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.05
    stalled = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_corr(h, n)
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        m = c_top.max()
        alpha = 50.0
        w = np.exp(alpha * (c_top - m))
        w /= w.sum()
        
        g = np.zeros(n)
        for i, ki in enumerate(idx):
            if w[i] < 1e-6:
                continue
            g += w[i] * lag_gradient(h, n, ki)
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        # Step size: adapt based on progress
        h_new = h - eta * g / gn
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
            eta = min(eta * 1.05, 0.2)
        else:
            stalled += 1
            if stalled > 50:
                eta *= 0.7
                stalled = 0
                if eta < 0.001:
                    break
    return best_h, best_c

def coord_descent_focused(h, n, deadline, top_k=10):
    """Coordinate descent that focuses on points affecting top-K lags"""
    best_c = compute_c5(h, n)
    max_passes = 500
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        idx = np.argsort(c)[::-1][:top_k]
        
        # Find points that affect these lags
        # For each critical lag, the affected points are in the valid range
        affected = set()
        for ki in idx:
            lag = ki - (n - 1)
            if lag >= 0:
                affected.update(range(0, n - lag))
                affected.update(range(lag, n))
            else:
                L = -lag
                affected.update(range(L, n))
                affected.update(range(0, n - L))
        
        improved = False
        for j in affected:
            if j >= n:
                continue
            # Try small adjustments to h[j]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                if h[j] + delta < 0 or h[j] + delta > 1:
                    continue
                h[j] += delta
                c_val = compute_c5(h, n)
                if c_val < best_c - 1e-12:
                    best_c = c_val
                    improved = True
                    break
                h[j] -= delta
        if not improved:
            break
    return h, best_c

def slsqp_optimize(h, n, alpha=100000, maxiter=2000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        return float((m + np.log(np.sum(np.exp(sh)))) / alpha * dx)
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        w = np.exp(sh) / np.sum(np.exp(sh))
        g = np.zeros(n)
        for ki in range(len(c)):
            if w[ki] < 1e-8:
                continue
            g += w[ki] * lag_gradient(x, n, ki)
        return g * dx / alpha
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_grad, args=(), method='SLSQP',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x if hasattr(res, 'x') else h, n)

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cval = compute_c5(h, n)
                        if cval < best_c - 1e-14:
                            best_c = cval
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    
    # Phase A: Subgradient descent
    t_sub = min(deadline - time.time(), 60)
    if t_sub > 10:
        h_sg, c_sg = subgrad_descent(h, n, max_iters=50000, top_k=15,
                                     deadline=time.time() + t_sub)
        h_sg = project_feasible(h_sg, n)
        c_sg = compute_c5(h_sg, n)
        if c_sg < best_c:
            h, best_c = h_sg, c_sg
    
    # Phase B: SLSQP
    if time.time() < deadline - 30:
        h_s = slsqp_optimize(h, n, alpha=100000, maxiter=2000)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    
    # Phase C: Focused coordinate descent
    if time.time() < deadline - 20:
        h_cd, c_cd = coord_descent_focused(h, n, time.time() + min(100, deadline - time.time() - 10), top_k=10)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    
    # Phase D: Pair search
    if time.time() < deadline - 10:
        h, best_c = pair_search(h, n, deadline - 3,
                                [0.01, 0.005, 0.002, 0.001, 0.0005], radius=40)
    
    # Phase E: Final SLSQP + pair
    if time.time() < deadline - 8:
        h_f = slsqp_optimize(h, n, alpha=500000, maxiter=2000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            h, best_c = h_f, c_f
    if time.time() < deadline - 3:
        h, best_c = pair_search(h, n, deadline - 2,
                                [0.0005, 0.0002, 0.0001], radius=30)
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

```python
def lag_gradient(h, n, k):
    """Compute ∂c[k]/∂h[m] for correlation index k"""
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # S = [0, n-lag)
        # Term 1: 1 for m in [lag, n)
        g[lag:] += 1.0
        # Term 2: -h[m+lag] for m in [0, n-lag)
        g[:n-lag] -= h[lag:]
        # Term 3: -h[m-lag] for m in [lag, n)
        g[lag:] -= h[:n-lag]
    else:
        L = -lag
        # S = [L, n)
        # Term 1: 1 for m in [0, n-L)
        g[:n-L] += 1.0
        # Term 2: -h[m-L] for m in [L, n)
        g[L:] -= h[:n-L]
        # Term 3: -h[m+L] for m in [0, n-L)
        g[:n-L] -= h[L:]
    return g
```
