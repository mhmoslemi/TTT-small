```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, n, alpha, mu):
    """Log-sum-exp smoothed objective with sum penalty. Returns (f, g)."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    s = alpha * c
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    f_lse = (sm + np.log(e.sum())) / alpha
    
    # Penalty
    diff = x.sum() - n / 2.0
    f_pen = mu * diff * diff
    f = f_lse + f_pen
    
    # Gradient of LSE part
    g = np.zeros(n)
    for j in range(n):
        # P(j) = sum_{k in valid range} w[k] * (1-x[j+k-(n-1)])
        k_lo = max(0, n - 1 - j)
        k_hi = min(2 * n - 2, 2 * n - 2 - j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            is_ = j + ks - (n - 1)
            g[j] += np.dot(w[ks], 1 - x[is_])
        # Q(j) = sum_{k in valid range} w[k] * x[j-k+(n-1)]
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + n - 1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            is2 = j - ks2 + (n - 1)
            g[j] -= np.dot(w[ks2], x[is2])
    g *= dx
    # Gradient of penalty
    g += 2 * mu * diff
    return f, g

def optimize_lbb(x, n, alpha, mu, maxiter=500):
    """Use L-BFGS-B to optimize."""
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda v: lse_obj_and_grad(v, n, alpha, mu)[0],
                   x, jac=lambda v: lse_obj_and_grad(v, n, alpha, mu)[1],
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return res.x

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary_fast(h_init, n, deadline, max_sweeps=200):
    """Faster greedy binary optimization using precomputed deltas."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                # Compute delta for swapping i (0->1) and j (1->0)
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i], h[best_j] = 1.0, 0.0
                # Update corr
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1 - h[jm[msk]])
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - i)
                if 0 <= k1 < 2*n-1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - best_j)
                if 0 <= k2 < 2*n-1:
                    delta[k2] += 1.0
                corr += delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
    return best_h, best_c
