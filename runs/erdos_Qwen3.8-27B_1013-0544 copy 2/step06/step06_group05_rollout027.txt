import numpy as np
from scipy.optimize import minimize
import time

_sle = {}

def compute_c5_val(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + (diff / s) * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h * (1.0 + diff / s), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def sle_obj(x):
    n = _sle['n']
    alpha = _sle['alpha']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = corr.max()
    scaled = np.clip(alpha * (corr - max_c), -100, 100)
    exp_c = np.exp(scaled)
    sum_exp = exp_c.sum()
    return max_c + np.log(sum_exp) / alpha

def sle_jac(x):
    n = _sle['n']
    alpha = _sle['alpha']
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = corr.max()
    scaled = np.clip(alpha * (corr - max_c), -100, 100)
    exp_c = np.exp(scaled)
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    j_arr = np.arange(n)
    grad = np.zeros(n)
    for k in range(n_corr):
        pk = p[k]
        if pk < 5e-8:
            continue
        s = k - (n - 1)
        idx_m = j_arr - s
        idx_p = j_arr + s
        vm = (idx_m >= 0) & (idx_m < n)
        vp = (idx_p >= 0) & (idx_p < n)
        g = np.zeros(n)
        g[vm] = (1.0 - x[idx_m[vm]])
        g[vp] -= x[idx_p[vp]]
        grad += pk * g * dx
    return grad

def constraint_fun(x):
    return x.sum() - len(x) / 2.0

def slsqp_optimize(h_init, n, alpha, maxiter=2000):
    _sle['n'] = n
    _sle['alpha'] = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(sle_obj, h_init.copy(), method='SLSQP', jac=sle_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def binary_sa(h_init, n, deadline, seed=42):
    dx = 2.0 / n
    cur = make_binary(h_init, n)
    cur_max = np.correlate(cur, 1.0 - cur, mode='full').max()
    best_h = cur.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    no_improve = 0
    restarts = 0
    max_restarts = 50

    while time.time() < deadline and restarts <= max_restarts:
        T0 = max(0.3, 2.5 * (0.75 ** restarts))
        T_end = 0.005
        iters = 80000
        for it in range(iters):
            if time.time() > deadline:
                return best_h, float(best_max * dx)
            if no_improve > 4000:
                break
            T = T0 * (T_end / T0) ** (it / iters)
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            cur[i] = 1.0
            cur[j] = 0.0
            new_max = np.correlate(cur, 1.0 - cur, mode='full').max()
            delta = new_max - cur_max
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                cur[i] = 0.0
                cur[j] = 1.0
                no_improve += 1
        restarts += 1
        if no_improve > 4000 and time.time() < deadline:
            for _ in range(rng.randint(3, min(15, n // 10))):
                oi = np.where(best_h == 1)[0]
                zi = np.where(best_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    best_h[rng.choice(oi)] = 0.0
                    best_h[rng.choice(zi)] = 1.0
            cur = best_h.copy()
            cur_max = np.correlate(cur, 1.0 - cur, mode='full').max()
            no_improve = 0

    return best_h, float(best_max * dx)

def guided_local_search(h_init, n, deadline, seed=42):
    """Local search biased toward reducing critical lags."""
    dx = 2.0 / n
    cur = make_binary(h_init, n)
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = corr.max()
    best_h = cur.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    no_improve = 0

    while time.time() < deadline:
        if no_improve > 2000:
            break
        critical = np.where(corr >= cur_max - 2.0)[0]
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for trial in range(80):
            if time.time() > deadline:
                return best_h, float(best_max * dx)
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            cur[i] = 1.0
            cur[j] = 0.0
            new_corr = np.correlate(cur, 1.0 - cur, mode='full')
            new_max = new_corr.max()
            crit_check = all(new_corr[k] <= cur_max + 0.5 for k in critical)
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                improved = True
                no_improve = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = cur.copy()
                break
            elif crit_check and new_max <= cur_max:
                corr = new_corr
                cur_max = new_max
                improved = True
                no_improve = 0
                break
            else:
                cur[i] = 0.0
                cur[j] = 1.0
                no_improve += 1
        if not improved and no_improve > 2000:
            break

    return best_h, float(best_max * dx)
