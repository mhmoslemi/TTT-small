import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    g = c * dx
    m = g.max()
    shifted = alpha * (g - m)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp  # softmax weights
    obj = m + np.log(sum_exp) / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    # For each lag k, we need dg_k/dh
    # k = n-1+j for j = 0, 1, ..., n-1
    # k = n-1-j for j = 1, 2, ..., n-1
    for j in range(n):
        wj = w[n-1+j] * dx  # weight for lag j (positive side)
        wneg = w[n-1-j] * dx if j > 0 else 0.0
        
        # For k = n-1+j: dc/dh[p] = (1-h[p-j])*[j<=p] - h[p+j]*[p<=n-1-j]
        if j == 0:
            # c_{n-1} = sum h[i]*(1-h[i])
            grad += wj * (1.0 - 2.0 * h)
        else:
            # Positive lag j
            # (1-h[p-j]) for p >= j
            grad[j:] += wj * (1.0 - h[:n-j])
            # -h[p+j] for p <= n-1-j
            grad[:n-j] += -wj * h[j:]
            
            # Negative lag -j (same as positive lag j by symmetry of the problem?)
            # Actually no, c_{n-1-j} is different from c_{n-1+j}
            # For k = n-1-j: dc/dh[p] = (1-h[p+j])*[p<=n-1-j] - h[p-j]*[p>=j]
            grad[:n-j] += wneg * (1.0 - h[j:])
            grad[j:] += -wneg * h[:n-j]
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    g = c * dx
    m = g.max()
    sh = alpha * (g - m)
    ms = sh.max()
    es = np.exp(sh - ms)
    ses = es.sum()
    obj = m + (ms + np.log(ses)) / alpha
    w = es / ses
    grad = np.zeros(n)
    # j=0
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    # j=1..n-1
    for j in range(1, n):
        wp = w[n-1+j] * dx
        wn = w[n-1-j] * dx
        grad[j:] += wp * (1.0 - h[:n-j])
        grad[:n-j] -= wp * h[j:]
        grad[:n-j] += wn * (1.0 - h[j:])
        grad[j:] -= wn * h[:n-j]
    return obj, grad

def opt_slsqp(h, n, alpha, maxiter, ftol):
    def f(x):
        return smooth_max_and_grad(x, n, alpha)[0]
    def jf(x):
        return smooth_max_and_grad(x, n, alpha)[1]
    def cf(x):
        return x.sum() - n/2.0
    def cjf(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=jf, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type':'eq','fun':cf,'jac':cjf}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project_feasible(res.x, n)

def opt_lbfgs(h, n, alpha, maxiter, lam):
    def f(x):
        o, _ = smooth_max_and_grad(x, n, alpha)
        return o + lam*(x.sum()-n/2.0)**2
    def jf(x):
        _, g = smooth_max_and_grad(x, n, alpha)
        return g + 2*lam*(x.sum()-n/2.0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=jf, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter':maxiter,'ftol':1e-18})
    return project_feasible(res.x, n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=200):
    m_range = np.arange(-(n-1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h==1)[0]]
    zeros_list = [int(x) for x in np.where(h==0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                if i == j:
                    continue
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im>=0)&(im<n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn>=0)&(imn<n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm>=0)&(jm<n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn>=0)&(jmn<n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                nc = (corr+delta).max()*dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                delta = np.zeros(2*n-1)
                im = i + m_range
                msk = (im>=0)&(im<n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn>=0)&(imn<n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm>=0)&(jm<n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn>=0)&(jmn<n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                delta[n-1+(j-i)] += 1.0
                delta[n-1+(i-j)] += 1.0
                h[i],h[j] = 1.0,0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    g = c * dx
    m = g.max()
    sh = alpha * (g - m)
    ms = sh.max()
    es = np.exp(sh - ms)
    ses = es.sum()
    obj = m + (ms + np.log(ses)) / alpha
    w = es / ses
    grad = np.zeros(n)
    grad += w[n - 1] * dx * (1.0 - 2.0 * h)
    for j in range(1, n):
        wp = w[n - 1 + j] * dx
        wn = w[n - 1 - j] * dx
        grad[j:] += wp * (1.0 - h[:n - j])
        grad[:n - j] -= wp * h[j:]
        grad[:n - j] += wn * (1.0 - h[j:])
        grad[j:] -= wn * h[:n - j]
    return obj, grad

def opt_slsqp(h, n, alpha, maxiter, ftol):
    def f(x):
        return smooth_max_and_grad(x, n, alpha)[0]
    def jf(x):
        return smooth_max_and_grad(x, n, alpha)[1]
    def cf(x):
        return x.sum() - n / 2.0
    def cjf(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=jf, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cjf}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def opt_lbfgs(h, n, alpha, maxiter, lam):
    def f(x):
        o, _ = smooth_max_and_grad(x, n, alpha)
        pen = lam * (x.sum() - n / 2.0) ** 2
        return o + pen
    def jf(x):
        _, g = smooth_max_and_grad(x, n, alpha)
        return g + 2.0 * lam * (x.sum() - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, jac=jf, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=80):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (j - i)] += 1.0
                delta[n - 1 + (i - j)] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
                break
        if not improved:
            break
    return best_h, best_c

def projected_grad_step(h, n, alpha, lr, deadline, max_steps):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_steps):
        if time.time() > deadline:
            break
        _, g = smooth_max_and_grad(h, n, alpha)
        h_new = project_feasible(h - lr * g, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c, best_h = c, h_new.copy()
        if c <= compute_c5(h, n) + 1e-12:
            h = h_new
        else:
            lr *= 0.5
        if it % 2000 == 0:
            lr *= 0.95
    return best_h, best_c
