def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def targeted_pair_search(h_init, n, deadline, max_sweeps=50, seed=42):
    """Targeted pair-swap local search focusing on worst lags."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    cur = h_init.copy()
    corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_c = cur_max
    best_h = cur.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(cur >= 0.999)[0]
        zeros = np.where(cur <= 0.001)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find worst lags (within a small tolerance of the max)
        threshold = cur_max - 3.0  # Allow moves that reduce near-worst lags too
        worst_lags = np.where(corr >= threshold)[0]
        
        for _ in range(n * 2):  # Try 2n random swaps per sweep
            if time.time() > deadline:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            # Targeted selection: prefer swaps that affect worst lags
            i = rng.choice(ones)
            j = rng.choice(zeros)
            
            cur[i] = 0.0
            cur[j] = 1.0
            new_corr = np.correlate(cur, 1.0 - cur, mode='full')
            new_max = new_corr.max()
            
            if new_max <= cur_max:
                # Accept (same or better)
                corr = new_corr
                cur_max = new_max
                improved = True
                if cur_max < best_c:
                    best_c = cur_max
                    best_h = cur.copy()
                # Update ones and zeros
                ones = np.where(cur >= 0.999)[0]
                zeros = np.where(cur <= 0.001)[0]
            else:
                # Reject
                cur[i] = 1.0
                cur[j] = 0.0
        
        if not improved:
            break
    
    return best_h, float(best_c * dx)

def annealed_search(h_init, n, deadline, max_iter=200000, seed=42):

    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        T *= gamma
        
        i = rng.choice(ones)

        cur[i] = 0.0
        cur[j] = 1.0
        new_corr = np.correlate(cur, 1.0 - cur, mode='full')
        new_max = new_corr.max()
        delta = new_max - cur_max

            cur_max = new_max

                best_h = cur.copy()
            # Update index sets
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    
    return best_h, float(best_c * dx)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(g)
        sh = alpha * (g - m)
        sm = np.max(sh)
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

- h'[l] = h[l] - δ

The change in C[n-1+k] (the correlation at lag k):
ΔC[k] = sum_m (h'[m](1-h'[m+k]) - h[m](1-h[m+k]))

Only terms m=j, m=l, m=j-k, m=l-k can change (and their counterparts m+k = j, l, i.e., m = j-k, l-k).

Let me enumerate:
- m=j: h'[j](1-h'[j+k]) - h[j](1-h[j+k]) = (h[j]+δ)(1-h[j+k]) - h[j](1-h[j+k]) = δ(1-h[j+k]) [if j+k < n]
- m=l: h'[l](1-h'[l+k]) - h[l](1-h[l+k]) = (h[l]-δ)(1-h[l+k]) - h[l](1-h[l+k]) = -δ(1-h[l+k]) [if l+k < n]
- m=j-k (if 0 ≤ j-k < n): h'[j-k](1-h'[j]) - h[j-k](1-h[j])
  = h[j-k](1-h[j]-δ) - h[j-k](1-h[j]) = -δ*h[j-k]
- m=l-k (if 0 ≤ l-k < n): h'[l-k](1-h'[l]) - h[l-k](1-h[l])
  = h[l-k](1-h[l]+δ) - h[l-k](1-h[l]) = δ*h[l-k]

So: ΔC[k] = δ * [(1-h[j+k])_valid - (1-h[l+k])_valid - h[j-k]_valid + h[l-k]_valid]
where h[a]_valid = h[a] if 0 ≤ a < n, else 0.

Let d_k = (1-h[j+k])_valid - (1-h[l+k])_valid - h[j-k]_valid + h[l-k]_valid

Then C'[k] = C[k] + δ * d_k.

We want: minimize max_k (C[k] + δ * d_k) subject to:
- 0 ≤ h[j] + δ ≤ 1  →  -h[j] ≤ δ ≤ 1-h[j]
- 0 ≤ h[l] - δ ≤ 1  →  h[l]-1 ≤ δ ≤ h[l]
- So: δ ∈ [max(-h[j], h[l]-1), min(1-h[j], h[l])]

This is a 1D convex optimization (minimize the max of linear functions). The optimal δ is:
- At a point where two lines C[k1] + δ*d[k1] = C[k2] + δ*d[k2] both achieve the maximum
- Or at a boundary point

To find it efficiently:
1. Evaluate f(δ) = max_k (C[k] + δ*d_k) at δ=0
2. Find the k* that achieves the max at δ=0
3. If d[k*] > 0: the max increases with δ, so we want smaller δ
   If d[k*] < 0: the max decreases with δ, so we want larger δ
4. Iterate: at the current δ, find the active k*, and move δ in the direction that reduces the max

This is essentially a subgradient method on the 1D problem, which converges in O(n) iterations (since there are only n+1 "breakpoints" where the active set changes).
