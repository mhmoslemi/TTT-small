Actually, for the SA, the main loop does:
1. Pick i, j (O(1))
2. Compute delta (O(n) with numpy overhead)
3. Compute new_corr = corr + delta (O(n))
4. Compute nc = new_corr.max() (O(n))
5.偶尔 recompute zeros_idx and ones_idx (O(n))

Steps 2-4 together are about 3-4 numpy operations on arrays of size ~500, which might take 5-10 microseconds each. So each SA step is maybe 20-50 microseconds. In 400 seconds, we can do about 8M-20M steps.

The previous code's SA also recomputes zeros_idx and ones_idx after every accepted move, which is O(n) in Python (using np.where). This is probably the main bottleneck.

**Optimization idea**: Instead of recomputing zeros_idx and ones_idx after every move, maintain them incrementally. When we swap i and j, we just need to move i from zeros to ones and j from ones to zeros. We can use a set or a list with position tracking.

But actually, the simplest optimization is: don't recompute every time. Instead, recompute every K steps (e.g., every 100 steps) or when the index is not in the current set.

```python
def fast_sa(h_bin, n, deadline, seed=42, t_start=0.05, t_end=0.0001):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    # Adaptive cooling
    steps_target = 5000000
    cool_rate = (t_end / t_start) ** (1.0 / steps_target)
    T = t_start
    step = 0
    no_improve = 0
    patience = 50000
    
    while time.time() < deadline:
        # Pick random i (zero) and j (one)
        # Expected O(1) since half are zeros
        i = rng.randint(n)
        while h[i] != 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] != 1:
            j = rng.randint(n)
        
        if i == j:
            continue
        
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        step += 1
        T *= cool_rate
        
        if no_improve > patience:
            # Restart
            nk = 5 + rng.randint(10)
```

Algorithm:

2. Find the max value M and the set of shifts S at which c[k] = M

   c. If yes, accept the swap and go to step 1
4. If no improving swap found, try swaps that keep max = M but reduce the number of shifts at M (to make it easier to find a swap that reduces M later)

```python
def gradient_lls(h, n, max_iter=500, deadline=None):
    """Minimize max correlation using smooth approximation + gradient"""
    dx = 2.0 / n
    alpha = 100.0
    lr = 0.01
    h = h.copy()
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full') * dx
        m = c.max()
        # Soft max gradient
        sh = alpha * (c - m)
        sm = sh.max()
        weights = np.exp(sh - sm)
        weights /= weights.sum()
        # Gradient of soft max w.r.t. h
        # c[k] = sum_i h[i](1-h[i+k]) * dx
        # dc/dh[j] = sum_k weights[k] * dc[k]/dh[j]
        # dc[k]/dh[j] = (1-h[j+k])*dx - h[j-k]*dx (with validity)
        grad = np.zeros(n)
        m_range = np.arange(-(n-1), n)
        for k_offset in range(len(weights)):
            k = k_offset - (n-1)
            w = weights[k_offset] * alpha * dx
            if w < 1e-10:
                continue
            # dc[k]/dh[j] for each j
            # = (1-h[j+k]) for valid j+k, minus h[j-k] for valid j-k
            jm = np.arange(n) + k
            msk = (jm >= 0) & (jm < n)
            grad[msk] += w * (1 - h[jm[msk]])
            jmn = np.arange(n) - k
            msk2 = (jmn >= 0) & (jmn < n)
            grad[msk2] -= w * h[jmn[msk2]]
        # Project gradient: maintain sum(h) = n/2
        grad -= grad.mean()
        # Clip to bounds
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        # Project to sum constraint
        h_new = project_feasible(h_new, n)
        if compute_c5(h_new, n) < compute_c5(h, n):
            h = h_new
    return h
```

def thue_morse_binary(n):
    """Generate Thue-Morse based binary sequence with exactly n/2 ones"""
    # Generate Thue-Morse sequence
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Adjust to have exactly n//2 ones
    target = n // 2
    ones_count = tm.sum()
    if ones_count > target:
        # Remove excess ones
        one_idx = np.where(tm == 1)[0]

        tm[remove] = 0

        add = rng.choice(zero_idx, size=target - ones_count, replace=False)
        tm[add] = 1

    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    cool_rate = (t_end / t_start) ** (1.0 / 2000000.0)
    T = t_start
    no_improve = 0
    
    while time.time() < deadline:
        i = rng.randint(n)
        while h[i] != 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] != 1:
            j = rng.randint(n)
        if i == j:
            continue
        delta = swap_delta(h, n, i, j, m_range)
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(8)
            for _ in range(nk):
                ii = rng.randint(n)
                while h[ii] != 0:
                    ii = rng.randint(n)
                jj = rng.randint(n)
                while h[jj] != 1:
                    jj = rng.randint(n)
                if ii != jj:
                    h[ii] = 1.0
                    h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = t_start * (0.3 + 0.4 * rng.random())
    return best_h, best_c

def best_improve_ls(h_bin, n, deadline, seed=42):
    """Best improvement local search: check all pairs, take best"""
    dx = 2.0 / n
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    while time.time() < deadline:
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_delta_c = 0
        best_i, best_j = -1, -1
        best_delta_arr = None
        for i in zeros:
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_delta_c - 1e-15:
                    best_delta_c = dc
                    best_i, best_j = i, j
                    best_delta_arr = delta
        if best_i >= 0:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = corr + best_delta_arr
            cur_c += best_delta_c
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
