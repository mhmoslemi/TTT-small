def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / exp_c.sum()
    
    j = np.arange(n)
    k = np.arange(2 * n - 1) - (n - 1)
    jk = j[:, None] + k[None, :]
    jm = j[:, None] - k[None, :]
    mask_k = (jk >= 0) & (jk < n)
    mask_m = (jm >= 0) & (jm < n)
    jk_c = np.clip(jk, 0, n - 1)
    jm_c = np.clip(jm, 0, n - 1)
    h_jk = h[jk_c]
    h_jm = h[jm_c]
    a = (p[None, :] * (1.0 - h_jk) * mask_k).sum(axis=1)
    b = (p[None, :] * h_jm * mask_m).sum(axis=1)
    return dx * (a - b)

def pd_optimize(h, n, alpha_start, alpha_end, lr, max_iters, deadline, eval_every=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_start
    alpha_mult = (alpha_end / alpha_start) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = smooth_gradient(h, n, alpha)
        h = h - lr * grad
        h = project_feasible(h, n)
        if (it + 1) % eval_every == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        alpha *= alpha_mult
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c
