def projected_gradient(h, n, alpha, max_steps=5000, deadline=None):

    best_h = h.copy()

    for it in range(max_steps):

        # Backtracking line search
        f0 = obj
        g = -grad
        t = step

            trial = project_feasible(trial, n)

def lp_oo_step(h, n, trust, deadline):

    cons = [cp.sum(hv) == n/2.0, hv >= 0, hv <= 1]

    cons.append(hv <= h + trust + eps)
    # Only add constraints for lags near the maximum to keep LP small
    threshold = c.max() - 0.02

        if c[ki] < threshold:
            continue
        lag = ki - center
        a = abs(lag)
        if a == 0:
            g = dx * (1.0 - 2.0 * h)
        elif a < n:
            if lag > 0:
                g = np.zeros(n)

            else:

                g = np.zeros(n)
                g[:n-M] += dx * (1.0 - h[M:])
                g[M:] += -dx * h[:n-M]
        else:
            continue

            break

    if hv.value is None:

    return project_feasible(hv.value, n)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    proj = np.clip(h - lam, 0, 1)
    r = target - proj.sum()
    if abs(r) > 1e-13:
        if r > 0:
            room = 1.0 - proj
            idx = np.where(room > 1e-14)[0]
            if len(idx) > 0:
                proj[idx] += r * room[idx] / room[idx].sum()
        else:
            idx = np.where(proj > 1e-14)[0]
            if len(idx) > 0:
                proj[idx] += r * proj[idx] / proj[idx].sum()
    return np.clip(proj, 0.0, 1.0)

        a = abs(lag)
        if a == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif a < n:
            t = np.zeros(n)
            if lag > 0:
                t[lag:] += 1.0 - h[:n - a]
                t[:n - a] += -h[a:]
            else:
                M = a
                t[:n - M] += 1.0 - h[M:]
                t[M:] += -h[:n - M]
            grad += wi * dx * t
    return obj, grad

def projected_gradient(h, n, alpha, max_steps=3000, deadline=None, ftol=1e-15):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    best_obj, _ = smooth_obj_grad(h, n, alpha)
    step = 1.0 / (alpha * 4.0 / n + 1.0)
    stale = 0
    for it in range(max_steps):
        if deadline is not None and time.time() > deadline:
            break
        obj, grad = smooth_obj_grad(h, n, alpha)
        if obj < best_obj - ftol:
            best_obj = obj
            stale = 0
        else:
            stale += 1
            if stale > 200:
                break
        g = -grad
        t = step
        improved = False
        for _ in range(20):
            trial = h + t * g
            trial = project_feasible(trial, n)
            f1, _ = smooth_obj_grad(trial, n, alpha)
            if f1 <= obj - 1e-4 * t * np.dot(grad, grad):
                h = trial
                improved = True
                break
            t *= 0.5
        if not improved:
            h = project_feasible(h + t * g, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        step = min(step * 1.02, 2.0 / (alpha * 4.0 / n + 1.0))
    return best_h, best_c

def _lag_grad(h, n, lag):
    dx = 2.0 / n
    a = abs(lag)
    if a == 0:
        return dx * (1.0 - 2.0 * h)
    if a >= n:
        return np.zeros(n)
    t = np.zeros(n)
    if lag > 0:
        t[lag:] += dx * (1.0 - h[:n - a])
        t[:n - a] += -dx * h[a:]
    else:
        M = a
        t[:n - M] += dx * (1.0 - h[M:])
        t[M:] += -dx * h[:n - M]
    return t

def lp_oo_step(h, n, trust):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    center = n - 1
    threshold = c.max() - 0.015
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1]
    cons.append(hv >= h - trust)
    cons.append(hv <= h + trust)
    n_cons = 0
    for ki in range(len(c)):
        if c[ki] < threshold:
            continue
        lag = ki - center
        a = abs(lag)
        if a == 0 or a >= n:
            continue
        g = _lag_grad(h, n, lag)
        cons.append(c[ki] + g @ (hv - h) <= tv)
        n_cons += 1
    if n_cons == 0:
        return h.copy()
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.GLOP, cp.CLARABEL, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and hv.value is not None:
                return project_feasible(hv.value, n)
        except Exception:
            continue
    return h.copy()

def local_refine(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h)
    center = n - 1
    for d in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
        improved = True
        while improved:
            if time.time() > deadline:
                return h, best_c
            improved = False
            c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            top_k = np.argsort(c)[-3:]
            for ki in top_k:
                lag = int(ki) - center
                a = abs(lag)
                if a == 0 or a >= n:
                    continue
                if lag > 0:
                    pairs = [(i, i + a) for i in range(0, n - a, 3)]
                else:
                    M = a
                    pairs = [(i, i + M) for i in range(0, n - M, 3)]
                for (i, j) in pairs:
                    if time.time() > deadline:
                        return h, best_c
                    if h[i] > h[j] + 1e-7:
                        delta = min(d, (h[i] - h[j]) * 0.5)
                        if delta < 1e-9:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        if compute_c5(h) < best_c - 1e-14:
                            best_c = compute_c5(h)
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    elif h[j] > h[i] + 1e-7:
                        delta = min(d, (h[j] - h[i]) * 0.5)
                        if delta < 1e-9:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        if compute_c5(h) < best_c - 1e-14:
                            best_c = compute_c5(h)
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
    return h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    # Phase 1: LP outer approximation
    if time.time() < deadline - 80:
        trust = 0.12
        for it in range(6):
            if time.time() > deadline - 80:
                break
            h_new = lp_oo_step(h, n, trust)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            trust *= 0.65
    # Phase 2: Projected gradient with increasing alpha
    for alpha in [300, 1500, 6000, 25000]:
        if time.time() > deadline - 40:
            break
        h_pg, c_pg = projected_gradient(best_h, n, alpha,
                                        max_steps=1200, deadline=deadline - 40)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 3: More LP with smaller trust
    if time.time() < deadline - 30:
        trust = 0.04
        for it in range(4):
            if time.time() > deadline - 30:
                break
            h_new = lp_oo_step(best_h, n, trust)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            trust *= 0.55
    # Phase 4: Final gradient polish
    if time.time() < deadline - 20:
        h_pg, c_pg = projected_gradient(best_h, n, 60000,
                                        max_steps=2000, deadline=deadline - 20)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
    # Phase 5: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_refine(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(380, (deadline - t0) * 0.33)
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    # Try other n values
    n_candidates = [200, 250, 350, 400]
    if h_init is not None:
        n_candidates = [x for x in n_candidates if x != n_init]
    for n_try in n_candidates:
        if time.time() > deadline - 90:
            break
        remaining = deadline - time.time() - 30
        if remaining < 70:
            break
        alloc = min(110, remaining * 0.25)
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_n(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Random restarts
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        attempts = 0
        while time.time() < deadline - 40 and attempts < 8:
            attempts += 1
            scale = 0.012 * (1.0 + 0.5 * attempts * 0.3)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, max(25, (deadline - time.time() - 20) * 0.35))
            if alloc < 22:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_pg, c_pg = projected_gradient(best_h, n, 100000,
                                        max_steps=2500, deadline=deadline - 10)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg
        best_h, best_c = local_refine(best_h, n, max(deadline - 1, time.time() + 8))
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
