def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, weights):
    """Compute weighted gradient of max correlation.
    weights: array of length 2n-1, same indexing as np.correlate output.
    """
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        d = idx - (n - 1)  # lag in [-n+1, n-1]
        w = weights[idx]
        if w < 1e-15:
            continue

            valid_k = valid_j - d  # = j-d, in [0, n-1]
            grad[j_start:j_end] += w * (1 - h[valid_k])

            valid_k2 = valid_j2 + d  # = j+d, in [0, n-1]

def proj_grad_descent(h, n, deadline, lr_init=0.01, lr_final=0.001, n_iters=50000):
    """Projected gradient descent with softmax-weighted subgradient."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = corr.max() * (2.0 / n)
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Softmax weights on correlations
        alpha = 5.0 + 5.0 * (it / n_iters)  # increasing sharpness
        shifted = alpha * (corr - corr.max())
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1); w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        # Compute weighted gradient
        # For efficiency, only use top 20 lags
        top_k = min(20, 2*n-1)
        top_idx = np.argsort(corr)[-top_k:]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()
        
        grad = np.zeros(n)
        for idx in top_idx:
            d = idx - (n - 1)
            wi = w_top[np.where(top_idx == idx)[0][0]]
            # First term
            js = max(0, d)
            je = min(n, d + n)
            if js < je:
                grad[js:je] += wi * (1 - h[js-d:je-d])
            # Second term
            js2 = max(0, -d)
            je2 = min(n, n - d)
            if js2 < je2:
                grad[js2:je2] -= wi * h[js2+d:je2+d]
        
        # Step
        lr = lr_init * (lr_final / lr_init) ** (it / n_iters)
        h = h - lr * grad
        
        # Project
        h = project_feasible(h, n)
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def weighted_gradient(h, n, corr, top_k=50):
    """Compute weighted gradient using top-k lags."""
    # Use softmax weighting
    alpha = 10.0
    max_c = corr.max()
    w = np.exp(alpha * (corr - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    
    # Get top indices
    top_idx = np.argsort(corr)[-top_k:]
    w_top = w[top_idx].copy()
    w_top_sum = w_top.sum()
    if w_top_sum < 1e-30:
        return np.zeros(n)
    w_top /= w_top_sum
    
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        d = idx - (n - 1)
        wi = w_top[i]
        # First term: (1-h[j-d]) for j in [max(0,d), min(n, d+n))
        js = max(0, d)
        je = min(n, d + n)
        if js < je:
            grad[js:je] += wi * (1 - h[js-d:je-d])
        # Second term: -h[j+d] for j in [max(0,-d), min(n, n-d))
        js2 = max(0, -d)
        je2 = min(n, n - d)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2+d:je2+d]
    return grad

def proj_grad_descent(h_init, n, deadline, n_iters=100000):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        cur_c = corr.max() * (2.0 / n)
        
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
        
        # Adaptive lr
        lr = 0.5 / (1.0 + it * 0.01)
        
        grad = weighted_gradient(h, n, corr, top_k=50)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check if we improved
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c = corr_new.max() * (2.0 / n)
        
        if new_c <= cur_c + 1e-12:
            h = h_new
        else:
            # Halve lr and retry
            h_new = h - lr * 0.1 * grad
            h_new = project_feasible(h_new, n)
            h = h_new
    
    return best_h, best_c

def binary_swap_search(h_init, n, deadline, max_sweeps=200):
    """Local search on binary solutions via swaps."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    
    ones_arr = np.where(h == 1)[0].astype(int)
    zeros_arr = np.where(h == 0)[0].astype(int)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try random swaps
        for _ in range(n):
            if time.time() > deadline:
                break
            ai = np.random.randint(len(zeros_arr))
            bi = np.random.randint(len(ones_arr))
            i = zeros_arr[ai]
            j = ones_arr[bi]
            
            # Compute delta
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            delta[k2] += 1.0
            
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                zeros_arr[ai] = j
                ones_arr[bi] = i
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            delta[k2] += 1.0
            nc = float((corr + delta).max() * dx)
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                za[ai] = j
                oa[bi] = i
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

1. The `proj_grad` function: `step = lr_t * grad` where `lr_t = base_lr * (1+0.0005*it)^(-0.5)` and `base_lr = 0.5/n`. For n=400, base_lr = 0.00125. At it=0, lr_t = 0.00125. The gradient components are in [-1, 1] (weighted sum of terms each in [-1,1]). So the step is at most 0.00125 per component. That seems reasonable - we're making small adjustments that should bring us closer to feasibility.

2. The line search: if the full step doesn't improve, try 5% of the step. This is very conservative but should make progress.

3. The stale counter: if we go 5000 iterations without improving, we stop. This prevents wasting time.

4. The alpha schedule: starts at alpha_init and increases to alpha_final over the first half of max_it. Higher alpha means more focus on the worst lags.

I think this should work well. The key advantage over the previous SLSQP approach is speed - each iteration is O(n²) for the correlate + O(top_k * n) for the gradient, vs O(n³) or worse for SLSQP. This allows many more iterations in the same time budget.

One concern: the learning rate might be too small, leading to very slow progress. Let me increase it slightly. Actually, let me think about what step size makes sense.

The objective change from a step `step` is approximately `grad · step = lr_t * ||grad||²`. With ||grad|| ~ 1 (typical), the improvement per step is ~ lr_t. To improve by 0.001 in C5 (which is ~0.001 * n/2 = 0.2 in correlation units), we need ~200 steps with lr_t = 0.00125. That's fast.

But the issue is that the gradient points in the direction that reduces the worst lag, but might increase other lags. The softmax weighting helps with this, but it's not perfect. The line search (accept/reject) handles this.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def weighted_gradient(h, n, corr, alpha, top_k):
    max_c = corr.max()
    w = np.exp(alpha * (corr - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= w_sum
    idx = np.argsort(corr)[-top_k:]
    wt = w[idx].copy()
    s = wt.sum()
    if s < 1e-30:
        return np.zeros(n)
    wt /= s
    grad = np.zeros(n)
    for ii in range(len(idx)):
        d = int(idx[ii]) - (n - 1)
        wi = wt[ii]
        js = max(0, d)
        je = min(n, d + n)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js - d:je - d])
        js2 = max(0, -d)
        je2 = min(n, n - d)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2 + d:je2 + d]
    return grad

def proj_grad(h_init, n, deadline, alpha_init=3.0, alpha_final=80.0, top_k=60):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    it = 0
    max_it = 300000
    base_lr = 1.0 / n
    stale = 0
    while it < max_it and time.time() < deadline:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(corr.max() * (2.0 / n))
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 8000:
            break
        frac = min(1.0, it / (max_it * 0.4))
        alpha = alpha_init + (alpha_final - alpha_init) * frac
        lr_t = base_lr * (1.0 + 0.0003 * it) ** (-0.5)
        grad = weighted_gradient(h, n, corr, alpha, top_k)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-15:
            break
        step = lr_t * grad
        h_new = project_feasible(h - step, n)
        c_new = compute_c5(h_new, n)
        if c_new <= cur_c + 1e-12:
            h = h_new
        else:
            h = project_feasible(h - 0.1 * step, n)
    return best_h, best_c

def binary_swap(h_init, n, deadline, max_sweeps=500, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    oa = np.where(h == 1)[0].astype(np.int64)
    za = np.where(h == 0)[0].astype(np.int64)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(2 * n):
            if time.time() > deadline:
                break
            ai = rng.randint(len(za))
            bi = rng.randint(len(oa))
            i = int(za[ai])
            j = int(oa[bi])
            delta = np.zeros(2 * n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            delta[msk] += h[jmn[msk]]
            delta[n - 1] -= 2.0
            k1 = n - 1 + (j - i)
            delta[k1] += 1.0
            k2 = n - 1 + (i - j)
            delta[k2] += 1.0
            nc = float((corr + delta).max() * dx)
            if nc < cur_c - 1e-15:
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = nc
                za[ai] = j
                oa[bi] = i
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
