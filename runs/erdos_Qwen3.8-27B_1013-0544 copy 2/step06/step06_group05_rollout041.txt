    """Best-improvement local search for binary h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_improve = 0
        best_pair = None
        
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                new_corr = np.correlate(h, 1.0 - h, mode='full')

                h[i] = 0.0

                if new_max < cur_max - best_improve:
                    best_improve = cur_max - new_max
                    best_pair = (i, j)

        h[j] = 0.0

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b

def local_search(h, n, deadline, max_iters=300):
    """Best-improvement 1-exchange local search."""
    cur_max = float(np.correlate(h, 1.0-h, mode='full').max())
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_imp = 0.0
        best_pair = None
        for i in zeros:
            for j in ones:
                h[i] = 1.0; h[j] = 0.0
                nm = np.correlate(h, 1.0-h, mode='full').max()
                h[i] = 0.0; h[j] = 1.0
                imp = cur_max - nm
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]] = 1.0
        h[best_pair[1]] = 0.0
        cur_max = float(np.correlate(h, 1.0-h, mode='full').max())
    return h, cur_max

def two_exchange_search(h, n, deadline, n_samples=400, max_iters=100, seed=42):
    rng = np.random.RandomState(seed)
    cur_max = float(np.correlate(h, 1.0-h, mode='full').max())
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_imp = 0.0
        best_mv = None
        for _ in range(n_samples):
            ii = rng.choice(len(zeros), 2, replace=False)
            jj = rng.choice(len(ones), 2, replace=False)
            i1, i2 = int(zeros[ii[0]]), int(zeros[ii[1]])
            j1, j2 = int(ones[jj[0]]), int(ones[jj[1]])
            h[i1]=1.0; h[i2]=1.0; h[j1]=0.0; h[j2]=0.0
            nm = float(np.correlate(h, 1.0-h, mode='full').max())
            h[i1]=0.0; h[i2]=0.0; h[j1]=1.0; h[j2]=1.0
            imp = cur_max - nm
            if imp > best_imp:
                best_imp = imp
                best_mv = (i1,i2,j1,j2)
        if best_mv is None:
            break
        h[best_mv[0]]=1.0; h[best_mv[1]]=1.0
        h[best_mv[2]]=0.0; h[best_mv[3]]=0.0
        cur_max = float(np.correlate(h, 1.0-h, mode='full').max())
    return h, cur_max

def slsqp_refine(h, n, alpha, maxiter, deadline):
    dx = 2.0/n
    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full')*dx
        m = g.max()
        sh = alpha*(g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh-sm).sum()))/alpha
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter, 'ftol':1e-18})
        return project_feasible(res.x, n)
    except:
        return h
