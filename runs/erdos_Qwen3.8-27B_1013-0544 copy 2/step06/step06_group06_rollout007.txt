    return float(np.max(c) * dx)

def compute_grad(h, n, lag):
    """Compute gradient of c_k with respect to h, where k = lag + (n-1)"""
    grad = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        # d/dh[j] = (1-h[j-lag]) for lag <= j <= n-1
        #          - h[j+lag] for 0 <= j <= n-1-lag
        valid_range = slice(0, n - lag)
        grad[lag:] += (1 - h[:n-lag])  # (1-h[j-lag]) for j = lag, ..., n-1
        grad[:n-lag] -= h[lag:]  # -h[j+lag] for j = 0, ..., n-1-lag
    else:
        m = -lag
        # c = sum_{i=m}^{n-1} h[i] * (1-h[i-m])
        # d/dh[j] = (1-h[j+m]) for 0 <= j <= n-1-m
        #          -h[j-m] for m <= j <= n-1
        grad[:n-m] += (1 - h[m:])  # (1-h[j+m]) for j = 0, ..., n-1-m
        grad[m:] -= h[:n-m]  # -h[j-m] for j = m, ..., n-1
    return grad

def project_feasible(h, target):
    """Project onto {x: sum(x) = target, 0 <= x <= 1}"""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0, 1)

def project_exact(h, target):
    """Exact projection using bisection on Lagrange multiplier"""
    h = np.array(h, dtype=float)
    lo, hi = -1.0, max(1.0 - h.min(), 0.0) + 1.0
    
    def f(lam):
        return np.clip(h - lam, 0, 1).sum() - target
    
    if f(0) == target:  # already at target with lam=0? No, f(0) = h.sum() - target
        pass
    
    # Bisection
    a, b = -max(h.max(), 1.0) - 1, max(1.0 - h.min(), 1.0) + 1
    fa = f(a)  # should be positive (sum too large when lam very negative)
    fb = f(b)  # should be negative (sum too small when lam large)
    
    for _ in range(60):
        mid = (a + b) / 2
        fm = f(mid)
        if fm > 0:
            a = mid
        else:
            b = mid
    
    lam = (a + b) / 2
    return np.clip(h - lam, 0, 1)

            if c_new < best_c:

def slsqp_polish(h0, n, deadline, alpha=100000):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        # Soft max
        shifted = c - m
        return m + np.log(np.exp(alpha * shifted).sum()) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()
        # gradient of soft max = sum_k weights[k] * grad(c_k)
        grad = np.zeros(n)
        for k in range(len(c)):
            if weights[k] > 1e-10:
                lag = k - (n-1)
                grad += weights[k] * dx * compute_grad(x, n, lag)
        return grad
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', 
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 1000, 'ftol': 1e-12})
    if res.success:
        return project_feasible(res.x, target)
    return h0

- Repeat

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_grad(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h

def thue_morse_init(n, seed=42):
    rng = np.random.RandomState(seed)
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_grad(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()

            if s < 1e-14:
