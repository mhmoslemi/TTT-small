    c = np.correlate(h, 1.0 - h, mode='full')

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            s = room.sum()
            if s < 1e-14:

            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

**Algorithm:**

3. Iterative reweighting:
   a. Compute overlaps L_k
   b. Set weights w_k ∝ exp(β * L_k / max_L) for large β
   c. Optimize: min_h Σ w_k L_k(h) subject to constraints
      This is: min_h Σ w_k * dx * Σ_i h[i](1-h[i+k])
      = min_h [dx * Σ_i h[i] * Σ_k w_k - dx * Σ_i Σ_k w_k h[i]h[i+k]]
      = constant - dx * h^T (Σ_k w_k A_k) h
      = constant - dx * h^T B h
      where B = Σ_k w_k A_k
      So we want to MAXIMIZE h^T B h subject to constraints.
      
      This is a quadratic optimization. If B is PSD, it's trivial (max at boundary). If not, it's non-convex.
      
      Actually, B is a sum of shift matrices weighted by positive weights. Each shift matrix A_k has 1s on the k-th diagonal. So B has entries B[i,j] = Σ_k w_k * 1_{j=i+k} = w_{j-i} (approximately). So B is a Toeplitz matrix (approximately circulant).
      
      For a symmetric Toeplitz matrix, the eigenvalues can be computed via FFT. If all eigenvalues are positive, B is PSD and the max of h^T B h over the hypercube with sum constraint is... complicated.
      
      Let me just use SLSQP to maximize h^T B h (or equivalently, minimize -h^T B h) with the constraints. This is a QP (quadratic objective, linear constraints) which SLSQP can handle well.

   d. Repeat with updated weights

This iterative reweighting + QP approach might be more effective than the log-sum-exp approach because:

import numpy as np
from scipy.optimize import minimize
from scipy.linalg import toeplitz
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            for j in list(ones_list):
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def reweight_qp(h_init, n, deadline, n_iter=30):
    """Iterative reweighting with QP subproblem."""
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    betas = np.linspace(5, 100, n_iter)
    
    for beta in betas:
        if time.time() > deadline - 15:
            break
        # Compute overlaps
        overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        max_ov = overlaps.max()
        if max_ov < 1e-15:
            break
        # Weights
        scaled = beta * overlaps / max_ov
        w = np.exp(scaled - scaled.max())
        w /= w.sum()
        
        # Build Toeplitz B
        first_col = w[n-1::-1]  # w[n-1], w[n-2], ..., w[0]
        first_row = w[n-1:2*n-1]  # w[n-1], w[n], ..., w[2n-2]
        B = toeplitz(first_col, first_row)
        
        # Maximize h^T B h s.t. sum(h)=n/2, 0<=h<=1
        # Using SLSQP
        def obj(x):
            return -x @ B @ x
        def obj_grad(x):
            return -2.0 * (B @ x)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c

    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()

    best_c = cur_c

    zeros_arr = np.where(h == 0)[0].copy()
    T = 0.02
    cool = 0.99998
    scale = 5000.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ai = rng.randint(len(zeros_arr))
        bi = rng.randint(len(ones_arr))
        i = zeros_arr[ai]
        j = ones_arr[bi]
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-10 and rng.random() < np.exp(scale * (cur_c - nc) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_arr[ai] = j
            ones_arr[bi] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        T *= cool

Actually, let me reconsider the projection step. The projection onto {h: sum(h) = n/2, 0 ≤ h ≤ 1} is not trivial. The simple approach (clip then adjust sum) works but might not be the true Euclidean projection. For the projected gradient method, using the true projection is important for convergence.

The true Euclidean projection onto {h: sum(h) = c, 0 ≤ h ≤ 1} can be computed by:
1. Find λ such that sum(clip(h - λ, 0, 1)) = c
2. Return clip(h - λ, 0, 1)

This is a 1D root-finding problem, which can be solved by bisection.

Let me implement this:

```python
def project_exact(h, n):
    """Exact Euclidean projection onto {x: sum(x)=n/2, 0<=x<=1}."""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
```

This is O(n) per bisection step, 100 steps, so O(100n) per projection. For n=400, that's 40,000 operations. Very fast.

from scipy.optimize import minimize
from scipy.linalg import toeplitz
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_exact(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greedy_binary(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_list):
            if time.time() > deadline:
                break
            for j in list(ones_list):
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                k2 = n - 1 + (i - j)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)

                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
