```
def project_sum_fast(h, S):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    S = float(S)
    for _ in range(3):
        diff = S - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cap = (1.0 - h[mask]).sum()
            if cap <= 1e-15:
                break
            s = diff / cap
            h[mask] += s * (1.0 - h[mask])
        else:
            mask = h > 0.0
            cap = h[mask].sum()
            if cap <= 1e-15:
                break
            s = diff / cap
            h[mask] += s * h[mask]
    h = np.clip(h, 0.0, 1.0)
    # final exact adjustment tiny
    diff = S - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            cap = (1.0-h).sum()
            if cap > 1e-15:
                h += (diff/cap)*(1.0-h)
                h = np.minimum(h,1.0)
        else:
            cap = h.sum()
            if cap > 1e-15:
                h += (diff/cap)*h
                h = np.maximum(h,0.0)
    return h
```

```
          if cap<=1e-15: break
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_r(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c(h, n):
    r = compute_r(h)
    return float(np.max(r) * (2.0 / n))


def sanitize_h(h, n):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or len(h) != n:
        h = np.resize(h, n)
    h = np.where(np.isfinite(h), h, 0.5)
    return np.clip(h, 0.0, 1.0)


def project_sum_fast(h, S):
    h = sanitize_h(h, len(h))
    S = float(S)
    for _ in range(3):
        diff = S - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0
            cap = (1.0 - h[mask]).sum()
            if cap <= 1e-15:
                break
            s = diff / cap
            h[mask] += s * (1.0 - h[mask])
        else:
            mask = h > 0.0
            cap = h[mask].sum()
            if cap <= 1e-15:
                break
            s = diff / cap
            h[mask] += s * h[mask]
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            cap = (1.0 - h).sum()
            if cap > 1e-15:
                h = np.minimum(1.0, h + (diff / cap) * (1.0 - h))
        else:
            cap = h.sum()
            if cap > 1e-15:
                h = np.maximum(0.0, h + (diff / cap) * h)
    return h


def grad_r_weighted(w, h, n):
    c = n - 1
    pw = np.concatenate(([0.0], np.cumsum(w), [0.0]))
    p = np.arange(n)
    lin = pw[p + n] - pw[p]

    t = np.zeros(n)
    t[0] = 2.0 * w[c]
    if n > 1:
        d = np.arange(1, n)
        t[d] = w[c - d] + w[c + d]

    quad = np.zeros(n)
    if abs(t[0]) > 1e-15:
        quad += t[0] * h
    nz = np.where(t > 1e-13)[0]
    for d in nz:
        if d == 0:
            continue
        td = t[d]
        m = n - d
        quad[:m] += td * h[d:]
        quad[d:] += td * h[:m]
    return lin - quad


def soft_core(h, alpha, n):
    r = np.correlate(h, 1.0 - h, mode="full")
    mr = float(np.max(r))
    a = alpha * (r - mr)
    a -= float(np.max(a))
    e = np.exp(a)
    se = float(e.sum())
    if se <= 0.0:
        p = np.full(len(r), 1.0 / len(r))
    else:
        p = e / se
    g = grad_r_weighted(p, h, n)
    return mr + np.log(se) / alpha, g


def soft_h_objective(x, n, alpha, mu, dx, target):
    h = np.asarray(x, dtype=float)
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    soft, g = soft_core(h, alpha, n)
    val = dx * soft
    grad = dx * g
    if mu > 0.0:
        dev = h.sum() - target
        val += mu * dev * dev
        grad = grad + 2.0 * mu * dev
    return float(val), grad


def make_bounds(n):
    return [(0.0, 1.0)] * n


def lbfgs_soft(h0, n, alpha, mu, maxfun=100, maxls=8):
    target = n * 0.5
    dx = 2.0 / n
    h0 = project_sum_fast(h0, target)
    mf = max(20, int(maxfun))
    opts = {
        "maxiter": mf,
        "maxfun": mf,
        "ftol": 1e-12,
        "gtol": 1e-8,
        "maxls": int(maxls),
    }
    try:
        res = minimize(
            soft_h_objective,
            h0,
            method="L-BFGS-B",
            jac=True,
            args=(n, alpha, mu, dx, target),
            bounds=make_bounds(n),
            options=opts,
        )
        h = np.asarray(res.x, dtype=float)
        if h is None or not np.all(np.isfinite(h)):
            h = h0
    except Exception:
        h = h0
    return project_sum_fast(h, target)

    dx = 2.0 / n

    best = compute_c(h, n)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    it = 0
    bad = 0
    resets = 0

    while time.time() < deadline:
        it += 1
        _, g = soft_core(h, alpha, n)
        g = dx * g
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mhat = m / (1.0 - b1 ** it)
        vhat = v / (1.0 - b2 ** it)
```
