def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    s = alpha * (c - M)
    sm = s.max()
    es = np.exp(s - sm)
    return M + (sm + np.log(es.sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = c.max()
    s = alpha * (c - M)
    sm = s.max()
    es = np.exp(s - sm)
    w = es / es.sum()
    # S1
    cumw = np.concatenate([[0.0], np.cumsum(w)])
    S1 = np.zeros(n)
    for m in range(n):
        S1[m] = cumw[2*n-1-m] - cumw[n-1-m]
    # S2
    corr_uf = np.correlate(w[n-1:], x, mode='full')
    S2 = corr_uf[:n][::-1]
    # S3
    corr_wf = np.correlate(w, x[::-1], mode='full')
    S3 = corr_wf[n-1:2*n-1]
    return dx * (S1 - S2 - S3)
