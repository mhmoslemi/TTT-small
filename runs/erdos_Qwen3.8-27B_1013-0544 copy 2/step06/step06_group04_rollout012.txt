def compute_c(h,n):

def project_vector(x, target, iters=45):
    n=len(x)
    if target <= 0.0:
       return np.zeros(n)
    if target >= float(n):
       return np.ones(n)
    lo=float(np.min(x))-1.0
    hi=float(np.max(x))+1.0
    for _ in range(iters):
       mid=0.5*(lo+hi)
       y=np.clip(x-mid,0.0,1.0)
       s=float(y.sum())
       if s > target:
          lo=mid
       else:
          hi=mid
    lam=0.5*(lo+hi)
    y=np.clip(x-lam,0.0,1.0)
    # adjust small
    for _ in range(20):
       diff=target - float(y.sum())
       if abs(diff) < 1e-14:
          break
       if diff > 0:
          slack=1.0-y
          idx=int(np.argmax(slack))
          y[idx]=min(1.0, y[idx]+diff)
       else:
          idx=int(np.argmax(y))
          y[idx]=max(0.0, y[idx]+diff)
    return y

import numpy as np
import time


def compute_c(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_vector(x, target, iters=40):
    x = np.asarray(x, dtype=float)
    n = x.size
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)

    xmin = float(np.min(x))
    xmax = float(np.max(x))
    s0 = float(np.sum(x))
    if xmin >= -1e-12 and xmax <= 1.0 + 1e-12 and abs(s0 - target) < 1e-12:
        return np.clip(x, 0.0, 1.0)

    lo = xmin - 1.0
    hi = xmax + 1.0
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - float(np.sum(y))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - y))
            y[idx] = min(1.0, y[idx] + diff)
        else:
            idx = int(np.argmax(y))
            y[idx] = max(0.0, y[idx] + diff)
    return y


def make_h_from_parent(parent, n):
    pn = int(parent.size)
    if pn < 1:
        return project_vector(np.full(n, 0.5), 0.5 * float(n), 60)
    xp = 2.0 * (np.arange(pn) + 0.5) / float(pn)
    xn = 2.0 * (np.arange(n) + 0.5) / float(n)
    h = np.interp(xn, xp, parent)
    return project_vector(h, 0.5 * float(n), 60)


def make_h_cos(n, coeffs, modes):
    pos = 2.0 * (np.arange(n) + 0.5) / float(n)
    s = np.zeros(n)
    for c, m in zip(coeffs, modes):
        if c != 0.0:
            s += float(c) * np.cos(np.pi * float(m) * pos)
    s = 0.5 * (s + s[::-1])
    s = np.clip(s, -1.0, 1.0)
    s -= float(np.mean(s))
    h = 0.5 + 0.5 * s
    return project_vector(h, 0.5 * float(n), 60)

def compute_grad_weights(x, n, p, ks):
    grad = np.zeros(n)
    if p.size == 0:
        return grad
    thr = max(1e-15, float(np.max(p)) * 1e-10)
    active = np.flatnonzero(p > thr)
    for idx in active:
        pu = float(p[idx])
        if pu <= 0.0:
            continue
        k = int(ks[idx])
        if k >= 0:
            a = k
            la = n - a
            grad[a:] += pu * (1.0 - x[:la])
            grad[:la] -= pu * x[a:]
        else:
            a = -k
            la = n - a
            grad[:la] += pu * (1.0 - x[a:])
            grad[a:] -= pu * x[:la]
    return grad


def smooth_objective(h, n, alpha):
    dx = 2.0 / float(n)
    f = np.correlate(h, 1.0 - h, mode="full") * dx
    z = alpha * f
    zm = float(np.max(z))
    s = float(np.sum(np.exp(z - zm)))
    return float((zm + np.log(s)) / alpha)


def run_lse(x, n, alpha, deadline, ks, target):
    x = project_vector(x, target, 50).copy()
    dx = 2.0 / float(n)
    best = x.copy()
    best_c = compute_c(x, n)

    for _ in range(100000):
        if time.time() >= deadline:
            break
        f = np.correlate(x, 1.0 - x, mode="full") * dx
        z = alpha * f
        zm = float(np.max(z))
        ez = np.exp(z - zm)
        p = ez / float(np.sum(ez))
        obj = float((zm + np.log(float(np.sum(ez)))) / alpha)
        exact = float(np.max(f))
        if exact < best_c - 1e-13:
            best = x.copy()
            best_c = exact

        grad = compute_grad_weights(x, n, p, ks) * dx
        norm = float(np.linalg.norm(grad))
        if norm < 1e-14:
            break

        lr = min(1.0, 0.5 / max(norm, 1e-14))
        accepted = False
        for _o in range(14):
            cand = project_vector(x - lr * grad, target, 32)
            objc = smooth_objective(cand, n, alpha)
            if objc < obj - 1e-12:
                x = cand
                accepted = True
                break
            lr *= 0.5
            if lr <= 1e-16:
                break
        if not accepted:
            break

    c = compute_c(x, n)
    if c < best_c:
        best = x.copy()
        best_c = c
    return best, best_c

        starts.append((make_h_cos(n, c2, modes5), n))

        for _rr in range(2):
            cc = np.zeros(7)
            cc[0] = 0.55 + 0.35 * rng.random()
            cc[1] = max(0.0, 0.45 - cc[0] + 0.25 * rng.random())
            cc[2] = 0.20 * rng.random() - 0.08
            cc[3] = 0.12 * rng.random() - 0.05
            cc[4] = 0.08 * rng.random() - 0.04
            cc[5] = 0.05 * rng.random() - 0.02
            starts.append((make_h_cos(n, cc, modes7), n))

    cands_all = []
    best_by_n = []
    counter = 0
    for h, n in starts:
        h = project_vector(h, 0.5 * float(n), 60)
        c = compute_c(h, n)
        cands_all.append((c, n, counter, h))
        counter += 1

        found = False
        for ii in range(len(best_by_n)):
            if best_by_n[ii][1] == n:
                if c < best_by_n[ii][0]:
                    best_by_n[ii] = (c, n, best_by_n[ii][2], h)
                found = True
                break
        if not found:
            best_by_n.append((c, n, counter, h))

    best_by_n.sort()
    cands_all.sort()

    selected = []
    used_sel_n = set()
    used_sel_key = set()
    for t in best_by_n:
        if len(selected) >= 5:
            break
        if t[1] not in used_sel_n:
            selected.append(t)
            used_sel_n.add(t[1])
            used_sel_key.add((t[1], t[2]))
    if len(selected) < 5:
        for t in cands_all:
            if len(selected) >= 5:
                break
            key = (t[1], t[2])
            if key not in used_sel_key:
                selected.append(t)
                used_sel_key.add(key)

    if not selected:
        n = 200
        h = make_h_cos(n, np.array([0.74, 0.26, 0.0, 0.0, 0.0]), modes5)
        return h, compute_c(h, n), n

    best_h = selected[0][3].copy()
    best_n = int(selected[0][1])
    best_c = float(selected[0][0])

    opt_total = max(60.0, float(budget_s) - 180.0)
    fracs = (0.34, 0.24, 0.16, 0.12, 0.10)

    for i in range(len(selected)):
        if time.time() > deadline - 50.0:
            break
        spend = fracs[i] * opt_total if i < len(fracs) else 0.05 * opt_total
        t_end = time.time() + spend
        if t_end < time.time() + 10.0 and time.time() + 10.0 < deadline - 30.0:
            t_end = time.time() + 10.0
        if t_end > deadline - 70.0:
            t_end = min(t_end, deadline - 70.0)
        if time.time() < t_end:
            h2, c2 = polish_candidate(selected[i][3], int(selected[i][1]), t_end, seed + i * 17 + 1)
            if c2 < best_c:
                best_h = h2
                best_n = int(selected[i][1])
                best_c = c2

    if time.time() < deadline - 45.0:
        b = binarize(best_h, best_n)
        t_end = min(deadline - 30.0, time.time() + 40.0)
        h2, c2 = polish_candidate(b, best_n, t_end, seed + 555)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    jitter = (0.03, 0.05, 0.02)
    for r in range(len(jitter)):
        if time.time() > deadline - 35.0:
            break
        pert = best_h + jitter[r] * rng.normal(size=best_n)
        pert = project_vector(pert, 0.5 * float(best_n), 60)
        t_end = min(deadline - 15.0, time.time() + 35.0)
        h2, c2 = polish_candidate(pert, best_n, t_end, seed + 1000 + r * 31)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = enforce_sum(best_h, best_n)
    best_c = compute_c(best_h, best_n)
    return best_h, float(best_c), int(best_n)
