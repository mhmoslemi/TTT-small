def compute_c(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return c, dx

def project(h, n):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0, 1)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = alpha * c.max()
    return float((c.max() + np.log(np.sum(np.exp(alpha * c - mx))) / alpha) * dx)
