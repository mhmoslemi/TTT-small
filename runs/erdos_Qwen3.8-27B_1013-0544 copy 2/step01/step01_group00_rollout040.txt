def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_constrained(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    projected = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        mask = projected < 0.9999 if diff > 0 else projected > 0.0001
        cnt = np.sum(mask)
        if cnt > 0:
            projected[mask] += diff / cnt
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def p_norm_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = np.sum(corr**p) ** (1.0/p)
    if obj < 1e-15:
        return np.zeros(n), 0.0
    scale = obj**(1.0-p)
    W = (corr**(p-1)) * scale
    one_minus_h = 1.0 - h
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):
        A[j] = np.dot(W[n-1 : 2*n-1-j], one_minus_h[j:])
        B[j] = np.dot(W[j : n+j][::-1], h)
    return A - B, float(obj)

def proj_grad_step(h, p, lr, max_backtrack=20):
    """One projected gradient step with backtracking."""
    n = len(h)
    target = n * 0.5
    grad, f_val = p_norm_grad(h, p)
    step = lr
    for _ in range(max_backtrack):
        h_new = h - step * grad
        h_new = project_constrained(h_new, target)
        _, f_new = p_norm_grad(h_new, p)
        if f_new < f_val or step < 1e-10:
            return h_new, f_new
        step *= 0.5
    return h_new, f_new

    n = len(h0)
    dx = 2.0/n
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    
    def obj(v):
        return v[-1]
    def ineq(v):
        hh = v[:n]
        cc = np.correlate(hh, 1.0-hh, mode='full') * dx
        return v[-1] - cc
    def eq(v):
        return np.sum(v[:n]) - n*0.5
    def eq_jac(v):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type':'ineq','fun':ineq}, {'type':'eq','fun':eq,'jac':eq_jac}],
                   options={'maxiter':max_iter, 'ftol':1e-10})
    h_new = project_constrained(np.clip(res.x[:n], 0, 1), n*0.5)
    return h_new, compute_c5(h_new)

    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)

    target = n * 0.5

    best_h = h.copy()

        h_mom = h_cur + (alpha_t - 1.0) / alpha_t * (h_cur - h_prev)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_constrained(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def sparse_pnorm_grad(h, p, m=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = np.sum(corr ** p) ** (1.0 / p)
    if obj < 1e-15:
        return np.zeros(n), 0.0
    top_idx = np.argpartition(corr, -m)[-m:]
    scale = obj ** (1.0 - p)
    g = np.zeros(n)
    omh = 1.0 - h
    for idx in top_idx:
        k = idx - (n - 1)
        w = corr[idx] ** (p - 1) * scale
        if k >= 0:
            g[:n - k] += w * omh[k:]
            g[k:] -= w * h[:n - k]
        else:
            kn = -k
            g[kn:] += w * omh[:n - kn]
            g[:n - kn] -= w * h[kn:]
    return g, float(obj)


def full_pnorm_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = np.sum(corr ** p) ** (1.0 / p)
    if obj < 1e-15:
        return np.zeros(n), 0.0
    W = (corr ** (p - 1)) * obj ** (1.0 - p)
    omh = 1.0 - h
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):
        A[j] = np.dot(W[n - 1:2 * n - 1 - j], omh[j:])
        B[j] = np.dot(W[j:n + j][::-1], h)
    return A - B, float(obj)


def pgd_optimize(h, p, n_iter, lr_init, m=5, use_momentum=True):
    n = len(h)
    target = n * 0.5
    h_cur = h.copy()
    h_prev = h.copy()
    lr = lr_init
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    
    for t in range(n_iter):
        if use_momentum:
            beta = 0.9
            h_mom = h_cur + beta * (h_cur - h_prev)
            h_mom = project_constrained(h_mom, target)
            grad, _ = sparse_pnorm_grad(h_mom, p, m)
        else:
            grad, _ = sparse_pnorm_grad(h_cur, p, m)
        
        h_new = h_cur - lr * grad
        h_new = project_constrained(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur - 1e-12:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        elif c5_new > c5_cur + 0.01:
            lr *= 0.5
            if lr < 1e-8:
                break
            h_new = h_cur - lr * grad
            h_new = project_constrained(h_new, target)
            c5_new = compute_c5(h_new)
        
        h_prev = h_cur
        h_cur = h_new
    
    return best_h, best_c5


def slsqp_tvar(h0, max_iter=400):
    n = len(h0)
    dx = 2.0 / n
    c5_cur = compute_c5(h0)
    v0 = np.concatenate([h0, [c5_cur + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]

    def obj(v):
        return v[-1]

    def ineq(v):
        hh = v[:n]
        return v[-1] - np.correlate(hh, 1.0 - hh, mode='full') * dx

    def eq(v):
        return np.sum(v[:n]) - n * 0.5

    def eqj(v):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': ineq},
                                {'type': 'eq', 'fun': eq, 'jac': eqj}],
                   options={'maxiter': max_iter, 'ftol': 1e-11})
    h_new = project_constrained(np.clip(res.x[:n], 0, 1), n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h, n_new):
    x_old = np.linspace(0, 1, len(h), endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_constrained(h_new, n_new * 0.5)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_constrained(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def sparse_pnorm_grad(h, p, m=6):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = np.sum(corr ** p) ** (1.0 / p)
    if obj < 1e-15:
        return np.zeros(n), 0.0
    top_idx = np.argpartition(corr, -m)[-m:]
    scale = obj ** (1.0 - p)
    g = np.zeros(n)
    omh = 1.0 - h
    for idx in top_idx:
        k = idx - (n - 1)
        w = corr[idx] ** (p - 1) * scale
        if k >= 0:
            g[:n - k] += w * omh[k:]
            g[k:] -= w * h[:n - k]
        else:
            kn = -k
            g[kn:] += w * omh[:n - kn]
            g[:n - kn] -= w * h[kn:]
    return g, float(obj)


def pgd_optimize(h, p, n_iter, lr_init, m=6, momentum=0.85):
    n = len(h)
    target = n * 0.5
    h_cur = h.copy()
    v = np.zeros(n)
    lr = lr_init
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    stall = 0
    for t in range(n_iter):
        grad, _ = sparse_pnorm_grad(h_cur, p, m)
        v = momentum * v + grad
        h_new = h_cur - lr * v
        h_new = project_constrained(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-11:
            c5_cur = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            stall += 1
            if c5_new > c5_cur + 0.005:
                lr *= 0.5
                if lr < 1e-9:
                    break
                h_new = h_cur - lr * v
                h_new = project_constrained(h_new, target)
                c5_new = compute_c5(h_new)
            if stall > 100:
                break
        h_cur = h_new
    return best_h, best_c5


def slsqp_tvar(h0, max_iter=400):
    n = len(h0)
    dx = 2.0 / n
    c5_cur = compute_c5(h0)
    v0 = np.concatenate([h0, [c5_cur + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]

    def obj(f):
        return f[-1]

    def ineq(f):
        hh = f[:n]
        return f[-1] - np.correlate(hh, 1.0 - hh, mode='full') * dx

    def eq(f):
        return np.sum(f[:n]) - n * 0.5

    def eqj(f):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': ineq},
                                {'type': 'eq', 'fun': eq, 'jac': eqj}],
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = project_constrained(np.clip(res.x[:n], 0, 1), n * 0.5)
    return h_new, compute_c5(h_new)


def coord_descent(h, n_passes=3, delta=0.005):
    n = len(h)
    target = n * 0.5
    c5_cur = compute_c5(h)
    for _ in range(n_passes):
        for i in range(n):
            for d in [delta, -delta, delta * 0.5, -delta * 0.5]:
                h_tr = h.copy()
                h_tr[i] = np.clip(h[i] + d, 0, 1)
                h_tr = project_constrained(h_tr, target)
                c5_tr = compute_c5(h_tr)
                if c5_tr < c5_cur - 1e-10:
                    h = h_tr
                    c5_cur = c5_tr
                    break
    return h, c5_cur

    h_new = np.interp(x_new, x_old, h)
    return project_constrained(h_new, n_new * 0.5)
