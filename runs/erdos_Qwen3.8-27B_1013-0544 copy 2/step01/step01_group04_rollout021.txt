def weighted_subgrad(h, n, top_k=3, rel_tol=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find all "active" shifts (within rel_tol of max)
    active = np.where(corr >= max_val * (1 - rel_tol))[0]
    # Limit to top_k
    if len(active) > top_k:
        active = active[np.argsort(corr[active])[-top_k:]]
    
    grad = np.zeros(n)
    for k in active:
        s = k - np.arange(n)
        mask = (s >= 0) & (s < n)
        grad[mask] += 1.0 - 2.0 * h[s[mask]]
    grad /= len(active)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(h - alpha, 0.0, 1.0)
    diff = result.sum() - target
    if abs(diff) > 1e-10:
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(free):
            result[free] -= diff / np.count_nonzero(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def subgrad_active(h, n, top_k=3, rel_tol=0.02):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val * (1.0 - rel_tol))[0]
    if len(active) > top_k:
        idx = np.argsort(corr[active])[-top_k:]
        active = active[idx]
    grad = np.zeros(n)
    for k in active:
        s = k - np.arange(n)
        mask = (s >= 0) & (s < n)
        grad[mask] += 1.0 - 2.0 * h[s[mask]]
    grad /= max(1, len(active))
    return grad

def slsqp_refine(h, n, dx, max_iter=200, ftol=1e-14):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        return compute_c5(x, dx)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        c5_new = compute_c5(res.x, dx)
        if c5_new < compute_c5(h, dx):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
