        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            step = min(step, 1.0 - h[mask].min())
        else:
            step = max(step, -h[mask].min())
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, tau, penalty, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * dx * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    w = exp_x / Z
    obj = (x_max + np.log(Z)) / tau
    lags = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-14:
            continue
        lag = lags[m]
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wm * (1.0 - h[:n-lag])
                grad[:n-lag] -= wm * h[lag:]
        else:
            L = -lag
            if n - L > 0:
                grad[:n-L] += wm * (1.0 - h[L:])
                grad[L:] -= wm * h[:n-L]
    grad *= dx
    if penalty > 0:
        s = h.sum() - target
        obj += penalty * s * s
        grad += 2.0 * penalty * s
    return obj, grad

def run_lbfgs(h0, tau, penalty, target, maxiter=400, gtol=1e-10):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))
    def fun(x):
        return smooth_max_obj_grad(x, tau, penalty, target)[0]
    def jac(x):
        return smooth_max_obj_grad(x, tau, penalty, target)[1]
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

def interpolate_to_n(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, h)

def projected_grad_desc(h0, tau, target, max_iter=2000, lr0=0.05):
    n = len(h0)
    h = h0.copy()
    best_obj = smooth_max_obj_grad(h, tau, 0, target)[0]
    for it in range(max_iter):
        obj, grad = smooth_max_obj_grad(h, tau, 0, target)
        if obj < best_obj:
            best_obj = obj
        lr = lr0 / (1.0 + 0.005 * it)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        obj_new, _ = smooth_max_obj_grad(h_new, tau, 0, target)
        if obj_new < obj:
            h = h_new
        else:
            lr *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = fix_sum(h_init, target0)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Annealing at initial resolution
    for tau in [10, 50, 200, 800, 3000, 10000, 30000]:
        if rem() < 40:
            break
        h_new = run_lbfgs(best_h, tau, penalty=500.0, target=target0, maxiter=500)
        h_new = fix_sum(h_new, target0)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 2: Resolution upsampling
    for n_try in [160, 200, 240, 320]:
        if rem() < 80:
            break
        if n_try % 2 != 0:
            continue
        target = n_try / 2.0
        h_t = interpolate_to_n(best_h, n_try)
        h_t = fix_sum(h_t, target)
        for tau in [200, 800, 3000, 10000]:
            if rem() < 25:
                break
            h_new = run_lbfgs(h_t, tau, penalty=500.0, target=target, maxiter=300)
            h_new = fix_sum(h_new, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
            h_t = h_new

    # Phase 3: Downsample and re-optimize (sometimes coarser is better)
    for n_down in [120, 100, 80]:
        if rem() < 50:
            break
        if n_down % 2 != 0:
            continue
        target = n_down / 2.0
        h_d = interpolate_to_n(best_h, n_down)
        h_d = fix_sum(h_d, target)
        for tau in [500, 2000, 8000]:
            if rem() < 20:
                break
            h_new = run_lbfgs(h_d, tau, penalty=500.0, target=target, maxiter=300)
            h_new = fix_sum(h_new, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
            h_d = h_new

    # Phase 4: Projected gradient descent at best resolution
    if rem() > 60:
        hn = len(best_h)
        thn = hn / 2.0
        for tau in [500, 2000, 8000]:
            if rem() < 20:
                break
            h_pg = projected_grad_desc(best_h.copy(), tau, thn, max_iter=3000, lr0=0.03)
            h_pg = fix_sum(h_pg, thn)
            c5 = evaluate_c5(h_pg)
            if c5 < best_c5:
                best_h = h_pg
                best_c5 = c5

    # Phase 5: Random restarts
    hn = len(best_h)
    thn = hn / 2.0
    for trial in range(40):
        if rem() < 20:
            break
        scale = 0.005 + 0.12 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = np.clip(hp, 0.0, 1.0)
        hp = fix_sum(hp, thn)
        h_new = run_lbfgs(hp, 2000, penalty=500.0, target=thn, maxiter=200)
        h_new = fix_sum(h_new, thn)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 6: Structured initializations
    if rem() > 40:
        x = np.linspace(0, 1, hn)
        structures = [
            1.0 - x,
            0.5 + 0.4 * np.sin(2 * np.pi * x),
            0.5 + 0.4 * np.sin(4 * np.pi * x),
            np.where(x < 0.5, 1.0, 0.0),
            np.where((x < 0.3) | (x > 0.7), 1.0, 0.0),
            0.5 + 0.3 * np.cos(2 * np.pi * x),
        ]
        for s in structures:
            if rem() < 15:
                break
            s = fix_sum(s.copy(), thn)
            h_new = run_lbfgs(s, 2000, penalty=500.0, target=thn, maxiter=300)
            h_new = fix_sum(h_new, thn)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 7: Final polish with high tau
    if rem() > 15:
        hn_f = len(best_h)
        tf = hn_f / 2.0
        h_new = run_lbfgs(best_h, 50000, penalty=1000.0, target=tf, maxiter=300)
        h_new = fix_sum(h_new, tf)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Final constraint enforcement
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn_final
