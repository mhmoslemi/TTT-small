def smooth_c5(h, dx, beta=200):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    max_val = np.max(corr)
    smooth_max = max_val + np.log(np.sum(np.exp(corr - max_val))) / beta
    return smooth_max

def smooth_c5_grad(h, dx, beta=200):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    weights = np.exp(corr - max_val)
    weights /= np.sum(weights)
    # Gradient of log-sum-exp / beta = sum of weighted gradients
    # grad of corr[k] * dx with respect to h[j]
    n = len(h)
    grad = np.zeros(n)
    # For each k (lag), the gradient contribution is weights[k] * beta_inv * dx * dcorr[k]/dh[j]
    inv_beta = 1.0 / beta
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        # d corr[k] / d h[j] for this lag
        n_min = max(0, -lag)
        n_max = min(n-1, n-1-lag)
        # First term: (1-h[j-lag]) for j in [n_min+lag, n_max+lag]
        j_start = max(0, n_min + lag)
        j_end = min(n-1, n_max + lag)
        for j in range(j_start, j_end + 1):
            grad[j] += inv_beta * dx * weights[k] * (1 - h[j-lag])
        # Second term: -h[j+lag] for j in [n_min, n_max]
        for j in range(n_min, n_max + 1):
            grad[j] += inv_beta * dx * weights[k] * (-h[j+lag])
    return grad

            h_try = project_feasible(best_h + rng.randn(n) * scale, n)
        res = minimize(
            lambda x: compute_c5(x, dx),
            h_try, method='SLSQP', bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000, 'ftol': 1e-16}
        )
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
        attempt += 1

    # Phase 4: Final L-BFGS-B with very high beta (last 10%)
    h_final = bgb_optimize(best_h.copy(), dx, max_iter=500, beta=2000.0, penalty_M=1.0)
    c5_final = compute_c5(h_final, dx)
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
