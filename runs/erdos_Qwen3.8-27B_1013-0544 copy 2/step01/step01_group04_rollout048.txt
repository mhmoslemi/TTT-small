        pop[i] = project_feasible(rng.rand(n), n)
    objs = np.array([compute_c5(pop[i], dx) for i in range(pop_size)])
    best_i = np.argmin(objs)
    best_h = pop[best_i].copy()
    best_val = objs[best_i]
    for gen in range(max_gens):
        if time.time() - start_time > time_budget:
            break
        progress = gen / max(max_gens - 1, 1)
        F = 0.85 - 0.55 * progress
        Cr = 0.8 - 0.4 * progress
        for i in range(pop_size):
            if time.time() - start_time > time_budget:
                break
            others = np.delete(np.arange(pop_size), i)
            a, b, c = rng.choice(len(others), size=3, replace=False)
            a, b, c = others[a], others[b], others[c]
            mutant = pop[a].copy() + F * (pop[b] - pop[c])
            mutant = project_feasible(mutant, n)
            r = rng.randint(n)
            mask = rng.rand(n) < Cr
            mask[r] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_feasible(trial, n)
            trial_obj = compute_c5(trial, dx)
            if trial_obj < objs[i]:
                pop[i] = trial
                objs[i] = trial_obj
                if trial_obj < best_val:
                    best_val = trial_obj
                    best_h = trial.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: DE global search at n=90
    p1_end = budget_s * 0.25
    h_de, c5_de = de_run(best_h, n, 50, 300, rng, start_time, start_time + p1_end)
    if c5_de < best_c5:
        best_h = h_de
        best_c5 = c5_de

    # Phase 2: SLSQP polishing with diverse perturbations
    p2_end = start_time + budget_s * 0.60
    attempt2 = 0
    while time.time() < p2_end:
        if attempt2 == 0:
            h_try = best_h.copy()
        elif attempt2 % 5 == 1:
            idx = rng.randint(n)
            h_try = best_h.copy()
            h_try[idx] = rng.rand()
            h_try = project_feasible(h_try, n)
        elif attempt2 % 5 == 2:
            idxs = rng.choice(n, size=3, replace=False)
            h_try = best_h.copy()
            h_try[idxs] = rng.rand(3)
            h_try = project_feasible(h_try, n)
        else:
            scale = 0.08 * (1.0 - attempt2 / 200.0) + 0.01
            h_try = best_h + rng.randn(n) * scale
            h_try = project_feasible(h_try, n)
        h_new = slsqp_polish(h_try, max_iter=350)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
        attempt2 += 1

    # Phase 3: Upsample via interpolation and optimize
    if time.time() < start_time + budget_s * 0.85:
        n2 = 2 * n
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up
        p3_end = start_time + budget_s * 0.85
        h_de2, c5_de2 = de_run(best_h, n2, 35, 150, rng, start_time, p3_end)
        if c5_de2 < best_c5:
            best_h = h_de2
            best_c5 = c5_de2
        h_new2 = slsqp_polish(best_h, max_iter=300)
        c5_new2 = compute_c5(h_new2, len(h_new2) and 2.0 / len(h_new2))
        if c5_new2 < best_c5:
            best_h = h_new2
            best_c5 = c5_new2

    # Phase 4: Final polish
    if time.time() < start_time + budget_s - 5:
        n_f = len(best_h)
        h_final = slsqp_polish(best_h, max_iter=600)
        c5_final = compute_c5(h_final, 2.0 / n_f)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    return (best_h, best_c5, len(best_h))
