            corr = np.correlate(h, 1-h, mode='full')
            C = np.max(corr) * dx
            # Active set: correlations within eps*dx of max

            # f_k(h) = sum_i h[i]*(1-h[i+k]) for appropriate k
            # df_k/dh[j] = sum over terms where j appears
            grad_sum = np.zeros(n)
            for k_idx in active:
                k = k_idx - (n - 1)  # convert to actual lag
                grad = np.zeros(n)
                if k >= 0:
                    for i in range(n - k):
                        grad[i] += (1 - h[i+k])      # d/dh[i] of h[i]*(1-h[i+k])
                        grad[i+k] += -h[i]           # d/dh[i+k] of h[i]*(1-h[i+k])
                else:
                    kk = -k
                    for i in range(n - kk):
                        grad[i+kk] += (1 - h[i])     # d/dh[i+kk] of h[i+kk]*(1-h[i])
                        grad[i] += -h[i+kk]          # d/dh[i] of h[i+kk]*(1-h[i])
                grad_sum += grad
            grad_sum /= len(active)
            # Project gradient to satisfy sum constraint: subtract mean
            grad_projected = grad_sum - np.mean(grad_sum)
            # Line search
            best_step = 0.01
            best_val = C
            for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
                h_new = np.clip(h - grad_projected * step, 0, 1)
                # Fix sum
                h_new = fix_sum(h_new, target_sum)
                val = compute_c5(h_new, dx)
                if val < best_val:
                    best_val = val
                    best_step = step
                    h_new_best = h_new.copy()
            if best_step > 0:
                h = h_new_best
        return h

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode="full")

    def compute_correlations(h):
        return np.correlate(h, 1 - h, mode="full")

    def fix_sum(h, target_sum):
        h = h.copy()
        diff = target_sum - np.sum(h)
        h += diff / len(h)
        return np.clip(h, 0.0, 1.0)

        corr = compute_correlations(h)
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        return (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx

        """Compute approximate subgradient of max_k corr(k)"""
        n = len(h)
        corr = compute_correlations(h)
        max_c = np.max(corr)
        # Active set
        active = corr >= max_c - eps_rel * max_c
        n_active = np.sum(active)
        n_full = 2 * n - 1
        # For each active correlation index, compute gradient
        # corr[j] corresponds to lag k = j - (n-1)
        grad = np.zeros(n)
        for j in np.where(active)[0]:
            k = j - (n - 1)  # actual lag
            g = np.zeros(n)
            if k >= 0:
                # corr = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
                # d/dh[i] = (1-h[i+k]) for i < n-k
                # d/dh[i+k] = -h[i] for i+k < n, i.e., i < n-k
                valid = np.arange(n - k)
                g[valid] += (1 - h[valid + k])
                g[valid + k] += (-h[valid])
            else:
                kk = -k
                # corr(-kk) = sum_{i=0}^{n-1-kk} h[i+kk] * (1-h[i])
                # d/dh[i+kk] = (1-h[i]) for i+kk < n
                # d/dh[i] = -h[i+kk] for i < n-kk
                valid = np.arange(n - kk)
                g[valid + kk] += (1 - h[valid])
                g[valid] += (-h[valid + kk])
            grad += g
        grad /= n_active
        return grad, max_c

        h_new = h - step * grad

        # This is not the exact Euclidean projection onto the polytope,
        # but it's a good approximation
        h_new = h_new - (np.sum(h_new) - target_sum) / len(h_new)
        h_new = np.clip(h_new, 0.0, 1.0)
        # Re-fix sum after clipping
        h_new = fix_sum(h_new, target_sum)
        return h_new

    def active_descent(h, dx, target_sum, n_iter=100, eps_rel=0.02):
        n = len(h)
        best_c5 = compute_c5(h, dx)
        h_best = h.copy()
        for it in range(n_iter):
            grad, max_c = gradient_of_max_corr(h, eps_rel)
            # Project gradient to be orthogonal to all-ones vector (sum constraint)
            grad_proj = grad - np.mean(grad)
            # Line search: try decreasing steps
            step = 0.1
            improved = False
            for _ in range(10):
                h_try = projected_gradient_step(h, grad_proj, target_sum, step)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    h_best = h.copy()

            if not improved:

        return h_best, best_c5

    One more idea: **pairwise swap optimization with exact evaluation**.
    
    For a pair (i, j) with current values h[i] = a, h[j] = b:

    Actually, I realize I need to be more careful about the projected gradient. The constraint is sum(h) = n/2, so the feasible set is the intersection of the hyperplane {sum = n/2} with the box [0,1]^n. The projection onto this set is not trivial.

    A simple approach: after taking a gradient step, clip to [0,1], then adjust the sum by adding a constant to all entries (and re-clip). This isn't the exact projection but works well in practice.

    A better approach: water-filling. After clipping, if the sum is too high, reduce the largest values first; if too low, increase the smallest values first.

    Let me implement a proper projection:

    ```python
    def project_to_box_sum(h, target_sum):
        """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}"""
        n = len(h)
        x = h.copy()
        # Iterate: clip, adjust, clip
        for _ in range(20):
            x = np.clip(x, 0, 1)
            diff = target_sum - np.sum(x)
            if abs(diff) < 1e-10:
                break
            # Distribute diff among free variables
            free = (x > 0) & (x < 1)
            if not np.any(free):
                break
            x[free] += diff / np.sum(free)
        x = np.clip(x, 0, 1)
        return x
    ```
