def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(corr.max())

def project_to_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h):
    """Convert to binary with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # Sort by value descending, take top n/2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def interp_h(h_old, n_new):

    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

    c5_cur = compute_c5(h0)

        return v[-1]

        return t - corr

        j[:n] = 1.0
        return j

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

    mask = h < 1.0 if diff > 0 else h > 0.0
    cnt = np.sum(mask)

    n = len(h0)
    c5_cur = compute_c5(h0)
    
    def t_obj(v):
        return v[-1]
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        return t - np.correlate(h, 1.0 - h, mode='full').max() * dx
    def t_eq(v):
        return np.sum(v[:n]) - n * 0.5
    def t_eq_jac(v):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})

    h_new = project_to_sum(h_new, n * 0.5)

            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 6: Smooth objective polish
    if time_left() > 40:
        for temp in [0.005, 0.001]:
            if time_left() < 30:
                break
            h_new, c5_new = smooth_optimize(h_best, temp, max_iter=150)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 7: Final SA + hill climb
    if time_left() > 60:
        h_bin2 = to_binary(h_best)
        h_sa, c5_sa = binary_sa(h_bin2, n_swaps=35000, t_start=0.002, t_end=0.00005, rng=rng)
        if c5_sa < c5_best:
            h_best = h_sa.copy()
            c5_best = c5_sa
        h_c, c5_c = binary_hill_climb(h_best, max_iters=15000, n_try=150, rng=rng)
        if c5_c < c5_best:
            h_best = h_c.copy()
            c5_best = c5_c

    # Final SLSQP squeeze
    if time_left() > 25:
        h_new, c5_new = slsqp_optimize(h_best, max_iter=150)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_to_sum(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
