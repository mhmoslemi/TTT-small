def project_feasible(h, n, tol=1e-12):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < tol:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-10:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_sweep(h, n, dx, grid, order=None):
    """One sweep of coordinate descent. Returns updated h and c5."""
    if order is None:
        order = np.arange(n)
    for i in order:
        # Save current value
        old_val = h[i]
        
        # Compute output at t=0, t=0.5, t=1
        h[i] = 0.0
        one_minus_h = 1.0 - h
        out0 = np.correlate(h, one_minus_h, mode='full')
        
        h[i] = 0.5
        one_minus_h = 1.0 - h
        out_half = np.correlate(h, one_minus_h, mode='full')
        
        h[i] = 1.0
        one_minus_h = 1.0 - h
        out1 = np.correlate(h, one_minus_h, mode='full')
        
        # Restore
        h[i] = old_val
        one_minus_h = 1.0 - h
        
        # Coefficients: f_m(t) = q[m]*t^2 + l[m]*t + c[m]
        c_coeff = out0
        l_coeff = out1 - out0
        q_coeff = 2.0 * (out_half - 0.5 * (out0 + out1))
        
        # Evaluate on grid
        t2 = grid * grid
        vals = q_coeff[None, :] * t2[:, None] + l_coeff[None, :] * grid[:, None] + c_coeff[None, :]
        max_vals = np.max(vals, axis=1)
        best_idx = np.argmin(max_vals)
        h[i] = grid[best_idx]
    
    # Enforce sum constraint
    h = project_feasible(h, n)
    return h, compute_c5(h, dx)
