def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    f = corr * dx
    fmax = np.max(f)
    f_shifted = f - fmax
    p = np.exp(f_shifted / mu)
    p /= np.sum(p)
    
    n_half = n - 1
    grad = np.zeros(n)
    for idx in range(len(p)):
        if p[idx] < 1e-8:

        else:
            kp = -k
            grad[:n-kp] += p[idx] * (1.0 - h[kp:])

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, dx, mu):
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    f = corr * dx
    fmax = np.max(f)
    f_shifted = f - fmax
    p = np.exp(f_shifted / mu)
    p /= np.sum(p)
    
    n_half = n - 1
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-10:
            continue
        k = idx - n_half
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    grad *= dx
    return grad, fmax

            best_c5 = c5_new
            no_improve = 0

            if no_improve > 100:
                alpha = max(0.001, alpha * 0.9)

        t = t_new
    
    return best_h, best_c5

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=300):

    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(
        lambda x: compute_c5(x, dx),
        h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, n)

        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, n, dx, mu):
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    f = corr * dx
    fmax = np.max(f)
    fs = f - fmax
    p = np.exp(fs / mu)
    ps = np.sum(p)
    if ps < 1e-300:
        p = np.zeros_like(f)
        p[np.argmax(f)] = 1.0
    else:
        p /= ps
    n_half = n - 1
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-10:
            continue
        k = idx - n_half
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    grad *= dx
    return grad, fmax

def fista_loop(h, n, dx, time_limit, rng, mu_init=0.03, mu_min=0.002):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    mu = mu_init
    t = 1.0
    h_prev = h.copy()
    h_curr = h.copy()
    y = h.copy()
    alpha = mu / (4.0 * dx)
    stall = 0

    while time.time() - start < time_limit:
        mu = max(mu_min, mu * 0.999)
        grad_y, _ = smooth_max_grad(y, n, dx, mu)
        h_new = project_feasible(y - alpha * grad_y, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
            alpha = min(alpha * 1.02, mu / (2.0 * dx))
        else:
            stall += 1
            if stall > 80:
                alpha = max(0.005, alpha * 0.85)
                mu = max(mu_min, mu * 0.95)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mf = (t - 1.0) / (t_new + 1.0)
        h_prev = h_curr
        h_curr = h_new
        y = project_feasible(h_curr + mf * (h_curr - h_prev), n)
        t = t_new

    return best_h, best_c5

def slsqp_polish(h, n, dx, target, iters=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    res = minimize(lambda x: compute_c5(x, dx), h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': iters, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def coord_descent(h, n, dx, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    target = n / 2.0
    while time.time() - start < time_limit:
        improved = False
        for j in rng.permutation(n):
            if time.time() - start > time_limit:
                break
            old = best_h[j]
            for frac in (0.0, 0.25, 0.5, 0.75, 1.0, old - 0.15, old - 0.05, old + 0.05, old + 0.15, 0.1, 0.9):
                cv = np.clip(frac, 0.0, 1.0)
                if abs(cv - old) < 1e-6:
                    continue
                th = best_h.copy()
                th[j] = cv
                th = project_feasible(th, n)
                c = compute_c5(th, dx)
                if c < best_c5 - 1e-14:
                    best_h = th
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: FISTA at original resolution
    t1 = budget_s * 0.35
    h_f, c5_f = fista_loop(best_h.copy(), n, dx, t1, rng)
    if c5_f < best_c5:
        best_h, best_c5 = h_f, c5_f

    # Phase 2: SLSQP polish at n
    t2 = budget_s * 0.55
    while elapsed() < t2:
        hp = slsqp_polish(best_h.copy(), n, dx, target, 300)
        cp = compute_c5(hp, dx)
        if cp < best_c5 - 1e-14:
            best_h, best_c5 = hp, cp
        else:
            break

    # Phase 3: Coordinate descent at n
    t3 = budget_s * 0.65
    if elapsed() < t3:
        hc, cc = coord_descent(best_h.copy(), n, dx, t3 - elapsed(), rng)
        if cc < best_c5:
            best_h, best_c5 = hc, cc

    # Phase 4: Upsample to 2n
    n2 = 2 * n
    dx2 = 2.0 / n2
    target2 = n2 / 2.0
    h_up = np.repeat(best_h, 2).copy()
    h_up = project_feasible(h_up, n2)
    t4 = budget_s * 0.85
    if elapsed() < t4:
        h_f2, c5_f2 = fista_loop(h_up.copy(), n2, dx2, min(t4 - elapsed(), 200), rng, mu_init=0.02, mu_min=0.002)
        c5_f2 = compute_c5(h_f2, dx2)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2
        while elapsed() < t4:
            hp2 = slsqp_polish(best_h.copy(), n2, dx2, target2, 250)
            cp2 = compute_c5(hp2, dx2)
            if cp2 < best_c5 - 1e-14:
                best_h, best_c5 = hp2, cp2
            else:
                break

    # Phase 5: Final polish
    nf = len(best_h)
    dxf = 2.0 / nf
    tf = nf / 2.0
    t5 = budget_s - 25
    if elapsed() < t5:
        h_f3, c5_f3 = fista_loop(best_h.copy(), nf, dxf, min(80, t5 - elapsed()), rng, mu_init=0.01, mu_min=0.001)
        if c5_f3 < best_c5:
            best_h, best_c5 = h_f3, c5_f3
        while elapsed() < t5:
            hp3 = slsqp_polish(best_h.copy(), nf, dxf, tf, 300)
            cp3 = compute_c5(hp3, dxf)
            if cp3 < best_c5 - 1e-14:
                best_h, best_c5 = hp3, cp3
            else:
                break

    return (best_h, best_c5, len(best_h))
