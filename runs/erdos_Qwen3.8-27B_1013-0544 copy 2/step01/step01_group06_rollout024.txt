    if rem > 80:
        for nn in [max(100, best_n - 50), min(300, best_n + 50)]:
            if rem < 60 or nn == best_n:
                break
            h_interp = project_box_sum(interp_to(best_h, nn), nn / 2.0)
            dl = time.time() + min(rem * 0.3, 120)
            try_optimize(h_interp, nn, dl, alpha0=60)
            rem = budget_s - (time.time() - t0) - 20

    # Phase 3: Final high-alpha polish
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        dl = t0 + budget_s - 5
        try_optimize(best_h, best_n, dl, alpha0=250)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
