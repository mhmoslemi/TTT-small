    _set_mode('softmax')
    for tau in [10000, 50000, 200000]:
        if rem() < 10:
            break
        x, c5 = _slsqp(best_h, n, _softmax_obj, tau, 500, 1e-16)
        if c5 is not None and c5 < best_c5:
            best_h, best_c5 = x, c5

    # Final constraint enforcement
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
