def smooth_max_grad(corr_values, temperature=0.01):
    """Compute gradient of smooth max (log-sum-exp) of corr_values."""
    shifted = corr_values - np.max(corr_values)
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    return weights

def active_set_projected_gradient(h, n, dx, target_sum, max_iter, time_limit, start_time, seed):
    """Active-set projected gradient with adaptive step size."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n_half = n - 1
    
    step_size = 0.005
    mu = 0.6
    momentum = np.zeros(n)
    stag = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        
        # Adaptive active set: use lags within tolerance of max
        tol = max(0.05, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 3:

        elif len(active) > 20:

        c5_new = compute_c5(h_new, dx)

            stag = 0
            # Occasionally increase step size if improving a lot

                h = project_feasible(h, target_sum)
                step_size = 0.004
                momentum = np.zeros(n)
                stag = 0
        h = h_new

def binary_local_search(h, n, dx, max_swaps=2000, time_limit=10, start_time=0):
    """Local search on binary-valued h (swap 0s and 1s)."""

        remove = rng_choice(ones_idx, n_ones - target)
        h_bin[remove] = 0

        zeros_idx = np.where(h_bin == 0)[0]
        add = rng_choice(zeros_idx, target - n_ones)
        h_bin[add] = 1

        # For same array, all lags from -(n-1) to (n-1) are valid
        # We want max of h(i)*(1-h(i+k)) which is n/2 - autocorr_k for binary
        # So minimize max_k (n/2 - autocorr_k) = n/2 - min_k autocorr_k
        # Maximize min_k autocorr_k
        return float(np.min(corr))
    
    best_min_ac = min_autocorr(h_bin)
    no_improve = 0
    
    for _ in range(max_swaps):
        if time.time() - start_time > time_limit:
            break
        i, j = rng_choice(n, 2, replace=False)
        if h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_min_ac = min_autocorr(h_bin)
        if new_min_ac > best_min_ac:
            best_min_ac = new_min_ac
            no_improve = 0
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            no_improve += 1
            if no_improve > 200:
                break
    
    return h_bin, best_min_ac

def rng_choice(n, k, replace=True):
    return np.random.choice(n, size=k, replace=replace)

def make_binary_from_pattern(n, pattern_type='sine', phase=0):
    """Generate binary sequences with good autocorrelation properties."""
    np.random.seed(0)
    if pattern_type == 'sine':
        x = np.linspace(0, 2*np.pi, n)
        threshold = 0.5
        h = (np.sin(x + phase) > threshold - 0.5).astype(float)
    elif pattern_type == 'quad':
        x = np.arange(n)
        # Quadratic residue-like pattern
        h = ((x * x + x) % n < n // 2).astype(float)
    elif pattern_type == 'alternating':
        h = (np.arange(n) % 2).astype(float)
    else:  # random
        h = (np.random.rand(n) < 0.5).astype(float)
    
    # Adjust to have exactly n/2 ones
    target = n // 2
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        h[np.random.choice(idx)] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
