import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(corr, t, dx):
    m = np.max(corr)
    shifted = t * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / t) * dx

def smooth_max_gradient(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = t * (corr - m)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    n = len(h)
    one_minus_h = 1.0 - h
    w_rev = weights[::-1]
    term1 = np.convolve(one_minus_h, w_rev, mode='full')
    term1 = term1[n-1:n-1+n]
    term2 = np.convolve(weights, h, mode='full')
    term2 = term2[n-1:n-1+n]
    grad = (term1 - term2) * dx
    return grad

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def lbfgs_optimize(h, dx, target_sum, t=50, maxiter=500):

        g = smooth_max_gradient(x, t, dx)
        g += 2e6 * (np.sum(x) - target_sum)
        return g

                       bounds=[(0.0, 1.0)] * n,

                                'gtol': 1e-12, 'maxls': 80})

    indices = rng.choice(n_freq - 1, n_pert, replace=False) + 1

    dH[indices] = scale * (rng.randn(n_pert) + 1j * rng.randn(n_pert))
    return np.fft.irfft(H + dH, n=n)

                best_h = h_p.copy()
                best_c5 = c5_p
    return best_h, best_c5

def group_refine(h, dx, target_sum, rng, n_rounds=40, group_size=6, t=80):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):

            hf[idx] = np.clip(x, 0.0, 1.0)
