_g_dx = 2.0 / 70

def project_feasible(x, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_corr_at_lag(h, L, n):
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh - h
    elif L > 0:
        grad[:n - L] += omh[L:]
        grad[L:] -= h[:n - L]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def smooth_obj(x):
    n = _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _g_beta
    shifted = corr - m
    exp_b = np.exp(beta * shifted)
    return (m + np.log(np.sum(exp_b)) / beta) * _g_dx
