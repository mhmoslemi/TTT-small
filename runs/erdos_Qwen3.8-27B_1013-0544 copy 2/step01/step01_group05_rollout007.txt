        return (start_time + budget_s - 5.0) - time.time()

    best_h = project_feasible(h0, target_sum)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: LSE annealing (main workhorse)
    h1, c5_1 = _run_lse_phase(best_h, target_sum, 3.0, 200.0, 5000, 0.8, 0.06, tleft)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Fine LSE with high lambda and momentum
    h2, c5_2 = _run_lse_phase(best_h, target_sum, 200.0, 800.0, 3000, 0.7, 0.01, tleft)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: SLSQP with analytic gradient
    if tleft() > 40:
        try:
            res = minimize(_obj, best_h, jac=_jac, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': _eq}],
                           options={'maxiter': 2000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 4: Basin hopping with local LSE descent
    if tleft() > 30:
        h = best_h.copy()
        for it in range(500):
            if tleft() < 5.0:
                break
            npert = rng.randint(3, 13)
            idx_p = rng.choice(n, size=npert, replace=False)
            h[idx_p] += rng.uniform(-0.1, 0.1, npert)
            h = project_feasible(h, target_sum)
            c5_t = compute_c5(h, dx)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
            # Local descent from perturbation
            step = 0.008
            for _ in range(200):
                if tleft() < 5.0:
                    break
                corr = np.correlate(h, 1.0 - h, mode='full')
                grad = lse_grad(corr, 600.0, h, n)
                h_new = project_feasible(h - step * grad, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < compute_c5(h, dx) - 1e-12:
                    h = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    step = min(step * 1.03, 0.02)
                else:
                    step *= 0.7
                    if step < 1e-9:
                        break

    # Phase 5: Final SLSQP polish
    if tleft() > 20:
        try:
            res = minimize(_obj, best_h, jac=_jac, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': _eq}],
                           options={'maxiter': 1500, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 6: Ultra-fine gradient steps
    if tleft() > 10:
        h = best_h.copy()
        step = 0.001
        for _ in range(5000):
            if tleft() < 3.0:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = lse_grad(corr, 1000.0, h, n)
            h_new = project_feasible(h - step * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                step *= 0.85
                if step < 1e-11:
                    break

    return (best_h, best_c5, n)
