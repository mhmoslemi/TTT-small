def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def lse_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50.0, 50.0)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-16:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            sp = -s
            grad[sp:] += w * (1.0 - h[:n - sp])
            grad[:n - sp] -= w * h[sp:]
    return grad

    h = v[:n]
    t = v[-1]
    dx = 2.0 / n

    return t - corr

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

    cons = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(lse_obj, h0.copy(), args=(temp,),
                   method='SLSQP', jac=lambda x, t: lse_grad(x, t),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})

    return h_new, compute_c5(h_new)

    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)

        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            if time_left() > 40:
                h_new2, c5_new2 = optimize_t(h_best, max_iter=300)
                if c5_new2 < c5_best:
                    h_best = h_new2
                    c5_best = c5_new2

    # Phase 4: Upscale to n=360
    if time_left() > 300:
        n_new = 360
        h_360 = interp_h(h_best, n_new)
        h_cur = h_360
        c5_cur = compute_c5(h_cur)
        if c5_cur < c5_best:
            h_best = h_cur.copy()
            c5_best = c5_cur

        # LSE at n=360 with decreasing temp
        for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
            if time_left() < 40:
                break
            h_new, c5_new = optimize_lse(h_cur, temp, max_iter=200)
            h_cur = h_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

        # Projected gradient with momentum at n=360
        if time_left() > 80:
            h_pg, c5_pg = proj_grad_steps(h_cur, 0.002, n_steps=3000, lr=0.005, momentum=0.9)
            if c5_pg < c5_best:
                h_best = h_pg
                c5_best = c5_pg
                h_cur = h_pg

        # More LSE after projected gradient
        if time_left() > 60:
            for temp in [0.005, 0.002, 0.001]:
                if time_left() < 30:
                    break
                h_new, c5_new = optimize_lse(h_cur, temp, max_iter=200)
                h_cur = h_new
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new

        # Perturbations at n=360
        for i in range(4):
            if time_left() < 40:
                break
            scale = 0.006 + 0.004 * i
            h_p = h_best + rng.randn(n_new) * scale
            h_p = project_h_bisect(h_p, n_new * 0.5)
            h_new, c5_new = optimize_lse(h_p, 0.003, max_iter=150)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new

    # Phase 5: Try n=540 if time permits
    if time_left() > 200:
        n_540 = 540
        h_540 = interp_h(h_best, n_540)
        h_cur = h_540
        c5_cur = compute_c5(h_cur)
        if c5_cur < c5_best:
            h_best = h_cur.copy()
            c5_best = c5_cur
        for temp in [0.01, 0.005, 0.002]:
            if time_left() < 50:
                break
            h_new, c5_new = optimize_lse(h_cur, temp, max_iter=150)
            h_cur = h_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Final: return best at its resolution
    h_best = project_h_bisect(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    if c5_final > 0.5:
        # Fallback to parent
        h_best = np.array(initial_h_values, dtype=float)
        h_best = project_h_bisect(h_best, len(h_best) * 0.5)
        c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
