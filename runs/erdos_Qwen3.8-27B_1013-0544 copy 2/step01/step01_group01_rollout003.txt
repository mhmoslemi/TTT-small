import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)


def compute_all_f(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def gradient_smooth_max(h, f_values, dx, n, beta):
    shifted = f_values - f_values.max()
    weights = np.exp(beta * shifted)
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights /= total

    grad = np.zeros(n)
    for idx in range(len(f_values)):
        if weights[idx] < 1e-8:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += weights[idx] * dx * (1 - 2 * h)
        elif k > 0:
            g = np.zeros(n)
            g[:n - k] += (1 - h[k:])
            g[k:] -= h[:n - k]
            grad += weights[idx] * dx * g
        else:
            kk = -k
            g = np.zeros(n)
            g[kk:] += (1 - h[:n - kk])
            g[:n - kk] -= h[kk:]
            grad += weights[idx] * dx * g

    return grad / beta * dx * beta  # the 1/beta from smooth_max derivative


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    n = n_points

    best_h = project_box_sum(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Gradual beta increase with projected gradient
    h = best_h.copy()
    betas = [5, 10, 20, 40, 80, 160, 320, 640]
    steps_per_beta = [200, 200, 300, 300, 400, 400, 500, 600]

    for bi, beta in enumerate(betas):
        if time_left() < 80:
            break
        max_iter = steps_per_beta[bi]
        step = 0.02
        stall = 0
        cur_c5 = compute_c5(h)

        for it in range(max_iter):
            if time_left() < 70:
                break
            f_values = compute_all_f(h, dx)
            cur_max = f_values.max()

            if cur_max < best_c5 - 1e-13:
                best_c5 = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

            if stall > 80:
                h = project_box_sum(h + np.random.randn(n) * 0.04, target_sum)
                stall = 0
                step = 0.03
                continue

            grad = gradient_smooth_max(h, f_values, dx, n, beta)

            # Normalize gradient for better step sizing
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad_dir = grad / gnorm

            # Backtracking line search
            accepted = False
            s = step
            for _ in range(12):
                h_new = project_box_sum(h - s * gnorm * grad_dir, target_sum)
                f_new = compute_all_f(h_new, dx)
                if f_new.max() < cur_max - 1e-14:
                    h = h_new
                    accepted = True
                    step = min(s * 1.1, 0.3)
                    break
                s *= 0.5

            if not accepted:
                stall += 5
                if stall > 150:
                    h = project_box_sum(h + np.random.randn(n) * 0.06, target_sum)
                    step = 0.05
                    stall = 0

    # Phase 2: SLSQP polish with increasing beta
    if time_left() > 30:
        bounds = Bounds(np.zeros(n), np.ones(n))

        for beta in [50, 100, 200, 400]:
            if time_left() < 15:

            def obj_func(hh, _beta=beta):
                corr = np.correlate(hh, 1.0 - hh, mode="full")
                return smooth_max_val(corr, _beta) * dx

            def obj_grad(hh, _beta=beta):
                corr = np.correlate(hh, 1.0 - hh, mode="full")
                f = corr * _beta
                fm = f.max()
                w = np.exp(f - fm)
                w /= w.sum()
                # d/dh of smooth_max * dx
                # = dx * (1/beta) * sum w_k * beta * df_k/dh = dx * sum w_k * df_k/dh
                g = np.zeros(n)
                for idx in range(len(w)):
                    if w[idx] < 1e-10:
                        continue
                    k = idx - (n - 1)
                    if k == 0:
                        g += w[idx] * dx * (1 - 2 * hh)
                    elif k > 0:
                        g[:n - k] += w[idx] * dx * (1 - hh[k:])
                        g[k:] -= w[idx] * dx * hh[:n - k]
                    else:
                        kk = -k
                        g[kk:] += w[idx] * dx * (1 - hh[:n - kk])
                        g[:n - kk] -= w[idx] * dx * hh[kk:]
                return g

            constraints = [{'type': 'eq',
                            'fun': lambda x: np.sum(x) - target_sum,
                            'jac': lambda x: np.ones(n)}]

            try:
                res = minimize(obj_func, best_h, jac=obj_grad, method='SLSQP',

                pass

    if time_left() > 10:
        h = best_h.copy()
        for sweep in range(500):
            if time_left() < 5:
                break
            improved = False
            for j in range(n):
                best_obj = compute_c5(h)
                best_val = h[j]
                offsets = [-0.3, -0.15, -0.07, -0.03, 0.03, 0.07, 0.15, 0.3]
                for d in offsets:
                    trial = h[j] + d
                    if trial < 0 or trial > 1:
                        continue
                    h[j] = trial
                    obj = compute_c5(h)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_val = trial
                h[j] = best_val
                if best_val != h[j] - 0:  # check if changed

                if abs(best_val - (h[j] - (best_val - best_val))) > 0:
                    pass
            # Simpler: just track if h changed
            h2 = project_box_sum(h, target_sum)
            if compute_c5(h2) < best_c5 - 1e-14:
                # Re-project properly
                break
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
            else:
                # Check convergence
                if sweep > 50:
                    # Gradually reduce step
                    break

    # Final verification
    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
