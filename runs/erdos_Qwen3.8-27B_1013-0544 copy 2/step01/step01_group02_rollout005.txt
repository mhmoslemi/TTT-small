import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 700:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_con(h, n):
    return h.sum() - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def coordinate_descent(h, n, max_sweeps, n_cand, time_limit, t0):
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time_limit - (time.time() - t0) < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = np.unique(np.concatenate([
                np.linspace(0.0, 1.0, n_cand + 1),
                np.clip([old + d for d in (-0.05, -0.02, 0.02, 0.05)], 0, 1),
                [old]
            ]))
            best_t = old
            best_c = best_c5
            best_hp = None
            for t in cands:
                if abs(t - old) < 1e-12:
                    continue
                h[i] = t
                hp = project_feasible(h, n)
                c = compute_c5(hp, n)
                h[i] = old
                if c < best_c - 1e-14:
                    best_c = c
                    best_t = t
                    best_hp = hp.copy()
            if best_hp is not None:
                h = best_hp
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h, n, t_start, t_end, n_iter, seed):
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.08, 0.08)
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        c_try = compute_c5(h, n)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h.copy()
        delta = c_try - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = c_try
        else:
            h[i] = hi - d
            h[j] = hj + d
    return best_h, best_c5

def slsqp_refine(h, n, alpha, maxiter, t0, time_limit):
    if time_limit - (time.time() - t0) < 15:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        hp = project_feasible(res.x, n)
        c = compute_c5(hp, n)
        if c < compute_c5(h, n):
            return hp, c
    except Exception:
        pass
    return h, compute_c5(h, n)

def randomized_local_search(h, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    no_improve = 0
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        d = rng.choice([-0.002, -0.005, -0.01, 0.002, 0.005, 0.01,
                        -0.02, 0.02, -0.03, 0.03])
        hi = best_h[i] + d
        hj = best_h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h_try = best_h.copy()
            h_try[i] = hi
            h_try[j] = hj
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 500:
                break
    return best_h, best_c5
