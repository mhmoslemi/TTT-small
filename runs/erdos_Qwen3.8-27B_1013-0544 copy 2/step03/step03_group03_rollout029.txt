A[j] = Σ_k w[k] · 1(0 ≤ j+lag_k < n) = Σ_k w[k] · 1(0 ≤ j + k - (n-1) < n)
= Σ_k w[k] · 1(n-1-j ≤ k ≤ 2n-2-j)

And B[j] = Σ_k w[k] · h[j+lag_k] · 1(0 ≤ j+lag_k < n)
= Σ_k w[k] · h[j + k - (n-1)] · 1(0 ≤ j + k - (n-1) < n)

Similarly for the second term:
C[j] = Σ_k w[k] · h[j-lag_k] · 1(0 ≤ j-lag_k < n)
= Σ_k w[k] · h[j - k + (n-1)] · 1(0 ≤ j - k + (n-1) < n)

Hmm, this is getting complicated with the boundaries. For large n and when the softmax weights are concentrated on a few lags, it's easier to just loop over those lags.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:lag+m])
            grad[lag:] -= h[:m]
    else:
        la = -lag
        m = n - la
        if m > 0:
            grad[la:] += (1.0 - h[:m])
            grad[:m] -= h[la:]
    return grad

def softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(shifted)
    s = w.sum()
    if s > 1e-300:
        w /= s
    return w

def smooth_objective(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_gradient(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = softmax_weights(corr, beta)
    thresh = 1e-8
    sig_idx = np.where(w > thresh)[0]
    grad = np.zeros(n)
    for idx in sig_idx:
        lag = idx - (n - 1)
        grad += w[idx] * lag_gradient(h, lag, n)
    return grad * dx

def sum_con_fun(h, target_sum):
    return h.sum() - target_sum

def sum_con_jac(h, target_sum):
    return np.ones(len(h))

def tl_time(t0, budget_s):
    return budget_s - (time.time() - t0)

def phase_nesterov(h0, target_sum, t0, budget_s, min_time, max_it, lr_init, beta_start, beta_end, momentum):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    best_local = compute_c5(h)
    lr = lr_init
    
    for it in range(max_it):
        if tl_time(t0, budget_s) < min_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        frac = min(it / max(max_it, 1), 1.0)
        beta_t = beta_start * (beta_end / beta_start) ** frac
        w = softmax_weights(corr, beta_t)
        thresh = 1e-10
        sig = np.where(w > thresh)[0]
        grad = np.zeros(n)
        for idx in sig:
            lag = idx - (n - 1)
            grad += w[idx] * lag_gradient(h, lag, n)
        grad *= dx
        
        lr_t = lr_init / (1.0 + 0.001 * it)
        v = momentum * v + (-lr_t * grad)
        h_new = project_feasible(h + v, target_sum)
        c_new = compute_c5(h_new)
        
        if c_new < best_local + 0.001:
            h = h_new
            if c_new < best_local:
                best_local = c_new
        else:
            v *= 0.4
            h = project_feasible(h + v, target_sum)
        
        if it % 5000 == 4999:
            c = compute_c5(h)
            if c < best_local:
                best_local = c

def phase_slsqp(h0, target_sum, t0, budget_s, min_time, beta_list, trials_per_beta):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target_sum, None)}]

        for trial in range(trials_per_beta):
            if tl_time(t0, budget_s) < min_time:
                break
            x0 = best_h.copy()
            if trial > 0:
                scale = 0.015 / trial

                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-16})

                if c5_val < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_val
            except Exception:
                pass
    
    return best_h, best_c5
