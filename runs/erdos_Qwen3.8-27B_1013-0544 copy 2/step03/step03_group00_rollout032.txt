import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_gtarget = 0.0

def _cfunc(x):
    return x.sum() - _gtarget

def _cjac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            h = np.clip(h + d / len(h), 0.0, 1.0)
        else:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_objg(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(len(corr))
    c = corr[idx]
    m = c.max()
    e = np.exp(beta * (c - m))
    sw = e.sum()
    w = e / sw
    val = (m + np.log(sw) / beta) * dx
    g = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = idx[i] - n + 1
        if s >= 0:
            a = np.arange(0, n - s)
            g[a] += wi * (1.0 - h[a + s])
            b = np.arange(s, n)
            g[b] -= wi * h[b - s]
        else:
            t = -s
            a = np.arange(t, n)
            g[a] += wi * (1.0 - h[a - t])
            b = np.arange(0, n - t)
            g[b] -= wi * h[b + t]
    g *= dx
    return val, g

class SMFunc:
    def __init__(self, beta, top_k):
        self.beta = beta
        self.top_k = top_k
    def fun(self, x):
        return sm_objg(x, self.beta, self.top_k)[0]
    def jac(self, x):
        return sm_objg(x, self.beta, self.top_k)[1]

def do_slsqp(h, beta, maxiter, top_k=None):
    global _gtarget
    n = len(h)
    _gtarget = n / 2.0
    target = _gtarget
    f = SMFunc(beta, top_k)
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(f.fun, h, jac=f.jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cfunc, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def local_search(h, max_sweeps=40):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_h = h.copy()
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-8:
                    continue
                h[i] = val
                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < c5 - 1e-14:
                    c5 = c
                    improved = True
                else:
                    h = old_h.copy()
        if not improved:
            break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def basin_hop(h_best, best_c5, n_hops, beta_list, maxiter,
              noise_base, noise_decay, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 40:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 25:
                break
            h_local = do_slsqp(h_local, beta, maxiter, top_k=20)
        c_local = compute_c5(h_local)
        h_local, c_local = local_search(h_local, max_sweeps=15)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5

def random_feasible(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n / 2.0)

    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    for beta in [10, 30, 80, 200, 500, 1000]:
