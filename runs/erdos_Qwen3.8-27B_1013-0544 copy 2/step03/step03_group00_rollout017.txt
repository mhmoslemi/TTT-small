        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def smoothmax_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_total = len(corr)
    if top_k is not None and top_k < m_total:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(m_total)
    v = corr[top_idx]
    m = v.max()
    exps = np.exp(beta * (v - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = top_idx[ii]
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        else:
            ll = -lag
            grad[ll:] += wi * (1.0 - h[:n - ll])
            grad[:n - ll] -= wi * h[ll:]
    grad *= dx
    return val, grad

def pgd_solve(h_init, beta, top_k, n_iter):
    """Projected gradient descent with adaptive step size."""
    n = len(h_init)
    target = n / 2.0
    h = project_box_sum(h_init.copy(), target)
    f, g = smoothmax_grad(h, beta, top_k)
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-12:
        return h, f
    step = 1.0 / gnorm
    best_h = h.copy()
    best_f = f
    
    for it in range(n_iter):
        # Gradient step
        h_new = project_box_sum(h - step * g, target)
        f_new, g_new = smoothmax_grad(h_new, beta, top_k)
        
        if f_new < f:
            # Try to increase step (BB-like)
            h = h_new
            f = f_new
            g = g_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
            # Adaptive step
            gnorm = np.linalg.norm(g)
            if gnorm > 1e-12:
                step = min(2.0 / gnorm, step * 1.1)
        else:
            # Decrease step
            step *= 0.7
            if step < 1e-10:
                break
    
    return best_h, best_f

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_grad(x, beta, top_k)[0]
    
    def objg(x):
        return smoothmax_grad(x, beta, top_k)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h):
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return h

    return project_box_sum(np.clip(h_new, 0.0, 1.0), n_new / 2.0)
