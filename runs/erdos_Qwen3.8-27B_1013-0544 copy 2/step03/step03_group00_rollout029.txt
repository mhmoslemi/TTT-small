```python
        else:

            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n

    m = np.max(corr)
```

```python
def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - n + 1  # actual shift
        # ∂C[s]/∂h[j] = (1-h[j-s]) * [0≤j-s<n] - h[j+s] * [0≤j+s<n]
        # For the first term: j-s valid means j from s to s+n-1, but j in [0,n-1]
        #   So j from max(0,s) to min(n-1, s+n-1)
        #   h[j-s] for j in [max(0,s), min(n-1,s+n-1)] is h[0 : min(n-1,s+n-1)-max(0,s)+1]
        # For the second term: j+s valid means j from -s to n-1-s, but j in [0,n-1]
        #   So j from max(0,-s) to min(n-1, n-1-s)
        #   h[j+s] for j in [max(0,-s), min(n-1,n-1-s)] is h[s+max(0,-s) : s+min(n-1,n-1-s)+1]
        
        # First term: (1-h[j-s]) for j where j-s is in [0, n-1]
        # j-s in [0, n-1] => j in [s, s+n-1], intersect [0, n-1]
        j_lo1 = max(0, s)
        j_hi1 = min(n - 1, s + n - 1)  # inclusive
        if j_hi1 >= j_lo1:
            # h[j-s] for j in [j_lo1, j_hi1] => h[j_lo1-s : j_hi1-s+1]
            grad[j_lo1:j_hi1+1] += wk * (1.0 - h[j_lo1-s:j_hi1-s+1])
        
        # Second term: -h[j+s] for j where j+s is in [0, n-1]
        # j+s in [0, n-1] => j in [-s, n-1-s], intersect [0, n-1]
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)  # inclusive
        if j_hi2 >= j_lo2:
            # h[j+s] for j in [j_lo2, j_hi2] => h[j_lo2+s : j_hi2+s+1]
            grad[j_lo2:j_hi2+1] += wk * (-h[j_lo2+s:j_hi2+s+1])
    
    grad *= dx
    return val, grad
```

```python
def pc_optimize(h_init, n_pcs, n_total, beta, maxiter=200):
    """Optimize a piecewise-constant function with n_pcs pieces, evaluated at n_total points."""
    piece_size = n_total // n_pcs
    # Initial values
    vals = np.zeros(n_pcs)
    for i in range(n_pcs):
        vals[i] = np.mean(h_init[i*piece_size:(i+1)*piece_size])
    
    def obj_from_vals(vals):
        h = np.repeat(vals, piece_size)
        # Adjust for remainder
        if len(h) < n_total:
            h = np.append(h, np.repeat(vals[-1], n_total - len(h)))
        h = project_feasible(h, n_total / 2.0)
        return h
    
    def obj(x):
        h = obj_from_vals(x)
        v, g = smoothmax_grad(h, beta)
        # Chain rule: d/dx_i = sum over j in piece i of g[j]
        grad_pc = np.zeros(n_pcs)
        for i in range(n_pcs):
            grad_pc[i] = g[i*piece_size:(i+1)*piece_size].sum()
        # Penalty for sum constraint
        s = h.sum() - n_total / 2.0
        v += 100.0 * s * s
        grad_pc += 100.0 * 2 * s / n_pcs  # approximate
        return v, grad_pc
    
    bounds = [(0.0, 1.0)] * n_pcs
    res = minimize(obj, vals, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter})
    return obj_from_vals(res.x)
```
