    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for it in range(n_steps):
        # Compute gradient: sum of p_k * grad_f_k
        g = 1.0 - h
        # Only compute for shifts where p_k > threshold
        active = np.where(p > 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            pk = p[k] * dx
            if lag >= 0:
                grad[:n-lag] += pk * g[lag:]
                grad[lag:] -= pk * h[:n-lag]
            else:
                la = -lag
                grad[la:] += pk * g[:n-la]
                grad[:n-la] -= pk * h[la:]
        # Adam step
        t += 1
        h_new, m, v = adam_update(h, grad, m, v, t, lr, 0.9, 0.999, 1e-8)
        h = project_feasible(h_new, target_sum)
    # Update p
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    p_new = p * np.exp(beta * (f - f.max()))
    p_new /= p_new.sum()
    return h, p_new

    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    shifted = beta * (corr - corr.max())
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        k = lag + (n - 1)
        wval = w[k]
        if wval < 1e-14:
            continue
        if lag >= 0:
            grad[:n - lag] += wval * g[lag:]
            grad[lag:] -= wval * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wval * g[:n - la]
            grad[:n - la] -= wval * h[la:]
    return dx * grad

def grad_weighted(h, p, dx):
    """Gradient of sum_k p_k * f_k(h)"""
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lag = k - (n - 1)
        pk = p[k] * dx
        if lag >= 0:

            grad[lag:] -= pk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += pk * g[:n - la]
            grad[:n - la] -= pk * h[la:]
    return grad

def adam_update(h, grad, m, v, t, lr, b1, b2, eps):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v

def _con_fun(h, target_sum):
    return h.sum() - target_sum

def _con_jac(h, target_sum):
    return np.ones(len(h))

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def _obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _smooth_max(corr, beta) * dx

def _grad_smooth(h, beta, dx):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    sh = beta * (corr - corr.max())
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        k = lag + (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if lag >= 0:
            grad[:n - lag] += wk * g[lag:]
            grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wk * g[:n - la]
            grad[:n - la] -= wk * h[la:]
    return dx * grad

def _grad_weighted(h, p, dx):
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)

        else:

    return grad

    v_hat = v / (1.0 - b2 ** t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps), m, v

    return h.sum() - ts

    return np.ones(len(h))
