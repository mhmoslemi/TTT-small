```python
def compute_grad(h, weights, n):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if abs(w) < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # h[j] * (1-h[j+lag]) for j = 0..n-1-lag
            # contribution from h[j]: (1-h[j+lag]) for j in 0..n-1-lag
            # contribution from (1-h[j]): -h[j-lag] for j in lag..n-1
            if n - 1 - lag >= 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n-lag]
        else:
            # lag < 0, let d = -lag
            d = -lag
            # h[j] * (1-h[j-d]) for j = d..n-1
            # contribution from h[j]: (1-h[j-d]) for j in d..n-1
            # contribution from (1-h[j]): -h[j+d] for j in 0..n-1-d
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] -= w * h[d:]
    return grad
```

```python
            res = minimize(_obj_func, h_cur, jac=_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=constr,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgrad_polish(h, target, n_pts, t0, time_limit):
    global _g_target
    _g_target = target
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    momentum = np.zeros(n_pts)
    for it in range(30000):
        if time.time() - t0 > time_limit:
            break
        step = 0.12 / (1.0 + 0.0004 * it)
        g = topk_subgrad(cur, K=5)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = -g / gn
        momentum = 0.7 * momentum + 0.3 * g
        mn = np.linalg.norm(momentum)
        if mn < 1e-12:
            break
        momentum /= mn
        accepted = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_trial = project_h(cur + step * shrink * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-12:
                cur = h_trial
                cur_c5 = c5_trial
                accepted = True
                break
        if not accepted:
            momentum *= 0.5
    return cur, cur_c5

def coord_polish(h, target, n_pts, t0, time_limit):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    while time.time() - t0 < time_limit:
        i = np.random.randint(n_pts)
        for delta in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2]:
            h_trial = cur.copy()
            h_trial[i] += delta
            h_trial = project_h(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-10:
                cur = h_trial
                cur_c5 = c5_trial
                break
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)

    n_base = len(h_parent)
    target_base = n_base / 2.0
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase A: SLSQP at base resolution with multiple starts
    starts = [h_parent.copy()]
    for _ in range(3):
        pert = h_parent + 0.1 * np.random.randn(n_base)
        starts.append(project_h(pert, target_base))

    for h_start in starts:
        if time.time() - t0 > budget_s * 0.25:
            break
        h_opt, c5_opt = slsqp_optimize(h_start, target_base, n_base, t0, budget_s * 0.25)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase B: Subgradient polish at base resolution
    if time.time() - t0 < budget_s * 0.55:
        h_sg, c5_sg = subgrad_polish(best_h, target_base, n_base, t0, budget_s * 0.55)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg.copy(), c5_sg

    # Phase C: Coordinate polish at base
    if time.time() - t0 < budget_s * 0.65:
        h_cp, c5_cp = coord_polish(best_h, target_base, n_base, t0, budget_s * 0.65)
        if c5_cp < best_c5:
            best_h, best_c5 = h_cp.copy(), c5_cp

    # Phase D: Higher resolution attempts
    for mult in [2.0, 2.5, 3.0]:
        if time.time() - t0 > budget_s * 0.75:
            break
        n_hi = int(n_base * mult)
        if n_hi > 1200:
            continue
        target_hi = n_hi / 2.0
        h_hi = project_h(interpolate_h(best_h, n_hi), target_hi)
        h_hi, c5_hi = slsqp_optimize(h_hi, target_hi, n_hi, t0, budget_s * 0.8)
        if c5_hi < best_c5:
            best_h, best_c5 = h_hi.copy(), c5_hi
        if time.time() - t0 < budget_s * 0.85:
            h_sg2, c5_sg2 = subgrad_polish(best_h, n_hi / 2.0, n_hi, t0, budget_s * 0.85)
            if c5_sg2 < best_c5:
                best_h, best_c5 = h_sg2.copy(), c5_sg2

    # Phase E: Final polish at best resolution
    n_best = len(best_h)
    target_best = n_best / 2.0
    if time.time() - t0 < budget_s * 0.95:
        h_final, c5_final = coord_polish(best_h, target_best, n_best, t0, budget_s * 0.95)
        if c5_final < best_c5:
            best_h, best_c5 = h_final.copy(), c5_final

    # Phase F: Binary rounding
    hb, cb = try_binary(best_h, len(best_h) / 2.0)
    if cb < best_c5:
        best_h, best_c5 = hb.copy(), cb

    return best_h, best_c5, int(len(best_h))
```
