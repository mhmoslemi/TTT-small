import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad_weights(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    w = e / e.sum()
    return w

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] += (-h[lag:]) * dx
    else:
        l = -lag
        grad[:n-l] += (1.0 - h[l:]) * dx
        grad[l:] += (-h[:n-l]) * dx
    return grad

def obj_smmax(h, dx, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def obj_smmax_grad(h, dx, beta, n, thresh=1e-10):
    c = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_grad_weights(c, beta)
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < thresh:
            continue
        lag = idx - center
        grad += w[idx] * grad_lag(h, lag, n, dx)
    return grad

def con_sum_fun(h, target):
    return np.sum(h) - target

def con_sum_jac(h, target):
    return np.ones(len(h))

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def slsqp_refine(h, target, dx, n, betas, maxiter, time_limit_start, budget_s):

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        h = best_h
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += (-h[lag:]) * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += (-h[:n - l]) * dx
    return grad

def obj_smmax(h, dx, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def obj_smmax_grad(h, dx, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    e = np.exp(x_shift)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    thresh = 1e-12
    for idx in range(len(c)):
        if w[idx] < thresh:
            continue
        lag = idx - center
        grad += w[idx] * grad_lag(h, lag, n, dx)
    return grad

def con_sum_fun(h, target):
    return float(np.sum(h) - target)

def con_sum_jac(h, target):
    return np.ones(len(h))

def subgrad_max(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for idx in active:
        lag = int(idx) - center
        grad += grad_lag(h, lag, n, dx)
    grad /= len(active)
    return grad, max_val

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def slsqp_phase(h, target, dx, n, betas, maxiter, start_t, budget):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in betas:
        if time.time() - start_t > budget:
            break
        f = partial(obj_smmax, dx=dx, beta=beta, n=n)
        g = partial(obj_smmax_grad, dx=dx, beta=beta, n=n)
        cf = partial(con_sum_fun, target=target)
        cj = partial(con_sum_jac, target=target)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(f, best_h, jac=g, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad_descent(h, target, n, start_t, budget, step_init=0.08, max_iters=8000):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    step = step_init
    stalled = 0
    for it in range(max_iters):
        if time.time() - start_t > budget:
            break
        grad, _ = subgrad_max(h_cur, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_unit = grad / gnorm
        improved = False
        for _ in range(30):
            h_trial = project_h(h_cur - step * grad_unit, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-15:
                h_cur = h_trial
                c5_cur = c5_trial
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                improved = True
                break
            step *= 0.5
        if improved:
            stalled = 0
            step = min(step * 1.15, 0.4)
        else:
            stalled += 1
            step *= 0.6
            if step < 0.003 and stalled > 50:
                T = 0.003 * np.exp(-it / 2000.0)
                h_trial = project_h(h_cur + np.random.normal(0, T, n), target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_cur + 2.0 * T:
                    h_cur = h_trial
                    c5_cur = c5_trial
                    step = 0.04
                    stalled = 0
            elif stalled > 200:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_t = time.time()

    def tl():
        return budget_s - (time.time() - start_t)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_parent = 250
        x = np.linspace(0, 2, n_parent)
        h_parent = project_h(0.5 + 0.2 * np.cos(np.pi * x), n_parent / 2.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Phase 1: SLSQP at parent resolution
    n = best_n
    target = n / 2.0
    dx = 2.0 / n
    h1, c5_1 = slsqp_phase(best_h, target, dx, n, [60, 120, 250, 500, 1000], 1500, start_t, budget_s * 0.3)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n

    # Phase 2: Subgradient descent at parent resolution
    if tl() > 60:
        h2, c5_2 = subgrad_descent(best_h, target, n, start_t, budget_s * 0.45, step_init=0.06, max_iters=10000)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n

    # Phase 3: Higher resolutions
    for n_try in [300, 350, 400]:
        if tl() < 100:
            break
        target_t = n_try / 2.0
        dx_t = 2.0 / n_try
        h_init = project_h(resample(best_h, n_try), target_t)
        h3, c5_3 = slsqp_phase(h_init, target_t, dx_t, n_try, [80, 200, 500], 1000, start_t, max(20, tl() - 40))
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n_try
            # Quick subgradient at this resolution
            if tl() > 40:
                h4, c5_4 = subgrad_descent(best_h, target_t, n_try, start_t, max(10, tl() - 20),
                                           step_init=0.04, max_iters=3000)
                if c5_4 < best_c5:
                    best_h, best_c5, best_n = h4, c5_4, n_try

    # Phase 4: Final polish at best resolution
    if tl() > 15:
        n_f = best_n
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        # SLSQP with very high beta
        hf, cf = slsqp_phase(best_h, target_f, dx_f, n_f, [500, 1000, 2000], 500, start_t, max(5, tl() - 10))
        if cf < best_c5:
            best_h, best_c5 = hf, cf
        # Tiny subgradient steps
        if tl() > 10:
            hfin, cfin = subgrad_descent(best_h, target_f, n_f, start_t, max(3, tl() - 2),
                                         step_init=0.01, max_iters=5000)
            if cfin < best_c5:
                best_h, best_c5 = hfin, cfin

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
