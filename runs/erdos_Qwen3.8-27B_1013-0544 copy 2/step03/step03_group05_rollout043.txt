Actually, I just realized something important. The `grad_lag_vec` function in the original code computes the gradient of `corr_lag * dx`. But when I use it in the LP, I need to be consistent. Let me be very careful about the math.

For lag ≥ 0:
f(h) = dx * sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])

∂f/∂h[j]:
- j appears as h[i+lag] when i = j-lag (valid for lag ≤ j ≤ n-1): contribution = dx * (1 - h[j-lag])
- j appears as h[i] when i = j (valid for 0 ≤ j ≤ n-1-lag): contribution = dx * (-h[j+lag])

So:
∂f/∂h[j] = dx * (1 - h[j-lag]) [if lag ≤ j ≤ n-1] + dx * (-h[j+lag]) [if 0 ≤ j ≤ n-1-lag]

For j in [lag, n-1-lag] (both terms active):
∂f/∂h[j] = dx * (1 - h[j-lag] - h[j+lag])

For j in [n-1-lag+1, n-1] (only first term):
∂f/∂h[j] = dx * (1 - h[j-lag])

For j in [0, lag-1] (only second term):
∂f/∂h[j] = dx * (-h[j+lag])

This matches what the original code does:
```
grad[lag:] += (1.0 - h[:n - lag]) * dx
grad[:n - lag] += -h[lag:] * dx
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag_vec(h, lag, n, dx):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[lag:] += (1.0 - h[:m]) * dx
        grad[:m] += -h[lag:] * dx
    else:
        l = -lag
        m = n - l
        grad[:m] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:m] * dx
    return grad

def corr_lag_val(h, lag, n, dx):
    if lag >= 0:
        m = n - lag
        return np.sum(h[lag:] * (1.0 - h[:m])) * dx
    else:
        l = -lag
        m = n - l
        return np.sum(h[:m] * (1.0 - h[l:])) * dx

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def sla_iteration(h0, n, target, dx, active_lags):
    """One SLPA iteration: solve LP with linearized constraints for active_lags."""
    center = n - 1
    A_rows = []
    b_vals = []
    for k in active_lags:
        lag = k - center
        g = grad_lag_vec(h0, lag, n, dx)
        val = corr_lag_val(h0, lag, n, dx)
        rhs = val - np.dot(g, h0)
        row = np.concatenate([g, [-1.0]])
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Variables: h[0], ..., h[n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n].copy()
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

_G_TARGET = 0
_G_DX = 0.0
_G_BETA = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag_vec(h, lag, n, dx):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[lag:] += (1.0 - h[:m]) * dx
        grad[:m] += -h[lag:] * dx
    else:
        l = -lag
        m = n - l
        grad[:m] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:m] * dx
    return grad

def corr_lag_val(h, lag, n, dx):
    if lag >= 0:
        m = n - lag
        return np.sum(h[lag:] * (1.0 - h[:m])) * dx
    else:
        l = -lag
        m = n - l
        return np.sum(h[:m] * (1.0 - h[l:])) * dx

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def obj_fun(h):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, _G_BETA) * _G_DX

def con_fun(h):
    return h.sum() - _G_TARGET

def con_jac(h):
    return np.ones(len(h))

def sla_solve(h0, n, target, dx, active_idx):
    center = n - 1
    A_rows = []
    b_vals = []
    for k in active_idx:
        lag = int(k) - center
        g = grad_lag_vec(h0, lag, n, dx)
        val = corr_lag_val(h0, lag, n, dx)
        rhs = val - np.dot(g, h0)
        A_rows.append(np.concatenate([g, [-1.0]]))
        b_vals.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq_arr = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq_arr,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq_arr,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None
    if res.status != 0 or res.x is None:
        return None
    return res.x[:n].copy()

def sla_optimize(h_init, n, target, dx, max_iters, time_fn, min_time):
    h0 = h_init.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    h_prev = None
    alpha = 1.0
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_vals = np.correlate(h0, 1.0 - h0, mode="full") * dx
        mx = np.max(corr_vals)
        active = np.where(corr_vals > mx * 0.90)[0]
        if len(active) < 3:
            active = np.argsort(corr_vals)[::-1][:5]
        h_lp = sla_solve(h0, n, target, dx, active)
        if h_lp is None:
            break
        h_cand = (1.0 - alpha) * h0 + alpha * h_lp
        h_cand = project_h(h_cand, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            alpha = min(1.0, alpha * 1.2)
        else:
            alpha = max(0.3, alpha * 0.7)
        if h_prev is not None and np.max(np.abs(h_cand - h_prev)) < 1e-10:
            break
        h_prev = h_cand.copy()
        h0 = h_cand
    return best_h, best_c5

def gradient_descent(h_init, n, target, dx, max_iters, time_fn, min_time, best_state, top_k):
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_val = np.correlate(cur, 1.0 - cur, mode="full") * dx
        idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[idx]
        temp = 250.0 + 100.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_lag_vec(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        step = 0.035 / (1.0 + 0.002 * it)
        improved = False
        for _ in range(20):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved:
            T = 0.0025 * np.exp(-it / (max(1, max_iters) * 0.6))
            if T > 1e-7:
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 4:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
    return cur, cur_c5

        h_pb = 0.5 + 0.22 * (np.exp(-((x - 0.35) / 0.25) ** 2) + np.exp(-((x - 1.65) / 0.25) ** 2))
        inits.append(project_h(h_pb, target))

            cur = project_h(h_init.copy(), target)
```
