import numpy as np
from scipy.optimize import minimize, Bounds
import time


def fft_correlate(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b[::-1], size)
    return np.fft.irfft(fa * fb)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smoothmax(h, beta, dx):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    sw = np.sum(weights)
    weights /= sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            M = -L
            grad[:n - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n - M]
    return grad * dx


def subgrad_active(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for L in lags:
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
        else:
            M = -L
            grad[:n - M] += (1.0 - h[M:])
            grad[M:] -= h[:n - M]
    return grad


def lbfgs_optimize(h_cur, n_pts, beta, dx, target, maxiter, pw):
    def obj(x):
        c = fft_correlate(x, 1.0 - x)
        return smooth_max(c, beta) * dx + pw * (x.sum() - target) ** 2

    def grad_obj(x):
        g = grad_smoothmax(x, beta, dx)
        g += 2.0 * pw * (x.sum() - target)
        return g

    bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    res = minimize(obj, h_cur, jac=grad_obj, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                           'gtol': 1e-12, 'maxls': 50})
    return res.x


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(128, 0.5)

    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at increasing resolutions
    resolutions = [256, 320, 400]
    for n_pts in resolutions:
        if tl() < 100:
            break
        target = n_pts / 2.0
        dx = 2.0 / n_pts
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_pts)
        h_cur = project_h(np.interp(x_new, x_old, best_h), target)
        cur_c5 = compute_c5(h_cur)
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

        for beta in [200, 600, 2000, 5000]:
            if tl() < 25:
                break
            pw = 50000.0
            try:
                h_new = lbfgs_optimize(h_cur, n_pts, beta, dx, target, 500, pw)
                h_new = project_h(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h_cur = h_new
                    cur_c5 = c5_new
            except Exception:
                pass

        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

    # Phase 2: Try random restart at current best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    dx_cur = 2.0 / n_cur
    for restart in range(3):
        if tl() < 60:
            break
        h_rand = np.random.uniform(0.1, 0.9, n_cur)
        h_rand = project_h(h_rand, target_cur)
        dx_r = 2.0 / n_cur
        try:
            h_r = lbfgs_optimize(h_rand, n_cur, 600, dx_r, target_cur, 300, 50000.0)
            h_r = project_h(h_r, target_cur)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r
        except Exception:
            pass

    # Phase 3: Subgradient active-set refinement
    n_fin = len(best_h)
    target_fin = n_fin / 2.0
    df = 2.0 / n_fin
    no_improve = 0
    for it in range(300000):
        if tl() < 3:
            break
        corr = fft_correlate(best_h, 1.0 - best_h)
        top_k = 5
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_lags = [int(i) - (n_fin - 1) for i in top_idx]
        g = subgrad_active(best_h, top_lags)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007]:
            ht = project_h(best_h - step * g, target_fin)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h = ht
                best_c5 = c5t
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve % 300 == 299:
                pert = np.random.randn(n_fin) * (0.005 + 0.005 * no_improve / 300)
                ht = project_h(best_h + pert, target_fin)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 3000:

    # Phase 4: Final L-BFGS-B polish with high beta
    if tl() > 40:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        dx_pol = 2.0 / n_pol
        pw = 1e5
        try:
            h_pol = lbfgs_optimize(best_h, n_pol, 8000, dx_pol, target_pol, 3000, pw)
            h_pol = project_h(h_pol, target_pol)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_h = h_pol
                best_c5 = c5_pol
        except Exception:
            pass

    # Phase 5: Final fine subgradient sweep
    n_fin = len(best_h)
    target_fin = n_fin / 2.0
    for it in range(100000):
        if tl() < 1:
            break
        corr = fft_correlate(best_h, 1.0 - best_h)
        top_idx = np.argsort(corr)[::-1][:5]
        top_lags = [int(i) - (n_fin - 1) for i in top_idx]

            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h = ht
                best_c5 = c5t
                break

    return best_h, best_c5, len(best_h)
