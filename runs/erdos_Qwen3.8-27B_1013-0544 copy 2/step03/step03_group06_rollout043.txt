    dx = 2.0 / n

def project_to_sum(h, target):
    """Project h onto {x : sum(x) = target, 0 <= x <= 1}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / (beta * len(x))

def _grad(x):
    dx = _opt_params['dx']
    beta = _opt_params['beta']
    target = _opt_params['target']
    pen = _opt_params['penalty']
    n = len(x)
    
    corr = np.correlate(x, 1.0 - x, mode="full")
    # Compute weights for smooth max
    shifted = np.clip(beta * (corr - np.max(corr)), -500, 500)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Compute dcorr/dh matrix (sparse structure)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-15:
            continue
        k = i - (n - 1)
        if k >= 0:
            # ∂corr_i/∂h[m] = (1 - h[m-k]) * I[k ≤ m] - h[m+k] * I[m ≤ n-1-k]
            if k == 0:
                grad += w * (1.0 - 2.0 * x)
            else:
                # First term: for m in [k, n-1], add (1 - x[m-k])
                grad[k:] += w * (1.0 - x[:n-k])
                # Second term: for m in [0, n-1-k], subtract x[m+k]
                grad[:n-k] -= w * x[k:]
        else:
            kp = -k
            # ∂corr_i/∂h[m] = (1 - h[m+kp]) * I[m ≤ n-1-kp] - h[m-kp] * I[m ≥ kp]
            # First term: for m in [0, n-1-kp], add (1 - x[m+kp])
            grad[:n-kp] += w * (1.0 - x[kp:])
            # Second term: for m in [kp, n-1], subtract x[m-kp]
            grad[kp:] -= w * x[:n-kp]
    
    # Add penalty gradient
    grad += 2.0 * pen * (x.sum() - target)
    
    return dx * grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_op = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / (beta * len(corr))

def _obj(x):
    dx = _op['dx']
    beta = _op['beta']
    target = _op['target']
    pen = _op['pen']
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, beta) * dx + pen * (x.sum() - target) ** 2

def _grad(x):
    dx = _op['dx']
    beta = _op['beta']
    target = _op['target']
    pen = _op['pen']
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    shifted = np.clip(beta * (corr - np.max(corr)), -500, 500)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-14:
            continue
        k = i - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * x)
            else:
                grad[k:] += w * (1.0 - x[:n - k])
                grad[:n - k] -= w * x[k:]
        else:
            kp = -k
            grad[:n - kp] += w * (1.0 - x[kp:])
            grad[kp:] -= w * x[:n - kp]
    grad += 2.0 * pen * (x.sum() - target)
    return dx * grad

        c5_new = compute_c5(h_opt)
        if c5_new < best_c5:
            best_h, best_c5 = h_opt, c5_new

    # Phase 2: Higher resolution exploration
    for n_hi in [240, 280, 320, 400, 480]:
        if rem() < 80:
            break
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_hi)
        h_hi = project_to_sum(np.interp(xn, xo, best_h), n_hi / 2.0)
        h_opt = runlbfgs(h_hi, n_hi, 1200, 1e-15)
        c5_new = compute_c5(h_opt)
        if c5_new < best_c5:
            best_h, best_c5 = h_opt, c5_new

    # Phase 3: Momentum subgradient at current best resolution
    nc = len(best_h)
    tg = nc / 2.0
    mom = np.zeros(nc)
    mc = 0.6
    noimp = 0
    for it in range(100000):
        if rem() < 10:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        ti = np.argsort(corr)[::-1][:5]
        tl = [int(i) - (nc - 1) for i in ti]
        g = subgrad_dir(best_h, tl)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        mom = mc * mom + (1 - mc) * g
        mn = np.linalg.norm(mom)
        if mn > 1e-10:
            mom /= mn
        found = False
        for stp in [0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008]:
            ht = project_to_sum(best_h - stp * mom, tg)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                best_h, best_c5 = ht, ct
                noimp = 0
                found = True
                break
        if not found:
            noimp += 1
            if noimp > 600 and noimp % 300 == 0:
                htry = project_to_sum(best_h + np.random.randn(nc) * 0.006, tg)
                ct = compute_c5(htry)
                if ct < best_c5:
                    best_h, best_c5 = htry, ct
                    noimp = 0
            if noimp > 10000:
                break

    # Phase 4: Random restarts with re-optimization
    nc = len(best_h)
    tg = nc / 2.0
    for rr in range(30):
        if rem() < 40:
            break
        noise_scale = 0.03 * (1.0 + rr * 0.15)
        h_pert = project_to_sum(best_h + np.random.randn(nc) * noise_scale, tg)
        h_opt = runlbfgs(h_pert, nc, 500, 1e-14)
        c5_new = compute_c5(h_opt)
        if c5_new < best_c5:
            best_h, best_c5 = h_opt, c5_new

    # Phase 5: Final intensive polish
    if rem() > 30:
        nf = len(best_h)
        tg = nf / 2.0
        h_pol = runlbfgs(best_h.copy(), nf, 3000, 1e-16)
        c5_new = compute_c5(h_pol)
        if c5_new < best_c5:
            best_h, best_c5 = h_pol, c5_new

    # Phase 6: Tiny subgradient refinement
    nf = len(best_h)
    tg = nf / 2.0
    for it in range(50000):
        if rem() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        ti = np.argsort(corr)[::-1][:4]
        tl = [int(i) - (nf - 1) for i in ti]
        g = subgrad_dir(best_h, tl)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        for stp in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            ht = project_to_sum(best_h - stp * g, tg)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-16:
                best_h, best_c5 = ht, ct
                break

    return best_h, best_c5, len(best_h)
