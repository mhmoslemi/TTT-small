    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for scale in [0.015, -0.015, 0.006, -0.006, 0.002, -0.002]:
                trial = h.copy()
                trial[j] += scale
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 4:
            break
    return h, best_c5

def shift_h(h, shift_frac):
    n = len(h)
    shift = int(round(shift_frac * n))
    return np.roll(h, shift)

def fourier_init(h_ref, n, n_modes, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, n_modes + 1):
        a = rng.uniform(-0.25, 0.25)
        b = rng.uniform(-0.25, 0.25)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = 0.4 * h + 0.6 * h_ref
    return project_h(h, n / 2.0)

def block_move_optimize(h, n, target, t0, budget, n_iter=2000):
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        bs = rng.randint(3, 15)
        start = rng.randint(0, max(1, n - bs))
        end = min(start + bs, n)
        for delta in [0.04, -0.04, 0.02, -0.02, 0.01, -0.01]:
            trial = h.copy()
            trial[start:end] += delta
            trial = project_h(trial, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                h = trial
                best_c5 = c_try
                improved = True
                break
        if not improved and it > 200:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(200, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base
    target = n_base / 2.0
    bounds = Bounds(np.zeros(n_base), np.ones(n_base))

    # Phase 1: Fourier-based diverse exploration
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        for s in range(6):
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            hp = fourier_init(best_h, n_base, 10, rng)
            hp = do_slsqp(hp, n_base, target, 80, bounds, 600, 1e-13)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_h, best_c5 = hp.copy(), cp

    # Phase 2: Shift exploration (exploit near-shift symmetry)
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        for s in range(10):
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            sh = rng.randint(1, max(2, n_base // 5))
            hp = shift_h(best_h, sh / float(n_base))
            hp = project_h(hp, target)
            hp = do_slsqp(hp, n_base, target, 150, bounds, 400, 1e-14)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_h, best_c5 = hp.copy(), cp

    # Phase 3: Main SLSQP with adaptive beta (increasing sharpness)
    for beta in [40, 100, 200, 400, 800, 1500, 3000, 5000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 12:
            break
        best_h = do_slsqp(best_h, n_base, target, beta, bounds, 4000, 1e-15)
        best_c5 = compute_c5(best_h)

    # Phase 4: Block moves for structural changes
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = block_move_optimize(best_h, n_base, target, t0, budget_s, 1500)

    # Phase 5: Gaussian perturbation multi-start
    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        for s in range(10):
            remaining = budget_s - (time.time() - t0)
            if remaining < 7:
                break
            scale = 0.035 / (s + 1)
            hp = project_h(best_h + rng.randn(n_base) * scale, target)
            hp = do_slsqp(hp, n_base, target, 250, bounds, 1200, 1e-14)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_h, best_c5 = hp.copy(), cp

    # Phase 6: Fine coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = coord_descent(best_h, target, n_base, 60, t0, budget_s)

    # Phase 7: Higher resolution refinement
    for n_hi in [n_base + 80, n_base + 160]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 40:
            break
        target_hi = n_hi / 2.0
        bounds_hi = Bounds(np.zeros(n_hi), np.ones(n_hi))
        cur_hi = project_h(resample(best_h, n_hi), target_hi)
        for beta in [100, 400, 1000, 2500]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            cur_hi = do_slsqp(cur_hi, n_hi, target_hi, beta, bounds_hi, 2000, 1e-14)
        cur_c5 = compute_c5(cur_hi)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_hi.copy(), cur_c5, n_hi

    # Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [5000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 4:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 3000, 1e-15)
            best_c5 = compute_c5(best_h)
        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = coord_descent(best_h, tf, nf, 50, t0, budget_s)

    return best_h, best_c5, best_n
