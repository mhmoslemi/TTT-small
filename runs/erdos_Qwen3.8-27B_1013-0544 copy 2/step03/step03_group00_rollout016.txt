import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def fista_optimize(h0, beta, max_iter, top_k=None, lr=None):
    """FISTA (accelerated projected gradient) for smoothmax objective."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    y = h0.copy()
    
    if lr is None:
        # Estimate Lipschitz constant
        # Use backtracking
        lr = 0.1
        # Try to estimate
        v, g = smoothmax_val_grad(h, beta, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-15:
            lr = 0.5 / gnorm * 0.1  # small initial step
        lr = min(lr, 0.05)
    
    for t in range(max_iter):
        y_next = h.copy()
        if t > 0:
            momentum = (t - 1) / (t + 2)
            y_next = h + momentum * (h - y)
        
        v, g = smoothmax_val_grad(y_next, beta, top_k)
        h_new = project_feasible(y_next - lr * g, target)
        y = h
        h = h_new
    
    return h

def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def lbfgsb_solve(h, beta, maxiter, top_k=None):
    """L-BFGS-B with penalty for sum constraint."""
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        penalty = 100.0 * (np.sum(x) - target) ** 2
        return v + penalty
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        gp = 200.0 * (np.sum(x) - target) * np.ones(n)
        return g + gp
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def bump_perturbation(h, rng, amplitude, width):
    """Add a Gaussian bump to h."""
    n = len(h)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    center = rng.uniform(0, 2)
    bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
    sign = rng.choice([-1, 1])
    return project_feasible(h + sign * bump, n / 2.0)

def basin_hop_structured(h_best, best_c5, n_hops, seed, t0, budget_s):

        if r < 0.3:
            # Gaussian noise

        elif r < 0.6:
            # Bump perturbation
            h_pert = bump_perturbation(h_best, rng, rng.uniform(0.05, 0.15), rng.uniform(0.1, 0.3))
        else:
            # Block swap

            j = rng.randint(0, n - 10)
            h_pert = h_best.copy()
            h_pert[i:i+10], h_pert[j:j+10] = h_pert[j:j+10], h_pert[i:i+10]
            h_pert = project_feasible(h_pert, n / 2.0)

        c_local = compute_c5(h_local)
        if c_local < best_c5:
