Since all w_k ≥ 0, this is minimized at v = 0. But with the sum constraint, we need sum(h) = n/2.

A practical approach: do the subgradient descent on the smooth max + penalty for the sum, with a large penalty.

= ∑_m h[m] * ∑_k p_k * (1-h[m+k])  (summing over valid k for each m)
= ∑_m h[m] * (1 - ∑_k p_k * h[m+k])  (where the inner sum is over valid k)

This IS separable in m! Each term only involves h[m] and other h values as coefficients.

min_h ∑_m [h[m] * (1 - ∑_k p_k * h[m+k]) * dx + μ * h[m]]
s.t. 0 ≤ h[m] ≤ 1, sum(h) = n/2

But wait, this is still coupled through the constraint sum(h) = n/2. Without that constraint, each h[m] is independently optimized.

For a single h[m], the objective is:
h[m] * (1 - S_m) * dx + μ * h[m]

where S_m = ∑_k p_k * h[m+k] (sum over valid k).

This is linear in h[m]! So the optimum is at h[m] = 0 or h[m] = 1, depending on the sign of the coefficient.

Coefficient = (1 - S_m) * dx + μ

If coefficient > 0: h[m] = 0
If coefficient < 0: h[m] = 1
If coefficient = 0: any value in [0,1]

This is great! It means for fixed p and μ, the optimal h is binary (or has some free variables). The sum constraint then determines which free variables get intermediate values.

This suggests a **dual ascent** approach:
1. Start with some p (uniform) and μ
2. Given p, μ: solve the inner problem (set h[m] = 0 or 1 based on sign, adjust free variables for sum constraint)
3. Update p: p_k ∝ exp(corr_k * dx / τ) (softmax over the correlations)
4. Update μ: via bisection to satisfy sum constraint
5. Repeat

This is essentially a **mirror descent / multiplicative weights** approach on the dual!

The dual variable p represents the "weights" on different lags, and we're finding the p that makes the inner problem's optimum have the smallest max correlation.

Let me formalize this:
- Primal: min_h max_k f_k(h) s.t. g(h) = 0, 0 ≤ h ≤ 1
- Dual: max_{p, μ} min_h ∑_k p_k f_k(h) + μ g(h) s.t. p_k ≥ 0, ∑p_k = 1
- The inner min is separable as shown above.

For the update:
- p_k ← p_k * exp(f_k(h) / τ) / Z (multiplicative weights)
- μ ← μ - η * (sum(h) - n/2) (gradient ascent on dual)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    w = e / e.sum()
    return w  # softmax weights

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr(h, lag, n, dx):
    """Gradient of corr at given lag (output index offset from center)."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def subgradient_descent(h, n, target, dx, max_iters, time_limit, initial_time):
    """Subgradient descent with momentum and adaptive weights."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur_c5
    best_h = cur.copy()
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for it in range(max_iters):
        if time.time() - initial_time > time_limit:
            break
        
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        center = n - 1
        
        # Adaptive: use top-k with softmax weights
        # Focus more on the top lags as we progress
        k_focus = min(3, max(1, 3 - it // 5000))
        top_idx = np.argsort(corr_val)[::-1][:k_focus]
        vals = corr_val[top_idx]
        
        # Temperature decreases over time (sharper focus)
        temp = 200.0 * (1.0 + 0.5 * it / max(1, max_iters))
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_corr(cur, lag, n, dx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Heavy ball momentum
        momentum = mu * momentum - grad
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm > 0:
            momentum /= momentum_norm
        
        # Adaptive step size (Barzilai-Borwein style)
        base_step = 0.05 / (1.0 + 0.01 * it)
        step = base_step
        
        improved = False
        for _ in range(30):
            h_trial = project_h(cur - step * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best:
                    best = cur_c5
                    best_h = cur.copy()
                break
            step *= 0.5
        
        if not improved:
            # Try without momentum
            step = base_step * 0.5
            for _ in range(15):
                h_trial = project_h(cur - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    if cur_c5 < best:
                        best = cur_c5
                        best_h = cur.copy()
                    break
                step *= 0.5
            
            if not improved and it % 500 == 0:
                # SA escape
                T = 0.003 * np.exp(-it / 3000.0)
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 2:
                    cur = h_trial
                    cur_c5 = c5_trial
    
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    remaining = budget_s
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(250, 0.5)
    
    n0 = len(h0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0
    
    # Try multiple resolutions
    n_list = [n0]
    if n0 != 200: n_list.append(200)
    if n0 != 250: n_list.append(250)
    if n0 != 300: n_list.append(300)
    n_list = sorted(set(n_list))
    n_list = [n for n in n_list if 100 <= n <= 400]
    
    for n in n_list:
        if time_left() < 60:
            break
        
        target = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        
        # Start from resampled best
        h_cur = project_h(resample(best_h, n), target)
        cur_c5 = compute_c5(h_cur)
        
        # Phase 1: SLSQP with smooth max, increasing beta
        for beta in [80, 160, 320, 640, 1280]:
            if time_left() < 20:
                break
            
            def obj(h_x):
                c = np.correlate(h_x, 1.0 - h_x, mode="full")
                return smooth_max(c, beta) * dx
            
            def obj_grad(h_x):
                c = np.correlate(h_x, 1.0 - h_x, mode="full")
                m = np.max(c)
                x_shift = np.clip(beta * (c - m), -500, 500)
                e = np.exp(x_shift)
                w = e / e.sum()
                grad = np.zeros(n)
                center = n - 1
                for i, wi in enumerate(w):
                    if wi < 1e-10:
                        continue
                    lag = i - center
                    grad += wi * grad_corr(h_x, lag, n, dx)
                return grad * dx
            
            def con_fun(h_x):
                return np.sum(h_x) - target
            
            def con_jac(h_x):
                return np.ones(n)
            
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                              constraints=cons, jac=obj_grad,
                              options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5:
                    h_cur = h_new
                    cur_c5 = c5_new
            except Exception:
                pass
        
        # Phase 2: Subgradient descent with momentum
        h_cur, cur_c5 = subgradient_descent(
            h_cur, n, target, dx,
            max_iters=20000,
            time_limit=time_left() - 20,
            initial_time=start_time
        )
        
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5
            best_n = n
    
    # Final fine polish on best solution
    if time_left() > 10:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        
        # Very fine subgradient
        cur = best_h.copy()
        cur_c5 = best_c5
        momentum = np.zeros(n_f)
        mu = 0.85
        
        for it in range(50000):
            if time_left() < 3:
                break
            
            corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
            corr_val = corr_raw * dx_f
            center = n_f - 1
            
            top_idx = np.argsort(corr_val)[::-1][:2]
            vals = corr_val[top_idx]
            temp = 500.0
            weights = np.exp((vals - vals[0]) * temp)
            weights /= weights.sum()
            
            grad = np.zeros(n_f)
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                grad += w * grad_corr(cur, lag, n_f, dx_f)
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            
            momentum = mu * momentum - grad
            mnorm = np.linalg.norm(momentum)
            if mnorm > 0:
                momentum /= mnorm
            
            step = 0.005
            for _ in range(20):
                h_trial = project_h(cur - step * momentum, target_f)
                c5_trial = compute_c5(h_trial)

                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_h = cur.copy()
                        best_c5 = cur_c5
                    break
                step *= 0.5
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    return e / e.sum()

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def subgrad_descent(h, n, target, dx, max_iters, deadline, start_t):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur_c5
    best_h = cur.copy()
    momentum = np.zeros(n)
    mu = 0.88
    center = n - 1
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_idx = np.argsort(corr_val)[::-1][:3]
        vals = corr_val[top_idx]
        temp = 250.0 + 100.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            if abs(w) > 1e-6:
                grad += w * grad_single_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        momentum = mu * momentum - grad
        mnorm = np.linalg.norm(momentum)
        if mnorm > 0:
            momentum /= mnorm
        base_step = 0.06 / (1.0 + 0.005 * it)
        step = base_step
        improved = False
        for _ in range(25):
            h_trial = project_h(cur - step * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best:
                    best = cur_c5
                    best_h = cur.copy()
                    stall = 0
                break
            step *= 0.5
        if not improved:
            step = base_step * 0.5
            for _ in range(15):
                h_trial = project_h(cur - step * grad, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    if cur_c5 < best:
                        best = cur_c5
                        best_h = cur.copy()
                        stall = 0
                    break
                step *= 0.5
            if not improved:
                stall += 1
                if stall > 200 and it % 500 == 0:
                    T = 0.002 * np.exp(-it / 5000.0)
                    if T > 1e-6:
                        noise = np.random.normal(0, T, n)
                        h_trial = project_h(cur + noise, target)
                        c5_trial = compute_c5(h_trial)
                        if c5_trial < cur_c5 + T * 3:
                            cur = h_trial
                            cur_c5 = c5_trial
                            stall = 0
    return best_h, best

def slsqp_smooth(h, n, target, dx, beta, maxiter):
    def obj(h_x):
        c = np.correlate(h_x, 1.0 - h_x, mode="full")
        return smooth_max_val(c, beta) * dx

    def obj_grad(h_x):
        c = np.correlate(h_x, 1.0 - h_x, mode="full")
        w = smooth_max_weights(c, beta)
        grad = np.zeros(n)
        center = n - 1
        for i, wi in enumerate(w):
            if wi < 1e-8:
                continue
            lag = i - center
            grad += wi * grad_single_lag(h_x, lag, n, dx)
        return grad * dx

    def con_fun(h_x):
        return np.sum(h_x) - target

    def con_jac(h_x):
        return np.ones(n)

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(res.x, target)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(250, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    n_candidates = sorted(set([n0, 200, 250, 300]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 400]

    for n in n_candidates:
        if tleft() < 80:
            break
        target = n / 2.0
        dx = 2.0 / n

        h_cur = project_h(resample_h(best_h, n), target)
        cur_c5 = compute_c5(h_cur)

        for beta in [60, 120, 250, 500, 1000, 2000]:
            if tleft() < 15:
                break
            h_new = slsqp_smooth(h_cur, n, target, dx, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h_cur = h_new
                cur_c5 = c5_new

        deadline = start_time + tleft() - 15
        h_cur, cur_c5 = subgrad_descent(h_cur, n, target, dx,
                                         max_iters=30000,
                                         deadline=deadline, start_t=start_time)

        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5
            best_n = n

    # Final fine polish
    if tleft() > 10:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        cur = best_h.copy()
        cur_c5 = best_c5
        momentum = np.zeros(n_f)
        mu = 0.82
        center = n_f - 1
        deadline = start_time + tleft() - 2
        for it in range(80000):
            if time.time() > deadline:
                break
            corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
            corr_val = corr_raw * dx_f
            top_idx = np.argsort(corr_val)[::-1][:2]
            vals = corr_val[top_idx]
            temp = 600.0
            weights = np.exp((vals - vals[0]) * temp)
            weights /= weights.sum()
            grad = np.zeros(n_f)
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                if w > 1e-6:
                    grad += w * grad_single_lag(cur, lag, n_f, dx_f)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            momentum = mu * momentum - grad
            mnorm = np.linalg.norm(momentum)
            if mnorm > 0:
                momentum /= mnorm
            step = 0.004
            for _ in range(20):
                h_trial = project_h(cur - step * momentum, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-16:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_h = cur.copy()
                        best_c5 = cur_c5
                    break
                step *= 0.5

    return best_h, best_c5, len(best_h)
