```python
def smooth_max_gradient(h, n, beta, lags=None):
    """Compute gradient of smooth_max(corr) where corr = np.correlate(h, 1-h, 'full')"""
    # corr has length 2n-1, indexed from -(n-1) to (n-1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(beta * (corr - corr.max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr[k] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            for i in range(n - lag):
                grad[i] += weights[k] * (1 - h[i + lag])
                grad[i + lag] += weights[k] * (-h[i])
        else:
            lag_abs = -lag
            # corr[k] = sum_{i=0}^{n-1-lag_abs} h[i+lag_abs] * (1-h[i])
            for i in range(n - lag_abs):
                grad[i + lag_abs] += weights[k] * (1 - h[i])
                grad[i] += weights[k] * (-h[i + lag_abs])
    
    return grad * (2.0 / n)  # multiply by dx
```

    from scipy.optimize import linprog
    # or use cvxpy

    #      sum(h) = target_sum

```python
import numpy as np
from scipy.optimize import minimize, linprog, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_for_lag(h, n, lag):
    """Compute gradient of g_lag(h) = sum_i h[i](1-h[i+lag]) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        mm = n - lag
        if mm > 0:
            g[:mm] += (1.0 - h[lag:])
            g[lag:] -= h[:mm]
    else:
        mm = n + lag
        if mm > 0:
            g[-mm:] += (1.0 - h[:mm])
            g[:mm] -= h[-mm:]
    return g

def all_lag_grads(h, n):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        grads[k] = grad_for_lag(h, n, lag)
    return corr, grads

def smooth_max_obj_grad(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    weights = exp_s / sum_exp
    f = (m + np.log(sum_exp) / beta) * dx
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mm = n - lag
            if mm > 0:
                grad[:mm] += w * (1.0 - x[lag:])
                grad[lag:] -= w * x[:mm]
        else:
            mm = n + lag
            if mm > 0:
                grad[-mm:] += w * (1.0 - x[:mm])
                grad[:mm] -= w * x[-mm:]
    return f, grad * dx

def constraint_eq(x, target_sum):
    return x.sum() - target_sum

def constraint_eq_jac(x, target_sum):
    return np.ones(len(x))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_subproblem(h_current, n, target_sum):
    """Solve the linearized subproblem using linprog"""
    corr = np.correlate(h_current, 1.0 - h_current, mode='full')
    n_lags = len(corr)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_rows = np.zeros((n_lags, n + 1))
    b_vals = np.zeros(n_lags)
    
    for k in range(n_lags):
        lag = k - (n - 1)
        grad_k = grad_for_lag(h_current, n, lag)
        g_k = corr[k]
        rhs = grad_k @ h_current - g_k
        A_rows[k, :n] = grad_k
        A_rows[k, -1] = -1.0
        b_vals[k] = rhs
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * n + [(-1.0, 10.0)]
    
    res = linprog(c, A_ub=A_rows, b_ub=b_vals, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

    else:

    n = len(h_init)

    h_init = project_feasible(h_init, target_sum)
    
    best_h = h_init.copy()

    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: LP outer approximation
    h = best_h.copy()

            break
        h_lp, t_lp = solve_lp_subproblem(h, n, target_sum)
        if h_lp is None:
            break

        c5_lp = compute_c5(h_lp)
        
        if c5_lp < best_c5 - 1e-12:
            h = h_lp.copy()
            best_h = h_lp.copy()
            best_c5 = c5_lp
            stagnant = 0
        else:

            improved = False

                h_test = (1.0 - alpha) * h + alpha * h_lp

                c5_test = compute_c5(h_test)
                if c5_test < best_c5 - 1e-12:
                    h = h_test.copy()
