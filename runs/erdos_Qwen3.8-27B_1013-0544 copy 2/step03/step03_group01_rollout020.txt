∂c_k/∂h[j] = dx * (1-h[j+k-n+1]) (when only the first index is valid)
∂c_k/∂h[j] = -dx * h[j-k+n-1] (when only the second index is valid)

For np.correlate(a, b, mode='full') where a, b have length n:
result[p] = Σ_i a[i] * b[i + p - (n-1)] for i + p - (n-1) in [0, n-1]

So for a=h, b=1-h:
result[p] = Σ_i h[i] * (1 - h[i + p - n + 1]) for i in [0,n-1] and i+p-n+1 in [0,n-1]

Let q = p - n + 1 (the "lag"):
result[q+n-1] = Σ_i h[i] * (1 - h[i+q]) for valid i and i+q

So the lag q = p - n + 1 ranges from -(n-1) to n-1.

For a specific lag q:
c_q = Σ_i h[i] * (1-h[i+q]) * dx (summing over valid i)

∂c_q/∂h[j] = dx * Σ_i [δ_{ij}(1-h[i+q]) - h[i]δ_{j,i+q}]
            = dx * [(1-h[j+q]) - h[j-q]] (when both j+q and j-q are in [0,n-1])
            = dx * [(1-h[j+q])] (when j+q is valid but j-q is not)
            = dx * [-h[j-q]] (when j-q is valid but j+q is not)

This gives us the gradient of each c_q. The gradient of max(c_q) is a convex combination of the gradients of the active c_q values.

I could use this for a projected subgradient method:
1. Compute c_q for all q
2. Find the active set A = {q : c_q is within ε of max}
3. Compute g = (1/|A|) Σ_{q∈A} ∇c_q
4. Take a step: h_new = h - lr * g
5. Project to feasible set
6. Repeat

This is O(n) per step (vs O(n²) for pair search), so it could be much faster.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, active_idx, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    ca = c[active_idx]
    m = ca.max()
    shifted = alpha * (ca - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp(h, n, active_idx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    def f(x):
        return obj_smooth(x, n, active_idx, alpha)
    try:
        res = minimize(f, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

def focal_optimize(h, n, deadline, top_m=3, alphas=None):
    if alphas is None:
        alphas = [1000, 5000, 20000, 100000, 500000, 2000000]
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Multiple focal iterations at this alpha
        for focal_iter in range(3):
            if time.time() > deadline:
                break
            c_vals = compute_all_c(best_h, n)
            top_idx = np.argsort(c_vals)[-top_m:]
            ht = slsqp(best_h, n, top_idx, alpha, 1500, 1e-14)
            c_new = compute_c5(ht, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = ht
    
    return best_h, best_c

def grad_local_search(h, n, deadline, max_iter=50000):
    """Projected subgradient method on active constraints"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.005
    n = len(best_h)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if lr < 1e-7:
            break
        
        c_vals = compute_all_c(best_h, n)
        m = c_vals.max()
        # Active set: within 0.5% of max
        active_mask = c_vals > m - 0.005
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c_vals)])
        
        # Compute gradient: average gradient of active c_q
        # c_q = Σ_i h[i]*(1-h[i+q])*dx where q = idx - (n-1)
        grad = np.zeros(n)
        for idx in active_idx:
            q = idx - (n - 1)  # lag
            for i in range(n):
                # ∂c_q/∂h[i] contribution
                # term 1: (1-h[i+q]) if i+q valid
                # term 2: -h[i-q] if i-q valid (from h[i-q]*(1-h[i]) when i→i-q)
                val = 0.0
                if 0 <= i + q < n:
                    val += (1.0 - best_h[i + q])
                if 0 <= i - q < n:
                    val -= best_h[i - q]
                grad[i] += val
        grad *= dx / len(active_idx)
        
        # Projected gradient step
        h_new = best_h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            lr = min(lr * 1.05, 0.02)
        else:
            lr *= 0.95
    
    return best_h, best_c

def pair_local(h, n, deadline, deltas=None):

    best_h = h.copy()

                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                            best_h[j] = hj
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                best_h[i] -= s * d
                                best_h[j] += s * d
    return best_h, best_c

    for _ in range(3):
