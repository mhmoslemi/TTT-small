        h_sg, c_sg = nesterov_subgrad(h_best, 5000, 0.02, 0.99995, beta_w=300)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
            if time_left() > 50:
                h_new = slsqp_optimize(h_best, n_final / 2.0, 2500, 400)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new, c

    # Phase 5: Final high-beta polish
    if time_left() > 40:
        h_final = slsqp_optimize(h_best, len(h_best) / 2.0, 4000, 400)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
