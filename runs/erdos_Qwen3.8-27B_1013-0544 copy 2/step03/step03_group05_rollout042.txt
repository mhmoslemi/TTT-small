def project_h(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif 0 < lag < n:

        grad[lag:] += -h[:n - lag] * dx
    elif -n < lag < 0:
        l = -lag
        grad[l:] += (1.0 - h[:n - l]) * dx
        grad[:n - l] += -h[l:] * dx
    return grad

def resample_h(h, n_new):
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def smooth_max_obj(x, dx, beta):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta) * dx

def smooth_max_jac_h(x, dx, beta):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -400, 400)
    w = np.exp(xs)
    w /= np.sum(w)
    n = len(x)
    center = n - 1
    grad = np.zeros(n)
    for i, w_i in enumerate(w):
        if w_i < 1e-12:
            continue
        lag = i - center
        grad += w_i * grad_lag(x, lag, n, dx)
    return grad * dx

def slsqp_pass(h, n, dx, beta, maxiter, ftol):
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(dx, beta),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x, target)
    except Exception:
        return h

def nesterov_opt(h, n, dx, max_iters, time_fn, tlimit, best_state,
                 top_k, temp0, step0, mom0):
    target = n / 2.0
    h = h.copy()
    h_ext = h.copy()
    cur_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    center = n - 1
    beta_m = mom0
    stall = 0

    for it in range(max_iters):
        if time_fn() < tlimit:
            break
        k_use = min(top_k + it // 400, 35)
        temp_it = temp0 + 25.0 * it / max(1, max_iters)
        top_idx = np.argsort(corr)[::-1][:k_use]
        vals = corr[top_idx]
        w = np.exp((vals - vals[0]) * temp_it)
        w /= np.sum(w)

        grad = np.zeros(n)
        for i, w_i in zip(top_idx, w):
            lag = int(i) - center
            grad += w_i * grad_lag(h, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_n = grad / gnorm

        step = step0 / (1.0 + 0.0004 * it)
        y = h_ext
        y_t = project_h(y - step * grad_n, target)
        c5_t = compute_c5(y_t)
        if c5_t < cur_c5 - 1e-15:
            h_ext = y_t
            prev_h = h
            h = y_t
            cur_c5 = c5_t
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            stall = 0
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = h.copy()
                best_state[2] = n
        else:
            h_t = project_h(h - step * grad_n, target)
            c5_t2 = compute_c5(h_t)
            if c5_t2 < cur_c5 - 1e-15:
                h = h_t
                cur_c5 = c5_t2
                h_ext = h + beta_m * (h - prev_h) if it > 0 else h.copy()
                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                stall = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = h.copy()
                    best_state[2] = n
            else:
                stall += 1
                if stall > 500:
                    T = 0.004 * np.exp(-stall / 5000.0)
                    h_t = project_h(h + np.random.normal(0, T, n), target)
                    c5_t3 = compute_c5(h_t)
                    if c5_t3 < cur_c5 + 3 * T:
                        h = h_t
                        h_ext = h
                        cur_c5 = c5_t3
                        stall = 0
                    elif stall > 2000:
                        h_ext = h.copy()
                        stall = 0

        beta_m = max(mom0 * (1.0 - 0.00005 * it), 0.4)
    return h, cur_c5, n

def fine_polish(h, n, dx, time_fn, tlimit, best_state):
    target = n / 2.0
    h = h.copy()
    cur_c5 = compute_c5(h)
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    for it in range(50000):
        if time_fn() < tlimit:
            break
        top_idx = np.argsort(corr)[::-1][:4]
        vals = corr[top_idx]
        w = np.exp((vals - vals[0]) * 400.0)
        w /= np.sum(w)
        grad = np.zeros(n)
        for i, w_i in zip(top_idx, w):
            lag = int(i) - center
            grad += w_i * grad_lag(h, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 0.003 / (1.0 + 0.0001 * it)
        improved = False
        for _ in range(15):
            h_t = project_h(h - step * grad / gnorm, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5 - 1e-14:
                h = h_t
                cur_c5 = c5_t
                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = c5_t
                    best_state[1] = h.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved and it % 500 == 0:
            i_rand = np.random.randint(n)
            for d in [0.005, -0.005]:
                h_t = h.copy()
                h_t[i_rand] += d
                h_t = project_h(h_t, target)
                c5_t = compute_c5(h_t)
                if c5_t < cur_c5 - 1e-14:
                    h = h_t
                    cur_c5 = c5_t
                    corr = np.correlate(h, 1.0 - h, mode="full") * dx
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = h.copy()
                        best_state[2] = n
    return h, cur_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(250, 0.5)

    best_c5 = compute_c5(h_parent)
    best_state = [best_c5, h_parent.copy(), len(h_parent)]

    # Phase 1: SLSQP at parent resolution with increasing beta
    n1 = len(h_parent)
    dx1 = 2.0 / n1
    h1 = project_h(h_parent.copy(), n1 / 2.0)
    for beta in [200, 500, 1200]:
        if time_left() < 30:
            break
        h1 = slsqp_pass(h1, n1, dx1, beta, 2000, 1e-14)
        c5_check = compute_c5(h1)
        if c5_check < best_state[0]:
            best_state[0] = c5_check
            best_state[1] = h1.copy()
            best_state[2] = n1

    # Phase 2: Nesterov subgradient at parent resolution
    if time_left() > 100:
        h1, _, _ = nesterov_opt(h1, n1, dx1, 80000, time_left, 80, best_state,
                                top_k=20, temp0=60.0, step0=0.035, mom0=0.85)

    # Phase 3: Try higher resolution (upsample and optimize)
    for n2 in [n1 + 50, n1 + 100, 400]:
        if time_left() < 150:
            break
        dx2 = 2.0 / n2
        h2 = project_h(resample_h(best_state[1], n2), n2 / 2.0)
        h2 = slsqp_pass(h2, n2, dx2, 400, 1500, 1e-14)
        c5_2 = compute_c5(h2)
        if c5_2 < best_state[0]:
            best_state[0] = c5_2
            best_state[1] = h2.copy()
            best_state[2] = n2
        if time_left() > 100:
            h2, _, _ = nesterov_opt(h2, n2, dx2, 40000, time_left, 60, best_state,
                                     top_k=15, temp0=50.0, step0=0.03, mom0=0.82)

    # Phase 4: Also try lower resolution
    n3 = n1 - 50
    if n3 >= 100 and time_left() > 80:
        dx3 = 2.0 / n3
        h3 = project_h(resample_h(best_state[1], n3), n3 / 2.0)
        h3 = slsqp_pass(h3, n3, dx3, 600, 1500, 1e-14)
        h3, _, _ = nesterov_opt(h3, n3, dx3, 30000, time_left, 40, best_state,
                                 top_k=12, temp0=70.0, step0=0.04, mom0=0.85)

    # Phase 5: SLSQP polish on best
    n_f = len(best_state[1])
    dx_f = 2.0 / n_f
    for beta in [1500, 4000]:
        if time_left() < 15:
            break
        h_f = slsqp_pass(best_state[1], n_f, dx_f, beta, 3000, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_state[0]:
            best_state[0] = c5_f
            best_state[1] = h_f
            best_state[2] = n_f

    # Phase 6: Ultra-fine final polish
    if time_left() > 20:
        n_p = len(best_state[1])
        dx_p = 2.0 / n_p
        fine_polish(best_state[1], n_p, dx_p, time_left, 5, best_state)

    return best_state[1], best_state[0], len(best_state[1])
