    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad

def slsqp_optimize(h_init, beta, maxiter, top_k=None):
    n = len(h_init)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_init, target)

def interp_resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def targeted_perturb(h, rng, strength):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_lag = np.argmax(corr)
    s = max_lag - n + 1
    score = np.zeros(n)
    i_lo = max(0, s)
    i_hi = min(n, n + s)
    if i_hi > i_lo:
        score[i_lo:i_hi] += h[i_lo:i_hi] * (1.0 - h[i_lo - s:i_hi - s])
    i_lo2 = max(0, -s)
    i_hi2 = min(n, n - s)
    if i_hi2 > i_lo2:
        score[i_lo2:i_hi2] += h[i_lo2 + s:i_hi2 + s] * (1.0 - h[i_lo2:i_hi2])
    mask = rng.random(n) < 0.4
    pert = rng.randn(n) * strength * score * mask
    return project_feasible(h + pert, target)

def periodic_init(n, freq, amp, rng):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
    h += rng.randn(n) * 0.02
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    def time_left():
        return deadline - time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = periodic_init(200, 7, 0.25, rng)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()
    n_best = len(h_best)

    # Phase 1: Quick SLSQP polish from initial
    for beta in [80, 200, 500]:
        if time_left() < 35:
            break
        h_cur = slsqp_optimize(h_cur, beta, 350, top_k=20)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            h_best = h_cur.copy()

    # Phase 2: Main search - simulated annealing style
    # Alternating: random perturb, targeted perturb, symmetry, diverse restart
    phase2_deadline = t0 + 0.55 * budget_s
    iter_count = 0
    while time.time() < phase2_deadline and time_left() > 40:
        iter_count += 1
        mode = iter_count % 5
        frac = iter_count / 100.0
        base_noise = 0.035 * max(0.3, 1.0 - frac)

        if mode == 0:
            h_pert = project_feasible(h_best + rng.randn(n_best) * base_noise, n_best / 2.0)
        elif mode == 1:
            h_pert = targeted_perturb(h_best, rng, base_noise * 1.5)
        elif mode == 2:
            h_pert = project_feasible(1.0 - h_best + rng.randn(n_best) * base_noise * 0.6, n_best / 2.0)
        elif mode == 3:
            h_pert = project_feasible(h_best + rng.randn(n_best) * base_noise * 0.7, n_best / 2.0)
        else:
            # Diverse restart with periodic structure
            freq = rng.randint(3, 15)
            h_pert = periodic_init(n_best, freq, rng.uniform(0.15, 0.35), rng)

        h_local = slsqp_optimize(h_pert, 350, 250, top_k=12)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 3: Multi-resolution exploration
    phase3_deadline = t0 + 0.80 * budget_s
    for n_try in [120, 180, 250, 300, 350]:
        if time.time() > phase3_deadline or time_left() < 50:
            break
        h_coarse = interp_resample(h_best, n_try)
        for beta in [100, 400]:
            if time_left() < 40:
                break
            h_coarse = slsqp_optimize(h_coarse, beta, 250, top_k=15)
        c = compute_c5(h_coarse)
        if c < best_c5:
            best_c5 = c
            h_best = h_coarse.copy()
            n_best = len(h_best)
        # Try symmetry
        h_sym = project_feasible(1.0 - h_coarse, n_try / 2.0)
        h_sym = slsqp_optimize(h_sym, 400, 200, top_k=15)
        c_sym = compute_c5(h_sym)
        if c_sym < best_c5:
            best_c5 = c_sym
            h_best = h_sym.copy()
            n_best = len(h_best)

    # Phase 4: Fine basin hopping at best resolution
    phase4_deadline = t0 + 0.93 * budget_s
    for hop in range(15):
        if time.time() > phase4_deadline or time_left() < 25:
            break
        noise = 0.012 * (0.8 ** hop)
        if hop % 3 == 0:
            h_pert = targeted_perturb(h_best, rng, noise * 2)
        elif hop % 3 == 1:
            h_pert = project_feasible(h_best + rng.randn(n_best) * noise, n_best / 2.0)
        else:
            h_pert = project_feasible(1.0 - h_best + rng.randn(n_best) * noise, n_best / 2.0)
        h_local = slsqp_optimize(h_pert, 600, 300, top_k=8)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()

    # Phase 5: Final high-beta polish
    if time_left() > 35:
        h_fin = slsqp_optimize(h_best, 1000, 500, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 6: Ultra-fine polish
    if time_left() > 25:
        h_fin2 = slsqp_optimize(h_best, 1500, 400, top_k=5)
        c_fin2 = compute_c5(h_fin2)
        if c_fin2 < best_c5:
            best_c5 = c_fin2
            h_best = h_fin2

    # Final feasibility check
    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, n_best
