import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def top_k_subgrad(h, corr, n, k_top):
    top_idx = np.argsort(corr)[-k_top:]
    w = np.exp(10.0 * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        grad += w[j] * subgrad_for_lag(h, lag, n)
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def compute_a_for_j(h, j, n):
    n_full = 2 * n - 1
    a = np.zeros(n_full)
    g = 1.0 - h
    ks = np.arange(n_full)
    idx1 = j + ks - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    if np.any(valid1):
        a[valid1] += g[idx1[valid1]]
    idx2 = j - ks + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    if np.any(valid2):
        a[valid2] -= h[idx2[valid2]]
    return a

def coord_optimal(h, j, n, corr):
    h_j = h[j]
    a = compute_a_for_j(h, j, n)
    lo = 0.0
    hi = 1.0
    offsets = corr - a * h_j
    def f_max(t):
        return np.max(a * (t - h_j) + offsets + a * h_j)
    def f_max2(t):
        return np.max(a * (t - h_j) + corr)
    def sg_range(t):
        vals = a * (t - h_j) + corr
        m = np.max(vals)
        active = vals >= m - 1e-12
        if not np.any(active):
            return 0.0, 0.0
        return np.min(a[active]), np.max(a[active])
    s_lo = sg_range(lo)
    if s_lo[0] >= 0:
        return lo
    s_hi = sg_range(hi)
    if s_hi[1] <= 0:
        return hi
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s_mid = sg_range(mid)
        if s_mid[0] >= 0:
            hi = mid
        elif s_mid[1] <= 0:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def coord_descent_sweep(h, n, target_sum, max_sweeps=20):
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            new_t = coord_optimal(h, j, n, np.correlate(h, 1.0 - h, mode='full'))
            if abs(new_t - old) > 1e-10:
                h[j] = new_t
                improved = True
        h = project_feasible(h, target_sum)
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent (exploits affine structure)
    if tl() > 200:
        h = best_h.copy()
        h = coord_descent_sweep(h, n, target_sum, max_sweeps=30)
        c = compute_c5(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c

    # Phase 2: Subgradient with top-k lags + momentum
    if tl() > 200:
        h = best_h.copy()
        velocity = np.zeros(n)
        beta_m = 0.85
        for it in range(50000):
            if tl() < 150:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_top = min(5, n)
            grad = top_k_subgrad(h, corr, n, k_top)
            lr = 0.006 / (1.0 + 0.0002 * it)
            velocity = beta_m * velocity + (1 - beta_m) * (-lr * grad)
            h_new = project_feasible(h + velocity, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5 + 0.003:
                h = h_new
                if c_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c_new
            else:
                velocity *= 0.4
                lr *= 0.5
            if it % 1000 == 999:
                h_cd = coord_descent_sweep(h.copy(), n, target_sum, max_sweeps=5)
                c_cd = compute_c5(h_cd)
                if c_cd < best_c5:
                    best_h = h_cd.copy()
                    best_c5 = c_cd
                    h = h_cd.copy()

    # Phase 3: SLSQP continuation with multiple betas
    bounds = [(0.0, 1.0)] * n
    def con_fun(h, ts):
        return h.sum() - ts
    def con_jac(h, ts):
        return np.ones(len(h))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]

    if tl() > 150:
        for beta in [50, 200, 1000, 5000, 20000]:
            if tl() < 60:
                break
            for trial in range(4):
                if tl() < 30:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * (0.01 / trial)
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 8000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 4: More coordinate descent after SLSQP
    if tl() > 100:
        h = best_h.copy()
        h = coord_descent_sweep(h, n, target_sum, max_sweeps=40)
        c = compute_c5(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c

    # Phase 5: Multi-resolution exploration
    if tl() > 100:
        res_candidates = [n, n // 2, n * 2, n - 10, n + 10, n + 30, n - 30, 105, 140, 175, 250, 300]
        res_candidates = sorted(set([r for r in res_candidates if 30 <= r <= 600 and r != n]))
        for n_try in res_candidates[:8]:
            if tl() < 50:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            # Coordinate descent at new resolution
            h_t = coord_descent_sweep(h_t, n_try, ts_t, max_sweeps=15)
            bnds_t = [(0.0, 1.0)] * n_try
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
            for beta in [500, 5000]:
                if tl() < 25:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 5000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        # Downsample or keep at this resolution
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        cons = cons_t
                except Exception:
                    pass

    # Phase 6: Final coordinate descent + SLSQP polish
    if tl() > 30:
        h = best_h.copy()
        h = coord_descent_sweep(h, n, target_sum, max_sweeps=30)
        c = compute_c5(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c

    if tl() > 15:
        for beta in [50000, 200000]:
            if tl() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Final micro subgradient
    if tl() > 10:
        h = best_h.copy()
        for it in range(3000):
            if tl() < 4:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (n - 1)
            grad = subgrad_for_lag(h, lag, n)
            lr = 0.0008 / (1.0 + 0.005 * it)
            h_new = project_feasible(h - lr * grad, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
            elif it > 500:
                break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
