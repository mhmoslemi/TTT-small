def subgradient_descent(h, target, max_iter, time_limit, t0):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    momentum = np.zeros(n)
    beta_m = 0.7
    step = 0.5
    best_c5 = c5_cur
    best_h = h_cur.copy()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad, val = active_subgradient(h_cur)
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        # Nesterov momentum
        momentum = beta_m * momentum - grad / gn
        # Backtracking line search
        for ls in range(20):
            h_trial = project_h(h_cur + step * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-14:
                h_cur = h_trial
                c5_cur = c5_trial
                step = min(step * 1.1, 2.0)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                break
            else:
                step *= 0.5
        else:
            # No improvement, reduce momentum
            momentum *= 0.5
            step = max(step, 0.01)
    
    return best_h, best_c5

def slsqp_refine(h, target, beta, max_iter):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    
    def constr(x):
        return np.sum(x) - target
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': constr, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            nf = len(h)
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_subgradient(h, k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for ki in top_idx:
        a = max(0, -ki)
        b = min(n, n - ki)
        if b <= a:
            continue

        idx1 = idx + ki
        grad[idx1] += (1.0 - h[idx])
        grad[idx] -= h[idx1]
    return grad, float(np.max(corr) * dx)

def smooth_max(x, beta):
