import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

        return h_old.copy()
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

    prev_h = None

            return np.ones(len(x))

            if sy > 1e-10:
                step = np.clip(float(s @ s) / sy, 0.005, 0.3)
            else:
                step = 0.06
        else:
            step = 0.08

        prev_grad = grad.copy()
        prev_h = cur.copy()

        velocity = 0.85 * velocity - step * grad_n
        vn = np.linalg.norm(velocity)
        if vn > 0.15:
            velocity *= 0.15 / vn

        improved = False
        for _ in range(25):
            h_trial = project_h(cur + velocity, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                no_improve = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = c5_trial
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            velocity *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                T = 0.004 * max(0.05, 1.0 - it / 8000.0)
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 6:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = c5_trial
                        best_state[1] = cur.copy()
                        best_state[2] = n
                no_improve = 0
                velocity = np.zeros(n)
                prev_grad = None
                prev_h = None
                max_no_improve = max(80, max_no_improve - 20)

    return cur, cur_c5


def slsqp_opt(h0, n, target, dx, deadline, betas):
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    bounds = Bounds(np.zeros(n), np.ones(n))

    for beta in betas:
        if time.time() > deadline:
            break

        def obj_func(x, b=beta):
            return smooth_max_val(np.correlate(x, 1.0 - x, mode="full") * dx, b)

        def obj_grad(x, b=beta):
            return compute_smooth_grad(x, dx, b, n)

        def con_fun(x):
            return np.sum(x) - target

        def con_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_func, cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return cur, cur_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(250, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    best_state = [best_c5, best_h.copy(), best_n]

    n1 = len(h_parent)
    target1 = n1 / 2.0
    dx1 = 2.0 / n1

    # Phase 1: Subgradient at parent resolution
    deadline1 = start_time + budget_s * 0.45
    h1, c5_1 = subgradient_opt(h_parent, n1, target1, dx1, deadline1, best_state)

    # Phase 2: SLSQP polish at parent resolution
    deadline2 = start_time + budget_s * 0.65
    h2, c5_2 = slsqp_opt(h1, n1, target1, dx1, deadline2, [150, 400, 1000, 2500])
    if c5_2 < best_state[0]:
        best_state[0] = c5_2
        best_state[1] = h2.copy()
        best_state[2] = n1

    # Phase 3: Try slightly higher resolution
    n3 = n1 + 50
    if n3 <= 400 and time.time() < start_time + budget_s * 0.82:
        deadline3 = start_time + budget_s * 0.82
        target3 = n3 / 2.0
        dx3 = 2.0 / n3
        h3 = project_h(resample(h2, n3), target3)
        h3, c5_3 = subgradient_opt(h3, n3, target3, dx3, deadline3, best_state)
        deadline3b = start_time + budget_s * 0.90
        h3b, c5_3b = slsqp_opt(h3, n3, target3, dx3, deadline3b, [200, 600])
        if c5_3b < best_state[0]:
            best_state[0] = c5_3b
            best_state[1] = h3b.copy()
            best_state[2] = n3

    # Phase 4: Also try a random perturbation restart at parent resolution
    if time.time() < start_time + budget_s * 0.95:
        deadline4 = start_time + budget_s * 0.95
        n4 = best_state[2]
        target4 = n4 / 2.0
        dx4 = 2.0 / n4
        h4 = project_h(best_state[1] + np.random.normal(0, 0.02, n4), target4)
        h4, c5_4 = subgradient_opt(h4, n4, target4, dx4, deadline4, best_state)

    # Final: ensure feasibility
    final_h = project_h(best_state[1], best_state[2] / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
