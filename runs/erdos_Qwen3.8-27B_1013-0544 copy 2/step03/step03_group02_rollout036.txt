import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(400):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def sum_eq_fun(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def smsm_obj(x, beta, n):
    c = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def as_obj_fun(xt, n):
    return xt[n]

def as_obj_jac(xt, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def as_ineq_fun(xt, lag, n):
    h = xt[:n]
    m = lag - (n - 1)
    if m >= 0:
        if m >= n:
            return 0.0
        return np.sum(h[:n-m] * (1.0 - h[m:])) - xt[n]
    else:
        la = -m
        if la >= n:
            return 0.0
        return np.sum(h[la:] * (1.0 - h[:n-la])) - xt[n]

def as_ineq_jac(xt, lag, n):
    h = xt[:n]
    m = lag - (n - 1)
    g = np.zeros(n + 1)
    g[n] = -1.0
    if m >= 0:
        if m < n:
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
    else:
        la = -m
        if la < n:
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    return g

def as_sum_eq_fun(xt, n, target):
    return np.sum(xt[:n]) - target

def as_sum_eq_jac(xt, n, target):
    g = np.zeros(n + 1)

def active_set_solve(h0, n, target, lags, t0, budget_s, maxiter=3000):
    if budget_s - (time.time() - t0) < 5:
        return h0
    cur_corr = get_corr(h0)
    t0_val = float(np.max(cur_corr)) + 0.01
    x0 = np.concatenate([h0, [t0_val]])
    bounds = Bounds(np.concatenate([np.zeros(n), [0.0]]),
                    np.concatenate([np.ones(n), [2.0]]))
    constraints = [{'type': 'eq', 'fun': as_sum_eq_fun, 'jac': as_sum_eq_jac,
                   'args': (n, target)}]
    for lag in lags:
        constraints.append({'type': 'ineq', 'fun': as_ineq_fun, 'jac': as_ineq_jac,
                           'args': (int(lag), n)})
    try:
        r = minimize(as_obj_fun, x0, jac=as_obj_jac, args=(n,), method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x[n] < 1.5:
            return project_h(r.x[:n], target)
    except Exception:
        pass
    return h0

def subgrad_descend(h, target, n, n_iter, step, t0, budget):
    mom = np.zeros(n)
    best_local = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[:n-lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n-lag]
            else:
                la = -lag
                grad[la:] += (1.0 - h[:n-la])
                grad[:n-la] -= h[la:]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        g = grad / gn
        mom = 0.88 * mom - step * g
        trial = project_h(h + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            h = trial
            best_local = c_try
            stall = 0
        else:
            stall += 1
            mom *= 0.3
            trial = project_h(h + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                h = trial
                best_local = c_try
                stall = 0
            else:
                mom *= 0.5
        if stall > 600:
            break
    return h, best_local

def coord_polish(h, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(h)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            for d in deltas:
                trial = h.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    h = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return h, best_c5

def slsqp_smooth(h0, n, target, beta, t0, budget_s, maxiter=2000):
    if budget_s - (time.time() - t0) < 5:
        return h0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (target,)}]
    try:

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except Exception:
        return h0

        cur = best_h.copy()
        corr = get_corr(cur)
        top_lags = np.argsort(corr)[::-1][:8]
        for iteration in range(15):
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur = active_set_solve(cur, n, target, top_lags, t0, budget_s, 5000)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n
            corr = get_corr(cur)
            new_max = int(np.argmax(corr))
            if new_max not in top_lags:
                top_lags = np.sort(np.append(top_lags, new_max))
                if len(top_lags) > 15:
                    idx_sorted = np.argsort(corr[top_lags])[::-1][:12]
                    top_lags = top_lags[idx_sorted]

    # Phase 5: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = coord_polish(best_h, target, n, 60, t0, budget_s)
        best_n = n

    # Phase 6: Multi-resolution search
    resolutions_to_try = []
    for r in [160, 200, 240, 280, 320, 400, 480, 560]:
        if r != n and r <= 640:
            resolutions_to_try.append(r)
    resolutions_to_try.sort(key=lambda r: -abs(r - n))

    for n_pts in resolutions_to_try:
        remaining = budget_s - (time.time() - t0)
        if remaining < 40:
            break
        target_r = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target_r)
        cur_c5 = compute_c5(cur)

        # Quick SLSQP
        for beta in [200, 500]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            cur = slsqp_smooth(cur, n_pts, target_r, beta, t0, budget_s, 1500)
            cur_c5 = compute_c5(cur)

        # Active set
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            corr = get_corr(cur)
            top_lags = np.argsort(corr)[::-1][:8]
            for iteration in range(10):
                remaining = budget_s - (time.time() - t0)
                if remaining < 10:
                    break
                cur = active_set_solve(cur, n_pts, target_r, top_lags, t0, budget_s, 2000)
                cur_c5 = compute_c5(cur)
                corr = get_corr(cur)
                new_max = int(np.argmax(corr))
                if new_max not in top_lags:
                    top_lags = np.sort(np.append(top_lags, new_max))
                    if len(top_lags) > 12:
                        idx_s = np.argsort(corr[top_lags])[::-1][:10]
                        top_lags = top_lags[idx_s]

        # Subgradient
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur, cur_c5 = subgrad_descend(cur, target_r, n_pts, 5000, 0.015, t0, budget_s)

        # Coord polish
        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            cur, cur_c5 = coord_polish(cur, target_r, n_pts, 30, t0, budget_s)

        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Phase 7: Final high-precision polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        nf = best_n
        tf = nf / 2.0
        for beta in [2000, 5000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            best_h = slsqp_smooth(best_h, nf, tf, beta, t0, budget_s, 5000)
            best_c5 = compute_c5(best_h)

        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            corr = get_corr(best_h)
            top_lags = np.argsort(corr)[::-1][:10]
            for iteration in range(10):
                remaining = budget_s - (time.time() - t0)
                if remaining < 8:
                    break
                best_h = active_set_solve(best_h, nf, tf, top_lags, t0, budget_s, 5000)
                best_c5 = compute_c5(best_h)
                corr = get_corr(best_h)
                nm = int(np.argmax(corr))
                if nm not in top_lags:
                    top_lags = np.sort(np.append(top_lags, nm))
                    if len(top_lags) > 12:
                        idx_s = np.argsort(corr[top_lags])[::-1][:10]
                        top_lags = top_lags[idx_s]

        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            best_h, best_c5 = subgrad_descend(best_h, tf, nf, 8000, 0.005, t0, budget_s)

        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = coord_polish(best_h, tf, nf, 80, t0, budget_s)

    return best_h, best_c5, best_n
