def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < 0:
        m = 0
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

            g[la:la+m] += (1.0 - h[:m])
            g[:m] -= h[la:la+m]
    return g

def sm_obj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    s = np.clip(beta * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / beta

def con_fun(x, ts):
    return x.sum() - ts

def con_jac(x, ts):
    return np.ones(len(x))

def nesterov_phase(h, n, dx, ts, t0, budget_s, remaining_s):
    best = compute_c5(h)
    v = np.zeros(n)
    mom = 0.85
    lr = 0.005
    stuck = 0
    it = 0
    while (budget_s - (time.time() - t0)) > remaining_s and it < 80000:
        it += 1
        if it % 200 == 0:
            mom = min(0.97, 0.85 + 0.00004 * it)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        tk = np.argsort(c)[-4:]
        w = np.exp(15.0 * (c[tk] - c[tk].max()))
        w /= w.sum()
        g = np.zeros(n)
        for j, p in enumerate(tk):
            g += w[j] * sg_lag(h, p - (n - 1), n)
        v = mom * v + (1.0 - mom) * (-lr * g)
        hn = project_feasible(h + v, ts)
        cn = compute_c5(hn)
        if cn < best:
            best = cn
            h = hn
            stuck = 0
            lr = min(lr * 1.03, 0.012)
        elif cn < best + 0.0003:
            h = hn
            stuck += 1
            if stuck > 500:
                lr *= 0.85
                stuck = 0
        else:
            v *= 0.3
            lr *= 0.6
            stuck += 1
        lr = max(lr, 3e-4)
    return h, best

def slsqp_phase(h, n, dx, ts, t0, budget_s, remaining_s, betas):
    best = compute_c5(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    for beta in betas:
        if (budget_s - (time.time() - t0)) < remaining_s:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
        for trial in range(3):
            if (budget_s - (time.time() - t0)) < remaining_s // 2:
                break
            x0 = h.copy()
            if trial > 0:
                x0 = x0 + np.random.randn(n) * (0.006 / trial)
                x0 = project_feasible(x0, ts)
            try:
                r = minimize(sm_obj, x0, args=(dx, beta), method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-16})
                hn = project_feasible(r.x, ts)
                c = compute_c5(hn)
                if c < best:
                    h, best = hn, c
            except Exception:
                pass
    return h, best

def _try_resolution(h_best, best_c5, n_best, t0, budget_s, n_try):
    if n_try < 60 or n_try > 800 or n_try == n_best:
        return None, best_c5
    dx_t = 2.0 / n_try
    ts_t = n_try / 2.0
    x_src = np.linspace(0, 2, len(h_best), endpoint=False)
    x_new = np.linspace(0, 2, n_try, endpoint=False)
    h_t = np.interp(x_new, x_src, h_best)
    h_t = project_feasible(h_t, ts_t)
    bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
    cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
    best_t = compute_c5(h_t)
    for beta in [300, 3000]:
        if (budget_s - (time.time() - t0)) < 20:
            break
        try:
            r = minimize(sm_obj, h_t, args=(dx_t, beta), method='SLSQP',
                         bounds=bnds_t, constraints=cons_t,
                         options={'maxiter': 3000, 'ftol': 1e-15})
            hn = project_feasible(r.x, ts_t)
            c = compute_c5(hn)
            if c < best_t:
                h_t, best_t = hn, c
        except Exception:
            pass
    if best_t < best_c5:
        return h_t, best_t
    return None, best_c5

def polish_phase(h, n, ts, t0, budget_s):
    best = compute_c5(h)
    for it in range(30000):
        if (budget_s - (time.time() - t0)) < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        p_w = np.argmax(corr)
        lag = p_w - (n - 1)
        g = sg_lag(h, lag, n)
        improved = False
        for lr_t in [0.0008, 0.0004, 0.0002, 0.0001, 5e-5]:
            hn = project_feasible(h - lr_t * g, ts)
            cn = compute_c5(hn)
            if cn < best:
                h, best = hn, cn
                improved = True
                break
        if not improved and it > 8000:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    ts = n / 2.0
    best_h = project_feasible(h_init.copy(), ts)
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Nesterov subgradient
    if rem() > 300:
        best_h, best_c5 = nesterov_phase(best_h, n, dx, ts, t0, budget_s, 250)

    # Phase 2: SLSQP refinement
    if rem() > 100:
        best_h, best_c5 = slsqp_phase(best_h, n, dx, ts, t0, budget_s, 60,
                                      [50, 200, 1000, 5000, 20000])

    # Phase 3: After SLSQP, another short Nesterov burst
    if rem() > 120:
        best_h, best_c5 = nesterov_phase(best_h, n, dx, ts, t0, budget_s, 100)

    # Phase 4: Higher beta SLSQP
    if rem() > 60:
        best_h, best_c5 = slsqp_phase(best_h, n, dx, ts, t0, budget_s, 40,
                                      [50000, 200000])

    # Phase 5: Resolution exploration
    if rem() > 80:
        n_current = len(best_h)
        candidates = []
        for off in [-20, -10, 10, 20, 30, 50, 70, 100]:
            candidates.append(n_current + off)
        for mult in [1.5, 2]:
            candidates.append(int(n_current * mult))
        candidates = sorted(set([c for c in candidates if 60 <= c <= 800]))
        for n_try in candidates[:6]:
            if rem() < 25:
                break
            h_new, c_new = _try_resolution(best_h, best_c5, n_current, t0, budget_s, n_try)
            if h_new is not None:
                best_h, best_c5 = h_new, c_new
                n_current = n_try
        # Refill best_h variable to use latest
        n_final = len(best_h)
        dx_final = 2.0 / n_final
        ts_final = n_final / 2.0

    # Phase 6: Final polish
    if rem() > 10:
        n_final = len(best_h)
        dx_final = 2.0 / n_final
        ts_final = n_final / 2.0
        best_h, best_c5 = polish_phase(best_h, n_final, ts_final, t0, budget_s)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
