            best_h = h_new.copy()

            h = h_new
            break
        h = h_new

import cvxpy as cp

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def build_linearized(h_cur, s, n):
    coeff = np.zeros(n)
    if s == 0:
        coeff[:] = -2.0 * h_cur
        const = 0.5 * n + np.dot(h_cur, h_cur)
    elif s > 0:
        idx1 = np.arange(s, n)
        coeff[idx1] += 1.0
        idx2 = np.arange(0, n - s)
        coeff[idx2] -= h_cur[idx2 + s]
        idx3 = np.arange(s, n)
        coeff[idx3] -= h_cur[idx3 - s]
        const = np.dot(h_cur[s:], h_cur[:n - s])
    else:
        t = -s
        idx1 = np.arange(0, n - t)
        coeff[idx1] += 1.0
        idx2 = np.arange(0, n - t)
        coeff[idx2] -= h_cur[idx2 + t]
        idx3 = np.arange(t, n)
        coeff[idx3] -= h_cur[idx3 - t]
        const = np.dot(h_cur[:n - t], h_cur[t:])
    return const, coeff

    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == 0.5 * n]
    for s in active_lags:
        c, coeff = build_linearized(h_cur, s, n)
        constraints.append(coeff @ h_var + c <= t / dx)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.GLOP, cp.CBC, cp.PDLP]:
        try:

            if h_var.value is not None:

        except Exception:

def slp_optimize(h_init, n, top_m, max_iter=200, tol=1e-8):
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h)
    prev_c = best_c
    for it in range(max_iter):
        corr = compute_corr(h)
        m = np.max(corr)
        top_idx = np.argsort(corr)[-top_m:]
        active_lags = (top_idx - (n - 1)).tolist()
        res = solve_lp(h, active_lags, n, dx)

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50, 50))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        elif s < 0:
            t = -s
            ms = np.arange(0, n - t)
            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):

    target = 0.5 * n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smoothmax_val_grad(x, beta)[0], h,
                       jac=lambda x: smoothmax_val_grad(x, beta)[1],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    from scipy.interpolate import interp1d
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, 0.5 * n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    elif 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_best = np.array(globals()['initial_h_values'], dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    n = len(h_best)
    best_c5 = compute_c5(h_best)

    # Phase 1: SLP with increasing active set size
    for top_m in [3, 5, 8, 12, 20, 30, 50]:
        if tl() < 60:
            break
        h_new, c_new = slp_optimize(h_best, n, top_m, max_iter=150)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: SLSQP polish
    for beta in [100, 300, 800]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: More SLP with larger active set
    for top_m in [40, 60, 80, 100, 150]:
        if tl() < 50:
            break
        h_new, c_new = slp_optimize(h_best, n, top_m, max_iter=100)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 4: Multi-restart with perturbation
    for trial in range(6):
        if tl() < 50:
            break
        scale = 0.005 + 0.003 * trial
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, 0.5 * n)
        h_new, c_new = slp_optimize(h_pert, n, 30, max_iter=80)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 5: SLSQP high-beta polish
    for beta in [500, 1500, 3000]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=400, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 6: Try lower resolution then upsample
    if tl() > 200:
        h_low = interp_to_n(h_best, 200)
        c_low = compute_c5(h_low)
        for top_m in [10, 30, 60]:
            if tl() < 100:
                break
            h_new, c_new = slp_optimize(h_low, 200, top_m, max_iter=100)
            if c_new < c_low:
                h_low, c_low = h_new, c_new
        h_up = interp_to_n(h_low, n)
        for top_m in [20, 50]:
            if tl() < 80:
                break
            h_new, c_new = slp_optimize(h_up, n, top_m, max_iter=100)
            if c_new < best_c5:
                h_best, best_c5 = h_new, c_new

    # Phase 7: Final SLP with very large active set
    if tl() > 60:
        h_new, c_new = slp_optimize(h_best, n, 200, max_iter=100)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 8: Final polish
    if tl() > 30:
        h_new = slsqp_refine(h_best, maxiter=300, beta=2000)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    h_best = project_feasible(h_best, 0.5 * len(h_best))
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
