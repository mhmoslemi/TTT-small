    """Project v onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    n = len(v)
    def project_with_lambda(lam):
        return np.clip(v - lam, 0.0, 1.0)

    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):

        proj = project_with_lambda(mid)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:

def corr_gradient(h, lag):
    """Gradient of corr at given lag w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # Term 1: h[m] * (1-h[m+lag]) -> derivative (1-h[m+lag]) for m where m+lag < n
    if lag >= 0:
        grad[:n-lag] += (1.0 - h[lag:])  # m from 0 to n-1-lag, i2 = m+lag
    else:
        grad[-lag:] += (1.0 - h[:n+lag])  # m from -lag to n-1, i2 = m+lag
    # Term 2: h[m-lag] * (1-h[m]) -> derivative -h[m-lag] for m where m-lag >= 0
    if lag >= 0:

        corr = np.correlate(h, 1.0 - h, mode="full")

        top_idx = np.argsort(corr)[::-1][:top_k]

        grad = np.zeros(n)

        # Actually standard Nesterov: y_{k+1} = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.mean(np.exp(s))) / beta) * dx

def _sm_con(x, n):
    return np.sum(x) - n / 2.0

def _sm_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, betas, maxiter=3000, ftol=1e-14):
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    for beta in betas:
        bounds = Bounds(np.zeros(n), np.ones(n))
        constr = [{'type': 'eq', 'fun': _sm_con, 'jac': _sm_jac, 'args': (n,)}]
        try:
            res = minimize(_sm_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=constr, args=(beta, dx),
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_h(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def coord_cd(h, n, t0, time_limit, max_rounds=200):
    target = n / 2.0
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    cur_max = np.max(corr)
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit - 5:
            break
        improved = False
        order = np.random.permutation(n)
        for m in order:
            if time.time() - t0 > time_limit - 5:
                break
            old = h_cur[m]
            # Try perturbations
            deltas = [0.05, -0.05, 0.1, -0.1, 1.0 - old, -old]
            best_d = 0.0
            best_m = cur_max
            for d_val in deltas:
                new_val = old + d_val
                if new_val < -1e-9 or new_val > 1 + 1e-9:
                    continue
                new_val = np.clip(new_val, 0, 1)
                if abs(new_val - old) < 1e-9:
                    continue
                h_cur[m] = new_val
                new_corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                new_max = np.max(new_corr)
                if new_max < best_m - 1e-11:
                    best_m = new_max
                    best_d = new_val - old
            if best_d != 0.0:
                h_cur[m] += best_d
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                cur_max = np.max(corr)
                improved = True
            else:
                h_cur[m] = old
        if not improved:
            break
    h_cur = project_h(h_cur, target)
    return h_cur

def binary_anneal(h_cont, n, t0, time_limit, n_iters=5000):
    """Simulated annealing on binary solutions derived from h_cont"""
    target = n // 2  # number of ones
    # Start from thresholding
    h_bin = (h_cont >= 0.5).astype(float)
    # Adjust count
    while h_bin.sum() > target:
        idx = np.random.choice(np.where(h_bin == 1)[0])
        h_bin[idx] = 0
    while h_bin.sum() < target:
        idx = np.random.choice(np.where(h_bin == 0)[0])
        h_bin[idx] = 1
    
    dx = 2.0 / n
    cur_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    T = 0.1  # initial temperature
    for it in range(n_iters):
        if time.time() - t0 > time_limit - 5:
            break
        # Flip: pick a 1 and a 0, swap them
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = h_bin.copy()
                best_c5 = cur_c5
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        T *= 0.999
    
    return best_h, best_c5

def adam_optimize(h_init, n, t0, time_limit, lr=0.01, beta1=0.9, beta2=0.999):

    m = np.zeros(n)  # first moment

    t = 0

        t += 1

        beta = 100.0 + t * 0.1
        corr = np.correlate(h, 1.0 - h, mode="full")

        s = np.clip(beta * (corr - cm), -500, 500)

        weights = np.exp(beta * (corr - cm)) / np.sum(np.exp(beta * (corr - cm)))

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.mean(np.exp(s))) / beta) * dx

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, betas, maxiter=3000, ftol=1e-14):
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_c5 = compute_c5(h_cur)
    for beta in betas:
        bounds = Bounds(np.zeros(n), np.ones(n))
        constr = [{"type": "eq", "fun": _con_fun, "jac": _con_jac, "args": (n,)}]
        try:
            res = minimize(_sm_obj, h_cur, method="SLSQP", bounds=bounds,
                          constraints=constr, args=(beta, dx),
                          options={"maxiter": maxiter, "ftol": ftol})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h_cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return h_cur

def coord_cd(h, n, t0, deadline, max_rounds=200):

    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    cur_max = np.max(corr)

        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for m in order:
            if time.time() > deadline:
                break
            old = h_cur[m]
            deltas = [1.0 - old, -old, 0.1, -0.1, 0.05, -0.05]
            best_mv = old
            best_mv_max = cur_max
            for dv in deltas:
                nv = old + dv
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                nv = float(np.clip(nv, 0.0, 1.0))
                if abs(nv - old) < 1e-9:
                    continue
                h_cur[m] = nv
                nc = np.max(np.correlate(h_cur, 1.0 - h_cur, mode="full"))
                if nc < best_mv_max - 1e-11:
                    best_mv_max = nc
                    best_mv = nv
            if best_mv != old:
                h_cur[m] = best_mv
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                cur_max = np.max(corr)
                improved = True
            else:

        if not improved:

def cont_to_binary(h_cont, n):
    target = n // 2
    # Take top n/2 values as 1s
    threshold = np.sort(h_cont)[::-1][target - 1] if target > 0 else 1.0
    h_bin = (h_cont >= threshold).astype(float)
    # If ties, adjust
    while h_bin.sum() > target:
        # Turn a 1 to 0 (prefer those closest to threshold)
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.mean(np.exp(s))) / beta) * dx

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_pass(h, n, betas, maxiter=3000, ftol=1e-14):
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    cur_c5 = compute_c5(h_cur)
    for beta in betas:
        bounds = Bounds(np.zeros(n), np.ones(n))
        constr = [{"type": "eq", "fun": _con_fun, "jac": _con_jac, "args": (n,)}]
        try:
            res = minimize(_sm_obj, h_cur, method="SLSQP", bounds=bounds,
                          constraints=constr, args=(beta, dx),
                          options={"maxiter": maxiter, "ftol": ftol})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h_cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return h_cur

def coord_cd(h, n, deadline, max_rounds=300):
    target = n / 2.0
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    cur_max = np.max(corr)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for m in order:
            if time.time() > deadline:
                break
            old = h_cur[m]
            deltas = [1.0 - old, -old, 0.08, -0.08, 0.04, -0.04]
            best_mv = old
            best_mx = cur_max
            for dv in deltas:
                nv = old + dv
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                nv = float(np.clip(nv, 0.0, 1.0))
                if abs(nv - old) < 1e-9:
                    continue
                h_cur[m] = nv
                nc = np.max(np.correlate(h_cur, 1.0 - h_cur, mode="full"))
                if nc < best_mx - 1e-11:
                    best_mx = nc
                    best_mv = nv
            if best_mv != old:
                h_cur[m] = best_mv
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                cur_max = np.max(corr)
                improved = True
            else:
                h_cur[m] = old
        if not improved:
            break
    h_cur = project_h(h_cur, target)
    return h_cur

def cont_to_binary(h_cont, n):
    target = n // 2
    sorted_h = np.sort(h_cont)[::-1]
    threshold = sorted_h[target - 1] if target > 0 else 1.0
    h_bin = (h_cont >= threshold).astype(float)
    while h_bin.sum() > target:
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0:
            break
        mi = ones[np.argmin(h_cont[ones])]
        h_bin[mi] = 0
    while h_bin.sum() < target:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) == 0:
            break
        mi = zeros[np.argmax(h_cont[zeros])]
        h_bin[mi] = 1
    return h_bin

def binary_anneal(h_cont, n, deadline, n_iters=30000):
    target = n // 2
    h_bin = cont_to_binary(h_cont, n)
    cur_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    T = 0.04
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-7 and np.random.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = h_bin.copy()
                best_c5 = cur_c5
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        T *= 0.9995
    return best_h, best_c5

def adam_optimize(h_init, n, deadline, lr=0.015):
    h = h_init.copy()
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    while time.time() < deadline:
        t += 1
        beta = 40.0 + 3.0 * t
        corr = np.correlate(h, 1.0 - h, mode="full")
        cm = np.max(corr)
        s = np.clip(beta * (corr - cm), -500, 500)
        exp_s = np.exp(s)
        w = exp_s / np.sum(exp_s)
        grad = np.zeros(n)
        active = np.where(w > 1e-4)[0]
        for li in active:
            d = int(li) - (n - 1)
            wi = w[li]
            if d > 0:
                grad[d:] += wi * (1.0 - h[:n - d])
                grad[:n - d] -= wi * h[d:]
            elif d < 0:
                dp = -d
                grad[:n - dp] += wi * (1.0 - h[dp:])
                grad[dp:] -= wi * h[:n - dp]
            else:
                grad += wi * (1.0 - 2.0 * h)
        m_arr = 0.9 * m_arr + 0.1 * grad
        v_arr = 0.999 * v_arr + 0.001 * grad * grad
        m_h = m_arr / (1.0 - 0.9**t)
        v_h = v_arr / (1.0 - 0.999**t)
        step = lr * m_h / (np.sqrt(v_h) + 1e-8)
        h = project_h(h - step, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    return best_h, best_c5
