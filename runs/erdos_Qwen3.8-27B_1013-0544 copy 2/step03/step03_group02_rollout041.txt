**Gradient-guided coordinate descent**:
1. Compute the gradient of the smooth max objective

The gradient of the smooth max of c_k with respect to h[j] is:
∂/∂h[j] smooth_max(c, β) = (1/β) * ∑_k w_k * ∂c_k/∂h[j]

where w_k = exp(β(c_k - max_c)) / ∑_m exp(β(c_m - max_c))

And ∂c_k/∂h[j] depends on whether j appears in the "first" or "second" position of the correlation at lag k.

This is computable in O(n) per coordinate (or O(n²) for all coordinates), which is affordable.

- The gradient component g[j] tells us the direction

- The step size can be determined by a line search or by the magnitude of g[j]

This is essentially doing gradient descent but one coordinate at a time (coordinate-wise gradient descent). The advantage is that we ensure feasibility after each update.

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15: break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new: return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def _sml_obj(h, beta, dx):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def _sc(h, t): return h.sum() - t
def _sj(h, t): return np.ones_like(h)

def _slsqp(h0, n, beta, mi=3000, ftol=1e-14):
    t = n / 2.0
    try:
        r = minimize(_sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
            bounds=Bounds(np.zeros(n), np.ones(n)),
            constraints=[{'type':'eq', 'fun':_sc, 'jac':_sj, 'args':(t,)}],
            options={'maxiter': mi, 'ftol': ftol})
        return project_h(r.x, t)
    except: return h0

def _smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    e = np.exp(xs)
    w = e / e.sum()
    grad = np.zeros(n)

        wp = w[p]
        if wp < 1e-8: continue
        lag = p - (n - 1)
        if lag > 0:
            l = lag
            m1 = idx >= l
            m2 = idx <= n - 1 - l
            grad[m1] += wp * (1.0 - h[idx[m1] - l])
            grad[m2] -= wp * h[idx[m2] + l]
        elif lag < 0:
            l = -lag
            m1 = idx <= n - 1 - l
            m2 = idx >= l
            grad[m1] += wp * (1.0 - h[idx[m1] + l])
            grad[m2] -= wp * h[idx[m2] - l]
        else:
            grad[:] += wp * (1.0 - 2.0 * h)
    return grad * dx

def _grad_cd(cur, target, n, sweeps, t0, budget_s):
    best = compute_c5(cur)
    beta = 300
    for sw in range(sweeps):
        if budget_s - (time.time() - t0) < 3: break
        g = _smooth_max_grad(cur, beta)
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if budget_s - (time.time() - t0) < 1: break
            if abs(g[j]) < 1e-12: continue
            direction = -np.sign(g[j])
            steps = [0.03, 0.015, 0.008, 0.05, 0.004, 0.1]

                trial[j] = np.clip(trial[j] + direction * s, 0, 1)
                trial = project_h(trial, target)
                cv = compute_c5(trial)
                if cv < best - 1e-15:
                    cur, best, improved = trial, cv, True
                    break
        if not improved: break
    return cur, best

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def _sml_obj(h, beta, dx):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def _sc(h, t):
    return h.sum() - t

def _sj(h, t):
    return np.ones_like(h)

def _slsqp(h0, n, beta, mi=3000, ftol=1e-14):
    t = n / 2.0
    try:
        r = minimize(_sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': _sc, 'jac': _sj, 'args': (t,)}],
                     options={'maxiter': mi, 'ftol': ftol})
        return project_h(r.x, t)
    except Exception:
        return h0

def _smg(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    e = np.exp(xs)
    w = e / e.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for p in range(len(c)):
        wp = w[p]
        if wp < 1e-8:
            continue
        lag = p - (n - 1)
        if lag > 0:
            l = lag
            m1 = idx >= l
            m2 = idx <= n - 1 - l
            if m1.any():
                grad[m1] += wp * (1.0 - h[idx[m1] - l])
            if m2.any():
                grad[m2] -= wp * h[idx[m2] + l]
        elif lag < 0:
            l = -lag
            m1 = idx <= n - 1 - l
            m2 = idx >= l
            if m1.any():
                grad[m1] += wp * (1.0 - h[idx[m1] + l])
            if m2.any():
                grad[m2] -= wp * h[idx[m2] - l]
        else:
            grad[:] += wp * (1.0 - 2.0 * h)
    return grad * dx

def _grad_cd(cur, target, n, sweeps, t0, budget_s):
    best = compute_c5(cur)
    beta = 300
    for sw in range(sweeps):
        if budget_s - (time.time() - t0) < 3:
            break
        g = _smg(cur, beta)
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if budget_s - (time.time() - t0) < 1:
                break
            if abs(g[j]) < 1e-12:
                continue
            direction = -np.sign(g[j])
            steps = (0.03, 0.015, 0.008, 0.05, 0.004, 0.002, 0.1)
            for s in steps:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + direction * s, 0, 1)
                trial = project_h(trial, target)
                cv = compute_c5(trial)
                if cv < best - 1e-15:
                    cur, best, improved = trial, cv, True
                    break
        if not improved:
            break
    return cur, best

def _plain_cd(cur, target, n, sweeps, t0, budget_s):
    best = compute_c5(cur)
    deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.06, -0.06)
    for sw in range(sweeps):
        if budget_s - (time.time() - t0) < 3:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if budget_s - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                cv = compute_c5(trial)
                if cv < best - 1e-15:
                    cur, best, improved = trial, cv, True
                    break
        if not improved:
            break
    return cur, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(320, 0.5)

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_start)

    np_parent = len(h_start)
    res_list = sorted(set([np_parent, 240, 200, 160, 120]), reverse=True)

    for n_pts in res_list:
        rem = budget_s - (time.time() - t0)
        if rem < 80 or n_pts < 60:
            break
        target = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # SLSQP with increasing smoothness
        for beta in [80, 200, 500]:
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            cur = _slsqp(cur, n_pts, beta, 2500)
            c = compute_c5(cur)
            if c < cur_c5:
                cur_c5 = c
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Gradient-guided coordinate descent
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            cur, cur_c5 = _grad_cd(cur, target, n_pts, 100, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Plain coordinate descent (catches what gradient CD misses)
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = _plain_cd(cur, target, n_pts, 50, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Perturbation restarts
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            for att in range(4):
                rem = budget_s - (time.time() - t0)
                if rem < 15:
                    break
                sc = 0.025 / (att + 1)
                hp = project_h(cur + np.random.randn(n_pts) * sc, target)
                hp = _slsqp(hp, n_pts, 300, 1500)
                cp = compute_c5(hp)
                if cp < cur_c5:
                    cur, cur_c5 = hp, cp
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts
                rem = budget_s - (time.time() - t0)
                if rem > 15:
                    cur, cur_c5 = _grad_cd(cur, target, n_pts, 50, t0, budget_s)
                    if cur_c5 < best_c5:
                        best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final high-precision polish on best solution
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        nf = len(best_h)
        tf = nf / 2.0
        for beta in [2000, 5000]:
            rem = budget_s - (time.time() - t0)
            if rem < 8:
                break
            best_h = _slsqp(best_h, nf, beta, 8000, 1e-15)
            best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = _grad_cd(best_h, len(best_h) / 2.0, len(best_h), 500, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h, best_c5 = _plain_cd(best_h, len(best_h) / 2.0, len(best_h), 100, t0, budget_s)

    return best_h, best_c5, best_n
