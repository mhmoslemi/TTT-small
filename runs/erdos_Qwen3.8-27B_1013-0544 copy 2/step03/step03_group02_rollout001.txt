def smooth_max_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    cs = np.clip(beta * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(cs)) / len(cs)) / beta) * dx

def sum_eq(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(smooth_max_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def make_perturbations(h0, n, target, n_perts, rng):
    perts = []
    x = np.linspace(0, 2, n)
    for i in range(n_perts):
        p = h0.copy()
        perm = i % 4
        amp = 0.04 * (1.0 + 0.3 * rng.rand())
        if perm == 0:
            p = p + amp * rng.randn(n)
        elif perm == 1:
            freq = rng.randint(1, 6)
            phase = rng.rand() * 2 * np.pi
            p = p + amp * np.sin(2 * np.pi * freq * x / 2 + phase)
        elif perm == 2:
            for _ in range(3):
                center = rng.rand() * n
                sigma = rng.rand() * 20 + 5

        else:

                p += amp * 0.5 * rng.randn(n)
        p = project_h(p, target)
        perts.append(p)

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(200, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    resolutions = sorted(set([n_base, 200, 240, 280]), key=lambda x: -x)
    resolutions = [r for r in resolutions if r <= max(n_base, 280)]

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 80:
            break
        target = n_pts / 2.0
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Generate diverse perturbations
        n_perts = 10
        perts = make_perturbations(cur, n_pts, target, n_perts, rng)
        perts.append(cur.copy())

        # Local optimize each perturbation
        per_budget = (remaining - 20) / (len(perts) + 2)
        for pi, pp in enumerate(perts):
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            alloc = min(per_budget, remaining - 10) / (len(perts) - pi)
            cur2 = do_slsqp(pp, n_pts, target, 150, 2000, 1e-14)
            cur2, c2 = pgd_optimize(cur2, target, t0, budget_s, max_time=max(3, alloc * 0.5))
            if c2 < cur_c5:
                cur, cur_c5 = cur2, c2
            if c2 < best_c5:
                best_h, best_c5, best_n = cur2.copy(), c2, n_pts

        # Additional PGD passes on best
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            for _ in range(3):
                remaining = budget_s - (time.time() - t0)
                if remaining < 8:
                    break
                cur, cur_c5 = pgd_optimize(cur, target, t0, budget_s, max_time=10)
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Coordinate descent polish
        remaining = budget_s - (time.time() - t0)
        if remaining > 12:
            cur, cur_c5 = coord_descent_adv(cur, target, t0, budget_s, max_time=10)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final high-precision polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 20 and best_h is not None:
        nf = len(best_h)
        tf = nf / 2.0
        for beta in [200, 500, 1000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, 5000, 1e-15)
            best_c5 = compute_c5(best_h)
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            best_h, best_c5 = pgd_optimize(best_h, tf, t0, budget_s, max_time=8)
        remaining = budget_s - (time.time() - t0)
        if remaining > 6:
            best_h, best_c5 = coord_descent_adv(best_h, tf, t0, budget_s, max_time=5)

    return best_h, best_c5, best_n
