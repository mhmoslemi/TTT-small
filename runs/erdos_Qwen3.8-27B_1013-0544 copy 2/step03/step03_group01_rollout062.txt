Actually, let me think about the "leveling" approach more carefully.

The idea: if the current C₅ is determined by a few "outlier" lags, we should focus on reducing those specific lags rather than trying to reduce all lags uniformly.

Algorithm:
1. Compute all correlation values
2. Find the lags where correlation > target (e.g., target = current_C5 - margin)
3. Compute the gradient of the sum of (c[k] - target)² for those lags
4. Take a gradient step in the -gradient direction
5. Project to feasible set
6. Update target (decrease it)
7. Repeat

This is like a "peaking" algorithm that focuses on the worst lags.

Let me implement this:

```python
def leveling_step(h, n, target, lr, top_k=5):
    """One step of leveling: reduce the top-k correlation values."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Get the indices of the top-k values
    top_indices = np.argsort(c)[-top_k:]
    # Compute gradient of sum_{i in top} (c[i] - target)^2
    # = 2 * sum_{i in top} (c[i] - target) * dc[i]/dh
    grad = np.zeros(n)
    for idx in top_indices:
        k = idx - (n - 1)  # convert to lag
        w = 2.0 * (c[idx] - target)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            # dc/dh[j] = (1-h[j+k]) for j in [0, n-1-|k|] - h[j-|k|] for j in [|k|, n-1]
            abs_k = abs(k)
            for j in range(n - abs_k):
                if k > 0:
                    grad[j] += w * (1.0 - h[j + k])
                else:
                    grad[j] += w * (1.0 - h[j - k])
            for j in range(abs_k, n):
                if k > 0:
                    grad[j] -= w * h[j - k]
                else:
                    grad[j] -= w * h[j + k]
    # Gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
def trig_h(params, n):
    """Generate h from trigonometric parameters.
    params = [a1, b1, a2, b2, ..., aM, bM]
    h(x) = 0.5 + sum_m [a_m cos(πmx) + b_m sin(πmx)]
    """
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    M = len(params) // 2
    for m in range(1, M+1):
        a = params[2*(m-1)]
        b = params[2*(m-1)+1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return h
```
