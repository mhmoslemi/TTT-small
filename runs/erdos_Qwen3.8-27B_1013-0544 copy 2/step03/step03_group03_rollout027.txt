def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:

        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:

        if cnt == 0:

    n = len(h)

    # corr[k] corresponds to lag d = k - (n-1)

    b = (2.0/n) * coeff

            elif hi < -1e-14:
                tlo = tm
            else:
                break
    return 0.5 * (tlo + thi)

def cd_pass(h, rng):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    order = rng.permutation(n)
    for i in order:
        t = _optimal_coord(h, i, corr)
        delta = t - h[i]
        if abs(delta) > 1e-15:
            coeff = _coeffs_for_i(h, i)
            corr += delta * coeff
            h[i] = t
    h = project_feasible(h, n / 2.0)
    return h

def cd_optimize(h, rng, deadline):
    h = h.copy()
    best = compute_c5(h)
    stagnant = 0
    while time.time() < deadline:
        h = cd_pass(h, rng)
        c = compute_c5(h)
        if c < best - 1e-10:
            best = c
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 200:
            break
    return h

def smooth_max(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def _slsqp_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def _con_eq(h, tgt):
    return h.sum() - tgt

def _con_jac(h, tgt):
    return np.ones(len(h))

def _random_binary_start(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    n_ones = h.sum()
    target = n / 2.0
    if n_ones > target:
        excess = int(n_ones - target)
        ones_idx = np.where(h > 0.5)[0]
        flip = rng.choice(ones_idx, size=min(excess, len(ones_idx)), replace=False)
        h[flip] = 0.0
    elif n_ones < target:
        deficit = int(target - n_ones)
        zeros_idx = np.where(h < 0.5)[0]
        flip = rng.choice(zeros_idx, size=min(deficit, len(zeros_idx)), replace=False)
        h[flip] = 1.0
    return project_feasible(h, target)

def _structured_start(n, rng):
    mode = rng.integers(0, 4)
    h = np.zeros(n)
    if mode == 0:
        blocks = rng.integers(3, 12)
        sizes = rng.dirichlet(np.ones(blocks) * 2) * n
        vals = rng.random(blocks)
        pos = 0
        for k in range(blocks):
            sz = int(round(sizes[k]))
            h[pos:pos + sz] = vals[k]
            pos += sz
        h[n - (pos - n):] = 0.0
    elif mode == 1:
        h = np.tile([1.0, 0.0], n // 2 + 1)[:n]
    elif mode == 2:
        h = np.linspace(0, 1, n) ** rng.uniform(0.3, 3.0)
    else:
        h = 0.5 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    n0 = len(best_h)
    best_h = project_feasible(best_h, n0 / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: CD on initial solution (45% of time)
    p1_deadline = t0 + 0.45 * budget_s
    h1 = cd_optimize(best_h, rng, p1_deadline)
    c = compute_c5(h1)
    if c < best_c5:
        best_h, best_c5 = h1, c

    # Phase 2: Resolution exploration (30% of time)
    p2_deadline = t0 + 0.75 * budget_s
    cur_n = len(best_h)
    candidates = sorted(set([max(60, cur_n + off) for off in [-60, -40, -30, -20, -10,
                                                             0, 10, 20, 30, 40, 60, 80, 100, 150]]))
    candidates += [100, 120, 140, 160, 180, 220, 250, 280, 320, 350, 400, 450, 500, 600, 700]
    candidates = sorted(set([x for x in candidates if 60 <= x <= 800]))

    for n_try in candidates:
        if time.time() > p2_deadline - 5:
            break
        ts = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_feasible(h_t, ts)
        sub_dl = min(p2_deadline - 2, time.time() + 60)
        h_t = cd_optimize(h_t, rng, sub_dl)
        c = compute_c5(h_t)
        if c < best_c5:
            best_h, best_c5 = h_t.copy(), c

    # Phase 3: Multi-start with random/structured inits (15% of time)
    p3_deadline = t0 + 0.90 * budget_s
    n_best = len(best_h)
    tgt_best = n_best / 2.0
    n_starts = 30
    per_start_time = max(5.0, (p3_deadline - time.time()) / n_starts - 1)

    for s in range(n_starts):
        if time.time() > p3_deadline - 3:
            break
        sub_dl = time.time() + per_start_time
        if s % 2 == 0:
            h_s = _random_binary_start(n_best, rng)
        else:
            h_s = _structured_start(n_best, rng)
        h_s = cd_optimize(h_s, rng, sub_dl)
        c = compute_c5(h_s)
        if c < best_c5:
            best_h, best_c5 = h_s.copy(), c

    # Phase 4: SLSQP polish (5% of time)
    p4_deadline = t0 + budget_s - 10
    n = len(best_h)
    dx = 2.0 / n
    tgt = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (tgt,)}]

    for beta in [50, 200, 1000, 5000, 20000]:
        if time.time() > p4_deadline:
            break
        for trial in range(2):
            if time.time() > p4_deadline - 5:
                break
            x0 = best_h.copy()
            if trial == 1:
                x0 += rng.standard_normal(n) * 0.01
                x0 = project_feasible(x0, tgt)
            try:
                res = minimize(_slsqp_obj, x0, args=(dx, beta), method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, tgt)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new.copy(), c
            except Exception:
                pass

    # Phase 5: Final CD polish (remaining time)
    h_f = cd_optimize(best_h, rng, t0 + budget_s - 3)
    c = compute_c5(h_f)
    if c < best_c5:
        best_h, best_c5 = h_f, c

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
