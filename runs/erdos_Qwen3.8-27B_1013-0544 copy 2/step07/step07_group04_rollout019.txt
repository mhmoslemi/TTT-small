    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):

            hi = mid

def _grad_c_shift(x, n, shift):
    """Gradient of c[shift] with respect to x."""
    g = np.zeros(n)
    dx = 2.0 / n
    if shift >= 0:
        m = n - shift
        g[shift:n] += (1.0 - x[:m])
        g[:m] -= x[shift:n]
    else:
        t = -shift
        m = n - t
        g[:m] += (1.0 - x[t:n])
        g[t:n] -= x[:m]
    return dx * g

    m = float(c.max())

    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha
