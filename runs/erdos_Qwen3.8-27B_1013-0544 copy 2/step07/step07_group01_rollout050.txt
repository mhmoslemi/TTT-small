    dx = 2.0 / n
    target = n / 2.0
    
    def objective(z):
        h = np.clip(z, 0, 1)
        c = np.correlate(h, 1-h, mode='full') * dx
        # p-norm approximation of max

            m = np.max(c)

    p2_end = t0 + 500
    best_h, best_c = multigrid_refine(best_h, n, p2_end)

    # Phase 3: Higher alpha L-BFGS-B (500-700s)
    p3_end = t0 + 700
    for a in [20000, 100000, 500000]:
        if time.time() > p3_end - 20:
            break
        h3 = lbfgsb(best_h, n, a, 500, p3_end - 10)
        h3 = project_feasible(h3, n)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: Momentum subgradient (700-900s)
    p4_end = t0 + 900
    h4, c4 = momentum_subgrad(best_h, n, p4_end, topk=5, alpha_w=50.0)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    # Phase 5: Random restarts + L-BFGS-B polish (900-1050s)
    p5_end = t0 + 1050
    rng = np.random.RandomState(seed + 1)
    for trial in range(12):
        if time.time() > p5_end - 10:
            break
        nl = 0.002 * (1 + trial * 0.4)
        h5 = best_h + nl * rng.randn(n)
        h5 = project_feasible(h5, n)
        h5 = lbfgsb(h5, n, 20000, 200, p5_end - 8)
        h5 = project_feasible(h5, n)
        h5 = lbfgsb(h5, n, 200000, 150, p5_end - 8)
        h5 = project_feasible(h5, n)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final polish (1050-deadline)
    if time.time() < deadline - 15:
        h6 = lbfgsb(best_h, n, 1000000, 300, deadline - 10)
        h6 = project_feasible(h6, n)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
