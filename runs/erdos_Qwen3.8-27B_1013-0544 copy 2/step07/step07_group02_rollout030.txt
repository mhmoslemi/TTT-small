def project_feasible(x,n):
    x=np.asarray(x,dtype=float)
    target=float(n)/2.0
    # First clip to [0,1]
    x=np.clip(x,0.0,1.0)
    s=x.sum()
    if abs(s-target) < 1e-15: return x
    # Bisection for projection onto box+sum
    lo=float(x.min()) - 1.0
    hi=float(x.max()) + 1.0
    for _ in range(80):
       mid=(lo+hi)*0.5
       y=np.clip(x-mid,0.0,1.0)
       sy=y.sum()
       if sy > target:
          lo=mid
       else:
          hi=mid
    y=np.clip(x-(lo+hi)*0.5,0.0,1.0)
    diff=target-y.sum()
    # distribute diff to a few interior points
    if diff>0:
       mask=y<1.0-1e-15
    else:
       mask=y>1e-15
    if mask.any():
       cnt=mask.sum()
       y[mask] += diff/cnt
       y=np.clip(y,0.0,1.0)
    # if residual due clip, iterative
    for _ in range(10):
       s=y.sum(); diff=target-s
       if abs(diff)<1e-15: break
       if diff>0:
          mask=y<1.0-1e-15
       else:
          mask=y>1e-15
       if not mask.any(): break
       y[mask] += diff/mask.sum()
       y=np.clip(y,0.0,1.0)
    return y

import numpy as np
import time
from scipy.optimize import minimize, linprog

_NS = 0
_DX = 1.0
_AL = 1.0
_TG = 0.0
_ON = None

def adj(x, n):
    t = 0.5 * float(n)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    for _ in range(25):
        d = t - float(np.sum(x))
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = x < 1.0 - 1e-12
        else:
            m = x > 1e-12
        if not m.any():
            break
        x[m] += d / float(np.sum(m))
        np.clip(x, 0.0, 1.0, out=x)
    return x

def proj(x, n):
    x = np.asarray(x, dtype=float)
    t = 0.5 * float(n)
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - t) < 1e-12:
        return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):

        y = np.clip(x - m, 0.0, 1.0)
        if float(y.sum()) > t:

        else:
            hi = m
    return adj(x - 0.5 * (lo + hi), n)

def vals(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def c5val(h, n):
    return float(np.max(vals(h, n)) * (2.0 / float(n)))

def add_g(g, h, n, k, w):
    if w <= 0.0:
        return
    if k >= 0:
        L = n - k
        if L > 0:
            g[:L] += w * (1.0 - h[k:k + L])
            g[k:k + L] -= w * h[:L]
    else:
        r = -k
        L = n - r
        if L > 0:
            g[r:r + L] += w * (1.0 - h[:L])
            g[:L] -= w * h[r:r + L]

def wgrad(h, n, p):
    g = np.zeros(n, dtype=float)
    idx = np.nonzero(np.asarray(p, dtype=float) > 1e-16)[0]
    for t in idx:
        ti = int(t)
        add_g(g, h, n, n - 1 - ti, float(p[ti]))
    return g

    m = len(c)
    cap = int(cap)
    if cap <= 0 or m == 0:

    if cap >= m:

    return np.argpartition(c, -cap)[-cap:]

    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    v = vals(h, _NS) * _DX
    m = float(np.max(v))

    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / _AL

def jac1(x):
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.ones(_NS, dtype=float)
    v = vals(h, _NS) * _DX
    m = float(np.max(v))
    e = np.exp(np.minimum(_AL * (v - m), 0.0))
    s = float(np.sum(e))
    if s <= 0.0:
        return np.zeros(_NS, dtype=float)
    p = e / s
    return wgrad(h, _NS, p) * _DX

    return float(np.sum(x) - _TG)

    return _ON.copy()

def slsqp(x, n, al, it, ft):

    x = proj(x, n)
    _NS = n
    _DX = 2.0 / float(n)
    _AL = float(al)
    _TG = 0.5 * float(n)

        hh = np.repeat(h, nn // n)
    else:
        xo = np.linspace(0.0, 2.0, n, endpoint=False)
        xn = np.linspace(0.0, 2.0, nn, endpoint=False)
        hh = np.interp(xn, xo, h)
    return proj(hh, nn)

def mk(n, seed):
    r = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = 0.5 + 0.30 * np.cos(np.pi * x) + 0.12 * np.sin(2.0 * np.pi * x) + 0.05 * r.normal(size=n)
    return proj(y, n)

def capf(n):
    if n <= 250:
        return 70
    if n <= 450:
        return 100
    if n <= 650:
        return 140
    return 180

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + max(30.0, float(budget_s) - 10.0)
    total = max(1.0, hard - t0)
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = mk(n, seed)

    if n < 10:
        n = 400
        h = mk(n, seed)
    if n > 960:
        h = reas(h, n, 960)
        n = 960
    h = proj(h, n)

    best_h = h.copy()
    best_n = int(n)
    best_c = c5val(best_h, best_n)

    t_s0 = min(hard - 5.0, t0 + 0.06 * total)
    for al in (30.0, 100.0):
        if time.time() >= t_s0 - 2.0:
            break
        ht = slsqp(best_h, best_n, al, 50, 1e-10)
        ct = c5val(ht, best_n)
        if ct < best_c - 1e-13:
            best_h = ht.copy()
            best_c = ct

    main_end = min(hard - min(240.0, 0.4 * total), t0 + 0.60 * total)
    if main_end < t_s0 + 10.0:
        main_end = min(hard - 10.0, t_s0 + max(20.0, 0.1 * total))
    hb, cb = lp_search(best_h, best_n, 0.08, main_end, capf(best_n))
    if cb < best_c - 1e-13:
        best_h = hb.copy()
        best_c = cb

    t_s2 = min(hard - 5.0, main_end + min(0.10 * total, 120.0))
    for al in (200.0, 1000.0, 5000.0):
        if time.time() >= t_s2 - 2.0:
            break
        ht = slsqp(best_h, best_n, al, 80, 1e-11)
        ct = c5val(ht, best_n)
        if ct < best_c - 1e-12:
            best_h = ht.copy()
            best_c = ct

    t_noise = min(hard - 5.0, t_s2 + min(0.14 * total, 160.0))
    trial = 0
    while time.time() < t_noise - 10.0:
        trial += 1
        sc = 0.015 + 0.03 * rng.rand()
        if trial == 1:
            sc *= 1.8
        nz = rng.randn(best_n)
        nz -= nz.mean()
        ini = proj(best_h + sc * nz, best_n)
        sub = min(t_noise - 5.0, time.time() + 20.0 + 10.0 * rng.rand())
        ht, ct = lp_search(ini, best_n, 0.06, sub, capf(best_n))
        if ct < best_c - 1e-13:
            best_h = ht.copy()
            best_c = ct

    for nn in (200, 600, 800):
        if hard - time.time() < 80.0 or nn == best_n or nn > 960:
            continue
        sub = min(hard - 60.0, time.time() + (70.0 if nn < best_n else 50.0))
        h2 = reas(best_h, best_n, nn)
        h2, ct = lp_search(h2, nn, 0.07, sub, capf(nn))
        if ct < best_c - 1e-13:
            best_h = h2.copy()
            best_n = nn
            best_c = ct
        if time.time() < sub - 5.0:
            ht = slsqp(best_h if best_n == nn else h2, nn, 1000.0, 60, 1e-11)
            ct = c5val(ht, nn)
            if ct < best_c - 1e-12:
                best_h = ht.copy()
                best_n = nn
                best_c = ct

    final_end = hard - 2.0
    if time.time() < final_end:
        hf, cf = lp_search(best_h, best_n, 0.004, final_end, max(180, capf(best_n)))
        if cf < best_c - 1e-12:
            best_h = hf.copy()
            best_c = cf

    best_h = adj(best_h, best_n)
    best_c = c5val(best_h, best_n)
    if not np.isfinite(best_c):
        best_h = mk(best_n, seed)
        best_c = c5val(best_h, best_n)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    return best_h, float(best_c), int(best_n)
