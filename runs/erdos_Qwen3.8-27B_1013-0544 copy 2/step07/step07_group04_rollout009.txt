    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        x = (np.arange(n0)+0.5)*(2.0/n0)
        h0 = project_box_sum(0.5 + 0.3*np.sin(3*np.pi*x) + 0.2*np.sin(7*np.pi*x), n0)

    h0 = project_box_sum(h0, n0)
    besth = h0.copy()
    bestc = float(compute_c(h0, n0).max())
    bestn = n0

    # Phase 1: L-BFGS-B polish on parent size
    for alpha, mu, it in ((30,200,250),(100,1000,200),(500,5000,150)):
        if time.time() > deadline - 60: break
        h2 = lbfgsb_phase(besth, bestn, alpha, mu, it)
        h2 = project_box_sum(h2, bestn)
        v2 = float(compute_c(h2, bestn).max())
        if v2 < bestc:
            bestc = v2; besth = h2

    # Phase 2: Intensive pair search
    p2_dl = deadline - 80
    if time.time() < p2_dl:
        besth, bestc = intensive_pair_search(besth, p2_dl, seed)
        bestn = len(besth)

    # Phase 3: Try another n for diversity
    for alt_n in (500, 350):
        if time.time() > deadline - 50: break
        ha = interpolate_h(besth, bestn, alt_n)
        ha = project_box_sum(ha + 0.003*rng.randn(alt_n), alt_n)
        h2 = lbfgsb_phase(ha, alt_n, 200.0, 2000.0, 150)
        h2 = project_box_sum(h2, alt_n)
        v2 = float(compute_c(h2, alt_n).max())
        if v2 < bestc - 1e-5:
            bestc = v2; besth = h2; bestn = alt_n
            if time.time() < deadline - 40:
                besth, bestc = intensive_pair_search(besth, deadline - 40, seed+77)
                bestn = len(besth)

    # Phase 4: Final L-BFGS-B + SLSQP polish
    for alpha, mu, it in ((1000,20000,120),(5000,100000,80)):
        if time.time() > deadline - 20: break
        h2 = lbfgsb_phase(besth, bestn, alpha, mu, it)
        h2 = project_box_sum(h2, bestn)
        v2 = float(compute_c(h2, bestn).max())
        if v2 < bestc:
            bestc = v2; besth = h2
    if time.time() < deadline - 15:
        h2 = slsqp_phase(besth, bestn, 3000.0, 100)
        h2 = project_box_sum(h2, bestn)
        v2 = float(compute_c(h2, bestn).max())
        if v2 < bestc:
            bestc = v2; besth = h2

    # Phase 5: Final pair search sweep
    if time.time() < deadline - 5:
        besth, bestc = intensive_pair_search(besth, deadline, seed+200)
        bestn = len(besth)

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
