import time
import cvxpy as cp

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

            if s < 1e-14:

            h = h + diff / s * h

def lse_obj_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + np.log(exp_shifted.sum()) / alpha
    w = exp_shifted / exp_shifted.sum()
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if w[m] < 1e-15:
            continue
        if k >= 0:
            j_lo = 0
            j_hi = n - k
            if j_hi > j_lo:
                grad[j_lo:j_hi] += w[m] * (1.0 - x[j_lo+k:j_hi+k])
            j_lo2 = k
            j_hi2 = n

        else:

            j_lo = 0

    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    # Compute gradient matrix
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            j_lo, j_hi = 0, n - k
            G[j_lo:j_hi, m] += (1.0 - h[j_lo+k:j_hi+k])
            j_lo2, j_hi2 = k, n
            G[j_lo2:j_hi2, m] -= h[j_lo2-k:j_hi2-k]
        else:
            k2 = -k
            j_lo, j_hi = 0, n
            G[j_lo:j_hi, m] += (1.0 - h[j_lo+k:j_hi+k])
            j_lo2, j_hi2 = 0, n - k2
            G[j_lo2:j_hi2, m] -= h[j_lo2+k2:j_hi2+k2]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'CLARABEL']:

            prob.solve(solver=sol)
            solved = True
            break

            continue

    d = delta.value

        if mc_try < best_mc - 1e-14:

            best_h = h_try
    return best_h, best_mc

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

_global_n = 400
_global_alpha = 50.0
_global_lam = 1e5

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = c_max + np.log(exp_s.sum()) / alpha
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        k = m - nm1
        if k >= 0:
            hi = n - k
            if hi > 0:
                grad[0:hi] += wm * (1.0 - x[k:hi + k])
            if k < n:
                grad[k:n] -= wm * x[0:n - k]
        else:
            k2 = -k
            grad[0:n] += wm * (1.0 - x[0:n - k2])
            if k2 < n:
                grad[0:n - k2] -= wm * x[k2:n]
    grad *= dx
    s = x.sum() - n / 2.0
    log_sum += 0.5 * lam * s * s
    grad += lam * s
    return log_sum, grad

def _obj_wrap(x):
    val, _ = lse_obj_grad(x, _global_n, _global_alpha, _global_lam)
    return val

def _grad_wrap(x):
    _, g = lse_obj_grad(x, _global_n, _global_alpha, _global_lam)
    return g

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    nm1 = n - 1
    for m in range(2 * n - 1):
        k = m - nm1
        if k >= 0:
            hi = n - k
            if hi > 0:
                G[0:hi, m] += (1.0 - h[k:hi + k])
            if k < n:
                G[k:n, m] -= h[0:n - k]
        else:
            k2 = -k
            G[0:n, m] += (1.0 - h[0:n - k2])
            if k2 < n:
                G[0:n - k2, m] -= h[k2:n]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'CLARABEL', 'GLPK']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam_f in [1.0, 0.5, 0.25]:
        h_try = h + lam_f * d
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def lbfgs_optimize(h, n, alpha, lam, maxiter):
    _global_n = n
    _global_alpha = alpha
    _global_lam = lam
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_wrap, h, jac=_grad_wrap, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return project_feasible(res.x, n)
