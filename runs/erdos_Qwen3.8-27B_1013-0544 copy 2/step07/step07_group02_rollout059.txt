def project_feasible(h, n):

def compute_gradients(h, n):
    """Compute G[i,m] = dC_m/dh_i for all i and m (m indexes the full correlation)"""
    ncorr = 2 * n - 1
    G = np.zeros((n, ncorr))
    for m in range(ncorr):
        k = m - (n - 1)
        # Contribution: C_m = sum_i h[i] * (1 - h[i+k]) for valid i
        # dC_m/dh[j] = (1-h[j+k]) [if 0<=j+k<n] - h[j-k] [if 0<=j-k<n]

        j1_lo = max(0, -k)

        if j2_hi > j2_lo:

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    """Optimize using log-sum-exp smoothed objective with SLSQP."""
    dx = 2.0 / n
    def obj(x):

        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    """Compute smooth max correlation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    ncorr = len(corr)
    shifted = alpha * (corr - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    sw = w.sum()
    logsumex = sm + np.log(sw)
    f = m + logsumex / alpha
    w = w / sw  # softmax weights
    
    # Gradient
    g = np.zeros(n)
    for m_idx in range(ncorr):
        ww = w[m_idx]
        if ww < 1e-14:
            continue
        k = m_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            g[i_lo:i_hi] += ww * (1.0 - h[i_lo+k:i_hi+k])
        i_lo2 = max(0, k)
        i_hi2 = min(n, n + k)
        if i_hi2 > i_lo2:
            g[i_lo2:i_hi2] -= ww * h[i_lo2-k:i_hi2-k]
    return f, g

def lbfgsb_augmented(h0, n, alpha, rho0, max_outer=8, max_inner=3000, tol=1e-8):
    """Augmented Lagrangian + L-BFGS-B for the smooth problem."""
    h = h0.copy()
    mu = 0.0
    rho = rho0
    for outer in range(max_outer):
        def obj(x):
            f, gf = smooth_obj_and_grad(x, n, alpha)
            aug = (x.sum() - n / 2.0)
            return f + mu * aug + 0.5 * rho * aug * aug
        def grad(x):
            f, gf = smooth_obj_and_grad(x, n, alpha)
            aug = (x.sum() - n / 2.0)
            return gf + (mu + rho * aug)
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-14})
        h = res.x
        aug = h.sum() - n / 2.0
        if abs(aug) < tol:
            break
        mu = mu + rho * aug
        rho = min(rho * 2.0, 1e6)
    return project_feasible(h, n)

def lp_step(h, n, eps, top_k=None):
    """LP trust region step with reduced constraints."""
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    ncorr = len(C)
    
    if top_k is not None and top_k < ncorr:
        idx = np.argsort(C)[::-1][:top_k]
    else:
        idx = np.arange(ncorr)
    
    # Compute gradients for selected shifts
    G_sel = np.zeros((n, len(idx)))
    for col, m_idx in enumerate(idx):
        k = m_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            G_sel[i_lo:i_hi, col] = 1.0 - h[i_lo+k:i_hi+k]
        i_lo2 = max(0, k)
        i_hi2 = min(n, n + k)
        if i_hi2 > i_lo2:
            G_sel[i_lo2:i_hi2, col] -= h[i_lo2-k:i_hi2-k]
    
    C_sel = C[idx]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C_sel + G_sel.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'CLARABEL', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.7, 0.5, 0.3, 0.1]:

        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

    # Phase 1: LP optimization with reduced constraints
    phase1_dl = t0 + 0.30 * total
    h_cur = best_h.copy()
    eps = 0.04
    no_improve = 0
    while time.time() < phase1_dl:
        h_new, _ = lp_step(h_cur, n, eps, top_k=min(n, 64))
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                eps *= 0.7
                if eps < 0.001:
                    break
                no_improve = 0
        h_cur = h_new

    # Phase 2: L-BFGS-B with augmented Lagrangian (key improvement)
    phase2_dl = t0 + 0.65 * total
    h_cur = best_h.copy()
    alpha_schedule = [300, 1000, 5000, 20000, 100000, 500000]
    for alpha in alpha_schedule:
        if time.time() > phase2_dl - 3:
            break
        remaining = phase2_dl - time.time() - 3
        if remaining < 3:
            break
        max_inner = max(50, int(remaining * 1.5))
        h_cur = lbfgsb_augmented(h_cur, n, alpha, rho0=50.0,
                                 max_outer=6, max_inner=min(max_inner, 4000), tol=1e-9)
        c5_new = compute_c5(h_cur, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()

    # Phase 3: Multi-resolution exploration
    phase3_dl = t0 + 0.82 * total
    for nh in [300, 500, 600, 800]:
        if time.time() > phase3_dl - 10:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        c5_h = compute_c5(hh, nh)
        if c5_h <= best_c5 + 0.002:
            eps_r = 0.03
            for _ in range(20):
                if time.time() > phase3_dl - 15:
                    break
                hh, _ = lp_step(hh, nh, eps_r, top_k=min(nh, 48))
                c5_new = compute_c5(hh, nh)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = hh.copy()
                    n = nh
            if time.time() < phase3_dl - 20:
                hh = lbfgsb_augmented(hh, nh, 50000, rho0=100.0,
                                      max_outer=5, max_inner=3000, tol=1e-9)
                c5_new = compute_c5(hh, nh)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = hh.copy()
                    n = nh

    # Phase 4: Final polish with very fine LP and L-BFGS-B
    if time.time() < dl - 10:
        h_cur = best_h.copy()
        eps_f = 0.003
        for _ in range(300):
            if time.time() > dl - 20:
                break
            h_new, _ = lp_step(h_cur, n, eps_f, top_k=32)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new

    if time.time() < dl - 5:
        h_fin = lbfgsb_augmented(best_h, n, 2000000, rho0=500.0,
                                 max_outer=5, max_inner=5000, tol=1e-10)
        c5_new = compute_c5(h_fin, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
