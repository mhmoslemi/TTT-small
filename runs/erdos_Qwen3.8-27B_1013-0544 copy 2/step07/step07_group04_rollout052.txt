def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.array(v, dtype=float).copy()
    target = 0.5 * n
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = target - v.sum()
        if abs(d) < 1e-14:
            break
        mask = (v > 1e-12) & (v < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            v[mask] += d / cnt
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj_grad(h, n, alpha):
    c = compute_c(h, n)
    m = float(c.max())
    if alpha < 1e-3:
        return m, c - m
    sh = np.clip(alpha * (c - m), -50, 500)
    e = np.exp(sh)
    se = e.sum()
    if se < 1e-300:
        return m, np.zeros_like(c)
    obj = m + np.log(se) / alpha
    q = e / se
    dx = 2.0 / n
    off = n - 1
    grad = np.zeros(n)
    nz = np.flatnonzero(q > 1e-12)
    for idx in nz:
        k = idx - off
        w = q[idx]
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            ml = n - k
            grad[k:] += w * (1.0 - h[:ml])
            grad[:ml] -= w * h[k:]
        else:
            k2 = -k
            ml = n - k2
            grad[:ml] += w * (1.0 - h[k2:])
            grad[k2:] -= w * h[:ml]
    return obj, dx * grad

def pair_search_fast(h, n, c, i, j):
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return False
    off = n - 1
    dx = 2.0 / n
    L = 2 * n - 1
    slope = np.zeros(L)
    for m in range(L):
        p = m - off
        ci = 0.0
        if 0 <= i + p < n:
            ci += 1.0 - h[i + p]
        if 0 <= i - p < n:
            ci -= h[i - p]
        cj = 0.0
        if 0 <= j + p < n:
            cj += 1.0 - h[j + p]
        if 0 <= j - p < n:
            cj -= h[j - p]
        slope[m] = dx * (ci - cj)
    x0 = h[i]
    a, b = lo, hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float((c + (c1 - x0) * slope).max())
    f2 = float((c + (c2 - x0) * slope).max())
    for _ in range(35):

            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float((c + (c1 - x0) * slope).max())
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float((c + (c2 - x0) * slope).max())
    xb = 0.5 * (a + b)
    fb = float((c + (xb - x0) * slope).max())
    f0 = float(c.max())
    if fb < f0 - 1e-14:
        h[i] = xb
        h[j] = S - xb
        c[:] += (xb - x0) * slope
        return True
    return False

def pair_descent(h, n, deadline, seed, max_no=600):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    no_imp = 0
    while time.time() < deadline and no_imp < max_no:
        if rng.random() < 0.8:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        if pair_search_fast(h, n, c, i, j):
            v = float(c.max())
            if v < best_c:
                best_c = v
                best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
    return project_box_sum(best_h, n), float(compute_c(best_h, n).max())

    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)
