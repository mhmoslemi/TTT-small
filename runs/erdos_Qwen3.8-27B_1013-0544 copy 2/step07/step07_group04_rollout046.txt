import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0.0:
        return m
    return m + np.log(s) / alpha

def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            s = k; m = n - s
            g[s:n] += w * (1.0 - x[:m])
            g[:m] -= w * x[s:n]
        elif k < 0:
            s = -k; m = n - s
            g[:m] += w * (1.0 - x[s:n])
            g[s:n] -= w * x[:m]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def sum_fun(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones(n)

def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 4 * maxiter,
                                "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_fun, "args": (n,), "jac": sum_jac}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c_base, i, j, n):
    x0 = float(h[i]); y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = np.linspace(lo, hi, 33)
    vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)
    k = int(np.argmin(vals))
    for _ in range(3):
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-12:
            break
        xs = np.linspace(a, b, 25)
        vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)
        k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-14:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        return xb, yb, bv
    return None

def pair_local_search(h, deadline, seed=1):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    if n < 2 or time.time() > deadline:
        return project_box_sum(besth, n), float(compute_c(besth, n).max())
    rng = np.random.RandomState(seed)
    noimp = 0
    off = n - 1
    sweep_d = 1
    while time.time() < deadline:
        if noimp > 600:
            # Add perturbation to escape local minimum
            h = h + 0.005 * rng.randn(n)
            h = project_box_sum(h, n)
            c = compute_c(h, n)
            noimp = 0
        mode = rng.random()
        if mode < 0.6:
            # Weighted random pair
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w = w / w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
            i = int(rng.randint(n - d))
        elif mode < 0.85:
            # Systematic sweep: pick next offset, random position
            i = int(rng.randint(n - sweep_d))
            d = sweep_d
            sweep_d += 1
            if sweep_d >= n:
                sweep_d = 1
        else:
            # Pure random
            d = int(rng.randint(1, n))
            i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i] = x; h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interpolate_h(h, nold, nnew):
    if nold == nnew:

    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)
