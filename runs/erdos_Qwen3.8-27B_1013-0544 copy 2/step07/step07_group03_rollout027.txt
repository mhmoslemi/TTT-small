    return float(np.max(c) * dx)

def get_corr(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1-h)[:int(abs(diff))+1]
            h[idx] = np.minimum(1.0, h[idx] + abs(diff)/len(idx))
        else:
            idx = np.argsort(h)[:int(abs(diff))+1]
            h[idx] = np.maximum(0.0, h[idx] + diff/len(idx))
    return h

def subgrad_step(h, n, top_k, lr):
    c = get_corr(h, n)
    top_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad[:] += 1.0 - 2.0 * h
            else:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            kk = -k
            if kk == 0:
                grad[:] += 1.0 - 2.0 * h
            else:
                grad[:n-kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n-kk]
    grad *= (2.0/n) / top_k
    return h - lr * grad, c.max()

def subgrad_optimize(h, n, deadline, max_iters=50000, top_k=5, lr_base=0.05):
    best_h = h.copy()
    best_c = float(get_corr(h, n).max())
    momentum = np.zeros(n)
    beta = 0.85
    G = np.zeros(n)
    t = 0
    stall = 0
    stall_limit = 3000
    rng = np.random.RandomState(123)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = get_corr(h, n)
        cur_c = float(c.max())
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best_h + 0.02 * rng.randn(n)
            h = project_feasible(h, n)
            G = np.zeros(n)
            momentum = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            if k >= 0:
                if k == 0:
                    grad[:] += 1.0 - 2.0 * h
                else:
                    grad[k:] += (1.0 - h[:n-k])
                    grad[:n-k] -= h[k:]
            else:
                kk = -k
                if kk == 0:
                    grad[:] += 1.0 - 2.0 * h
                else:
                    grad[:n-kk] += (1.0 - h[kk:])
                    grad[kk:] -= h[:n-kk]
        grad *= (2.0/n) / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum**2
        lr = lr_base / (np.sqrt(G) + 1e-7)
        h_new = project_feasible(h - lr * momentum, n)
        c_check = get_corr(h_new, n).max()
        if c_check < cur_c + 0.01:
            h = h_new
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=1000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        lse = m + sm/alpha + np.log(np.exp(shifted-sm).sum())/alpha
        return lse
    def obj_grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        weights = np.exp(shifted - sm)
        weights /= weights.sum()
        # gradient of lse = d/dx [m + (1/alpha)log(sum exp(alpha(c-m)))]
        # = (1/alpha) * sum(weights_k * alpha * dc_k/dx) + dc_max/dx * (1 - sum(weights_k))
        # approximately = sum(weights_k * dc_k/dx) (since weights sum to 1)
        grad = np.zeros(n)
        for idx, w in zip(np.arange(len(c)), weights):
            if w < 1e-10:
                continue
            k = idx - (n-1)
            if k >= 0:
                if k == 0:
                    grad += w * (1.0 - 2.0*x)
                else:
                    grad[k:] += w * (1.0 - x[:n-k])
                    grad[:n-k] -= w * x[k:]
            else:
                kk = -k
                if kk == 0:
                    grad += w * (1.0 - 2.0*x)
                else:
                    grad[:n-kk] += w * (1.0 - x[kk:])
                    grad[kk:] -= w * x[:n-kk]
        grad *= dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        # Use SLSQP
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
