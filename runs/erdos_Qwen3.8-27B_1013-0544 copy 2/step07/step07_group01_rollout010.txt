    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            rs = room.sum()
            if rs < 1e-14:
                break
            h = h + diff / rs * room
        else:
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(x, n, alpha, penalty=0.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    p = np.exp(shifted - sm)
    ps = p.sum()
    lse_val = (sm + np.log(ps)) / alpha
    obj = m + lse_val
    if penalty > 0:
        obj += penalty * (x.sum() - n / 2.0) ** 2
    
    # Gradient
    p = p / ps
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-12:

                grad[l:] -= w * x[:n - l]

                grad[ll:] += w * (1.0 - x[:n - ll])
                grad[:n - ll] -= w * x[ll:]
    grad *= dx
    if penalty > 0:

    return best_h, best_c

    dx = 2.0 / n

    sm = sh.max()

    if time.time() < deadline - 60:
        for nl in [200, 300, 500]:
            if time.time() > deadline - 55:
                break
            if nl == n:
                continue
            xo = np.linspace(0, 2, n)
            xl = np.linspace(0, 2, nl)
            hl = np.interp(xl, xo, best_h)
            hl = project_feasible(hl, nl)
            for a in [10000, 100000, 1000000, 10000000]:
                if time.time() > deadline - 50:
                    break
                hl = lbfgs_opt(hl, nl, a, 5.0, 3000)
            ch = compute_c5(hl, nl)
            if ch < best_c:
                best_c, best_h, n = ch, hl.copy(), nl

    # Final polish
    if time.time() < deadline - 10:
        hp = lbfgs_opt(best_h, n, 500000000, 20.0, 3000)
        cp = compute_c5(hp, n)
        if cp < best_c:
            best_c, best_h = cp, hp.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
