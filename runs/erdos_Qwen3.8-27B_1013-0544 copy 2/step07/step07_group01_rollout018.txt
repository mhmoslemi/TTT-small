from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    am = ac.max()
    return (am + np.log(np.exp(ac - am).sum())) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ac = alpha * c
    am = ac.max()
    p = np.exp(ac - am)
    ps = p.sum()
    w = p / ps
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[:n - lag] += wk * dx * (1.0 - x[lag:])
            grad[lag:] -= wk * dx * x[:n - lag]
        elif -n < lag < 0:
            ll = -lag
            grad[ll:] += wk * dx * (1.0 - x[:n - ll])
            grad[:n - ll] -= wk * dx * x[ll:]
    return grad

def lbfgs_opt(h, n, alpha, maxiter=2000, ftol=1e-15, gtol=1e-8):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_opt(h, n, alpha, maxiter=1000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad(h, n, deadline, alpha=5000, mu=0.92):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 3.0
    mom = np.zeros(n)
    stall = 0
    for _ in range(1000000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        ac = alpha * c
        am = ac.max()
        p = np.exp(ac - am)
        ps = p.sum()
        if ps < 1e-30:
            break
        w = p / ps
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[:n - lag] += w[k] * (1.0 - h[lag:])
                grad[lag:] -= w[k] * h[:n - lag]
            elif -n < lag < 0:
                ll = -lag
                grad[ll:] += w[k] * (1.0 - h[:n - ll])
                grad[:n - ll] -= w[k] * h[ll:]
        grad *= dx
        mom = mu * mom - lr * grad
        h_new = project_feasible(h + mom, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.01, dx * 10.0)
        else:
            stall += 1
            if stall > 200:
                lr *= 0.9
                if lr < dx * 0.05:
                    lr = dx * 1.0
                    stall = 0
        h = h_new
    return best_h, best_c

def coarse_to_fine(h_start, n_target, deadline):

    grids = []

        g *= 2

    if grids[0] > n_cur:
        # Downsample to start
        h = np.interp(np.linspace(0, 2, grids[0]), np.linspace(0, 2, n_cur), h)
        h = project_feasible(h, grids[0])
        n_cur = grids[0]
    
    for ng in grids:
        if time.time() > deadline:
            break
        if ng != n_cur:
            h = np.interp(np.linspace(0, 2, ng), np.linspace(0, 2, n_cur), h)
            h = project_feasible(h, ng)

        # Optimize at this resolution

        h = np.interp(np.linspace(0, 2, n_target), np.linspace(0, 2, n_cur), h)
        h = project_feasible(h, n_target)
    return h
