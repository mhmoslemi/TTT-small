import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full") * (2.0 / n)

def project_to_binary(h, n):
    """Project a continuous h to binary with exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(-h)[:n//2]
    out = np.zeros(n, dtype=float)
    out[idx] = 1.0
    return out

def binary_local_search(h, n, seed, deadline):
    """Directed local search for binary h."""
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    c = compute_c(h, n)
    c_max = float(c.max())
    best_h = h.copy()
    best_c = c_max
    no_improve = 0
    
    while time.time() < deadline and no_improve < 5000:
        # Weight lags by their current value (focus on bottleneck)
        c_pos = c.copy()
        c_pos[c_pos < 0] = 0
        w = c_pos + 1e-10
        w = w / w.sum()
        
        # Pick a lag
        k = int(rng.choice(len(c), p=w))
        # In full correlation, index k corresponds to lag k - (n-1)
        lag = k - (n - 1)
        
        if lag == 0:
            # Zero lag is always 0 for binary, skip
            no_improve += 1
            continue
        
        # Find positions contributing to this lag
        if lag > 0:
            # "1-to-0": h[i]=1, h[i+lag]=0 for i in [0, n-1-lag]
            valid_i = np.arange(0, n - lag)
        else:
            # "0-to-1": h[i]=0, h[i+|lag|]=1 for i in [0, n-1+lag]
            lag_abs = -lag
            valid_i = np.arange(0, n - lag_abs)
        
        if len(valid_i) == 0:
            no_improve += 1
            continue
        
        if lag > 0:
            contributors = valid_i[(h[valid_i] == 1) & (h[valid_i + lag] == 0)]
        else:
            lag_abs = -lag
            contributors = valid_i[(h[valid_i] == 0) & (h[valid_i + lag_abs] == 1)]
        
        if len(contributors) == 0:
            no_improve += 1
            continue
        
        # Pick a random contributor
        i = int(contributors[rng.randint(len(contributors))])
        
        # Find a swap partner j
        if lag > 0:
            # We want to set h[i]=0. Find j where h[j]=0, j≠i
            # Ideally j is far from i to minimize side effects
            zeros = np.where(h == 0)[0]
            zeros = zeros[zeros != i]
            # Prefer zeros that are far from i and not near i+lag
            if len(zeros) == 0:
                no_improve += 1
                continue
            j = int(zeros[rng.randint(len(zeros))])
        else:
            lag_abs = -lag
            # We want to set h[i]=1. Find j where h[j]=1, j≠i
            ones = np.where(h == 1)[0]
            ones = ones[ones != i]
            if len(ones) == 0:
                no_improve += 1
                continue
            j = int(ones[rng.randint(len(ones))])
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_c(h, n)
        new_max = float(new_c.max())
        
        if new_max < c_max - 1e-15:
            c = new_c
            c_max = new_max
            no_improve = 0
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
        else:
            # Undo
            h[i], h[j] = h[j], h[i]
            no_improve += 1
    
    return best_h, best_c
