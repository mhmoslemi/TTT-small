def compute_grad_max(h, n, k_star):
    offset = k_star  # m index in correlate output
    grad = np.zeros(n)
    # First term: (1-h[j-offset]) for j where 0 <= j-offset < n, i.e., offset <= j < n+offset
    lo1 = max(0, offset)
    hi1 = min(n, n + offset)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - offset])
    # Second term: -h[j+offset] for j where 0 <= j+offset < n, i.e., -offset <= j < n-offset
    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] -= h[idx + offset]
    return grad

                    corr += delta
                    cur_c = nc

                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)

                    break

        if not improved:

            best_h = h.copy()

def proj_subgrad(h, n, dl, step_init=0.5):
    """Projected subgradient method for max of correlations."""
    dx = 2.0 / n
    h = h.copy()
    step = step_init / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(200000):

            break

        k_star = int(c.argmax())

        grad = np.zeros(n)
        lo1 = max(0, offset)
        hi1 = min(n, n + offset)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx - offset])
        lo2 = max(0, -offset)
        hi2 = min(n, n - offset)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx + offset]
        # Step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = float(np.max(np.correlate(h_new, 1-h_new, mode='full')) * dx)
        c_old = float(c[k_star] * dx)
        if c_new < c_old:
            h = h_new
            step = min(step * 1.05, 2.0/n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            step *= 0.7
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
