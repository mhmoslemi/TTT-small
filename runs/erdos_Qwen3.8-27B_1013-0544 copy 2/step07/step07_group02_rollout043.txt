        if abs(diff) < 1e-14:

            if s < 1e-14:

def compute_gradients(h, n):
    """Compute gradient matrix: G[i, m] = d overlap[m] / d h[i]"""
    n_corr = 2 * n - 1
    G = np.zeros((n, n_corr))
    for m in range(n_corr):
        k = m - (n - 1)
        # d/dh[i] of sum_j h[j]*(1-h[j+k]):
        # Term 1: i = j, contributes (1-h[i+k]) if i+k in range
        # Term 2: i = j+k, contributes -h[i-k] if i-k in range
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    n_corr = 2 * n - 1
    G = np.zeros((n, n_corr))
    for m in range(n_corr):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = np.array(delta.value).flatten()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try

    h_sg, c5_sg = subgrad_descent(h_sg, n, num_steps=3000, lr0=0.1, decay=0.9998)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 4: LP from subgradient result + noise restarts
    p4_dl = t0 + 0.82 * total
    h = best_h.copy()
    eps = 0.03
    for trial in range(50):
        if time.time() > p4_dl:
            break
        if trial > 0:
            noise_scale = 0.02 * (1.0 + 0.3 * rng.rand())
            h = best_h + noise_scale * rng.randn(n)
            h = project_feasible(h, n)
            eps = 0.04
        for _ in range(200):
            if time.time() > p4_dl:
                break
            h_new, mc_new = lp_step(h, n, eps)
            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            eps = max(eps * 0.998, 0.003)

    # Phase 5: Multi-resolution
    p5_dl = t0 + 0.90 * total
    for n_try in [200, 300, 500]:
        if time.time() > p5_dl - 45:
            break
        xo = np.linspace(0, 2, best_n)
        xn = np.linspace(0, 2, n_try)
        h_try = np.interp(xn, xo, best_h)
        h_try = project_feasible(h_try, n_try)
        c5_try = compute_c5(h_try, n_try)
        eps_r = 0.03
        for _ in range(150):
            if time.time() > p5_dl - 30:
                break
            h_try, mc_new = lp_step(h_try, n_try, eps_r)
            c5_try = mc_new * 2.0 / n_try
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            h_try = project_feasible(h_try, n_try)
            eps_r = max(eps_r * 0.998, 0.003)

    # Phase 6: SLSQP final polish with increasing sharpness
    if time.time() < dl - 8:
        h_fin = best_h.copy()
        for alpha in [20000, 200000, 2000000]:
            if time.time() > dl - 8:
                break
            h_fin = slsqp_optimize(h_fin, best_n, alpha, maxiter=800)
            c5_fin = compute_c5(h_fin, best_n)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 7: Final tight LP
    if time.time() < dl - 3:
        h = best_h.copy()
        eps = 0.004
        while time.time() < dl - 3:
            h_new, mc_new = lp_step(h, best_n, eps)
            c5_new = mc_new * 2.0 / best_n
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            eps = max(eps * 0.998, 0.0005)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
