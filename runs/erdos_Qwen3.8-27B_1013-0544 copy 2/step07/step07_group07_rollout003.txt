    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def smooth_autocorr_grad(h, n, beta):
    A = np.correlate(h, h, mode='full')
    maxA = A.max()
    ev = np.exp(-beta * (A - maxA))
    p = ev / ev.sum()
    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)[:, None]
    lag_idx = lags[None, :]
    idx1 = j_idx - lag_idx
    idx2 = j_idx + lag_idx
    g = np.zeros(n)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g += np.where(v1, h[np.clip(idx1, 0, n - 1)] * p[None, :], 0.0).sum(axis=1)
    g += np.where(v2, h[np.clip(idx2, 0, n - 1)] * p[None, :], 0.0).sum(axis=1)
    return g

def adam_smooth_ascend(h, n, beta_schedule, steps_per_beta, lr_start, lr_end):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    total_betas = len(beta_schedule)
    for bi, beta in enumerate(beta_schedule):
        lr = lr_start * (lr_end / lr_start) ** (bi / max(total_betas - 1, 1))
        for _ in range(steps_per_beta):
            t += 1
            g = smooth_autocorr_grad(h, n, beta)
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g ** 2
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            h = h + lr * mh / (np.sqrt(vh) + eps)
            h = project_feasible(h, n)
    return h

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

def barrier_slsqp(h, n, t, maxiter, ftol):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        excess = np.maximum(c - t, 0)
        return 0.5 * np.sum(excess ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greed_binary(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_set = set(int(x) for x in np.where(h == 1)[0])
    zeros_set = set(int(x) for x in np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in list(zeros_set):
            if time.time() > deadline:
                break
            for j in list(ones_set):
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_set.discard(j)
                    zeros_set.discard(i)
                    ones_set.add(i)
                    zeros_set.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def gradient_path(h, n, dl, max_beta=20000, steps=150):
    betas = np.logspace(np.log10(5), np.log10(max_beta), 8)
    for beta in betas:
        if time.time() > dl - 3:
            break
        h = adam_smooth_ascend(h, n, [beta], steps, 0.02, 0.005)
    return h

        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    p4 = t0 + 0.85 * total

        0.02 * np.sin(2 * np.pi * 3.0 * x + 0.5),
        0.015 * np.cos(2 * np.pi * 7.0 * x),

    best_c = compute_c5(best_h, n)
