    h = np.round(h)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n//2:
        # Remove excess ones
        excess = len(ones) - n//2
        to_remove = rng.choice(ones, excess, replace=False)
        h[to_remove] = 0
        zeros = np.where(h == 0)[0]
    elif len(ones) < n//2:
        deficit = n//2 - len(ones)
        to_add = rng.choice(zeros, deficit, replace=False)
        h[to_add] = 1
        ones = np.where(h == 1)[0]
    
    c = np.correlate(h, 1-h, mode='full')
    best_c = c.copy()
    best_h = h.copy()
    best_max = c.max()
    
    for it in range(max_iters):
        T = T_init * (T_final/T_init) ** (it / max_iters)
        m_star = c.argmax()
        
        # Compute f(a) and g(b)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # f(a) = h[a-m*] + h[a+m*] - 1, with boundary=0
        a_m = ones_idx - m_star
        a_p = ones_idx + m_star
        f_vals = -1.0
        valid_m = (a_m >= 0) & (a_m < n)
        f_vals[valid_m] += h[a_m[valid_m]]
        valid_p = (a_p >= 0) & (a_p < n)
        f_vals[valid_p] += h[a_p[valid_p]]
        
        # g(b) = 1 - h[b+m*] - h[b-m*], with boundary=0
        b_p = zeros_idx + m_star
        b_m = zeros_idx - m_star
        g_vals = 1.0
        valid_bp = (b_p >= 0) & (b_p < n)
        g_vals[valid_bp] -= h[b_p[valid_bp]]
        valid_bm = (b_m >= 0) & (b_m < n)
        g_vals[valid_bm] -= h[b_m[valid_bm]]
        
        # Try top candidates
        best_improvement = 0
        best_swap = None
        K = min(20, len(ones_idx))
        K2 = min(20, len(zeros_idx))
        top_a = ones_idx[np.argsort(f_vals)[:K]]
        top_b = zeros_idx[np.argsort(g_vals)[:K2]]
        
        for a in top_a:
            for b in top_b:
                if a == b:
                    continue
                # Compute full delta
                delta = np.zeros(2*n-1)
                # a: 1→0
                am_all = a + np.arange(-(n-1), n)
                valid = (am_all >= 0) & (am_all < n)
                delta[valid] -= (1 - h[am_all[valid]])
                am2_all = a - np.arange(-(n-1), n)
                valid2 = (am2_all >= 0) & (am2_all < n)
                delta[valid2] += h[am2_all[valid2]]
                # b: 0→1
                bm_all = b + np.arange(-(n-1), n)
                valid3 = (bm_all >= 0) & (bm_all < n)
                delta[valid3] += (1 - h[bm_all[valid3]])
                bm2_all = b - np.arange(-(n-1), n)
                valid4 = (bm2_all >= 0) & (bm2_all < n)
                delta[valid4] -= h[bm2_all[valid4]]
                
                new_max = np.max(c + delta)
                if new_max < c.max() - 1e-12:
                    improvement = c.max() - new_max
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_swap = (a, b)
                elif T > 0.0001 and rng.rand() < np.exp(-0.5/T):
                    # Accept with probability (simulated annealing)
                    if best_swap is None:
                        best_swap = (a, b)
                        best_improvement = -1  # mark as SA acceptance
        
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            c = np.correlate(h, 1-h, mode='full')
            if c.max() < best_max:
                best_max = c.max()
                best_h = h.copy()
    
    return best_h, best_max

    p1_dl = t0 + 0.25 * total
    h_b = binary_project(h, n, rng)
    mc_b = float(np.correlate(h_b, 1 - h_b, mode='full').max())
    c5_b = mc_b * 2.0 / n
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_mc = mc_b
    
    n_iters_p1 = 200000
    while time.time() < p1_dl:
        remaining = p1_dl - time.time()
        iters = min(n_iters_p1, int(remaining * 5000))
        if iters < 1000:
            break
        h_b, mc_b = binary_sa(h_b, n, rng, iters, 1.0, 0.05, 10)
        c5_b = mc_b * 2.0 / n
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_mc = mc_b
        n_iters_p1 = max(10000, n_iters_p1 // 3)
    
    # Phase 2: LP polish (20% time)
    p2_dl = t0 + 0.45 * total
    h_cur = best_h.copy()
    eps = 0.02
    for _ in range(500):
        if time.time() > p2_dl:
            break
        h_cur, mc_new = lp_step(h_cur, n, eps)
        c5_new = mc_new * 2.0 / n
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_cur.copy()
            best_mc = mc_new
        elif mc_new > best_mc - 1e-6:
            eps = max(eps * 0.75, 0.002)
    
    # Phase 3: Multiple restarts (35% time)
    p3_dl = t0 + 0.80 * total
    for trial in range(30):
        if time.time() > p3_dl:
            break
        # Different perturbation strategies
        if trial % 3 == 0:
            h_start = best_h + 0.08 * rng.randn(n)
        elif trial % 3 == 1:
            h_start = best_h + 0.15 * rng.randn(n)
        else:
            # Random restart with structure
            h_start = np.full(n, 0.5)
            h_start += 0.3 * rng.randn(n)
        
        h_start = project_feasible(h_start, n)
        h_b = binary_project(h_start, n, rng)
        
        iters_sa = 50000
        while time.time() < p3_dl - 5:
            h_b, mc_b = binary_sa(h_b, n, rng, iters_sa, 0.5, 0.02, 8)
            c5_b = mc_b * 2.0 / n
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_mc = mc_b
            iters_sa = max(5000, iters_sa // 2)
            if iters_sa <= 5000:
                break
        
        # Quick LP polish
        h_cur = h_b.copy()
        for _ in range(20):
            if time.time() > p3_dl - 2:
                break
            h_cur, mc_new = lp_step(h_cur, n, 0.01)
            c5_new = mc_new * 2.0 / n
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
                best_mc = mc_new
    
    # Phase 4: Final fine LP polish (20% time)
    h_cur = best_h.copy()
    eps = 0.005
    for _ in range(500):
        if time.time() > deadline:
            break
        h_cur, mc_new = lp_step(h_cur, n, eps)
        c5_new = mc_new * 2.0 / n
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_cur.copy()
            best_mc = mc_new
        eps = max(eps * 0.997, 0.0003)
    
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
