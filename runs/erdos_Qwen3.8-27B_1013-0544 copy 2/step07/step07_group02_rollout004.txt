def compute_gradients(h, n):

            s = room.sum()

                break

        else:

            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):

    G = np.zeros((n, n_full))
    for m in range(n_full):

def weighted_grad(h, n, w):
    """Compute sum_m w[m] * G[:,m] without building full G."""
    n_full = 2 * n - 1
