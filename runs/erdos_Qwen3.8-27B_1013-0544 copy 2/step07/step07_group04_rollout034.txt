    # Find top-k active indices
    idx = np.argsort(c)[::-1][:top_k]
    # Candidate points: corners + pairwise intersections
    candidates = [(a_lo, b_lo), (a_lo, b_hi), (a_hi, b_lo), (a_hi, b_hi)]
    # Also the current point
    candidates.append((0.0, 0.0))
    # Pairwise intersections of active constraints
    for p in range(len(idx)):
        for q in range(p + 1, len(idx)):
            m1, m2 = idx[p], idx[q]
            A = np.array([gi[m1], gj[m1]], dtype=float)
            B = np.array([gi[m2], gj[m2]], dtype=float)
            det = A[0] * B[1] - A[1] * B[0]
            if abs(det) < 1e-14: continue
            # c[m1]+a*A[0]+b*A[1] = c[m2]+a*B[0]+b*B[1]
            rhs = c[m2] - c[m1]
            # a*(A[0]-B[0]) + b*(A[1]-B[1]) = rhs
            dA = A - B
            det2 = dA[0] * 1.0 - dA[1] * 0.0  # line: dA[0]*a + dA[1]*b = rhs
            if abs(dA[0]) > 1e-14:
                # For each b in [b_lo, b_hi], a = (rhs - dA[1]*b) / dA[0]
                # Check if a is in [a_lo, a_hi]
                for bp in [b_lo, b_hi, 0.0]:
                    ap = (rhs - dA[1] * bp) / dA[0]
                    if a_lo - 1e-9 <= ap <= a_hi + 1e-9:
                        candidates.append((np.clip(ap, a_lo, a_hi), np.clip(bp, b_lo, b_hi)))
            elif abs(dA[1]) > 1e-14:
                for ap in [a_lo, a_hi, 0.0]:
                    bp = (rhs - dA[0] * ap) / dA[1]
                    if b_lo - 1e-9 <= bp <= b_hi + 1e-9:
                        candidates.append((np.clip(ap, a_lo, a_hi), np.clip(bp, b_lo, b_hi)))
    # Evaluate all candidates
    best_val = float(c.max())
    best_ab = (0.0, 0.0)
    for ap, bp in candidates:
        ap = float(np.clip(ap, a_lo, a_hi))
        bp = float(np.clip(bp, b_lo, b_hi))
        val = float((c + ap * gi + bp * gj).max())
        if val < best_val - 1e-10:
            best_val = val
            best_ab = (ap, bp)
    if best_val < float(c.max()) - 1e-10:
        return float(x0 + best_ab[0]), float(y0 + best_ab[1]), best_val
    return None

def pair_local_search(h, deadline, seed=1, use_2d=True):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    noimp = 0
    maxno = 1200
    while time.time() < deadline and noimp < maxno:
        # Choose lag
        if rng.random() < 0.8:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = None
        if use_2d and rng.random() < 0.4:
            res = pair_2d_search(h, c, i, j, n)
        if res is None:
            res = pair_1d_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interpolate_h(h, nold, nnew):
    if nold == nnew: return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    work = h.copy()
    workc = bestc
    if n < 4 or time.time() > deadline:
        return project_box_sum(besth, n), float(compute_c(besth, n).max())
    phases = (
        (50.0, 50.0, 300),
        (200.0, 200.0, 250),
        (800.0, 800.0, 200),
        (3000.0, 3000.0, 150),
        (15000.0, 15000.0, 120),
    )
    for alpha, mu, maxit in phases:
        if time.time() > deadline - 10: break
        h2 = project_box_sum(work, n)
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(pen_obj, h2, args=(n, alpha, mu), method="L-BFGS-B",
                           jac=pen_grad, bounds=bounds,
                           options={"maxiter": maxit, "maxfun": 3 * maxit,
                                    "ftol": 1e-15, "gtol": 1e-8})
            if res.x is not None: work = project_box_sum(res.x, n)
        except Exception:
            pass
        v2 = float(compute_c(work, n).max())
        if v2 < bestc:
            bestc = v2
            besth = work.copy()
        if v2 <= workc + 1e-6:
            workc = v2
        else:
            work = besth.copy()
            workc = bestc
    # Pair search
    pdl = deadline - 20
    if time.time() < pdl:
        besth, bestc = pair_local_search(besth, pdl, seed=seed, use_2d=True)
        work = besth.copy()
        workc = bestc
    # Extra L-BFGS
    for alpha, mu, maxit in ((15000.0, 15000.0, 100), (60000.0, 60000.0, 80)):
        if time.time() > deadline - 8: break
        h2 = project_box_sum(work, n)
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(pen_obj, h2, args=(n, alpha, mu), method="L-BFGS-B",
                           jac=pen_grad, bounds=bounds,
                           options={"maxiter": maxit, "maxfun": 2 * maxit,
                                    "ftol": 1e-15, "gtol": 1e-8})
            if res.x is not None: work = project_box_sum(res.x, n)
        except Exception:
            pass
        v2 = float(compute_c(work, n).max())
        if v2 < bestc:
            bestc = v2
            besth = work.copy()
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def final_polish(h, n, seed, deadline):
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    curh = h.copy()
    curc = bestc
    for alpha, maxit in ((5000, 150), (20000, 100), (80000, 80)):
        if time.time() > deadline - 10: break
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(pen_obj, curh, args=(n, alpha, alpha), method="L-BFGS-B",
                           jac=pen_grad, bounds=bounds,
                           options={"maxiter": maxit, "maxfun": 2 * maxit,
                                    "ftol": 1e-15, "gtol": 1e-9})
            if res.x is not None: curh = project_box_sum(res.x, n)
        except Exception:
            pass
        v2 = float(compute_c(curh, n).max())
        if v2 < bestc:
            bestc = v2
            besth = curh.copy()
        curc = v2
    if time.time() < deadline - 5:
        besth, bestc = pair_local_search(besth, deadline - 1, seed=seed + 200, use_2d=True)
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.49 * np.sin(3.0 * np.pi * x)
    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0
    # Try multiple n values
    n_candidates = [n0]
    for m in (300, 350, 450):
        if m not in n_candidates and 8 <= m <= 600:
            n_candidates.append(m)
    n_candidates = sorted(set(n_candidates))
    L = len(n_candidates)
    # Allocate time
    if L == 1:
        fracs = [0.90]
    else:
        fracs = [0.40] + [(0.75 - 0.40) / (L - 1)] * (L - 1)
    span = max(1.0, deadline - t0)
    for k_idx, nh in enumerate(n_candidates):
        if time.time() > deadline - 15: break
        dl = t0 + span * min(0.88, sum(fracs[:k_idx + 1])) - 3
        dl = min(dl, deadline - 12)
        if dl < time.time() + 8: continue
        if k_idx == 0:
            hinit = besth.copy()
        else:
            hinit = interpolate_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        hopt, copt = optimize_n(hinit, nh, seed + 13 * k_idx, dl)
        if copt < bestc:
            bestc = copt
            besth = hopt.copy()
            bestn = nh
    # Final polish on best
    if time.time() < deadline - 10:
        hfin, cfin = final_polish(besth, bestn, seed + 777, deadline - 2)
        if cfin < bestc:
            bestc = cfin
            besth = hfin.copy()
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
