def c5val(h, n):

def proj(a, n):
    a = np.asarray(a, float).ravel()
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]

    lo, hi = a.min() - 0.5, a.max() + 0.5

        if np.clip(a - m, 0, 1).sum() > t: lo = m
        else: hi = m
    h = np.clip(a - (lo + hi) * 0.5, 0, 1)
    for _ in range(50):
        d = t - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1 - h; s = r.sum()
            if s < 1e-14: break
            h += d * r / s
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d * h / s
    h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0: h[int(np.argmax(1 - h))] += d
        else: h[int(np.argmax(h))] += d
    return np.clip(h, 0, 1)

def lse_fg(x, n, beta, pen):
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, mode="full") * dx, 0)
    m = c.max()
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(m), np.zeros(n)
    p = e / s
    F = m + np.log(s) / beta
    A = np.convolve(p[::-1], 1 - h, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if pen > 0:
        d = h.sum() - n * 0.5
        F += 0.5 * pen * d * d
        g = g + pen * d
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(F), g

    bnds = [(0, 1)] * n

        if time.time() > dl - 5: break
        _st = min(dl - 2, time.time() + per)
        try:
            r = minimize(lse_fg, h, method="L-BFGS-B", jac=True,
                         bounds=bnds, args=(n, bt, pen), callback=_cb,
                         options={"maxiter": 200000, "ftol": 1e-15, "gtol": 1e-10, "maxls": 30})
            h = proj(r.x, n)
        except Exception:
            pass
        c = c5val(h, n)
        if c < bc:
            bc, best = c, h.copy()
        # Small perturbation to escape plateaus
        rng = np.random.RandomState(int(bt * 13 + 7))
        h = proj(h + 1e-5 * rng.randn(n), n)
    return best, bc

                base_i[k, mb] -= hb[b[mb]]
