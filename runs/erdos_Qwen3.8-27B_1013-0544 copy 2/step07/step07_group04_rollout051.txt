    f = 1.0 - 2.0 * h
    delta_grid = np.linspace(lo_d, hi_d, 256)
    vals = pair_fast_eval(c, f, i, j, delta_grid)
    k = np.argmin(vals)
    if k == 0 or k == len(delta_grid) - 1:
        return None
    a, b = delta_grid[k - 1], delta_grid[k + 1]
    delta_fine = np.linspace(a, b, 128)
    vals_fine = pair_fast_eval(c, f, i, j, delta_fine)
    k2 = np.argmin(vals_fine)
    best_val = float(vals_fine[k2])
    if best_val < float(c.max()) - 1e-12:
        d_best = float(delta_fine[k2])
        return x0 + d_best, y0 - d_best, best_val
    return None

def local_search(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    best = h.copy()
    bestc = float(c.max())
    off = n - 1
    no_imp = 0
    max_no_imp = 20000
    restart_count = 0

    while time.time() < deadline:
        if no_imp > max_no_imp:
            if restart_count < 20:
                h = project_box_sum(best + 0.02 * rng.randn(n), n)
                c = compute_c(h, n)
                no_imp = 0
                restart_count += 1
            else:
                break

        pos = c[off + 1:]
        neg = c[off - 1::-1]
        w = np.maximum(pos, neg) + 1e-10
        w /= w.sum()
        d = int(rng.choice(n - 1, p=w)) + 1
        i = int(rng.randint(n - d))
        j = i + d

        res = pair_optimize(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i] = x
            h[j] = y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc - 1e-14:
                bestc = v
                best = h.copy()
            no_imp = 0
        else:
            no_imp += 1

    best = project_box_sum(best, n)
    bestc = float(compute_c(best, n).max())
    return best, bestc

def subgrad_optimize(h, n, deadline, k=8):
    h = project_box_sum(h, n)
    dx = 2.0 / n
    off = n - 1
    best = h.copy()
    bestc = float(compute_c(h, n).max())
    v_mom = np.zeros(n)
    step = 0.003
    it = 0

    while time.time() < deadline:
        c = compute_c(h, n)
        sorted_c = np.sort(c)
        threshold = sorted_c[-k]
        active = c >= threshold - 1e-12
        n_act = int(active.sum())
        g = np.zeros(n)
        for idx in np.where(active)[0]:
            lag = idx - off
            w = 1.0 / n_act
            if lag == 0:
                g += w * (1.0 - 2.0 * h)
            elif lag > 0:
                g[:n - lag] += w * (1.0 - h[lag:])
                g[lag:] -= w * h[:n - lag]
            else:
                s = -lag
                g[:n - s] += w * (1.0 - h[s:])
                g[s:] -= w * h[:n - s]
        g *= dx
        v_mom = 0.85 * v_mom - step * g
        h = project_box_sum(h + v_mom, n)
        c_new = float(compute_c(h, n).max())
        if c_new < bestc:
            bestc = c_new
            best = h.copy()
        it += 1
        if it % 100 == 0:
            step *= 0.97
        if it % 500 == 0 and step < 1e-5:
            step = 0.002
            v_mom = np.zeros(n)
            h = project_box_sum(best + 0.01 * np.random.randn(n), n)
    return best, bestc

def slsqp_final(h, n, deadline):
    alpha = 8000.0
    dx = 2.0 / n
    off = n - 1

    def obj(x):
        c = compute_c(x, n)
        m = float(c.max())
        sh = alpha * (c - m)
        return m + float(np.exp(sh).sum()) / alpha

    def grad(x):
        c = compute_c(x, n)
        m = float(c.max())
        sh = alpha * (c - m)
        e = np.exp(sh)
        q = e / e.sum()
        g = np.zeros(n)
        for idx in np.flatnonzero(q > 1e-10):
            lag = idx - off
            w = q[idx]
            if lag == 0:
                g += w * (1.0 - 2.0 * x)
            elif lag > 0:
                g[:n - lag] += w * (1.0 - x[lag:])
                g[lag:] -= w * x[:n - lag]
            else:
                s = -lag
                g[:n - s] += w * (1.0 - x[s:])
                g[s:] -= w * x[:n - s]
        return dx * g

    def sfun(x):
        return x.sum() - 0.5 * n

    def sjac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': sfun, 'jac': sjac}],
                       options={'maxiter': 300, 'ftol': 1e-15})
        if hasattr(res, 'x') and res.x is not None:
            h2 = project_box_sum(res.x, n)
            if compute_c(h2, n).max() < compute_c(h, n).max():
                return h2
    except Exception:
        pass
    return h

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.4 * np.sin(3.0 * np.pi * x)
        hp = project_box_sum(hp, n0)

    besth = hp.copy()
    bestc = float(compute_c(besth, n0).max())
    bestn = n0

    # Phase 1: momentum subgradient (25% time)
    dl1 = t0 + 0.25 * (deadline - t0)
    h1, c1 = subgrad_optimize(besth, bestn, dl1, k=8)
    if c1 < bestc:
        bestc = c1
        besth = h1

    # Phase 2: intensive local search (55% time)
    dl2 = t0 + 0.80 * (deadline - t0)
    h2, c2 = local_search(besth, bestn, dl2, seed=seed + 7)
    if c2 < bestc:
        bestc = c2
        besth = h2

    # Phase 3: try other resolutions (10% time)
    dl3 = t0 + 0.90 * (deadline - t0)
    for m in (320, 480):
        if time.time() > dl3 - 10:
            break
        hint = interpolate_h(besth, bestn, m)
        hint = project_box_sum(hint + 0.003 * rng.randn(m), m)
        hc, cc = local_search(hint, m, dl3, seed=seed + m)
        if cc < bestc:
            bestc = cc
            besth = hc
            bestn = m

    # Phase 4: SLSQP polish + more local search
    if time.time() < deadline - 10:
        hfin = slsqp_final(besth, bestn, deadline - 5)
        cfin = float(compute_c(hfin, bestn).max())
        if cfin < bestc:
            bestc = cfin
            besth = hfin

    if time.time() < deadline - 3:
        hls, cls = local_search(besth, bestn, deadline, seed=seed + 999)
        if cls < bestc:
            bestc = cls
            besth = hls

    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
