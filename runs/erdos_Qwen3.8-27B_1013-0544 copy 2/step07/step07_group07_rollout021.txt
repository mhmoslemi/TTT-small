    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())

    excess = np.maximum(c - t, 0.0)

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):

def run_slsqp_with_obj(h, n, obj_func, maxiter=4000, ftol=1e-16):
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < float('inf'):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def barrier_slsqp(h, n, t, maxiter=4000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - t, 0.0)
        return 0.5 * np.sum(excess ** 2)
    return run_slsqp_with_obj(h, n, obj, maxiter)

def lse_slsqp(h, n, alpha, maxiter=4000):

If I fix which lags are "active" (i.e., achieve the maximum), I can set up:
min C5
s.t. c_k <= C5 for active k
     sum(h) = n/2
     0 <= h <= 1

But c_k is still quadratic. However, if I linearize c_k around the current solution, I get a linear program!

**Linearized barrier approach**:
1. At current solution h*, compute c*_k for all k
2. Identify active set A = {k : c*_k > C5 - epsilon}
3. Linearize c_k around h*: c_k ≈ c*_k + grad_k * (h - h*)
4. Solve: min C5 s.t. c*_k + grad_k*(h-h*) <= C5 for k in A, sum(h)=n/2, 0<=h<=1
5. This is a linear program!
6. Update h and repeat

This is essentially a cutting-plane / sequential linear programming approach. It might work well for the final polish.
