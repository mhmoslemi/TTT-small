def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    return h_b

def find_best_swap(h_b, corr, n):
    """Find the best single swap (i: 0->1, j: 1->0) that minimizes max(corr).
    Returns (i, j, new_max) or (None, None, None) if no improvement."""
    L = 2 * n - 1
    zeros = np.where(h_b == 0)[0]
    ones = np.where(h_b == 1)[0]
    n_z = len(zeros)
    n_o = len(ones)
    cur_max = corr.max()
    
    # Precompute delta contributions
    # For zero i: effect[i, idx] = (1-h[i+idx-(n-1)]) if valid - h[i-idx+(n-1)] if valid
    # For one j: effect_j[idx] = -(1-h[j+idx-(n-1)]) if valid + h[j-idx+(n-1)] if valid
    
    # Vectorized computation
    # For each zero i, the "delta from i" at correlation index idx:
    # = (1-h[i+k]) if 0<=i+k<n where k=idx-(n-1)
    #   - h[i-k] if 0<=i-k<n
    # = (1-h[i+idx-(n-1)]) if n-1-i <= idx < 2n-1-i
    #   - h[i-idx+(n-1)] if i-(n-1) <= idx <= i
    
    # Build using advanced indexing
    # For zeros: positions i, and the range of valid idx
    # This is a band matrix structure
    
    effect_z = np.zeros((n_z, L))
    for iz in range(n_z):
        i = zeros[iz]
        # First term: (1-h[i+k]) for k such that 0<=i+k<n
        lo1 = max(0, n-1-i)  # min idx where i+idx-(n-1) >= 0
        hi1 = min(L, 2*n-1-i)  # max idx where i+idx-(n-1) < n
        if lo1 < hi1:
            p = np.arange(lo1, hi1)
            effect_z[iz, lo1:hi1] += (1.0 - h_b[i + p - (n-1)])
        # Second term: -h[i-k] for k such that 0<=i-k<n
        lo2 = max(0, i-(n-1))
        hi2 = min(L, i+1)
        if lo2 < hi2:
            p = np.arange(lo2, hi2)
            effect_z[iz, lo2:hi2] -= h_b[i - p + (n-1)]
    
    effect_o = np.zeros((n_o, L))
    for jo in range(n_o):
        j = ones[jo]
        # First term: -(1-h[j+k])
        lo1 = max(0, n-1-j)
        hi1 = min(L, 2*n-1-j)
        if lo1 < hi1:
            p = np.arange(lo1, hi1)
            effect_o[jo, lo1:hi1] -= (1.0 - h_b[j + p - (n-1)])
        # Second term: +h[j-k]
        lo2 = max(0, j-(n-1))
        hi2 = min(L, j+1)
        if lo2 < hi2:
            p = np.arange(lo2, hi2)
            effect_o[jo, lo2:hi2] += h_b[j - p + (n-1)]
    
    # For each (i,j) pair: new_max = max(corr + effect_z[i] + effect_o[j])
    best_val = cur_max
    best_iz, best_jo = -1, -1
    for iz in range(n_z):
        a = corr + effect_z[iz]  # shape (L,)
        # For all j: max(a + effect_o[j])
        scores = (a[None, :] + effect_o).max(axis=1)
        j_min = np.argmin(scores)
        if scores[j_min] < best_val:
            best_val = scores[j_min]
            best_iz, best_jo = iz, j_min
    
    if best_iz < 0:
        return None, None, None
    return zeros[best_iz], ones[best_jo], best_val

            lo1 = max(0, n-1-j); hi1 = min(L, 2*n-1-j)

            if cur_c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def compute_swap_delta(i, j, h_b, n):
    """Compute correlation delta for swapping i(0->1) and j(1->0)."""
    L = 2 * n - 1
    delta = np.zeros(L)
    lo = max(0, n - 1 - i); hi = min(L, 2 * n - 1 - i)
    if lo < hi:
        p = np.arange(lo, hi)
        delta[lo:hi] += (1.0 - h_b[i + p - (n - 1)])
    lo = max(0, i - (n - 1)); hi = min(L, i + 1)
    if lo < hi:
        p = np.arange(lo, hi)
        delta[lo:hi] -= h_b[i - p + (n - 1)]
    lo = max(0, n - 1 - j); hi = min(L, 2 * n - 1 - j)
    if lo < hi:
        p = np.arange(lo, hi)
        delta[lo:hi] -= (1.0 - h_b[j + p - (n - 1)])
    lo = max(0, j - (n - 1)); hi = min(L, j + 1)
    if lo < hi:
        p = np.arange(lo, hi)
        delta[lo:hi] += h_b[j - p + (n - 1)]
    return delta

def find_best_swap(h_b, corr, n):
    L = 2 * n - 1
    zeros = np.where(h_b == 0)[0]
    ones = np.where(h_b == 1)[0]
    n_z, n_o = len(zeros), len(ones)
    cur_max = corr.max()
    effect_o = np.zeros((n_o, L))
    for jo in range(n_o):
        j = ones[jo]
        lo = max(0, n - 1 - j); hi = min(L, 2 * n - 1 - j)
        if lo < hi:
            p = np.arange(lo, hi)
            effect_o[jo, lo:hi] -= (1.0 - h_b[j + p - (n - 1)])
        lo = max(0, j - (n - 1)); hi = min(L, j + 1)
        if lo < hi:
            p = np.arange(lo, hi)
            effect_o[jo, lo:hi] += h_b[j - p + (n - 1)]
    best_val = cur_max
    best_iz, best_jo = -1, -1
    for iz in range(n_z):
        i = zeros[iz]
        a_i = corr.copy()
        lo = max(0, n - 1 - i); hi = min(L, 2 * n - 1 - i)
        if lo < hi:
            p = np.arange(lo, hi)
            a_i[lo:hi] += (1.0 - h_b[i + p - (n - 1)])
        lo = max(0, i - (n - 1)); hi = min(L, i + 1)
        if lo < hi:
            p = np.arange(lo, hi)
            a_i[lo:hi] -= h_b[i - p + (n - 1)]
        scores = (a_i[None, :] + effect_o).max(axis=1)
        j_min = np.argmin(scores)
        if scores[j_min] < best_val:
            best_val = scores[j_min]
            best_iz, best_jo = iz, j_min
    if best_iz < 0:
        return None, None
    return zeros[best_iz], ones[best_jo]

def greedy_binary_search(h_init, n, deadline, max_iters=10000):
    dx = 2.0 / n
    h_b = to_binary(h_init, n)
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = find_best_swap(h_b, corr, n)
        if i is None:
            stall += 1
            if stall > 5:
                zeros = np.where(h_b == 0)[0]
                ones = np.where(h_b == 1)[0]
                rng = np.random.RandomState(it * 31 + 7)
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                stall = 0
            else:
                break
        delta = compute_swap_delta(i, j, h_b, n)
        h_b[i] = 1.0
        h_b[j] = 0.0
        corr += delta
        cur_c = corr.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

def simulated_annealing(h_init, n, deadline, max_iters=50000):
    dx = 2.0 / n
    h_b = to_binary(h_init, n)
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    T = 0.5
    T_min = 0.01
    cool_rate = (T_min / T) ** (1.0 / max_iters)
    rng = np.random.RandomState(12345)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.where(h_b == 0)[0]
        ones = np.where(h_b == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = compute_swap_delta(i, j, h_b, n)
        new_c = (corr + delta).max() * dx
        dE = new_c - cur_c
        if dE <= 0 or rng.random() < np.exp(-dE / max(T, 1e-10)):
            h_b[i] = 1.0
            h_b[j] = 0.0
            corr += delta
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h_b.copy()
        T *= cool_rate
    return best_h, best_c
