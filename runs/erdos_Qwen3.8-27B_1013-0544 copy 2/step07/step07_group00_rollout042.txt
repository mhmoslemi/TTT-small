def project_feasible(y, n):
    if n <= 0: return y
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * n
    # If close and bounds okay, fix tiny violations
    if np.all(y >= -1e-15) and np.all(y <= 1.0 + 1e-15):
       y = np.clip(y,0.0,1.0)
       if abs(y.sum()-target) <= 1e-11:
           return y
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
       mid = 0.5*(lo+hi)
       s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
       if s > target:
          lo = mid
       else:
          hi = mid
    lam = 0.5*(lo+hi)
    h = np.clip(y - lam, 0.0, 1.0)
    # correct residual
    diff = target - float(h.sum())
    if diff > 1e-12:
       # add to variables with largest room
       room = 1.0 - h
       order = np.argsort(room)[::-1]
       for idx in order:
          if diff <= 1e-15: break
          add = min(diff, room[idx])
          if add > 0:
             h[idx] += add; diff -= add
    elif diff < -1e-12:
       order = np.argsort(h)[::-1]
       for idx in order:
          if diff >= -1e-15: break
          sub = min(-diff, h[idx])
          if sub > 0:
             h[idx] -= sub; diff += sub
    return h
```

def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    if n <= 0:
        return y
    target = 0.5 * n
    if np.all(y >= -1e-14) and np.all(y <= 1.0 + 1e-14):
        z = np.clip(y, 0.0, 1.0)
        diff = target - float(z.sum())
        if abs(diff) <= 1e-12:
            return z
        step = diff / float(n)
        if float(np.min(z) + step) >= -1e-15 and float(np.max(z) + step) <= 1.0 + 1e-15:
            z = z + step
            return np.clip(z, 0.0, 1.0)
        y = z
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 1e-15: break
                room = 1.0 - float(h[idx])
                if room <= 0.0: continue
                add = min(diff, room)
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                if diff >= -1e-15: break
                room = float(h[idx])
                if room <= 0.0: continue
                sub = min(-diff, room)
                h[idx] -= sub
                diff += sub
    return h
```

def weighted_gradient(h, n, w):
    g = np.zeros(n, dtype=float)
    center = n - 1
    for idx in range(len(w)):
        wv = float(w[idx])
        if wv == 0.0:
            continue
        k = idx - center
        if k >= 0:
            m = n - k
            if m <= 0: continue
            g[k:] += wv * (1.0 - h[:m])
            g[:m] -= wv * h[k:]
        else:
            r = -k
            m = n - r
            if m <= 0: continue
            g[:m] += wv * (1.0 - h[r:])
            g[r:] -= wv * h[:m]
    return g

def smooth_fg(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    maxc = float(np.max(c))
    z = alpha * (c - maxc)
    e = np.exp(z)

def lbfgsb_refine(h, n, pairs, maxiter, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in pairs:
        if time.time() > deadline - 5.0:
            break
        try:
            res = minimize(penalty_obj, best_h, jac=penalty_grad, args=(n, alpha, float(mu)),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': int(maxiter), 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 40})

            continue

        nc = compute_c5(nh, n)
        if nc < best_c:
            best_h = nh
            best_c = nc
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time
import math

def centers(n):
    return (np.arange(n, dtype=float) + 0.5) * (2.0 / n)

def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    target = 0.5 * n
    if np.all(y >= -1e-14) and np.all(y <= 1.0 + 1e-14):
        z = np.clip(y, 0.0, 1.0)
        d = target - float(z.sum())
        if abs(d) <= 1e-12:
            return z
        s = d / float(n)
        if float(np.min(z) + s) >= -1e-15 and float(np.max(z) + s) <= 1.0 + 1e-15:
            return np.clip(z + s, 0.0, 1.0)
        y = z
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(y - m, 0.0, 1.0))) > target:
            lo = m
        else:
            hi = m
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-12:
        if d > 0.0:
            i = int(np.argmax(1.0 - h))
            h[i] += min(d, 1.0 - float(h[i]))
        else:
            i = int(np.argmax(h))
            h[i] -= min(-d, float(h[i]))
    return h

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def prefix_suffix_max(c):
    return np.maximum.accumulate(c), np.maximum.accumulate(c[::-1])[::-1]

def weighted_gradient(h, n, w):
    g = np.zeros(n, dtype=float)
    center = n - 1
    for idx, wv in enumerate(w):
        if wv == 0.0:
            continue
        k = idx - center
        if k >= 0:
            m = n - k
            if m <= 0:
                continue
            g[k:] += wv * (1.0 - h[:m])
            g[:m] -= wv * h[k:]
        else:
            r = -k
            m = n - r
            if m <= 0:
                continue
            g[:m] += wv * (1.0 - h[r:])
            g[r:] -= wv * h[:m]
    return g

def smooth_fg(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    e = np.exp(alpha * (c - mx))
    s = float(np.sum(e))
    dx = 2.0 / n
    if s <= 0.0:
        w = np.full(c.shape, 1.0 / len(c), dtype=float)
        return mx * dx, weighted_gradient(x, n, w) * dx
    w = e / s
    return (mx + math.log(s) / alpha) * dx, weighted_gradient(x, n, w) * dx

def penalty_fg(x, n, alpha, mu):
    v, g = smooth_fg(x, n, alpha)
    s = float(x.sum()) - 0.5 * n
    v += mu * s * s
    g = g + 2.0 * mu * s
    return v, g

def penalty_obj(x, n, alpha, mu):
    return penalty_fg(x, n, alpha, mu)[0]

def penalty_grad(x, n, alpha, mu):
    return penalty_fg(x, n, alpha, mu)[1]

def lbfgsb_refine(h, n, pairs, maxiter, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in pairs:
        if time.time() > deadline - 5.0:
            break
        try:
            res = minimize(penalty_obj, best_h, jac=penalty_grad, args=(n, alpha, float(mu)),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': int(maxiter), 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 40})
        except Exception:
            continue
        nh = project_feasible(res.x, n)
        nc = compute_c5(nh, n)
        if nc < best_c:
            best_h, best_c = nh, nc
    return best_h, best_c

def coord_feasible(W, loW, c_pos, a_pos, c_neg, a_neg, c0, a0, h0):
    if W < loW - 1e-13:
        return False
    lo_x, hi_x = 0.0, 1.0
    if c_pos.size > 0:
        hi_x = min(hi_x, float(np.min((W - c_pos + a_pos * h0) / a_pos)))
    if c_neg.size > 0:
        lo_x = max(lo_x, float(np.max((W - c_neg + a_neg * h0) / a_neg)))
    if hi_x + 1e-12 < lo_x:
        return False
    if c0 is None:
        return True
    disc = a0 * a0 - 4.0 * (W - c0)
    if disc < -1e-10:
        return True
    if disc < 0.0:
        disc = 0.0
    sd = math.sqrt(disc)
    x1 = h0 + 0.5 * (a0 - sd)
    x2 = h0 + 0.5 * (a0 + sd)
    return (lo_x <= x1 + 1e-10) or (hi_x >= x2 - 1e-10)

def point_value(c_line, a_line, c0, a0, h0, U, xv):
    val = U
    if c_line.size > 0:
        val = max(val, float(np.max(c_line + a_line * (xv - h0))))
    if c0 is not None:
        d = xv - h0
        val = max(val, c0 + a0 * d - d * d)
    return val

def coordinate_update(h, c, p, n, pref, suff, binary_iters):
    h0 = float(h[p])
    m_lo = min(p, n - 1 - p)
    m_hi = max(p + n - 1, 2 * n - 2 - p)
    c_aff = c[m_lo:m_hi + 1]
    U = 0.0
    if m_lo > 0:
        U = float(pref[m_lo - 1])
    if m_hi + 1 < len(c):
        U = max(U, float(suff[m_hi + 1]))
    c_mx = float(np.max(c_aff))
    if c_mx <= U + 1e-12:
        return h0
    ms = np.arange(m_lo, m_hi + 1, dtype=int)
    sh = ms - (n - 1)
    kf = p + sh
    kb = p - sh
    a = np.zeros(ms.shape, dtype=float)
    vb = (kb >= 0) & (kb < n)
    if np.any(vb):
        a[vb] += 1.0 - h[kb[vb]]
    vf = (kf >= 0) & (kf < n)
    if np.any(vf):
        a[vf] -= h[kf[vf]]
    zmask = (ms == n - 1)
    if np.any(zmask):
        c0 = float(c_aff[zmask][0])
        a0 = float(a[zmask][0])
        lmask = ~zmask
    else:
        c0, a0 = None, 0.0
        lmask = np.ones(ms.shape, dtype=bool)
    cl, al = c_aff[lmask], a[lmask]
    tol = 1e-12
    zs = np.abs(al) < tol
    const_mx = float(np.max(cl[zs])) if zs.any() else 0.0
    loW = max(U, const_mx)
    hiW = max(U, c_mx)
    if hiW <= loW + 1e-13:
        return h0
    pos = al > tol
    neg = al < -tol
    cp, ap = cl[pos], al[pos]
    cn, an = cl[neg], al[neg]
    loB, hiB = loW, hiW
    for _ in range(int(binary_iters)):
        W = 0.5 * (loB + hiB)
        if coord_feasible(W, loW, cp, ap, cn, an, c0, a0, h0):
            hiB = W
        else:
            loB = W
    Ws = 0.5 * (loB + hiB)
    cands = [0.0, 1.0, h0]
    Wc = Ws + 1e-10
    lo_x, hi_x = 0.0, 1.0
    if cp.size > 0:
        hi_x = min(hi_x, float(np.min((Wc - cp + ap * h0) / ap)))
    if cn.size > 0:
        lo_x = max(lo_x, float(np.max((Wc - cn + an * h0) / an)))
    if hi_x >= lo_x:
        cands.append(lo_x)
        cands.append(hi_x)
        cands.append(0.5 * (lo_x + hi_x))
    if c0 is not None:
        disc = a0 * a0 - 4.0 * (Ws - c0)
        if disc >= -1e-10:
            if disc < 0.0:
                disc = 0.0
            sd = math.sqrt(disc)
            cands.append(h0 + 0.5 * (a0 - sd))
            cands.append(h0 + 0.5 * (a0 + sd))
    best_x, best_val = h0, hiW
    for xv in cands:
        if xv < -1e-12 or xv > 1.0 + 1e-12:
            continue
        xv = float(np.clip(xv, 0.0, 1.0))
        val = point_value(cl, al, c0, a0, h0, U, xv)
        if val < best_val - 1e-13:
            best_val, best_x = val, xv
    return best_x

def coord_cd(h, n, max_passes, deadline, block, recompute_every, binary_iters):
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    no_improve = 0
    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        pref, suff = prefix_suffix_max(c)
        updated = False
        for p in range(n):
            if time.time() > deadline:
                break
            dp = p > 0 and p % block == 0
            dr = p > 0 and p % recompute_every == 0
            if dp or dr:
                if dp:
                    h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                pref, suff = prefix_suffix_max(c)
            nx = coordinate_update(h, c, p, n, pref, suff, binary_iters)
            if abs(nx - h[p]) > 1e-13:
                h[p] = nx
                updated = True
        h = project_feasible(h, n)
        cn = compute_c5(h, n)
        if cn < best_c - 1e-13:
            best_h, best_c, no_improve = h.copy(), cn, 0
        elif not updated:
            break
        else:
            no_improve += 1
            if no_improve == 3:
                h = project_feasible(best_h, n)
            if no_improve >= 6:
                break
    return best_h, best_c

def random_start(n, rng):
    return project_feasible(rng.uniform(0.0, 1.0, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + total - 5.0
    rng = np.random.RandomState(seed)
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 500
        h = np.full(n, 0.5)
    if n < 20:
        n = 200
        h = np.full(n, 0.5)
    if n > 1000:
        h = np.interp(centers(1000), centers(n), h)
        n = 1000
    h = project_feasible(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)
    primary = [(50, 1.0), (500, 10.0), (2000, 100.0), (10000, 1000.0), (50000, 1000.0)]
    d1 = max(t0 + 30.0, deadline - min(700.0, max(120.0, 0.65 * total)))
    ho, co = lbfgsb_refine(h, n, primary, 160, d1)
    if co < best_c:
        best_h, best_c = ho, co
    d2 = max(t0 + 30.0, deadline - min(500.0, max(80.0, 0.45 * total)))
    ho, co = coord_cd(best_h, len(best_h), 250, d2, 20, 1 if len(best_h) <= 1000 else 4, 40)
    if co < best_c:
        best_h, best_c = ho, co
    ncur = len(best_h)
    if time.time() < deadline - 320.0 and ncur > 180:
        nco = 160
        hco = project_feasible(np.interp(centers(nco), centers(ncur), best_h), nco)
        d1c = min(deadline - 200.0, time.time() + 70.0)
        d2c = min(deadline - 120.0, time.time() + 70.0)
        hco, _ = lbfgsb_refine(hco, nco, [(200, 10.0), (2000, 100.0), (20000, 1000.0)], 80, d1c)
        hco, _ = coord_cd(hco, nco, 60, d2c, 8, 1, 38)
        hup = project_feasible(np.interp(centers(ncur), centers(nco), hco), ncur)
        d3c = min(deadline - 80.0, time.time() + 60.0)
        d4c = min(deadline - 40.0, time.time() + 60.0)
        hup, co = lbfgsb_refine(hup, ncur, [(1000, 100.0), (10000, 1000.0)], 80, d3c)
        hup, co = coord_cd(hup, ncur, 50, d4c, 10, 1 if ncur <= 1000 else 4, 38)
        if co < best_c:
            best_h, best_c = hup, co
    for nt in (400, 600, 800):
        if nt == len(best_h):
            continue
        if time.time() > deadline - 280.0:
            break
        ncur = len(best_h)
        hint = project_feasible(np.interp(centers(nt), centers(ncur), best_h), nt)
        d1n = min(deadline - 180.0, time.time() + 70.0)
        d2n = min(deadline - 90.0, time.time() + 70.0)
        ho, co = lbfgsb_refine(hint, nt, [(200, 10.0), (2000, 100.0), (20000, 1000.0)], 80, d1n)
        ho, co = coord_cd(ho, nt, 60, d2n, 16, 1 if nt <= 1000 else 4, 38)
        if co < best_c:
            best_h, best_c = ho, co
    ncur = len(best_h)
    if time.time() < deadline - 160.0:
        for r in range(2):
            if time.time() > deadline - 100.0:
                break
            sig = 0.015 + 0.03 * r
            hp = best_h + rng.normal(0.0, sig, ncur)
            if r == 1:
                hp = 0.6 * hp + 0.4 * random_start(ncur, rng)
            hp = project_feasible(hp, ncur)
            d1r = min(deadline - 80.0, time.time() + 40.0)
            d2r = min(deadline - 40.0, time.time() + 40.0)
            ho, co = lbfgsb_refine(hp, ncur, [(200, 10.0), (5000, 1000.0)], 60, d1r)
            ho, co = coord_cd(ho, ncur, 25, d2r, 12, 1 if ncur <= 1000 else 4, 35)
            if co < best_c:
                best_h, best_c = ho, co
    ncur = len(best_h)
    if time.time() < deadline - 180.0:
        ho, co = lbfgsb_refine(best_h, ncur, [(20000, 1000.0), (100000, 1000.0), (300000, 1000.0)],
                               150, deadline - 50.0)
        if co < best_c:
            best_h, best_c = ho, co
    ncur = len(best_h)
    ho, co = coord_cd(best_h, ncur, 5000, deadline - 1.0,
                      12 if ncur <= 800 else 24, 1 if ncur <= 1000 else 4, 45)
    if co < best_c:
        best_h, best_c = ho, co
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
