def polish_lse(h, n, betas, deadline, stage_time=35.0):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5.0:
            break
        _stop_time = min(deadline - 3.0, time.time() + stage_time)
        try:
            res = minimize(lbfgs_func, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, float(beta), 1e4),
                          callback=stop_callback,
                          options={"maxiter": 300000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9})
            work = project_feasible(res.x, n)
        except Exception:
            pass
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
        for v in (best[::-1], 1.0 - best[::-1], 0.5*(best + best[::-1])):
            cv = compute_c5(project_feasible(v, n), n)
            if cv < bc:
                bc, best = cv, project_feasible(v, n)
        if bc < compute_c5(work, n) - 1e-12:
            work = best.copy()
    return best, bc

def binary_swap_improve(h, n, deadline, max_sweeps=50, chunk=4096):
    if n % 2 != 0:
        return top_n_binary(h, n), compute_c5(top_n_binary(h, n), n)
    h = top_n_binary(h, n).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    M = int(round(corr.max()))
    best_h, best_M = h.copy(), M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.nonzero(h == 1)[0]
        zeros = np.nonzero(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        zi = np.repeat(zeros, len(ones))
        oj = np.tile(ones, len(zeros))
        improved = False
        for st in range(0, len(zi), chunk):
            if time.time() > deadline:
                break
            en = min(st + chunk, len(zi))
            i_arr = zi[st:en]
            j_arr = oj[st:en]
            di1 = n - 1 + (j_arr - i_arr)
            di2 = n - 1 + (i_arr - j_arr)
            valid1 = (di1 >= 0) & (di1 < 2*n - 1)
            valid2 = (di2 >= 0) & (di2 < 2*n - 1)
            # Delta for each lag: removing 1 at j, adding 1 at i
            # c_new[k] = c[k] - (1-h[(j+k)%(n)]) - h[(j-k+n)%(n)] + (1-h[(i+k)%(n)]) + h[(i-k+n)%(n)]
            # Simplified: for linear correlation
            # Let's just compute the max by checking all lags affected
            # Faster: precompute base
            base = np.zeros(2*n-1)
            for p in range(n):
                f = p + np.arange(-(n-1), n)
                mf = (f >= 0) & (f < n)
                base[mf] += (1.0 - h[f[mf]]) * h[p]
                b = p - np.arange(-(n-1), n)
                mb = (b >= 0) & (b < n)
                base[mb] -= h[b[mb]] * h[p]
            # This is getting complex. Let's just do direct evaluation for chunks.
            break
        # Fallback: simple random swaps
        for _ in range(200):
            if time.time() > deadline:
                break
            ii = int(ones[np.random.randint(len(ones))])
            jj = int(zeros[np.random.randint(len(zeros))])
            h[ii], h[jj] = h[jj], h[ii]
            nc = np.correlate(h, 1.0 - h, mode="full")
            nM = int(round(nc.max()))
            if nM < M:
                M = nM
                corr = nc
                if M < best_M:
                    best_M, best_h = M, h.copy()
                improved = True
            else:
                h[ii], h[jj] = h[jj], h[ii]
        if not improved:
            break
    return best_h, best_M * (2.0 / float(n))

def simulated_anneal(h, n, deadline, total_iters=80000):
    h = top_n_binary(h, n).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(np.float64)
    M = int(corr.max())
    best_h, best_M = h.copy(), M
    ones = np.nonzero(h == 1)[0]
    zeros = np.nonzero(h == 0)[0]
    T = 4.0
    T_min = 0.05
    n_steps = max(1, total_iters)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / float(n_steps)
        T = T_min + (4.0 - T_min) * (1.0 - frac) ** 2.0
        ii = int(ones[np.random.randint(len(ones))])
        jj = int(zeros[np.random.randint(len(zeros))])
        h[ii], h[jj] = 0.0, 1.0
        nc = np.correlate(h, 1.0 - h, mode="full")
        nM = int(nc.max())
        delta = nM - M
        if delta <= 0 or np.random.random() < np.exp(-float(delta) / max(T, 1e-10)):
            M = nM
            corr = nc
            idx_ones = np.where(h[ii:jj+1] if ii < jj else h[jj:ii+1] == 0)[0] if False else None
            ones = np.nonzero(h == 1)[0]
            zeros = np.nonzero(h == 0)[0]
            if M < best_M:
                best_M, best_h = M, h.copy()
        else:
            h[ii], h[jj] = 1.0, 0.0
    return best_h, best_M * (2.0 / float(n))

def try_resolution(h_best, c_best, n_best, n_new, deadline):
    x0 = np.linspace(0.0, 2.0, n_best)
    x1 = np.linspace(0.0, 2.0, n_new)
    hr = np.interp(x1, x0, h_best)
    h_proj = project_feasible(hr, n_new)
    hc = compute_c5(h_proj, n_new)
    hh, cc = polish_lse(h_proj, n_new, [100.0, 500.0, 2000.0, 8000.0], deadline, 25.0)
    if cc < hc:
        hc, h_proj = cc, hh
    if n_new % 2 == 0:
        hb = top_n_binary(h_proj, n_new)
        bc = compute_c5(hb, n_new)
        if time.time() < deadline - 30:
            hb2, bc2 = simulated_anneal(hb, n_new, deadline, total_iters=30000)
            if bc2 < bc:
                hb, bc = hb2, bc2
        if bc < hc:
            hc, h_proj = bc, hb
        if time.time() < deadline - 20:
            hh2, cc2 = polish_lse(h_proj, n_new, [5000.0, 20000.0], deadline, 15.0)
            if cc2 < hc:
                hc, h_proj = cc2, hh2
            if n_new % 2 == 0:
                hb3 = top_n_binary(h_proj, n_new)
                if compute_c5(hb3, n_new) < hc:
                    hc, h_proj = compute_c5(hb3, n_new), hb3
    return h_proj, hc, n_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    end = t0 + max(60.0, total_budget - 15.0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        if n0 < 4 or not np.all(np.isfinite(h0)):
            raise ValueError("bad")
    except Exception:
        n0 = 400
        h0 = np.zeros(n0)
        h0[::2] = 1.0

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Stage 1: LSE polish from initial
    if time.time() < end - 300:
        hh, cc = polish_lse(best_h, best_n, [50.0, 200.0, 1000.0, 5000.0], end - 300, 30.0)
        if cc < best_c:
            best_c, best_h = cc, hh

    # Stage 2: Binary + SA
    if best_n % 2 == 0 and time.time() < end - 250:
        hb = top_n_binary(best_h, best_n)
        bc = compute_c5(hb, best_n)
        if bc < best_c:
            best_c, best_h = bc, hb
        if time.time() < end - 200:
            hb2, bc2 = simulated_anneal(best_h, best_n, end - 200, total_iters=60000)
            if bc2 < best_c:
                best_c, best_h = bc2, hb2

    # Stage 3: LSE polish after SA
    if time.time() < end - 150:
        hh, cc = polish_lse(best_h, best_n, [2000.0, 10000.0, 50000.0], end - 150, 30.0)
        if cc < best_c:
            best_c, best_h = cc, hh
        if best_n % 2 == 0:
            hb = top_n_binary(best_h, best_n)
            bc = compute_c5(hb, best_n)
            if bc < best_c:
                best_c, best_h = bc, hb

    # Stage 4: Try n=200
    if time.time() < end - 180:
        h200, c200, n200 = try_resolution(best_h, best_c, best_n, 200, end - 170)
        if c200 < best_c:
            best_c, best_h, best_n = c200, h200, n200

    # Stage 5: Try n=300
    if best_n != 300 and best_n % 2 == 0 and time.time() < end - 120:
        h300, c300, n300 = try_resolution(best_h, best_c, best_n, 300, end - 110)
        if c300 < best_c:
            best_c, best_h, best_n = c300, h300, n300

    # Stage 6: More SA on best if time remains
    if best_n % 2 == 0 and time.time() < end - 60:
        n_sa = min(50000, max(10000, int((end - time.time() - 30) * 3000)))
        hb3, bc3 = simulated_anneal(best_h, best_n, end - 50, total_iters=n_sa)
        if bc3 < best_c:
            best_c, best_h = bc3, hb3

    # Stage 7: Final LSE polish
    if time.time() < end - 30:
        hh, cc = polish_lse(best_h, best_n, [10000.0, 50000.0, 100000.0], end - 10, 15.0)
        if cc < best_c:
            best_c, best_h = cc, hh
        if best_n % 2 == 0:
            hb = top_n_binary(best_h, best_n)
            bc = compute_c5(hb, best_n)
            if bc < best_c:
                best_c, best_h = bc, hb

    # Final variant check
    for v in (best_h[::-1], 1.0 - best_h[::-1], 0.5*(best_h + best_h[::-1])):
        vv = project_feasible(v, best_n)
        cv = compute_c5(vv, best_n)
        if cv < best_c:
            best_c, best_h = cv, vv

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
