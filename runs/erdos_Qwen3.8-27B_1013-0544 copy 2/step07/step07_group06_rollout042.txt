def swap_delta_vec(h, one_h, i, j, n):
    """Vectorized computation of correlation delta for swap (i:0->1, j:1->0)."""
    u = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # (1-h)[i-u] where 0 <= i-u < n
    idx1 = i - u  # shape (2n-1,)
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.zeros(2*n-1)
    term1[mask1] = one_h[idx1[mask1]]
    
    # (1-h)[j-u] where 0 <= j-u < n
    idx2 = j - u
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.zeros(2*n-1)
    term2[mask2] = one_h[idx2[mask2]]
    
    # h[i+u] where 0 <= i+u < n
    idx3 = i + u
    mask3 = (idx3 >= 0) & (idx3 < n)
    term3 = np.zeros(2*n-1)
    term3[mask3] = h[idx3[mask3]]
    
    # h[j+u] where 0 <= j+u < n
    idx4 = j + u
    mask4 = (idx4 >= 0) & (idx4 < n)
    term4 = np.zeros(2*n-1)
    term4[mask4] = h[idx4[mask4]]
    
    delta = term1 - term2 - term3 + term4
    
    # Special terms
    k0 = n - 1  # u=0
    delta[k0] -= 2.0
    k_pos = n - 1 + (i - j)
    if 0 <= k_pos <= 2*n-2:
        delta[k_pos] += 1.0
    k_neg = n - 1 + (j - i)
    if 0 <= k_neg <= 2*n-2:
        delta[k_neg] += 1.0
    
    return delta
