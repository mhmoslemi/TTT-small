def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)
        rem = diff
        for p in idx:
            if abs(rem) < 1e-15:
                break
            if rem > 0:
                add = min(rem, 1.0 - h[p])
                if add > 0:
                    h[p] += add
                    rem -= add
            else:
                sub = min(-rem, h[p])
                if sub > 0:
                    h[p] -= sub
                    rem += sub
        # if one value left due roundoff, set? maybe

def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s <= 1e-15: break
            h += (d / s) * r
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += (d / s) * h
    h = np.clip(h,0,1)
    d = target - h.sum()
    if abs(d) > 1e-13:
       idx = np.argsort(1.0-h if d>0 else h)
       if d>0: idx=idx[::-1]
       rem=d
       for p in idx:
          if abs(rem)<1e-15: break
          if rem>0:
             a=min(rem,1-h[p]); h[p]+=a; rem-=a
          else:
             a=min(-rem,h[p]); h[p]-=a; rem+=a
    d=target-h.sum()
    if abs(d)>1e-15:
       if d>0:
          if (1-h).max()>0: h[np.argmax(1-h)] += d
       else:
          if h.max()>0: h[np.argmax(h)] += d
    return np.clip(h,0,1)

import numpy as np
from scipy.optimize import minimize
import time
import math


def correlate_counts(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h, n):
    return float(np.max(correlate_counts(h, n)) * (2.0 / n))


def project_feasible(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(70):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h)[::-1] if d > 0 else np.argsort(h)
        rem = d
        for p in idx:
            if abs(rem) < 1e-15:
                break
            if rem > 0:
                a = min(rem, 1.0 - h[p])
                h[p] += a
                rem -= a
            else:
                a = min(-rem, h[p])
                h[p] -= a
                rem += a
    d = target - h.sum()
    if abs(d) > 1e-15:
        if d > 0 and (1.0 - h).max() > 0:
            h[int(np.argmax(1.0 - h))] += d
        elif d < 0 and h.max() > 0:
            h[int(np.argmax(h))] += d
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):
    hb = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        hb[np.argsort(h)[-k:]] = 1.0
    return hb


def make_diff_idx(n):
    idx = np.arange(n)
    return (idx[None, :] - idx[:, None] + n - 1).astype(np.int32)


def smooth_fun(x, aux):
    n = aux[0]
    diff_idx = aux[1]
    alpha = aux[2]
    pen = aux[3]
    dx = aux[4]
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    z = np.clip(alpha * dx * (c - m), -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        s = 1.0
    w = e / s
    W = w[diff_idx]
    grad_counts = W.sum(axis=1) - W.dot(h) - W.T.dot(h)
    grad = dx * grad_counts
    val = m * dx + math.log(s) / alpha
    if pen > 0.0:
        sd = h.sum() - 0.5 * n
        if sd != 0.0:
            val += 0.5 * pen * sd * sd
            grad = grad + (pen * sd)
    return val, grad


def run_minimize(cur, aux, bounds, maxit):
    maxit = int(maxit)
    opts = {
        "maxiter": maxit,
        "ftol": 1e-15,
        "gtol": 1e-9,
        "maxls": 20,
        "maxfun": int(max(200, 50 * maxit)),
    }
    try:
        res = minimize(smooth_fun, cur, jac=True, method="L-BFGS-B",
                       bounds=bounds, args=aux, options=opts)
    except Exception:
        return cur
    x = getattr(res, "x", None)
    if x is None:
        return cur
    x = np.asarray(x, dtype=float)
    if x.shape != (len(cur),) or not np.all(np.isfinite(x)):
        return cur
    return x


def lbgfs_phase(h, n, diff_idx, deadline, schedule):

def build_features(h, n):
    L = 2 * n - 1
    F = np.zeros((n, L), dtype=float)
    for r in range(1, n):
        ip = n - 1 + r
        F[:n - r, ip] += 1.0 - h[r:n]
        F[r:n, ip] += -h[:n - r]
        ineg = n - 1 - r
        F[r:n, ineg] += 1.0 - h[:n - r]
        F[:n - r, ineg] += -h[r:n]
    return F


def update_features(F, i, j, n):
    rows = np.arange(n)
    m1 = rows != i
    c1 = n - 1 + (i - rows)
    c2 = n - 1 + (rows - i)
    F[rows[m1], c1[m1]] -= 1.0
    F[rows[m1], c2[m1]] -= 1.0
    m2 = rows != j
    c1 = n - 1 + (j - rows)
    c2 = n - 1 + (rows - j)
    F[rows[m2], c1[m2]] += 1.0
    F[rows[m2], c2[m2]] += 1.0

    dx = 2.0 / n

    while time.time() < deadline and sweeps < max_sweeps:

        if zeros.size == 0 or ones.size == 0:
            break
        best_row = cur
        bi = -1
        bj = -1
        for i in zeros:
            base = c + F[int(i)]
            cand = base[None, :] - F[ones]

            col1 = n - 1 + (ones - i)
            col2 = n - 1 + (i - ones)

            rms = cand.max(axis=1)
            kk = int(np.argmin(rms))

                best_row = float(rms[kk])
                bi = int(i)
                bj = int(ones[kk])

        c = c + F[bi] - F[bj]

def binary_kick(h, n, b, rng):
    h = h.astype(float).copy()
    zeros = list(np.flatnonzero(h < 0.5))
    ones = list(np.flatnonzero(h >= 0.5))
    for _ in range(b):
        if not zeros or not ones:
            break
        ip = int(rng.randint(len(zeros)))
        jp = int(rng.randint(len(ones)))
        i = zeros.pop(ip)
        j = ones.pop(jp)
        h[i] = 1.0
        h[j] = 0.0
        zeros.append(j)
        ones.append(i)
    return h
