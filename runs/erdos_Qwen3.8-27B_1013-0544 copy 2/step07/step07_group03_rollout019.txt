def active_gradient(h,c,F, beta=100.0, width=0.05, max_active=16):
    idx = np.flatnonzero(c >= F - width)
    if len(idx) ==0: idx = np.array([int(np.argmax(c))])
    if len(max_active) > max_active:
       # top max_active among idx (sort by c)
       order = np.argsort(c[idx])[::-1][:max_active]
       idx=idx[order]
    # weights: softmax beta*(c-F)
    z = beta*(c[idx]-F)
    z -= z.max()
    w = np.exp(z)
    s = w.sum()
    if s <=0: w=np.full(len(idx),1.0/len(idx))
    else: w/=s
    G=np.zeros(n)
    for idxk, wk in zip(idx,w):

import numpy as np
import time

COARSE = np.array([
    0.5, 0.25, 0.75, 1.0, 0.125, 0.875, 0.0625,
    0.9375, 0.03125, 0.96875, 0.1875, 0.3125,
    0.6875, 0.625, 0.375, 0.09375, 0.015625,
    0.01, 0.05
], dtype=float)

REFINE = np.array([
    0.5, 1.5, 2.5, 0.25, 0.75, 1.25, 0.15, 2.0
], dtype=float)

FULLFR = np.array([
    0.9, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.75, 0.1
], dtype=float)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(250):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)


def eval_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def build_indices(n):
    ar = np.arange(n)
    base = n - 1
    idx_sh = np.empty((n, n), dtype=np.int64)
    idx_ba = np.empty((n, n), dtype=np.int64)
    for p in range(n):
        idx_sh[p] = p - ar + base
        idx_ba[p] = ar - p + base
    return idx_sh, idx_ba


def apply_pair(cn, h, omh, n, idx_sh, idx_ba, i, j, delta):
    d = float(delta)
    d2 = d * d
    cn[idx_sh[i]] += d * omh
    cn[idx_ba[i]] -= d * h
    cn[n - 1] -= d2
    nd = -d
    cn[idx_sh[j]] += nd * omh
    cn[idx_ba[j]] -= nd * h
    cn[n - 1] -= d2
    cn[j - i + n - 1] += d2
    cn[i - j + n - 1] += d2


def line_search_pair(h, omh, c, F, i, j, n, idx_sh, idx_ba):
    maxd = min(float(h[j]), float(1.0 - h[i]))
    if maxd <= 1e-14:
        return None

    bestf = F
    bestd = 0.0
    bestc = None

    for frac in COARSE:
        d = frac * maxd
        if d <= 1e-15:
            continue
        if d > maxd * (1.0 + 1e-12):
            d = maxd
        cn = c.copy()
        apply_pair(cn, h, omh, n, idx_sh, idx_ba, i, j, d)
        f = float(cn.max())
        if f < bestf - 1e-12:
            bestf = f
            bestd = d
            bestc = cn

    if bestc is not None:
        for mult in REFINE:
            d = bestd * float(mult)
            if d <= 1e-15:
                continue
            if d > maxd:
                d = maxd
            cn = c.copy()
            apply_pair(cn, h, omh, n, idx_sh, idx_ba, i, j, d)
            f = float(cn.max())
            if f < bestf - 1e-13:
                bestf = f
                bestd = d
                bestc = cn

    if bestc is None:
        return None
    return (i, j, bestd, bestc, bestf)


def add_lag_grad(G, h, k, w):
    n = len(h)
    if k == 0:
        G += w * (1.0 - 2.0 * h)
    elif k > 0:
        m = k
        G[m:] += w * (1.0 - h[:n - m])
        G[:n - m] -= w * h[m:]
    else:
        m = -k
        G[:n - m] += w * (1.0 - h[m:])
        G[m:] -= w * h[:n - m]


def active_gradient(h, c, F, topk, beta):
    n = len(h)
    if topk <= 1:
        idx = int(np.argmax(c))
        G = np.zeros(n, dtype=float)
        add_lag_grad(G, h, idx - (n - 1), 1.0)
        return G

    top = np.argsort(c)[::-1][:topk]
    z = beta * (c[top] - F)
    z -= z.max()
    w = np.exp(z)
    s = w.sum()
    if s <= 1e-300:
        w = np.full(topk, 1.0 / float(topk), dtype=float)
    else:
        w /= s

    G = np.zeros(n, dtype=float)
    base = n - 1
    for idx, wk in zip(top, w):
        if wk > 1e-8:
            add_lag_grad(G, h, int(idx) - base, float(wk))
    return G


def full_step(h, c, F, G):
    n = len(h)
    d = -G
    d = d - d.mean()
    pos = d > 1e-12
    neg = d < -1e-12
    if not (np.any(pos) or np.any(neg)):
        return None

    lmax = 1e100
    if np.any(pos):
        lmax = min(lmax, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lmax = min(lmax, float(np.min(h[neg] / (-d[neg]))))
    if not np.isfinite(lmax) or lmax <= 1e-13:
        return None

    besth = None
    bestc = None
    bestf = F

    for f in FULLFR:
        lam = float(f) * lmax
        if lam <= 1e-14:
            continue
        if lam >= lmax:
            lam = lmax * 0.999999
        hn = h + lam * d
        if np.min(hn) < -1e-10 or np.max(hn) > 1.0 + 1e-10:
            continue
        hn = np.clip(hn, 0.0, 1.0)
        cn = np.correlate(hn, 1.0 - hn, mode="full").astype(float)
        fn = float(cn.max())
        if fn < bestf - 1e-12:
            bestf = fn
            besth = hn
            bestc = cn

    if besth is None:
        return None
    return (besth, bestc, bestf)


def local_polish(h0, n, idx_sh, idx_ba, deadline, rng):
    h = project_feasible(h0, n)
    omh = 1.0 - h
    c = np.correlate(h, omh, mode="full").astype(float)
    F = float(c.max())

    best_h = h.copy()
    best_F = F

    it = 0
    stall = 0
    rebuilt = 0
    P = 4
    top_modes = ((1, 0.0), (5, 60.0), (12, 20.0))

    while time.time() < deadline:
        it += 1

        if F < best_F - 1e-13:
            best_F = F
            best_h = h.copy()

        if rebuilt >= 40 or it % 80 == 0:
            c = np.correlate(h, omh, mode="full").astype(float)
            F = float(c.max())
            rebuilt = 0

        if it % 120 == 30:
            hs = 0.5 * (h + h[::-1])
            cs = np.correlate(hs, 1.0 - hs, mode="full").astype(float)
            fs = float(cs.max())
            if fs < F - 1e-12:
                h = hs
                omh = 1.0 - h
                c = cs
                F = fs
                rebuilt = 0
                stall = 0

        topk, beta = top_modes[it % 3]
        G = active_gradient(h, c, F, topk, beta)
        improved = False

        lowp = min(P, n)
        low = np.argpartition(G, lowp - 1)[:lowp]
        high = np.argpartition(G, n - lowp)[-lowp:]

        best_res = None
        best_fn = F
        for i in low:
            ii = int(i)
            for j in high:
                jj = int(j)
                if ii == jj:
                    continue
                res = line_search_pair(h, omh, c, F, ii, jj, n, idx_sh, idx_ba)
                if res is not None and res[4] < best_fn - 1e-13:
                    best_fn = res[4]
                    best_res = res

        if best_res is not None:
            i, j, d, cn, fn = best_res
            h[i] += d
            h[j] -= d
            omh[i] -= d
            omh[j] += d
            c = cn
            F = fn
            improved = True
            rebuilt += 1

        if not improved:
            resf = full_step(h, c, F, G)
            if resf is not None:
                h, c, F = resf
                omh = 1.0 - h
                rebuilt = 0
                improved = True

        if not improved:
            rand_tries = 16 if stall < 60 else 4
            for _ in range(rand_tries):
                ii = int(rng.randint(n))
                jj = int(rng.randint(n))
                if ii == jj:
                    continue
                res = line_search_pair(h, omh, c, F, ii, jj, n, idx_sh, idx_ba)
                if res is not None:
                    i, j, d, cn, fn = res
                    h[i] += d
                    h[j] -= d
                    omh[i] -= d
                    omh[j] += d
                    c = cn
                    F = fn
                    improved = True
                    rebuilt += 1
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 120:
                break

    best_h = project_feasible(best_h, n)
    c = np.correlate(best_h, 1.0 - best_h, mode="full").astype(float)
    best_F = float(c.max())
    return best_h, best_F


def make_default(n):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.49 * np.cos(np.pi * x)
    return project_feasible(h, n)


def make_u(n):
    t = max(1, n // 8)
    edge = 0.85
    h = np.empty(n, dtype=float)
    h[:t] = edge
    h[-t:] = edge
    denom = float(n) - 2.0 * float(t)
    if denom < 1e-12:
        mid = 0.5
    else:
        mid = (0.5 * float(n) - 2.0 * float(t) * edge) / denom
        mid = float(np.clip(mid, 0.0, 1.0))
    h[t:n - t] = mid
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 8.0)
    deadline = t0 + total
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or len(h0) < 2:
        n = 400
        h0 = make_default(n)
    else:
        n = len(h0)

    h0 = project_feasible(h0, n)
    idx_sh, idx_ba = build_indices(n)

    c0 = np.correlate(h0, 1.0 - h0, mode="full").astype(float)
    best_F = float(c0.max())
    best_h = h0.copy()

    sym_h = project_feasible(0.5 * (h0 + h0[::-1]), n)
    sym_F = eval_raw(sym_h)
    sym_was_best = (sym_F < best_F - 1e-13)
    if sym_was_best:
        best_h = sym_h.copy()
        best_F = sym_F

    d1 = t0 + 0.32 * total
    if time.time() < min(deadline - 5.0, d1):
        best_h, best_F = local_polish(best_h, n, idx_sh, idx_ba, min(deadline - 5.0, d1), rng)

    if not sym_was_best and time.time() < deadline - 40.0:
        end = min(deadline - 20.0, time.time() + 0.10 * total)
        if end - time.time() > 15.0:
            sh, sF = local_polish(sym_h, n, idx_sh, idx_ba, end, rng)
            if sF < best_F - 1e-14:
                best_h = sh
                best_F = sF

    scales = [0.005, 0.01, 0.02, 0.008, 0.03, 0.05, 0.015, 0.08, 0.12, 0.2]
    slice_s = max(18.0, min(70.0, 0.055 * total))

    for s in scales:
        if time.time() > deadline - 50.0:
            break
        end = min(deadline - 30.0, time.time() + slice_s)
        if end - time.time() < 15.0:
            break
        hst = project_feasible(best_h + float(s) * rng.randn(n), n)
        sh, sF = local_polish(hst, n, idx_sh, idx_ba, end, rng)
        if sF < best_F - 1e-14:
            best_h = sh
            best_F = sF

    if time.time() < deadline - 60.0:
        hu = make_u(n)
        hu_F = eval_raw(hu)
        if hu_F < best_F + 2.0:
            end = min(deadline - 25.0, time.time() + 55.0)
            if end - time.time() > 12.0:
                sh, sF = local_polish(hu, n, idx_sh, idx_ba, end, rng)
                if sF < best_F - 1e-14:
                    best_h = sh
                    best_F = sF

    if time.time() < deadline - 3.0:
        best_h, best_F = local_polish(best_h, n, idx_sh, idx_ba, deadline - 2.0, rng)

    if time.time() < deadline - 80.0 and n < 1000:
        nh = min(2 * n, 1000)
        if nh > n:
            xo = np.linspace(0.0, 2.0, n)
            xn = np.linspace(0.0, 2.0, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            idxn_sh, idxn_ba = build_indices(nh)
            end = min(deadline - 5.0, time.time() + 70.0)
            if end - time.time() > 20.0:
                hh2, hhF = local_polish(hh, nh, idxn_sh, idxn_ba, end, rng)
                if hhF * (2.0 / float(nh)) < best_F * (2.0 / float(n)) - 1e-12:
                    best_h = hh2
                    best_F = hhF
                    n = nh

    best_h = project_feasible(best_h, n)
    cf = np.correlate(best_h, 1.0 - best_h, mode="full").astype(float)
    best_C = float(cf.max()) * (2.0 / float(n))
    return best_h, best_C, int(n)
