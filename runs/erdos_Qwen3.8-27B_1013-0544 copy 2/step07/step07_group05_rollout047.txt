import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    s = float(y.sum())
    if abs(s - target) < 1e-14:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            y = y.copy()
            y[mask] += diff / cnt
        else:
            y = y + diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def softmax_w(c, beta):
    mc = float(np.max(c))
    z = beta * (c - mc)
    np.clip(z, -700.0, 0.0, out=z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mc, w
    w = e / s
    f = mc + np.log(s) / beta
    return f, w

def grad_weighted_corr(h, w, n):
    dx = 2.0 / n
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk < 1e-15:
            continue

            grad[kk:] -= wk * h[:n - kk]
    return grad * dx

def obj_fun(x):

    n = st['n']
    dx = st['dx']

    f, w = softmax_w(c, st['beta'])
    g = grad_weighted_corr(x, w, n)
    st['cx'] = x.copy()
    st['cg'] = g.copy()
    return float(f)

def jac_fun(x):
    st = _st
    if st['cx'] is not None and np.array_equal(x, st['cx']):
        return st['cg'].copy()
    obj_fun(x)
    return st['cg'].copy()

def refine_lbfgs(h, beta, maxiter, deadline):

def active_grad_descent(h, n, top_m, lr, nsteps, deadline):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    cur = erdos_c5(h, n)
    dx = 2.0 / n
    for step in range(nsteps):
        if time.time() > deadline - 6.0:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_m:]

        w[top_idx] = 1.0 / top_m
        g = grad_weighted_corr(h, w, n)
        # Backtracking line search
        alpha = lr
        for _ in range(20):
            candidate = h - alpha * g
            candidate = project_box_sum(candidate, target)
            cc = erdos_c5(candidate, n)
            if cc < cur - 1e-15 * alpha:
                h = candidate
                cur = cc
                break
            alpha *= 0.5
        else:
            break
    return h, cur

def pair_line(h, n, i, j, cur, deadline):

    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    def obj_delta(d):
        hh = h.copy()
        hh[i] += d
        hh[j] -= d
        return erdos_c5(hh, n)
    cs = obj_delta(lo)
    if cs < best_c:
        best_c = cs
        best_d = lo
    cs = obj_delta(hi)
    if cs < best_c:
        best_c = cs
        best_d = hi
    try:
        if time.time() < deadline - 1.0:
            rs = so.minimize_scalar(obj_delta, bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 30, 'xatol': 1e-11})

                cs2 = obj_delta(float(rs.x))
                if cs2 < best_c:
                    best_c = cs2
                    best_d = float(rs.x)

        pass
    if best_c < cur - 1e-15:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return np.clip(hh, 0.0, 1.0), best_c

def pair_polish(h, n, deadline, seed, n_pairs=40, epochs=4):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= n_pairs:
        top = order[:n_pairs]
    else:
        need = n_pairs - int(order.size)
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for ep in range(epochs):
        if time.time() > deadline - 5.0:
            break
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):
                pairs.append((int(top[a]), int(top[b])))
        rng.shuffle(pairs)
        for i, j in pairs:

                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline)
            if cc < cur - 1e-15:
                h = hh
                cur = cc
                improved = True
        if not improved:
            break
    return h, cur
