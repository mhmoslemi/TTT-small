from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline):

    def obj(x):
        if time.time() > deadline:
            return 1e10
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return _smooth_max(c, alpha)
    def con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def lbfgs_opt(h, n, alpha, maxiter, deadline):

            return 1e10

        c = np.correlate(xc, 1.0 - xc, mode='full') * dx
        return _smooth_max(c, alpha) + pen * (xc.sum() - n / 2.0) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def multi_perturb_opt(h, n, rng, alpha, maxiter, deadline, n_trials=6):

        c_new = compute_c5(h_new, n)
        if c_new < best_c:

    return best_h, best_c
