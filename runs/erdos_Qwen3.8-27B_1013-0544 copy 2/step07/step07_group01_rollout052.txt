                       constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def downsample(h, n, m):
    xf = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    xc = np.linspace(0, 2, m, endpoint=False) + 1.0 / m
    return project_feasible(np.interp(xc, xf, h), m)

def upsample(h, n, m):
    xf = np.linspace(0, 2, m, endpoint=False) + 1.0 / m
    xc = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    return project_feasible(np.interp(xf, xc, h), m)

def adam_subgrad(h, n, deadline, lr0=0.005):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    mu1, mu2 = 0.9, 0.999
    while time.time() < deadline:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        elif lag < 0 and -lag < n:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll])
            grad[:n - ll] -= h[ll:]
        grad *= dx
        lr = lr0 / (1.0 + 0.005 * t)
        m = mu1 * m + (1 - mu1) * grad
        v = mu2 * v + (1 - mu2) * grad * grad
        h = project_feasible(h - lr * m / (np.sqrt(v) + 1e-8), n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def binary_init(n, rng):
    pos = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-grid coarse-to-fine with SLSQP + gradients
    for ng in [48, 96, 160, 240]:
        if time.time() > deadline - 350:
            break
        h_c = downsample(best_h, n, ng)
        for a in [200, 2000, 20000]:
            if time.time() > deadline - 300:
                break
            h_c = slsqp_optimize(h_c, ng, a, 500, 1e-12)
        h_f = upsample(h_c, ng, n)
        cf = compute_c5(h_f, n)
        if cf < best_c:
            best_c, best_h = cf, h_f.copy()

    # Phase 2: SLSQP with increasing alpha (analytical gradient = fast)
    alphas = [500, 2000, 8000, 30000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 180:
            break
        h2 = slsqp_optimize(best_h, n, a, 800, 1e-14)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Adam subgradient for escape
    if time.time() < deadline - 130:
        h3, c3 = adam_subgrad(best_h, n, deadline - 130, lr0=0.004)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()
        if time.time() < deadline - 110:
            h3p = slsqp_optimize(best_h, n, 100000, 500, 1e-14)
            c3p = compute_c5(h3p, n)
            if c3p < best_c:
                best_c, best_h = c3p, h3p.copy()

    # Phase 4: Binary rounding + SLSQP refine
    if time.time() < deadline - 100:
        h_bin = (best_h > 0.45).astype(float)
        h_bin = project_feasible(h_bin, n)
        h_bin = slsqp_optimize(h_bin, n, 5000, 500, 1e-13)
        h_bin = slsqp_optimize(h_bin, n, 50000, 500, 1e-13)
        cb = compute_c5(h_bin, n)
        if cb < best_c:
            best_c, best_h = cb, h_bin.copy()

    # Phase 5: Multiple restarts (binary init + noisy perturbation)
    p5_end = deadline - 25
    for trial in range(40):
        if time.time() > p5_end:
            break
        if trial % 3 == 0:
            h5 = binary_init(n, rng)
        elif trial % 3 == 1:
            h5 = best_h + (0.004 + 0.002 * trial) * rng.randn(n)
            h5 = project_feasible(h5, n)
        else:
            h5 = best_h.copy()
            flip = rng.randint(n // 4, n // 2)
            idx = rng.choice(n, size=flip, replace=False)
            h5[idx] = 1.0 - h5[idx]
            h5 = project_feasible(h5, n)
        h5 = slsqp_optimize(h5, n, 5000, 400, 1e-13)
        h5 = slsqp_optimize(h5, n, 80000, 400, 1e-13)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Final high-alpha polish
    if time.time() < deadline - 5:
        for a in [2000000, 20000000]:
            if time.time() > deadline - 3:
                break
            h6 = slsqp_optimize(best_h, n, a, 600, 1e-15)
            c6 = compute_c5(h6, n)
            if c6 < best_c:
                best_c, best_h = c6, h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
