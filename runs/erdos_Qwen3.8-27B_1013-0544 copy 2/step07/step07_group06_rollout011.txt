def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):

    return float(F), g

    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:

            best = work.copy()

    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())

        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):

    return float(F), g

        F += 0.5 * pen * dev * dev
        g = g + pen * dev
