def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_max(h, n, alpha=100.0):

    w /= w.sum()
    # Gradient: for each j, sum over k of w[k] * dc[k]/dh[j]
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        # dc[k_idx]/dh[j] = (1-h[j+k]) - h[j-k] (with bounds)
        # j range: max(0,-k) to min(n-1, n-1-k)

        j2 = min(n, n - k)
        # Term 1: (1 - h[j+k]) for j in [j1, j2-1] where j+k in [0,n-1]
        # j+k ranges from j1+k to j2-1+k
        # Since k = k_idx - (n-1), j+k = j + k_idx - (n-1)
        # For j in [j1, j2-1]: j+k in [j1+k, j2-1+k]
        # j1+k = max(0,-k)+k, j2-1+k = min(n-1, n-1-k)+k
        # If k>=0: j1=0, j2=n-k, so j+k in [k, n-1] -> all valid
        # If k<0: j1=-k, j2=n, so j+k in [0, n-1+k] -> all valid

            if j_hi >= j_lo:

def lp_step(h, n, eps, solver='GLOP'):

    mc = float(C.max())
    # Only compute gradient for active constraints (top few)
    G = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        j1 = max(0, -k)
        j2 = min(n, n - k)
        if j2 > j1:
            idx1 = j1 + k
            idx2 = j2 - 1 + k
            G[j1:j2] += (1.0 - h[idx1:idx2+1])

            j_hi = j2 - 1

                G[j_lo:j_hi+1] -= h[j_lo-k:j_hi+1-k]

            j_lo = j1
            j_hi = min(j2 - 1, n - 1 + k)
            if j_hi >= j_lo:
                G[j_lo:j_hi+1] -= h[j_lo-k:j_hi+1-k]
    
    # Full LP with all constraints

        j2 = min(n, n - k)

        delta <= eps,

        if mc_try < best_mc - 1e-14:

            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha if shifted.max() < 500 else m
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_c(h, n):
    """Compute all partial derivatives dc[k_idx]/dh[j] for the max gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    grad = np.zeros((n, 2 * n - 1))
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        j1 = max(0, -k)
        j2 = min(n, n - k)
        if j2 <= j1:
            continue
        idx1 = j1 + k
        idx2 = j2 - 1 + k
        grad[j1:j2, k_idx] = 1.0 - h[idx1:idx2 + 1]
        if k >= 0:
            j_lo = max(j1, k)
            j_hi = j2 - 1
            if j_hi >= j_lo:
                grad[j_lo:j_hi + 1, k_idx] -= h[j_lo - k:j_hi + 1 - k]
        else:
            j_lo = j1
            j_hi = min(j2 - 1, n - 1 + k)
            if j_hi >= j_lo:
                grad[j_lo:j_hi + 1, k_idx] -= h[j_lo - k:j_hi + 1 - k]
    return grad, c * dx

def lp_step_active(h, n, eps, n_active=50, solver='GLOP'):
    """LP step using only the most active constraints."""
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    Cdx = C * (2.0 / n)
    # Select active constraints: top n_active by value
    thresh = mc - 0.005
    active = np.where(Cdx >= thresh)[0]
    if len(active) < 5:
        active = np.argsort(Cdx)[-n_active:]
    if len(active) > n_active:
        top_n = n_active
        active = active[np.argsort(Cdx[active])[-top_n:]]
    
    # Build gradient for active constraints only
    G = np.zeros((n, len(active)))
    for idx, k_idx in enumerate(active):
        k = k_idx - (n - 1)
        j1 = max(0, -k)
        j2 = min(n, n - k)
        if j2 <= j1:
            continue
        idx1 = j1 + k
        idx2 = j2 - 1 + k
        G[j1:j2, idx] = 1.0 - h[idx1:idx2 + 1]
        if k >= 0:
            j_lo = max(j1, k)
            j_hi = j2 - 1
            if j_hi >= j_lo:
                G[j_lo:j_hi + 1, idx] -= h[j_lo - k:j_hi + 1 - k]
        else:
            j_lo = j1
            j_hi = min(j2 - 1, n - 1 + k)
            if j_hi >= j_lo:
                G[j_lo:j_hi + 1, idx] -= h[j_lo - k:j_hi + 1 - k]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    ca = Cdx[active]
    constraints = [
        ca + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'GLPK', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
        h_try = h + lam * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        if sm > 500:
            return m
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        if sm > 500:
            # Subgradient of max
            k_max = np.argmax(g)
            k = k_max - (n - 1)
            grad = np.zeros(n)
            j1 = max(0, -k)
            j2 = min(n, n - k)
            if j2 > j1:
                idx1 = j1 + k
                idx2 = j2 - 1 + k
                grad[j1:j2] = 1.0 - h[idx1:idx2 + 1]
            kk = k
            if kk >= 0:
                j_lo = max(j1, kk)
                j_hi = j2 - 1
                if j_hi >= j_lo:
                    grad[j_lo:j_hi + 1] -= h[j_lo - kk:j_hi + 1 - kk]
            else:
                j_lo = j1
                j_hi = min(j2 - 1, n - 1 + kk)
                if j_hi >= j_lo:
                    grad[j_lo:j_hi + 1] -= h[j_lo - kk:j_hi + 1 - kk]
            return grad * 1.0
        w = np.exp(shifted - sm)
        w /= w.sum()
        grad = np.zeros(n)
        for mi in range(2 * n - 1):
            if w[mi] < 1e-10:
                continue
            k = mi - (n - 1)
            j1 = max(0, -k)
            j2 = min(n, n - k)
            if j2 <= j1:
                continue
            idx1 = j1 + k
            idx2 = j2 - 1 + k
            grad[j1:j2] += w[mi] * (1.0 - x[idx1:idx2 + 1])
            if k >= 0:
                j_lo = max(j1, k)
                j_hi = j2 - 1
                if j_hi >= j_lo:
                    grad[j_lo:j_hi + 1] -= w[mi] * x[j_lo - k:j_hi + 1 - k]
            else:
                j_lo = j1
                j_hi = min(j2 - 1, n - 1 + k)
                if j_hi >= j_lo:
                    grad[j_lo:j_hi + 1] -= w[mi] * x[j_lo - k:j_hi + 1 - k]
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def nesterov_phase(h, n, alphas, iters_per_alpha, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        x = best_h.copy()
        x_prev = best_h.copy()
        c5_curr = best_c5
        for t in range(1, iters_per_alpha + 1):
            if time.time() > deadline:
                break
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
            # Compute smoothed gradient
            c = np.correlate(y, 1.0 - y, mode='full') * (2.0 / n)
            cm = c.max()
            shifted = alpha * (c - cm)
            sm = shifted.max()
            if sm > 500:
                w = np.zeros(len(c))
                w[np.argmax(c)] = 1.0
            else:
                w = np.exp(shifted - sm)
                w /= w.sum()
            grad = np.zeros(n)
            for mi in range(2 * n - 1):
                if w[mi] < 1e-12:
                    continue
                k = mi - (n - 1)
                j1 = max(0, -k)
                j2 = min(n, n - k)
                if j2 <= j1:
                    continue
                idx1 = j1 + k
                idx2 = j2 - 1 + k
                grad[j1:j2] += w[mi] * (1.0 - y[idx1:idx2 + 1])
                if k >= 0:
                    j_lo = max(j1, k)
                    j_hi = j2 - 1
                    if j_hi >= j_lo:
                        grad[j_lo:j_hi + 1] -= w[mi] * y[j_lo - k:j_hi + 1 - k]
                else:
                    j_lo = j1
                    j_hi = min(j2 - 1, n - 1 + k)
                    if j_hi >= j_lo:
                        grad[j_lo:j_hi + 1] -= w[mi] * y[j_lo - k:j_hi + 1 - k]
            eta = 1.0 / (1.0 + 0.05 * alpha * t * 0.001)
            x_new = y - eta * grad
            x_new = project_feasible(x_new, n)
            c5_new = compute_c5(x_new, n)
            if c5_new < c5_curr - 1e-14:
                c5_curr = c5_new
            x_prev = x
            x = x_new
        if c5_curr < best_c5 - 1e-13:
            best_c5 = c5_curr
            best_h = x.copy()
    return best_h, best_c5
