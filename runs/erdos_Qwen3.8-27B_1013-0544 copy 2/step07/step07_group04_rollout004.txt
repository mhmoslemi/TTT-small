def lse_weights(c, alpha):
    if alpha <= 0.0:
        return np.ones_like(c)/float(c.size)
    M=float(c.max())
    e=np.exp(alpha*np.minimum(c-M,0.0))
    s=float(e.sum())
    if s <= 0.0:
        q=np.zeros_like(c); q[int(np.argmax(c))]=1.0; return q
    return e/s

def weighted_gradient(h,w,n):
    dx=2.0/n; off=n-1
    wp=w[off:off+n]
    wn=w[off::-1]
    cross=np.zeros(n)
    if n>1:
        for d in range(1,n):
            S=wp[d]+wn[d]
            if S != 0.0:
                cross[:n-d] += S*h[d:]
                cross[d:] += S*h[:n-d]
    const=np.zeros(n)
    if n>1:
        const[1:] += np.cumsum(wp[1:])
        const[:n-1] += np.cumsum(wn[1:])[::-1]
    return dx*(w[off]*(1.0-2.0*h)+const-cross)

def lse_obj(h,n,alpha):
    c=compute_c(h,n); M=float(c.max())
    if alpha <= 0.0: return M
    e=np.exp(alpha*np.minimum(c-M,0.0)); s=float(e.sum())
    if s <=0.0: return M
    return M+float(np.log(s))/alpha

def lse_grad(h,n,alpha):
    c=compute_c(h,n)
    q=lse_weights(c,alpha)
    return weighted_gradient(h,q,n)

def pen_obj(h,n,alpha,mu):
    v=lse_obj(h,n,alpha)
    d=float(h.sum()-0.5*n)
    return v+0.5*mu*d*d

    d=float(h.sum()-0.5*n)

import numpy as np
from scipy.optimize import minimize
import time
def compute_c(h,n): return np.correlate(h,1.0-h,mode='full')*(2.0/n)
def project_box_sum(v,n):
    v=np.nan_to_num(np.asarray(v,float).reshape(-1))
    if v.size!=n:
        w=np.zeros(n); k=min(n,v.size); w[:k]=v[:k]; v=w
    t=0.5*n; lo=float(v.min())-1.0; hi=float(v.max())+1.0
    for _ in range(60):
        mid=0.5*(lo+hi); s=np.clip(v-mid,0.0,1.0).sum()
        if s>t: lo=mid
        else: hi=mid
    v=np.clip(v-0.5*(lo+hi),0.0,1.0)
    for _ in range(40):
        d=t-v.sum()
        if abs(d)<1e-14: break
        m=(v>1e-12)&(v<1.0-1e-12); c=int(m.sum())
        if c: v[m]+=d/c
        else: v+=d/n
        v=np.clip(v,0.0,1.0)
    return v
def lse_weights(c,a):
    if a<=0.0: return np.ones_like(c)/float(c.size)
    M=float(c.max()); e=np.exp(a*np.minimum(c-M,0.0)); s=float(e.sum())
    if s<=0.0:
        q=np.zeros_like(c); q[int(np.argmax(c))]=1.0; return q
    return e/s
def weighted_gradient(h,w,n):
    dx=2.0/n; off=n-1
    wp=w[off:off+n]; wn=w[off::-1]
    cross=np.zeros(n)
    if n>1:
        for d in range(1,n):
            S=wp[d]+wn[d]
            if S!=0.0:
                cross[:n-d]+=S*h[d:]
                cross[d:]+=S*h[:n-d]
    const=np.zeros(n)
    if n>1:
        const[1:]+=np.cumsum(wp[1:])
        const[:n-1]+=np.cumsum(wn[1:])[::-1]
    return dx*(w[off]*(1.0-2.0*h)+const-cross)
def lse_obj(h,n,a):
    c=compute_c(h,n); M=float(c.max())
    if a<=0.0: return M
    e=np.exp(a*np.minimum(c-M,0.0)); s=float(e.sum())
    if s<=0.0: return M
    return M+float(np.log(s))/a
def lse_grad(h,n,a):
    q=lse_weights(compute_c(h,n),a)
    return weighted_gradient(h,q,n)
def pen_obj(h,n,a,mu):
    v=lse_obj(h,n,a); d=float(h.sum()-0.5*n)
    return v+0.5*mu*d*d
def pen_grad(h,n,a,mu):
    g=lse_grad(h,n,a); d=float(h.sum()-0.5*n)
    return g+mu*d
def sum_fun(h,n): return float(h.sum()-0.5*n)
def sum_jac(h,n): return np.ones_like(h)
def lbfgsb_phase(h,n,a,mu,mi):
    h=project_box_sum(h,n); b=[(0.0,1.0)]*n
    try:
        r=minimize(pen_obj,h,args=(n,a,mu),method='L-BFGS-B',jac=pen_grad,bounds=b,
                  options={'maxiter':mi,'maxfun':4*mi,'ftol':1e-14,'gtol':1e-8})
        if r.x is not None: return project_box_sum(r.x,n)
    except Exception: pass
    return h
def slsqp_phase(h,n,a,mi):
    h=project_box_sum(h,n); b=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':sum_fun,'args':(n,),'jac':sum_jac}]
    try:
        r=minimize(lse_obj,h,args=(n,a),method='SLSQP',jac=lse_grad,bounds=b,constraints=cons,
                  options={'maxiter':mi,'ftol':1e-12})
        if r.x is not None: return project_box_sum(r.x,n)
    except Exception: pass
    return h

def compute_c(h,n):
    return np.correlate(h,1.0-h,mode='full')*(2.0/n)

def project_box_sum(v,n):

        else: hi=mid
    v=np.clip(v-0.5*(lo+hi),0.0,1.0)
    for _ in range(40):

def lse_weights(c,a):

    return e/s

def weighted_gradient(h,w,n):

    cross=np.zeros(n)
    if n>1:

                cross[:n-d]+=S*h[d:]
                cross[d:]+=S*h[:n-d]
    const=np.zeros(n)
    if n>1:
        const[1:]+=np.cumsum(wp[1:])
        const[:n-1]+=np.cumsum(wn[1:])[::-1]
    return dx*(w[off]*(1.0-2.0*h)+const-cross)

def lse_obj(h,n,a):

    if a<=0.0: return M
    e=np.exp(a*np.minimum(c-M,0.0)); s=float(e.sum())
    if s<=0.0: return M
    return M+float(np.log(s))/a

def lse_grad(h,n,a):

    return weighted_gradient(h,q,n)

def pen_obj(h,n,a,mu):
    v=lse_obj(h,n,a); d=float(h.sum()-0.5*n)
    return v+0.5*mu*d*d

def pen_grad(h,n,a,mu):

    return g+mu*d

def sum_fun(h,n):
