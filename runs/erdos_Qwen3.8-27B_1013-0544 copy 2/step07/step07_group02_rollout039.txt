```
def soft_val_grad(h, n, alpha):
    h=np.asarray(h,float)
    a=float(alpha)
    if a < 1e-6: a=1e-6
    P=float(np.sum(h))
    prefix=np.empty(n+1, dtype=float)
    prefix[0]=0.0
    np.cumsum(h, out=prefix[1:])
    p=np.correlate(h,h,mode='full')
    pl=p[n-1:2*n-1]
    idx=n-np.arange(n)
    Cpos=prefix[idx]-pl
    Cneg=np.empty(n, dtype=float)
    Cneg[0]=Cpos[0]
    if n>1:
        Cneg[1:]=P-prefix[1:]-pl[1:]
    max_raw=float(max(Cpos.max(), Cneg.max()))
    zpos=a*(Cpos-max_raw)
    zneg=a*(Cneg-max_raw)
    ppos=np.exp(zpos)
    pneg=np.exp(zneg)
    p0=float(ppos[0])
    total=p0 + float(np.sum(ppos[1:])) + (float(np.sum(pneg[1:])) if n>1 else 0.0)
    if total <= 0.0 or not np.isfinite(total):
        total=1.0
    wpos=ppos/total
    wneg=pneg/total
    wpos[0]=0.0
    wneg[0]=0.0
    w0=p0/total
    if n>1:
        cum=np.cumsum(wpos[::-1])[::-1]
        suffix=np.empty(n+1, dtype=float)
        suffix[:n]=cum
        suffix[n]=0.0
        ind_pos=suffix[1:]
    else:
        ind_pos=np.zeros(1)
    ind_neg=np.cumsum(wneg)
    s=wpos+wneg
    s[0]=0.0
    L=np.convolve(h,s,mode='full')[:n]
    R=np.correlate(h,s,mode='full')[n-1::-1]
    g=w0*(1.0-2.0*h) + ind_pos + ind_neg - (L+R)
    val=max_raw + np.log(total)/a
    return float(val), g.astype(float, copy=False)
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def make_even(x, lo, hi):
    x = int(round(float(x)))
    if x % 2 != 0:
        x += 1
    if x < lo:
        x = lo
    if x > hi:
        x = hi
    if x % 2 != 0:
        x -= 1
    return x


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float)
    bad = ~np.isfinite(y)
    if bad.any():
        y = np.where(bad, 0.0, y)
    y = np.clip(y, 0.0, 1.0)
    S = 0.5 * float(n)
    if abs(float(np.sum(y)) - S) <= 1e-12:
        return y.copy()

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    tau = 0.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        x = np.clip(y - tau, 0.0, 1.0)
        s = float(np.sum(x)) - S
        if abs(s) <= 1e-13:
            break
        if s < 0.0:
            hi = tau
        else:
            lo = tau

    x = np.clip(y - tau, 0.0, 1.0)
    diff = S - float(np.sum(x))
    if abs(diff) > 1e-13:
        interior = np.nonzero((x > 1e-10) & (x < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            x[interior] = x[interior] + diff / float(len(interior))
            x = np.clip(x, 0.0, 1.0)

    diff = S - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            if x[idx] + diff < 1.0:
                x[idx] = x[idx] + diff
            else:
                x[idx] = 1.0
        else:
            idx = int(np.argmax(x))
            if x[idx] + diff > 0.0:
                x[idx] = x[idx] + diff
            else:
                x[idx] = 0.0
    return x


def c5_value(h, n):
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * (2.0 / float(n)))


def interpolate_h(h, n_to):
    n = len(h)
    if n == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n)
    x_new = np.linspace(0.0, 2.0, n_to)
    return np.interp(x_new, x_old, h)


def soft_val_grad(h, n, alpha, need_grad=True):
    h = np.asarray(h, dtype=float)
    a = float(alpha)
    if a < 1e-6:
        a = 1e-6

    P = float(np.sum(h))
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])

    p = np.correlate(h, h, mode="full")
    pl = p[n - 1:2 * n - 1]
    idx = n - np.arange(n)
    Cpos = prefix[idx] - pl

    Cneg = np.empty(n, dtype=float)
    Cneg[0] = Cpos[0]
    if n > 1:
        Cneg[1:] = P - prefix[1:] - pl[1:]

    max_raw = float(max(np.max(Cpos), np.max(Cneg)))
    zpos = a * (Cpos - max_raw)
    zneg = a * (Cneg - max_raw)
    ppos = np.exp(zpos)
    pneg = np.exp(zneg)
    p0 = float(ppos[0])
    total = p0 + float(np.sum(ppos[1:]))
    if n > 1:
        total += float(np.sum(pneg[1:]))
    if (not np.isfinite(total)) or total < 1.0:
        total = 1.0

    val = max_raw + np.log(total) / a
    if not need_grad:
        return float(val), None

    wpos = ppos / total
    wneg = pneg / total
    wpos[0] = 0.0
    wneg[0] = 0.0
    w0 = p0 / total

    pw = np.cumsum(wpos)
    ind_pos = pw[n - 1::-1]
    ind_neg = np.cumsum(wneg)

    s = wpos + wneg
    s[0] = 0.0
    L = np.convolve(h, s, mode="full")[:n]
    R = np.correlate(h, s, mode="full")[n - 1::-1]

    g = w0 * (1.0 - 2.0 * h) + ind_pos + ind_neg - (L + R)
    return float(val), g


def soft_objective(x, n, alpha, mu):
    val, _ = soft_val_grad(x, n, alpha, False)
    s = float(np.sum(x)) - 0.5 * float(n)
    return float(val + float(mu) * s * s)


def soft_gradient(x, n, alpha, mu):
    _, g = soft_val_grad(x, n, alpha, True)
    s = float(np.sum(x)) - 0.5 * float(n)
    return g + (2.0 * float(mu) * s)


def lbfgsb_refine(h, n, alpha, mu, maxiter, t_end):
    rem = t_end - time.time()
    if rem < 4.0:
        return project_box_sum(h, n)

    x0 = project_box_sum(h, n)
    scale = max(1.0, 400.0 / float(n))
    allowed = int(min(350, max(10, rem * 10.0 * scale)))
    mi = int(min(int(maxiter), allowed))
    if mi < 10:
        return x0

    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": mi,
        "maxfun": mi * 10,
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxls": 50,
        "iprint": -1,
    }

    try:
        res = minimize(
            soft_objective,
            x0,
            args=(n, float(alpha), float(mu)),
            method="L-BFGS-B",
            fprime=soft_gradient,
            bounds=bounds,
            options=opts,
        )
        return project_box_sum(res.x, n)
    except Exception:
        return x0


def refine_sequence(h, n, alphas, mu, maxiter, t_end):
    x = project_box_sum(h, n)
    for a in alphas:
        if time.time() >= t_end - 2.0:
            break
        x = lbfgsb_refine(x, n, a, mu, maxiter, t_end)
    return x


def make_n_list(n0):
    n0 = make_even(n0, 64, 800)
    res = []
    res.append(n0)
    fracs = (0.5, 0.75, 0.9, 1.25, 1.5)
    for f in fracs:
        res.append(make_even(int(round(f * float(n0))), 64, 800))
    fixed = (96, 128, 160, 200, 256, 320, 400, 512, 640)
    for v in fixed:
        res.append(v)
    out = []
    for v in res:
        v = make_even(v, 64, 800)
        if v not in out:
            out.append(v)
    return out


def binary_split(b, n):
    ones = []
    zeros = []
    for i in range(n):
        if b[i] > 0.5:
            ones.append(i)
        else:
            zeros.append(i)
    return ones, zeros


def binary_from_fractional(h, n):
    x = project_box_sum(h, n)
    order = np.argsort(-x, kind="mergesort")
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    return b


def binary_counts(b):
    return np.correlate(b, 1.0 - b, mode="full").astype(float, copy=False)


def apply_flip(c, b, i, delta):
    n = len(b)
    ctr = n - 1
    if i > 0:
        c[ctr - i:ctr] += -delta * b[i - 1::-1]
    if i + 1 < n:
        c[ctr + 1:ctr + n - i] += delta * (1.0 - b[i + 1:])
    b[i] += delta


def binary_tail_score(c):
    mx = float(np.max(c))
    d = c - (mx - 1.0)
    return float(np.sum(np.maximum(d, 0.0)))


def binary_top_lags(c):
    m = len(c)
    k = min(40, m)
    if k == m:
        out = []
        for i in range(m):
            out.append(i)
        return out
    idx = np.argpartition(c, -k)[-k:]
    out = []
    for v in idx:
        out.append(int(v))
    return out


def binary_propose(b, c, top_lags, ones, zeros, rng):
    n = len(b)
    if len(ones) == 0 or len(zeros) == 0:
        return None

    if top_lags and rng.rand() < 0.5:
        idx = int(top_lags[rng.randint(len(top_lags))])
        q = idx - (n - 1)
        found = None

        if q > 0:
            m = n - q
            mask = (b[:m] > 0.5) & (b[q:] <= 0.5)
            pos = np.flatnonzero(mask)
            if len(pos) > 0:
                i = int(pos[rng.randint(len(pos))])
                j = i + q
                if rng.rand() < 0.5:
                    z = int(zeros[rng.randint(len(zeros))])
                    if z != i:
                        found = (i, z)
                else:
                    o = int(ones[rng.randint(len(ones))])
                    if o != j:
                        found = (o, j)

        elif q < 0:
            l = -q
            mask = (b[l:] > 0.5) & (b[:n - l] <= 0.5)
            pos = np.flatnonzero(mask) + l
            if len(pos) > 0:
                i = int(pos[rng.randint(len(pos))])
                j = i - l
                if rng.rand() < 0.5:
                    z = int(zeros[rng.randint(len(zeros))])
                    if z != i:
                        found = (i, z)
                else:
                    o = int(ones[rng.randint(len(ones))])
                    if o != j:
                        found = (o, j)

        if found is not None:
            return found

    a = int(ones[rng.randint(len(ones))])
    z = int(zeros[rng.randint(len(zeros))])
    return (a, z)


def binary_local_search(b0, n, t_end, rng):
    b = binary_from_fractional(b0, n)
    ones, zeros = binary_split(b, n)
    c = binary_counts(b)

    cur_mx = float(np.max(c))
    cur_tail = binary_tail_score(c)
    cur_score = cur_mx + 0.005 * cur_tail

    best_b = b.copy()
    best_c = c.copy()
    best_mx = cur_mx
    best_score = cur_score

    top = binary_top_lags(c)
    T = 0.3
    it = 0

    while time.time() < t_end:
        if it > 0 and it % 250 == 0:
            top = binary_top_lags(c)
            T = max(0.02, T * 0.97)
            if it % 2500 == 0:
                ones, zeros = binary_split(b, n)

        pair = binary_propose(b, c, top, ones, zeros, rng)
        if pair is None:
            break
        a, z = pair

        if not (b[a] > 0.5 and b[z] <= 0.5):
            if b[a] <= 0.5 and b[z] > 0.5:
                a, z = z, a
            else:
                it += 1
                continue

        da = 1.0 - 2.0 * float(b[a])
        dz = 1.0 - 2.0 * float(b[z])

        c_trial = c.copy()
        apply_flip(c_trial, b, a, da)
        apply_flip(c_trial, b, z, dz)

        mx = float(np.max(c_trial))
        tail = binary_tail_score(c_trial)
        score = mx + 0.005 * tail

        accept = False
        if mx < cur_mx - 1e-12 or score < cur_score - 1e-12:
            accept = True
        elif T > 1e-4:
            d = score - cur_score
            if d > 0.0 and d < 50.0 * T:
                prob = float(np.exp(-d / T))
                if rng.rand() < prob:
                    accept = True
            elif d <= 0.0:
                accept = True

        b[z] -= dz
        b[a] -= da

        if accept:
            b[a] += da
            b[z] += dz

            try:
                ones.remove(a)
            except ValueError:
                pass
            try:
                zeros.remove(z)
            except ValueError:
                pass
            ones.append(z)
            zeros.append(a)

            c = c_trial
            cur_mx = mx
            cur_tail = tail
            cur_score = score

            c5_cur = 2.0 * cur_mx / float(n)
            c5_best = 2.0 * best_mx / float(n)
            if c5_cur < c5_best - 1e-13 or (abs(c5_cur - c5_best) <= 1e-13 and score < best_score - 1e-12):
                best_b = b.copy()
                best_c = c_trial.copy()
                best_mx = cur_mx
                best_score = score

        it += 1

    return best_b, 2.0 * float(best_mx) / float(n), best_score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 10.0)
    hard = t0 + total
    cont_end = t0 + 0.42 * total
    bin_end = min(t0 + 0.78 * total, hard - 45.0)
    if cont_end > hard - 30.0:
        cont_end = hard - 30.0
    if bin_end < cont_end + 10.0:
        bin_end = max(cont_end + 10.0, hard - 15.0)
    if bin_end > hard - 10.0:
        bin_end = hard - 10.0

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.ndim != 1 or len(h0) < 16:
            raise ValueError("bad initial")
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)

    n0 = make_even(n0, 64, 800)
    if len(h0) != n0:
        h0 = interpolate_h(h0, n0)
    h0 = project_box_sum(h0, n0)

    best_h = h0.copy()
    n_best = n0
    best_c = c5_value(best_h, n_best)

    h_cur = refine_sequence(best_h, n_best, (20.0, 100.0, 500.0), 5000.0, 220, cont_end)
    cv = c5_value(h_cur, n_best)
    if cv < best_c - 1e-13:
        best_h = h_cur.copy()
        best_c = cv

    ns = make_n_list(n_best)
    for nn in ns:
        if time.time() >= cont_end - 10.0:
            break
        h_local = interpolate_h(best_h, nn)
        h_local = refine_sequence(h_local, nn, (20.0, 200.0), 5000.0, 180, cont_end)
        cv = c5_value(h_local, nn)
        if cv < best_c - 1e-13:
            best_h = h_local.copy()
            n_best = nn
            best_c = cv

        if nn <= 256 and time.time() < cont_end - 25.0:
            h_rand = project_box_sum(0.5 + 0.25 * rng.randn(nn), nn)
            h_rand = refine_sequence(h_rand, nn, (30.0, 300.0), 5000.0, 120, cont_end)
            cv = c5_value(h_rand, nn)
            if cv < best_c - 1e-13:
                best_h = h_rand.copy()
                n_best = nn
                best_c = cv

    if time.time() < bin_end - 25.0:
        cand = []
        vals = (
            n_best,
            make_even(n_best // 2, 64, 800),
            make_even((3 * n_best) // 2, 64, 800),
            256,
            320,
            400,
            512,
        )
        for v in vals:
            v = make_even(v, 64, 800)
            if v not in cand:
                cand.append(v)
        for v in make_n_list(n_best):
            v = make_even(v, 64, 800)
            if v not in cand:
                cand.append(v)

        for nn in cand:
            if time.time() >= bin_end - 25.0:
                break
            alloc = min(bin_end - time.time() - 15.0, 170.0)
            if alloc < 12.0:
                break

            b0 = binary_from_fractional(interpolate_h(best_h, nn), nn)
            b_best_b, b_c5, _ = binary_local_search(b0, nn, time.time() + alloc, rng)

            if b_c5 < best_c - 1e-13:
                best_h = b_best_b.astype(float).copy()
                n_best = nn
                best_c = b_c5

            if time.time() < hard - 60.0:
                hb = project_box_sum(b_best_b, nn)
                hb = refine_sequence(hb, nn, (100.0, 800.0), 5000.0, 150, hard)
                cv = c5_value(hb, nn)
                if cv < best_c - 1e-13:
                    best_h = hb.copy()
                    n_best = nn
                    best_c = cv

    if time.time() < hard - 5.0:
        h_fin = refine_sequence(best_h, n_best, (50.0, 300.0, 1500.0), 5000.0, 150, hard)
        cv = c5_value(h_fin, n_best)
        if cv < best_c - 1e-13:
            best_h = h_fin.copy()
            best_c = cv

    best_h = project_box_sum(best_h, n_best)
    best_c = c5_value(best_h, n_best)
    return best_h, float(best_c), int(n_best)
```
