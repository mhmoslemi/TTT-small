        delta[off + ks] -= dj[:,None] * h[j-ks][None,:]
    
    return np.max(c_base[None,:] + dx*delta, axis=1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        v[m] += d / c if c else d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    q = e / e.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    nz = np.flatnonzero(q > 1e-15)
    for idx in nz:
        k = idx - off
        w = q[idx]
        if k > 0:
            g[:n-k] += w * (1.0 - x[k:])
            g[k:] -= w * x[:n-k]
        elif k < 0:
            kk = -k
            g[kk:] += w * (1.0 - x[:n-kk])
            g[:n-kk] -= x[kk:] * w
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = x.sum() - 0.5 * n
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = x.sum() - 0.5 * n
    return lse_grad(x, n, alpha) + mu * d

def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 4 * maxiter,
                                "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
                 "jac": lambda x: np.ones(n)}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_grad, bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def pair_eval_fast(c_base, h, i, j, xs, n):
    dx = 2.0 / n
    off = n - 1
    x0, y0 = h[i], h[j]
    S = x0 + y0
    di = (xs - x0)[:, None]
    dj = ((S - xs) - y0)[:, None]
    L = 2 * n - 1
    delta = np.zeros((len(xs), L))
    k_lo1, k_hi1 = max(-(n-1), -i), min(n-1, n-1-i)
    if k_lo1 <= k_hi1:
        ks = np.arange(k_lo1, k_hi1 + 1)
        delta[:, off + ks] += di * (1.0 - h[i + ks][None, :])
    k_lo2, k_hi2 = max(-(n-1), i-n+1), min(n-1, i)
    if k_lo2 <= k_hi2:
        ks = np.arange(k_lo2, k_hi2 + 1)
        delta[:, off + ks] -= di * h[i - ks][None, :]
    k_lo1j, k_hi1j = max(-(n-1), -j), min(n-1, n-1-j)
    if k_lo1j <= k_hi1j:
        ks = np.arange(k_lo1j, k_hi1j + 1)
        delta[:, off + ks] += dj * (1.0 - h[j + ks][None, :])
    k_lo2j, k_hi2j = max(-(n-1), j-n+1), min(n-1, j)
    if k_lo2j <= k_hi2j:
        ks = np.arange(k_lo2j, k_hi2j + 1)
        delta[:, off + ks] -= dj * h[j - ks][None, :]
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c, i, j, n):
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo, hi = max(0.0, S - 1.0), min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = np.array([lo, hi, x0])
    a, b = max(lo, x0 - 0.03), min(hi, x0 + 0.03)
    if b - a > 1e-12:
        xs = np.unique(np.concatenate([xs, np.linspace(lo, hi, 16), np.linspace(a, b, 48)]))
    vals = pair_eval_fast(c, h, i, j, xs, n)
    for _ in range(2):
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1] if len(xs) > 1 else xs[0]
        elif k == len(xs) - 1:
            a, b = (xs[-2] if len(xs) > 1 else xs[0]), xs[-1]
        else:
            a, b = xs[k-1], xs[k+1]
        if b - a < 1e-11:
            break
        xs = np.linspace(a, b, 64)
        vals = pair_eval_fast(c, h, i, j, xs, n)
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(np.clip(S - xb, lo, hi))
        return xb, yb
    return None

def pair_local_search(h, deadline, seed=1, max_noimp=3000):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    if n < 4 or time.time() > deadline:
        return h, bestc
    rng = np.random.RandomState(seed)
    noimp = 0
    off = n - 1
    L2 = L = 1000
    while time.time() < deadline and noimp < max_noimp:
        if rng.random() < 0.8:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-10
            w /= w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            h[i], h[j] = res
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc - 1e-15:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
        if L % L2 == 0:
            L2 *= 2
        L += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)
