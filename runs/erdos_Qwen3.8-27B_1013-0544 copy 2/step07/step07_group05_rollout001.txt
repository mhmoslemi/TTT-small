```
def project_box_sum(x,target):
    x=np.asarray(x,float)
    n=x.size
    y=np.clip(x,0,1)
    # bisection for theta
    lo=float(np.min(x)) - 1.0
    hi=float(np.max(x)) + 1.0
    for _ in range(60):
       mid=0.5*(lo+hi)
       y=np.clip(x-mid,0,1)
       s=y.sum()
       if s>target: lo=mid
       else: hi=mid
    y=np.clip(x-0.5*(lo+hi),0,1)
    diff=target-y.sum()
    for _ in range(4):
       if abs(diff)<1e-13: break
       mask=(y>1e-12)&(y<1-1e-12)
       cnt=mask.sum()
       if cnt: y[mask]+=diff/cnt
       else: y[0]+=diff
       y=np.clip(y,0,1)
       diff=target-y.sum()
    return y
```

```
def grad_weighted(h,w,n,dx):
    if n == 1:
        return (1.0 - 2.0*h) * w[0] * dx
    mid=n-1
    wp=np.zeros(n); wn=np.zeros(n)
    wp[1:] = w[mid+1:]
    wn[1:] = w[:mid][::-1]
    pref_p=np.cumsum(wp)
    pref_n=np.cumsum(wn)
    idx=np.arange(n)
    T = pref_p + pref_n[n-1-idx]
    s=wp+wn
    kern=np.zeros(2*n-1)
    kern[mid+1:] = s[1:]
    if mid>0:
        kern[:mid][::-1] = s[1:]
    neigh=np.convolve(h,kern,mode='full')[mid:mid+n]
    return (T + w[mid]*(1.0 - 2.0*h) - neigh) * dx
```

```python
import time
import numpy as np
import scipy.optimize as so

OPT = {}
SLOPT = {}


def is_better(cand, cur):
    return np.isfinite(cand) and cand < cur


def erdos_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def correlations(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def project_box_sum(x, target):
    x = np.array(x, dtype=float).copy()
    if x.size == 0:
        return x
    target = float(np.clip(target, 0.0, float(x.size)))
    x = np.nan_to_num(x, 0.0, 1.0)
    y = np.clip(x, 0.0, 1.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt:
            y[mask] += diff / cnt
        else:
            y[0] += diff
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def resample_n(h, n, nh):
    h = np.asarray(h, dtype=float)
    if nh == n:
        return project_box_sum(h, 0.5 * nh)
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, h)
    return project_box_sum(hh, 0.5 * nh)


def softmax_values(c, beta):
    c = np.asarray(c, dtype=float)
    maxc = float(np.max(c))
    if beta <= 0.0:
        w = np.full(c.shape, 1.0 / c.size)
        return maxc, w
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    s = float(ew.sum())
    f = maxc + float(np.log(s)) / beta
    if s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / s
    return f, w


def grad_weighted(h, w, n, dx):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    if n == 1:
        return (1.0 - 2.0 * h) * w[0] * dx
    mid = n - 1
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    pref_p = np.cumsum(wp)
    pref_n = np.cumsum(wn)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    s = wp + wn
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    if mid > 0:
        kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (consts + w[mid] * (1.0 - 2.0 * h) - neigh) * dx


def opt_fun(x):
    n = int(OPT['n'])
    c = correlations(x, n)
    f, w = softmax_values(c, float(OPT['beta']))
    g = grad_weighted(x, w, n, 2.0 / n)
    res = float(x.sum() - OPT['target'])
    rho = float(OPT['rho'])
    if rho != 0.0:
        f += 0.5 * rho * res * res
        g = g + rho * res
    return float(f), g


def refine_lbfgs(h, beta, rho, maxiter, deadline):
    h = project_box_sum(np.asarray(h, dtype=float), 0.5 * len(h))
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0 or n < 2:
        return h, erdos_c5(h, n)
    OPT['n'] = n
    OPT['target'] = target
    OPT['beta'] = float(beta)
    OPT['rho'] = float(rho)
    if n > 500:
        n_iter = max(8, int(maxiter * 0.55))
    else:
        n_iter = int(maxiter)
    mx = int(min(n_iter, max(8, int(rem / 2.5))))
    try:
        res = so.minimize(
            opt_fun, h, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': mx, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30, 'disp': False}
        )
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def sl_obj(x):
    n = int(SLOPT['n'])
    c = correlations(x, n)
    f, w = softmax_values(c, float(SLOPT['beta']))
    g = grad_weighted(x, w, n, 2.0 / n)
    SLOPT['cache_x'] = np.asarray(x, dtype=float).copy()
    SLOPT['cache_g'] = g
    return float(f)


def sl_jac(x):
    cx = SLOPT.get('cache_x')
    if cx is not None and np.array_equal(np.asarray(x, dtype=float), cx):
        return SLOPT['cache_g'].copy()
    sl_obj(x)
    return SLOPT['cache_g'].copy()


def sl_con_fun(x):
    return float(x.sum() - SLOPT['target'])


def sl_con_jac(x):
    return np.ones(int(SLOPT['n']), dtype=float)


def refine_slsqp(h, beta, maxiter, deadline):
    h = project_box_sum(np.asarray(h, dtype=float), 0.5 * len(h))
    n = len(h)
    if n > 480 or n < 2:
        return h, erdos_c5(h, n)
    rem = deadline - time.time()
    if rem < 8.0:
        return h, erdos_c5(h, n)
    SLOPT['n'] = n

    SLOPT['beta'] = float(beta)

    SLOPT['cache_g'] = None

    try:

            constraints=[{'type': 'eq', 'fun': sl_con_fun, 'jac': sl_con_jac}],
            options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-12}
        )
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)


def pair_A(i, j, h, n, lag):
    A = np.zeros(2 * n - 1)
    b = 1.0 - h
    q = i - lag
    m = (q >= 0) & (q < n)
    A[m] += b[q[m]]
    p = i + lag
    m = (p >= 0) & (p < n)
    A[m] -= h[p[m]]
    q = j - lag
    m = (q >= 0) & (q < n)
    A[m] -= b[q[m]]
    p = j + lag
    m = (p >= 0) & (p < n)
    A[m] += h[p[m]]
    return A


def pair_val(d, c, A, qidx, qscale):
    v = c + d * A
    m = float(np.max(v))
    add = d * d * qscale
    for q in qidx:
        vv = v[q] + add
        if vv > m:
            m = vv
    return m


def one_pair(h, n, i, j, c, cur, lag, deadline, use_scalar):

    best = cur
    bd = 0.0
    span = hi - lo
    pts = [lo, hi, 0.5 * (lo + hi), lo + 0.25 * span, lo + 0.75 * span]
    if lo <= 0.0 <= hi:

            bd = float(d)
    if use_scalar and span > 1e-10 and time.time() < deadline - 1.0:
        try:

                pair_val, args=(c, A, qidx, qscale),

            )
            if rs.x is not None:
                d = float(rs.x)
                if d < lo:
                    d = lo
                elif d > hi:
                    d = hi
                val = pair_val(d, c, A, qidx, qscale)
                if val < best:
                    best = val
                    bd = d

            pass
    if bd < lo:
        bd = lo
    elif bd > hi:

    if best < cur - 1e-13 and abs(bd) > 1e-14:
        new_i = vi + bd

        h[i] = float(np.clip(new_i, 0.0, 1.0))

        c = c + bd * A
        add = bd * bd * qscale
        for q in qidx:

            c = correlations(h, n)

        g = grad_weighted(h, w, n, 2.0 / n)
        g = g - float(np.mean(g))
        recv = np.where(h < 1.0 - 1e-7)[0]
        don = np.where(h > 1e-7)[0]
        if recv.size == 0 or don.size == 0:
            break
        R = min(K, int(recv.size))
        D = min(K, int(don.size))
        ridx = recv[np.argsort(g[recv])[:R]]
        didx = don[np.argsort(-g[don])[:D]]
        if R == 0 or D == 0:
            break
        score = (g[didx][:, None] - g[ridx][None, None])
        score *= np.minimum(h[didx][:, None], 1.0 - h[ridx][None, :])
        flat = np.argsort(score.ravel())[::-1]
        pairs = []
        for fl in flat[:max_pairs]:
            a = int(fl) // R
            b = int(fl) % R
            i = int(ridx[b])
            j = int(didx[a])
            if i != j and h[j] > 1e-9 and h[i] < 1.0 - 1e-9:
                pairs.append((i, j))
        rng.shuffle(pairs)
        acc = 0
        for i, j in pairs:
            if time.time() > deadline - 2.0:
                break
            ok, _, c2, cur2 = one_pair(h, n, i, j, c, cur, lag, deadline, True)
            if ok:
                c = c2
                cur = cur2
                acc += 1
                if acc >= 25:
                    c = correlations(h, n)
                    cur = float(np.max(c))
                    acc = 0
        if acc == 0:
            for _ in range(36):
                if time.time() > deadline - 2.0:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                ok, _, c2, cur2 = one_pair(h, n, i, j, c, cur, lag, deadline, False)
                if ok:
                    c = c2
                    cur = cur2
                    acc += 1
                    if acc > 18:
                        c = correlations(h, n)
                        cur = float(np.max(c))
                        break
            if acc == 0:
                fail += 1
                if fail >= 4:
                    break
            else:
                fail = 0
        else:
            fail = 0
    h = project_box_sum(h, 0.5 * n)
    c = correlations(h, n)
    cur = float(np.max(c))
    return h, cur


def nearest_ns(cands, target, limit):
    scored = []
    for x in cands:
        scored.append((abs(int(x) - int(target)), int(x)))
    scored.sort()
    out = []
    for _, x in scored[:limit]:
        if x not in out:
            out.append(x)
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)
    seed = int(seed)

    try:
        h_in = np.asarray(initial_h_values, dtype=float)
        if h_in.size == 0:
            raise RuntimeError
        h = h_in.copy()

        h = np.full(320, 0.5, dtype=float)

    total = max(30.0, deadline - t0)

    end2 = t0 + 0.55 * total
    end_explore = deadline - 80.0

    sched = (
        (20.0, 1000.0, 45),
        (100.0, 1000.0, 40),
        (500.0, 5000.0, 36),
        (2000.0, 10000.0, 32),

    )
    for beta, rho, mx in sched:
        if time.time() > end1 - 4.0:

        cand, c = refine_lbfgs(best_h, beta, rho, mx, deadline)
        if is_better(c, best_c):
            best_h = cand.copy()

    if time.time() < end1 - 6.0:
        cand, c = refine_slsqp(best_h, 5000.0, 18, deadline)

    if time.time() < end2 - 12.0:
        best_h, best_c = pair_polish_fast(best_h, best_n, end2, seed + 1)

    alts = nearest_ns((280, 320, 360, 420, 480), best_n, 3)
    for nh in alts:

            continue
        t = time.time()
        limit = min(end_explore, t + 85.0)
        hh = resample_n(best_h, best_n, nh)
        c = erdos_c5(hh, nh)
        d1 = min(limit, time.time() + 32.0)
        hh, c = refine_lbfgs(hh, 1200.0, 10000.0, 35, d1)
        if is_better(c, best_c + 0.05):
            d2 = min(limit, time.time() + 26.0)
            hh, c = refine_lbfgs(hh, 30000.0, 100000.0, 28, d2)
            if is_better(c, best_c):
                best_h = hh.copy()
                best_n = nh
                best_c = c
            d3 = min(limit, time.time() + 32.0)
            if time.time() < d3 - 8.0:
                hh, c = pair_polish_fast(hh, nh, d3, seed + nh)
                if is_better(c, best_c):
                    best_h = hh.copy()
                    best_n = nh
                    best_c = c

    if time.time() < end_explore - 45.0:
        for thr in (0.35, 0.5, 0.65):
            if time.time() > end_explore - 40.0:
                break
            hh = (best_h > thr).astype(float)
            hh = project_box_sum(hh, 0.5 * best_n)
            c = erdos_c5(hh, best_n)
            if c < best_c + 0.08:
                d = min(end_explore - 20.0, time.time() + 35.0)
                hh, c = refine_lbfgs(hh, 800.0, 10000.0, 28, d)
                if c < best_c + 0.03:
                    d = min(end_explore - 15.0, time.time() + 25.0)
                    hh, c = pair_polish_fast(hh, best_n, d, seed + int(thr * 100))
                    if is_better(c, best_c):
                        best_h = hh.copy()
                        best_n = best_n
                        best_c = c

    if time.time() < end_explore - 35.0:
        rng = np.random.RandomState(seed + 777)
        x = np.linspace(0.0, 2.0, best_n)
        f = np.arange(1, 7)
        a = 0.35 * rng.standard_normal(6)
        b = 0.35 * rng.standard_normal(6)
        hh = 0.5 + (a * np.cos(np.pi * f[:, None] * x[None, :]) +
                    b * np.sin(np.pi * f[:, None] * x[None, :])).sum(axis=0)

            if is_better(c, best_c):
                best_h = hh.copy()
                best_c = c
                d = min(end_explore - 12.0, time.time() + 25.0)
                hh, c = pair_polish_fast(hh, best_n, d, seed + 999)

                    best_c = c

        cand, c = refine_lbfgs(best_h, 400000.0, 100000.0, 18, deadline)

    if time.time() < deadline - 15.0:

    best_c = float(erdos_c5(best_h, best_n))
    return best_h, best_c, int(best_n)
```
