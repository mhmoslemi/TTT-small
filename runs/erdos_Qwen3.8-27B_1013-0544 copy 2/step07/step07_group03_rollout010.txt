                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                # Apply swap
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < 2*n-1)
                delta[msk] += 1.0 - h_b[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < 2*n-1)
                delta[msk] -= h_b[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < 2*n-1)
                delta[msk] -= 1.0 - h_b[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < 2*n-1)
                delta[msk] += h_b[jmn[msk]]
                delta[n - 1] -= 2.0
                delta[n - 1 + (best_j - i)] += 1.0
                delta[n - 1 + (i - best_j)] += 1.0
                h_b[i] = 1.0
                h_b[best_j] = 0.0
                corr += delta
                cur_c = best_nc
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c

def double_swap_ls(h_b, n, deadline, max_iters=5000):
    """Double-swap local search: try pairs of swaps."""
    dx = 2.0 / n
    cur_h = h_b.copy()
    cur_c = np.correlate(cur_h, 1 - cur_h, mode='full').max() * dx
    best_h = cur_h.copy()
    best_c = cur_c
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Find best single swap
        best_i1, best_j1 = -1, -1
        best_c1 = cur_c - 1e-15
        corr = np.correlate(cur_h, 1 - cur_h, mode='full').astype(float)
        
        # For efficiency, sample a subset
        sample_size = min(50, len(zeros_list))
        sample_zeros = np.random.choice(zeros_list, sample_size, replace=False)
        sample_ones = np.random.choice(ones_list, sample_size, replace=False)
        
        for i1 in sample_zeros:
            for j1 in sample_ones:
                trial = cur_h.copy()
                trial[i1] = 1.0
                trial[j1] = 0.0
                c1 = np.correlate(trial, 1 - trial, mode='full').max() * dx
                if c1 < best_c1:
                    best_c1 = c1
                    best_i1, best_j1 = i1, j1
        
        if best_i1 < 0:
            break
        
        # Fixed first swap, find best second swap
        cur_h[best_i1] = 1.0
        cur_h[best_j1] = 0.0
        ones.add(best_i1)
        ones.discard(best_j1)
        zeros.discard(best_i1)
        zeros.add(best_j1)
        ones_list = list(ones)
        zeros_list = list(zeros)
        
        best_i2, best_j2 = -1, -1
        best_c2 = cur_c - 1e-15  # We want total improvement over original
        sample_zeros2 = np.random.choice(zeros_list, min(50, len(zeros_list)), replace=False)
        sample_ones2 = np.random.choice(ones_list, min(50, len(ones_list)), replace=False)
        
        for i2 in sample_zeros2:
            if i2 == best_i1:
                continue
            for j2 in sample_ones2:
                if j2 == best_j1:
                    continue
                trial = cur_h.copy()
                trial[i2] = 1.0
                trial[j2] = 0.0
                c2 = np.correlate(trial, 1 - trial, mode='full').max() * dx
                if c2 < best_c2:
                    best_c2 = c2
                    best_i2, best_j2 = i2, j2
        
        if best_i2 >= 0:
            cur_h[best_i2] = 1.0
            cur_h[best_j2] = 0.0
            ones.add(best_i2)
            ones.discard(best_j2)
            zeros.discard(best_i2)
            zeros.add(best_j2)
            ones_list = list(ones)
            zeros_list = list(zeros)
        else:
            # Undo first swap
            cur_h[best_i1] = 0.0
            cur_h[best_j1] = 1.0
            ones.discard(best_i1)
            ones.add(best_j1)
            zeros.add(best_i1)
            zeros.discard(best_j1)
            ones_list = list(ones)
            zeros_list = list(zeros)
            break
        
        cur_c = np.correlate(cur_h, 1 - cur_h, mode='full').max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = cur_h.copy()
    
    return best_h, best_c

def sa_binary(h_b, n, deadline, max_iters=100000, T_init=0.03, T_min=0.001):
    """Simulated annealing on binary representation."""
    dx = 2.0 / n
    cur_h = h_b.copy()
    cur_c = np.correlate(cur_h, 1 - cur_h, mode='full').max() * dx
    best_h = cur_h.copy()
    best_c = cur_c
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max_iters)
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Choose move
        r = rng.random()
        if r < 0.4:
            # Single swap
            i = rng.choice(zeros)
            j = rng.choice(ones)
            trial = cur_h.copy()
            trial[i] = 1.0
            trial[j] = 0.0
            nc = np.correlate(trial, 1 - trial, mode='full').max() * dx
            dc = nc - cur_c
            if dc < 0 or rng.random() < np.exp(-dc / T):
                cur_h = trial
                cur_c = nc
                ones[ones.index(j)] = i
                zeros[zeros.index(i)] = j
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = cur_h.copy()
        elif r < 0.7:
            # Double swap
            if len(zeros) < 2 or len(ones) < 2:
                T *= cooling
                continue
            i1 = rng.choice(zeros)
            i2 = rng.choice([z for z in zeros if z != i1])
            j1 = rng.choice(ones)
            j2 = rng.choice([o for o in ones if o != j1])
            trial = cur_h.copy()
            trial[i1] = 1.0
            trial[i2] = 1.0
            trial[j1] = 0.0
            trial[j2] = 0.0
            nc = np.correlate(trial, 1 - trial, mode='full').max() * dx
            dc = nc - cur_c
            if dc < 0 or rng.random() < np.exp(-dc / T):
                cur_h = trial
                cur_c = nc
                # Update lists
                for idx in [i1, i2]:
                    if idx in zeros:
                        zeros.remove(idx)
                    if idx not in ones:
                        ones.append(idx)
                for idx in [j1, j2]:
                    if idx in ones:
                        ones.remove(idx)
                    if idx not in zeros:
                        zeros.append(idx)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = cur_h.copy()
        else:
            # Block move: flip a small block (maintaining sum)
            blen = rng.randint(3, min(15, n // 10))
            start = rng.randint(0, n - blen)
            block = cur_h[start:start+blen].copy()
            n_ones_in = int(block.sum())
            n_zeros_in = blen - n_ones_in
            # Swap within block to maintain total
            if n_ones_in > 0 and n_zeros_in > 0:
                n_swap = min(n_ones_in, n_zeros_in, blen // 2)
                if n_swap > 0:
                    one_pos = [start + p for p in range(blen) if block[p] == 1]
                    zero_pos = [start + p for p in range(blen) if block[p] == 0]
                    flip_o = rng.choice(one_pos, min(n_swap, len(one_pos)), replace=False)
                    flip_z = rng.choice(zero_pos, min(n_swap, len(zero_pos)), replace=False)
                    trial = cur_h.copy()
                    trial[flip_o] = 0.0
                    trial[flip_z] = 1.0
                    nc = np.correlate(trial, 1 - trial, mode='full').max() * dx
                    dc = nc - cur_c
                    if dc < 0 or rng.random() < np.exp(-dc / T):
                        cur_h = trial
                        cur_c = nc
                        for idx in flip_o:
                            ones.remove(idx)
                            zeros.append(idx)
                        for idx in flip_z:
                            zeros.remove(idx)
                            ones.append(idx)
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = cur_h.copy()
        T = max(T * cooling, T_min)
    
    return best_h, best_c

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
