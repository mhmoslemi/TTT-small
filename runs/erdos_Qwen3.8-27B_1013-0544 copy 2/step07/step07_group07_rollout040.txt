def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full') * dx
    # softmax of alpha*c
    amc = alpha * c
    amc_shifted = amc - amc.max()
    w = np.exp(amc_shifted) / np.exp(amc_shifted).sum()
    # w has length 2n-1, indexed from -(n-1) to n-1
    # gradient_j = sum_k w[k] * [(1-h[j+k])*1_{j+k<n} - h[j-k]*1_{j-k>=0}]
    grad = np.zeros(n)
    for j in range(n):
        # first part: sum over k where j+k < n
        # k ranges from -(n-1) to n-1, but we need j+k in [0,n-1]
        # so k in [-j, n-1-j]
        # w index: k + (n-1)
        k_min = max(-(n-1), -j)
        k_max = min(n-1, n-1-j)
        # (1-h[j+k]) for k in [k_min, k_max]
        # j+k ranges from j+k_min to j+k_max = [max(0, j-n+1+j)... ] 
        # Actually j+k_min >= 0 by construction of k_min
        # j+k_max = j + n-1-j = n-1, good
        idx1 = np.arange(j+k_min, j+k_max+1)  # indices into h
        widx1 = np.arange(k_min + (n-1), k_max + (n-1)+1)  # indices into w
        grad[j] += np.dot(w[widx1], 1-h[idx1])
        
        # second part: sum over k where j-k >= 0, i.e., k <= j
        # k in [k_min2, k_max2] where k_max2 = min(n-1, j), k_min2 = max(-(n-1), 0-(j-k... ))
        # Actually we need j-k >= 0 so k <= j, and j-k < n so k > j-n
        # k in [max(-(n-1), j-n+1), min(n-1, j)]
        k_min2 = max(-(n-1), j-n+1)
        k_max2 = min(n-1, j)
        if k_max2 >= k_min2:
            idx2 = np.arange(j-k_max2, j-k_min2+1)  # j-k for k in [k_min2, k_max2]
            widx2 = np.arange(k_min2 + (n-1), k_max2 + (n-1)+1)

    return grad

def compute_c5(h, n):

def compute_corr(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

    n = len(h)

def hinge_obj_grad(h, t, mu):
    """Smooth hinge: 0.5 * sum(max(0, c_k - t)^2) + mu*(sum(h)-n/2)^2"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    excess = np.maximum(c - t, 0.0)
    obj = 0.5 * np.sum(excess ** 2) + mu * (h.sum() - n / 2.0) ** 2
    # Gradient: for c_k > t, grad contribution is (c_k - t) * dc_k/dh
    grad = np.zeros(n)
    mask = c > t
    if mask.any():
        w = np.zeros_like(c)
        w[mask] = c[mask] - t
        # Same gradient computation as before but with w instead of softmax
        for j in range(n):
            m_lo = max(-(n-1), -j)
            m_hi = min(n-1, n-1-j)
            if m_hi >= m_lo:
                idx_h = np.arange(j + m_lo, j + m_hi + 1)
                idx_w = np.arange(m_lo + (n-1), m_hi + (n-1) + 1)
                grad[j] += np.dot(w[idx_w], 1 - h[idx_h])
            m_lo2 = max(-(n-1), j - n + 1)
            m_hi2 = min(n-1, j)
            if m_hi2 >= m_lo2:
                idx_h2 = np.arange(j - m_hi2, j - m_lo2 + 1)
                idx_w2 = np.arange(m_lo2 + (n-1), m_hi2 + (n-1) + 1)
                grad[j] -= np.dot(w[idx_w2], h[idx_h2])
    grad += 2 * mu * (h.sum() - n / 2.0)
    return obj, grad

def hinge_lbfgsb(h, n, t, mu, maxiter):
    def fun_g(x):
        return hinge_obj_grad(x, t, mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_g, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxfun': 80000})
        return project_feasible(res.x, n)
    except Exception:
        return h

    zeros_list = list(np.where(h == 0)[0])

        improved = False
        for i in zeros_list:

                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)

                nc = (corr + delta).max() * dx
                if nc < best_nc:

                    best_j = j
            if best_j >= 0:
                j = best_j
                delta = np.zeros(2 * n - 1)

def compute_grad_fft(h, w, n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def build_W_arr(w, n, N):
    """Build circular arrangement of weight array w (length 2n-1) into array of length N."""
    arr = np.zeros(N)
    arr[0:n] = w[n-1:]
    if N - (n-1) > n:
        arr[n:N-(n-1)] = 0
    arr[N-(n-1):] = w[:n-1]
    return arr

def fft_gradient(h, w, n):
    """Compute gradient using FFT. w has length 2n-1."""
    N = 1
    while N < 2 * n - 1 + 1:
        N *= 2
    W_arr = build_W_arr(w, n, N)
    Wp_arr = build_W_arr(w[::-1], n, N)
    
    f = 1.0 - h
    F = np.fft.rfft(f, N)
    Wf = np.fft.rfft(W_arr)
    grad1 = np.fft.irfft(F * np.conj(Wf), N)[:n]
    
    H = np.fft.rfft(h, N)
    Wpf = np.fft.rfft(Wp_arr)
    grad2 = np.fft.irfft(H * np.conj(Wpf), N)[:n]
    
    return grad1 - grad2

def lse_obj_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    amc = alpha * c
    amc_max = amc.max()
    amc_shifted = amc - amc_max
    exp_vals = np.exp(amc_shifted)
    w = exp_vals / exp_vals.sum()
    logsum = amc_max + np.log(exp_vals.sum())
    obj = logsum / alpha + mu * (h.sum() - n / 2.0) ** 2
    grad = fft_gradient(h, w, n)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return obj, grad

def hinge_obj_grad(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    excess = np.maximum(c - t, 0.0)
    obj = 0.5 * np.sum(excess ** 2) + mu * (h.sum() - n / 2.0) ** 2
    w = np.zeros_like(c)
    mask = c > t
    w[mask] = c[mask] - t
    grad = fft_gradient(h, w, n)
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return obj, grad

def lse_lbfgsb(h, n, alpha, mu, maxiter):
    def fun_g(x):
        return lse_obj_grad(x, alpha, mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_g, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def hinge_lbfgsb(h, n, t, mu, maxiter):

    bounds = [(0.0, 1.0)] * n
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})

                jmn = j - m_range
