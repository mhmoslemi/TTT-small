import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    sh = alpha * (corr - m)
    sm2 = np.max(sh)
    val = m + (sm2 + np.log(np.exp(sh - sm2).sum())) / alpha
    val += penalty * (x.sum() - n / 2.0) ** 2
    return val

def _grad(x, n, alpha, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    sh = alpha * (corr - m)
    sm2 = np.max(sh)
    p = np.exp(sh - sm2)
    p /= p.sum()
    g = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-12:
            continue
        if 0 <= l < n:
            g[:n-l] += w * (1.0 - x[l:])
            g[l:] -= w * x[:n-l]
        elif -n < l < 0:
            ll = -l
            g[ll:] += w * (1.0 - x[:n-ll])
            g[:n-ll] -= w * x[ll:]
    g *= dx
    g += 2.0 * penalty * (x.sum() - n / 2.0)
    return g

def lbfgsb_opt(h, n, alpha, penalty, maxiter, pgtol):
    bounds = [(0.0, 1.0)] * n
    res = fmin_l_bfgs_b(_obj, h, fprime=_grad,
                        args=(n, alpha, penalty),
                        bounds=bounds, maxiter=maxiter,
                        pgtol=pgtol, factr=1e12, disp=0)
    return project_feasible(res, n)

def softgrad_step(h, n, alpha, lr, momentum, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sm2 = sh.max()
    p = np.exp(sh - sm2)
    p /= p.sum()
    g = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-12:
            continue
        if 0 <= l < n:
            g[:n-l] += w * (1.0 - h[l:])
            g[l:] -= w * h[:n-l]
        elif -n < l < 0:
            ll = -l
            g[ll:] += w * (1.0 - h[:n-ll])
            g[:n-ll] -= w * h[ll:]
    g *= dx
    momentum = mu * momentum - lr * g
    h = project_feasible(h + momentum, n)
    return h, momentum

def softgrad_loop(h, n, alpha, deadline, mu, lr_init, stall_limit):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = lr_init
    momentum = np.zeros(n)
    stall = 0
    max_stall = 0
    for step in range(500000):
        if time.time() > deadline:
            break
        h, momentum = softgrad_step(h, n, alpha, lr, momentum, mu)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.5 * dx)
        else:
            stall += 1
            if stall > stall_limit:
                lr *= 0.9
                if lr < dx * 0.01:
                    lr = dx * 0.1
                    stall = 0
                if lr < dx * 0.001:
                    break
    return best_h, best_c

def multi_res_opt(h, n, deadline, best_c, best_h, rng):
    for nl in [n // 2, n // 4, n // 8]:
        if nl < 16 or time.time() > deadline - 30:
            break
        xo = np.linspace(0, 2, n)
        xl = np.linspace(0, 2, nl)
        hl = np.interp(xl, xo, h)
        hl = project_feasible(hl, nl)
        for a in [1000, 10000, 100000]:
            if time.time() > deadline - 10:
                break
            hl = lbfgsb_opt(hl, nl, a, 500.0, 200, 1e-12)
        hl_hi = np.interp(xo, xl, hl)
        hl_hi = project_feasible(hl_hi, n)
        hl_hi = lbfgsb_opt(hl_hi, n, 100000, 500.0, 300, 1e-13)
        ch = compute_c5(hl_hi, n)
        if ch < best_c:
            best_c = ch
            best_h = hl_hi.copy()
    return best_h, best_c
