    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

            for j in ones:

                ik = i + m_range

                i_ik = i - m_range
                valid_i_ik = (i_ik >= 0) & (i_ik < n)
                Bi[valid_i_ik] -= h_b[i_ik[valid_i_ik]]
                
                Bj = np.ones(2*n-1)
                Bj[valid_jk] -= h_b[jk[valid_jk]]
                j_ik = j - m_range
                valid_j_ik = (j_ik >= 0) & (j_ik < n)
                Bj[valid_j_ik] -= h_b[j_ik[valid_j_ik]]

                valid_ik = (ik >= 0) & (ik < n)

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_B_all(h, i, n):
    k_vals = np.arange(-(n - 1), n)
    i_plus = i + k_vals
    i_minus = i - k_vals
    vp = (i_plus >= 0) & (i_plus < n)
    vm = (i_minus >= 0) & (i_minus < n)
    B = np.ones(2 * n - 1)
    B[vp] -= h[i_plus[vp]]
    B[vm] -= h[i_minus[vm]]
    return B

def binary_1swap(h_init, n, deadline, max_sweeps=1000):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h_b == 0)[0]
        ones = np.where(h_b == 1)[0]
        for i in zeros:
            Bi = compute_B_all(h_b, i, n)
            best_j = -1
            best_nc = cur_c - 1e-15
            for j in ones:
                Bj = compute_B_all(h_b, j, n)
                nc = (corr + Bi - Bj).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                Bj = compute_B_all(h_b, best_j, n)
                h_b[i] = 1.0
                h_b[best_j] = 0.0
                corr += Bi - Bj
                cur_c = best_nc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c

def binary_2swap(h_init, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    h_b = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h_b[idx[:n // 2]] = 1.0
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones_arr = np.where(h_b == 1)[0]
    zeros_arr = np.where(h_b == 0)[0]
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try 2-swaps: pick two zeros and two ones
        for attempt in range(200):
            if time.time() > deadline:
                break
            za, zb = rng.choice(len(zeros_arr), 2, replace=False)
            a, b = zeros_arr[za], zeros_arr[zb]
            pa, pb = rng.choice(len(ones_arr), 2, replace=False)
            p, q = ones_arr[pa], ones_arr[pb]
            Ba = compute_B_all(h_b, a, n)
            Bb = compute_B_all(h_b, b, n)
            Bp = compute_B_all(h_b, p, n)
            Bq = compute_B_all(h_b, q, n)
            delta = Ba + Bb - Bp - Bq
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-14:
                h_b[a] = 1.0
                h_b[b] = 1.0
                h_b[p] = 0.0
                h_b[q] = 0.0
                corr += delta
                cur_c = nc
                improved = True
                ones_arr = np.where(h_b == 1)[0]
                zeros_arr = np.where(h_b == 0)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        if not improved and sweep > 10:
            # Also try 1-swaps between attempts
            for i in zeros_arr[:50]:
                Bi = compute_B_all(h_b, i, n)
                best_j = -1
                best_nc = cur_c - 1e-15
                for j in ones_arr:
                    Bj = compute_B_all(h_b, j, n)
                    nc = (corr + Bi - Bj).max() * dx
                    if nc < best_nc:
                        best_nc = nc
                        best_j = j
                if best_j >= 0:
                    Bj = compute_B_all(h_b, best_j, n)
                    h_b[i] = 1.0
                    h_b[best_j] = 0.0
                    corr += Bi - Bj
                    cur_c = best_nc
                    improved = True
                    ones_arr = np.where(h_b == 1)[0]
                    zeros_arr = np.where(h_b == 0)[0]
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h_b.copy()
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 5
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Penalty coordinate descent (fast)
    p1 = t0 + 0.15 * total
    for mu in [5.0, 10.0, 20.0, 50.0, 100.0]:
        if time.time() > p1 - 2:
            break
        h_cd, c_cd = coord_descent(best_h, n, mu, p1, max_sweeps=200)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd.copy()

    # Phase 2: SLSQP with increasing alpha
    p2 = t0 + 0.35 * total
    alphas = [500, 2000, 20000, 200000, 2e6, 2e7]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Binary local search (1-swap)
    p3 = t0 + 0.60 * total
    hb, cb = binary_1swap(best_h, n, p3, max_sweeps=500)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: Binary 2-swap search
    p4 = t0 + 0.75 * total
    hb2, cb2 = binary_2swap(best_h, n, p4, max_sweeps=100)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()

    # Phase 5: Multi-restart SLSQP from perturbed best
    p5 = t0 + 0.88 * total
    scales = [0.005, 0.01, 0.02, 0.05, 0.1, 0.015, 0.03, 0.08, 0.15, 0.04]
    for si in range(len(scales)):
        if time.time() > p5 - 3:
            break
        h = best_h + scales[si] * rng.randn(n)
        h = project_feasible(h, n)
        for alpha in [1e5, 1e6, 1e7]:
            if time.time() > p5 - 2:
                break
            h = slsqp_optimize(h, n, alpha, 1000, 1e-15)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
                break
        # Quick binary refinement
        hb, cb = binary_1swap(h, n, min(time.time() + 30, p5 - 2), max_sweeps=100)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 6: Final polish
    if time.time() < dl - 10:
        h = best_h.copy()
        for alpha in [5e6, 5e7, 5e8]:
            if time.time() > dl - 5:
                break
            h = slsqp_optimize(h, n, alpha, 1500, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
        hb, cb = binary_1swap(best_h, n, dl - 1, max_sweeps=200)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
