        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + (diff / s) * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / s), 0, 1)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_greedy(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                imn = i - m_range
                msk2 = (imn >= 0) & (imn < n)
                jm = j + m_range
                msk3 = (jm >= 0) & (jm < n)
                jmn = j - m_range
                msk4 = (jmn >= 0) & (jmn < n)
                delta = (np.where(msk, 1.0, 0.0) + np.where(msk2, -1.0, 0.0)
                         + np.where(msk3, -1.0, 0.0) + np.where(msk4, 1.0, 0.0))
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def binary_sa(h_init, n, deadline, rng, n_iter=400000):
    m_range = np.arange(-(n - 1), n)
    msk_i_plus = np.array([(i >= 0) & (i < n) for i in m_range])
    msk_i_minus = np.array([(i >= 0) & (i < n) for i in -m_range])
    msk_j_plus = msk_i_plus
    msk_j_minus = msk_i_minus
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    base = cur_c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = base * 0.4 * (1.0 - it / float(n_iter)) ** 1.5 + base * 0.0001
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        delta = (np.where(msk_i_plus & (i + m_range >= 0) & (i + m_range < n), 1.0, 0.0)
                 + np.where(msk_i_minus & (i - m_range >= 0) & (i - m_range < n), -1.0, 0.0)
                 + np.where(msk_j_plus & (j + m_range >= 0) & (j + m_range < n), -1.0, 0.0)
                 + np.where(msk_j_minus & (j - m_range >= 0) & (j - m_range < n), 1.0, 0.0))
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < len(delta):
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < len(delta):
            delta[k2] += 1.0
        nc = (corr + delta).max() * dx
        dc = nc - cur_c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            ones_list.remove(j)
            zeros_list.remove(i)
            ones_list.append(i)
            zeros_list.append(j)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        shifted = np.clip(shifted, -500, 500)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, threshold, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(g - threshold, 0)
        return 0.5 * np.sum(excess ** 2) + 0.01 * g.max()
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def refine_continuous(h, n, deadline):
    c = compute_c5(h, n)
    alphas = [50000, 200000, 1000000, 5000000, 20000000, 100000000, 500000000]
    for al in alphas:
        if time.time() > deadline - 5:
            break
        mi = 3000 if al < 1e6 else 5000
        h = lse_slsqp(h, n, al, mi, 1e-15)
        c = compute_c5(h, n)
    return h, c

def barrier_push(h, n, c_cur, deadline):
    for i in range(15):
        if time.time() > deadline - 3:

        t = c_cur * (1.0 - 0.002 * (i + 1))

        c = compute_c5(h, n)
        if c < c_cur:
            c_cur = c
    return h, c_cur
