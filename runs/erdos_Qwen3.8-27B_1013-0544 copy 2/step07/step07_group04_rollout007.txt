        v = np.clip(v, 0.0, 1.0)
    return v

def grad_term(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        m = n - k
        g[k:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[k:]
    else:
        k = -k
        m = n - k
        g[:m] += dx * (1.0 - h[k:])
        g[k:] -= dx * h[:m]
    return g

def topk_gradient(h, n, topk=5, temp=50.0):
    c = compute_c(h, n)
    off = n - 1
    k_idx = np.argpartition(c, -topk)[-topk:]
    vals = c[k_idx]
    maxv = vals.max()
    wts = np.exp(-temp * (maxv - vals))
    wts /= wts.sum()
    g = np.zeros(n)
    for idx, w in zip(k_idx, wts):
        g += w * grad_term(h, n, int(idx) - off)
    return g, float(maxv)

def gradient_step(h, n, topk=5, temp=50.0, max_ls=20):
    g, f0 = topk_gradient(h, n, topk, temp)
    c = compute_c(h, n)
    f0 = float(c.max())
    step = 1.0
    for _ in range(max_ls):
        h2 = project_box_sum(h - step * g, n)
        f2 = float(compute_c(h2, n).max())
        if f2 < f0 - 1e-15:
            return h2, f2
        step *= 0.5
        if step < 1e-12:
            break
    return h, f0

def pair_search(h, c_base, i, j, n):
    if i == j:
        return None
    x0, y0 = float(h[i]), float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    d = j - i
    L = 2 * n - 1
    
    def eval_pair(x):
        y = S - x
        dx1 = x - x0
        dx2 = y - y0
        delta = np.zeros(L)
        delta[i:i+n] += dx1 * omhrev
        si = n - 1 - i
        delta[si:si+n] += dx1 * h
        delta[j:j+n] += dx2 * omhrev
        sj = n - 1 - j
        delta[sj:sj+n] += dx2 * h
        # self-interaction correction for i
        idx0 = off
        delta[idx0] += x0*(1-x0) + y0*(1-y0) - x*(1-x) - y*(1-y)
        # cross terms
        if d != 0:
            delta[off - d] += x0*(1-y0) - x*(1-y)
            delta[off + d] += y0*(1-x0) - y*(1-x)
        return c_base + dx * delta
    
    xs = np.linspace(lo, hi, 128)
    cands = np.array([eval_pair(x) for x in xs])
    best = np.max(cands, axis=1)
    k = int(np.argmin(best))
    # refine
    if 0 < k < len(xs)-1:
        a, b = xs[k-1], xs[k+1]
    elif k == 0:
        a, b = xs[0], xs[1]
    else:
        a, b = xs[-2], xs[-1]
    xs2 = np.linspace(a, b, 64)
    cands2 = np.array([eval_pair(x) for x in xs2])
    best2 = np.max(cands2, axis=1)
    k2 = int(np.argmin(best2))
    bv = float(best2[k2])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(xs2[k2])
        yb = float(S - xb)
        if yb < 0: yb, xb = 0.0, S
        elif yb > 1: yb, xb = 1.0, S - 1.0
        return xb, yb, bv
    return None

def pair_local_search(h, deadline, seed=1, max_noimp=500):
    n = len(h)
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    noimp = 0
    
    while time.time() < deadline and noimp < max_noimp:
        pos = c[n-1+1:]
        neg = c[n-1-1::-1] if n > 2 else np.array([0.0])
        w = np.maximum(pos, neg) + 1e-10
        w /= w.sum()
        d = int(rng.choice(n-1, p=w)) + 1
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search(h, c, i, j, n)
        if res is not None:
            x, y, _ = res
            h[i], h[j] = x, y
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc, besth = v, h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())

def optimize_n(h, n, seed, deadline):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    
    # Phase 1: gradient steps
    topks = [3, 5, 8, 10]
    temps = [200.0, 50.0, 10.0, 3.0]
    for tk, tp in zip(topks, temps):
        for _ in range(60):
            if time.time() > deadline - 10:
                break
            h2, f2 = gradient_step(h, n, topk=tk, temp=tp)
            if f2 < bestc:
                bestc, besth = f2, h2
            if time.time() > deadline - 10:
                break
        h = besth
    
    # Phase 2: pair search
    pl = deadline - 40
    if time.time() < pl:
        besth, bestc = pair_local_search(besth, pl, seed=seed)
    
    # Phase 3: more gradient with higher topk
    if time.time() < deadline - 15:
        h = besth.copy()
        for _ in range(40):
            if time.time() > deadline - 15:
                break
            h2, f2 = gradient_step(h, n, topk=15, temp=5.0)
            if f2 < bestc:
                bestc, besth = f2, h2
            h = besth
    
    # Phase 4: final pair polish
    if time.time() < deadline - 5:
        besth, bestc = pair_local_search(besth, deadline, seed=seed+77, max_noimp=300)
    
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def interp_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        n0 = len(hp)
    except Exception:
        n0 = 400
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        hp = 0.5 + 0.45 * np.sin(3.0 * np.pi * x)
        hp = project_box_sum(hp + 0.01 * rng.randn(n0), n0)
    
    hp = project_box_sum(hp, n0)
    besth = hp.copy()
    bestc = float(compute_c(hp, n0).max())
    bestn = n0
    
    # Try multiple n values
    n_list = [n0]
    for m in (256, 300, 400, 500, 600):
        if m != n0 and 8 <= m <= 700:
            n_list.append(m)
    n_list = sorted(set(n_list))[:5]
    
    L = len(n_list)
    # Allocate time: first gets 50%, rest split
    if L == 1:
        fr = [0.85]
    else:
        fr = [0.45]
        rem = 0.40 / (L - 1)
        for _ in range(L - 1):
            fr.append(rem)
    
    span = max(1.0, deadline - t0)
    cum = 0.0
    for k, nh in enumerate(n_list):
        if time.time() > deadline - 20:
            break
        cum += fr[k]
        dl_phase = t0 + span * min(0.92, cum) - 3
        if dl_phase > deadline - 20:
            dl_phase = deadline - 20
        if dl_phase < time.time() + 5:
            continue
        
        if k == 0:
            hinit = besth.copy()
        else:
            hinit = interp_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        
        hopt, copt = optimize_n(hinit, nh, seed + 13 * k, dl_phase)
        if copt < bestc:
            bestc, besth, bestn = copt, hopt, nh
    
    # Final polish on best solution
    if time.time() < deadline - 5:
        hfin, cfin = pair_local_search(besth, deadline, seed=9999, max_noimp=2000)
        if cfin < bestc:
            bestc, besth = cfin, hfin
    
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
