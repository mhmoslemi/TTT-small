def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return val
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

        lag = k_idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad[:] += w * (1.0 - h)
                grad[:] -= w * h
            elif lag < n:
                grad[lag:] += w * (1.0 - h[:n-lag])
                grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            if l < n:
                grad[:n-l] += w * (1.0 - h[l:])
                grad[l:] -= w * h[:n-l]
    grad *= dx
    return grad, corr

def fista_opt(h, n, alpha, deadline, lr_init=0.005):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        g, _ = smooth_grad(y, n, alpha)
        x_new = project_feasible(y - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - x)
        x = x_new
        t_k = t_new
        c = compute_c5(x, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = x.copy()
            stall = 0
            lr = min(lr * 1.003, 0.5)
        else:
            stall += 1
            if stall > 150:
                lr *= 0.92
                if lr < 1e-6:
                    lr = 0.003
                    stall = 0
                x = best_h.copy()
                y = best_h.copy()
                t_k = 1.0
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cfun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_single_lag(h, n, lag, maxiter=1500):
    dx = 2.0 / n
    def obj(x):
        if lag >= 0:
            if lag == 0:
                return np.sum(x * (1 - x)) * dx
            return np.sum(x[:n-lag] * (1 - x[lag:])) * dx
        else:
            l = -lag
            return np.sum(x[l:] * (1 - x[:n-l])) * dx
    def cfun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-grid coarse optimization
    t1 = t0 + 170
    xo = np.linspace(0, 2, n)
    for nc in [40, 60, 80, 120]:
        if time.time() > t1 - 15:
            break
        xc = np.linspace(0, 2, nc)
        inits = [np.interp(xc, xo, best_h)]
        hb = np.zeros(nc)
        hb[rng.choice(nc, nc // 2, replace=False)] = 1.0
        inits.append(hb)
        period = max(2, nc // 12)
        hp = np.array([1.0 if i % period < period // 2 else 0.0 for i in range(nc)])
        inits.append(hp)
        for hc0 in inits:
            if time.time() > t1 - 8:
                break
            hc = project_feasible(hc0.copy(), nc)
            hc = slsqp_smooth(hc, nc, 300, 600, 1e-10)
            hc = slsqp_smooth(hc, nc, 3000, 600, 1e-11)
            h_up = np.interp(xo, xc, hc)
            h_up = project_feasible(h_up, n)
            c = compute_c5(h_up, n)
            if c < best_c:
                best_c = c
                best_h = h_up.copy()

    # Phase 2: FISTA fine-grid optimization
    t2 = t0 + 480
    for alpha in [300, 1500, 8000]:
        if time.time() > t2 - 20:
            break
        h2, c2 = fista_opt(best_h.copy(), n, alpha, min(time.time() + 90, t2), lr_init=0.004)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: SLSQP with increasing alpha
    t3 = t0 + 780
    for alpha in [5000, 30000, 200000]:
        if time.time() > t3 - 35:
            break
        h3 = slsqp_smooth(best_h.copy(), n, alpha, 2500, 1e-13)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Active-lag sequential optimization
    t4 = deadline - 80
    for round in range(6):
        if time.time() > t4:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)
        cmax = corr.max()
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        h4 = slsqp_single_lag(best_h.copy(), n, lag, maxiter=1200)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()
        else:
            # Try second worst lag
            idx = np.argsort(corr)[::-1]
            for k2 in idx[1:3]:
                if time.time() > t4 - 5:
                    break
                lag2 = k2 - (n - 1)
                h4b = slsqp_single_lag(best_h.copy(), n, lag2, maxiter=800)
                c4b = compute_c5(h4b, n)
                if c4b < best_c:
                    best_c = c4b
                    best_h = h4b.copy()
                    break

    # Phase 5: Noise restarts + SLSQP polish
    t5 = deadline - 20
    for trial in range(18):
        if time.time() > t5 - 3:
            break
        nl = 0.0015 + trial * 0.0008
        h5 = best_h + nl * rng.randn(n)
        h5 = project_feasible(h5, n)
        h5 = slsqp_smooth(h5, n, 20000, 800, 1e-13)
        h5 = slsqp_smooth(h5, n, 200000, 600, 1e-14)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Final high-alpha polish
    if time.time() < deadline - 10:
        h6 = slsqp_smooth(best_h.copy(), n, 2000000, 2000, 1e-15)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
