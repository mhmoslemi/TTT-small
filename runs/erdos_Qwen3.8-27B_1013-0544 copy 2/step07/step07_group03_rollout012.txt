import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    if abs(float(h.sum()) - target) < 1e-12:
        return h
    lo = -float(h.min())
    hi = 1.0 - float(h.max())
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h + mid, 0.0, 1.0).sum()) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return np.clip(h, 0.0, 1.0)


def project_fast(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = float(h.sum())
            if s < 1e-14:
                break
            h = h + diff / s * h
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) > 1e-8:
        return project_feasible(h, n)
    return h


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def subgrad_o(k, h, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    return g


def slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    sm = float(np.max(sh))
    return m + (sm + float(np.log(np.exp(sh - sm).sum()))) / alpha


def slsqp_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(np.max(c))
    sh = alpha * (c - m)
    sm = float(np.max(sh))
    ew = np.exp(sh - sm)
    w = ew / ew.sum()
    nz = np.argsort(w)[-64:]
    g = np.zeros(n)
    for idx in nz:
        ww = float(w[idx])
        if ww <= 1e-12:
            continue
        k = (n - 1) - int(idx)
        g += ww * subgrad_o(k, x, n)
    return g


def sum_eq(x, n):
    return float(x.sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones((1, n))


def slsqp_once(h, n, alpha, maxiter, ftol):
    h = project_feasible(h, n)
    try:
        res = minimize(
            slsqp_obj,
            h,
            args=(n, alpha),
            method="SLSQP",
            bounds=[(0.0, 1.0)] * n,
            jac=slsqp_jac,
            constraints=[{"type": "eq", "fun": sum_eq, "jac": sum_jac, "args": (n,)}],
            options={"maxiter": maxiter, "ftol": ftol},
        )
        return project_feasible(res.x, n)
    except Exception:
        return h


def continuous_subgrad(h, n, deadline, top_k=3, lr_base=0.08, stall_limit=5000):
    h = project_feasible(h, n)
    best = h.copy()
    best_c = compute_c5(h, n)
    G = np.zeros(n)
    mom = np.zeros(n)
    beta = 0.9
    stall = 0
    t = 0
    dx = 2.0 / n
    while time.time() < deadline:
        t += 1
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        cur = float(np.max(c))
        if cur < best_c:
            best_c = cur
            best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            h = best + 0.02 * np.random.RandomState(t).standard_normal(n)
            h = project_fast(h, n)
            G = np.zeros(n)
            mom = np.zeros(n)
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for idx in top_idx:
            k = (n - 1) - int(idx)
            g += subgrad_o(k, h, n)
        g *= dx / float(top_k)
        mom = beta * mom + (1.0 - beta) * g
        G = beta * G + (1.0 - beta) * (mom * mom)
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_fast(h - lr * mom, n)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx)
        if c_new <= cur + 1e-7:
            h = h_new
        elif time.time() < deadline:
            h = project_fast(h - 0.25 * lr * mom, n)
    return best, best_c


def continuous_polish(h, n, deadline):
    h = project_feasible(h, n)
    best = h.copy()
    best_c = compute_c5(h, n)
    alphas = [10000.0, 100000.0, 1000000.0, 10000000.0]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        h2 = slsqp_once(h, n, alpha, 800, 1e-15)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best = h2.copy()
            h = h2
    if time.time() < deadline - 10:
        h2, c2 = continuous_subgrad(h, n, deadline, top_k=5, lr_base=0.05)
        if c2 < best_c:
            best_c = c2
            best = h2.copy()
    return best, best_c


def binary_from_values(h, n):
    h = np.asarray(h, dtype=float)
    x = np.zeros(n)
    target = n // 2
    idx = np.argsort(h)[::-1]
    x[idx[:target]] = 1.0
    return x


def o_from_binary(x):
    return np.correlate(x, 1.0 - x, mode="full")[::-1].astype(float)


def build_AE_active(x, active):
    n = x.size
    lag = active - (n - 1)
    s = np.arange(n)[:, None]
    L = s + lag[None, :]
    R = s - lag[None, :]
    vp = (L >= 0) & (L < n)
    vr = (R >= 0) & (R < n)
    Lc = np.clip(L, 0, n - 1)
    Rc = np.clip(R, 0, n - 1)
    A = vp * (1.0 - x[Lc]) - vr * x[Rc]
    E = vr * x[Rc] - vp * (1.0 - x[Lc])
    return A, E


def apply_swap_o(o, x, i, j):
    n = x.size
    M = 2 * n - 1
    mr = np.arange(-(n - 1), n)
    Li = np.clip(i + mr, 0, n - 1)
    Ri = np.clip(i - mr, 0, n - 1)
    vpi = (i + mr >= 0) & (i + mr < n)
    vqi = (i - mr >= 0) & (i - mr < n)
    d = vpi * (1.0 - x[Li]) - vqi * x[Ri]
    Lj = np.clip(j + mr, 0, n - 1)
    Rj = np.clip(j - mr, 0, n - 1)
    vpj = (j + mr >= 0) & (j + mr < n)
    vqj = (j - mr >= 0) & (j - mr < n)
    d += vqj * x[Rj] - vpj * (1.0 - x[Lj])
    d[n - 1] -= 2.0
    dd = j - i
    idx = n - 1 + dd
    if 0 <= idx < M:
        d[idx] += 1.0
    idx = n - 1 - dd
    if 0 <= idx < M:
        d[idx] += 1.0
    return np.rint(o + d)


def binary_find_step(x, o, cur, full=False, max_checked=8, deadline=None):
    n = x.size
    if cur <= 0:
        return False, None, None, None
    active = np.where(o >= cur - 2.0)[0]
    active = active[active != n - 1]
    if active.size == 0:
        return False, None, None, None
    A_all, E_all = build_AE_active(x, active)
    zeros = np.where(x < 0.5)[0]
    ones = np.where(x >= 0.5)[0]
    if zeros.size == 0 or ones.size == 0:
        return False, None, None, None
    base = o[active]
    for i in zeros:
        if deadline is not None and time.time() > deadline:
            return False, None, None, None
        P = base + A_all[i]
        scores = np.max(P[None, :] + E_all[ones, :], axis=1)
        pos = np.where(scores < cur)[0]
        if pos.size == 0:
            continue
        if full:
            cand = pos[np.argsort(scores[pos])]
        else:
            m = max_checked
            if m > 0 and pos.size > m:
                cand_all = np.argpartition(scores, m)[:m]
                cand = cand_all[scores[cand_all] < cur]
            else:
                cand = pos
            cand = cand[np.argsort(scores[cand])]
        for oi in cand:
            if deadline is not None and time.time() > deadline:
                return False, None, None, None
            if scores[oi] >= cur:
                continue
            ii = int(i)
            jj = int(ones[oi])
            new_o = apply_swap_o(o, x, ii, jj)
            new_cur = float(new_o.max())
            if new_cur < cur:
                x[ii] = 1.0
                x[jj] = 0.0
                return True, x, new_o, new_cur
    return False, None, None, None


def perturb_binary(x, n, rng, k):
    x = x.copy()
    k = int(min(k, n // 2))
    if k <= 0:
        return x
    zeros = np.where(x < 0.5)[0]
    ones = np.where(x >= 0.5)[0]
    for _ in range(k):
        if zeros.size == 0 or ones.size == 0:
            break
        ia = int(rng.randint(zeros.size))
        ja = int(rng.randint(ones.size))
        i = int(zeros[ia])
        j = int(ones[ja])
        x[i] = 1.0
        x[j] = 0.0
        zeros = np.where(x < 0.5)[0]
        ones = np.where(x >= 0.5)[0]
    return x


def low_pass_binary(n, seed, smooth):
    rng = np.random.RandomState(seed)
    g = rng.standard_normal(n)
    l = max(1, min(int(smooth), n // 2))
    t = np.arange(-l, l + 1)
    kernel = np.exp(-np.abs(t) / float(l))
    kernel /= kernel.sum()
    vals = np.convolve(g, kernel, "same")
    x = np.zeros(n)
    idx = np.argsort(vals)[::-1]
    x[idx[: n // 2]] = 1.0
    return x


def binary_search(x, n, deadline, rng):

    o = o_from_binary(x)
    cur = float(o.max())
    best_x = x.copy()

    kick_sizes = [1, 2, 3, 5, 8, 12, 20, 30, 50, 80, 120, 200, 300, 500, 800]

        ok, x, o, cur = binary_find_step(x, o, cur, full=False, max_checked=8, deadline=deadline)
        if ok:

                progress = 0
            else:

        kk = kick_sizes[min(progress, len(kick_sizes) - 1)]

        ok = o_from_binary(xk)

def clean_binary(h, n):
    h = np.asarray(h, dtype=float)
    hb = np.rint(h)
    cnt = int(np.sum(hb))
    target = n // 2
    if cnt == target:
        return hb
    if cnt > target:
        ones = np.where(hb > 0.5)[0]
        if ones.size:
            sel = ones[int(np.argmin(h[ones]))]
            hb[sel] = 0.0
    else:
        zeros = np.where(hb < 0.5)[0]
        if zeros.size:
            sel = zeros[int(np.argmax(h[zeros]))]
            hb[sel] = 1.0
    return hb
