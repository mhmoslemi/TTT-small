    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    h0 = h[p]
    m_f_lo = max(0, n - 1 - p)
    m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
    m_b_lo = max(0, p)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

sum_k #{i: h[i] ≠ h[i+k]} = sum_i h[i]*(1-h[i]) + ...

  Total for p: h[p+m-n+1] - (1-h[p-m+n-1]) [with appropriate bounds]

- From h[q]: was contributing h[q]=0 to first factor and 1-h[q]=1 to second factor. After: h[q]=1, so first factor contributes 1, second factor contributes 0.
  - First factor: was 0, now 1*(1-h[q-m+n-1]) = 1-h[q-m+n-1]. Change: +(1-h[q-m+n-1])
  - Second factor: was h[q+m-n+1]*1, now 0. Change: -h[q+m-n+1]
  Total for q: (1-h[q-m+n-1]) - h[q+m-n+1]

Combined change: [h[p+m-n+1] - (1-h[p-m+n-1])] + [(1-h[q-m+n-1]) - h[q+m-n+1]]

This is O(n) to compute for all affected m values. With numpy vectorization, a single swap evaluation is fast.

For SA, we'd do:
1. Randomly pick a 1-position p and a 0-position q
2. Compute ΔE (change in C5 = max change)
3. Accept with probability min(1, exp(-ΔE/T))

The issue is that C5 = max_k c[k], so ΔE is the change in the max, which is not simply the max of individual changes (a previously sub-maximal shift might become the new max).

For correctness, we need to recompute the full max after each proposed swap, which is O(n). But that's fine for SA.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def one_coord_optimize(c, h, n, p, grid_size):
    """Find optimal delta for coordinate p given current correlation c."""
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return 0.0, float(c.max())
    m_f_lo = max(0, n - 1 - p)
    m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
    m_b_lo = max(0, p)
    m_b_hi = min(2 * n - 2, p + n - 1)
    m_lo = min(m_f_lo, m_b_lo)
    m_hi = max(m_f_hi, m_b_hi)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag = (ms == n - 1)
    c_aff = c[ms]
    umax = 0.0
    if m_lo > 0:
        umax = max(umax, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        umax = max(umax, float(c[m_hi + 1:].max()))
    dg = np.linspace(-h0, 1.0 - h0, grid_size)
    nv = c_aff[None, :] + dg[:, None] * a[None, :]
    nv -= (dg[:, None] ** 2) * diag[None, :]
    tm = np.maximum(nv.max(axis=1), umax)
    bi = int(np.argmin(tm))
    return float(dg[bi]), float(tm[bi])
