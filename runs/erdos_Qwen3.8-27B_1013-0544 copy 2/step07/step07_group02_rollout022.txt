        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['CLARABEL', 'ECOS', 'GLOP']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.5, 0.3, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_smooth(h, n, alpha, maxiter, ftol, dl):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol,
                       'disp': 0})
        if time.time() > dl:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps):

def smooth_obj(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    logsum = sm + np.log(exp_s.sum())
    return c_max + logsum / alpha + mu * (x.sum() - n/2.0) ** 2

def smooth_grad(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for m_idx in active:
        k = m_idx - (n - 1)
        wm = w[m_idx]
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1e > i1s:
            g[i1s:i1e] += wm * (1.0 - x[i1s+k:i1e+k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2e > i2s:
            g[i2s:i2e] -= wm * x[i2s-k:i2e-k]
    g += 2.0 * mu * (x.sum() - n/2.0)
    return g

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'disp': 0})
        if time.time() > dl:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h
