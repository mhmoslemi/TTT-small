import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    return m + np.log(s) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    q = e / s
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            m2 = n - k
            g[k:n] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[k:n]
        elif k < 0:
            s2 = -k
            m2 = n - s2
            g[:m2] += w * (1.0 - x[s2:n])
            g[s2:n] -= w * x[:m2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def lbfgsb_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 4 * maxiter,
                                "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 24)))
    a = max(lo, x0 - 0.04)
    b = min(hi, x0 + 0.04)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 96)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    base_p = ax1 * (1.0 - y0) + x0 * ax1
    exact_p = xs * (1.0 - yv) - x0 * (1.0 - y0)
    delta[:, idxp] += exact_p - base_p
    idxm = off + d
    base_m = ax2 * (1.0 - x0) + y0 * ax2
    exact_m = yv * (1.0 - xs) - y0 * (1.0 - x0)
    delta[:, idxm] += exact_m - base_m
    return np.max(c_base[None, :] + dx * delta, axis=1)

def pair_search(h, c_base, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, c_base, i, j, xs, n, x0, y0, S)
    for _ in range(3):
        if len(xs) <= 1:
            break
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:
            a, b = xs[-2], xs[-1]
        else:
            a, b = xs[k - 1], xs[k + 1]
        if b - a < 1e-11:
            break
        xs2 = np.linspace(a, b, 64)
        vals = pair_eval(h, c_base, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(S - xb)
        if yb < 0.0:
            yb = 0.0
            xb = S - yb
        elif yb > 1.0:
            yb = 1.0
            xb = S - yb
        return xb, yb, bv
    return None

def subgrad_phase(h, n, deadline, seed):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    curh = h.copy()
    rng = np.random.RandomState(seed)
    base_step = 0.005
    noimp = 0
    while time.time() < deadline:
        c = compute_c(curh, n)
        max_c = float(c.max())
        # Active set: lags within 5% of max
        thresh = max_c - 0.05 * max_c
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        g = np.zeros(n)
        off = n - 1
        for idx in active:
            k = idx - off
            w = 1.0 / len(active)
            if k > 0:
                m2 = n - k
                g[k:n] += w * (1.0 - curh[:m2])
                g[:m2] -= w * curh[k:n]
            elif k < 0:
                s2 = -k
                m2 = n - s2
                g[:m2] += w * (1.0 - curh[s2:n])
                g[s2:n] -= w * curh[:m2]
            else:
                g += w * (1.0 - 2.0 * curh)
        g *= (2.0 / n)
        # Adaptive step size
        step = base_step * (1.0 + 0.1 * np.sin(noimp * 0.1))
        newh = curh - step * g
        newh = project_box_sum(newh, n)
        newc = compute_c(newh, n)
        newmax = float(newc.max())
        if newmax < bestc:
            bestc = newmax
            besth = newh.copy()
            noimp = 0
        else:
            noimp += 1
            if noimp > 200:
                base_step *= 0.8
                noimp = 0
                base_step = max(base_step, 0.0005)
            # Occasionally perturb to escape
            if noimp > 100 and rng.random() < 0.1:
                pert = 0.01 * rng.randn(n)
                newh = project_box_sum(curh + pert, n)
        curh = newh
    return project_box_sum(besth, n), float(compute_c(besth, n).max())

def slsqp_phase(h, n, alpha, maxiter):

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    return m + np.log(float(np.exp(sh).sum())) / alpha

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    q = e / float(e.sum())
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            m2 = n - k
            g[k:n] += w * (1.0 - x[:m2])
            g[:m2] -= w * x[k:n]
        elif k < 0:
            s2 = -k
            m2 = n - s2
            g[:m2] += w * (1.0 - x[s2:n])
            g[s2:n] -= w * x[:m2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    return lse_grad(x, n, alpha) + mu * float(x.sum() - 0.5 * n)

def lbfgsb_phase(h, n, alpha, mu, maxiter):

    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_grad, bounds=bounds,

                                "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def pair_xs(lo, hi, x0):
    xs = np.array([lo, hi, x0], dtype=float)
    if hi - lo > 1e-12:
        xs = np.concatenate((xs, np.linspace(lo, hi, 24)))
    a = max(lo, x0 - 0.04)
    b = min(hi, x0 + 0.04)
    if b - a > 1e-12:
        xs = np.concatenate((xs, np.linspace(a, b, 96)))
    xs = np.unique(xs)
    xs = xs[(xs >= lo - 1e-12) & (xs <= hi + 1e-12)]
    return np.clip(xs, lo, hi)

def pair_eval(h, c_base, i, j, xs, n, x0, y0, S):
    L = 2 * n - 1
    off = n - 1
    dx = 2.0 / n
    omh = 1.0 - h
    omhrev = omh[::-1]
    delta = np.zeros((len(xs), L), dtype=float)
    ax1 = xs - x0
    ax2 = x0 - xs
    yv = S - xs
    delta[:, i:i + n] += ax1[:, None] * omhrev[None, :]
    si = n - 1 - i
    delta[:, si:si + n] += ax2[:, None] * h[None, :]
    delta[:, j:j + n] += ax2[:, None] * omhrev[None, :]
    sj = n - 1 - j
    delta[:, sj:sj + n] += ax1[:, None] * h[None, :]
    idx0 = off
    base_self = ax1 * (1.0 - x0) + x0 * ax2 + ax2 * (1.0 - y0) + y0 * ax1
    exact_self = xs * (1.0 - xs) + yv * (1.0 - yv) - (x0 * (1.0 - x0) + y0 * (1.0 - y0))
    delta[:, idx0] += exact_self - base_self
    d = j - i
    idxp = off - d
    delta[:, idxp] += xs * (1.0 - yv) - x0 * (1.0 - y0) - (ax1 * (1.0 - y0) + x0 * ax1)
    idxm = off + d
    delta[:, idxm] += yv * (1.0 - xs) - y0 * (1.0 - x0) - (ax2 * (1.0 - x0) + y0 * ax2)
    return np.max(c_base[None, :] + dx * delta, axis=1)

        return None

    cm = float(c_base.max())
    if bv < cm - 1e-14:

        if yb < 0.0:
            yb = 0.0; xb = S
        elif yb > 1.0:
            yb = 1.0; xb = S - 1.0
