def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    if float(h.sum()) == target:
        return h
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    mu = 0.0
    for _ in range(60):
        mu = 0.5 * (lo + hi)
        p = np.clip(h - mu, 0.0, 1.0)
        s = float(p.sum())
        if s > target:
            lo = mu
        else:
            hi = mu
    p = np.clip(h - mu, 0.0, 1.0)
    diff = target - float(p.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - p
            sr = float(room.sum())
            if sr > 1e-14:
                p += (diff / sr) * room
        else:
            sp = float(p.sum())
            if sp > 1e-14:
                p += (diff / sp) * p
        p = np.clip(p, 0.0, 1.0)
    diff = target - float(p.sum())
    if abs(diff) > 1e-16:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        if free.any():
            p[free] += diff / int(free.sum())
            p = np.clip(p, 0.0, 1.0)
        if abs(target - float(p.sum())) > 1e-12:
            p += (target - float(p.sum())) / max(n, 1)
            p = np.clip(p, 0.0, 1.0)
    return p
