import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    logsumexp = np.log(np.sum(np.exp(s)))
    f = (2.0 / n) * (m + logsumexp / alpha)
    w = np.exp(s - logsumexp)
    grad = np.zeros(n)
    coeff = (2.0 / n) / alpha
    for j in range(n):
        k_hi = max(j + n - 1, 2 * n - 2 - j)
        ks = np.arange(0, k_hi + 1)
        idx1 = j - ks + (n - 1)
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + ks - (n - 1)
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(len(ks))
        t1[v1] = (1.0 - x[idx1[v1]]) * w[ks[v1]]
        t2 = np.zeros(len(ks))
        t2[v2] = x[idx2[v2]] * w[ks[v2]]
        grad[j] = coeff * (t1.sum() - t2.sum())
    return f, grad

def slsqp_optimize(h, n, alphas, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        def obj(x):
            return _smooth_obj_and_grad(x, n, alpha)[0]
        def jac(x):
            return _smooth_obj_and_grad(x, n, alpha)[1]
        try:
            res = minimize(obj, best_h, jac=jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid_n=25):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = 0
            k_hi_p = max(p + n - 1, 2 * n - 2 - p)
            m_hi = k_hi_p
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            a[vf] += (1.0 - h[jf[vf]])
            a[vb] -= h[jb[vb]]
            diag = np.where(ms == (n - 1))[0]
            c_aff = c[ms]
            unaff = 0.0
            if m_hi < 2 * n - 2:
                unaff = float(c[m_hi + 1:].max())
            d_grid = np.linspace(-h0, 1.0 - h0, grid_n)
            best_d, best_v = 0.0, float(c.max())
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_aff + d * a
                if len(diag) > 0:
                    nv[diag] -= d * d
                v = max(unaff, float(nv.max()))
                if v < best_v:
                    best_v, best_d = v, d
            if 1 <= np.argmin([max(unaff, float((c_aff + d * a).max() - d*d if len(diag)>0 else 0))]) < grid_n - 1:
                pass
            # Parabolic refinement
            if best_d != 0.0:
                lo = max(-h0, best_d - 2.0 * h0 / grid_n)
                hi = min(1.0 - h0, best_d + 2.0 * h0 / grid_n)
                for _ in range(5):
                    d1 = lo + (hi - lo) / 3.0
                    d2 = lo + 2.0 * (hi - lo) / 3.0
                    v1 = max(unaff, float((c_aff + d1 * a).max() - d1*d1 if len(diag)>0 else 0))
                    v2 = max(unaff, float((c_aff + d2 * a).max() - d2*d2 if len(diag)>0 else 0))
                    if v1 < v2:
                        hi = d2
                    else:
                        lo = d1
                d_star = (lo + hi) / 2.0
                nv_s = c_aff + d_star * a
                if len(diag) > 0:
                    nv_s[diag] -= d_star * d_star
                v_star = max(unaff, float(nv_s.max()))
                if v_star < best_v:
                    best_d, best_v = d_star, v_star
            if abs(best_d) > 1e-12:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                d = h[p] - h0
                c[ms] += d * a
                if len(diag) > 0:
                    c[ms][diag] -= d * d
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def binary_search_hill(n, rng, budget_iter=2000):
    """Hill-climb a binary vector to maximize min autocorrelation."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(c.max()) * 2.0 / n
    for _ in range(budget_iter):
        i = rng.randint(n)
        if h[i] == 1.0:
            candidates = np.where(h == 0)[0]
        else:
            candidates = np.where(h == 1)[0]
        if len(candidates) == 0:
            break
        j = candidates[rng.randint(len(candidates))]
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(c.max()) * 2.0 / n
        if c5 >= best_c5:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        else:
            best_c5 = c5
    return h, best_c5
