    # For term2: g2[k,j] = h0[j-shifts[k]] (valid if 0<=j-shifts[k]<n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h0, n):
    m = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    j_idx = np.arange(n)
    jp = j_idx[None, :] + shifts[:, None]
    jm = j_idx[None, :] - shifts[:, None]
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    t1 = np.where(vp, 1.0 - h0[np.clip(jp, 0, n - 1)], 0.0)
    t2 = np.where(vm, h0[np.clip(jm, 0, n - 1)], 0.0)
    return t1 - t2

def top_lag_step(h, n, max_step=0.01):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    k_star = int(np.argmax(c))
    g = compute_grad_matrix(h, n)
    b_ub = c_max - c
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-max_step, max_step)] * n
    c_obj = g[k_star]
    res = linprog(c_obj, A_ub=g, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success or res.fun >= -1e-15:
        return None
    return res.x

def topk_lag_step(h, n, k=5, max_step=0.01):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    g = compute_grad_matrix(h, n)
    topk = np.argsort(c)[::-1][:k]
    b_ub = c_max - c
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-max_step, max_step)] * n
    c_obj = g[topk].mean(axis=0)
    res = linprog(c_obj, A_ub=g, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success or res.fun >= -1e-15:
        return None
    return res.x

def lse_slsqp(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        shifted = alpha * c
        sm = shifted.max()
        return sm / alpha + np.log(np.exp(shifted - sm).sum()) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def barrier_slsqp(h, n, t_target, maxiter=2500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        excess = np.maximum(c - t_target, 0)
        return 0.5 * np.dot(excess, excess)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def do_top_lag_phase(h, n, best_c, dl, max_iters=40, eps=0.005):
    for it in range(max_iters):
        if time.time() > dl:
            break
        d = top_lag_step(h, n, max_step=eps)
        if d is None:

            if c_try < best_c - 1e-16:
                h, best_c = h_try, c_try
                improved = True
                break
        if not improved:
            break
    return h, best_c
