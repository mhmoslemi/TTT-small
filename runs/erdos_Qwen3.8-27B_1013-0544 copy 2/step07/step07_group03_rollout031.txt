def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

                delta = np.zeros(2 * n - 1)
                im = i + s_range
                msk = (im >= 0) & (im < n)
                delta[msk] -= h_b[im[msk]]
                jmn = j - s_range
                msk2 = (jmn >= 0) & (jmn < n)
                delta[msk2] += h_b[jmn[msk2]]
                jm = j + s_range
                msk3 = (jm >= 0) & (jm < n)
                delta[msk3] += h_b[jm[msk3]]
                imn = i - s_range
                msk4 = (imn >= 0) & (imn < n)
                delta[msk4] -= h_b[imn[msk4]]
                
                h_b[i], h_b[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                # Update ones and zeros
                ones = np.delete(ones, best_local)
                ones = np.append(ones, i)
                zeros = zeros[zeros != i]
                zeros = np.append(zeros, j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
                # Need to recompute B since h_b changed
                # Actually, B only changes at positions i and j
                # B[i, :] = h_b[i+s] + h_b[i-s] (changed because h_b[i] changed)
                # B[j, :] = h_b[j+s] + h_b[j-s] (changed because h_b[j] changed)
                # But also, B[k, s] for k ≠ i,j might change if s = j-k or s = k-j
                # This is complex, so let's just recompute B each sweep
                # Actually, let's recompute B after each swap... that's expensive
                # Alternative: just recompute B at the start of each sweep
                break  # Restart the sweep with updated B
        
        if not improved:
            # Perturb
            k_pert = rng.randint(3, 8)
            for _ in range(k_pert):
                idx_z = rng.randint(len(zeros))
                idx_o = rng.randint(len(ones))
                ii = zeros[idx_z]
                jj = ones[idx_o]
                delta = np.zeros(2 * n - 1)
                im = ii + s_range
                msk = (im >= 0) & (im < n)
                delta[msk] -= h_b[im[msk]]
                jmn = jj - s_range
                msk2 = (jmn >= 0) & (jmn < n)
                delta[msk2] += h_b[jmn[msk2]]
                jm = jj + s_range
                msk3 = (jm >= 0) & (jm < n)
                delta[msk3] += h_b[jm[msk3]]
                imn = ii - s_range
                msk4 = (imn >= 0) & (imn < n)
                delta[msk4] -= h_b[imn[msk4]]
                h_b[ii], h_b[jj] = 1.0, 0.0
                corr += delta
                cur_c = corr.max() * dx
                ones = np.delete(ones, idx_o)
                ones = np.append(ones, ii)
                zeros = np.delete(zeros, idx_z)
                zeros = np.append(zeros, jj)
    
    return best_h, best_c

Option 2 is better. After a swap, B is only slightly wrong (only entries involving the swapped positions are wrong). Since we're doing greedy improvement, the first swap in a sweep is the most important, and subsequent swaps use slightly outdated B. This should be fine for convergence.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def apply_swap(h_b, corr, i, j, s_range, n):
    delta = np.zeros(2 * n - 1)
    im = i + s_range
    msk = (im >= 0) & (im < n)
    delta[msk] -= h_b[im[msk]]
    jmn = j - s_range
    msk2 = (jmn >= 0) & (jmn < n)
    delta[msk2] += h_b[jmn[msk2]]
    jm = j + s_range
    msk3 = (jm >= 0) & (jm < n)
    delta[msk3] += h_b[jm[msk3]]
    imn = i - s_range
    msk4 = (imn >= 0) & (imn < n)
    delta[msk4] -= h_b[imn[msk4]]
    h_b[i], h_b[j] = 1.0, 0.0
    corr += delta
    return corr

def binary_greedy(h_b, n, deadline, max_sweeps=8000, perturb_k=5):
    dx = 2.0 / n
    s_range = np.arange(-(n - 1), n)
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones = list(np.where(h_b == 1)[0])
    zeros = list(np.where(h_b == 0)[0])
    stuck_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        rng = np.random.RandomState(sweep * 31 + 7)
        zeros_shuffled = zeros[:]
        rng.shuffle(zeros_shuffled)
        improved = False
        
        for i in zeros_shuffled:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                ncor = np.zeros(2 * n - 1)
                im = i + s_range
                msk = (im >= 0) & (im < n)
                ncor[msk] -= h_b[im[msk]]
                jmn = j - s_range
                msk2 = (jmn >= 0) & (jmn < n)
                ncor[msk2] += h_b[jmn[msk2]]
                jm = j + s_range
                msk3 = (jm >= 0) & (jm < n)
                ncor[msk3] += h_b[jm[msk3]]
                imn = i - s_range
                msk4 = (imn >= 0) & (imn < n)
                ncor[msk4] -= h_b[imn[msk4]]
                nc = (corr + ncor).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                corr = apply_swap(h_b, corr, i, best_j, s_range, n)
                cur_c = best_nc
                ones.remove(best_j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(best_j)
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h_b.copy()
        
        if not improved:
            stuck_count += 1
            k_pert = min(perturb_k + stuck_count, 15)
            for _ in range(k_pert):
                if len(zeros) > 0 and len(ones) > 0:
                    ii = zeros[rng.randint(len(zeros))]
                    jj = ones[rng.randint(len(ones))]
                    corr = apply_swap(h_b, corr, ii, jj, s_range, n)
                    cur_c = corr.max() * dx
                    ones.remove(jj)
                    ones.append(ii)
                    zeros.remove(ii)
                    zeros.append(jj)
            stuck_count = 0
        else:
            stuck_count = max(0, stuck_count - 1)
    
    return best_h, best_c

def subgrad_optimize(h, n, deadline, max_iters=20000, top_k=3, lr_base=0.08):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    G = np.zeros(n)
    momentum = np.zeros(n)
    beta = 0.9
    t = 0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            h = best_h + 0.01 * np.random.RandomState(t).randn(n)
            h = project_feasible(h, n)
            G[:] = 0.0
            momentum[:] = 0.0
            stall = 0
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            if k == 0:
                grad[:] += 1.0 - 2.0 * h
            elif k > 0:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
            else:
                kk = -k
                grad[:n - kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n - kk]
        grad = dx * grad / top_k
        momentum = beta * momentum + (1 - beta) * grad
        G += momentum ** 2
        lr = lr_base / (np.sqrt(G) + 1e-8)
        h_new = project_feasible(h - lr * momentum, n)
        c_chk = np.correlate(h_new, 1 - h_new, mode='full') * dx
        if c_chk.max() < cur_c + 0.003:
            h = h_new
    return best_h, best_c

    t0 = time.time()

        c = compute_c5(h, n)

    # Subgradient

            hh = project_feasible(hh, nn)

        h = slsqp_optimize(h, n, 1e8, 1000, 1e-16)
        c = compute_c5(h, n)

    return best_h, best_c, n
