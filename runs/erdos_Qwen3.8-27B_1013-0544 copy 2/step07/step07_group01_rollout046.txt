    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_exact(h, n):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    g_max = g.max()
    sh = alpha * (g - g_max)

def slsqp_with_t(h, n, alpha, maxiter, ftol, deadline):
    """Minimize t subject to smooth max constraint and box/sum constraints"""
    dx = 2.0 / n
    
    def obj(x):
        # x = [h_0, ..., h_{n-1}, t]
        return x[n]
    
    def ineq_constraint(x):
        h_part = x[:n]
        t = x[n]
        g = np.correlate(h_part, 1.0 - h_part, mode='full') * dx
        sh = alpha * (g - t)
        # log-sum-exp for numerical stability
        sh_max = sh.max()
        return np.log(np.exp(sh - sh_max).sum()) + sh_max  # ≤ 0 means all g_k ≤ t (approx)
    
    def eq_constraint(x):
        return x[:n].sum() - n/2
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    # Initial guess for t
    t0 = compute_c5(h, n) + 0.05
    x0 = np.append(h, t0)
    
    cons = [
        {'type': 'ineq', 'fun': lambda x: -ineq_constraint(x)},  # negated for ineq (≥0)
        {'type': 'eq', 'fun': eq_constraint}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_res = project_exact(res.x[:n], n)
        return h_res
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_exact(h, n):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=n/2} via bisection."""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def slsqp_t_var(h, n, alpha, maxiter, ftol, deadline):
    """Minimize t s.t. logsumexp(alpha*(g-t))<=0, sum(h)=n/2, 0<=h<=1."""
    dx = 2.0 / n

    def obj(x):
        return x[n]

    def ineq_fun(x):
        hp = x[:n]
        t = x[n]
        g = np.correlate(hp, 1.0 - hp, mode='full') * dx
        sh = alpha * (g - t)
        sh_max = sh.max()
        return -(np.log(np.exp(sh - sh_max).sum()) + sh_max)

    def eq_fun(x):
        return x[:n].sum() - n / 2.0

    t0 = compute_c5(h, n) + 0.05
    x0 = np.append(h.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun},
        {'type': 'eq', 'fun': eq_fun}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < deadline:
            return project_exact(res.x[:n], n)
    except Exception:
        pass
    return h


def momentum_subgrad(h, n, deadline, mu0=0.85):
    """Momentum subgradient with exact projection and adaptive step."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    mu = mu0
    lr = dx * 3.0
    momentum = np.zeros(n)
    stall = 0
    for step in range(20000000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        elif lag < 0 and -lag < n:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll])
            grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h = project_exact(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, dx * 30.0)
        else:
            stall += 1
            if stall > 300:
                lr *= 0.93
                mu = max(mu - 0.0005, 0.5)
                if lr < dx * 0.02:
                    lr = dx * 0.5
                    mu = min(mu + 0.05, 0.95)
                    stall = 0
    return best_h, best_c


def smooth_grad(h, n, alpha, deadline, mu=0.8):
    """Gradient descent on log-sum-exp smooth with exact projection."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 1.0 / (alpha * dx * 0.5)
    momentum = np.zeros(n)
    for step in range(500000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        sh = alpha * (corr - corr.max())
        p = np.exp(sh)
        p /= p.sum()
        S_plus = np.zeros(n)
        S_minus = np.zeros(n)
        for li in range(2 * n - 1):
            l = li - (n - 1)
            w = p[li]
            if w < 1e-10:
                continue
            if l >= 0 and l < n:
                S_plus[:n - l] += w * h[l:]
                S_minus[l:] += w * h[:n - l]
            elif l < 0 and -l < n:
                ll = -l
                S_plus[ll:] += w * h[:n - ll]
                S_minus[:n - ll] += w * h[ll:]
        grad = dx * ((1.0 - S_plus) - S_minus)
        momentum = mu * momentum - lr * grad
        h = project_exact(h + momentum, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
