    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + diff / (tot + 1e-30))
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + diff / (tot + 1e-30))
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    # Softmax weights
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    wsum = weights.sum()
    obj = (m + mx / alpha + np.log(wsum) / alpha) * dx
    return obj
