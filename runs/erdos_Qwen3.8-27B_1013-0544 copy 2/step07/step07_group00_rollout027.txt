def solve_1d_minimax(c_base_n1, a, t_lo, t_hi):
    """
    Solve min_t max(c_base_n1 + t - t^2, max_i (c_base[m_i] + a_i * t))
    over t ∈ [t_lo, t_hi].
    
    a: array of linear coefficients for each affected m ≠ n-1
    c_base_n1: base value of c[n-1] (without h[p]'s contribution)
    """
    n_affected = len(a)
    if n_affected == 0:
        # Only the quadratic term
        # g(t) = c_base_n1 + t - t^2, minimize over [t_lo, t_hi]
        # g is concave, so minimum is at a boundary
        t_candidates = [t_lo, t_hi]
    else:
        # Find all intersection points between g(t) and l_i(t)
        # g(t) = l_i(t) → c_base_n1 + t - t^2 = c_base[m_i] + a_i * t
        # → t^2 + (a_i - 1) * t + (c_base[m_i] - c_base_n1) = 0

        for i in range(n_affected):

            A = 1.0
            B = a[i] - 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
