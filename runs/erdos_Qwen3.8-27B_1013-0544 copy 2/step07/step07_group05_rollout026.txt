def c5_val(h, n):

def proj(x, tgt):

            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(tgt - y.sum())
    for _ in range(5):

        c = int(m.sum())

    z = beta * (c - mc)

    e = np.exp(z)
    s = float(e.sum())

    wp = np.zeros(n)
    wn = np.zeros(n)

    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kn = np.zeros(2 * n - 1)
    kn[mid + 1:] = s[1:]
    kn[:mid][::-1] = s[1:]
    nb = np.convolve(h, kn, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - nb) * dx
