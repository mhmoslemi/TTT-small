def compute_corr(h, n):
    dx = 2.0 / float(n)

    h = np.clip(x, 0.0, 1.0)

    B = np.convolve(P, h, mode="full")[n-1:2*n-1]

    dev = h.sum() - 0.5 * n
    F += 0.5 * pen * dev * dev

    """Take a subgradient step to reduce the max correlation."""
    c = compute_corr(h, n)
    M = c.max()
    # Get indices of top correlations
    top_idx = np.argsort(-c)[:n_active]
    # Compute subgradient: average of gradients of active constraints
    dx = 2.0 / float(n)
    g = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)  # actual shift
        # c[k_idx] = sum_i h[i]*(1-h[i+k])*dx
        # d/dc[h[j]] = dx*(1-h[j+k]) + dx*(-h[j-k]) (with bounds)
        # First term: h[j] appears as h[i] with i=j, need j+k < n

        g[mask1] += dx * (1.0 - h[np.arange(n)[mask1] + k])
        # Second term: h[j] appears as h[i+k] with i=j-k, need j-k >= 0
        mask2 = (np.arange(n) - k) >= 0
        mask2 = mask2 & ((np.arange(n) - k) < n)
        g[mask2] -= dx * h[np.arange(n)[mask2] - k]
    g /= n_active
    h_new = project_feasible(h - step * g, n)
    return h_new

    """Generate Thue-Morse binary sequence of length n."""
    t = np.zeros(n)

    return t

            j >>= 1
        r[i] = f
    return r

            hw = project_feasible(res.x, best_n)
            c = compute_c5(hw, best_n)
            if c < best_c:
                best_c = c
                best_h = hw
        except Exception:
            pass

    # Final variant check
    for cand in [best_h, best_h[::-1], 1.0 - best_h[::-1]]:
        pc = project_feasible(cand, best_n)
        cc = compute_c5(pc, best_n)
        if cc < best_c:
            best_c = cc
            best_h = pc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
