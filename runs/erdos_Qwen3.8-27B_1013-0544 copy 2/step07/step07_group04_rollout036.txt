        if hasattr(r, 'x') and r.x is not None:

def pair_eval(h, c, i, j, xr, n):
    """Evaluate max c after changing h[i] to xr and h[j] to S-xr."""
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    dxi = xr - x0
    dxj = S - xr - y0
    dx = 2.0 / n
    off = n - 1
    omh = 1.0 - h
    L = 2 * n - 1
    delta = np.zeros(L)
    # h[i] changes by dxi
    t1 = np.arange(0, n - i)
    if len(t1) > 0:
        delta[off+t1] += dxi * omh[i:i+len(t1)]
    t2 = np.arange(max(1, i-n+1), i+1) if i > 0 else np.array([], dtype=int)
    if len(t2) > 0:
        delta[off+t2] -= dxi * h[i-t2]
    # h[j] changes by dxj
    t3 = np.arange(0, n - j)
    if len(t3) > 0:
        delta[off+t3] += dxj * omh[j:j+len(t3)]
    t4 = np.arange(max(1, j-n+1), j+1) if j > 0 else np.array([], dtype=int)
    if len(t4) > 0:
        delta[off+t4] -= dxj * h[j-t4]
    return float(np.max(c + dx * delta))

def pair_search_fast(h, c, i, j, n):
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-14:
        return False
    cm0 = float(c.max())
    best_x = x0
    best_v = cm0
    xs = np.linspace(lo, hi, 21)
    for x in xs:
        v = pair_eval(h, c, i, j, x, n)
        if v < best_v:
            best_v = v
            best_x = x
    if best_v < cm0 - 1e-13:
        mid = (hi - lo) / 21.0
        a = max(lo, best_x - mid)
        b = min(hi, best_x + mid)
        xs2 = np.linspace(a, b, 31)
        for x in xs2:
            v = pair_eval(h, c, i, j, x, n)
            if v < best_v:
                best_v = v
                best_x = x
        h[i] = best_x
        h[j] = S - best_x
        return True
    return False

def pair_loop(h, n, deadline, seed):
