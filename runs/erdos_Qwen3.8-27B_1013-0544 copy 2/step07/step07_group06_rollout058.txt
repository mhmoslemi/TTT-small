      h = np.asarray(h, dtype=float).ravel()
      if h.size != n:

      target = n // 2
      idx = np.argsort(-h)
      hb = np.zeros(n)

      k = np.arange(2*n-1) - (n-1)  # shifts: -(n-1) to (n-1)
