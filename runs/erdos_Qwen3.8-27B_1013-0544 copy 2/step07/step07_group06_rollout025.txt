from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_correlate_max(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c))

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    # Bisection for threshold
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0
    return hb

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

    best = work.copy()
    bc = compute_c5(best, n)

def binary_greedy(h, n, deadline, max_sweeps=30, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        # Compute base for all possible swaps
        base = np.zeros((n, 2 * n - 1), dtype=float)
        l = np.arange(-(n - 1), n, dtype=int)
        for p in range(n):
            f = p + l
            mf = (f >= 0) & (f < n)
            base[p, mf] = 1.0 - h[f[mf]]
            b = p - l
            mb = (b >= 0) & (b < n)
            base[p, mb] -= h[b[mb]]
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            arr[:, d1] += 1.0
            d2 = n - 1 + (i - j)
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv = int(i[loc])
                jv = int(j[loc])
                h[iv] = 1.0
                h[jv] = 0.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(float(corr.max())))
                if M < bestM:
                    bestM = M
                    besth = h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def top_n_binary(h, n):
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0
    return hb

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

_stop_time = 0.0

def _stop_cb(*a):

def polish_solution(h, n, betas, pen, deadline, stage_time):

    best = work.copy()

def binary_sa(h0, n, deadline, max_iters=200000, seed=42):

        return hb, compute_c5(hb, n)

    besth = h.copy()
    bestM = M
    rng = np.random.RandomState(seed)
    T0, T1 = 2.0, 0.005
    for it in range(max_iters):
        if time.time() > deadline:
            break
        alpha = it / max(1, max_iters - 1)
        T = T0 * (T1 / T0) ** alpha
        ones = np.nonzero(h == 1)[0]
        zeros = np.nonzero(h == 0)[0]
        if ones.size < 2 or zeros.size < 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        newM = float(new_corr.max())
        delta = newM - M
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            M = newM
            corr = new_corr
            if M < bestM:
                bestM = M
                besth = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return besth, bestM * (2.0 / float(n))

def binary_greedy(h, n, deadline, max_sweeps=25, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = float(corr.max())
    besth = h.copy()
    bestM = M
    l = np.arange(-(n - 1), n, dtype=int)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        base = np.zeros((n, 2 * n - 1), dtype=float)
        for p in range(n):
            f = p + l
            mf = (f >= 0) & (f < n)
            base[p, mf] = 1.0 - h[f[mf]]
            b = p - l
            mb = (b >= 0) & (b < n)
            base[p, mb] -= h[b[mb]]
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            arr[:, d1] += 1.0
            d2 = n - 1 + (i - j)
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                h[int(i[loc])] = 1.0
                h[int(j[loc])] = 0.0
                corr = arr[loc].astype(float, copy=True)
                M = float(corr.max())
                if M < bestM:
                    bestM = M
                    besth = h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))
