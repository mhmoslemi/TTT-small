   x=project_feasible(x,n)
   best=x.copy(); bestc=compute_c5(x)
   m=np.zeros(n); v=np.zeros(n); t=0
   b1,b2=0.9,0.999

      for it in range(iters):
         if time.time()>deadline: break
         F,c,d=fdc(x,n,mode,param)

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).ravel()
    if z.size != n:
        z = np.resize(z, n)
    z = np.nan_to_num(z, copy=False, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(12):
        slo = float(np.sum(np.clip(z - lo, 0.0, 1.0)))
        shi = float(np.sum(np.clip(z - hi, 0.0, 1.0)))
        if slo > target and shi < target:
            break
        if slo <= target:
            lo *= 2.0
        if shi >= target:
            hi *= 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            h[mask] += diff / float(np.count_nonzero(mask))
        else:
            i = int(np.argmax(np.minimum(h, 1.0 - h)))
            h[i] = min(1.0, max(0.0, h[i] + diff))
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):

        order = np.argsort(h)[::-1]
        hb = np.zeros(n)

def fdc(x, n, mode, param):
    x = np.asarray(x, dtype=float).ravel()

def grad_from_d(x, d, n):
    dx = 2.0 / n
    pd = np.concatenate(([0.0], np.cumsum(d)))
    lin = pd[n:] - pd[:n]
    c1 = np.convolve(x, d, mode="full")
    c2 = np.convolve(x, d[::-1], mode="full")
    y = c1[n - 1:n - 1 + n] + c2[n - 1:n - 1 + n]
    return (lin - y) * dx

def _smooth_fun(x):

    return float(F)

    _, _, d = fdc(x, G_N, G_MODE, G_PARAM)
    g = grad_from_d(x, d, G_N)
    if not np.all(np.isfinite(g)):

    return g

    return np.ones(len(x), dtype=float)

def slsqp_step(x, n, mode, param, maxiter):
    if maxiter <= 0:
        return project_feasible(x, n)
    x = project_feasible(x, n)
    set_gp(n, mode, param)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]
    try:
        res = minimize(
            _smooth_fun, x,
            jac=_smooth_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1.0e-13, "disp": False},
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return x

        c2[t2] += 1.0
    return c2, float(c2.max())

    return best, bestc

def make_perturb(h, n, rng, limit=8):
    x = np.linspace(0.0, 2.0, n)
    out = []
    for amp in (0.004, 0.008, 0.015, 0.030):
        out.append(h + amp * rng.randn(n))
    for f in (1.5, 3.0, 5.5, 9.0):
        out.append(h + 0.020 * np.sin(2.0 * np.pi * f * x + 0.9))
        out.append(h + 0.018 * np.cos(2.0 * np.pi * f * x + 1.7))
    out.append(1.0 - h + 0.010 * rng.randn(n))
    out.append(h[::-1] + 0.010 * rng.randn(n))
    res = []
    for a in out:
        res.append(project_feasible(a, n))
        if len(res) >= limit:
            break
    return res

def hinge_push(x, n, deadline):
    x = project_feasible(x, n)
    best = x.copy()
    bestc = compute_c5(x)
    offs = (1.0e-5, 2.0e-5, 5.0e-5, 1.0e-4)
    for off in offs:
        if time.time() > deadline - 20.0:
            break
        t = min(bestc - off, 0.38078)
        if t >= bestc - 1.0e-7:
            continue
        x = slsqp_step(x, n, "hinge", (t, 200.0), 180)
        c = compute_c5(x)
        if c < bestc:
            bestc = c
            best = x.copy()
        else:
            x = best.copy()
        if time.time() < deadline - 50.0:
            x2, ac = adam_smooth(
                best, n,
                [("hinge", (t, 200.0), 70), ("lse", 5.0e-5, 50)],
                deadline - 40.0,
            )
            if ac < bestc:
                bestc = ac
                best = x2.copy()
        if bestc <= 0.38075:
            break
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 15.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    hp = None
    try:
        hp = np.asarray(initial_h_values, dtype=float).ravel().copy()
    except Exception:
        hp = None

    if hp is None or hp.size == 0:
        n = 400
        x0 = np.linspace(0.0, 2.0, n)
        hp = 0.5 + 0.5 * np.sin(2.0 * np.pi * x0)
    n = int(hp.size)
    if n < 24:
        n2 = 24
        hp = np.interp(np.linspace(0.0, 2.0, n2), np.linspace(0.0, 2.0, n), hp)
        n = n2
    if n > 1000:
        n2 = 960
        hp = np.interp(np.linspace(0.0, 2.0, n2), np.linspace(0.0, 2.0, n), hp)
        n = n2
    if n % 2 != 0:
        n2 = n - 1
        hp = np.interp(np.linspace(0.0, 2.0, n2), np.linspace(0.0, 2.0, n), hp)
        n = n2

    hp = project_feasible(hp, n)
    best_h = hp.copy()
    best_c = compute_c5(best_h)

    dl1 = t0 + 0.22 * total
    if time.time() < dl1 - 5.0:
        bh, bc = refine(best_h, n, dl1, heavy=True, do_adam=False)
        if bc < best_c:
            best_c, best_h = bc, bh

    dl2 = t0 + 0.36 * total
    if n % 2 == 0 and time.time() < dl2 - 35.0:
        hb, cb = binary_local_search(best_h, n, min(dl2, time.time() + 180.0), rng, batch=70)
        if cb < best_c:
            best_c, best_h = cb, hb
            dl2b = min(deadline, dl2 + 0.05 * total)
            if time.time() < dl2b - 15.0:
                bh, bc = refine(best_h, n, dl2b, heavy=False, do_adam=True)
                if bc < best_c:
                    best_c, best_h = bc, bh

    dl3 = t0 + 0.60 * total
    perts = make_perturb(best_h, n, rng, limit=8)
    for idx, s in enumerate(perts):
        if time.time() > dl3 - 35.0:
            break
        span = 45.0 - 6.0 * max(0, idx - 3)
        tele = min(dl3, time.time() + max(20.0, span))
        bh, bc = refine(s, n, tele, heavy=False, do_adam=True)
        if bc < best_c:
            best_c, best_h = bc, bh

    dl4 = t0 + 0.76 * total
    for ns in (128, 160, 200, 256):
        if ns >= n or time.time() > dl4 - 90.0:
            continue
        xold = np.linspace(0.0, 2.0, n)
        xnew = np.linspace(0.0, 2.0, ns)
        hn = project_feasible(np.interp(xnew, xold, best_h), ns)
        tele = min(dl4 - 30.0, time.time() + 70.0)
        bn, cn = refine(hn, ns, tele, heavy=False, do_adam=True)
        if cn < best_c:
            best_c, best_h, n = cn, bn, ns
        if time.time() < dl4 - 45.0:
            rb = np.zeros(ns)
            rb[rng.choice(ns, ns // 2, replace=False)] = 1.0
            br, cr = binary_local_search(rb, ns, min(dl4 - 30.0, time.time() + 45.0), rng, batch=130)
            if cr < best_c:
                best_c, best_h, n = cr, br, ns

    dl5 = t0 + 0.90 * total
    if time.time() < dl5 - 80.0 and n < 512:
        nh = min(600, max(480, n + 80))
        if nh % 2 != 0:
            nh += 1
        xold = np.linspace(0.0, 2.0, n)
        xnew = np.linspace(0.0, 2.0, nh)
        hn = project_feasible(np.interp(xnew, xold, best_h), nh)
        bh, bc = refine(hn, nh, dl5, heavy=False, do_adam=False)
        if bc < best_c:
            best_c, best_h, n = bc, bh, nh

    dlf = deadline - 15.0
    if time.time() < dlf - 25.0:
        best_h, best_c = hinge_push(best_h, n, dlf)

    if time.time() < deadline - 18.0:
        fin = [
            ("pnorm", 512, 70),
            ("lse", 5.0e-5, 90),
            ("lse", 1.0e-5, 100),
        ]
        if time.time() < deadline - 35.0:
            t = min(best_c - 8.0e-6, 0.38078)
            if t < best_c - 1.0e-8:
                fin.insert(0, ("hinge", (t, 250.0), 120))
        bh, bc = slsqp_polish(best_h, n, fin, deadline)
        if bc < best_c:
            best_c, best_h = bc, bh

    if np.all(best_h >= -1.0e-12) and np.all(best_h <= 1.0 + 1.0e-12) and abs(float(best_h.sum()) - 0.5 * n) < 1.0e-11:
        best_h = np.clip(best_h, 0.0, 1.0)
    else:
        best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
