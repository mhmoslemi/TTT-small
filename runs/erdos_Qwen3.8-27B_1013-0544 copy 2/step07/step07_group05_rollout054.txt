import scipy.optimize as so

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:

        else:

        else:

def softmax_values(c, beta):
    maxc = float(np.max(c))

    ew = np.exp(z)
    ws = float(ew.sum())

    if ws <= 0.0:

    else:

    return f, w


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def subgradient_step(h, n, lr, deadline):
    """One subgradient step with Nesterov momentum."""
    dx = 2.0 / n
    c = erdos_c5_vec(h, n)
    maxc = float(np.max(c))
    # Subgradient: gradient of the max-correlation lag
    top_idx = np.where(c > maxc - 1e-6)[0]
    # Average gradient over top lags
    w = np.zeros_like(c)
    for idx in top_idx:
        w[idx] = 1.0
    w /= len(top_idx)
    g = smooth_grad_from_w(h, w, n, dx)
    return h - lr * g


def nesterov_loop(h, n, deadline, lr_init=0.01, momentum=0.9, seed=0):
    """Nesterov accelerated subgradient method."""
    rng = np.random.RandomState(int(seed))
    cur = erdos_c5(h, n)
    v = np.zeros_like(h)
    lr = lr_init
    for it in range(200):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = h + momentum * v
        g = subgradient_step(y, n, lr, deadline)
        # Project the step

        new_c = erdos_c5(h_new, n)
        if new_c < cur - 1e-10:
            v = h_new - h
            h = h_new
            cur = new_c
        else:
            v = momentum * v * 0.5
        lr *= 0.98
    return h, cur

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo, hi = float(np.min(x)) - 1.0, float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / int(mask.sum())
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

    mc = float(np.max(c))
    z = np.clip(beta * (c - mc), -745.0, 0.0)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = mc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def grad_from_w(h, w, n, dx):
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

    n, dx = int(st['n']), float(st['dx'])

    f, w = softmax_mx(c, float(st['beta']))
    g = grad_from_w(x, w, n, dx)
    return float(f), g

    except Exception:

def s_obj(x):

    s_obj(x)
    return st['cg'].copy()

def s_con_j(x):
    return np.ones(int(SS['n']), dtype=float)

def refine_slsqp(h, beta, maxiter, deadline):

    target = 0.5 * n

    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)

    h = st['h'].copy()

    st['cg'] = g[st['start']:st['start']+st['k']].copy()

    return st['cg'].copy()

import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'beta': 1.0}
SS = {'n': 1, 'dx': 1.0, 'beta': 1.0, 'cx': None, 'cg': None}
BLK = {'h': None, 'k': 1, 'start': 0, 'n': 1, 'beta': 1.0, 'tgt': 0.0, 'cx': None, 'cg': None}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo, hi = float(np.min(x)) - 1.0, float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(4):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / int(mask.sum())
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_mx(c, beta):
    mc = float(np.max(c))
    z = np.clip(beta * (c - mc), -745.0, 0.0)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = mc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w


def grad_from_w(h, w, n, dx):
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    n = int(LS['n'])
    dx = LS['dx']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_mx(c, LS['beta'])
    g = grad_from_w(x, w, n, dx)
    return float(f), g


def refine_lbfgs(h, beta, maxiter, deadline):
    n = len(h)
    target = 0.5 * n
    if time.time() > deadline - 3.0:
        return h, erdos_c5(h, n)
    LS['n'], LS['dx'], LS['beta'] = n, 2.0 / n, float(beta)
    mx = int(min(maxiter, max(15, int((deadline - time.time()) / 3.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(l_fg, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 50, 'maxcor': 20,
                                   'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def s_obj(x):
    n, dx = SS['n'], SS['dx']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_mx(c, SS['beta'])
    g = grad_from_w(x, w, n, dx)
    SS['cx'], SS['cg'] = x.copy(), g
    return float(f)


def s_jac(x):
    if SS['cx'] is not None and np.array_equal(x, SS['cx']):
        return SS['cg'].copy()
    s_obj(x)
    return SS['cg'].copy()


def s_con(x):
    return float(x.sum() - 0.5 * SS['n'])


def s_con_j(x):
    return np.ones(SS['n'], dtype=float)


def refine_slsqp(h, beta, maxiter, deadline):
    n = len(h)
    if time.time() > deadline - 6.0:
        return h, erdos_c5(h, n)
    SS['n'], SS['dx'], SS['beta'] = n, 2.0 / n, float(beta)

def block_cd(h, n, deadline, seed, n_rounds, beta_lo=2000.0, beta_hi=20000.0):

    cur = erdos_c5(h, n)

        if time.time() > deadline:
            break
        prog = it / max(1, n_rounds)
        k_max = max(4, int(20 - 12 * prog))
        k_min = max(3, int(8 - 4 * prog))
        k_max = min(k_max, n - 2)
        k_min = min(k_min, k_max)
        k = rng.randint(k_min, k_max + 1)
        start = rng.randint(0, n - k)
        tgt = float(h[start:start + k].sum())

        BLK['h'], BLK['k'], BLK['start'] = h, k, start

            if res.x is not None:

                nc = erdos_c5(h, n)

                    cur = nc

            pass
