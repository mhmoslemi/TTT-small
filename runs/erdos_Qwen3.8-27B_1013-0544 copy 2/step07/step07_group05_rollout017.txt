    idx = np.arange(n)
    consts = pref_p + pref_n[n-1-idx]
    return (w[mid]*(1.0-2.0*h) + consts - neigh) * dx

def obj_grad_topk(x):
    n = ST['n']
    k = ST['k']
    dx = ST['dx']
    c = np.correlate(x, 1.0-x, mode='full') * dx
    top_idx = np.argsort(c)[::-1][:k]
    # Object value: mean of top k
    f = float(np.mean(c[top_idx]))
    # Gradient
    w = np.zeros(2*n-1)
    w[top_idx] = 1.0/k
    g = compute_weighted_grad(x, w, n)
    return f, g

Let me also think about whether to use the "source" formulation or "dest" formulation for the smooth max:
- Source: f = (1/beta) * log(sum exp(beta*c[k])) - close to max

1. It's exactly the max when k=1
2. It's the mean when k=2n-1 (very smooth)
3. For intermediate k, it captures the "active set" better than softmax (which gives equal weight spread over all lags)

Let me also think about the pairwise step more carefully. The previous code:
- Picks top 25 (or 15 for n>500) fractional coordinates
- Tries all pairs among them (up to 25*24/2 = 300 pairs)
- Does 3 passes

My improvement:
- In each pass, go through ALL coordinates in random order
- For each coordinate i, pick the 5-10 best j candidates (based on |h[i]-h[j]| and position)
- Try the pair update
- This is O(n * log(n)) per pass instead of O(m^2)

Actually, let me think about what makes a good pair (i,j):
- Moving mass from i to j: delta > 0 means h[i] decreases, h[j] increases
- This is good if h[i] is "causing" high correlations and h[j] is "absorbing" them
- A rough heuristic: i should be where h is "too high" and j where h is "too low"
- But it's not just about the value - it's about the position in the correlation structure

For a practical approach, let me just do random pairs but with more of them per sweep, and multiple sweeps. With n=400, if I do 5000 random pairs per sweep and each takes ~0.1ms, that's 0.5s per sweep. I can do 10+ sweeps in 5 seconds.

import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def weighted_grad(h, w, n):
    dx = 2.0 / n
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    if mid + 1 < len(w):
        w_pos[1:] = w[mid + 1:]
    if mid > 0:
        w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    kern = np.zeros(2 * n - 1)
    if mid + 1 < len(kern):
        kern[mid + 1:] = s[1:]
    if mid > 0:
        kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def topk_obj_grad(x, n, k):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top_idx = np.argsort(c)[::-1][:k]
    f = float(np.mean(c[top_idx]))
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k
    g = weighted_grad(x, w, n)
    return f, g

def refine_topk(h, k, n, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    rem = deadline - time.time()
    if rem < 3.0 or n < 2:
        return h, erdos_c5(h, n)

    def _obj(x):
        f, _ = topk_obj_grad(x, n, k)
        return f

    def _jac(x):
        _, g = topk_obj_grad(x, n, k)
        return g

    bounds = [(0.0, 1.0)] * n
    mx = int(min(maxiter, max(10, int(rem / 2.0))))
    try:
        res = so.minimize(_obj, h, jac=_jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 50, 'disp': False})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def refine_softmax(h, beta, n, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    rem = deadline - time.time()
    if rem < 3.0 or n < 2:
        return h, erdos_c5(h, n)
    dx = 2.0 / n
    mid = n - 1

    def _soft_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        maxc = float(np.max(c))
        z = beta * (c - maxc)
        np.clip(z, -745.0, 0.0, out=z)
        ew = np.exp(z)
        ws = float(ew.sum())
        f = maxc + np.log(ws) / beta
        w = ew / ws
        g = weighted_grad(x, w, n)
        return f, g

    def _obj(x):
        return _soft_obj(x)[0]

    def _jac(x):
        return _soft_obj(x)[1]

    bounds = [(0.0, 1.0)] * n
    mx = int(min(maxiter, max(10, int(rem / 2.5))))
    try:
        res = so.minimize(_obj, h, jac=_jac, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxls': 50, 'disp': False})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def subgrad_project(h, n, steps, deadline, lr_init=0.01):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    cur = erdos_c5(h, n)
    lr = lr_init
    for s in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0
        g = weighted_grad(h, w, n)
        h = h - lr * g
        h = project_box_sum(h, target)
        c_new = erdos_c5(h, n)
        if c_new < cur:
            cur = c_new
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.7
            h = project_box_sum(h + lr * g, target)
            cur = erdos_c5(h, n)
    return h, cur

def pair_eval(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, n)

def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    if time.time() > deadline:
        return h, cur
    for frac in (0.0, 0.5, 1.0):
        d = lo + frac * (hi - lo)
        c = pair_eval(d, h, i, j, n)
        if c < best_c:
            best_c = c
            best_d = d
    try:
        if time.time() < deadline - 0.1:
            def _p(d):
                return pair_eval(d, h, i, j, n)
            rs = so.minimize_scalar(_p, bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 15, 'xatol': 1e-10})
            if rs.x is not None:
                c2 = pair_eval(float(rs.x), h, i, j, n)
                if c2 < best_c:
                    best_c = c2
                    best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return hh, float(best_c)
    return h, cur

def pair_sweep(h, n, deadline, seed, n_pairs=3000, n_sweeps=3):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    cur = erdos_c5(h, n)
    pairs_total = n_pairs * n_sweeps
    pairs_done = 0
    for sw in range(n_sweeps):
        if time.time() > deadline - 2.0:
            break
        idx = np.arange(n)
        rng.shuffle(idx)
        for a in range(0, n - 1, 3):
            i = int(idx[a])
            j = int(idx[(a + 3) % n])
            if i == j:
                continue
            hh, cc = pair_line(h, n, i, j, cur, deadline - 1.0)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
            pairs_done += 1
            if pairs_done >= pairs_total:
                break
        if pairs_done >= pairs_total:
            break
    return h, cur

def resample(h, n_old, n_new):
    x = np.linspace(0.0, 2.0, n_old)
    y = np.linspace(0.0, 2.0, n_new)
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * n_new)
