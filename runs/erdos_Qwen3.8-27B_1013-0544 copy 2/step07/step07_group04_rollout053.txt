    return project_box_sum(best_h, n), float(compute_c(best_h, n).max())

Wait, there's a problem. When I accept a worsening move, the current h is now worse. The next pair_search will try to improve from this worse state. But pair_search finds the best single-pair improvement from the current state. If the current state is worse than best, the "improvement" might still be worse than best, and we'd accept it with SA probability.

This creates a random walk in the space, which is the point of SA. But it means we need to be careful:
- Always track best_h and best_c
- The current h might be worse than best
- We should occasionally revert to best if we've been stuck too long

This is exactly what I have with the `no_imp > max_no` check.

Let me also add a "jump" mechanism: if we've been stuck for a while, make a random perturbation and re-optimize briefly.

OK let me just write the complete code now. I'll be concise.
</think>

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt: lo = mid
        else: hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        d = tgt - v.sum()
        if abs(d) < 1e-14: break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        v[m if c else slice(None)] += d / (c if c else n)
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0: return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    return m + np.log(s) / alpha if s > 0 else m

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c); q[int(c.argmax())] = 1.0
    else:
        q = e / s
    g = np.zeros(n); dx = 2.0 / n; off = n - 1
    omh = 1.0 - x; omhr = omh[::-1]
    for idx in np.flatnonzero(q):
        k = int(idx) - off; w = q[idx]
        if k > 0:
            m2 = n - k; g[k:n] += w * omh[:m2]; g[:m2] -= w * x[k:n]
        elif k < 0:
            m2 = n + k; g[:m2] += w * omh[-k:]; g[-k:] -= w * x[:m2]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def sfun(x, n): return float(x.sum() - 0.5 * n)
def sjac(x, n): return np.ones(n)

def pobj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_obj(x, n, alpha) + 0.5 * mu * d * d

def pgrd(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return lse_grad(x, n, alpha) + mu * d

def lbfgs(h, n, alpha, mu, mi):
    try:
        r = minimize(pobj, h, args=(n, alpha, mu), method="L-BFGS-B",
                     jac=pgrd, bounds=[(0.0, 1.0)] * n,
                     options={"maxiter": mi, "maxfun": 4*mi, "ftol": 1e-15, "gtol": 1e-9})
        if r.x is not None: return project_box_sum(r.x, n)
    except Exception: pass
    return h

def slsqp(h, n, alpha, mi):
    try:
        cons = [{"type": "eq", "fun": sfun, "args": (n,), "jac": sjac}]
        r = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                     jac=lse_grad, bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={"maxiter": mi, "ftol": 1e-14})
        if r.x is not None: return project_box_sum(r.x, n)
    except Exception: pass
    return h

def pair_eval(h, cb, i, j, xs, n, x0, y0, S):
    L = 2*n - 1; off = n - 1; dx = 2.0 / n
    omh = 1.0 - h; omhr = omh[::-1]
    d = np.zeros((len(xs), L))
    a1 = xs - x0; a2 = x0 - xs; yv = S - xs
    d[:, i:i+n] += a1[:, None] * omhr[None, :]
    si = n - 1 - i
    d[:, si:si+n] += a2[:, None] * h[None, :]
    d[:, j:j+n] += a2[:, None] * omhr[None, :]
    sj = n - 1 - j
    d[:, sj:sj+n] += a1[:, None] * h[None, :]
    idx0 = off
    bs = a1*(1.0-x0) + x0*a2 + a2*(1.0-y0) + y0*a1
    es = xs*(1.0-xs) + yv*(1.0-yv) - (x0*(1.0-x0) + y0*(1.0-y0))
    d[:, idx0] += es - bs
    dd = j - i
    idxp = off - dd
    d[:, idxp] += a1*(1.0-yv) - x0*(1.0-y0) - a2*(1.0-x0) + y0*a2
    idxm = off + dd
    d[:, idxm] += a2*(1.0-x0) + y0*a2 - yv*(1.0-xs) + y0*(1.0-x0)
    return np.max(cb[None, :] + dx * d, axis=1)

def pair_xs(lo, hi, x0):
    xs = [lo, hi, x0]
    if hi - lo > 1e-12:
        xs.extend(np.linspace(lo, hi, 16).tolist())
    a = max(lo, x0 - 0.03); b = min(hi, x0 + 0.03)
    if b - a > 1e-12:
        xs.extend(np.linspace(a, b, 64).tolist())
    xs = np.clip(np.unique(np.array(xs)), lo, hi)
    return xs

def pair_search(h, cb, i, j, n):
    x0 = float(h[i]); y0 = float(h[j]); S = x0 + y0
    lo = max(0.0, S - 1.0); hi = min(1.0, S)
    if hi - lo < 1e-13: return None
    xs = pair_xs(lo, hi, x0)
    vals = pair_eval(h, cb, i, j, xs, n, x0, y0, S)
    for _ in range(2):
        if len(xs) <= 1: break
        k = int(np.argmin(vals))
        if k == 0: a, b = xs[0], xs[1]
        elif k == len(xs)-1: a, b = xs[-2], xs[-1]
        else: a, b = xs[k-1], xs[k+1]
        if b - a < 1e-11: break
        xs2 = np.linspace(a, b, 48)
        vals = pair_eval(h, cb, i, j, xs2, n, x0, y0, S)
        xs = xs2
    k = int(np.argmin(vals)); bv = float(vals[k])
    if bv < float(cb.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi)); yb = float(S - xb)
        yb = float(np.clip(yb, 0.0, 1.0)); xb = float(np.clip(S - yb, lo, hi))
        return xb, S - xb, bv
    return None

def pair_sa(h, cb, deadline, seed, T0=0.3):
    n = len(h)
    h = project_box_sum(h, n)
    cb = compute_c(h, n)
    best_c = float(cb.max()); best_h = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1; t_start = time.time()
    total = max(deadline - t_start, 1.0)
    no_imp = 0; max_no = 800
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total)
        T = T0 * (0.001) ** frac
        if rng.random() < 0.82:
            pos = cb[off+1:]; neg = cb[off-1::-1]
            w = np.maximum(pos, neg) + 1e-12; w /= w.sum()
            d = int(rng.choice(n-1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d)); j = i + d
        res = pair_search(h, cb, i, j, n)
        if res is not None:
            x, y, val = res
            if val < best_c or rng.random() < np.exp(-(max(0.0, val - best_c)) / max(T, 1e-14)):
                h[i], h[j] = x, y
                cb = compute_c(h, n)
                cur = float(cb.max())
                if cur < best_c:
                    best_c = cur; best_h = h.copy(); no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > max_no:
            h = best_h.copy(); cb = compute_c(h, n); no_imp = 0
            max_no = int(max_no * 1.3)
    return project_box_sum(best_h, n), float(compute_c(best_h, n).max())

def interp_h(h, no, nn):
    if no == nn: return h.copy()
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    return np.interp(xn, xo, h)
