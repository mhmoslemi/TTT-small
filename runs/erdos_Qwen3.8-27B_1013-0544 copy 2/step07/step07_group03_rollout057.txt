def find_best_swap(corr, flip_deltas, zeros, ones, n):
    """Find the swap (i from zeros, j from ones) that minimizes the new max."""
    best_val = np.inf
    best_i = -1
    best_j = -1
    
    for i in zeros:
        base = corr + flip_deltas[i]  # shape (2n-1,)
        # For all j in ones, compute max(base - flip_deltas[j])
        # flip_deltas[ones] has shape (len(ones), 2n-1)
        diffs = base[None, :] - flip_deltas[ones]  # shape (len(ones), 2n-1)

        if vals[idx] < best_val:

            best_i = i
            best_j = ones[idx]

            corr = new_corr
            cur_max = exact_max
            ones.remove(best_j); ones.append(best_i)
            zeros.remove(best_i); zeros.append(best_j)
            flip_deltas = compute_flip_deltas(h_b, n)
            T *= cool
            if cur_max < best_max:
                best_max = cur_max
                best_h = h_b.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            T *= cool

    return best_h, best_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP continuous optimization (25% time)
    p1 = t0 + 0.25 * total
    alphas = [1000, 5000, 50000, 500000, 5e6, 5e7, 5e8]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p1 - 3:
            break
        h = slsqp_optimize(h, n, alpha, 4000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Binary local search (55% time)
    p2 = t0 + 0.80 * total
    h_b = make_binary(best_h, n)
    hb, cb = binary_swap_search(h_b, n, p2, max_iters=3000, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: SLSQP polish on best (10% time)
    p3 = t0 + 0.92 * total
    h = best_h.copy()
    for alpha in [5e6, 5e7, 5e8, 5e9]:
        if time.time() > p3 - 2:
            break
        h = slsqp_optimize(h, n, alpha, 3000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Another binary search round on polished solution
    if time.time() > dl - 30:
        pass
    else:
        h_b = make_binary(best_h, n)
        hb, cb = binary_swap_search(h_b, n, dl - 5, max_iters=500, seed=seed + 1)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 5: Try higher resolution if time permits
    if time.time() < dl - 60 and n <= 500:
        nh = n + 100
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = slsqp_optimize(hh, nh, 5e6, 2000, 1e-15)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            h_b = make_binary(hh, nh)
            hb2, cb2 = binary_swap_search(h_b, nh, dl - 5, max_iters=500, seed=seed + 2)
            if cb2 < best_c:
                best_c, best_h, n = cb2, hb2.copy(), nh
            elif ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
