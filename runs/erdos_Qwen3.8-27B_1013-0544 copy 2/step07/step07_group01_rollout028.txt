        h[idx[:target - n_ones]] = 1

def flip_effect(h, n, j):
    """Compute the change in correlation if we flip bit j."""
    corr_len = 2 * n - 1
    dc = np.zeros(corr_len)
    # j is currently 0 or 1; flipping changes it
    val = h[j]
    # For shift s (position p = (n-1) + s in full correlation):
    # c[p] = sum_i h[i+s]*(1-h[i])
    # Effect of changing h[j]:
    # Case A: j appears as i+s (i.e., i = j-s), contribution h[j]*(1-h[j-s])
    #   Before: val*(1-h[j-s]), After: (1-val)*(1-h[j-s])
    #   Change: (1-2*val)*(1-h[j-s])
    #   Valid: 0 <= j-s <= n-1-s, i.e., s in [j-(n-1-s)... ] 
    #   Actually: i = j-s must be in [0, n-1], and i+s = j must be in [0, n-1] (automatic)
    #   So: 0 <= j-s <= n-1 means j-(n-1) <= s <= j
    s_min_A = max(0, j - (n - 1))
    s_max_A = min(j, n - 1)
    for s in range(s_min_A, s_max_A + 1):
        p = (n - 1) + s
        i = j - s
        dc[p] += (1 - 2 * val) * (1 - h[i])
    
    # Case B: j appears as i (i.e., i = j), contribution h[j+s]*(1-h[j])
    #   Before: h[j+s]*(1-val), After: h[j+s]*val
    #   Change: (2*val - 1)*h[j+s]
    #   Valid: 0 <= j+s <= n-1 means -j <= s <= n-1-j
    s_min_B = max(-(j), 0)  # s can be negative
    s_max_B = min(n - 1 - j, n - 1)
    for s in range(s_min_B, s_max_B + 1):
        p = (n - 1) + s
        dc[p] += (2 * val - 1) * h[j + s]
    
    return dc

def coordinate_descent_swap(h, n, deadline, max_steps=5000):
    """Coordinate descent using swaps (maintains binary and sum constraint)."""
    h = make_binary(h, n)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_mc = corr.max()
    stall = 0

        best_improvement = 0

        best_new_mc = best_mc
        
        # For efficiency, only try a subset if n is large
        if len(zeros) * len(ones) > 50000:
            # Subsample

            # Just do all of them, should be OK for n <= 500
            pass

                # Effect of swapping: flip j (0->1) and flip k (1->0)
                dc_j = flip_effect(h, n, j)  # flip from 0 to 1
                dc_k = flip_effect(h, n, k)  # flip from 1 to 0
                new_corr = corr + dc_j + dc_k
                new_mc = new_corr.max()
                if new_mc < best_new_mc - 1e-15:
                    best_new_mc = new_mc
                    best_jk = (j, k)
        
        if best_jk is not None:

            best_mc = corr.max()
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:

    return best_h, best_mc * (2.0 / n)

def slsqp_opt(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': ftol})

    hsg, csg = subgrad_smooth(hsg, n0, p4, mu=0.92)
    if csg < best_c:
        best_c, best_h = csg, hsg.copy()

    # Phase 5: Multi-start perturbations
    p5 = deadline - 80
    for trial in range(6):
        if time.time() > p5 - 10:
            break
        for nv in [0.005, 0.02, 0.05]:
            hp = best_h + nv * rng.randn(n0)
            hp = project_feasible(hp, n0)
            for a in [10000, 200000]:
                if time.time() > p5 - 5:
                    break
                hp = slsqp_opt(hp, n0, a, 1500, 1e-14)
            cp = compute_c5(hp, n0)
            if cp < best_c:
                best_c, best_h = cp, hp.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 15:
        hf = best_h.copy()
        for a in [500000, 5000000, 50000000]:
            if time.time() > deadline - 8:
                break
            hf = slsqp_opt(hf, n0, a, 3000, 1e-15)
            cf = compute_c5(hf, n0)
            if cf < best_c:
                best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n0)
    best_c = compute_c5(best_h, n0)
    return best_h, best_c, n0
