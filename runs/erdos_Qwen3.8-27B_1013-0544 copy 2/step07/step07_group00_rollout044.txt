import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    sh_max = shifted.max()
    lse = sh_max + np.log(np.sum(np.exp(shifted - sh_max)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter=1000):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for a in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, a), method='SLSQP',
                          bounds=[(0, 1)] * n,
                          constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=200, deadline=None, grid=200):
    h = h.copy()

    for it in range(max_passes):

            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p, p - n + 1)
            m_hi = min(2 * n - 2, n - 1 + p, p + n - 1)
            if m_lo >= m_hi:
                continue
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            qf = p + ms - (n - 1)
            vf = (qf >= 0) & (qf < n)
            if np.any(vf):
                a[vf] += (1.0 - h[qf[vf]])
            qb = p - ms + (n - 1)
            vb = (qb >= 0) & (qb < n)
            if np.any(vb):
                a[vb] -= h[qb[vb]]
            ca = c[ms]
            diag = (ms == n - 1)

            dlo, dhi = -h0, 1.0 - h0

def lp_linearization(h, n, n_iter=50, deadline=None):
    """Iterative linearization: solve LP to get improved h."""
    from scipy.optimize import linprog
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        # Build linearization constraints: c[m] + grad[m].@(h_new - h) <= t
        # grad[m, p] = 1 - h[qf_valid] - h[qb_valid]
        # For efficiency, compute A matrix where A[m, p] = grad[m,p]
        A = np.ones((2 * n - 1, n))
        for m in range(2 * n - 1):
            qf = np.arange(n) + m - (n - 1)
            vf = (qf >= 0) & (qf < n)
            qb = np.arange(n) - m + (n - 1)
            vb = (qb >= 0) & (qb < n)
            A[m, vf] -= h[qf[vf]]
            A[m, vb] -= h[qb[vb]]
        # LP: min t s.t. A @ h_new - t + (A @ h - c) <= 0, 0<=h<=1, sum(h)=n/2
        # Variables: [h_new (n), t (1)]
        # c_obj: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        # Constraints: A @ h_new - t <= c - A @ h  (for each m)
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = c - A @ h
        # Equality: sum(h_new) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                new_h = project_feasible(res.x[:n], n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    h, best_c = new_h, new_c
                else:
                    break
        except Exception:
            break
    return h, best_c
