def pair_search_direct(h, c_base, i, j, n):
    """Find optimal split of h[i]+h[j] using golden section search."""
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-12:
        return None
    
    # Golden section search
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    ga = lo
    gb = hi
    gc = gb - invphi * (gb - ga)
    gd = ga + invphi * (gb - ga)
    
    h[i] = gc
    h[j] = S - gc
    c_curr = compute_c(h, n)
    fc = float(c_curr.max())
    
    h[i] = gd
    h[j] = S - gd
    c_curr = compute_c(h, n)
    fd = float(c_curr.max())
    
    for _ in range(35):
        if gb - ga < 1e-11:
            break
        if fc < fd:
            gb = gd
            gd = gc
            fd = fc
            gc = gb - invphi * (gb - ga)
            h[i] = gc
            h[j] = S - gc
            fc = float(compute_c(h, n).max())
        else:
            ga = gc
            gc = gd
            fc = fd
            gd = ga + invphi * (gb - ga)
            h[i] = gd
            h[j] = S - gd
            fd = float(compute_c(h, n).max())
    
    t_star = 0.5 * (ga + gb)
    h[i] = t_star
    h[j] = S - t_star
    v = float(compute_c(h, n).max())
    
    if v < float(c_base.max()) - 1e-12:
        return v
    else:
        h[i] = x0
        h[j] = y0
        return None
