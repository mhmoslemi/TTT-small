import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h, target, dx, betas, max_iter_per_beta, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 15:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter_per_beta, 'ftol': 1e-20})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def weighted_subgrad(h, target, time_fn, max_iters, m_factor, momentum, min_time=3):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    step = 0.012

    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_factor, max(3, n // 8))
        top_idx = np.argsort(corr)[::-1][:m_act]
        top_vals = corr[top_idx]
        lags = [int(i) - (n - 1) for i in top_idx]
        wsum = np.sum(top_vals) + 1e-15
        weights = top_vals / wsum

        grad = np.zeros(n)
        for w, lag in zip(weights, lags):
            if lag == 0:
                grad += w * (1.0 - 2.0 * best)
            elif lag > 0:
                grad[lag:] += w * (1.0 - best[:-lag])
                grad[:n - lag] -= w * best[lag:]
            else:
                mm = -lag
                grad[:n - mm] += w * (1.0 - best[mm:])
                grad[mm:] -= w * best[:n - mm]

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (grad / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn

        found = False
        s = step
        for _ in range(12):
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                step = min(s * 1.15, 0.035)

                break

                break

def local_move_refine(h, target, time_fn, max_rounds, min_time=5):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(max_rounds):
        if time_fn() < min_time:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:8]
        improved = False
        for idx in top_idx:
            if time_fn() < min_time:
                break
            lag = int(idx) - (n - 1)
            if lag == 0:
                continue
            if lag > 0:
                pairs = [(i + lag, i) for i in range(n - lag)]
            else:
                mm = -lag
                pairs = [(i, i + mm) for i in range(n - mm)]
            for (i, j) in pairs[::max(1, len(pairs)//50)]:
                old_i = best[i]
                old_j = best[j]
                for di, dj in [(-0.01, 0.01), (0.01, -0.01), (-0.005, 0.005)]:

                    test = best.copy()
                    test[i] = ni
                    test[j] = nj

                        best_c5 = c5t
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5
