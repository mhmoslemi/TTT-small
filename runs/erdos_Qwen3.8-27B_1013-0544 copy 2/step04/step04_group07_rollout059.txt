import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        z[zeros[order[:int(diff)]]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:int(-diff)]]] = 0.0
    return z

def smooth_max(x, beta):
    m = np.max(x)
    s = np.sum(np.exp(np.clip(beta * (x - m), -500, 500)))
    return m + np.log(s / len(x)) / beta

def slsqp_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0
