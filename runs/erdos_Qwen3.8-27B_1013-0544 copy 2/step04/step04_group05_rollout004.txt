    top_idx = np.argsort(corr)[::-1][:n_active]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta):
    """Compute smooth max of correlation * dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    obj = (m + np.log(exp_shifted.sum() / len(corr)) / beta) * dx
    
    # Gradient: only compute for lags with significant weight
    threshold = 1e-8
    active = np.where(weights > threshold)[0]
    
    grad = np.zeros(n)
    for idx in active:
        w = weights[idx]
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        elif -n < lag < 0:
            ml = -lag
            grad[:n-ml] += w * (1.0 - h[ml:])
            grad[ml:] -= w * h[:n-ml]
    
    grad *= dx
    return obj, grad

                     step_init, momentum, use_adaptive=True):

    n = len(h_init)
    x = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    mu = momentum
    step = step_init
    no_improve = 0
    patience = 2000
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        
        # Nesterov lookahead
        x_look = x + mu * vel
        x_look = np.clip(x_look, 0.0, 1.0)
        
        obj, grad = smooth_max_obj_grad(x_look, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Gradient step
        vel = mu * vel - step * grad
        x_new = x + vel
        x_new = project_h(x_new, target)
        
        c5_new = compute_c5(x_new)
        
        if c5_new < best_c5 - 1e-14:
            best = x_new.copy()
            best_c5 = c5_new
            x = x_new
            no_improve = 0
        else:
            no_improve += 1
            if use_adaptive and no_improve % 50 == 0:
                step *= 0.8

For the "full" mode, the lags range from -(n-1) to +(n-1).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    weights = exp_vals / s
    
    obj = (m + np.log(s / len(corr)) / beta) * dx
    
    active = np.where(weights > 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        w = weights[idx]
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        elif -n < lag < 0:
            ml = -lag
            grad[:n-ml] += w * (1.0 - h[ml:])
            grad[ml:] -= w * h[:n-ml]
    grad *= dx
    return obj, grad

        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        elif -n < lag < 0:
            ml = -lag
            grad[:n-ml] += w * (1.0 - h[ml:])
            grad[ml:] -= w * h[:n-ml]
    grad *= dx
    return obj, grad

def fista_optimize(h_init, target, beta, time_fn, max_iters, step_init,
                   restart_every=2500, restart_scale=0.55):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(best)
    t = 1.0
    step = step_init
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        if it > 0 and it % restart_every == 0:
            y = x.copy()
            t = 1.0
            step *= restart_scale
            if step < 1e-7:
                step = step_init * 0.05
        obj, grad = smooth_max_obj_and_grad(y, beta)
        x_new = project_h(y - step * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = np.clip(y_new, 0.0, 1.0)
        t = t_new
        x = x_new
        c5_new = compute_c5(x)
        if c5_new < best_c5 - 1e-14:
            best = x.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0:
                step *= 0.9
            if no_improve > 6000:
                break
    return best, best_c5

def slsqp_with_grad(h_init, target, dx, beta, max_iter, time_fn):
    n = len(h_init)
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_fn(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -400, 400)
        return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

    def grad_fn(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -400, 400)
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()
        active = np.where(weights > 1e-10)[0]
        grad = np.zeros(n)
        for idx in active:
            w = weights[idx]
            lag = idx - (n - 1)
            if lag == 0:
                grad += w * (1.0 - 2.0 * h_in)
            elif 0 < lag < n:
                grad[lag:] += w * (1.0 - h_in[:-lag])
                grad[:n-lag] -= w * h_in[lag:]
            elif -n < lag < 0:
                ml = -lag
                grad[:n-ml] += w * (1.0 - h_in[ml:])
                grad[ml:] -= w * h_in[:n-ml]
        grad *= dx
        return grad

    def con_fun(h_in):
        return h_in.sum() - target

    def con_jac(h_in):
        return np.ones(n)

    try:
        res = minimize(obj_fn, h_init, jac=grad_fn, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0

    # Phase 1: FISTA at initial resolution with increasing beta
    for beta in [150, 800, 4000]:
        if rem() < 30:
            break
        best_h, best_c5 = fista_optimize(
            best_h, target0, beta, rem,
            max_iters=15000, step_init=0.04 / np.sqrt(n0),
            restart_every=2500, restart_scale=0.55
        )

    # Phase 2: SLSQP with analytical gradient at initial resolution
    if rem() > 40:
        h_s, c5_s = slsqp_with_grad(best_h, target0, dx0, 5000, 2000, rem)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Upscale to 2*n0
    if rem() > 200:
        n2 = n0 * 2
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)

        for beta in [300, 1500, 8000]:
            if rem() < 50:
                break
            h2, c5_2 = fista_optimize(
                h2, target2, beta, rem,
                max_iters=12000, step_init=0.03 / np.sqrt(n2),
                restart_every=2500, restart_scale=0.55
            )

        if rem() > 60:
            h2, c5_2 = slsqp_with_grad(h2, target2, dx2, 8000, 2000, rem)

        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 4: Final FISTA with very high beta
    n_f = len(best_h)
    target_f = n_f / 2.0
    for beta in [15000, 40000]:
        if rem() < 15:
            break
        best_h, best_c5 = fista_optimize(
            best_h, target_f, beta, rem,
            max_iters=8000, step_init=0.01 / np.sqrt(n_f),
            restart_every=2000, restart_scale=0.5
        )

    # Phase 5: Final SLSQP polish
    if rem() > 25:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        h_s, c5_s = slsqp_with_grad(best_h, target_f, dx_f, 30000, 1500, rem)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 6: Ultra-fine FISTA
    if rem() > 15:
        n_f = len(best_h)
        target_f = n_f / 2.0
        best_h, best_c5 = fista_optimize(
            best_h, target_f, 80000, rem,
            max_iters=6000, step_init=0.005 / np.sqrt(n_f),
            restart_every=1500, restart_scale=0.4
        )

    return best_h, best_c5, len(best_h)
