    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:

            s = room.sum()

            s = h.sum()
            if s < 1e-14:

def analytic_grad_at_lag(h, n, lag):
    """
    Compute ∂g(k)/∂h[i] for a given lag k.
    The correlation index j = lag + (n-1).
    For lag >= 0: ∂g/∂h[i] = 1(i>=lag)*(1-h[i-lag]) - 1(i+lag<=n-1)*h[i+lag]
    For lag < 0: let k=-lag>0: ∂g/∂h[i] = 1(i<=n-1-k)*(1-h[i+k]) - 1(i>=k)*h[i-k]
    """
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        # i >= k: contributes (1 - h[i-k])
        if k < n:
            grad[k:] += (1 - h[:n-k])
        # i + k <= n-1, i.e., i <= n-1-k: contributes -h[i+k]
        if n-1-k >= 0:
            grad[:n-k] -= h[k:]
    else:
        k = -lag
        # i <= n-1-k: contributes (1 - h[i+k])
        if n-1-k >= 0:
            grad[:n-k] += (1 - h[k:])
        # i >= k: contributes -h[i-k]
        if k < n:
            grad[k:] -= h[:n-k]
    return grad * (2.0 / n)

def subgrad_optimize(h, n, deadline, max_iter=100000):
    """Projected subgradient with backtracking and momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iter):

        g = compute_g(h, n)
        m = g.max()

            h = h  # don't update

                lr = 0.1
                v = np.zeros(n)
        
        if m < best_c:
            best_c = m
            best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_and_grad(h, n, alpha):
    """Compute softmax approximation of max(g) and its gradient."""
    g = compute_g(h, n)
    # Numerically stable log-sum-exp
    gm = g.max()
    exp_g = np.exp(alpha * (g - gm))
    sum_exp = exp_g.sum()
    obj = gm + np.log(sum_exp) / alpha
    
    # Weights p_j = exp(alpha * (g_j - gm)) / sum_exp
    p = exp_g / sum_exp
    
    # Gradient: sum_j p_j * dg_j/dh
    grad = np.zeros(n)
    dx = 2.0 / n
    for j in range(2 * n - 1):
        if p[j] < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            k = lag
            if k < n:
                grad[k:] += p[j] * (1 - h[:n-k])
            if n - 1 - k >= 0:
                grad[:n-k] -= p[j] * h[k:]
        else:
            k = -lag
            if n - 1 - k >= 0:
                grad[:n-k] += p[j] * (1 - h[k:])
            if k < n:
                grad[k:] -= p[j] * h[:n-k]
    grad *= dx
    return obj, grad

def lfgbbs_smooth(h, n, deadline, alpha_start=10, alpha_end=10000, n_rounds=5):
    """Optimize using L-BFGS-B with increasing alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for round_i in range(n_rounds):
        if time.time() > deadline:
            break
        # Interpolate alpha
        frac = round_i / max(1, n_rounds - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Define objective and gradient for L-BFGS-B
        # We'll handle the sum constraint via penalty
        penalty = 1e4
        
        def obj_grad(x):
            x_clipped = np.clip(x, 0, 1)
            obj, grad = softmax_obj_and_grad(x_clipped, n, alpha)
            # Add penalty for sum constraint
            s = x_clipped.sum() - n / 2.0
            obj += penalty * s * s
            grad += 2 * penalty * s * np.ones(n)
            return obj, grad
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    return best_h, best_c

    best_c = compute_c5(h, n)
    lr = 0.1

        # Backtracking line search
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * factor
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < obj:
                h = h_new
                break
        
        if compute_c5(h, n) < best_c:
            best_c = compute_c5(h, n)
            best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.9
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_grad(h, n, alpha):
    """Compute gradient of log-sum-exp approximation to max(g)."""
    g = compute_g(h, n)
    gm = g.max()
    exp_g = np.exp(alpha * (g - gm))
    s = exp_g.sum()
    p = exp_g / s  # softmax weights
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-16:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            k = lag
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
            if n - 1 - k >= 0:
                grad[:n-k] -= w * h[k:]
        else:
            k = -lag
            if n - 1 - k >= 0:
                grad[:n-k] += w * (1.0 - h[k:])
            if k < n:
                grad[k:] -= w * h[:n-k]
    return grad * dx

def projected_grad_optimize(h, n, deadline, alphas):
    """Optimize with projected gradient and backtracking line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 0.1
        momentum_vec = np.zeros(n)
        mu = 0.8
        
        for it in range(20000):
            if time.time() > deadline:
                break
            
            grad = softmax_grad(h, n, alpha)
            
            # Momentum update
            momentum_vec = mu * momentum_vec + grad
            direction = momentum_vec
            
            # Backtracking line search
            found = False
            base_obj = compute_c5(h, n)
            for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]:
                step = lr * factor
                h_new = project_feasible(h - step * direction, n)
                c_new = compute_c5(h_new, n)
                if c_new < base_obj:
                    h = h_new
                    found = True
                    break
            
            if found:
                c_now = compute_c5(h, n)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
                lr = min(lr * 1.1, 2.0)
            else:
                lr *= 0.8
                mu = mu * 0.95
                if lr < 1e-10:
                    break
    
    return best_h, best_c

def binary_search_local(h_init, n, deadline, max_iters=200000):
    """Binary local search via random 1-swap moves."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    rng = np.random.RandomState(123)
    
    accept_temp = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            ones.remove(j); ones.add(i)
            zeros.remove(i); zeros.add(j)
        else:
            h[i], h[j] = 0.0, 1.0
            # Occasionally accept worse moves (simulated annealing)
            if rng.rand() < np.exp(-(c_new - best_c) / accept_temp):
                ones.remove(j); ones.add(i)
                zeros.remove(i); zeros.add(j)
            accept_temp *= 0.999
    
    return h, best_c
