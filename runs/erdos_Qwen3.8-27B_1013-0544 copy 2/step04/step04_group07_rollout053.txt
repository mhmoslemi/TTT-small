            cur = slsqp_optimize(cur, n_pts, target, beta, maxiter=min(5000, int(rem*80)), ftol=1e-18)
            update(cur)

        rem = budget_s - (time.time() - t0)
        if rem > 40:
            cur, _ = projected_gradient(cur, n_pts, target, 2000, 3000, t0, budget_s)
            update(cur)

        # Binary local search
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            hb, _ = binary_local_search(cur, target, 200, t0, budget_s, rng)
            if compute_c5(hb) < best_c5:
                cur = hb
            cur = slsqp_optimize(cur, n_pts, target, 1000, maxiter=3000, ftol=1e-18)
            update(cur)

        # Iterated local search: perturb and re-optimize
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            for noise in [0.15, 0.08, 0.04]:
                rem = budget_s - (time.time() - t0)
                if rem < 25:
                    break
                perturbed = project_h(cur + rng.standard_normal(n_pts) * noise, target)
                perturbed = slsqp_optimize(perturbed, n_pts, target, 500, maxiter=3000, ftol=1e-17)
                update(perturbed)

    # Final multi-resolution polish at best
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        nf = len(best_h)
        tf = nf / 2.0
        for beta in [5000, 10000]:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            polished = slsqp_optimize(best_h, nf, tf, beta, maxiter=8000, ftol=1e-19)
            update(polished)

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, _ = projected_gradient(best_h, nf, tf, 5000, 3000, t0, budget_s)
        update(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        nf = len(best_h)
        tf = nf / 2.0
        hb, _ = binary_local_search(best_h, tf, 100, t0, budget_s, rng)
        if compute_c5(hb) < best_c5:
            best_h = hb
        best_h = slsqp_optimize(best_h, nf, tf, 5000, maxiter=5000, ftol=1e-19)
        update(best_h)

    # Try a fresh binary initialization at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        nf = len(best_h)
        tf = nf / 2.0
        for attempt in range(5):
            rem = budget_s - (time.time() - t0)
            if rem < 10:
                break
            bh = gen_binary_init(nf, rng)
            bh, _ = binary_local_search(bh, tf, 100, t0, budget_s, rng)
            bh = slsqp_optimize(bh, nf, tf, 2000, maxiter=5000, ftol=1e-18)
            update(bh)

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, _ = projected_gradient(best_h, nf, tf, 5000, 2000, t0, budget_s)
        update(best_h)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
