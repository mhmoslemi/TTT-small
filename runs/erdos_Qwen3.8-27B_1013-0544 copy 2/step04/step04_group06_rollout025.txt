    h = np.clip(h, 0.0, 1.0)

        else:

def get_subgradient(h, n_active):
    """Compute subgradient of max correlation using top n_active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]

        else:

            g[:n - m2] += (1.0 - h[m2:])

Actually, let me think about this problem from a different angle. The correlation at lag k is:
corr_k = ∫₀^{2-k} h(x)(1-h(x+k)) dx (for 0 ≤ k ≤ 2)

And also for negative lags (which give the same values by symmetry of the full correlation).

The "full" correlation in numpy goes from lag -(n-1) to +(n-1), so we have 2n-1 values. But due to the specific structure, corr_{-k} and corr_k are related.

Actually, np.correlate(h, 1-h, mode="full")[j] = sum_i h[i] * (1-h[i + (j - (n-1))])
For j = n-1 (lag 0): sum_i h[i] * (1-h[i]) = sum h[i] - sum h[i]² 
For j > n-1 (positive lag k = j-(n-1)): sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
For j < n-1 (negative lag): sum_{i=0}^{n-1+k} h[i+k] * (1-h[i]) = sum_{i=k}^{n-1} h[i] * (1-h[i-k])

So corr_{-k} = sum_{i=k}^{n-1} h[i](1-h[i-k]) and corr_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k]).

These are not the same in general (unless h has special symmetry).

For the max, we need to consider all 2n-1 lags.

def project_sum(h, target):

def gradient_max_corr(h, n_active):
    """Compute subgradient of max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)

        if lag == 0:
            g += (1.0 - 2.0 * h)

        else:

            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]

    v = np.zeros(n)  # second moment
    b1, b2 = 0.9, 0.999

    eps = 1e-8
    no_improve = 0
    cool = 0

        if it % 1000 == 0 and it > 0:

                v = np.zeros(n)

        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g

        vhat = v / (1 - b2**(it+1))
        h_new = project_sum(h - lr * mhat / (np.sqrt(vhat) + eps), target)

            h = h_new  # still move for continuity

def slsqp_optimize(h_init, beta, maxiter=2000):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0, 1)] * n
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        bc = np.clip(beta * (corr - m), -300, 300)
        e = np.exp(bc)
        se = float(e.sum())
        return (m + np.log(se / len(corr)) / beta) * dx
    
    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        bc = np.clip(beta * (corr - m), -300, 300)
        e = np.exp(bc)
        se = float(e.sum())
        w = e / se
        g = np.zeros(n)
        nc = len(corr)
        for j in range(nc):

                g[:n - lag] -= w[j] * x[lag:]
            else:

                g[:n - m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n - m2]
        return g * dx

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

Actually, I can precompute the gradient as a sum over all lags:
g[i] = sum_k w_k * (contribution of h[i] to corr_k)

For a given i, h[i] appears in corr_k when:

- k > 0: h[i] appears as h[j+k] with j+k=i, i.e., j=i-k, contributing -h[i-k] if i-k >= 0

- k = 0: contributes (1-2h[i])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad(h, n_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g / len(top_idx), corr

def slsqp_opt(h_init, beta, maxiter=2000):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0, 1)] * n

    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        bc = np.clip(beta * (corr - m), -300, 300)
        e = np.exp(bc)
        se = float(e.sum())
        return (m + np.log(se / len(corr)) / beta) * dx

    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        bc = np.clip(beta * (corr - m), -300, 300)
        e = np.exp(bc)
        se = float(e.sum())
        w = e / se
        g = np.zeros(n)
        nc = len(corr)
        for j in range(nc):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - x[:-lag])
                g[:n - lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n - m2]
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)
