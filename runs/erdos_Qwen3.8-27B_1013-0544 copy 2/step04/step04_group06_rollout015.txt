    """One sweep of coordinate descent with exact 1D subproblem solving."""
    n = len(h)
    nc = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    idx_order = np.random.permutation(n)
    
    for j in idx_order:
        old_val = h[j]
        
        # Compute B[idx] = d(corr[idx])/d(h[j])
        B = np.zeros(nc)
        
        # Contribution from h[j] as the "h[i]" factor:
        # corr[idx] includes h[j] * (1 - h[j+k]) where k = idx - (n-1)
        # This is valid when 0 ≤ j < n (true) and 0 ≤ j+k < n
        # j+k = j + idx - (n-1), so 0 ≤ j + idx - (n-1) < n
        # → n-1-j ≤ idx ≤ n-1-j+n-1 = 2n-2-j
        ka_lo = n - 1 - j
        ka_hi = 2 * n - 2 - j
        ka_lo = max(ka_lo, 0)
        ka_hi = min(ka_hi, nc - 1)
        if ka_lo <= ka_hi:

            # So h[j+k] = h[0], h[1], ..., h[n-1] (but clipped to valid range)
            start = j + ka_lo - (n - 1)
            length = ka_hi - ka_lo + 1
            B[ka_lo:ka_hi+1] += (1.0 - h[start:start+length])

        # corr[idx] includes h[i] * (1 - h[j]) where i+k = j, i = j-k = j - idx + (n-1)

            # Let me just compute i for each idx

        # 1D minimization
        A = corr - old_val * B

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_grad(h, beta):
    """Objective and gradient for SLSQP."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-11:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_optimize(h_init, beta, maxiter=2000):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[0]
    
    def grad(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)
