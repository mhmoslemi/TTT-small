```python
def find_optimal_delta(corr, g, delta_lo, delta_hi, n_evals=50):
    """Minimize max_k(corr[k] + delta*g[k]) for delta in [delta_lo, delta_hi]"""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    a, b = delta_lo, delta_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + c * g)
    fd = np.max(corr + d * g)
    for _ in range(n_evals):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * g)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + d * g)
    return (a + b) / 2
```

```python
def pc_c5(x_switch, heights):
    """Compute C5 for piecewise-constant h"""
    M = len(heights)
    # x_switch has M+1 entries (including 0 and 2)
    # heights has M entries
    dx_segments = np.diff(x_switch)  # width of each segment
    # sum constraint: sum(heights * dx_segments) should = 1
    # C5 = max over shifts s in [0,2] of integral h(x)(1-h(x+s))dx
    # For piecewise constant, this is a piecewise-constant function of s
    # The "events" (where it changes) are at s = x_a - x_b for segments a,b
    # So we only need to evaluate at the midpoints between events
    
    # For each pair of segments (a, b), the overlap contribution is:
    # if segment a is at [x_a, x_a+w_a] and segment b is at [x_b, x_b+w_b]
    # then at shift s, segment a overlaps with segment b when:
    # x_b < x_a + s < x_b + w_b and x_a + s > x_b
    # i.e., x_b - x_a < s < x_b + w_b - x_a
    # The overlap length is: min(x_a+s, x_a+w_a) - max(x_a+s, x_b) ... 
    
    # Actually, let me just discretize the shift and compute directly
    n_shift = 200  # number of shift values to check
    shifts = np.linspace(0.001, 1.999, n_shift)  # shifts in (0, 2)
    # For each shift s, compute the overlap integral
    # overlap(s) = integral h(x) * (1-h(x+s)) dx over x where both x and x+s are in [0,2]
    # = integral_{max(0, -s)}^{min(2, 2-s)} h(x) * (1-h(x+s)) dx
    
    # For piecewise constant:
    # split [0, min(2,2-s)] into subintervals where h(x) and h(x+s) are both constant
    # The breakpoints are the x_switch values and x_switch - s values
    
    c5 = 0
    for s in shifts:
        lo = 0.0
        hi = min(2.0, 2.0 - s)
        if hi <= lo:
            continue
        # breakpoints
        bps = np.unique(np.concatenate([x_switch, x_switch - s, [lo, hi]]))
        bps = bps[(bps >= lo - 1e-12) & (bps <= hi + 1e-12)]
        bps = np.clip(bps, lo, hi)
        total = 0.0
        for i in range(len(bps)-1):
            mid = (bps[i] + bps[i+1]) / 2
            # find which segment mid is in
            seg_x = np.searchsorted(x_switch, mid, side='right') - 1
            seg_x = np.clip(seg_x, 0, M-1)
            # find which segment mid+s is in
            mid_s = mid + s
            seg_s = np.searchsorted(x_switch, mid_s, side='right') - 1
            seg_s = np.clip(seg_s, 0, M-1)
            total += heights[seg_x] * (1 - heights[seg_s]) * (bps[i+1] - bps[i])
        c5 = max(c5, total)
    return c5
```

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

def pc_overlap(xsw, heights, s):

    M = len(heights)

    lo = max(0.0, -s)
    hi = min(2.0, 2.0 - s)

    bps = np.unique(np.sort(np.concatenate([xsw, xsw - s, [lo, hi]])))

    bounds = []
    for i in range(M-1):
        bounds.append((0.01, 1.99))  # switching points (will be project to ordered)
    for i in range(M-1):
        bounds.append((0.01, 0.99))  # free heights
    
    n_params = 2*M - 2
    
    # Initial guess
    if x0_sw is not None:
        x0 = np.concatenate([x0_sw, x0_ht])
    else:
        # Equal-width segments, alternating heights
        x0 = np.zeros(n_params)
        for i in range(M-1):
            x0[i] = (i+1) * 2.0 / M  # equal width switching
        for i in range(M-1):
            x0[M-1+i] = 1.0 if i % 2 == 0 else 0.0  # alternating
        # Adjust to satisfy bounds
        x0 = np.clip(x0, [b[0] for b in bounds], [b[1] for b in bounds])
    
    # Need to handle the ordering constraint on switching points
    # Use a reparameterization: sw_i = cumsum of positive variables
    # Or just use penalty
    
    # Actually, let's use a simpler approach: optimize with Nelder-Mead
    # which doesn't need gradients or bounds (we just restart if infeasible)
    
    from scipy.optimize import minimize as sp_min
    
    best_params = x0.copy()
    best_val = obj(x0)
    
    for attempt in range(5):
        if time_fn() < 10:
            break
        res = sp_min(obj, best_params, method='Nelder-Mead',
                     options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
        # Check feasibility
        sw = res.x[:M-1]
        hts_free = res.x[M-1:2*M-2]
        xsw = np.concatenate([[0.0], sw, [2.0]])
        widths = np.diff(xsw)
        if np.all(np.diff(sw) > -1e-6) and np.all(sw > 0) and np.all(sw < 2):
            remaining = 1.0 - np.sum(hts_free * widths[:M-1])
            h_last = remaining / widths[M-1]
            if 0 <= h_last <= 1 and np.all(hts_free >= -0.01) and np.all(hts_free <= 1.01):
                val = res.fun
                if val < best_val:
                    best_val = val
                    best_params = res.x.copy()
        # Perturb for next attempt
        best_params = best_params + np.random.randn(n_params) * 0.1
        best_params = np.clip(best_params, [b[0] for b in bounds], [b[1] for b in bounds])
    
    return best_val, best_params

def pc_to_array(params, M, n):
    """Convert optimized params to n-point array."""
    sw = params[:M-1]
    hts_free = params[M-1:2*M-2]
    xsw = np.concatenate([[0.0], sw, [2.0]])
    widths = np.diff(xsw)
    remaining = 1.0 - np.sum(hts_free * widths[:M-1])
    h_last = remaining / widths[M-1]
    heights = np.concatenate([hts_free, [h_last]])
    h = pc_h_to_values(xsw, heights, n)
    return h

def smooth_max_val_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    # Gradient (only for active correlations)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def proj_grad_descent(h, target, n_iter, time_fn, betas):
    """Projected gradient descent with adaptive smooth max."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    for beta in betas:
        for _beta_iter in range(n_iter // len(betas)):
            if time_fn() < 2:
                return best, best_c5
            val, grad = smooth_max_val_and_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            # Backtracking line search
            step = 0.1
            improved = False
            for _ in range(20):
                h_new = project_h(h - step * grad, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    if c5_new < best_c5:
                        best = h_new.copy()
                        best_c5 = c5_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                no_imp += 1
                if no_imp > 50:
                    break
            else:
                no_imp = 0
    return best, best_c5
