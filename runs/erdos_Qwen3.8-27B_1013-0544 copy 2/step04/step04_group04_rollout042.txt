    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_v = np.exp(shifted)
    sum_e = np.sum(exp_v)
    w = exp_v / sum_e
    val = (max_c + np.log(sum_e) / beta) * dx
    s = h.sum() - n / 2.0
    val += penalty * s * s

            m = -lag
            grad[:n - m] += wk * (1.0 - h[m:])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    xc = np.clip(x, 0, 1)
    corr = np.correlate(xc, 1.0 - xc, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_v = np.exp(shifted)
    sum_e = np.sum(exp_v)
    w = exp_v / sum_e
    val = (max_c + np.log(sum_e) / beta) * dx
    s = xc.sum() - n / 2.0
    val += penalty * s * s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k] * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * xc)
        elif 0 < lag:
            grad[lag:] += wk * (1.0 - xc[:-lag])
            grad[:n - lag] -= wk * xc[lag:]
        else:
            m = -lag
            grad[:n - m] += wk * (1.0 - xc[m:])
            grad[m:] -= wk * xc[:n - m]
    grad += 2.0 * penalty * s
    return val, grad

def _lbfgs(h, deadline, betas, min_left=20):
    n = len(h)
    bc5 = compute_c5(h)
    for beta, pen, mit in betas:
        if time.time() > deadline - min_left:
            break
        try:
            r = minimize(_obj, h, method='L-BFGS-B', args=(beta, pen),
                        bounds=[(0, 1)] * n,
                        options={'maxiter': mit, 'ftol': 1e-18, 'maxls': 100})
            h2 = project_h(r.x, n / 2.0)
            c2 = compute_c5(h2)
            if c2 < bc5 - 1e-15:
                h, bc5 = h2, c2
        except Exception:
            pass
    return h, bc5

def _subgrad(h, deadline, n_iter, top_k=10):
    n = len(h)
    tgt = n / 2.0
    bc5 = compute_c5(h)
    steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    ni = 0
    for it in range(n_iter):
        if time.time() > deadline - 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(3, n // 30))
        ti = np.argsort(corr)[::-1][:nk]
        g = np.zeros(n)
        for idx in ti:
            lag = idx - (n - 1)
            if lag == 0:
                g += 1.0 - 2.0 * h
            elif 0 < lag:
                g[lag:] += 1.0 - h[:-lag]
                g[:n - lag] -= h[lag:]
            else:
                m = -lag
                g[:n - m] += 1.0 - h[m:]
                g[m:] -= h[:n - m]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        fnd = False
        for st in steps:
            ht = project_h(h - st * g, tgt)
            c = compute_c5(ht)
            if c < bc5 - 1e-14:
                h, bc5, ni = ht, c, 0
                fnd = True
                break
        if not fnd:
            ni += 1
            if ni > 400:
                sc = min(0.05, 0.004 * (1 + ni / 400))

                if compute_c5(ht) < bc5:

            if ni > 2500:

    return h, bc5
