import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def focused_subgrad(h, target, n, n_iter, t0, budget, top_k=8):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    step = 0.01
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        for _ in range(10):
            trial = project_h(cur - step * grad, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                stall = 0
                break
            step *= 0.7
        else:
            stall += 1
            step = max(step * 0.5, 0.001)
            if step < 0.0005:
                step = 0.02
            trial = project_h(cur - step * grad, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                stall = 0
        if stall > 200:
            break
    return cur, best_c5

def anneal_binary(h_bin, n, t0, budget, n_iter=80000, seed=42):
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    cur_max = float(np.max(compute_lags(h)))
    best_h = h.copy()
    best_max = cur_max
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_max = float(np.max(compute_lags(h)))
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = 0.15 * (0.005) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_max = new_max
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_max

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                if cur[j] + d < -0.01 or cur[j] + d > 1.01:

                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

        c_try = compute_c5(cur_alt)
        if c_try < best_c5:
            best_h = cur_alt.copy()
            best_c5 = c_try
            best_n = n_alt
            n = n_alt
            target = target_alt

    # Phase 7: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h = do_slsqp(best_h, n, target, 100000, maxiter=10000, ftol=1e-18)
        best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = focused_subgrad(best_h, target, n, 8000, t0, budget_s, top_k=5)

    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        best_h, best_c5 = coord_descent(best_h, target, n, 100, t0, budget_s)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
