    dx = 2.0 / n

    # d(corr_k)/dh[i] for lag s = k-(n-1):
    #   = (1-h[i+s])*[i+s<n] - h[i-s]*[i-s>=0]
    
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n-s]
        else:
            la = -s
            grad[la:] += wk * (1.0 - h[:n-la])
            grad[:n-la] -= wk * h[la:]
    return grad

        wi = w[idx]*dx
        if k >= 0:
            g[:n-k] += wi*(1.0-h[k:])
            g[k:] -= wi*h[:n-k]
        else:
            la = -k
            g[la:] += wi*(1.0-h[:n-la])
            g[:n-la] -= wi*h[la:]
    return g

def slsqp_h(h0, n, target, beta, maxiter=10000):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type':'eq', 'fun': lambda x: x.sum()-target, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(smooth_max_val, h0, args=(beta,n), jac=lambda x: smooth_max_grad(x,beta,n),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter':maxiter, 'ftol':1e-18})
        return project_h(r.x, target)
    except:
        return h0

def penalty_val(h, t0, lam, n):
    c = corr_full(h)*(2.0/n)
    ex = np.maximum(c - t0, 0)
    return float(np.sum(ex**2) + lam*(h.sum()-n/2.0)**2)
