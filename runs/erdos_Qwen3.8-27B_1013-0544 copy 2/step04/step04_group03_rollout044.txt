import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def get_worst_lags(h, top_k=8):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    return idx

def grad_lag(h, lag_idx, n):
    lag = int(lag_idx) - (n - 1)
    n_cent = n - abs(lag)
    if lag >= 0:
        out = np.zeros(n)
        out[lag:n] += (1.0 - h[:n_cent])
        out[lag:n] -= h[lag:n]
        return out
    else:
        la = -lag
        out = np.zeros(n)
        out[:n_cent] += (1.0 - h[la:])
        out[:n_cent] -= h[:n_cent]
        return out

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_fn(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.8
            mom *= 0.3
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                cur = trial
                best_c5 = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return cur, best_c5

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.03, -0.03, 0.015, -0.015, 0.008, -0.008]
    for sw in range(max_sweeps):
        rem = budget - (time.time() - t0)
        if rem < 2:
            break
        improved = False
        for j in range(n):
            if rem < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def structured_binary(n, seed):
    rng = np.random.default_rng(seed)
    pattern = rng.random(n)
    return binary_snap(pattern + 0.01 * np.sin(np.linspace(0, 4*np.pi, n)), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = 0.5 + 0.3 * np.sin(np.linspace(0, 3*np.pi, n_init))
        h_parent = project_h(h_parent, n_init / 2.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Phase 1: Binary optimization at multiple resolutions
    for n1 in [192, 256]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        rng = np.random.default_rng(seed + n1)
        parent_res = project_h(resample_fn(h_parent, n1), n1 / 2.0)
        starts = [
            binary_snap(parent_res, n1),
            structured_binary(n1, seed + n1 + 1),
        ]
        # Add a few more random starts for diversity
        for s_idx in range(3):
            starts.append(binary_snap(0.5 + rng.random(n1) * 0.3, n1))

        for s_i, start in enumerate(starts):
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            time_alloc = min(rem * 0.4 / max(len(starts) - s_i, 1), 120)
            h_b, b_max = block_anneal(start, n1, t0, t0 + time_alloc,
                                       n_iter=int(time_alloc * 3000), rng=np.random.default_rng(seed + n1 + s_i))
            h_b = project_h(h_b, n1 / 2.0)
            c_try = compute_c5(h_b)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_b.copy()
                best_n = n1

    # Phase 2: Continuous refinement at best resolution
    n_cur = best_n
    target = n_cur / 2.0
    h_cur = best_h + 0.03 * np.random.randn(n_cur)
    h_cur = project_h(h_cur, target)

    for beta in [100, 400, 1500, 5000, 15000]:
        rem = budget_s - (time.time() - t0)
        if rem < 12:
            break
        h_cur = do_slsqp(h_cur, n_cur, target, beta, maxiter=10000)
        c_try = compute_c5(h_cur)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_cur.copy()

    # Phase 3: Focused subgradient descent
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_cur, c_try = focused_subgrad(best_h, target, n_cur, 30000, t0, budget_s, top_k=16)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_cur.copy()

    # Phase 4: SLSQP after subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        for beta in [5000, 20000, 60000]:
            rem = budget_s - (time.time() - t0)
            if rem < 10:
                break
            best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=8000)
            best_c5 = compute_c5(best_h)

    # Phase 5: Try going to n=384 if parent was better
    rem = budget_s - (time.time() - t0)
    if rem > 50 and best_n < 384:
        n2 = 384
        h2 = project_h(resample_fn(best_h, n2), n2 / 2.0)
        for beta in [400, 2000, 8000]:
            rem = budget_s - (time.time() - t0)
            if rem < 10:
                break
            h2 = do_slsqp(h2, n2, n2 / 2.0, beta, maxiter=6000)
            c_try = compute_c5(h2)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h2.copy()
                best_n = n2
                n_cur = n2
                target = n2 / 2.0

    # Phase 6: Multi-perturbation restarts
    rem = budget_s - (time.time() - t0)
    rng2 = np.random.default_rng(seed + 999)
    for att in range(8):
        if rem < 15:
            break
        scale = 0.02 * (1.0 + 0.5 * att)
        hp = best_h + rng2.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp = do_slsqp(hp, n_cur, target, 1000, maxiter=4000)
        hp, c_hp = focused_subgrad(hp, target, n_cur, 5000, t0, budget_s, top_k=10)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = hp.copy()

    # Phase 7: Final coordinate descent polish
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 400, t0, budget_s)

    # Phase 8: Ultra-fine SLSQP
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h = do_slsqp(best_h, n_cur, target, 50000, maxiter=12000)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 300, t0, budget_s)

    return best_h, best_c5, best_n
