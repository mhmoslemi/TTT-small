def slsqp_run(h0, n, beta, maxiter=3000):
    target = n / 2.0
    bounds = [(0, 1)] * n

        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 500)
        return (m + np.log(np.exp(shifted).sum()) / beta) * (2.0 / n)
    
    def grad(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(shifted)
        w = e / e.sum()
        g = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * xc)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - xc[:-lag])
                g[:n-lag] -= w[j] * xc[lag:]
            else:
                ml = -lag
                g[:n-ml] += w[j] * (1.0 - xc[ml:])
                g[ml:] -= w[j] * xc[:n-ml]
        return g * (2.0 / n)
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq, 'jac': eqj}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(np.clip(res.x, 0, 1), target)
