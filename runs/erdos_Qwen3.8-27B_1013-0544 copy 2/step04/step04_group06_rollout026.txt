def softmax_obj_grad(h, beta, active_lags=None):
    """Compute softmax-approximated max and its gradient.
    If active_lags is given, only use those lags for the max."""
    n = len(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if active_lags is not None:
        corr_used = corr[active_lags]
    else:
        corr_used = corr
    
    m = np.max(corr_used)
    # Smooth max approximation
    bc = np.clip(beta * (corr_used - m), -500, 0)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se) / beta) * dx  # Note: log(sum(exp)) not log(sum(exp)/len)
    # Wait, the standard log-sum-exp: log(sum(exp(x_i))) ≈ max(x_i) + log(sum(exp(x_i - max)))
    # So smooth_max ≈ max + log(sum(exp(x_i - max))) / beta * beta... 
    # Let me redo: smooth_max(beta * x) / beta = (max + log(sum(exp(beta*(x_i - max)))))/beta
    # = max/beta + log(sum(exp(beta*(x_i - max))))/beta
    # Hmm, that's not right either.
    # 
    # The smoothed max: (1/beta) * log(sum(exp(beta * x_i)))
    # = (1/beta) * (beta * m + log(sum(exp(beta * (x_i - m)))))
    # = m + (1/beta) * log(sum(exp(beta * (x_i - m))))
    #
    # So val should be: (m + log(se) / beta) * dx
    # where se = sum(exp(beta * (corr_used - m)))
    
    # Gradient
    w = e / se
    g = np.zeros(n)
    
    if active_lags is not None:
        for idx, j in zip(active_lags, range(len(active_lags))):
            if w[j] < 1e-12:
                continue
            lag = idx - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - h[:-lag])
                g[:n - lag] -= w[j] * h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - h[m2:])
                g[m2:] -= w[j] * h[:n - m2]
    else:
        for j in range(nc):
            if w[j] < 1e-12:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - h[:-lag])
                g[:n - lag] -= w[j] * h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - h[m2:])
                g[m2:] -= w[j] * h[:n - m2]
    
    return val, g * dx

def slsqp_step(h, n, beta, maxiter, active_lags=None):
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        x = np.clip(x, 0, 1)
        return softmax_obj_grad(x, beta, active_lags)[0]
    
    def grad(x):
        x = np.clip(x, 0, 1)
        return softmax_obj_grad(x, beta, active_lags)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n_active, max_iter, t0, budget, use_bb=True):
    """Subgradient method with optional Barzilai-Borwein step sizes."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    prev_h = best.copy()
    prev_g = None
    bb_alpha = 0.01  # initial step
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        # Use symmetry: only need to check lags 0 to n-1 (indices n-1 to 2n-2)
        # But for the gradient, we need all active lags
        # Find top lags from the full correlation
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Determine step size
        if use_bb and prev_g is not None:
            s = best - prev_h
            y = g - prev_g
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-10 and sy > 1e-10:
                bb_alpha = sy / yy
                bb_alpha = np.clip(bb_alpha, 1e-5, 0.1)
        
        improved = False
        steps = [bb_alpha * 2, bb_alpha, bb_alpha * 0.5, bb_alpha * 0.25, 0.005, 0.001, 0.0003]
        for step in steps:
            if step <= 0:
                continue
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        
        prev_h = best.copy()
        prev_g = g.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 200 and it % 100 == 99:
                pert = np.random.randn(n) * 0.005
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 5000:
                break
        else:
            no_improve = 0
    
    return best, best_c5

        best_h = np.full(480, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    # Phase 1: SLSQP warmup with increasing beta
    for beta in [50, 200, 800, 3000]:
        if tl() < 120:
            break
        h_new = slsqp_step(best_h, n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, target, 12, 400, t0, budget_s)
    # Phase 2: Active-set SLSQP focusing on critical lags
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    top_lags = np.argsort(corr)[::-1][:30]
    for beta in [1000, 4000, 12000]:
        if tl() < 180:
            break
        h_new = slsqp_step(best_h, n, beta, maxiter=2000, active_lags=top_lags)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_lags = np.argsort(corr)[::-1][:30]
        best_h, best_c5 = subgrad_polish(best_h, target, 20, 600, t0, budget_s)
    # Phase 3: Intensive subgradient with large active set and BB steps
    best_h, best_c5 = subgrad_polish(best_h, target, 30, 8000, t0, budget_s, use_bb=True)
    # Phase 4: Multi-start with diverse perturbations
    for restart in range(6):
        if tl() < 130:
            break
        if restart == 0:
            h_try = best_h.copy()
        elif restart == 1:
            h_try = project_sum(best_h + np.random.randn(n) * 0.025, target)
        elif restart == 2:
            h_try = best_h.copy()
            blk = n // 5
            st = np.random.randint(0, n - blk)
            h_try[st:st+blk] = 1.0 - h_try[st:st+blk]
            h_try = project_sum(h_try, target)
        elif restart == 3:
            half = n // 2
            h_try = best_h.copy()
            h_try[:half] += 0.04 * np.random.randn(half)
            h_try[half:] -= 0.04 * np.random.randn(half)
            h_try = project_sum(h_try, target)
        elif restart == 4:
            freqs = np.abs(np.fft.rfftfreq(n) * n)
            mask = (freqs > 2) & (freqs < n // 4)
            fourier = np.fft.rfft(best_h)
            fourier[mask] += np.random.randn(mask.sum()) * 0.015
            h_try = project_sum(np.fft.irfft(fourier, n), target)
        else:
            h_try = project_sum(np.roll(best_h, np.random.randint(1, n//20)), target)
        for beta in [800, 6000]:
            if tl() < 60:
                break
            h_new = slsqp_step(h_try, n, beta, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new
    # Phase 5: Final high-precision SLSQP + subgradient
    if tl() > 60:
        h_new = slsqp_step(best_h, n, 40000, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    best_h, best_c5 = subgrad_polish(best_h, target, 25, 10000, t0, budget_s, use_bb=True)
    # Phase 6: Micro-polish with very small steps
    for it in range(20000):
        if tl() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[lag:] += (1.0 - best_h[:-lag])
                g[:n - lag] -= best_h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best_h[m2:])
                g[m2:] -= best_h[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for step in [0.003, 0.001, 0.0004, 0.00015, 0.00005, 0.00002]:
            ht = project_sum(best_h - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                improved = True
                break
        if not improved and it % 800 == 0:
            ht = project_sum(best_h + np.random.randn(n) * 0.001, target)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h = ht
                best_c5 = c5_t if False else c5t
    return best_h, best_c5, n
