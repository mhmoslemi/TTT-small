def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            if tot < 1e-14:

    sh = alpha * (c - m)
    mx = sh.max()

        d = rng.uniform(-0.03, 0.03)
        hi = cur_h[i] + d
        hj = cur_h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cur_h[i], cur_h[j] = hi, hj
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = hi - d, hj + d
    return best_h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=3000, ftol=1e-14)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    
    if time.time() < deadline - 5:
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                                          [0.01, 0.005, 0.002, 0.001])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                                          [0.01, 0.005, 0.002], radius=40)
    
    if time.time() < deadline - 8:
        best_h, best_c = simulated_anneal(best_h, n, min(deadline - 5, time.time() + 30), rng)
    
    if time.time() < deadline - 3:
        ht = slsqp_optimize(best_h, n, 500000, maxiter=5000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    
    if time.time() < deadline - 1:
        if n <= 120:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                          [0.002, 0.001, 0.0005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                          [0.002, 0.001], radius=30)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
