def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_sweep(h, corr, n_bisect=35, rng=None):
    n = len(h)
    size = 2 * n - 1
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    k_range = np.arange(-(n - 1), n)
    for i in order:
        i = int(i)
        v = h[i]
        idx_r = i + k_range
        idx_l = i - k_range
        valid_r = (idx_r >= 0) & (idx_r < n)
        valid_l = (idx_l >= 0) & (idx_l < n)
        hr = np.zeros(size)
        hl = np.zeros(size)
        if valid_r.any():
            hr[valid_r] = h[idx_r[valid_r]]
        if valid_l.any():
            hl[valid_l] = h[idx_l[valid_l]]
        a = 1.0 - hr - hl
        b = corr - a * v
        lo, hi = 0.0, 1.0

                break
