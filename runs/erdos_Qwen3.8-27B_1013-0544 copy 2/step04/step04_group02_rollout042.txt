def fourier_h(coeffs, x):
    """Evaluate h(x) from Fourier coefficients.
    coeffs = [a1, b1, a2, b2, ..., aK, bK]
    h(x) = 0.5 + sum_k [ak*cos(pi*k*x) + bk*sin(pi*k*x)]
    """
    n = len(x)
    K = len(coeffs) // 2
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        h += coeffs[2*(k-1)] * np.cos(np.pi * k * x)
        h += coeffs[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def fourier_c5(coeffs, n_eval):

    h = fourier_h(coeffs, x)

        return 1.0  # penalty
    h = np.clip(h, 0, 1)
    # Re-normalize
    target = n_eval / 2.0
    h = project_feasible(h, target)
    return compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_fourier(coeffs, x):
    K = len(coeffs) // 2
    h = np.full(len(x), 0.5)
    for k in range(1, K + 1):
        h += coeffs[2*(k-1)] * np.cos(np.pi * k * x)
        h += coeffs[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def fourier_obj(coeffs, n_eval, beta, lam):
    x = np.linspace(0, 2, n_eval, endpoint=False)
    h = eval_fourier(coeffs, x)
    # Penalty for bound violations
    pen = 0.0
    under = np.clip(-h, 0, None)
    over = np.clip(h - 1, 0, None)
    pen = lam * (np.sum(under**2) + np.sum(over**2))
    if pen > 10:
        return pen
    h = np.clip(h, 0, 1)
    target = n_eval / 2.0
    h = project_feasible(h, target)
    corr = compute_corr(h)
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0/n_eval) + pen

def fourier_optimize(start_coeffs, K, n_eval, deadline, seed):
    rng = np.random.RandomState(seed)
    n_par = 2 * K
    bounds = Bounds(-0.5, 0.5)
    best_coeffs = start_coeffs.copy()
    best_val = fourier_obj(start_coeffs, n_eval, 100, 10)
    
    for attempt in range(5):
        if time.time() > deadline:
            break
        x0 = start_coeffs.copy()
        if attempt > 0:
            x0 += rng.randn(n_par) * 0.05 * (attempt / 5.0)
            x0 = np.clip(x0, -0.5, 0.5)
        
        for beta in [50, 200, 1000, 5000]:
            if time.time() > deadline:
                break
            try:
                res = minimize(fourier_obj, x0, args=(n_eval, beta, 10),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-14})
                val = res.fun
                if val < best_val:
                    best_val = val
                    best_coeffs = res.x.copy()
                    x0 = res.x.copy()
            except Exception:
                pass
    
    return best_coeffs, best_val

def grad_at_lag(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        L = lag
        if L > 0:
            m1 = np.arange(L, n)
            g[m1] = (1.0 - h[m1 - L])
        m2 = np.arange(0, n - L)
        g[m2] += (-h[m2 + L])
    else:
        J = -lag
        m1 = np.arange(0, n - J)
        g[m1] = (1.0 - h[m1 + J])
        m2 = np.arange(J, n)
        g[m2] += (-h[m2 - J])
    return g
