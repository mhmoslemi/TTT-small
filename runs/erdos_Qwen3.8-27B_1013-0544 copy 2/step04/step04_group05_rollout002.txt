def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgrad_opt(h, target, time_fn, max_iters, n_lags,
                step_init, step_min, step_decay, decay_every,
                momentum, perturb_every, perturb_scale):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    step = step_init
    for it in range(max_iters):
        if time_fn() < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(n_lags, max(5, n // 8))
        top = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top]
        g = compute_subgrad(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        d = -vel / vn
        found = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= decay_every:
                step = max(step_min, step * step_decay)
                no_improve = 0
            if no_improve >= 900 and it % perturb_every == 0:
                for scale in [perturb_scale, perturb_scale * 0.4]:
                    p = np.random.randn(n) * scale
                    ht = project_h(best + p, target)
                    c5t = compute_c5(ht)
                    if c5t < best_c5:
                        best = ht
                        best_c5 = c5t
                        vel *= 0.3
                        no_improve = 0
                        step = step_init
                        break
    return best, best_c5

def slsqp_polish(h, target, dx, betas, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 10:
            break
        try:
            res = minimize(_slsqp_objective, best, args=(beta, dx),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_constraint,
                                         'args': (target,), 'jac': _slsqp_jac}],
                           options={'maxiter': max_iter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def _slsqp_objective(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def _slsqp_constraint(h_in, target):
    return h_in.sum() - target

def _slsqp_jac(h_in):
    return np.ones(len(h_in))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(400)
        best_h[::2] = 1.0

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0

    # Phase 1: Brief SLSQP polish
    if rem() > 60:
        best_h, best_c5 = slsqp_polish(best_h, target0, dx0,
                                       [200, 800, 3000], 4000, rem)

    # Phase 2: Main subgradient - moderate lags, fast steps
    best_h, best_c5 = subgrad_opt(
        best_h, target0, rem,
        max_iters=120000, n_lags=40,
        step_init=0.015, step_min=0.00003, step_decay=0.85, decay_every=120,
        momentum=0.86, perturb_every=350, perturb_scale=0.008
    )

    # Phase 3: Second pass - more lags, slower momentum
    if rem() > 80:
        best_h, best_c5 = subgrad_opt(
            best_h, target0, rem,
            max_iters=100000, n_lags=60,
            step_init=0.008, step_min=0.00002, step_decay=0.88, decay_every=180,
            momentum=0.90, perturb_every=450, perturb_scale=0.005
        )

    # Phase 4: Third pass - many lags, high momentum, tiny steps
    if rem() > 60:
        best_h, best_c5 = subgrad_opt(
            best_h, target0, rem,
            max_iters=80000, n_lags=80,
            step_init=0.004, step_min=0.000008, step_decay=0.92, decay_every=250,
            momentum=0.93, perturb_every=500, perturb_scale=0.003
        )

    # Phase 5: SLSQP burst at high beta
    if rem() > 40:
        best_h, best_c5 = slsqp_polish(best_h, target0, dx0,
                                       [2000, 8000], 5000, rem)

    # Phase 6: Try n=600 if beneficial
    n_cur = len(best_h)
    if rem() > 150 and n_cur <= 400:
        n2 = 600
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        if rem() > 60:
            h2, c5_2 = slsqp_polish(h2, target2, dx2, [500, 2000], 4000, rem)
        if rem() > 50:
            h2, c5_2 = subgrad_opt(
                h2, target2, rem,
                max_iters=60000, n_lags=80,
                step_init=0.006, step_min=0.00002, step_decay=0.88, decay_every=150,
                momentum=0.91, perturb_every=400, perturb_scale=0.004
            )
        if rem() > 30:
            h2, c5_2 = subgrad_opt(
                h2, target2, rem,
                max_iters=40000, n_lags=100,
                step_init=0.003, step_min=0.00001, step_decay=0.92, decay_every=250,
                momentum=0.94, perturb_every=500, perturb_scale=0.002
            )
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 7: Final ultra-fine polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    if rem() > 10:
        best_h, best_c5 = subgrad_opt(
            best_h, target_f, rem,
            max_iters=30000, n_lags=80,
            step_init=0.002, step_min=0.000003, step_decay=0.94, decay_every=300,
            momentum=0.94, perturb_every=600, perturb_scale=0.0015
        )

    return best_h, best_c5, len(best_h)
