        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += (1.0 - h[m2:])
        g[m2:] -= h[:n - m2]
    return g

def top_k_lags(h, k):
    """Return indices of top-k correlation lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    return np.argsort(corr)[-k:] - (n - 1)

def obj_lse(h, beta, top_lags):
    """Log-sum-exp of top lags, approximating max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Use all lags for LSE (stable computation)
    bc = beta * corr
    bc -= np.max(bc)
    lse = np.max(corr) + np.log(np.sum(np.exp(bc))) / beta
    return lse * dx, corr

def grad_lse(h, beta, top_lags):
    """Gradient of the LSE objective (only for top lags)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    
    # Compute weights for top lags
    bc = beta * corr[top_lags]
    bc -= bc.max()
    e = np.exp(bc)
    # Approximate: assume only top lags contribute significantly
    se_total = np.exp(beta * (np.max(corr) - np.max(corr)))  # =1 for max
    # Better: compute full LSE denominator approximately
    all_bc = beta * corr
    all_max = np.max(all_bc)
    se_full = np.sum(np.exp(all_bc - all_max))
    
    w = e / np.sum(e)  # normalized weights among top lags
    scale = np.exp(all_max - all_max) / se_full  # = 1/se_full... 
    # Actually: LSE = max(corr) + log(sum(exp(beta*(corr-max))))/beta
    # d/dh LSE = sum_k exp(beta*(c_k-max))/(sum exp(beta*(c_k-max))) * dc_k/dh
    #           = sum_k w_k * dc_k/dh  where w_k = exp(beta*(c_k-max))/sum(exp(...))
    
    # For the full LSE:
    w_full = np.exp(all_bc - all_max) / se_full
    # But we only compute gradients for top_lags
    # Approximate: w_full for non-top lags is ~0 when beta is large
    
    g = np.zeros(n)
    for j in range(len(top_lags)):
        wl = w_full[top_lags[j]]
        if wl < 1e-10:
            continue
        g += wl * lag_gradient(h, top_lags[j])
    
    return g * dx

def lse_optimize(h_init, beta, n_top=10, maxiter=200):
    """Optimize LSE objective using L-BFGS-B with augmented Lagrangian."""
    n = len(h_init)
    target = n / 2.0
    
    # Precompute which lags are top
    top = top_k_lags(h_init, n_top)
    
    lam = 0.0
    rho = 10.0
    x = h_init.copy()
    
    for aug_it in range(10):
        def fun_and_grad(x):
            h = np.clip(x, 0, 1)
            val, corr = obj_lse(h, beta, top)
            g = grad_lse(h, beta, top)
            s = x.sum() - target
            # Penalize out-of-bounds
            g = g + lam + 2.0 * rho * s  # (this broadcasts)
            val_pen = val + lam * s + rho * s * s
            return val_pen, g
        
        res = minimize(fun_and_grad, x, jac=True, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': maxiter, 'gtol': 1e-10})
        x = res.x
        s = x.sum() - target
        if abs(s) < 1e-8:
            break
        lam = lam + rho * s
        if abs(s) > 0.5:
            rho = min(rho * 2, 10000)
        x = project_sum(x, target)
    
    x = project_sum(x, target)
    return x

def subgrad_momentum(h, target, n_active=8, max_iter=5000, t0=None, budget=0):
    """Subgradient with momentum and adaptive step."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)  # momentum
    momentum = 0.9
    step = 0.03
    no_improve = 0
    it = 0
    
    while it < max_iter:
        if t0 is not None and budget - (time.time() - t0) < 3:
            break
        it += 1
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[-n_active:]
        
        g = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            g += lag_gradient(best, lag)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Momentum update
        v = momentum * v - step * g
        v = momentum * v  # Apply momentum to the velocity
        
        # Actually, let's use a simpler Polyak-type step
        # step_k = (C5_target - c5_prev) / ||g|| ... but we don't know C5_target
        # Use diminishing step
        step_k = step / (1 + 0.001 * it)
        
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.04]:
            s = step_k * mult
            ht = project_sum(best - s * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 300 and it % 100 == 50:
                pert = np.random.randn(n) * (0.01 + 0.01 * no_improve / 300)
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 5000:
                break
        else:
            no_improve = 0
    
    return best, best_c5
