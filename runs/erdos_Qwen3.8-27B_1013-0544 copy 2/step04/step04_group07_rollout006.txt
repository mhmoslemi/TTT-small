import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def de_optimize(h0, n, target, t0, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    pop_size = min(max(n // 3, 40), 120)
    pop = np.zeros((pop_size, n))
    pop[0] = h0
    for i in range(1, pop_size):
        if i < pop_size // 2:
            scale = 0.03 + 0.2 * (i / (pop_size // 2 + 1.0))
            pop[i] = project_h(h0 + scale * (rng.rand(n) - 0.5) * 2, target)
        else:
            pop[i] = project_h(0.5 + 0.45 * (rng.rand(n) - 0.5) * 2, target)
    fitness = np.array([compute_c5(p) for p in pop])
    best_idx = int(np.argmin(fitness))
    best_fit = fitness[best_idx]
    no_improve = 0
    gen = 0
    while time.time() - t0 < time_limit:
        gen += 1
        F = max(0.25, 0.8 * (0.995 ** gen))
        gen_imp = False
        active_range = min(pop_size, max(10, pop_size // 2))
        for i in range(active_range):
            if time.time() - t0 > time_limit:
                break
            r = rng.rand()
            cands = np.array([j for j in range(pop_size) if j != i])
            if r < 0.35 and best_idx != i:
                bc = np.array([j for j in cands if j != best_idx])
                a, b = rng.choice(bc, 2, replace=False)
                v = pop[best_idx] + F * (pop[a] - pop[b])
            elif r < 0.75:
                a, b, c = rng.choice(cands, 3, replace=False)
                v = pop[a] + F * (pop[b] - pop[c])
            else:
                v = pop[i] + F * 0.6 * (rng.rand(n) - 0.5)
            cr = rng.uniform(0.5, 0.9)
            jr = rng.randint(n)
            mask = rng.rand(n) < cr
            mask[jr] = True
            trial = np.where(mask, v, pop[i])
            trial = project_h(trial, target)
            fit = compute_c5(trial)
            if fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = fit
                if fit < best_fit - 1e-15:
                    best_fit = fit
                    best_idx = i
                    gen_imp = True
        if gen_imp:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                for _ in range(4):
                    idx = rng.randint(pop_size)
                    pop[idx] = project_h(0.5 + 0.5 * (rng.rand(n) - 0.5) * 2, target)
                    fitness[idx] = compute_c5(pop[idx])
                    if fitness[idx] < best_fit:
                        best_fit = fitness[idx]
                        best_idx = idx
                no_improve = 0
            if no_improve > 200:
                break
    return pop[best_idx].copy(), best_fit

def periodic_de(h0, n, target, t0, time_limit, seed=43):
    """DE with constraint h[i] = h[i + n//2] (period 1)."""
    rng = np.random.RandomState(seed)
    half = n // 2
    h_half = h0[:half].copy()
    pop_size = min(max(half // 3, 30), 100)
    pop = np.zeros((pop_size, half))
    pop[0] = h_half
    for i in range(1, pop_size):
        if i < pop_size // 2:
            scale = 0.03 + 0.15 * (i / (pop_size // 2 + 1.0))
            pop[i] = np.clip(h_half + scale * (rng.rand(half) - 0.5) * 2, 0, 1)
        else:
            pop[i] = np.clip(0.5 + 0.4 * (rng.rand(half) - 0.5) * 2, 0, 1)
    pop_target = half / 2.0
    pop = np.array([project_h(p, pop_target) for p in pop])
    
    def eval_full(p):
        full = np.concatenate([p, p])
        return compute_c5(full)
    
    fitness = np.array([eval_full(p) for p in pop])
    best_idx = int(np.argmin(fitness))
    best_fit = float(fitness[best_idx])
    no_improve = 0
    gen = 0
    while time.time() - t0 < time_limit:
        gen += 1
        F = max(0.25, 0.7 * (0.995 ** gen))
        gen_imp = False
        for i in range(pop_size):
            if time.time() - t0 > time_limit:
                break
            cands = np.array([j for j in range(pop_size) if j != i])
            r = rng.rand()
            if r < 0.4 and best_idx != i:
                bc = np.array([j for j in cands if j != best_idx])
                a, b = int(rng.choice(bc, 2, replace=False)[0]), int(rng.choice(bc, 2, replace=False)[1])
                v = pop[best_idx] + F * (pop[a] - pop[b])
            else:
                a, b, c = rng.choice(cands, 3, replace=False)
                v = pop[a] + F * (pop[b] - pop[c])
            cr = rng.uniform(0.5, 0.85)
            jr = rng.randint(half)
            mask = rng.rand(half) < cr
            mask[jr] = True
            trial = np.where(mask, v, pop[i])
            trial = project_h(trial, pop_target)
            fit = eval_full(trial)
            if fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = fit
                if fit < best_fit - 1e-15:
                    best_fit = fit
                    best_idx = i
                    gen_imp = True
        if gen_imp:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break
    best_half = pop[best_idx].copy()
    return np.concatenate([best_half, best_half]), best_fit

def subgrad_polish(h0, n, target, t0, time_limit, seed=44):
    """Subgradient method with smoothing for final polish."""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.01
    for it in range(50000):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        thr = mc - 0.002 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        if len(idx) == 0:
            idx = np.array([np.argmax(corr)])
        if len(idx) > 30:
            idx = idx[np.argsort(corr[idx])[::-1][:30]]
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - h[lag:])
                g[lag:] -= h[:n - lag]
            else:
                la = -lag
                g[la:] += (1.0 - h[:n - la])
                g[:n - la] -= h[la:]
        g /= max(len(idx), 1)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        h = project_h(h - step * g, target)
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, step = h.copy(), c, min(step * 1.05, 0.03)
        else:
            step *= 0.95
    return best_h, best_c5

def coord_polish(h0, n, target, t0, time_limit, seed=45):
    h = h0.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    deltas = (0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005)
    for _ in range(20000):
        if time.time() - t0 > time_limit:
            break
        j = rng.randint(n)
        for d in deltas:
            trial = h.copy()
            trial[j] += d
            trial = project_h(trial, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                h, best_c5 = trial, c
                break
    return h, best_c5

def slsqp_polish(h0, n, target, t0, time_limit):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr)) * (2.0 / n)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        thr = mc - 0.001 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        if len(idx) > 20:
            idx = idx[np.argsort(corr[idx])[::-1][:20]]
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - x[lag:])
                g[lag:] -= x[:n - lag]
            else:
                la = -lag
                g[la:] += (1.0 - x[:n - la])
                g[:n - la] -= x[la:]
        g /= max(len(idx), 1)
        return g * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': 3000, 'ftol': 1e-14})
        return project_h(r.x, target)
    except:
        return h0

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
