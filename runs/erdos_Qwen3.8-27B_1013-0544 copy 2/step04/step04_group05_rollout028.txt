import numpy as np
from scipy.optimize import minimize, Bounds
import time, math

_g_beta = 100.0
_g_dx = 0.005
_g_target = 200.0
_g_n = 400
_g_t0 = 0.0
_g_budget = 1100.0

def _rem():
    return _g_budget - (time.time() - _g_t0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _g_obj(h):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    shifted = np.clip(_g_beta * (c - m), -400, 400)
    val = math.log(float(np.sum(np.exp(shifted))) / len(c)) / _g_beta
    return (m + val) * _g_dx

def _g_con(h):
    return h.sum() - _g_target

def _g_jac(h):
    return np.ones(_g_n)

def weighted_subgrad(h, top_idx, weights):
    n = len(h)
    offset = n - 1
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        lag = int(top_idx[i]) - offset
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad

def subgrad_loop(h_init, target, max_iters, top_k, base_temp, nev_beta, momentum, pert_every, pert_scale):
    n = len(h_init)
    best = h_init.copy()
    best_c5 = compute_c5(best)
    y = best.copy()
    vel = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if _rem() < 3:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        prog = min(it / max_iters, 1.0)
        temp = base_temp * (1.0 + 4.0 * (1.0 - prog) ** 2)
        k = max(15, int(top_k * (1.0 - 0.4 * prog)))
        top_idx = np.argsort(corr)[::-1][:k]
        top_vals = corr[top_idx].astype(float)
        x_sh = top_vals / temp
        x_sh -= np.max(x_sh)
        w = np.exp(x_sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k) / k
        else:
            w /= ws

        g = weighted_subgrad(y, top_idx, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn

        alpha = 0.012 * (1.0 + 0.5 * (1.0 - prog))
        improved = False
        for _ in range(18):
            ht = project_h(best + alpha * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                y = np.clip(ht + nev_beta * (ht - best), 0.0, 1.0)
                best = ht
                best_c5 = c5t
                no_improve = 0
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= pert_every:
                scale = pert_scale * (1.0 + 0.3 * (no_improve // pert_every))
                p = np.random.randn(n) * scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    y = best.copy()
                    vel *= 0.2
                    no_improve = 0
                elif no_improve > pert_every * 25:
                    break
    return best, best_c5

        except Exception:
            pass
    return best, best_c5
