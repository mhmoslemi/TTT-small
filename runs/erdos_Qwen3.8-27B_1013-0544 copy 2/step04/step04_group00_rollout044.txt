            s = h.sum()
            if s < 1e-13: break
            h = h + diff / s * h

def compute_subgrad_max(h, n, top_m=5):
    """Compute subgradient of max correlation at h.
    Uses top_m active lags with softmax weighting."""
    g = compute_all_g(h, n)
    m = g.max()
    # Softmax weights for top active lags
    alpha = 100.0
    w = np.exp(alpha * (g - m))
    w = w / w.sum()
    # Only keep top_m
    idx = np.argsort(w)[::-1][:top_m]
    w = w[idx]
    w = w / w.sum()
    
    # Compute gradient for each active lag
    grad = np.zeros(n)
    for wi, mi in zip(w, idx):
        lag = mi - (n - 1)
        # ∂c[mi]/∂h[j] = (1-h[j+lag]) if 0≤j+lag<n minus h[j-lag] if 0≤j-lag<n
        # Vectorized:
        # First term: (1-h[j+lag]) for valid j
        if lag >= 0:
            # j+lag < n means j < n-lag
            valid = np.arange(n - lag)
            grad[valid] += wi * (1 - h[valid + lag])
            # j-lag >= 0 means j >= lag
            if lag > 0:
                valid2 = np.arange(lag, n)
                grad[valid2] -= wi * h[valid2 - lag]
        else:
            # lag < 0, let l = -lag
            l = -lag
            # j+lag < n means j < n+l (always true since j < n)
            # but j+lag >= 0 means j >= l
            valid = np.arange(l, n)
            grad[valid] += wi * (1 - h[valid + lag])  # j+lag = j-l
            # j-lag >= 0 means j >= -lag = l... wait
            # j - lag = j + l, need 0 <= j+l < n, so j < n-l
            valid2 = np.arange(n - l)
            grad[valid2] -= wi * h[valid2 + l]  # h[j-lag] = h[j+l]

    h_prev = h.copy()
    h_cur = h.copy()
    step = 0.05

            break

        y = h_cur + (t - 1) / t_new * (h_cur - h_prev)

                break

                best_h = h_cur.copy()
                patience = 0

            patience = 0
            h_prev = h_cur.copy()
            t = 1.0

    best_c = compute_c5(best_h, n)
    t = 1.0
    h_prev = h.copy()
    h_cur = h.copy()
    step = 0.03
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = h_cur + (t - 1) / t_new * (h_cur - h_prev)
        y = np.clip(y, 0, 1)
        g = compute_all_g(y, n)
        m = g.max()
        w = np.exp(80.0 * (g - m))
        w = w / w.sum()
        top_idx = np.argsort(w)[::-1][:top_m]
        w = w[top_idx]
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, mi in zip(w, top_idx):
            lag = int(mi) - (n - 1)
            if lag >= 0:
                if lag < n:
                    v = np.arange(n - lag)
                    grad[v] += wi * (1 - y[v + lag])
                if lag > 0:
                    v2 = np.arange(lag, n)
                    grad[v2] -= wi * y[v2 - lag]
            else:
                l = -lag
                if l < n:
                    v = np.arange(l, n)
                    grad[v] += wi * (1 - y[v - l])
                if l > 0:
                    v2 = np.arange(n - l)
                    grad[v2] -= wi * y[v2 + l]
        c_check = compute_c5(y, n)
        found = False
        for _ in range(30):
            h_new = project_feasible(y - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_check - 1e-12:
                found = True
                break
            step *= 0.7
        if found:
            h_prev = h_cur
            h_cur = h_new
            t = t_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            step *= 0.5
            stall += 1
        if stall > 300:
            step = max(step * 4, 0.02)
            stall = 0
            h_prev = h_cur.copy()
            t = 1.0
    return best_h, best_c

def free_coord_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    free_mask = (best_h > 0.02) & (best_h < 0.98)
    free_idx = np.where(free_mask)[0]
    if len(free_idx) == 0:
        return best_h, best_c
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for idx in free_idx:
                for direction in (1.0, -1.0):
                    h2 = best_h.copy()
                    h2[idx] = np.clip(h2[idx] + direction * ss, 0, 1)
                    h2 = project_feasible(h2, n)
                    c2 = compute_c5(h2, n)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                        best_h = h2
                        improved = True
                        free_mask = (best_h > 0.02) & (best_h < 0.98)
                        free_idx = np.where(free_mask)[0]
                        break
    return best_h, best_c

def targeted_pair_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    g = compute_all_g(best_h, n)
    m = g.max()
    active = np.where(g > m - 0.005)[0]
    active_lags = [int(k) - (n - 1) for k in active]
    active_coords = set()
    for lag in active_lags:
        if lag >= 0:
            active_coords.update(range(0, n - lag))
            if lag > 0:
                active_coords.update(range(lag, n))
        else:
            l = -lag
            active_coords.update(range(l, n))
            if l > 0:
                active_coords.update(range(0, n - l))
    active_coords = list(active_coords)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for d in deltas:
        if time.time() > deadline:
            break
        prog = True
        while prog and time.time() < deadline:
            prog = False
            for i in active_coords:
                if time.time() > deadline:
                    break
                for j in active_coords:
                    if j <= i:
                        continue
                    for s in (1.0, -1.0):
                        hi, hj = best_h[i] + s * d, best_h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            best_h[i], best_h[j] = hi, hj
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                prog = True
                            else:
                                best_h[i], best_h[j] = hi - s * d, hj + s * d
                        if time.time() > deadline:
                            break
                    if time.time() > deadline:
                        break
    return best_h, best_c

def binary_refine(h, n, deadline, max_sweeps=100):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
                    ones.remove(j)
                    ones.append(i)
                    zeros = [x for x in range(n) if x not in ones]
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c

def try_resolution(h_best, n_best, c_best, n_new, deadline):
    if time.time() > deadline or n_new == n_best:
        return h_best, n_best, c_best
    if n_new > n_best:
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, n_new)
        h_new = project_feasible(np.interp(xn, xo, h_best), n_new)
    else:
        idx = np.linspace(0, n_best - 1, n_new).astype(int)
        h_new = project_feasible(h_best[idx], n_new)
    h_opt, c_opt = active_set_slsqp(h_new, n_new, deadline, [1000, 10000, 100000], 800)
    if c_opt < c_best:
        return h_opt, n_new, c_opt
    return h_best, n_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = project_feasible(np.full(n, 0.5), n)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha (40%)
    p1 = t0 + 0.40 * (budget_s - 5)
    best_h, best_c = active_set_slsqp(best_h, n, p1,
                                       [200, 1000, 5000, 20000, 100000, 500000, 2000000], 800)

    # Phase 2: Nesterov subgradient (15%)
    p2 = t0 + 0.55 * (budget_s - 5)
    h2, c2 = subgrad_nesterov(best_h.copy(), n, p2, top_m=8, max_iter=15000)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: SLSQP again from improved point (10%)
    p3 = t0 + 0.65 * (budget_s - 5)
    best_h, best_c = active_set_slsqp(best_h, n, p3,
                                       [100000, 500000, 2000000, 5000000, 10000000], 1200)

    # Phase 4: Multi-resolution (10%)
    p4 = t0 + 0.75 * (budget_s - 5)
    for nh in [200, 300, 350]:
        if time.time() > p4 - 30:
            break
        best_h, n, best_c = try_resolution(best_h, n, best_c, nh, p4 - 30)

    # Phase 5: Free coordinate descent (10%)
    p5 = t0 + 0.85 * (budget_s - 5)
    best_h, best_c = free_coord_descent(best_h, n, p5)

    # Phase 6: Targeted pair refinement (remaining)
    best_h, best_c = targeted_pair_refine(best_h, n, dl - 15)

    # Phase 7: Binary refinement as final attempt
    if time.time() < dl - 10:
        hb, cb = binary_refine(best_h.copy(), n, dl - 10, max_sweeps=200)
        if cb < best_c:
            best_c = cb
            best_h = hb

    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
