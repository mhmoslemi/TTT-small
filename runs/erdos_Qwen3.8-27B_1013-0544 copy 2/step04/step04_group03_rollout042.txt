def slp_step(h0, n, target, t0=None, budget=None, max_outer=20):
    """One SLP iteration: linearize and solve LP."""
    h = h0.copy()
    dx = 2.0 / n
    
    for outer in range(max_outer):
        if t0 is not None and budget - (time.time() - t0) < 3:
            break
        # Compute current overlaps
        raw_corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = raw_corr * dx
        t_cur = np.max(overlaps)
        
        # Linearize: overlap_k(h) ≈ overlap_k(h0) + grad_k · (h - h0)
        # For the LP, we use the current h as the linearization point
        # grad_k[t] = d(overlap_k)/d(h[t])
        
        # Solve LP: minimize t
        # s.t. for all k: overlap_k(h0) + grad_k · δ ≤ t
        #      (where δ = h_new - h, so overlap at h_new ≈ overlap_k(h0) + grad_k·δ)
        #      h ≤ 1, h ≥ 0 (i.e., -h ≤ δ ≤ 1-h)
        #      sum(δ) = 0
        #      t ≥ 0
        
        n_c = 2 * n - 1
        # Variables: δ[0..n-1], t
        # c = [0, ..., 0, 1] (minimize t)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Inequality constraints: A_ub @ x ≤ b_ub
        # For each k: grad_k · δ - t ≤ -overlap_k(h)
        # i.e., [grad_k, -1] · [δ, t] ≤ -overlap_k
        A_ub = np.zeros((n_c, n + 1))
        for j in range(n_c):
            g = d_overlap_dh(h, j) * dx  # gradient of overlap_j wrt h
            A_ub[j, :n] = g
            A_ub[j, n] = -1.0
        b_ub = -overlaps
        
        # Box constraints: -h[i] ≤ δ[i] ≤ 1-h[i]
        # h[i] + δ[i] ≤ 1 => δ[i] ≤ 1-h[i]
        # -δ[i] ≤ h[i] => δ[i] ≥ -h[i]
        # Combine with the A_ub above
        A_box = np.zeros((2 * n, n + 1))
        for i in range(n):
            A_box[i, i] = 1.0  # δ[i] ≤ 1-h[i]
            A_box[n + i, i] = -1.0  # -δ[i] ≤ h[i]
        b_box = np.concatenate([1.0 - h, h])
        
        A_full = np.vstack([A_ub, A_box])
        b_full = np.concatenate([b_ub, b_box])
        
        # Equality: sum(δ) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * n + [(0, None)], method='highs')
        
        if res.success:
            h_new = h + res.x[:n]
            h_new = project_h(h_new, target)
            t_new = compute_c5(h_new)
            if t_new < t_cur + 1e-10:
                h = h_new
            else:
                # Damped step
                alpha = 0.5
                h_new = project_h(h + alpha * res.x[:n], target)
                h = h_new
        else:
            break
        
        if t_cur - compute_c5(h) < 1e-10:
            break
    
    return h

            G[j, :n-la] += -h[la:]
