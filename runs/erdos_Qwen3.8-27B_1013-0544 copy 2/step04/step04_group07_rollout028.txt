def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:

        z[zeros[np.argsort(h[zeros])[::-1]][:int(diff)]] = 1.0
    elif diff < -1e-10:

        z[ones[np.argsort(h[ones])][:int(-diff)]] = 0.0

    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

                best_c5 = cur_c5

            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cool

def greedy_binary_ls(h, target, max_sweeps, t0, budget):
    """Greedy binary local search using current max lag as proxy."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    dx = 2.0 / n
    corr = compute_corr(h)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget - 5:
            break
        # Find active lags (top few)
        top_k = 5
        sorted_lags = np.argsort(corr)[::-1]
        active = sorted_lags[:top_k]
        
        best_c5_est = cur_c5
        best_pair = None
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        # For each pair (i, j), estimate new C5
        for i in ones:
            for j in zeros:
                # Compute delta at active lags
                est = 0
                for k_idx in active:
                    lag = k_idx - (n - 1)
                    delta = 0
                    # h[i]: 1→0
                    if lag >= 0:
                        if lag <= n - 1 - i and i + lag < n:
                            delta -= (1 - h[i + lag])
                        if lag <= i:
                            delta += h[i - lag]
                    else:

                        if la <= i:
                            delta -= (1 - h[i - la])
                        if la <= n - 1 - i and i + la < n:
                            delta += h[i + la]
                    # h[j]: 0→1
                    if lag >= 0:
                        if lag <= n - 1 - j and j + lag < n:
                            delta += (1 - h[j + lag])
                        if lag <= j:
                            delta -= h[j - lag]
                    else:
                        la = -lag
                        if la <= j:
                            delta += (1 - h[j - la])
                        if la <= n - 1 - j and j + la < n:
                            delta -= h[j + la]
                    est = max(est, (corr[k_idx] + delta) * dx)
                
                if est < best_c5_est - 1e-15:
                    best_c5_est = est
                    best_pair = (i, j)

        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
