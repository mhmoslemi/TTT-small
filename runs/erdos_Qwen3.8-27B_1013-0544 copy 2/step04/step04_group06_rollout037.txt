def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on shift."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient(h, top_k=5):
    """Compute subgradient of max correlation, using top-k active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    # Find top-k lags
    top_idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g, corr

    """SLSQP with log-sum-exp smooth approximation."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0, 1)] * n

    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        return (m + np.log(e.sum() / len(corr)) / beta) * dx

    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        nc = len(corr)

            if w[j] < 1e-11:

            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:

                g[:n - lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n - m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n - m2]
        return g * dx

    def eq(x):
        return np.sum(x) - target

    cons = [{'type': 'eq', 'fun': eq, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)  # velocity for momentum
    step = 0.02  # initial step size
    no_improve = 0
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        
        # Nesterov lookahead

        g, corr = subgradient(h_look, top_k)

        g /= gn

        for _ in range(15):

                best = h_new
                best_c5 = c5_new
                no_improve = 0
                improved = True
