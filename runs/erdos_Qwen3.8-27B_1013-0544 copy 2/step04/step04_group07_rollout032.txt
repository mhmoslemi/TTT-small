def get_subgrad(h, n, num_active=50):
    """Get subgradient of the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.005 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]

        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / max(len(idx), 1), corr, mc

    # Gradient of regularization
    g += 2.0 * lam * (0.5 - h)
    return g

    return np.sum(h) - target

def sum_constraint_jac(h, target):
    return np.ones(h.shape)

def run_slsqp(h0, n, target, beta, lam, bounds, maxiter, ftol, t0, budget):

            lambda h: obj_fn(h, beta, 2.0/n, lam),
            h0,

    except Exception:
        pass
    return h0

def binary_snap(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()

        z[ones[np.argsort(h[ones])][:int(-diff)]] = 0.0

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:

        h = np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s)) / len(s)) / beta

def corr_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    w = np.exp(s)

The idea: minimize t subject to:

These are valid upper bounds but not tight.

A tighter approach: use the McCormick envelope. For z = a*b with a,b ∈ [0,1]:
z ≤ a, z ≤ b, z ≥ a+b-1, z ≥ 0

But here I have z = h[i]*(1-h[i+k]), which is a product of two variables.

Let u = h[i], v = 1-h[i+k], w = u*v.
Then: w ≤ u, w ≤ v, w ≥ u+v-1, w ≥ 0.

And v = 1-h[i+k], so u+v-1 = h[i] + 1 - h[i+k] - 1 = h[i] - h[i+k].

So: w ≥ h[i] - h[i+k], w ≥ 0, w ≤ h[i], w ≤ 1-h[i+k].

The LP would be:
minimize t
subject to:
  ∑ w[i,k] ≤ t/n for all k (where w[i,k] bounds h[i]*(1-h[i+k]))
  w[i,k] ≤ h[i]
  w[i,k] ≤ 1-h[i+k]
  w[i,k] ≥ h[i] - h[i+k]
  w[i,k] ≥ 0
  ∑ h[i] = n/2
  0 ≤ h[i] ≤ 1

This is a valid LP relaxation! The optimal value of this LP is a lower bound on C₅. If the relaxation is tight (which it often is for such problems), then the LP solution gives the optimal h.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s)) / len(s)) / beta

def corr_smooth_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -300, 300)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    n1 = n - 1
    thresh = 1e-10
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < thresh:
            continue
        la = abs(ki - n1)
        if la == 0:
            g += wi * (1.0 - 2.0 * h)
        else:
            g[la:] += wi * (1.0 - h[:n - la])
            g[:n - la] -= wi * h[la:]
    return g

def slsqp_opt(h0, n, target, beta, maxiter, ftol):
    dx = 2.0 / n
    def f(h):
        return smooth_max_val(np.correlate(h, 1.0 - h, mode="full"), beta) * dx
    def fp(h):
        return corr_smooth_grad(h, beta) * dx
    def cfun(h):
        return h.sum() - target
    def cfjac():
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cfjac}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(f, h0, jac=fp, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def backtracking_momentum(h0, n, target, beta, t0, time_limit, n_iter=20000, lr0=0.02, mom=0.85):
    h = h0.copy()
    v = np.zeros(n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() - t0 > time_limit - 3:
            break
        g = corr_smooth_grad(h, beta) * dx
        lr = lr0 / (1.0 + 0.005 * it)
        v = mom * v - lr * g
        h_trial = project_h(h + v, target)
        f0 = smooth_max_val(np.correlate(h, 1 - h, mode="full"), beta)
        f1 = smooth_max_val(np.correlate(h_trial, 1 - h_trial, mode="full"), beta)
        if f1 > f0 + 0.5 * lr0 * np.dot(g, h_trial - h):
            v *= -0.5
            h_trial = project_h(h + v, target)
        h = h_trial
        if it % 50 == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h.copy(), c
    c = compute_c5(h)
    if c < best_c5:
        best_h, best_c5 = h.copy(), c
    return best_h, best_c5

def resample_to(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def binary_snap(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        sorted_z = zeros[np.argsort(h[zeros])[::-1]]
        z[sorted_z[:int(np.ceil(diff))]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        sorted_o = ones[np.argsort(h[ones])]
        z[sorted_o[:int(np.ceil(-diff))]] = 0.0
    return project_h(z, target)

def variance_penalty_obj(h, beta, dx, lam, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    base = smooth_max_val(corr, beta) * dx
    top = np.sort(corr)[::-1][:top_k]
    var_pen = np.var(top) * lam
    return base + var_pen

            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            cur = slsqp_opt(cur, n_r, target, beta, 3000, 1e-14)
            if n_r == n_parent:
                c = compute_c5(cur)
                if c < best_c5:
                    best_h, best_c5 = cur.copy(), c
        h_work = cur

    # Phase 2: Backtracking momentum at parent resolution
    n_f = n_parent
    target_f = n_f / 2.0
    cur = project_h(resample_to(best_h, n_f), target_f)
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        cur, c = backtracking_momentum(cur, n_f, target_f, 1000, t0, min(rem, 120) + t0, n_iter=15000, lr0=0.01, mom=0.85)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: High-beta SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        for beta in [2000, 5000, 10000]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            cur = slsqp_opt(cur, n_f, target_f, beta, 5000, 1e-15)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5 = cur.copy(), c

    # Phase 4: Multi-start perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        for trial in range(8):
            rem = budget_s - (time.time() - t0)
            if rem < 12:
                break
            scale = 0.02 + 0.01 * trial
            noise = rng.normal(0, scale, n_f)
            h_pert = project_h(best_h + noise, target_f)
            h_new = slsqp_opt(h_pert, n_f, target_f, 2000, 2000, 1e-14)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new.copy(), c

    # Phase 5: Try alternative resolutions
    for n_alt in [n_f - 32, n_f + 32, n_f - 64, n_f + 64]:
        rem = budget_s - (time.time() - t0)
        if rem < 30 or n_alt < 128:
            continue
        target_alt = n_alt / 2.0
        h_alt = project_h(resample_to(best_h, n_alt), target_alt)
        h_alt = slsqp_opt(h_alt, n_alt, target_alt, 2000, 3000, 1e-14)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c5:
            best_h, best_c5 = h_alt.copy(), c_alt
            n_f = n_alt
            target_f = target_alt

    # Phase 6: Binary snap + re-optimize
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        hb = binary_snap(best_h, target_f)
        cb = compute_c5(hb)
        if cb < best_c5:
            best_h, best_c5 = hb, cb
        best_h = slsqp_opt(best_h, n_f, target_f, 5000, 4000, 1e-15)
        best_c5 = compute_c5(best_h)

    # Phase 7: Final momentum + SLSQP
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        best_h, best_c5 = backtracking_momentum(best_h, n_f, target_f, 3000, t0, min(rem - 5, 40) + t0, n_iter=5000, lr0=0.005, mom=0.8)
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h = slsqp_opt(best_h, n_f, target_f, 10000, 3000, 1e-15)
        best_c5 = compute_c5(best_h)

    best_h = project_h(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_f
