def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_pnorm(h, p):
    """Compute smooth max using p-norm of top values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    thresh = np.percentile(corr, 50)
    above = np.maximum(corr - thresh, 0)
    val = (np.sum(above**p)) ** (1.0/p) + thresh
    return val * dx

    n = len(h)

            grad += (1.0 - 2.0 * h)

            grad[lag:] += (1.0 - h[:-lag])

def slsqp_smooth_max(h, beta):

    m = np.max(corr)

    n = len(h)
    slopes = np.zeros(2*n-1)
    
    # lag = 0
    slopes[n-1] = 1.0 - 2.0 * h[i]
    
    # Positive lags: 1 to min(i, n-1) for term 1
    max_lag_pos_t1 = min(i, n-1)
    if max_lag_pos_t1 > 0:
        lags = np.arange(1, max_lag_pos_t1 + 1)
        idx_pos = lags + n - 1
        slopes[idx_pos] += (1.0 - h[i - lags])
    
    # Positive lags: 1 to min(n-1-i, n-1) for term 2
    max_lag_pos_t2 = min(n-1-i, n-1)
    if max_lag_pos_t2 > 0:
        lags = np.arange(1, max_lag_pos_t2 + 1)
        idx_pos = lags + n - 1
        slopes[idx_pos] -= h[i + lags]
    
    # Negative lags (m = -lag): 1 to min(n-1-i, n-1) for term 1
    max_lag_neg_t1 = min(n-1-i, n-1)
    if max_lag_neg_t1 > 0:
        m_vals = np.arange(1, max_lag_neg_t1 + 1)
        idx_neg = (n-1) - m_vals
        slopes[idx_neg] += (1.0 - h[i + m_vals])
    
    # Negative lags: 1 to min(i, n-1) for term 2
    max_lag_neg_t2 = min(i, n-1)
    if max_lag_neg_t2 > 0:
        m_vals = np.arange(1, max_lag_neg_t2 + 1)
        idx_neg = (n-1) - m_vals
        slopes[idx_neg] -= h[i - m_vals]
    
    return slopes

                h[i] = new_hi
        corr = np.correlate(h, 1.0 - h, mode="full")
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
    return best_h, best_c5

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def slsqp_round(h, betas, maxiter, time_fn):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 15:
            break
        def obj_f(x):
            v, _ = smooth_max_obj_grad(x, beta)
            return v
        def grad_f(x):
            _, g = smooth_max_obj_grad(x, beta)
            return g
        def con_f(x):
            return x.sum() - target
        def con_jac(x):
            return np.ones(len(x))
        constraints = [{'type': 'eq', 'fun': con_f, 'jac': con_jac}]
        try:
            res = minimize(obj_f, h_cur, method='SLSQP', bounds=bounds,
                          jac=grad_f, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgrad_polish(h, target, n_iter, time_fn, top_k, steps_list):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(3, n // 25))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        found = False
        for step in steps_list:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < compute_c5(h) - 1e-14:
                h = ht
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp > 200:
                sc = 0.003 * min(5, no_imp // 200 + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < compute_c5(h):
                    h = ht
                    no_imp = 0
            if no_imp > n_iter // 8:
                break
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Coordinate descent at current resolution
    target = n_best / 2.0
    h_cd, c5_cd = coord_descent(best_h, 2000, time_left, target, K=10, alpha=0.8)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd, c5_cd

    # Phase 2: SLSQP polish
    if time_left() > 30:
        h_s, c5_s = slsqp_round(best_h, [200, 1000, 5000, 20000], 3000, time_left)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Multi-resolution exploration
    for n_res in [150, 200, 300, 400, 500, 700, 900]:
        if time_left() < 80:
            break
        if n_res == n_best:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        # Coordinate descent at new resolution
        h_r, c5_r = coord_descent(h_res, 1500, time_left, n_res / 2.0, K=8, alpha=0.9)
        if time_left() > 40:
            h_r, c5_r = slsqp_round(h_r, [500, 3000, 10000], 2000, time_left)
        if c5_r < best_c5:
            best_h, best_c5, n_best = h_r, c5_r, n_res

    # Phase 4: More coordinate descent with different relaxation
    target = len(best_h) / 2.0
    if time_left() > 30:
        h_cd2, c5_cd2 = coord_descent(best_h, 3000, time_left, target, K=12, alpha=0.6)
        if c5_cd2 < best_c5:
            best_h, best_c5 = h_cd2, c5_cd2

    # Phase 5: Subgradient polish
    if time_left() > 20:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        best_h, best_c5 = subgrad_polish(best_h, len(best_h)/2.0,
                                          int(time_left() * 80), time_left, 5, steps)

    # Phase 6: Final SLSQP
    if time_left() > 15:
        h_f, c5_f = slsqp_round(best_h, [10000, 50000, 150000], 5000, time_left)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Phase 7: Final coordinate descent
    if time_left() > 10:
        target = len(best_h) / 2.0
        h_f2, c5_f2 = coord_descent(best_h, 2000, time_left, target, K=15, alpha=1.0)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2

    # Final projection and verification
    target = len(best_h) / 2.0
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
