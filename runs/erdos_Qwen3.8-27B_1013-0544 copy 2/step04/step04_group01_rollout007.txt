    """Change h[j] to new_val and redistribute the difference."""
    h_new = h.copy()
    diff = new_val - h[j]
    h_new[j] = new_val
    # Redistribute -diff across all other coordinates
    others = np.ones(n)
    others[j] = 0
    # Proportional redistribution (clipped to maintain bounds)
    for _ in range(100):
        # How much can each other coordinate absorb?
        room_up = (1 - h_new) * others
        room_down = h_new * others
        if diff > 0:
            # Need to decrease others by diff total

    h_new *= target / h_new.sum()

        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=40):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht

    if n <= 120:
        if time.time() < deadline - 5:
            h, best_c = pair_search(h, n, deadline - 5,
                                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    else:
        if time.time() < deadline - 5:
            h, best_c = near_pair_search(h, n, deadline - 5,
                                         [0.02, 0.01, 0.005, 0.002, 0.001], radius=50)

    if time.time() < deadline - 3:
        ht = slsqp(h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    if n <= 120 and time.time() < deadline - 2:
        h, best_c = pair_search(h, n, deadline - 1,
                                [0.001, 0.0005, 0.0002, 0.0001])
    elif time.time() < deadline - 2:
        h, best_c = near_pair_search(h, n, deadline - 1,
                                     [0.001, 0.0005, 0.0002], radius=30)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial construction
    alloc1 = min(200, (deadline - t0) * 0.25)
    h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try intercepting at different n values
    x_init = np.linspace(0, 2, n_init)
    for n_try in [100, 150, 250]:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.2)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with perturbation
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.03 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended pair search on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005,
                                          0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 3,
                                              [0.01, 0.005, 0.002, 0.001,
                                               0.0005, 0.0002, 0.0001],
                                              radius=60)

    # Phase 5: Final SLSQP polish + pair search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp(best_h, n, 1000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            if n <= 150:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.0005, 0.0002, 0.0001],
                                                  radius=40)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
