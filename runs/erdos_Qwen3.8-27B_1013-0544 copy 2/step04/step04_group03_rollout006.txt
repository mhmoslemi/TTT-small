import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def do_slsqp(h0, n, target, beta, maxiter=5000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def generate_thue_morse(n):
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    if t.sum() != n / 2:
        if t.sum() > n / 2:
            ones = np.where(t == 1)[0]
            t[ones[:int(t.sum() - n / 2)]] = 0
        else:
            zeros = np.where(t == 0)[0]
            t[zeros[:int(n / 2 - t.sum())]] = 1
    return t

def generate_shifted_tm(n, shift):
    t = generate_thue_morse(n)
    return np.roll(t, shift)

    stall = 0

        if budget - (time.time() - t0) < 3:
            break
        max_idx = np.argmax(cur_corr)
        max_val = cur_corr[max_idx]
        lag = int(max_idx) - (n - 1)
        improved = False
        ones = np.where(cur == 1)[0]

            break

        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)
