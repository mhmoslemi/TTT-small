import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def binary_c5(h, n):
    """Compute C5 for a binary (or general) h vector."""
    return compute_c5(h, n)

def compute_N(h, n):
    """Compute N_k = sum_i h[i+k]*(1-h[i]) for each shift k (linear correlation)."""
    # Returns array of length 2n-1, indexed as correlate output
    return np.correlate(h, 1-h, mode='full').astype(float)

def make_beatty(n, density, offset=0):
    """Create a Beatty sequence (Sturmian word) of length n."""
    m = int(round(n * density))
    h = np.zeros(n)
    # Place ones at positions floor(i * n / m + offset) for i = 0, ..., m-1
    # Actually, Beatty sequence: positions are floor((i+1)*n/m) for i=0,...,m-1
    positions = np.floor(np.arange(m) * n / m + offset).astype(int)
    positions = positions % n  # wrap around
    h[positions] = 1.0
    return h

def make_random_binary(n, rng):
    """Random binary sequence with exactly n/2 ones."""
    indices = rng.permutation(n)
    h = np.zeros(n)
    h[indices[:n//2]] = 1.0
    return h

def binary_swap_search(h, n, deadline, max_sweeps=500):
    """Local search for binary solutions using single swaps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    N = compute_N(h, n)  # length 2n-1
    current_max = N.max()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Try all swaps
        best_delta = 0
        best_a, best_b = -1, -1
        
        for a in ones:
            for b in zeros:
                # Compute effect of swapping a and b
                # a: 1 -> 0, b: 0 -> 1
                new_max = current_max
                # We need to check all k
                # For efficiency, only check k where N_k is near the max
                # But for correctness, let's do all k
                
                # Compute ΔN_k for all k
                # For k > 0 (index n-1+k in the correlate output):
                # ΔN = -1_{h[a-k]=0, valid} + 1_{h[a+k]=1, valid} 
                #       + 1_{h[b-k]=0, valid} - 1_{h[b+k]=1, valid}
                
                # This is O(n) per swap, too slow for O(n^2) swaps

                h[a], h[b] = 0.0, 1.0
                N_new = np.correlate(h, 1-h, mode='full')
                c_new = float(N_new.max())
                h[a], h[b] = 1.0, 0.0
                
                if c_new < current_max - 1e-10:
                    best_delta = c_new - current_max
                    best_a, best_b = a, b
                    current_max = c_new
                    improved = True
        
        if not improved:

            best_c = float(current_max * 2.0 / n)

```python
def binary_swap_search_fi(h, n, deadline, max_sweeps=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    N = np.correlate(h, 1-h, mode='full').astype(float)
    current_max_val = N.max()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for a in ones[:]:
            for b in zeros[:]:
                h[a], h[b] = 0.0, 1.0
                c_new = float(np.correlate(h, 1-h, mode='full').max())
                h[a], h[b] = 1.0, 0.0
                if c_new < current_max_val - 1e-10:
                    # Apply this swap
                    h[a], h[b] = 0.0, 1.0
                    ones.remove(a)
                    zeros.remove(b)
                    ones.append(b)
                    zeros.append(a)
                    current_max_val = c_new
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    best_c = current_max_val * 2.0 / n
    return h, best_c
```
