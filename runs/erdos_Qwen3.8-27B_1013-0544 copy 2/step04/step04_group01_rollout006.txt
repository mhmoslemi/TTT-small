Also, I want to add a key improvement: **using the gradient to guide the pair search**. Instead of trying all pairs uniformly, I can:
1. Compute the gradient of the smooth max objective
2. For each i, the best j to pair with is the one that maximizes the directional derivative in the (e_i - e_j) direction
3. This gives a much more efficient search

The directional derivative in direction (e_i - e_j) is: grad[i] - grad[j]
We want to minimize, so we want grad[i] - grad[j] to be as negative as possible, i.e., grad[j] >> grad[i].

So the best pairs are (i, j) where grad[j] - grad[i] is large (move mass from high-gradient to low-gradient positions).

Wait, actually for the direction d * (e_i - e_j), the directional derivative is d * (grad[i] - grad[j]). We want this to be negative (decreasing the objective), so we want grad[i] < grad[j], i.e., move mass from j (high gradient) to i (low gradient).

But this is just gradient descent! The issue is that the constraint set (box + linear equality) makes the gradient descent step infeasible. The pair search is a way to do projected gradient descent with elementary directions.

So a smarter pair search would:
1. Compute grad
2. Sort indices by gradient value
3. Try pairs where the lower-gradient index receives mass from the higher-gradient index
4. This should find improvements much faster

            break
        _, grad = smooth_obj_grad(h, n, alpha)
        order = np.argsort(grad)
        pairs = []
        for a in range(n):
            for b in range(a + 1, min(n, a + 80)):
                i, j = order[a], order[b]
                diff = grad[j] - grad[i]
                if diff > 1e-12:
                    pairs.append((diff, i, j))
        pairs.sort(reverse=True)
        improved = False
        for diff, i, j in pairs[:top_k]:
            if time.time() > deadline:
                return h, best_c
            for d in steps:
                hi = h[i] + d; hj = h[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c; improved = True; break
                    else:
                        h[i], h[j] = hi - d, hj + d
        if not improved:
            break
    return h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 12:
            break
        ht = slsqp_opt(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c; best_h = ht

    if time.time() < deadline - 12:
        ht = lbfgs_opt(best_h, n, 200000, maxiter=5000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c; best_h = ht

    if time.time() < deadline - 8:
        best_h, best_c = grad_pair_search(best_h, n, deadline - 8, top_k=400)

    if n <= 150 and time.time() < deadline - 5:
        best_h, best_c = pair_search(best_h, n, deadline - 5,
                                     [0.02, 0.01, 0.005, 0.002, 0.001])
    elif time.time() < deadline - 5:
        best_h, best_c = pair_search(best_h, n, deadline - 5,
                                     [0.02, 0.01, 0.005], max_passes=15)

    if time.time() < deadline - 8:
        ht = slsqp_opt(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c; best_h = ht

    if time.time() < deadline - 3:
        best_h, best_c = grad_pair_search(best_h, n, deadline - 3,
                                          top_k=500, steps=(0.005, 0.002, 0.001, 0.0005))

    if n <= 150 and time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                     [0.001, 0.0005, 0.0002, 0.0001])
    elif time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, n, deadline - 0.5,
                                     [0.001, 0.0005], max_passes=10)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 100

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = (deadline - t0) * 0.45
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150, 200]:
            if n_try == n_init or time.time() > deadline - 25:
                continue
            remaining = deadline - time.time()
            alloc = min(55, remaining * 0.18)
            if alloc < 15:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt; best_h = h_opt
                x_init = x_new; n_init = n_try

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(3, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.02 + 0.12 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 10) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt; best_h = h_opt

    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002], max_passes=10)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
