import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _corr_gradient(h, corr, w, n):
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += wj * (1.0 - h[idx + lag])
            grad[idx + lag] -= wj * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += wj * (1.0 - h[idx])
            grad[idx] -= wj * h[idx + la]
    return grad

def lbfgs_obj(x, dx, beta, pen, ts):
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = smooth_max(corr, beta) * dx
    val += pen * (x.sum() - ts) ** 2
    return val

def lbfgs_grad(x, dx, beta, pen, ts):
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -700.0, 700.0))
    w /= w.sum()
    grad = _corr_gradient(x, corr, w, n) * dx
    grad += 2.0 * pen * (x.sum() - ts) * np.ones(n)
    return grad

def lbfgs_refine(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    c5_best = compute_c5(h)
    h_best = h.copy()
    pen = 500.0
    betas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    h_curr = h.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        h_curr = h_best.copy()
        try:
            res = minimize(lbfgs_obj, h_curr, jac=lbfgs_grad, args=(dx, beta, pen, ts),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 80})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass
        # Try with perturbation
        if time.time() < deadline:
            h_pert = project_feasible(h_best + 0.02 * np.random.randn(n), ts)
            try:
                res = minimize(lbfgs_obj, h_pert, jac=lbfgs_grad, args=(dx, beta, pen, ts),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 80})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    h_best = h_new
                    c5_best = c5_new
            except Exception:
                pass
    return h_best, c5_best

def binary_sa(h_init, n, deadline, seed):
    np.random.seed(seed)
    b = (h_init > 0.5).astype(np.float64)
    n_on = int(b.sum())
    target = n // 2
    if n_on > target:
        idx = np.where(b == 1)[0]
        b[np.random.choice(idx, n_on - target, replace=False)] = 0
    elif n_on < target:
        idx = np.where(b == 0)[0]
        b[np.random.choice(idx, target - n_on, replace=False)] = 1
    c5_cur = compute_c5(b)
    c5_best = c5_cur
    b_best = b.copy()
    T = 0.005
    for it in range(500000):
        if time.time() > deadline:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        c5_new = compute_c5(b)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-8)):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                b_best = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        T *= 0.9999
    return b_best, c5_best

def slsqp_polish(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    c5 = compute_c5(h)
    for beta in [100000, 500000, 2000000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(lambda x: smooth_max(np.correlate(np.clip(x,0,1), 1-np.clip(x,0,1), 'full'), beta) * dx,
                          h, method='SLSQP', bounds=bounds, constraints=con,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h, c5 = h_new, c5_new
        except Exception:
            pass
    return h, c5

def subsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h, best_c5, best_n = None, float('inf'), 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:

    if parent is None:

    # Binarize differently

        b_t = (parent > thresh).astype(np.float64)
        b_t = project_feasible(b_t, ts)
        starts.append(b_t)
    for idx, s in enumerate(starts):
        if time.time() > d2:

    d4 = t0 + budget_s * 0.97

        # Subgradient micro-polish
