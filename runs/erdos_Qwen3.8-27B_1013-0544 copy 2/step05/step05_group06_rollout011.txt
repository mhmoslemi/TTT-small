            break

            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):

        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta

    for sweep in range(max_sweeps):

            break
        F = compute_F(h, n)
        g = compute_g(h, n)

        K_eff = min(K, n_shifts)
        top_idx = np.argsort(g)[::-1][:K_eff]

        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for a in range(K_eff):
                    for b in range(a + 1, K_eff):
                        m1, m2 = top_idx[a], top_idx[b]
                        ds = s[m2] - s[m1]

                            dc = (g[m1] - g[m2]) / ds

                    if cur_max < best_c:

                    top_idx = np.argsort(g)[::-1][:K_eff]

        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], m_idx[v4]] += delta

def pair_descent(h, n, deadline, K=8, max_sweeps=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    n_shifts = 2 * n - 1
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        cur_max = g.max()
        K_eff = min(K, n_shifts)
        top_idx = np.argsort(g)[::-1][:K_eff]
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                cands = [d_lo, d_hi]
                for a in range(K_eff):
                    for b in range(a + 1, K_eff):
                        m1, m2 = top_idx[a], top_idx[b]
                        ds = s[m2] - s[m1]
                        if abs(ds) > 1e-15:
                            dc = (g[m1] - g[m2]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                bd = 0.0
                bv = cur_max
                for d in cands:
                    d = float(np.clip(d, d_lo, d_hi))
                    v = float(np.max(g + d * s))
                    if v < bv - 1e-13:
                        bv = v
                        bd = d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    g += bd * s
                    update_F(F, i, j, bd, n, dm)
                    cur_max = g.max()
                    if cur_max < best_c:
                        best_c = cur_max
                        best_h = h.copy()
                    top_idx = np.argsort(g)[::-1][:K_eff]
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return best_h, best_c

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    dl = t0 + total
    rng = np.random.RandomState(seed)

    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP warm-up

    for alpha in [200, 1000, 5000, 25000, 100000]:

            break
        h = slsqp_optimize(h, n, alpha, p1, maxiter=200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 2: Pair descent K=8
    p2 = t0 + 0.38 * total
    h, c = pair_descent(best_h.copy(), n, p2, K=8, max_sweeps=500)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 3: SLSQP high alpha
    p3 = t0 + 0.46 * total
    if time.time() < p3:
        h = slsqp_optimize(best_h.copy(), n, 500000, p3, maxiter=400)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 4: Pair descent K=16
    p4 = t0 + 0.68 * total
    h, c = pair_descent(best_h.copy(), n, p4, K=16, max_sweeps=120)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 5: Pair descent K=24
    p5 = t0 + 0.82 * total
    h, c = pair_descent(best_h.copy(), n, p5, K=24, max_sweeps=40)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    # Phase 6: Perturb + reoptimize
    p6 = t0 + 0.90 * total
    for scale in [0.15, 0.08, 0.04]:
        if time.time() > p6:
            break
        h_p = best_h + rng.randn(n) * scale
        h_p = project_feasible(h_p, n)
        h_p = slsqp_optimize(h_p, n, 50000, p6, maxiter=100)
        h_p, c = pair_descent(h_p, n, p6, K=8, max_sweeps=25)
        if c < best_c:
            best_c = c
            best_h = h_p.copy()
    # Phase 7: Final SLSQP
    p7 = t0 + 0.95 * total
    if time.time() < p7:
        h = slsqp_optimize(best_h.copy(), n, 20000000, p7, maxiter=500)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 8: Final pair descent
    if time.time() < dl - 3:
        h, c = pair_descent(best_h.copy(), n, dl - 2, K=16, max_sweeps=60)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
