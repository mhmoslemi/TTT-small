    # c_old[k] + d[k]*delta is a line for each k
    # g(delta) = max_k (c_old[k] + d[k]*delta)
    # Find delta in [delta_lo, delta_hi] that minimizes g
    # Use subgradient bisection
    for _ in range(50):
        delta_mid = (delta_lo + delta_hi) / 2
        vals = c_old + d * delta_mid
        max_val = vals.max()
        active = np.where(np.isclose(vals, max_val, atol=1e-10))[0]
        subgrad = d[active].mean()
        if subgrad > 0:
            delta_hi = delta_mid
        elif subgrad < 0:
            delta_lo = delta_mid
        else:
            break
    return (delta_lo + delta_hi) / 2

        if abs(diff) < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

    act0 = c_vals >= m0 - 1e-12
