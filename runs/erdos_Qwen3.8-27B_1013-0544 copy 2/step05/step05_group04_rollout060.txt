def compute_d(i, j, h, n):
    m = np.arange(2*n-1)
    # c[m, i]
    idx1_i = i - m + (n-1)  # for (1-h[...]) term
    valid1_i = (idx1_i >= 0) & (idx1_i < n)
    idx2_i = i + m - (n-1)  # for -h[...] term
    valid2_i = (idx2_i >= 0) & (idx2_i < n)
    
    c_i = np.zeros(2*n-1)
    c_i[valid1_i] = 1.0 - h[idx1_i[valid1_i]]
    c_i[valid2_i] -= h[idx2_i[valid2_i]]
    
    # c[m, j]
    idx1_j = j - m + (n-1)
    valid1_j = (idx1_j >= 0) & (idx1_j < n)
    idx2_j = j + m - (n-1)
    valid2_j = (idx2_j >= 0) & (idx2_j < n)
    
    c_j = np.zeros(2*n-1)
    c_j[valid1_j] = 1.0 - h[idx1_j[valid1_j]]
    c_j[valid2_j] -= h[idx2_j[valid2_j]]
    
    return c_i - c_j
```

    # d[m] is nonzero only where c[m,i] or c[m,j] is nonzero
    # c[m,i] nonzero for m in [max(0, i-(n-1)), min(2n-2, i+n-1)]
    # which simplifies to m in [max(0, i-n+1), min(2n-2, i+n-1)]
    # For i in [0,n-1]: m in [0, min(2n-2, i+n-1)] = [0, 2n-2] (since i+n-1 >= n-1 and 2n-2 >= n-1)
    # Actually more precisely:
    # First term valid: 0 <= i-m+(n-1) < n → i-n+1 <= m <= i+n-1
    # Second term valid: 0 <= i+m-(n-1) < n → n-1-i <= m <= 2n-2-i
    # Union: m in [max(0, i-n+1, n-1-i), min(2n-2, i+n-1, 2n-2-i)]
    # For i in [0, n-1]: i-n+1 >= -(n-1) >= 0-... let's just use the full range
    # The full range of m where d could be nonzero is [0, 2n-2]
    
    m = np.arange(2*n-1)
    dx = 2.0/n
    
    idx1_i = i - m + (n-1)
    v1i = (idx1_i >= 0) & (idx1_i < n)
    idx2_i = i + m - (n-1)
    v2i = (idx2_i >= 0) & (idx2_i < n)
    
    idx1_j = j - m + (n-1)
    v1j = (idx1_j >= 0) & (idx1_j < n)
    idx2_j = j + m - (n-1)
    v2j = (idx2_j >= 0) & (idx2_j < n)
    
    d = np.zeros(2*n-1)
    d[v1i] += dx * (1.0 - h[idx1_i[v1i]])
    d[v2i] -= dx * h[idx2_i[v2i]]
    d[v1j] -= dx * (1.0 - h[idx1_j[v1j]])
    d[v2j] += dx * h[idx2_j[v2j]]
    
    return d
```

```python
def find_optimal_delta(corr, d, lo, hi):
    if lo >= hi:
        return 0.0
    # f(delta) = max(corr + delta * d), convex in delta
    # Use golden section or ternary search
    # For speed, do a coarse search then refine
    n_pts = 20

    vals = np.max(corr[None, :] + candidates[:, None] * d[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_delta
