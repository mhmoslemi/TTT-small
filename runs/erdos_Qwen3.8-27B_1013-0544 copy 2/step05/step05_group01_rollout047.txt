Actually, let me think about this differently. The problem is:
min_h max_k f_k(h)

where f_k(h) = c[k] (the correlation at lag k).

A good approach is the **bisection method**:
- Guess a value T
- Check if there exists h with max_k f_k(h) ≤ T
- This is: minimize max_k (f_k(h) - T) or equivalently check feasibility of f_k(h) ≤ T for all k

The feasibility check is a non-convex problem (since f_k is not convex), so bisection doesn't directly work.

min_h (1/|S|) sum_{k in S} f_k(h)

where S is the set of top-m lags. This is a smooth (though non-convex) objective that approximates the max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, k):
    m = n - 1
    offset = k - m
    g = np.zeros(n)
    lo1 = max(0, -offset)
    hi1 = min(n, n - offset)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1 + offset:hi1 + offset])
    lo2 = max(0, offset)
    hi2 = min(n, n + offset)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    return g

def nesterov_subgrad(h, n, deadline, n_iter=30000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.08
    beta = 0.9
    for it in range(n_iter):
        if time.time() > deadline:
            break
        y = h + beta * v
        y_clip = np.clip(y, 0.0, 1.0)
        c = np.correlate(y_clip, 1 - y_clip, mode='full')
        k_star = np.argmax(c)
        g = grad_lag(y_clip, n, k_star)
        v = beta * v - lr * g
        h = h + v
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        if it % 100 == 0:
            lr = max(0.001, lr * 0.995)
    return best_h, best_c

        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        return float(x.sum() - n / 2.0)
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
