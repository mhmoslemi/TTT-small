def project_feasible(h, n):

            if s < 1e-14:

def projected_grad_descent(h, n, deadline, alpha=50000, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = analytic_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = lr / gn
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:

                found = True
                break

            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.05, 10.0)

  Change: +(1-h[i+shift_k])

  Valid when 0 ≤ i-shift_k < n, i.e., i-(n-1) ≤ shift_k ≤ i

         + [(1-h[i+shift_k]) * I(-i≤shift_k≤n-1-i)]
         - [h[i-shift_k] * I(i-(n-1)≤shift_k≤i)]

This is O(n) to compute per swap (only n shifts are affected for each position change).

To evaluate if a swap improves C₅, we need max_k (overlaps[k] + delta[k]). Since only O(n) values of k change, and the rest stay the same, we can:
1. Compute the new max over the changed k values
2. Compare with the max over unchanged k values (which is the old max, unless the old max was at a changed k)

So the algorithm per swap is O(n) for computing delta, and O(n) for finding the new max (or O(1) if we maintain a sorted structure).

    ng = 2 * n - 1
    delta = np.zeros(ng)
    
    # j as source: k in [n-1-j, 2n-2-j]
    lo = max(0, n-1-j)
    hi = min(ng-1, 2*n-2-j)
    if lo <= hi:
        shifts = np.arange(lo, hi+1) - (n-1)
        delta[lo:hi+1] -= (1.0 - hb[j + shifts])
    
    # j as target: k in [j, j+n-1]
    lo = max(0, j)
    hi = min(ng-1, j+n-1)
    if lo <= hi:
        shifts = np.arange(lo, hi+1) - (n-1)
        delta[lo:hi+1] += hb[j - shifts]
    
    # i as source: k in [n-1-i, 2n-2-i]
    lo = max(0, n-1-i)
    hi = min(ng-1, 2*n-2-i)
    if lo <= hi:
        shifts = np.arange(lo, hi+1) - (n-1)
        delta[lo:hi+1] += (1.0 - hb[i + shifts])
    
    # i as target: k in [i, i+n-1]
    lo = max(0, i)

        delta[lo:hi+1] -= hb[i - shifts]

    if 0 <= k2 < ng:

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
