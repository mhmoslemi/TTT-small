def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_c5(hb, n):
    # Fast computation for binary case
    c = np.correlate(hb, 1 - hb, mode='full')
    return float(np.max(c) * (2.0 / n))

def anneal_binary(hb, n, deadline, max_iters=500000, T0=0.5, T1=0.01):
    """Simulated annealing on binary solution"""
    best_hb = hb.copy()
    best_c = binary_c5(hb, n)
    cur_c = best_c
    cur_hb = hb.copy()
    ones = set(np.where(cur_hb == 1)[0].tolist())
    zeros = set(np.where(cur_hb == 0)[0].tolist())
    
    np.random.seed(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max_iters)
        if T < 0.001:
            break
        
        # Random swap
        i = np.random.choice(list(zeros))
        j = np.random.choice(list(ones))
        
        cur_hb[i] = 1.0
        cur_hb[j] = 0.0
        new_c = binary_c5(cur_hb, n)
        delta = new_c - cur_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c = new_c
            ones.add(i)
            ones.discard(j)
            zeros.discard(i)
            zeros.add(j)
            if new_c < best_c:
                best_c = new_c
                best_hb = cur_hb.copy()
        else:
            cur_hb[i] = 0.0
            cur_hb[j] = 1.0
    
    return best_hb, best_c

def sa_binary(hb, n, deadline, n_iter, T0, T1, rng):
    dx = 2.0 / n
    cur = hb.copy()
    cur_c = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) * dx
    best = cur.copy()
    best_c = cur_c
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (float(it) / float(n_iter))
        if T < 1e-6:
            T = 1e-6
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            cur[i] = 0.0
            cur[j] = 1.0
    
    return best, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):

def proj_grad(h, n, deadline, alpha=100000, max_iter=30000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute gradient of softmax-approximated max
        g_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        m = g_vals.max()
        sh = alpha * (g_vals - m)
        sh -= sh.max()
        w = np.exp(sh)
        w /= w.sum()
        
        # Gradient of the sum w[k] * corr[k] w.r.t. h
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # d corr[k] / d h[j] = (1-h[j+k-(n-1)]) - h[j-k+1-(n-1)] (for valid indices)
        # Simplifying: d corr[n-1+L] / d h[j] = (1-h[j+L]) - h[j-L] (for valid j+L, j-L)
        grad = np.zeros(n)
        for L in range(-(n-1), n):
            wk = w[n-1+L]
            if wk < 1e-12:
                continue
            for j in range(n):
                dj = 0.0
                jpL = j + L
                if 0 <= jpL < n:
                    dj += (1.0 - h[jpL])
                jL = j - L
                if 0 <= jL < n:
                    dj -= h[jL]
                grad[j] += wk * dj
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = lr / gnorm
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.85
            stall += 1
            if lr < 1e-14 or stall > 150:
                break
        else:
            stall = 0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.02, 20.0)
    return best_h, best_c
