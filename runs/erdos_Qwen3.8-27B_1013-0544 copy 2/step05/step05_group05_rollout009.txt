                best_h = h.copy()
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def fast_binary_ls(h, n, deadline):
    h = make_binary(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    pattern = 1 - 2 * h
    cur_max = float(corr.max())
    L = np.array([min(n, 2 * n - 1 - k) for k in range(n)])
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    buf = np.zeros(2 * n - 1)
    streak = 0
    while time.time() < deadline:
        best_nm = cur_max
        best_pair = None
        for i in ones:
            Li = L[i]
            for j in zeros:
                Lj = L[j]
                buf[:] = corr
                buf[i:i + Li] -= pattern[:Li]
                buf[j:j + Lj] += pattern[:Lj]
                nm = float(buf.max())
                if nm < best_nm - 1e-15:
                    best_nm = nm
                    best_pair = (i, j)
        if best_pair is None:
            if streak > 0:
                break
            streak += 1
            if streak > 3:
                break
            continue
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = buf.copy()
        pattern[i] = 1.0
        pattern[j] = -1.0
        cur_max = best_nm
        ones.remove(i)
        zeros.remove(j)
        zeros.append(i)
        ones.append(j)
        streak = 0
    return h, cur_max * (2.0 / n)

    zeros = np.where(h == 0)[0].tolist()

    n_iter = 5000000
    t0 = time.time()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / n_iter
        T = 0.3 * (0.0005 / 0.3) ** frac
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        Li = L[i]
        Lj = L[j]
        buf[:] = corr

        buf[j:j + Lj] += pattern[:Lj]
        nm = float(buf.max())
        delta = nm - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):

            pattern[i] = 1.0

def cont_refine(h, n, deadline, n_iter=50000):
    h = h.astype(float).copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [10, 50, 200, 1000, 5000, 20000]
    lr = 0.05
    for alpha in alphas:
        if time.time() > deadline:
            break
        for it in range(n_iter):
            if time.time() > deadline:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = corr.max()
            sm = m + (corr.max() - m + np.log(np.exp((corr - m) / alpha).sum())) * 0
            sh = (corr - m) / alpha
            w = np.exp(sh)
            w /= w.sum()
            pat = 1 - 2 * h
            grad = np.zeros(n)
            for p in range(n):
                Lp = min(n, 2 * n - 1 - p)
                grad[p] = np.dot(w[p:p + Lp], pat[:Lp])
            grad *= dx
            h_new = project_feasible(h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
            else:
                if it % 5000 == 0:
                    lr *= 0.5
                if lr < 1e-8:
                    break
    return best_h, best_c
