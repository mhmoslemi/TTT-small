```python
def simulated_annealing_binary(hb, n, deadline, t0=1.0, cooling=0.9999, max_iter=None):
    """SA on binary solution with 1-swap moves"""
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    current_h = hb.copy()
    current_c = best_c
    temp = t0
    ones = np.where(current_h == 1)[0]
    zeros = np.where(current_h == 0)[0]
    rng = np.random.RandomState(42)
    
    for it in range(max_iter or 10**9):
        if time.time() > deadline:
            break
        # Choose a random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Apply swap
        current_h[i] = 0
        current_h[j] = 1
        new_c = compute_c5(current_h, n)
        
        # Accept or reject
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            # Accept
            current_c = new_c
            # Update sets
            ones = np.where(current_h == 1)[0]  # This is expensive!
            zeros = np.where(current_h == 0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            # Revert
            current_h[i] = 1
            current_h[j] = 0
        
        temp *= cooling
    
    return best_h, best_c
```

```python
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def linearized_minmax(h, n, deadline, r_init=0.05, r_min=0.001, max_iter=50000):
    """Linearized min-max with adaptive trust region."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

        b = np.zeros(n_lags)

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

                b[k] += h[i] * (1 - h[j])

                A[k, j] += -h[i]

            h_var <= 1,
            cp.sum(h_var) == n / 2,
            h_var >= h - r,
            h_var <= h + r,
        ]
        for k in range(n_lags):
            constraints.append(A[k] @ h_var + b[k] * dx <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
```

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_linearization(h, n):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    b = np.zeros(n_lags)
    for k in range(n_lags):
        ell = k - (n - 1)
        i_min = max(0, -ell)
        i_max = min(n - 1, n - 1 - ell)
        i = np.arange(i_min, i_max + 1)
        ip = i + ell
        f_val = np.sum(h[i] * (1 - h[ip]))
        A[k, i] += (1 - h[ip])
        A[k, ip] += -h[i]
        b[k] = f_val - np.dot(A[k], h)
    return A, b
