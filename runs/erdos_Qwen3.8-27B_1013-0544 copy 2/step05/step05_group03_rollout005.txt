import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    return m + np.log(np.exp(shifted).sum()) / alpha

def slsqp(h, n, alpha, maxiter, deadline):
    def obj(x):
        return soft_max_obj(x, n, alpha)
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':cons}],
                      options={'maxiter':maxiter,'ftol':1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_run(hb, n, deadline, t0, cooling, seed_val):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    h = hb.copy()
    c = best_c
    t = t0
    rng = np.random.RandomState(seed_val)
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    while time.time() < deadline:
        i = rng.randint(len(zeros_list))
        j = rng.randint(len(ones_list))
        zi = zeros_list[i]
        oj = ones_list[j]
        h[zi] = 1.0
        h[oj] = 0.0
        new_c = compute_c5(h, n)
        delta = new_c - c
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            c = new_c
            zeros_list[i] = oj
            ones_list[j] = zi
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[zi] = 0.0
            h[oj] = 1.0
        t *= cooling
    return best_h, best_c

def greedy_search(hb, n, deadline, max_sweeps=100):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    h = hb.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                h[i] = 1.0
                h[j] = 0.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    zeros.discard(i)
                    zeros.add(j)
                    ones.discard(j)
                    ones.add(i)
                    improved = True
                    break
                h[i] = 0.0
                h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def perturb(hb, n, k, rng):
    start = hb.copy()
    for _ in range(k):
        oi = np.where(start == 1)[0]
        zi = np.where(start == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        o = oi[rng.randint(len(oi))]
        z = zi[rng.randint(len(zi))]
        start[o] = 0.0
        start[z] = 1.0
    return start

    p1 = t0 + 0.15 * total

        if c_try < best_c:

    # Phase 2: Initial greedy (5%)
    p2 = t0 + 0.20 * total
    hg, cg = greedy_search(best_h.copy(), n, p2, 40)
    if cg < best_c:

        best_h = hg.copy()

    # Phase 3: Iterated SA with greedy polish (60%)
    p3 = t0 + 0.80 * total
    sa_configs = [

    sa_round = 0
    while time.time() < p3 - 10:

        t_s, cool = cfg
        if sa_round == 0:
            start = best_h.copy()
        else:
            k = min(1 + sa_round // 3, 4)
            start = perturb(best_h, n, k, rng)
        h_s, c_s = sa_run(start, n, p3 - 5, t_s, cool, 1000 + sa_round * 7919)
        if c_s < best_c:
            best_c = c_s

            if cp < best_c:
                best_c = cp
                best_h = hp.copy()

            for i in range(target - n_ones):
                best_h[zi[i]] = 1.0
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
