```python
    # s = F[i] - F[j], the direction
    # g = current values of the correlation
    # Find t in [lo, hi] that minimizes max_k (g[k] + t*s[k])
    
    # At t=0, the top K indices matter
    top = np.argsort(g)[::-1][:K]
    
    # The optimal t is either lo, hi, or a crossing point of two top lines
    # Line m: g[m] + t*s[m]
    # Crossing of m1, m2: g[m1] + t*s[m1] = g[m2] + t*s[m2]
    # t = (g[m2] - g[m1]) / (s[m1] - s[m2])
    
    candidates = [lo, hi]
    for a in range(K):
        for b in range(a+1, K):
            m1, m2 = top[a], top[b]
            ds = s[m1] - s[m2]
            if abs(ds) > 1e-15:

                if lo < t_cross < hi:
                    candidates.append(t_cross)

            best_t = t
    return best_t, best_val
```

```python
def pair_descent(h, n, deadline, K=5, max_sweeps=200):
    dx = 2.0 / n
    q = 1.0 - 2.0 * h
    F = compute_F(h, n)
    g = compute_g(h, n)
    m = 2 * n - 1
    best_c = g.max()
    best_h = h.copy()
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Determine pair order
        # Bias toward pairs with large |F[i] - F[j]| on active set
        top = np.argsort(g)[::-1][:K]
        # Random order with bias
        order = np.arange(n)
        rng.shuffle(order)
        
        improved = 0
        for idx_a in range(n):
            if time.time() > deadline:
                break
            i = order[idx_a]
            for idx_b in range(idx_a + 1, n):
                j = order[idx_b]
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                # Find optimal t
                # g_new[k] = g[k] + t * s[k]
                # We want min_t max_k g_new[k]
                # Check top K + a few more
                check = np.union1d(top, np.argsort(g + d_lo * s)[::-1][:K])
                check = np.union1d(check, np.argsort(g + d_hi * s)[::-1][:K])
                
                candidates = [d_lo, d_hi]
                for a in range(len(check)):
                    for b in range(a + 1, len(check)):
                        m1, m2 = check[a], check[b]
                        ds = s[m1] - s[m2]
                        if abs(ds) > 1e-15:
                            t_cross = (g[m2] - g[m1]) / ds
                            if d_lo - 1e-15 < t_cross < d_hi + 1e-15:
                                candidates.append(t_cross)
                
                best_t = 0.0
                best_val = np.max(g + 0.0 * s)  # = g.max()
                for t in candidates:
                    t = float(np.clip(t, d_lo, d_hi))
                    val = np.max(g + t * s)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if abs(best_t) > 1e-15:
                    h[i] += best_t
                    h[j] -= best_t
                    g += best_t * s
                    update_F(F, q, i, j, best_t, n)
                    if g.max() < best_c - 1e-14:
                        best_c = g.max()
                        best_h = h.copy()
                    top = np.argsort(g)[::-1][:K]

        g = compute_g(h, n)
        c = g.max()
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            break
    return best_h, best_c
```

```python
            if abs(s[top1]) < 1e-10 and abs(s[top[1] if len(top)>1 else 0]) < 1e-10:

            check = np.argsort(g)[::-1][:K + 2]
            candidates = [d_lo, d_hi]
            nc = len(check)

                for b in range(a + 1, nc):

                        if d_lo < t_cross < d_hi:

                val = float(np.max(g + t * s))
                if val < best_val - 1e-14:

                    best_t = t

                h[j] -= best_t
                g += best_t * s

                top = np.argsort(g)[::-1][:K]
                top1 = top[0]

        F = compute_F(h, n)

        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            break
    return best_h, best_c
```

For the L-BFGS-B with penalty:

The gradient of the log-sum-exp approximation:
L(x) = max(g) + (1/alpha) * log(sum(exp(alpha * (g - max(g)))))

∂L/∂h[m] = (1/|A|) * sum_{k in A} ∂g_k/∂h[m] + softmax weights...

Actually, let me just use the softmax:
L(x) = (1/alpha) * log(sum_k exp(alpha * g_k))

∂L/∂h[m] = sum_k softmax(alpha*g)_k * ∂g_k/∂h[m]
          = sum_k softmax(alpha*g)_k * F[m,k]

where F[m,k] = ∂g_k/∂h[m] (computed as above).

This is a weighted sum of the rows of F, with weights from the softmax. Very efficient to compute!

The penalty gradient: ∂/∂h[m] of mu*(sum-n/2)^2 = 2*mu*(sum-n/2)

So total gradient: sum_k softmax_k * F[m,k] + 2*mu*(sum-n/2) for all m.

This is O(n*m) = O(n*2n) = O(n^2) per gradient evaluation, which is fine for n=250.

And L-BFGS-B only stores O(n) Hessian information, so it's memory efficient.

```python
def lbfgs_optimize(h, n, alpha, mu, deadline, maxiter=500):
    dx = 2.0 / n
    q = 1.0 - 2.0 * h
    
    def obj(x):
        gx = compute_g(x, n)
        amax = alpha * gx
        sm = amax.max()
        return (sm + np.log(np.exp(amax - sm).sum())) / alpha + mu * (x.sum() - n/2)**2
    
    def grad(x):
        gx = compute_g(x, n)
        amax = alpha * gx
        sm = amax.max()
        w = np.exp(amax - sm)
        w /= w.sum()
        # gradient = sum_k w_k * F[:, k]
        # F[m,k] = dx * (1-2*q[k-m]) for valid k-m
        # So sum_k w_k * F[m,k] = dx * sum_k w_k * q[k-m] (with sign flip)
        # = dx * (w * q)[m] (convolution-like)
        # Actually: sum_k w_k * (1-2*q[k-m]) = (1-2) ... no
        # sum_k w_k * dx * (1-2*q[k-m]) for valid k-m in [0,n)
        # = dx * (sum_k w_k - 2 * sum_k w_k * q[k-m])
        # = dx * (1 - 2 * (w conv q)[m]) ... not quite a convolution
        
        # Let's just use the F matrix approach
        qx = 1.0 - 2.0 * x
        F_x = compute_F_from_q(qx, n)  # precomputed from q
        g_obj = (F_x @ w).ravel()  # shape (n,)... wait, F is (n, m), w is (m,)
        g_obj = F_x.T @ w  # no, F[m,k] * w[k] summed over k = (F @ w)[m]
        g_obj = F_x @ w
        g_obj += 2.0 * mu * (x.sum() - n/2)
        return g_obj
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h
```

Wait, I need to be careful. `compute_F` returns F where F[m,k] = ∂g_k/∂h[m]. Then the gradient of the objective with respect to h[m] is:
∂L/∂h[m] = sum_k w_k * ∂g_k/∂h[m] + 2*mu*(sum-n/2)
          = sum_k w_k * F[m,k] + 2*mu*(sum-n/2)
          = (F @ w)[m] + 2*mu*(sum-n/2)

where F @ w is a vector of length n (matrix-vector product). This is O(n*m) = O(n*2n) per evaluation.

But I also need to recompute F each time x changes (since F depends on h). So each gradient evaluation is:
1. Compute q = 1-2*x: O(n)
2. Compute F from q: O(n^2) (but vectorizable)
3. Compute g = correlate(x, 1-x): O(n^2) (via FFT would be O(n log n))
4. Compute softmax: O(m) = O(n)
5. Matrix-vector product F @ w: O(n*m) = O(n^2)

Total: O(n^2) per gradient. With n=250, that's about 62,500 operations. L-BFGS-B might need 100-500 iterations, each with 1-10 gradient evaluations. So total: 100-5000 gradient evals × 62,500 = 6M-312M ops. Should be a few seconds.

```python
def compute_F_from_q(q, n):
    dx = 2.0 / n
    m = 2 * n - 1
    F = np.zeros((n, m))
    for i in range(n):

    return F
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def build_F(q, n):
    dx = 2.0 / n
    m = 2 * n - 1
    F = np.zeros((n, m))
    idx = np.arange(n)
    for i in range(n):
        F[i, i:i+n] = dx * q
    return F

def update_F_diag(F, q, p, dq, n):
    dx = 2.0 / n
    idx = np.arange(n)
    F[idx, idx + p] += dx * dq
    q[p] += dq

def lbfgs_phase(h, n, deadline):
    dx = 2.0 / n
    m = 2 * n - 1
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        mu = max(1.0, n * 0.1)
        def obj(x):
            gx = compute_g(x, n)
            ag = alpha * gx
            sm = ag.max()
            val = (sm + np.log(np.exp(ag - sm).sum())) / alpha
            val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad(x):
            gx = compute_g(x, n)
            ag = alpha * gx
            sm = ag.max()
            w = np.exp(ag - sm)
            w /= w.sum()
            qx = 1.0 - 2.0 * x
            Fx = build_F(qx, n)
            g_obj = Fx @ w
            g_obj += 2.0 * mu * (x.sum() - n / 2.0)
            return g_obj
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = project_feasible(res.x, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def pair_descent(h, n, deadline, K=5, full_pass=True):
    q = 1.0 - 2.0 * h
    F = build_F(q, n)
    g = compute_g(h, n)
    best_c = float(g.max())
    best_h = h.copy()
    rng = np.random.RandomState(7)
    stall = 0
    for sweep in range(10000):
        if time.time() > deadline:
            break
        top = np.argsort(g)[::-1][:K]
        improved = 0
        if full_pass:
            order = np.arange(n)
            rng.shuffle(order)
            pairs_range = list(range(n))
        else:
            pairs_range = list(range(0, n, 2))
            rng.shuffle(pairs_range)
        for ia in pairs_range:
            if time.time() > deadline:
                break
            i = order[ia] if full_pass else ia
            for ib in range(ia + 1, len(pairs_range)):
                j = order[ib] if full_pass else ib
                if j <= i:
                    continue
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                check = top.copy()
                candidates = [d_lo, d_hi]
                nc = len(check)
                for a in range(nc):
                    for b in range(a + 1, nc):
                        m1, m2 = int(check[a]), int(check[b])
                        ds = s[m1] - s[m2]
                        if abs(ds) > 1e-15:
                            tc = (g[m2] - g[m1]) / ds
                            if d_lo < tc < d_hi:
                                candidates.append(tc)
                best_t = 0.0
                best_val = float(g.max())
                for t in candidates:
                    t = float(np.clip(t, d_lo, d_hi))
                    val = float(np.max(g + t * s))
                    if val < best_val - 1e-14:

                        best_t = t
                if abs(best_t) > 1e-13:
                    h[i] += best_t
                    h[j] -= best_t
                    g += best_t * s
                    update_F_diag(F, q, i, -2.0 * best_t, n)
                    update_F_diag(F, q, j, 2.0 * best_t, n)
                    top = np.argsort(g)[::-1][:K]
                    if float(g.max()) < best_c - 1e-14:

                        best_h = h.copy()

        q = 1.0 - 2.0 * h

        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0

    zeros = set(range(n)) - ones

                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.discard(o)
                    ones.add(z)
                    zeros.discard(z)
                    zeros.add(o)
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
```
