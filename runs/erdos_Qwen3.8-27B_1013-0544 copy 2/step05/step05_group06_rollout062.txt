def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, alpha, n):
    dx = 2.0 / n
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def smooth_grad(x, alpha, n):
    dx = 2.0 / n
    g = compute_g(x, n)
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w /= w.sum()
    grad = np.zeros(n)
    nm1 = n - 1
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        d = m_idx - nm1
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wm * (1.0 - x[i_lo + d:i_hi + d])
        i_lo2 = max(0, d)
        i_hi2 = min(n, n + d)
        if i_lo2 < i_hi2:
            grad[i_lo2 - d:i_hi2 - d] -= wm * x[i_lo2:i_hi2]
    return grad * dx

def lbfgsb_solve(h, n, alphas, deadline):
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(
                fun=lambda x: smooth_obj(x, alpha, n),
                x0=best_h.copy(),
                jac=lambda x: smooth_grad(x, alpha, n),
                bounds=bounds,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            h_new = project_feasible(res.x, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def fast_pair_descent(h, n, deadline, K=12, max_sweeps=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    nm1 = n - 1
    dm = np.arange(2 * n - 1) - nm1
    m_idx = np.arange(2 * n - 1)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Compute F
        F = np.zeros((n, 2 * n - 1))
        for mk in range(2 * n - 1):
            d = dm[mk]
            i_lo = max(0, d)
            i_hi = min(n, n + d)
            if i_lo < i_hi:
                F[i_lo:i_hi, mk] += (1.0 - h[i_lo - d:i_hi - d])
            j_lo = max(0, -d)
            j_hi = min(n, n - d)
            if j_lo < j_hi:
                F[j_lo:j_hi, mk] -= h[j_lo + d:j_hi + d]
        
        g = compute_g(h, n)
        cur_max = g.max()
        top = np.argsort(g)[::-1][:K]
        improved = 0
        
        for i in range(n):
            if time.time() > deadline:
                break
            hi_val = h[i]
            for j in range(i + 1, n):
                hj_val = h[j]
                d_lo = max(-hi_val, hj_val - 1.0)
                d_hi = min(1.0 - hi_val, hj_val)
                if d_hi - d_lo < 1e-13:
                    continue
                s = F[i] - F[j]
                # Build candidate deltas
                cands = [d_lo, d_hi]
                for a in range(K):
                    for b in range(a + 1, K):
                        ta = top[a]
                        tb = top[b]
                        ds = s[tb] - s[ta]
                        if abs(ds) > 1e-15:
                            dc = (g[ta] - g[tb]) / ds
                            if d_lo + 1e-14 < dc < d_hi - 1e-14:
                                cands.append(dc)
                best_d = 0.0
                best_v = cur_max
                for dc in cands:
                    dc = float(np.clip(dc, d_lo, d_hi))
                    g_new = g + dc * s
                    v = float(g_new.max())
                    if v < best_v - 1e-13:
                        best_v = v
                        best_d = dc
                        g = g_new
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    # Update F
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i + dm[v1], m_idx[v1]] -= best_d
                    v2 = (i - dm >= 0) & (i - dm < n)
                    F[i - dm[v2], m_idx[v2]] -= best_d
                    v3 = (j + dm >= 0) & (j + dm < n)
                    F[j + dm[v3], m_idx[v3]] += best_d
                    v4 = (j - dm >= 0) & (j - dm < n)
                    F[j - dm[v4], m_idx[v4]] += best_d
                    cur_max = best_v
                    top = np.argsort(g)[::-1][:K]
                    improved += 1
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c

                for a in range(K):
                    for b in range(a + 1, K):
                        ta, tb = tp[a], tp[b]
                        ds = s[tb] - s[ta]
                        if abs(ds) > 1e-15:
                            dc = (g[ta] - g[tb]) / ds
                            if dlo + 1e-14 < dc < dhi - 1e-14:
                                cands.append(dc)
                bd, bv = 0.0, cm
                for dc in cands:
                    dc = float(np.clip(dc, dlo, dhi))
                    nv = float((g + dc * s).max())
                    if nv < bv - 1e-13:
                        bv, bd = nv, dc
                        if bv < cm - 5e-4:
                            break
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    g += bd * s
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i + dm[v1], mi[v1]] -= bd
                    v2 = (i - dm >= 0) & (i - dm < n)
                    F[i - dm[v2], mi[v2]] -= bd
                    v3 = (j + dm >= 0) & (j + dm < n)
                    F[j + dm[v3], mi[v3]] += bd
                    v4 = (j - dm >= 0) & (j - dm < n)
                    F[j - dm[v4], mi[v4]] += bd
                    cm = bv
                    tp = np.argsort(g)[::-1][:K]
                    imp += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < bc:
            bc = c
            bh = h.copy()
        if imp == 0:
            st += 1
            if st >= 3:
                break
        else:
            st = 0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    bh = h0.copy()
    bc = compute_c5(bh, n)
    # Phase 1: L-BFGS-B continuation
    p1 = t0 + 0.25 * (budget_s - 10)
    h, c = lbfgs_opt(bh.copy(), n, [30, 100, 300, 1000, 3000, 10000, 50000], p1)
    if c < bc:
        bh, bc = h, c
    # Phase 2: Pair descent
    p2 = t0 + 0.60 * (budget_s - 10)
    h, c = pair_desc(bh.copy(), n, p2, K=12, maxsw=200)
    if c < bc:
        bh, bc = h, c
    # Phase 3: Multi-restart with perturbation
    p3 = t0 + 0.85 * (budget_s - 10)
    for sc in [0.06, 0.03, 0.015, 0.005]:
        if time.time() > p3:
            break
        hp = project_feasible(bh + rng.normal(0, sc, n), n)
        sd1 = min(p3, time.time() + 0.15 * (p3 - t0))
        h2, c2 = lbfgs_opt(hp, n, [2000, 10000], sd1)
        sd2 = min(p3, time.time() + 0.12 * (p3 - t0))
        h2, c2 = pair_desc(h2, n, sd2, K=10, maxsw=40)
        if c2 < bc:
            bh, bc = h2, c2
    # Phase 3b: Try different resolution
    if time.time() < p3 - 20:
        n2 = 200
        xc = np.linspace(0, 2, n, endpoint=False)
        xf = np.linspace(0, 2, n2, endpoint=False)
        h_alt = np.interp(xf, xc, bh)
        h_alt = project_feasible(h_alt, n2)
        sd3 = min(p3, time.time() + 0.15 * (p3 - t0))
        h3, c3 = lbfgs_opt(h_alt, n2, [1000, 5000], sd3)
        sd4 = min(p3, time.time() + 0.12 * (p3 - t0))
        h3, c3 = pair_desc(h3, n2, sd4, K=12, maxsw=40)
        if c3 < bc:
            bh, bc, n = h3, c3, n2
    # Phase 4: Final polish
    if time.time() < dl - 5:
        h, c = pair_desc(bh.copy(), n, dl - 3, K=12, maxsw=50)
        if c < bc:
            bh, bc = h, c
    bh = project_feasible(bh, n)
    bc = compute_c5(bh, n)
    return bh, bc, n
