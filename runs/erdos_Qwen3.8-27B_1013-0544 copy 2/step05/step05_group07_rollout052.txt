import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, deadline, rng, max_iter=None):
    """Simulated annealing on binary vectors with exactly n/2 ones."""
    m = n // 2
    h = h_init.copy()
    # Make binary
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:m]] = 1.0
    
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    cur_c = best_c
    cur_h = h_bin.copy()
    
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    
    T = 0.05
    T_min = 1e-7
    cooling = 0.9999
    
    if max_iter is None:
        max_iter = 1000000
    
    it = 0
    while it < max_iter and time.time() < deadline:
        # Batch for speed: do a few iterations at once
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h_bin[i] = 1.0
        h_bin[j] = 0.0
        new_c = compute_c5(h_bin, n)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        
        T = max(T * cooling, T_min)
        it += 1
    
    return best_h, best_c

def lbfgs_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g *= dx
        g += 2e5 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def block_swap_sa(h_init, n, deadline, rng):

    m = n // 2

    cur_c = best_c

    T_min = 1e-7

        # Find a random position for the 1-block

        zeros_pos = np.where(cur_h == 0)[0]
        
        if len(ones_pos) < blk or len(zeros_pos) < blk:

        # Pick a random consecutive block of 1s

        block_1 = [start_1 + d for d in range(blk) if start_1 + d < n]
        # Make sure they're all 1s

            it += 1

        # Pick a random consecutive block of 0s

        # Swap

            cur_h[p] = 0
        for p in block_0:
            cur_h[p] = 1
        
        new_c = compute_c5(cur_h, n)

        else:

                cur_h[p] = 0

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    m = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:m]] = 1.0
    return h_bin

def fast_binary_sa(h_init, n, deadline, rng, T_init=0.03):
    h = make_binary(h_init, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T_init
    T_min = 1e-9
    cooling = 0.99999
    no_improve = 0
    restart_every = 400000
    while time.time() < deadline:
        i_idx = rng.randint(len(zeros))
        j_idx = rng.randint(len(ones))
        i = int(zeros[i_idx])
        j = int(ones[j_idx])
        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0.0:
            accept = True
        elif delta > 50.0 * T:
            accept = False
        else:
            accept = (rng.random() < np.exp(-delta / T))
        if accept:
            cur_c = new_c
            zeros[i_idx] = j
            ones[j_idx] = i
            no_improve += 1
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = 0.0
            h[j] = 1.0
        T = max(T * cooling, T_min)
        if no_improve > restart_every:
            h = best_h.copy()
            n_pert = max(n // 8, 5)
            for _ in range(n_pert):
                oc = np.where(h == 1)[0]
                zc = np.where(h == 0)[0]
                if len(oc) > 0 and len(zc) > 0:
                    h[zc[rng.randint(len(zc))]] = 1.0
                    h[oc[rng.randint(len(oc))]] = 0.0
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            cur_c = compute_c5(h, n)
            T = T_init * 0.4
            no_improve = 0
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 1e6 * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n - k])
        g[:n - k] += -x[k:]
    else:
        g[:n + k] += (1.0 - x[-k:])
        g[-k:] += -x[:n + k]
    g *= dx
    g += 2e6 * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alphas):
    best = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        try:
            fun = partial(smooth_obj, n=n, alpha=alpha)
            jac = partial(smooth_grad, n=n, alpha=alpha)
            res = minimize(fun, best.copy(), method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-11})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best = h_new.copy()
        except Exception:
            pass
    return best, best_c

def coordinate_descent(h, n, deadline, step):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    for _ in range(10):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 1:
                break
            for j in range(n - 1, i, -1):
                for s in (1.0, -1.0):
                    hi = h[i] + s * step
                    hj = h[j] - s * step
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - s * step, hj + s * step
        if not improved:
            break
    return best_h, best_c
