def compute_gradient_fast(h, w, n):
    # grad[j] = sum_{k=0}^{n-1-j} w[k]*(1-h[j+k]) - sum_{k=1}^{j} w[k]*h[j-k]
    # First part: sum_{k=0}^{n-1-j} w[k] - sum_{k=0}^{n-1-j} w[k]*h[j+k]
    # = W_prefix[n-j] - (h cross-correlate w)[j]
    # Second part: -(convolve(w, h))[j] + w[0]*h[j]  (for j >= 1)
    
    W_prefix = np.cumsum(w)  # W_prefix[i] = sum_{k=0}^{i} w[k]
    # W_prefix[n-1-j] = sum_{k=0}^{n-1-j} w[k]
    
    # conv1[j] = sum_{k=0}^{n-1-j} h[j+k]*w[k] = np.correlate(h, w, 'valid')[j]
    conv1 = np.correlate(h, w, mode='valid')  # length n
    
    # conv2[j] = sum_{k=1}^{j} w[k]*h[j-k]
    # np.convolve(w, h, 'full')[j] = sum_{k=0}^{min(j,n-1)} w[k]*h[j-k] (for j < n)
    conv_full = np.convolve(w, h, mode='full')  # length 2n-1
    conv2 = conv_full[:n].copy()
    conv2[np.arange(n)] -= w[0] * h  # subtract k=0 term
    conv2[0] = 0  # no k>=1 terms for j=0
    
    # First part
    W_terms = W_prefix[n-1 - np.arange(n)]  # sum_{k=0}^{n-1-j} w[k] for each j
    first_part = W_terms - conv1
    
    # Second part (negative)
    second_part = -conv2
    
    return first_part + second_part

    best_c = compute_c5(h, n)

        c = compute_c_arr(h, n)
        c_max = c.max()

        if c_max < best_c - 1e-15:
            best_c = c_max * (2.0/n)

        alpha = alpha_init * (alpha_final / alpha_init) ** progress

        shifted = alpha * (c - c_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:

        else:
            w = w / w_sum

            if c_new.max() < c_max * 0.99:
                lr = min(lr * 1.1, 0.01)
        
        n_iter += 1
    
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, w, n):
    """Gradient of smooth max over full correlation. O(n^2) direct."""
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            if k >= 0:
                if k <= j:
                    g += w[m] * (1.0 - h[j - k])
                if j <= n - 1 - k:
                    g -= w[m] * h[j + k]
            else:
                kp = -k
                if j <= n - 1 - kp:
                    g += w[m] * (1.0 - h[j + kp])
                if kp <= j:
                    g -= w[m] * h[j - kp]
        grad[j] = g
    return grad

def nesterov_gradient(h, n, deadline, alpha_start=200, alpha_end=50000, lr_start=0.002):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    momentum = np.zeros(n)
    beta = 0.85
    lr = lr_start
    n_iter = 0
    max_iter = 8000
    while time.time() < deadline and n_iter < max_iter:
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        if c_max * (2.0 / n) < best_c - 1e-15:
            best_c = float(c_max * 2.0 / n)
            best_h = h.copy()
        progress = min(1.0, n_iter / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        shifted = alpha * (c - c_max)
        mx = shifted.max()
        if mx > 400:
            shifted -= mx
        exp_vals = np.exp(shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = exp_vals / w_sum
        grad = compute_grad(h, w, n)
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if c_new.max() > c_max * 1.08 + 1e-10:
            lr = max(lr * 0.5, 1e-9)
            momentum *= 0.5
        else:
            h = h_new
            if n_iter % 50 == 0:
                lr = min(lr * 1.02, 0.02)
        n_iter += 1
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 400:
            sh -= mx
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search(h, n, deadline, base_delta=0.001):
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [base_delta * (0.5 ** i) for i in range(10)]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        passes = 0
        while improved and time.time() < deadline and passes < 500:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    for sign in [1, -1]:
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
    return h, best_c

def local_search_near(h, n, deadline, base_delta=0.001, radius=60):
    best_c = compute_c5(h, n)
    h = h.copy()
    deltas = [base_delta * (0.5 ** i) for i in range(8)]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        passes = 0
        while improved and time.time() < deadline and passes < 300:
            improved = False
            passes += 1
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                jlo = max(0, i - radius)
                jhi = min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    for sign in [1, -1]:
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
    return h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:

    # Phase 3: Another gradient pass from refined solution
    if time.time() < deadline - 300:
        h2, c2 = nesterov_gradient(best_h, n, deadline - 280, alpha_start=1000, alpha_end=100000, lr_start=0.001)
