def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def compute_grad_max(h, n):
    """Compute subgradient of max_k g_k(h) w.r.t. h"""
    g = compute_g(h, n)
    m = g.max()
    k = int(np.argmax(g))
    lag = k - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += dx * (1.0 - h[:n - lag])
        grad[:n - lag] -= dx * h[lag:]
    else:
        to = -lag
        grad[:n - to] += dx * (1.0 - h[to:])
        grad[to:] -= dx * h[:n - to]
    return grad, m

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s
        else:
            mask = h > 1e-15
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s * h[mask] / max(h[mask].sum(), 1e-15)
    return np.clip(h, 0.0, 1.0)

    y = h0.copy()
    t = 1.0

    L = 2.0 / n  # crude Lipschitz constant
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute gradient at y

        # FISTA update

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = h[mask] * (target - h[~mask].sum()) / h[mask].sum() if h[mask].sum() > 0 else h[mask]
    return np.clip(h, 0.0, 1.0)

def pair_cd(h, n, deadline, n_pairs_per_sweep=100, n_sweeps=50, seed=42):
    """Pair coordinate descent with ternary search"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g = compute_g(best_h, n)
        
        for _ in range(n_pairs_per_sweep):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            # Compute D(k) = effect per unit delta
            # A(k) for position i, B(k) for position j
            # D = A - B
            # A[k_offset] = (1-h[i+lag]) if i+lag<n else 0, minus h[i-lag] if i-lag>=0
            # where lag = k - (n-1), k is the index in g
            
            # For g indexed 0 to 2n-2, lag = idx - (n-1)
            lags = np.arange(2*n-1) - (n-1)  # range from -(n-1) to n-1
            
            # A(lag) for coord i:
            # (1-h[i+lag]) if 0<=i+lag<n (as the h[m] term)
            # - h[i-lag] if 0<=i-lag<n (as the (1-h[m+lag]) term)
            A = np.zeros(2*n-1)
            mask1 = (i + lags >= 0) & (i + lags < n)
            A[mask1] += (1.0 - best_h[i + lags[mask1]])
            mask2 = (i - lags >= 0) & (i - lags < n)
            A[mask2] -= best_h[i - lags[mask2]]
            
            B = np.zeros(2*n-1)
            mask1 = (j + lags >= 0) & (j + lags < n)
            B[mask1] += (1.0 - best_h[j + lags[mask1]])
            mask2 = (j - lags >= 0) & (j - lags < n)
            B[mask2] -= best_h[j - lags[mask2]]
            
            D = A - B
            
            # Find range of delta
            s = best_h[i] + best_h[j]
            lo = max(-best_h[i], best_h[j] - 1.0)
            hi = min(1.0 - best_h[i], best_h[j])
            if hi - lo < 1e-12:
                continue
            
            # Ternary search for min of max_k (g[k] + delta * D[k])
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = np.max(g + c_ * D)
            fd = np.max(g + d_ * D)
            
            for _ in range(50):
                if fc < fd:
                    b = d_; d_ = c_; fd = fc
                    c_ = b - gr * (b - a)
                    fc = np.max(g + c_ * D)
                else:
                    a = c_; c_ = d_; fc = fd
                    d_ = a + gr * (b - a)
                    fd = np.max(g + d_ * D)
            
            delta_opt = (a + b) / 2.0
            new_g = g + delta_opt * D
            new_c = np.max(new_g) * (2.0 / n)
            
            if new_c < best_c - 1e-14:
                best_h[i] += delta_opt
                best_h[j] -= delta_opt
                g = new_g  # Update g (approximately - ignores k=0 quadratic)
                best_c = new_c
                improved = True
        
        if not improved and sweep > 5:
            break
        
        # Recompute g exactly periodically
        if sweep % 10 == 9:
            g = compute_g(best_h, n)
            best_c = compute_c5(best_h, n)
    
    return best_h, best_c

def active_subgrad(h, n, deadline, max_iter=50000, eps=0.001):
    """Subgradient using active set (all lags within eps of max)"""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(123)
    lr = 0.01
    momentum = np.zeros(n)
    rho = 0.5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 5000 == 0:
            lr = max(0.001, lr * 0.9)
        
        g = compute_g(best_h, n) * (2.0 / n)
        m = g.max()
        
        # Active set: lags within eps of max
        active = np.where(g >= m - eps)[0]
        
        # Averaged subgradient over active set
        lags = active - (n - 1)
        grad = np.zeros(n)
        for lag in lags:
            if lag >= 0:
                grad[lag:] += (1.0 - best_h[:n-lag])
                grad[:n-lag] -= best_h[lag:]
            else:
                to = -lag
                grad[:n-to] += (1.0 - best_h[to:])
                grad[to:] -= best_h[:n-to]
        grad /= (len(active) * (2.0/n))
        
        # Momentum step
        momentum = rho * momentum + (1-rho) * grad
        best_h = project_feasible(best_h - lr * momentum, n)
        
        c = compute_c5(best_h, n)
        if c < best_c:
            best_c = c
    
    return best_h, best_c

def slsqp_smooth(h0, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        mg = g.max()
        s = g - mg
        return mg + np.log(np.exp(s - s.max()).sum() / len(s) + np.exp(s.max()-s.max())) / alpha
    
    # Simpler: log-sum-exp / alpha + max
    def obj2(x):
        g = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        mg = g.max()
        s = g - mg
        return mg + (s.max() + np.log(np.exp(s-s.max()).sum())) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj2, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h0

def binary_ls(h0, n, deadline, seed=99):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    rng = np.random.RandomState(seed)
    
    for sweep in range(100000):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(hb == 0)[0]
        ones = np.where(hb == 1)[0]
        rng.shuffle(zeros)
        rng.shuffle(ones)
        
        for i in zeros[:50]:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = best_c
            for j in ones[:50]:
                hb[i] = 1.0; hb[j] = 0.0
                c = compute_c5(hb, n)
                if c < best_nc:
                    best_nc = c; best_j = j
                hb[i] = 0.0; hb[j] = 1.0
            if best_j >= 0:
                hb[i] = 1.0; hb[best_j] = 0.0
                best_c = best_nc
                improved = True
                break
        if not improved:
            break
    return hb, best_c

Wait, I have a bug in the pair_cd function. When I compute D = A - B, I need to be careful about the indexing.

Let me re-derive: g is indexed 0 to 2n-2, where index idx corresponds to lag = idx - (n-1).

g[idx] = sum_{m: valid} h[m] * (1-h[m + lag])

where lag = idx - (n-1), and "valid" means 0 ≤ m < n and 0 ≤ m+lag < n.

If lag ≥ 0: valid m are 0 to n-1-lag
If lag < 0: valid m are -lag to n-1 (i.e., m+lag ≥ 0 means m ≥ -lag)

Now, h[i] appears in g[idx] when:
1. m = i and 0 ≤ i + lag < n (i.e., -i ≤ lag < n-i)
   Contribution: h[i] * (1-h[i+lag])
   Derivative w.r.t. h[i]: (1-h[i+lag])

2. m + lag = i, i.e., m = i - lag, and 0 ≤ i-lag < n and 0 ≤ i < n (the latter is always true)
   Also need 0 ≤ m < n: 0 ≤ i-lag < n
   And the "valid" condition: if lag ≥ 0, need m ≤ n-1-lag, i.e., i-lag ≤ n-1-lag, i.e., i ≤ n-1 (always true)
   If lag < 0, need m ≥ -lag, i.e., i-lag ≥ -lag, i.e., i ≥ 0 (always true)
   Contribution: h[i-lag] * (1-h[i])
   Derivative w.r.t. h[i]: -h[i-lag]

So A(idx) = derivative of g[idx] w.r.t. h[i]:
- Term 1: (1-h[i+lag]) if -i ≤ lag < n-i
- Term 2: -h[i-lag] if 0 ≤ i-lag < n

For the full index range, lag goes from -(n-1) to n-1.

Term 1 valid when: lag ≥ -i and lag ≤ n-1-i
Term 2 valid when: lag ≤ i and lag ≥ i-(n-1) = i-n+1

Since i is in [0, n-1]:
- Term 1: lag ∈ [-i, n-1-i]
- Term 2: lag ∈ [i-n+1, i]

In terms of idx = lag + (n-1):
- Term 1: idx ∈ [n-1-i, 2n-2-i]
- Term 2: idx ∈ [i+1, n-1+i]

Let me verify with a small example. n=5, i=2.
- Term 1: idx ∈ [5-1-2, 10-2-2] = [2, 6]
- Term 2: idx ∈ [3, 5-1+2] = [3, 6]

Let me verify: for lag = -2 (idx = 2):
g[2] = g(lag=-2) = sum_{m=2}^{4} h[m]*(1-h[m-2])
h[2] appears as m=2: h[2]*(1-h[0]), derivative = (1-h[0]) ✓ (term 1, since i+lag = 2-2 = 0, valid)
h[2] appears as m+lag=2 → m=4: h[4]*(1-h[2]), derivative = -h[4] ✓ (term 2, since i-lag = 2-(-2) = 4, valid)

For lag = 2 (idx = 6):

h[2] appears as m=2: h[2]*(1-h[4]), derivative = (1-h[4]) ✓ (term 1, i+lag = 4, valid)
h[2] appears as m+lag=2 → m=0: h[0]*(1-h[2]), derivative = -h[0] ✓ (term 2, i-lag = 0, valid)
