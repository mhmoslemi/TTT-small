- The slope changes from negative to positive
- I.e., the maximum line switches from one with positive slope to one with negative slope

A better approach: find all pairs of lines that intersect at the minimum of the upper envelope. But that's O(n²) which is expensive.

Alternative: use the fact that at the optimum, there exist two (or more) active constraints k1, k2 such that:
- c_k1 + t*g_k1 = c_k2 + t*g_k2 (they're tied for max)
- g_k1 and g_k2 have opposite signs (or one is zero)

So t* = (c_k2 - c_k1) / (g_k1 - g_k2) for the active pair.

Wait, if sg > 0 (the active constraint has positive slope), then increasing t would increase f, so we should go left (decrease t). If sg < 0, we should go right. This is correct for bisection on the subgradient.

But there's a subtlety: the subgradient at a point is the set of g_k for all k that achieve the max. So if multiple k are tied, the subgradient is the set {g_k : k in argmax}. The condition for optimality is 0 ∈ conv({g_k : k in argmax}).

For bisection, I'll use: if max of active g_k's < 0, go right; if min of active g_k's > 0, go left; otherwise, we're at the optimum.

    if time.time() < deadline - 4:
        h, best_c = coordinate_descent(h, n, max_sweeps=1500,
                                       deadline=deadline - 2, randomized=True, rng=rng)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(125, 0.5)
        n_init = 125

    best_h = None
    best_c = float('inf')

    # Main optimization on initial construction
    alloc1 = min(250, (deadline - t0) * 0.35)
    h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Try other n values by interpolation
    for n_try in [125, 150, 200]:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(130, (deadline - time.time()) * 0.25)
        if alloc < 50:
            break
        h_opt, c_opt = full_optimize(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Multi-restart with structured perturbation
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        n_restarts = max(4, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.015 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(70, (deadline - time.time() - 25) * 0.3)
            if alloc < 25:
                break
            h_opt, c_opt = full_optimize(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final intensive polish
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h, best_c = coordinate_descent(best_h, n, max_sweeps=3000,
                                       deadline=deadline - 8, randomized=True, rng=rng)
        if time.time() < deadline - 4:
            ht = slsqp_optimize(h, n, 500000, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                h = ht
        if time.time() < deadline - 2:
            h, best_c = coordinate_descent(h, n, max_sweeps=2000, deadline=deadline - 1)
        best_h = project_feasible(h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
