    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, n, alpha):
    """Gradient of (1/alpha)*log(sum exp(alpha*corr_k))"""
    corr = np.correlate(h, 1 - h, mode='full')
    a = alpha * corr
    a_max = a.max()
    e = np.exp(a - a_max)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        l = k - (n - 1)
        if l >= 0:
            grad[l:] += wk * (1.0 - h[:n-l])
            if n-l > 0:
                grad[:n-l] -= wk * h[l:]
        else:
            lp = -l
            if n-lp > 0:
                grad[:n-lp] += wk * (1.0 - h[lp:])
                grad[lp:] -= wk * h[:n-lp]
    return grad

def projected_grad_descent(h, n, alpha, lr, n_steps, deadline):
    """Projected gradient descent on smooth max with projection"""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_smooth(h, n, alpha)
        # project gradient step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            # decrease learning rate slightly
        else:
            # backtracking
            lr *= 0.5
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                h = h_new
            if lr < 1e-10:
                break
    return best_h, best_c

        if mx > 500:

        return m + np.log(np.sum(np.exp(shifted))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = corr.max()
    s = alpha * (corr - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def grad_smooth(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    a = alpha * corr
    am = a.max()
    e = np.exp(a - am)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        l = k - (n - 1)
        if l >= 0:
            if n - l > 0:
                grad[l:] += wk * (1.0 - h[:n - l])
                grad[:n - l] -= wk * h[l:]
        else:
            lp = -l
            if n - lp > 0:
                grad[:n - lp] += wk * (1.0 - h[lp:])
                grad[lp:] -= wk * h[:n - lp]
    return grad

def lfbbg(h, n, alpha, pen, maxiter=5000):
    target = n / 2.0
    dx = 2.0 / n
    def f(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        val = smooth_max_val(corr, alpha)
        cv = x.sum() - target
        return val + pen * cv * cv
    def g(x):
        gs = grad_smooth(x, n, alpha) * dx
        cv = x.sum() - target
        return gs + 2.0 * pen * cv * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, method='L-BFGS-B', bounds=bounds, jac=g,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_opt(h, n, alpha, maxiter=2000):
    target = n / 2.0
    dx = 2.0 / n
    def f(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max_val(corr, alpha)
    def g(x):
        return grad_smooth(x, n, alpha) * dx
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target,
           'jac': lambda x: np.ones(n)}
    try:
        res = minimize(f, h, method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-14}, jac=g)
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pgd(h, n, alpha, lr_init, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    dx = 2.0 / n
    for step in range(8000):
        if time.time() > deadline:
            break
        g = grad_smooth(h, n, alpha) * dx
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-13:
            break
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c

def near_pair(h, n, deadline, radius=60):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c
