import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_loop(h, n, deadline, max_iters=2000000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step_size = 0.02
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        k_star = int(np.argmax(c))
        K = k_star
        idx = np.arange(n)
        p1 = idx + K - n + 1
        p2 = idx - K + n - 1
        valid1 = (p1 >= 0) & (p1 < n)
        valid2 = (p2 >= 0) & (p2 < n)
        grad = np.zeros(n)
        grad[valid1] += (1.0 - h[p1[valid1]])
        grad[valid2] -= h[p2[valid2]]
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn
        improved = False
        for ss_mult in [1.0, 0.5, 0.2, 0.1]:
            ss = step_size * ss_mult
            trial = h - ss * grad
            trial = project_feasible(trial, n)
            tc = compute_c5(trial, n)
            if tc < best_c:
                best_c = tc
                best_h = trial.copy()
                h = trial.copy()
                improved = True
                break
            elif tc < compute_c5(h, n):
                h = trial.copy()
                improved = True
                break
        if improved:
            stall = 0
            step_size = min(step_size * 1.05, 0.08)
        else:
            stall += 1
            step_size *= 0.6
            if step_size < 1e-7:
                break
            if stall > 500:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=50, max_passes=300):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for d in deltas:
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                j_lo = max(0, i - radius)
                j_hi = min(n, i + radius + 1)
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return best_h, best_c

def multi_subgrad_restart(h, n, deadline, rng, n_restarts=3):
    best_h, best_c = h.copy(), compute_c5(h, n)
    per_restart = max(20, (deadline - time.time()) / n_restarts)
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        if r == 0:
            h_cur = h.copy()
        else:
            scale = 0.005 * (r + 1) * 0.5
            h_cur = h + rng.normal(0, scale, len(h))
            h_cur = project_feasible(h_cur, n)
        h_sub, c_sub = subgradient_loop(h_cur, n, time.time() + per_restart)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub.copy()
    return best_h, best_c

    return best_h, best_c, n
