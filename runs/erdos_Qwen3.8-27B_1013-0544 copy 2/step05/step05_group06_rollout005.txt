    p /= p.sum()

        if pk < 1e-15:
            continue

        # ∂g[k]/∂h[j] = dx * [(1-h[j-c]) if 0<=j-c<n else 0] - dx * [h[j+c] if 0<=j+c<n else 0]
        # Term 1: (1-h[j-c]) for 0<=j-c<n => c<=j<n+c
        if c >= 0:
            # j from c to n-1: h[j-c] = h[0..n-1-c]
            grad[c:] += pk * dx * (1.0 - h[:n-c])
        else:
            # j from 0 to n-1-c: h[j-c] = h[-c..n-1]
            grad[:n+c] += pk * dx * (1.0 - h[-c:])
        # Term 2: -h[j+c] for 0<=j+c<n => -c<=j<n-c
        if c >= 0:
            # j from 0 to n-1-c: h[j+c] = h[c..n-1]
            grad[:n-c] -= pk * dx * h[c:]
        else:
            # j from -c to n-1: h[j+c] = h[0..n-1+c]
            grad[-c:] -= pk * dx * h[:n+c]
    
    return grad
