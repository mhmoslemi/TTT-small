    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        # Softmax approximation of max
        shifted = alpha * (g - g.max())
        return g.max() + np.log(np.exp(shifted).sum()) / alpha

            # Score for j: sum over active k of (D[i,k] - D[j,k]) * g[k] (positive = good)
            # Or more simply: D[j, argmax_g] should be small (or D[i, argmax_g] should be large)
            
            k_star = top_idx[0]  # Most active constraint
            # For direction 1 (move from i to j): want D[j, k_star] < D[i, k_star]
            # For direction 2 (move from j to i): want D[i, k_star] < D[j, k_star]
            
            # Find best j for direction 1:
            scores1 = D[i, k_star] - D[:, k_star]  # positive means j is better
            # Also consider other active constraints
            for kk in top_idx[1:min(3, K)]:
                scores1 += D[i, kk] - D[:, kk]
            scores1[i] = -np.inf
            top_j_idx1 = np.argsort(scores1)[::-1][:top_j]
            
            # Find best j for direction 2:
            scores2 = D[:, k_star] - D[i, k_star]
            for kk in top_idx[1:min(3, K)]:
                scores2 += D[:, kk] - D[i, kk]
            scores2[i] = -np.inf
            top_j_idx2 = np.argsort(scores2)[::-1][:top_j]
            
            # Try all candidate (i, j) pairs
            candidates_j = set(top_j_idx1.tolist() + top_j_idx2.tolist())
            for j in candidates_j:
                if j == i:
                    continue
                # Both directions
                for direction in [1, -1]:
                    if direction == 1:
                        # h[i] -= delta, h[j] += delta
                        s = D[j] - D[i]
                        d_lo = 0.0
                        d_hi = min(h[i], 1.0 - h[j])
                    else:
                        # h[i] += delta, h[j] -= delta
                        s = D[i] - D[j]
                        d_lo = 0.0
                        d_hi = min(1.0 - h[i], h[j])
                    
                    if d_hi < 1e-14:
                        continue
                    
                    # Find best delta
                    best_val = current_max
                    best_d = 0.0
                    for d in [d_lo, d_hi]:
                        val = np.max(g + d * s)
                        if val < best_val:
                            best_val = val
                            best_d = d
                    for a in range(K):
                        for b in range(K):
                            if a == b:
                                continue
                            m1, m2 = top_idx[a], top_idx[b]
                            ds = s[m1] - s[m2]
                            if abs(ds) < 1e-15:
                                continue
                            dc = (g[m2] - g[m1]) / ds
                            if d_lo <= dc <= d_hi:
                                val = np.max(g + dc * s)
                                if val < best_val:
                                    best_val = val
                                    best_d = dc
                    
                    if best_d > 0 and best_val < current_max - 1e-13:
                        if direction == 1:
                            h[i] -= best_d
                            h[j] += best_d
                        else:
                            h[i] += best_d
                            h[j] -= best_d
                        g += best_d * s
                        current_max = g.max()
                        top_idx = np.argsort(g)[::-1][:K]
                        sweep_improved += 1
        
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        if sweep_improved == 0:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_D(h, n):
    num_lags = 2 * n - 1
    dm = np.arange(num_lags) - (n - 1)
    D = np.zeros((n, num_lags))
    for k in range(num_lags):
        d = dm[k]
        if d == 0:
            D[:, k] = 1.0 - 2.0 * h
        else:
            i_lo1 = max(0, -d)
            i_hi1 = min(n, n - d)
            if i_lo1 < i_hi1:
                D[i_lo1:i_hi1, k] += (1.0 - h[i_lo1 + d:i_hi1 + d])
            i_lo2 = max(0, d)
            i_hi2 = min(n, n + d)
            if i_lo2 < i_hi2:
                D[i_lo2:i_hi2, k] -= h[i_lo2 - d:i_hi2 - d]
    return D

def smooth_max_val(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    shifted = alpha * (g - g.max())
    return g.max() + np.log(np.exp(shifted).sum()) / alpha

def smooth_max_grad(x, n, alpha):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    shifted = alpha * (g - g.max())
    e = np.exp(shifted)
    w = e / e.sum()
    D = compute_D(x, n)
    return D.T @ w * (2.0 / n)

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        shifted = alpha * (g - g.max())
        return g.max() + np.log(np.exp(shifted).sum()) / alpha

    def grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        shifted = alpha * (g - g.max())
        e = np.exp(shifted)
        w = e / e.sum()
        D = compute_D(x, n)
        return D.T @ w

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def full_pair_descent(h, n, deadline, max_sweeps=200, K=6):

    for sweep in range(max_sweeps):
