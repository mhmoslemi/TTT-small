import cvxpy as cp

def run_packing():
    n = 26
    # Initial positions using hexagonal packing
    centers = []
    for r in range(int(np.sqrt(n)) + 1):
        for c in range(int(np.sqrt(n)) + 1):
            x = c + r / 2
            y = r
            if x < n and y < n:
                centers.append([x, y])

    # Verify we have enough points
    if len(centers) < n:
        for i in range(n - len(centers)):
            centers.append([i, i])

    centers = np.array(centers[:n]) / (np.max(centers) + 1e-12)

    # Define variables
    radii = cp.Variable(n)
    # All radii must be non-negative
    constraints = [radii >= 0]

    # Distance constraint between all pairs of circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            # Ensure circles do not overlap
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Ensure circles stay within the unit square
    for i in range(n):
        x, y = centers[i]
        # Left boundary
        constraints.append(x - radii[i] >= -1e-12)
        # Right boundary
        constraints.append(x + radii[i] <= 1 + 1e-12)
        # Bottom boundary
        constraints.append(y - radii[i] >= -1e-12)
        # Top boundary
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # Maximize the sum of the radii
    obj = cp.Maximize(cp.sum(radii))

    # Formulate and solve the problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS)

    if not prob.status == 'optimal':
        raise RuntimeError(f"Optimization failed: {prob.status}")

    # Convert to NumPy arrays
    centers = np.array(centers)
    radii = np.array(radii.value).flatten()

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
