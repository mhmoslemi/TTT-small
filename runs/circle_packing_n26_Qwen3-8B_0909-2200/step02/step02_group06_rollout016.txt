import numpy as np
import math
from scipy.optimize import minimize

def run_packing():

    # This arrangement will fit 26 circles, with the radii computed as a function of the distance between circles and the boundary.

    # Define initial centers and radii
    centers = []
    radii = []

    # Row spacing in hexagonal packing
    h = math.sqrt(3) / 2  # Height between rows
    # We'll use a vertical spacing slightly smaller than h to allow for more rows
    vertical_spacing = h - 0.01  # A small reduction to fit more rows

    # Number of circles per row
    row_sizes = [5, 5, 5, 5, 4, 2]  # Total = 26
    row_centers = []
    
    # Generate the coordinates for the hexagonal packing
    for i in range(len(row_sizes)):
        row = []
        for j in range(row_sizes[i]):
            x = 0.05 + j * (0.90 / (row_sizes[i] - 1)) if row_sizes[i] > 1 else 0.45
            y = i * vertical_spacing + 0.1 if i % 2 == 0 else i * vertical_spacing + 0.1 + 0.05
            row.append((x, y))
        row_centers.append(row)

    # Extract all centers and compute radii
    for row in row_centers:
        for x, y in row:
            centers.append([x, y])

    # Now compute optimal radii for all circles
    # Place a few circles near corners and edges to maximize radii

    # Function to compute the total sum of radii
    def total_radius(radii):
        total = 0
        for i in range(len(radii)):
            # Use a simple radius based on distance to nearest boundary and nearest neighbor
            min_dist_to_boundary = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            for j in range(len(radii)):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = math.hypot(dx, dy)
                    min_dist_to_neighbor = dist - (radii[i] + radii[j])  # For optimization, we can assume radii[j] is similar to radii[i]
            radius = min(min_dist_to_boundary, min_dist_to_neighbor)
            total += radius
        return -total  # Negative because we are minimizing

    # Optimization
    initial_guess = [0.05 for _ in range(26)]
    result = minimize(total_radius, initial_guess, method='L-BFGS-B', bounds=[(0.01, 0.5) for _ in range(26)])

    # Return the centers and optimal radii
    centers = np.array(centers)
    radii = np.array(optimal_radii)
