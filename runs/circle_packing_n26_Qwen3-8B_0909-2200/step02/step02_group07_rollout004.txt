import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    n = 26
    # Initial guess for centers: Fixed hexagonal pattern with positions (x, y) in [0, 1]^2
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        # Place centers with hexagonal packing
        if row % 2 == 0:
            x = col * (1.0 / (5)) + 0.05

            x = (col + 0.5) * (1.0 / (5)) - 0.05

        centers[i] = [x, y]

    # Define objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because scipy minimize

    # Define constraints: circles must not overlap, must be inside the square
    def constraint_overlaps(radii):

        distances = squareform(pdist(centers))

            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(n):

    res = minimize(
