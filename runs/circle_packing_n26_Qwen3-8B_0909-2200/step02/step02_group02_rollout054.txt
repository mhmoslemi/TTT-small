import numpy as np

def run_packing():

    centers = []
    radii = []

    # Helper function to compute distance squared between two points
    def dist_sq(p1, p2):
        return np.sum((np.array(p1) - np.array(p2)) ** 2)

    # Initialize first circle in the center
    x = 0.5
    y = 0.5
    centers.append([x, y])
    radii.append(0.0)

    for _ in range(1, n):
        best_radius = 0.0
        best_pos = None

        # Try placing the new circle in the center of the available space
        try_pos = [0.5, 0.5]

            dist = np.sqrt(dist_sq(try_pos, centers[i]))

            if max_radius > best_radius:
                best_radius = max_radius
                best_pos = try_pos

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
