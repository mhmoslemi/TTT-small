import numpy as np
from scipy.optimize import minimize

def run_packing():

    # We'll use a nonlinear optimization approach to maximize the sum of radii

    # Initial guess for the centers (hexagonal grid with some randomness)
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26)

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints:
    # 1. Boundary: Each circle must be fully contained in [0, 1] x [0, 1]
    # 2. Non-overlap: For all i != j, distance between centers >= radii[i] + radii[j]

    # Define all constraints
    constraints = []

    # Boundary constraint: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    def boundary_constraint(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return [x - r, x + r - 1, y - r, y + r - 1]
        return cons

    for i in range(26):
        cons = boundary_constraint(i)
        constraints.append({'type': 'ineq', 'fun': cons})

    # Non-overlap constraint: distance >= r_i + r_j
    def overlap_constraint(i, j):
        def cons(radii):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (radii[i] + radii[j])
        return cons

    for i in range(26):
        for j in range(i + 1, 26):
            cons = overlap_constraint(i, j)
            constraints.append({'type': 'ineq', 'fun': cons})

    # Run optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints
    )

    # Return centers, radii, and sum of radii
    return centers, result.x, np.sum(result.x)
