import numpy as np

def run_packing():

    This function computes the optimal packing of 26 circles in a unit square.
    The strategy is to create a 5x5 grid of circles with minimal spacing between them,
    and then add a 26th circle that fits in the corner without overlapping.
    """

    # Grid dimensions
    grid_rows = 5
    grid_cols = 5
    grid_spacing = 1.0 / (grid_rows - 1)  # Spacing between center points

    # Create the 5x5 grid centers
    centers = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * grid_spacing
            y = i * grid_spacing
            centers.append([x, y])

    # Compute the radius for the 25 grid circles
    # The minimum distance to adjacent circles determines the maximum radius
    # Here, we use the distance between centers of adjacent circles.

    # Radius for grid circles is determined by the minimum spacing between circle centers
    # The minimal distance between centers for adjacent circles is grid_spacing * sqrt(2) (diagonal)
    # We allow for a small tolerance to ensure non-overlapping
    radius_grid = grid_spacing / np.sqrt(2) - 1e-6  # Subtract a tiny value to ensure non-overlapping

    # Initialize radii with the grid radius
    radii = np.full(grid_rows * grid_cols, radius_grid)

    # Add the 26th circle near the corner, trying to maximize its radius without overlapping.
    # We try placing it in the bottom-left corner (0, 0)
    x26 = 0.0 + 1e-6
    y26 = 0.0 + 1e-6
    centers.append([x26, y26])

    # Compute maximal radius for the 26th circle without overlapping with grid circles
    def max_radius_corner():
        max_r = 0.0
        for i in range(grid_rows * grid_cols):
            dist = np.sqrt((centers[i][0] - x26)**2 + (centers[i][1] - y26)**2)
            max_r = min(max_r, dist - 1e-6)
        return max_r

    # Try to compute maximum radius for corner circle
    r26 = max_radius_corner()

    # Update radii
    radii = np.append(radii, r26)

    # Now, return the computed (centers, radii, sum_radii)
    centers = np.array(centers)
    sum_radii = np.sum(radii)

    # Validate the packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:

        return True, "ok"

    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(message)

    return centers, radii, sum_radii
