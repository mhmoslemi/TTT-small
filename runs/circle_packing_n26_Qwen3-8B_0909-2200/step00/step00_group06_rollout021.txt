import cvxpy as cp

    right_pad = 0.2
    bottom_pad = 0.2
    top_pad = 0.2
    width = 1 - left_pad - right_pad
    height = 1 - bottom_pad - top_pad

    # Define the "ideal" positions for a hexagonal grid in the inner area
    # Let's assume we have a grid with 5 columns and 6 rows
    row_count = 6
    col_count = 5
    col_spacing = width / (col_count - 1)
    row_spacing = height / (row_count - 1)
    row_height = np.sqrt(3) / 2 * row_spacing  # Height of a hexagonal grid

    # Generate initial centers
    centers = []
    radii = []
    for i in range(row_count):

            centers.append([x, y])

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.zeros(n)

    # Define optimization variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Define constraints
    constraints = []
    for i in range(n):
        # Constraint: circle i stays within the unit square
        constraints.append(

            constraints.append(dist >= r[i] + r[j] - 1e-12)

    constraints.append(r >= 0)

    problem.solve(solver=cp.ECOS)

    sum_radii = np.sum(radii)
