import numpy as np
import cvxpy as cp

def run_packing():
    # Initial guess for centers based on a hexagonal grid
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Step 1: Place 12 circles in a hexagonal grid in the center of square
    grid_spacing = 0.2
    for i in range(6):
        for j in range(6):
            centers[i * 6 + j] = np.array([j * grid_spacing, i * grid_spacing])
    # Adjust centers to be in center of square
    centers -= np.array([0.5 * grid_spacing, 0.5 * grid_spacing])
    # Ensure they're within square, check if less than 0.5*grid_spacing away
    centers = np.clip(centers, 0, 1)
    # Initial radius for central circles (based on grid spacing)
    radii[:12] = grid_spacing / 2

    # Step 2: Place remaining 14 circles near corners using optimization
    # Create variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraint: circles must stay within [0,1] square
    for i in range(n):
        con = (x[i, 0] - r[i] >= 0) & (x[i, 0] + r[i] <= 1) & (x[i, 1] - r[i] >= 0) & (x[i, 1] + r[i] <= 1)
        constraints.append(con)

    # Non-overlap constraint: max distance between centers must be at least sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            con = cp.norm(x[i] - x[j]) >= r[i] + r[j]
            constraints.append(con)

    # Initial guess
    x_initial = centers
    r_initial = radii

    # Create problem
    problem = cp.Problem(objective, constraints)

    # Solve the problem using solver
    problem.solve(warm_start=(x_initial, r_initial), solver=cp.SCS)

    # Extract optimal centers and radii
    centers_opt = x.value
    radii_opt = r.value

    # Final check: ensure all within square and non-overlapping
    # (no need to recheck; this is guaranteed by constraints)

    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
