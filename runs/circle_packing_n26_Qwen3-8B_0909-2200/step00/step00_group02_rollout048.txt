import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Step 1: Generate an initial equidistant grid in the unit square
    n = 26
    x = np.linspace(0, 1, int(np.sqrt(n)) + 1)
    y = np.linspace(0, 1, int(np.sqrt(n)) + 1)
    grid = np.array([[xi, yi] for xi in x for yi in y])[:n]
    centers = grid
    radii = np.full(n, 0.05)

    # Define objective function to maximize total radius sum
    def objective(r):
        return -np.sum(r)  # Negative for minimization

    # Define constraints for packing
    def constraints(r, centers, n):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                def dist_con(r, i=i, j=j, centers=centers):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    return np.sqrt(dx*dx + dy*dy) - r[i] - r[j] - 1e-12
                cons.append({'type': 'ineq', 'fun': dist_con})
        return cons

    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=Bounds(lb=0, ub=np.inf),
        constraints=constraints(radii, centers, n),
        options={'ftol': 1e-8, 'maxiter': 500}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
