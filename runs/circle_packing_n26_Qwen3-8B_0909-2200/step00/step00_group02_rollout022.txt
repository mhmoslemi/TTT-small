import scipy.optimize as optimize

            y = row * (1 / (row_count - 1)) if row_count > 1 else 0.5
            positions.append((x, y))

    centers = np.array(positions)

    radii = np.full(n_circles, radius_initial)

    def objective(radii):

    def constraint_circle_in_square(radii):

                dist = np.sqrt(dx ** 2 + dy ** 2)
