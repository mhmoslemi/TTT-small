import scipy.optimize as optimize

        [1, 1],
        [0, 1]

    edges = np.array([
        [0.25, 0],
        [0.5, 0],
        [0.75, 0],
        [0, 0.25],
        [0, 0.5],
        [0, 0.75],
        [0.25, 1],
        [0.5, 1],
        [0.75, 1],
        [1, 0.25],
        [1, 0.5],
        [1, 0.75]
    ])
    
    # Interior: hexagonal packing of 10 circles
    interior = np.array([
        [0.2, 0.2],
        [0.3, 0.3],
        [0.5, 0.3],
        [0.7, 0.3],
        [0.8, 0.2],
        [0.3, 0.5],
        [0.5, 0.5],
        [0.7, 0.5],
        [0.2, 0.7],
        [0.5, 0.7]
    ])
    
    # Combine all positions
    centers = np.vstack((corners, edges, interior))
    
    # Initial radii: based on distance to neighbors
    def compute_initial_radii(centers):
        n = centers.shape[0]
        radii = np.zeros(n)
        for i in range(n):

            for j in range(n):
                if i == j:
                    continue
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < min_dist:
                    min_dist = dist
            radii[i] = min_dist / 2.0
        return radii
    
    radii = compute_initial_radii(centers)
    
    # Define objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Define constraints:
    def constraint_func(radii):

        for i in range(len(radii)):
            x, y = centers[i]

            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

    result = optimize.minimize(
        fun=objective,

    sum_radii = np.sum(radii_result)
