import scipy.optimize as opt

def run_packing():

    n = 26

    max_rows = 5
    max_cols = 6

    radii = np.full(n, 0.1)

    def objective(r):

    def constraint_func(r, c):
        dist = np.sqrt(np.sum((c[:, np.newaxis] - c[np.newaxis, :])**2, axis=-1))
