import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance as dist
from scipy.optimize import Bounds

def run_packing():

        return -np.sum(radii)  # We maximize sum of radii by minimizing -sum

    def constraint_dist(centers, radii, i, j):
        if i == j:
            return 0
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])

    def constraint_boundary(centers, radii, i):
        x, y = centers[i]
        r = radii[i]
        return x - r >= -1e-12 and x + r <= 1 + 1e-12 and y - r >= -1e-12 and y + r <= 1 + 1e-12

    def hex_grid(n, side_length):

            row = i // 2
            col = i % 2
            x = col * (side_length * np.sqrt(3))
            y = row * (side_length * 1.5)
            centers.append([x, y])
        return np.array(centers)

    # Adjust for unit square
    def scale_to_unit(centers):
        min_x = np.min(centers[:, 0])
        max_x = np.max(centers[:, 0])
        min_y = np.min(centers[:, 1])
        max_y = np.max(centers[:, 1])
        scaling_factor = 1.0 / max(max_x - min_x, max_y - min_y)
        return centers * scaling_factor

    # Start with a moderate hexagonal grid
    n = 26
    initial_centers = hex_grid(n, 0.1)
    initial_centers = scale_to_unit(initial_centers)

    constraints = []
    for i in range(n):

        args=(),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints

    centers = initial_centers
