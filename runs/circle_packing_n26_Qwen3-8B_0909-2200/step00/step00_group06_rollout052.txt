import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define centers and radii for a hybrid packing approach
    # First, create a grid-based layout with some circles nested inside others
    # and some placed near corners and edges to utilize space efficiently

    # Grid layout: 5 rows and 5 columns of circles with spacing
    grid_centers = []
    grid_radii = []
    for i in range(5):
        for j in range(5):
            x = i / 4.5
            y = j / 4.5
            r = 0.1
            grid_centers.append([x, y])
            grid_radii.append(r)

    # Add circles near corners (adjusted to avoid overlap)
    corner_centers = [
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1]
    ]
    corner_radii = [0.15] * 4

    # Add circles near the midpoints of the sides
    side_centers = [
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]
    ]
    side_radii = [0.15] * 4

    # Add some larger circles in the center
    center_centers = [
        [0.5, 0.5]
    ]
    center_radii = [0.2]

    # Combine all centers and radii
    centers = np.array(grid_centers + corner_centers + side_centers + center_centers)
    radii = np.array(grid_radii + corner_radii + side_radii + center_radii)

    # Optimize the packing by adjusting positions and radii
    def optimize_packing(centers, radii):
        # Initial guess: use given centers and radii
        initial_guess = np.hstack((centers.flatten(), radii))

        # Define bounds for positions (must stay within unit square) and radii (non-negative)
        bounds = []
        for i in range(centers.shape[0]):
            bounds.append((0, 1))  # x-coordinate
            bounds.append((0, 1))  # y-coordinate
            bounds.append((0, 1))  # radius

        # Now optimize the radii and positions
        def objective(x):
            # Unpack x into centers and radii
            flat_centers = x[:centers.shape[0] * 2].reshape(centers.shape)
            flat_radii = x[centers.shape[0] * 2:]

            # Check for boundary violations
            for i in range(centers.shape[0]):
                x, y = flat_centers[i]
                r = flat_radii[i]
                if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                    return np.inf

            # Check for circle overlaps
            for i in range(centers.shape[0]):
                for j in range(i + 1, centers.shape[0]):
                    dx = flat_centers[i, 0] - flat_centers[j, 0]
                    dy = flat_centers[i, 1] - flat_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < flat_radii[i] + flat_radii[j] - 1e-12:
                        return np.inf

            # Return negative sum of radii to maximize it
            return -np.sum(flat_radii)

        # Set bounds
        bounds = Bounds([0, 0] * centers.shape[0], [1, 1] * centers.shape[0])

        # Run the optimization
        result = minimize(objective, x0=initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        return result.x[:centers.shape[0] * 2].reshape(centers.shape), result.x[centers.shape[0] * 2:]

    optimized_centers, optimized_radii = optimize_packing(centers, radii)

    # Compute total sum of radii
    sum_radii = np.sum(optimized_radii)

    return (optimized_centers, optimized_radii, sum_radii)
