import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    # Initialize a hexagonal packing with equal radii for the first pass
    n = 26
    num_rows = int(np.ceil(np.sqrt(n)))
    num_cols = int(np.ceil(n / num_rows))
    radius_initial = 0.1
    centers = np.zeros((n, 2))
    radii = np.full(n, radius_initial)

    # Generate a hexagonal grid
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i, 0] = x / (num_cols - 1) if num_cols > 1 else 0.5
        centers[i, 1] = y / (num_rows - 1) if num_rows > 1 else 0.5

    # Define objective function (maximize sum of radii)
    def objective(x, centers, radii):
        # x is a flat vector containing [r0, r1, ..., r25]
        new_radii = np.array(x).reshape((n,))
        total = np.sum(new_radii)
        return -total  # Minimize negative total to maximize

    # Constraints: boundary, distance between centers, and non-negativity
    def constraints_func(x, centers, radii):
        new_radii = np.array(x).reshape((n,))
        # Check if any radius is negative
        if np.any(new_radii < 0):
            return np.inf
        # Check if circles are within unit square
        for i in range(n):
            r = new_radii[i]
            if not (0 <= centers[i, 0] - r and centers[i, 0] + r <= 1):
                return np.inf
            if not (0 <= centers[i, 1] - r and centers[i, 1] + r <= 1):
                return np.inf
        # Check for overlapping circles
        dists = squareform(pdist(centers))
        for i in range(n):

                if dists[i, j] < new_radii[i] + new_radii[j] - 1e-12:
                    return np.inf
        return 0

    # Convert centers and radii to flat variables
    flat_centers = centers.flatten()
    flat_radii = radii.flatten()
    initial_guess = flat_radii

    bounds = [(0, 1) for _ in range(n)]

    # Define the optimization problem
    def objective_func(x):
        return objective(x, centers, radii)

        return constraints_func(x, centers, radii)

    # Run optimization
    result = minimize(

        x0=initial_guess,
        bounds=bounds,
        method='SLSQP',
