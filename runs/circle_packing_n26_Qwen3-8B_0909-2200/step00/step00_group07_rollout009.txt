import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():

    def hexagonal_grid(n_rows, n_cols, grid_spacing=1.0):
        centers = []
        for i in range(n_rows):
            for j in range(n_cols):
                x = j * grid_spacing
                y = i * grid_spacing * np.sqrt(3) / 2
                centers.append([x, y])
        return np.array(centers)

    # Initial hexagonal grid with 10 rows and 10 columns
    # Grid spacing is chosen to fit into the square
    n_rows = 5
    n_cols = 5
    grid_spacing = 1.0 / (np.sqrt(3) / 2 * n_rows)

    centers = hexagonal_grid(n_rows, n_cols, grid_spacing)
    centers = ((centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0)) * 1.0)
    
    # Add corner circles
    corner_circles = [
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.9],
        [0.9, 0.1],
        [0.5, 0.1],
        [0.5, 0.9]
    ]
    centers = np.vstack((centers, corner_circles))
    n_circles = len(centers)

    # Initial guess for radii - same for all
    initial_radii = np.array([0.15] * n_circles)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization to maximize radii
    bounds = [(0.0, 1.0) for _ in range(n_circles)]
    
    result = opt.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
    )

    radii = result.x
    centers = centers[np.argsort(radii)[::-1]]
    radii = radii[np.argsort(radii)[::-1]]

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
