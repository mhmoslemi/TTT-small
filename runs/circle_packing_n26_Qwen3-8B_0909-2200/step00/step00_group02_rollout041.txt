import scipy.optimize as opt

def run_packing():

    n = 26
    grid_size = 5
    grid_offset = 1 / grid_size

    # Generate a coarse grid as initial centers
    x_centers = np.linspace(0 + grid_offset, 1 - grid_offset, grid_size)
    y_centers = np.linspace(0 + grid_offset, 1 - grid_offset, grid_size)

    # Create all combinations of x and y centers for 25 circles
    centers = np.array([(x, y) for x in x_centers for y in y_centers])

    centers = np.vstack((centers, (0.5, 0.5)))

    # Step 2: Define radius variables (initial guess: equal radii)
    initial_radii = np.full(n, 1 / (grid_size * grid_size))

    # Dictionary to store pairwise distance constraints
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            # Constraint: distance between centers[i] and centers[j] >= radii[i] + radii[j]
            def constraint_func(radii):
                xi, yi = centers[i]
                xj, yj = centers[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': constraint_func})

    # Boundary constraints for all circles to stay within unit square
    for i in range(n):
        x, y = centers[i]
        # Constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        def x_min(radii):
            return x - radii[i]
        def x_max(radii):
            return 1 - x - radii[i]
        def y_min(radii):
            return y - radii[i]

            return 1 - y - radii[i]
        constraints.append({'type': 'ineq', 'fun': x_min})

        constraints.append({'type': 'ineq', 'fun': y_min})
        constraints.append({'type': 'ineq', 'fun': y_max})

    # Define variables
    bounds = [(0, None) for _ in range(n)]

    # Perform optimization
    result = opt.minimize(
        fun=lambda r: -np.sum(r),  # Maximize sum of radii
        x0=initial_radii,
        method='SLSQP',
