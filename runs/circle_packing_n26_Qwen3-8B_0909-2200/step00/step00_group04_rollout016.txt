import numpy as np
import scipy.optimize as opt

def run_packing():
    # Start with a grid layout to seed the positions
    n = 26
    rows = 5
    cols = 6
    spacing = 1.0 / (cols + 1)
    
    # Grid-based initial positions
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * spacing + spacing * 0.5
        y = row * spacing + spacing * 0.5
        centers[i] = [x, y]
        # Initial radius is uniform but will be optimized later
        radii[i] = 0.1
    
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total
    
    def constraints(radii, centers):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons = dist - (radii[i] + radii[j])
                constraints.append(cons)
        return np.array(constraints)
    
    # Define bounds for radii
    bounds = [(0, 1)] * n
    
    # Initial guess for radii (based on grid spacing)
    initial_radii = np.full(n, 0.1)
    
    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=opt.Constraint(
            type='ineq',
            fun=lambda x: constraints(x, centers)
        )
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
