import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():

    num_circles = 26

    tolerance = 1e-5

    # Triangle spacing: s = sqrt(3) / 3 ≈ 0.577
    s = np.sqrt(3) / 3
    num_rows = 6
    num_cols = 5
    circles = []

    for i in range(num_rows):
        for j in range(num_cols):
            x = j * s
            y = i * (s * np.sqrt(3) / 2)
            circles.append((x, y))

    centers = np.array(circles)
    radii = np.full(len(circles), 1.0 / (2 * np.sqrt(3)))

    # Ensure all circles are within [0,1]^2
    def scale_to_unit_square(centers):
        min_x = np.min(centers[:, 0])
        max_x = np.max(centers[:, 0])
        min_y = np.min(centers[:, 1])
        max_y = np.max(centers[:, 1])
        scale_x = (1 - min_x) / (max_x - min_x) if (max_x - min_x) > 0 else 1
        scale_y = (1 - min_y) / (max_y - min_y) if (max_y - min_y) > 0 else 1
        return centers * np.array([scale_x, scale_y])

    centers = scale_to_unit_square(centers)
    radii *= np.sqrt(3) / 3  # Adjust radii to fit in unit square

    centers = np.round(centers, 8)
    radii = np.round(radii, 8)

    def objective(radii):
        return -np.sum(radii)

    def non_overlapping_constraints(x):

                dist = np.linalg.norm(centers[i] - centers[j])

                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0.0

    def boundary_constraints(x):
