import numpy as np
from scipy.optimize import minimize

def run_packing():

    rows = 5
    cols = 5
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers.append([x, y])
    
    # Handle the extra circle (central one)
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    radii = np.full(26, 0.1)  # Initial guess

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative to maximize

    # Define constraints for boundary and non-overlapping
    def boundary_constraints(radii):
        for i in range(26):
            x, y = centers[i]
            if not (0 <= x - radii[i] and x + radii[i] <= 1 and 0 <= y - radii[i] and y + radii[i] <= 1):
                return False
        return True

    # Define pairwise non-overlapping constraints
    def non_overlap_constraints(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Combine all constraints
    def constraints(radii):
        if not boundary_constraints(radii):
            return np.inf
        if not non_overlap_constraints(radii):
            return np.inf
        return 0

    # Use optimization to refine radii
    result = minimize(
        objective,
        radii,
        bounds=[(0, 1)] * 26,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
    )

    if not result.success:
        raise RuntimeError("Optimization failed.")

    # Final centers and radii
    return centers, result.x, np.sum(result.x)
