import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial grid points
    # 5x5 grid, creating 25 points
    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)
    centers = np.array([[xi, yi] for xi in x for yi in y])
    # Add one extra point near the bottom-right corner
    centers = np.vstack([centers, [0.9, 0.1]])

    # Initialize radii for all centers: try equal radii initially
    radii = np.full(len(centers), 0.05)

    # Define constraints for the optimization
    n = len(centers)
    constraints = []

    # Constraint: each circle must touch the boundary
    def boundary_constraint(i):
        def c(x):
            x_i = x[i * 2]
            y_i = x[i * 2 + 1]
            r_i = x[n * 2 + i]
            return x_i - r_i + 1e-12, y_i - r_i + 1e-12, x_i + r_i - 1 + 1e-12, y_i + r_i - 1 + 1e-12
        return c

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 2] - x[n * 2 + i] + 1e-12})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 2 + 1] - x[n * 2 + i] + 1e-12})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[n * 2 + i] - x[i * 2] + 1e-12})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[n * 2 + i] - x[i * 2 + 1] + 1e-12})

    # Constraint: non-overlapping between circles
    def dist_constraint(i, j):
        def c(x):
            x_i = x[i * 2]
            y_i = x[i * 2 + 1]
            r_i = x[n * 2 + i]
            x_j = x[j * 2]
            y_j = x[j * 2 + 1]
            r_j = x[n * 2 + j]
            return np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2) - r_i - r_j + 1e-12
        return c

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist_constraint(i, j)(x)})

    # Optimization problem: maximize sum of radii
    def objective(x):
        return -np.sum(x[n * 2:])

    # Bounds: all x and y between 0 and 1, radii ≥ 0
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y for each center
        bounds.append((0.0, 1.0))  # radius for each circle

    # Initial guess: x and y as centers, radius as 0.05
    x0 = np.array([centers[i][0] for i in range(n)])
    x0 = np.concatenate([x0, [centers[i][1] for i in range(n)]])
    x0 = np.concatenate([x0, radii])

    # Solve the optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )

    # Extract the final solution
    x_final = result.x
    centers_final = np.array([[
        x_final[i * 2],
        x_final[i * 2 + 1]
    ] for i in range(n)])
    radii_final = x_final[n * 2:]

    # Return centers, radii, and sum of radii
    return (centers_final, radii_final, np.sum(radii_final))
