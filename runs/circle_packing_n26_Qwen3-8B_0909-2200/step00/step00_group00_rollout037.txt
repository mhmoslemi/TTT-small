from scipy.optimize import minimize
from scipy.optimize import Bounds

    col_count = 5

            centers[i * col_count + j] = [x, y]

        return -np.sum(radii)  # Minimize negative sum to maximize sum

            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
