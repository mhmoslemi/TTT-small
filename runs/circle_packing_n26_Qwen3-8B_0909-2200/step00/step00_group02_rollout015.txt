import numpy as np
import scipy.optimize as optimize

def run_packing():

    grid_size = 4
    grid_spacing = 1 / (grid_size + 1)  # Spacing between centers of adjacent circles
    core_radii = np.full(grid_size * grid_size, 0.1)

    core_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = grid_spacing * (i + 1)

            core_centers.append([x, y])
    core_centers = np.array(core_centers)

    all_centers = np.vstack((core_centers, corner_centers))
    all_radii = np.concatenate((core_radii, corner_radii))

    # Step 4: Optimization: maximize sum of radii
    def objective(x):
        # x is a vector of radii updates to be optimized
        updated_radii = all_radii + x
        updated_radii[updated_radii < 0] = 0  # Ensure non-negativity

    bounds = [(-0.5, 0.5) for _ in range(len(all_radii))]  # Allow up to 0.5 increase in radii
