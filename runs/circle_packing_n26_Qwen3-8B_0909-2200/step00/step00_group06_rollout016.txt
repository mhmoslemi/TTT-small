import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles and variables
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize total radius sum
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: All circles within the unit square [0,1]×[0,1]
    constraints = []

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints += [
            x - r >= 0,
            x + r <= 1,
            y - r >= 0,
            y + r <= 1
        ]

    # Constraints: No two circles overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints += [dist >= radii[i] + radii[j]]

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.OSQP, verbose=False)

    # Extract results
    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)

    return (centers_val, radii_val, sum_radii)
