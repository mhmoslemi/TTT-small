```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Step 1: Initial guess - try a cluster of tightly packed circles
    # Define 26 circle centers as a grid with perturbations
    centers = np.random.rand(26, 2)
    # Ensure centers are near a compact cluster
    centers += np.array([[0.5, 0.5]] * 26)
    centers /= 2.0

    # Initial guess for radii (small, but nonzero)
    radii = np.random.rand(26) * 0.05
    radii = np.clip(radii, 0, 0.1)

    # Objective function: minimize negative sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    constraints = []

    def non_overlap(x, y, r_i, r_j):
        if r_i == r_j:
            return np.sqrt((x - y)**2) >= r_i + r_j - 1e-8
        return np.sqrt((x - y)**2) >= r_i + r_j - 1e-8

    # Boundary constraints: for each circle, x - r_i >= 0, x + r_i <= 1, similarly for y
    for i in range(26):
        def boundary(i):
            def cons(x):
                center = x[:2*i+2].reshape(26, 2)
                radius = x[2*i+2]
                if (center[i, 0] - radius < -1e-12 or
                    center[i, 0] + radius > 1 + 1e-12 or
                    center[i, 1] - radius < -1e-12 or
                    center[i, 1] + radius > 1 + 1e-12):
                    return 1
                return 0
            return {'type': 'ineq', 'fun': cons}
        constraints.append(boundary(i))

    # Non-overlap constraints
    for i in range(26):
        for j in range(i + 1, 26):
            def non_overlap_c(i, j):
                def cons(x):
                    center = x[:2*i+2].reshape(26, 2)
                    radius = x[2*i+2]
                    radius2 = x[2*j+2]
                    dist = np.sqrt((center[i, 0] - center[j, 0])**2 + 
                                   (center[i, 1] - center[j, 1])**2)
                    return dist - radius - radius2 + 1e-8
                return {'type': 'ineq', 'fun': cons}
            constraints.append(non_overlap_c(i, j))

    # Bounds: radii must be >= 0
    bounds = [(0, 1) for _ in range(26 * 2 + 26)]

    # Initial guess: centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Solve the optimization
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Extract centers and radii
    x_opt = result.x
    centers_final = x_opt[:26 * 2].reshape(26, 2)
    radii_final = x_opt[26 * 2:]

    # Validate the packing
    valid, message = validate_packing(centers_final, radii_final)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    sum_radii = np.sum(radii_final)
    return centers_final, radii_final, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
