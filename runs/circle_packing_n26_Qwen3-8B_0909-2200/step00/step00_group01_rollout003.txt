def run_packing():
    # Number of circles
    n = 26
    # Initial random centers within the unit square
    centers = np.random.rand(n, 2)
    # Initial uniform radii
    radii = np.full(n, 0.03)
    
    # Optimization parameters
    learning_rate = 0.01
    num_iterations = 1000
    tolerance = 1e-5

    # Optimization loop
    for _ in range(num_iterations):
        # Compute pairwise distances
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dists[i, j] = np.sqrt(dx**2 + dy**2)
        
        # Compute forces between circles
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = dists[i, j]
                    if dist < 1e-12:
                        continue
                    force = (radii[i] + radii[j]) / dist
                    forces[i, 0] += force * dx
                    forces[i, 1] += force * dy

        # Update centers
        centers -= learning_rate * forces

        # Ensure centers stay within [0, 1]^2
        centers = np.clip(centers, 0, 1)

        # Compute radii based on minimum distances to boundaries
        radii = np.zeros(n)
        for i in range(n):
            dist_to_boundary = np.array([
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            ])
            radii[i] = np.min(dist_to_boundary)
            if radii[i] < 0:
                radii[i] = 0

        # Ensure no overlapping between circles
        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    # Adjust radii to resolve overlap
                    radii[i] = (dist - 1e-12) / 2
                    radii[j] = (dist - 1e-12) / 2

        # Check for convergence
        if np.max(np.abs(forces)) < tolerance:

    # Validate the packing

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
