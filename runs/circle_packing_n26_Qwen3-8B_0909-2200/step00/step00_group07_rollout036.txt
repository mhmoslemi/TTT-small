import cvxpy as cp

def run_packing():

    num_circles = 26
    grid_rows = 5
    grid_cols = 5

    # Define a grid with varying row spacing
    x_coords = np.linspace(0.1, 0.9, grid_cols)
    y_coords = np.linspace(0.1, 0.9, grid_rows)

    # Initial grid positions (this is a starting point, not necessarily optimal)
    centers = np.array([
        [x, y] for i, (x, y) in enumerate(zip(x_coords, y_coords))
    ])

    # Initial radii guess (based on spacing)
    # For simplicity, estimate radii as the minimum distance to the nearest grid neighbor
    dist = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dist, np.inf)
    min_dist = np.min(dist, axis=1)
    radii = min_dist / 2.5

    # Define optimization variables
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles)

    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(num_circles):
        # Center must lie within the square
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)
        # Radius must be non-negative
        constraints.append(r[i] >= 0)

    # Non-overlap constraint between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            # Euclidean distance between centers must be >= r_i + r_j
            # To represent this with CVXPY, we use the following:
            # distance = sqrt((x_i - x_j)^2 + (y_i - y_j)^2)
            # So: distance >= r_i + r_j
            distance = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(distance >= r[i] + r[j] - 1e-8)

    # Initial guess
    x_initial = centers
    r_initial = radii

    # Define the optimization problem
    prob = cp.Problem(obj, constraints)

    # Add initial guess
    prob = prob.copy()

    prob.constraints[0].dual_vars = cp.Parameter((num_circles, 2), value=x_initial)

    # Solve the optimization problem
    prob.solve()

    centers_opt = x.value

    # Return the result
    return (centers_opt, radii_opt, sum_radii)
