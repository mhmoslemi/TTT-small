import scipy.optimize as opt

def run_packing():

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

        centers[i] = [x, y]
        radii[i] = 0.1  # Initial low radius

    # Optimization bounds
    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    # Objective function: Maximize sum of radii
    def objective(x):
        return -np.sum(x[2::3])  # Negative for maximization

    # Constraints: Non-overlapping circles
    def constraints(x):
        # Extract positions and radii
        centers = x[0::3].reshape(-1, 2)
        radii = x[2::3]
        
        # Check boundary constraints
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return np.inf  # Penalize for violating boundary constraints
        
        # Check non-overlapping constraint
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)

                    return np.inf  # Penalize for overlapping circles

        x0,
        bounds=bounds,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
