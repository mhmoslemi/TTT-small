import scipy.optimize as optimize

def run_packing():

    centers = []
    radii = []
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 5.0 + 0.05
        y = row / 5.0 + 0.05
        centers.append([x, y])
    
    # Define optimization variables: centers (x, y) and radii
    n = 26
    initial_centers = np.array(centers)
    initial_radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # Boundary constraints
        for i in range(n):
            if radii[i] < 0:
                return False
            if initial_centers[i, 0] - radii[i] < -1e-12 or \
               initial_centers[i, 0] + radii[i] > 1 + 1e-12 or \
               initial_centers[i, 1] - radii[i] < -1e-12 or \

                return False
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
