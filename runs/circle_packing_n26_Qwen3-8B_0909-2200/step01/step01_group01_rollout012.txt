import numpy as np
import random
from scipy.optimize import minimize

def run_packing():
    # Define parameters
    num_circles = 26
    max_iterations = 100
    tol = 1e-6

    # Initial placement in a staggered (hexagonal) grid
    # We'll have 7 rows with 4, 5, 6, 5, 4, 5, 4 circles respectively
    # Total area covered by grid will be 26 circles
    grid_rows = 7
    grid_cols = [4, 5, 6, 5, 4, 5, 4]

    # Spacing factor between centers
    spacing_factor = 0.33

    # Initial angles for staggered grid
    angles = np.linspace(0, np.pi, grid_rows, endpoint=False)

    centers = []
    for i, col_count in enumerate(grid_cols):
        row_angle = angles[i]
        for j in range(col_count):
            r = spacing_factor * (i + 1)
            x = (j + 0.5) * spacing_factor * (i + 1)
            y = r * np.sin(row_angle)
            centers.append([x, y])

    centers = np.array(centers)

    # Initial equal radii
    radii = np.full(len(centers), 0.1)

    # Define objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Define constraints: no overlap between circles
    def constraint_overlap(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return -1
        return 0

    # Define constraint to ensure circles are inside the unit square
    def constraint_inside(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Define bounds for radii
    bounds = [(0, 1.0) for _ in range(len(centers))]

    # Optimization problem
    prob = {
        'fun': objective,
        'bounds': bounds,
        'constraints': [
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_inside}
        ]
    }

    # Perform optimization
    result = minimize(**prob, method='SLSQP', options={'maxiter': max_iterations, 'ftol': tol, 'eps': 1e-6})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
