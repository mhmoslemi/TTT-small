import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for circle positions using a hexagonal grid
    # Place 5 rows with 5 circles per row
    centers = []
    radii = []
    for i in range(5):
        for j in range(5):
            x = j + 0.1
            y = i + 0.1
            centers.append([x, y])
            radii.append(0.05)

    # 1 extra circle to be inserted dynamically
    # We'll add it slightly near the bottom right
    centers.append([0.85, 0.85])
    radii.append(0.05)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to minimize: -sum of radii (maximizing sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap
    def non_overlap(radii):
        dist = squareform(pdist(centers))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1.0
        return 0.0

    # Constraint for boundary
    def in_square(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or \
               y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1.0
        return 0.0

    # Combine constraints: function returns 0 if constraints are satisfied, -1 otherwise
    def constraint(radii):
        return non_overlap(radii) + in_square(radii)

    # Optimization: minimize -sum(radii) with constraints
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        jac=lambda x: np.array([1.0] * len(x)),
        constraints=[{'type': 'ineq', 'fun': constraint}],
        bounds=[(0.0, 0.25) for _ in range(len(radii))],
        options={'maxiter': 1000}
    )

    if not res.success:
        raise RuntimeError("Optimization failed with message: %s" % res.message)

    # Final result
    best_radii = res.x
    sum_radii = np.sum(best_radii)
    return centers, best_radii, sum_radii
