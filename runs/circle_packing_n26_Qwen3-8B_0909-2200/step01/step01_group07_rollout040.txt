def generate_hexagonal_grid(n, side_length, center):
    """
    Generate a hexagonal grid of size n, with spacing based on side_length.
    """
    grid = []
    for i in range(n):
        x = center[0] + (i // 3) * side_length
        y = center[1] + (i % 3) * side_length
        grid.append((x, y))
    return np.array(grid)
