def compute_total_sum(radii):

def check_overlap(centers, radii, eps=1e-8):
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - eps:
                return False
    return True

def check_boundary(centers, radii, eps=1e-8):

        r = radii[i]
        if (x - r < -eps or x + r > 1 + eps or

            return False
    return True
