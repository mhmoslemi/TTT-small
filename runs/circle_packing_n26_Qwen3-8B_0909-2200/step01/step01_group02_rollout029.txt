import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26

    # Create variables for circle centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, non_negative=True)

    # Constraints: all circles within the unit square
    constraints = []

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r <= 1e-8)
        constraints.append(x + r <= 1 + 1e-8)
        constraints.append(y - r <= 1e-8)
        constraints.append(y + r <= 1 + 1e-8)

    # Non-overlap between all pairs of circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            constraint = cp.sqrt(dx**2 + dy**2) >= radii[i] + radii[j] - 1e-8
            constraints.append(constraint)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.SCS)
    except:
        return None, None, 0.0

    # Extract the results
    centers_val = np.array(centers.value)
    radii_val = np.array(radii.value)

    # Final validation step
    for i in range(n):

            if dist < radii_val[i] + radii_val[j] - 1e-6:

    sum_radii = np.sum(radii_val)
    return centers_val, radii_val, sum_radii
