import scipy.optimize as optimize

def run_packing():

    grid_centers = []

        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            grid_centers.append([x, y])
    
    # Add corner circles
    corners = [[0.0, 0.0], [1.0, 0.0], [1.0, 1.0], [0.0, 1.0]]
    grid_centers += corners

    # Final grid
    centers = np.array(grid_centers)
    radii = np.zeros(len(centers))

    # Attempt to optimize the radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - r_i - r_j + 1e-8
        return cons

    constraints = []

    bounds = [(0.0, 1.0) for _ in range(len(centers))]

    result = optimize.minimize(
