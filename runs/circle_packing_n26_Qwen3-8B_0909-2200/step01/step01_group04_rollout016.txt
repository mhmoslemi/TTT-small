import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables
    n = 26
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints: Each circle must be within the unit square
    constraints = []

    for i in range(n):
        # Left and right boundaries
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        # Bottom and top boundaries
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlap constraint: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j])

    # Solve the optimization
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract results
    centers = x.value
    radii = r.value

    # Ensure values are finite (in case of numerical issues)
    if not np.isfinite(centers).all() or not np.isfinite(radii).all():
        raise ValueError("Optimization failed to produce finite values")

    return centers, radii, np.sum(radii)
