import math
import random

def run_packing():

    radii = []

        x = i / 12.0
        y = 0.25
        centers.append([x, y])
        radii.append(0.05)

    # Second row of 13 circles, offset horizontally and elevated vertically
    for i in range(13):
        x = (i + 0.5) / 12.0
        y = 0.5
        centers.append([x, y])

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Recursive function to adjust radii based on spacing and boundaries
    def adjust_radii(centers, radii, max_iterations=100):
        for _ in range(max_iterations):
            for i in range(len(centers)):
                x, y = centers[i]
                r = radii[i]

                # Check boundaries
                if x - r < -1e-8:
                    radii[i] = x
                if x + r > 1.0 + 1e-8:
                    radii[i] = 1.0 - x
                if y - r < -1e-8:
                    radii[i] = y
                if y + r > 1.0 + 1e-8:
                    radii[i] = 1.0 - y

            # Adjust radii based on spacing with neighbors
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-8:
                        # Reduce radii to remove overlap
                        overlap = (radii[i] + radii[j]) - dist
                        ratio = 0.5
                        radii[i] -= overlap * ratio
                        radii[j] -= overlap * ratio
                        # Ensure non-negative radii
                        radii[i] = max(0.0, radii[i])
                        radii[j] = max(0.0, radii[j])

            # Small perturbation to prevent circles from getting stuck in local minima

    adjust_radii(centers, radii)

    sum_radii = np.sum(radii)
