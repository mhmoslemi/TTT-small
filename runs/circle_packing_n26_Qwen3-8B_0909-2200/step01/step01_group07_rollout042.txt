import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def create_hexagonal_grid(num_circles):
    rows = int(np.ceil(num_circles / 3))
    cols = np.ceil(num_circles / rows)
    centers = np.empty((num_circles, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= num_circles:
                break
            x = c / (cols - 1) if cols > 1 else 0.5
            y = r / (rows - 1) if rows > 1 else 0.5
            # Shift hex grid to center
            y += r % 2 * 0.25
            centers[idx] = [x, y]
            idx += 1
    return centers

def run_packing():
    num_circles = 26
    centers = create_hexagonal_grid(num_circles)

    # Initial radii guesses: based on grid spacing
    spacing = np.min(np.sqrt(np.sum((centers[0] - centers[1]) ** 2)))
    initial_radii = np.full(num_circles, spacing / 2)

    # Objective function to maximize total radius sum
    def objective(radii, centers):
        return -np.sum(radii)  # Negative for maximization

    # Constraints:
    # 1. Each circle must be fully inside the unit square
    # 2. No overlaps between circles
    # 3. Radii must be non-negative

    def constraint_inside(radii, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    def constraint_overlap(radii, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Use a solver to optimize radii
    bounds = [(0, 1) for _ in range(num_circles)]
    result = optimize.minimize(

        initial_radii,
        args=(centers,),
        bounds=bounds,
        constraints=[

            {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
        ],
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
