import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    # plus one additional circle at the bottom right
    
    # Row and column spacings
    spacing = 1.0 / 5.0  # 5 columns, 5 rows
    row_offset = spacing * np.sqrt(3) / 2  # Offset for hexagonal packing

    # Generate centers for the first 25 circles in a hexagonal pattern
    centers = []
    radii = []
    for i in range(5):  # Rows

                y = i * spacing
            else:
                y = i * spacing + row_offset

    radii = np.array(radii)

        return -np.sum(radii)  # Minimize negative sum to maximize sum

        return cons

    x0 = radii.copy()

    cons = []

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
