import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles to pack
    n = 26

    # Create a hexagonal packing pattern of circles within the unit square
    # We will place them in rows, offsetting between rows to minimize spacing

    # Starting with an initial guess of circle positions
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define the spacing between centers for nearest neighbor circles
    delta_x = 1 / (np.ceil(np.sqrt(n)))  # Horizontal spacing
    delta_y = delta_x * np.sqrt(3) / 2  # Vertical spacing between offset rows

    row_count = np.ceil(np.sqrt(n))
    col_count = np.ceil(n / row_count)

    # Place the circles in a hexagonal lattice
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col * delta_x
        y = row * delta_y
        centers[i, 0] = x
        centers[i, 1] = y

    # Now, we want to find the optimal radii for these centers
    # Variables
    radii = cp.Variable(n)
    # Objective: maximize the sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints:
    # 1. Radii must be non-negative
    constraints = [
        radii >= 0,
    ]

    # 2. Each circle must be fully within the unit square
    for i in range(n):
        x_i, y_i = centers[i]
        constraint = (x_i - radii[i] >= 0) & (x_i + radii[i] <= 1) & \
                    (y_i - radii[i] >= 0) & (y_i + radii[i] <= 1)
        constraints.append(constraint)

    # 3. Circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraint = dist >= radii[i] + radii[j] - 1e-8
            constraints.append(constraint)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve()

    # Extract the optimal centers and radii
    centers = centers.value
    radii = radii.value

    # Calculate total sum of radii
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
