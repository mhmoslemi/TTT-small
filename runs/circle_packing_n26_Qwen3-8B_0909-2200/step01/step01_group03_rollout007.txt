import math
from scipy.optimize import minimize
from scipy.spatial import distance

    # First create a hexagonal grid with m rows and n columns
    # We'll use a hybrid arrangement for 26 circles

    centers = []
    radii = []
    for i in range(26):
        # Use a hexagonal packing pattern for better efficiency
        if i % 2 == 0:
            row = i // 2
            col = i % 2
        else:
            row = (i - 1) // 2
            col = (i - 1) % 2

        if col == 0:
            y = row / 2
        else:
            y = (row + 0.5) / 2

        x = col / 2
        centers.append([x, y])

    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum

    # Define constraints for non-overlapping and boundary conditions
    def constraint_nonoverlap(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1.0 + 1e-8 or

        return 0

    result = minimize(

        method='SLSQP',
