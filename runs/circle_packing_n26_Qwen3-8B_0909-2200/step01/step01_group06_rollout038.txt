import scipy.optimize as opt

def run_packing():

    def initial_centers(n_circles):

        x = grid[:-1]
        y = grid[:-1]
        centers = np.array([(xi, yi) for xi in x for yi in y])
        return centers

    # Objective: Maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints:
    # 1. No circle goes outside the square
    # 2. No circle has negative radius
    # 3. No two circles overlap

    def constraint_radius_non_negative(radii):
        return np.all(radii >= 0)

    def constraint_inside_square(centers, radii):
        return np.all((centers[:, 0] - radii >= 0) & (centers[:, 0] + radii <= 1) &
                      (centers[:, 1] - radii >= 0) & (centers[:, 1] + radii <= 1))

    def constraint_no_overlap(centers, radii):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
