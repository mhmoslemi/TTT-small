import numpy as np

def run_packing():

    # The pattern will have 5 rows with varying numbers of circles:
    # Row 0: 5 circles
    # Row 1: 6 circles
    # Row 2: 5 circles
    # Row 3: 6 circles
    # Row 4: 5 circles
    # 5 rows with total of 5 + 6 + 5 + 6 + 5 = 27 circles, but we need 26, so remove one circle

    # Define the vertical spacing between rows in a hexagonal grid
    row_spacing = 0.5  # Height of a hexagonal row
    col_spacing = 0.5  # Horizontal spacing between adjacent circles

    centers = []
    radii = []

    # Generate the hexagonal grid with 5 rows
    for row in range(5):
        for col in range(5 + (row % 2)):
            # Compute coordinates
            x = col * col_spacing
            y = row * row_spacing
            # Ensure x and y stay within [0, 1]
            if x > 1 - 1e-12:
                x = 1 - 1e-12
            if y > 1 - 1e-12:
                y = 1 - 1e-12
            centers.append([x, y])

    # Remove the last circle to reduce total to 26 circles (total would be 27)
    centers.pop()

    # Compute radii for each circle based on nearest boundary and other circles
    for i in range(len(centers)):
        x, y = centers[i]
        min_dist = float('inf')

        # Check distance to boundaries
        min_dist = min(min_dist, x, 1 - x, y, 1 - y)

        # Check distances to other circles
        for j in range(len(centers)):
            if i == j:
                continue
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            min_dist = min(min_dist, dist)

        # Radius is the minimum distance to boundary or other circle, divided by 2
        r = min_dist / 2
        radii.append(r)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Compute total sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
