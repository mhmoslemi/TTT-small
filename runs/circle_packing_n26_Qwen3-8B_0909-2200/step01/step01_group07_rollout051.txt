import scipy.optimize as optimize

def run_packing():

    def hexagonal_grid(n):
        row_count = int(np.ceil(np.sqrt(n)))
        col_count = int(np.ceil(n / row_count))
        centers = []
        for i in range(row_count):
            for j in range(col_count):
                x = j + 0.5 * i
                y = i + 0.5 * j
                if x < 1 and y < 1:
                    centers.append([x, y])
        return np.array(centers)

    initial_centers = hexagonal_grid(26)
    initial_radii = np.full(26, 0.1)

    def constraint_function(radii, centers):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Optimization
    result = optimize.minimize(

        x0=initial_radii,
        bounds=[(0, 1)] * 26,
        constraints=[{"type": "ineq", "fun": lambda x: constraint_function(x, initial_centers)}]

    centers = initial_centers
