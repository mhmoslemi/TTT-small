import numpy as np
import scipy.optimize as optimize
import cvxpy as cp

def run_packing():

    num_circles = 26
    rows, cols = 5, 5  # 5x5 grid with 25 circles, one extra circle for optimization
    grid = np.linspace(0.1, 0.9, cols)
    grid_y = np.linspace(0.1, 0.9, rows)
    centers = np.array([np.tile(grid, rows) + np.repeat(grid_y, cols), np.repeat(grid_y, cols)]).T
    radii = np.full(num_circles, 0.05)

    # Define variables for optimization
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles, non_negative=True)

    # Constraints
    constraints = []
    for i in range(num_circles):
        # Boundary constraints
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlapping constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)

    # Extract results
    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    # Ensure the output is in the required format
    return (centers, radii, sum_radii)
