from scipy.optimize import minimize, Bounds
from scipy.optimize import NonlinearConstraint

def run_packing():

    num_rows = 4
    num_cols = 6
    initial_centers = []
    radii_initial = []

    # Hexagonal grid spacing: h = 1 / sqrt(3) (densest packing in 2D for same radii)
    # However, instead of equal radii, we will later optimize them
    h = 1.0 / (np.sqrt(3))  # Horizontal distance between centers
    v = h * (np.sqrt(3) / 2)  # Vertical distance between rows (hexagonal)
    offset = h / 2  # X offset for even rows

    for row in range(num_rows):
        for col in range(num_cols):
            x = col * h + (row % 2) * offset
            y = row * v
            initial_centers.append([x, y])
            radii_initial.append(0.02)  # Initial guess for radius

    initial_centers = np.array(initial_centers)
    radii_initial = np.array(radii_initial)

    # To maximize the total sum of radii, we minimize the negative of the sum
    def objective(radii):
        return -np.sum(radii)

    # Boundary constraint: Each circle must be inside the unit square
    def boundary_inequality(radii):
        return np.sum((initial_centers[:, 0] - radii) >= 0) + np.sum((initial_centers[:, 0] + radii) <= 1) + \

    def non_overlap_inequality(radii):

    ]

    # Run the optimization
    result = minimize(

        x0=radii_initial,
        bounds=bounds,
        constraints=constraints,
