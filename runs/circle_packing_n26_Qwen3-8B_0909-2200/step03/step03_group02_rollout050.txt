import math
from scipy.optimize import minimize

    row_counts = [5, 6, 6, 5, 4]  # Total 26 circles

    centers = []

            dist_right = 1 - x_coords[i]
            dist_bottom = y_coords[i]
            dist_top = 1 - y_coords[i]
            min_boundary = min(dist_left, dist_right, dist_bottom, dist_top)
