import math

from scipy.optimize import minimize

def run_packing():

    num_circles = 26

    # Generate initial centers using a staggered arrangement
    centers = []
    for i in range(num_circles):
        x = i / (num_circles - 1)
        y = 0.5
        centers.append([x, y])
    centers = np.array(centers)

    # Flatten centers for optimization
    initial_guess = []
    for x, y in centers:
        initial_guess.append(x)
        initial_guess.append(y)
    initial_guess = np.array(initial_guess)

    # Optimization bounds
    bounds = []
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y

        x_coords = x[0::2]
        y_coords = x[1::2]

        # Reconstruct centers
        centers_opt = np.column_stack((x_coords, y_coords))

        # Compute radii for each circle
        radii = []
        for i in range(num_circles):
            x_i = centers_opt[i][0]
            y_i = centers_opt[i][1]

            # Distance to boundaries
            dist_left = x_i
            dist_right = 1.0 - x_i
            dist_bottom = y_i
            dist_top = 1.0 - y_i
            min_boundary_dist = min(dist_left, dist_right, dist_bottom, dist_top)

            # Distance to nearest neighbor
            min_dist_neighbor = float('inf')

                    continue

                dy = centers_opt[i][1] - centers_opt[j][1]
                dist = math.sqrt(dx**2 + dy**2)

            # Max radius is the minimum of boundary distance and half the nearest neighbor distance
            max_radius = min(min_boundary_dist, min_dist_neighbor / 2)

        # Compute total radii
        total_radii = sum(radii)
        return -total_radii  # Minimize negative of total radii to maximize it

    result = minimize(

        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',

    sum_radii = sum(radii)
