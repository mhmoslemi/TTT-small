import math
from scipy.optimize import minimize

def run_packing():
    # Number of circles
    n = 26

    np.random.seed(42)
    initial_centers = np.random.rand(n, 2)

    def objective(x):
        # Flatten x into coordinates: x[0::2] for x-coordinates and x[1::2] for y-coordinates
        x_coords = x[0::2]
        y_coords = x[1::2]

        # Calculate radii for each circle
        radii = []
        for i in range(n):
            # Distance to boundaries
            dist_left = x_coords[i]
            dist_right = 1 - x_coords[i]

            min_boundary = min(dist_left, dist_right, dist_bottom, dist_top)

                if i == j:
                    continue
                dx = x_coords[i] - x_coords[j]
                dy = y_coords[i] - y_coords[j]
                dist = math.sqrt(dx ** 2 + dy ** 2)
                if dist < min_dist:
                    min_dist = dist

            # Radius is the minimum of boundary distance and half the distance to the nearest neighbor
            r = min(min_boundary, min_dist / 2)
            radii.append(r)

        return -sum(radii)  # Maximize sum of radii by minimizing negative sum

    # Optimization with constraints
