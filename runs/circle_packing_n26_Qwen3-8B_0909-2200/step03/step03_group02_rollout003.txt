import math
from scipy.optimize import differential_evolution

def run_packing():

    n = 26

    # Define bounds for the square [0,1]x[0,1]
    bounds = [(0.0, 1.0) for _ in range(2 * n)]  # Each circle has (x, y)

    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array where x[0::2] contains x-coordinates and x[1::2] y-coordinates
        x_coords = x[0::2]
        y_coords = x[1::2]

        # Initialize radii and constraints
        radii = []
        for i in range(n):
            # Compute distance to boundaries
            dist_left = x_coords[i]
            dist_right = 1 - x_coords[i]
            dist_bottom = y_coords[i]
            dist_top = 1 - y_coords[i]
            min_boundary = min(dist_left, dist_right, dist_bottom, dist_top)

            # Compute distance to other circles
            min_dist = float('inf')
            for j in range(n):
                if i == j:
                    continue
                dx = x_coords[i] - x_coords[j]
                dy = y_coords[i] - y_coords[j]
                dist = math.sqrt(dx ** 2 + dy ** 2)
                if dist < min_dist:

            r = min(min_boundary, min_dist / 2)
            radii.append(r)
