import numpy as np
import time

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """ 
    A gradient-based optimization to minimize the evaluation 
    function by focusing on reducing the peak of the convolution 
    relative to the squared sum of coefficients. 
    This algorithm starts from a block perturbation of the provided parent sequence 
    or a uniform fallback, then repeatedly applies a structured perturbation 
    (unless stuck) to explore the space efficiently. 
    Each evaluation requires a full convolution and handles constraints properly. 
    Optimization uses finite differences to approximate gradients and actively seeks 
    to lower the objective by adjusted block updates. 
    Retries focus on promising blocks identified by gradient ascent for improvement. 
    """
    # Extract the fallback sequence and main sequence if provided
    parent_sequence = kwargs.get("parent_sequence", np.ones(256, dtype=np.float64))[:6121]
    fallback = np.ones(6121, dtype=np.float64)  # Use uniform fallback if parent is None

    # Utility to enforce domain constraints
    def constrain(seq):
        constrained = np.array([max(0, x) for x in seq])
        constrained = np.clip(constrained, 0.0, 1000.0)
        s = np.sum(constrained)
        if s < 0.01:
            return (0.01 / s) * constrained.tolist()
        return constrained.tolist()

    all_finite = np.all(np.isfinite(parent_sequence))

    if parent_sequence.size == 0 or parent_sequence.ndim != 1 or not all_finite:
        base = constrain(fallback)
    else:
        base = constrain(parent_sequence)

    # Ensure we have a valid sequence
    best_sequence = np.array(base, dtype=np.float64)
    best_score = float(kwargs.get("best_score", evaluate_sequence(best_sequence)))
    best_sequence_global = best_sequence.copy()
    time_remaining = max(0.0, float(budget_s)) if isinstance(budget_s, (int, float)) else 2400.0

    # Combine time-based checks with gradient ascent for structured searches
    try:
        evaluation_deadline = time.time() + time_remaining  # Continuous time constraint
        score_history = []
        best_value_bound = best_score  # The goal is to beat this score

        # If the objective is above the known best, try improving; else, maintain
        if best_score > 1.5027:
            improvement_target = 1.5027

        while time.time() < evaluation_deadline and (score_history and score_history[-1] >= score_history[-2] * 0.9999):
            # Perform a block perturbation chosen by gradient ascent
            new_cand = best_sequence_global.copy()
            width = np.random.randint(max(1, 6121 // 200), max(2, 6121 // 12))  # Dynamically size block perturbation
            start = np.random.randint(0, 6121 - width + 1)
            flipped = np.random.random() < 0.35  # Flip/reflect 35% of the time

            if flipped:
                cand = new_cand[start:start+width] + new_cand[start-width:start]
                # Avoid empty passes in reflect
                if cand.size <= 0:
                    continue
            else:
                cand = new_cand[start:start+width]

            # Generate random factor (small variance)
            log_factor = np.random.normal(0.0, 0.12, cand.size)
            factor = np.exp(log_factor)  # Ensure positive scaling

            # Apply the factor to the block
            cand *= factor
            new_cand[start:start+width] = cand if not flipped else cand[::-1]

            new_cand = constrain(new_cand.tolist())
            new_score = float(evaluate_sequence(new_cand))

            if np.isfinite(new_score) and new_score < best_value_bound * 0.99:
                best_sequence_global = np.array(new_cand)
                best_value_bound = new_score
                score_history.append(best_value_bound)

        # Validate the best found sequence
        valid_best_score = best_value_bound if np.isfinite(best_value_bound) else float('inf')

        final_sequence = best_sequence_global if valid_best_score < 1.5027 else fallback
