  import numpy as np

def two_dot_crossover(a: list[float], b: list[float]) -> tuple[list[float], list[float]]:
    """
    Perform two-point crossover on two parent sequences.

    Converts the input lists into numpy arrays for easier manipulation.
    Selects two random indices as cut points and swaps the segment between them.
    Returns two child sequences.
    """
    # Convert lists to numpy arrays
    a_arr = np.array(a, dtype=np.float64)
    b_arr = np.array(b, dtype=np.float64)
    n = len(a_arr)
    # Generate two distinct random cut points (i < j)
    i = int(np.random.random() * n)
    j = int(np.random.random() * n)
    if i >= j:
        i, j = j, i + 1 if i + 1 < n else i
    # Create two children by swapping the segment from i to j
    child1 = np.concatenate([a_arr[:i], b_arr[i:j], a_arr[j:]])
    child2 = np.concatenate([b_arr[:i], a_arr[i:j], b_arr[j:]])
    return child1.tolist(), child2.tolist()

def block_perturbation(candidate: list[float], rng=None) -> list[float]:
    """
    Perform a block perturbation on a candidate sequence.
    candidate is a list of floats to be disturbed locally.
    rng is a random number generator for reproducibility.
    """
    if not hasattr(rng, 'random'):
        rng = np.random.default_rng(seed=42)
    n = len(candidate)
    # Choose a random width for the block
    low_width = max(1, n // 200)
    high_width = max(2, n // 12)
    width = int(rng.integers(low_width, high_width))
    # Choose a random start index
    start_idx = int(rng.integers(0, n - width))
    # Generate uniform random factor between [0.5, 1.5]
    scale = rng.uniform(0.5, 1.5, width)
    # Multiply the block by the scale factor
    perturbed = candidate.copy()
    perturbed[start_idx:start_idx + width] *= scale
    # With probability 5%, mirror the entire sequence and average
    if rng.random() < 0.05:
        perturbed = 0.5 * (np.array(perturbed) + np.array(perturbed)[::-1])
    # Clip the perturbed sequence to [0, 1000]
    perturbed = np.clip(perturbed, 0.0, 1000.0).tolist()
    return perturbed

def mutate_sequence(candidate: list[float], mutation_rate: float = 0.3) -> list[float]:
    """
    Apply Gaussian perturbation mutation.
    Each element has a probability of mutation_rate to be Gaussian noise.
    """
    n = len(candidate)
    # Apply noise to a fraction of elements
    noisy_mask = np.random.uniform(0, 1, n) < mutation_rate
    # Generate multiplicative Gaussian noise for the noisy_mask
    exponent = np.random.normal(0.0, 0.1, n)[noisy_mask]
    scale = np.exp(exponent) - 0.0  # Avoid zero
    # Mutate only where noisy_mask is True
    perturbed = candidate.copy()
    for i in range(n):
        if noisy_mask[i]:
            perturbed[i] *= scale
            # Ensure non-negative after multiplication
            if perturbed[i] < 0:
                perturbed[i] = np.abs(perturbed[i])
    return perturbed
