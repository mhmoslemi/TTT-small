import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8

    # Surrogate function for r_max estimation
    def surrogate(coeffs):
        coeffs = np.array(coeffs)
        coeffs = np.pad(coeffs, (0, C4_MAX_COEFF_COUNT - len(coeffs)), mode='constant')
        t0 = time.time()
        # Numerical estimation of r_max
        def Q(z):
            H = np.polynomial.hermite.hermval(z, [1.0])
            return sum(coeffs[k] * H**(4*k) for k in range(len(coeffs)))
        
        # Find roots of Q(z)/z^2
        def f(z):
            if z == 0:
                return np.inf
            return Q(z) / (z**2)
        
        # Use Newton-Raphson to find r_max
        def find_r_max():
            z = 1.0
            for _ in range(100):
                f_z = f(z)
                f_prime_z = (f(z + 1e-6) - f(z - 1e-6)) / (2 * 1e-6)
                if np.abs(f_prime_z) < 1e-6:
                    break
                z -= f_z / f_prime_z
            return z
        
        r_max = find_r_max()
        return r_max**2 / (2 * np.pi)
    
    # Optimization with random restarts
    best_coeffs = None
    best_bound = float('inf')
    
    for _ in range(10):
        if time.time() - t0 > budget_s:
            break
        start = time.time()
        init = np.random.rand(C4_MAX_COEFF_COUNT)
        res = minimize(lambda x: surrogate(x), init, method='L-BFGS-B', bounds=[(0, 10) for _ in range(C4_MAX_COEFF_COUNT)])
        if res.success:
            bound = res.fun
            if bound < best_bound:
                best_bound = bound
                best_coeffs = res.x
        print(f"Eval time: {time.time() - start:.2f}s, Current best: {best_bound}")
    
    return best_coeffs
