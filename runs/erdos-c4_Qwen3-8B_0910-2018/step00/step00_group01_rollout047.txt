Use a gradient-based search with random restarts on a smaller coefficient set, using numerical approximations of r_max to guide the optimization, and validate promising candidates with exact symbolic computation.

import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = np.array([1.0, 0.0, 0.0])
    best_value = float('inf')
    start_time = time()
    
    def evaluate(coeffs):
        t = sp.symbols('t')
        Q = 0
        for k in range(len(coeffs)):
            Q += coeffs[k] * sp.herite(4*k, t)
        Q = Q.as_poly()
        Q = Q - Q.as_poly().coeffs()[0]  # Ensure Q(0) = 0
        if Q.as_poly().all_coeffs()[-1] < 0:
            Q *= -1
        Q = Q.as_poly().as_expr() / t**2
        roots = sp.roots(Q)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if len(real_roots) > 0 else 1.0
        return r_max**2 / (2 * np.pi)
    
    def surrogate(coeffs):
        t = sp.symbols('t')
        Q = 0
        for k in range(len(coeffs)):
            Q += coeffs[k] * sp.herite(4*k, t)
        Q = Q.as_poly()
        Q = Q - Q.as_poly().coeffs()[0]
        if Q.as_poly().all_coeffs()[-1] < 0:
            Q *= -1
        Q = Q.as_poly().as_expr() / t**2
        roots = sp.nsolve(Q, t, 1.0)
        while roots == None:
            roots = sp.nsolve(Q, t, 2.0)
        r_max = roots.evalf()
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return -surrogate(coeffs)
    
    for _ in range(50):
        if time() - start_time > budget_s - 10:
            break
        coeffs = np.random.uniform(-1.0, 1.0, size=6)
        res = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-2, 2) for _ in range(6)])
        if time() - start_time > budget_s - 10:
            break
        coeffs = res.x
        val = evaluate(coeffs)
        if val < best_value:
            best_value = val
            best_coeff = coeffs
    
    return best_coeff
