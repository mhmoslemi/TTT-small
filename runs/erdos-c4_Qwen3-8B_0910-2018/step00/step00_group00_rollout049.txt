import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermitian
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.random.rand(8)  # Start with 8 random coefficients

    def objective(coeffs):
        # Build Hermite polynomial Q(z)
        coeffs = np.array(coeffs, dtype=float)
        m = len(coeffs)
        H = [hermitian(4 * k, 1) for k in range(m)]
        Q = sum(c * h for c, h in zip(coeffs, H))

        # Enforce Q(0) = 0
        const_term = Q[0]
        coeffs[0] -= const_term / (4**0)  # Adjust c_0 to make Q(0) = 0

        # Find r_max, the largest real root where Q(z)/z² changes sign
        def func(z):
            z2 = z ** 2
            return Q(z) / z2 if z2 != 0 else np.inf

        try:
            result = root_scalar(func, bracket=(1, 100), method='brentq')
            r_max = result.root
            return r_max ** 2 / (2 * np.pi)
        except:
            return float('inf')

    # Use L-BFGS-B with restarts
    best_coeffs = initial_coefficients
    best_value = float('inf')
    start_time = time.time()

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, initial_coefficients, method='L-BFGS-B', bounds=[(0, 1) for _ in range(8)])
        if res.success:
            current_value = res.fun
            if current_value < best_value:
                best_value = current_value
                best_coeffs = res.x

    return best_coeffs
