import numpy as np
from scipy.optimize import minimize
from sympy import symbols, simplify, diff, HermiteH, solve, roots, symbols, Rational, N
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial = np.array([0.4, 0.3, 0.2])
    best_coeffs = initial.copy()
    best_bound = float('inf')
    
    def evaluate_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        c_m = -Q / (HermiteH(4*m, 0) if m > 0 else 1)
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m)) + c_m * HermiteH(4*m, z)
        Q = Q / z**2
        try:
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = np.array([N(r) for r in real_roots])
            real_roots = np.sort(np.unique(real_roots))
            r_max = real_roots[-1] if len(real_roots) > 0 else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(coeffs):
        nonlocal best_bound
        if len(coeffs) > 8:
            return float('inf')
        b = evaluate_bound(coeffs)
        if b < best_bound:
            best_bound = b
            best_coeffs = coeffs.copy()
        return b
    
    def optimize():
        for _ in range(10):
            result = minimize(objective, initial, bounds=[(-10, 10) for _ in range(8)], method='L-BFGS-B', options={'maxiter': 100})
            if result.success:
                initial = result.x
            else:
                break
    
    start_time = time.time()
    optimize()
    remaining_time = max(0, budget_s - (time.time() - start_time))

            for j in range(i+1, 8):
                new_coeffs = best_coeffs.copy()

                new_coeffs[j] -= 0.1
                b = evaluate_bound(new_coeffs)

                    best_bound = b
                    best_coeffs = new_coeffs
