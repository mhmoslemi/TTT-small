import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate(coeffs):
        from sympy import symbols, Hermite, expand, solve, Poly
        z = symbols('z')
        H = [Hermite(k) for k in range(1, len(coeffs)+1)]
        Q = sum(c * H[i] for i, c in enumerate(coeffs))
        Q = Q.subs(z, 0)
        Q = Q - Q.as_independent(z)[1].as_coefficient(z**2)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if r > 0:
                return r**2 / (2 * np.pi)
        return float('inf')
    
    def get_finite_coeffs(coeffs):
        finite_coeffs = [c for c in coeffs if np.isfinite(c)]
        return finite_coeffs
    
    def optimize_coefficients():
        initial = np.random.uniform(-1, 1, size=3)
        result = minimize(lambda x: evaluate(x), x0=initial, bounds=[(-5, 5) for _ in range(3)], method='L-BFGS-B', options={'maxiter': 100})

        best_bound = evaluate(best_coeffs)

            best_coeffs = get_finite_coeffs(best_coeffs)
        return best_coeffs
    
    best_coeffs = optimize_coefficients()
