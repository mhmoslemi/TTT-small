import numpy as np
from scipy.optimize import root_scalar
from functools import partial
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coeff = np.array([1.0, 0.5, -0.3])  # Initial guess with 3 coefficients
    
    def objective(coeffs):
        m = len(coeffs)
        # Construct Q(z) = sum c_k H_{4k}(z)
        # Use SymPy for exact evaluation of Hermite polynomials
        import sympy as sp
        z = sp.symbols('z')
        H = [sp.hermite(4*k, z) for k in range(m)]
        Q = sum(coeffs[k] * H[k] for k in range(m))
        
        # Enforce Q(0) = 0 and adjust sign for positivity at infinity
        if Q.subs(z, 0) != 0:
            coeffs = [-c for c in coeffs]
            Q = -Q
        
        # Remove z^2 factor
        Q = Q.as_leading_coefficient() * Q.as_poly().as_expr() / z**2
        
        # Find largest positive real root with sign change
        def f(x):
            return Q.subs(z, x).evalf()
        
        try:
            res = root_scalar(f, bracket=[1, 100], method='brentq', xtol=1e-8)
            r_max = res.root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Use a small population of candidate vectors for evolution
    def evolve(population, mutation_rate=0.1):
        new_pop = []
        for vec in population:
            new_vec = vec + np.random.normal(0, mutation_rate, len(vec))
            new_pop.append(new_vec)
        return new_pop
    
    population = [initial_coeff.copy() for _ in range(20)]
    best = initial_coeff
    best_score = objective(initial_coeff)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for vec in population:
            score = objective(vec)
            if score < best_score:
                best_score = score
                best = vec.copy()
        
        # Evolve better population
        population = evolve(population)
    
    # Ensure at most 8 coefficients
    if len(best) > 8:
        best = best[:8]
    return best
