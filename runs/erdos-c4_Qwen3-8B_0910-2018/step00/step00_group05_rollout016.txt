import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

        from scipy.optimize import Bounds
        def surr_func(x):
            return np.sum((x ** 2) * np.sin(x))
        return minimize(surr_func, coeffs, bounds=Bounds(-5, 5), method='L-BFGS-B').fun

        e_x = np.exp(x / temp)
        return e_x / e_x.sum()
    
    def generate_candidate():
        temp = 1.0
        base = np.random.randn(8)
        weights = softmax(base, temp)
        coeffs = np.random.randn(8) * weights
        return coeffs
    
    best_coeffs = generate_candidate()
    best_value = objective(best_coeffs)
    best_time = 0
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        candidates = [generate_candidate() for _ in range(10)]
        surrogate_values = [fast_surrogate(c) for c in candidates]
        idx = np.argmin(surrogate_values)

        try:
            val = objective(candidate)
