import numpy as np
from scipy.optimize import minimize
from sympy import Hermite, symbols, Eq, solve
from time import time

def run(seed=42, budget_s=600, **kwargs):

    initial_coefficients = np.array([1.0, 1.0, 1.0])  # Initial guess
    best_coeff = initial_coefficients
    best_bound = float('inf')
    time_start = time()
    
    def evaluate_coeffs(c):
        coeffs = np.array(c)
        m = len(coeffs)
        z = symbols('z')

        # Enforce Q(0) = 0

        Q = Q / z**2

        roots = solve(Q, z)

        real_roots = [r for r in real_roots if r > 0 and r != 0]
        # Remove even multiplicity roots
        unique_roots = []
        for r in real_roots:
            if not unique_roots or abs(r - unique_roots[-1]) > 1e-8:
                unique_roots.append(r)
        if not unique_roots:
            return float('inf')
        r_max = max(unique_roots)
        return r_max**2 / (2 * np.pi)
    
    def objective(x):

        if bound < best_bound:
            best_bound = bound
            best_coeff = x
