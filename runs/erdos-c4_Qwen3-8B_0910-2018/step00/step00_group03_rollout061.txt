import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import root_scalar
from scipy.optimize import minimize
from scipy import special

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    def evaluate_upper_bound(coeffs):
        m = len(coeffs)
        # Compute Hermite polynomials H_4k(z) up to needed degree
        coeffs = np.array(coeffs, dtype=np.float64)
        # Generate Hermite polynomials up to H_4m
        max_deg = 4*m
        z = np.linspace(-100, 100, 10000)
        H = np.zeros((max_deg + 1, len(z)))
        for n in range(max_deg + 1):
            H[n] = special.hermite(n, z)
        # Build Q(z)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * H[4*k]
        # Normalize by z^2 to make Q(0) = 0
        Q /= z**2
        # Find r_max: largest positive real root where Q(z) changes sign
        def sign_change(x):
            return np.sign(Q[np.abs(z - x) < 1e-3])
        r_max = None
        for x in z[z > 0]:
            if np.abs(x) < 1e-3:
                continue
            if np.sign(Q[np.abs(z - x) < 1e-3]) != 0:
                continue
            if np.sign(Q[np.abs(z - x) < 1e-3]) != np.sign(Q[np.abs(z - x) < 1e-3]):
                r_max = x
                break
        if r_max is None:
            # Fallback to numerical root finding
            try:
                root = root_scalar(lambda x: Q[np.abs(z - x) < 1e-3], bracket=[0, 100], method='brentq')
                r_max = root.root
            except:
                r_max = 100
        return r_max ** 2 / (2 * np.pi)

    def objective(coeffs):
        return evaluate_upper_bound(coeffs)

    def update_coefficients(coeffs):
        # Expand to 8 coefficients if needed
        if len(coeffs) < 8:
            new_coeffs = np.zeros(8, dtype=np.float64)
            new_coeffs[:len(coeffs)] = coeffs
            return new_coeffs
        return coeffs

    def random_search():
        best_coeffs = np.random.rand(3) - 0.5
        best_bound = objective(best_coeffs)
        for _ in range(200):
            coeffs = np.random.rand(3) - 0.5
            current_bound = objective(coeffs)
            if current_bound < best_bound:
                best_coeffs = coeffs
                best_bound = current_bound
        return update_coefficients(best_coeffs)

    def optimize_coeffs(coeffs):
        result = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(len(coeffs))], tol=1e-6)
        return result.x

    initial_coeffs = np.array([0.5, -0.25, 0.25], dtype=np.float64)
    best_coeffs = initial_coeffs
    best_bound = objective(best_coeffs)

    for _ in range(10):
        coeffs = np.random.rand(3) - 0.5
        current_bound = objective(coeffs)
        if current_bound < best_bound:
            best_coeffs = coeffs
            best_bound = current_bound

    # Optimize the best found coefficients
    best_coeffs = optimize_coeffs(best_coeffs)
    best_coeffs = update_coefficients(best_coeffs)

    return best_coeffs
