import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, Hermite, expand, solve, Eq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, -1.0, 1.0])
    
    def objective(c):
        c = np.array(c, dtype=np.float64)
        if len(c) > 8:
            return float('inf')
        try:
            # Fast numerical surrogate
            z = np.linspace(-10, 10, 1000)
            H = np.array([Hermite(4*k)(z) for k in range(len(c))]).T
            Q = H @ c
            Q /= z**2
            r_max = np.max(z[np.abs(Q) > 1e-6])
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def exact_eval(c):
        c = np.array(c, dtype=np.float64)
        if len(c) > 8:
            return float('inf')
        try:
            z = symbols('z')
            H = [Hermite(4*k)(z) for k in range(len(c))]
            Q = sum(c[k] * H[k] for k in range(len(c)))
            Q = Q - Q.subs(z, 0)
            if Q.as_coefficients_dict().get(z**2, 0) != 0:
                Q = Q / z**2
            Q = Q.as_numerical()
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            real_roots.sort()
            for i in range(len(real_roots) - 1):
                if real_roots[i+1] - real_roots[i] > 1e-6:
                    r_max = real_roots[i+1]
                    return r_max**2 / (2 * np.pi)
            return float('inf')
        except:
            return float('inf')
    
    def generate_candidates(n_samples=100):
        candidates = []
        for _ in range(n_samples):
            c = np.random.uniform(-2, 2, size=8)
            candidates.append(c)
        return candidates
    
    def optimize():
        start = time.time()
        best = initial_coefficients
        best_score = objective(best)
        for _ in range(50):
            candidates = generate_candidates(n_samples=20)
            scores = [objective(c) for c in candidates]
            idx = np.argmin(scores)
            best = candidates[idx]
            best_score = scores[idx]
            if time.time() - start > budget_s - 10:
                break
        return best
    
    best = optimize()
    # Refine with exact evaluation
    final_coeff = best
    final_coeff = np.clip(final_coeff, -2, 2)
    return final_coeff
