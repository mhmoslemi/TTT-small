import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, simplify
from sympy.polys import root-lo

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, 0.5, 0.5])

    def objective(coeffs):
        coeffs = np.clip(coeffs, -10, 10)
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positive at infinity
        if Q.as_leading_term(z) < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().remove_term(2)
        # Find r_max
        try:
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            valid_roots = []
            for r in real_roots:
                if r > 0:
                    valid_roots.append(r)
            if not valid_roots:
                return float('inf')
            r_max = valid_roots[-1]
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    def surrogate_objective(coeffs):
        return objective(coeffs)

    def optimize():
        result = minimize(surrogate_objective, initial_coefficients, method='L-BFGS-B', bounds=[(-10, 10) for _ in initial_coefficients])
        best_coeffs = result.x
        return best_coeffs

    start_time = time.time()
    best_coeffs = optimize()
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.1, len(best_coeffs))
        new_coeffs = best_coeffs + perturb
        new_coeffs = np.clip(new_coeffs, -10, 10)
        try:
            score = objective(new_coeffs)
            if score < objective(best_coeffs):
                best_coeffs = new_coeffs
        except:
            pass
        time.sleep(0.1)

    return best_coeffs
