import matplotlib.pyplot as plt

import time

def run(seed=42, budget_s=600, **kwargs):

        m = len(c)
        H = [lambda x: (x**2)**k for k in range(m)]

        Q = Q - Q(0)
        
        def f(z):
            return Q(z)
        
        try:
            roots = np.roots([1] + [-c[0] if k == 0 else -c[k] for k in range(m)])

            positive_roots = np.unique(positive_roots)
            
            def sign_change(z):
                return np.sign(f(z))
            
            for r in positive_roots:
                if np.isclose(f(r), 0):
                    continue
                if np.sign(f(r - 1e-5)) != np.sign(f(r + 1e-5)):
                    return r**2 / (2 * np.pi)
            
            return float('inf')
        except:
            return float('inf')
    
    def exact_check(c):
        m = len(c)
        z = symbols('z')
        H = [Hermitian(z, k) for k in range(m)]
