import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    def evaluate(coeffs):
        import sympy
        z = sympy.symbols('z')
        H = [sympy.hermite(n, z) for n in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q.subs(z, 0).as_numerical()
        if Q[0] < 0:
            Q = -Q
        Q = Q.as_poly()
        roots = Q.all_roots()
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate(x)
    
    def f(x):
        return -objective(x)
    
    def initial_guess():
        return np.random.rand(3)
    
    best = initial_guess()
    best_val = objective(best)
    for n in range(3, 9):
        if time.time() > budget_s:
            break
        bounds = Bounds([0]*n, [1]*n)
        res = minimize(f, best, method='L-BFGS-B', bounds=bounds)
        if res.success:
            best = res.x
            best_val = objective(best)
            print(f"Try {n} coeffs: {best_val}")
        else:
            print(f"Minimize failed for {n} coeffs")
    
    return best
