import matplotlib.pyplot as plt
from scipy.optimize import minimize, root_scalar
import sympy as sp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    initial_coefficients = np.random.rand(C4_MAX_COEFF_COUNT)
    
    def evaluate_coefficients(coeffs):
        t0 = time()
        z = sp.symbols('z')
        H = [sp.herite(n, z) for n in range(4*C4_MAX_COEFF_COUNT + 1)]
        Q = sum(coeffs[k] * H[4*k] for k in range(len(coeffs)))
        Q = Q - Q.as_leads()[0] * z**2
        roots = sp.solveset(Q, z, domain=sp.Reals)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted(real_roots)
        for root in real_roots:
            if np.isclose(root, 0):
                real_roots.remove(root)
        filtered_roots = []
        for i in range(len(real_roots)):
            if i == 0:
                if real_roots[i] > 0:
                    filtered_roots.append(real_roots[i])
            else:
                if real_roots[i] > 0 and real_roots[i] > real_roots[i-1]:
                    filtered_roots.append(real_roots[i])
        if not filtered_roots:
            return float('inf')
        r_max = filtered_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_coefficients(coeffs)
    
    def evaluate_and_check(coeffs, tolerance=1e-8):

            return val
        except:
            return float('inf')

            options={'ftol': 1e-12, 'gtol': 1e-12}
        )

    start_time = time()

    best_value = evaluate_and_check(best_coeffs)
