import numpy as np
from scipy.optimize import minimize
from scipy.optimize import fsolve
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, simplify, Poly, roots
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def numerical_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        
        # Enforce Q(0) = 0 and positive at infinity
        Q = Q.subs(z, 0).as_coefficients_dict()[1] * Q / Q.subs(z, 0)
        
        # Remove z^2 factor
        Q = Q.as_poly().as_expr() / z**2
        
        # Find r_max numerically
        try:
            def f(r):
                return Q.subs(z, r)
            
            r_min = 0.1
            r_max = 100
            try:
                root = root_scalar(f, bracket=[r_min, r_max], method='brentq')
                r_max = root.root
            except:
                r_max = 100
            
            if r_max < 0.1:
                return float('inf')
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def exact_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        
        # Enforce Q(0) = 0 and positive at infinity
        Q = Q.subs(z, 0).as_coefficients_dict()[1] * Q / Q.subs(z, 0)
        
        # Remove z^2 factor
        Q = Q.as_poly().as_expr() / z**2
        
        # Find r_max exactly
        try:
            poly = simplify(Q)
            roots = roots(poly)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            r_max = 0
            for r in real_roots:
                if np.isclose(r, 0):
                    continue
                if r > r_max:
                    r_max = r
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def fitness(coeffs):
        return exact_bound(coeffs)
    
    def objective(x):
        return numerical_bound(x)
    
    def gradient(x):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x1 = x.copy()
            x2 = x.copy()
            x1[i] += eps
            x2[i] -= eps
            grad[i] = (objective(x1) - objective(x2)) / (2 * eps)
        return grad
    
    def optimize():
        initial_guess = np.random.rand(4)
        result = minimize(objective, initial_guess, jac=gradient, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(4)], options={'maxiter': 100})
        return result.x
    
    start_time = time.time()
    best_coeffs = optimize()
    while time.time() - start_time < budget_s - 1:
        new_guess = best_coeffs + np.random.normal(0, 0.1, 4)
        new_guess = np.clip(new_guess, -10, 10)
        new_score = numerical_bound(new_guess)
        if new_score < numerical_bound(best_coeffs):
            best_coeffs = new_guess
        time.sleep(0.1)
    
    return best_coeffs
