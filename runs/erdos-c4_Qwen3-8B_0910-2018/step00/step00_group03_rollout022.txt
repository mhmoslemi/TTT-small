Use gradient descent with a fast numerical proxy to find promising coefficient sets, then refine with exact symbolic computation. Start with 3 coefficients, expand up to 8 if needed. Balance exploration and exploitation to find low U(c).

import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, diff, solve, simplify
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, 1.0, 1.0])
    best_coeff = initial_coefficients
    best_score = float('inf')
    start_time = time.time()
    
    def numerical_proxy(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2*np.pi)
    
    def objective(coeffs):
        return numerical_proxy(coeffs)
    
    def refine(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        Q = Q.as_poly()
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2*np.pi)
    
    def search(coeffs):
        result = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in coeffs])
        return result.x
    
    def expand_coeffs(coeffs):
        m = len(coeffs)
        return np.concatenate((coeffs, np.zeros(8 - m)))
    
    for expansion in range(3):
        current_coeffs = expand_coeffs(best_coeff)
        for _ in range(10):
            if time.time() - start_time > budget_s - 10:
                break
            current_coeffs = search(current_coeffs)
            current_score = refine(current_coeffs)
            if current_score < best_score:
                best_score = current_score
                best_coeff = current_coeffs
        if len(best_coeff) < 8:
            best_coeff = expand_coeffs(best_coeff)
    
    return best_coeff
