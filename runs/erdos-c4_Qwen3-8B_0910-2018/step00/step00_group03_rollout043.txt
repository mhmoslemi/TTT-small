import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, 1.0, 1.0])
    best_coeff = initial_coefficients
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score
        t0 = time.time()
        # Fast numerical surrogate
        c = np.clip(c, 1e-6, 1e6)
        c /= c.sum()
        t1 = time.time()
        if t1 - t0 > 1:
            return 1000
        # Exact check with heuristic cost
        c = np.log(c)
        c = np.exp(c)
        c /= c.sum()
        return 1000
    
    result = minimize(objective, initial_coefficients, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)], options={'maxiter': 100})
    best_coeff = result.x
    best_coeff /= best_coeff.sum()
    best_coeff = np.clip(best_coeff, 1e-6, 1e6)
    best_coeff = np.log(best_coeff)
    best_coeff = np.exp(best_coeff)
    best_coeff /= best_coeff.sum()
    return best_coeff[:8]
