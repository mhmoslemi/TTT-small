import numpy as np
from scipy.optimize import newton
from sympy import symbols, HermiteH, simplify, solve, roots, Eq, pi
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(c):
        m = len(c)
        z = symbols('z')
        Q = 0
        for k in range(m):
            Q += c[k] * HermiteH(4*k, z)
        Q = simplify(Q)
        
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        
        # Make polynomial positive at infinity
        leading_term = Q.as_leading_term(z)
        if leading_term.as_coefficient(z) < 0:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q / z**2
        Q = simplify(Q)
        
        # Find real roots
        roots_real = []
        for r in roots(Q, z, domain='R'):
            if isinstance(r, tuple):
                for root in r:
                    roots_real.append(root)
            else:
                roots_real.append(r)
        
        roots_real = sorted([r for r in roots_real if r > 0])
        r_max = None
        for r in roots_real:
            if r > 0:
                r_max = r
                break
        
        if r_max is None:
            return float('inf')
        
        return r_max**2 / (2 * pi)
    
    def score(c):
        try:
            return evaluate_coefficients(c)
        except:
            return float('inf')
    
    def search():
        best = np.array([1.0, 1.0, 1.0])
        best_score = score(best)
        
        for _ in range(10):

                candidates.append(best[i] - 0.1)

                print(f"Improved score: {best_score}")
