import numpy as np
from scipy.optimize import minimize
from scipy.special import hermition
from sympy import symbols, Rational, simplify, solve, Eq, exp, pi
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(c):
        c = np.array(c, dtype=np.float64)
        n = len(c)
        coeffs = np.zeros(n + 1)
        coeffs[:n] = c
        # Construct Q(z) = sum c_k H_{4k}(z)
        z = symbols('z')
        Q = sum(coeffs[k] * hermition(4*k)(z) for k in range(n))
        # Enforce Q(0) = 0 and positive at infinity
        Q = Q - Q.as_coefficients_dict()[z**0] * z**2
        # Find real roots of Q(z)/z^2
        expr = Q / z**2
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        # Sort and find sign changes
        real_roots.sort()
        sign_changes = []
        for i in range(1, len(real_roots)):
            if np.sign(expr.subs(z, real_roots[i-1])) != np.sign(expr.subs(z, real_roots[i])):
                sign_changes.append(real_roots[i])
        if not sign_changes:
            return float('inf')
        r_max = max(sign_changes)
        return r_max**2 / (2 * pi)
    
    def exact_objective(c):
        c = np.array(c, dtype=np.float64)
        n = len(c)
        coeffs = np.zeros(n + 1)
        coeffs[:n] = c
        z = symbols('z')
        Q = sum(coeffs[k] * hermition(4*k)(z) for k in range(n))
        Q = Q - Q.as_coefficients_dict()[z**0] * z**2
        expr = Q / z**2
        roots = solve(expr, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        sign_changes = []
        for i in range(1, len(real_roots)):
            if np.sign(expr.subs(z, real_roots[i-1])) != np.sign(expr.subs(z, real_roots[i])):
                sign_changes.append(real_roots[i])
        if not sign_changes:
            return float('inf')
        r_max = max(sign_changes)
        return r_max**2 / (2 * pi)
    
    def num_eval(c):
        c = np.array(c, dtype=np.float64)
        n = len(c)
        z = np.linspace(-10, 10, 100000)
        Q = np.zeros_like(z)
        for k in range(n):
            H = hermition(4*k)(z)
            Q += c[k] * H
        Q /= z**2
        sign_changes = np.where(np.sign(Q[1:]) != np.sign(Q[:-1]))[0]
        if not sign_changes.size:
            return float('inf')
        r_max = z[sign_changes[0]] if sign_changes[0] > 0 else z[-1]
        return r_max**2 / (2 * pi)
    
    def optimize():
        initial_guess = np.random.rand(8)
        result = minimize(
            num_eval,
            initial_guess,
            method='L-BFGS-B',
            bounds=[(-10, 10) for _ in range(8)],
            options={'maxiter': 100, 'disp': False}
        )
        final_c = result.x
        # Validate with exact method
        final_bound = exact_objective(final_c)
        return final_c
    
    start_time = time.time()
    best_c = optimize()
    while time.time() - start_time < budget_s:
        new_c = best_c + np.random.normal(0, 0.1, best_c.shape)
        new_c = np.clip(new_c, -10, 10)
        new_bound = exact_objective(new_c)
        if new_bound < exact_objective(best_c):
            best_c = new_c
    return best_c.tolist()
