import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from sympy import symbols, Eq, solve, HermiteH, expand

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        c = np.array(x)
        m = len(c)
        
        # Build Hermite polynomial
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(m))
        
        # Ensure Q(0) = 0 and make it positive infinity
        if Q.subs(z, 0) != 0:
            raise ValueError("Q(0) != 0")
        if Q.as_leading_coefficient() < 0:
            Q *= -1
        
        # Remove z^2 factor
        Q = Q.as_poly().as_expr()
        Q = Q.subs(z, z**2).as_poly().as_expr()
        
        # Find r_max numerically
        try:
            r_max = root_scalar(lambda r: Q.subs(z, r), bracket=[0.1, 100], method='brentq', rtol=1e-8)
            return r_max.root**2 / (2*np.pi)
        except:
            return float('inf')
    
    # Initial guess
    initial_guess = np.random.rand(4)
    
    # Optimization
    result = minimize(objective, initial_guess, method='L-BFGS-B', options={'gtol': 1e-8, 'ftol': 1e-8})
    
    # Return best coefficients
    return result.x
