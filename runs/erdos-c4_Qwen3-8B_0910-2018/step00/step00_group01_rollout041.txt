from scipy.optimize import minimize, root_scalar
from scipy.special import hermitian
import time

# Precompute Hermite polynomials up to H_32
def hermite_basis(n):
    return [hermitian(2*k).tolist()[0] for k in range(n+1)]

def evaluate_Q(coeffs, z):
    H = hermite_basis(len(coeffs))
    return sum(c * H[i](z) for i, c in enumerate(coeffs))

def compute_r_max(coeffs):

        return evaluate_Q(coeffs, z)

def objective(coeffs):

    if r == np.inf:

    return r**2 / (2 * np.pi)
