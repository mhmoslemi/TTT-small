Use random search with gradient ascent on a fast numerical proxy to find promising coefficient vectors, then refine with exact symbolic calculations to verify the upper bound.
</strategy>

import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize.minpack import root_scalar
from scipy.optimize import Bounds
from scipy.optimize import BoundsError
from scipy.optimize import minimize_scalar
from scipy.optimize import Bounds
from scipy.optimize import fsolve
from sympy import symbols, HermiteH, simplify, Eq, solve
from sympy.polys.polytools import factor_list
from sympy import PolynomialRing, QQ, Poly, discriminant, roots, N
from sympy.abc import z

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_upper = float('inf')
    best_coeffs = np.array([1.0, 1.0, 1.0])
    time_used = 0

    def eval_proxy(coeffs):
        nonlocal time_used
        if time_used >= budget_s:
            return best_upper
        start = time.time()
        try:
            m = len(coeffs)
            # Construct symbolic Q(z) = sum_{k=0}^{m-1} c_k H_{4k}(z)
            z_var = symbols('z')
            Q = 0
            for k in range(m):
                Q += coeffs[k] * HermiteH(4*k, z_var)
            # Ensure Q(0) = 0
            Q = Q.subs(z_var, 0)
            # Flip sign if needed to make Q(z)/z^2 positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q *= -1
            # Remove z^2 factor (equivalent to Q(z)/z^2)
            # Compute Q(z^2) / z^2
            Q_z2 = Q.subs(z_var, z_var**2)
            # Compute Q_z2 / z^2 = Q(z^2)/z^2
            # But we need to find r_max such that Q(z^2)/z^2 changes sign
            # Find real roots of Q_z2 = 0
            # First, substitute z^2 = t to get Q(t)
            t = symbols('t')
            Q_t = Q_z2.subs(z_var, t)
            # Find real roots of Q_t = 0
            roots_expr = roots(Q_t, t)
            real_roots = [r for r in roots_expr if r.is_real]
            # Sort real roots and find r_max
            real_roots = np.sort([float(r) for r in real_roots if r > 0])
            if len(real_roots) == 0:
                return float('inf')
            # Find r_max where Q_z2 changes sign
            # Try binary search on interval [0, max_real]
            max_real = real_roots[-1] if len(real_roots) > 0 else 1
            try:
                r_max = root_scalar(lambda x: Q_z2.subs(z_var, x), bracket=[0.1, max_real], method='brentq').root
                upper_bound = r_max**2 / (2 * np.pi)
                return upper_bound
            except:
                return float('inf')
        finally:
            time_used += time.time() - start
        return float('inf')

    def fitness(coeffs):
        return eval_proxy(coeffs)

    def update_best(coeffs, current_upper):
        nonlocal best_upper, best_coeffs
        if current_upper < best_upper:
            best_upper = current_upper
            best_coeffs = coeffs.copy()

    for _ in range(100):
        if time_used >= budget_s:
            break
        # Random search with perturbations
        coeffs = np.random.uniform(-1, 1, size=8)
        current_upper = fitness(coeffs)
        update_best(coeffs, current_upper)

    # Try gradient ascent from best_coeffs
    def objective(coeffs):
        return fitness(coeffs)

    def grad_objective(coeffs):
        # Estimate gradient using finite differences
        grad = np.zeros_like(coeffs)
        eps = 1e-6
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            val_plus = objective(coeffs_plus)
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            val_minus = objective(coeffs_minus)
            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad

    # Gradient ascent
    for _ in range(100):
        if time_used >= budget_s:
            break
        grad = grad_objective(best_coeffs)
        best_coeffs = best_coeffs - 0.1 * grad
        current_upper = objective(best_coeffs)
        update_best(best_coeffs, current_upper)

    return best_coeffs
