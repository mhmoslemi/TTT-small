import random
from scipy.optimize import minimize
from sympy import symbols, Hermite, expand, solve, Poly

def run(seed=42, budget_s=600, **kwargs):

        coeffs = np.array(coeffs)
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        Q = Q.as_coefficients_dict()
        c_m = -Q[0] if 0 in Q else 0
        coeffs = np.append(coeffs, c_m)
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m+1))
        Q = Q.as_poly()
        zeros = solve(Q, z)
        real_roots = [r.evalf() for r in zeros if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def exact_upper_bound(coeffs):
        from sympy import Rational
        coeffs = [Rational(c) for c in coeffs]
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(m))
        Q = Q.as_poly()

            scores = np.array([surrogate_upper_bound(ind) for ind in population])

    best_coeffs = refine_candidate(best_coeffs)
