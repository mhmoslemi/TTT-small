import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, roots, simplify
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([0.5, 0.5, 0.5])
    best_coeff = initial_coefficients
    best_bound = float('inf')
    start_time = time()
    
    def eval_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q - Q.as_leading_term(z**2)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() == 0:
                continue
            if (Q.subs(z, r - 1e-6).evalf() * Q.subs(z, r + 1e-6).evalf()) < 0:
                r_max = r
                break
        else:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    def fast_surrogate(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q - Q.as_leading_term(z**2)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() == 0:
                continue
            if (Q.subs(z, r - 1e-3).evalf() * Q.subs(z, r + 1e-3).evalf()) < 0:
                r_max = r
                break
        else:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return fast_surrogate(coeffs)
    
    def exact_objective(coeffs):
        return eval_bound(coeffs)
    
    def optimize(coeffs):
        res = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-5, 5) for _ in range(len(coeffs))], options={'max_iter': 100})
        return res.x
    
    def refine(coeffs):
        for _ in range(5):
            res = optimize(coeffs)
            if exact_objective(res) < exact_objective(coeffs):
                coeffs = res
            else:
                break
        return coeffs
    
    for _ in range(20):

            break
        refined = refine(best_coeff)
        current_bound = exact_objective(refined)
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = refined
    
    return best_coeff
