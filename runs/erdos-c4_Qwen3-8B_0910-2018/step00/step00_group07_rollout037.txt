import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, S
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, 1.0, 1.0])
    best_coeff = initial_coefficients
    best_bound = float('inf')
    
    def objective(coeffs):
        nonlocal best_coeff, best_bound
        coeffs = np.array(coeffs, dtype=float)
        # Construct Q(z)
        z = symbols('z')
        Q = 0
        for k in range(len(coeffs)):
            Q += coeffs[k] * HermiteH(4*k, z)
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity
        if Q.as_leading_term(z) < 0:
            Q = -Q
        # Remove z² factor
        Q = Q.as_poly().div(z**2)[1]
        # Find roots
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        # Filter for sign change
        sign_changes = []
        for r in real_roots:
            if r > 0:
                sign_changes.append(r)
        if not sign_changes:
            return float('inf')
        r_max = max(sign_changes)
        bound = r_max**2 / (2 * np.pi)
        return bound

    def inner_search(coeffs):
        result = minimize(lambda x: objective(x), x0=coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in coeffs])
        return result.x

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        # Random search
        trial_coeffs = np.random.uniform(-10, 10, size=8)
        bound = objective(trial_coeffs)
        if bound < best_bound:
            best_bound = bound
            best_coeff = trial_coeffs
        # Local search
        local_coeffs = inner_search(best_coeff)
        bound = objective(local_coeffs)
        if bound < best_bound:
            best_bound = bound
            best_coeff = local_coeffs
    return best_coeff
