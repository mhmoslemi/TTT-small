import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, Eq, solve, symbols, diff, simplify
import time

def hermite_evaluate(coeffs, z):
    z_sym = symbols('z')
    h = sum(c * Hermite(4*k, z_sym) for k, c in enumerate(coeffs))
    return h

def exact_bound(coeffs):
    z = symbols('z')
    h = hermite_evaluate(coeffs, z)
    Q = h / z**2
    # Find sign change
    roots = solve(Q, z)
    real_roots = [r.evalf() for r in roots if r.is_real]
    real_roots = sorted([r for r in real_roots if r != 0])
    positive_roots = [r for r in real_roots if r > 0]
    for r in positive_roots:
        if r > 0:
            r_max = r
            break
    return r_max**2 / (2 * np.pi)

def numerical_surrogate(coeffs):
    z = symbols('z')
    h = hermite_evaluate(coeffs, z)
    Q = h / z**2
    # Find real positive roots numerically
    from scipy.optimize import root_scalar
    def f(z):
        return Q.subs(z, z).evalf()
    try:
        root = root_scalar(f, bracket=[1, 100], method='brentq')
        return root.root**2 / (2 * np.pi)
    except:
        return float('inf')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coeffs = np.array([1., 1., 1.])
    
    def objective(x):
        return numerical_surrogate(x)
    
    result = minimize(
        objective,
        x0=initial_coeffs,
        method='L-BFGS-B',
        bounds=[(-10, 10) for _ in range(8)],
        options={'maxiter': 100}
    )
    
    best_coeffs = result.x
    # Final refinement with exact evaluation
    final_bound = exact_bound(best_coeffs)
    return best_coeffs
