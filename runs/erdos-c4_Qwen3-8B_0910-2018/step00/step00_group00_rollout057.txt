import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, 0.0, 0.0])
    best_coefficients = initial_coefficients.copy()
    best_bound = float('inf')

    def evaluate_fast(coefficients):
        coeffs = coefficients.copy()
        coeffs = np.concatenate((coefficients, [0.0]))
        m = len(coeffs)
        H = np.zeros((m, m))
        for k in range(m):
            H[:, k] = np.polysum([1.0, 0.0], [0, 1, 0, 0]) if k == 0 else np.polysum([1.0, 0.0], [0, 1, 0, 0])
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q[0] = 0.0
        Q = Q[1:]
        r = np.roots(Q)
        r_real = r[np.isreal(r)]
        r_pos = r_real[r_real > 0]
        r_max = np.max(r_pos)
        if np.isfinite(r_max):
            return r_max**2 / (2 * np.pi)
        return 10.0

    def objective(x):
        return evaluate_fast(x)

    def constraints(x):
        return np.sum(x) - 1.0

    def bounds(x):
        return np.sum(x**2) - 1.0

    def non_linear_constraint(x):
        return np.sum(x**2) - 1.0

    def optimize_with_surrogate():
        nonlocal best_coefficients, best_bound
        x0 = np.random.randn(8)
        x0 = x0 / np.linalg.norm(x0)
        result = minimize(objective, x0, bounds=Bounds(-10, 10), constraints=LinearConstraint([[1.0]], [1.0], [1.0]), method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
        if np.isfinite(result.fun):
            candidate = result.x
            eval_result = evaluate_fast(candidate)
            if eval_result < best_bound:
                best_bound = eval_result
                best_coefficients = candidate

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        optimize_with_surrogate()
    return best_coefficients
