import numpy as np
from scipy.optimize import minimize, fsolve
from scipy.optimize.optimize import OptimizeResult
from sympy import symbols, hermite, diff, solve, simplify
from itertools import product
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_value = float('inf')
    max_iter = 100
    iteration = 0
    start_time = time.time()

    def evaluate(coeffs):
        nonlocal best_value
        if len(coeffs) > 8:
            return best_value
        try:
            z = symbols('z')
            Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
            Q = Q - Q.as_leading_term(z**2)
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            else:
                return best_value
            value = r_max**2 / (2 * np.pi)
            return value
        except:
            return best_value

    def obj_fn(coeffs):
        return evaluate(coeffs)

    while time.time() - start_time < budget_s and iteration < max_iter:
        coeffs = np.random.uniform(-1, 1, size=8)
        result = minimize(obj_fn, x0=coeffs, method='L-BFGS-B', bounds=[(-1, 1)]*8)
        if result.fun < best_value:
            best_value = result.fun
            best_coeff = result.x
        iteration += 1

    return best_coeff
