from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

                return res / (z**2)
            
            z_vals = np.linspace(-10, 10, 1000)

            Q = sum(x[k] * hermite(4*k, z) for k in range(len(x)))
