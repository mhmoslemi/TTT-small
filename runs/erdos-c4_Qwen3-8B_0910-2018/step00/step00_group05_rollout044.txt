import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, hermite, solve, Poly, Rational, simplify, diff
from itertools import product

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_bound = float('inf')
    start_time = time.time()

    def evaluate_candidate(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q - Q.as_coefficients_dict()[z**2]*z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        unique_roots = sorted(set(real_roots))
        r_max = None
        for r in unique_roots:
            if r > 0:
                r_max = r
                break
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)

    def objective(x):
        nonlocal best_bound
        if time.time() - start_time > budget_s:
            return best_bound
        try:
            val = evaluate_candidate(x)
            if val < best_bound:
                best_bound = val
                best_coeff = x.copy()
            return val
        except:
            return float('inf')

    def grid_search():
        nonlocal best_bound
        for m in range(1, 9):
            for coeffs in product(np.linspace(-10, 10, 50), repeat=m):
                if np.sum(coeffs) == 0:
                    continue
                try:
                    val = evaluate_candidate(coeffs)
                    if val < best_bound:
                        best_bound = val
                        best_coeff = np.array(coeffs)
                except:
                    pass

    def optimize():
        nonlocal best_bound
        for _ in range(100):
            if time.time() - start_time > budget_s:
                return
            try:
                res = minimize(objective, np.random.uniform(-10, 10, 8), method='L-BFGS-B')
                if res.fun < best_bound:
                    best_bound = res.fun
                    best_coeff = res.x.copy()
            except:
                pass

    grid_search()
    optimize()

    return best_coeff
