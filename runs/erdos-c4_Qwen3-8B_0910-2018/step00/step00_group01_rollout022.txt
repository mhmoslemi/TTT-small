import numpy as np
from scipy.optimize import minimize, fsolve
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeffs = np.random.rand(8)
    best_score = float('inf')
    
    def objective(coeffs):
        nonlocal best_score
        t0 = time.time()
        coeffs = np.clip(coeffs, 0, 1)
        coeffs = coeffs / np.sum(coeffs)
        
        def H4k(z):
            return np.polynomial.hermite.hermval(z, [0]*4 + [1])
        
        Q = np.zeros_like(coeffs)
        for k in range(8):
            Q += coeffs[k] * H4k(k)
        
        # Enforce Q(0) = 0
        Q0 = Q[0]
        Q[0] -= Q0
        # Make positive at infinity
        if Q[-1] < 0:
            Q *= -1
        
        # Remove z² factor
        def Q_z2(z):
            return Q[z**2]
        
        # Find r_max via root finding
        try:
            r_max = fsolve(lambda r: Q_z2(r), 10)
            r_max = max(r_max)
        except:
            r_max = 1000
        
        r_max = r_max.real
        score = r_max**2 / (2 * np.pi)
        if score < best_score:
            best_score = score
            best_coeffs = coeffs.copy()
        return score
    
    for _ in range(100):
        result = minimize(objective, best_coeffs, method='L-BFGS-B', options={'maxiter': 10})
        if time.time() - t0 > budget_s:
            break
    
    return best_coeffs
