from scipy.optimize import minimize
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):

    initial_coefficients = np.array([1.0, -1.0, 1.0])
    n = len(initial_coefficients)
    if n > 8:
        n = 8
    if n < 1:
        n = 1

    # Define Hermite polynomials up to order 4*7=28
    def hermite(n):
        from scipy.special import hermite
        return hermite(n, np.polynomial.hermite.hermite)

    def evaluate_coefficients(coeffs):
        from sympy import symbols, expand, solve, Eq
        z = symbols('z')
        Q = sum(c * hermite(4 * i)(z) for i, c in enumerate(coeffs))
        Q = Q.as_poly()
        # Enforce Q(0) = 0
        if Q.subs(z, 0) != 0:
            raise ValueError("Q(0) must be zero")
        # Normalize to be positive at infinity
        leading_term = Q.as_expr().as_poly().coeffs[-1]
        if leading_term < 0:
            coeffs = [-c for c in coeffs]
        # Remove z^2 factor
        Q = Q.as_expr().as_poly().div(z**2)[0]
        # Find real roots of Q(z)
        roots = solve(Q, z)
        # Filter real roots, exclude even multiplicity
        real_roots = [r.evalf() for r in roots if r.is_real and not r.is_muli]
        if not real_roots:
            return np.inf
        r_max = max(real_roots)
        if r_max < 0:
            r_max = -r_max
        return r_max**2 / (2 * np.pi)

    def objective(coeffs):
        return evaluate_coefficients(coeffs)

    def fast_surrogate(coeffs):
        from scipy.optimize import minimize
        # Fast numerical estimation
        def Q_numerical(z, coeffs):
            from scipy.special import hermi
            return sum(c * hermi(4 * i, z) for i, c in enumerate(coeffs))
        # Numerically find r_max
        def f(z):
            return Q_numerical(z, coeffs)
        # Use Newton method to find sign change
        z = np.linspace(0, 100, 1000)
        sign_changes = np.sign(f(z))[1:] * np.sign(f(z))[:-1]
        idx = np.where(sign_changes < 0)[0]
        if not idx.size:
            return np.inf
        r_max = z[idx[0]] + (z[idx[0]] - z[idx[0]-1]) * np.sign(f(z[idx[0]]))
        return r_max**2 / (2 * np.pi)

    # Main optimization loop
    def optimize_coefficients():
        best_coeffs = initial_coefficients[:n].copy()
        best_value = np.inf

            return fast_surrogate(coeffs)

        res = minimize(
            surrogate_coeffs,
            initial_coefficients[:n],
