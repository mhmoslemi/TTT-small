from scipy.optimize import minimize
from scipy.optimize import Bounds
from sympy import symbols, Eq, solve, simplify, zeros, diff, HermiteH
from sympy.polynomials import Polynomial
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def eval_surrogate(c):
        m = len(c)
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_expr()
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q
        Q = Q / z**2
        Q = Q.as_expr()
        
        def f(z_val):
            try:
                val = Q.subs(z, z_val)
                return val.evalf()
            except:
                return 0
        
        roots = []
        for z in np.linspace(-100, 100, 10000):
            prev = f(z)
            curr = f(z + 1e-6)
            if prev * curr < 0:
                roots.append(z)
        
        if not roots:
            return float('inf')
        
        r_max = max(roots)
        return r_max**2 / (2 * np.pi)
    
    def exact_check(c):
        m = len(c)
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_expr()
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        
        def find_r_max(Q):
            from sympy import Nroots
            roots = Nroots(Q)
            real_roots = [r for r in roots if r.is_real]
            real_roots = np.array([float(r) for r in real_roots])
            real_roots = np.sort(real_roots)

            return float('inf')

        return r_max**2 / (2 * np.pi)

        print("Best candidate:", best_coeff)
        return best_coeff
