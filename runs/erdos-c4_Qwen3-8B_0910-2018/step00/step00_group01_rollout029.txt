from scipy.optimize import minimize, root_scalar
from scipy.special import hermite
import time

    def eval_Q(z, coeffs):

        Q = np.zeros_like(z)

            H = hermite(4*k, z)
            Q += coeffs[k] * H
        return Q

            Q *= -1

                continue

                continue
        return r_max ** 2 / (2 * np.pi)

        return compute_U(coeffs)

        return [(-10, 10) for _ in coeffs]
