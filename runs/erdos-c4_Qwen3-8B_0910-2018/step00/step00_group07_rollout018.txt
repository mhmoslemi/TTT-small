import numpy as np
import sympy as sp
from scipy.optimize import minimize
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, -1.0, 1.0])
    best_coeff = initial_coefficients
    best_bound = float('inf')
    time_used = 0

    def evaluate(coeffs):
        nonlocal time_used
        start = time.time()
        try:
            # Numerical surrogate
            def Q(z):
                h = np.polyspace([0, 4 * len(coeffs) - 4], 4)
                poly = sum(coeffs[k] * h[4 * k] for k in range(len(coeffs)))
                return poly

            def numeric_evaluate(coeffs):
                symbol = sp.symbols('z')
                h = [sp.hermeval(symbol, sp.hermitelist(range(4 * len(coeffs))))[4 * k] for k in range(len(coeffs))]
                poly = sum(coeffs[k] * h[k] for k in range(len(coeffs)))
                poly = poly - poly.subs(symbol, 0)
                poly = poly.as_poly()
                return poly

            # Use numerical root-finding
            def get_r_max(coeffs):
                symbol = sp.symbols('z')
                h = [sp.hermeval(symbol, sp.hermitelist(range(4 * len(coeffs))))[4 * k] for k in range(len(coeffs))]
                poly = sum(coeffs[k] * h[k] for k in range(len(coeffs)))
                poly = poly - poly.subs(symbol, 0)

                # Enforce positivity at infinity
                leading_term = poly.as_expr().as_ordered_terms()[0]
                if leading_term.coeff(symbol, 2 * len(coeffs) - 2) < 0:
                    coeffs = -np.copy(coeffs)
                    poly = -poly

                # Remove z^2 factor
                poly = poly.as_poly()
                if poly.LC() < 0:
                    poly = -poly
                poly = poly.as_expr()

                # Find real roots
                root_list = sp.solvers.solve(poly, symbol)
                real_roots = [r.evalf() for r in root_list if r.is_real]
                real_roots = sorted([r for r in real_roots if r > 0])

                # Find sign change
                r_max = 0
                for i in range(len(real_roots) - 1):
                    if np.sign(poly.subs(symbol, real_roots[i])) != np.sign(poly.subs(symbol, real_roots[i + 1])):
                        r_max = real_roots[i + 1]
                        break
                return r_max

            r_max = get_r_max(coeffs)
            if r_max == 0:
                return 0.0

            return r_max ** 2 / (2 * np.pi)

        except Exception as e:
            print(f"Error evaluating: {e}")
            return float('inf')

        finally:
            time_used += time.time() - start

    def fitness_function(x):
        nonlocal time_used
        if time_used >= budget_s:
            return float('inf')
        return evaluate(x)

    def random_search(iterations=100):
        nonlocal time_used
        for _ in range(iterations):
            if time_used >= budget_s:
                break
            coeffs = np.random.rand(len(initial_coefficients)) * 2 - 1
            score = fitness_function(coeffs)
            if score < best_bound:
                best_bound = score
                best_coeff = coeffs
            time_used += 0.1

    def gradient_descent():
        nonlocal time_used
        res = minimize(fitness_function, initial_coefficients, method='L-BFGS-B', bounds=[(-2, 2) for _ in range(len(initial_coefficients))])
        if res.success and res.fun < best_bound:
            best_bound = res.fun
            best_coeff = res.x
        time_used += 50

    def local_search():
        nonlocal time_used
        for i in range(len(best_coeff)):
            if time_used >= budget_s:
                break
            for step in [0.1, 0.01, 0.001]:
                new_coeff = np.copy(best_coeff)
                new_coeff[i] += step
                score = fitness_function(new_coeff)
                if score < best_bound:
                    best_bound = score
                    best_coeff = new_coeff
                new_coeff[i] -= 2 * step
                score = fitness_function(new_coeff)
                if score < best_bound:
                    best_bound = score
                    best_coeff = new_coeff
                time_used += 0.5

    try:
        random_search(100)
        gradient_descent()
        local_search()
    except:
        pass

    return best_coeff
