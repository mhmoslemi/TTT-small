import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
from bisect import bisect_left
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial guess with 3 coefficients
    initial_coefficients = np.array([1.0, 0.5, -0.3])

    # Define the Hermite polynomials of order 4k
    def hermite(n):
        if n == 0:
            return lambda z: 1.0
        elif n == 1:
            return lambda z: 2*z
        elif n == 2:
            return lambda z: 4*z**2 - 2
        elif n == 3:
            return lambda z: 8*z**3 - 12*z
        elif n == 4:
            return lambda z: 16*z**4 - 48*z**2 + 12
        else:
            raise ValueError("Hermite polynomial not implemented for this order")

    # Build Q(z) = sum c_k H_{4k}(z)
    def build_Q(coeffs):
        terms = [coeffs[i] * hermite(4 * i)(z) for i in range(len(coeffs))]
        return sum(terms)

    # Evaluate the uncertainty bound via the described method
    def evaluate_uncertainty(coeffs):
        # Build Q(z)
        Q = build_Q(coeffs)
        # Roots finding and sign change
        # Numerical approximation for r_max
        steps = 1000
        z_values = np.linspace(-10, 10, steps)
        Q_values = [Q(z) for z in z_values]
        sign_changes = np.sign(np.gradient(Q_values))
        positive_sign_changes = np.where(sign_changes > 0)[0]
        if not positive_sign_changes.size:
            return float('inf')
        r_max = z_values[positive_sign_changes[0]]
        # Return U = r_max^2 / (2 * np.pi)
        return r_max**2 / (2 * np.pi)

    # Define cost function (to minimize uncertainty)
    def cost(coeffs):
        try:
            return evaluate_uncertainty(coeffs)
        except:
            return float('inf')

    # Define a function to get the best coefficients in the allowed range
    def best_coefficients():
        best = np.inf
        best_coeffs = initial_coefficients.copy()
        # Try with up to 8 coefficients
        for n in range(1, 9):
            # Start with initial guess and refine
            if n == 3:
                coeffs = initial_coefficients.copy()
            else:
                coeffs = np.random.uniform(-1, 1, n)
            # Optimization
            result = minimize(
                cost,
                x0=coeffs,
                bounds=Bounds([-10, 10], [-10, 10]),
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if result.success and result.fun < best:
                best = result.fun
                best_coeffs = result.x
        return best_coeffs

    start_time = time.time()
    result = best_coefficients()
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_coefficients  # Use initial if timeout
    return result
