import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, sqrt, N, simplify, Poly, roots

def run(seed=42, budget_s=600, **kwargs):

    def fast_bound(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.as_expr()
        Q = Q.subs(z, 0).cancel()
        Q = Q.as_poly().as_expr()
        Q = Q / z**2
        roots = solve(Q, z)
        sign_changes = 0
        for r in roots:
            if r.is_real and r > 0:
                sign_changes += 1
        if sign_changes == 0:
            return float('inf')
        r_max = max([r.evalf() for r in roots if r.is_real and r > 0])
        return (r_max**2) / (2 * np.pi)
    
    def exact_bound(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))

        Q = Q.subs(z, 0).cancel()
        Q = Q.as_poly().as_expr()
        Q = Q / z**2
        roots = solve(Q, z)

        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    def objective(c):
        return exact_bound(c)

        return fast_bound(c)
