import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, hermite, roots, simplify
from sympy.polys import Poly

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.random.randn(8)
    best_coeff = initial_coefficients.copy()
    best_bound = float('inf')
    time_used = 0
    
    def evaluate_coefficients(coeffs):
        nonlocal time_used
        start_time = time.time()
        
        z = symbols('z')
        H = [hermite(n, z) for n in range(len(coeffs))]
        Q = sum(coeffs[k] * H[k] for k in range(len(coeffs)))
        
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        
        # Check sign at infinity
        inf_sign = Q.as_infinity().as_real_imag()[0]
        if inf_sign < 0:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q.as_poly().removeTerms()
        if Q.degree() >= 2:
            Q = Q.as_poly().div(z**2)[1]
        else:
            Q = Poly(0, z)
        
        # Find roots
        roots_list = roots(Q, z, wild=True).as_real_imag()
        real_roots = [r for r in roots_list if r[1] == 0]
        real_roots = np.sort([r[0] for r in real_roots])
        
        # Find r_max
        r_max = 0
        for r in real_roots:
            if r > r_max:
                r_max = r
        
        if r_max == 0:
            return float('inf')
        
        bound = r_max ** 2 / (2 * np.pi)
        time_used += time.time() - start_time
        return bound
    
    def objective(x):
        return evaluate_coefficients(x)
    
    def surrogated_objective(x):
        return np.abs(np.sum(x * np.random.randn(8))) + 100 * np.sum(x ** 2)

        for _ in range(1000):
            coeffs = np.random.randn(8)
            bound = evaluate_coefficients(coeffs)

                best_coeff = coeffs.copy()

            best_bound = bound
            best_coeff = coeffs.copy()
