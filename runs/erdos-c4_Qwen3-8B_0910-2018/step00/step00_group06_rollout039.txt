import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Eq, solve, simplify
from itertools import product
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, -1.0, 1.0])
    
    def objective(coeffs):
        coeffs = np.array(coeffs)
        # Apply Hermite polynomials
        z = symbols('z')
        H = [hermite(i, z) for i in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity
        if Q.as_leading_coefficient() < 0:
            Q *= -1
        # Remove z^2 factor
        Q = Q.as_poly().as_dict()
        z2_factor = 1
        for degree, coeff in Q.items():
            if degree == 2:
                z2_factor = coeff
        Q = sum(coeff * z**(degree - 2) for degree, coeff in Q.items())
        # Find roots
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted(real_roots)
        # Find r_max
        r_max = None
        for i in range(len(real_roots) - 1):
            if real_roots[i+1] - real_roots[i] > 1e-6:
                r_max = real_roots[i+1]
                break
        if r_max is None:
            r_max = real_roots[-1] if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def loss(coeffs):
        return objective(coeffs)
    
    def initial_search():
        best = initial_coefficients
        best_loss = loss(best)
        for n in range(2, 8+1):
            for signs in product([1, -1], repeat=n):
                trial = np.array([s * 1.0 for s in signs])
                loss_val = loss(trial)
                if loss_val < best_loss:
                    best = trial
                    best_loss = loss_val
        return best
    
    def optimize(coeffs):
        result = minimize(loss, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in coeffs])
        return result.x
    
    best_coeffs = initial_search()
    for _ in range(3):
        if time.time() > budget_s - 10:
            break
        best_coeffs = optimize(best_coeffs)
    
    return best_coeffs.tolist()
