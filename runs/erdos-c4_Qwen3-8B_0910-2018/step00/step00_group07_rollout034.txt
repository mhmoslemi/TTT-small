import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.optimize import minimize_scalar
from scipy.special import hermitian

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coefficients = np.array([1.0, -1.0, 1.0])
    bounds = [(-100, 100) for _ in range(8)]
    
    def objective(c):
        try:
            Q = np.zeros(1000)
            for k in range(len(c)):
                H = hermitian(4*k)
                Q += c[k] * H(Q)
            Q[0] = 0.0
            Q /= Q[-1]
            def f(z):
                return Q[z]
            try:
                r_max = root_scalar(f, bracket=[0, 100], method='brentq').root
                return r_max**2 / (2 * np.pi)
            except:
                return np.inf
        except:
            return np.inf
    
    def adjust_coefficients(c):
        return minimize(
            fun=objective,
            x0=c,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'gtol': 1e-8}
        ).x
    
    best_coeff = initial_coefficients
    best_U = np.inf
    
    for _ in range(20):
        new_coeff = adjust_coefficients(best_coeff)
        new_U = objective(new_coeff)
        if new_U < best_U:
            best_U = new_U
            best_coeff = new_coeff
    
    return best_coeff
