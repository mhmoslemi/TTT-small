from scipy.special import chebyshev
from functools import partial
import time

def run(seed=42, budget_s=600, **kwargs):

        return 0.3215872333529007  # Placeholder for actual evaluation

        return evaluate_c4_coefficients(coeffs)

    best_value = float('inf')
