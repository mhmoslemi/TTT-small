import numpy as np
from scipy.optimize import Bounds, minimize
from scipy.special import hermitedomain
import time
from sympy import symbols, hermite, simplify, solve, roots

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    best_count = 1
    total_time = 0
    
    for count in range(1, 9):
        print(f"Trying with {count} coefficients...")
        if count > 8:
            continue
        
        def objective(x):
            nonlocal total_time
            start_time = time.time()
            try:
                # Construct Hermite polynomials
                z = symbols('z')
                H = [hermite(k, z) for k in range(count)]
                
                # Compute Q(z)
                Q = sum(x[k] * H[k] for k in range(count))
                
                # Enforce Q(0) = 0 and positive at infinity
                Q = Q - Q.subs(z, 0)
                
                # Find r_max
                expr = Q / z**2
                roots_expr = roots(expr)
                r_max = None
                for r in roots_expr:
                    if r.is_real and r > 0:
                        r_max = r
                if r_max is None:
                    return float('inf')
                
                # Compute upper bound
                upper_bound = r_max**2 / (2 * np.pi)
                return upper_bound
            except:
                return float('inf')
            finally:
                elapsed = time.time() - start_time
                total_time += elapsed
                if total_time > budget_s:
                    print(f"Time budget exceeded. Stopping.")
                    return best_coeff
    
        # Initialize with low-rank configurations
        x0 = np.array([0.5] * count)
        result = minimize(objective, x0, bounds=Bounds(lb=-1.0, ub=1.0), method='L-BFGS-B', options={'maxiter': 100})
        current_bound = result.fun
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = result.x
            best_count = count
        print(f"Best bound: {best_bound} with {best_count} coefficients")
    
    return best_coeff
