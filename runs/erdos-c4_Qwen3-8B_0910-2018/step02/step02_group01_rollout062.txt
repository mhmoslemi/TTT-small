def evaluate_c4_coefficients(coefficients):
    z = symbols('z')
    Q = sum(c * HermiteH(4*k, z) for k, c in enumerate(coefficients))
    Q = Q - Q.as_poly().subs(z, 0)
    Q = Q / z**2

    real_roots = [r for r in real_roots if not np.isclose(r, 0)]
    real_roots.sort()
    if not real_roots:
        return float('inf')
    r_max = real_roots[-1]
    return r_max**2 / (2 * np.pi)
