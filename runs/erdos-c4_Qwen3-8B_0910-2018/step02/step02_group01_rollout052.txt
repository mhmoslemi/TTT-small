import numpy as np
from scipy.optimize import differential_evolution, minimize
from sympy import symbols, Hermite, simplify, solve, RealDomain
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    best_count = 1

    for count in range(1, 9):
        print(f"Trying with {count} coefficients...")
        
        def objective(x):
            c = x
            coeffs = np.array([c[i] for i in range(count)])
            Q = 0
            for k in range(count):
                Q += coeffs[k] * Hermite(4*k, symbols('z'))
            Q = simplify(Q)
            Q = Q.as_poly().as_expr()
            Q = Q.subs(symbols('z'), 0)
            Q = Q.as_numer_denom()[0]
            if Q == 0:
                return float('inf')
            z = symbols('z')
            poly = simplify(Q)
            roots = solve(poly, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            real_roots = sorted(real_roots)
            r_max = None
            for i in range(1, len(real_roots)):
                if real_roots[i] > 0:
                    r_max = real_roots[i]
                    break
            if r_max is None:
                return float('inf')
            bound = r_max**2 / (2 * np.pi)
            return bound
        
        result = differential_evolution(objective, bounds=[(-1, 1) for _ in range(count)], strategy='best1bin', popsize=15, maxiter=100, mutation=(0.5, 1), recombination=0.9, tol=1e-10, seed=seed)
        current_bound = result.fun
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = result.x
            best_count = count
        print(f"Best bound: {best_bound} with {best_count} coefficients")
    
    return best_coeff
