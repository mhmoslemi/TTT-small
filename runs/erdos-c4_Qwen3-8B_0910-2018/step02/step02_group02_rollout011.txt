from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
import time
from numba import jit

@jit(nopython=True)
def hermite_eval(c, z):
    H = np.zeros_like(z)
    H[0] = 1.0
    for k in range(1, len(c)):
        H[k] = 2 * (2*k-1) * H[k-1] - 2*(k-1)*H[k-2]
    return np.sum(c * H)

def evaluate_c4_coefficients(c):
    z = np.linspace(-10, 10, 10000)
    Q = hermite_eval(c, z)
    Q /= z**2
    roots = np.roots(np.polynomial.Polynomial.fromroots(z[np.where(np.abs(np.diff(np.sign(Q)))>0)]))
    r_max = np.max(roots[roots>0])
    return r_max**2 / (2*np.pi)

def run(seed=42, budget_s=600, **kwargs):

    def objective(c):
        return evaluate_c4_coefficients(c)
    
    def gradient(c):

        grad = np.zeros_like(c)
        for i in range(len(c)):
            perturbed = c.copy()
            perturbed[i] += eps
            grad[i] = (objective(perturbed) - objective(c)) / eps
        return grad

        candidates = np.random.choice(range(len(pop)), size=k, replace=False)
        best_idx = candidates[np.argmin(fitnesses[candidates])]
        return pop[best_idx]
    
    def crossover(a, b):
        point = np.random.randint(1, len(a))
        return np.concatenate([a[:point], b[point:]])

        for i in range(len(ind)):

                ind[i] += np.random.uniform(-0.5, 0.5)
