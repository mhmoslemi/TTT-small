    best_bound = float('inf')
    
    def evaluate_coefficients(c):

        def hermite(n, z):
            return mp.hermite(n, z)
        
        def Q(z):
            val = 0
            for k in range(m):
                val += c[k] * hermite(4*k, z)
            return val

        # Remove z^2 factor
        def Q_normalized(z):
            return Q(z) / z**2
        
        # Find r_max
        try:
            roots = nroots(Q_normalized)
            real_roots = [r for r in roots if np.isreal(r)]
            real_roots = [r.real for r in real_roots]
            real_roots = sorted(real_roots)
            
            # Find where sign changes
            sign_changes = []
            for i in range(len(real_roots) - 1):
                if np.sign(Q_normalized(real_roots[i])) != np.sign(Q_normalized(real_roots[i+1])):
                    sign_changes.append((real_roots[i], real_roots[i+1]))
            
            if sign_changes:
                r_max = max(sign_changes, key=lambda x: x[1])[1]
                return r_max**2 / (2 * np.pi)

                return float('inf')
