from scipy.optimize import Bounds, minimize
from sympy import symbols, simplify, hermite, solve, Eq, sqrt, pi
from itertools import product

def run(seed=42, budget_s=600, **kwargs):

        unique_roots = []
        for r in real_roots:

    def objective(x):
