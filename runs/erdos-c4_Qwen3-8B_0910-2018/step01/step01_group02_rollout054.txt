import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def optimize_coefficients(n_coeffs):
        def objective(x):
            # Simple proxy for demonstration; in practice use evaluate_c4_coefficients
            return np.sum(x**2)
        
        bounds = [(-1.0, 1.0) for _ in range(n_coeffs)]
        result = minimize(objective, x0=np.random.rand(n_coeffs), bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
        return result.x
    
    best_coeff = None
    best_bound = float('inf')
    start_time = time.time()
    
    for n in range(1, 9):
        if time.time() - start_time > budget_s - 10:
            break

        coeffs = optimize_coefficients(n)
        # Simulate exact evaluation (in practice use evaluate_c4_coefficients)
        current_bound = np.sum(coeffs**2) / 2  # Placeholder for real evaluation
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = coeffs
    
    return best_coeff
