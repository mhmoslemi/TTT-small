import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time
import sympy as sp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = np.array([1.0])
    best_bound = float('inf')
    best_count = 1
    
    for count in range(1, 9):
        print(f"Trying with {count} coefficients...")
        start_time = time.time()

        if count == 1:
            init_coeff = np.array([1.0])
        else:
            init_coeff = np.random.rand(count)
        
        def objective(x):
            # Construct polynomial Q(z)
            z = sp.symbols('z')
            H = [sp.hermeval(z, sp.hermitelist(range(4*k + 1))) for k in range(count)]
            Q = sum(x[k] * H[k] for k in range(count))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Flip sign to ensure positive at infinity
            if Q.as_infinite_limit(z) > 0:
                Q = -Q
            
            # Remove z^2 factor and find largest positive real root
            Q = Q / z**2
            roots = sp.solvers.solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            real_roots = [r for r in real_roots if sp.solvers.root_multiset(Q, r).multiplicity % 2 == 1]
            
            if not real_roots:
                return float('inf')
            
            r_max = max(real_roots)
            bound = r_max**2 / (2 * np.pi)
            return bound
        
        # Optimize with multi-start
        results = []
        for _ in range(5):
            x0 = np.random.rand(count) * 2 - 1
            res = minimize(objective, x0, bounds=[(-1, 1) for _ in range(count)], method='L-BFGS-B', options={'maxiter': 100})
            results.append(res.x)
        
        for x in results:
            if np.isfinite(objective(x)):
                current_bound = objective(x)
                if current_bound < best_bound:
                    best_bound = current_bound
                    best_coeff = x
                    best_count = count
        
        elapsed = time.time() - start_time
        print(f"Current best bound: {best_bound}, Elapsed: {elapsed:.2f}s")
    
    return best_coeff
