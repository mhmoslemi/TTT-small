import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):

    def hermite_eval(z, coeffs):
        H = np.polynomial.hermite.hermder([0]*len(coeffs))  # Precompute derivatives

        for k in range(len(coeffs)):
            Q += coeffs[k] * np.polynomial.hermite.hermval(z, H[k])
        return Q
    
    def find_r_max(coeffs):
        def Q(z):
            return hermite_eval(z, coeffs)
        
        def Q_normalized(z):
            return Q(z) / (z**2 + 1e-10)
        
        try:

            for r in real_roots:
                if r > 0:
                    if np.sign(Q_normalized(r - 1e-5)) != np.sign(Q_normalized(r + 1e-5)):
                        return r**2
            return np.inf
        except:
            return np.inf
    
    def evaluate_c4_coeffs(coeffs):
        return find_r_max(coeffs)

        try:
            return evaluate_c4_coeffs(coeffs)
