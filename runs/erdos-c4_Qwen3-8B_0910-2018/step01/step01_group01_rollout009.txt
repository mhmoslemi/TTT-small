import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(n):
        if n == 0:
            return lambda z: 1.0
        elif n == 1:
            return lambda z: 2*z
        elif n == 2:
            return lambda z: 4*z**2 - 2
        elif n == 3:
            return lambda z: 8*z**3 - 12*z
        elif n == 4:
            return lambda z: 16*z**4 - 48*z**2 + 12
        elif n == 5:
            return lambda z: 32*z**5 - 160*z**3 + 120*z
        elif n == 6:
            return lambda z: 64*z**6 - 480*z**4 + 960*z**2 - 120
        elif n == 7:
            return lambda z: 128*z**7 - 1344*z**5 + 4480*z**3 - 4480*z
        elif n == 8:
            return lambda z: 256*z**8 - 3584*z**6 + 17920*z**4 - 35840*z**2 + 5040
        else:
            raise ValueError("Hermite polynomial not implemented for n = {}".format(n))
    
    def evaluate_c4_coeffs(coeffs):
        Q = lambda z: sum(c * hermite_poly(k)(z) for k, c in enumerate(coeffs))
        # Ensure Q(0) = 0
        Q = lambda z: Q(z) - Q(0)
        # Make Q positive at infinity
        if Q(100) > 0:
            pass
        elif Q(-100) > 0:
            Q = lambda z: -Q(z)
        else:
            Q = lambda z: -Q(z)
        # Remove z^2 factor
        Q = lambda z: Q(z) / z**2 if z != 0 else 0
        # Find r_max
        def f(z):
            return Q(z)
        # Numerical root-finding
        roots = []
        for z in np.linspace(-100, 100, 1000):
            if z == 0:
                continue
            if np.sign(f(z)) != np.sign(f(z + 1e-5)):
                # Potential root
                try:
                    r = opt.root_scalar(lambda x: f(x), bracket=[z - 1e-5, z + 1e-5], method='brentq').root
                    if np.isclose(f(r), 0):
                        roots.append(r)
                except:
                    pass
        if not roots:
            return np.inf
        roots = sorted(roots)
        r_max = roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        try:
            return evaluate_c4_coeffs(coeffs)
        except:
            return np.inf
    
    def bounds(n):
        return Bounds([-10.0 for _ in range(n)], [10.0 for _ in range(n)])
    
    best_coeffs = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    best_score = float('inf')

    for n in range(4, 9):
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with {n} coefficients")
        initial_guess = np.random.uniform(-1.0, 1.0, n)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds(n))
        if result.success:
            score = result.fun
            if score < best_score:
                best_score = score
                best_coeffs = result.x
                print(f"New best score: {best_score}")
    
    return best_coeffs
