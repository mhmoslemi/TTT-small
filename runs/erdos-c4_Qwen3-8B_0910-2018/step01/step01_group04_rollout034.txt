import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import minimize_scalar
from scipy.optimize import dual_annealing

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(n, z):
        if n == 0:
            return 1.0
        elif n == 1:
            return 2*z
        elif n == 2:
            return 4*z**2 - 2
        elif n == 3:
            return 8*z**3 - 12*z
        elif n == 4:
            return 16*z**4 - 48*z**2 + 12
        return 0.0
    
    def eval_Q(z, coeffs):
        Q = 0.0
        for k in range(len(coeffs)):
            Q += coeffs[k] * hermite_poly(4*k, z)
        return Q
    
    def find_r_max(coeffs):
        from scipy.optimize import root_scalar
        
        def f(z):
            return eval_Q(z, coeffs) / (z**2) if z != 0 else float('inf')
        
        try:
            root = root_scalar(f, bracket=[0.1, 100], method='brentq')
            return root.root
        except:
            return 100.0
    
    def evaluate_c4_coefficients(coeffs):
        r_max = find_r_max(coeffs)
        return r_max**2 / (2 * np.pi)
    
    def surrogate_objective(coeffs):
        return -evaluate_c4_coefficients(coeffs)
    
    def init_chebyshev(n):
        return np.array([np.cos(k * np.pi / (n + 1)) for k in range(n + 1)])
    
    best_coeffs = np.array([1.0])
    best_value = float('inf')
    best_m = 1
    
    for m in range(1, 9):
        bounds = [(-10.0, 10.0) for _ in range(m)]
        if m == 1:
            guess = np.array([1.0])
        else:
            guess = init_chebyshev(m - 1)
        
        start_time = time.time()
        result = minimize(
            surrogate_objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
        )
        end_time = time.time()
        
        if end_time - start_time > budget_s:
            continue
        
        refined_coeffs = result.x
        exact_value = evaluate_c4_coefficients(refined_coeffs)
        
        if exact_value < best_value:
            best_value = exact_value
            best_coeffs = refined_coeffs
            best_m = m
    
    return best_coeffs
