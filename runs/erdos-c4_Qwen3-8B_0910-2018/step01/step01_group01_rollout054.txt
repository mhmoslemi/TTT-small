import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, linsolve, solve, simplify

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.zeros(8)
    
    def objective(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_ordered_terms()[0]  # Remove constant term
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [float(r) for r in real_roots if r > 0]
        if not real_roots:
            return np.inf
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def initial_guess():
        return np.array([1.0, 0.5, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02])
    
    result = minimize(objective, initial_guess(), method='L-BFGS-B', bounds=[(-10, 10)]*8)
    return result.x
