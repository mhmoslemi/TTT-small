import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def exact_bound(coeffs):
        # Replace with actual evaluate_c4_coefficients in production
        return np.sum(coeffs ** 2) / 2  # Placeholder for real evaluation
    
    def objective(x, penalty=1e-3):
        # Numerical surrogate with penalty for sign changes
        return np.sum(x ** 2) + penalty * np.sum(np.abs(np.diff(x)))
    
    best_coeff = np.random.rand(1)
    best_bound = float('inf')
    total_time = 0
    
    for n in range(1, 9):
        if total_time >= budget_s:
            break
        print(f"Trying with {n} coefficients...")
        bounds = Bounds(lb=-1.0, ub=1.0)
        x0 = np.random.rand(n)
        start_time = time.time()
        
        # Use L-BFGS-B with adaptive penalty
        result = minimize(
            lambda x: objective(x, penalty=1e-3),
            x0,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        elapsed = time.time() - start_time
        total_time += elapsed
        
        # Simulate exact evaluation
        current_bound = exact_bound(result.x)
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = result.x
    
    return best_coeff
