from scipy.optimize import minimize, Bounds
import time
from fractions import Fraction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        from sympy import symbols, Eq, solve, hermite, roots, S
        z = symbols('z')
        Q = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.subs(z, 0).as_numerical_value()
        Q = Q.as_numerical_value()
        
        # Ensure Q(0) = 0
        if Q != 0:
            coeffs = [c * Q.as_numerical_value() for c in coeffs]
        
        # Make Q positive at infinity
        leading_term = sum(c * hermite(4*k, z).as_leading_coefficient() for k, c in enumerate(coeffs))
        if leading_term < 0:
            coeffs = [-c for c in coeffs]
        
        Q = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q / z**2
        
        # Find real roots
        expr = Q.as_numerical_value()
        root_list = solve(expr, z, domain=S.Reals)
        
        r_max = 0
        for r in root_list:
            if r.is_real and r > 0:
                if r > r_max:
                    r_max = r
        
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_coefficients(x)
    
    best_coeff = None
    best_value = float('inf')
    
    # Try with increasing number of coefficients up to 8
    for count in range(1, 9):
        print(f"Trying with {count} coefficients...")
