import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return np.sum(x**2)
    
    best_coeff = np.array([1.0])
    best_bound = float('inf')
    
    def evaluate_with_count(n):
        bounds = Bounds(lb=-1.0, ub=1.0)
        x0 = np.random.rand(n) * 2 - 1
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
        return result.x
    
    for count in range(1, 9):
        print(f"Trying with {count} coefficients...")
        start_time = time.time()
        coeffs = evaluate_with_count(count)
        elapsed = time.time() - start_time
        
        # Simulate bound estimation
        current_bound = np.sum(coeffs**2) / 2
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = coeffs
        print(f"Current best bound: {best_bound}, Elapsed: {elapsed:.2f}s")
    
    return best_coeff
