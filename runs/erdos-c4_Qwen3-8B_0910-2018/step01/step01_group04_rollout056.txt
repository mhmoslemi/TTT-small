import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import minimize_scalar
from sympy import symbols, HermitePolynomial, simplify, solve, exp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coeffs(coeffs):
        m = len(coeffs)
        c = coeffs
        # Construct Q(z) = sum_{k=0}^m c_k H_{4k}(z)
        z = symbols('z')
        Q = 0
        for k in range(m):
            Q += c[k] * HermitePolynomial(4*k, z)
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity by flipping sign if necessary
        if Q.subs(z, 1000) < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find largest positive real root where Q(z) changes sign
        # Convert to numerical form
        Q_num = simplify(Q)
        # Use numerical root finding
        def f(x):
            return Q_num.subs(z, x)
        # Use binary search to find r_max
        low, high = 0, 100
        for _ in range(100):
            mid = (low + high) / 2
            if f(mid) > 0:
                low = mid
            else:
                high = mid
        r_max = high
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_coeffs(coeffs)
    
    best_coeffs = np.array([1.0])
    best_value = float('inf')
    
    for m in range(1, 9):
        # Initial grid search
        grid = np.logspace(-2, 2, 100)
        for c0 in grid:
            for c1 in grid:
                for c2 in grid:
                    for c3 in grid:
                        for c4 in grid:
                            for c5 in grid:
                                for c6 in grid:
                                    for c7 in grid:
                                        coeffs = np.array([c0, c1, c2, c3, c4, c5, c6, c7])[:m]
                                        if m == 1:
                                            if np.abs(coeffs[0]) < 1e-6:
                                                continue
                                        try:
                                            val = objective(coeffs)
                                            if val < best_value:
                                                best_value = val
                                                best_coeffs = coeffs
                                        except:
                                            continue
        # Gradient descent refinement
        if m <= 8:
            bounds = [(-10.0, 10.0) for _ in range(m)]
            guess = np.random.rand(m) * 2.0 - 1.0
            start_time = time.time()
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=bounds,
                options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
            )
            end_time = time.time()
            if end_time - start_time < budget_s:
                coeffs = result.x
                val = objective(coeffs)
                if val < best_value:
                    best_value = val
                    best_coeffs = coeffs
    
    return best_coeffs
