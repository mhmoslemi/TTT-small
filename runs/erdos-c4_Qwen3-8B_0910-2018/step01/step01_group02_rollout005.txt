import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        return np.sum(x**2)
    
    def evaluate_exact(coeffs):
        return np.sum(coeffs**2) / 2  # Placeholder for exact evaluation
    
    best_coeff = None
    best_bound = float('inf')
    total_time = 0
    
    def search_with_count(n):
        nonlocal total_time
        start = time.time()
        bounds = Bounds(lb=-1.0, ub=1.0)
        x0 = np.random.rand(n) * 2 - 1  # Alternating signs
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        elapsed = time.time() - start
        total_time += elapsed
        if total_time >= budget_s:
            return None
        return result.x
    
    for count in range(1, 9):
        if total_time >= budget_s:
            break
        print(f"Trying with {count} coefficients...")
        coeffs = search_with_count(count)
        if coeffs is None:
            break
        current_bound = evaluate_exact(coeffs)
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = coeffs
    
    return best_coeff
