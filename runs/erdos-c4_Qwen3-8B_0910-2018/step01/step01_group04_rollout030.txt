import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import _checkfinite

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        # Placeholder for exact evaluation; in practice, this would call the authoritative function
        return np.sum(coeffs**2)
    
    def surrogate_objective(coeffs):
        # Fast numerical estimation
        return np.sum(coeffs**2)
    
    best_coeffs = None
    best_value = float('inf')
    total_time = 0
    
    for m in range(1, 9):
        bounds = [(-10.0, 10.0) for _ in range(m)]
        initial_guess = np.random.uniform(-1.0, 1.0, m)
        
        start_time = time.time()
        result = minimize(
            surrogate_objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 200}
        )
        elapsed_time = time.time() - start_time
        total_time += elapsed_time
        
        if total_time >= budget_s:
            break
        
        refined_coeffs = result.x
        exact_value = evaluate_c4_coefficients(refined_coeffs)
        
        if exact_value < best_value:
            best_value = exact_value
            best_coeffs = refined_coeffs
    
    return best_coeffs
