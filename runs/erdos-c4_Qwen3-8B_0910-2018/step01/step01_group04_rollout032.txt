import numpy as np
import time
from scipy.optimize import minimize
from scipy.integrate import quad
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(n):
        if n == 0:
            return np.poly1d([1.0])
        elif n == 1:
            return np.poly1d([1.0, 0.0])
        else:
            p = np.zeros(n + 1)
            p[0] = 1.0
            for k in range(1, n + 1):
                p = np.convolve(p, [1.0, 0.0, 1.0])
                p = p / np.sqrt(2**(k-1))
            return np.poly1d(p)
    
    def evaluate_bound(coeffs):
        m = len(coeffs)
        c = coeffs
        H = [hermite_poly(k) for k in range(m)]
        Q = sum(c[k] * H[k] for k in range(m))
        Q = Q / (Q[2] ** 2)
        r_max = 0.0
        def f(z):
            return Q(z)
        try:
            roots = np.roots(Q.coef)
            real_roots = [r for r in roots if np.isclose(r.imag, 0)]
            real_roots = sorted(real_roots)
            for r in real_roots:
                if r > 0:
                    r_max = r
            return r_max ** 2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(coeffs):
        return evaluate_bound(coeffs)
    
    # Initial guess with orthogonal basis
    def init_coeffs(m):
        return np.random.randn(m)
    
    best_coeffs = None
    best_value = float('inf')
    
    for m in range(1, 9):
        bounds = [(-10.0, 10.0) for _ in range(m)]
        guess = init_coeffs(m)
        
        start_time = time.time()
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 200}
        )
        end_time = time.time()
        
        if end_time - start_time > budget_s:
            continue
        
        current_value = result.fun
        if current_value < best_value:
            best_value = current_value
            best_coeffs = result.x
    
    return best_coeffs
