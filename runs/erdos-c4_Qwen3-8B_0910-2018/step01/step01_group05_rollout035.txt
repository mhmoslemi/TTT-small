import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from functools import partial
import time
from sympy import symbols, HermiteH, Poly, solve

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    best_coeffs = np.random.rand(8)
    best_value = float('inf')
    time_remaining = budget_s
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        poly = 0
        for k in range(len(coeffs)):
            poly += coeffs[k] * HermiteH(4*k, z)
        # Ensure Q(0) = 0
        poly = poly.subs(z, 0)
        c_m = -poly
        poly += c_m
        # Flip sign if needed
        if poly.subs(z, 100).as_numerical_value() < 0:
            poly *= -1
        # Remove z^2 factor
        poly = poly.as_poly().as_ordered_terms()
        if len(poly) > 2:
            poly = poly[2]
        else:
            poly = 0
        # Find roots
        roots = solve(poly, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 100
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def grid_search(n_coeff):
        grid = np.linspace(-1, 1, 100)
        best = np.inf
        best_coeffs = np.random.rand(n_coeff)
        for i in range(len(grid)):
            for j in range(len(grid)):
                for k in range(len(grid)):
                    if n_coeff == 3:
                        coeffs = np.array([grid[i], grid[j], grid[k]])
                    elif n_coeff == 4:
                        coeffs = np.array([grid[i], grid[j], grid[k], grid[i]])
                    elif n_coeff == 5:
                        coeffs = np.array([grid[i], grid[j], grid[k], grid[i], grid[j]])
                    elif n_coeff == 6:
                        coeffs = np.array([grid[i], grid[j], grid[k], grid[i], grid[j], grid[k]])
                    elif n_coeff == 7:
                        coeffs = np.array([grid[i], grid[j], grid[k], grid[i], grid[j], grid[k], grid[i]])
                    elif n_coeff == 8:
                        coeffs = np.array([grid[i], grid[j], grid[k], grid[i], grid[j], grid[k], grid[i], grid[j]])
                    value = objective(coeffs)
                    if value < best:
                        best = value
                        best_coeffs = coeffs
        return best_coeffs, best
    
    for n_coeff in range(1, C4_MAX_COEFF_COUNT + 1):
        if time_remaining <= 0:
            break
        start = time.time()
        coeffs, value = grid_search(n_coeff)
        time_used = time.time() - start
        time_remaining -= time_used
        if value < best_value:
            best_value = value
            best_coeffs = coeffs
    
    return best_coeffs
