import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coeffs(coeffs):
        return coeffs.copy()
    
    def objective(coeffs):
        return np.sum(coeffs ** 2)
    
    def bounds(n):
        return Bounds([-10.0 for _ in range(n)], [10.0 for _ in range(n)])
    
    best_coeffs = np.array([1.0])
    best_score = float('inf')
    start_time = time.time()
    
    for n in range(1, 9):
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with {n} coefficients")
        initial_guess = np.random.uniform(-1.0, 1.0, n)
        result = opt.minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds(n))
        if result.success:
            score = result.fun
            if score < best_score:
                best_score = score
                best_coeffs = result.x
                print(f"New best score: {best_score}")
    
    return best_coeffs
