import time
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import dual_annealing

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_Q(coeffs, z):
        # Compute Hermite polynomials H_{4k}(z)
        H = [1.0]  # H_0(z) = 1
        for k in range(1, len(coeffs)*4 + 1):
            H_k = np.polysub(np.polymul([1.0, -2.0*z], H[-1]), np.polymul([1.0, -z], H[-2]))
            H.append(H_k)
        
        # Construct Q(z)
        Q = np.zeros(len(H[-1]) + 1)
        for k in range(len(coeffs)):
            H_k = H[4*k]
            Q = np.polynomial.polypadd(Q, H_k)
            Q = np.polynomial.polymul(Q, coeffs[k])
        
        # Normalization: Q(0) = 0 and ensure positivity at infinity
        if Q[0] != 0:
            Q = np.polysub(Q, Q[0])
        
        # Find r_max
        def f(x):
            return np.polyval(Q, x) / (x**2)
        
        # Find largest positive real root where sign changes
        try:
            roots = np.roots(Q)
            real_roots = roots[np.isreal(roots)]
            real_roots = np.sort(real_roots)
            
            # Check sign changes across real_roots
            r_max = None
            for i in range(len(real_roots) - 1):
                if np.sign(f(real_roots[i])) != np.sign(f(real_roots[i+1])):
                    r_max = real_roots[i+1]
                    break
            
            if r_max is None:
                r_max = real_roots[-1]
            
            if r_max < 0:
                r_max = None
            
            if r_max is not None:
                return r_max**2 / (2 * np.pi)
            else:
                return float('inf')
        except:
            return float('inf')
    
    def objective(coeffs):
        return evaluate_Q(coeffs, 0)

    for m in range(1, 6):
        bounds = [(-10.0, 10.0) for _ in range(m)]

    return best_coeffs
