import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import sympy as sp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_with_symbolics(coeffs):
        m = len(coeffs)
        z = sp.symbols('z')
        Q = sum(coeffs[k] * sp.hermites(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = sp.solvers.solve(sp.simplify(Q), z)
        real_roots = [r.evalf() for r in roots if r.is_real and r != 0]
        real_roots = np.array([r for r in real_roots if r > 0])
        if len(real_roots) == 0:
            return np.inf
        r_max = real_roots[-1]
        return r_max**2 / (2*np.pi)
    
    def objective(coeffs):
        return evaluate_with_symbolics(coeffs)
    
    def gradient(coeffs):
        eps = 1e-6
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            perturbed = coeffs.copy()
            perturbed[i] += eps
            grad[i] = (objective(perturbed) - objective(coeffs)) / eps
        return grad
    
    best_coeffs = np.array([1.0])
    best_value = objective(best_coeffs)
    
    for num_coeff in range(1, 9):

        for candidate in candidates:
            res = minimize(
                objective,
                candidate,
                method='L-BFGS-B',
                jac=gradient,
                bounds=Bounds(-100, 100),

            )
            if res.fun < best_value:
                best_value = res.fun
                best_coeffs = res.x
    
    return best_coeffs
