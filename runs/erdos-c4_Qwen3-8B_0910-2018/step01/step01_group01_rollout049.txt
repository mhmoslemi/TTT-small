import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(z, coeffs):
        h = np.zeros_like(z)
        for k, c in enumerate(coeffs):
            h += c * np.polyspecial.hermite(4*k, z)
        return h
    
    def evaluate_c4_coeffs(coeffs):
        z = np.linspace(-10, 10, 1000)
        Q = hermite_poly(z, coeffs)
        z_squared = z**2
        Q_div_z_squared = Q / z_squared
        r_max = 0
        for i in range(len(z)-1):
            if np.sign(Q_div_z_squared[i]) != np.sign(Q_div_z_squared[i+1]):
                r_max = z[i+1]
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        try:
            return evaluate_c4_coeffs(coeffs)
        except:
            return np.inf
    
    def bounds(n):
        return Bounds([-10.0 for _ in range(n)], [10.0 for _ in range(n)])
    
    best_coeffs = np.array([1.0])
    best_score = float('inf')
    start_time = time.time()
    
    def initial_grid_search():
        nonlocal best_coeffs, best_score
        for c in np.linspace(-10, 10, 20):
            score = objective(np.array([c]))
            if score < best_score:
                best_score = score
                best_coeffs = np.array([c])
    
    def bayesian_optimize(n):
        nonlocal best_coeffs, best_score
        initial_guess = np.random.uniform(-10, 10, n)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds(n))
        if result.success:
            score = result.fun
            if score < best_score:
                best_score = score
                best_coeffs = result.x
    
    initial_grid_search()
    
    for n in range(2, 9):
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with {n} coefficients")
        initial_guess = np.random.uniform(-10, 10, n)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds(n))
        if result.success:
            score = result.fun
            if score < best_score:
                best_score = score
                best_coeffs = result.x
    
    return best_coeffs
