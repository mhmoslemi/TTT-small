import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):

    def hermite_poly(x, coeffs):
        return np.sum([coeffs[i] * np.polyspecial.hermetepoly(4*i, x) for i in range(len(coeffs))])

            return Q(z) / (z**2)

            return evaluate_c4_coeffs(coeffs)
