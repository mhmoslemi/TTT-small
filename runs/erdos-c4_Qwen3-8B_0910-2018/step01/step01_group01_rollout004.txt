import scipy.optimize as opt
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_poly(z, coeffs):
        H = [1.0]
        for k in range(len(coeffs)):
            H.append(2 * (2*k + 1) * H[k] * z - 2*k * H[k-1] if k > 0 else 0)
        return np.polynomial.hermite.hermval(z, H)
    
    def evaluate_c4_coeffs(coeffs):
        H = [1.0]
        for k in range(len(coeffs)):
            H.append(2 * (2*k + 1) * H[k] * z - 2*k * H[k-1] if k > 0 else 0)
        Q = sum(c * H[4*k + 1] for k, c in enumerate(coeffs))
        Q = Q / z**2
        roots = np.roots(Q)
        r_max = None
        for root in roots:
            if np.isreal(root) and root > 0:
                r = root.real
                if r_max is None or r > r_max:
                    r_max = r
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        try:
            return evaluate_c4_coeffs(coeffs)
        except:
            return np.inf
    
    bounds = [(-10.0, 10.0) for _ in range(8)]
    
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=100, tol=1e-10, mutation=(0.5, 1), recombination=0.9)
    
    return result.x
