import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with 2 coefficients
    initial_coeff = np.random.rand(2)
    
    # Store best coefficients and bound
    best_coeff = initial_coeff
    best_bound = float('inf')
    best_n = 2
    
    # GP surrogate model
    kernel = C(1.0, (1e-5, 1e5)) * RBF(1.0, (1e-5, 1e5))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
    
    def objective(x, n):
        # Numerical surrogate to estimate U(c)
        # This will be replaced with exact calculation in evaluate_c4_coefficients
        # Use a simple proxy for demonstration
        return np.sum(x**2) / 2
    
    # Define bounds for coefficients
    bounds = Bounds(lb=-1.0, ub=1.0)
    
    # Bayesian optimization with adaptive coefficient count
    def search_with_count(n):
        print(f"Trying with {n} coefficients...")
        x0 = np.random.rand(n)
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        print(f"Search with {n} coefficients completed.")
        return result.x
    
    # Adaptive coefficient count based on previous results
    for count in range(2, 9):
        print(f"Trying with {count} coefficients...")
        start_time = time.time()
        
        # Fit GP on previous results
        if count > 2:
            X = np.array([np.random.rand(count) for _ in range(10)])
            y = np.array([objective(x, count) for x in X])
            gp.fit(X, y)
        
        # Sample from GP predictive distribution
        X_pred = np.array([np.random.rand(count) for _ in range(10)])
        y_pred, y_std = gp.predict(X_pred, return_std=True)
        candidates = X_pred[np.argsort(y_pred)]
        
        # Evaluate top candidates
        best_candidate = None
        best_val = float('inf')
        for candidate in candidates[:5]:
            val = objective(candidate, count)
            if val < best_val:
                best_val = val
                best_candidate = candidate
        
        # Use best candidate as starting point
        x0 = best_candidate
        result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        elapsed = time.time() - start_time
        
        # Simulate exact evaluation (in real code, this would be done via evaluate_c4_coefficients)
        # For now, use the objective value as a proxy
        current_bound = result.fun
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = result.x
            best_n = count
        print(f"Current best bound: {best_bound}, Elapsed: {elapsed:.2f}s")
    
    return best_coeff
