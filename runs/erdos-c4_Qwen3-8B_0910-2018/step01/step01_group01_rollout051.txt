import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_coeffs(n):
        return np.random.uniform(0.1, 1.0, n)
    
    def evaluate_c4_coeffs(coeffs):
        return coeffs
    
    def objective(coeffs):
        try:
            res = evaluate_c4_coeffs(coeffs)
            return res[0]
        except:
            return np.inf
    
    def bounds(n):
        return [(-1.0, 1.0) for _ in range(n)]
    
    best_coeffs = np.array([1.0])
    best_score = float('inf')
    start_time = time.time()
    
    for n in range(1, 9):
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with {n} coefficients")
        initial_guess = hermite_coeffs(n)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds(n))
        if result.success:
            score = result.fun
            if score < best_score:
                best_score = score
                best_coeffs = result.x
                print(f"New best score: {score}")
    
    return best_coeffs
