import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_eval(z, coeffs):
        from scipy.special import hermite
        H = np.array([hermite(k)(z) for k in range(len(coeffs))])
        return np.dot(coeffs, H)
    
    def evaluate_Q(z, coeffs):
        Q = hermite_eval(z, coeffs)
        return Q / z**2
    
    def find_r_max(coeffs):
        from scipy.optimize import root_scalar
        try:
            result = root_scalar(
                lambda z: evaluate_Q(z, coeffs),
                bracket=[1, 100],
                method='brentq'
            )
            return result.root
        except:
            return 100
    
    def compute_bound(coeffs):
        r_max = find_r_max(coeffs)
        return r_max**2 / (2 * np.pi)
    
    def surrogate_objective(coeffs):
        return -compute_bound(coeffs)
    
    def exact_objective(coeffs):
        return compute_bound(coeffs)
    
    best_coeffs = None
    best_value = float('inf')
    total_time = 0
    
    for m in range(3, 6):
        bounds = [(-5.0, 5.0) for _ in range(m)]
        initial_guesses = [np.random.rand(m) * 2 - 1 for _ in range(10)]
        
        for guess in initial_guesses:
            start_time = time.time()
            result = minimize(
                surrogate_objective,
                guess,
                method='L-BFGS-B',
                bounds=bounds,
                options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100}
            )
            elapsed = time.time() - start_time
            total_time += elapsed
            
            if total_time >= budget_s:
                break
            
            refined_guess = result.x
            exact_value = exact_objective(refined_guess)
            
            if exact_value < best_value:
                best_value = exact_value
                best_coeffs = refined_guess
    
    return best_coeffs
