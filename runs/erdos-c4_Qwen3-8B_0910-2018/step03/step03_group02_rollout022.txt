import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(coeffs):
        result = evaluate_c4_coefficients(coeffs)
        return result
    
    def bounded_optimizer(start, bounds, max_iter=100):
        result = minimize(objective, start, bounds=bounds, method='L-BFGS-B', tol=1e-10, options={'maxiter': max_iter})
        return result.x
    
    def grid_search(n_points=5, dim=8):
        points = np.linspace(-2, 2, n_points)
        grid = np.array(np.meshgrid(*[points for _ in range(dim)])).T.reshape(-1, dim)
        return grid
    
    start_time = time.time()
    best_coeffs = np.random.rand(8) * 2 - 1
    best_value = objective(best_coeffs)
    
    grid = grid_search(n_points=5, dim=8)
    for coeffs in grid:
        if time.time() - start_time > budget_s - 10:
            break
        if np.sum(np.abs(coeffs)) == 0:
            continue
        try:
            optimized = bounded_optimizer(coeffs, Bounds(-10, 10), max_iter=100)
            value = objective(optimized)
            if value < best_value:
                best_value = value
                best_coeffs = optimized
        except:
            continue
    
    return best_coeffs
