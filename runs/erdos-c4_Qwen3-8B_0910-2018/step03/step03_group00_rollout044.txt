from scipy.optimize import minimize

import time

def run(seed=42, budget_s=600, **kwargs):

    def hermite_poly(n):
        if n == 0:
            return lambda z: 1
        elif n == 1:
            return lambda z: 2*z
        elif n == 2:
            return lambda z: 4*z**2 - 2
        elif n == 3:
            return lambda z: 8*z**3 - 12*z
        else:
            raise ValueError("Hermite polynomial not implemented for n > 3")
    
    def evaluate_c4_coefficients(coeffs):
        from sympy import symbols, expand, diff, simplify, solve, Eq
        z = symbols('z')

        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if len(real_roots) > 0 else 1
        return r_max**2 / (2 * np.pi)
    
    def chebyshev_param(n_coeff):
        x = np.linspace(-1, 1, n_coeff)
        return 0.5 * (1 - x)
    
    best_coeffs = np.array([0.5, 0.5, 0.5])

            pass

    return best_coeffs
