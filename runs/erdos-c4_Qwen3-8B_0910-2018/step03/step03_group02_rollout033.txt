from scipy.optimize import minimize, Bounds
from sympy import symbols, HermitePolynomial, Poly, solve, S
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8

    def get_hermite_coeffs(n):
        z = symbols('z')
        return [HermitePolynomial(n, z).as_poly().all_coeffs()[::-1] for n in range(C4_MAX_COEFF_COUNT)]

    def construct_Q(coeffs):
        z = symbols('z')
        Q = sum(c * HermitePolynomial(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.as_poly().as_expr()
        return Q

    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * HermitePolynomial(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.as_poly().as_expr()
        
        Q = Q - Q.as_poly().coeffs()[0] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        
        if not real_roots:
            return float('inf')
        
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)

    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)

    def refine_guess(coeffs):

        coeffs[1] = 1.0
        return coeffs
