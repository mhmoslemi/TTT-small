import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import qmc
from functools import partial
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    
    def evaluate_c4_coefficients(coeffs):
        from scipy.optimize import root_scalar
        from sympy import symbols, hermite, simplify, Poly, roots
        from fractions import Fraction
        
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        Q = Q.subs(z, 0).as_numerical_integral()
        Q = Q / z**2
        try:
            r_max = root_scalar(lambda x: Q.subs(z, x), bracket=[0, 10], method='brentq').root
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def surrogate(coeffs):
        return np.sum((coeffs**2) * np.linspace(1, 0.1, len(coeffs)))
    
    best_value = float('inf')
    best_coeffs = np.zeros(C4_MAX_COEFF_COUNT)
    
    start_time = time.time()
    
    # Lattice-based initialization
    sample = qmc.qmcsample(np.linspace(-1, 1, C4_MAX_COEFF_COUNT), 20, random_state=seed)
    for init in sample:
        coeffs = 0.5 * (1 - 2 * init)
        result = minimize(
            lambda x: surrogate(x),
            x0=coeffs,
            method='L-BFGS-B',
            bounds=[(-1, 1)]*C4_MAX_COEFF_COUNT,
            options={'ftol': 1e-8, 'gtol': 1e-8}
        )
        coeffs = result.x
        exact_value = evaluate_c4_coefficients(coeffs)
        if exact_value < best_value:
            best_value = exact_value
            best_coeffs = coeffs
    
    # Refinement with Bayesian optimization
    from skopt import BayesSearchSpace, BayesianOptimization
    space = BayesSearchSpace({

        func=partial(evaluate_c4_coefficients, coeffs),
        search_space=space,
        n_iter=20,
        random_state=seed
    )

            break
        x = optimizer.ask()

    return best_coeffs
