from scipy.optimize import minimize
from scipy.optimize import Bounds
import time
from sympy import symbols, Eq, solve, hermite, Poly, SturmSequence, roots, Interval
from sympy.calculus.util import minimum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        Q = Q - Q.as_numerical().subs(z, 0)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        for i in range(len(real_roots)):
            if i > 0 and real_roots[i] == real_roots[i-1]:
                continue
            if i < len(real_roots) - 1 and real_roots[i] == real_roots[i+1]:
                continue
            r_max = real_roots[i]
            return r_max**2 / (2 * np.pi)
        return float('inf')
    
    def grid_search(start, end, step, dim):
        best = np.inf
        best_coeffs = np.random.rand(dim) * 2 - 1
        for a in np.arange(start, end, step):
            for b in np.arange(start, end, step):
                for c in np.arange(start, end, step):
                    for d in np.arange(start, end, step):
                        coeffs = np.array([a, b, c, d])
                        val = evaluate_c4_coefficients(coeffs)
                        if val < best:
                            best = val
                            best_coeffs = coeffs
