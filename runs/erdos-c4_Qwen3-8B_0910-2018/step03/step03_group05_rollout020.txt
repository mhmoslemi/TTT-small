from scipy.optimize import minimize, Bounds
from functools import lru_cache
import time
from scipy.special import hermite
from sympy import symbols, Eq, solve, simplify, N

def run(seed=42, budget_s=600, **kwargs):

    best_coeffs = np.array([])
    best_value = float('inf')
    start_time = time.time()
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def evaluate_c4_coefficients(coeffs):
        from sympy import symbols, Eq, solve, simplify, N, Poly, roots
        z = symbols('z')
        H = [simplify(hermite(k, z)) for k in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        Q = Q.as_poly()
        roots = N(roots(Q, z))
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([float(r) for r in real_roots if r > 0])
        r_max = real_roots[-1] if len(real_roots) > 0 else 1.0
        return r_max**2 / (2 * np.pi)
    
    def hybrid_optimizer(initial_guess, bounds, max_iter=100):
        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': max_iter})
        return result.x
    
    for n_coeff in range(1, 9):
