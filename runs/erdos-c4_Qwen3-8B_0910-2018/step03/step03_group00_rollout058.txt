import numpy as np
from scipy.optimize import minimize
from scipy.special import hermite
from numpy.polynomial.hermite_e import HermiteE
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def construct_Q(coeffs):
        x = np.linspace(-1, 1, 1000)
        H = np.array([HermiteE(4*k)(x) for k in range(len(coeffs))])
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q /= x**2
        return Q
    
    def find_r_max(Q):
        Q = Q[Q > -1e-10]
        roots = np.roots(Q)
        real_roots = roots[np.isreal(roots)]
        real_roots = np.sort(real_roots)
        for r in real_roots:
            if r > 0 and np.abs(Q[np.abs(x - r) < 1e-5]) < 1e-5:
                return r**2
        return np.inf
    
    def objective(coeffs):
        x = np.linspace(-1, 1, 1000)
        H = np.array([HermiteE(4*k)(x) for k in range(len(coeffs))])
        Q = np.sum(coeffs[:, np.newaxis] * H, axis=0)
        Q[0] = 0
        Q /= x**2
        Q[0] = 0
        r_max = find_r_max(Q)
        return r_max / (2 * np.pi)
    
    best_value = float('inf')

                best_value = res.fun
                best_coeffs = res.x
                
    return best_coeffs
