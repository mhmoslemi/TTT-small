import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, Eq, solve, hermite, Rational

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.as_expr()
        Q = Q.subs(z, 0)
        Q = Q.as_numer_denom()[0]
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        real_roots = [r for r in real_roots if not r.is_zero]
        real_roots = sorted(real_roots)
        r_max = real_roots[-1] if len(real_roots) > 0 else 1
        return r_max**2 / (2 * np.pi)
    
    def bounded_optimizer(start, bounds, max_iter=100):
        result = minimize(objective, start, bounds=bounds, method='L-BFGS-B', tol=1e-12, options={'maxiter': max_iter})
        return result.x
    
    def refine_guess(coeffs):
        coeffs = np.copy(coeffs)
        coeffs[1] = 1.0
        return coeffs
    
    start_time = time.time()
    best_coeffs = np.array([Rational(1, 1), Rational(1, 1)])
    best_value = evaluate_c4_coefficients(best_coeffs)
    
    initial_guess = np.array([Rational(1, 1), Rational(1, 1)])
    refined_guess = refine_guess(initial_guess)
    best_coeffs = refined_guess
    best_value = evaluate_c4_coefficients(best_coeffs)
    
    for m in range(2, 6):
        for _ in range(50):
            if time.time() - start_time > budget_s - 10:
                break

            current_coeffs = np.concatenate((current_coeffs, np.array([1.0])))
            current_coeffs = current_coeffs[:m]
            optimized = bounded_optimizer(current_coeffs, Bounds([-10, -10], [10, 10]))
            value = evaluate_c4_coefficients(optimized)
            if value < best_value:
                best_value = value
                best_coeffs = optimized
    
    return best_coeffs
