import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import fsolve
from functools import lru_cache
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeffs = np.array([0.5, 0.5])
    best_value = float('inf')
    start_time = time.time()
    
    def check_bounds(coeffs):
        return np.abs(coeffs).sum() < 1e-6
    
    def analytical_bound(z, coeffs):
        H = np.polymul([1, 0, 0], [1, 0, 0])  # Placeholder for Hermite polynomial
        Q = np.zeros_like(coeffs)
        for k, c in enumerate(coeffs):
            Q += c * H[4*k]
        Q /= z**2
        return Q
    
    def find_r_max(coeffs):
        def f(z):
            return analytical_bound(z, coeffs)
        try:
            root = fsolve(f, 1.0)
            return root[0]**2
        except:
            return float('inf')
    
    def objective(coeffs):
        nonlocal best_value
        if check_bounds(coeffs):
            return float('inf')
        r_max = find_r_max(coeffs)
        if r_max == float('inf'):
            return float('inf')
        return r_max / (2 * np.pi)
    
    def hybrid_optimizer(initial_guess, bounds, max_iter=100):
        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': max_iter})
        return result.x
    
    for n_coeff in range(1, 9):
        if time.time() - start_time > budget_s - 10:
            break
        for _ in range(5):
            angles = np.linspace(0, 2*np.pi, n_coeff, endpoint=False)
            initial = np.cos(angles) * np.random.rand(n_coeff)
            result = hybrid_optimizer(initial, Bounds(-1, 1))
            current_value = objective(result)
            if current_value < best_value:
                best_value = current_value
                best_coeffs = result
    
    return best_coeffs
