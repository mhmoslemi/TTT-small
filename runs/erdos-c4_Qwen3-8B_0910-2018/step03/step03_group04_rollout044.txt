import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.special import hermite

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    best_count = 1
    
    for count in range(1, 9):
        print(f"Testing {count} coefficients...")
        
        if count == 1:
            x0 = np.array([0.5])
        else:
            x0 = np.random.uniform(-1, 1, count)
        
        def objective(x):
            coeffs = np.array(x, dtype=np.float64)
            z = np.linspace(-10, 10, 1000)
            Q = np.zeros_like(z)
            for k in range(len(coeffs)):
                Q += coeffs[k] * hermite(4*k, z)
            Q /= z**2
            sign_changes = np.where(np.diff(np.sign(Q)) != 0)[0]
            r_max = 10.0
            for root_idx in sign_changes:
                root = z[root_idx]
                if root > 0 and root < r_max:
                    r_max = root
            return r_max**2 / (2 * np.pi)
        
        result = minimize(objective, x0, bounds=[(-1, 1) for _ in range(count)], method='L-BFGS-B', options={'maxiter': 100})
        current_bound = result.fun
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeff = result.x
            best_count = count
        print(f"Best bound: {best_bound} with {best_count} coefficients")
    
    return best_coeff
