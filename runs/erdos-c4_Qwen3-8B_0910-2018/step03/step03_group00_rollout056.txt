def run(seed=42, budget_s=600, **kwargs):

    def evaluate_c4_coefficients(coeffs):
        z = sp.symbols('z')
        Q = sum(c * sp.hermite.H(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.subs(z, 0)
        Q = Q.as_numerical()
        Q = Q / z**2
        roots = sp.solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        r_max = max(real_roots) if real_roots else 0
        return r_max**2 / (2 * np.pi)
    
    best_value = float('inf')
    best_coeffs = np.array([0.5, 0.5, 0.5])
    
    def chebyshev_param(n_coeff):
        x = np.linspace(-1, 1, n_coeff)
        return 0.5 * (1 - x)
    
    start_time = time.time()
