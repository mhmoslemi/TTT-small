import numpy as np
from scipy.optimize import minimize
from sympy import symbols, diff, hermite, Lambda, nsolve
from functools import lru_cache
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeffs = np.array([])
    best_value = float('inf')
    start_time = time.time()
    
    def evaluate(coeffs):
        z = symbols('z')
        poly = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        poly = poly.as_poly()

        roots = [r for r in poly.all_roots() if r.is_real and r > 0]
        roots = [r for r in roots if np.isclose(poly.evalf(subs=z).subs(z, r), 0, atol=1e-8)]
        if not roots:
            return float('inf')
        r_max = max(roots)
        return r_max**2 / (2 * np.pi)
    
    def fast_evaluate(coeffs):

        poly = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        poly = poly / z**2
        sign_changes = 0
        prev_sign = np.sign(poly[0])
        for val in poly[1:]:
            if np.sign(val) != prev_sign:
                sign_changes += 1
                prev_sign = np.sign(val)
        if sign_changes == 0:
            return float('inf')
        r_max = np.max(z[np.abs(poly) > 1e-6])
