import numpy as np
import sympy as sp
from scipy.optimize import Bounds, minimize, gp
from scipy.optimize import LinearConstraint, NonlinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_coeffs = np.array([1.0, 0.0])
    best_coeffs = initial_coeffs
    best_value = float('inf')
    total_time = 0

    def evaluate_coeffs(coeffs):
        coeffs = np.array(coeffs, dtype=np.float64)
        m = len(coeffs)
        z = sp.symbols('z')
        Q = sum(coeffs[k] * sp.hermite.H(4*k, z) for k in range(m))
        Q = Q.as_poly().as_expr()
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = sp.solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [float(r) for r in real_roots]
        real_roots = sorted(real_roots)
        r_max = None
        for r in real_roots:
            if r > 0:
                r_max = r
                break
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)

    # Grid search with adaptive refinement
    for n in [2, 4, 6, 8]:
        grid = np.array([np.linspace(-1, 1, 20) for _ in range(n)]).T
        for coeffs in grid:
            value = evaluate_coeffs(coeffs)
            if value < best_value:
                best_value = value
                best_coeffs = coeffs

    # Bayesian optimization with custom kernel
    class CustomKernel:
        def __init__(self, lengthscale=1.0):
            self.lengthscale = lengthscale

        def __call__(self, X1, X2):
            dist = np.sqrt(np.sum((X1 - X2)**2, axis=1))
            return np.exp(-dist / self.lengthscale)

    class CustomAcquisition:
        def __init__(self, kernel, X, y, bounds):
            self.kernel = kernel
            self.X = X
            self.y = y
            self.bounds = bounds

        def __call__(self, x):
            K = self.kernel(self.X, x.reshape(1, -1))
            K_XX = self.kernel(self.X, self.X)
            K_inv = np.linalg.pinv(K_XX)
            mu = np.mean(self.y)
            sigma = np.sqrt(np.diag(K @ K_inv @ K.T))
            acq = mu - 1.96 * sigma
            return acq

    def objective(coeffs):
        return evaluate_coeffs(coeffs)

    bounds = Bounds(-10, 10)
    X = np.array([np.random.uniform(-10, 10, size=8) for _ in range(100)])
    y = np.array([objective(x) for x in X])
    kernel = CustomKernel(lengthscale=1.0)
    acq = CustomAcquisition(kernel, X, y, bounds)

    for _ in range(20):
        time_start = time.time()
        if total_time >= budget_s:
            break
        candidate = np.random.uniform(-10, 10, size=8)
        acq_val = acq(candidate)
        if acq_val < 0:
            continue
        best_coeffs = candidate
        total_time += time.time() - time_start

    return best_coeffs
