import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.stats import qmc
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    
    def evaluate_c4_coefficients(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    # Generate initial points using Latin hypercube sampling
    samples = qmc.LatinHypercube(d=8, random_state=seed).generate(samples=16)
    samples = np.clip(samples * 10 - 5, -10, 10)
    
    # Initialize Gaussian process model
    kernel = ConstantKernel(1.0) * RBF(length_scale=1.0, length_scale_bounds=(0.1, 10.0))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
    
    def optimize_coefficients():
        budget = budget_s
        time_start = time.time()
        best_x = samples[0]
        best_f = objective(best_x)
        
        for i in range(200):
            if time.time() - time_start > budget:
                break
            
            # Predict next point
            y = np.array([objective(x) for x in samples])
            gp.fit(samples, y)
            next_x = gp.predict(np.atleast_2d(samples.mean(axis=0)), return_std=True)[0]
            
            # Evaluate at current best
            next_x = np.clip(next_x, -10, 10)
            
            # Use L-BFGS-B for local refinement
            result = minimize(
                fun=objective,
                x0=next_x,
                method='L-BFGS-B',
                bounds=[(-10, 10) for _ in range(8)],
                options={'maxfun': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            new_f = result.fun
            if new_f < best_f:
                best_f = new_f
                best_x = result.x
                
            samples = np.vstack((samples, result.x))
        
        return best_x
    
    best_coeffs = optimize_coefficients()
    return best_coeffs
