from scipy.optimize import minimize
from functools import partial
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_coeff = 6  # Use 6 free coefficients for better balance
    
    # Define symbolic variables
    z = symbols('z')
    H = [1]  # H_0(z)
    for n in range(1, 24):
        H.append(diff(H[-1], z) * 2 * z - 2 * n * H[-1])

    H_4k = [simplify(H[4*k]) for k in range(n_coeff)]

            Q -= Q.subs(z, 0)

        if Q.as_infinite_limit(z) > 0:
            Q = -Q
        # Remove z^2 factor (numerically approximate leading term)
        leading_term = Q.as_leading_coefficient()
        Q = Q / leading_term
        # Find r_max numerically
        try:
            from scipy.optimize import root_scalar
            try:
                r_max = root_scalar(lambda x: Q.subs(z, x), bracket=[1, 100], method='brentq').root

                r_max = 100
            return r_max**2 / (2 * np.pi)

            return float('inf')
