import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time
from sympy import symbols, hermite, solve, S, N

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    time_start = time.time()
    
    def evaluate_coefficients(c):
        z = symbols('z')
        poly = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        poly = poly.as_coefficients_dict()
        poly = sum(k * v for k, v in poly.items())  # Convert to standard form
        poly = poly.as_poly()
        poly = poly.remove_order()
        poly = poly.as_poly()
        poly = poly / z**2
        roots = solve(poly, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if np.isclose(r, 0):
                continue
            try:
                if np.sign(poly.subs(z, r - 1e-6)) != np.sign(poly.subs(z, r + 1e-6)):
                    return (r**2) / (2 * np.pi)
            except:
                continue
        return float('inf')
    
    for count in range(1, 9):
        print(f"Testing {count} coefficients...")
        if count == 1:
            c = np.array([0.5])
        else:
            c = np.random.uniform(-1, 1, count)
        c = np.clip(c, -1, 1)
        
        def objective(x):
            return evaluate_coefficients(x)
        
        bounds = Bounds(lb=-1.0, ub=1.0)
        result = minimize(objective, c, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        
        if result.fun < best_bound:
            best_bound = result.fun
            best_coeff = result.x
    
    return best_coeff
