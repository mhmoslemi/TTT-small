@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)  # BH index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # compute masks for out-of-range on i and j
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute linear offsets for left and out start
    batch_offset = pid_batch * stride_left_batch

    # Pointers for left (i,k)
    # k loop will be inside
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # left_ptr at (i,k)
        left_ptrs = left_ptr + batch_offset + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_block = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right_ptr at (k,j) - transposed view
        # for right, K dimension as first, J as second
        right_ptrs = right_ptr + batch_offset + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_block = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left_block, right_block)

    # store output
    out_ptrs = out_ptr + batch_offset + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul forward pass with a fused Triton kernel.

The heavy cubic operation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix multiplication using a
custom Triton kernel.  All other operations (LayerNorms,
linear projections, gating) are performed with PyTorch for
simplicity and numerical stability.

The kernel processes each (batch, hidden) pair independently
and tiles the i-, j- and k- dimensions (BLOCK_M x BLOCK_N x BLOCK_K).
It accumulates the result in float32 and writes the output as
float32.  The surrounding code reshapes tensors to a
[batch*hidden, N, N] view so the kernel can treat the problem as a
standard batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: computes out = left @ right^T for each batch*head
# ------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                      # [BH, N, N] left matrix (i,k)
    right_ptr,                     # [BH, N, N] right matrix (i,k) – will be read transposed
    out_ptr,                       # [BH, N, N] output matrix (i,j)
    N,                             # sequence length
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)      # combined batch * hidden index
    pid_i = tl.program_id(0)          # i-tile index
    pid_j = tl.program_id(1)          # j-tile index

    # Starting coordinates of the tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Range of indices for this tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds (for the tail tiles)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Batch offset (same for left, right, out)
    batch_offset = pid_batch * stride_left_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile L[i,k]
        left_ptrs = left_ptr + batch_offset + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # Load right tile transposed R_T[k,j] = right[j,k]
        # Note: we stride over k using stride_right_k and over j using stride_right_i
        right_ptrs = right_ptr + batch_offset + \
                     offs_k[:, None] * stride_right_k + \
                     offs_j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # Rank‑K update
        acc += tl.dot(left_tile, right_tile)

    # Write the result
    out_ptrs = out_ptr + batch_offset + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------
# Entry point used by the evaluator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]        # [hidden_dim, dim]
    left = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 5) TriMul cubic update via Triton kernel
    # --------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension to the second axis and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_view = left_perm.view(BH, N, N)
    right_view = right_perm.view(BH, N, N)

    out_view = torch.empty_like(left_view)  # same dtype & device

    # Block sizes – chosen to fit typical GPU register budgets
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        BH                         # batch*head dimension
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_view,
        right_view,
        out_view,
        N,
        left_view.stride(0), left_view.stride(1), left_view.stride(2),
        right_view.stride(0), right_view.stride(1), right_view.stride(2),
        out_view.stride(0), out_view.stride(1), out_view.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------
    # 7) Apply output gate and final linear projection
    # --------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"]   # [dim, hidden_dim]
    out = F.linear(out, to_out_w)        # bias is False in the reference

    return out
