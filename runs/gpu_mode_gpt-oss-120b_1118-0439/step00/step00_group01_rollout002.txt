@triton.jit
def triton_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N, hidden_heads,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,   # note B is transposed: we need to read B rows as j index
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # a_ptr points to start of A flattened.
    # b_ptr points to start of B.
    # For each head h, compute C[h, i, j] = sum_k A[h, i, k] * B[h, j, k]
    # we treat B transposed in j,k order.
    head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute offsets within head blocks
    a_ptrs = a_ptr + head * stride_ah + offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + head * stride_bh + offs_j[:, None] * stride_bj + tl.arange(0, BLOCK_K)[None, :] * stride_bk
    # Note: B is accessed as B[h, j, k] (same as original B shape)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (tl.arange(0, BLOCK_K)[None,:] + k*BLOCK_K < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & (tl.arange(0, BLOCK_K)[None,:] + k*BLOCK_K < N), other=0.0)
        # multiply-add across k block
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store result
    c_ptrs = c_ptr + head * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))
```

Actually A layout: dimension order (head, i, k). So offset = head * stride_ah + i * stride_ai + k * stride_ak. For contiguous layout: stride_ak = 1 (contiguous innermost dimension). stride_ai = N (since moving to next row moves across columns). stride_ah = N*N.

C shape same as A: (head, i, j). strides: stride_cj = 1, stride_ci = N, stride_ch = N*N.

def custom_kernel(data):
    x, mask, weights, config = data
    # x shape [B,N,N,dim]
    # mask shape [B,N,N] maybe None (if not provided)?
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear proj left and right (no bias)
    # x_norm shape [B,N,N,dim]; weight shapes: left_proj.weight shape [hidden_dim, dim] per Linear: out_features x in_features.
    # left_proj = x_norm @ left_proj_weight.T
    # So left = torch.einsum('bnsc,hd->bnsh', x_norm, left_proj_weight) -> shape [B,N,N,hidden_dim]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.matmul(x_norm, left_proj_weight.t())  # shape [B,N,N,hidden_dim]
    right = torch.matmul(x_norm, right_proj_weight.t())
    
    # apply mask if not nomask (the config may indicate mask)
    if mask is not None:
        mask = mask.unsqueeze(-1).float()
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.matmul(x_norm, left_gate_weight.t()))
    right_gate = torch.sigmoid(torch.matmul(x_norm, right_gate_weight.t()))
    out_gate = torch.sigmoid(torch.matmul(x_norm, out_gate_weight.t()))

    left = left * left_gate
    right = right * right_gate
    
    # TriMul using triton kernel
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    left_flat = left_perm.view(total_heads, N, N)
    right_flat = right_perm.view(total_heads, N, N)
    out_flat = torch.empty_like(left_flat)  # shape [total_heads,N,N]

    # call triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_matmul_outgoing_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        total_heads,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # out shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate
    # final linear
    to_out_weight = weights["to_out.weight"]  # shape [dim, hidden_dim]
    out = torch.matmul(out, to_out_weight.t())  # shape [B,N,N,dim]

    return out
```

import triton
import triton.language as tl

@triton.jit
def triton_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N, total_heads,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets within N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute pointer bases for A and B
    a_ptrs = a_ptr + head * stride_ah + offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + head * stride_bh + offs_j[:, None] * stride_bj + tl.arange(0, BLOCK_K)[None, :] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < N), other=0.0)
        # dot product across k (inner dimension)
        acc += tl.dot(a, b)
        # increment pointers for next k block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store result
    c_ptrs = c_ptr + head * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj

"""
TriMul "outgoing" forward pass implemented with a fused Triton GEMM kernel.
The main computation is the einsum:
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
which is equivalent to a batch of matrix multiplications
    C_h = A_h @ B_h^T
for each head h = batch * hidden_dim.
All other operations (LayerNorms, linear projections, gates) are performed in PyTorch.
The Triton kernel computes all C_h efficiently in a single launch.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    head = tl.program_id(0)  # total_heads = B * hidden_dim
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute block coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current head (no k offset yet)
    a_base = a_ptr + head * stride_ah
    b_base = b_ptr + head * stride_bh

    # Pointers to the start of the current (i,k) and (j,k) tiles
    a_tile_ptr = a_base + offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_tile_ptr = b_base + offs_j[:, None] * stride_bj + tl.arange(0, BLOCK_K)[None, :] * stride_bk

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A and B tiles (with masking for out‑of‑bounds elements)
        a = tl.load(
            a_tile_ptr + k_offset * stride_ak,
            mask=(offs_i[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k_offset < N),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + k_offset * stride_bk,
            mask=(offs_j[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k_offset < N),
            other=0.0,
        )
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write the result back to C, applying the same bound checks
    c_base = c_ptr + head * stride_ch
    c_tile_ptr = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(
        c_tile_ptr,
        acc,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, same device as input).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    eps = 1e-5

    # -------------------------------------------------------------------------
    # 1. Initial LayerNorm
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (config["dim"],), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    # shape: [B, N, N, hidden_dim]
    left = torch.matmul(x, left_proj_w.t())
    right = torch.matmul(x, right_proj_w.t())

    # -------------------------------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # mask shape: [B, N, N] -> broadcast to last dim
        mask = mask.unsqueeze(-1).to(dtype)
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.matmul(x, left_gate_w.t()))
    right_gate = torch.sigmoid(torch.matmul(x, right_gate_w.t()))
    out_gate = torch.sigmoid(torch.matmul(x, out_gate_w.t()))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core TriMul: out = left @ right^T for each head = batch * hidden_dim
    # -------------------------------------------------------------------------
    B, N, _, hidden_dim = left.shape
    total_heads = B * hidden_dim

    # Move the hidden dimension to the batch axis so we can treat the problem
    # as a regular batched GEMM.
    left_ = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_.view(total_heads, N, N)
    right_flat = right_.view(total_heads, N, N)

    out_flat = torch.empty_like(left_flat)  # [total_heads, N, N]

    # Triton block sizes – chosen to fit well into shared memory for N up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.matmul(out, to_out_w.t())

    return out

    Arguments

    b_base = b_ptr + head * stride_bh
