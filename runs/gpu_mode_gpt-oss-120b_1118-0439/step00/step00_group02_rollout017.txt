    out = torch.empty_like(left)
    # compute strides

    stride_right = right.stride()

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)
    # compute batch and hidden id
    b = pid_hb // H
    h = pid_hb % H

    # offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask dims
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # masks for left and right loads
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_j[:, None] & mask_k[None, :]

        # pointer arithmetic
        a_ptrs = left_ptr + b * stride_lb + i[:, None] * stride_li + k[None, :] * stride_lk + h * stride_lh
        b_ptrs = right_ptr + b * stride_rb + j[:, None] * stride_ri + k[None, :] * stride_rk + h * stride_rh

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # accumulate dot product over k
        acc += tl.dot(a, b, trans_b=True)

    mask_out = mask_i[:, None] & mask_j[None, :]
    out_ptrs = out_ptr + b * stride_ob + i[:, None] * stride_oi + j[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_out)

    Performs layer‑norm, linear projections with gates, the core

    """

    hidden_dim = config["hidden_dim"]

        bias=weights["to_out_norm.bias"],
    )
    #--- Output gate
    out = out * out_gate
    #--- Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy O(N³·C_z) computation
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is performed by a custom Triton kernel that computes a batched
matrix‑multiply for each channel c (hidden dimension). The rest of the
module – LayerNorm, linear projections, sigmoid gates and the final
projection – are executed with standard PyTorch ops.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr,                # ptr to left tensor      [B,N,N,Cz]
    right_ptr,               # ptr to right tensor     [B,N,N,Cz]
    out_ptr,                 # ptr to output tensor    [B,N,N,Cz]
    B, N, Cz,                # runtime dimensions
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides (batch,i,k,c)
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k reduction
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_hb = tl.program_id(2)              # combined batch‑channel tile

    # derive batch index b and channel index c
    b = pid_hb // Cz
    c = pid_hb % Cz

    # i and j coordinates for this tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for tail handling
    mask_i = i < N
    mask_j = j < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Build masks for the two loads
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_j[:, None] & mask_k[None, :]

        # Pointer arithmetic
        a_ptrs = (left_ptr + b * stride_lb
                  + i[:, None] * stride_li
                  + k[None, :] * stride_lk
                  + c * stride_lh)
        b_ptrs = (right_ptr + b * stride_rb
                  + j[:, None] * stride_ri
                  + k[None, :] * stride_rk
                  + c * stride_rh)

        # Load tiles (out‑of‑bounds elements are zeroed)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate dot product over the k‑dimension.
        # right tile is transposed on‑the‑fly (trans_b=True).
        acc += tl.dot(a, b, trans_b=True)

    # Write the result tile
    mask_out = mask_i[:, None] & mask_j[None, :]
    out_ptrs = (out_ptr + b * stride_ob
                + i[:, None] * stride_oi
                + j[None, :] * stride_oj
                + c * stride_oh)
    tl.store(out_ptrs, acc, mask=mask_out)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched per‑channel matrix multiplication implementing the
    TriMul core operation.
    """
    assert left.shape == right.shape
    B, N, _, C = left.shape
    out = torch.empty_like(left)

    # strides of the contiguous tensors
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * C,
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, C,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            input_tensor : [B, N, N, dim]
            mask_tensor  : [B, N, N] (may be None)
            weights      : dict with all linear / layer‑norm parameters
            config       : dict containing at least 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over channels)
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul (batched per‑channel GEMM via Triton)
    out = _tri_mul_triton(left, right)

    # -----------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 7. Output gate
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8. Final linear projection to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
