"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The core O(N³) operation:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

Implemented as a batched matrix multiplication:
    left_perm (B, D, N, N) @ right_perm.transpose(-2, -1) (B, D, N, N)
The kernel works on the flattened batch dimension (B*D) and uses
BLOCK_M/N/K tiling (64×64×64) with accumulation in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers to A, B, C tensors
    M, N, K,                        # matrix dimensions (all = seq_len)
    stride_am, stride_ak,           # A: row stride, col stride
    stride_bk, stride_bn,           # B (transposed): row (k) stride, col (j) stride
    stride_cm, stride_cn,           # C: row stride, col stride
    stride_batch,                   # offset between consecutive (batch,head) matrices (= N*N)
    BLOCK_M: tl.constexpr,          # tile size in M dimension
    BLOCK_N: tl.constexpr,          # tile size in N dimension
    BLOCK_K: tl.constexpr,          # tile size in K dimension
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # flattened batch*head index

    # --------------------------------------------------------------- #
    # Compute block start offsets
    # --------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointer for this (b,head) pair
    a_base = a_ptr + pid_b * stride_batch
    b_base = b_ptr + pid_b * stride_batch
    c_base = c_ptr + pid_b * stride_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over K dimension in BLOCK_K sized chunks
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K block offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (transposed view): shape [BLOCK_K, BLOCK_N]
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------- #
    # Write back result
    # --------------------------------------------------------------- #
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------------- #
# Python entry point expected by the test harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of weight tensors (see function body)
            - config: dict containing at least 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------- #
    # LayerNorm on the input
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # Optional mask (skip if config says no mask or mask is None)
    # --------------------------------------------------------------------- #
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # Gating mechanisms
    # --------------------------------------------------------------------- #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # Prepare tensors for the Triton kernel:
    #   shape [B, hidden_dim, N, N] (contiguous) -> treat each (b,head) as a
    #   separate matrix multiplication.
    # --------------------------------------------------------------------- #
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)               # [B, D, N, N]

    B, D, N, _ = left_perm.shape

    # Strides (int64 values)
    stride_am = left_perm.stride(2)   # row stride (N)
    stride_ak = left_perm.stride(3)   # col stride (1)

    stride_bk = right_perm.stride(3)  # k‑row stride (1)
    stride_bn = right_perm.stride(2)  # j‑col stride (N)

    stride_cm = out_perm.stride(2)    # row stride (N)
    stride_cn = out_perm.stride(3)    # col stride (1)

    stride_batch = N * N               # distance between successive (b,head) matrices

    # Tile sizes – chosen to fit well on modern GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * D,
    )
    # Launch the kernel
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # Reshape back to original layout and apply the remaining layers
    # --------------------------------------------------------------------- #
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # LayerNorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # bias is None for this layer

    return out
