@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    eps,
    M, C,
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr
):
    pid = tl.program_id(axis=0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)
    mask_m = offs_m < M
    mask_c = offs_c < C
    x_ptrs = x_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc)
    x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_c[None, :], other=0.0)
    # Compute mean
    # sum across C dimension
    sum_x = tl.sum(x, axis=1)  # shape (BLOCK_M)
    mean = sum_x / C
    # compute variance
    diff = x - mean[:, None]
    sum_sq = tl.sum(diff * diff, axis=1)
    var = sum_sq / C
    inv_std = tl.rsqrt(var + eps) # rsqrt faster
    # normalize
    y = (x - mean[:, None]) * inv_std[:, None]
    # apply weight/bias
    if weight_ptr != 0:
        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
        y = y * w[None, :]
    if bias_ptr != 0:
        b = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0)
        y = y + b[None, :]
    # store
    y_ptrs = y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc)
    tl.store(y_ptrs, y, mask=mask_m[:, None] & mask_c[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # ensure contiguous
    input_tensor = input_tensor.contiguous()
    # optionally apply mask fill zeros through broadcasting

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5

    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias, eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # linear: shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask if present
    if not config.get("nomask", False):
        # mask shape [B,N,N]; unsqueeze last dim
        mask = mask_tensor.unsqueeze(-1)  # broadcast
        left = left * mask
        right = right * mask

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # TriMul (einsum)
    # Reshape to (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    # left: B,N,N,H; want left_perm = (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).reshape(B*hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).reshape(B*hidden_dim, N, N)  # swapped j,k
    out_perm = torch.bmm(left_perm, right_perm)  # shape (B*H,N,N)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps)

    # multiply with out_gate
    out_norm = out_norm * out_gate

    # final linear to output dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return output

"""
custom_kernel for the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]) using a Triton
   kernel that computes the mean/variance per (b,i,j) position and
   applies the learned weight and bias.
2. Linear projections:
   * left  = x·W_left   (no bias)
   * right = x·W_right
   * left_gate  = σ(x·W_left_gate)
   * right_gate = σ(x·W_right_gate)
   * out_gate   = σ(x·W_out_gate)
3. Optional element‑wise mask (broadcasted on the channel dimension).
4. Gating:
   left  = left  * left_gate
   right = right * right_gate
5. Tri‑multiplicative update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix multiplication:
   `out = (left.permute(0,3,1,2)  @  right.permute(0,3,2,1))`
   where the hidden dimension is folded into the batch size.
6. Layer‑norm on the hidden dimension (`out`) using the same Triton kernel.
7. Multiply with `out_gate`, then a final linear projection to the original
   channel dimension.
All heavy element‑wise operations (layer‑norm) are fused in Triton,
while the O(N³·H) reduction is performed with an efficient batched GEMM
(`torch.bmm`).

The function signature follows the problem statement:
    custom_kernel(data) -> torch.Tensor
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for LayerNorm over the last dimension.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr,                # input / output pointers
    weight_ptr, bias_ptr,        # scale and bias (1‑D, size C)
    eps,                         # epsilon for numerical stability
    M, C,                        # M = B*N*N, C = channel dimension
    stride_xm, stride_xc,        # strides of X (M, C)
    stride_ym, stride_yc,        # strides of Y (M, C)
    BLOCK_M: tl.constexpr,       # rows per program
    BLOCK_C: tl.constexpr,       # channels per program (set >= max C)
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_c = tl.arange(0, BLOCK_C)                         # [BLOCK_C]

    mask_m = offs_m < M
    mask_c = offs_c < C

    # Load X (shape BLOCK_M x BLOCK_C) with proper masking
    x_ptrs = X_ptr + (offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc)
    X = tl.load(x_ptrs,
                mask=mask_m[:, None] & mask_c[None, :],
                other=0.0)  # zero‑pad out‑of‑bounds channels

    # ----- mean -----
    sum_x = tl.sum(X, axis=1)               # sum over C
    mean = sum_x / C

    # ----- variance (masked) -----
    diff = X - mean[:, None]                # broadcast mean
    diff_sq = diff * diff
    mask_c_f = tl.where(mask_c, 1.0, 0.0)   # 1.0 for valid channels, else 0
    sum_sq = tl.sum(diff_sq * mask_c_f[None, :], axis=1)
    var = sum_sq / C

    inv_std = tl.rsqrt(var + eps)           # 1 / sqrt(var + eps)

    # ----- normalize -----
    Y = diff * inv_std[:, None]

    # ----- affine transformation -----
    w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)   # weight
    b = tl.load(bias_ptr + offs_c,   mask=mask_c, other=0.0)   # bias
    Y = Y * w[None, :] + b[None, :]

    # ----- store -----
    y_ptrs = Y_ptr + (offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc)
    tl.store(y_ptrs,
             Y,
             mask=mask_m[:, None] & mask_c[None, :])

# ----------------------------------------------------------------------
# Python wrapper around the Triton LayerNorm kernel
# ----------------------------------------------------------------------
def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 64,
                     BLOCK_C: int = 384) -> torch.Tensor:
    """
    LayerNorm over the last dimension using Triton.
    x:    [B, N, N, C] (contiguous float32)
    weight, bias: 1‑D tensors of size C
    Returns tensor of same shape.
    """
    assert x.is_contiguous()
    B, N, N2, C = x.shape
    assert N == N2
    M = B * N * N2

    # flatten leading dimensions for the kernel: shape (M, C)
    x_flat = x.view(M, C)
    out_flat = torch.empty_like(x_flat)

    # strides for the flattened view
    stride_xm = x_flat.stride(0)          # equals C
    stride_xc = x_flat.stride(1)          # 1
    stride_ym = out_flat.stride(0)
    stride_yc = out_flat.stride(1)

    # ensure weight/bias are 1‑D contiguous
    weight = weight.contiguous()
    bias = bias.contiguous()

    # launch kernel (1‑D grid)
    grid = lambda META: (triton.cdiv(M, META['BLOCK_M']),)

    layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        eps,
        M,
        C,
        stride_xm,
        stride_xc,
        stride_ym,
        stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
    )
    return out_flat.view(B, N, N2, C)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Basic configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    device = input_tensor.device
    input_tensor = input_tensor.contiguous()
    if mask_tensor is not None:
        mask_tensor = mask_tensor.contiguous()

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over `dim`)
    # ------------------------------------------------------------------
    x_norm = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        BLOCK_M=64,
        BLOCK_C=384,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections and gates
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # optional mask (broadcast on channel dim)
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).type_as(left)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gating values
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Tri‑multiplicative update (einsum equivalent)
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to batch dimension = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H,N,N)
    right_perm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H,N,N), swapped axes
    out_perm = torch.bmm(left_perm, right_perm)                         # (B*H,N,N)
    # reshape back to [B,N,N,hidden_dim]
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)        # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        BLOCK_M=64,
        BLOCK_C=384,
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Gated output and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    output = F.linear(out_norm, weights["to_out.weight"])  # [B,N,N,dim]
    return output
