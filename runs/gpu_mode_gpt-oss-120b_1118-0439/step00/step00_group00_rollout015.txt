@triton.jit
def layernorm_gate_kernel(
    out_ptr, out_gate_ptr, out_norm_weight_ptr, out_norm_bias_ptr,
    B, N, hidden, eps,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_weight_h, stride_bias_h,
    BLOCK_H: tl.constexpr
):
   # Compute global index:
   b = tl.program_id(0)
   i = tl.program_id(1)
   j = tl.program_id(2)

   # Compute base pointer for out
   out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j

   gate_ptrs = out_gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j

   # accumulate sum and sumsq across hidden dim
   sum = tl.float32(0.0)
   sumsq = tl.float32(0.0)
   # loop over hidden dimension
   for offset_h in range(0, hidden, BLOCK_H):
        cur_h = offset_h + tl.arange(0, BLOCK_H)
        mask = cur_h < hidden
        out_vals = tl.load(out_ptrs + cur_h*stride_out_h, mask=mask, other=0.0).to(tl.float32)
        sum += tl.sum(out_vals, axis=0)
        sumsq += tl.sum(out_vals * out_vals, axis=0)
   mean = sum / hidden
   var = sumsq / hidden - mean * mean
   rstd = 1.0 / tl.sqrt(var + eps)

   # Now compute normalized values and apply weight & bias and out_gate
   for offset_h in range(0, hidden, BLOCK_H):
        cur_h = offset_h + tl.arange(0, BLOCK_H)
        mask = cur_h < hidden
        out_val = tl.load(out_ptrs + cur_h*stride_out_h, mask=mask, other=0.0).to(tl.float32)
        gate_val = tl.load(gate_ptrs + cur_h*stride_gate_h, mask=mask, other=0.0).to(tl.float32)
        # weight,bias
        w = tl.load(out_norm_weight_ptr + cur_h*stride_weight_h, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(out_norm_bias_ptr + cur_h*stride_bias_h, mask=mask, other=0.0).to(tl.float32)
        normed = (out_val - mean) * rstd
        normed = normed * w + b
        out_scaled = normed * gate_val
        tl.store(out_ptrs + cur_h*stride_out_h, out_scaled.to(tl.float16) if out_ptrs.dtype==... else out_scaled)

@triton.jit
def layernorm_gate_kernel(out_ptr, out_gate_ptr, weight_ptr, bias_ptr,
                         B, N, hidden, eps,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                         stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                         stride_weight_h, stride_bias_h,
                         BLOCK_H: tl.constexpr):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # base pointers
    out_ptr_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j
    gate_ptr_base = out_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # compute mean & var
    acc = tl.float32(0.0)
    acc2 = tl.float32(0.0)
    for offset in range(0, hidden, BLOCK_H):
        cur = offset + tl.arange(0, BLOCK_H)
        mask = cur < hidden
        o = tl.load(out_ptr_base + cur * stride_out_h, mask=mask, other=0.0).to(tl.float32)
        acc += tl.sum(o, axis=0)
        acc2 += tl.sum(o * o, axis=0)

    mean = acc / hidden
    var = acc2 / hidden - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # compute output
    for offset in range(0, hidden, BLOCK_H):
        cur = offset + tl.arange(0, BLOCK_H)
        mask = cur < hidden
        o = tl.load(out_ptr_base + cur * stride_out_h, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(gate_ptr_base + cur * stride_gate_h, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur * stride_weight_h, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr + cur * stride_bias_h, mask=mask, other=0.0).to(tl.float32)

        norm = (o - mean) * rstd
        norm = norm * w + b_
        out = norm * g
        tl.store(out_ptr_base + cur * stride_out_h, out, mask=mask)

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data

    # config:
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("epsilon", 1e-5)

    # Move all to same device
    device = input_tensor.device

    # Ensure contiguous
    x = input_tensor.contiguous()
    if mask is not None:
        mask = mask.contiguous()

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input x
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        # mask shape [B, N, N]
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_
    # else no mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batch GEMM
    # Permute to (B, hidden, i, k) and (B, hidden, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transposed k<->j

    B, hidden, N, _ = left_perm.shape
    left_flat = left_perm.reshape(B * hidden, N, N)
    right_flat = right_perm.reshape(B * hidden, N, N)

    # Compute matmul
    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*hidden, N, N)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # B,N,N,hidden

    # Apply to_out_norm and gating via Triton kernel
    out_normed = out.clone()

    # Launch kernel
    # Compute strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_normed.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    # weight and bias strides
    stride_w_h = to_out_norm_weight.stride(0)  # weight is 1D
    stride_b_h = to_out_norm_bias.stride(0)

    # grid: (B, N, N)
    grid = (B, N, N)
    layernorm_gate_kernel[grid](
        out_normed,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        B, N, hidden_dim, eps,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        stride_w_h, stride_b_h,
        BLOCK_H=64
    )

    # Final linear
    out_final = torch.nn.functional.linear(out_normed, to_out_weight)  # shape B,N,N,dim

    return out_final

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]) using the provided
   norm.weight / norm.bias.
2. Compute three linear projections on the normalized tensor:
   left, right (both → hidden_dim) and three gating projections
   left_gate, right_gate, out_gate (all → hidden_dim).
3. Apply the optional pairwise mask to left/right, then multiply by
   the respective gates.
4. Perform the “outgoing” triangular multiplication:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is realised as a batched matrix multiplication
   (left × rightᵀ) per hidden dimension using a single torch.bmm
   on tensors of shape (B*hidden, N, N).
5. Apply a second Layer‑norm (to_out_norm) on the hidden dimension and
   fuse the multiplication by out_gate.  This step is executed by a
   custom Triton kernel that:
   * computes mean/variance over the hidden dimension,
   * normalises, scales by to_out_norm.weight/bias,
   * multiplies element‑wise with out_gate.
6. Final linear projection back to the original channel dimension
   using to_out.weight.

Only the second Layer‑norm + gating step is performed in Triton – the
rest uses PyTorch for simplicity and readability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_gate_kernel(
    out_ptr,                     # *float32   (output tensor)
    gate_ptr,                    # *float32   (out_gate)
    weight_ptr,                  # *float32   (to_out_norm.weight)
    bias_ptr,                    # *float32   (to_out_norm.bias)
    B, N, hidden, eps,           # i32 scalars
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,   # i64 strides
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_w_h, stride_b_h,                                 # i64 (always 1)
    BLOCK_H: tl.constexpr,                                   # compile‑time tile size
):
    """Layer‑norm over the hidden dimension followed by gating.

    Each program instance handles one (b, i, j) position and loops over
    the hidden dimension in chunks of BLOCK_H.
    """
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Base pointers for this (b,i,j) entry
    base_out = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j
    base_gate = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # ----- first pass: compute mean and variance -----
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, hidden, BLOCK_H):
        cur = off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask = cur < hidden
        val = tl.load(base_out + cur * stride_out_h, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(val)
        sum_sq  += tl.sum(val * val)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalise, apply weight/bias and gate -----
    for off in range(0, hidden, BLOCK_H):
        cur = off + tl.arange(0, BLOCK_H)
        mask = cur < hidden

        o = tl.load(base_out + cur * stride_out_h, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(base_gate + cur * stride_gate_h, mask=mask, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + cur * stride_w_h, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr   + cur * stride_b_h, mask=mask, other=0.0).to(tl.float32)

        norm = (o - mean) * rstd
        norm = norm * w + b_
        out  = norm * g

        tl.store(base_out + cur * stride_out_h, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("epsilon", 1e-5)

    # ------------------------------------------------------------------
    # Extract weight tensors (all are expected on the same device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    x_norm = F.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)    # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Outgoing triangular multiplication via batched GEMM
    # ------------------------------------------------------------------
    # Permute to shape (B, hidden, N, N) and apply the required transpose
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    left_flat  = left_perm.reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.reshape(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: (i,k) @ (k,j)ᵀ  → (i,j)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + gating (implemented in Triton)
    # ------------------------------------------------------------------
    out_normed = out.clone()   # allocate output buffer

    # Strides for the tensors
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_normed.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    stride_w_h = to_out_norm_weight.stride(0)   # always 1
    stride_b_h = to_out_norm_bias.stride(0)     # always 1

    # Launch the Triton kernel: one program per (b, i, j)
    grid = (B, N, N)
    layernorm_gate_kernel[grid](
        out_normed,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        B, N, hidden_dim, eps,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        stride_w_h, stride_b_h,
        BLOCK_H=64,
    )

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_normed, to_out_weight)   # shape [B, N, N, dim]

    return out_final
