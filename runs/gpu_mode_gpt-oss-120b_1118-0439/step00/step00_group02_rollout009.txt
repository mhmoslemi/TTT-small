def trmul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bc = tl.program_id(0)   # batch * hiddenDim index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # compute batch idx and hidden idx
    b = pid_bc // H
    h = pid_bc % H

    # compute start offsets for i and j
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # define bounds
    i_max = tl.minimum(i_offset + BLOCK_M, N)
    j_max = tl.minimum(j_offset + BLOCK_N, N)

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        # compute actual size for block K
        cur_k = tl.minimum(k_offset + BLOCK_K, N) - k_offset

        # load left tile: shape (BLOCK_M, cur_k)
        left_ptrs = left_ptr + (b * stride_left_b + i_offset * stride_left_i + k_offset * stride_left_k + h * stride_left_h)
        # compute offsets: we need grid of offsets rows and cols
        # left tile rows: i_range = tl.arange(0, BLOCK_M) if i < i_max else disabled
        # left tile columns: k_range = tl.arange(0, cur_k)
        i_range = tl.arange(0, BLOCK_M)
        k_range = tl.arange(0, cur_k)
        # maybe we should create base pointer for left for each i
        left_offsets = i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k

        # load right tile: shape (BLOCK_N, cur_k)
        # right indices: (b, j, k, h)
        right_ptrs = right_ptr + (b * stride_right_b + j_offset * stride_right_i + k_offset * stride_right_k + h * stride_right_h)

        right_offsets = j_range[:, None] * stride_right_i + k_range[None, :] * stride_right_k

        # compute contribution
        acc += tl.dot(left_tile, right_tile, trans_b=True)  # left (M,K) dot right (N,K).T -> (M,N)

    # store result
    out_ptrs = out_ptr + (b * stride_out_b + i_offset * stride_out_i + j_offset * stride_out_k + h * stride_out_h)

    mask_j = j_range < (j_max - j_offset)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )  # shape B,N,N,dim

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if exist and not nomask
    if not config.get("nomask", False) and mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # B,N,N,1
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # Triton kernel launch
    B, N, _, _ = left.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    grid = (
        (B * hidden_dim),                                   # program id0: batch*channel
        ( (N + BLOCK_M - 1) // BLOCK_M ),                  # program id1: row tile
        ( (N + BLOCK_N - 1) // BLOCK_N ),                  # program id2: col tile
    )

    # compute strides
    stride_left = left.stride()  # (B_stride, i_stride, j_stride, h_stride)
    stride_right = right.stride()
    stride_out = out.stride()

    trmul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply out_gate gating
    out_gated = out_norm * out_gate

    # Final linear projection to dim
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

    return out_final

"""
TriMul forward implementation with a Triton kernel.

The kernel computes the core O(N³·Cₕ) operation:
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

for each batch `b` and hidden channel `h`.  The computation is tiled over the
(i, j, k) axes with BLOCK_M × BLOCK_N × BLOCK_K tiles.  All LayerNorms,
linear projections and gating are performed in PyTorch; the heavy
pair‑wise reduction is off‑loaded to the Triton kernel to demonstrate a
fusion that PyTorch’s JIT cannot achieve alone.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled pairwise reduction (the einsum in TriMul)
# ----------------------------------------------------------------------
@triton.jit
def trmul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h].
    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_bc = tl.program_id(0)   # batch * hidden-channel index
    pid_m  = tl.program_id(1)   # tile row (i)
    pid_n  = tl.program_id(2)   # tile column (j)

    # ----------- decode batch and hidden indices ----------
    b = pid_bc // H
    h = pid_bc % H

    # ----------- tile origins ----------
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # range vectors inside a tile
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # validity masks for the i and j dimensions
    i_valid = i_base + i_range < N
    j_valid = j_base + j_range < N

    # base addresses (b and h are fixed for the whole tile)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 (input may be FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- loop over k dimension ----------
    for k_base in range(0, N, BLOCK_K):
        # mask for the k dimension of the current tile
        k_valid = (k_base + k_range) < N

        # ---- load left tile (i, k) ----
        left_offs = (i_base + i_range)[:, None] * stride_left_i + \
                    (k_base + k_range)[None, :] * stride_left_k
        left = tl.load(left_base + left_offs,
                       mask = i_valid[:, None] & k_valid[None, :],
                       other = 0.0).to(tl.float32)

        # ---- load right tile (j, k) ----
        right_offs = (j_base + j_range)[:, None] * stride_right_i + \
                     (k_base + k_range)[None, :] * stride_right_k
        right = tl.load(right_base + right_offs,
                        mask = j_valid[:, None] & k_valid[None, :],
                        other = 0.0).to(tl.float32)

        # ---- accumulate dot product (left @ rightᵀ) ----
        # left: (M, K), right: (N, K) -> we need (M, N)
        acc += tl.dot(left, right, trans_b=True)

    # ----------- write the result ----------
    out_offs = (i_base + i_range)[:, None] * stride_out_i + \
               (j_base + j_range)[None, :] * stride_out_j
    tl.store(out_base + out_offs,
             acc,
             mask = i_valid[:, None] & j_valid[None, :])

# ----------------------------------------------------------------------
# Host function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).

    Arguments:
        data: tuple
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask_tensor : torch.Tensor of shape [B, N, N] (may be None)
            - weights     : dict of model parameters
            - config      : dict containing at least {"dim", "hidden_dim", "nomask"}

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract configuration and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm before projections
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projection weights (bias=False)
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # Gate linear weights
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    # Output LayerNorm and final linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape: B,N,N)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)                 # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction using Triton
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # (B,N,N,hidden_dim)

    # Tile sizes – chosen to fit typical B200 shared memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                      # program id 0: batch * hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,        # program id 1: i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # program id 2: j‑tiles
    )

    # Strides for each dimension (contiguous layout = (B,N,N,hidden_dim))
    sL = left.stride()   # (b,i,k,h)
    sR = right.stride()
    sO = out.stride()

    trmul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        sL[0], sL[1], sL[2], sL[3],
        sR[0], sR[1], sR[2], sR[3],
        sO[0], sO[1], sO[2], sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # reasonable default for B200
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_gated = out_norm * out_gate

    # Final linear mapping hidden_dim -> dim
    output = torch.nn.functional.linear(out_gated, to_out_weight)

    return output
