import torch
import triton
import triton.language as tl

@triton.jit
def outgoing_matmul_kernel(
    A, B, C,
    BATCH, N, D,
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bj, stride_Bk, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_idx = pid // D
    d_idx = pid % D

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers for this batch and channel
    a_ptr = A + batch_idx * stride_Ab + d_idx * stride_Ad
    b_ptr = B + batch_idx * stride_Bb + d_idx * stride_Bd
    c_ptr = C + batch_idx * stride_Cb + d_idx * stride_Cd

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_K = tl.minimum(N - k, BLOCK_K)
        # compute offsets
        # A offsets: (i, k)
        a_offsets = a_ptr + (i0 + tl.arange(0, BLOCK_M))[:, None] * stride_Ai + (k + tl.arange(0, BLOCK_K))[None, :] * stride_Ak
        # B offsets: (j, k)
        b_offsets = b_ptr + (j0 + tl.arange(0, BLOCK_N))[:, None] * stride_Bj + (k + tl.arange(0, BLOCK_K))[None, :] * stride_Bk

        # masks
        mask_a = (i0 + tl.arange(0, BLOCK_M)[:, None] < N) & (k + tl.arange(0, BLOCK_K)[None, :] < N)
        mask_b = (j0 + tl.arange(0, BLOCK_N)[:, None] < N) & (k + tl.arange(0, BLOCK_K)[None, :] < N)

        a = tl.load(a_offsets, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_offsets, mask=mask_b, other=0.0).to(tl.float32)

    tl.store(c_offsets, acc, mask=mask_c)

"""
TriMul (outgoing) forward kernel.

The module performs:
    1. LayerNorm on the input tensor [B, N, N, C].
    2. Linear projections to a hidden dimension H (no bias).
    3. Optional masking of the projected tensors.
    4. Sigmoid gates applied to the projections.
    5. Core update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       (i.e. a batched matrix multiply left @ rightᵀ for every hidden
       channel).  This O(N³·H) step is implemented in Triton.
    6. LayerNorm on the hidden tensor, gated by out_gate, and a final
       linear projection back to C.

All heavy arithmetic is in float32; the Triton kernel works on contiguous
float32 tensors and tiles the (i, j) plane with BLOCK_M×BLOCK_N tiles while
loop‑ing over the reduction dimension k in BLOCK_K steps.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def outgoing_matmul_kernel(
    A, B, C,                     # pointers to left, right, output
    BATCH, N, H,                 # batch size, seq_len, hidden dim
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bj, stride_Bk, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h] for a single hidden
    channel h.  The kernel processes one (batch, hidden) pair per program
    (grid dim 0) and tiles the (i, j) plane using grid dims 1 and 2.
    """
    pid = tl.program_id(0)               # fused index over batch * hidden
    batch_idx = pid // H                 # batch index
    h_idx = pid % H                      # hidden channel index

    pid_i = tl.program_id(1)             # tile over i (rows)
    pid_j = tl.program_id(2)             # tile over j (cols)

    i0 = pid_i * BLOCK_M                 # start row
    j0 = pid_j * BLOCK_N                 # start col

    # Base pointers for this (batch, hidden) pair
    a_base = A + batch_idx * stride_Ab + h_idx * stride_Ad
    b_base = B + batch_idx * stride_Bb + h_idx * stride_Bd
    c_base = C + batch_idx * stride_Cb + h_idx * stride_Cd

    # Accumulator for the BLOCK_M × BLOCK_N tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        # Offsets for the current tile
        i_off = i0 + tl.arange(0, BLOCK_M)[:, None]      # (M,1)
        j_off = j0 + tl.arange(0, BLOCK_N)[:, None]      # (N,1)
        k_off = k + tl.arange(0, BLOCK_K)[None, :]       # (1,K)

        # Compute memory addresses
        a_ptr = a_base + i_off * stride_Ai + k_off * stride_Ak
        b_ptr = b_base + j_off * stride_Bj + k_off * stride_Bk

        # Masks for out‑of‑bounds loads
        mask_a = (i_off < N) & (k_off < N)
        mask_b = (j_off < N) & (k_off < N)

        # Load tiles (float32)
        a = tl.load(a_ptr, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptr, mask=mask_b, other=0.0).to(tl.float32)

        # Accumulate dot-product over the K dimension
        acc += tl.dot(a, b)   # (M,K) · (N,K)ᵀ -> (M,N)

    # Write the result back
    c_ptr = c_base + i0 * stride_Ci + j0 * stride_Cj
    mask_c = (i0 + tl.arange(0, BLOCK_M)[:, None] < N) & (j0 + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(c_ptr, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, C]
        - mask_tensor  : torch.Tensor[bs, N, N] (0/1 mask) or None
        - weights_dict : dict of model weights (see skeleton)
        - config_dict  : contains "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, C]  (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask (if present) ----------
    if mask is not None:
        # broadcast mask to hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core O(N³·H) update via Triton ----------
    B, N, _, _ = left.shape  # _ is hidden_dim
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,H]

    # Strides (int64)
    s_Ab, s_Ai, s_Ak, s_Ad = left.stride()
    s_Bb, s_Bj, s_Bk, s_Bd = right.stride()
    s_Cb, s_Ci, s_Cj, s_Cd = out.stride()

    # Tiling parameters (tuned for typical GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # batch * hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over j
    )

    outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_Ab,
        s_Ai,
        s_Ak,
        s_Ad,
        s_Bb,
        s_Bj,
        s_Bk,
        s_Bd,
        s_Cb,
        s_Ci,
        s_Cj,
        s_Cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Final LayerNorm, gating and output projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
