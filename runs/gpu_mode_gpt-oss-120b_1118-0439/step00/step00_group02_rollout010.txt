def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # 1. LayerNorm over dim (last dim) using weight & bias.
    # compute mean & var per element (B,N,N) across dim
    # We can just use torch.nn.functional.layer_norm with weight & bias (per channel)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # 2. Linear projections left/right
    # weight shapes
    left_proj_weight = weights["left_proj.weight"] # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    # compute linear without bias
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # 3. Apply mask if not None or if config["nomask"]?
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    # 4. Compute gating values
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate
    # 5. Compute core multiplication using Triton kernel
    # left and right are float32; we may cast to float16 for speed.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)  # same shape
    # compute strides
    # left.stride() returns tuple
    stride0, stride1, stride2, stride3 = left_fp16.stride()
    # Provide kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    trimul_matmul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        bs, N, hidden_dim,
        stride0, stride1, stride2, stride3,
        stride0, stride1, stride2, stride3,
        stride0, stride1, stride2, stride3,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    out = out_fp16.to(torch.float32)  # convert back
    # 6. Output layernorm (hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    # 7. Linear to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    b = batch_idx // H
    h = batch_idx % H

    # compute start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # coordinate offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # offsets for left and right
        # left[b, i, k, h]
        left_ptrs = left_ptr + b*stride_l0 + h*stride_l3 + offs_i[:, None]*stride_l1 + offs_k[None, :]*stride_l2
        # right[b, j, k, h]
        right_ptrs = right_ptr + b*stride_r0 + h*stride_r3 + offs_j[:, None]*stride_r1 + offs_k[None, :]*stride_r2

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + b*stride_o0 + h*stride_o3 + offs_i[:, None]*stride_o1 + offs_j[None, :]*stride_o2
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Triton kernel that computes, for each batch b and hidden channel h,
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    where left and right have shape [B, N, N, H].
    The kernel tiles the i and j dimensions (BLOCK_M, BLOCK_N) and iterates
    over k in steps of BLOCK_K.
    """
    pid_batch = tl.program_id(0)   # combined batch index (b * H + h)
    pid_m = tl.program_id(1)       # i‑tile
    pid_n = tl.program_id(2)       # j‑tile

    # decompose combined index into batch b and hidden dimension h
    b = pid_batch // H
    h = pid_batch % H

    # start coordinates of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)  -> indices (i,k)
        left_ptrs = left_ptr + \
            b * stride_b + h * stride_h + \
            offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: shape (BLOCK_K, BLOCK_N) -> indices (k,j)
        right_ptrs = right_ptr + \
            b * stride_b + h * stride_h + \
            offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiplication of the two tiles
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # write the result back to memory (output shape (i,j))
    out_ptrs = out_ptr + \
        b * stride_b + h * stride_h + \
        offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys “dim”, “hidden_dim”, “nomask” (bool)
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------- LayerNorm over the last dimension ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -------- Linear projections (no bias) ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Optional mask ----------
    if (mask is not None) and (not nomask):
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------- Cast to FP16 for the heavy computation ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)

    # -------- Triton kernel launch ----------
    stride_b, stride_i, stride_k, stride_h = left_fp16.stride()   # (B,N,N,H) layout
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden_dim,                     # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N        # j‑tiles
    )
    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Back to FP32
    out = out_fp16.to(torch.float32)

    # -------- Post‑processing ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Triton kernel for the core TriMul update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    The tensors have shape [B, N, N, H] (batch, seq_i, seq_j/k, hidden).
    The kernel tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N) and
    iterates over k in steps of BLOCK_K, accumulating in FP32.
    """
    pid_batch = tl.program_id(0)   # combined (b * H + h)
    pid_m = tl.program_id(1)       # i‑tile index
    pid_n = tl.program_id(2)       # j‑tile index

    # decode batch and hidden indices
    b = pid_batch // H
    h = pid_batch % H

    # start offsets for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (i, k) → shape (M, K)
        left_ptrs = left_ptr + \
            b * stride_b + h * stride_h + \
            offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: (k, j) → shape (K, N)
        # note the swap of strides: rows use stride_k, columns use stride_i
        right_ptrs = right_ptr + \
            b * stride_b + h * stride_h + \
            offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # tile‑wise matrix multiply
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # write the result (output layout is (i, j))
    out_ptrs = out_ptr + \
        b * stride_b + h * stride_h + \
        offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using Triton for the
    heavy N³ matrix‑multiplication.  The function receives:
        data = (input_tensor, mask, weights, config)

    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the N³ update
    # ------------------------------------------------------------------
    # Cast to FP16 for the compute‑intensive part
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)

    # Strides in elements for a tensor of shape [B, N, N, H]
    stride_b, stride_i, stride_k, stride_h = left_fp16.stride()

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        B * hidden_dim,                     # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N        # j‑tiles
    )
    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)   # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 6. Post‑processing: norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
