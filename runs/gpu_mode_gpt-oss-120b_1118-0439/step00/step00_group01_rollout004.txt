import triton
import triton.language as tl

@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute the actual batch and hidden dimension index
    total_bd = B * D
    b = pid_b // D
    d = pid_b % D

    # Compute starting indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Create range offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (left)
        a_ptrs = left_ptr + b*stride_l_b + d*stride_l_d + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (right transposed)
        # right index: (b, k, j, d)
        b_ptrs = right_ptr + b*stride_r_b + d*stride_r_d + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute matrix multiplication block
        acc += tl.dot(a, b)

        # Increment pointers? Not needed because we recompute each iteration.

    # Write back results
    out_ptrs = out_ptr + b*stride_o_b + d*stride_o_d + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward implementation with a Triton kernel.

The heavy part of the TriMul operation is the tensor contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

for each batch `b` and hidden channel `d`.  This is a per‑channel
batched matrix multiplication (left @ rightᵀ).  The kernel below
computes this contraction using a tiled GEMM algorithm with three
compile‑time block sizes (BLOCK_M, BLOCK_N, BLOCK_K).  All other
operations (LayerNorms, linear projections, sigmoids, masking) are
performed with native PyTorch kernels.

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns the final Tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: per‑channel batched matmul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bd = tl.program_id(2)         # combined batch * hidden channel

    # Resolve batch and hidden-channel indices
    b = pid_bd // D
    d = pid_bd % D

    # Starting indices of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets inside the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: left[b, i, k, d]
        a_ptrs = left_ptr + b * stride_l_b + d * stride_l_d \
                 + (offs_i[:, None] * stride_l_i) \
                 + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed: right[b, j, k, d] -> we need right[b, k, j, d]
        b_ptrs = right_ptr + b * stride_r_b + d * stride_r_d \
                 + (offs_k[:, None] * stride_r_k) \
                 + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # GEMM on the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write the result tile to global memory
    out_ptrs = out_ptr + b * stride_o_b + d * stride_o_d \
               + (offs_i[:, None] * stride_o_i) \
               + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module using a custom Triton kernel.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (ignored if config["nomask"] is True)
            - weights_dict: mapping from weight names to torch.Tensor
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights onto the correct device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (all in FP32)
    # ------------------------------------------------------------------
    # Linear maps: dim -> hidden_dim
    left = F.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Gates: sigmoid(Linear(dim -> hidden_dim))
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3) Optional masking (mask broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask may be bool or float; ensure correct dtype and broadcast
        mask = mask.to(dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape          # left shape = [B, N, N, hidden_dim]
    out = torch.empty_like(left)     # allocate output tensor

    # Ensure contiguous tensors for correct stride extraction
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Extract strides (in element counts)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # Tile sizes – chosen to fit well on B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (i-tiles, j-tiles, batch*hidden)
    total_bd = B * hidden_dim
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        total_bd,
    )

    # Launch kernel
    triu_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_weight)       # hidden_dim -> dim, no bias

    return out
