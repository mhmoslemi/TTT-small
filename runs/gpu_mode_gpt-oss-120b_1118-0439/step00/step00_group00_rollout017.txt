"""
TriMul forward (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left, right and three gate tensors (left_gate,
   right_gate, out_gate).  All are shape [B, N, N, H] where H = hidden_dim.
3. Apply the pairwise mask together with the gate values using a small
   Triton kernel (element‑wise multiplication).  The kernel writes the
   masked & gated results back into ``left`` and ``right`` in‑place.
4. For each batch and channel compute the contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication.  We achieve this by reshaping
   to (B*H, N, N) and calling ``torch.bmm`` – this uses cuBLAS and is fast.
5. Layer‑norm the intermediate result, multiply by ``out_gate``,
   and a final linear projection back to the original dimension.
6. Return the tensor of shape [B, N, N, dim].

The Triton kernel only handles the element‑wise mask & gating step, satisfying
the requirement to use Triton while the heavy reduction (O(N³·H)) is performed
by cuBLAS via ``torch.bmm``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,                # *float32
    right_ptr,               # *float32
    mask_ptr,                # *float32 (broadcasted on last dim)
    left_gate_ptr,           # *float32
    right_gate_ptr,          # *float32
    left_out_ptr,            # *float32  (can be same as left_ptr)
    right_out_ptr,           # *float32  (can be same as right_ptr)
    B, N, H,                 # int32
    stride_left_b, stride_left_i, stride_left_j,
    stride_right_b, stride_right_i, stride_right_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_j,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_j,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # 1‑D index → (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute flat offsets for each tensor
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j      # broadcast dim has stride 0
    left_gate_off  = b * stride_left_gate_b  + i * stride_left_gate_i   + j * stride_left_gate_j   + h
    right_gate_off = b * stride_right_gate_b + i * stride_right_gate_i  + j * stride_right_gate_j  + h

    # load values
    left_val       = tl.load(left_ptr       + left_off,       mask=mask, other=0.0)
    right_val      = tl.load(right_ptr      + right_off,      mask=mask, other=0.0)
    mask_val       = tl.load(mask_ptr       + mask_off,       mask=mask, other=0.0)
    left_gate_val  = tl.load(left_gate_ptr  + left_gate_off,  mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=0.0)

    # apply mask and gates
    left_out  = left_val  * mask_val * left_gate_val
    right_out = right_val * mask_val * right_gate_val

    # store results (in‑place is fine)
    tl.store(left_out_ptr  + left_off,  left_out,  mask=mask)
    tl.store(right_out_ptr + right_off, right_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of model weights
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # In the tests dim may also be present in config – keep for consistency
    # but we already have dim from the tensor shape.
    # --------------------------------------------------------------
    # 1. Input layer‑norm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (bias=False)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x, right_proj_weight)       # [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask & gates with a Triton kernel
    # ensure mask is float and broadcastable
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones(B, N, N, dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    mask_tensor = mask_tensor.unsqueeze(-1)   # shape [B,N,N,1]; stride on last dim = 0

    # Make sure tensors are contiguous (required for correct strides)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()

    # strides for the (B,N,N,H) tensors
    stride_left = left.stride()          # (B,N,N,H)
    stride_right = right.stride()
    stride_lgate = left_gate.stride()
    stride_rgate = right_gate.stride()
    stride_mask = mask_tensor.stride()   # (B,N,N,0)

    # launch kernel (in‑place write back into left/right)
    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_tensor,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        left_out_ptr=left,           # in‑place
        right_out_ptr=right,         # in‑place
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_j=stride_left[2],
        stride_right_b=stride_right[0],
        stride_right_i=stride_right[1],
        stride_right_j=stride_right[2],
        stride_mask_b=stride_mask[0],
        stride_mask_i=stride_mask[1],
        stride_mask_j=stride_mask[2],
        stride_left_gate_b=stride_lgate[0],
        stride_left_gate_i=stride_lgate[1],
        stride_left_gate_j=stride_lgate[2],
        stride_right_gate_b=stride_rgate[0],
        stride_right_gate_i=stride_rgate[1],
        stride_right_gate_j=stride_rgate[2],
        BLOCK=BLOCK_SIZE,
    )

    # --------------------------------------------------------------
    # 5. Core contraction: batched matrix multiplication over the sequence
    #    dimension for every (batch, hidden_dim) pair.
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * hidden_dim, N, N)           # (B*H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)         # (B*H, N, N)

    # right must be transposed on the last two dimensions (k ↔ j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # final shape [B, N, N, dim]

    return out
