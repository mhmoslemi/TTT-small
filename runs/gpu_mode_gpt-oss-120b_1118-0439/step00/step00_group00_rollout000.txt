def custom_kernel(data):
    x, mask, weights, config = data
    B,N,N_,C = x.shape
    assert N == N_
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on x: using weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use F.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (C,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).type_as(left)  # broadcast
        left = left * mask_unsq
        right = right * mask_unsq

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert left, right to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output buffer
    out_fp16 = torch.empty_like(left_fp16)  # same shape

    # Launch kernel
    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * num_blocks_m * num_blocks_n

    tri_mul_kernel[(total_programs,)](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, # adjust
    )

    # Convert out to float32
    out = out_fp16.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,

):

    # compute block indices
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode: pid -> (b, d, block_m, block_n)
    # total per channel = num_blocks_m * num_blocks_n

    d_idx = tmp % D
    tmp = tmp // D
    b_idx = tmp % B
    tmp = tmp // B

    block_n = tmp // num_blocks_m

    j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):

        mask_k = kk < N

"""
TriMul (outgoing) kernel implementation.

The core operation computes for every batch b and channel d:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where left/right are projected, gated and (optionally) masked tensors of shape
[B, N, N, hidden_dim]. This is a batched matrix multiplication L @ Rᵀ
performed independently for each hidden channel. The heavy reduction over the
sequence dimension N is fused into a Triton kernel. All surrounding
LayerNorms, linear projections and gating are carried out with PyTorch
(keeping the code simple and memory‑efficient). The kernel works in FP16
with FP32 accumulation and supports arbitrary batch size, sequence length and
hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    # strides of left: (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides of right: (b,j,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides of out: (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # tile counts
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id -> (batch, channel, block_m, block_n)
    tmp = pid
    d_idx = tmp % D
    tmp = tmp // D
    b_idx = tmp % B
    tmp = tmp // B
    block_m = tmp % num_blocks_m
    block_n = tmp // num_blocks_m

    # start indices for the tile
    i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        kk = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = kk < N

        # left: shape (BLOCK_M, BLOCK_K)   -> (i, k)
        left = tl.load(
            left_ptr
            + b_idx * stride_lb
            + i[:, None] * stride_li
            + kk[None, :] * stride_lk
            + d_idx * stride_ld,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right: shape (BLOCK_K, BLOCK_N)  -> (k, j)
        right = tl.load(
            right_ptr
            + b_idx * stride_rb
            + kk[:, None] * stride_rk
            + j[None, :] * stride_rj
            + d_idx * stride_rd,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # multiply‑accumulate
        acc += tl.dot(left, right)

    # write result (store as FP16, matching the output buffer dtype)
    out = acc.to(tl.float16)
    out_ptrs = (
        out_ptr
        + b_idx * stride_ob
        + i[:, None] * stride_oi
        + j[None, :] * stride_oj
        + d_idx * stride_od
    )
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[bs, N, N, dim]
        The processed tensor.
    """
    # ----------------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # LayerNorm on the input
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ----------------------------------------------------------------------
    # Linear projections (biasless)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ----------------------------------------------------------------------
    # Optional mask
    # ----------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # Gating (sigmoid on linear transforms of the normalized input)
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Core TriMul via Triton (L @ Rᵀ per hidden channel)
    # ----------------------------------------------------------------------
    # Cast to FP16 for the kernel (saves bandwidth, accumulation in FP32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size: one program per (batch, channel, tile_i, tile_j)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * num_blocks_m * num_blocks_n

    tri_mul_kernel[
        (total_programs,)
    ](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Back to FP32 for remaining ops
    out = out_fp16.to(torch.float32)

    # ----------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    out_scaled = out_norm * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = F.linear(out_scaled, to_out_weight)   # (B,N,N,dim)

    return output
```
