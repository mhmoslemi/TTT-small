def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    # parameters
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm on dim
    # compute mean and var across dim axis (last)
    x = input_tensor
    # shape [B,N,N,dim]
    B, N, _, D = x.shape # D==dim
    # compute mean, var
    # use torch.mean/var for speed
    x_norm = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + eps)
    x_norm = x_norm * weights['norm.weight'] + weights['norm.bias']

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

    # apply mask; ensure mask shape [B,N,N,1] broadcast to hidden
    mask_exp = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # compute gates
    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()
    left = left * left_gate
    right = right * right_gate

"""
TriMul (outgoing) forward implementation.

The heavy part of the TriMul operator is the einsum:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
where left/right have shape (B, N, N, H) (H = hidden_dim).
We implement this as a batched matrix multiplication with an
additional hidden dimension using a custom Triton kernel.  The
kernel processes a tile of the output (i,j) and a tile of the
hidden dimension, accumulating dot products over the k‑dimension.
All other operations (layer‑norms, linear projections, gates) are
performed with PyTorch for simplicity and flexibility.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Batched matmul over the k‑dimension with an extra hidden axis.

    Input tensors have shape (B, N, N, H) with layout:
        left  [b, i, k, h]
        right [b, j, k, h]   (k is summed, j is the second output axis)
    Output shape: (B, N, N, H)  →  out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_m = tl.program_id(0)          # tile in i (rows)
    pid_n = tl.program_id(1)          # tile in j (cols)
    pid_bh = tl.program_id(2)         # combine batch and hidden‑tile

    # split batch & hidden‑tile
    num_hidden_tiles = tl.cdiv(H, BLOCK_H)
    batch_id = pid_bh // num_hidden_tiles
    hidden_tile = pid_bh % num_hidden_tiles
    hid_start = hidden_tile * BLOCK_H

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = hid_start + tl.arange(0, BLOCK_H)

    # runtime bounds checks
    valid_m = offs_m < N
    valid_n = offs_n < N
    valid_h = offs_h < H

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        valid_k = offs_k < N

        # ---------- left tile (M x K x H) ----------
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + offs_m[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=valid_m[:, None, None] & valid_k[None, :, None] & valid_h[None, None, :],
            other=0.0,
        )

        # ---------- right tile (K x N x H) ----------
        # we load it transposed w.r.t. the original (j,k) layout so
        # that the dot product aligns K dimensions.
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + offs_n[None, :, None] * stride_r_j
            + offs_k[:, None, None] * stride_r_k
            + offs_h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=valid_k[:, None, None] & valid_n[None, :, None] & valid_h[None, None, :],
            other=0.0,
        )

        # accumulate dot products per hidden slice
        for hh in range(0, BLOCK_H):
            # left_slice: (M, K), right_slice: (K, N)
            left_slice = left[:, :, hh]    # (BLOCK_M, BLOCK_K)
            right_slice = right[:, :, hh]  # (BLOCK_K, BLOCK_N)
            acc[:, :, hh] += tl.dot(left_slice, right_slice)

    # ---------- write the result ----------
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + offs_m[:, None, None] * stride_o_i
        + offs_n[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=valid_m[:, None, None] & valid_n[None, :, None] & valid_h[None, None, :],
    )

                "to_out.weight"
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] – the transformed pair representation.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    # compute mean/var manually (no torch.nn.LayerNorm to stay on‑GPU)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]   # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (mask may be all‑ones)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core cubic update via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # strides in elements
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # kernel tile sizes – tuned for typical GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8  # hidden‑dimension tile

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    hidden_tiles = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (grid_m, grid_n, B * hidden_tiles)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=lsb,
        stride_l_i=lsi,
        stride_l_k=lsk,
        stride_l_h=lsh,
        stride_r_b=rsb,
        stride_r_j=rsj,
        stride_r_k=rsk,
        stride_r_h=rsh,
        stride_o_b=osb,
        stride_o_i=osi,
        stride_o_j=osj,
        stride_o_h=osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 6. LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_h) / torch.sqrt(var_h + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
