    left_ptr, right_ptr, out_ptr,
    B, N, D,

    stride_right_b, stride_right_i, stride_right_k, stride_right_d,

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)   # combined batch and d block

    d_block = pid_z % total_d_blocks

    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul forward pass (outgoing version) implemented with a Triton kernel.

The kernel computes the core "pairwise" multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where `left` and `right` are the linear projections of the normalized input,
already masked and gated.  The rest of the module (LayerNorms, gating,
final projection) is performed with PyTorch for simplicity.  The Triton kernel
tiles the i, j and hidden dimensions and processes the sequence dimension k
in blocks, using `tl.dot` for the per‑channel matrix multiplication.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # batch, sequence length, hidden dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """TriMul X = Σ_k L[i,k,:] * R[j,k,:]   (batched, per hidden channel)."""
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # combined batch & d‑block

    # split pid_z into batch index and hidden‑block index
    num_d_blocks = tl.cdiv(D, BLOCK_D)
    batch = pid_z // num_d_blocks
    d_block = pid_z % num_d_blocks

    # thread‑local coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # accumulator for the tile (M×N×D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over the K dimension in chunks
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # -------- load left[b, i, k, d] ----------
        left_ptrs = left_ptr \
            + batch * stride_left_b \
            + i[:, None, None] * stride_left_i \
            + k[None, :, None] * stride_left_k \
            + d[None, None, :] * stride_left_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # -------- load right[b, j, k, d] ----------
        right_ptrs = right_ptr \
            + batch * stride_right_b \
            + j[:, None, None] * stride_right_i \
            + k[None, :, None] * stride_right_k \
            + d[None, None, :] * stride_right_d
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # for each hidden channel in the block, perform a matmul over K
        for db in range(0, BLOCK_D):
            # shape: (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
            left_mat = left[:, :, db]
            right_mat = right[:, :, db]
            # tl.dot expects (M, K) @ (K, N); we transpose the right operand
            acc[:, :, db] += tl.dot(left_mat, tl.trans(right_mat))

    # ---------- write result ----------
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + i[:, None, None] * stride_out_i \
        + j[None, :, None] * stride_out_j \
        + d[None, None, :] * stride_out_d
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of model weights (layernorm, linear weights, etc.)
            - config: dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the input -----
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if any) -----
    if nomask:
        mask_unsq = torch.ones_like(x[..., :1], dtype=dtype, device=device)
    else:
        mask_unsq = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ----- Gating -----
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # Make tensors contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # shape is [B, N, N, hidden_dim]

    # ----- Allocate output for the hidden dimension -----
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ----- Launch Triton kernel for the core einsum -----
    # strides (in elements, not bytes)
    lsb, lsi, lsk, lsd = left.stride()
    rsb, rsi, rsk, rsd = right.stride()
    osb, osi, osj, osd = out_hidden.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 8   # hidden dimension is a multiple of 8 in all tests
    BLOCK_K = 32

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    d_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid_z = B * d_blocks

    tri_mul_kernel[
        (grid_i, grid_j, grid_z)
    ](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        lsb,
        lsi,
        lsk,
        lsd,
        rsb,
        rsi,
        rsk,
        rsd,
        osb,
        osi,
        osj,
        osd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # ----- Final LayerNorm, gating and output projection -----
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply the output gate (broadcast over i, j)
    out_hidden = out_hidden * out_gate

    # Linear projection back to original dimension
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])

    return out
