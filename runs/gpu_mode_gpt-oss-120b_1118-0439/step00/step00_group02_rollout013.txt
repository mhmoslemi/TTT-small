    input_tensor, mask, weights, config = data  # mask shape (B,N,N)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    norm_bias = weights["norm.bias"]

    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()

def custom_kernel(data):
    """
    Outgoing TriMul forward
    Computes out[b,i,j,:] = sum_k left[b,i,k,:] * right[b,j,k,:]
    where left and right are linear projections of the normalized input,
    multiplied by per‑position gates and an optional mask. The core
    N‑by‑N reduction is performed with a fused Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over the last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask if present
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left = left_proj * mask_unsq
        right = right_proj * mask_unsq
    else:
        left = left_proj
        right = right_proj

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for Triton kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, D = left_fp16.shape  # D == hidden_dim

    # Allocate output (float32 for accumulation)
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Triton block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * D,

"""
Outgoing TriMul forward (AlphaFold3).

The module first applies a LayerNorm to the 4‑D tensor [B, N, N, C],
projects it to the hidden dimension, multiplies by three learned gates
and an optional pairwise mask, and finally computes the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

for every batch and hidden channel.  The N‑by‑N reduction over the
sequence dimension k is performed by a tiled batched GEMM written in
Triton (one kernel instance per (batch, hidden) pair).  The result
is normalised, gated and projected back to the original channel size.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _outgoing_gemm_kernel(
    left_ptr,          # pointer to left   [B, N, N, D]   (float16)
    right_ptr,         # pointer to right  [B, N, N, D]   (float16)
    out_ptr,           # pointer to output [B, N, N, D]   (float32)
    batch_hidden,      # B * D
    N,                 # sequence length
    D,                 # hidden dimension
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: for each (b, d) compute left @ rightᵀ over the K dimension."""
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bn = tl.program_id(2)         # combined batch⋅hidden index

    # decode batch and hidden channel from the flattened index
    b = pid_bn // D
    d = pid_bn % D

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulation buffer (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = tl.arange(0, BLOCK_K) + k * BLOCK_K
        mask_k = cur_k < N

        # left: (i, k)  -> shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr \
                + b * stride_l_batch \
                + d * stride_l_d \
                + offs_i[:, None] * stride_l_i \
                + cur_k[None, :] * stride_l_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right transposed: (k, j)  -> shape (BLOCK_K, BLOCK_N)
        b_ptr = right_ptr \
                + b * stride_r_batch \
                + d * stride_r_d \
                + cur_k[:, None] * stride_r_k \
                + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # store result
    c_ptr = out_ptr \
            + b * stride_o_batch \
            + d * stride_o_d \
            + offs_i[:, None] * stride_o_i \
            + offs_j[None, :] * stride_o_j
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict of torch.Tensors containing all learnable parameters
            - config: dict with keys "dim" (C) and "hidden_dim" (C_z)

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm over the input channel dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ---------- Linear projections ----------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------- Apply mask (if supplied) ----------
    if mask is not None:
        # make mask broadcastable to the hidden dimension
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left_proj * mask_exp
        right = right_proj * mask_exp
    else:
        left = left_proj
        right = right_proj

    # ---------- Apply element‑wise gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to float16 for the heavy contraction ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, D = left_fp16.shape  # D == hidden_dim

    # Allocate output tensor (float32 for accumulation)
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # ---------- Strides (elements, not bytes) ----------
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # ---------- Triton kernel launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * D,                                 # one kernel per (batch, hidden) pair
    )

    _outgoing_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B * D,
        N,
        D,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    # LayerNorm over the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Output gate
    out = out * out_gate

    # Final linear projection back to original channel size
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
