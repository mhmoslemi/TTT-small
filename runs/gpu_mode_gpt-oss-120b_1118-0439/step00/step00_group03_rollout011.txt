def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # get config values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    B, N, N2, C = input_tensor.shape
    assert N == N2
    device = input_tensor.device
    # layernorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask_tensor is not None and mask_tensor.numel() > 0:
        mask_4d = mask_tensor.unsqueeze(-1).type_as(left)
        left = left * mask_4d
        right = right * mask_4d
    # gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # convert to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # final projection
    out = F.linear(out_norm, weights["to_out.weight"])

"""
TriMul forward (outgoing) implementation using Triton for the O(N³) einsum.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ⁽ᴮ×ᴺ×ᴺ×C⁾.
2. Apply two linear projections (left/right) to obtain tensors of shape
   (B, N, N, H) where H = hidden_dim.
3. Optionally mask the projected tensors.
4. Compute gate tensors (sigmoid(linear(x))) and multiply them onto the
   projected tensors.
5. Perform the core operation:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication L × Rᵀ for each (b, h).
   It is realised with a custom Triton kernel that tiles the (i, j) output
   plane with BLOCK_M × BLOCK_N tiles and iterates over the k‑dimension in
   blocks of size BLOCK_K.
6. Layer‑norm the result over the hidden dimension, apply the output gate,
   and a final linear projection back to the original channel dimension C.
7. Return the tensor of shape (B, N, N, C) (float32).

The Triton kernel works on fp16 data (left/right are cast to half) while
accumulating in fp32, which matches the precision of the reference
implementation. The host code uses PyTorch for the cheap operations
(layer‑norms, linear layers, sigmoids) and calls the kernel only for the
expensive N³ step.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched L × Rᵀ for each hidden dimension.

    left  : (B, N, N, H)  – indexed as (b, i, k, h)
    right : (B, N, N, H)  – indexed as (b, j, k, h)
    out   : (B, N, N, H)  – indexed as (b, i, j, h)
    """
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_hb = tl.program_id(2)  # combined batch * hidden index

    # -------------------------------------------------------------
    # Decode batch and hidden indices
    hb = pid_hb
    batch_idx = hb // H
    hidden_idx = hb % H

    # -------------------------------------------------------------
    # Compute offsets within the i‑ and j‑ dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the beginning of the current (b, h) slice
    left_base  = left_ptr  + batch_idx * stride_left_b  + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_b   + hidden_idx * stride_out_h

    # -------------------------------------------------------------
    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for the k dimension
        mask_k = offs_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # Load a tile of right transposed: shape (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h] → (k, j) layout for tl.dot
        right_ptrs = right_base \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # Accumulate
        acc += tl.dot(left_tile, right_tile)

    # -------------------------------------------------------------
    # Store the result
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Execute the forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model weights (float32)
            - config: dict with keys "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, C]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Mask (if provided and not a no‑mask run)
    if mask_tensor is not None and mask_tensor.numel() > 0 and not config.get("nomask", False):
        mask_4d = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_4d
        right = right * mask_4d

    # -----------------------------------------------------------------
    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core Triton kernel (batched matrix multiplication)
    B, N, _, _ = left.shape  # N is sequence length

    # Cast to half for the kernel (accumulation stays in fp32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tiling parameters – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        # left strides
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # right strides
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # out strides
        out_fp32.stride(0),
        out_fp32.stride(1),
        out_fp32.stride(2),
        out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # -----------------------------------------------------------------
    # 6. Output layer‑norm over hidden dimension
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim channels
    out = F.linear(out_norm, weights["to_out.weight"])

    return out
