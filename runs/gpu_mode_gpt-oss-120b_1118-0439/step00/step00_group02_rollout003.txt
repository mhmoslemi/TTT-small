@triton.jit
def triton_trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    # Compute block indices
    b = tl.program_id(2)
    i_block = tl.program_id(0)
    j_block = tl.program_id(1)

    # offsets
    i0 = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k0_base = tl.arange(0, BLOCK_K)  # inside loop

    h0 = tl.arange(0, BLOCK_H)

    # masks for i,j,h within range
    mask_i = i0 < N
    mask_j = j0 < N
    mask_h = h0 < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K + k0_base
        mask_k = k0 < N

        # compute offsets for left: left[b,i,k,h]
        left_ptrs = left_ptr + b * stride_l_b \
                    + i0[:, None, None] * stride_l_i \
                    + k0[None, :, None] * stride_l_k \
                    + h0[None, None, :] * stride_l_h

        # compute offsets for right: right[b,j,k,h] (no transpose)
        right_ptrs = right_ptr + b * stride_r_b \
                     + j0[:, None, None] * stride_r_i \
                     + k0[None, :, None] * stride_r_k \
                     + h0[None, None, :] * stride_r_h

        # load with mask; produce FP16 -> cast to FP32
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0).to(tl.float32)

        # Multiply-accumulate across k dimension
        # left: (M,K,H), right: (N,K,H)
        # we want acc[i,j,h] += sum_k left[i,k,h] * right[j,k,h]
        # Use broadcasting: left[:, :, h] * right[:, :, h] but need broadcast across i and j.
        # We can compute for each k slice:
        #   a = left[:, k, :] shape (M,H)
        #   b = right[:, k, :] shape (N,H)
        #   acc += a[:, None, :] * b[None, :, :]
        # So loop over k within BLOCK_K.
        for kk in range(BLOCK_K):
            # ensure we don't go out of range for last tile when k exceeds N
            # but we already mask out-of-range values as 0 so it's okay
            a = left[:, kk, :]  # (M, H)

"""
custom_kernel for the outgoing TriMul operator.

The kernel focuses on the cubic‑complexity part:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

It tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N) and reduces over k in
BLOCK_K‑sized chunks, while also processing a small block of the hidden
dimension (BLOCK_H) at once.  All other operations (LayerNorm, projections,
gates, output norm and final linear) are performed with PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ reduction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_outgoing_kernel(
    left_ptr,          # *fp16 left  [B, N, N, H]
    right_ptr,         # *fp16 right [B, N, N, H]
    out_ptr,           # *fp32 out   [B, N, N, H]

    B, N, H,           # runtime constants

    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ per hidden slice.
    left shape : (B, N, N, H)  -> indexed as (b, i, k, h)
    right shape: (B, N, N, H)  -> indexed as (b, j, k, h)
    out shape  : (B, N, N, H)  -> (b, i, j, h)
    """
    # program ids
    b = tl.program_id(2)                     # batch index
    pid_i = tl.program_id(0)                  # i‑tile
    pid_j = tl.program_id(1)                  # j‑tile

    # tile offsets
    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h0 = tl.arange(0, BLOCK_H)                # hidden offset (within block)

    # range masks
    mask_i = i0 < N
    mask_j = j0 < N
    mask_h = h0 < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # ------------------ load left ------------------
        # left[b, i, k, h]
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + i0[:, None, None] * stride_l_i \
            + k0[None, :, None] * stride_l_k \
            + h0[None, None, :] * stride_l_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)   # (M, K, H)

        # ------------------ load right -----------------
        # right[b, j, k, h]
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + j0[:, None, None] * stride_r_i \
            + k0[None, :, None] * stride_r_k \
            + h0[None, None, :] * stride_r_h
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)   # (N, K, H)

        # ------------------ MAC over k -----------------
        # Perform a reduction over the K dimension.
        # Loop over each k inside the tile (unrolled manually).
        for kk in range(0, BLOCK_K):
            a = left[:, kk, :]          # (M, H)
            b_ = right[:, kk, :]        # (N, H)
            # Broadcast (M,1,H) * (1,N,H) => (M,N,H)
            acc += a[:, None, :] * b_[None, :, :]

    # ------------------ store -------------------------
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + i0[:, None, None] * stride_o_i \
        + j0[None, :, None] * stride_o_j \
        + h0[None, None, :] * stride_o_h

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones if nomask=True)
            - weights: dict with all model weights (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    input_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim]

    # --------------------------------------------------------------
    # Linear projections (dim → hidden)
    # --------------------------------------------------------------
    left_proj = F.linear(input_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(input_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Gates (sigmoid activations)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(input_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(input_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(input_norm, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # Apply optional mask (only on the projections, as in the reference)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # Apply gated element‑wise multiplication
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # Cast to float16 for the heavy N³ kernel
    # --------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Allocate output buffer (float32 for accumulation)
    out_hidden = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton kernel launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # 128 hidden → 16 blocks; works for any hidden ≤ 256

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs)

    # Strides for left/right/out tensors
    l_s0, l_s1, l_s2, l_s3 = left_h.stride()
    r_s0, r_s1, r_s2, r_s3 = right_h.stride()
    o_s0, o_s1, o_s2, o_s3 = out_hidden.stride()

    triton_trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_hidden,
        bs, N, hidden,
        l_s0, l_s1, l_s2, l_s3,
        r_s0, r_s1, r_s2, r_s3,
        o_s0, o_s1, o_s2, o_s3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # Continue with the remaining TriMul pipeline (LayerNorm, gating,
    # final projection) using PyTorch.
    # --------------------------------------------------------------
    # out_hidden is now [B,N,N,hidden] in float32
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )   # [B,N,N,hidden]

    # Apply the output gate
    out_gated = out_norm * out_gate

    # Final linear projection back to dim
    final_out = F.linear(out_gated, weights["to_out.weight"])

    return final_out
