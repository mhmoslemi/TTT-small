import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_j, stride_left_gate_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_j, stride_right_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * D
    mask = offs < total
    # compute coordinates
    # Offsets to compute (b,i,j,d)
    # using integer division
    # Use 64-bit ints: tl.int64
    offs_i = offs // D
    d = offs % D
    offs_ij = offs_i // N
    j = offs_i % N
    i = offs_ij % N
    b = offs_ij // N

    # pointers
    left = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + d * stride_left_d
    left_gate = left_gate_ptr + b * stride_left_gate_b + i * stride_left_gate_i + j * stride_left_gate_j + d * stride_left_gate_d
    right = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + d * stride_right_d
    right_gate = right_gate_ptr + b * stride_right_gate_b + i * stride_right_gate_i + j * stride_right_gate_j + d * stride_right_gate_d

    left_val = tl.load(left, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate, mask=mask, other=0.0)
    right_val = tl.load(right, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate, mask=mask, other=0.0)

    left_out = left_val * left_gate_val
    right_out = right_val * right_gate_val

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel defined above

def custom_kernel(data):
    """
    Triton-accelerated outgoing triangle multiplicative update.
    ...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # mask handling
    use_mask = not config.get("nomask", False) and mask is not None
    if use_mask:
        # Convert mask to same dtype and ensure shape [B,N,N]
        mask_tensor = mask.to(dtype=dtype)
    else:
        mask_tensor = torch.empty(0, device=device, dtype=dtype)  # placeholder

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape (..., hidden_dim)
    right = F.linear(x, right_proj_weight)

    # Gates pre-sigmoid
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask and gating with Triton kernel (in-place)
    B, N, _, _ = left.shape  # left shape (B,N,N,hidden_dim)
    # ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # launch kernel
    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    mask_ptr = mask_tensor.data_ptr() if use_mask else torch.empty(1, device=device, dtype=dtype).data_ptr()
    mask_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_ptr,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_tensor.stride(0) if use_mask else 0,
        mask_tensor.stride(1) if use_mask else 0,
        mask_tensor.stride(2) if use_mask else 0,
        HAS_MASK=use_mask,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1) # (B, D, N, N) transpose k

"""
custom_kernel: Triton‑accelerated outgoing triangle multiplicative update (TriMul).

The kernel fuses the mask application and gating (sigmoid) multiplication for the
left and right projected tensors.  The heavy N³ bilinear product is performed
with a batched matrix multiplication (torch.matmul) on the permuted tensors.
Only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HAS_MASK: tl.constexpr,        # compile‑time constant
    BLOCK_SIZE: tl.constexpr,     # compile‑time constant
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)

    total = B * N * N * D
    valid = offs < total

    # ---- decode linear index into (b,i,j,d) ----
    # total_per_batch = N * N * D
    # total_per_i   = N * D
    # total_per_j   = D
    b = offs // (N * N * D)
    rem = offs % (N * N * D)

    i = rem // (N * D)
    rem = rem % (N * D)

    j = rem // D
    d = rem % D

    # ----- pointers -----
    left = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + d * stride_left_d
    left_gate = left_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + d * stride_gate_d
    right = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + d * stride_right_d
    right_gate = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + d * stride_rgate_d

    left_val = tl.load(left, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate, mask=valid, other=0.0)
    right_val = tl.load(right, mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate, mask=valid, other=0.0)

    left_out = left_val * left_gate_val
    right_out = right_val * right_gate_val

    if HAS_MASK:
        # mask has shape (B,N,N) – broadcast over the hidden dimension
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
                           mask=valid, other=1.0)
        left_out = left_out * mask_val
        right_out = right_out * mask_val

    tl.store(left, left_out, mask=valid)
    tl.store(right, right_out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or ``None``
        - weights      : dict of torch.Tensor containing all model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the updated pair representation.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ---------- Gate linear layers + sigmoid ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ---------- Optional mask ----------
    use_mask = (not nomask) and (mask is not None)
    if use_mask:
        mask_tensor = mask.to(dtype=dtype)
    else:
        # dummy tensor – never accessed when HAS_MASK=False
        mask_tensor = torch.empty(1, device=device, dtype=dtype)

    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if use_mask:
        mask_tensor = mask_tensor.contiguous()

    B, N, _, _ = left.shape  # left.shape == (B,N,N,hidden_dim)

    # ---------- Triton kernel: mask & gating (in‑place) ----------
    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_tensor,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_tensor.stride(0) if use_mask else 0,
        mask_tensor.stride(1) if use_mask else 0,
        mask_tensor.stride(2) if use_mask else 0,
        HAS_MASK=use_mask,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ---------- Bilinear aggregation across k (i.e. matmul) ----------
    # left/right: (B,N,N,hidden_dim) -> (B,hidden_dim,N,N)
    left_perm = left.permute(0, 3, 1, 2)        # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1)      # (B, D, N, N)   (transpose k dimension)
    out_perm = torch.matmul(left_perm, right_perm)  # (B, D, N, N)
    out = out_perm.permute(0, 2, 3, 1)                # (B, N, N, D)

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out
