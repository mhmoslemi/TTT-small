@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid = tl.program_id(0) # linear id across total blocks (batch * Mblocks * Nblocks)
   # compute batch index
   batch_stride = tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
   batch_idx = pid // batch_stride
   block_idx = pid % batch_stride
   block_i = block_idx // tl.cdiv(N, BLOCK_N)
   block_j = block_idx % tl.cdiv(N, BLOCK_N)
   # offsets
   offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask within range
   mask_i = offs_i < M
   mask_j = offs_j < N
   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
   # loop over k
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       k_base = k * BLOCK_K
       offs_k = k_base + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K
       # load tiles
       a = tl.load(a_ptr + batch_idx * stride_a_batch + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak,
                   mask=mask_i[:, None] & mask_k[None, :], other=0.0)
       b = tl.load(b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn,
                   mask=mask_k[:, None] & mask_j[None, :], other=0.0)
       # compute dot
       acc += tl.dot(a, b)
   # store results
   c = acc.to(tl.float16) if ??? else compute to float32? We'll keep float32 as final dtype.
   tl.store(c_ptr + batch_idx * stride_c_batch + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn,
            c, mask=mask_i[:, None] & mask_j[None, :])

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,

):

    # Compute batch/block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # Offsets for i and j
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

            a_ptr + batch_idx * stride_az + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],

"""
TriMul (outgoing) forward pass with a Triton‐based batched GEMM.

The heavy part of TriMul is the tensor contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of independent matrix multiplications
    out_d = left_d @ right_d.T   (left_d,right_d : [N,N])
for each hidden channel d and each batch element.
We fuse the batched GEMM with Triton and keep the rest of the
operations (LayerNorms, Linear projections, gating) in PyTorch for
readability and correctness.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B^T) for tensors of shape
#   A : [BATCH, M, K]   (float16)
#   B : [BATCH, K, N]   (float16)   – already transposed
#   C : [BATCH, M, N]   (float16)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,      # A: batch, M, K
    stride_bz, stride_bk, stride_bn,      # B: batch, K, N
    stride_cz, stride_cm, stride_cn,      # C: batch, M, N
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------
    # Decode a 3‑D grid (batch, block_i, block_j)
    # ------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_id  = pid %  blocks_per_batch
    block_i   = block_id // num_blocks_n
    block_j   = block_id %  num_blocks_n

    # ------------------------------------------------------
    # Offsets within the M and N dimensions
    # ------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_ptr + batch_idx * stride_az
                    + offs_i[:, None] * stride_am
                    + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: shape [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_ptr + batch_idx * stride_bz
                    + offs_k[:, None] * stride_bk
                    + offs_j[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate dot product (fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------
    # Write back C tile (convert to fp16)
    # ------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + batch_idx * stride_cz
                + offs_i[:, None] * stride_cm
                + offs_j[None, :] * stride_cn,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# --------------------------------------------------------------
# Entry point used by the grader
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]          # C_z
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on input (dim)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # mean / var over channel dim
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Rearrange tensors so that the GEMM is (B*hidden, N, N) @ (B*hidden, N, N)^T
    #   left  : (B, hidden, i, k)
    #   right : (B, hidden, k, j)   (k and j swapped)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)  k,j

    BH = bs * hidden
    left_flat  = left_perm.reshape(BH, N, N)   # [BH, M, K]
    right_flat = right_perm.reshape(BH, N, N)  # [BH, K, N] (already transposed)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    M, K, N_ = N, N, N   # square matrices, all equal to sequence length

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_ + BLOCK_N - 1) // BLOCK_N
    total_blocks = BH * num_blocks_m * num_blocks_n

    # Strides for contiguous tensors
    stride_az = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bz = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cz = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    batched_gemm_kernel[(total_blocks,)](
        left_flat, right_flat, out_flat,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        M, N_, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)
    out = out.to(torch.float32)  # back to fp32 for subsequent layers

    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(-1, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    # ------------------------------------------------------------------
    # Apply final gating and projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
