@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0) # over combined (b * hidden_dim)
    total = B * hidden_dim
    if pid >= total:
        return
    b = pid // hidden_dim
    d = pid % hidden_dim

    # tile indices for rows i and cols j
    pid_m = tl.program_id(axis=1) # row tile index
    pid_n = tl.program_id(axis=2) # col tile index

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers
    # left: shape (N, N) dims i,k (row i, column k)
    left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    # right: shape (N, N) dims j,k but we need right[j,k,d] -> we will need to treat right as accessed at (j,k) with same d (since hidden_dim same)
    # In out = left[i,k]*right[j,k] sum over k: right[j,k] needed. So we use j index as col. We need to load tile of right for rows j, k? Actually for each block N,N we need j tile (col) and k tile: we treat right transposed across k (the sum axis). Because we want right[j,k] not right[k,j]. So we load right_tile as shape (BLOCK_N, BLOCK_K) with rows j across N and columns k across K.
    # For right, pointer indexing: j dimension uses stride_right_i? In original shape right dims: (batch, N, N, hidden_dim) where shape indexes: (b, j, k, d). Strides: stride_i over j dimension (second N), stride_k over k dimension (third N). Both sampling on memory accordingly.     right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + offs_n[:, None] * stride_right_i + offs_k[None, :] * stride_right_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load slices
        left = tl.load(left_ptrs, mask=(offs_m[:,None] < N) & (offs_k[None,:] < N), other=0.0)

        right_ptrs += BLOCK_K * stride_right_k

    tl.store(out_ptrs, acc.to(tl.float16) if acc.dtype != tl.float16 else acc, mask=(offs_m[:,None] < N) & (offs_n[None,:] < N))

    # Set eps=1e-5.

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Step 2: left_proj = linear(x): no bias. weight shape [hidden_dim, dim].

"""
TriMul forward with a custom Triton kernel for the core N^3 einsum.

The kernel computes, for each batch b and hidden channel d,
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batched matrix multiplication of
left[..., :, :] @ right[..., :, :].T for each hidden dimension.
All other operations (LayerNorms, linear projections, gates) are done
with PyTorch for simplicity; the kernel handles the heavy O(N³) work.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel: compute out = left @ right^T for every (batch, hidden) pair
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # sizes: batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_comb = tl.program_id(0)   # batch * hidden index
    pid_m    = tl.program_id(1)   # i tile
    pid_n    = tl.program_id(2)   # j tile

    total = B * H
    if pid_comb >= total:
        return

    # decode batch and hidden dim
    b = pid_comb // H
    d = pid_comb % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers for left: shape (i, k)
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k

    # Pointers for right: we need (k, j) layout for the dot product
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for _ in range(num_tiles_k):
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)
        acc += tl.dot(left, right)  # (M,K) @ (K,N) -> (M,N)

        # advance k‑tiles
        left_ptrs  += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_k

    # store result
    out_base = out_ptr + b * stride_o_b + d * stride_o_d
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of TriMul (outgoing version) using a Triton kernel for the
    multiplicative update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : Tensor[bs, N, N, dim] (float32)
        - mask        : Tensor[bs, N, N] (float32 or bool) or None
        - weights     : dict of model weights (float32)
        - config      : dict containing "dim" and "hidden_dim"

    Returns
    -------
    Tensor[bs, N, N, dim]
        Result of the TriMul forward pass.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask if present
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core multiplicative update via Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # cast to FP16 for better throughput inside kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements)
    s_l_b, s_l_i, s_l_k, s_l_d = left_fp16.stride()
    s_r_b, s_r_i, s_r_k, s_r_d = right_fp16.stride()
    s_o_b, s_o_i, s_o_j, s_o_d = out.stride()

    # block sizes – tuned for typical B200 GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_d,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 7. Apply output gate and final projection
    # -----------------------------------------------------------------
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
