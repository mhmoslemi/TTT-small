    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Linear weights
    left_proj_w = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_w = weights["to_out.weight"]        # shape [dim, hidden_dim] (output dim)
    # Norm weights
    norm_w = weights["norm.weight"]   # shape [dim]
    norm_b = weights["norm.bias"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # eps for layernorm
    eps = 1e-5

    # Normalize input
    # Compute mean and var across dim (last dimension)
    # flatten batch and N,N dims
    x = input_tensor
    # compute mean per element across D
    # shape (B,N,N) -> broadcast

@triton.jit
def batched_matmul_kernel(A, B, C,
                          stride_ab, stride_am, stride_ak,
                          stride_bb, stride_bk, stride_bn,
                          stride_cb, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   # program ID
   pid_b = tl.program_id(0)  # batch index
   pid_m = tl.program_id(1)  # block row index
   pid_n = tl.program_id(2)  # block col index

   # Compute start offsets
   off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # Pointers start for each batch
   a_ptr = A + pid_b * stride_ab + off_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
   b_ptr = B + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + off_n[None, :] * stride_bn

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over k
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptr, mask=(off_m < M) & (tl.arange(0, BLOCK_K) + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptr, mask=(tl.arange(0, BLOCK_K) + k*BLOCK_K < K) & (off_n < N), other=0.0)
        # Compute dot product of (M x K) and (K x N)
        acc += tl.dot(a, b)
        a_ptr += BLOCK_K * stride_ak
        b_ptr += BLOCK_K * stride_bk

   # Write back
   c_ptr = C + pid_b * stride_cb + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
   mask = (off_m[:, None] < M) & (off_n[None, :] < N)
   tl.store(c_ptr, acc, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # retrieve shapes
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm normalization
    # compute mean/var across dim
    # x_norm = (input-t_mean)/sqrt(var+eps) * norm_weight + norm_bias
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # Projections
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Linear transforms
    left = torch.nn.functional.linear(x_norm, left_proj_w)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    if mask is not None:
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched GEMM
    # Permute to bring hidden dim before sequence dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose last two dims of right
    right_T = right_perm.transpose(2, 3).contiguous()
    # Flatten batch*hidden_dim
    batch_total = bs * hidden_dim
    left_flat = left_perm.reshape(batch_total, N, N)
    right_flat = right_T.reshape(batch_total, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch_total,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output back to [bs, N, N, hidden_dim]
    out = out_flat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Apply to_out_norm (layernorm over hidden_dim channel)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # compute mean and var across hidden_dim
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)

    out_final = torch.nn.functional.linear(out_norm, to_out_w)  # returns [bs,N,N,dim]

TriMul forward kernel (outgoing version) with a Triton GEMM.
The implementation follows the reference PyTorch module:

1. LayerNorm on the input tensor (last dimension).
2. Linear projections (left/right) and gating (left/right/out) using the
   provided weights.  Mask (if any) is applied after projection.
3. The core multiplicative update is:
       out[i,j] = Σ_k left[i,k] * right[j,k]
   which can be expressed as a batch of matrix multiplications:
       out_batch = left_batch @ right_batch.T
   where each batch corresponds to a (batch, hidden_dim) pair.
   This step is accelerated with a custom Triton kernel.
4. LayerNorm on the intermediate result, element‑wise gating, and a final
   linear projection back to the original dimension.

All heavy computation (the N³ term) is performed by the Triton kernel,
while the remaining element‑wise operations use native PyTorch for
convenience.

import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[b] = A[b] @ B[b].T   (A,B,C are 3‑D tensors)."""
    pid_b = tl.program_id(0)          # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)          # row block
    pid_n = tl.program_id(2)          # col block

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of the current block
    a_ptrs = A + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k_offset < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] + k_offset < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)                      # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak            # advance K‑slice
        b_ptrs += BLOCK_K * stride_bk

    # Write the result
    c_ptrs = C + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)


def custom_kernel(data):
    """TriMul forward pass (outgoing)."""
    # Unpack inputs
    input_tensor, mask, weights, config = data
    eps = 1e-5

    # Shapes / config
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b                 # shape [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core multiplicative update (batched GEMM) ----------
    # Rearrange tensors to [B * hidden_dim, N, N] with batch dim first
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    # Transpose the inner matrix of right (k <-> j)
    right_T = right_perm.transpose(2, 3).contiguous()      # [B, H, N, N] where dim2/3 swapped

    batch_total = bs * hidden_dim
    left_flat = left_perm.reshape(batch_total, N, N)
    right_flat = right_T.reshape(batch_total, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Block sizes – chosen to fit the B200 memory hierarchy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ---------- Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_out) * torch.rsqrt(var_out + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- Gating & final projection ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
