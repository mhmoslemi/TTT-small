def custom_kernel(data):
    x, mask, weights, config = data
    # maybe mask is None: handle accordingly

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # layernorm on x
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    
    # Linear projections:
    # x shape [B,N,N,dim]
    x_flat = x.view(-1, dim) # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"])  # weight shape (hidden_dim, dim)
    left = left.view(x.shape[0], x.shape[1], x.shape[2], hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"])
    right = right.view(... same shape)
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(...).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(...).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(...).sigmoid()
    
    # apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask
    # apply gates
    left = left * left_gate
    right = right * right_gate

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM kernel.

The module performs:
1. LayerNorm on the input tensor (dim = C).
2. Linear projections to a hidden dimension H for left/right tensors and
   their sigmoid gates.
3. Optional masking of left/right.
4. Multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication per hidden channel.
   Implemented with a custom Triton kernel that processes (B*H) independent
   GEMMs of shape (N,N) @ (N,N)^T.
5. LayerNorm on the hidden dimension, gating with out_gate, and final linear
   projection back to the original dimension.

The kernel loads blocks of A (left) and B (right) in fp16, accumulates in fp32,
and writes the result in fp16 before casting back to fp32 in Python.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the contracted einsum:                               #
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]                   #
# This is a batched GEMM over the k dimension for each (b,h) pair.      #
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, H,                           # runtime sizes
    stride_ah, stride_ai, stride_ak, stride_ad,   # strides for A (left)
    stride_bh, stride_bj, stride_bk, stride_bd,   # strides for B (right)
    stride_ch, stride_ci, stride_cj, stride_cd,   # strides for C (out)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------- #
    # program IDs
    pid = tl.program_id(0)       # combined batch*hidden
    pid_i = tl.program_id(1)     # block row (i)
    pid_j = tl.program_id(2)     # block col (j)

    # Decompose pid into batch index and hidden index
    h = pid % H
    b = pid // H

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + b * stride_ah
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
            + h * stride_ad
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (transposed version): shape (BLOCK_K, BLOCK_N)
        # B is stored as [b, j, k, h]; we need B_T[k, j]
        b_ptrs = (
            b_ptr
            + b * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
            + h * stride_bd
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate the block matrix product
        acc += tl.dot(a, b)

    # Write back the result in fp16 (the output tensor is allocated as fp16)
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + b * stride_ch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
        + h * stride_cd
    )
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul einsum.
    left / right: [B, N, N, H] (float16 preferred)
    Returns out: [B, N, N, H] (float16)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous memory and half precision for bandwidth
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides (in elements) for each tensor
    stride_ah, stride_ai, stride_ak, stride_ad = left.stride()
    stride_bh, stride_bj, stride_bk, stride_bd = right.stride()
    stride_ch, stride_ci, stride_cj, stride_cd = out.stride()

    # Tile sizes (tuned for B200; adjust if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, num_tile_rows, num_tile_cols)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_ad,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_bd,
        stride_ch,
        stride_ci,
        stride_cj,
        stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Cast back to fp32 for the remaining pipeline
    return out.to(torch.float32)


def custom_kernel(data):
    """
    Entry point for the Triton-accelerated TriMul (outgoing) operation.
    `data` is a tuple:
        (input, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------- #
    # 2. Linear projections + sigmoid gates
    # ------------------------------------------------------------------- #
    x_flat = x.reshape(-1, dim)               # (B*N*N, dim)

    # Projections
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).sigmoid()

    # Reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------- #
    # 3. Optional masking (mask shape: [B, N, N])
    # ------------------------------------------------------------------- #
    if mask is not None:
        # Ensure mask is broadcastable and same dtype
        mask = mask.to(dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4. Multiplicative update via fused GEMM (the expensive part)
    # ------------------------------------------------------------------- #
    out = tri_mul_gemm(left, right)           # (B, N, N, H) float32

    # ------------------------------------------------------------------- #
    # 5. Output layernorm, gating, and final projection back to dim
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out_flat = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, weights["to_out.weight"])
    out = out.view(B, N, N, dim)

    return out
```
