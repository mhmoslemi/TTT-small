Pseudo kernel body:

```
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B_D, N,
    stride_left_g, stride_left_m, stride_left_k,
    stride_right_g, stride_right_k, stride_right_n,
    stride_out_g, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_g = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        a_ptrs = left_ptr + pid_g * stride_left_g + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + pid_g * stride_right_g + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # store result
    out_ptrs = out_ptr + pid_g * stride_out_g + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: Triton-accelerated forward pass of the TriMul (outgoing) module.

The algorithm follows the reference PyTorch implementation:
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension (left, right) and three gating
   projections (left_gate, right_gate, out_gate).
3. Masking and gating applied element‑wise.
4. Core O(N³) operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM for each (batch, hidden) pair with a
   Triton kernel (no explicit Python loops).
5. Post‑processing: LayerNorm, out_gate multiplication and final linear
   projection back to the original channel dimension.

The Triton kernel performs a fused matrix multiplication for each group
g = batch * hidden_dim, handling arbitrary sequence lengths N up to 1024.
All tensors are kept in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# --------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    N,                                          # sequence length
    stride_left_g, stride_left_m, stride_left_k,
    stride_right_g, stride_right_k, stride_right_n,
    stride_out_g, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[g, i, j] = Σ_k left[g, i, k] * right[g, k, j] for all groups g."""
    pid_g = tl.program_id(0)      # group = batch * hidden_dim
    pid_m = tl.program_id(1)      # block row (i)
    pid_n = tl.program_id(2)      # block column (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to A (left) and B (right_T) sub‑tiles
        a_ptrs = left_ptr + pid_g * stride_left_g \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + pid_g * stride_right_g \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n

        # Load with masks, fill OOB with 0
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Fused dot product over K
        acc += tl.dot(a, b)

    # Store the result tile
    out_ptrs = out_ptr + pid_g * stride_out_g \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor:  torch.Tensor [B, N, N] (float or bool)
            - weights_dict: dict with all model weights (torch.Tensor)
            - config_dict: dict with keys "dim", "hidden_dim", "nomask", ...

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask_f = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the core contraction
    # ------------------------------------------------------------------
    B, N, _, D = left.shape                     # D == hidden_dim
    # Prepare tensors for the kernel: shape (B, D, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, D, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims → [B, D, N, N]

    # Flatten batch and hidden dimension into a single group dimension
    G = B * D
    left_g = left_perm.view(G, N, N)
    right_g = right_perm.view(G, N, N)
    out_g = torch.empty_like(left_g)

    # Strides of the contiguous (G, N, N) tensors
    s_left_g = left_g.stride(0)
    s_left_m = left_g.stride(1)
    s_left_k = left_g.stride(2)

    s_right_g = right_g.stride(0)
    s_right_k = right_g.stride(1)   # row (k)
    s_right_n = right_g.stride(2)   # col (j)

    s_out_g = out_g.stride(0)
    s_out_m = out_g.stride(1)
    s_out_n = out_g.stride(2)

    # Block sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        G,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left_g,
        right_g,
        out_g,
        N,
        s_left_g, s_left_m, s_left_k,
        s_right_g, s_right_k, s_right_n,
        s_out_g, s_out_m, s_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before subsequent ops
    torch.cuda.synchronize()

    # Reshape back to original layout [B, N, N, hidden_dim]
    out = out_g.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing (norm, out_gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
