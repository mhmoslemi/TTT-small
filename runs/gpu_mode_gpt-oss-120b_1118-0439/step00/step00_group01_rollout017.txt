def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    d_idx,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute base offsets for this block and dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers start
    # For left: (b, i, k, d)
    left_ptrs = left_ptr + pid_b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + d_idx * stride_left_d

    # For right: (b, j, k, d) [swap i and j]
    right_ptrs = right_ptr + pid_b * stride_right_b + offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k + d_idx * stride_right_d

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load blocks
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < N), other=0.0)
        # Compute block matmul: sum_{k_block} left(i,k)*right(j,k)

        right_ptrs += BLOCK_K * stride_right_k

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # config: dim, hidden_dim, maybe also nomask
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True) # bool
    eps = 1e-5
    
    # Cast to appropriate dtype
    x = input_tensor
    if not x.is_contiguous():
        x = x.contiguous()
    # mask handling
    if nomask or mask_tensor is None:
        mask = torch.ones((x.shape[0], x.shape[1], x.shape[2]), dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)   # shape (B,N,N,1)
    
    # Load weights tensors, ensure correct dtype
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm on x
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask to projections
    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Triton kernel for tri-mul

    out_norm = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out_norm * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)

"""
TriMul (outgoing) forward pass for AlphaFold3 implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Project the normalized tensor with three linear layers (left, right, out‑gate)
   and compute sigmoid gates for left, right and output.
3. Apply a binary mask (if provided) to the left/right projections,
   then multiply them by their respective gates.
4. Compute the core pairwise multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each hidden‑dim channel.
   It is implemented in a Triton kernel that tiles the i/j dimensions
   (BLOCK_M × BLOCK_N) and iterates over k (BLOCK_K) and the hidden
   dimension d (BLOCK_D) in loops, using `tl.dot` for the inner
   reduction.
5. Apply a second layer‑norm on the hidden dimension, multiply by the out‑gate,
   and finally linearly project back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batch of matrix multiplications for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)               # batch dimension
    pid_i = tl.program_id(1)               # i‑tile
    pid_j = tl.program_id(2)               # j‑tile

    # --------------------------------------------------------------
    # 1) Offsets for the i‑ and j‑tiles
    # --------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # --------------------------------------------------------------
    # 2) Loop over hidden‑dim tiles
    # --------------------------------------------------------------
    num_d = tl.cdiv(H, BLOCK_D)
    for d_iter in range(0, num_d):
        d_base = d_iter * BLOCK_D
        d = d_base + tl.arange(0, BLOCK_D)
        mask_d = d < H

        # Accumulator for this d‑tile (shape: D × M × N)
        acc = tl.zeros((BLOCK_D, BLOCK_M, BLOCK_N), dtype=tl.float32)

        # ----------------------------------------------------------
        # 3) Loop over reduction axis k
        # ----------------------------------------------------------
        num_k = tl.cdiv(N, BLOCK_K)
        for k_iter in range(0, num_k):
            k_base = k_iter * BLOCK_K
            k = k_base + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # Pointers for the left / right blocks (shape D×M×K and D×N×K)
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + i[None, :, None] * stride_left_i
                + k[None, None, :] * stride_left_k
                + d[:, None, None] * stride_left_d
            )
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + j[None, :, None] * stride_right_j
                + k[None, None, :] * stride_right_k
                + d[:, None, None] * stride_right_d
            )

            # Load with proper masking
            left = tl.load(
                left_ptrs,
                mask=mask_i[None, :, None] & mask_k[None, None, :] & mask_d[:, None, None],
                other=0.0,
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[None, :, None] & mask_k[None, None, :] & mask_d[:, None, None],
                other=0.0,
            )

            # Accumulate dot product over K
            acc += tl.dot(left, right)          # (D, M, N)

        # ----------------------------------------------------------
        # 4) Store the computed tile to the output tensor
        # ----------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + i[None, :, None] * stride_out_i
            + j[None, None, :] * stride_out_j
            + d[:, None, None] * stride_out_d
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[None, :, None] & mask_j[None, None, :] & mask_d[:, None, None],
        )


def trmul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the Triton‑accelerated bilinear update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tunable tile sizes (work well for N up to 1024 and H up to 384)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    trmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if provided (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", True):
        mask = None
    else:
        mask = mask_tensor.to(x.dtype)
    if mask is not None:
        mask = mask.unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # Compute gates
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # Apply gates to projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core TriMul operation (Triton)
    # ------------------------------------------------------------------
    inter = trmul_forward(left, right)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Output post‑processing
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Hidden‑dim LayerNorm
    out = torch.nn.functional.layer_norm(
        inter,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Multiply by output gate
    out = out * out_gate
    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
