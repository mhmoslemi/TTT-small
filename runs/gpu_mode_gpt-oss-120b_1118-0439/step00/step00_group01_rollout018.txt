@triton.jit
def layernorm_gate_kernel(
    out_ptr, out_gate_ptr, norm_weight_ptr, norm_bias_ptr,
    out_norm_gated_ptr,
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w, stride_norm_b,
    stride_out_norm_gated_b, stride_out_norm_gated_i, stride_out_norm_gated_j, stride_out_norm_gated_h,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # position index
    pos = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    total = B * N * N
    mask_pos = pos < total

    # compute b,i,j index from pos
    batch_idx = pos // (N * N)
    pos_in_batch = pos % (N * N)
    i_idx = pos_in_batch // N
    j_idx = pos_in_batch % N

    # compute base offsets
    base_out = batch_idx * stride_out_b + i_idx * stride_out_i + j_idx * stride_out_j
    base_gate = batch_idx * stride_gate_b + i_idx * stride_gate_i + j_idx * stride_gate_j
    base_outnorm = batch_idx * stride_out_norm_gated_b + i_idx * stride_out_norm_gated_i + j_idx * stride_out_norm_gated_j

    # first compute mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H
        # load out values
        vals = tl.load(out_ptr + base_out[:, None] + cur_h, mask=mask_pos[:, None] & mask_h[None,:], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    eps = 1e-5
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: compute normalized and gated, store result
    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        vals = tl.load(out_ptr + base_out[:, None] + cur_h, mask=mask_pos[:, None] & mask_h[None,:], other=0.0)
        gate_vals = tl.load(out_gate_ptr + base_gate[:, None] + cur_h, mask=mask_pos[:, None] & mask_h[None,:], other=0.0)

        # load norm weight and bias
        w = tl.load(norm_weight_ptr + cur_h, mask=mask_h, other=0.0)
        b = tl.load(norm_bias_ptr + cur_h, mask=mask_h, other=0.0)

        # normalize
        normalized = (vals - mean[:, None]) * inv_std[:, None] * w + b
        gated = normalized * gate_vals

        # store to out_norm_gated
        tl.store(out_norm_gated_ptr + base_outnorm[:, None] + cur_h,
                 gated,
                 mask=mask_pos[:, None] & mask_h[None,:])

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # For stability, we convert to float32.
    x = input_tensor

    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight.view(1,1,1,-1) + norm_bias.view(1,1,1,-1)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gamma = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (bs,N,N,hidden_dim)
    right_gamma = torch.nn.functional.linear(x_norm, right_proj_weight)

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply mask if given (maybe mask is bool or float)
    if mask is not None:
        mask_f = mask.to(dtype=x_norm.dtype).unsqueeze(-1)  # shape (bs,N,N,1)
    else:
        mask_f = torch.ones_like(x_norm[..., :1])
    # Apply mask to projected left/right
    left = left_gamma * mask_f
    right = right_gamma * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute out via batched GEMM across hidden_dim
    # reshape for bmm
    # (bs, hidden_dim, N, N) -> (bs*hidden_dim, N, N)

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    H = hidden_dim

    right_flat = right_perm.view(B * H, N, N)

    # reshape back to (bs, N, N, hidden_dim)

        stride_gate_i=out_gate.stride(1),
        stride_gate_j=out_gate.stride(2),
        stride_gate_h=out_gate.stride(3),

        stride_norm_b=to_out_norm_bias.stride(0),
        stride_out_norm_gated_b=out_norm_gated.stride(0),
        stride_out_norm_gated_i=out_norm_gated.stride(1),
        stride_out_norm_gated_j=out_norm_gated.stride(2),
        stride_out_norm_gated_h=out_norm_gated.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,

import torch
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel: fused LayerNorm (over hidden dim) + out_gate
# -----------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    # pointers
    out_ptr,                # [B, N, N, H]  (input of this stage)
    out_gate_ptr,           # [B, N, N, H]  (gate)
    norm_weight_ptr,        # [H]
    norm_bias_ptr,          # [H]
    out_norm_gated_ptr,     # [B, N, N, H]  (output of this stage)
    # shape / strides
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_outnorm_b, stride_outnorm_i, stride_outnorm_j, stride_outnorm_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,  # positions per program
    BLOCK_H: tl.constexpr,  # hidden‑dim tile
):
    pid = tl.program_id(0)

    # ---------- 1D indexing of (b,i,j) ----------
    total = B * N * N
    pos = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_pos = pos < total

    # safe indices (avoid OOB address computation)
    batch_idx = tl.where(mask_pos, pos // (N * N), 0)
    pos_in_batch = tl.where(mask_pos, pos % (N * N), 0)
    i_idx = tl.where(mask_pos, pos_in_batch // N, 0)
    j_idx = tl.where(mask_pos, pos_in_batch % N, 0)

    # base offsets for the three tensors (no hidden dim yet)
    base_out    = batch_idx * stride_out_b    + i_idx * stride_out_i    + j_idx * stride_out_j
    base_gate   = batch_idx * stride_gate_b   + i_idx * stride_gate_i   + j_idx * stride_gate_j
    base_outnorm= batch_idx * stride_outnorm_b+ i_idx * stride_outnorm_i+ j_idx * stride_outnorm_j

    # ---------- First pass: compute mean & var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_H):
        cur_h = k_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # load out[..., h]
        out_vals = tl.load(
            out_ptr + base_out[:, None] + cur_h,
            mask=mask_pos[:, None] & mask_h[None, :],
            other=0.0,
        )
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

    eps = 1e-5
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- Second pass: normalize, gate, store ----------
    for k_start in range(0, H, BLOCK_H):
        cur_h = k_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # load values
        out_vals  = tl.load(
            out_ptr + base_out[:, None] + cur_h,
            mask=mask_pos[:, None] & mask_h[None, :],
            other=0.0,
        )
        gate_vals = tl.load(
            out_gate_ptr + base_gate[:, None] + cur_h,
            mask=mask_pos[:, None] & mask_h[None, :],
            other=0.0,
        )
        # load norm parameters
        w = tl.load(norm_weight_ptr + cur_h, mask=mask_h, other=0.0)  # (BLOCK_H,)
        b = tl.load(norm_bias_ptr   + cur_h, mask=mask_h, other=0.0)  # (BLOCK_H,)

        # normalize
        normed = (out_vals - mean[:, None]) * inv_std[:, None] * w + b
        # apply gate
        gated = normed * gate_vals

        # store
        tl.store(
            out_norm_gated_ptr + base_outnorm[:, None] + cur_h,
            gated,
            mask=mask_pos[:, None] & mask_h[None, :],
        )

# -----------------------------------------------------------
# Main entry point (PyTorch + Triton)
# -----------------------------------------------------------
def custom_kernel(data):
    """
    TriMul 'outgoing' forward pass.
    Uses PyTorch for linear projections and the heavy 3‑D multiplication,
    and a Triton kernel for the final LayerNorm + out_gate fusion.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------- LayerNorm on the input (dim) --------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean / var over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w.view(1, 1, 1, -1) + norm_b.view(1, 1, 1, -1)

    # -------- Linear projections --------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # -------- Gates --------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # -------- Apply mask (if any) and gates --------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
    else:
        mask_f = torch.ones_like(x_norm[..., :1])
    left = left_proj * mask_f
    right = right_proj * mask_f
    left = left * left_gate
    right = right * right_gate

    # -------- triangular multiplicative update (outgoing) ----------
    # reshape to (B*H, N, N) and perform batched matmul
    left_t = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    left_flat = left_t.view(bs * hidden_dim, seq_len, seq_len)
    right_flat = right_t.view(bs * hidden_dim, seq_len, seq_len)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    # out shape: (B, N, N, H)

    # -------- fused LayerNorm + out_gate via Triton ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm_gated = torch.empty_like(out)

    total_positions = bs * seq_len * seq_len
    BLOCK_M = 128   # positions processed per program
    BLOCK_H = 32    # hidden‑dim tile
    grid = (triton.cdiv(total_positions, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_ptr=out,
        out_gate_ptr=out_gate,
        norm_weight_ptr=to_out_norm_w,
        norm_bias_ptr=to_out_norm_b,
        out_norm_gated_ptr=out_norm_gated,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        stride_gate_b=out_gate.stride(0),
        stride_gate_i=out_gate.stride(1),
        stride_gate_j=out_gate.stride(2),
        stride_gate_h=out_gate.stride(3),
        stride_outnorm_b=out_norm_gated.stride(0),
        stride_outnorm_i=out_norm_gated.stride(1),
        stride_outnorm_j=out_norm_gated.stride(2),
        stride_outnorm_h=out_norm_gated.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # -------- final linear projection back to dim --------
    # to_out.weight shape: (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_norm_gated, weights["to_out.weight"])

    return out_final
