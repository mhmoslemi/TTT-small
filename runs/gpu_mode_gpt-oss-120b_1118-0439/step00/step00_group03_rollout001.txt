@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,  # can be nullptr (0) if no mask
    out_left_ptr, out_right_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    has_mask: tl.constexpr,
):

    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)

"""
TriMul forward pass (outgoing version) with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (C = dim).
2. Linear projections:
   * left  = Linear(x_norm, left_proj_weight)      -> [B,N,N,hidden_dim]
   * right = Linear(x_norm, right_proj_weight)    -> [B,N,N,hidden_dim]
   * left_gate  = sigmoid(Linear(x_norm, left_gate_weight))
   * right_gate = sigmoid(Linear(x_norm, right_gate_weight))
   * out_gate   = sigmoid(Linear(x_norm, out_gate_weight))   -> [B,N,N,hidden_dim]
3. Fuse the element‑wise mask, left‑gate and right‑gate multiplications in a
   Triton kernel (no arithmetic beyond elementwise ops).
4. Compute the core O(N³) operation:
      out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   which is a batched matrix multiplication per hidden dimension:
      left_perm  = left.permute(0,3,1,2)   # [B, H, N, N]
      right_perm = right.permute(0,3,1,2)  # [B, H, N, N]
      out_perm   = torch.matmul(left_perm, right_perm.transpose(-1,-2))
      out        = out_perm.permute(0,2,3,1)   # [B,N,N,H]
5. Second Layer‑norm over the hidden dimension, multiply by out_gate and
   project back to the original dimension with a final linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,                # *float32
    right_ptr,               # *float32
    left_gate_ptr,           # *float32
    right_gate_ptr,          # *float32
    mask_ptr,                # *float32 (optional)
    out_left_ptr,            # *float32
    out_right_ptr,           # *float32
    total_elements: tl.int64,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Fuse mask, left‑gate and right‑gate multiplication."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values (out of bounds loads return 0.0 because of mask)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    else:
        mask_val = tl.full((BLOCK,), 1.0, tl.float32)

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask_tensor: torch.Tensor, shape [B, N, N] (may be all‑ones)
            - weights_dict: dict of torch.Tensors with keys matching the
              module definition.
            - config_dict: dict with at least "dim", "hidden_dim",
              and "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]  # (dim,)
    norm_b = weights["norm.bias"]    # (dim,)

    # mean / var over the channel dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b  # broadcast over B,N,N

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = F.linear(x_norm, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_w) # [B,N,N,hidden_dim]

    left_gate_w = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"] # (hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"]     # (hidden_dim, dim)

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ------------------------------------------------------------------
    # 3) Fuse mask & gate multiplication with Triton
    # ------------------------------------------------------------------
    # Prepare mask (broadcast to hidden_dim) if needed
    if nomask:
        has_mask_flag = 0
        mask_expanded = torch.empty(0, dtype=dtype, device=device)  # dummy
    else:
        has_mask_flag = 1
        # mask shape [B,N,N] -> [B,N,N,1] -> broadcast to hidden_dim
        mask_expanded = mask_tensor.to(dtype).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # Allocate output buffers
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    total_elements = left.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total_elements, meta["BLOCK"]),)

    _mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expanded,
        out_left_ptr=left_out,
        out_right_ptr=right_out,
        total_elements=total_elements,
        BLOCK=BLOCK,
        HAS_MASK=has_mask_flag,
        num_warps=4,
    )
    # Synchronize to make sure kernel finishes before next ops
    torch.cuda.synchronize()

    # Replace with gated & masked tensors
    left = left_out
    right = right_out

    # ------------------------------------------------------------------
    # 4) Core O(N³) multiplication: left @ right^T per hidden dim
    # ------------------------------------------------------------------
    # Permute to [B, H, N, N] for batched gemm
    left_perm = left.permute(0, 3, 1, 2)   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2) # B, H, N, N

    # Batched matrix multiplication (B*H independent GEMMs)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # B, H, N, N

    # Back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5) Second LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"]    # (hidden_dim,)

    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply out_gate (same shape as out_norm)
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"]  # (dim, hidden_dim)
    output = F.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return output
