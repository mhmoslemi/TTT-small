def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Extract weight tensors (assumed on same device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over channel dim
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=1e-5)

    # Linear projections (float32)
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (bs, seq_len, seq_len, hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if any (mask shape bs, seq_len, seq_len)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape (bs, seq_len, seq_len, 1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output fp16
    out_fp16 = torch.empty_like(left_fp16)  # same shape

    # Compute strides
    stride_batch, stride_i, stride_k, stride_d = left_fp16.stride()
    # stride_d should be 1
    # We'll ignore stride_d in kernel

    # Choose block sizes adaptively based on seq_len
    # Use 64 for large seq_len else maybe 32 for small
    if seq_len <= 128:
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32
    elif seq_len <= 256:
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

    # Compute grid dimensions
    grid = ( (seq_len + BLOCK_M - 1) // BLOCK_M,
             (seq_len + BLOCK_N - 1) // BLOCK_N,
             bs * hidden_dim)

    trimul_outgoing_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        bs, seq_len, hidden_dim,
        stride_batch, stride_i, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Cast back to float32
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm and out_gate
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul outgoing kernel (AlphaFold3 style) – fast batched matrix‑multiply
======================================================================

The heavy part of the TriMul forward pass is the einsum

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix multiplication (L @ Rᵀ) for every hidden
dimension *d*.  This implementation:

* Performs LayerNorms and linear projections in PyTorch (float32).
* Casts the projected tensors to fp16 and runs a custom Triton kernel
  that computes the batched mat‑mul with a 3‑D grid:
    - program_id(0) → i‑tile
    - program_id(1) → j‑tile
    - program_id(2) → flattened (batch, hidden_dim) index
* The kernel tiles the N×N matrices with configurable BLOCK sizes,
  accumulates in fp32, and writes back in fp16.
* The result is cast back to float32, followed by the remaining
  LayerNorm, gating and output linear projection (all in PyTorch).

The whole function conforms to the required ``custom_kernel`` entry point.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *fp16   [B, N, N, H]
    right_ptr,         # *fp16   [B, N, N, H]
    out_ptr,           # *fp16   [B, N, N, H]
    B, N, H,           # int32 dimensions
    stride_batch,      # int64 stride for batch dimension
    stride_i,          # int64 stride for first N dimension (i / j)
    stride_k,          # int64 stride for second N dimension (k)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)            # block index along dimension i
    pid_j = tl.program_id(1)            # block index along dimension j
    pid_z = tl.program_id(2)            # flattened (batch, hidden) index

    # ---- unpack batch and hidden index ------------------------------------
    b = pid_z // H                      # batch id
    h = pid_z % H                       # hidden (head) id

    # ---- compute offsets for this program ---------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i_offset < N
    j_valid = j_offset < N

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_batch + h
    right_base = right_ptr + b * stride_batch + h
    out_base   = out_ptr   + b * stride_batch + h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- tiled K‑loop ------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < N

        # left tile:      shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] * stride_i + k_offset[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        #   we need right[j, k] → stored as right[b, j, k, h]
        right_ptrs = right_base + k_offset[:, None] * stride_k + j_offset[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=k_valid[:, None] & j_valid[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # fp16 × fp16 → fp32 accumulate

    # ---- write out ---------------------------------------------------------
    out_ptrs = out_base + i_offset[:, None] * stride_i + j_offset[None, :] * stride_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
              - mask: torch.Tensor [B, N, N] (float32 or bool, on CUDA)
              - weights: dict of Tensors containing all linear / LN parameters
              - config: dict with keys "dim" and "hidden_dim" (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # weight tensors
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ----------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4. Gating (float32)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Core O(N³) multiplication via Triton (fp16)
    # ----------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)  # same shape [B,N,N,hidden_dim]

    # strides of a contiguous tensor [B, N, N, H]
    stride_batch, stride_i, stride_k, _ = left_fp16.stride()  # (..., H) stride = 1

    # Choose tile size (tuned for the given N)
    if N <= 128:
        BLOCK_M = BLOCK_N = BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to float32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear (float32)
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
