def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # epsilon
    eps = 1e-5

    # LayerNorm over dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present (mask shape [B,N,N]) else skip
    if mask is not None:
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton kernel
    out = triton_tri_mul(left, right, bs, seq_len, hidden_dim)  # to define

    # Output normalization
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    # Final projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

```
@triton.jit
def trilmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and hidden dimension
    b = pid0 // H
    h = pid0 % H

    # Compute block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Create pointers base for b and h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Mask for valid i/j within N dimension
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < N

        # Pointers to left tile (i,k)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Pointers to right tile (k,j)
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # Store results
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton.

The module performs:
1. LayerNorm over the channel dimension.
2. Linear projections to a hidden dimension (left/right) and corresponding sigmoidal gates.
3. Optional masking of the projected tensors.
4. A dense pairwise update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ for every hidden channel d.
   This O(N³) step is executed by a custom Triton kernel.
5. LayerNorm on the hidden dimension, multiplication by an outgoing gate and a final linear
   projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ  (per batch and per hidden channel)
# ----------------------------------------------------------------------------- #
@triton.jit
def _trilmul_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    out_ptr,             # *float32
    B, N, H,             # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)               # combined batch * hidden
    pid_i = tl.program_id(1)               # block row   (i)
    pid_j = tl.program_id(2)               # block col   (j)

    # Decode batch index (b) and hidden channel (h)
    b = pid0 // H
    h = pid0 % H

    # Starting coordinates of the current block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within the block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers that already contain batch and hidden offsets
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Masks for out‑of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)  -- indexed by (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile:  shape (BLOCK_K, BLOCK_N)  -- indexed by (k, j)
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Accumulate the dot product
        acc += tl.dot(left_tile, right_tile)

    # Store the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def _trilmul_torch(left: torch.Tensor, right: torch.Tensor,
                   block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    `left` and `right` are expected to be contiguous tensors of shape
    [B, N, N, H] and dtype float32.
    Returns a tensor of the same shape containing the batched matrix product
    left @ rightᵀ per hidden channel.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides (int64) – PyTorch tensors are row‑major (contiguous)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    grid = (B * H,
            (N + block_m - 1) // block_m,
            (N + block_n - 1) // block_n)

    _trilmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,   # sR_i plays the role of stride over the “j” dimension
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------------- #
# Public entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Performs the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (or None)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict containing at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs and model parameters
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm parameters (over 'dim')
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    # Projection weights
    left_proj_w   = weights["left_proj.weight"].to(device, dtype)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype)

    # Output norm + projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)
    to_out_w      = weights["to_out.weight"].to(device, dtype)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # --------------------------------------------------------------------- #
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 3. Optional masking (mask is broadcast over the hidden dim)
    # --------------------------------------------------------------------- #
    if not config.get("nomask", False) and mask is not None:
        mk = mask.to(dtype).unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mk
        right = right * mk

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear transforms of the normalized input)
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core TriMul contraction via Triton (out = left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Ensure contiguous layout for optimal pointer arithmetic
    left  = left.contiguous()
    right = right.contiguous()
    out = _trilmul_torch(left, right,
                         block_m=64, block_n=64, block_k=64)   # shape [B, N, N, H]

    # Release large temporaries early
    del left, right

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm on the hidden dimension
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # --------------------------------------------------------------------- #
    # 7. Multiply by the outgoing gate
    # --------------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 8. Final linear projection back to the original channel dimension
    # --------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
