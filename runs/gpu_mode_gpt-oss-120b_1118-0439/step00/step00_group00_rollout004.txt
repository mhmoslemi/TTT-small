def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    slice_id = tl.program_id(0)  # combined batch*hidden
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    b = slice_id // H
    h = slice_id % H

    # compute offsets
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset_base = 0

    # mask for i and j boundary
    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Compute pointers
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_offset[:, None]*stride_left_i + k_offset[None, :]*stride_left_k
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + k_offset[:, None]*stride_right_k + j_offset[None, :]*stride_right_j

        # load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left_tile, right_tile)

    # apply mask to output (if we need to zero out beyond N)
    # store
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offset[:, None]*stride_out_i + j_offset[None, :]*stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

"""
TriMul (outgoing) forward pass accelerated with Triton.

The heavy part of the operation is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left and right are projected and gated versions of the input.
The contraction is equivalent to a batched matrix multiplication
between left[..., :, :, d] and right[..., :, :, d].T for each hidden
channel d.  This kernel computes that product using a tiled GEMM in
Triton, while the surrounding LayerNorm, linear projections, and
sigmoids are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # sizes: batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    #  Program IDs:
    #   pid0: combined (batch, hidden) slice
    #   pid1: block row   (i)
    #   pid2: block col   (j)
    slice_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # decode batch and hidden indices
    b = slice_id // H
    h = slice_id % H

    # offsets within the N×N matrix
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for this (b,h) slice
    base_l = left_ptr + b * stride_l_b + h * stride_l_h
    base_r = right_ptr + b * stride_r_b + h * stride_r_h
    base_o = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for the current tile
        # left: shape (BLOCK_M, BLOCK_K) -> rows i, cols k
        left_tile_ptr = base_l + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        # right: we need (BLOCK_K, BLOCK_N) where rows are k and cols are j
        right_tile_ptr = base_r + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_j

        # masked loads (out‑of‑bounds entries are zero)
        left_tile = tl.load(left_tile_ptr,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)  # (M,K)·(K,N) -> (M,N)

    # write the result back, respecting matrix bounds
    out_tile_ptr = base_o + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_tile_ptr, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version) with a Triton
    kernel for the inner N×N contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of model parameters (see keys below)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         ``nomask`` (bool) indicates whether a mask should be applied.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N]  ->  [B, N, N, 1] for broadcasting
        m = mask.to(dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half precision before the heavy contraction (saves bandwidth)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Output of the contraction lives in float32
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_fp16, right_fp16, out_hidden,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm, output gate, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out_hidden,
                            (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    out_gated = out_norm * out_gate   # element‑wise gate

    # Final linear projection back to original dimension
    output = F.linear(out_gated, to_out_weight)   # shape [B, N, N, dim]

    return output
