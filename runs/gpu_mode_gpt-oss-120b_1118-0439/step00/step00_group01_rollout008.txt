@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    b = tl.program_id(0) # batch index (flattened B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute block offsets for output
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # load left block
        left_ptrs = left_ptr + b * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0).to(tl.float32)
        # load right block (note right shape (batch, K, N)), we need k, n
        right_ptrs = right_ptr + b * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0).to(tl.float32)
        acc += tl.dot(left, right)
    # write output
    out_ptrs = out_ptr + b * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    # store only if within bounds
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask)

"""
Custom Triton kernel implementation of the outgoing TriMul operator.
The kernel computes the batched matrix multiplication required for
the einsum ``out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]``.
The surrounding linear projections, gating, and layer‑norms are
performed with PyTorch for simplicity, while the heavy O(N³) inner
product is fused in a single Triton kernel.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the TriMul einsum.

    left  : [batch, M, K]
    right : [batch, K, N]
    out   : [batch, M, N]
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the absolute coordinate of the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_idx * stride_left_batch \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # Load a tile of right: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back the result.
    c_ptrs = out_ptr + batch_idx * stride_out_batch \
             + offs_m[:, None] * stride_out_m \
             + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc, mask=(mask_m[:, None] & mask_n[None, :]))


def tri_mul_matmul(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Perform the core TriMul multiplication using the Triton kernel.

    Args:
        left:  Tensor of shape [B, N, N, H] (float32).
        right: Tensor of shape [B, N, N, H] (float32).
        BLOCK_*: Tile sizes for the kernel.

    Returns:
        Tensor of shape [B, N, N, H] (float32).
    """
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) so that the hidden dimension becomes a batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two axes of right so that we get (B, H, K, N).
    right_T = right_perm.transpose(2, 3).contiguous()   # (B, H, N, N) -> (B, H, N, N) with axes swapped

    # Collapse the batch and hidden dimensions.
    left_flat = left_perm.view(-1, N, N)   # (B*H, M=N, K=N)
    right_flat = right_T.view(-1, N, N)   # (B*H, K=N, N)

    batch = left_flat.shape[0]

    out_flat = torch.empty_like(left_flat)  # (B*H, N, N)

    # Strides for the three tensors.
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to original layout.
    out_perm = out_flat.view(B, H, N, N)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict mapping weight names to torch.Tensor
            - config: dict containing configuration (dim, hidden_dim, ...)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean / var over the last dim (dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # ----- Linear projections -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # ----- Apply mask (if present) -------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # broadcast to hidden dim
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gating -------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core TriMul multiplication (batched GEMM) --------------------------
    out = tri_mul_matmul(left, right)

    # ----- Output LayerNorm ---------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ----- Apply output gate --------------------------------------------------
    out_norm = out_norm * out_gate

    # ----- Final linear projection -------------------------------------------
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)

"""
Outgoing TriMul operator with a fused Triton kernel.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left_{b,d} and right_{b,d} are (N×N) matrices.
The kernel below computes this GEMM for every (batch, hidden) pair in a
single Triton launch. All surrounding LayerNorms, linear projections,
and gating are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the TriMul einsum.

    left  : [batch, M, K]   (float32)
    right : [batch, K, N]   (float32)
    out   : [batch, M, N]   (float32)
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block coordinates.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = left_ptr + batch_idx * stride_left_batch \
                 + offs_m[:, None] * stride_left_m \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        b_ptrs = right_ptr + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = out_ptr + batch_idx * stride_out_batch \
             + offs_m[:, None] * stride_out_m \
             + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc, mask=(mask_m[:, None] & mask_n[None, :]))


def _tri_mul_matmul(left: torch.Tensor, right: torch.Tensor,
                    BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Core TriMul multiplication using the Triton kernel.

    Args:
        left : Tensor [B, N, N, H] (float32)
        right: Tensor [B, N, N, H] (float32)

    Returns:
        Tensor [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape

    # (B, H, N, N) – make hidden dimension a batch dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Right needs its last two axes swapped so that we can multiply
    #   left[i,k] * right_T[k,j] .
    right_T = right_perm.transpose(2, 3).contiguous()   # (B, H, N, N) → (B, H, K, N)

    # Collapse the batch and hidden dimensions.
    left_flat = left_perm.view(-1, N, N)          # (B*H, M=N, K=N)
    right_flat = right_T.view(-1, N, N)           # (B*H, K=N, N)

    batch = left_flat.shape[0]

    out_flat = torch.empty_like(left_flat)        # (B*H, N, N)

    # Strides for the three tensors (all contiguous → simple strides).
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H).
    out_perm = out_flat.view(B, H, N, N)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor [B, N, N, dim]
            - mask        : Tensor [B, N, N] (may be None)
            - weights     : dict of model weights
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias                     # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3) Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # broadcast to hidden dim
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core TriMul multiplication (batched GEMM)
    # --------------------------------------------------------------
    out = _tri_mul_matmul(left, right)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # --------------------------------------------------------------
    # 7) Output gate
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate

    # --------------------------------------------------------------
    # 8) Final linear projection (hidden → dim)
    # --------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return output
