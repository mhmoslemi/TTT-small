"""
TriMul outgoing kernel for AlphaFold3 implemented with Triton.

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections → left, right (hidden_dim) and three sigmoidal gates.
3. Apply mask and gates to left/right.
4. Core computation: for each channel d,
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication left @ rightᵀ per channel.
   Implemented as a tiled GEMM in Triton (BLOCK_M×BLOCK_N output tile,
   BLOCK_K inner reduction, BLOCK_D channels per block).
5. Layer‑norm the hidden result, multiply by out_gate and project back to dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_kernel(
    left_ptr,          # half*  [B, N, N, C] (i, k, c)
    right_ptr,         # half*  [B, N, N, C] (j, k, c) – we load transposed (k, j)
    out_ptr,           # float* [B, N, N, C] (i, j, c)
    B, N, C,
    stride_lb, stride_li, stride_lk, stride_lc,
    stride_rb, stride_rj, stride_rk, stride_rc,
    stride_ob, stride_oi, stride_oj, stride_oc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]"""
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_c = tl.program_id(2)

    # ----- tile offsets -------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    num_c_tiles = tl.cdiv(C, BLOCK_D)
    batch = pid_c // num_c_tiles
    c_tile = pid_c % num_c_tiles
    c0 = c_tile * BLOCK_D

    # ----- range vectors ------------------------------------------------
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    d = tl.arange(0, BLOCK_D)

    # ----- bounds -------------------------------------------------------
    i_valid = i0 + i < N
    j_valid = j0 + j < N
    c_valid = c0 + d < C

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- main k‑loop --------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k0 = kk * BLOCK_K
        k_valid = (k0 + k) < N

        # load left tile: shape (M, K, D)
        left_ptrs = (
            left_ptr
            + batch * stride_lb
            + (i0 + i[:, None, None]) * stride_li
            + (k0 + k[None, :, None]) * stride_lk
            + (c0 + d[None, None, :]) * stride_lc
        )
        left_mask = i_valid[:, None, None] & k_valid[None, :, None] & c_valid[None, None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # load right tile transposed (K, N, D) -> we treat it as (K, N) per channel
        right_ptrs = (
            right_ptr
            + batch * stride_rb
            + (k0 + k[:, None, None]) * stride_rk
            + (j0 + j[None, :, None]) * stride_rj
            + (c0 + d[None, None, :]) * stride_rc
        )
        right_mask = k_valid[:, None, None] & j_valid[None, :, None] & c_valid[None, None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ----- channel‑wise dot product ----------------------------------
        for di in range(0, BLOCK_D):
            left_mat = left_tile[:, :, di]      # (M, K)
            right_mat = right_tile[:, :, di]    # (K, N)
            acc[:, :, di] += tl.dot(left_mat, right_mat)
        # end channel loop
    # end k‑loop

    # ----- store ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_ob
        + (i0 + i[:, None, None]) * stride_oi
        + (j0 + j[None, :, None]) * stride_oj
        + (c0 + d[None, None, :]) * stride_oc
    )
    store_mask = i_valid[:, None, None] & j_valid[None, :, None] & c_valid[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor   [B, N, N, dim]
            mask         : torch.Tensor   [B, N, N]   (bool/float)
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ---------- Mask handling ----------
    if nomask:
        mask_tensor = torch.ones((B, N, N), device=device, dtype=x_norm.dtype)
    else:
        mask_tensor = mask.to(x_norm.dtype)
    mask_exp = mask_tensor.unsqueeze(-1)  # [B, N, N, 1]

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # proj: [B,N,N,hidden_dim]
    left = F.linear(x_norm, left_proj_w)          # (hidden_dim, dim)
    right = F.linear(x_norm, right_proj_w)

    # apply mask
    left = left * mask_exp
    right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the GEMM kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ---------- Triton kernel launch ----------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # block sizes (tuned for A100/B200, but work for others)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32  # channels per block

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((hidden_dim + BLOCK_D - 1) // BLOCK_D),
    )

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lc = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rc = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oc = out.stride()

    outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lc,
        s_rb,
        s_rj,
        s_rk,
        s_rc,
        s_ob,
        s_oi,
        s_oj,
        s_oc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ---------- Post‑processing ----------
    # hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # apply out_gate
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, to_out_w)

    return out
