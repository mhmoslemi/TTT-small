def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # LayerNorm (norm)
    # Compute mean and var across last dim (dim)
    # input shape: (B, N, N, dim)
    B, N, _, C = input_tensor.shape
    x = input_tensor

    # Compute mean and var
    # Using torch.mean and var (unbiased=False)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if mask is None:
        mask_exp = None
    else:
        mask = mask.to(x.dtype)  # maybe float
        mask_exp = mask.unsqueeze(-1)  # shape (B,N,N,1)

    # Apply mask if needed
    if mask_exp is not None:
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # At this point left and right have shape (B,N,N,hidden_dim)
    # Compute out using Triton kernel:
    #   out = left_i_k * right_j_k^T sum over k
    B = x.shape[0]
    # Permute for kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, D, N, N) transposed second-last dims (k,j)

    # Flatten batch*hidden_dim as M
    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)
    out_flat = torch.empty_like(left_flat)  # same dtype

    # setup strides
    stride_aa = left_flat.stride(0)  # batch stride (M)
    stride_am = left_flat.stride(1)  # row stride (i)
    stride_ak = left_flat.stride(2)  # col stride (k)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # k dimension (since it's rows in B_T)
    stride_bn = right_flat.stride(2)  # N dimension

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)  # row i stride
    stride_cn = out_flat.stride(2)  # col j stride

    # Determine grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             M )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ak, stride_aa,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cc,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape (B,N,N,hidden_dim)

    # Apply to_out_norm (layernorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Compute layernorm per hidden_dim
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_normed = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_normed = out_normed * to_out_norm_weight + to_out_norm_bias

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    # shape out_normed: (B,N,N,hidden_dim)
    out_final = torch.nn.functional.linear(out_normed, to_out_weight)  # shape (B,N,N,dim)

    return out_final

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections:
   * left  = x @ left_proj.weight.T
   * right = x @ right_proj.weight.T
3. Optional pairwise mask (B,N,N) is broadcast to [B,N,N,1] and applied
   to both `left` and `right`.
4. Compute element‑wise gates (sigmoid of linear layers) and apply them:
   left  = left  * left_gate
   right = right * right_gate
5. Core O(N³) update:
     out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of
   `left` (N×N) with the transposed `right` (N×N) for each hidden
   channel.  The multiplication is performed by a custom Triton kernel.
6. Layer‑norm over the hidden dimension, gated by `out_gate`.
7. Final linear projection back to the original channel dimension.

The Triton kernel processes a batch of matrices in parallel:
   - program ids (pid_x, pid_y, pid_m) index output tile (i‑block, j‑block)
     and the matrix index `m = b * hidden_dim + d`.
   - Inside the kernel we iterate over the K dimension in tiles, load
     sub‑blocks with `tl.load`, accumulate with `tl.dot` and finally
     `tl.store` the result.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ak, stride_aa,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batch GEMM: C = A @ B  (A,B shape: [M, N, N])"""
    pid_m = tl.program_id(2)   # matrix index (batch * hidden_dim)
    pid_i = tl.program_id(0)   # block row index
    pid_j = tl.program_id(1)   # block col index

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointer for the selected matrix
    a_batch = a_ptr + pid_m * stride_aa
    b_batch = b_ptr + pid_m * stride_bb
    c_batch = c_ptr + pid_m * stride_cc

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Pointers for A and B sub‑tiles
        a_ptrs = a_batch + (offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn)

        # Load tiles (zero‑pad OOB)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store result tile
    c_ptrs = c_batch + (offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul outgoing forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with required weight tensors
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, on same device as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast to ...1)
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul (batched GEMM) using Triton
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # Prepare tensors for the kernel: shape (B, hidden_dim, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, D, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()       # (B, D, N, N)   (transposed inner N)

    # Flatten batch and hidden dim into a single matrix batch dimension
    M = B * hidden_dim
    left_flat = left_t.view(M, N, N)
    right_flat = right_t.view(M, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides in element units
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tunable block sizes (empirically good for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ak, stride_aa,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
```
