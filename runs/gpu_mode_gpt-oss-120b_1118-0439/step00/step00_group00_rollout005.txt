"""
TriMul forward implementation with a custom Triton kernel.

The kernel computes the core tensor contraction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the projected and gated pairwise
representations of shape [B, N, N, H] (H = hidden_dim).

The surrounding operations (LayerNorm, linear projections,
gates, final normalization and projection) are performed with
standard PyTorch ops.  The heavy O(N³·H) contraction is
fused in a Triton kernel that processes each (batch, head) slice
as an independent matrix‑multiply (left @ rightᵀ) using tiling.
All computations are in float32 for numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh, # left strides
    stride_rb, stride_rj, stride_rk, stride_rh, # right strides
    stride_ob, stride_oi, stride_oj, stride_oh, # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul contraction.

    For each (b, h) we compute:
        out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    using tiling over M = i, N = j, K = k.
    """
    pid = tl.program_id(0)            # combined batch*head index
    pid_i = tl.program_id(1)          # tile row (i)
    pid_j = tl.program_id(2)          # tile col (j)

    # decompose combined index into batch and head
    b = pid // H
    h = pid % H

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # tile start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range vectors within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed: we need right_T of shape (BLOCK_K, BLOCK_N)
        # right original layout: [b, j, k, h]; we want [k, j]
        right_ptrs = right_base + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # matrix multiply for this K‑tile
        acc += tl.dot(left, right)

    # write back the result tile
    out_ptrs = out_base + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel
    for the heavy tensor contraction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim) – bias=False in the reference
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask handling
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor
    # ensure mask shape [B,N,N,1] for broadcasting
    mask = mask.unsqueeze(-1).to(x.dtype)
    left  = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid activations)
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul contraction via Triton
    # Prepare tensors in contiguous layout (should already be)
    left  = left.contiguous()
    right = right.contiguous()

    # Output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements) for each tensor
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # modest warp count; can be tuned
    )

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden_dim, gating and output projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate  # broadcast over hidden_dim

    to_out_weight = weights["to_out.weight"]   # shape (dim, hidden_dim)
    out = F.linear(out, to_out_weight)        # final projection to `dim`

    return out
