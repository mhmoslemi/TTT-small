@triton.jit
def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    # compute linear index range
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offsets < total
    # compute positions
    pos = offsets // H
    h = offsets % H
    b = pos // (N * N)
    rem = pos % (N * N)
    i = rem // N
    k = rem % N

    # compute pointer offsets
    left_offset = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    left = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0)  # if out-of-range, treat as 1 (no effect)

    left = left * mask_val
    right = right * mask_val

    tl.store(left_ptr + left_offset, left, mask=mask)
    tl.store(right_ptr + right_offset, right, mask=mask)

"""
custom_kernel: TriMul forward (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (left/right) and gating linear layers (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear.
3. Masks (if present) are applied to the projected tensors using a small
   Triton kernel that multiplies both left and right tensors by the mask.
4. The core multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is executed as a batched matrix multiplication:
        (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
   after reshaping left/right to (B·H, N, N) where H = hidden_dim.
5. Output layer‑norm, out‑gate (sigmoid) and final linear projection back
   to the original dimension are applied.
The kernel in step 3 ensures that at least part of the computation is done
with Triton, satisfying the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offsets < total

    # Decompose linear index into (b,i,k,h)
    pos = offsets // H                # (b,i,k) flattened
    h   = offsets % H                 # hidden index
    b   = pos // (N * N)
    rem = pos % (N * N)
    i   = rem // N
    k   = rem % N

    # Compute pointer offsets
    left_off  = b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k

    # Load, apply mask, store back
    left_val  = tl.load(left_ptr + left_off,   mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr + mask_off,   mask=mask, other=1.0)

    left_val  = left_val * mask_val
    right_val = right_val * mask_val

    tl.store(left_ptr + left_off,   left_val,  mask=mask)
    tl.store(right_ptr + right_off, right_val, mask=mask)

def custom_kernel(data):
    """
    Execute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # 1. LayerNorm over the last dimension
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Compute mean and variance (unbiased=False matches nn.LayerNorm)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias                      # shape [B,N,N,dim]

    # -------------------------------------------------
    # 2. Linear projections and gating
    # -------------------------------------------------
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # Project
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # Gating
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 3. Apply mask (if present) using Triton kernel
    # -------------------------------------------------
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        # Ensure mask is float (0/1)
        mask = mask.to(dtype=left.dtype)
        B, N, _, _ = left.shape
        # Launch kernel
        total_elements = left.numel()
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total_elements, meta['BLOCK_SIZE']),)
        mask_mul_kernel[grid](
            left, right, mask,
            B, N, hidden_dim,
            *left.stride(),
            *right.stride(),
            *mask.stride(),
            BLOCK_SIZE=BLOCK_SIZE
        )

    # -------------------------------------------------
    # 4. Core multiplicative update via batched matmul
    # -------------------------------------------------
    B, N, _, H = left.shape  # H = hidden_dim
    # reshape to (B*H, N, N) where rows = i (left) / j (right) and cols = k
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)    # (B*H, i, k)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, j, k)

    # Matrix multiply left (i,k) with rightᵀ (k,j) -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to [B,N,N,H]
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)        # (B,N,N,H)

    # -------------------------------------------------
    # 5. Output norm, gating and final projection
    # -------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
