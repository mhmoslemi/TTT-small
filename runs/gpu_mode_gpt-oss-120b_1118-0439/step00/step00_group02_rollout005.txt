import torch.nn.functional as F

def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    # config contains 'dim' and 'hidden_dim', also maybe 'nomask'
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # layernorm weight bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Proj weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # to_out_norm weight,bias
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # final linear weight
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm on last dim
    # Use F.layer_norm
    x_norm = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # 2. Linear projections
    left = F.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # 3. Apply mask if present
    if mask is not None:
        mask_uns = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_uns
        right = right * mask_uns

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Transpose right for multiplication
    right_t_fp16 = right_fp16.transpose(1,2)  # swap the two N axes

    # Permute to bring hidden dim to second dimension (batch, hidden, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_t_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # N == second N
    batch_hidden = B * H

    # reshape to (batch_hidden, N, N)
    A = left_perm.view(batch_hidden, N, N)
    B_mat = right_perm.view(batch_hidden, N, N)

    # Allocate output float32
    out_mat = torch.empty_like(A, dtype=torch.float32)

    # Launch Triton kernel
    _batched_matmul_kernel[A.shape[0]](
        A,
        B_mat,

        out_mat.stride(1), out_mat.stride(2),  # stride_cm, stride_cn

        num_warps=4,

"""
TriMul forward (outgoing) implementation with Triton.
---------------------------------------------------
The heavy part of the TriMul operator is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication:
    left₍b,d₎ ∈ ℝ^{N×N},  right₍b,d₎ ∈ ℝ^{N×N}
    out₍b,d₎ = left₍b,d₎ @ right₍b,d₎ᵀ

We flatten the batch and hidden dimensions (B·H) and launch a
standard Triton GEMM kernel (BLOCK_M×BLOCK_N tiles, BLOCK_K inner
tiles).  The kernel works on fp16 inputs and accumulates in fp32,
producing fp32 results.

All other operations (LayerNorm, linear projections, gates, final
normalisation and projection) are performed with PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_bb, stride_cb,       # batch stride for A, B, C
    stride_am, stride_ak,                 # A row / col stride
    stride_bk, stride_bn,                 # B row (K) / col (N) stride
    stride_cm, stride_cn,                 # C row / col stride
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers with correct strides and batch offset
        a_ptrs = a_ptr + pid_batch * stride_ab + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + pid_batch * stride_bb + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Guarded loads (mask out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_ptr + pid_batch * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask_store = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim]   (float32)
            - mask:  torch.Tensor [B, N, N]       (float32 or bool)
            - weights: dict of torch.Tensor containing all model weights
            - config: dict with keys "dim", "hidden_dim", possibly "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------- unpack -------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device

    # -------- layer norm on dim --------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # -------- linear projections --------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # -------- mask (optional) ----------------------------------------
    if mask is not None and mask.numel() > 0:
        # ensure same dtype as left/right
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------- gates ---------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------- convert to fp16 for GEMM -------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Transpose right on the two sequence axes (i ↔ k)
    right_t_fp16 = right_fp16.transpose(1, 2)   # shape [B,N,N,hidden]

    # Bring hidden dimension to the second axis for batched GEMM
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right_t_fp16.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Flatten batch+hidden into a single batch dimension
    A = left_perm.view(batch_hidden, N, N)   # fp16
    B_mat = right_perm.view(batch_hidden, N, N)  # fp16

    # Output buffer (fp32 accumulation)
    out_mat = torch.empty_like(A, dtype=torch.float32, device=device)

    # ----- launch Triton GEMM kernel --------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        A, B_mat, out_mat,
        A.stride(0), B_mat.stride(0), out_mat.stride(0),
        A.stride(1), A.stride(2),
        B_mat.stride(1), B_mat.stride(2),
        out_mat.stride(1), out_mat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -------- reshape back to [B, N, N, hidden] -----------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]

    # -------- final LayerNorm, gating, and projection ----------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                      # element‑wise gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
