def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N2, C = input_tensor.shape
    assert N == N2

    # LayerNorm on dim (C)
    eps = config.get("layernorm_eps", 1e-5)
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (C,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Compute left/right proj
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask (if provided)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    else:
        mask_ = None

    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # compute triple product using Triton
    out = triton_compute(left, right, mask)  # But we already applied mask to left/right; we don't need extra mask for output

    # After kernel, out shape [B,N,N,H], maybe as float32
    # Normalize out (to_out_norm)
    out_norm = torch.nn.functional.layer_norm(out, (config["hidden_dim"],), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply output gate:
    out_norm = out_norm * out_gate

    # Final linear projection
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

def batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_abc, stride_ai, stride_ak, stride_ah,
    stride_bbc, stride_bi, stride_bk, stride_bh,
    stride_cc, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # a: [BC, N, K] (where K = N)
    # b: [BC, N, K] (original right matrix). We will treat as B_T within kernel
    # c: [BC, N, N] output
    bc = tl.program_id(axis=0) # combined batch channel index
    pid_m = tl.program_id(axis=1) # row block
    pid_n = tl.program_id(axis=2) # col block

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute if within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to a and b base for batch channel
    a_batch_ptr = a_ptr + bc * stride_abc
    b_batch_ptr = b_ptr + bc * stride_bbc

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (transposed!). Original B: shape (N, N) where axes = (j, k)
        # We need tile B_T[ k, j ] i.e., B[ j, k ]
        # Offsets: rows = k (original column dimension), cols = j (original row)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi) # Wait stride_bk is stride for original k dimension (col). Need compute pointers correctly.
        # Actually we need pointer offset = j * stride_? + k * stride_?; original B[bc, j, k] -> offset = j*stride_bi + k*stride_bk.
        # For B_T, we want entries B[bc, j, k] but we treat them as shape (k, j); so we load with (offs_k, offs_n). So pointer = bc_base + offs_n * stride_bi + offs_k * stride_bk.
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    c_ptrs = c_ptr + bc * stride_cc + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # Normalization
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # shape [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze for broadcasting
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    else:
        mask_ = torch.ones_like(left) # all ones

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication using Triton kernel
    out = triton_matmul_outgoing(left, right, B, N, hidden_dim)

    # Output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def triton_matmul_outgoing(left,right, B,N,H):
    # left/right shape [B,N,N,H]
    # flatten bc dims
    left_contig = left.permute(0,3,1,2).contiguous()
    right_contig = right.permute(0,3,1,2).contiguous()
    BC = B * H
    left_flat = left_contig.view(BC, N, N)
    right_flat = right_contig.view(BC, N, N)

    out_flat = torch.empty_like(left_flat)  # shape [BC, N, N]

    # Get strides
    a_stride_bc, a_stride_i, a_stride_k = left_flat.stride()
    b_stride_bc, b_stride_i, b_stride_k = right_flat.stride()
    c_stride_bc, c_stride_i, c_stride_j = out_flat.stride()

    # Kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BC,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        B=BC,
        N=N,
        H=H,
        stride_abc=a_stride_bc,
        stride_ai=a_stride_i,
        stride_ak=a_stride_k,
        stride_bbc=b_stride_bc,
        stride_bi=b_stride_i,
        stride_bk=b_stride_k,
        stride_cc=c_stride_bc,
        stride_ci=c_stride_i,
        stride_cj=c_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back: out shape [B,N,N,H] by view
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

"""
TriMul outgoing update kernel (AlphaFold3 style) using Triton.

The kernel computes the core O(N³) operation:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left/right are the projected and gated pairwise representations.
We flatten the batch and hidden dimensions (B*H) and perform a batched
matrix multiplication of A @ Bᵀ with a custom Triton kernel. All other
operations (LayerNorm, Linear projections, gating) are performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_abc, stride_ai, stride_ak,
    stride_bbc, stride_bi, stride_bk,
    stride_cc, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.

    Shapes (after flattening batch * hidden):
        A: [BC, N, N]   (row-major)
        B: [BC, N, N]   (row-major, used as transposed)
        C: [BC, N, N]

    Parameters
    ----------
    a_ptr, b_ptr, c_ptr : pointers to the flattened tensors.
    B, N, H : int32 scalars (BC = B*H, sequence length N).
    stride_* : int32 strides of the tensors.
    BLOCK_M, BLOCK_N, BLOCK_K : compile‑time tile sizes.
    """
    # program ids
    bc = tl.program_id(axis=0)          # batch*hidden index
    pid_m = tl.program_id(axis=1)       # block row (i)
    pid_n = tl.program_id(axis=2)       # block col (j)

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this BC slice
    a_base = a_ptr + bc * stride_abc
    b_base = b_ptr + bc * stride_bbc
    c_base = c_ptr + bc * stride_cc

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: (M, K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile: we need B[j, k] (original layout) => (K, N)
        b_ptrs = b_base + (offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (M, K) dot (K, N) => (M, N)

    # write back C tile
    c_ptrs = c_base + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _triton_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel for the outgoing triangular update.
    left/right: [B, N, N, H] tensors (float32, contiguous).
    Returns out of shape [B, N, N, H] (float32).
    """
    B, N, _, H = left.shape
    device = left.device

    # Bring hidden dimension to the batch axis for a simple 2‑D matmul view
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides in element counts
    a_stride_bc, a_stride_i, a_stride_k = left_flat.stride()
    b_stride_bc, b_stride_i, b_stride_k = right_flat.stride()
    c_stride_bc, c_stride_i, c_stride_j = out_flat.stride()

    # Tile sizes – chosen to fit nicely into registers/shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        B=B * H,
        N=N,
        H=H,
        stride_abc=a_stride_bc,
        stride_ai=a_stride_i,
        stride_ak=a_stride_k,
        stride_bbc=b_stride_bc,
        stride_bi=b_stride_i,
        stride_bk=b_stride_k,
        stride_cc=c_stride_bc,
        stride_ci=c_stride_i,
        stride_cj=c_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Updated tensor of shape [B, N, N, C] (float32, same device as input).
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        # ensure mask is float (0/1) and broadcastable
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core triangular multiplication (Triton) ----------
    out = _triton_matmul_outgoing(left, right)                # [B,N,N,hidden]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- Apply out gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to C ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
