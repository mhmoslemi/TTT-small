@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Am, stride_Ai, stride_Ak,
                          stride_Bm, stride_Bk, stride_Bj,
                          stride_Cm, stride_Ci, stride_Cj,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0) # id for each M (combination of batch and hidden dim)
    # compute offsets for A and B
    # compute block indices:
    block_i = tl.program_id(axis=1) # row block index
    block_j = tl.program_id(axis=2) # col block index
    # compute offsets for rows and cols within block
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for k in range(0, K, BLOCK_K):
        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid * stride_Am + i[:, None] * stride_Ai + (k + tl.arange(0, BLOCK_K))[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=(i[:, None] < N) & (k + tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        # Load B block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid * stride_Bm + (k + tl.arange(0, BLOCK_K))[:, None] * stride_Bk + j[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=(k + tl.arange(0, BLOCK_K)[:, None] < K) & (j[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # store result
    c_ptrs = C_ptr + pid * stride_Cm + i[:, None] * stride_Ci + j[None, :] * stride_Cj
    tl.store(c_ptrs, acc, mask=(i[:, None] < N) & (j[None, :] < N))

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ai, stride_ak,
                        stride_bm, stride_bk, stride_bj,
                        stride_cm, stride_ci, stride_cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # axes
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1) # block row
    pid_j = tl.program_id(2) # block col

    # compute ranges
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        # read A

        # read B

        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_j[None, :]), other=0.0)
        # accumulate

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ai, stride_ak,
                        stride_bm, stride_bk, stride_bj,
                        stride_cm, stride_ci, stride_cj,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load A (M_i x K)
        a_ptrs = A_ptr + batch_id * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & (offs_k[None, :] < K)), other=0.0)

        # Load B (K x N)
        b_ptrs = B_ptr + batch_id * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_j[None, :]), other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = C_ptr + batch_id * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The core of TriMul is an O(N³) tensor contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension `d` (and each batch) this is exactly a matrix
multiplication L_d @ R_dᵀ where
    L_d[b][i, k] = left[b, i, k, d]
    R_d[b][k, j] = right[b, j, k, d]

We compile a Triton kernel that performs a batched GEMM over the combined
dimension M = B * hidden_dim.  The kernel tiles the output with
BLOCK_M × BLOCK_N blocks and accumulates over K in tiles of size BLOCK_K.
All other operations (LayerNorm, Linear projections, gating and the final
output linear) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C[m] = A[m] @ B[m]   for m in 0..M-1
    A, B, C are interpreted as (M, N, K), (M, K, N), (M, N, N) respectively.
    """
    batch_id = tl.program_id(0)            # 0 .. M-1
    pid_i = tl.program_id(1)               # row block index
    pid_j = tl.program_id(2)               # col block index

    # row / column indices within the output matrix
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # ---- load a block of A (M_i × K) ----
        a_ptrs = a_ptr + batch_id * stride_am \
                 + offs_i[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & (offs_k[None, :] < K)),
                    other=0.0)

        # ---- load a block of B (K × N) ----
        b_ptrs = b_ptr + batch_id * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_j[None, :]),
                    other=0.0)

        # ---- accumulate ----
        acc += tl.dot(a, b)

    # ---- write back ----
    c_ptrs = c_ptr + batch_id * stride_cm \
             + offs_i[:, None] * stride_ci \
             + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim]
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- load weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm over the channel dimension -----
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps)

    # ----- Linear projections -----
    left = F.linear(x, left_proj_weight, None)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight, None)

    # ----- optional mask -----
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- gating signals -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight, None))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight, None))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight, None))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Triton batched GEMM: compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------------
    # Rearrange tensors so that the hidden dimension becomes part of the batch.
    # After permute+contiguous each matrix is stored as (M, N, N)
    # where M = B * hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, D, N, N] (transpose over the two N axes)

    M = B * hidden_dim
    left_mat = left_perm.view(M, N, N)      # (M, N, N)
    right_mat = right_perm.view(M, N, N)    # (M, N, N)   (K×N after transpose)
    out_mat = torch.empty_like(left_mat)    # (M, N, N)

    # strides for the three tensors (they are contiguous)
    stride_am, stride_ai, stride_ak = left_mat.stride()
    stride_bm, stride_bk, stride_bj = right_mat.stride()
    stride_cm, stride_ci, stride_cj = out_mat.stride()

    # tiling parameters – chosen for good occupancy on A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        M, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- output layer norm, gating and final linear -----
    out = F.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight, None)   # final projection back to dim

    return out
