```python
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute batch and head index
    b = pid_h // H
    h = pid_h % H

    # offsets for i and j
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i_offset < N
    j_mask = j_offset < N

    # base pointers for left and right for this batch and head
    left_batch_head_ptr = left_ptr + b * stride_left_batch + h * stride_left_h
    right_batch_head_ptr = right_ptr + b * stride_right_batch + h * stride_right_h
    out_batch_head_ptr = out_ptr + b * stride_out_batch + h * stride_out_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left block [M, K]
        left_ptrs = left_batch_head_ptr + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # Load right block [N, K] (j dimension)
        right_ptrs = right_batch_head_ptr + j_offset[:, None] * stride_right_j + k_offset[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # Compute dot product (M x K) * (N x K)^T => (M, N)
        acc += tl.dot(left, right, trans_b=True)

    # Store result
    out_ptrs = out_batch_head_ptr + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute batch and hidden index
    b = pid_h // H
    h = pid_h % H

    # offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base = out_ptr + b * stride_o_batch + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        left_ptrs = left_base + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        right_ptrs = right_base + j_off[:, None] * stride_r_i + k_off[None, :] * stride_r_k

        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        # convert to fp16 for matmul, accumulate as fp32
        left = left.to(tl.float16)
        right = right.to(tl.float16)

def custom_kernel(data):
    """
    Triton-accelerated forward pass for TriMul (outgoing version).
    Performs layernorm, linear projections, gating, and the core
    pairwise multiplication over the sequence dimension using a
    fused matrix‑multiply kernel.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Ensure contiguous
    x = input_tensor

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ---------- Mask ----------
    if mask is not None:
        # mask may be bool, convert to float
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    right = right * right_gate

"""
TriMul (outgoing version) forward pass with Triton acceleration.

The heavy part is the pairwise multiplication over the sequence dimension:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

We treat each hidden channel `d` (and batch `b`) as an independent matrix
multiply L_d @ R_dᵀ where L_d and R_d are (N×N) matrices.  The Triton
kernel tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and reduces over k in
BLOCK_K chunks, loading the two operands in FP16, accumulating in FP32,
and storing the result in FP32.  The surrounding operations
(LayerNorm, linear projections, sigmoid gates) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid:
        program_id(0) -> i‑tile
        program_id(1) -> j‑tile
        program_id(2) -> combined (b, h) index
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # Decode batch and hidden-channel index
    b = pid_h // H
    h = pid_h % H

    # Offsets for the i and j tiles
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i_off < N
    j_mask = j_off < N

    # Base pointers for the current (b, h)
    left_base = left_ptr + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base = out_ptr + b * stride_o_batch + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # Load a (M×K) tile from left and (N×K) tile from right
        left_ptrs = left_base + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        right_ptrs = right_base + j_off[:, None] * stride_r_i + k_off[None, :] * stride_r_k

        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float16)
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0).to(tl.float16)

        # Multiply‑accumulate: (M×K) • (N×K)ᵀ → (M×N)
        acc += tl.dot(left, right, trans_b=True)

    # Write the result tile
    out_ptrs = out_base + i_off[:, None] * stride_o_i + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32) on the same device.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- Input LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        # mask: [bs, N, N]  ->  [bs, N, N, 1] for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the kernel ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ---------- Triton kernel for the einsum ----------
    out = torch.empty(bs, N, N, hidden_dim, dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    l_batch, l_i, l_k, l_h = left_fp16.stride()
    r_batch, r_i, r_k, r_h = right_fp16.stride()
    o_batch, o_i, o_j, o_h = out.stride()

    # Tile sizes – chosen to fit well on B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        bs * hidden_dim,                       # (batch, hidden) programs
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        bs,
        N,
        hidden_dim,
        l_batch,
        l_i,
        l_k,
        l_h,
        r_batch,
        r_i,
        r_k,
        r_h,
        o_batch,
        o_i,
        o_j,
        o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Apply out gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
