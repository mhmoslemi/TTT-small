def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Input: [B,N,N,dim] dtype float32.
    # Move to GPU? Already.
    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # pack: layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask handling
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # reshape for kernel
    # permute to [B, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # compute right transpose
    right_T = right_flat.transpose(1, 2).contiguous()

    # allocate out
    out_flat = torch.empty_like(left_flat)  # shape [BH,N,N]

    # get strides
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_T.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # define kernel grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # final linear to get dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_la, stride_lm, stride_lk,
    stride_ra, stride_rm, stride_rk,
    stride_oa, stride_om, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids: batch hidden
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    left_base = left_ptr + pid_batch * stride_la
    right_base = right_ptr + pid_batch * stride_ra
    out_base = out_ptr + pid_batch * stride_oa

    # pointers to rows/cols
    left_ptrs = left_base + (offs_m[:, None] * stride_lm) + (offs_k[None, :] * stride_lk)
    right_ptrs = right_base + (offs_k[:, None] * stride_rm) + (offs_n[None, :] * stride_rk)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for k block
        cur_k = k * BLOCK_K
        k_offset = cur_k + offs_k
        # mask
        mask_k = k_offset < K

        # load tiles
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(left_tile, right_tile)
        # increment pointers for next tile
        left_ptrs += BLOCK_K * stride_lk
        right_ptrs += BLOCK_K * stride_rm

    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_om) + (offs_n[None, :] * stride_ok)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_la, stride_li, stride_lk,
    stride_ra, stride_rk, stride_rn,
    stride_oa, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    left_batch = left_ptr + pid_batch * stride_la
    right_batch = right_ptr + pid_batch * stride_ra
    out_batch = out_ptr + pid_batch * stride_oa

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of loops over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        left_tile = tl.load(left_batch + (offs_m[:, None] * stride_li) + (offs_k[None, :] * stride_lk),

    # Ensure mask handling if config.get('nomask', False):
    nomask = config.get('nomask', False)
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm over dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    # Mask
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f

    right_gate = F.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = F.linear(x, weights['out_gate.weight']).sigmoid()

    right = right * right_gate

    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_la = left_flat.stride(0)

    stride_lk = left_flat.stride(2)

    BLOCK_N = 64
    BLOCK_K = 64

"""
TriMul (outgoing) forward implementation.

The heavy O(N³) part:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

is expressed as a batched GEMM:
    out[b, d] = left[b, d] @ right[b, d].T
for each (batch, hidden) pair.  The GEMM is computed by a Triton
kernel with tiling (BLOCK_M, BLOCK_N, BLOCK_K).  All other
operations (LayerNorms, Linear projections, gates, masking) are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float32   [batch_hidden, M, K]
    right_ptr,              # *float32   [batch_hidden, K, N] (implemented via stride swap)
    out_ptr,                # *float32   [batch_hidden, M, N]
    M, N, K,                # compile‑time dimensions (M = N = K = seq_len)
    stride_la, stride_li, stride_lk,   # strides for left (batch, row i, col k)
    stride_ra, stride_rk, stride_rn,   # strides for right (batch, row k, col n)
    stride_oa, stride_oi, stride_oj,   # strides for output (batch, row i, col j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)              # output row tile
    pid_n = tl.program_id(2)              # output col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    left_batch  = left_ptr  + pid_batch * stride_la
    right_batch = right_ptr + pid_batch * stride_ra
    out_batch   = out_ptr   + pid_batch * stride_oa

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps (compile‑time loop)
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_tile = tl.load(
            left_batch
            + (offs_m[:, None] * stride_li)
            + (cur_k[None, :] * stride_lk),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: shape (BLOCK_K, BLOCK_N)
        # note: stride_rk is the stride for the K‑dimension (original column stride)
        #       stride_rn is the stride for the N‑dimension (original row stride)
        right_tile = tl.load(
            right_batch
            + (cur_k[:, None] * stride_rk)
            + (offs_n[None, :] * stride_rn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(left_tile, right_tile)

    # store the result
    out_ptrs = out_batch + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors containing all linear / LN parameters
        - config       : dict with at least keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if present and not disabled)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        m = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections on the *original* normalized input)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplication (batched GEMM) via Triton
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden, N, N] and flatten batch*hidden dimension.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    total = B * H

    left_flat = left.view(total, N, N)     # [B*hidden, N, N]
    right_flat = right.view(total, N, N)   # same layout
    out_flat = torch.empty_like(left_flat) # allocate output

    # Strides (in elements)
    stride_la = left_flat.stride(0)
    stride_li = left_flat.stride(1)   # row i stride (N)
    stride_lk = left_flat.stride(2)   # col k stride (1)

    stride_ra = right_flat.stride(0)
    # For the GEMM we need right as (K, N) -> swap the last two axes via stride tricks
    stride_rk = right_flat.stride(2)   # stride for K (original column stride = 1)
    stride_rn = right_flat.stride(1)   # stride for N (original row stride = N)

    stride_oa = out_flat.stride(0)
    stride_oi = out_flat.stride(1)
    stride_oj = out_flat.stride(2)

    # Tile sizes – chosen to fit typical B200 SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel.  M, N, K are compile‑time constants for Triton.
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_la, stride_li, stride_lk,
        stride_ra, stride_rk, stride_rn,
        stride_oa, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
