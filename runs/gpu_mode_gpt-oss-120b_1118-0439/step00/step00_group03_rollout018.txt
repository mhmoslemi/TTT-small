def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Move to device
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias
    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    # Mask
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Prepare for kernel: permute to (B, H, N, N)
    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()
    # Convert to half for kernel input
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    B, H, N, _ = left_fp16.shape
    # Output buffer
    out_intermediate = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    stride_a_batch = left_fp16.stride(0)
    stride_a_hidden = left_fp16.stride(1)

    stride_b_i = right_fp16.stride(2)
    stride_b_j = right_fp16.stride(3)

"""
TriMul forward kernel (outgoing version) using Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last channel dimension.
2. Linear projections:
   * left  = Linear(x, left_proj.weight)
   * right = Linear(x, right_proj.weight)
3. Apply optional mask (broadcast over the hidden dimension).
4. Compute sigmoid gates and apply them:
   left  *= sigmoid(Linear(x, left_gate.weight))
   right *= sigmoid(Linear(x, right_gate.weight))
5. Rearrange tensors to (B, H, N, N) and cast to half‑precision.
6. Compute the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM `left @ right.T` using a custom Triton kernel.
7. Permute back to (B, N, N, H) and apply a second LayerNorm
   (`to_out_norm`) followed by the output gate (`out_gate`).
8. Final linear projection to the original channel size `dim`.

The heavy O(N³) part (step 6) is fully executed in Triton, while
the remaining point‑wise operations are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_j,
    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B⁽ᵀ⁾   where A,B shape = (B,H,N,N)."""
    pid_i = tl.program_id(0)               # block row
    pid_j = tl.program_id(1)               # block col
    pid_h = tl.program_id(2)               # combined batch‑head index

    # Recover batch and head indices
    b_idx = pid_h // hidden_dim
    h_idx = pid_h - b_idx * hidden_dim

    # Base pointers for this (batch, head)
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_hidden
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_hidden
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_hidden

    # Tile start offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Index vectors (may go out‑of‑bounds, masks will hide them)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_j
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile transposed (Bᵀ): shape (BLOCK_K, BLOCK_N)
        # Bᵀ is accessed with stride_b_j for the k‑dimension and stride_b_i for the j‑dimension
        b_ptrs = b_base + offs_k[:, None] * stride_b_j + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -------- LayerNorm on input --------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # [B, N, N, dim]

    # -------- Linear projections --------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Optional mask ----------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).type_as(left)         # broadcast to hidden dim
        left = left * mask_
        right = right * mask_

    # -------- Gates --------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------- Prepare tensors for Triton GEMM --------
    # Permute to (B, H, N, N) and make them contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Cast to half for better bandwidth
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (float32)
    out_int = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # Strides (int values)
    stride_a_batch = left_fp16.stride(0)
    stride_a_hidden = left_fp16.stride(1)
    stride_a_i = left_fp16.stride(2)
    stride_a_j = left_fp16.stride(3)

    stride_b_batch = right_fp16.stride(0)
    stride_b_hidden = right_fp16.stride(1)
    stride_b_i = right_fp16.stride(2)
    stride_b_j = right_fp16.stride(3)

    stride_c_batch = out_int.stride(0)
    stride_c_hidden = out_int.stride(1)
    stride_c_i = out_int.stride(2)
    stride_c_j = out_int.stride(3)

    # Tiling parameters (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
        B * hidden_dim                         # batch * head
    )

    # Launch Triton kernel
    tri_mul_gemm_kernel[grid](
        left_fp16, right_fp16, out_int,
        N, N, N,
        stride_a_batch, stride_a_hidden, stride_a_i, stride_a_j,
        stride_b_batch, stride_b_hidden, stride_b_i, stride_b_j,
        stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
        hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel finishes before moving on (optional, sync on next torch op)
    torch.cuda.synchronize()

    # -------- Restore original layout --------
    # out_int currently (B, H, N, N) -> permute to (B, N, N, H)
    out = out_int.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # -------- Second LayerNorm (to_out_norm) --------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # -------- Output gate --------
    out_norm = out_norm * out_gate

    # -------- Final linear projection --------
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
