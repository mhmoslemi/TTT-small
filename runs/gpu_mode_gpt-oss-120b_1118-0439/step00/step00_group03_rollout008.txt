@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # Compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Create masks to avoid out-of-bounds loads
    # We'll create mask for M,N dimensions
    # For each tile, will check if off within M,N.

    a_ptr_base = A_ptr + batch_idx * stride_Ab
    b_ptr_base = B_ptr + batch_idx * stride_Bb
    c_ptr_base = C_ptr + batch_idx * stride_Cb

    # Load A and B tiles
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        # mask k within K
        k_in_bounds = cur_k < K
        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr_base + (offs_m[:, None] * stride_Am) + (cur_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & k_in_bounds[None, :], other=0.0)
        # B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr_base + (cur_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)
        b = tl.load(b_ptrs, mask=k_in_bounds[:, None] & (offs_n[None, :] < N), other=0.0)
        # compute dot
        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_ptr_base + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5) # optional

    # retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Input normalization
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask handling
    nomask = config.get("nomask", False)
    if not nomask:
        # mask shape [B,N,N]; unsqueeze last dim
        mask_exp = mask.unsqueeze(-1)   # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base pointers offset by batch
    a_batch_ptr = A_ptr + batch_idx * stride_Ab
    b_batch_ptr = B_ptr + batch_idx * stride_Bb
    c_batch_ptr = C_ptr + batch_idx * stride_Cb

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Mask for out-of-bounds
        m_mask = offs_m < M
        n_mask = offs_n < N
        k_mask = cur_k < K

        # Load A tile (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_Am) + (cur_k[None, :] * stride_Ak)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)

"""
Custom Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The forward pass follows the reference PyTorch implementation:
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and linear gates (left/right/out).
3. Optional mask multiplication.
4. Apply gates to the projected tensors.
5. Compute the core O(N³) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
   C = A @ Bᵀ with A,B ∈ ℝ^{(B·H) × N × N}, H = hidden_dim.
   This step is performed by a fused Triton kernel.
6. Layer‑norm over the hidden dimension, apply the out‑gate,
   and a final linear projection back to the original dimension.
All tensors stay on the GPU, and the heavy O(N³) part is accelerated
by Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,   # batch, row, col strides for A
    stride_Bb, stride_Bk, stride_Bn,   # batch, row(K), col(N) strides for B (already transposed)
    stride_Cb, stride_Cm, stride_Cn,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B, where each batch corresponds to a (M×K)·(K×N) multiply."""
    batch_id = tl.program_id(0)
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + batch_id * stride_Ab
    b_batch_ptr = B_ptr + batch_id * stride_Bb
    c_batch_ptr = C_ptr + batch_id * stride_Cb

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Bounds checks
        m_valid = offs_m < M
        n_valid = offs_n < N
        k_valid = cur_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_Am) + (cur_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=m_valid[:, None] & k_valid[None, :], other=0.0)

        # Load B tile: (BLOCK_K, BLOCK_N)  (B is already transposed)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & n_valid[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out the result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=m_valid[:, None] & n_valid[None, :])

        data: Tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask:        torch.Tensor [B, N, N] (may be unused)
            weights:     dict of model weights
            config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask has shape [B, N, N]; we broadcast on dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM C = A @ Bᵀ
    # A : left, B : right (to be transposed on the last two axes)
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, N, N, hidden_dim
    # bring hidden_dim forward and make memory contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & head dimension: [B*H, N, N]
    left_mat = left_perm.view(B * hidden_dim, N_seq, N_seq)
    right_mat = right_perm.view(B * hidden_dim, N_seq, N_seq)

    # Transpose the last two dimensions of right (so we compute A @ Bᵀ)
    right_t = right_mat.transpose(1, 2).contiguous()

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_cnt = left_mat.shape[0]
    M = N_seq
    N = N_seq
    K = N_seq

    grid = (
        batch_cnt,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        M,
        N,
        K,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_t.stride(0),
        right_t.stride(1),
        right_t.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden dimension, apply out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
