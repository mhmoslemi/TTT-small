    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # dims
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape (B,N,N). Convert to float.
        mask_f = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gate using Triton kernel
    import triton
    import triton.language as tl

    @triton.jit
    def gate_kernel(gated_left_ptr, gated_right_ptr,
                    left_ptr, left_gate_ptr,
                    right_ptr, right_gate_ptr,
                    N_ELEMS,
                    BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < N_ELEMS
        l = tl.load(left_ptr + offs, mask=mask, other=0.0)
        lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
        r = tl.load(right_ptr + offs, mask=mask, other=0.0)
        rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
        gl = l * lg
        gr = r * rg
        tl.store(gated_left_ptr + offs, gl, mask=mask)
        tl.store(gated_right_ptr + offs, gr, mask=mask)

    # flatten
    left_flat = left.reshape(-1).contiguous()
    right_flat = right.reshape(-1).contiguous()
    left_gate_flat = left_gate.reshape(-1).contiguous()
    right_gate_flat = right_gate.reshape(-1).contiguous()
    gated_left_flat = torch.empty_like(left_flat)
    gated_right_flat = torch.empty_like(right_flat)

    total = left_flat.shape[0]
    BLOCK = 32768
    grid = (triton.cdiv(total, BLOCK),)
    gate_kernel[grid](
        gated_left_flat,
        gated_right_flat,
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        total,
        BLOCK_SIZE=BLOCK
    )

    gated_left = gated_left_flat.view_as(left)
    gated_right = gated_right_flat.view_as(right)

    # Contract via batched matmul
    # Permute to (B, H, N, N)
    left_perm = gated_left.permute(0, 3, 1, 2).contiguous()
    right_perm = gated_right.permute(0, 3, 1, 2).contiguous()
    # Reshape to (B*H, N, N)
    left_batch = left_perm.view(B * hidden_dim, N, N)
    right_batch = right_perm.view(B * hidden_dim, N, N)

    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

"""
TriMul (outgoing) forward pass using a fused Triton kernel for gating.

Algorithm
---------
1. Layer‑normalize the pair representation `x` (shape [B,N,N,C]).
2. Linear projections → `left` and `right` (shape [B,N,N,H]).
3. Apply optional mask (broadcast on the hidden dimension).
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
5. Use a Triton kernel to fuse the element‑wise product:
       gated_left  = left  * left_gate
       gated_right = right * right_gate
6. Contract over the second sequence dimension:
       out[b,i,j,d] = Σ_k gated_left[b,i,k,d] * gated_right[b,j,k,d]
   i.e. a batched matrix‑multiply `left @ rightᵀ` for every hidden head.
7. Layer‑normalize the contracted tensor, multiply by `out_gate`,
   and finally linearly project back to `C` (original `dim`).

The heavy element‑wise gating is performed on GPU by a Triton kernel,
while the N³ contraction is executed with an optimized batched GEMM
(`torch.bmm`), which maps to cuBLAS / cuTensorCore kernels.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fused element‑wise gating (left*gate, right*gate)
# -------------------------------------------------------------------------
@triton.jit
def _gate_kernel(
    gated_left_ptr, gated_right_ptr,   # output pointers
    left_ptr, left_gate_ptr,           # left and its gate
    right_ptr, right_gate_ptr,         # right and its gate
    N_ELEMS: tl.int64,                 # total number of elements
    BLOCK_SIZE: tl.constexpr           # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    l  = tl.load(left_ptr       + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    gl = l * lg
    gr = r * rg

    tl.store(gated_left_ptr  + offs, gl, mask=mask)
    tl.store(gated_right_ptr + offs, gr, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as inputs)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Layer‑norm on channel dimension C
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid on a linear transform of the normalized input)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5. Fuse element‑wise gating with a Triton kernel
    # -----------------------------------------------------------------
    # Flatten tensors to 1‑D contiguous buffers for the kernel
    left_flat      = left.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat= right_gate.contiguous().view(-1)

    gated_left_flat  = torch.empty_like(left_flat)
    gated_right_flat = torch.empty_like(right_flat)

    total_elems = left_flat.numel()
    BLOCK = 32768  # tunable, must be a multiple of 32 for good occupancy

    grid = (triton.cdiv(total_elems, BLOCK),)
    _gate_kernel[grid](
        gated_left_flat,
        gated_right_flat,
        left_flat,
        left_gate_flat,
        right_flat,
        right_gate_flat,
        total_elems,
        BLOCK_SIZE=BLOCK
    )

    # Reshape back to original 4‑D tensors
    gated_left  = gated_left_flat.view_as(left)
    gated_right = gated_right_flat.view_as(right)

    # -----------------------------------------------------------------
    # 6. Contraction over the second sequence dimension (batched GEMM)
    #    out[b,i,j,d] = Σ_k gated_left[b,i,k,d] * gated_right[b,j,k,d]
    #    i.e. for each head d: left @ rightᵀ
    # -----------------------------------------------------------------
    # Move hidden dimension to batch axis: (B, H, N, N)
    left_perm  = gated_left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = gated_right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch & head dimensions → (B*H, N, N)
    left_batch  = left_perm.view(B * hidden_dim, N, N)
    right_batch = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))  # (B*H, N, N)

    # Restore the original layout (B, N, N, H)
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output layer‑norm, gating, and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
