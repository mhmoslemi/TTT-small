def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias)

    # linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((B,N,N), dtype=left.dtype, device=device)
    else:
        mask_tensor = mask.to(left.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)
    left = left * mask_tensor
    right = right * mask_tensor

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # Compute core multiplication via Triton kernel
    # Rearrange left/right for batched matmul
    # left shape: (B,N,N,hidden)
    # Convert to (B, hidden, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    # Right transposed for matmul: (B, hidden, K, N)
    right_t = right_perm.transpose(2,3).contiguous()
    # Reshape to (B*hidden, N, N) etc.
    BH = B * hidden_dim
    A = left_perm.view(BH, N, N)   # shape (B*H, M=N, K=N)
    B_t = right_t.view(BH, N, N)   # shape (B*H, K=N, N)

    # Create output tensor
    out_mat = torch.empty_like(A, dtype=torch.float32) # shape (B*H,N,N)
    # If A dtype is float16 maybe cast; we will keep float32 for simplicity.

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B_t,
        C_ptr=out_mat,
        BATCH=BH,
        M=N,
        N=N,
        K=N,
        stride_a_batch=A.stride(0),
        stride_a_m=A.stride(1),
        stride_a_k=A.stride(2),
        stride_b_batch=B_t.stride(0),
        stride_b_k=B_t.stride(1),
        stride_b_n=B_t.stride(2),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape (B,N,N,hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
custom_kernel: Triton‑accelerated Triangle Multiplicative Update (outgoing)

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections → left, right ∈ ℝ^{B×N×N×H}.
   Apply optional mask (broadcast over H) and element‑wise gates
   (sigmoid of separate linear layers).
3. Core operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication per hidden channel:
   for each d (and batch) compute left_d @ right_dᵀ .
   Implemented with a Triton kernel (batched GEMM).
4. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to C dimensions.

Only the heavy O(N³·H) multiplication is performed in Triton; all
other lightweight ops use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    BATCH, M, N, K,               # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B   where A:[BATCH,M,K], B:[BATCH,K,N], C:[BATCH,M,N]"""
    pid_batch = tl.program_id(axis=0)
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this block
    a_ptrs = (
        A_ptr
        + pid_batch * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # k‑range for this tile
        k_valid = (k_tile * BLOCK_K + offs_k) < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

        # advance pointers to next k‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of all linear / layer‑norm parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm over the last dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for batched GEMM ----------
    # bring hidden dim forward, make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (j,k)

    # transpose right so its inner dim is K (k) and outer dim N (j)
    right_t = right_perm.transpose(2, 3).contiguous()   # [B, H, K, N]

    B, H, N, _ = left_perm.shape
    batch_hidden = B * hidden_dim

    A = left_perm.view(batch_hidden, N, N)          # (B*H, M=N, K=N)
    B_mat = right_t.view(batch_hidden, N, N)        # (B*H, K=N, N)

    # output buffer
    out_mat = torch.empty_like(A, dtype=torch.float32)  # (B*H, N, N)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[
        grid
    ](
        A_ptr=A,
        B_ptr=B_mat,
        C_ptr=out_mat,
        BATCH=batch_hidden,
        M=N,
        N=N,
        K=N,
        stride_a_batch=A.stride(0),
        stride_a_m=A.stride(1),
        stride_a_k=A.stride(2),
        stride_b_batch=B_mat.stride(0),
        stride_b_k=B_mat.stride(1),
        stride_b_n=B_mat.stride(2),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Final LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Output gate ----------
    out = out * out_gate

    # ---------- Output projection ----------
    out = F.linear(out, weights["to_out.weight"])  # no bias

    return out
