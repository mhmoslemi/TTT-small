def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm of input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

                         (dim,),
                         weight=norm_weight,
                         bias=norm_bias,
                         eps=1e-5)

        # linear projections
        left = F.linear(x, weights["left_proj.weight"])
        right = F.linear(x, weights["right_proj.weight"])

        if not nomask:
            mask = mask_tensor.float().unsqueeze(-1)  # [B,N,N,1]
            left = left * mask
            right = right * mask

        # gates
        left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
        out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

        left = left * left_gate
        right = right * right_gate

        # cast to half for kernel
        left_h = left.to(torch.float16)
        right_h = right.to(torch.float16)

        # core matmul
        out_h = tri_mul_triton(left_h, right_h,
                               block_m=64,
                               block_n=64,
                               block_k=64,
                               block_d=32)   # hidden_dim divisible by 32; if not we still handle tail

        # cast back to float32
        out = out_h.to(torch.float32)

        # final norm, gate, linear
        to_out_norm_weight = weights["to_out_norm.weight"]
        to_out_norm_bias = weights["to_out_norm.bias"]
        out = F.layer_norm(out,
                           (hidden_dim,),
                           weight=to_out_norm_weight,
                           bias=to_out_norm_bias,
                           eps=1e-5)
        out = out * out_gate
        out = F.linear(out, weights["to_out.weight"])
    return out

"""
Triangle Multiplicative Update (outgoing) – highly‑optimized forward pass.

The heavy part is the tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is a batched matrix‑multiply per hidden channel.
A custom Triton kernel computes this contraction for many (i, j) tiles
and a small block of hidden channels (BLOCK_D) at a time.
All other ops (LayerNorms, linear projections, sigmoids) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, D,                         # runtime sizes
    stride_a_i, stride_a_k, stride_a_d,   # strides for A (i, k, d)
    stride_b_j, stride_b_k, stride_b_d,   # strides for B (j, k, d) – note j is first dim
    stride_c_i, stride_c_j, stride_c_d,   # strides for C (i, j, d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute a block of the TriMul contraction."""
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_d = tl.program_id(2)   # hidden‑channel tile

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = pid_d * BLOCK_D

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over the summed dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: shape (M, K, D)  == left[b, i, k, d]
        a_ptrs = a_ptr + (
            offs_i[:, None, None] * stride_a_i +
            offs_k[None, :, None] * stride_a_k +
            offs_d[None, None, :] * stride_a_d
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # (M, K, D)

        # B block transposed to (K, N, D) == right[b, j, k, d] indexed as (k, j, d)
        b_ptrs = b_ptr + (
            offs_k[:, None, None] * stride_b_k +
            offs_j[None, :, None] * stride_b_j +
            offs_d[None, None, :] * stride_b_d
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # (K, N, D)

        # Per‑channel matrix multiply: acc[:, :, d] += a[:, :, d] @ b[:, :, d]
        for dk in range(0, BLOCK_D):
            a_slice = a[:, :, dk]       # (M, K)
            b_slice = b[:, :, dk]       # (K, N)
            acc[:, :, dk] = acc[:, :, dk] + tl.dot(a_slice, b_slice)

    # Store the result (C) – still FP32
    c_ptrs = c_ptr + (
        offs_i[:, None, None] * stride_c_i +
        offs_j[None, :, None] * stride_c_j +
        offs_d[None, None, :] * stride_c_d
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M=32, BLOCK_N=32, BLOCK_K=32, BLOCK_D=8) -> torch.Tensor:
    """
    Triton implementation of the core contraction.
    left, right: [B, N, N, D] (any dtype, preferably float16)
    Returns: [B, N, N, D] same dtype as input (float32 accumulator is cast back).
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        (D + BLOCK_D - 1) // BLOCK_D,
    )
    # Strides for a single batch slice (shape [N, N, D])
    stride_i = left.stride(0)   # i‑stride
    stride_k = left.stride(1)   # k‑stride
    stride_d = left.stride(2)   # d‑stride

    for b in range(B):
        a_ptr = left[b]
        b_ptr = right[b]
        c_ptr = out[b]
        tri_mul_kernel[grid](
            a_ptr, b_ptr, c_ptr,
            N, D,
            stride_i, stride_k, stride_d,           # A strides (i,k,d)
            stride_i, stride_k, stride_d,           # B strides (j,k,d) – layout identical
            stride_i, stride_k, stride_d,           # C strides (i,j,d)
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    data = (input, mask, weights, config)
    Returns tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input & output channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension
    nomask = config.get("nomask", True)

    # unpack weights
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    with torch.no_grad():
        # 1. Input LayerNorm
        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_w,
            bias=norm_b,
            eps=1e-5,
        )  # [B, N, N, dim]

        # 2. Linear projections (no bias)
        left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
        right = F.linear(x, right_proj_w)

        # 3. Optional mask
        if not nomask:
            mask = mask_tensor.float().unsqueeze(-1)  # [B,N,N,1]
            left = left * mask
            right = right * mask

        # 4. Gates (sigmoid of linear transforms)
        left_gate = torch.sigmoid(F.linear(x, left_gate_w))
        right_gate = torch.sigmoid(F.linear(x, right_gate_w))
        out_gate = torch.sigmoid(F.linear(x, out_gate_w))

        # 5. Apply gates to projections
        left = left * left_gate
        right = right * right_gate

        # 6. Core contraction – run in half precision for speed
        left_h = left.to(torch.float16).contiguous()
        right_h = right.to(torch.float16).contiguous()
        out_h = tri_mul_triton(
            left_h,
            right_h,
            BLOCK_M=32,
            BLOCK_N=32,
            BLOCK_K=32,
            BLOCK_D=8,          # 8 divides both 128 and 384
        )  # [B,N,N,hidden_dim] float32

        # 7. Post‑normalisation, gating and final projection
        out = F.layer_norm(
            out_h,
            (hidden_dim,),
            weight=to_out_norm_w,
            bias=to_out_norm_b,
            eps=1e-5,
        )
        out = out * out_gate
        out = F.linear(out, to_out_w)   # back to [B,N,N,dim]

    return out
