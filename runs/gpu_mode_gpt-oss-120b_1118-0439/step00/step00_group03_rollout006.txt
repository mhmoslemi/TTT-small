def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,N2,dim = input_tensor.shape
    assert N == N2
    hidden_dim = config["hidden_dim"]
    # LayerNorm of input
    # Using torch.nn.functional.layer_norm: but we need weight and bias shape [dim]
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # linear projections
    # linear util: torch.nn.functional.linear(x, weight, bias=None)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # apply mask if provided (mask shape B,N,N)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # core multiplication
    # allocate out tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)
    # call triton kernel
    launch_trimul(left, right, out, B, N, hidden_dim,
                  left.stride(), right.stride(), out.stride())
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # apply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm on dim channel dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # core multiplication
    out = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim), dtype=torch.float32, device=device)
    _trimul_kernel_launcher(left, right, out)
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # apply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    i0 = pid0 * BLOCK_M
    j0 = pid1 * BLOCK_N
    batch = pid2 // D
    d = pid2 % D

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base pointers for this batch and channel
    left_base = left_ptr + batch * stride_lb + d * stride_ld
    right_base = right_ptr + batch * stride_rb + d * stride_rd
    out_base = out_ptr + batch * stride_ob + d * stride_od

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # index ranges
        i = i0 + tl.arange(0, BLOCK_M)
        j = j0 + tl.arange(0, BLOCK_N)
        kk = k0 + tl.arange(0, BLOCK_K)

        # bounds
        mask_i = i < N
        mask_j = j < N
        mask_k = kk < N

        # load left tile (M,K)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes the O(N³) einsum:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to a batched matrix multiplication
    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

All other operations (LayerNorm, Linear projections, gates) are performed
with PyTorch for simplicity.  The heavy N³ reduction is off‑loaded to
Triton, tiled with BLOCK_M×BLOCK_N×BLOCK_K and parallelised over the
batch and hidden‑dimension axes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for the core N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # batch, seq_len, hidden_dim
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block index over i (rows)
    pid_n = tl.program_id(1)      # block index over j (cols)
    pid_z = tl.program_id(2)      # combined batch*hidden_dim

    # start indices of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # decode batch and hidden dimension from pid_z
    batch = pid_z // D
    d = pid_z % D

    # offsets for this tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current batch & hidden channel
    left_base  = left_ptr  + batch * stride_lb + d * stride_ld
    right_base = right_ptr + batch * stride_rb + d * stride_rd
    out_base   = out_ptr   + batch * stride_ob + d * stride_od

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (M, K)  – indices (i, k)
        left = tl.load(

            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,

    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N]  (may be None or empty)
            - weights: dict with the LayerNorm and Linear weights (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core N³ multiplication (Triton) ----------
    out = torch.empty((input_tensor.shape[0], input_tensor.shape[1],
                       input_tensor.shape[2], hidden_dim),
                      dtype=torch.float32,
                      device=device)
    _run_trimul(left, right, out)   # out now contains the einsum result

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear back to dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
