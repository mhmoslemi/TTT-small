@triton.jit
def outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    i_base = i * BLOCK_M
    j_base = j * BLOCK_N

    # Offsets for each dimension
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    mask_j = offs_j < N
    mask_k = offs_k < N

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Cast to appropriate dtype (float32)
    x = input_tensor  # shape [B,N,N,dim]
    # LayerNorm
    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Apply mask if exists or not.
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask_unsq
        right_proj = right_proj * mask_unsq
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Compute outgoing multiplication via Triton
    out = outgoing_triton(left, right)  # shape [B,N,N,hidden_dim]
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Multiply by out_gate
    out = out * out_gate
    # Final linear to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight) # shape [B,N,N,dim]
    return out
```

"""
TriMul outgoing implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections and per‑position sigmoid gates:
   - left = (x @ W_left) * mask * sigmoid(x @ W_left_gate)
   - right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)
3. Core "outgoing" update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM for each hidden channel `d`.  The
   computation is tiled over the sequence dimensions (i, j) and
   reduced over `k` inside a Triton kernel.
4. Layer‑norm the result over the hidden dimension, multiply by the
   `out_gate` (sigmoid(x @ W_out_gate)), and finish with a final
   linear projection back to `C`.
5. Return a tensor of shape [B,N,N,C].

The Triton kernel tiles the (i, j) plane with BLOCK_M × BLOCK_N
elements and accumulates over K in steps of BLOCK_K.  For each
(k, d) pair we compute the outer product `left[i,k,d] * right[j,k,d]`
and accumulate it into an (M,N,D) accumulator.  The kernel works
with arbitrary batch size (B ≤ 2) and hidden dimension `D` (≤ 128 in
tests).  Bounds are masked so that partially‑filled tiles are handled
correctly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ multiplication (outgoing update)
# ----------------------------------------------------------------------
@triton.jit
def outgoing_mul_kernel(
    left_ptr,          # *float32  [B, N, N, D]  (i, k, d)
    right_ptr,         # *float32  [B, N, N, D]  (j, k, d)
    out_ptr,           # *float32  [B, N, N, D]  (i, j, d)

    B, N, D,           # runtime parameters

    # strides for left tensor (b,i,k,d)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    # strides for right tensor (b,j,k,d)
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    # strides for out tensor (b,i,j,d)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile along i (row)
    pid_j = tl.program_id(2)          # tile along j (col)

    # ------------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_base = kb * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[b, i, k, d]
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            pid_b * stride_l_b + \
            offs_i[:, None, None] * stride_l_i + \
            offs_k[None, :, None] * stride_l_k + \
            tl.arange(0, BLOCK_D)[None, None, :] * stride_l_d

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)                     # (M, K, D)

        # ------------------------------------------------------------------
        # Load right[b, j, k, d]
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + \
            pid_b * stride_r_b + \
            offs_j[:, None, None] * stride_r_j + \
            offs_k[None, :, None] * stride_r_k + \
            tl.arange(0, BLOCK_D)[None, None, :] * stride_r_d

        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)                     # (N, K, D)

        # ------------------------------------------------------------------
        # Outer‑product accumulation over the K dimension inside the block
        # ------------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            # The loads already zero‑out out‑of‑bounds k entries,
            # so we can safely multiply without extra masking.
            l = left[:, kk, :]               # (M, D)
            r = right[:, kk, :]              # (N, D)
            # Broadcast to (M, N, D) and accumulate
            acc += l[:, None, :] * r[None, :, :]   # (M, N, D)

    # ------------------------------------------------------------------
    # Write the tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_o_b + \
        offs_i[:, None, None] * stride_o_i + \
        offs_j[None, :, None] * stride_o_j + \
        tl.arange(0, BLOCK_D)[None, None, :] * stride_o_d

    store_mask = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def trigram_mul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul update using Triton.
    left, right: tensors of shape [B, N, N, D] (float32, contiguous)
    Returns: tensor of shape [B, N, N, D] (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # Tile sizes – tuned for the B200.  D equals the hidden dimension.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = D  # compile‑time constant per launch

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Strides (in element counts)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    outgoing_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        # left strides [b,i,k,d]
        l_strides[0],
        l_strides[1],
        l_strides[2],
        l_strides[3],
        # right strides [b,j,k,d]
        r_strides[0],
        r_strides[1],
        r_strides[2],
        r_strides[3],
        # out strides [b,i,j,d]
        o_strides[0],
        o_strides[1],
        o_strides[2],
        o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out

    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and masks
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    if mask is not None:
        # mask shape [B,N,N] -> broadcast to hidden_dim
        mask_exp = mask.unsqueeze(-1).to(dtype)                # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core outgoing update (Triton)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = trigram_mul_outgoing(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
```
