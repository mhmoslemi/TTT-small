```
def outgoing_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_batch = tl.program_id(2)
    
    b = pid_batch // H
    h = pid_batch % H

    # Compute actual i and j offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask out-of-range
    i_in_bounds = i_offset < N
    j_in_bounds = j_offset < N

    # Pointers to left and right start
    left_base = left_ptr + b*stride_l_b + h*stride_l_h
    right_base = right_ptr + b*stride_r_b + h*stride_r_h
    out_base = out_ptr + b*stride_o_b + h*stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute number of blocks for K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        # k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for k range
        k_mask = k_offset < N

        # pointers per tile
        left_ptrs = left_base + (k_offset[:, None] * stride_l_i) + (i_offset[None, :] * stride_l_k)   # left[b, k, i, h] using swapped strides
```

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Normalize over last dim:
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                      weight=norm_weight, bias=norm_bias)
    # Projections (linear WITHOUT bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x, left_proj_weight)
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # shape [bs, N, N, 1]
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute outgoing multiplication: out[b,i,j,h] = sum_k left[b,k,i,h] * right[b,k,j,h]
    # We'll use Triton.
    out = triton_outgoing_mul(left, right)  # custom function using kernel

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Apply out_gate (broadcast over i,j)
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input ``x`` using ``norm`` parameters.
2. Project to a hidden dimension (no bias) with two linear layers:
   left = x @ W_left,   right = x @ W_right.
3. Optionally mask the projected tensors.
4. Compute three sigmoid gates (left, right, out) from ``x``.
5. Apply element‑wise gates to the projected tensors.
6. Core operation (outgoing update):
      out[b,i,j,h] = Σ_k left[b,k,i,h] * right[b,k,j,h]
   This is a batched matrix multiplication where the first two
   axes of ``left`` and ``right`` are swapped.  It is implemented
   in Triton for high performance.
7. Layer‑norm the hidden result, multiply by the output gate,
   and finally project back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the outgoing triangular multiplicative update.
# ----------------------------------------------------------------------
@triton.jit
def outgoing_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)    # block index over i
    pid_n = tl.program_id(1)    # block index over j
    pid_bh = tl.program_id(2)   # combined batch*hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden index
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # Compute offsets for the output tile (i, j)
    # --------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # --------------------------------------------------------------
    # Accumulator (use float32 for numeric stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left_tile[i,k] corresponds to left[b,k,i,h]
        # Row dimension i uses stride_l_k, column dimension k uses stride_l_i
        left_ptrs = left_base + i[:, None] * stride_l_k + k[None, :] * stride_l_i
        # right_tile[k,j] corresponds to right[b,k,j,h]
        right_ptrs = right_base + k[:, None] * stride_r_i + j[None, :] * stride_r_k

        left_tile = tl.load(left_ptrs,
                            mask=(i_mask[:, None] & k_mask[None, :]),
                            other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=(j_mask[:, None] & k_mask[None, :]),
                             other=0.0)

        # A: (BLOCK_M, BLOCK_K), B: (BLOCK_K, BLOCK_N)
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # Write the result back
    # --------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc,
             mask=i_mask[:, None] & j_mask[None, :])


def triton_outgoing_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing triangular multiplicative update using Triton.

    Args:
        left:  Tensor of shape (B, N, N, H)
        right: Tensor of shape (B, N, N, H)

    Returns:
        out: Tensor of shape (B, N, N, H)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous tensors for predictable strides
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * H                                   # batch * hidden
    )

    outgoing_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the tester
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (or None)
            - weights: dict of all linear / layer‑norm parameters
            - config: dict containing at least 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core outgoing triangular multiplication (Triton)
    # ------------------------------------------------------------------
    # Result shape: (B, N, N, hidden_dim)
    out_hidden = triton_outgoing_mul(left, right)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate

    out = F.linear(out_hidden, weights["to_out.weight"])
    return out
