@triton.jit
def outgoing_matmul_kernel(A_ptr, B_ptr, C_ptr,
                           M, N, K,
                           stride_a_batch, stride_a_m, stride_a_k,
                           stride_b_batch, stride_b_n, stride_b_k,
                           stride_c_batch, stride_c_m, stride_c_n,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute block masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to base of the batch (slice)
    a_base = A_ptr + pid_batch * stride_a_batch
    b_base = B_ptr + pid_batch * stride_b_batch
    c_base = C_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for this k block
        cur_k_offset = cur_k * BLOCK_K + offs_k
        # Load A block [M, K]
        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (cur_k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (cur_k_offset[None, :] < K), other=0.0)
        # Load B block [N, K]
        b_ptrs = b_base + (offs_n[:, None] * stride_b_n) + (cur_k_offset[None, :] * stride_b_k)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (cur_k_offset[None, :] < K), other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)

    # Write out result
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask (if provided)
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)  # shape B,N,N,1
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare tensors for Triton matmul
    B, N, _, _ = left.shape  # shape B,N,N, hidden_dim
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Merge batch and hidden dims
    S = B * hidden_dim
    left_flat = left_perm.view(S, N, N)
    right_flat = right_perm.view(S, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)   # row dimension (j)
    stride_b_k = right_flat.stride(2)   # column dimension (k)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    grid = (S,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    outgoing_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape B,N,N,hidden_dim

    # Output layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing update (AlphaFold3) – Triton‐accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) to a hidden dimension H:
      L = X·W_left   ,   R = X·W_right
3. Apply an optional pairwise mask (shape B×N×N) to L and R.
4. Compute three sigmoid gates from X:
      gL = σ(X·W_left_gate), gR = σ(X·W_right_gate),
      gO = σ(X·W_out_gate)
   and multiply:
      L ← L ⊙ gL,   R ← R ⊙ gR.
5. For every batch b and hidden channel h compute
      out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h] .
   This is a batch of matrix‑multiplications A·Bᵀ where
      A = L_{b,h} ∈ ℝ^{N×N},  B = R_{b,h} ∈ ℝ^{N×N}.
   The heavy ω = N³ work is done in a Triton kernel that
   processes each (b,h) slice independently.
6. Layer‑norm over the hidden dimension, multiply by gO and
   a final linear projection back to C channels.

Only the N³ matrix multiplication is off‑loaded to Triton; the
remaining operations stay in PyTorch to keep the implementation
compact and numerically stable.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _outgoing_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ  for many independent slices.
    A : [S, M, K]   (row‑major)
    B : [S, N, K]   (row‑major, we treat each row as a vector over K)
    C : [S, M, N]
    """
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current slice
    a_base = A_ptr + batch_id * stride_a_batch
    b_base = B_ptr + batch_id * stride_b_batch
    c_base = C_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for tile_k in range(0, num_k_tiles):
        cur_k = tile_k * BLOCK_K + offs_k
        k_mask = cur_k < K

        # Load A[M, K]
        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0)

        # Load B[N, K]  (each row corresponds to an output‑column index)
        b_ptrs = b_base + (offs_n[:, None] * stride_b_n) + (cur_k[None, :] * stride_b_k)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & k_mask[None, :], other=0.0)

        acc += tl.dot(a, b)   # (M,K) • (N,K) → (M,N)

    # Write the tile
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of torch.Tensors containing all linear / layer‑norm parameters
            - config: dict with keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # C
    hidden_dim = config["hidden_dim"]  # H
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and broadcastable
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Kernel: batched matmul over the pairwise dimension (outgoing update)
    #    Compute C_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, H]

    # Rearrange to (B, H, N, N) → contiguous → (S, N, N) where S = B * H
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    S = B * hidden_dim

    left_flat  = left_perm.view(S, N, N)
    right_flat = right_perm.view(S, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Tile size – works well for N up to 1024 on an A200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for the flattened tensors
    stride_a_batch = left_flat.stride(0)
    stride_a_m     = left_flat.stride(1)
    stride_a_k     = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n     = right_flat.stride(1)   # row (output‑column) dimension
    stride_b_k     = right_flat.stride(2)   # reduction dimension

    stride_c_batch = out_flat.stride(0)
    stride_c_m     = out_flat.stride(1)
    stride_c_n     = out_flat.stride(2)

    grid = (
        S,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _outgoing_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                         # M, N, K (all equal to sequence length)
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
