"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` using the provided `norm` weights.
2. Linear projections:
   * `left = x @ left_proj.weight`
   * `right = x @ right_proj.weight`
3. Compute sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. **Fused Triton kernel** – for each element (b,i,k) multiply:
       left  = left  * left_gate  * mask   (if a mask is supplied)
       right = right * right_gate * mask
   The kernel works on a flattened `[b,i,k]` axis and processes the
   hidden dimension in tiles (default 32).  This fuses the element‑wise
   mask and gating operations.
5. Pairwise multiplicative update:
   `out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]`
   Implemented as a batched GEMM:
       `left_perm = left.permute(0,3,1,2).reshape(B*H,N,N)`
       `right_perm = right.permute(0,3,1,2).reshape(B*H,N,N)`
       `out_perm = torch.bmm(left_perm, right_perm.transpose(1,2))`
   reshaped back to `[B,N,N,H]`.
6. Apply the second LayerNorm (`to_out_norm`), the output gate
   (`out_gate`) and the final linear projection (`to_out`) to obtain
   the result of shape `[B, N, N, dim]`.

Only part of the computation (mask + gating) is performed in a custom
Triton kernel; the heavy O(N³) reduction is delegated to cuBLAS via a
batched GEMM, which is highly optimized on the target B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_outl_b, stride_outl_i, stride_outl_k, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_k, stride_outr_h,
    BLOCK_ELEM: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)          # blocks over the flattened (b,i,k) axis
    pid_h = tl.program_id(1)        # blocks over the hidden dimension

    start = pid * BLOCK_ELEM
    offs = start + tl.arange(0, BLOCK_ELEM)
    valid = offs < (B * N * N)

    # convert flattened index to (b,i,k)
    idx = tl.where(valid, offs, 0)
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    k = rem % N

    # hidden‑dim tile start
    h_start = pid_h * BLOCK_H
    h_range = tl.arange(0, BLOCK_H)
    h_valid = (h_start + h_range) < H

    # base pointer offsets (without the hidden dimension)
    base_l = b * stride_l_b + i * stride_l_i + k * stride_l_k
    base_r = b * stride_r_b + i * stride_r_i + k * stride_r_k
    base_lg = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k
    base_rg = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k
    base_outl = b * stride_outl_b + i * stride_outl_i + k * stride_outl_k
    base_outr = b * stride_outr_b + i * stride_outr_i + k * stride_outr_k

    # pointer offsets for the hidden‑dim slice
    h_idx = h_start + h_range
    left_ptrs = left_ptr + base_l[:, None] + h_idx[None, :] * stride_l_h
    right_ptrs = right_ptr + base_r[:, None] + h_idx[None, :] * stride_r_h
    lg_ptrs = left_gate_ptr + base_lg[:, None] + h_idx[None, :] * stride_lg_h
    rg_ptrs = right_gate_ptr + base_rg[:, None] + h_idx[None, :] * stride_rg_h

    load_mask = valid[:, None] & h_valid[None, :]
    left = tl.load(left_ptrs, mask=load_mask, other=0.0)
    right = tl.load(right_ptrs, mask=load_mask, other=0.0)
    left_gate = tl.load(lg_ptrs, mask=load_mask, other=0.0)
    right_gate = tl.load(rg_ptrs, mask=load_mask, other=0.0)

    if HAS_MASK:
        mask_ptrs = mask_ptr + b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
        mask_val = tl.load(mask_ptrs, mask=valid, other=0.0)          # shape (BLOCK_ELEM,)
        mask_vec = mask_val[:, None]                                 # broadcast over H
        left = left * left_gate * mask_vec
        right = right * right_gate * mask_vec
    else:
        left = left * left_gate
        right = right * right_gate

    # store the fused results
    outl_ptrs = out_left_ptr + base_outl[:, None] + h_idx[None, :] * stride_outl_h
    outr_ptrs = out_right_ptr + base_outr[:, None] + h_idx[None, :] * stride_outr_h
    tl.store(outl_ptrs, left, mask=load_mask)
    tl.store(outr_ptrs, right, mask=load_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be None)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict: contains at least "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask & gate – fused Triton kernel when a mask is present
    # ------------------------------------------------------------------
    if not nomask:
        # ensure contiguous tensors for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_tensor = mask.contiguous().to(dtype)

        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        # strides (int64)
        s_l = left.stride()
        s_r = right.stride()
        s_lg = left_gate.stride()
        s_rg = right_gate.stride()
        s_outl = left_out.stride()
        s_outr = right_out.stride()
        s_mask = mask_tensor.stride()

        BLOCK_ELEM = 256      # tunable – number of (b,i,k) elements per program
        BLOCK_H = 32          # tile size for the hidden dimension

        total_elements = B * N * N
        hidden_tiles = (hidden_dim + BLOCK_H - 1) // BLOCK_H
        grid = ((total_elements + BLOCK_ELEM - 1) // BLOCK_ELEM, hidden_tiles)

        _mask_gate_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_tensor,
            left_out, right_out,
            B, N, hidden_dim,
            s_l[0], s_l[1], s_l[2], s_l[3],
            s_r[0], s_r[1], s_r[2], s_r[3],
            s_lg[0], s_lg[1], s_lg[2], s_lg[3],
            s_rg[0], s_rg[1], s_rg[2], s_rg[3],
            s_mask[0], s_mask[1], s_mask[2],
            s_outl[0], s_outl[1], s_outl[2], s_outl[3],
            s_outr[0], s_outr[1], s_outr[2], s_outr[3],
            BLOCK_ELEM=BLOCK_ELEM,
            BLOCK_H=BLOCK_H,
            HAS_MASK=1,
        )
        left = left_out
        right = right_out
    else:
        # No mask – simple element‑wise multiplication in PyTorch
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for a strided batched gemm
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_perm: (B*H, N, N) = left_perm @ right_permᵀ
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
