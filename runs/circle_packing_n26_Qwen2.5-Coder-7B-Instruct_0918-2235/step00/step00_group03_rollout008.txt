import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def compute_all_radii(centers, radii, eps=1e-12):
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y)
        dists = cdist(centers, centers[i].reshape(1, 2), 'euclidean').flatten()
        dists[i] = np.inf  # Avoid using the distance to itself
        min_dist_nei = dists.min() - radii - eps
        r = min(r, min_dist_nei)
        if r < 0:
            r = 0
        radii[i] = r
    return radii

def refine_all(centers, eps=1e-9, max_iter=200):
    radii = np.full(len(centers), 1e-3)
    for _ in range(max_iter):
        new_radii = compute_all_radii(centers, radii, eps)
        if np.allclose(new_radii, radii, atol=eps):
            break
        radii = new_radii
    return radii

def anneal_one_start(centers, rng, steps=500, init_step=0.07, min_step=0.01):
    r = min_step
    bounds = [(0, 1)] * len(centers)
    for step in range(steps):
        new_centers = centers.copy()
        i = rng.choice(range(len(centers)))
        dx, dy = rng.uniform(-r, r, size=2)
        new_centers[i, 0] = np.clip(centers[i, 0] + dx, 0, 1)
        new_centers[i, 1] = np.clip(centers[i, 1] + dy, 0, 1)
        new_radii = refine_all(new_centers)
        new_sum = np.sum(new_radii)
        if new_sum > sum(np.sum(radii)):
            centers = new_centers
            radii = new_radii
        else:
            if new_sum > sum(np.sum(centers)):
                p = np.exp((new_sum - sum(np.sum(centers))) / 0.25)
                if rng.random() < p:
                    centers = new_centers
                    radii = new_radii
        r *= 0.995  # Annealing schedule
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng()
    centers = rng.random((26, 2))
    radii = np.ones(26) * 1e-3
    best_total, best_centers, best_radii = None, None, 0.
    for _ in range(30):
        centers, radii, cur_total = anneal_one_start(centers, rng)
        if cur_total > best_total or best_total is None:
            best_total, best_centers, best_radii = cur_total, centers, radii
    return best_centers, best_radii, best_total
