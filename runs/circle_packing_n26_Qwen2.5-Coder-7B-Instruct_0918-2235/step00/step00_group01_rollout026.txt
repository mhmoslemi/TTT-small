def place_circle(pts, rad, order, k, MIN_R, S):
    pts_trial = np.random.random((S, 2))
    candidates = []
    for x, y in pts_trial:
        if all(cdist([(x, y)], pts, 'euclidean') > (rad + MIN_R)[order[list(order).index(k) - :k]]) and 0 <= x <= 1 - MIN_R and 0 <= y <= 1 - MIN_R:
            clearance = min(min(min(x, 1 - x, y, 1 - y), min(np.sqrt((x - px) ** 2 + (y - py) ** 2) - rad[j] for j, (px, py) in enumerate(pts)))) for i in range(k))
            if clearance < 1e-6:
                clearance = 1e-6
            candidates.append((x, y, clearance))
    return max(candidates, key=lambda c: c[2]) if candidates else (np.random.random(2), MIN_R)

def propose_move(x, T, σ):
    Δ = np.random.normal(0, σ, 2)
    x_prime = np.clip(x + Δ, min(x, 1 - x), max(x, 1 - x))
    return x_prime, y_prime

def calculate_acceptance_ratio(pts, rad, i, ε_w, ε_p):
    U_old = U_wall_old(pts[i], rad[i]) + sum(U_pair_old(pts[i], rad[i], rad[j]) for j in range(len(rad)) if j != i)
    U_new = U_wall(pts[i], rad[i]) + sum(U_pair(pts[i], rad[i], rad[j]) for j in range(len(rad)) if j != i)
    ΔU = U_new - U_old
    return 1 if ΔU <= 0 else np.exp(-ΔU / T)

def U_wall(x, r):
    return 0.5 * ε_w * (1 / (r + min(x, 1 - x, min(x, 1 - x)))) ** 2

def U_wall_old(x, r):
    return 0.5 * ε_w * (1 / (r + min(x, 1 - x, min(x, 1 - x)))) ** 1

def U_pair(x1, r1, x2, r2):
    return 0.5 * ε_p * (1 / (r1 + r2 - np.sqrt((x1 - x2) ** 2)) ** 2)

def U_pair_old(x1, r1, x2, r2):
    return 0.5 * ε_p * (1 / (r1 + r2 - np.sqrt((x1 - x2) ** 2)) ** 1)

def max_radius(pts, rad, i, MIN_R):
    radial_force_to_wall = 2 / min(pts[i, 0] + rad[i], 1 - pts[i, 0] - rad[i], pts[i, 1] + rad[i], 1 - pts[i, 1] - rad[i])
    radial_force_to_other_circles = min(2 / min(np.sqrt(np.sum((pts[i] - pts[j]) ** 2)) - rad[j] - rad[i] for j in range(len(rad)) if j != i), np.inf)
    return pts[i], min(radial_force_to_wall, radial_force_to_other_circles) - MIN_R
