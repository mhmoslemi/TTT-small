import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def lp_radii_fixup(x, centers):
    n = x.shape[0] // 2
    constraint_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            constraint_matrix[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraint_matrix[j, i] = constraint_matrix[i, j]
    constraint_matrix -= np.triu(np.ones((n, n)), k=1)
    constraint_matrix = np.where(constraint_matrix < 0, np.zeros_like(constraint_matrix), constraint_matrix)
    r = np.sum(x)
    constraints = [{'type': 'ineq', 'fun': lambda r, m, i: r[i] - m.min()},
                  {'type': 'ineq', 'fun': lambda r, m, i: m.max() - r[i]},
                  {'type': 'ineq', 'fun': lambda r, m, i: np.sum(m) - r}]
    result = linprog(c=constraint_matrix.flatten(), A_le=np.hstack((np.ones((n*(n-1)//2, 2*n)), constraint_matrix)), b_le=np.zeros(n*(n-1)//2 + 2*n + 1), bounds=(0, 0.5))
    updated_radii = result.x[:n]
    return r - (r - np.sum(updated_radii))

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    np.random.seed(42)  # for reproducibility
    for i in range(n):
        dists = np.random.rand(4) * np.array([1, 1, 1, 1])
        while True:
            offsets = dists * np.random.rand(4)
            if offsets[0] + offsets[2] <= 1 and offsets[1] + offsets[3] <= 1:
                centers[i] = offsets
                break
    initial_r = np.min(centers[:, 0] - centers[:, 1]) - 1e-10
    initial_r = np.min([initial_r, 0.5])
    objectives = [np.sum(radii) - lp_radii_fixup(radii, centers) for radii in [initial_r * np.ones(n)]]
    for _ in range(8):
        jitter = np.random.rand(n, 2) * 0.01
        centers += jitter
        radii = lp_radii_fixup(radii, centers)
        objectives.append(np.sum(radii))
    i_max = np.argmax(objectives)
    centers = centers[i_max] + jitter[i_max] * (objectives[i_max] < objectives[i_max-1])
    for _ in range(20):
        result = minimize(lambda x: lp_radii_fixup(x[:n], x[n:]), np.hstack((radii, centers.ravel())), method='L-BFGS-B', bounds=[(0, 0.5) for _ in range(n)] + [(0, 1) for _ in range(2*n)])
        if result.success:
            radii = result.x[:n]
            centers = result.x[n:].reshape(-1, 2)
            if validate_packing(centers, radii)[0]:
                return centers, radii, np.sum(radii)
    return centers, radii, np.sum(radii)
