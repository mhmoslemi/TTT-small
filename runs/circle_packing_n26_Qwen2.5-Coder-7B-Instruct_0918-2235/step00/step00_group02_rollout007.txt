import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform
import random

def calc_max_radii(C):
    n = C.shape[0]
    dists = squareform(pdist(C))
    r = np.zeros(n)
    A = []
    b = []
    for i in range(n):
        for j in range(i+1, n):
            A.append([1, 1, -1, -1])
            b.append(dists[i][j])
            A.append([0, 0, 1, -1])
            b.append(dists[i][j])
            A.append([0, 0, -1, 1])
            b.append(dists[i][j])
            A.append([1, 0, 0, 0])
            b.append(C[i][0])
            A.append([1, 0, 0, 0])
            b.append(1 - C[i][0])
            A.append([0, 1, 0, 0])
            b.append(C[i][1])
            A.append([0, 1, 0, 0])
            b.append(1 - C[i][1])
    A = np.array(A)
    b = np.array(b)
    res = linprog(np.ones(n), A_ub=-A, b_ub=-b, bounds=(0, 1), method='highs', options={'tol': 1e-9})
    if not res.success:
        return -np.inf, r
    rad = res.x
    return -np.sum(rad), rad

def objective(x):
    n = 26
    radius_vec = x[n:]
    centers = x[:n].reshape(n, 2)
    rad_sum = -np.sum(radius_vec)
    if np.any(radius_vec < 0) or not all(0 <= centers[:, 0] <= 1 and 0 <= centers[:, 1] <= 1):
        return -np.inf
    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(centers[i] - centers[j]) < radius_vec[i] + radius_vec[j] + 1e-12:
                return -np.inf
    return rad_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    init_grid = np.array([
        [0.05 + 0.18 * i + 0.09 * j % 2 * 0.18, 0.05 + 0.156 * j] for i in range(6) for j in range(6)
        [0.05 + 0.18 * i + 0.09 * (j + 1) % 2 * 0.18, 0.05 + 0.156 * (j + 1)] for i in range(6) for j in range(6)
        [0.05 + 0.18 * i + 0.09 * (j + 2) % 2 * 0.18, 0.05 + 0.156 * (j + 2)] for i in range(6) for j in range(6)
        [0.05 + 0.18 * i + 0.09 * (j + 3) % 2 * 0.18, 0.05 + 0.156 * (j + 3)] for i in range(6) for j in range(6)
        [0.05 + 0.18 * i + 0.09 * (j + 4) % 2 * 0.18, 0.05 + 0.156 * (j + 4)] for i in range(6) for j in range(6)
    ])[:-2]
    
    centers = init_grid[random.sample(range(len(init_grid)), 26)]
    temperature = 0.1
    cooling_factor = 0.995
    max_iterations = 50000
    sigma = 0.02
    
    for _ in range(max_iterations):
        if temperature < 1e-5:
            temperature = 1e-5
        
        i = random.randint(0, 25)
        C_prime = centers.copy()
        C_prime[i] += np.array([np.random.normal(0, sigma), np.random.normal(0, sigma)])
        C_prime[i] = np.clip(C_prime[i], 0, 1)
        
        _, r_prime = calc_max_radii(C_prime)
        new_sum = -np.sum(r_prime)
        
        if new_sum > objective(centers):
            centers = C_prime.copy()
        else:
            if np.random.rand() < np.exp((new_sum - objective(centers)) / temperature):
                centers = C_prime.copy()
        
        temperature *= cooling_factor
    
    _, radii = calc_max_radii(centers)
    
    return centers, radii, -objective(centers)

# Run the function and print the result
centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
