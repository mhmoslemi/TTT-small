import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centers, placed_radii):
    num_candidates = candidates.shape[0]
    num_existing = len(placed_centers)
    placed_centres_cpy = np.array(placed_centers)
    placed_radii_cpy = np.array(placed_radii)
    
    if num_existing == 0:
        dists = np.sqrt(np.sum(candidates**2, axis=1))
        r_max = np.minimum(dists, candidates[:, 0]) * np.minimum(dists, candidates[:, 1])
        return np.minimum(r_max, 1 - candidates[:, 0]) * np.minimum(r_max, 1 - candidates[:, 1])
    
    distances = np.array([np.sqrt(np.sum((candidates[:, np.newaxis, :] - placed_centres_cpy[i])[..., 0]**2 + (candidates[:, np.newaxis, :] - placed_centres_cpy[i])[..., 1]**2)) 
                          for i in range(num_existing)])
    r_max = np.minimum(distances - placed_radii_cpy[:, np.newaxis], candidates[:, 0]) * np.minimum(distances - placed_radii_cpy[:, np.newaxis], candidates[:, 1])
    r_max = np.minimum(r_max, 1 - candidates[:, 0]) * np.minimum(r_max, 1 - candidates[:, 1])
    r_max = np.maximum(r_max, 0.0)
    return np.min(r_max, axis=0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    squares = np.array([(i * h + epsilon, j * h + epsilon, (i + 1) * h - epsilon, (j + 1) * h - epsilon) for i in range(21) for j in range(21)])
    center_candidates = (squares[:, 0] + squares[:, 2]) / 2, (squares[:, 1] + squares[:, 3]) / 2
    candidate_centers = np.array(center_candidates).T
    
    placed_centers = []
    placed_radii = []
    while len(placed_centers) < 26:
        r_candidates = compute_r_max(candidate_centers, placed_centers, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centers.append(candidate_centers[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidate_centers - candidate_centers[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidate_centers = candidate_centers[~too_close]
    
    def lp_solve(centers, radii):
        n = centers.shape[0]
        r = cp.Variable(n, nonneg=True)
        constraints = [cp.MatMul(cp.diag(2 * r) - cp.eye(n), centers) <= cp.norm(centers, axis=1) + 1e-9, cp.norm(centers, axis=1) - r >= epsilon]
        objective = cp.Maximize(cp.sum(r))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-9)
        return np.array(r.value)
    
    radii = lp_solve(np.array(placed_centers), np.array(placed_radii))
    total_radius_sum = np.sum(radii)
    
    for _ in range(5000):
        indices_to_move = np.argmax(np.random.rand(26) / (np.array(radii).max() / np.random.rand(26)))[0]
        alpha = 0.02 * (1 - _ / 5000)
        proposed_center = np.clip(placed_centers[indices_to_move] + np.random.uniform(-alpha, alpha, size=2), 0, 1)
        proposed_radii = lp_solve(np.concatenate([placed_centers[:indices_to_move], [proposed_center], placed_centers[indices_to_move + 1:]]), np.concatenate([placed_radii[:indices_to_move], [0.0], placed_radii[indices_to_move + 1:]]))
        if np.sum(proposed_radii) > np.sum(radii):
            placed_centers[indices_to_move] = proposed_center
            placed_radii[indices_to_move] = proposed_radii
    
    return np.array(placed_centers), np.array(placed_radii), np.sum(placed_radii)
