import numpy as np
from scipy.optimize import minimize

def make_pair_indices(n):
    return [(i, j) for i in range(n) for j in range(i + 1, n)]

def ineqs(v):
    n = len(v) // 3
    pairs = make_pair_indices(n)
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    res = np.zeros(len(pairs) + 4*n)
    for i in range(n):
        res[4*i] = xs[i] - rs[i]
        res[4*i + 1] = 1 - xs[i] - rs[i]
        res[4*i + 2] = ys[i] - rs[i]
        res[4*i + 3] = 1 - ys[i] - rs[i]
    for (i, j) in pairs:
        dx, dy = xs[i] - xs[j], ys[i] - ys[j]
        res[4*n + i*j + j] = dx**2 + dy**2 - (rs[i] + rs[j])**2
    return res

def objective(v):
    return -np.sum(v[2::3])  # Minimize negative sum to maximize sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r0 = 0.02
    xs = np.random.uniform(r0, 1-r0, n)
    ys = np.random.uniform(r0, 1-r0, n)
    rs = np.repeat(r0, n)
    v0 = np.hstack((xs, ys, rs))
    bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n
    constraints = [{'type': 'ineq', 'fun': ineqs}]
    res = minimize(objective, v0, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    
    if not res.success:
        raise Exception('Optimization failed')
    
    xs = res.x[:n]
    ys = res.x[n:2*n]
    rs = res.x[2*n:]
    centers = np.column_stack((xs, ys))
    radii = rs
    sum_r = rs.sum()
    
    return centers, radii, sum_r
