import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_cycles = 50
    num_trials = 40
    alpha = 0.8
    eps_stop = 1e-4

    def compute_distances(centers):
        d = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        np.fill_diagonal(d, np.inf)
        return d

    def construct_linprog_matrix(centers):
        d = compute_distances(centers)
        rho = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
        A_ub = []
        b_ub = []
        for i in range(n):
            for j in range(i + 1, n):
                A_ub.append([-1, 0, 1, 0])
                b_ub.append(d[i, j])
                A_ub.append([0, -1, 0, 1])
                b_ub.append(d[i, j])
        A_ub = np.array(A_ub)
        bounds = [(0, r) for r in rho]
        return A_ub, b_ub, bounds

    def solve_inner_lp(centers):
        A_ub, b_ub, bounds = construct_linprog_matrix(centers)
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return centers, res.fun
        return res.x, res.fun
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    for _ in range(num_trials):
        centers = np.random.rand(n, 2)
        current_sum = 0
        step = 0.1
        circles = np.arange(n)
        while step > eps_stop:
            for i in circles:
                old_pos = centers[i]
                delta = np.random.uniform(-step, step, size=2)
                new_pos = old_pos + delta
                new_pos = np.clip(new_pos, 0, 1)
                centers[i] = new_pos
                new_radii, new_sum = solve_inner_lp(centers)
                if new_sum <= current_sum + 1e-9:
                    centers[i] = old_pos
                else:
                    current_sum = new_sum
            step *= alpha
        if current_sum > best_sum:
            best_centers, best_radii = centers, new_radii
            best_sum = current_sum
    return best_centers, best_radii, best_sum
