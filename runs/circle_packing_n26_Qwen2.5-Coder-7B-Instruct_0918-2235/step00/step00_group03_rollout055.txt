def energy(vars, lambda_):
    n = len(vars) // 3
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    penalty = 0.0
    for i in range(n):
        if x[i] - r[i] < -1e-12:
            penalty += (-(x[i] - r[i]) ** 2)
        if 1 - x[i] - r[i] < -1e-12:
            penalty += (-(1 - x[i] - r[i]) ** 2)
        if y[i] - r[i] < -1e-12:
            penalty += (-(y[i] - r[i]) ** 2)
        if 1 - y[i] - r[i] < -1e-12:
            penalty += (-(1 - y[i] - r[i]) ** 2)
        for j in range(i + 1, n):
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            if dist - r[i] - r[j] < -1e-12:
                penalty += (-(dist - r[i] - r[j]) ** 2)
    
    sum_radii = -np.sum(r)
    return sum_radii + lambda_ * penalty
