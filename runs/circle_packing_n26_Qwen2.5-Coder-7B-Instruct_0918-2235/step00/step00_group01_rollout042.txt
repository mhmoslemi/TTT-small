import numpy as np
from scipy.spatial import distance

def dist2(i, j):
    return np.sum((pts[i] - pts[j]) ** 2)

def shuffle_randomly():
    return np.random.permutation(n).astype(int)

def place_circle(idx):
    rad_new = np.inf
    for _ in range(S):
        x, y = np.random.rand(2)
        outside_boundaries = x < MIN_R or x > 1 - MIN_R or y < MIN_R or y > 1 - MIN_R
        no_overlap = all(dist2(idx, j) > (rad_j + rad_new) ** 2 for j in range(n) if rad_j > 0)
        if outside_boundaries or not no_overlap:
            continue
        clearance = min(min(np.abs(np.array([x, y]) - np.array(pts[j])), np.array([1 - x, 1 - y]))) - rad_new
        rad_new = min(rad_new, clearance)
    rad[idx] = rad_new
    pts[idx] = (x, y)
    return rad_new

def shake_leaves():
    for idx in range(n):
        dx, dy = np.random.uniform(-SHAKE, SHAKE, size=2)
        x_new, y_new = pts[idx] + (dx, dy)
        x_new = min(max(x_new, rad_new), 1 - rad_new)
        y_new = min(max(y_new, rad_new), 1 - rad_new)
        update_radius(idx, (x_new, y_new))
        if rad[idx] >= rad_new:
            pts[idx] = (x_new, y_new)
            rad[idx] = rad_new
    return rad_idx

def update_radius(idx, center):
    (x_new, y_new) = center
    d_walls = np.array([x_new, 1 - x_new, y_new, 1 - y_new])
    rad_new = min(d_walls, dist2(idx, pts[j]) - rad[j] ** 2 for j in range(n) if rad[j] > 0)
    rad_new = min(rad_new, rad[idx])
    rad[idx] = rad_new

def optimize():
    while True:
        temp_idx, max_idx = np.unravel_index(np.argmax(delta_U), (n, n))
        if delta_U[temp_idx, max_idx] > 0:
            if np.random.rand() < np.exp(-delta_U[temp_idx, max_idx] / T):
                update_radius(temp_idx, pts[max_idx])
                pts[max_idx], pts[temp_idx] = pts[temp_idx], pts[max_idx]
            else:
                rad_idx[temp_idx], rad_idx[max_idx] = rad_idx[max_idx], rad_idx[temp_idx]
        T *= cooling_rate
        if T < T_final:
            break

def validate_and_improve():
    for i in range(n):
        rad_new = min(pts[i, 0] - 0.5, 0.5 - pts[i, 0], pts[i, 1] - 0.5, 0.5 - pts[i, 1])
        for j in range(n):
            if j == i:
                continue
            rad_new = min(rad_new, (rad[j] + 0.0001 + dist2(i, j)) ** 0.5)
        rad[i] = rad_new
    return rad.sum()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    global pts, rad, delta_U, T, cooling_rate, T_final, T_init, n, S, SHAKE, MIN_R, R_min
    n = 26
    S = 5000
    SHAKE = 1e-4
    MIN_R = 1e-8
    R_min = 1e-6
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    
    np.random.seed(42)

    # Phase 1: Random greedy insertion
    order = shuffle_randomly()
    for t, k in enumerate(order):
        rad[k] = place_circle(k)
    pts = pts[order]
    rad = rad[order]

    rad_idx = rad.copy()
    delta_U = np.zeros((n, n))
    T = T_init
    cooling_rate = 0.999
    T_final = 1e-4

    # Phase 2: Force-field relaxation
    optimize()

    # Post-processing & refinement
    rad_idx = rad.copy()
    validate_and_improve()
    validate_and_improve()
    validate_and_improve()
    validate_and_improve()
    validate_and_improve()

    return pts, rad, validate_and_improve()
