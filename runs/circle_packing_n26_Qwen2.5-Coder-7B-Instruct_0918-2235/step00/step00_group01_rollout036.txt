import numpy as np

def run_packing():
    np.random.seed(12345)
    n = 26
    P = 60
    w = 0.7298
    c1 = 1.49618
    c2 = 1.49618
    G = 200
    v_max = np.array([1, 1, 0.5])
    lambda_multiplier = 1e6
    best = -np.inf
    best_v = None

    # Initialize population
    pop = np.random.uniform(0, 1, (P, 2*n)) * v_max
    pop[:, n:] = np.random.uniform(0, 0.01, (P, n))
    velocities = np.zeros_like(pop)
    p_best = pop.copy()
    p_best_fitness = np.zeros(P)
    g_best = pop[0]
    g_best_fitness = -np.inf

    # Evaluate initial population
    for i in range(P):
        fitness = objective_function(pop[i])
        if fitness > g_best_fitness:
            g_best_fitness = fitness
            g_best = pop[i]
        if fitness > p_best_fitness[i]:
            p_best[i] = pop[i]
            p_best_fitness[i] = fitness

    for _ in range(G):
        for i in range(P):
            velocities[i] = w * velocities[i] + c1 * np.random.rand(n) * (p_best[i] - pop[i]) + c2 * np.random.rand(n) * (g_best - pop[i])
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            pop[i] += velocities[i]
            pop[i] = np.clip(pop[i], np.zeros(2*n), v_max)
            fitness = objective_function(pop[i])
            if fitness > p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best[i] = pop[i]
            if fitness > g_best_fitness:
                g_best_fitness = fitness
                g_best = pop[i]
        
        lambda_multiplier *= 1.01
        
        # Early stopping if stagnation
        if g_best_fitness - best <= 1e-2:
            break

        best = g_best_fitness

    # Post-processing
    best_x = g_best
    best_r = np.clip(best_x[n:], 0, 0.5)
    best_x[n:] = best_r
    best_x = np.clip(best_x, np.zeros(2*n), v_max)

    for _ in range(10):
        for i in range(n):
            best_x[n+i] = best_x[n+i] + np.random.uniform(-1e-4, 1e-4, size=(2,))
            best_x[n+i] = np.clip(best_x[n+i], np.zeros(2*n), v_max)
            best_r[n:] = compute_radii(best_x[:n], best_x[n:])
            if objective_function(best_x) > objective_function(best_x):
                break
            best_x[n+i] -= 2 * np.random.uniform(-1e-4, 1e-4, size=(2,))

    return best_x[:n].reshape(-1, 2), best_r, np.sum(best_r)

def objective_function(x):
    radii = x[n:]
    penalty = 0
    for i in range(n):
        if x[i*n] < radii[i] or x[i*n] > 1 - radii[i]:
            penalty -= x[i*n] - radii[i]
        if x[i*n+1] < radii[i] or x[i*n+1] > 1 - radii[i]:
            penalty -= x[i*n+1] - radii[i]
        for j in range(i+1, n):
            dist = np.sqrt((x[i*n] - x[j*n])**2 + (x[i*n+1] - x[j*n+1])**2)
            if dist < radii[i] + radii[j]:
                penalty -= (radii[i] + radii[j]) - dist
    return np.sum(radii) - lambda_multiplier * penalty

def compute_radii(x, r):
    for i in range(n):
        r[i] = min(x[i*2], 1-x[i*2], x[i*2+1], 1-x[i*2+1], *[np.sqrt((x[i*2] - x[j*2])**2 + (x[i*2+1] - x[j*2+1])**2) - r[j] for j in range(n) if i != j])
    return r
