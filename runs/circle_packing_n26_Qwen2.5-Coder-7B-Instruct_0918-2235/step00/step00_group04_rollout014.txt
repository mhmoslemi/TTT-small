import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
from scipy.spatial import distance
import time

def init_hex_grid():
    """Build a 6x5 hexagonal lattice and select 26 points uniformly at random from it."""
    hexagon_radius = 0.3
    square_length = 1.0
    hex_width = hexagon_radius * 2 * np.sqrt(3)
    hex_height = hexagon_radius * 3

    outer_radius = np.max([hex_width / 2, hex_height / 2]) * 1.1
    inner_radius = outer_radius * 0.8
    structure_radius = max(outer_radius, inner_radius)

    while True:
        grid_size = int(np.ceil(square_length / structure_radius))
        x = np.linspace(-structure_radius, structure_radius, grid_size)
        y = np.linspace(-structure_radius, structure_radius, grid_size)
        circles = np.array([(xi, yi) for yi in y for xi in x if (xi**2 + yi**2)**0.5 <= structure_radius])
        if len(circles) >= 26:
            break

    np.random.shuffle(circles)
    centres = circles[:26]
    radii = np.full_like(centres[:, 0], 0.01)
    return centres, radii

def pack_and_optimize():
    """Optimizes the packing of 26 circles in a unit square using continuous penalty method."""
    def objective(z):
        k = len(z) // 3
        x, y, r = z[:k], z[k:2*k], z[2*k:]
        boundary_term = np.sum(np.maximum(0, r - x)**2) + \
                         np.sum(np.maximum(0, r - (1 - x))**2) + \
                         np.sum(np.maximum(0, r - y)**2) + \
                         np.sum(np.maximum(0, r - (1 - y))**2)
        overlap_term = np.sum(np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - distance.pdist(z[:2*k].reshape(k, 2)).flatten())**2)
        return -np.sum(r) + 1e6 * (boundary_term + overlap_term)

    def jacobian(z):
        k = len(z) // 3
        x, y, r = z[:k], z[k:2*k], z[2*k:]
        grad_x = -np.gradient(r, x, edge_order=2) - 2 * np.maximum(0, r - x) * np.heaviside(r - x, 0) + \
                  -np.gradient(r, y, edge_order=2) - 2 * np.maximum(0, r - (1 - x)) * np.heaviside(r - (1 - x), 0)
        grad_y = -np.gradient(r, x, edge_order=2) - 2 * np.maximum(0, r - y) * np.heaviside(r - y, 0) + \
                  -np.gradient(r, y, edge_order=2) - 2 * np.maximum(0, r - (1 - y)) * np.heaviside(r - (1 - y), 0)
        grad_r = -np.ones_like(r) + 2 * np.sum(np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - distance.pdist(z[:2*k].reshape(k, 2)).flatten())**2, axis=1)
        return np.concatenate([grad_x, grad_y, grad_r])

    centres, radii = init_hex_grid()
    z = np.hstack((centres[:, 0], centres[:, 1], radii))
    bounds = [(0, 1)] * 130 + [(0, None)] * 26
    res = minimize(objective, z, jac=jacobian, bounds=bounds, method='L-BFGS-B', options={'gtol': 1e-6, 'maxiter': 1500})
    
    return res.x[:26].reshape(-1, 2), np.maximum(res.x[26:52], 0), np.sum(np.maximum(res.x[52:], 0))
