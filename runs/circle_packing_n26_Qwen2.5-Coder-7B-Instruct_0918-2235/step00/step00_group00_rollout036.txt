import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    n_cands = len(candidates)
    n_placed = len(placed_centres)
    dists = np.linalg.norm(candidates[:, np.newaxis, :] - placed_centres[np.newaxis, :, :], axis=-1)
    max_radii = np.minimum.reduce([
        dists - placed_radii[:, None],
        candidates[:, 0] + 1e-9,
        1 - candidates[:, 0] - 1e-9,
        candidates[:, 1] + 1e-9,
        1 - candidates[:, 1] - 1e-9
    ], axis=0)
    return np.maximum(0.0, max_radii.min(axis=1))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_h = 0.05
    eps = 1e-6
    candidates = np.array([(x, y) for x in np.arange(eps, 1-eps, grid_h) for y in np.arange(eps, 1-eps, grid_h)])
    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]

        if feasible_idx.size == 0:
            break

        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])

        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    n = len(placed_centres)
    C = np.array(placed_centres)
    r = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(r.sum())
    constraints = [
        r <= placed_centres[:, 0] + 1e-9,
        r <= 1 - placed_centres[:, 0] - 1e-9,
        r <= placed_centres[:, 1] + 1e-9,
        r <= 1 - placed_centres[:, 1] - 1e-9,
        cp.sum(r - placed_centres[:, np.newaxis] - placed_centres[np.newaxis, :], axis=1) <= (r + placed_radii[:, np.newaxis] + placed_radii[np.newaxis, :]).T
    ]
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-9)

    return C, r.value, r.value.sum()

centers, radii, sum_radii = run_packing()
