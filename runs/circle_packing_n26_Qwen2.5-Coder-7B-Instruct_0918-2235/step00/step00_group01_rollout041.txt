import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.stats import norm

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    order = np.random.permutation(n)
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    MIN_R = 1e-8

    def dist2(i, j):
        return np.sum((pts[i] - pts[j]) ** 2)

    for t, k in enumerate(order):
        S = 5000
        samples = np.random.rand(S, 2)
        valid_samples = []
        for x, y in samples:
            if all(dist2(x, y, pts[j]) >= (rad[j] + MIN_R) ** 2 for j in range(t)) and \
               x >= MIN_R and x <= 1 - MIN_R and y >= MIN_R and y <= 1 - MIN_R:
                valid_samples.append((x, y))
        if not valid_samples:
            rad /= 1.001
            continue
        valid_samples = np.array(valid_samples)
        clearance = np.minimum(np.minimum(valid_samples[:, 0], 1 - valid_samples[:, 0]),
                                np.minimum(valid_samples[:, 1], 1 - valid_samples[:, 1]))
        clearance -= np.minimum(dist2(valid_samples, pts[:t]).reshape(S, t), rad[:t] ** 2)
        if clearance.min() < MIN_R:
            rad /= 1.001
            continue
        best_idx = clearance.argmax()
        pts[k] = valid_samples[best_idx]
        rad[k] = clearance[best_idx]

    shake_scale = 1e-4
    pts += (np.random.rand(n, 2) - 0.5) * shake_scale
    pts = np.maximum(0, np.minimum(pts, 1 - rad[:, np.newaxis]))

    T_init = 1.0
    T_final = 1e-4
    cooling_rate = 0.999
    epsilon_w = 1.0
    epsilon_p = 1.0
    T = T_init

    while T > T_final:
        i = np.random.randint(n)
        dx, dy = np.random.randn(2) * T * 0.01
        x, y = pts[i] + dx, pts[i] + dy
        x = np.clip(x, rad[i], 1 - rad[i])
        y = np.clip(y, rad[i], 1 - rad[i])

        clearance = min(min(x, 1 - x, y, 1 - y), np.min([np.sqrt(dist2(i, j)) - rad[j] for j in range(n) if j != i]))
        new_radius = min(clearance, rad[i])
        if new_radius < rad[i]:
            epsilon_w *= 0.999
        else:
            epsilon_p *= 0.999
        new_U = epsilon_w * 0.5 * (1 / clearance) ** 0.5 + epsilon_p * 0.5 * sum((rad[j] + rad[i] - dist2(i, j)) ** 2 for j in range(n) if j != i)
        old_U = epsilon_w * 0.5 * (1 / rad[i]) ** 0.5 + epsilon_p * 0.5 * sum((rad[j] + rad[i] - dist2(i, j)) ** 2 for j in range(n) if j != i)
        delta_U = new_U - old_U
        if delta_U < 0 or np.random.rand() < np.exp(-delta_U / T):
            pts[i] = [x, y]
            rad[i] = new_radius

        T *= cooling_rate

    feasible = True
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(dist2(i, j))
            if dist < rad[i] + rad[j] - 1e-12:
                feasible = False
                break

    if not feasible:
        raise Exception("Pack failed to feasible state: Overlapping circles found!")

    pts += (np.random.rand(n, 2) - 0.5) * shake_scale

    final_rad = rad.copy()
    for i in range(n):
        r_i = min(min(pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1]), np.min([np.sqrt(dist2(i, j)) - final_rad[j] for j in range(n) if j != i]))
        final_rad[i] = max(MIN_R, r_i)

    return pts, final_rad, np.sum(final_rad)
