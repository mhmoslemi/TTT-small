import numpy as np
from scipy.spatial.distance import pdist, squareform

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)  # Fixed seed for reproducibility

    P = 60  # population size
    w = 0.7298  # inertia weight
    c1 = 1.49618  # cognitive constant
    c2 = 1.49618  # social constant
    G = 200  # maximum generations

    n_circles = 26
    bounds = np.array([(0, 1), (0, 1), (0, 0.5)])

    # Initial particle population
    pos = np.hstack([np.random.rand(P, n_circles, 2), np.zeros((P, n_circles, 1))])
    vel = np.zeros_like(pos)

    lambda_penalty = 1000  # Initial penalty multiplier

    best_particle = None
    best_value = -np.inf

    for gen in range(G):
        for i in range(P):
            # Evaluate new particle fitness
            fitness = np.sum(pos[i, :, 2]) - lambda_penalty * penalty(pos[i, :, :])
            if fitness > best_value:
                best_value = fitness
                best_particle = pos[i]

        # Update global best
        g_best = best_particle

        # Update particles
        for i in range(P):
            # Update velocity
            vel[i] = w * vel[i] + c1 * np.random.rand(n_circles, 3) * (g_best[i] - pos[i]) + c2 * np.random.rand(n_circles, 3) * (best_particle[i] - pos[i])
            # Apply velocity cap
            vel[i] = np.clip(vel[i], -bounds, bounds)
            pos[i] = pos[i] + vel[i]
            # Clip positions to bounds
            pos[i, :, :2] = np.clip(pos[i, :, :2], 0, 1)
            pos[i, :, 2] = np.clip(pos[i, :, 2], 0, 0.5)
        # Gradually increase penalty at each generation
        lambda_penalty *= 1.02

        # Check for premature convergence
        if gen % 15 == 14 and fitness == best_value:
            print("Early stopping: no improvement for 15 generations.")
            break

    # Refine the best particle with deterministic gradient descent
    def radius_loss(x):
        centers, radii = x[:, :n_circles, :2], x[:, :n_circles, 2]
        distances = scipy.spatial.distance_matrix(centers, centers)
        overlap = radii[:, np.newaxis] + radii[np.newaxis, :] - distances
        overlap_dist = np.maximum(overlap, 0.0)
        return -np.sum(radii) + np.sum(overlap_dist * 1e6)

    def radius_grad(x):
        centers, radii = x[:, :n_circles, :2], x[:, :n_circles, 2]
        distances = scipy.spatial.distance_matrix(centers, centers)
        overlap = radii[:, np.newaxis] + radii[np.newaxis, :] - distances
        distance_gradients = np.zeros_like(x)
        distance_gradients[:, :n_circles, 2] = -np.sum(np.where(overlap > 0, (overlap_dist * np.sign(centers[:, np.newaxis, :2] - centers[np.newaxis, :, :2]) / distances[:, np.newaxis, np.newaxis] + overlap_dist[:, :, np.newaxis] / distances[:, np.newaxis, :]), 0), axis=2)
        return distance_gradients.flatten()

    best_radius_update = scipy.optimize.minimize(radius_loss, best_particle.flatten(), method='BFGS', jac=radius_grad, options={'maxiter': 100}, bounds=[(-bounds[0], bounds[1]*2) for _ in range(2), (0,)] * n_circles)

    final_packing = best_radius_update.x.reshape((P, n_circles, 3))
    best_centers = final_packing[:, :, :2]
    best_radii = final_packing[:, :, 2]
    sum_radii = np.sum(best_radii)
    
    # Validation
    valid, message = validate_packing(best_centers, best_radii)
    if not valid:
        raise ValueError(message)

    return best_centers, best_radii, sum_radii

def penalty(x):
    # Penalty function if not feasible
    _, _, sum_radii = run_packing()
    return sum_radii * 1e6  # Slack penalty if sum_radii is below target

centers, radii, sum_radii = run_packing()
