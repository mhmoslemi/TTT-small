```python
import numpy as np
import cvxpy as cp
import scipy.spatial.distance

def compute_r_max(candidates, placed_centres, placed_radii):
    def feasible_r_max(x, y):
        r = np.min([
            x, 1-x, y, 1-y,
            np.min([
                np.sqrt((x - cx)**2 + (y - cy)**2) - rj
                for cx, cy, rj in zip(placed_centres[:, 0], placed_centres[:, 1], placed_radii)
            ])
        ])
        return np.maximum(0, r)

    return np.vectorize(feasible_r_max)(*candidates.T)

def run_packing():
    h = 0.05
    epsilon = 1e-6
    grid = np.array([(i * h + epsilon, j * h + epsilon)
                     for i in range(21) for j in range(21)])
    candidates = np.array([
        p for p in grid if any([
            not (x < 0 or x > 1 or y < 0 or y > 1)
            for x, y in [p]
        ])
    ])

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]

        if feasible_idx.size == 0:
            break

        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = scipy.spatial.distance.cdist(candidates[best:best+1], candidates)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[np.where(~too_close[0, :])[1]]

    C = np.array(placed_centres)
    M = np.ones((26, 26)) - np.eye(26)
    r = cp.Variable(26)
    
    constraints = [
        r >= 0,
        r <= C[:, 0],
        r <= 1 - C[:, 0],
        r <= C[:, 1],
        r <= 1 - C[:, 1],
        M @ r <= scipy.spatial.distance.cdist(C, C).flatten(),
    ]

    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)

    result = problem.solve(solver=cp.ECOS, abstol=1e-9)
    if result != np.inf:
        return placed_centres, r.value, result
    else:
        return placed_centres, np.zeros(26), 0.0
```
