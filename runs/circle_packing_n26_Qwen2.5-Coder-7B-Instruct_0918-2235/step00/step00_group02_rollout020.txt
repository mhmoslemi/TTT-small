import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, differential_evolution

def build_lp(C, r):
    n = C.shape[0]
    dists = squareform(pdist(C))
    A = np.empty((n, n))
    max_radius = C[:, 0] + C[:, 1][:, np.newaxis]
    max_radius = np.minimum(max_radius, np.fmin(1-C[:, 0], 1-C[:, 1]))
    A = np.vstack((dists - r[:, None] - r[None, :],
                    np.abs(C[:, 0] - r[:, None]),
                    np.abs(C[:, 1] - r[:, None]),
                    max_radius[:, None] - r[:, None]))
    b = np.hstack((np.zeros(n**2), 1, 1, max_radius))
    return A, b

def compute_sum_radius(C, r):
    return -np.sum(r)

def de_obj(C):
    n = C.shape[0]
    r = np.ones(n)
    A, b = build_lp(C, r)
    lp_result = linprog(-r, A_ub=A, b_ub=b, method='highs', options={'disp': False})
    if lp_result.status != 0:
        return 1e6
    return -np.sum(lp_result.x)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    C = 0.5 + 0.4 * np.random.rand(N*2).reshape(N, 2)

    C[:, 1] = np.clip(C[:, 1], 0, 1)
    bounds = [(0, 1)] * 2 * N
