import numpy as np
from scipy.optimize import minimize

def pack_circles():
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                    y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    def energy(vars, lambda_val):
        n = 26
        sum_r = np.sum(vars[n:])
        overlap_penalties = np.sum(np.maximum(0, 1 - vars[n:]))
        boundary_penalties = np.sum(lambda_val * np.maximum(0, vars[:n] - vars[n:]))
        boundary_penalties += np.sum(lambda_val * np.maximum(0, 1 - vars[:n] - vars[n:]))
        return -sum_r + lambda_val * (overlap_penalties + boundary_penalties)

    def gradient(vars, lambda_val):
        n = 26
        grad = np.zeros(2 * n)
        grad[n:] = -np.ones(n)
        overlap_grads = np.zeros(n)
        boundary_grads = np.zeros(n)
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist > (vars[i] + vars[j]):
                    continue
                p_ij = lambda_val * 2 * (vars[i] + vars[j] - dist) * (-centers[i, 0] + centers[j, 0]) / dist
                overlap_grads[i] += p_ij
                overlap_grads[j] += p_ij
            
            for axis in range(2):
                compile_inf = np.hypot(centers[i, 0] - 0.5 * (1 if axis == 0 else 0),
                                    centers[i, 1] - 0.5 * (1 if axis == 1 else 0))
                pp_bd = lambda_val * 2 * (compile_inf - vars[i])
                boundary_grads[i] += pp_bd

        grad[:n] = boundary_grads
        grad[n:] += overlap_grads
        return grad

    def optimize_for_lambda(lambda_val, init_vars):
        return minimize(energy, init_vars, method='L-BFGS-B', jac=gradient, args=(lambda_val,),
                       bounds=np.array([(0, 1)] * n + [(0, 0.5)] * n))

    n = 26
    min_radius = 1e-4
    max_radius = 0.5
    init_centers = np.random.rand(n, 2)
    init_radii = np.full(n, min_radius)
    init_vars = np.hstack([init_centers.ravel(), init_radii])

    lambda_0 = 1.0
    lambda_max = 1e5
    rho = 1.5
    tol = 1e-8
    max_iter = 1000
    best_sum_radii = np.inf
    best_vars = None

    for _ in range(100):
        lambda_val = lambda_0
        vars = init_vars.copy()
        for _ in range(max_iter):
            result = optimize_for_lambda(lambda_val, vars)
            vars = result.x
            if result.fun < best_sum_radii - tol:
                best_sum_radii = result.fun
                best_vars = vars
            if lambda_val >= lambda_max:
                break
            lambda_val *= rho

    best_centers = best_vars[:n].reshape((n, 2))
    best_radii = best_vars[n:]

    return best_centers, best_radii, best_sum_radii

# Run the packing algorithm
centers, radii, sum_radii = pack_circles()

# Validate and print results
is_valid, message = validate_packing(centers, radii)
print(f"Validity: {is_valid}")
print(f"Sum of radii: {sum_radii:.5f}")
