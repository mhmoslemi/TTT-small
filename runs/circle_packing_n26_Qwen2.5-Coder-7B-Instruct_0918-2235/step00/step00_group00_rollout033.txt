import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(0)
    step_size_start = 0.15
    step_size_end = 0.001
    temperature_start = 1.0
    temperature_end = 1e-4
    max_iter = 20000
    
    def inflation_step(centers, radii):
        for i in range(n):
            valid_radii = []
            for j in range(n):
                if j != i:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist - radii[j] > 0:
                        valid_radii.append(dist - radii[j])
            r_new = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1], valid_radii)
            r_new = max(r_new, 0)
            radii[i] = r_new
        return centers, radii

    def proposal(centers, max_step):
        centers_new = centers.copy()
        i = np.random.randint(n)
        delta_x = np.random.uniform(-max_step, max_step)
        delta_y = np.random.uniform(-max_step, max_step)
        centers_new[i][0] += delta_x
        centers_new[i][1] += delta_y
        centers_new[i] = np.clip(centers_new[i], 0, 1)
        return centers_new

    def accept(centers_new, radii_new, S_new, S_curr, temperature):
        if S_new > S_curr:
            return True, centers_new, radii_new
        accept_prob = np.exp((S_new - S_curr) / temperature)
        if np.random.rand() < accept_prob:
            return True, centers_new, radii_new
        return False, centers, radii

    centers = np.random.rand(n, 2)
    while np.max(np.linalg.norm(centers - centers[:, np.newaxis], axis=2)) < 0.1:
        centers = np.random.rand(n, 2)

    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = -np.inf

    for k in range(max_iter):
        centers_new = proposal(centers, step_size_start * (step_size_end / step_size_start) ** (k / max_iter))
        centers_new, radii = inflation_step(centers_new, best_radii.copy())
        S_new = np.sum(radii)
        S_curr = np.sum(best_radii)
        temperature = temperature_start * (temperature_end / temperature_start) ** (k / max_iter)

        accepted, centers_new, radii_new = accept(centers_new, radii, S_new, S_curr, temperature)

        if accepted and S_new > best_sum:
            best_centers = centers_new.copy()
            best_radii = radii_new.copy()
            best_sum = S_new

        step_size_start = step_size_start * (step_size_end / step_size_start) ** (k / max_iter)
        temperature = temperature_start * (temperature_end / temperature_start) ** (k / max_iter)

    return best_centers, best_radii, best_sum
