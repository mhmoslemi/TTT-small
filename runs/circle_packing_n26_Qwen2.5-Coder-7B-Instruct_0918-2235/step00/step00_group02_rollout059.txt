import numpy as np
from scipy.optimize import linprog, minimize
from scipy.stats import norm

def calc_max_radii(C):
    n = C.shape[0]
    A_eq = np.zeros((n, n))
    for i in range(n):
        A_eq[i, i] = 2
        for j in range(i + 1, n):
            dist_ij = np.linalg.norm(C[i] - C[j])
            A_eq[i, j] = dist_ij
            A_eq[j, i] = dist_ij
    
    b_eq = np.zeros(n)
    for i in range(n):
        b_eq[i] = min(C[i, 0], 1 - C[i, 0], C[i, 1], 1 - C[i, 1])
    
    res = linprog(c=-np.ones(n), A_eq=A_eq, b_eq=b_eq, method='highs', tol=1e-9)
    if not res.success:
        return np.zeros(n), -np.inf
    
    r = -res.fun
    if np.any(r <= 0):
        r = np.zeros(n)
    
    return r, r.sum()

def run_packing():
    np.random.seed(42)
    n = 26
    T0 = 0.1
    alpha = 0.995
    N_max = 50000
    sigma = 0.02
    best_sum = 0
    best_C = None
    best_r = None
    
    def objective(x):
        C = x.reshape(n, 2)
        r, S = calc_max_radii(C)
        if S > 2.636:
            S = 1e20  # Overlarge value to reject
        return -S
    
    def acceptance(Δ, T):
        return Δ >= 0 or np.exp(Δ / T) > np.random.rand()
    
    C = np.random.rand(n, 2) * 0.8 + 0.1  # Hexagonal grid seed
    S = np.sum(calc_max_radii(C)[0])
    best_S = S
    best_C = C
    best_r = calc_max_radii(C)[0]
    T = T0
    
    for k in range(N_max):
        i = np.random.randint(n)
        C_new = C.copy()
        C_new[i] += norm.rvs(scale=sigma, size=2)
        C_new[i] = np.clip(C_new[i], 0, 1)
        r_new, S_new = calc_max_radii(C_new)
        Δ = S_new - S
        
        if acceptance(Δ, T):
            C = C_new
            S = S_new
            if S > best_S:
                best_S = S
                best_C = C
                best_r = r_new
        
        T *= alpha
        
        if S > 2.636 or k == N_max - 1:
            break
    
    np.clip(best_r, 0, None, out=best_r)
    return best_C, best_r, np.sum(best_r)
