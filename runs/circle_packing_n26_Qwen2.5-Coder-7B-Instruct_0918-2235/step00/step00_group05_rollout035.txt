import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance_matrix

def generate_hexagonal_centers(lattice_width,的世界梦 labyrinth_type, steps=26):
    centers = []
    center_radius = lattice_width * np.sqrt(3) / 6
    if labyrinth_type == 'triangle':
        for r in range(steps):
            height = center_radius * np.sqrt(3) * (r + 1)
            row_shift = lattice_width * r / 2
            for c in range(steps - r):
                x = row_shift + c * lattice_width * (3 * center_radius)
                y = height
                if y > 1:
                    break
                centers.append([x, y])
    elif labyrinth_type == 'square':
        height = steps * lattice_width
        step = lattice_width / (steps - 1)
        for r in range(steps):
            for c in range(steps):
                x = c * step
                y = r * step
                if x > 1:
                    break
                if y > 1:
                    break
                centers.append([x, y])
    else:
        height = np.sqrt(3) * lattice_width * steps / 2
        row_shift = lattice_width * np.sqrt(3) / 6
        for r in range(steps):
            height = row_shift * (r + 1)
            for c in range(steps):
                x = (r + c) * lattice_width
                y = height
                if x > 1:
                    break
                if y > 1:
                    break
                centers.append([x, y])
    return np.array(centers)

def solve_lps(centers, r0=1/3):
    n = len(centers)
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    for i, (x, y) in enumerate(centers):
        A_eq[i, i] = -1
        b_eq[i] = -x
        if y <= 1 - x:
            A_eq[i, i] += 1
            b_eq[i] += 1 - x
        if x <= 1 - y:
            A_eq[i, i] += 1
            b_eq[i] += 1 - y
        for j in range(i):
            dist = distance_matrix([centers[i]], [centers[j]])[0][0]
            A_eq[i, j] = 1
            A_eq[j, i] = 1
            b_eq[i] += dist
            b_eq[j] += dist
    result = linprog(np.ones(n), A_ub=A_eq, b_ub=b_eq, bounds=(0.0001, None))
    return result.x, result.fun

def nelder_mead(centers, max_iter=100, tolerance=1e-8):
    def objective(params):
        centers = generate_hexagonal_centers(params[0], 'triangle')
        _, radii = solve_lps(centers)
        return -radii.sum()
    bounds = [(0, np.pi/6), (-0.05, 0.05), (-0.05, 0.05)]
    result = minimize(objective, x0=[np.pi/6, 0, 0], method='nelder-mead', bounds=bounds, options={'maxiter': max_iter, 'xtol': tolerance})
    return result.x.tolist()

def run_packing(steps=26, lattice_width=0.03):
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(100):
        theta = np.random.uniform(0, np.pi/6)
        dx, dy = np.random.uniform(-0.05, 0.05, 2)
        centers = generate_hexagonal_centers(lattice_width, 'triangle')
        centers += [[theta, dx, dy]]
        _, radii = solve_lps(centers)
        if radii.sum() > best_sum:
            best_sum = radii.sum()
            best_centers = centers
            best_radii = radii

    return best_centers, best_radii, best_sum
