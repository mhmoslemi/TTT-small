import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    n = len(placed_centres)
    k = len(candidates)
    r = np.maximum(0, np.minimum(
        np.sqrt(np.sum((candidates[:, np.newaxis, :] - placed_centres[None, :, :]) ** 2, axis=-1)) - placed_radii[None, :],
        candidates[:, 0] - placed_radii[None, :],
        placed_centres[:, 0] - candidates[:, 0],
        candidates[:, 1] - placed_radii[None, :],
        placed_centres[:, 1] - candidates[:, 1],
    ).min(axis=1))
    return r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    grid_spacing = 0.05
    clearance = 1e-6
    h = grid_spacing
    k = int(2 / h)
    candidates = np.array([(x * h, y * h) for x in range(k) for y in range(k)])
    candidates = candidates[(candidates[:, 0] >= clearance) & (candidates[:, 0] <= 1 - clearance) &
                             (candidates[:, 1] >= clearance) & (candidates[:, 1] <= 1 - clearance)]

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    if len(placed_centres) != 26:
        return np.zeros((26, 2)), np.zeros(26), 0.0

    C = np.array(placed_centres)
    N = len(C)
    r = cp.Variable(N)
    one_over_r = {i: cp.Variable() for i in range(N)}
    constraints = [cp.sum(r) <= 26]
    for i in range(N):
        constraints.append(r[i] <= C[i, 0])
        constraints.append(r[i] <= 1 - C[i, 0])
        constraints.append(r[i] <= C[i, 1])
        constraints.append(r[i] <= 1 - C[i, 1])
        for j in range(i + 1, N):
            constraints.append(r[i] + r[j] <= np.linalg.norm(C[i] - C[j]))
        constraints.append(r[i] >= 0)
        one_over_r[i] >= 1 / (r[i] + clearance)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-9)

    if prob.status != cp.OPTIMAL:
        return C, r.value, np.sum(r.value)

    Cadjusted = np.zeros((N, 2))
    for i in range(N):
        Candidate = C.copy()
        old_rplain = r.value[i]
        r_adjusted = old_rplain
        changes = np.full(len(C), 0)
        changes[i] = 1
        prime = 0.02
        old_sum_r = r_adjusted
        prev_sum_r = old_sum_r
        ct = 0
        flag = True
        while flag:
            flag = False
            prev_sum_r = old_sum_r
            for j in range(len(changes)):
                if changes[j] > 0:
                    dt = np.random.uniform(-prime, prime, size=2)
                    dt = dt / np.linalg.norm(dt)
                    Cadjusted[j, :] = C[j, :] + dt
                    Cadjusted[j, 0] = np.clip(Cadjusted[j, 0], 0 + clearance, 1 - clearance)
                    Cadjusted[j, 1] = np.clip(Cadjusted[j, 1], 0 + clearance, 1 - clearance)
                    radjusted = np.array([np.maximum(0, np.minimum(
                        np.sqrt(np.sum((Cadjusted[:, np.newaxis, :] - C[j, :]) ** 2, axis=-1)), (np.clip(Cadjusted[:, 0], 0 + clearance, 1 - clearance)) - 1e-9,
                        np.clip(C[j, 0], 0 + clearance, 1 - clearance) - np.clip(Cadjusted[:, 0], 0 + clearance, 1 - clearance),
                        np.clip(Cadjusted[:, 1], 0 + clearance, 1 - clearance) - 1e-9,
                        np.clip(C[j, 1], 0 + clearance, 1 - clearance) - np.clip(Cadjusted[:, 1], 0 + clearance, 1 - clearance),
                    ).min(axis=1)] * N)
                    for dx in range(26):
                        if dx != j:
                            radjusted[j] = (np.maximum(0, np.minimum(
                                np.sqrt(np.sum((C[j, : np.newaxis] - C[dx, :]) ** 2, axis=-1)), (np.clip(C[dx, 0], 0 + clearance, 1 - clearance)) - 1e-9,
                                np.clip(C[j, 0], 0 + clearance, 1 - clearance) - np.clip(C[dx, 0], 0 + clearance, 1 - clearance),
                                np.clip(C[dx, 1], 0 + clearance, 1 - clearance) - 1e-9,
                                np.clip(C[j, 1], 0 + clearance, 1 - clearance) - np.clip(C[dx, 1], 0 + clearance, 1 - clearance),
                            ).min(axis=1)))
                    sum_radjusted = np.sum(radjusted)
                    if sum_radjusted > old_sum_r:
                        r_adjusted = radjusted
                        flag = True
                        old_sum_r = sum_radjusted
                        changes[j] = 0

            if sum_radjusted > old_sum_r:
                old_sum_r = sum_radjusted
                r_adjusted = radjusted
                Cadjusted = Cadjusted
            else:
                if ct <= 1:
                    prime = 0.02
                elif ct <= 2:
                    prime = 0.01
                elif ct <= 3:
                    prime = 0.005
                else:
                    prime = 0.001
                    flag = False
                Cadjusted = C.copy()
            ct += 1
        r.value[i] = r_adjusted[i]
        C = Cadjusted

    r = cp.Variable(N)
    constraints = [cp.sum(r) == 26]
    for i in range(N):
        constraints.append(r[i] <= C[i, 0])
        constraints.append(r[i] <= 1 - C[i, 0])
        constraints.append(r[i] <= C[i, 1])
        constraints.append(r[i] <= 1 - C[i, 1])
        for j in range(i + 1, N):
            constraints.append(r[i] + r[j] <= np.linalg.norm(C[i] - C[j]))
        constraints.append(r[i] >= 0)
        for k in range(N):
            constraints.append(r[k] == abs(one_over_r[k]))

    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-9)

    maxvol = prob.value
    rmax = cp.Variable(N)
    constraints = [cp.sum(rmax) == 26]
    for i in range(N):
        constraints.append(rmax[i] <= C[i, 0])
        constraints.append(rmax[i] <= 1 - C[i, 0])
        constraints.append(rmax[i] <= C[i, 1])
        constraints.append(rmax[i] <= 1 - C[i, 1])
        for j in range(i + 1, N):
            constraints.append(rmax[i] + rmax[j] <= np.linalg.norm(C[i] - C[j]))
        constraints.append(rmax[i] >= 0)
        for k in range(N):
            constraints.append(rmax[k] == abs(one_over_r[k]))
    prob = cp.Problem(cp.Maximize(cp.sum(rmax)), constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-9)
    Cmaxvol = C
    rmaxvol = cp.value(rmax)
    sum_rmaxvol = cp.value(cp.sum(rmax))

    return Cmaxvol, rmaxvol, sum_rmaxvol
