import numpy as np
import cvxpy as cp

def greedy_insertion(candidates, placed_centres, placed_radii):
    r_candidates = np.min(np.linalg.norm(candidates[:, np.newaxis, :] - np.atleast_2d(placed_centres), axis=2) - placed_radii[:, np.newaxis] - 1e-9, axis=1)
    r_candidates(np.where(r_candidates < 0)) = 0
    feasible_idx = np.where(r_candidates > 0)[0]
    if feasible_idx.size == 0:
        return False, None, None
    best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
    placed_centres.append(candidates[best])
    placed_radii.append(r_candidates[best])
    dists = np.linalg.norm(candidates - candidates[best], axis=1)
    too_close = dists < (r_candidates[best] + 1e-9)
    candidates = candidates[~too_close]
    return True, np.array(placed_centres), np.array(placed_radii)

def lp_radius_optimization(centers, radii):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r <= cp.max(cp.concat([centers[:, 0] - r, r, centers[:, 1] - r, r], axis=1), axis=1), r >= 0]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= np.linalg.norm(centers[i] - centers[j]))
    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-9)
    return r.value

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    candidates = np.array([(i*h + epsilon, j*h + epsilon) for i in range(1, 19) for j in range(1, 19)])
    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        success, new_placed_centres, new_placed_radii = greedy_insertion(candidates, placed_centres, placed_radii)
        if not success:
            break

    new_placed_centres, new_placed_radii = np.array(new_placed_centres), np.array(new_placed_radii)

    for _ in range(5000):
        accepted = False
        temperature = 0.02
        for _ in range(100):
            index = np.random.randint(26)
            dx, dy = np.random.uniform(-temperature, temperature, 2)
            new_centers = new_placed_centres.copy()
            new_centers[index] += np.array([dx, dy])
            new_centers[index] = np.clip(new_centers[index], 0, 1)
            new_radii = lp_radius_optimization(new_centers, new_placed_radii)
            if np.sum(new_radii) > np.sum(new_placed_radii):
                new_placed_centres, new_placed_radii = new_centers, new_radii
                accepted = True
            temperature *= 0.995
        if not accepted:
            break

    return new_placed_centres, new_placed_radii, np.sum(new_placed_radii)
