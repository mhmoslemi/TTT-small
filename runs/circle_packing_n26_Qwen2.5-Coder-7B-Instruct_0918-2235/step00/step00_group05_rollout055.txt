import numpy as np
from scipy.optimize import linear_sum_assignment, linprog, minimize
from scipy.spatial.distance import cdist

def lp_radii_fixup(center, radii, L):
    n = len(center) // 2
    Awall = np.zeros((4 * n, n))
    Jwall = np.zeros((4 * n, n))
    for i in range(n):
        Awall[4 * i, i] = 1
        Jwall[4 * i, i] = 1
        Awall[4 * i + 1, i] = -1
        Jwall[4 * i + 1, i] = -1
    for i in range(n):
        Awall[4 * i + 2, i] = 1
        Jwall[4 * i + 2, i] = 1
        Awall[4 * i + 3, i] = -1
        Jwall[4 * i + 3, i] = -1
    rmax = np.minimum(center[:n], center[n:])
    Awall_obj = Awall - 2 * rmax[:, np.newaxis]
    rmax_Ok = np.sum(np.maximum(Awall_obj, 0), axis=0)
    Awall_wall = np.hstack((Awall, -0.5 * np.eye(n)))
    Awall_wall[2 * n:, n:] *= -1
    Awall_radius = np.vstack((Awall_wall, -Awall_obj))
    Jwall_wall = np.hstack((Jwall, -0.5 * np.eye(n)))
    Jwall_wall[2 * n:, n:] *= -1
    Jwall_radius = np.vstack((Jwall_wall, -Jwall_obj))
    Aeq_radius = np.zeros((4 * n, n))
    Feq_radius = np.zeros(4 * n)
    b = np.zeros(4 * n + 2 * n)
    b[:2*n] = L
    res = linprog(-rmax_wall, A_eq=Aeq_radius, b_eq=Feq_radius, A_ub=Awall_radius, b_ub=b, method='highs')
    return res.x[:n]

def gradient_penalty(center, radii, L):
    sum_radii = np.sum(radii)
    L_wall = np.zeros(4 * len(center) // 2)
    L_overlap = np.zeros(len(center) // 2 * (len(center) // 2 - 1) // 2)
    for i in range(len(center) // 2):
        L_wall[4 * i] = max(0, radii[i] - center[i])
        L_wall[4 * i + 1] = max(0, radii[i] - (1 - center[i]))
        L_wall[4 * i + 2] = max(0, radii[i] - center[i + len(center) // 2])
        L_wall[4 * i + 3] = max(0, radii[i] - (1 - center[i + len(center) // 2]))
    L_max = 0
    for i in range(len(center) // 2):
        for j in range(i + 1, len(center) // 2):
            L_dist = cdist([center[i]], [center[j + len(center) // 2]], 'euclidean')[0][0]
            L_overlap_ij = max(0, radii[i] + radii[j] - L_dist)
            L_max = max(L_max, L_overlap_ij)
    return sum_radii - L_wall.sum() - L_overlap.sum() * L_max

def shake_up(center, radii, k, delta):
    n = len(center) // 2
    for _ in range(k):
        indices = np.random.choice(n, size=k, replace=False)
        center[indices] += np.random.uniform(-delta, delta, 2 * k)
        center[indices] = np.clip(center[indices], 0, 1)
    return center, radii

def gradient_descent_with_penalty(center, radii, L, tol, max_iter):
    n = len(center) // 2
    step_size = 0.1
    for _ in range(max_iter):
        gradient = np.zeros(2 * n + n)
        for i in range(n):
            xi, yi, ri = center[i], center[i + n], radii[i]
            for j in range(i + 1, n):
                xj, yj, rj = center[j], center[j + n], radii[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                if dist < ri + rj - 1e-12:
                    penalty_ij = 1
                    gradient[i*n:i*n+2] -= penalty_ij * ((xi - xj) / dist)
                    gradient[j*n:j*n+2] += penalty_ij * ((xi - xj) / dist)
                else:
                    penalty_ij = 0
                    gradient[i*n:i*n+2] -= penalty_ij * ((xi - xj) / dist)
                    gradient[j*n:j*n+2] += penalty_ij * ((xi - xj) / dist)
            wall_penalty = penalty_wall(xi, yi, ri)
            gradient[i*n:i*n+2] -= wall_penalty * (np.sign(xi - ri - 0.5) + np.sign(xi - (1 - ri) - 0.5))
            gradient[i*n+n] += wall_penalty
        delta = step_size * gradient / np.linalg.norm(gradient)
        center -= delta
        L_new = lp_radii_fixup(center, radii, L)
        if L_new.sum() > L.sum():
            step_size *= 2
        else:
            step_size *= 0.5
        if np.linalg.norm(gradient) < tol:
            return center, radii
    return center, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    center = np.random.uniform(0, 1, 2 * n)
    radii = np.random.uniform(0, 0.5, n)
    L = lp_radii_fixup(center, radii, 1e4)
    best_center = center.copy()
    best_radii = radii.copy()
    best_sum = L.sum()
    for _ in range(8):
        center_jitter = np.random.uniform(-0.01, 0.01, 2 * n)
        center += center_jitter
        center[:n] = np.maximum(np.minimum(center[:n], 1), 0)
        center[n:] = np.maximum(np.minimum(center[n:], 1), 0)
        L = lp_radii_fixup(center, radii, 1e4)
        center, radii = gradient_descent_with_penalty(center, radii, L, 1e-12, 2000)
        radii = lp_radii_fixup(center, radii, L)
        center, radii, sum_radii = shake_up(center, radii, 3, 0.01)
        if sum_radii > best_sum:
            best_center = center.copy()
            best_radii = radii.copy()
            best_sum = sum_radii
    return best_center, best_radii, best_sum
