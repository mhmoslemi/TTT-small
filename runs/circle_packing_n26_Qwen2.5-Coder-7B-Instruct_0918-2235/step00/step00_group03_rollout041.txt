import numpy as np
from scipy.optimize import linprog
from itertools import permutations

def compute_distances(centers):
    n = centers.shape[0]
    distances = np.zeros((n, n))
    for i, (x1, y1) in enumerate(centers):
        for j in range(i + 1, n):
            x2, y2 = centers[j]
            d = np.sqrt((x2 - x1)**2 + (y2 - y1)**2)
            distances[i, j] = d
            distances[j, i] = d
    return distances
