import numpy as np
from scipy.optimize import differential_evolution

def penalized_objective(v):
    n = v.shape[0] // 3
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            penalty += np.maximum(0, r[i] + r[j] - dij)**2
    penalty += np.maximum(0, r - x)**2
    penalty += np.maximum(0, r - (1 - x))**2
    penalty += np.maximum(0, r - y)**2
    penalty += np.maximum(0, r - (1 - y))**2
    return -np.sum(r) + 1000 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    result = differential_evolution(penalized_objective, bounds, popsize=10, maxiter=50, tol=1e-12, disp=False)
    centers = result.x[:n] + result.x[2*n:]
