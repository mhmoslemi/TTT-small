import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

# Helper function to compute the objective value for the LP
def lp_objective(centers):
    n = centers.shape[0] // 2
    c = np.ones(n)
    r_bounds = [(0, None)] * n
    x, y = centers[:n], centers[n:]
    d = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :]) ** 2 + (y[:, np.newaxis] - y[np.newaxis, :]) ** 2)
    A_ub = np.vstack([-np.eye(n), np.eye(n), -np.eye(n), np.eye(n), np.zeros_like(d) + d])
    b_ub = np.hstack([x, 1 - x, y, 1 - y, d.flatten()])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    return -result.fun

    r_bounds = [(0, None)] * n
    c = np.ones(n)
