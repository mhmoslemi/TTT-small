import numpy as np
from scipy import optimize as opt

def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    r = np.ones(n_circles)
    A_ub = []
    b_ub = []
    
    for i in range(n_circles):
        A_ub.append([0] * n_circles)
        A_ub[i][i] = 1
        b_ub.append(centres_i[i, 0])
        A_ub.append([0] * n_circles)
        A_ub[i][i] = 1
        b_ub.append(1 - centres_i[i, 0])
        A_ub.append([0] * n_circles)
        A_ub[i][i] = 1
        b_ub.append(centres_i[i, 1])
        A_ub.append([0] * n_circles)
        A_ub[i][i] = 1
        b_ub.append(1 - centres_i[i, 1])
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = abs(centres_i[i, 0] - centres_i[j, 0])
            dy = abs(centres_i[i, 1] - centres_i[j, 1])
            d = np.sqrt(dx**2 + dy**2)
            A_ub.append([0] * n_circles)
            A_ub[-1][i] = 1
            A_ub[-1][j] = 1
            b_ub.append(d)
    
    c = -np.ones(n_circles)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    n_particles = 30
    n_circles = 26
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    max_iter = 3000
    vel_clip = 0.1
    alpha = 0.999
    
    dx = 1/5.0
    grid = np.array([(x + np.random.normal(0, 0.01), y + np.random.normal(0, 0.01))
                      for x in np.linspace dx, 5, 5)
                      for y in np.linspace dx, 5, 2)])
    grid = grid[:25]
    
    centres = np.tile(grid, (n_particles, 1, 1)) + np.random.normal(0, 1e-2, (n_particles, n_circles, 2))
    velocities = np.zeros_like(centres)
    pbests = centres.copy()
    pbest_sums = np.array([solve_radius_lp(centres_i)[1] for centres_i in centres])
    gbest_idx = np.argmax(pbest_sums)
    gbest = pbests[gbest_idx]
    gbest_sum = pbest_sums[gbest_idx]
    
    for iteration in range(max_iter):
        for i, pos_i in enumerate(centres):
            velocities[i] = inertia * velocities[i] + cognitive * np.random.rand() * (pbests[i] - pos_i) + social * np.random.rand() * (gbest - pos_i)
            if np.linalg.norm(velocities[i]) > vel_clip:
                velocities[i] /= np.linalg.norm(velocities[i]) * vel_clip
            pos_i += velocities[i]
            pos_i = np.clip(pos_i, 0, 1)
            radii_i, sum_radii_i = solve_radius_lp(pos_i)
            if sum_radii_i > pbest_sums[i]:
                pbests[i] = pos_i
                pbest_sums[i] = sum_radii_i
                if sum_radii_i > gbest_sum:
                    gbest = pbests[i]
                    gbest_sum = sum_radii_i
        if iteration % 100 == 0:
            print(f"Iteration {iteration}, Sum of radii: {gbest_sum}")
        if iteration % 500 == 0:
            for i in np.random.choice(range(n_particles), 4):
                centres[i] = np.tile(grid, (1, 1, 1)) + np.random.normal(0, 1e-2, (1, n_circles, 2))
                velocities[i] = np.zeros_like(centres[i])
    
    return gbest, solve_radius_lp(gbest)[0], gbest_sum
