import cma
import numpy as np
from scipy.optimize import linprog

# Validate packing as per the question statement
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

# Objective function for CMA-ES (radius sum minus penalties)
def objective(z):
    n = len(z) // 2
    centres = z[:n].reshape(n, 2)
    radii = z[n:]
    # Compute penalties
    wall_penalty = np.sum(np.sum(np.maximum(0, radii - np.minimum(centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1])), axis=1) ** 2)
    overlap_penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            overlap_penalty += np.sum(np.maximum(0, (radii[i] + radii[j]) - dist) ** 2)
    penalty = wall_penalty + overlap_penalty
    # Use large penalties to ensure infeasibility only at machine precision
    objective_value = np.sum(radii) - 1e4 * penalty if penalty < 1e-12 else np.inf
    return objective_value

# Solve maximum radius linear program for fixed centre set
def refine_radii(centres):
    n = centres.shape[0]
    # Set up linear programming problem
    A = np.empty((4 + n * (n - 1), 2 * n))
    b = np.empty(4 + n * (n - 1))
    
    # x_ij <= min(x_i, 1 - x_i, y_i, 1 - y_i)
    A[:n * 2] = np.vstack([np.eye(2 * n), -np.eye(2 * n)])
    b[:n * 2] = np.hstack([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]])
    
    # Radii sum constraint (arbitrary large value except infinity)
    A[n * 2] = np.ones(2 * n)
    b[n * 2] = 26
    
    # Symmetry and circle proximity constraints
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist_matrix[i, j] = (np.sum((centres[i] - centres[j]) ** 2))**0.5
    for i in range(n):
        A[i * 2 + 2:, 2 * i:2 * (i + 1)] = [[1, 0], [0, 1], [-1, 0], [0, -1]] + np.array([[0, 0]] * (n - i - 1)) * dist_matrix[i, i + 1:]
        A[i * 2 + 2:, 2 * j:2 * (j + 1)] = [[0, 0]] * i + np.array([[1, 0], [0, 1], [-1, 0], [0, -1]]) + [[0, 0]] * (n - j - 1) * dist_matrix[j, j + 1:]
        b[i * 2 + 2:] = dist_matrix[i, j] + np.array([np.inf] * (n - i - 1) + [np.inf] * (n - j - 1))
    
    # Solve linear program
    res = linprog([1e-10] * (2 * n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    radii = res.x[:n]
    return radii

# Main function to run packing
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Generate initial centers using a hexagonal lattice
    a = np.sqrt(2 * (1 - 1 / np.sqrt(3))) / 2
    b = 2 * a / np.sqrt(3)
    centres = np.array([(i % 3) * a + 0.5 * (i % 3) % 2 * b + np.random.uniform(-0.01, 0.01, size=1),
                        (i // 3) * b + np.random.uniform(-0.01, 0.01, size=1)])
    r = refine_radii(centres)
    z = np.concatenate([centres.flatten(), r])
    
    # Set up CMA-ES
    es = cma.CMAEvolutionStrategy(z, 0.1, options={'bounds': [(0, 1)] * 2 * n + [(0, None)] * n})
    es.optimize(objective, verb_disp=False)
    
    # Optional SA "shake-up" if needed
    best_z = es.result[0]
    refine_radii(best_z[:n * 2])
    for _ in range(200):
        shake_indices = np.random.choice(n, 3)
        shake_displacements = np.random.uniform(-0.01, 0.01, size=(n, 2))
        for idx in shake_indices:
            best_z[idx * 2:idx * 2 + 2] += shake_displacements[idx]
        refine_radii(best_z[:n * 2])
        if objective(best_z) > objective(es.result[0]):
            es.result[0] = best_z.copy()
        else:
            break
    
    # Validation
    if validate_packing(es.result[0][:n * 2].reshape(n, 2), es.result[0][n:])[0]:
        return es.result[0][:n * 2].reshape(n, 2), es.result[0][n:], es.result[0][n:].sum()
    else:
        # Fallback: re-compute optimal radii on last known feasible centre set
        refine_radii(es.result[0][:n * 2])
        return es.result[0][:n * 2].reshape(n, 2), es.result[0][n:], es.result[0][n:].sum()
