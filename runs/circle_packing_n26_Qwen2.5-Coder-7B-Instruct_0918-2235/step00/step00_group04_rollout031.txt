import numpy as np
import scipy.optimize as opt

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # maximize sum of radii
    A_ub = np.empty((4 * n + n * (n - 1) // 2, n))
    b_ub = np.empty(4 * n + n * (n - 1) // 2)

    # inequalities: r_i <= x_i
    np.fill_diagonal(A_ub, 1)
    b_ub[:n] = centers[:, 0]

    # inequalities: r_i <= 1 - x_i
    A_ub[n:2*n, ::2] = -1
    b_ub[n:2*n] = centers[:, 0]

    # inequalities: r_i <= y_i
    np.fill_diagonal(A_ub[2*n:n*2], 1)
    b_ub[2*n:n*2] = centers[:, 1]

    # inequalities: r_i <= 1 - y_i
    A_ub[n*2:3*n, ::2] = -1
    b_ub[n*2:3*n] = centers[:, 1]

    # inequalities: r_i + r_j <= d_ij
    idx = 3 * n
    for i in range(n):
        for j in range(i + 1, n):
            d_ij_squared = np.sum((centers[i] - centers[j]) ** 2)
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij_squared
            idx += 1

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        raise ValueError("LP failed")
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    # Create a 5x6 grid and delete 4 points to form 26 centers
    grid = np.mgrid[0.1:1:0.2, 0.1:1:0.2].reshape(2, -1).T
    shuffle_indices = np.arange(len(grid))
    np.random.shuffle(shuffle_indices)
    centers = grid[shuffle_indices][:n, :]
    current_radii, current_sum = compute_lp_solution(centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    epsilon = 1e-12

    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        delta = np.random.uniform(-step, step, size=2)
        new_center = centers[i] + delta
        new_center = np.clip(new_center, 0, 1)
        new_centers = centers.copy()
        new_centers[i] = new_center
        try:
            new_radii, new_sum = compute_lp_solution(new_centers)
        except ValueError:
            new_sum = -np.inf
        if new_sum > current_sum + epsilon:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
                best_sum = current_sum
        if iter % K_JUMP == 0:
            new_centers = np.random.uniform(0, 1, size=(n, 2))
            new_radii, new_sum = compute_lp_solution(new_centers)
            if new_sum > best_sum:
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
                best_sum = new_sum
        if iter % N_REDUCE == 0 and step > 1e-4:
            step *= 0.9

    return best_centers, best_radii, best_sum
