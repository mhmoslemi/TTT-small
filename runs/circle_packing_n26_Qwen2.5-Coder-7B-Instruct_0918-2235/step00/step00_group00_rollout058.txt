import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    n = len(placed_centres)
    m = len(candidates)
    dists = np.array([np.linalg.norm(candidates - centre, axis=1) for centre in placed_centres])
    r_max = np.minimum(np.minimum(
        np.min(dists - np.expand_dims(placed_radii, axis=1), axis=1), candidates[:, 0]
    ), np.minimum(candidates[:, 1], 1 - candidates[:, 0]))
    return np.maximum(r_max, np.zeros(m))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    candidates = np.array([(x, y) for x in np.arange(h, 1, h) for y in np.arange(h, 1, h)])
    candidates = candidates[(candidates[:, 0] >= epsilon) & (candidates[:, 0] <= 1 - epsilon) &
                            (candidates[:, 1] >= epsilon) & (candidates[:, 1] <= 1 - epsilon)]

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    prob = cp.Problem()
    n = len(placed_centres)
   radii = cp.Variable(n, nonneg=True)
    constraints = [radii <= placed_centres[:, 0],
                   radii <= 1 - placed_centres[:, 0],
                   radii <= placed_centres[:, 1],
                   radii <= 1 - placed_centres[:, 1]]
    dists_sq = np.sum((np.array(placed_centres) - np.outer(placed_centres[:, None], np.ones(n))) ** 2, axis=2)
    dists = np.maximum(dists_sq, 1e-12)
    for i in range(n):
        constraints.extend([radii[i] + radii[j] <= dists[i, j] for j in range(i + 1, n)])
    prob += cp.Maximize(cp.sum(radii))
    prob.solve(solver=cp.ECOS, abstol=1e-9)

    placed_radii = prob.value
    return np.array(placed_centres), placed_radii, np.sum(placed_radii)
