def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Build initial centre grid and radii
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)

    # 2. Define objective and Jacobian
    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, lambda_B, lambda_O)

    def jacobian(z):
        x, y, r = unpack(z)
        return compute_penalty_gradient(x, y, r, lambda_B, lambda_O)

    # 3. Bounds
    bounds = [(0,1)]*52 + [(0,None)]*26

    # 4. Run trust-constr optimiser
    res = minimize(objective, z,
                   jac=jacobian,
                   method='trust-constr',
                   bounds=bounds,
                   options={'gtol':1e-6, 'ftol':1e-8,
                            'maxiter':3000,
                            'verbose':0})

    # 5. Extract best found
    centres_opt, radii_opt = unpack(res.x)

    # 6. Optional LP tightening
    radii_opt, _ = solve_lp(centres_opt)

    sum_r = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_r

def init_hex_grid():
    n = 52
    scale = np.sqrt(7) / 2
    centres = np.array([])
    for i in range(5):
        for j in range(3):
            if i % 2 == j % 2:
                cx = (i + j / 2) * scale / (1 + scale)
                cy = (i + j) * np.sqrt(3) / (2 * (1 + scale))
                centres = np.append(centres, [cx, cy])
    centres = centres.reshape(52, 2)
    mask = np.random.choice(52, 26, replace=False)
    centres = centres[mask]
    radii = np.full(26, 0.01)
    return centres, radii

def build_variable_vector(centres, radii):
    return np.concatenate((centres.flatten(), radii))

def unpack(z):
    n = 26
    x = z[:52]
    r = z[52:]
    return x.reshape(n, 2), x.reshape(n, 2), r

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    n = len(x)
    boundary_penalties = lambda_B * np.sum(np.maximum(0, r - x) + np.maximum(0, r - (1 - x)) + np.maximum(0, r - y) + np.maximum(0, r - (1 - y)))
    overlap_penalties = lambda_O * np.sum(np.maximum(0, r[i] + r[j] - np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)) for i in range(n) for j in range(i + 1, n))
    return -np.sum(r) + boundary_penalties + overlap_penalties

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    n = len(x)
    grad = np.zeros_like(r)
    for i in range(n):
        b2x = lambda_B * np.maximum(0, r[i] - x[i])
        b2y = lambda_B * np.maximum(0, r[i] - y[i])
        if b2x > 0:
            grad[i] -= b2x
        if b2y > 0:
            grad[i] -= b2y
        for j in range(i + 1, n):
            d_ij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            o2 = lambda_O * np.maximum(0, r[i] + r[j] - d_ij)
            if o2 > 0:
                grad[i] -= o2 * (x[i] - x[j]) / d_ij
                grad[j] -= o2 * (x[j] - x[i]) / d_ij
    return np.concatenate((grad, grad))

def solve_lp(centres):
    n = len(centres)
    m = np.zeros((n, n), dtype=int)
    np.fill_diagonal(m, 2)
