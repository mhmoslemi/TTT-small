import numpy as np
import cvxpy as cp
import math
import random

def compute_optimal_radii(centers):
    n = len(centers)
    r = cp.Variable(n, nonneg=True)
    constraints = []
    x = centers[:,0]
    y = centers[:,1]
    for i in range(n):
        constraints.append(r[i] <= x[i])
        constraints.append(r[i] <= 1 - x[i])
        constraints.append(r[i] <= y[i])
        constraints.append(r[i] <= 1 - y[i])
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i]-centers[j])
            constraints.append(r[i] + r[j] <= d)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if r.value is None:
        raise RuntimeError("LP unsolved")
    radii = np.maximum(r.value, 0.0)
    return radii

def run_packing():
    np.random.seed(42)
    random.seed(42)

    n_circles = 26

    centres = np.random.uniform(0.05, 0.95, (n_circles, 2))

    current_centres = centres
    current_radii = optimise_radii(current_centres)
    current_sum = current_radii.sum()

    best_centres = current_centres.copy()
    best_radii   = current_radii.copy()
    best_sum     = current_sum

    n_iter = 20000
    step_start = 0.15      
    step_end   = 0.001     
    temp_start = 1.0       
    temp_end   = 1e-4      

    for k in range(n_iter):
        frac = k / n_iter
        step = step_start * ((step_end / step_start) ** frac)
        temp = temp_start * ((temp_end / temp_start) ** frac)

        i = np.random.randint(n_circles)
        new_centres = current_centres.copy()
        new_centres[i, 0] += (np.random.rand() * 2 - 1) * step
        new_centres[i, 1] += (np.random.rand() * 2 - 1) * step
        new_centres[i] = np.clip(new_centres[i], 0.0, 1.0)

        new_radii = optimise_radii(new_centres)
        new_sum = new_radii.sum()

        if new_sum > current_sum:
            accept = True
        else:
            accept_prob = math.exp((new_sum - current_sum) / max(temp, 1e-12))
            accept = random.random() < accept_prob

        if accept:
            current_centres = new_centres
            current_radii   = new_radii
            current_sum     = new_sum

            if current_sum > best_sum + 1e-12:
                best_centres = current_centres.copy()
                best_radii   = current_radii.copy()
                best_sum     = current_sum

    final_radii = optimise_radii(best_centres)
    final_sum   = final_radii.sum()

    return best_centres, final_radii, final_sum
