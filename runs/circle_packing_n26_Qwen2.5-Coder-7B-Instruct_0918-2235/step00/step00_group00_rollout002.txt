import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def inflation_rule(centers, radii):
    n = len(centers)
    new_radii = np.zeros(n)
    for i in range(n):
        r_i_max = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if j != i:
                dist = np.linalg.norm(centers[i] - centers[j])
                r_i_max = min(r_i_max, dist - radii[j])
        new_radii[i] = max(0, min(r_i_max, radii[i]))
    return new_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    centers = centers / np.max(centers, axis=0)  # Ensure coverage within the unit square
    centers[np.random.rand(n, 2) < 0.1] *= 0.9  # Avoid immediate degeneration

    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = 0

    s_start, s_end = 0.15, 0.001
    T_start, T_end = 1.0, 1e-4
    N = 20000

    for k in range(N):
        i = np.random.randint(n)
        delta = (np.random.rand(2) * 2 - 1) * [s_start * (s_end / s_start) ** (k / N), s_start * (s_end / s_start) ** (k / N)]
        delta = np.clip(delta, -1, 1)
        centers[i] += delta
        centers[i] = np.clip(centers[i], 0, 1)

        new_radii = inflation_rule(centers, best_radii)
        new_sum = np.sum(new_radii)

        if new_sum > best_sum:
            best_radii = new_radii.copy()
            best_sum = new_sum
            if np.all(best_centers == centers):
                best_centers = centers.copy()

        if new_sum > best_sum or np.exp((new_sum - best_sum) / T_start) > np.random.rand():
            best_centers = centers.copy()
            best_sum = new_sum

        best_centers = centers.copy() if new_sum > best_sum else best_centers
        best_sum = np.sum(best_radii) if new_sum > best_sum else best_sum

        s_start *= s_end / s_start
        T_start *= T_end / T_start

    return best_centers, best_radii, best_sum
