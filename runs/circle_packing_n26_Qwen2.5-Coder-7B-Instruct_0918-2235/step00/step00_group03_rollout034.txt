import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def compute_distances(centers):
        n = centers.shape[0]
        d = np.zeros((n, n))
        for i in range(n):
            for j in range(i+1, n):
                d[i, j] = np.sum((centers[i] - centers[j]) ** 2) ** 0.5
        return d

    def compute_max_radius(centers, i):
        return np.min([centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]])

    def coordinate_ascent(centers, step_size=0.1, max_cycles=30, alpha=0.8, eps_stop=1e-4):
        n = centers.shape[0]
        rho = compute_max_radius(centers)
        bounds = [(0, r) for r in rho]
        A = np.zeros((n*(n-1)//2, n))
        b = np.zeros(n*(n-1)//2)
        for i in range(n-1):
            A[i*(n-1)//2+i, i] = 1
            for j in range(i+1, n):
                A[i*(n-1)//2+i, j] = 1
                b[i*(n-1)//2+i] = compute_distances(centers)[i, j]

        current_sum = 0
        for trial in range(40):
            centers = np.random.rand(n, 2)
            current_sum = 0
            for _ in range(max_cycles):
                for i in np.random.permutation(n):
                    old_pos = centers[i].copy()
                    delta = np.random.uniform(-step_size, step_size, size=2)
                    centers[i] = np.clip(old_pos + delta, 0, 1)
                    d = compute_distances(centers)
                    rho = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
                    bounds = np.array([(0, r) for r in rho])
                    r = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs').x
                    current_sum = np.sum(r)
                    if not all(r <= rho):
                        centers[i] = old_pos
            if current_sum > 2.636:
                return centers, r, current_sum
        return centers, r, 2.636  # Fallback to target sum if all trials failed
