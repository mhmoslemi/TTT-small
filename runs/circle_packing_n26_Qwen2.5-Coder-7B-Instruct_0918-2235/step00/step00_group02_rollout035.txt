import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist

N = 26
bounds = [(0, 1), (0, 1)] * N

def build_lp(center_coords):
    C = np.array(center_coords).reshape(N, 2)
    dists = cdist(C, C, 'euclidean')
    lhs_sparse = np.zeros((N * N, N * 2))
    rhs_sparse = np.zeros(N * N)
    for i in range(N):
        for j in range(i + 1, N):
            lhs_sparse[i * 2 * (j - i) + 0, i] = 1
            lhs_sparse[i * 2 * (j - i) + 0, j] = 1
            lhs_sparse[i * 2 * (j - i) + 1, i] = -1
            lhs_sparse[i * 2 * (j - i) + 1, j] = -1
            rhs_sparse[i * 2 * (j - i)] = dists[i, j]
        lhs_sparse[i * 2 * (N - i) + N - 1, i] = 1
        rhs_sparse[i * 2 * (N - i)] = C[i, 0]
        lhs_sparse[i * 2 * (N - i) + N, i] = 1
        rhs_sparse[i * 2 * (N - i) + 1] = C[i, 1]
        lhs_sparse[i * 2 * (N - i) + N + 1, i] = -1
        rhs_sparse[i * 2 * (N - i) + 2] = 1 - C[i, 0]
        lhs_sparse[i * 2 * (N - i) + N + 2, i] = -1
        rhs_sparse[i * 2 * (N - i) + 3] = 1 - C[i, 1]
    return lhs_sparse, rhs_sparse

def compute_sum_radius(center_coords):
    lhs_sparse, rhs_sparse = build_lp(center_coords)
    try:
        result = linprog(np.ones(N), A_ub=lhs_sparse, b_ub=rhs_sparse, bounds=bounds, method='highs')
        if result.success:
            return -result.fun
        else:
            return float('inf')
    except Exception:
        return float('inf')

    return compute_sum_radius(center_coords)
