import numpy as np
from scipy.spatial.distance import cdist

def dist2(p1, p2=None):
    if p2 is None:
        return np.sum(p1 ** 2, axis=1)
    else:
        return np.sum((p1 - p2) ** 2, axis=1)

def run_packing():
    n = 26
    np.random.seed(42)
    pts = np.random.rand(n, 2)
    rad = np.zeros(n)
    MIN_R = 1e-8

    # Phase-1: Random greedy insertion
    order = np.random.permutation(n)
    for t in order:
        S = 5000
        while True:
            candidates = np.random.rand(S, 2)
            valid = np.ones(S, dtype=bool)
            for j in range(n):
                valid &= dist2(candidates, pts[j]) > rad[j] ** 2 + MIN_R
                valid &= candidates[:, 0] > MIN_R
                valid &= candidates[:, 0] < 1 - MIN_R
                valid &= candidates[:, 1] > MIN_R
                valid &= candidates[:, 1] < 1 - MIN_R
            if np.any(valid):
                min_clearance = np.inf * np.ones(S)
                for j in range(n):
                    clearance = np.clip(np.sqrt(dist2(candidates, pts[j]) - rad[j] ** 2), a_min=MIN_R, a_max=None)
                    clearance[min_clearance > clearance] = clearance[min_clearance > clearance]
                    min_clearance = np.minimum(min_clearance, clearance)
                idx = np.argmax(min_clearance)
                rad[t] = min_clearance[idx]
                pts[t] = candidates[idx]
                break
            else:
                S = max(S // 2, 10)
        if rad[t] < MIN_R:
            rad[t] = MIN_R

    # Phase-2: Force-field relaxation
    T_init = 1.0
    T_final = 1e-4
    cooling_rate = 0.999
    ε_w = 1.0
    ε_p = 1.0

    def get_potential(points, radii):
        u_wall = 0.5 * ε_w / (radii + 1e-8) ** 1
        u_pair = 0.5 * ε_p / (radii[:, np.newaxis] + radii - dist2(points, points[:, np.newaxis]) + 1e-8) ** 2
        return np.sum(u_wall + np.triu(u_pair, k=1))

    while T > T_final:
        # Select a random circle
        i = np.random.randint(n)
        # Propose a move
        Δ = 0.01 * T * np.random.normal(size=2)
        x_new, y_new = pts[i] + Δ
        x_new = np.clip(x_new, rad[i], 1 - rad[i])
        y_new = np.clip(y_new, rad[i], 1 - rad[i])
        points_new = pts.copy()
        points_new[i] = [x_new, y_new]
        # Recompute radius for i
        clearance_wall = np.clip(np.min(np.sqrt(dist2(points_new[i], pts)) - rad + 1e-8), a_min=1e-8, a_max=None)
        clearance_pair = np.clip(np.min(np.sqrt(dist2(points_new[i], points_new)) - rad - 1e-8), a_min=1e-8, a_max=None)
        rad_new = np.min([clearance_wall, clearance_pair])
        # Acceptance criterion
        dU = get_potential(points_new, rad_new) - get_potential(pts, rad)
        if dU <= 0 or np.random.rand() < np.exp(-dU / T):
            pts[i] = [x_new, y_new]
            rad[i] = rad_new
        T *= cooling_rate

    # Post-processing: Radius maximization
    for i in range(n):
        clearance_wall = np.clip(np.min(np.sqrt(dist2(pts[i], pts)) - rad + 1e-8), a_min=1e-8, a_max=None)
        clearance_pair = np.clip(np.min(np.sqrt(dist2(pts[i], pts)) - rad - 1e-8), a_min=1e-8, a_max=None)
        rad[i] = np.min([clearance_wall, clearance_pair])

    return pts, rad, np.sum(rad)
