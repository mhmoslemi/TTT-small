import numpy as np
from scipy.spatial.distance import cdist

def heapify_v(vector, reverse=False):
    for i in range(len(vector) // 2, -1, -1):
        sift_down(vector, i, reverse)

def sift_down(vector, index, reverse=False):
    size = len(vector)
    while True:
        left = 2 * index + 1
        right = 2 * index + 2
        target = index

        if left < size and ((vector[left] > vector[target]) if reverse else (vector[left] < vector[target])):
            target = left

        if right < size and ((vector[right] > vector[target]) if reverse else (vector[right] < vector[target])):
            target = right

        if target != index:
            vector[index], vector[target] = vector[target], vector[index]
            index = target
        else:
            break

def heapify_e(vector, map_dict, reverse=False):
    for i in range(len(map_dict) // 2, -1, -1):
        sift_down_e(vector, map_dict, i, reverse)

def sift_down_e(vector, map_dict, index, reverse=False):
    size = len(vector)
    while True:
        left = 2 * index + 1
        right = 2 * index + 2
        target = index

        if left < size and ((vector[left] > vector[target]) if reverse else (vector[left] < vector[target])):
            target = left

        if right < size and ((vector[right] > vector[target]) if reverse else (vector[right] < vector[target])):
            target = right

        if target != index:
            vector[index], vector[target] = vector[target], vector[index]
            map_dict[target], map_dict[index] = map_dict[index], map_dict[target]
            index = target
        else:
            break

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    MAX_ITER = 200
    EPS = 1e-12
    RESTARTS = 30
    best_sum = 0.0
    best_centers = np.zeros((26, 2))
    best_radii = np.zeros((26,))
    
    for _ in range(RESTARTS):
        centers = np.random.rand(26, 2)
        radii = np.full(26, 1e-3)
        new_radii = refine_radii(centers, radii)
        if new_radii is None:
            continue
        sum_radii = new_radii.sum()
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = new_radii
            
        # Simulated Annealing
        step = 0.07
        min_step = 0.01
        for _ in range(500):
            T = 0.25 * (1 - step / 500)
            step -= (step - min_step) / 500
            for _ in range(30):
                i = np.random.randint(26)
                old_pos = centers[i].copy()
                new_pos = old_pos + np.random.uniform(-step, step, 2)
                new_pos = np.clip(new_pos, 0, 1)
                centers[i] = new_pos.copy()
                new_radii = refine_radii(centers, radii)
                if new_radii is None:
                    continue
                new_sum = new_radii.sum()
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = new_radii.copy()
                else:
                    if new_sum > sum_radii:
                        sum_radii = new_sum
                centers[i] = old_pos
    
    return best_centers, best_radii, best_sum

def refine_radii(centers, radii, eps=1e-12, max_iter=200):
    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        for i in range(26):
            x, y = centers[i]
            r = min(x - eps, 1 - x, y - eps, 1 - y)
            dists = cdist(centers[:i], centers[i:i+1], 'euclidean').flatten()
            dists = dists + radii[:i]
            new_r = np.min([r] + list(dists))
            new_r = max(0, new_r)
            new_radii[i] = new_r
        if np.allclose(radii, new_radii, atol=eps):
            break
        radii = new_radii
    else:
        return None
    return new_radii
