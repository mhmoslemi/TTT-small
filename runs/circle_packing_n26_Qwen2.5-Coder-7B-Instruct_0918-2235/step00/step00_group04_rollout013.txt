```python
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

# Constants
N = 26
epsilon = 1e-12

# Initial centres (hexagonal lattice)
hex_offset = np.array([[np.sqrt(3) / 2, 0.5], [np.sqrt(3) / 2, -0.5]])
hex_points = np.dstack([np.tile(np.arange(N), (2, 1)), np.repeat(np.arange(N), 2.)])
hex_points -= np.mean(hex_points, axis=0)
min_dist = np.min(pdist(hex_points, 'euclidean'))
offset = min_dist * 0.95
hex_points[1::2] += hex_offset  # Duplicate and shift offset
hex_points = hex_points[:N, :]

# Projection function to keep points within [0,1]
def project(centers):
    return np.clip(centers, 0, 1)

# Function to evaluate objective and gradient
def objective_and_gradient(centers):
    centers = project(centers)
    d = squareform(pdist(centers))
    r = np.zeros(N)
    c = linprog(-r, A_ub=np.vstack([np.eye(N), -(centers), 1 - centers, centers, d / 2]), b_ub=np.hstack([np.zeros(N), centers, 1 - centers, np.zeros(N), d + epsilon]).T, method='highs')['x']
    from scipy.optimize import linprog
    dual = linprog(None, A_eq=np.eye(N), b_eq=r, method='highs')['dual']
    if dual is None:
        dual = np.zeros_like(r)
    grad_x = np.zeros(N)
    grad_y = np.zeros(N)
    for i in range(N):
        for j in range(i + 1, N):
            if r[i] + r[j] > d[i, j] - epsilon:
                u = (centers[i, 0] - centers[j, 0]) / d[i, j]
                v = (centers[i, 1] - centers[j, 1]) / d[i, j]
                grad_x[i] += dual[j] * u
                grad_x[j] -= dual[j] * u
                grad_y[i] += dual[j] * v
                grad_y[j] -= dual[j] * v
    return -np.sum(r), -(np.hstack([grad_x, grad_y]))

# Run optimization
result = minimize(objective_and_gradient, hex_points.flatten(), method='L-BFGS-B', bounds=[(0, 1)] * 52, callback=lambda x: print(f"Sum: {-x.fun:.3f}"))

# Scale and return result
radii = np.clip(np.abs(result.x[N:]), 0, None)
center_indices = np.argsort(-radii)
best_radii =radii[center_indices][:26]
best_centers = project(result.x[:N][center_indices][:26].reshape((26,2)))
sum_radii = np.sum(best_radii)
print(f"Sum of radii: {sum_radii:.3f}")

return best_centers, best_radii, sum_radii
```
