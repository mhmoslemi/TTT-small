import numpy as np
from scipy.optimize import linprog

def clip(v, lo, hi):
    return np.clip(v, lo, hi)

def solve_LP(centers):
    n = len(centers)
    c = -np.ones(n)  # Maximize sum of radii
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append([1, 0, -1 if i % 2 == 0 else 1])
        b_ub.append(centers[i, 0 if i % 2 == 0 else 1])
        
        A_ub.append([-1, 0, 1 if i % 2 == 0 else -1])
        b_ub.append(1 - centers[i, 0 if i % 2 == 0 else 1])
        
        for j in range(i + 1, n):
            d_ij = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            A_ub.append([0, 0, 1 if i % 2 == 0 and j % 2 == 0 else -2 if i % 2 == 0 and j % 2 != 0 else -2 if i % 2 != 0 and j % 2 == 0 else 2 if i % 2 != 0 and j % 2 != 0 else d_ij**2])
            b_ub.append(d_ij**2)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return result.x, -result.fun


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    centers = np.array([[0.1 * (i % 6) + 0.05, 0.1 * (i // 6) + 0.05] for i in range(30) if (i % 6, i // 6) != (0, 4)])
    np.random.shuffle(centers)
    centers = centers[:26]
    
    current_centers = centers
    current_radii, current_sum = solve_LP(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
    
    step = 0.1
    for iter in range(20000):
        i = np.random.randint(26)
        
        new_center = clip(current_centers[i] + np.random.uniform(-step, step, size=2), 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_center
        
        new_radii, new_sum = solve_LP(new_centers)
        
        if new_sum > current_sum + 1e-12:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
        
        if iter % 500 == 0:
            step *= 0.9
        
        if iter % 1000 == 0:
            random_centers = np.array([[0.1 * (i % 6) + 0.05, 0.1 * (i // 6) + 0.05] for i in range(30) if (i % 6, i // 6) != (0, 4)])
            np.random.shuffle(random_centers)
            random_centers = random_centers[:26]
            random_radii, random_sum = solve_LP(random_centers)
            if random_sum > best_sum:
                best_centers, best_radii, best_sum = random_centers, random_radii, random_sum
    
    return best_centers, best_radii, best_sum
