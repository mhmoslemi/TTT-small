import numpy as np
from scipy.optimize import minimize

def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])
        cons.append(1 - xs[i] - rs[i])
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i in range(n):
        for j in range(i+1,n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist2 = dx*dx + dy*dy
            cons.append(dist2 - (rs[i] + rs[j])**2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r0 = 0.08
    radious = np.ones(n) * r0

    # Generate random initial coordinates with non-overlapping circles
    xs = np.random.uniform(r0, 1 - r0, n)
    ys = np.random.uniform(r0, 1 - r0, n)
    
    # Construct initial guess as a flat array
    x = np.hstack((xs, ys, radious))
    
    bounds = [(0, None), (0, None)] * n + [(None, 0.5)]

    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]

    # Optimization options
    options = {'ftol': 1e-9, 'maxiter': 5000, 'disp': False}
    
    res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint_all}], options=options)

    if not res.success:
        raise Exception("Optimization failed")

    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    
    centers = np.column_stack((xs, ys))
    sum_r = rs.sum()
    
    return centers, rs, sum_r
