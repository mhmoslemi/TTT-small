import numpy as np
from scipy.optimize import linprog, differential_evolution, bounds as.Bounds

def hex_grid(radius, packing_density):
    cell_size = 2 * radius / np.sqrt(3)
    side_length = int(np.ceil(np.sqrt(24 * packing_density / np.sqrt(3) / np.pi + 16) + 2))
    side_length = max(1, min(side_length, 13))  # Maximum 13 circles along one edge to fit in unit square
    x = np.arange(side_length) * cell_size
    y = np.arange(side_length) * cell_size * np.sqrt(3) / 2
    X, Y = np.meshgrid(x, y)
    centers = np.array([X.ravel(), Y.ravel()]).T
    cells = centers.shape[0]
    in_square = centers[:, 0] >= 0
    in_square &= centers[:, 0] <= 1
    in_square &= centers[:, 1] >= 0
    in_square &= centers[:, 1] <= 1
    centers = centers[in_square[:, 0] & in_square[:, 1]]
    center_indices = np.arange(len(centers))
    np.random.shuffle(center_indices)
    centers = centers[center_indices]
    centers = centers[max(0, len(centers) - 26):]
    return centers, radius

def add_circles_around_quadrant(centers, radius, quadrant, max_iter=10):
    if len(centers) == 26:
        return centers, []

    midpoint = np.array([0.5, 0.5])
    radius_factor = 0.85

    missing_indices = np.array([i for i in range(26) if i not in np.arange(len(centers))])
    if len(missing_indices) == 0:
        return centers, []

    mid = np.sum(centers[missing_indices], axis=0) / len(missing_indices)

    for _ in range(max_iter):
        if len(missing_indices) == 0:
            break

        angle = np.linspace(0, 2 * np.pi, len(missing_indices) + 1)[:-1]
        n = len(angle)
        candidate_centers = mid + radius_factor * radius * np.array([np.cos(angle), np.sin(angle)]).T

        valid_indices = []
        for j, c in enumerate(candidate_centers):
            is_inside = c[0] >= 0 and c[0] <= 1 and c[1] >= 0 and c[1] <= 1
            if is_inside and not any(np.allclose(c, centers[k], atol=1e-12) for k in range(len(centers))):
                valid_indices.append(j)

        if len(valid_indices) > 0:
            new_center = candidate_centers[valid_indices[0]]
            centers = np.vstack((centers, new_center))
            missing_indices = np.delete(missing_indices, valid_indices[0])

    return centers, []

def solve_radii_fixed_centres(centers):
    n = len(centers)
    A = []
    b = []

    for r1, (x1, y1) in enumerate(centers):
        A.append([x1, y1, 1])
        b.append(1 - min(x1, 1 - x1, y1, 1 - y1))

        for r2 in range(r1 + 1, n):
            d = np.linalg.norm(centers[r1] - centers[r2])
            A.append([ centers[r2][0] - centers[r1][0], centers[r2][1] - centers[r1][1], 1 ])
            b.append(d - centers[r2][1] + centers[r1][1] - min(centers[r2][0], 1 - centers[r2][0], centers[r1][0], 1 - centers[r1][0]))

    lb = [0] * n
    result = linprog([1] * n, A_ub=A, b_ub=b, bounds=Bounds(0, None), method='highs')
    if not result.success:
        raise ValueError('LP infeasible')

    return np.maximum(result.x, 0), sum(result.x)

def penalty_centres_fixed_radii(centers, radii):
    n = len(centers)
    pen_vec = np.zeros(n)

    for i in range(n):
        pen_vec[i] = np.sum(np.maximum(0, radii + radii[i] - np.linalg.norm(centers - centers[i], axis=1)))
        pen_vec[i] += np.sum(np.maximum(0, radii[i] - min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])))

    return pen_vec

def coordinated_ascent(init_centers, max_iter=15):
    n = 26
    centers = init_centers
    best_centers = centers.copy()
    best_sum_radii = np.sum(solve_radii_fixed_centres(centers)[0])

    for _ in range(max_iter):
        radii, sum_radii = solve_radii_fixed_centres(centers)
        pen_vec = penalty_centres_fixed_radii(centers, radii)
        penalty = np.sum(pen_vec)

        annealing = 50
        step_size = 0.1

        while penalty > 0:
            random_subset = np.random.choice(n, size=int(n / 5), replace=False)
            old_centers = centers.copy()
            centers[random_subset, :] += step_size * (np.random.rand(*centers[random_subset].shape) - 0.5)
            new_radii, new_sum_radii = solve_radii_fixed_centres(centers)
            new_pen_vec = penalty_centres_fixed_radii(centers, new_radii)
            new_penalty = np.sum(new_pen_vec)

            if new_penalty < penalty * (1 - annealing / max_iter):
                penalty = new_penalty
            else:
                centers = old_centers

        if new_sum_radii > best_sum_radii:
            best_sum_radii = new_sum_radii
            best_centers = centers.copy()

    return best_centers, best_sum_radii

def local_audit_and_refine(centers, radii):
    n = 26
    for _ in range(10):
        for i in range(n):
            j = np.random.choice([k for k in range(n) if k != i])
            old_centers = centers.copy()
            old_sum_radii = np.sum(radii)
            centers[i:j+1] = centers[j,i:-1]
            radii[i:j+1] = radii[j,i:-1]
            new_sum_radii = np.sum(solve_radii_fixed_centres(centers)[0])
            if new_sum_radii > old_sum_radii:
                continue
            else:
                centers = old_centers

    return centers, np.sum(radii)

def run_packing():
    best_centers = None
    best_sum_radii = -np.inf

    np.random.seed(42)
    for seed in range(10):
        centers, radius = hex_grid(radius=0.05, packing_density=26)
        centers, _ = add_circles_around_quadrant(centers, radius, quadrant=1)
        centers = differential_evolution(penalty_centres_fixed_radii, bounds=Bounds(0, 1), args=(radii,), popsize=100).x.reshape(-1, 2)
        centers, sum_radii = coordinated_ascent(centers)
        centers, sum_radii = local_audit_and_refine(centers, solve_radii_fixed_centres(centers)[0])

        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers

    return best_centers, solve_radii_fixed_centres(best_centers)[0], best_sum_radii
