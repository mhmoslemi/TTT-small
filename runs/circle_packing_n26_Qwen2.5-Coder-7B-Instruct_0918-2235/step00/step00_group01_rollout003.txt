import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def objective_u(u):
    x = u[:26]
    y = u[26:]
    n = len(x)
    c = np.ones(n)
    A_pos = -np.eye(n)
    b_pos = np.zeros(n)
    A_bound = np.zeros((4*n, n))
    b_bound = np.zeros(4*n)
    for i in range(n):
        A_bound[4*i, i] = 1
        b_bound[4*i] = 1 - x[i]
        A_bound[4*i+1, i] = 1
        b_bound[4*i+1] = 1 - y[i]
        A_bound[4*i+2, i] = -1
        b_bound[4*i+2] = x[i]
        A_bound[4*i+3, i] = -1
        b_bound[4*i+3] = y[i]
    A = np.vstack((np.ones((n, n)), A_pos, A_bound))
    b = np.hstack((np.zeros(n), b_pos, b_bound))
    result = linprog(c, A_ub=A, b_ub=b, bounds=np.zeros(n), method='highs')
    return -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    result = differential_evolution(objective_u, bounds, popsize=20, maxiter=200, tol=1e-12)
    u_opt = result.x
    x = u_opt[:26]
    y = u_opt[26:]
    n = len(x)
    c = np.ones(n)
    A_pos = -np.eye(n)
    b_pos = np.zeros(n)
    A_bound = np.zeros((4*n, n))
    b_bound = np.zeros(4*n)
    for i in range(n):
        A_bound[4*i, i] = 1
        b_bound[4*i] = 1 - x[i]
        A_bound[4*i+1, i] = 1
        b_bound[4*i+1] = 1 - y[i]
        A_bound[4*i+2, i] = -1
        b_bound[4*i+2] = x[i]
        A_bound[4*i+3, i] = -1
        b_bound[4*i+3] = y[i]
    A = np.vstack((np.ones((n, n)), A_pos, A_bound))
    b = np.hstack((np.zeros(n), b_pos, b_bound))
    result = linprog(c, A_ub=A, b_ub=b, bounds=np.zeros(n), method='highs')
    radii_opt = result.x
    sum_radii = radii_opt.sum()
    return np.column_stack((x, y)), radii_opt, sum_radii

centers, radii, sum_radii = run_packing()
