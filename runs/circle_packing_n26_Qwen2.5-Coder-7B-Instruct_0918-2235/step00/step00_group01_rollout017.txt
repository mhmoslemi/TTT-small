import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_solution(centers, boundary_constraint=True):
    n = centers.shape[0]
    radii_bound = np.array([[0, None] for _ in range(n)])
    
    A_ineq = []
    b_ineq = []
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append(np.zeros(n))
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(np.linalg.norm(centers[i] - centers[j]))
            
    if boundary_constraint:
        A_ineq.append(np.ones(n))
        b_ineq.append(1)
        A_ineq.append(np.ones(n))
        b_ineq.append(1 - centers[0, 0])
        A_ineq.append(np.ones(n))
        b_ineq.append(1 - centers[0, 1])
        A_ineq.append(np.ones(n))
        b_ineq.append(1 - centers[1, 0])
        A_ineq.append(np.ones(n))
        b_ineq.append(1 - centers[2, 1])
    
    c = np.ones(n)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=radii_bound, method='highs')
    return result.x

def objective_function(centers):
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    r = lp_solution(centers)
    if np.any(r <= 0):
        return -np.inf
    sum_radii = r.sum()
    distances = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + 
                          (y[:, np.newaxis] - y[np.newaxis, :])**2)
    overlap_constraints = r[:, np.newaxis] + r[np.newaxis, :] - distances
    if np.any(overlap_constraints > 1e-12):
        return -np.inf
    if np.any(x < 0) or np.any(x > 1) or np.any(y < 0) or np.any(y > 1):
        return -np.inf
    return -sum_radii

def run_packing():
    np.random.seed(0)
    bounds = [(0, 1), (0, 1)] * 26
    result = differential_evolution(objective_function, bounds=bounds, popsize=20, maxiter=200, tol=1e-12)
    centers = np.array(result.x).reshape(26, 2)
    radii = lp_solution(centers)
    sum_radii = radii.sum()
    final_result = minimize(objective_function, centers, method='Nelder-Mead', bounds=bounds, options={'maxfev': 1000})
    centers = final_result.x.reshape(26, 2)
    radii = lp_solution(centers)
    sum_radii = radii.sum()
    return centers, radii, sum_radii
