import numpy as np
from scipy.optimize import linprog

def run_packing():
    n = 26
    R = 40
    s0 = 0.1
    alpha = 0.8
    epsilon = 1e-9
    
    best_sum = -np.inf
    for trial in range(R):
        centres = np.random.rand(n, 2)
        r = np.random.rand(n) * 0.1
        dists = np.zeros((n, n))
        
        best_sum_trial = -np.inf
        best_positions_trial = None
        best_radii_trial = None
        
        s = s0
        step_count = 0
        
        while s >= 1e-4 and step_count < 5:
            for i in np.random.permutation(n):
                # Clip movement to boundary
                delta = np.random.uniform(-s, s, size=2)
                centre_new = np.clip(centres[i] + delta, 0, 1)
                
                # Recompute distances and constraints
                dists.fill(0)
                np.fill_diagonal(dists, np.inf)
                dists[:i, i+1:] = dists[i+1:, i] = np.sqrt(
                    np.sum((centres[:i, np.newaxis, :] - centre_new[np.newaxis, :]) ** 2, axis=2)
                )
                dists_i = dists[i, np.where(dists[i] < np.inf)]
                rho_i = np.minimum(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
                bounds = np.column_stack((np.zeros(n), np.minimum(rhos_i, rho_i)))
                
                # Solve linear program
                c = -np.ones(n)
                A_ub = (np.where(dists < np.inf, 1, 0).sum(axis=0) > 1).astype(int)
                row_idx, col_idx = np.triu_indices(n, k=1)
                A_ub[row_idx, col_idx] = 1
                A_ub[col_idx, row_idx] = 1
                A_lb = np.zeros((n, n))
                b_lb = np.zeros(n)
                b_ub = dists_i
                constraints = [{'type': 'ineq', 'fun': lambda r, A_ub=A_ub, b_ub=b_ub: A_ub @ r - b_ub},
                               {'type': 'ineq', 'fun': lambda r: r}]
                result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_lb=A_lb, b_lb=b_lb, bounds=bounds, options={'disp': False})
                
                if result.success:
                    if result.fun > best_sum_trial:
                        best_sum_trial = result.fun
                        best_positions_trial = np.round(centre_new, decimals=12)
                        best_radii_trial = result.x
                    centres[i] = centre_new
            s *= alpha
            step_count += 1
        
        if best_sum_trial > best_sum:
            best_sum = best_sum_trial
            best_positions = best_positions_trial
            best_radii = best_radii_trial
    
    return best_positions, best_radii, best_sum
