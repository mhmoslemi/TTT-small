import numpy as np
import scipy.optimize as optimize
import cma

def refine_radii(centres):
    n = centres.shape[0]
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        A[i, i] = 1
        b[i] = min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
        for j in range(i + 1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            A[i, j] = -1
            A[j, i] = -1
            b[j] = dist
    r, exitflag = optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not exitflag in [0, 2, 4]:
        r = refine_radii(centres)  # Retry if non-optimal
    return np.maximum(r, 0)

def objective(z):
    n = len(z) // 2
    centres = z[:n].reshape((n, 2))
    r = refine_radii(centres)
    return np.sum(r)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def initial_mean():
    n = 26
    a = b = c = np.sqrt(2) / 3  # Side length of hexagon
    jitter = 0.01  # Jitter for centers
    centres = np.zeros((n, 2))
    idx = 0
    centres[idx, :] = a / (2 * b) + jitter * np.random.rand(2)
    idx += 1
    centres[idx:idx + 8, 0] = a / (2 * b) + jitter * np.random.RandomState().rand(8)
    centres[idx:idx + 8, 1] = - centres[idx] + jitter * np.random.RandomState().rand(8)
    idx += 8
    centres[idx:idx + 8, 1] = + centres[idx] + jitter * np.random.RandomState().rand(8)
    idx += 8
    centres[idx:idx + 6, 0] = a - (a - centres[idx-1, 0] + jitter * np.random.rand(6))
    centres[idx:idx + 6, 1] = b
    centres[idx:idx + 6, 1] = b + jitter * np.random.rand(6)
    centres[idx:idx + 6, 1] = - b + jitter * np.random.rand(6)
    centres[idx:idx + 6, 1] = a - (a - centres[idx-1, 1] + jitter * np.random.rand(6))
    centres[idx:idx + 6, 0] = b
    centres[idx:idx + 6, 0] = - b + jitter * np.random.rand(6)
    centres[idx:idx + 6, 0] = a - (a - centres[idx-1, 0] + jitter * np.random.rand(6))
    return centres

def simulate_annealing(best_z):
    n = len(best_z) // 2
    best_f = objective(best_z)
    for _ in range(200):
        idxs = np.random.choice(n, 3)
        d = 0.01
        delta_z = np.zeros(n * 2)
        for i in idxs:
            delta_z[i * 2] += np.random.uniform(-d, d)
            delta_z[i * 2 + 1] += np.random.uniform(-d, d)
        new_z = best_z + delta_z
        new_z = np.clip(new_z, np.zeros(n * 2), [1]*n + [0.5]*n)
        new_f = objective(new_z)
        if new_f > best_f:
            best_z = new_z
            best_f = new_f
    return best_z

def run_packing():
    n = 26
    mean = initial_mean()
    mean = np.concatenate((mean, np.zeros(n)))  # Append initial radii guess
    bounds = np.concatenate((np.zeros(n * 2), np.zeros(n) + 0.5))  # [0,1]^2 for centres, [0,0.5] for radii
    bounds = np.column_stack((bounds, bounds))
    es = cma.CMAEvolutionStrategy(mean, n/2)
    es.optimize(objective, CMA_options={'bounds': (bounds[:, 0], bounds[:, 1])}, CMA callbacks={'after-generation': None})
    best_z = es.result.x
    best_f = es.result.fvalue
    best_gd = es.result.gd
    while best_gd > 1e-9:
        best_z = simulate_annealing(best_z)
        best_f = objective(best_z)
        best_gd = 0.5 * np.sum((best_z[:-n]-best_z[:-n])/np.ones(n*2))**2
    circles = validate_packing(best_z[:n], best_z[n:])
    if not circles[0]:
        for i in range(100):
            r_lp = refine_radii(best_z[:n])
            best_z[n:] = r_lp
            circles = validate_packing(best_z[:n], best_z[n:])
            if circles[0]:
                break
    return best_z[:n], best_z[n:], np.sum(best_z[n:] + 1e-15)
