import numpy as np
from scipy.optimize import linprog

def solve_inner_lp(positions):
    n = positions.shape[0]
    A_eq, b_eq, A_ub, b_ub = [], [], [], []
    bounds = []

    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.linalg.norm(positions[i] - positions[j])
            A_eq.append([1, 1])
            b_eq.append(d_ij)
            bounds.extend([(0, np.min(positions[i]).min()), (0, np.min(positions[i]).min())])

    objective = -np.ones(n)
    result = linprog(objective, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_cycles = 50
    step_size = 0.1
    alpha = 0.8
    eps_stop = 1e-4
    num_trials = 40
    best_sum = float('-inf')
    rng = np.random.RandomState(42)

    for _ in range(num_trials):
        positions = rng.uniform(0, 1, (n, 2))
        current_sum = -1

        for _ in range(max_cycles):
            for i in range(n):
                old_pos = positions[i]
                delta = rng.uniform(-step_size, step_size, size=2)
                positions[i] = np.clip(old_pos + delta, 0, 1)
                radii, new_sum = solve_inner_lp(positions)

                if new_sum <= current_sum + 1e-9:
                    positions[i] = old_pos
                else:
                    current_sum = new_sum

            step_size *= alpha
            if step_size < eps_stop:
                break

        if current_sum > best_sum:
            best_positions = positions.copy()
            best_sum = current_sum

    return best_positions, solve_inner_lp(best_positions)[0], best_sum
