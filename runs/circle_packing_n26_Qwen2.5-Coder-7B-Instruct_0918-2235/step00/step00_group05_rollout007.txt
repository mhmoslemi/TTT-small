import numpy as np
from scipy.optimize import linprog, differential_evolution

def hex_grid(radius, packing_density):
    rows = np.round(np.sqrt(3 * packing_density / np.pi))
    cols = np.round((1.5 * rows) / np.tan(np.pi / 3))
    s = 2 * radius / np.sqrt(3)
    grid = np.array([
        (x, y)
        for y in np.arange(s, 1 + 1e-12, s * np.sqrt(3))
        for x in np.arange(s / 2, 1 + 1e-12, s)
    ])
    if len(grid) < packing_density:
        return None
    return grid

def add_circles_around_quadrant(center, grid):
    x, y = center
    quadrant_grid = np.array([
        (x + dx * np.tan(np.pi / 3), y + dy * np.tan(np.pi / 3))
        for dx in np.linspace(-1, 1, 4)
        for dy in np.linspace(-1, 1, 4)
        if abs(dx) + abs(dy) < 1.5 and dx**2 + dy**2 < 1
    ])
    return np.add(quadrant_grid, center)

def solve_radii_fixed_centres(centres):
    n = centres.shape[0]
    A = np.array([
        np.hstack([centres[:, [0]], np.zeros(n), -centres[:, [1]], np.zeros(n)])
        for x, y in centres
    ])
    b = np.max(centres, axis=1)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            A = np.vstack([A, np.hstack([centres[i]], np.zeros(n), -centres[j], np.zeros(n))])
            b = np.append(b, d - 1e-12)
    A = np.vstack([A, -np.eye(n)])
    b = np.append(b, np.zeros(n))
    x = linprog(np.ones(n), A_ub=A, b_ub=b, method='highs').x
    return x, np.sum(x)

def penalty_centres_fixed_radii(centres, radii, eps=1e-12):
    total_penalty = 0
    n = len(centres)
    dists = np.linalg.norm(centres[None, :] - centres[:, None], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            penalty = max(0, radii[i] + radii[j] - dists[i, j] + eps)
            total_penalty += penalty
        penalty = max(0, max(centres[i]) - min(radii[i], 1 - max(centres[i])) + eps)
        total_penalty += penalty
    return total_penalty

def coordinated_ascent(init_centres, max_iter=15):
    n = init_centres.shape[0]
    best_sum_radii = -np.inf
    best_centres = None
    best_radii = None
    for _ in range(max_iter):
        x, sum_radii = solve_radii_fixed_centres(init_centres)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centres = init_centres.copy()
            best_radii = x.copy()
        centre_offset = np.random.rand(n, 2) * 2 - 1
        centre_offset *= (1 - init_centres.max(axis=0) - 1e-12) * 0.5
        init_centres += centre_offset
    return best_centres, best_radii, best_sum_radii

def local_audit_and_refine(centres, radii):
    n = centres.shape[0]
    for _ in range(10):  # Simulated Annealing
        swap_indices = np.random.choice(n, size=2, replace=False)
        new_centres = centres.copy()
        new_centres[swap_indices] = centres[swap_indices[::-1]]
        x, sum_radii = solve_radii_fixed_centres(new_centres)
        if sum_radii > np.sum(radii):
            return new_centres, x, sum_radii
    return centres, radii, np.sum(radii)

def run_packing():
    np.random.seed(42)
    init_centres = np.array([
        (0.5 + np.random.rand() * 0.1, 0.5 + np.random.rand() * 0.1) for _ in range(26)
    ])
    best_centres, best_radii, sum_radii = None, None, -np.inf
    for seed in range(10):
        np.random.seed(seed)
        init_centres = np.array([
            (0.5 + np.random.rand() * 0.1, 0.5 + np.random.rand() * 0.1) for _ in range(26)
        ])
        centres, radii, sum_radii = coordinated_ascent(init_centres)
        centres, radii, sum_radii = local_audit_and_refine(centres, radii)
        if sum_radii > best_sum_radii:
            best_centres, best_radii, sum_radii = centres, radii, sum_radii
    return best_centres, best_radii, sum_radii
