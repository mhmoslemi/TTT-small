import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for the centers and radii
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place circles around the boundary
    for i in range(n):
        angle = 2 * np.pi * i / n
        x = 0.5 + 0.45 * np.cos(angle)
        y = 0.5 + 0.45 * np.sin(angle)
        centers[i] = [x, y]
        radii[i] = 0.15
    
    # Sum of initial radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
