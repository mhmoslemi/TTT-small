import numpy as np
from scipy.spatial.distance import cdist

def dist2(i, j, pts):
    return np.sum((pts[i] - pts[j]) ** 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pts = np.zeros((n, 2), dtype=np.float64)
    rad = np.zeros(n, dtype=np.float64)

    # Phase 1: Random greedy insertion
    np.random.seed(42)
    order = np.random.permutation(n)
    for t in order:
        S = 5000
        samples = np.random.rand(S, 2)
        feasible = []
        for x, y in samples:
            if all(dist2(i, t, pts) > rad[i]**2 + 1e-12 for i in range(t)) and \
               x > 1e-8 and x < 1 - 1e-8 and y > 1e-8 and y < 1 - 1e-8:
                clearance = min(x, 1-x, y, 1-y)
                feasible.append((x, y, clearance))
        if feasible:
            x, y, clearance = max(feasible, key=lambda p: p[2])
            rad[t] = clearance
            pts[t] = (x, y)
        else:
            rad[t] = 1e-6

    # Shake
    for i in range(n):
        dx, dy = np.random.randn(2) * 1e-4
        x, y = pts[i] + (dx, dy)
        x = min(max(x, rad[i]), 1 - rad[i])
        y = min(max(y, rad[i]), 1 - rad[i])
        clearance = min(x, 1-x, y, 1-y)
        r = min(clearance, min(rad[j] for j in range(n) if j != i))
        if r > rad[i]:
            rad[i] = r
            pts[i] = (x, y)

    # Phase 2: Force-field relaxation

        y = min(max(y, rad[i]), 1 - rad[i])
        clearance = min(x, 1-x, y, 1-y)
        r = min(clearance, min(rad[j] for j in range(n) if j != i))
        if r > rad[i]:

                        (epsilon_p * (1 / (rad[i] - cdist([[pts[i][0], pts[i][1]], pts, 'euclidean')[:, 1])**2))
            if np.random.rand() < np.exp(-delta_U / T):
                rad[i] = r
                pts[i] = (x, y)

        T *= cooling_rate

    # Post-processing
    for i in range(n):
        clearance = min(pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1]
                       ) - rad[i if i == 0 else i - 1]
        rad[i] = min(clearance, min(rad[j] for j in range(n) if j != i))
        collective_clearance = min(clearance, min(rad[j] for j in range(n) if j != i))
        rad[i] = collective_clearance
