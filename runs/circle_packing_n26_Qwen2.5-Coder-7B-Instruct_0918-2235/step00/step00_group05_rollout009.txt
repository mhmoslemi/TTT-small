import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, highs

def lp_radii_fixup(center, pack):
    n = pack.shape[0]

    A = []
    b = []
    for i in range(n):
        A.append([2*center[i,0], 2*center[i,1], 1])
        b.append(min(center[i,0], 1-center[i,0], center[i,1], 1-center[i,1]))
        for j in range(i):
            r_i, r_j = pack[i, 2], pack[j, 2]
            d_ij = pack[i, :2] - pack[j, :2]
            A.append([2*d_ij[0], 2*d_ij[1], 2])
            b.append(2*r_i + 2*r_j)
    A = np.array(A)

    sol = linprog(-1, A_ub=A, b_ub=b, bounds=(0, 0.5), method='highs')
    return pack.copy()

def gradient(center, var, pack):
    n = pack.shape[0]
    g = np.zeros(var.shape)

    for i in range(n):
        r_i = pack[i, 2]
        half_r_i = r_i / 2
        f_ij = lambda x, y: (x**2 + y**2)**0.5 if x**2 + y**2 > 0 else 0
        g_xi = sum(-2 * (pack[i, :2] - pack[j, :2])[0] * f_ij pack[i, :2] - pack[j, :2])[0] for j in range(n) if i != j) / 2
        g_yi = sum(-2 * (pack[i, :2] - pack[j, :2])[1] * f_ij pack[i, :2] - pack[j, :2])[1] for j in range(n) if i != j) / 2
        g_i = np.array([g_xi, g_yi, 1])
        c_i = half_r_i + pack[i, 2]
        g[i] = g_i * c_i

    return g

def shake_up(center, pack):
    k = min(3, pack.shape[0])
    indices = np.random.choice(range(pack.shape[0]), k, replace=False)
    dx, dy = np.random.uniform(-0.01, 0.01, size=(k, 2))
    center[indices] += np.hstack((dx, dy, np.zeros((k, 1))))
    pack[indices] = np.column_stack((center[indices], lp_radii_fixup(center, pack)[indices]))
    return pack

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial setup
    center = np.random.rand(n, 2)
    center[:, 0] = np.clip(center[:, 0], 0.05, 0.95)
    center[:, 1] = np.clip(center[:, 1], 0.05, 0.95)
    pack = np.column_stack((center, 0.5 * np.ones((n, 1))))

    # Optimize many seeds 
    best_centers = pack.copy()
    best_sum = 0
    for _ in range(8):
        pack = best_centers.copy()
        result = minimize(
            lambda x: -np.sum(linprog(-1, A_ub=-(x[2*n:].reshape(-1, 3)), b_ub=x[:2*n], bounds=(0, 0.5), method='highs').x) +
                      np.sum(x[2*n:]),
            [np.tile(center, (1, 3)).flatten(), lp_radii_fixup(center, pack)],
            method='L-BFGS-B'
        )
        pack = result.x.reshape(n, 3)
        pack[:, :2] = result.x[:2*n].reshape(n, 2)
        pack[:, 2] = lp_radii_fixup(result.x[:2*n].reshape(n, 2), pack)[:, 2]
        if np.sum(pack[:, 2]) > best_sum:
            best_centers = pack.copy()
            best_sum = np.sum(best_centers[:, 2])
    
    # Optional shake-up
    pack = shake_up(np.column_stack((best_centers[:, :2], np.zeros((n, 1)))), best_centers)
    
    # Validation
    pack[:, :2] -= 1e-10  # Move inwards slightly to ensure validity
    pack[:, 2] -= 1e-10
    
    return best_centers[:, :2], best_centers[:, 2], np.sum(best_centers[:, 2])
