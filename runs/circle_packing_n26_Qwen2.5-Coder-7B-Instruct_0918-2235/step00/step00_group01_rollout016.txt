import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_radii(centers):
    n = len(centers)
    c = np.ones(n)
    A_ub = np.empty((n + n * (n - 1) // 2, n))
    b_ub = np.empty(n + n * (n - 1) // 2)
    
    offsets = np.triu_indices(n, k=1)
    A_ub[:n, :] = np.eye(n)
    A_ub[n:, offsets[0]] = -1
    A_ub[n:, offsets[1]] = -1
    b_ub[:n] = centers[:, 0]
    b_ub[n:] = centers[:, 1]
    d = np.sqrt((centers[:, None, 0] - centers[None, :, 0]) ** 2 +
               (centers[:, None, 1] - centers[None, :, 1]) ** 2)
    b_ub[n + n * (n - 1) // 2:] = 1 - d.flat[n * (n - 1) // 2:]
    
    r_bounds = [(0, None)] * n
    res = linprog(c, A_ub, b_ub, bounds=r_bounds, method='highs')
    return res.x.sum(), res.x

def objective_DE(u):
    n = len(u) // 2
    centers = u.reshape((n, 2))
    return -lp_radii(centers)[0]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    result = differential_evolution(objective_DE, bounds, popsize=20, maxiter=200, tol=1e-12)
    
    centers_DE = result.x.reshape((26, 2))
    sum_radii_DE, radii_DE = lp_radii(centers_DE)
    
    result_local = minimize(lambda u: -lp_radii(u.reshape((26, 2)))[0], centers_DE.flatten(),
                          method='Nelder-Mead', bounds=bounds)
    centers_NM = result_local.x.reshape((26, 2))
    sum_radii_NM, radii_NM = lp_radii(centers_NM)
    
    if sum_radii_NM > sum_radii_DE:
        return centers_NM, radii_NM, sum_radii_NM
    
    return centers_DE, radii_DE, sum_radii_DE

centers, radii, sum_radii = run_packing()
print(sum_radii)
