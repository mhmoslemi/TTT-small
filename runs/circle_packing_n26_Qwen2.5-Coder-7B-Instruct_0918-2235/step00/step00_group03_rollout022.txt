import numpy as np
from scipy.spatial.distance import cdist
from numpy.random import default_rng

np.random.seed(42)
rng = default_rng()

def refine_radii(centres, radii):
    n = len(radii)
    max_radii = np.minimum(centres[:, 0] - radii, 1 - centres[:, 0] - radii)
    max_radii = np.minimum(max_radii, 1 - centres[:, 1] - radii)
    max_radii = np.minimum(max_radii, centres[:, 1] - radii)

    dist_matrix = cdist(centres, centres)
    np.fill_diagonal(dist_matrix, np.inf)
    min_dists = np.min(dist_matrix - radii[:, np.newaxis], axis=1) - radii - 1e-12
    max_radii = np.minimum(max_radii, min_dists)
    max_radii = np.maximum(max_radii, 0)

    return np.minimum(radii + max_radii, 0.5)

def anneal_one_start(init_centres, steps=500, init_step=0.07, min_step=0.01):
    n = len(init_centres)
    best_centres = init_centres
    best_radii = np.full(n, 1e-3)
    best_sum = np.sum(refine_radii(init_centres, best_radii))

    for _ in range(steps):
        step_size = max(init_step - init_step * _, min_step)
        for i in range(n):
            old_pos = init_centres[i]
            delta = (-step_size + 2 * step_size * rng.random()) * 2
            new_pos = old_pos + [delta, delta]
            new_pos = np.clip(new_pos, 0, 1)
            new_radii = refine_radii(np.vstack([new_pos, centre for centre in init_centres if centre is not old_pos]), best_radii)
            new_sum = np.sum(new_radii)
            if new_sum > best_sum:
                best_centres = new_pos
                best_radii = new_radii
                best_sum = new_sum
        init_centres = best_centres

    return best_centres, best_radii, best_sum

def run_packing(RESTARTS=30) -> tuple[np.ndarray, np.ndarray, float]:
    best_total = 0
    best_centers = np.zeros((26, 2))
    best_radii = np.zeros(26)

    for _ in range(RESTARTS):
        init_centres = rng.random((26, 2))
        new_centers, new_radii, new_total = anneal_one_start(init_centres)
        if new_total > best_total:
            best_total = new_total
            best_centers = new_centers
            best_radii = new_radii

    return best_centers, best_radii, best_total.astype(float)
