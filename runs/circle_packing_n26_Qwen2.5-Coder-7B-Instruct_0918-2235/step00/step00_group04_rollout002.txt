```python
import numpy as np
from scipy.optimize import linprog

# PSO parameters
n_particles = 30
inertia = 0.72984
cognitive = 1.49445
social = 1.49445
max_iter = 3000
vel_clip = 0.1
alpha = 0.999

# Data structures
centres = np.zeros((n_particles, 26, 2))
velocities = np.zeros((n_particles, 26, 2))
pbests = np.zeros((n_particles, 26, 2))
pbest_sums = np.zeros((n_particles,))
gbest = np.zeros((26, 2))
gbest_sum = -np.inf

# Linear programming wrapper
def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    d_ij = np.sqrt(((centres_i[None, :, :2] - centres_i[:, None, :2]) ** 2).sum(axis=2))
    c = -np.ones(n_circles)
    A_ub = np.zeros((n_circles * 5, n_circles))
    b_ub = np.zeros(n_circles * 5)
    A_ub[:n_circles, np.arange(n_circles)] = np.eye(n_circles)
    A_ub[n_circles:2 * n_circles, np.arange(n_circles)] = np.eye(n_circles)
    b_ub[:n_circles] = centres_i[:, 0]
    b_ub[n_circles:2 * n_circles] = 1 - centres_i[:, 0]
    A_ub[2 * n_circles:3 * n_circles, np.arange(n_circles)] = np.eye(n_circles)
    A_ub[3 * n_circles:, np.arange(n_circles)] = np.eye(n_circles)
    b_ub[2 * n_circles:] = centres_i[:, 1]
    b_ub[3 * n_circles:] = 1 - centres_i[:, 1]
    np.fill_diagonal(A_ub[4 * n_circles:], np.ones((n_circles - 1)))
    d_ij[np.triu_indices(n_circles, k=1)] = -d_ij[np.triu_indices(n_circles, k=1)]
    A_ub[4 * n_circles:] += d_ij[np.triu_indices(n_circles, k=1)]
    b_ub[4 * n_circles:] += (centres_i[:, None, 0] + centres_i[None, :, 0])[np.triu_indices(n_circles, k=1)]
    b_ub[4 * n_circles:] += (centres_i[:, None, 1] + centres_i[None, :, 1])[np.triu_indices(n_circles, k=1)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n_circles, method='highs')
    return res.x, -res.fun

# Initial seed configuration
np.random.seed(42)
grid = np.array([[i, j] for i in np.arange(0, 1, 1 / 5.0) for j in np.arange(0, 1, 1 / 5.0)])
grid = grid[np.random.choice(len(grid), 30, replace=False)]
centres = grid + 0.01 * np.random.randn(n_particles, 26, 2)
velocities = np.zeros_like(centres)

# PSO loop
for iteration in range(max_iter):
    for particle in range(n_particles):
        velocities[particle] *= alpha
        for circle in range(26):
            r1, r2 = np.random.rand(2)
            velocities[particle, circle] += inertia * velocities[particle, circle] + cognitive * r1 * (pbests[particle, circle] - centres[particle, circle]) + social * r2 * (gbest[circle] - centres[particle, circle])
            velocities[particle, circle] = np.clip(velocities[particle, circle], -vel_clip, vel_clip)
            centres[particle, circle] += velocities[particle, circle]
            centres[particle, circle] = np.clip(centres[particle, circle], 0, 1)
        radii, sum_radii = solve_radius_lp(centres[particle])
        if sum_radii > pbest_sums[particle]:
            pbests[particle] = centres[particle]
            pbest_sums[particle] = sum_radii
        if sum_radii > gbest_sum:
            gbest = pbests[particle]
            gbest_sum = sum_radii

# Final optimisation of global best
gbest_radii, gbest_sum = solve_radius_lp(gbest)

# Output
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    return gbest, gbest_radii, gbest_sum

result = run_packing()
print(result)
```
