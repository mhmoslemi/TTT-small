import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(z):
    centers = z[:52].reshape(-1, 2)
    radii = z[52:]
    return sum(radii) - penalize(centers, radii)

def refine_radii(centers, radii):
    A = np.zeros((centers.shape[0], centers.shape[0]))
    b = np.zeros(centers.shape[0])
    for i in range(centers.shape[0]):
        A[i, i] = 2
        b[i] = 2 * min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            A[i, j] = -1
            A[j, i] = -1
            dist = np.linalg.norm(centers[i] - centers[j])
            b[i] += dist - radii[i]
            b[j] += dist - radii[j]
    r = np.linalg.solve(A, b)
    return r

def penalize(centers, radii):
    wall_penalty = sum([
        max(0, r - min(x, 1 - x, y, 1 - y))**2 for x, y, r in zip(centers[:, 0], centers[:, 1], radii)
    ])
    overlap_penalty = sum([
        max(0, (r1 + r2) - np.linalg.norm(c1 - c2))**2 for i, c1, r1 in enumerate(centers)
        for j, c2, r2 in enumerate(centers)
        if i < j
    ])
    return wall_penalty + overlap_penalty
