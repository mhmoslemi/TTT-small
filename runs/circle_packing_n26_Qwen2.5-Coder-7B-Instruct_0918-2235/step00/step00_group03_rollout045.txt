import numpy as np

def refine_radii(centers, radii, eps=1e-12, max_iter=200):
    n = len(centers)
    dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    
    for i in range(max_iter):
        new_radii = np.zeros(n)
        for j in range(n):
            r Lim = np.min([centers[j, 0] - radii[j], 1 - centers[j, 0] - radii[j]])
            r Lim = np.min([centers[j, 1] - radii[j], 1 - centers[j, 1] - radii[j]])
            min_dist = np.min(dist_matrix[j, :] - radii - eps)
            new_radii[j] = np.min([r Lim, min_dist])

        diff = np.linalg.norm(new_radii - radii)
        if diff < eps:
            break

        radii = new_radii

    return radii

def anneal_one_start(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):

    centres = init_centres.copy()
    radii = np.full(len(centres), 1e-3)
    best_radii, best_sum = refine_radii(centres, radii.copy()), np.sum(refine_radii(centres, radii.copy()))
    base_centres, base_radii = best_centres.copy(), best_radii.copy()
    sum = best_sum
    
    for stepnum in range(steps):
        T = 0.25 * (1 - stepnum / steps)
        step = init_step * (1 - stepnum / steps)
        
        i = rng.choice(len(centres))
        delta = (-step, step)
        old_pos = centres[i].copy()
        centres[i] += rng.choice(delta, 2)
        max(0, min(1, centres[i, 0])), max(0, min(1, centres[i, 1])))

        new_radii = refine_radii(centres, radii.copy())
        new_sum = np.sum(new_radii)

        if new_sum > sum or rng.random() < np.exp((new_sum - sum) / T):
            sum = new_sum
            base_radii, base_sum = new_radii.copy(), sum
        else:
            centres[i] = old_pos

    return base_centres, base_radii, base_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng()
    centers, radii, sum_radii = None, None, -np.inf
    RESTARTS = 30
    ANNEAL_STEPS = 500

    for _ in range(RESTARTS):
        init_centres = rng.random(size=(26, 2))
        result = anneal_one_start(rng, init_centres, ANNEAL_STEPS)
        if np.sum(result[1]) > sum_radii:
            centers, radii, sum_radii = result

    return centers, radii, sum_radii
