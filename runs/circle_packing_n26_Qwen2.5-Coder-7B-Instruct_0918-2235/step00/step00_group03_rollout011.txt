import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_cycles = 30
    num_trials = 40
    eps_stop = 1e-4
    delta = 0.2
    step = 0.1
    alpha = 0.8
    rng = np.random.default_rng()

    def get_distances(centers):
        A = np.zeros((n, n))
        np.fill_diagonal(A, np.inf)
        np.fill_diagonal(A + np.triu(np.ones((n, n)) * -1, k=1), 1)
        return np.maximum(0, A)

    def get_radii_fixed_positions(centers):
        A_ub = get_distances(centers)
        b_ub = A_ub.sum(axis=1)
        bounds = [(0, min(c, 1-c)) for c in centers[:, 0] if c < 1]
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            raise ValueError("LP solve failed")
        return -res.fun

    def validate_layout(centers, radii):
        d = get_distances(centers)
        if not np.allclose(d, radii[:, None] + radii[None, :], atol=1e-12):
            raise ValueError("Radii do not match distances")
        if not np.all(centers[:, 0] >= 0) or not np.all(centers[:, 0] <= 1):
            raise ValueError("Centres out of range [0,1]")
        if not np.all(centers[:, 1] >= 0) or not np.all(centers[:, 1] <= 1):
            raise ValueError("Centres out of range [0,1]")
        if not np.all(radii >= 0):
            raise ValueError("Negative radii")
        if np.any(d < radii[:, None] + radii[None, :] - 1e-12):
            raise ValueError("Overlap")
        return True

    best_sum = -np.inf
    best_centers = None

    for trial in range(num_trials):
        centers = rng.uniform(0, 1, size=(n, 2))
        current_sum = get_radii_fixed_positions(centers)
        for cycle in range(max_cycles):
            for i in range(n):
                old_pos = centers[i]
                delta = rng.uniform(-delta, delta, size=2)
                centers[i] = np.clip(old_pos + delta, 0, 1)
                new_sum = get_radii_fixed_positions(centers)
                if new_sum <= current_sum + 1e-9:
                    centers[i] = old_pos
                else:
                    current_sum = new_sum
            step *= alpha
            if step < eps_stop:
                break
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()

    if not validate_layout(best_centers, np.ones(n) * best_sum / n):
        raise ValueError("Validation failed")

    radii = np.ones(n) * best_sum / n
    return best_centers, radii, best_sum
