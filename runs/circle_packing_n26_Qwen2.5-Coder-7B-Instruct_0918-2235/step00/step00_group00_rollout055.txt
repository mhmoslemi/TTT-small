import numpy as np
import math

def compute_max_radius(bd, min_dist):
    return min([bd / 2, min_dist / 2])

def update_radii(centers, radii, EPS=1e-8):
    for i in range(len(radii)):
        x, y = centers[i]
        r_nw = compute_max_radius(x, centers[i] - centers[0])
        r_ne = compute_max_radius(1 - x, centers[0] - centers[i])
        r_sw = compute_max_radius(y, centers[1] - centers[i])
        r_se = compute_max_radius(1 - y, centers[1] - centers[i])
        r_best = min(r_nw, r_ne, r_sw, r_se)
        for j in range(len(radii)):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                r_best = min(r_best, d - radii[j])
        if radii[i] > r_best - EPS:
            radii[i] = r_best
    return np.maximum(radii, 0)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.random.rand(26, 2)
    radii = np.zeros(26)
    for _ in range(50):
        radii = update_radii(centers, radii)
    while True:
        new_centers = centers + np.random.uniform(-0.1, 0.1, (26, 2))
        new_centers = np.clip(new_centers, 0, 1)
        new_radii = update_radii(new_centers, radii.copy())
        new_sum = np.sum(new_radii)
        current_sum = np.sum(radii)
        if new_sum > current_sum:
            centers, radii = new_centers, new_radii
        elif np.random.rand() < np.exp((new_sum - current_sum) / 1e-4):
            centers, radii = new_centers, new_radii
        sum_radii = np.sum(radii)
        is_valid, _ = validate_packing(centers, radii)
        if is_valid:
            return centers, radii, sum_radii
        else:
            print("Validation failed, retrying...")
            centers = np.random.rand(26, 2)
            radii = np.zeros(26)
