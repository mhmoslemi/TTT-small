def refine_radii(centers):
    n = centers.shape[0]
    c = np.c_[centers, np.zeros(n)]
    A = []
    b = []
    for i in range(n):
        A.append([1]*n + [0]*n)
        b.append(centers[i, 0])
        
        A.append([-1]*n + [0]*n)
        b.append(-1 + centers[i, 0])
        
        A.append([0]*n + [1]*n)
        b.append(centers[i, 1])
        
        A.append([0]*n + [-1]*n)
        b.append(-1 + centers[i, 1])
        
        for j in range(n):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                A.append([1,-1]*n)
                b.append(dist)
                
                A.append([-1,1]*n)
                b.append(dist)
                
    A = np.array(A)
    b = np.array(b)
    
    bounds = [(0, None)] * (2*n) + [(0, None)] * n
    result = linprog(np.ones(3*n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    radii = result.x[-n:]
    return radii
