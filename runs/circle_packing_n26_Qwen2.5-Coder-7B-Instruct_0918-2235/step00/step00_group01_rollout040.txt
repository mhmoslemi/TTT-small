import numpy as np
from scipy.optimize import linprog, differential_evolution

def objective(u):
    x = u[:26]
    y = u[26:]
    n = 26
    r_bounds = np.array([[0, 1]] * n).T
    
    c = np.ones(n)
    
    d = np.sqrt((x[:, None] - x) ** 2 + (y[:, None] - y) ** 2)
    A_ub = np.zeros((n * n, n * 2))
    b_ub = np.zeros(n * n)
    
    for i in range(n):
        A_ub[i * n + i, i * 2] = 1
        A_ub[i * n + i, i * 2 + 1] = 0
        b_ub[i * n + i] = 1 - x[i]
    
        A_ub[i * n + i + 1, i * 2] = 0
        A_ub[i * n + i + 1, i * 2 + 1] = 1
        b_ub[i * n + i + 1] = 1 - y[i]
    
        for j in range(i + 1, n):
            A_ub[i * n + j * 2 - i, i * 2] = -1
            A_ub[i * n + j * 2 - i, i * 2 + 1] = 0
            A_ub[i * n + j * 2 - i, (j - 1) * 2] = 1
            A_ub[i * n + j * 2 - i, (j - 1) * 2 + 1] = 0
            b_ub[i * n + j * 2 - i] = d[i, j]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    return -res.fun

def run_packing():
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    popsize = 20
    maxiter = 200
    tol = 1e-12
    
    result = differential_evolution(objective, bounds, popsize=popsize, maxiter=maxiter, tol=tol)
    x = result.x[:26]
    y = result.x[26:]
    
    n = 26
    r_bounds = np.array([[0, None]] * n).T
    
    c = np.ones(n)
    
    d = np.sqrt((x[:, None] - x) ** 2 + (y[:, None] - y) ** 2)
    A_ub = np.zeros((n * n, n))
    b_ub = np.zeros(n * n)
    
    for i in range(n):
        A_ub[i * n + i, i] = 1
    
        for j in range(i + 1, n):
            A_ub[i * n + j * 2 - i, i] = -1
            A_ub[i * n + j * 2 - i, j] = 1
            b_ub[i * n + j * 2 - i] = d[i, j]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    
    centers = np.column_stack([x, y])
    radii = res.x
    sum_radii = res.fun
    
    return centers, radii, sum_radii
