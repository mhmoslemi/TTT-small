import numpy as np
from scipy.optimize import linprog

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.zeros(n)
    dists = np.zeros((n, n))
    alpha = 0.8
    s = 0.1
    eps = 1e-9
    best_sum = -np.inf
    restarts = 40
    max_cycles = 30

    def compute_distances(centers):
        return np.sqrt((centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0]) ** 2 +
                      (centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1]) ** 2)

    def solve_lp(centers, radii, dists):
        A_ub = np.zeros((2 * n, n))
        b_ub = np.zeros(2 * n)
        c = -np.ones(n)
        bounds = np.array([(0, np.minimum(x, 1-x, y, 1-y)) for x, y in centers])

        for i in range(n):
            for j in range(i + 1, n):
                A_ub[i * 2, j] = 1
                A_ub[i * 2 + 1, j] = 1
                b_ub[i * 2] = dists[i, j]
                b_ub[i * 2 + 1] = dists[i, j]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'tol': 1e-8})
        if res.success:
            return res.x, res.success
        else:
            return None, res.success

    for _ in range(restarts):
        current_sum = 0
        for cycle in range(max_cycles):
            for i in range(n):
                c_i_prime = np.clip(centers[i] + np.random.uniform(-s, s, size=2), 0, 1)
                dists[:, i] = compute_distances(centers)
                dists[i] = compute_distances(c_i_prime.reshape(-1, 2))[:, 0]
                new_radii, success = solve_lp(centers, radii, dists)
                if success and new_radii.sum() > current_sum + eps:
                    centers[i] = c_i_prime
                    radii = new_radii
                    current_sum = new_radii.sum()
                else:
                    dists[:, i] = compute_distances(centers)[:, i]

            if cycle > 0 and current_sum - last_sum < eps:
                s /= 2

            last_sum = current_sum
            if s < 1e-4:
                break

        if current_sum > best_sum:
            best_sum = current_sum
            best_positions = centers
            best_radii = radii

    return best_positions, best_radii, best_sum
