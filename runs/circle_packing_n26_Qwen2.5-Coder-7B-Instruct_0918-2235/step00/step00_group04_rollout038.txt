import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def compute_lp_solution(centers):
    n = centers.shape[0]
    A_ub = np.zeros((2 * n + n * (n - 1), n))
    b_ub = np.zeros(2 * n + n * (n - 1))
    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = centers[i, 0]
        A_ub[i + n, i] = 1
        b_ub[i + n] = 1 - centers[i, 0]
        A_ub[i + 2 * n, i] = 1
        b_ub[i + 2 * n] = centers[i, 1]
        A_ub[i + 2 * n, n + i] = -1
        A_ub[i + 2 * n + 1, i] = 1
        b_ub[i + 2 * n + 1] = 1 - centers[i, 1]
    A_ub = A_ub[a > 0] for a in b_ub if a>0].T
    b_ub = b_ub[b_ub>0]
    c = -np.ones(n)
    res = linprog(c, A_ub, b_ub, method='highs')
    if not res.success:
        raise ValueError("LP solve unsuccessful. This should not happen.")
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    grid = np.array([[i/5, j/6] for i in range(5) for j in range(6)])
    centers = grid[np.random.choice(len(grid), n, replace=False)]
    centers = centers[np.random.choice(len(centers), n, replace=False)]
    current_centers = centers
    while True:
        shuffle_indices = np.arange(n)
        np.random.shuffle(shuffle_indices)
        current_centers = current_centers[shuffle_indices]
        lp_solution, current_sum = compute_lp_solution(current_centers)
        if np.sum(lp_solution) > current_sum:
            current_sum = np.sum(lp_solution)
            break
    best_centers = current_centers
    best_radii = lp_solution
    best_sum = current_sum
    max_iter = 20000
    k_jump = 500
    n_reduce = 500
    step = 0.1
    for iter in range(max_iter):
        i = np.random.randint(n)
        dx, dy = np.random.uniform(-step, step, size=2)
        new_centers = current_centers.copy()
        new_centers[i] += np.array([dx, dy])
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        lp_solution, new_sum = compute_lp_solution(new_centers)
        if new_sum > current_sum + 1e-8:
            current_centers = new_centers
            current_sum = new_sum
            best_centers = new_centers
            best_radii = lp_solution
            best_sum = new_sum
        if iter % k_jump == 0:
            new_centers = grid[np.random.choice(len(grid), n, replace=False)]
            new_centers = np.random.rand(n, 2) * [1, 1]
            new_centers = np.clip(new_centers, 0, 1)
            lp_solution, new_sum = compute_lp_solution(new_centers)
            if new_sum > best_sum:
                best_centers = new_centers
                best_radii = lp_solution
                best_sum = new_sum
        if (iter + 1) % n_reduce == 0 and step > 1e-8:
            step *= 0.9
            print(f"Step reduced to {step}")
        if step < 1e-4:
            break
    return best_centers, best_radii, best_sum
