def objective(v, λ):
    penalty = 0
    for i in range(0, len(v), 3):
        if v[i] - v[i+2] < 0:
            penalty += -(v[i] - v[i+2])
        if 1 - v[i] - v[i+2] < 0:
            penalty += -(1 - v[i] - v[i+2])
        if v[i+1] - v[i+2] < 0:
            penalty += -(v[i+1] - v[i+2])
        if 1 - v[i+1] - v[i+2] < 0:
            penalty += -(1 - v[i+1] - v[i+2])
        for j in range(i+3, len(v), 3):
            if (np.sqrt((v[i]-v[j])**2 + (v[i+1]-v[j+1])**2) - v[i+2] - v[j+2]) < 0:
                penalty += -((v[i+2] + v[j+2]) - np.sqrt((v[i]-v[j])**2 + (v[i+1]-v[j+1])**2))
    return np.sum(v[2::3]) - λ * penalty

def radius_to_clearance(coords):
    x, y, r = coords
    r = min(x, y, 1-x, 1-y)
    for i in range(26):
        if i*3 != coords[0]:
            dist = np.sqrt((x-coords[3*i])**2 + (y-coords[3*i+1])**2)
            r = min(r, dist-coords[3*i+2])
    return r
