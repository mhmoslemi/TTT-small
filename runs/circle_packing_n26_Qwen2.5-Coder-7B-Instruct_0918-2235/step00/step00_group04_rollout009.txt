import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum of radii
    A = []
    b = []
    
    # r_i <= x_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(centers[i, 0])
    
    # r_i <= 1 - x_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - centers[i, 0])
    
    # r_i <= y_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(centers[i, 1])
    
    # r_i <= 1 - y_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - centers[i, 1])
    
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dist_ij)
    
    A = np.array(A)
    b = np.array(b)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        radii = res.x
        sum_radii = -res.fun
    else:
        raise ValueError("LP solver failed to converge")
    
    centers = np.array(centers)
    return centers, radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # For reproducibility
    n = 26
    gap = 0.16
    
    # Create a 5x6 grid
    grid = np.array([[i * gap, j * gap] for i in range(6) for j in range(5)])
    centers = grid.flat[:n].reshape(n, 2)
    
    # Randomly delete 4 points to get exactly 26 centers
    indices = np.random.choice(np.arange(n), n - 22, replace=False)
    centers = np.delete(centers, indices, axis=0)
    
    # Generate random perturbations to avoid duplicate positions
    noise = 0.01 * np.random.randn(2 * n)
    centers = centers + noise.reshape(-1, 2)
    centers = np.clip(centers, 0, 1)
    
    current_centers = centers
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers = current_centers
    best_radii = current_radii
    best_sum = current_sum
    
    step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    eps = 1e-12
    
    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        new_x = np.clip(current_centers[i, 0] + (np.random.rand() * 2 * step - step), 0, 1)
        new_y = np.clip(current_centers[i, 1] + (np.random.rand() * 2 * step - step), 0, 1)
        new_centers = np.array(current_centers)
        new_centers[i] = [new_x, new_y]
        
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum + eps:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = current_centers
                best_radii = current_radii
                best_sum = new_sum
        else:
            if iter % N_REDUCE == 0 and iter > 1000:
                step *= 0.9
        
        if iter % K_JUMP == 0 and iter > 1000:
            new_centers = np.random.rand(n, 2)
            new_centers = np.clip(new_centers, 0, 1)
            new_radii, new_sum = compute_lp_solution(new_centers)
            if new_sum > best_sum:
                best_centers = new_centers
                best_radii = new_radii
                best_sum = new_sum
    
    return best_centers, best_radii, best_sum

# Example run
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of radii: {sum_radii}")
