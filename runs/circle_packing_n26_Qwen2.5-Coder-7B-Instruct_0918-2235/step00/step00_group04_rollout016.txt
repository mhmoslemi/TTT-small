from scipy.optimize import linprog

def compute_lp_solution(centers):

    b = np.empty(n + 4*n*(n-1)//2 + 3*n)

    row_idx = 0
    for i in range(n):
        # r_i <= x_i
        A[row_idx, i] = 1
        b[row_idx] = centers[i, 0]
        row_idx += 1
        # r_i <= 1 - x_i
        A[row_idx, i] = 1
        b[row_idx] = 1 - centers[i, 0]
        row_idx += 1
        # r_i <= y_i
        A[row_idx, i] = 1

        row_idx += 1

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # Fix random seed for reproducibility
    n = 26
    # Create initial grid and randomly drop 4 points
    grid = np.array([[x, y] for x in np.linspace(0.1, 0.9, 5) for y in np.linspace(0.1, 0.9, 5)])
    centers = np.random.choice(grid, size=(26,), replace=False)
    np.random.shuffle(centers)
    
    current_centers = centers
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
    
    step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    ε = 1e-12
    
    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        new_x = np.clip(current_centers[i, 0] + np.random.uniform(-step, step), 0, 1)
        new_y = np.clip(current_centers[i, 1] + np.random.uniform(-step, step), 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = [new_x, new_y]
        
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum + ε:
            current_centers = new_centers.copy()
            current_radii = new_radii
            current_sum = new_sum
            
            if new_sum > best_sum:
                best_centers = current_centers.copy()
                best_radii = current_radii
                best_sum = new_sum
        
        if (iter + 1) % K_JUMP == 0:
            # Perform large perturbation every few hundred iterations
            new_centers = np.random.rand(n, 2)
            new_centers[:, 0] *= 0.8
            new_centers[:, 0] += 0.1
            new_centers[:, 1] *= 0.8
            new_centers[:, 1] += 0.1
            new_radii, new_sum = compute_lp_solution(new_centers)
            
            if new_sum > best_sum:
                best_centers = new_centers.copy()
                best_radii = new_radii
                best_sum = new_sum
        
        if (iter + 1) % N_REDUCE == 0:
            step *= 0.9
    
    return best_centers.reshape((n, 1)), best_radii, best_sum
