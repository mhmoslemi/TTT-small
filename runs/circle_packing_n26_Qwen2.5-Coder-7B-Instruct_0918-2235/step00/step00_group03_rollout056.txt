import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    max_cycles = 50
    num_trials = 40
    step_size_init = 0.1
    step_size_decay = 0.8
    step_size_min = 1e-4
    eps_stop = 1e-4

    avg_sum_radii = 0.0

    for trial in range(num_trials):
        centers = np.random.rand(num_circles, 2)
        best_sum = -np.inf

        for cycle in range(max_cycles):
            step_size = step_size_init * (step_size_decay ** cycle)
            if step_size < step_size_min:
                break

            for i in np.random.permutation(num_circles):
                old_pos = centers[i]
                delta = np.random.uniform(-step_size, step_size, size=2)
                centers[i] = np.clip(old_pos + delta, 0, 1)

                r = _get_radii_from_positions(centers)
                sum_radii = np.sum(r)

                if sum_radii > best_sum + eps_stop:
                    best_sum = sum_radii
                else:
                    centers[i] = old_pos

            avg_sum_radii += best_sum / num_trials

    return centers, r, avg_sum_radii

def _get_radii_from_positions(centers):
    dists = squareform(pdist(centers))
    radii = np.zeros(len(centers))
    bounds = [(0, min(c, 1 - c)) for c in centers]
    A_ub, b_ub = _build_linear_constraints(centers, dists)

    res = linprog(-np.ones(len(radii)), A_ub=A_ub, b_ub=b_ub, bounds=bounds)

    if not res.success:
        raise ValueError(f"Linear programming problem failed: {res.message}")

    return np.maximum(res.x, 0)
    
def _build_linear_constraints(centers, dists):
    A_ub = []
    b_ub = []
    n = len(centers)

    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([1, 0] * i + [1] + [0] * (j - i - 1) + [1] + [0] * (n - j - 1))
            b_ub.append(dists[i, j])

    return np.array(A_ub), np.array(b_ub)
