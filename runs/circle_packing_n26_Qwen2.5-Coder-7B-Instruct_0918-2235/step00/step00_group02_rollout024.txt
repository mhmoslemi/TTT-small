import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_sum_radii(positions):
    n = 26
    positions = positions.reshape(n, 2)
    dmat = np.empty((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dmat[i, j] = dmat[j, i] = np.sqrt((positions[i, 0] - positions[j, 0]) ** 2 + (positions[i, 1] - positions[j, 1]) ** 2)
    
    A_ub = np.empty((n * 3 + n * (n - 1) // 2, n))
    b_ub = np.empty(n * 3 + n * (n - 1) // 2)
    
    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = 0.5
    
    offset = n * 3
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[offset + (i * n + j), i] = 1
            A_ub[offset + (i * n + j), j] = 1
            b_ub[offset + (i * n + j)] = dmat[i, j]
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return -res.fun
    else:
        return -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 52
    result = differential_evolution(compute_sum_radii, bounds, popsize=20, recombination=0.7, maxiter=100, disp=False)
    positions = result.x.reshape(26, 2)
    r = np.zeros(26)
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt((positions[i, 0] - positions[j, 0]) ** 2 + (positions[i, 1] - positions[j, 1]) ** 2)
            r[i] = dist / 2 - 1e-12
            r[j] = dist / 2 - 1e-12
    return positions, r, np.sum(r)
