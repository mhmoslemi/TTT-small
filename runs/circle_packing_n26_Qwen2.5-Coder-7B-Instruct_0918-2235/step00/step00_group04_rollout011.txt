import numpy as np
from scipy.optimize import linprog


def solve_radius_lp(centres):
    n_circles = centres.shape[0]
    c = -np.ones(n_circles)
    A_nei = np.zeros((n_circles * (n_circles - 1) // 2, 2 * n_circles))
    b_nei = np.zeros(n_circles * (n_circles - 1) // 2)
    A_bound = np.zeros((4 * n_circles, 2 * n_circles))
    b_bound = np.zeros(4 * n_circles)

    di = np.tile(centres, (n_circles, 1, 1)) - np.swapaxes(centres, 0, 1)
    di = di[:, ~np.eye(n_circles, dtype=bool), :]
    d2 = np.sum(di**2, axis=2)
    np.fill_diagonal(d2, np.inf)
    d = np.sqrt(d2)
    idx_nei = d2 <= 1
    d_nei = d[idx_nei]
    A_nei[idx_nei, np.arange(2 * n_circles) % 2::2] = centres[idx_nei]
    A_nei[idx_nei, np.arange(2 * n_circles) % 2] = -centres[idx_nei]
    np.fill_diagonal(A_nei, d_nei)

    lb = np.zeros(2 * n_circles)
    ub = np.multiply(centres.flatten(), 2)

    ub[centres < 0.5] = np.inf
    ub[centres > 0.5] = np.inf

    res = linprog(c, A_ub=np.vstack((A_bound, A_nei)), b_ub=np.hstack((b_bound, b_nei)),
                  bounds=(lb, ub), method='highs', options={'tol': 1e-5})
    return -res.x[:n_circles], -res.fun


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_particles = 30
    n_circles = 26
    n spatial_dims = 2

    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    alpha = 0.999
    max_iter = 3000
    vel_clip = 0.1

    np.random.seed(42)

    grid = np.mgrid[0:1:5j, 0:1:6j].T.reshape(-1, 2)
    grid = grid[~np.all(grid == 0, axis=1) & ~np.all(grid == 1, axis=1)]
    np.random.shuffle(grid)
    grid = grid[:n_circles, :]
    grid += np.random.normal(0, 0.01, grid.shape)

    centres = np.repeat(grid[np.newaxis:,:,:], n_particles, axis=0)
    velocities = np.zeros_like(centres)
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres.mean(0)
    gbest_sum = solve_radius_lp(gbest)[1]

    for it in range(max_iter):
        for p in range(n_particles):
            r1, r2 = np.random.uniform(0, 1, 2)
            velocities[p] = inertia * velocities[p] + cognitive * r1 * (pbests[p] - centres[p]) + \
                             social * r2 * (gbest - centres[p])

            velocities[p] = np.clip(velocities[p], -vel_clip, vel_clip)
            centres[p] += velocities[p]

            centres[p, 0] = np.clip(centres[p, 0], 0, 1)
            centres[p, 1] = np.clip(centres[p, 1], 0, 1)

            radii, sum_radii = solve_radius_lp(centres[p])
            if sum_radii > pbest_sums[p]:
                pbests[p], pbest_sums[p] = centres[p], sum_radii

        if np.max(pbest_sums - gbest_sum) < 1e-6:
            gbest_sum = np.max(pbest_sums)
            gbest = pbests[np.argmax(pbest_sums)]

        if (it + 1) % 500 == 0:
            swap_indices = np.random.choice(n_particles, int(n_particles * 0.1), replace=False)
            centres[swap_indices] = np.repeat(grid[np.newaxis, :, :], int(n_particles * 0.1), axis=0) + \
                                   np.random.normal(0, 0.01, centres[swap_indices].shape)

    best_centres, best_radii = gbest, solve_radius_lp(gbest)[0]
    return best_centres, best_radii, best_radii.sum()
