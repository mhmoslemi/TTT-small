import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def fixed_random_star_points(n, s=1.0):
    theta = np.linspace(0, 2 * np.pi, n + 1)[:-1]
    x = s * np.cos(theta)
    y = s * np.sin(theta)
    x += s * np.random.random(n) - 0.5
    y += s * np.random.random(n) - 0.5
    return np.array([x, y]).T

def objective_with_grad(triplet):
    centers = triplet[:52].reshape(26, 2)
    radii = triplet[52:]

    A_ub = build_A_ub(centers)
    b_ub = build_b_ub(centers, radii)
    res = minimizeobjective(radii, A_ub, b_ub, centers)
    if res.success:
        return -np.sum(radii), res.jac
    else:
        # Fallback to random initialization
        return -np.inf, np.zeros_like(triplet)

def minimizeobjective(radii, A_ub, b_ub, centers):
    from scipy.optimize import linprog
    result = linprog(-radii, A_ub=A_ub, b_ub=b_ub, bounds=(0, 1))
    return result

def build_A_ub(centers):
    n = centers.shape[0]
    A1 = np.hstack([np.eye(n), np.zeros((n, n))])
    A2 = np.hstack([np.zeros((n, n)), np.eye(n)])
    A3 = np.zeros((n, 2*n))
    A4 = np.zeros((n, 2*n))
    for i in range(n):
        for j in range(i+1, n):
            if (centers[i] - centers[j]).dot(centers[i] - centers[j]) <= (1e-12)**2:
                continue
            d = np.linalg.norm(centers[i] - centers[j])
            A3[i, j] = centers[i, 0] - centers[j, 0] / d
            A3[i, j+n] = centers[i, 1] - centers[j, 1] / d
            A4[j, i] = centers[i, 0] - centers[j, 0] / d
            A4[j, i+n] = centers[i, 1] - centers[j, 1] / d
    A_ub = np.vstack([A1, A2, A3, A4])
    return A_ub

def build_b_ub(centers, radii):
    n = centers.shape[0]
    b1 = centers[:, 0]
    b2 = 1 - centers[:, 0]
    b3 = centers[:, 1]
    b4 = 1 - centers[:, 1]
    b5 = np.sum(centers**2, axis=1) + 2 * np.outer(radii, radii) + 2 * radii**2 - 2 * centers @ centers.T - 1
    b_ub = np.hstack([b1, b2, b3, b4, b5])
    return b_ub

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(2*n).reshape(n, 2)
    np.clip(centers, 0, 1, out=centers)

    # Initial guess: radii set to values that allow circles to just touch
    radii = np.clip(0.5 * np.ones(n) * np.random.rand(n), 0.1, 0.5)
    initial_guess = np.hstack([centers.ravel(), radii])

    result = minimize(objective_with_grad, initial_guess, method='L-BFGS-B', bounds=[(0,1)]*2*n + [(0,None)]*n)
    if result.success:
        final_centers = result.x[:2*n].reshape(n, 2)
        final_radii = result.x[2*n:]
        return final_centers, final_radii, -result.fun
    else:
        return np.zeros((n,2)), np.zeros(n), 0
