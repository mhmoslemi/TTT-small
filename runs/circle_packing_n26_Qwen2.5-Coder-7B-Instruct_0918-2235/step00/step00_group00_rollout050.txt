import numpy as np
import cvxpy as cp

def distance_matrix(coords):
    n = coords.shape[0]
    return np.sqrt((coords[:, None, :] - coords[None, :, :]).square().sum(axis=2))

def compute_r_max(candidates, placed_centres, placed_radii):
    r_max = np.minimum(
        np Minimum([np.linalg.norm(candidates - p, axis=1) - r for p, r in zip(placed_centres, placed_radii)], axis=0),
        1 - candidates[:, 0],
        candidates[:, 0],
        1 - candidates[:, 1],
        candidates[:, 1]
    )
    return np.maximum(0, r_max)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    h = 0.05
    epsilon = 1e-6
    
    # Generate a dense regular grid over the unit square
    x_coords = np.arange(epsilon, 1, h)
    y_coords = np.arange(epsilon, 1, h)
    candidates = np.stack(np.meshgrid(x_coords, y_coords), axis=-1).reshape(-1, 2)
    
    placed_centres = []
    placed_radii = []
    
    while len(placed_centres) < n:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    C = np.array(placed_centres)
    Rs = np.array(placed_radii)
    
    # Linear Programme to get optimal radii given fixed centres
    obj = cp.Minimize(cp.sum(Rs))
    constraints = [Rs <= x for x in C[:, 0]], [Rs <= 1 - x for x in C[:, 0]], \
                 [Rs <= y for y in C[:, 1]], [Rs <= 1 - y for y in C[:, 1]], \
                 [Rs + Rs[j] <= distance_matrix(C)[i, j] for i in range(n) for j in range(i+1, n)]
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-9)
    
    if np.isnan(problem.value):
        return np.zeros((n, 2)), np.zeros(n), 0
    
    radii = Rs + problem.value
    sum_radii = np.sum(radii)
    
    # Local refinement
    alpha = 0.02
    for _ in range(5000):
        move = np.random.uniform(-alpha, alpha, (n, 2))
        C_new = np.clip(C + move, 0, 1)
        Rs_new = np.array([R + (C_new[i] - C[i]).sqrt() - r for i, (R, r) in enumerate(zip(radii, placed_radii))])
        Rs_new = np.maximum(0, Rs_new)
        r_sum_new = np.sum(Rs_new)
        if r_sum_new > sum_radii:
            C = C_new
            radii = Rs_new
            sum_radii = r_sum_new
        alpha *= 0.999
    
    return C, radii, sum_radii
