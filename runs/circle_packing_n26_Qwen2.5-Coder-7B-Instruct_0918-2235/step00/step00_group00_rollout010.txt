import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters for simulated annealing
    n = 26
    N = 20000
    s_start = 0.15
    s_end = 0.001
    T_start = 1.0
    T_end = 1e-4
    
    # Initial random positions and zero radii
    centers = np.random.rand(n, 2)
    radii = np.zeros(n)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    
    for k in range(N):
        # Annealing schedule
        s_k = s_start * (s_end / s_start) ** (k / N)
        T_k = T_start * (T_end / T_start) ** (k / N)
        
        # Select a circle randomly
        i = np.random.randint(n)
        
        # Propose a new center
        new_center = centers[i] + np.random.uniform(-s_k, s_k, 2)
        new_center = np.clip(new_center, 0, 1)
        
        # Inflation step
        new_centers = np.where(np.arange(n) == i, new_center, centers)
        new_radii = np.zeros(n)
        while True:
            prev_radii = new_radii.copy()
            for j in range(n):
                for k in range(j + 1, n):
                    dist = np.linalg.norm(new_centers[j] - new_centers[k])
                    if dist > 1e-12:
                        new_radii[j] = min(1 - new_centers[j, 0], new_centers[j, 0], 1 - new_centers[j, 1], new_centers[j, 1],
                                           dist / 2 - prev_radii[k])
                        new_radii[k] = min(1 - new_centers[k, 0], new_centers[k, 0], 1 - new_centers[k, 1], new_centers[k, 1],
                                           dist / 2 - prev_radii[j])
                new_radii[j] = max(0, new_radii[j])
            updated = np.allclose(prev_radii, new_radii)
            if updated:
                break
        
        new_sum = np.sum(new_radii)
        
        # Acceptance test
        if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T_k):
            centers[i] = new_center
            radii = new_radii.copy()
            
            # Update best found configuration
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    # Validate the final packing
    if not validate_packing(best_centers, best_radii)[0]:
        raise ValueError("Failed to validate packed configuration")
    
    return best_centers, best_radii, best_sum
