import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

def build_lp(C):
    N = C.shape[0]
    distances = cdist(C, C)
    A = np.vstack((distances, C[:,0], C[:,1], C[:,0], C[:,1], np.ones(N)))
    b = np.hstack((distances.ravel() - 2 * np.eye(N).ravel(), C[:,0], C[:,1], 1 - C[:,0], 1 - C[:,1], np.ones(N)))
    c = -np.ones(N)
    bounds = [(0, None) for _ in range(N*2)]
    return A, b, c, bounds

def validate_and_correct_r(r):
    r = np.clip(r, 0, None)
    return r

def compute_sum_radius(centers, radii):
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def de_obj(C):
        try:

            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
