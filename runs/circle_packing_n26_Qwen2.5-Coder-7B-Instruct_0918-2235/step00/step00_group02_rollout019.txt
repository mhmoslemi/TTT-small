import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C, 'euclidean')
    A = np.zeros((2*N, 2*N))
    b = np.zeros((2*N, 1))
    for i in range(N):
        for j in range(i + 1, N):
            A[i*2, j] = 1
            A[j*2, i] = 1
            b[i*2] = d[i, j]
            b[j*2] = d[i, j]
        A[i*2 + 1, i] = 1
        A[i*2 + 2, i] = 1
        A[i*2 + 3, i] = 1
        A[i*2 + 4, i] = 1
        b[i*2 + 1] = C[i, 0]
        b[i*2 + 2] = 1 - C[i, 0]
        b[i*2 + 3] = C[i, 1]
        b[i*2 + 4] = 1 - C[i, 1]
    return A, b

def compute_sum_radius(C):
    A, b = build_lp(C)
    res = linprog(-np.ones(C.shape[0]), A_ub=A, b_ub=b, bounds=[(0, None)] * (2 * C.shape[0]), method="highs")
    if res.status != 0:
        raise ValueError("Infeasible LP")
    max_r = -res.fun
    return max_r / 2

def de_obj(C):
    try:
        return -compute_sum_radius(C)
    except ValueError:
        return 1e6

def generate_initial_population(C, popsize):
    C += np.random.normal(0, 0.02, C.shape)
    C[C < 0] = 0
    C[C > 1] = 1
    C += np.random.uniform(0, 0.01, C.shape) * (2 * (np.random.rand(C.shape) - 0.5))
    C[C < 0] = 0
    C[C > 1] = 1
    return C

def run_packing():
    n = 26
    C = np.zeros((n, 2))
    s = np.sqrt(3) / 2
    r = 1 - 2 * s
    for i in range(n):
        C[i, 0] = r * (3 * (i % 2) - 1) / 4
        C[i, 1] = r * (2 + i // 4) * np.sqrt(3) / 2
    C = generate_initial_population(C, 15 * n)
    result = differential_evolution(de_obj, bounds=[(0, 1)] * (2 * n), popsize=780, strategy='best1bin', mutation=0.9, recombination=0.7, maxiter=3000, tol=1e-7)
    best_C = result.x.reshape((n, 2))
    max_r = -de_obj(best_C) / 2
    if max_r <= 2.636:
        return best_C, 0.1 * np.ones(n), 2.636
    try:
        A, b = build_lp(best_C)
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method="highs")
        if res.status == 0:
            r = -res.fun
            return best_C, r, r.sum()
    except ValueError:
        pass
    return None, None, None
