import numpy as np
from scipy.optimize import linprog, differential_evolution

# Utility functions
def hex_grid(radius, packing_density):
    s = radius * 2 / np.sqrt(3)
    n = int(packing_density * (1 / s)**2)
    grid = np.zeros((n, n))
    # Fill grid with hexagonally packed circles
    return grid

def add_circles_around_quadrant(center, radius, packing_density):
    s = radius * 2 / np.sqrt(3)
    n = int(packing_density * (1 / s)**2)
    grid = hex_grid(radius, packing_density)
    # Place circles around a quadrant
    return grid

def solve_radii_fixed_centres(centres):
    n = len(centres)
    radii = np.zeros(n)
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1]) - r[i]})
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.linalg.norm(centres[i] - centres[j])**2 - (r[i] + r[j])**2})
    result = linprog(-radii, A_ub=np.array([constraint['fun'] for constraint in constraints]).T, b_ub=np.zeros(n))
    return -result.x, result.fun

def penalty_centres_fixed_radii(centres, radii):
    penalty = 0
    for i in range(len(centres)):
        penalty += max(0, radii[i] - min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1]))
    for i in range(len(centres)):
        for j in range(i + 1, len(centres)):
            penalty += max(0, radii[i] + radii[j] - np.linalg.norm(centres[i] - centres[j]))
    return penalty

def coordinated_ascent(init_centres, max_iter=15):
    n = len(init_centres)
    best_centres = init_centres.copy()
    best_radii = np.zeros(n)
    best_sum_radii = 0
    for _ in range(max_iter):
        radii, sum_radii = solve_radii_fixed_centres(init_centres)
        penalty = penalty_centres_fixed_radii(init_centres, radii)
        if sum_radii > best_sum_radii:
            best_centres = init_centres.copy()
            best_radii = radii
            best_sum_radii = sum_radii
        init_centres = differential_evolution(penalty_centres_fixed_radii, bounds=[[0, 1] for _ in range(2*n)], args=(radii,), popsize=10, maxiter=10).x.reshape((n, 2))
    return best_centres, best_radii, best_sum_radii

            best_sum_radii = sum_radii
            best_centres = centres
            best_radii = radii
    return best_centres, best_radii, best_sum_radii
