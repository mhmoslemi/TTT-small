import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    trials = 40
    best_sum = -np.inf

    for trial in range(trials):
        centers = np.random.rand(n, 2)
        radii = np.zeros(n)
        dists = np.zeros((n, n))
        A_ub = np.zeros((n * (n - 1) // 2, n))
        b_ub = np.zeros(n * (n - 1) // 2)
        c = -np.ones(n)
        bounds = np.zeros((n, 2))

        for i in range(n):
            x_i, y_i = centers[i]
            r_i = np.minimum(x_i, 1 - x_i, y_i, 1 - y_i)
            bounds[i] = [0, r_i]

        for _ in range(100):
            new_centers = np.array(centers)
            new_radius_sum = 0
            
            for i in range(n):
                A_ub.fill(0)
                b_ub.fill(0)
                dists.fill(0)
                
                x_i, y_i = centers[i]
                r_i = np.minimum(x_i, 1 - x_i, y_i, 1 - y_i)
                j = 0

                for k in range(n):
                    if k == i:
                        continue

                    x_k, y_k = centers[k]
                    d_ij = np.sqrt((x_i - x_k) ** 2 + (y_i - y_k) ** 2)
                    dists[j] = d_ij
                    b_ub[j] = d_ij
                    A_ub[j, i] = 1
                    A_ub[j, k] = 1
                    j += 1

                new_radius_sum += r_i
                sol = linprog(c=c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
                if sol.success and sol.fun < new_radius_sum:
                    new_centers[i] = sol.x
                    new_radius_sum = sol.fun

            if new_radius_sum > best_sum:
                best_sum = new_radius_sum
                centers = new_centers

    return centers, np.maximum(0, np.minimum(1, np.sqrt(best_sum / 26) - 1e-6)), best_sum
