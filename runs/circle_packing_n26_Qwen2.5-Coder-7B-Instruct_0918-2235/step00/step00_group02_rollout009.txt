import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_sum_radii(positions):
    n = 26
    x = positions[:n]
    y = positions[n:]
    d_mat = np.sqrt((x[:, np.newaxis] - x) ** 2 + (y[:, np.newaxis] - y) ** 2)
    r = (d_mat / 2)
    r[np.triu_indices(n, k=1)] = 0
    r = np.diag(r.sum(axis=1)) - r
    c = -np.ones(n * 2)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([1 if i == j else 0 for j in range(n)])
        b_ub.append(1)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([1 if i == idx else 0 for idx in range(n)])
            A_ub.append([1 if j == idx else 0 for idx in range(n)])
            b_ub.extend([1, 1])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(0, 1)] * (n * 2))
    if result.success:
        return -result.fun
    else:
        return -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([[0.5, 0.5]])
    for i in range(4):
        centers = np.vstack((centers, [0.1 + i * 0.2, 0.1]))
        centers = np.vstack((centers, [0.1 + i * 0.2, 0.9]))
    for i in range(4):
        centers = np.vstack((centers, [0.1, 0.1 + i * 0.2]))
        centers = np.vstack((centers, [0.9, 0.1 + i * 0.2]))
    radii = np.full(n, 0.1)
    for i in range(n):
        if i == 0:
            radii[i] = 0.2
    positions = centers.flatten()
    result = differential_evolution(compute_sum_radii, bounds=[(0, 1)] * (n * 2), popsize=20, polish=False, maxiter=1000)
    if result.success:
        centers = result.x[:n].reshape(n, 2)
        radii = -compute_sum_radii(result.x[:n].flatten()) / n
        return centers, radii, sum(radii)
    else:
        return centers, radii, sum(radii)
