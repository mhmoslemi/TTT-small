import numpy as np
from scipy.optimize import linprog

def run_packing():
    n = 26
    max_attempts = 40
    max_cycles = 50
    step_size = 0.1
    alpha = 0.8
    eps_stop = 1e-4

    best_sum = -np.inf
    best_positions = None
    best_radii = None

    def random_positions():
        return np.random.rand(n, 2)

    def compute_distances(centers):
        dists = np.zeros((n, n))
        for i in range(n):
            # Direct distances are modified to exclude precise boundary contacts
            dists[i, i:] = np.linalg.norm((centers[i] - centers[i + 1:]), axis=1)
        dists = np.maximum(dists, 1e-6)
        return dists

    def compute_radii(centers):
        dists = compute_distances(centers)
        bounds = np.array([[0, min(pos, 1 - pos)] for pos in centers.flatten()])
        A_ub = np.zeros((n * (n - 1) // 2, n))
        b_ub = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n - 1):
            for j in range(i + 1, n):
                A_ub[idx, [i, j]] = 1
                b_ub[idx] = dists[i, j]
                idx += 1
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            raise ValueError('LP solve failed')
        radii = res.x
        return radii, res.fun * -1

    def update_positions(centers, radii):
        nonlocal step_size
        improved = False
        for _ in range(n):
            i = np.random.randint(0, n)
            dx, dy = np.random.uniform(-step_size, step_size, size=2)
            centers[i] += [dx, dy]
            centers[i, :] = np.clip(centers[i], 0.0, 1.0)
            new_radii, new_sum = compute_radii(centers)
            if new_sum > best_sum + 1e-9:
                best_sum = new_sum
                best_positions = centers.copy()
                best_radii = new_radii.copy()
                improved = True
        if not improved:
            step_size *= alpha
            if step_size < eps_stop:
                return False
        return True

    for _ in range(max_attempts):
        centers = random_positions()
        while not update_positions(centers, compute_radii(centers)[0]):
            pass

    return best_positions, best_radii, best_sum

# Define the run_packing function
run_packing()
