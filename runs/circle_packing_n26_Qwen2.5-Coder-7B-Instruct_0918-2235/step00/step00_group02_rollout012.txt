import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist

N = 26
BOUND = 1.0

def build_lp(center_matrix):
    C = center_matrix
    pairwise_distances = cdist(C, C)
    N = len(C)
    
    A = []
    b = []
    
    for i in range(N):
        A.append([-1] * N)
        b.append(0)
    
    for i in range(N):
        for j in range(i + 1, N):
            A.append([1, -1])
            b.append(pairwise_distances[i, j])
            A.append([-1, 1])
            b.append(pairwise_distances[i, j])
    
    for i in range(N):
        for j in range(i + 1, N):
            A.append([1, 0])
            b.append(C[i, 0] + pairwise_distances[i, j])
            A.append([0, 1])
            b.append(C[i, 1] + pairwise_distances[i, j])
            A.append([-1, 0])
            b.append(1 - C[i, 0] + pairwise_distances[i, j])
            A.append([0, -1])
            b.append(1 - C[i, 1] + pairwise_distances[i, j])
    
    A = np.array(A)
    b = np.array(b)
    
    return A, b

def de_obj(centers):
    centers = centers.reshape((N, 2))
    A, b = build_lp(centers)
    result = linprog([-1] * N, A_ub=A, b_ub=b, bounds=[(0, BOUND) for _ in range(len(centers))])
    if result.success:
        return result.fun
    else:
        return 1e6

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.empty((N, 2))
    centers[::2, 0] = np.linspace(1/6, 5/6, N//2)
    centers[1::2, 0] = np.concatenate([np.linspace(1/3, 2/3, N//2), np.linspace(2/3, 1/3, N//2)])
    centers[::2, 1] = np.concatenate([np.ones(N//2), np.zeros(N//2)])
    centers[1::2, 1] = np.concatenate([np.zeros(N//2), np.ones(N//2)])
    
    popsize = 15 * N
    bounds = [(-np.inf, np.inf) for _ in range(N * 2)]
    strategy = 'best1bin'
    mutation = 0.9
    recombination = 0.7
    maxfev = 3000
    tol = 1e-7
    
    de_result = differential_evolution(de_obj, bounds, popsize=popsize, strategy=strategy, mutation=mutation, recombination=recombination, maxfev=maxfev, tol=tol)
    
    if de_result.fun < 2.636:
        centers = de_result.x.reshape((N, 2))
        A, b = build_lp(centers)
        result = linprog([-1] * N, A_ub=A, b_ub=b, bounds=[(0, BOUND) for _ in range(len(centers))])
        if result.success:
            radii = -result.fun
    
    validate_result, message = validate_packing(centers, radii)
    if not validate_result:
        raise Exception(message)
    
    return centers, radii, sum(radii)
