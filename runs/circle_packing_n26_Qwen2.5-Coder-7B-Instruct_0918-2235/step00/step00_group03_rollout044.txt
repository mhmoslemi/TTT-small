import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    R = 40
    s0 = 0.1
    alpha = 0.8
    epsilon = 1e-9
    best_sum = -np.inf

    for _ in range(R):
        # Initialize random positions within [0,1] square
        centers = np.random.rand(n, 2)
        radii, dists = compute_radii_and_dists(centers)
        current_sum = np.sum(radii)
        steps_since_improvement = 0

        while s0 > 1e-4:
            s = s0
            for _ in range(n):
                accept_move = False
                while not accept_move and s > 1e-4:
                    i = np.random.randint(n)
                    delta = np.random.uniform(-s, s, size=2)
                    new_centers = centers.copy()
                    new_centers[i] = np.clip(new_centers[i] + delta, 0, 1)
                    new_radii, _ = compute_radii_and_dists(new_centers)
                    new_sum = np.sum(new_radii)

                    if new_sum > current_sum + epsilon:
                        centers = new_centers
                        radii = new_radii
                        current_sum = new_sum
                        accept_move = True
                    else:
                        s *= alpha

            if current_sum <= best_sum:
                steps_since_improvement += 1
                if steps_since_improvement >= 5:
                    s *= 0.5
            else:
                best_sum = current_sum
                steps_since_improvement = 0

            s0 *= alpha

    return centers, radii, best_sum

def compute_radii_and_dists(centers):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
            dists[j, i] = dists[i, j]

    radii = np.zeros(n)
    bounds = [(0, None) for _ in range(n)]
    for i in range(n):
        bounds[i] = (0, min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]))

    from scipy.optimize import linprog
    A = np.zeros((n * (n - 1) // 2, 2 * n))
    b = np.zeros(n * (n - 1) // 2)

    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i * 2:i * 2 + 2] = [1, 1]
            A[row, j * 2:j * 2 + 2] = [-1, -1]
            b[row] = dists[i, j]
            row += 1

    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise ValueError("LP solver failed to find a feasible solution")

    radii[:n] = -res.x[:n]
    return radii, dists
