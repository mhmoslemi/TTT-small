import numpy as np
from scipy.optimize import linprog, differential_evolution

def hex_grid(radius, packing_density):
    s = 2 * radius / np.sqrt(3)
    max_radius = 0.5 * s * packing_density
    centers = []
    x_offset = 0
    for i in range(8):
        y_offset = (np.sqrt(3) / 3) * s * i
        cx = np.arange(0, 1, s) + x_offset
        cy = np.arange(0, 1, s * np.sqrt(3))
        centers.extend(np.array(list(zip(np.repeat(cx, len(cy)), np.tile(cy, len(cx)))))[::2])
    return np.array(centers) * packing_density

def add_circles_around_quadrant(centers, quadrant_size):
    quadrant_size_factor = (1 / quadrant_size) ** 2
    return centers

def solve_radii_fixed_centres(centers):
    n = centers.shape[0]
    A = np.vstack([np.eye(n), -np.eye(n)])
    b = np.concatenate([np.min(centers[:, 0], axis=0)*(np.ones(n), np.zeros(n)), np.min(centers[:, 1], axis=0)*(np.zeros(n), np.ones(n))])
    c = np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n)
    return res.x, np.sum(res.x)

def penalty_centres_fixed_radii(centers, radii):
    penalties = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            penalties.append(max(0, radii[i] + radii[j] - dist))
        penalties.append(max(0, radii[i] - min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])))
    return np.sum(penalties)

def coordinated_ascent(init_centres, max_iter=15):
    n = init_centres.shape[0]
    best_centres = init_centres
    best_radii = np.zeros(n)
    best_sum = 0
    for _ in range(max_iter):
        # Fix radii and optimize centres
        radii, sum_radii = solve_radii_fixed_centres(best_centres)
        def penalty(x):
            return penalty_centres_fixed_radii(x.reshape(n, 2), radii)
        result = differential_evolution(penalty, bounds=[(0, 1)] * 2 * n)
        new_centres = result.x.reshape(n, 2)
        # Update only if better
        if sum_radii > best_sum:
            best_centres = new_centres
            best_radii = radii
            best_sum = sum_radii
    return best_centres, best_radii, best_sum

def local_audit_and_refine(centres, radii):
    n = centres.shape[0]
    for _ in range(10):
        i = np.random.randint(n)
        radii_list = list(radii)
        removed_radius = radii_list[i]
        del radii_list[i]
        added_radius = (1 - np.sum(radii_list)) / (n - 1)
        for j in range(min(n, 6)):
            radii_list[j] += removed_radius * (1 - j / 6)
        radii_list.append(added_radius)
        new_centres, _, _ = coordinated_ascent(centres, max_iter=5)
        if np.sum(radii_list) > best_sum:
            best_centres = new_centres
            best_radii = np.array(radii_list)
            best_sum = np.sum(best_radii)
    return best_centres, best_radii, best_sum

def run_packing():
    best_sum = 0
    best_centres = None
    best_radii = None
    for seed in range(10):
        np.random.seed(seed)
        radius = 0.03
        packing_density = 1.5
        centers = hex_grid(radius, packing_density)
        centers = add_circles_around_quadrant(centers, 0.25)
        centers = np.pad(centers, [(0, n - len(centers)), (0, 0)])
        centre_radius_max = np.min([np.min(centers[:, 0], axis=0), 1 - np.min(centers[:, 0], axis=0), np.min(centers[:, 1], axis=0), 1 - np.min(centers[:, 1], axis=0)])
        centre_radius_max = min(centre_radius_max * packing_density, 0.04)
        centres = centres * centre_radius_max
        centres += np.random.uniform(0, 0.01, size=centres.shape)
        optimal_centres, optimal_radii, optimal_sum = coordinated_ascent(centres, max_iter=20)
        optimal_centres, optimal_radii, optimal_sum = local_audit_and_refine(optimal_centres, optimal_radii)
        if optimal_sum > best_sum:
            best_centres = optimal_centres
            best_radii = optimal_radii
            best_sum = optimal_sum
    return best_centres, best_radii, best_sum
