import numpy as np
import scipy.optimize

def solve_radius_lp(centres):
    n = centres.shape[0]
    A_ub = np.zeros((n * n, n))
    b_ub = np.zeros(n * n)
    
    idx = 0
    for i in range(n):
        rmin = np.sqrt(centres[i]**2)
        A_ub[idx, i] = 1
        b_ub[idx] = rmin
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - rmin
        idx += 1
        rmin = np.sqrt((1 - centres[i])**2)
        A_ub[idx, i] = 1
        b_ub[idx] = rmin
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - rmin
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centres[i] - centres[j])**2))
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d
            idx += 1
    
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
    return res.x, res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_particles, n_circles = 30, 26
    inertia, cognitive, social, max_iter, vel_clip, alpha = 0.729, 1.49445, 1.49445, 3000, 0.1, 0.999
    
    seed = 42
    np.random.seed(seed)
    
    # Construct a 5x6 grid and randomly discard 4 points to leave exactly 26 sites
    grid = np.mgrid[0:1:5j, 0:1:5j][:,0:30].reshape((n_circles, 2)) - 0.25
    mask = np.random.choice([0, 1], size=n_circles, p=[4/25, 21/25])
    grid = grid[mask]
    
    centres = np.zeros((n_particles, n_circles, 2)) + grid[np.random.randint(0, 26, n_particles)][:, None, :]
    velocities = np.zeros((n_particles, n_circles, 2))
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres[np.argmin(pbest_sums), :, :]
    gbest_sum = pbest_sums[np.argmin(pbest_sums)]
    
    alpha_history = np.full(max_iter, alpha)
    
    for it in range(max_iter):
        alpha_history[it] *= alpha
        
        for p in range(n_particles):
            for i in range(n_circles):
                r1, r2 = np.random.rand(2)
                velocities[p, i] *= inertia
                velocities[p, i] += cognitive * r1 * (pbests[p, i] - centres[p, i])
                velocities[p, i] += social * r2 * (gbest[i] - centres[p, i])
                velocities[p, i] = np.clip(velocities[p, i], -vel_clip, vel_clip)
                centres[p, i] += velocities[p, i]
                centres[p, i] = np.clip(centres[p, i], 0, 1)
            
            radii, sum_radii = solve_radius_lp(centres[p])
            
            if sum_radii > pbest_sums[p]:
                pbests[p] = centres[p]
                pbest_sums[p] = sum_radii
            
            if sum_radii > gbest_sum:
                gbest = centres[p]
                gbest_sum = sum_radii
        
        for p in range(n_particles):
            velocities[p] /= alpha_history[it]
    
    return gbest, np.zeros(n_circles), gbest_sum
