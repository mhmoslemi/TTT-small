import numpy as np
from scipy.optimize import linprog, differential_evolution

def objective(u):
    x = u[:26]
    y = u[26:]
    c = np.ones(26)
    
    d = np.sqrt(((x[:, np.newaxis] - x) ** 2 + (y[:, np.newaxis] - y) ** 2))
    np.fill_diagonal(d, np.inf)
    r_bounds = np.zeros(2 * 26)
    A_ub = np.zeros((26 * 26 + 26 * 4, 2 * 26))
    b_ub = np.zeros(26 * 26 + 26 * 4)
    
    index = 0
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub[index, i * 2] = 1
            A_ub[index, j * 2] = 1
            b_ub[index] = d[i, j]
            index += 1
    
    for i in range(26):
        A_ub[index, i * 2] = 1
        A_ub[index, (i * 2) + 1] = 0
        b_ub[index] = x[i]
        index += 1
        A_ub[index, i * 2] = 0
        A_ub[index, (i * 2) + 1] = 1
        b_ub[index] = y[i]
        index += 1
        A_ub[index, i * 2] = 1
        A_ub[index, (i * 2) + 1] = 0
        b_ub[index] = 1 - x[i]
        index += 1
        A_ub[index, i * 2] = 0
        A_ub[index, (i * 2) + 1] = 1
        b_ub[index] = 1 - y[i]
        index += 1
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    if result.success:
        return -result.fun
    else:
        return np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    result = differential_evolution(objective, bounds=bounds, popsize=20, maxiter=200)
    u = result.x
    x = u[:26]
    y = u[26:]
    
    c = np.ones(26)
    d = np.sqrt(((x[:, np.newaxis] - x) ** 2 + (y[:, np.newaxis] - y) ** 2))
    np.fill_diagonal(d, np.inf)
    A_ub = np.zeros((26 * 26 + 26 * 4, 2 * 26))
    b_ub = np.zeros(26 * 26 + 26 * 4)
    index = 0
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub[index, i * 2] = 1
            A_ub[index, j * 2] = 1
            b_ub[index] = d[i, j]
            index += 1
    
    for i in range(26):
        A_ub[index, i * 2] = 1
        A_ub[index, (i * 2) + 1] = 0
        b_ub[index] = x[i]
        index += 1
        A_ub[index, i * 2] = 0
        A_ub[index, (i * 2) + 1] = 1
        b_ub[index] = y[i]
        index += 1
        A_ub[index, i * 2] = 1
        A_ub[index, (i * 2) + 1] = 0
        b_ub[index] = 1 - x[i]
        index += 1
        A_ub[index, i * 2] = 0
        A_ub[index, (i * 2) + 1] = 1
        b_ub[index] = 1 - y[i]
        index += 1
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    centers = np.vstack((x, y)).T
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
