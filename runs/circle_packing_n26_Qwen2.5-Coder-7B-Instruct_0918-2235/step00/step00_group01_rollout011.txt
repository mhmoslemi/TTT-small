        valid = np.ones(S, dtype=bool)

            invalid = cdist(candidates, [pts[j]], 'euclidean') < (rad[j] + MIN_R)
            valid = np.logical_and(valid, ~invalid)

        valid = np.logical_and(valid, candidates[:, 0] <= 1 - MIN_R)
        valid = np.logical_and(valid, candidates[:, 1] >= MIN_R)
        valid = np.logical_and(valid, candidates[:, 1] <= 1 - MIN_R)
        
        if not valid.any():
            rad[k] = MIN_R
            continue
        
        clearance = np.zeros(S)
        clearance[valid] = np.minimum(
            np.minimum(candidates[valid, 0], 1 - candidates[valid, 0]),
            np.minimum(candidates[valid, 1], 1 - candidates[valid, 1])
        )

            rad[k] = MIN_R
            continue

        idx = np.argmax(clearance)
        pts[k] = candidates[idx]
