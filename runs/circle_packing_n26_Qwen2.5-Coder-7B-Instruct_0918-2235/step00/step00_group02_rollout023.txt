import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_sum_radii(positions):
    n = 26
    positions = positions.reshape(n, 2)
    dmat = np.sqrt(np.sum((positions[:, np.newaxis, :] - positions[np.newaxis, :, :]) ** 2, axis=2))
    dmat -= np.eye(n) * 1e-10
    
    # Radius constraints
    A = np.vstack((np.ones(n), -np.ones(n)))
    b = -npmin(dmat) + np.max(positions[:, 0]) - np.min(positions[:, 0])
    A_eq = np.eye(n)
    b_eq = np.zeros(n)
    
    # Solve for radii using linear programming
    res = linprog(np.ones(n), A_ub=np.vstack((A, dmat)), b_ub=np.hstack((b, np.zeros(n))), A_eq=A_eq, b_eq=b_eq, method='highs')
    if res.success:
        return -res.fun
    else:
        return -np.inf

def run_packing():
    # Initial positions from hex packing approximation
    hex_positions = [
        (0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9),
        (0.5, 0.1), (0.5, 0.9), (0.1, 0.5), (0.9, 0.5),
        (0.3, 0.1), (0.7, 0.1), (0.3, 0.9), (0.7, 0.9),
        (0.05, 0.05), (0.95, 0.05), (0.45, 0.2), (0.55, 0.2), (0.45, 0.4), (0.55, 0.4),
        (0.15, 0.3), (0.85, 0.3), (0.65, 0.5), (0.35, 0.5), (0.15, 0.7), (0.85, 0.7),
        (0.45, 0.6), (0.55, 0.6), (0.45, 0.8), (0.55, 0.8), (0.45, 0.9), (0.55, 0.9)
    ]
    initial_positions = np.array(hex_positions + [-1] * 10)
    
    # Perform hill climbing to improve positions
    def hill_climb(positions):
        n = 26
        best_positions = positions.copy()
        best_sum_radii = compute_sum_radii(positions)
        
        for _ in range(100):
            for i in range(n):
                for j in [-1, 0, 1]:
                    for k in [-1, 0, 1]:
                        new_positions = positions.copy()
                        new_positions[i, j] += np.random.normal(0, 0.01)
                        new_sum_radii = compute_sum_radii(new_positions)
                        if new_sum_radii > best_sum_radii:
                            best_positions = new_positions
                            best_sum_radii = new_sum_radii
        return best_positions, best_sum_radii
    
    # Run differential evolution with initial positions
    bounds = [(0, 1)] * 52
    result = differential_evolution(compute_sum_radii, bounds, popsize=20, maxiter=100, polish=True)
    if result.success:
        best_positions = result.x
    else:
        best_positions, _ = hill_climb(initial_positions)
    
    return best_positions[:n], compute_sum_radii(best_positions)
