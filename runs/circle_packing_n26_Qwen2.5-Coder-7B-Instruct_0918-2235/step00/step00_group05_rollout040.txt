import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist

def lp_radii_fixup(radii, centers, overlap_tolerance=1e-12):
    n = len(radii)
    A = np.zeros((n*(n-1), 2*n))
    b = np.zeros((n*(n-1)))
    for i in range(n-1):
        for j in range(i+1, n):
            A[i*(n-1)+j-1, i] = 1
            A[i*(n-1)+j-1, j] = 1
            A[i*(n-1)+j-1, n+i] = -1
            A[i*(n-1)+j-1, n+j] = -1
            b[i*(n-1)+j-1] = radii[i] + radii[j] - overlap_tolerance
    res = linprog(np.ones(n), A_ub=A, b_ub=b, bounds=None, method='highs')
    return res.x if res.success else radii

def total_penalties(centers, radii):
    wall_penalty = np.sum([
        np.maximum(0, radii - min(centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]))
    ])
    overlap_penalty = 0
    for i in range(len(centers)):
        for j in range(i+1, len(centers)):
            dist = np.linalg.norm(centers[i] - centers[j])
            overlap_penalty += max(0, (radii[i] + radii[j]) - dist)
    return wall_penalty, overlap_penalty

def penalty_function(x, n=26):
    centres = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    wall_penalty, overlap_penalty = total_penalties(centres, radii)
    return wall_penalty + overlap_penalty

def objective_function(x, n=26):
    centres = x[:2*n].reshape((n,2))
    radii = x[2*n:]
    objective_value = np.sum(radii) - penalty_function(x, n)
    return -objective_value  # We use negative because we are using a minimizer

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial guess = np.zeros(6*n)
    best_value = -np.inf
    best_centers = np.zeros((n,2))
    best_radii = np.zeros(n)
    
    for _ in range(8):
        initial_guess[:2*n] = np.array([
            np.random.uniform(0.01, 0.99, n) * np.sin(np.linspace(0, 2*np.pi, n + 1))[:-1],
            np.random.uniform(0.01, 0.99, n) * np.cos(np.linspace(0, 2*np.pi, n + 1))[:-1]
        ])
        initial_guess[2*n:] = np.floor(np.arange(n) * 0.5 / n)
        
        res = minimize(objective_function, initial_guess, method='L-BFGS-B', bounds=
                       [(0.0001, 0.99),]*(2*n) + [(0, 0.5)]*n,
                       args=(n,))
        centers = res.x[:2*n].reshape((n,2))
        radii = res.x[2*n:]
        objective_value = -res.fun
        wall_penalty, overlap_penalty = total_penalties(centers, radii)
        
        if objective_value > best_value:
            best_value = objective_value
            best_centers = centers
            best_radii = radii
    
    return best_centers, best_radii, best_value + penalty_function(best_centers, best_radii)
