import numpy as np
from scipy.spatial import ConvexHull
from scipy.optimize import minimize

def objective(packing):
    centers, radii = packing
    return -np.sum(radii)  # Maximize sum of radii

def constraint(packing):
    centers, radii = packing
    hull = ConvexHull(centers)
    radius = np.max(hull.points[np.argmax(hull.volume)])
    return 1 - radius  # Ensure circles fit within unit square

def is_valid(centers, radii):
    n = centers.shape[0]

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_guess_centers = np.random.random((26, 2))
    initial_guess_radii = 0.01 * np.ones(26)
    initial_guess = np.concatenate((initial_guess_centers.flatten(), initial_guess_radii))

    bounds = []
    for i in range(26):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))  # Limit radius to a fraction of the side length of the unit square

    constraints = [{'type': 'ineq', 'fun': constraint}]
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)

    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0

    centers = result.x[:52].reshape((26, 2))
    radii = result.x[52:]
    if not is_valid(centers, radii):
        return np.zeros((26, 2)), np.zeros(26), 0

    return centers, radii, -result.fun
