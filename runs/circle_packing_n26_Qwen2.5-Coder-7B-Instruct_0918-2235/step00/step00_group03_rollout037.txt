import numpy as np
from scipy.spatial.distance import cdist
from numpy.random import default_rng

rng = default_rng(42)

def compute_all_radii(centers, radii, dist_matrix, eps=1e-12):
    n = centers.shape[0]
    for i in range(n):
        r_x = min(centers[i, 0] - radii[i], 1 - centers[i, 0] - radii[i])
        r_y = min(centers[i, 1] - radii[i], 1 - centers[i, 1] - radii[i])
        r = min(r_x, r_y)
        r_dist = np.min(dist_matrix[i] - radii, axis=0) - radii[i]
        r = min(r, r_dist)
        if r < 0:
            r = 0
        radii[i] = r
    return radii

def refine_all(centers, radii, eps=1e-12, max_iter=200):
    dist_matrix = cdist(centers, centers)
    for _ in range(max_iter):
        changes = radii.copy()
        for i in range(centers.shape[0]):
            dist_matrix[:, i] -= radii[i]
            b_x = np.minimum(centers[i, 0] - radii[i], 1 - centers[i, 0] - radii[i])
            b_y = np.minimum(centers[i, 1] - radii[i], 1 - centers[i, 1] - radii[i])
            new_radii = np.minimum(b_x, b_y)
            new_radii = np.minimum(new_radii, dist_matrix[:, i] - eps)
            new_radii = np.maximum(new_radii, 0)
            changes[i] = new_radii
        radii = np.minimum(radii, changes)
    return radii

def generate_initial_centres(n, rng):
    return rng.random((n, 2))

def anneal_one_start(initial_centers, rng, steps=500, init_step=0.07, min_step=0.01):
    centers = initial_centers.copy()
    radii = np.full(26, 1e-3)
    dist_matrix = cdist(centers, centers)
    refine_all(centers, radii, dist_matrix=dist_matrix)
    base_sum = np.sum(radii)
    for step in range(steps):
        T = 0.25 * (1 - step / steps)
        stepsize = init_step * (1 - step / steps)
        i = rng.integers(0, centers.shape[0])
        dx, dy = rng.random(2) * 2 * stepsize - stepsize
        centers[i] = np.clip(centers[i] + np.array([dx, dy]), 0, 1)
        dist_matrix = cdist(centers, centers)
        refine_all(centers, radii, dist_matrix=dist_matrix)
        new_sum = np.sum(radii)
        if new_sum > base_sum or np.exp((new_sum - base_sum) / T) > rng.random():
            base_sum = new_sum
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    RESTARTS = 30
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    for _ in range(RESTARTS):
        initial_centers = generate_initial_centres(n, rng)
        centers, radii, sum = anneal_one_start(initial_centers, rng)
        if sum > best_sum:
            best_sum = sum
            best_centers = centers
            best_radii = radii
    return best_centers, best_radii, best_sum
