import numpy as np
from scipy.optimize import linear_sum_assignment, minimize
from scipy.spatial import distance_matrix

def calc_max_radii(C):
    if not (0 <= C).all() or not (C <= 1).all():
        return np.zeros(26), -1e27
    D = distance_matrix(C, C)
    D -= np.identity(26) * 1e-8
    A = np.empty((2 * 26, 26))
    A[:26, :] = np.eye(26)
    A[26:, :] = -np.eye(26)
    b = np.empty(2 * 26)
    b[:26] = 0.5 * (1 - D.diagonal())
    std_devs = 0.005 * np.ones(26)
    std_devs[D.diagonal()] = 1
    b[26:] = D.diagonal() + 2 * std_devs
    b[26:] = np.minimum(b, 1)
    c = np.ones(26)
    res = minimize(lambda x: -x.dot(c), b, method='SLSQP', 
                   bounds=[(b[i] if i < 26 else b[i], np.inf) for i in range(26)],
                   constraints={'type': 'ineq', 'fun': lambda x: b-x})
    if not res.success:
        return np.zeros(26), -1e27
    r = np.maximum(0, b[:26] - res.x)
    r = np.minimum(r, 0.5)
    s = r.sum()
    if not validate_packing(C, r)[0]:
        s = -1e27
    return r, s

def run_packing():
    n = 26
    target_sum = 2.636
    initial_temp = 0.1
    cooling_factor = 0.995
    max_iterations = 50000
    step_size = 0.02
    restart_threshold = 1000
    restarts = 5
    
    initial_layout = np.array([
        [0.18 * i + 0.1, 0.156 * j + 0.1]
        for i in range(5) for j in range([(6, 6, 6, 5, 6)][i])
    ])
    if initial_layout.shape[0] != n:
        initial_layout = np.array(
            [[0.2, 0.2], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
             [0.2, 0.6], [0.6, 0.6], [0.2, 0.8], [0.4, 0.8], [0.8, 0.8],
             [0.3, 0.1], [0.3, 0.3], [0.5, 0.1], [0.5, 0.3], [0.7, 0.1], [0.7, 0.3],
             [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9],
             [0.35, 0.1], [0.45, 0.1], [0.55, 0.1], [0.65, 0.1], [0.45, 0.9], [0.55, 0.9], [0.65, 0.9]]
        )
    current_C = initial_layout
    current_r, current_s = calc_max_radii(current_C)
    
    T = initial_temp
    for k in range(max_iterations):
        i = np.random.randint(n)
        current_C_proposed = current_C.copy()
        current_C_proposed[i, :] += step_size * np.random.randn(2)
        np.clip(current_C_proposed, 0, 1, out=current_C_proposed)
        current_r_proposed, current_s_proposed = calc_max_radii(current_C_proposed)
        
        if current_s_proposed > current_s or np.exp((current_s_proposed - current_s) / T) > np.random.rand():
            current_C = current_C_proposed
            current_r = current_r_proposed
            current_s = current_s_proposed
        
        T *= cooling_factor
        if k % restart_threshold == 0 and k < (5 * max_iterations):
            current_C = np.random.uniform(0.05, 0.95, size=(n, 2))
            current_r, current_s = calc_max_radii(current_C)
        
        if current_s >= target_sum:
            break
    
    final_C = current_C.copy()
    final_r, _ = calc_max_radii(final_C)
    
    return final_C, final_r, final_r.sum()
