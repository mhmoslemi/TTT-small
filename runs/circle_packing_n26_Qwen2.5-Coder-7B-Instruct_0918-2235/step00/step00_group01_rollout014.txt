import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])  # lower
        cons.append(1 - xs[i] - rs[i])  # upper
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i, j in combinations(range(n), 2):
        dx = xs[i] - xs[j]
        dy = ys[i] - ys[j]
        dist2 = dx * dx + dy * dy
        cons.append(dist2 - (rs[i] + rs[j]) ** 2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

def generate_initial_guess():
    n = 26
    r0 = 0.02
    xs = np.random.uniform(r0, 1 - r0, n)
    ys = np.random.uniform(r0, 1 - r0, n)
    rs = np.full(n, r0)
    while not check_feasibility(xs, ys):
        xs = np.random.uniform(r0, 1 - r0, n)
        ys = np.random.uniform(r0, 1 - r0, n)
    v = np.hstack([xs, ys, rs])
    return v

def check_feasibility(xs, ys):
    n = len(xs)
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < (np.minimum(xs[i], xs[j]) + np.minimum(ys[i], ys[j]) + min(xs[i], xs[j]) + min(ys[i], ys[j])):
                return False
    return True

def run_packing():
    v0 = generate_initial_guess()
    bnds = ((0, 1),) * len(v0)  # 26*x, 26*y, 26*r
    res = minimize(objective, v0, bounds=bnds, constraints={'type': 'ineq', 'fun': constraint_all}, options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    if not res.success:
        raise Exception("Optimization failed")
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs, ys))
    radii = rs
    sum_r = rs.sum()
    return centers, radii, sum_r
