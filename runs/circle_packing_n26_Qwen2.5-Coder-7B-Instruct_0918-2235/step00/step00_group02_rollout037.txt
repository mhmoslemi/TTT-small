import numpy as np

def calc_max_radii(C):
    num_circles = C.shape[0]
    A = np.empty((num_circles * (num_circles - 1), 2 * C.shape[0]))
    b = np.empty(num_circles * (num_circles - 1))
    
    idx = 0
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            A[idx, 2*i:2*i+2] = 1
            A[idx, 2*j:2*j+2] = -1
            b[idx] = np.linalg.norm(C[i] - C[j])
            idx += 1

    for i in range(num_circles):
        A[idx, 2*i:2*i+2] = 1
        b[idx] = C[i, 0]
        idx += 1

        A[idx, 2*i:2*i+2] = 1
        b[idx] = 1 - C[i, 0]
        idx += 1

        A[idx, 2*i:2*i+2] = 1
        b[idx] = C[i, 1]
        idx += 1

        A[idx, 2*i:2*i+2] = 1
        b[idx] = 1 - C[i, 1]
        idx += 1

    A = np.minimum(A, b.reshape(-1, 1), where=A < b.reshape(-1, 1))
    r = np.linalg.norm(C, axis=1)
    r = np.minimum(r, 0.5)

    sol = np.linalg.solve(A, b)
    r = np.minimum(r, np.clip(sol, 0, None))
    return r, np.sum(r)

def run_packing():
    centers = 0.5 * np.ones((26, 2))  # Start in the center of the unit square
    radii = np.zeros(26)
    radius_sum = np.sum(radii)
    
    def objective(x):
        nonlocal centers, radii, radius_sum
        centers = x.reshape((26, 2))
        radii, radius_sum = calc_max_radii(centers)
        return -radius_sum
    
    def acceptance_delta(new_sum, old_sum, temp):
        return (new_sum - old_sum) / temp
    
    T = 0.1
    alpha = 0.995
    max_iter = 50000
    sigma = 0.02
    trials_without_improvement = 0
    
    for _ in range(max_iter):
        i = np.random.randint(26)
        C_i = centers[i]
        C_i_new = C_i + np.random.normal(0, sigma, size=2)
        C_i_new = np.clip(C_i_new, 0, 1)
        
        centers[i] = C_i_new
        r_new, _ = calc_max_radii(centers)
        new_sum = np.sum(r_new)
        
        if objective(r_new) < objective(radii):
            radii = r_new
            radius_sum = new_sum
        else:
            prob = np.exp(acceptance_delta(new_sum, radius_sum, T))
            if np.random.rand() < prob:
                radii = r_new
                radius_sum = new_sum
            else:
                centers[i] = C_i
        
        T *= alpha
        trials_without_improvement += 1
        
        if new_sum > 2.636:
            break
        
        if trials_without_improvement > 1000:
            centers = 0.5 * np.ones((26, 2))
    
    if objective(radii) < -2e-5:
        raise Exception('Fallback to default configuration')
    
    r_optimal = calc_max_radii(centers)[0]
    return centers, r_optimal, np.sum(r_optimal)
