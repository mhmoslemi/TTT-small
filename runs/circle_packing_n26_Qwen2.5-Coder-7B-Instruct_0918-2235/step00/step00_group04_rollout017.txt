import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # maximize sum radii

    A = []  # inequality constraints will be added here
    b = []  # inequality constraints right-hand sides

    for i in range(n):
        A.append([1, 0, 0] + [-1]*n)  # r_i <= x_i
        b.append(centers[i][0])
        
        A.append([0, 1, 0] + [-1]*n)  # r_i <= y_i
        b.append(centers[i][1])

        A.append([-1, 0, 0] + [1]*n)  # -r_i <= 1 - x_i
        b.append(1 - centers[i][0])

        A.append([0, -1, 0] + [1]*n)  # -r_i <= 1 - y_i
        b.append(1 - centers[i][1])

    pairwise_constraints = list(range(n, 3*n))  # use different indices to avoid overlap with previously added constraints
    for i in range(n):
        for j in range(i + 1, n):
            A.append([-1]*3 + [1]*2 + [-1]*n)  # r_i + r_j <= d_ij
            A[-1][3] = 2  # coefficients for r_i and r_j
            A[-1][4] = 2  # coefficients for r_i and r_j
            A[-1][i] = 2  # positions
            A[-1][j] = 2  # positions
            dist = np.linalg.norm(centers[i] - centers[j])
            b.append(dist)

    A = np.array(A)
    b = np.array(b)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x, np.sum(res.x)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 20000
    K_JUMP = 500
    N_REDUCE = 500
    epsilon = 1e-12

    np.random.seed(42)
    
    # Generate initial centers: 5x6 grid minus 4 points
    grid = np.array([(x, y) for x in np.linspace(0, 1, 7) for y in np.linspace(0, 1, 7)])
    center_indices = np.random.choice(np.arange(49), 26, replace=False)
    initial_centers = grid[center_indices]

    # Complete local search
    current_centers = initial_centers
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum

    step = 0.1

    for iter in range(max_iter):
        # Select one of the circles
        i = np.random.randint(n)
        
        # Move circle by a small random step in x- and y-directions
        step_x = np.random.uniform(-step, step)
        step_y = np.random.uniform(-step, step)
        new_centers = current_centers.copy()
        new_centers[i] = [np.clip(current_centers[i, 0] + step_x, 0, 1), 
                          np.clip(current_centers[i, 1] + step_y, 0, 1)]

        # Solve LP with new centers
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        # Accept move if it improves sum of radii
        if new_sum > current_sum + epsilon:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = current_centers, current_radii, current_sum

        # Occasionally (every K_JUMP iterations) perform large perturbation
        if iter % K_JUMP == 0:
            random_centers = initial_centers + np.random.uniform(-0.1, 0.1, size=(n, 2))
            random_centers = np.clip(random_centers, 0, 1)
            new_radii, new_sum = compute_lp_solution(random_centers)
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = random_centers, new_radii, new_sum

        # Schedule to reduce step size every N_REDUCE iterations
        if iter % N_REDUCE == 0:
            step *= 0.9

    return best_centers, best_radii, best_sum
