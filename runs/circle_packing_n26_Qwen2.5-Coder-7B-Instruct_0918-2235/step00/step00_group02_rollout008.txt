import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

def compute_sum_radii(positions):
    n = 26
    positions = positions.reshape(n, 2)
    radii = np.zeros(n)
    max_distance = 1.0
    rmax = 0.1
    A = np.ones((n, n))
    np.fill_diagonal(A, 0)
    A_ub = -A
    b_ub = -rmax * np.ones(n)

    for i in range(n):
        for j in range(i):
            dist = np.linalg.norm(positions[i] - positions[j])
            A_ub[i + j * n] = dist - rmax
            b_ub[i + j * n] = dist
        
    A_ub = A_ub.reshape(n * n, n)
    A_eq = np.eye(n)
    b_eq = np.zeros(n)
    bounds = [(0, 1) for _ in range(n * 2)]
    
    try:
        result = linprog(radii, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not result.success:
            raise Exception("Linprog did not converge")
        radii = result.x
        return -np.sum(radii)
    except Exception as e:
        return float('-inf')

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_positions = np.random.rand(26, 2)
    result = differential_evolution(compute_sum_radii, bounds=[(0, 1)] * 52, popsize=20, maxfev=10000, seed=0)
    positions = result.x.reshape(26, 2)
    radii = np.zeros(26)
    for i in range(26):
        for j in range(i+1, 26):
            dist = np.linalg.norm(positions[i] - positions[j])
            r_i = 0.5 * (dist - 1e-12)
            if r_i > 0:
                radii[i] = r_i
    return positions, radii, -result.fun

# Run the packing function
centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
