```python
import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog

# Define constants
n = 26
eps = 1e-9
alpha = 0.8
s = 0.1
max_cycles = 5
max_trials = 40

# Define initial conditions
centres = np.random.rand(n, 2)
r = np.zeros(n)
dists = np.zeros((n, n))
simplex_A_ub = np.zeros((n*(n-1)//2, n))
simplex_b_ub = np.zeros(n*(n-1)//2)
    
# Build the system of inequalities and equalities
def build_system(centres):
    global simplex_A_ub, simplex_b_ub
    for i in range(n):
        for j in range(i+1, n):
            dists[i, j] = dists[j, i] = np.linalg.norm(centres[i] - centres[j])
            simplex_A_ub[i*(n-1)//2 + j - (i-1)*i//2][i] = 1
            simplex_A_ub[i*(n-1)//2 + j - (i-1)*i//2][j] = 1
            simplex_b_ub[i*(n-1)//2 + j - (i-1)*i//2] = dists[i, j]
        
def solve_radii(centres):
    build_system(centres)
    bounds = [(0, min(c, 1-c) + eps) for c in np.linalg.norm(centres, axis=1)]
    
    c = -np.ones(n)
    res = linprog(c, A_ub=simplex_A_ub, b_ub=simplex_b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("Linprog failed to converge.")
    
    return -res.fun

def accept_move(centres_new, radii_new, centres_old, radii_old, new_sum):
    global s
    if new_sum >= radii_old.sum() + eps:
        return True
    else:
        s *= alpha
        return False

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_centres = None

    for _ in range(max_trials):
        centres = np.random.rand(n, 2)
        for cycle in range(max_cycles):
            for k in range(n):
                proposed_centre = np.clip(centres[k] + s * np.random.rand(2) - 0.5, 0, 1)
                centres[k] = proposed_centre
                new_sum = solve_radii(centres)
                
                if accept_move(centres, solve_radii(centres), best_centres, best_sum, new_sum):
                    best_sum = new_sum
                    best_centres = centres.copy()
                    
            if accept_move(centres, solve_radii(centres), best_centres, best_sum, new_sum):
                best_sum = new_sum
                best_centres = centres.copy()
                
            s *= alpha

    return best_centres, solve_radii(best_centres), best_sum
```
