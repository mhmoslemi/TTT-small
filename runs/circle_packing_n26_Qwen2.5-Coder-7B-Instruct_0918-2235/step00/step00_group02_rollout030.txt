import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def calc_max_radii(C):
    n = C.shape[0]
    C = C.reshape(n, 2)
    D = cdist(C, C, 'euclidean')
    bounds = [(0, 1)] * 2 * n
    x0 = np.zeros(2 * n)
    
    def objective(x):
       .rad = x.reshape(n, 2)
        rad[:, 0] = np.minimum(rad[:, 0], C[:, 0])
        rad[:, 0] = np.maximum(rad[:, 0], 1 - C[:, 0])
        rad[:, 1] = np.minimum(rad[:, 1], C[:, 1])
        rad[:, 1] = np.maximum(rad[:, 1], 1 - C[:, 1])
        rad = np.minimum(rad, D - np.eye(n))
        valid = rad >= 0
        if not valid.all():
            return -np.inf
        return -np.sum(rad, axis=1).sum()
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, options={'maxiter': 10000})
    if res.success:
        rad = res.x.reshape(n, 2)
        rad[:, 0] = np.minimum(rad[:, 0], C[:, 0])
        rad[:, 0] = np.maximum(rad[:, 0], 1 - C[:, 0])
        rad[:, 1] = np.minimum(rad[:, 1], C[:, 1])
        rad[:, 1] = np.maximum(rad[:, 1], 1 - C[:, 1])
        rad = np.minimum(rad, D - np.eye(n))
        rad[rad < 0] = 0
        return rad, res.fun
    else:
        return np.zeros(n), -np.inf
