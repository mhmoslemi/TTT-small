import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = centers.shape[0]
    
    A = []   # Inequality constraints matrix
    b = []   # Inequality constraints rhs
    
    for i in range(n):
        A.append([-1, 0, 0, 0, 0, 0])     # Ensure r_i <= x_i
        b.append(centers[i, 0])
        
        A.append([0, -1, 0, 0, 0, 0])     # Ensure r_i <= 1 - x_i
        b.append(1 - centers[i, 0])
        
        A.append([0, 0, -1, 0, 0, 0])     # Ensure r_i <= y_i
        b.append(centers[i, 1])
        
        A.append([0, 0, 0, -1, 0, 0])     # Ensure r_i <= 1 - y_i
        b.append(1 - centers[i, 1])
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A.append([0, 0, 0, 0, -1, 0])  # Ensure r_i + r_j <= d_ij
            b.append(d)
            
            A.append([0, 0, 0, 0, 0, -1])  # Ensure r_i + r_j <= d_ij
            b.append(d)

            A.append([0, 0, 0, 0, 1, 0])   # Ensure r_i + r_j <= d_ij
            b.append(d)
            
            A.append([0, 0, 0, 0, 0, 1])   # Ensure r_i + r_j <= d_ij
            b.append(d)

    A = np.array(A)
    b = np.array(b)
        
    c = -np.ones(n)  # Maximize sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        
    assert res.success, "LP did not converge"
        
    radii = res.x
    sum_radii = -res.fun  # Negate since linprog maximizes c^T x by default

    return radii, sum_radii
