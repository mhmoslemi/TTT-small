from scipy.optimize import minimize

def objective(x):
    r = np.sqrt(x[1::2])  # Radii
    centers = x[0::2].reshape(-1, 2)  # Centers
    return -np.sum(r)  # Negative of sum of radii to maximize

def constraint(x):
    r = np.sqrt(x[1::2])  # Radii
    centers = x[0::2].reshape(-1, 2)  # Centers
    sum_r = np.sum(r)
    if sum_r > 2.636:
        return sum_r - 2.636
    else:
        return 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(n * 3)
    bounds = [(0, 1) for _ in range(2 * n)]
    for i in range(n):
        bounds.append((0, None))  # Radius bounds

    cons = {
        'type': 'eq',
        'fun': constraint
    }

    result = minimize(objective, x0, bounds=bounds, Constraints=cons, method='SLSQP')
    
    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0

    r = np.sqrt(result.x[1::2])  # Radii
    centers = result.x[0::2].reshape(-1, 2)  # Centers
    sum_radii = np.sum(r)

    validate_result, _ = validate_packing(centers, r)
    if not validate_result:
        return np.zeros((n, 2)), np.zeros(n), 0
