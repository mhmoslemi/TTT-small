import numpy as np
from scipy.optimize import linprog, minimize

def objective(c):
    n = len(c) // 2
    c = c.reshape((n, 2))
    r = solve_lp(c)
    return -np.sum(r)

def gradient(c):
    n = len(c) // 2
    c = c.reshape((n, 2))
    r, dual = solve_lp(c, return_dual=True)
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for i in range(n):
        grad_x[i] = dual[i]
        grad_y[i] = dual[i]
        for j in range(n):
            if i != j:
                dx, dy = c[i] - c[j]
                d = np.sqrt(dx ** 2 + dy ** 2)
                u, v = dx / d, dy / d
                grad_x[i] += dual[n + i * n + j] * u
                grad_y[i] += dual[n + i * n + j] * v
                grad_x[j] -= dual[n + i * n + j] * u
                grad_y[j] -= dual[n + i * n + j] * v
    return -np.hstack((grad_x, grad_y))

def solve_lp(c, return_dual=False):
    n = len(c) // 2
    A_ub = np.zeros((2*n*n, n*2))
    b_ub = np.zeros(2*n*n)
    
    # Boundary constraints
    for k in range(n):
        A_ub[k, 2*k] = 1
        A_ub[k, 2*k + 1] = 0
        b_ub[k] = c[k, 0]
        A_ub[n + k, 2*k] = 0
        A_ub[n + k, 2*k + 1] = 1
        b_ub[n + k] = c[k, 1]
        
    # Pair constraints
    count = 2*n
    for i in range(n):
        for j in range(i+1, n):
            dx, dy = c[i] - c[j]
            d = np.sqrt(dx ** 2 + dy ** 2)
            A_ub[count, 2*i] = 1
            A_ub[count, 2*i + 1] = 0
            A_ub[count, 2*j] = -1
            A_ub[count, 2*j + 1] = 0
            b_ub[count] = d
            count += 1
    
    res = linprog(np.ones(2*n), A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(0, 1)] * (2*n))
    if res.success and res.dual is not None:
        r = res.x
        dual = res.dual
    else:
        r = np.ones(n) * 1e-6  # Default small radius
        dual = np.zeros(2*n*n)
        
    if return_dual:
        return r, dual
    else:
        return r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([
        0.085, 0.165, 0.255, 0.345, 0.435, 0.525, 0.615, 0.705, 0.795, 0.885,
        0.085, 0.165, 0.255, 0.345, 0.435, 0.525, 0.615, 0.705, 0.795, 0.885,
        0.085, 0.165, 0.255, 0.345, 0.435, 0.525, 0.615, 0.705, 0.795, 0.885
    ]).reshape((26, 2))
    res = minimize(objective, centers, method='L-BFGS-B', jac=gradient, bounds=[(0, 1)] * 52)
    if not res.success:
        res = minimize(objective, centers + np.random.rand(26, 2) * 0.1, method='L-BFGS-B', jac=gradient, bounds=[(0, 1)] * 52)
    radii = solve_lp(res.x)
    sum_radii = np.sum(radii)
    return res.x, radii, sum_radii
