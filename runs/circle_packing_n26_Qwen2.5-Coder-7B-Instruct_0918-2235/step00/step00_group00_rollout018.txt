import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Parameters
    n_circles = 26
    step_start, step_end = 0.15, 0.001
    temp_start, temp_end = 1.0, 1e-4
    iterations = 20000

    # Initialize
    centers = np.random.rand(n_circles, 2)
    radii = np.zeros(n_circles)
    
    def max_radius(i):
        x, y = centers[i]
        return min(x, 1-x, y, 1-y, min(np.linalg.norm(centers[i] - centers[j]) - radii[j] for j in range(n_circles) if j != i))
    
    def inflation_step-update(radii):
        r_new = np.zeros_like(radii)
        for i in range(n_circles):
            r_new[i] = max_radius(i)
        return np.clip(r_new, 0, None)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = 0
    
    for s, t in zip(np.geomspace(step_start, step_end, iterations), np.geomspace(temp_start, temp_end, iterations)):
        i = np.random.randint(n_circles)
        delta = (np.random.rand(2) - 0.5) * s
        centers[i] = np.clip(centers[i] + delta, 0, 1)
        radii = inflation_step(centers, radii)
        new_sum = np.sum(radii)
        
        if new_sum > np.sum(best_radii) or (np.random.rand() < np.exp((new_sum - np.sum(best_radii)) / t)):
            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = new_sum
    
    return best_centers, best_radii, np.sum(best_radii)
