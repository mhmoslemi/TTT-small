import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform

def objective(v, n):
    x = v[:n].reshape(-1, 2)
    r = v[n:].reshape(-1, 1)
    dists = squareform(pdist(x))
    overlap = np.sum(np.maximum(0, r + r.T - dists) ** 2)
    boundary = np.sum(np.maximum(0, r - x[:, 0]) ** 2) + \
               np.sum(np.maximum(0, r - (1 - x[:, 0])) ** 2) + \
               np.sum(np.maximum(0, r - x[:, 1]) ** 2) + \
               np.sum(np.maximum(0, r - (1 - x[:, 1])) ** 2)
    return -np.sum(r) + 1000 * (overlap + boundary)

def backpack(v, n, sol):
    x = v[:n].reshape(-1, 2)
    r = v[n:].reshape(-1, 1)
    sum_r = np.sum(r)
    dists = squareform(pdist(x))
    overlap = np.sum(np.maximum(0, r + r.T - dists) ** 2)
    boundary = np.sum(np.maximum(0, r - x[:, 0]) ** 2) + \
               np.sum(np.maximum(0, r - (1 - x[:, 0])) ** 2) + \
               np.sum(np.maximum(0, r - x[:, 1]) ** 2) + \
               np.sum(np.maximum(0, r - (1 - x[:, 1])) ** 2)
    return sum_r if overlap == 0 and boundary == 0 else -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    best = None
    best_sol = None
    for seed in range(10):
        np.random.seed(seed)
        res = differential_evolution(objective, bounds, args=(n,), popsize=10, max generation=50, tol=1e-12,
                                  callback=lambda x, minimize_results: backpack(x, n, minimize_results),
                                  disp=False)
        if best is None or res.fun > best:
            best = res.fun
            best_sol = (res.x[:n].reshape(-1, 2), res.x[n:].reshape(-1, 1))
    
    centers, radii = best_sol
    sum_radii = -best
    return centers, radii.flatten(), sum_radii
