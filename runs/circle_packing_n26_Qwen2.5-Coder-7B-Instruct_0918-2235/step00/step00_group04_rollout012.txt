def dual_to_gradient(dual, pos, coords):

    for i in range(n_circles):

                if i < j:
                    dS_dx[i * 2 + dim] += dual[mapping[1][i + j * (n_circles - i - 1) - i * (i + 1) // 2 - 1]] * (pos[i, dim] - pos[j, dim]) / distance(pos[i], pos[j])
                    dS_dx[j * 2 + dim] -= dual[mapping[1][i + j * (n_circles - i - 1) - i * (i + 1) // 2 - 1]] * (pos[i, dim] - pos[j, dim]) / distance(pos[i], pos[j])
                else:
                    dS_dx[i * 2 + dim] += dual[mapping[1][j + i * (n_circles - j - 1) - j * (j + 1) // 2 - 1]] * (pos[i, dim] - pos[j, dim]) / distance(pos[i], pos[j])
