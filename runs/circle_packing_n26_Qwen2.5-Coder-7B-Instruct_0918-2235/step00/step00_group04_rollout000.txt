import numpy as np
from scipy.optimize import linprog

def solve_LP(centers):
    n = len(centers)
    c = -np.ones(n)  # Maximize sum of radii

    # Build A_ub and b_ub for linear constraints
    A_ub = np.empty((4 * n + n * (n - 1) // 2, n))
    b_ub = np.empty(4 * n + n * (n - 1) // 2)

    count = 0
    for i in range(n):
        A_ub[count, i] = 1
        b_ub[count] = centers[i, 0]
        count += 1
        A_ub[count, i] = 1
        b_ub[count] = 1 - centers[i, 0]
        count += 1
        A_ub[count, i] = 1
        b_ub[count] = centers[i, 1]
        count += 1
        A_ub[count, i] = 1
        b_ub[count] = 1 - centers[i, 1]
        count += 1

    # Add pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[count, i] = 1
            A_ub[count, j] = 1
            b_ub[count] = dist
            count += 1

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(0, None)] * n)
    if not res.success:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='simplex')

    if not res.success:
        # Fallback: use a smaller step and recompute
        step_size = 0.05 * step_size
        random_move(centers, step_size)
        return solve_LP(centers, step_size)

    return res.x, -res.fun

def random_move(centers, step_size):
    n = len(centers)
    i = np.random.randint(n)
    delta = 2 * step_size * (np.random.rand(2) - 0.5)
    centers[i] += delta
    centers[i, centers[i] < 0] = 0
    centers[i, centers[i] > 1] = 1

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    random.seed(0)
    n = 26
    step_step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    ε = 1e-12

    centers = np.zeros((n, 2))
    grid = np.arange(0, 1.1, 0.1)
    grid = np.delete(grid, [2, 5, 7])  # Remove two points to get 26 centers

    for i in range(n):
        row = np.random.choice(grid)
        col = np.random.choice(grid)
        while np.any(np.sqrt((centers[:i, 0] - col) ** 2 + (centers[:i, 1] - row) ** 2) < 2 * step_size):
            row = np.random.choice(grid)
            col = np.random.choice(grid)
        centers[i] = [row, col]

    current_r, current_sum = solve_LP(centers)
    best_r = current_r.copy()
    best_sum = current_sum

    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        random_move(centers, step_size)

        new_r, new_sum = solve_LP(centers)
        if new_sum > current_sum + ε:
            current_r = new_r
            current_sum = new_sum
            if new_sum > best_sum:
                best_r = current_r.copy()
                best_sum = new_sum
        else:
            # Optionally adjust step size if not improving
            if iter % N_REDUCE == 0:
                if step_step > 1e-4:
                    step_step *= 0.9

        if iter % K_JUMP == 0:
            # Randomly repopulate
            generate_new_centers(centers, step_step)
            new_r, new_sum = solve_LP(centers)
            if new_sum > best_sum:
                best_r = new_r.copy()
                best_sum = new_sum
            step_size = step_step

    return centers, best_r, best_sum
