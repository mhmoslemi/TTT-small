import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize the centers and radii of the circles
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.1  # Start with small radii

    # Sum of radii
    sum_radii = radii.sum()

    # Validation function
    def validate(centers, radii):
        for i in range(26):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Ensure non-overlap and boundary constraints
    while not validate(centers, radii)[0]:
        centers = np.random.rand(26, 2)
        radii = np.random.rand(26) * 0.1  # Reset radii

    # Validate the final packing
    assert validate(centers, radii)[0]

    return centers, radii, sum_radii
