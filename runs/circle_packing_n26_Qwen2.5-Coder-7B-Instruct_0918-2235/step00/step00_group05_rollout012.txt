def generate_hexagon_lattice(n1, n2, s):
    b1 = np.array([s, 0])
    b2 = np.array([s / 2, np.sqrt(3) * s / 2])
    lattice_points = []
    for i in range(n1):
        for j in range(n2):
            lattice_points.append(np.dot(b1, i) + np.dot(b2, j))
    return lattice_points
