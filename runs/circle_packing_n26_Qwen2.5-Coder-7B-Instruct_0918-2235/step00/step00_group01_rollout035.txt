import numpy as np
from scipy.optimize import minimize

def penalty(v, λ):
    n = len(v) // 3
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    pen = np.sum(np.maximum(0, centers[:, 0] - radii))
    pen += np.sum(np.maximum(0, 1 - centers[:, 0] - radii))
    pen += np.sum(np.maximum(0, centers[:, 1] - radii))
    pen += np.sum(np.maximum(0, 1 - centers[:, 1] - radii))
    pen += np.sum(np.maximum(0, centers - centers[:, np.newaxis] - radii[:, np.newaxis][:, np.newaxis]))
    pen -= λ * pen
    return pen

def objective(v, λ):
    n = len(v) // 3
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    return -np.sum(radii) + λ * penalty(v, λ)
