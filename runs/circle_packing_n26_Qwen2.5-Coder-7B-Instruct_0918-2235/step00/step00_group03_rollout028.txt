import numpy as np
from scipy.optimize import linprog
from scipy.spatial import distance_matrix

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_radius = 0.5
    initial_positions = np.random.rand(n, 2)
    initial_radii = np.zeros(n)
    step_size = 0.1
    alpha = 0.85
    tolerance = 1e-9
    best_sum = -np.inf
    best_positions = None
    best_radii = None

    for trial in range(40):
        centres = initial_positions.copy()
        radii = initial_radii.copy()
        step_size = 0.1
        improving_cycles = 0

        while step_size > 1e-4 and improving_cycles < 5:
            for i in np.random.permutation(n):
                centre = centres[i]
                delta = np.random.uniform(-step_size, step_size, 2)
                centre = np.clip(centre + delta, 0, 1)
                centre = centre.reshape(2, 1)

                A1 = np.c_[np.ones(n)[:, None], -np.eye(n)]
                A2 = np.c_[np.zeros(n)[:, None], np.eye(n)]
                A3 = np.c_[-np.ones(n)[:, None], np.zeros(n)[:, None]]

                A_ub = np.vstack([A1, A2, A3])
                b_ub = np.hstack([np.inf * np.ones(n), radii, centres[:, i]])

                # Compute pairwise distances
                dists = distance_matrix(centres, centres)
                np.fill_diagonal(dists, np.inf)
                b_ub = np.where(dists < np.inf, dists - radii, b_ub)

                c = -np.ones(n)
                bounds = [(0, None)] * n

                res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

                if res.success:
                    radii = np.maximum(res.x, 0)
                    for j in range(n):
                        if radii[j] > max_radius:
                            radii[j] = max_radius

                    new_sum = np.sum(radii)
                    if new_sum > best_sum + tolerance:
                        best_sum = new_sum
                        best_positions = centres.copy()
                        best_radii = radii.copy()
                        improving_cycles += 1
                    elif step_size > 1e-5:
                        improving_cycles = 0

            step_size *= alpha

    return best_positions, best_radii, best_sum
