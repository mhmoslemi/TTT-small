import numpy as np
from scipy.spatial import Delaunay

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Define the initial centers as a hexagonal lattice
    side_length = np.sqrt(3) / 3
    num_rows = int(np.ceil(1 / side_length)) + 1
    num_cols = int(np.ceil(1 / (2 * side_length))) + 1
    centers = []
    
    for row in range(num_rows):
        for col in range(num_cols):
            if (row + col) % 2 == 0:
                x = col * side_length
                y = row * side_length + (col % 2) * side_length * 0.5
            else:
                x = col * side_length + side_length / 2
                y = row * side_length
            if x <= 1 and y <= 1:
                centers.append([x, y])
    
    centers = np.array(centers)
    
    # Step 2: Pack circles using a Delaunay triangulation
    triangulation = Delaunay(centers)
    centers = triangulation.points
    
    # Step 3: Calculate radii
    radii = np.zeros(len(centers))
    target_sum_radii = 2.636
    
    # Step 4: Iterate to optimize radii
    for iteration in range(100):
        for i in range(len(centers)):
            radius = np.sqrt((centers[i][0] - 0.5) ** 2 + (centers[i][1] - 0.5) ** 2)
            if radius > 0.5:
                radius = 0.5
            else:
                radius = 0.5 - radius
            radii[i] = radius
        
        if np.sum(radii) > target_sum_radii:
            target_sum_radii = np.sum(radii)
        
        if np.sum(radii) < target_sum_radii:
            target_sum_radii = np.sum(radii)
    
    return centers, radii, np.sum(radii)
