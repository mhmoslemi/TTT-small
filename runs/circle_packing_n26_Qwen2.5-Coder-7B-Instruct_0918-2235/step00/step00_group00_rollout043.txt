import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial import distance_matrix

def objective(v, lambda_penalty):
    n = len(v) // 3
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    penalty = 0
    for i in range(n):
        for j in range(i+1, n):
            dij = distance_matrix([centers[i]], [centers[j]])[0][0]
            penalty += np.maximum(0, radii[i] + radii[j] - dij)**2
        penalty += np.maximum(0, radii[i] - centers[i][0])**2
        penalty += np.maximum(0, radii[i] - (1 - centers[i][0]))**2
        penalty += np.maximum(0, radii[i] - centers[i][1])**2
        penalty += np.maximum(0, radii[i] - (1 - centers[i][1]))**2
    return -np.sum(radii) + lambda_penalty * penalty

def run_packing():
    np.random.seed(0)
    n = 26
    initial_state = np.random.rand(52)
    initial_state[:2*n] = np.clip(0.5 * initial_state[:2*n], 0, 1)
    initial_state[2*n:] = 0.25  # Starting point far from optimal, collision avoidance
    lambda_penalty = 1000

    best_r_sum = 0
    best_centers = None
    best_radii = None

    for _ in range(15):
        bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n
        res = differential_evolution(objective, bounds, args=(lambda_penalty,), popsize=10, maxiter=50, disp=False)
        if res.success and res.fun > best_r_sum:
            best_r_sum = -res.fun
            best_centers = res.x[:2*n].reshape(n, 2)
            best_radii = res.x[2*n:]

    # Final optimization to get exact radii
    def lp_objective(x):
        return -np.sum(x)
    lp_bounds = [(0, None)] * n
    lp_res = minimize(lp_objective, best_radii, method='SLSQP', bounds=lp_bounds, constraints={'type': 'ineq',
                                                                                          'fun': lambda x: best_r_sum + objective(np.concatenate([best_centers.flatten(), x]), lambda_penalty)[1]})
    best_radii = lp_res.x

    return best_centers, best_radii, np.sum(best_radii)
