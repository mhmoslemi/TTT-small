import numpy as np
from scipy.optimize import minimize

def build_variable_vector(centres, radii):
    return np.concatenate((centres[:, 0], centres[:, 1], radii))

def unpack(z):
    x = z[:52].reshape(26, 2)[:, 0]
    y = z[:52].reshape(26, 2)[:, 1]
    r = z[52:]
    return x, y, r

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
   _nb = compute_boundary_penalty(x, y, r, lambda_B)
    no = compute_overlap_penalty(x, y, r, lambda_O)
    return -np.sum(r) + _nb + _no

def compute_boundary_penalty(x, y, r, lambda_B):
    return lambda_B * np.sum(
        np.maximum(0, r - x) ** 2 +
        np.maximum(0, r - (1 - x)) ** 2 +
        np.maximum(0, r - y) ** 2 +
        np.maximum(0, r - (1 - y)) ** 2
    )

def compute_overlap_penalty(x, y, r, lambda_O):
    n = len(r)
    no = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(x[i] - x[j] + (y[i] - y[j]) * 1j)
            no += max(0, r[i] + r[j] - d_ij) ** 2
    return lambda_O * no

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)

    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, 1e6, 1e6)

    def jacobian(z):
        x, y, r = unpack(z)
        nb = compute_boundary_penalty_gradient(x, y, r, 1e6)
        no = compute_overlap_penalty_gradient(x, y, r, 1e6)
        return np.concatenate((nb[0][:, 0], nb[0][:, 1], nb[1], no[0][:, 0], no[0][:, 1], no[1]))

    def compute_boundary_penalty_gradient(x, y, r, lambda_B):
        nb = np.zeros((2*n, 3))
        for i in range(26):
            nb[i * 3, 0] = -2 * lambda_B * max(0, r[i] - x[i]) * (1 if r[i] > x[i] else 0)
            nb[i * 3 + 1, 1] = -2 * lambda_B * max(0, r[i] - (1 - x[i])) * (1 if r[i] > (1 - x[i]) else 0)
            nb[i * 3 + 2, 2] = -2 * lambda_B * max(0, r[i] - y[i]) * (1 if r[i] > y[i] else 0)
        return nb, np.zeros(52)

    def compute_overlap_penalty_gradient(x, y, r, lambda_O):
        no = np.zeros((2*n, 3))
        for i in range(26):
            for j in range(i + 1, 26):
                d_ij = np.linalg.norm(x[i] - x[j] + (y[i] - y[j]) * 1j)
                if d_ij > 0:
                    no[i * 3, 0] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
                    no[j * 3, 0] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
                    no[i * 3 + 1, 1] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
                    no[j * 3 + 1, 1] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
                    no[i * 3 + 2, 2] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
                    no[j * 3 + 2, 2] += -2 * lambda_O * max(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij * (1 if r[i] + r[j] > d_ij else 0)
        return no, np.zeros(52)

    bounds = [(0, 1)] * 52 + [(0, None)] * 26

    res = minimize(objective, z, jac=jacobian,
                   method='trust-constr', bounds=bounds,
                   options={'gtol': 1e-6, 'ftol': 1e-8, 'maxiter': 3000, 'verbose': 0})

    centres_opt, radii_opt = unpack(res.x)
    return centres_opt, radii_opt, np.sum(radii_opt)

def init_hex_grid():
    radius = 0.01
