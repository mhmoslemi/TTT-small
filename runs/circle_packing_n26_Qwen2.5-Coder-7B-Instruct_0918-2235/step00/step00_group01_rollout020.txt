def compute_minimum_clearances(centers, radii):
    min_radii = np.ones_like(radii) * 1e-10
    for i in range(26):
        x_i, y_i = centers[i]
        r_i = radii[i]
        distances = np.hypot(x_i - centers[:, 0], y_i - centers[:, 1])
        clearances = np.maximum(x_i - r_i, 1 - x_i - r_i, y_i - r_i, 1 - y_i - r_i, distances - (radii + radii) - 1e-12)
        min_radii[i] = np.min(clearances)
    return min_radii
