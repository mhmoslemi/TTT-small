import numpy as np
from scipy.optimize import linprog, minimize

def lp_dual_to_gradient(radii, lams, dists, i, j):
    dual_active = i < j
    coef = lams[j] if dual_active else lams[i]
    xi, yi, xj, yj = dists[i, j]
    normalized_vec_diff = [(xj - xi) / dists[i, j],
                            (yj - yi) / dists[i, j]]
    grad_x = coef * normalized_vec_diff[0]
    grad_y = coef * normalized_vec_diff[1]
    return grad_x, grad_y

def objective_and_gradient(cents):
    n = len(cents) // 2
    cents = cents.reshape((-1, 2))
    xs, ys = cents[:, 0], cents[:, 1]

    # linear programming constraints	
    # lhs of constraints: x_i <= 1, 1 - x_i <= 1, y_i <= 1, count
    # rhs vectors: np.ones(2n), 1 - np.ones(2n), np.ones(2n)
    
    A_ub = np.zeros((4*n, 2*n))
    bounds = [(0, None), ] * 2*n
    
    for i in range(n):
        A_ub[4*i] = xs - 1
        A_ub[4*i+1] = 1 - xs
        A_ub[4*i+2] = ys - 1
        A_ub[4*i+3] = 1 - ys
        bounds.extend([(0, None), ] * 2)
    
    # Distances squared matrix
    dists_squared = (xs[:, np.newaxis] - xs)**2 + (ys[:, np.newaxis] - ys)**2
    dists = np.sqrt(dists_squared + 1e-12)
    
    eq = np.zeros((n*(n-1)//2, 2*n))
    h = 0.5  # diagonal matrix with 0.5s
    for i in range(n):
        for j in range(i+1, n):
            idx = i * n + j
            delta_x = h[i, j]
            delta_y = h[i, j]
            
            A_ub[4*n+idx] = 2 * np.array([delta_x, delta_y])
            bounds.extend([(0, None), ] * 2)
            eq[idx, 2*i:2*i+2] = [2 * delta_x, 2 * delta_y]
            eq[idx, 2*j:2*j+2] = [-2 * delta_x, -2 * delta_y]
    
    A_eq = eq
    b_eq = np.concatenate((np.ones(n), np.ones(n*(n-1)//2)))
    lams = linprog([
       np.ones(2*n), 
        -np.ones(2*n), 
        -np.ones(n), 
        -np.ones(n)
    ], 
    A_ub=A_ub, 
    b_ub=b_eq, 
    A_eq=A_eq, 
    b_eq=b_eq, 
    method='highs').dual
    
    if lams is None:
        lams = np.zeros(A_ub.shape[0])
    
    radii = linprog(
        np.ones(2*n),
        A_ub=A_ub,
        b_ub=b_eq,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=(0, None),
        method='highs'
    ).x
    
    lams = linprog(np.ones(2*n), A_ub=A_ub, b_ub=b_eq, A_eq=A_eq, b_eq=b_eq, method='highs').dual
    
    if lams is None:
        lams = np.zeros(A_ub.shape[0])
    
    sum_radii = np.sum(radii)
    n, y, z = np.indices((n, n, 2))
    dists = np.sqrt(((xs[:, np.newaxis] - xs[y, z])**2 
                    + (ys[:, np.newaxis] - ys[y, z])**2 
                    + 1e-12))
    dists_norm = dists / dists
    grad_cents = np.zeros((2*n, 2*n))
    
    for i in range(n):
        for j in range(i, n):
            lams_ij = lams[j * n + i] if i < j else lams[i * n + j]
            if lams_ij > 0:
                if i == j:
                    grad_cents[2*i:2*i+2, 2*i:2*i+2] += lams_ij * np.array([[y[i+1]-y[i], z[i+1]-z[i]],
                                                                  [z[i]-z[i+1], -y[i]-y[i+1]]]) / dists[i, j]
                else:
                    grad_cents[2*i:2*i+2, 2*j:2*j+2] += lams_ij * np.array([[y[j]-y[i], z[j]-z[i]],
                                                                  [z[j]-z[i], y[j]-y[i]]]) / dists[i, j]
                    grad_cents[2*j:2*j+2, 2*i:2*i+2] -= lams_ij * np.array([[y[j]-y[i], z[j]-z[i]],
                                                                  [z[j]-z[i], y[j]-y[i]]]) / dists[i, j]
    
    return -sum_radii, grad_cents.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    mins = np.zeros((n,), dtype=int)
    maxs = np.zeros((n,), dtype=int)
    for i in range(n):
        mins[i] = (i//5) * 5
        maxs[i] = 5 * ((i//5) + 1)
    
    # generate random grid placement
    xs = np.random.randint(mins[::10], maxs[::10], size=n)
    ys = np.random.randint(mins[1::10], maxs[1::10], size=n)
    
    # create initial arrangement
    centres = np.zeros((n*2,))
    centres[::2] = xs
    centres[1::2] = ys
    
    # run L-BFGS-B optimization
    result = minimize(objective_and_gradient, centres, bounds=[(0, 1)] * len(centres), jac=True, method='L-BFGS-B')
    centre_coords, opt_radii = result.x.reshape(-1, 2), np.exp(result.fun)
    
    radii = np.clip(opt_radii, 0, None)
    sum_radii = np.sum(radii)
    
    return centre_coords, radii, sum_radii
