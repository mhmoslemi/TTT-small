import numpy as np
import random

def radius_inflation(circles, radii, epsilon=1e-8, max_iter=50):
    n = circles.shape[0]
    for _ in range(max_iter):
        biggest_change = 0
        for i in range(n):
            r_new = min(circles[i, 0], 1 - circles[i, 0], circles[i, 1], 1 - circles[i, 1])
            r_min = float('inf')
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(circles[i] - circles[j])
                    r_min = min(r_min, dist - radii[j])
            r_new = min(r_new, r_min)
            r_new = max(r_new, 0)
            biggest_change = max(biggest_change, abs(radii[i] - r_new))
            radii[i] = r_new
        if biggest_change < epsilon:
            break
    return radii

def simulated_annealing(circles, step_sizes, temperatures, max_iter=20000):
    n = circles.shape[0]
    radii = np.zeros(n)
    best_sum = -np.inf
    best_circles = circles.copy()
    for k in range(max_iter):
        i = random.randint(0, n - 1)
        orig_x, orig_y = circles[i, :]
        circle_delta = np.array([np.random.uniform(-step_sizes[k], step_sizes[k]) for _ in range(2)])
        circles[i, :2] = np.clip(orig_x + circle_delta[0], 0, 1), np.clip(orig_y + circle_delta[1], 0, 1)
        updated_radii = radius_inflation(circles, radii)
        new_sum = np.sum(updated_radii)
        if new_sum > best_sum:
            best_sum = new_sum
            best_circles = circles.copy()
        else:
            accept = np.random.rand() < np.exp((new_sum - best_sum) / temperatures[k])
            if accept:
                best_sum = new_sum
                best_circles = circles.copy()
            else:
                circles[i, :2] = orig_x, orig_y
        circles = best_circles.copy()
    return circles, radius_inflation(circles, radii), best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    circles = np.array([[random.uniform(0, 1) for _ in range(2)] for _ in range(26)])
    step_sizes = np.linspace(0.15, 0.001, 20000)
    temperatures = np.linspace(1.0, 1e-4, 20000)
    final_circles, final_radii, final_sum = simulated_annealing(circles, step_sizes, temperatures)
    return final_circles, final_radii, final_sum
