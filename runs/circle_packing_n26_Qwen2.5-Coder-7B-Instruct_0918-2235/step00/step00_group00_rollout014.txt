import numpy as np
import scipy.optimize as opt

def compute_r_max(candidates, placed_centres, placed_radii):
    dists_to_existing = np.linalg.norm(candidates[:, np.newaxis, :] - placed_centres[np.newaxis, :, :], axis=2)
    radii_max = np.min(dists_to_existing - placed_radii[:, np.newaxis], axis=1)
    radii_max = np.minimum(radii_max, candidates[:, 0] - 0.000001)
    radii_max = np.minimum(radii_max, 1.0 - candidates[:, 0] - 0.000001)
    radii_max = np.minimum(radii_max, candidates[:, 1] - 0.000001)
    radii_max = np.minimum(radii_max, 1.0 - candidates[:, 1] - 0.000001)
    return np.maximum(radii_max, 0.0)

def lp_optimize_radii(centers, radii):
    num_circles = centers.shape[0]
    objective = np.ones(num_circles)
    constraints = []
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + placed_radii[i] - 1.0})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[j]})
    solution = opt.linprog(objective, A_ub=np.vstack(constraints), bounds=(0.0, None))
    return solution.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_size = 40
    cell_size = 1.0 / grid_size
    epsilon = 1e-6
    candidates = np.array([[x + (i + 0.5) * cell_size, y + (j + 0.5) * cell_size]
                           for i in range(grid_size) for j in range(grid_size)
                           if x + cell_size < 1 and y + cell_size < 1]
                          )
    candidates -= 0.5 * cell_size
    placed_centres = []
    placed_radii = []
    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        too_close = np.linalg.norm(candidates[np.newaxis, :] - candidates[best], axis=2) < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close.reshape(-1)]
    total_rad = sum(placed_radii)
    if total_rad == 0:
        return (np.zeros((26, 2)), np.zeros((26,)), 0.0)
    refined_centers = np.array(placed_centres)
    current_rad = np.array(placed_radii)
    for _ in range(5000):
        move = np.random.uniform(-0.02, 0.02, size=(26, 2))
        new_centers = refined_centers + move
        scales = lp_optimize_radii(new_centers, current_rad)
        if np.sum(current_rad) + np.sum(scales) > total_rad:
            refined_centers = new_centers
            current_rad = scales
    return (refined_centers, current_rad, np.sum(current_rad))
