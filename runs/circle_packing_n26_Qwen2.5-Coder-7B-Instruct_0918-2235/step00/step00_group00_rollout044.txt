import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    grid = np.mgrid[0:h:21j, 0:h:21j].reshape(2, -1).T
    candidates = grid[np.all(grid > epsilon, axis=1) & np.all(grid < 1 - epsilon, axis=1)]
    
    placed_centres = []
    placed_radii = []

    def compute_r_max(candidates, placed_centres, placed_radii):
        C = np.array(placed_centres)
        R = np.array(placed_radii)
        r_max = np.minimum.reduce([np.maximum(0, (candidates[:, None, :] - C) ** 2).sum(axis=2) - R[:, None] + epsilon**2, candidates[:, 0], 1 - candidates[:, 0], candidates[:, 1], 1 - candidates[:, 1]], axis=0).T
        return r_max

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    circles = np.array(placed_centres)
    radii = np.array(placed_radii)

    def opt_radii_fixed_circles(circles):
        n = circles.shape[0]
        R = cp.Variable(n)
        constraints = [R <= circles[:, 0],
                        R <= 1 - circles[:, 0],
                        R <= circles[:, 1],
                        R <= 1 - circles[:, 1]]
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(R[i] + R[j] <= np.linalg.norm(circles[i] - circles[j]))
        constraints.append(R >= 0)
        objective = cp.Maximize(R.sum())
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-9)
        return R.value

    radii = opt_radii_fixed_circles(circles)

    for i in range(5000):
        alpha = max(0.02 - i * 0.00004, 0.0005)
        for _ in range(20):
            idx = np.random.randint(circles.shape[0])
            delta = np.random.uniform(-alpha, alpha, size=2)
            new_circle = circles[idx] + delta

                new_radii = opt_radii_fixed_circles(np.vstack([circles[:idx], new_circle, circles[idx+1:]]))
                if new_radii.sum() > radii.sum():
                    circles[idx] = new_circle
                    radii = new_radii

    return circles, radii, radii.sum()
