import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C)
    boundary = np.array([C[:, 0], C[:, 0] - 1, C[:, 0], C[:, 1],
                          C[:, 1], C[:, 1] - 1, C[:, 1]]).T
    boundary = boundary[:, np.any(boundary >= 0, axis=1)]
    rows = np.tile(np.arange(N), N * 2)
    cols = np.repeat(np.arange(N), N * 2)
    data = np.concatenate([np.ones(N * 2), boundary[:, 1]])
    A = np.vstack([d[rows != cols], boundary[rows == cols], 
                   boundary[cols > rows], boundary[cols < rows]])
    A[:, 0] *= -1
    b = np.hstack([-np.zeros(N * 2), boundary[:, 2], boundary[:, 2], 
                   boundary[:, 2]])
    return A, b

def compute_sum_radius(C, r):
    return -np.sum(r)

def de_obj(C):
    try:
        A, b = build_lp(C)
        res = linprog(np.ones_like(C).flatten(), A_ub=A, b_ub=b, method='highs')
        if not res.success:
            return 1e6
        r = res.x.reshape(C.shape[0])
        r = np.maximum(r, 0)
        return compute_sum_radius(C, r)
    except Exception:
        return 1e6

def run_packing():
    N = 26
    C = 0.5 * np.ones((N, 2))
    popsize = 15 * N
    best_C = None
    best_f = -1e6
    maxfev = 3000
    tol = 1e-7
    strategy = 'best1bin'
    mutation = 0.9
    recombination = 0.7

    for _ in range(maxfev):
        population = C + 0.02 * np.random.randn(popsize, 2)
        population = np.clip(population, 0, 1)
        population = np.vstack([population, np.random.uniform(0, 1, (10, 2))])
        population = np.clip(population, 0, 1)

        fitness = np.array([de_obj(C) for C in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] > best_f:
            best_f = fitness[best_idx]
            best_C = population[best_idx]

        if best_f > 2.636:
            break

    if best_C is not None:
        A, b = build_lp(best_C)
        res = linprog(np.ones_like(best_C).flatten(), A_ub=A, b_ub=b, method='highs')
        if res.success:
            r = res.x.reshape(N)
            r = np.maximum(r, 0)
            centers = best_C
            radii = r
            sum_radii = np.sum(radii)
        else:
            return np.zeros((N, 2)), np.zeros(N), 0
    else:
        # Fallback to baseline packing
        centers = np.array([[[0.1, 0.1]],
