import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C, 'euclidean')
    A = np.vstack((np.eye(N), -np.eye(N), 
                  np.ones((N, 2)), -np.ones((N, 2)), 
                  np.ones((N, 2)), -np.ones((N, 2)), 
                  np.ones((N, 2)), -np.ones((N, 2)), 
                  d))
    b = np.hstack((C.max(axis=1) - C.min(axis=1), 
                   np.zeros(N), C[:, 0], 1 - C[:, 0], 
                   C[:, 1], 1 - C[:, 1], C[:, 0], 1 - C[:, 0]))
    return A, b

def de_obj(C):
    A, b = build_lp(C)
    try:
        res = linprog(-np.ones(C.shape[0]), A_ub=A, b_ub=b, method='highs', options={'basics': 'matrix'})
        return -res.fun  # Negative because we minimized the negative sum
    except Exception:
        return 1e6  # Infeasible or numerical issues, return very small objective

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    popsize = 15 * N
    C = np.array([[i/N, j/N] for i in range(N) for j in range(N)]).reshape(N, 2) + 0.02 * np.random.randn(N, 2)
    best_C = None
    maxfev = 3000

   # Initial population with some random perturbations and clamping
    C += np.random.randn(N, 2) * 0.02
    C = np.clip(C, 0, 1)

    # Differential Evolution settings
    bounds = [(0, 1)] * (2 * N)
    result = differential_evolution(de_obj, bounds, popsize=popsize, strategy='best1bin', mutation=0.9, recombination=0.7, maxfev=maxfev)

    if result.success and result.fun > 2.636:
        best_C = result.x.reshape(N, 2)
        A, b = build_lp(best_C)
        res = linprog(np.ones(C.shape[0]), A_ub=A, b_ub=b, method='highs', options={'basics': 'matrix'})
        radii = np.maximum(0, res.x)
        sum_radii = np.sum(radii)
    else:
        # Fallback to a baseline packing
        C = np.array([[i/N, j/N] for i in range(N) for j in range(N)]).reshape(N, 2) + 0.02 * np.random.randn(N, 2)
        C = np.clip(C, 0, 1)
        best_C = C
        radii = 0.1 * np.ones(N)
