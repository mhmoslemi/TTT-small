import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Constraints order: (x - r, 1 - x - r, y - r, 1 - y - r, distance^2 - (r + r')^2)
    def constraints_func(v):
        k = np.array([1., -2., 1.], dtype=v.dtype)  # boundary constraint multiplier
    
        xs = v[::3]
        ys = v[1::3]
        rs = v[2::3]
    
        n = len(rs)
        cons = []
    
        for i in range(n):
            cons.append(xs[i] - rs[i])
            cons.append(1 - xs[i] - rs[i])
            cons.append(ys[i] - rs[i])
            cons.append(1 - ys[i] - rs[i])
            cons.extend((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2 - (rs[i] + rs[j])**2
                        for j in range(i + 1, n))
    
        return np.concatenate(cons)

    def objective(v):
        rs = v[2::3]
        return -np.sum(rs)  # Minimize negative sum, maximizing sum
    
    np.random.seed(42)  # reproducing results
    max_iter = 2000
    ftol = 1e-9
    disp = False
    epsilon = 1e-8

    # Initial feasible guess of small radii
    v0 = np.concatenate((np.random.uniform(low=0.06, high=0.94, size=n), 
                          np.random.uniform(low=epsilon, high=0.5, size=n) * 0.02))
    
    res = minimize(objective, v0,
                     bounds=[0,1]*n + [0,0.5],
                     method=None,  # Let minimize() choose the method
                     jac=None,
                     hess='cs',
                     hessp=None,
                     constraints={'type': 'ineq', 'fun': constraints_func},
                     tol=ftol,
                     options={'disp': disp, 'maxiter': max_iter})

    if not res.success:
        raise Exception('Optimization failed! Solution was not feasible.')

    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    sum_r = np.sum(rs)

    return np.column_stack((xs, ys)), rs, sum_r
