import numpy as np
import scipy.optimize as opt
import cma


def refine_radii(centers):
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        A[i, i] = 1
        b[i] = min(center[i] for center in centers)
    for i in range(n):
        for j in range(i + 1, n):
            A[i, j] = A[j, i] = 1
            b[i] += np.linalg.norm(centers[i] - centers[j])
            b[j] += np.linalg.norm(centers[j] - centers[i])
    return opt.linprog(-np.ones(n), A_ub=A, b_ub=b, A_eq=None, b_eq=None, bounds=(0, 1), method='highs').x


def initial_mean():
    n = 26
    r = 0.1  # Initial radius guess
    a = np.array([0.0, 0.33, 0.67, 1.0])
    b = np.array([r, r, r, r])
    centers = np.zeros((n, 2))
    for i in range(4):
        for j in range(9):
            x = (i * a[j] + (j % 2) * b[j]) * (1 / (2 * b[j]))
            y = (i * b[j] + (j // 2)) * (1 / (2 * b[j]))
            centers[i * 9 + j] = [x, y]
    centers += np.random.uniform(-0.01, 0.01, (n, 2))
    r0 = refine_radii(centers)
    return centers.flatten().tolist() + r0.tolist()


def objective(z):
    centers = np.array(z[:52]).reshape((26, 2))
    radii = refine_radii(centers)
    return np.sum(radii)

    n_trials = 200
