import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    m = len(placed_centres)
    dists = np.sum((candidates[:, np.newaxis, :] - placed_centres) ** 2, axis=2)
    min_dists = np.min(dists, axis=1)
    small_clearance = 1e-6
    r_max = np.clip(0.5 * (min_dists - placed_radii) - small_clearance, 0, None)
    borders = np.min(np.stack((candidates[:, 0], candidates[:, 1], 1 - candidates[:, 0], 1 - candidates[:, 1])), axis=0)
    r_max = np.minimum(r_max, borders)
    return r_max

def run_packing() -> tuple:
    h = 0.05
    epsilon = 1e-6
    grid = np.arange(0, 1, h, dtype=float)
    candidates = np.array([[x, y] for x in grid for y in grid])
    candidates = candidates[np.any(candidates > epsilon, axis=1) & np.any(candidates < 1 - epsilon, axis=1)]

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + epsilon + 1e-9)
        candidates = candidates[~too_close]

    if len(placed_centres) < 26:
        return np.zeros((26, 2)), np.zeros(26), 0.0

    C = np.array(placed_centres)
    N = len(placed_centres)
    r = cp.Variable(N, nonneg=True)
    objective = cp.Maximize(cp.sum(r))
    constraints = [r <= C[:, 0], r <= 1 - C[:, 0], r <= C[:, 1], r <= 1 - C[:, 1]]
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(r[i] + r[j] <= np.linalg.norm(C[i] - C[j]) - 1e-9)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-9)

    return C, r.value, np.sum(r.value)

centers, radii, sum_radii = run_packing()
