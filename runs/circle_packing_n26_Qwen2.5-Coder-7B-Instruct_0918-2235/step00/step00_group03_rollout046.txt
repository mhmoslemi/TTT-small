import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def refine_radii(centres, radii, eps=1e-12, max_iter=200):
    n = len(centres)
    for _ in range(max_iter):
        update = np.zeros_like(radii)
        for i in range(n):
            b_i = np.min([centres[i][0] - radii[i], 1 - centres[i][0] - radii[i],
                           centres[i][1] - radii[i], 1 - centres[i][1] - radii[i]])
            deltas = np.maximum(0, cdist([centres[i]], centres) - radii - eps)
            deltas[i] = np.inf
            update[i] = np.min([b_i, np.min(deltas)])
        radii = np.maximum(0, radii + update)
        if np.max(np.abs(update)) < eps:
            break
    return radii

def simulate_annealing(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):
    n = len(init_centres)
    best_centres = init_centres.copy()
    current_centres = init_centres.copy()
    radii = np.full(n, 1e-3)
    
    current_best_sum = np.sum(refine_radii(current_centres, radii))
    best_sum = current_best_sum
    
    for step in range(steps):
        step_size = init_step * (1 - step / steps)
        for _ in range(n):
            i = rng.integers(0, n)
            old_pos = current_centres[i].copy()
            current_centres[i][0] += rng.uniform(-step_size, step_size)
            current_centres[i][1] += rng.uniform(-step_size, step_size)
            current_centres[i] = current_centres[i].clip(0, 1)
            new_radii = refine_radii(current_centres, radii)
            new_sum = np.sum(new_radii)
            if new_sum > current_best_sum:
                current_best_sum = new_sum
            else:
                current_centres[i] = old_pos
                new_radii = refine_radii(current_centres, radii)
                new_sum = np.sum(new_radii)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = current_centres.copy()
        
        if step % 100 == 0:
            print(f"Step {step}/{steps}, Best sum: {best_sum}")
    
    return best_centres, radii, best_sum

def run_packing(seed=42) -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(seed)
    n = 26
    best_total = float('-inf')
    best_centers = None
    best_radii = None
    
    for _ in range(30):
        init_centres = rng.uniform(size=(n, 2))
        init_centres[:, 0] = init_centres[:, 0] / np.sum(init_centres[:, 0])
        init_centres[:, 1] = init_centres[:, 1] / np.sum(init_centres[:, 1])
        centres, radii, sum = simulate_annealing(rng, init_centres)
        if sum > best_total:
            best_total = sum
            best_centers = centres
            best_radii = radii
    
    return best_centers, best_radii, best_total
