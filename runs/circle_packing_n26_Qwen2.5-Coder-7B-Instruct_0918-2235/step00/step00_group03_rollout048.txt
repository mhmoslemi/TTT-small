import numpy as np
from scipy.optimize import minimize, check_grad
from scipy.spatial.distance import cdist
from scipy.stats import unituniform

def softplus(x):
    return np.log(1 + np.exp(x))

def penalty(x, r, boundary_penalty):
    penalties = 0.0
    for i in range(r.size):
        penalties += softplus(boundary_penalty * max(0, x[i, 0] - r[i]))
        penalties += softplus(boundary_penalty * max(0, r[i] - (1 - x[i, 0])))
        penalties += softplus(boundary_penalty * max(0, x[i, 1] - r[i]))
        penalties += softplus(boundary_penalty * max(0, r[i] - (1 - x[i, 1])))
    for i in range(r.size):
        for j in range(i + 1, r.size):
            dist = np.linalg.norm(x[i] - x[j]) - r[i] - r[j]
            penalties += softplus(max(0, r[i] + r[j] - dist))
    return penalties

def energy(vars, boundary_penalty):
    x = vars[0:26].reshape(-1, 2)
    r = vars[26:]
    return -np.sum(r) + penalty(x, r, boundary_penalty)

def gradient(vars, boundary_penalty):
    x = vars[0:26].reshape(-1, 2)
    r = vars[26:]
    num_vars = 52
    grad = np.zeros(num_vars)
    for i in range(r.size):
        grad[26+i] = -1.0
        grad[26+i] += 4 * boundary_penalty * r[i] * softplus(boundary_penalty * max(0, x[i, 0] - r[i]))
        grad[26+i] += 4 * boundary_penalty * r[i] * softplus(boundary_penalty * max(0, r[i] - (1 - x[i, 0])))
        grad[26+i] += 4 * boundary_penalty * r[i] * softplus(boundary_penalty * max(0, x[i, 1] - r[i]))
        grad[26+i] += 4 * boundary_penalty * r[i] * softplus(boundary_penalty * max(0, r[i] - (1 - x[i, 1])))
    dists = cdist(x, x)
    r_dists = r + r[:, np.newaxis]
    mask = dists > 0
    grad += np.where(mask & (r_dists < dists), 2 * boundary_penalty * (r_dists - dists) * softplus(boundary_penalty * (r_dists - dists)), 0).sum(axis=1)
    return grad.reshape(num_vars, 1)

def optimize(vars, boundary_penalty):
    options = {'disp': False, 'maxiter': 1000, 'tol': 1e-3}
    result = minimize(energy, vars, method='L-BFGS-B', jac=gradient, args=(boundary_penalty,), bounds=[(0, 1) for _ in range(vars.size)], options=options)
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    radius_bound = 0.5
    boundary_penalty = 1.0

    best_sum_r = 0
    best_vars = None
    for restart in restarts:
        vars = np.hstack((restart.flat, radius_bound * np.ones(26)))
        for lambda_power in range(0, int(np.log(max_lambda) / np.log(2)) + 1):
            lambda_ = 2 ** lambda_power
            vars = optimize(vars, lambda_)
            current_sum_r = np.sum(vars[26:])

                best_sum_r = current_sum_r
