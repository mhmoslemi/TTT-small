import numpy as np
import cvxpy as cp


def compute_r_max(candidates, placed_centres, placed_radii):
    m = len(placed_centres)
    r_max = np.zeros(candidates.shape[0])
    for i in range(candidates.shape[0]):
        r_good = np.inf
        for cr, c in zip(placed_radii, placed_centres):
            dist = np.linalg.norm(candidates[i] - c)
            if dist - cr - 1e-12 > 0:
                r_good = min(r_good, dist - cr - 1e-12)
        r_max[i] = min(r_good, candidates[i, 0], 1 - candidates[i, 0], candidates[i, 1], 1 - candidates[i, 1])
    return r_max


def greedily_place_circles():
    np.random.seed(42)
    h = 0.05
    epsilon = 1e-6
    candidates = np.array([[x, y] for x in np.arange(epsilon, 1, h) for y in np.arange(epsilon, 1, h)])
    
    placed_centres = []
    placed_radii = []
    
    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best_idx = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best_idx])
        placed_radii.append(r_candidates[best_idx])
        dists = np.linalg.norm(candidates - candidates[best_idx], axis=1)
        too_close = dists < (r_candidates[best_idx] + 1e-9)
        candidates = candidates[~too_close]
    
    return np.array(placed_centres), np.array(placed_radii)


def refine_placement(centers, radii):
    n = centers.shape[0]
    C = cp.Variable((n, 2), value=centers)
    R = cp.Variable((n,), nonneg=True, value=radii)
    
    objective = cp.Maximize(cp.sum(R))
    constraints = [
        R <= C[:, 0],
        R <= 1 - C[:, 0],
        R <= C[:, 1],
        R <= 1 - C[:, 1],
        [R[i] + R[j] <= cp.sqrt(np.sum((C[i] - C[j]) ** 2)) for i in range(n) for j in range(i + 1, n)]
    ]
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-9)
    
    return C.value, R.value


def run_packing() -> tuple:
    centers, radii = greedily_place_circles()
    for _ in range(5000):
        alpha = 0.02 - 0.0018 * np.random.rand()
        centers -= alpha * np.random.randn(26, 2)
        centers = np.clip(centers, 0, 1)
        centers_opt, radii_opt = refine_placement(centers, radii)
        if radii_opt.sum() > radii.sum():
            centers, radii = centers_opt, radii_opt

    return centers, radii, radii.sum()
