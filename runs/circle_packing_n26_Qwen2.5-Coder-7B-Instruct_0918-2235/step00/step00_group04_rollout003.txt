import numpy as np
from scipy.optimize import linprog

N_CIRCLES = 26
N_PARTICLES = 30
INERTIA = 0.729
COGNITIVE = 1.49445
SOCIAL = 1.49445
MAX_ITER = 3000
VEL_CLIP = 0.1
ALPHA = 0.999
STDDEV = 0.01
GRID_SIZE = 5.0

def solve_radius_lp(centers):
    n_circles = centers.shape[0]
    distances = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    diagonal = np.arange(n_circles)
    diagonal_distances = distances[diagonal, diagonal] + 1e-12
    inequalities = []
    for i in range(n_circles):
        inequalities.extend([
            np.concatenate([[1], centers[i, :], [1] * (n_circles - i - 1)]),
            np.concatenate([[1] * (i + 1), centers[i, :], [0] * (n_circles - i - 1)]),
            np.concatenate([[1] * (i + 1), centers[i, :], [1], [0] * (n_circles - i - 2)])
        ])
    inequalities.extend([
        np.insert(centers[i, :], i, [0.0] * 2),
        np.insert(centers[i, :], i, [1.0] * 2),
    ] for i in range(n_circles))
    A_ub = np.array(inequalities)
    b_ub = np.concatenate([centers[i, 0] for i in range(n_circles)],
                          [centers[i, 1] for i in range(n_circles)],
                          [1.0 for _ in range(n_circles)],
                          [1.0 for _ in range(n_circles)],
                          diagonal_distances + 1e-12])
    c = -np.ones(n_circles)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n_circles, method='highs')
    return result.x, -result.fun

def generate_seed():
    grid = np.array([[i / GRID_SIZE for i in range(6)] for j in range(5)])
    np.random.shuffle(grid)
    grid = grid[:26] + np.random.normal(0, STDDEV, (26, 2))
    grid = np.clip(grid, 0, 1)
    return grid * (1 / GRID_SIZE)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    centres = generate_seed()
    velocities = np.zeros((N_PARTICLES, N_CIRCLES, 2))
    pbests = [None] * N_PARTICLES
    pbest_sums = np.zeros(N_PARTICLES)
    gbest = np.zeros((N_CIRCLES, 2))
    gbest_sum = 0.0

    for _ in range(MAX_ITER):
        for p in range(N_PARTICLES):
            if pbest_sums[p] > gbest_sum:
                gbest = pbests[p].copy()
                gbest_sum = pbest_sums[p]

            velocities[p] = INERTIA * velocities[p] + COGNITIVE * np.random.rand() * (pbests[p] - centres[p]) + \
                             SOCIAL * np.random.rand() * (gbest - centres[p])

            velocities[p] = np.clip(velocities[p], -VEL_CLIP, VEL_CLIP)
            centres[p] += velocities[p]

            centres[p] = np.clip(centres[p], (0, 0), (1, 1))

        for p in range(N_PARTICLES):
            radii, sum_radii = solve_radius_lp(centres[p])
            if sum_radii > pbest_sums[p]:
                pbests[p] = centres[p].copy()
                pbest_sums[p] = sum_radii

        if pbest_sums[np.argmin(pbest_sums)] > gbest_sum and np.random.rand() < 0.02:
            np.random.shuffle(centres)

    _, best_sum = solve_radius_lp(gbest)
    return gbest, _, best_sum
