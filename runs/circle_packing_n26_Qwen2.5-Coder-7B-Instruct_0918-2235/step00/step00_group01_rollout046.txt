import numpy as np
from scipy.optimize import minimize


def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])
        cons.append(1 - xs[i] - rs[i])
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist2 = dx * dx + dy * dy
            cons.append(dist2 - (rs[i] + rs[j]) ** 2)
    return np.array(cons)


def objective(v):
    return -np.sum(v[2::3])


def initial_guess(n=26, r0=0.02):
    xs, ys = np.random.uniform(r0, 1 - r0, (2, n))
    rs = np.full(n, r0, dtype=float)
    occupied = np.zeros((n, n), dtype=bool)
    while True:
        feasible = True
        for i in range(n):
            if rr[i] > 0:
                continue
            x = xs[i]
            y = ys[i]
            rr[i] = r0
            for j in range(i + 1, n):
                if rr[j] > 0:
                    dx = xs[j] - x
                    dy = ys[j] - y
                    dist = np.sqrt(dx * dx + dy * dy)
                    if (rr[i] + rr[j] > dist) or (any(np.abs(xs - x) < (rr[i] + rr[j])) or any(np.abs(ys - y) < (rr[i] + rr[j]))):
                        rr[i] = r0
                        rr[j] = r0
                        feasible = False
                        break
            if not feasible:
                break
        if feasible:
            return np.concatenate((xs, ys, rr))


def run_packing():
    r0 = 0.08  # Smaller radius to start within the constraints
    n = 26
    bounds = [(0, 0.5) for _ in range(26)] + [(0, 1) for _ in range(26)] + [(0, 0.5) for _ in range(26)]
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
   
    results = minimize(objective, initial_guess(n, r0), method='SLSQP', bounds=bounds, constraints=({'type': 'ineq', 'fun': constraint_all}), options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    
    if not results.success:
        raise ValueError("Optimization did not converge successfully.")

    centers = results.x[0::3].reshape(-1, 2)
    radii = results.x[2::3]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
