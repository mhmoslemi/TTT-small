import numpy as np
from scipy.optimize import linprog, bounds
from scipy.spatial.distance import pdist, squareform

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum of radii
    radius_constraints = []
    for i in range(n):
        radius_constraints.append([1] + [0]*i + [-1] + [0]*(n-i-1))
        radius_constraints.append([1] + [0]*i + [-1] + [0]*(n-i-1))
        radius_constraints.append([1] + [0]*i + [0] + [-1]*(n-i-1))
        radius_constraints.append([1] + [0]*i + [0] + [-1]*(n-i-1))
    for i in range(n):
        for j in range(i+1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            row = [0]*i + [1] + [0]*(j-i-1) + [1] + [0]*(n-j-1) + [-2]
            radius_constraints.append(row)

    A_ub = np.array(radius_constraints)
    b_ub = np.array([x for x, y in centers] + [x for x, y in centers] + [2*(1 - x) for x, y in centers] + [2*y for x, y in centers] + [dist_ij**2 for dist_ij in squareform(pdist(centers))])
    
    lp_result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds(0, 1), method='highs')
    return np.array(lp_result.x), -lp_result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
   step_size = 0.1
    
    # Generate initial centers using a grid and random deletions
    grid = np.mgrid[0.1:1:5.1, 0.1:1:6.1].T.reshape(-1, 2)
    centers = np.random.choice(grid, size=n, replace=False)
    
    current_centers = centers.copy()
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers = centers.copy()
    best_radii = current_radii
    best_sum = current_sum
    
    max_iter = 20000
    step_reduction_interval = 5000
    step = step_size

    for iter in range(max_iter):
        i = np.random.randint(n)
        new_pos = current_centers[i] + np.random.uniform(-step, step, size=2)
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_pos
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum + 1e-8:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = new_centers
                best_radii = new_radii
                best_sum = new_sum
        if iter % step_reduction_interval == 0:
            step *= 1/10
        
    return best_centers, best_radii, best_sum
