import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # Step 2.1: Random permutation of indices
    order = np.random.permutation(n)
    
    # Helper function to calculate squared distance
    def dist2(i, j):
        return np.sum((centers[i] - centers[j]) ** 2)
    
    # Step 2.2: Iterative placement
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    MIN_R = 1e-8
    
    for t in range(n):
        k = order[t]
        S = 5000
        candidates = np.random.rand(S, 2)  # Sample points uniformly in the unit square
        
        # Filter infeasible positions
        feasible = True
        while feasible:
            filtered_cands = []
            for x, y in candidates:
                if np.all(dist2(k, np.where(centers != 0)[0]) >= (radii[np.where(centers != 0)[0]] + MIN_R) ** 2) and x > MIN_R and x < 1 - MIN_R and y > MIN_R and y < 1 - MIN_R:
                    filtered_cands.append([x, y])
            if filtered_cands:
                candidates = np.array(filtered_cands)
                feasible = False
        
        # Choose the point that maximizes the locally minimal clearance
        cleansing_ratios = np.zeros(len(filtered_cands))
        for idx, (x, y) in enumerate(filtered_cands):
            cleansing_ratio = (
                min(1 - x, x, 1 - y, y)  # Distance to walls
                / np.sqrt(dist2(k, np.where(centers != 0)[0]) - (radii[np.where(centers != 0)[0]] ** 2)).min()
            )
            cleansing_ratios[idx] = cleansing_ratio
        
        # Select the point with the largest clearance
        clearance_idx = np.argmax(conserving_ratios)
        x, y = filtered_cands[clearance_idx]
        centers[k] = [x, y]
        radii[k] = min(1 - x, x, 1 - y, y)
        
    # Optional: local shake
    shake_strength = 1e-4
    for _ in range(10):
        for i in range(n):
            for d in range(2):
                centers[i][d] += (np.random.rand() * 2 - 1) * shake_strength
                if centers[i][d] < 0:
                    centers[i][d] = 0
                elif centers[i][d] > 1:
                    centers[i][d] = 1
    
    # Step 3: Force-field relaxation & simulated annealing
    T_init = 1.0
    T_final = 1e-4
    cooling_rate = 0.999
    epsilon_w = 1.0
    epsilon_p = 1.0
    
    T = T_init
    while T > T_final:
        for _ in range(200000):
            i = np.random.randint(n)
            translate_radius = 0.01 * T
            dx, dy = np.random.randn(2) * translate_radius
            new_x, new_y = centers[i] + np.array([dx, dy])
            new_x = np.clip(new_x, radii[i], 1 - radii[i])
            new_y = np.clip(new_y, radii[i], 1 - radii[i])
            new_rad = min(1 - new_x, new_x, 1 - new_y, new_y)
            
            if new_rad < radii[i]:
                epsilon_w *= 0.95
                epsilon_p *= 0.95
            else:
                new_potential = sum([
                    0.5 * epsilon_w * ((new_rad + min(1 - new_x, new_x, 1 - new_y, new_y))**-1)**2,
                    sum(0.5 * epsilon_p * ((new_rad + radii[j] - dist2(i, j))**-1)**2 for j in range(n) if i != j)
                ])
                old_potential = sum([
                    0.5 * epsilon_w * ((radii[i] + min(1 - centers[i, 0], centers[i, 0], 1 - centers[i, 1], centers[i, 1]))**-1)**2,
                    sum(0.5 * epsilon_p * ((radii[i] + radii[j] - dist2(i, j))**-1)**2 for j in range(n) if i != j)
                ])
                
                if new_potential <= old_potential:
                    centers[i] = np.array([new_x, new_y])
                    radii[i] = new_rad
                else:
                    if np.random.rand() < np.exp((old_potential - new_potential) / T):
                        centers[i] = np.array([new_x, new_y])
                        radii[i] = new_rad
        
        T *= cooling_rate
    
    # Step 4: Post-processing & refinement
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    maximal_radius_sum = sum(min(1 - np.min(centers[:i, 0]), np.min(centers[:i, 0]), 1 - np.min(centers[:i, 1]), np.min(centers[:i, 1])) for i in range(n))

    return centers, radii, maximal_radius_sum
