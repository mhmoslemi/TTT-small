import numpy as np
from scipy.optimize import minimize
from itertools import combinations

# Function to initialize a hexagonal lattice and drop points to form 26 circles
def init_hex_grid():
    # Create a 6x5 grid of points
    points = np.array([[(np.sqrt(3)/6) * i + (1 if i % 2 else 0.5),
                         (1/2) * i] for i in range(15)])
    # Randomly select 4 points to drop, leaving 26 centers
    centers = np.vstack(points[np.random.choice(points.shape[0], 26, replace=False)])
    # Initialize all radii very small
    radii = np.full(26, 0.01)
    return centers, radii

# Helper functions to unpack and pack variables
def unpack(z):
    return z[:52].reshape(-1, 2), z[52:]

def build_variable_vector(centers, radii):
    return np.hstack((centers.ravel(), radii))

# Function to compute the penalty objective
def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    penalty_B = np.sum(np.maximum(0, r - x)**2 +
                       np.maximum(0, r - (1 - x)**2) +
                       np.maximum(0, r - y)**2 +
                       np.maximum(0, r - (1 - y)**2))
    penalty_O = 0
    for i, j in combinations(range(26), 2):
        x_i, y_i = x[i], y[i]
        x_j, y_j = x[j], y[j]
        d_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        penalty_O += np.maximum(0, r[i] + r[j] - d_ij)**2
    return -np.sum(r) + lambda_B * penalty_B + lambda_O * penalty_O

# Function to compute the Jacobian of the penalty objective
def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    grad_x = np.zeros(52)
    grad_y = np.zeros(52)
    grad_r = np.zeros(26)
    
    for i in range(26):
        r_i = r[i]
        for dim in [0, 1]:
            grad_x[i * 2 + dim] = -4 * r_i * (1 if i * 2 + dim == 2 * np.where(r > x)[0] else 0)
            grad_y[i * 2 + dim] = -4 * r_i * (1 if i * 2 + dim == 2 * np.where(r > y)[0] else 0)
        for j in range(i + 1, 26):
            x_i, y_i = x[i], y[i]
            x_j, y_j = x[j], y[j]
            d_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            if d_ij > r_i + r_j:
                continue
            factor = -2 * (r_i + r_j - d_ij) * ((x_i - x_j) / d_ij)
            if x_i < x_j:
                grad_x[i * 2] += factor
            else:
                grad_x[j * 2] -= factor
            if y_i < y_j:
                grad_y[i * 2 + 1] += factor
            else:
                grad_y[j * 2 + 1] -= factor

    grad_r = np.zeros(26)
    for i in range(26):
        for j in range(i + 1, 26):
            r_i, r_j = r[i], r[j]
            x_i, y_i = x[i], y[i]
            x_j, y_j = x[j], y[j]
            d_ij = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            if d_ij > r_i + r_j:
                continue
            grad_r[i] -= factor
            grad_r[j] -= factor

    return np.hstack((grad_x, grad_y, grad_r))

# Main packing function
def run_packing():
    lambda_B = 1e6
    lambda_O = 1e6

    # Initialize centres and radii
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)

    # Define objective and gradient
    def objective(z):
        return compute_penalty_objective(*unpack(z), lambda_B, lambda_O)

    def jacobian(z):
        return compute_penalty_gradient(*unpack(z), lambda_B, lambda_O)

    # Bounds
    bounds = [(0, 1)] * 52 + [(0, None)] * 26

    # Run optimizer
    result = minimize(objective, z, jac=jacobian, method='trust-constr',
                     bounds=bounds, options={'gtol': 1e-6, 'ftol': 1e-8,
                                            'maxiter': 3000})

    # Extract result
    centres_opt, radii_opt = unpack(result.x)
    sum_radii = np.sum(radii_opt)

    return centres_opt, radii_opt, sum_radii
