import numpy as np

# Constants
NUM_CIRCLES = 26
s_initial = 0.08
THETA_RANGE = np.linspace(0, np.pi/6, 51)
DXDY_RANGE = np.linspace(-0.025, 0.025, 51)

def generate_centres(theta, dx, dy, s):
    b1 = np.array([s, 0])
    b2 = np.array([s/2, s*np.sqrt(3)/2])
    indices = np.arange(-4, 5), np.arange(0, 3)
    centres = []
    for n1 in indices[0]:
        for n2 in indices[1]:
            centre = n1*b1 + n2*b2
            centre = np.dot([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]], centre)
            centre += np.array([dx, dy])
            centres.append(centre)
    return np.array(centres)

def solve_lps(centres):
    n = len(centres)
    A = np.eye(2*n, 2*n)
    A[:n, :n] = -np.eye(n)
    A[n:, :n] = -np.eye(n)
    A[:n, n:] = 2*np.column_stack([centres, np.zeros((n, 2))])[:, 0::2]
    A[n:, n:] = 2*np.column_stack([centres, np.zeros((n, 2))])[:, 1::2]
    b = np.hstack([centres[:, 0], centres[:, 1], 2])
    bounds = [(0, None) for _ in range(2*n)]
    radii = np.maximum(np.linalg.solve(A, b), 0)
    sum_radii = np.sum(radii)
    return radii / sum_radii, sum_radii  # Normalize for comparison

def nelder_mead(x, tolerance=1e-7):
    """
    Find the minimum of a function using the Nelder–Mead method.
    :param x: Initial guess for the minimizer
    :param tolerance: Accuracy to achieve in the minimizer
    :return: Best candidate point and sum of radii
    """
    # Simplex setup
    n = len(x)
    max_iter = 1000
    a = np.array([0.001, 0.001])
    r = 1.0
    s = 0.5
    t = 2.0
    lower_bound = np.array([-np.inf] * n)
    upper_bound = np.array([np.inf] * n)
    
    # Initialize centroid
    c = np.mean(x[:-1], axis=0)
    
    # Main loop
    for i in range(max_iter):
        # Evaluate points
        vals = [solve_lps(generate_centres(x[i], x[i+1], x[i+2], s_initial))[1] for i in range(n+1)]

        if np.all(vals <= vals[0]):
            return x[np.argmin(vals)], min(vals)
    
        # Reflection
        xr = c + (1 + r) * (x[-1] - c)
        vr = solve_lps(generate_centres(xr[0], xr[1], xr[2], s_initial))[1]
        
        if vr < vals[-1]:
            s *= t
        else:
            if vr < vals[-2]:
                # Expansion
                xe = c + s * (xr - c)
                ve = solve_lps(generate_centres(xe[0], xe[1], xe[2], s_initial))[1]
                
                if ve < vr:
                    x[-1] = xe
                else:
                    x[-1] = xr
            else:
                s *= r
                x[1:] = ([c + s * (j - c) for j in x[1:]] + [c])
                
                if vr < vals[-1]:
                    x[-1] = xr
    
    # Fallback on worst candidate
    return x[np.argmin(vals)], min(vals)

def run_packing():
    best_z, best_sum = None, -np.inf
    
    # 1. Coarse grid search
    for theta in THETA_RANGE:
        for dx in DXDY_RANGE:
            for dy in DXDY_RANGE:
                centres = generate_centres(theta, dx, dy, s_initial)
                if any((c < 0).any(axis=1) or (c > 1).any(axis=1) for c in centres):
                    continue
                radii, s = solve_lps(centres)
                if s > best_sum:
                    best_sum, best_z = s, (centres, radii)

    # 2. Local refinement with Nelder–Mead
    for n in range(100):
        seed = np.random.random(6)
        modseed = 0.02 * (seed - 0.5)
        (theta, dx, allowed) = modseed
        y = generate_centres(theta, dx, allowed, s_initial)
        y radii, s = solve_lps(y)
        update_best_if_needed(best_z, y, s)

    # 3. Final jiggle
    for i in range(200):
        seed = np.random.random(6)
        y = generate_centres(seed[0], seed[1], seed[2], s_initial)
        y radii, s = solve_lps(y)
        update_best_if_needed(best_z, y, s)

    # 4. Final LP re-solve to guarantee optimal radii
    best_centres, _ = best_z
    best_radii, best_sum = solve_lps(best_centres)

    return best_centres, best_radii, best_sum

# Function to update best solution
def update_best_if_needed(best_z, candidate, candidate_sum):
    global best_sum, best_z
    if candidate_sum > best_sum:
        best_sum, best_z = candidate_sum, candidate

# Top level function call
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of radii: {sum_radii}")
