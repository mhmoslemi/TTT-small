import numpy as np
from scipy.optimize import minimize, linprog

# Define the validation function
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def lp_radii_fixup(centers, radii):
    n = centers.shape[0]
    A = np.zeros((n * n, n))
    b = np.zeros(n * n)
    for i in range(n):
        A[i * n + i, i] = 1
        b[i * n + i] = radii[i]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A[i * n + j, i] = 1
            A[i * n + j, j] = 1
            b[i * n + j] = dist - radii[i] - radii[j]
    result = linprog(-radii, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)])
    return result.x

def obj_func(x):
    n = x.size // 3
    radii = x[:n]
    walls = x[n:2*n]
    overlaps = x[2*n:]
    P_wall = np.sum(walls)
    P_overlap = np.sum(overlaps)
    return -np.sum(radii) + 1e4 * P_wall + 1e4 * P_overlap

    walls = x[n:2*n]
    overlaps = x[2*n:]
    gradradii = np.ones(n)
    gradwalls = np.zeros(n)
    gradoverlaps = np.zeros(n)
    for i in range(n):
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if radii[i] + radii[j] > dist:
                    gradradii[i] -= 1
                    gradradii[j] -= 1
    gradwalls.fill(0)
    gradoverlaps.fill(0)
    return np.concatenate((gradradii, gradwalls, gradoverlaps))

def constraint_wall(x):
    n = x.size // 3
    radii = x[:n]
    walls = x[n:2*n]
    return walls - radii

def constraint_overlap(x):
    n = x.size // 3
    radii = x[:n]
    walls = x[n:2*n]
    overlaps = x[2*n:]
    for i in range(n):
        for j in range(i + 1, n):

            if radii[i] + radii[j] > dist:
                overlaps[i * n + j] = 1
            else:
                overlaps[i * n + j] = 0
    return overlaps

def initial_centers_poisson(density=10):
    n = 26
    cx, cy = np.random.rand(2 * n)
    centers = np.vstack((cx, cy)).T
    return centers

def initial_radii_fixed(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        if r <= 0:
            radii[i] = 0
        else:
            radii[i] = r
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers_poisson = initial_centers_poisson()
    radii_fixed = initial_radii_fixed(centers_poisson)
    centers_fixed = centers_poisson + radii_fixed[:, np.newaxis]
    centers = centers_fixed * 0.5

    best_centers = centers
    best_radii = radii_fixed
    best_sum = np.sum(best_radii)

    n = centers.shape[0]
    lower_bound = np.zeros(3 * n)
    upper_bound = np.full(3 * n, np.inf)

    for i in range(n):
        lower_bound[3 * i] = -radii_fixed[i]
        upper_bound[3 * i] = centers_fixed[i, 0]
        lower_bound[3 * i + 1] = -radii_fixed[i]
        upper_bound[3 * i + 1] = centers_fixed[i, 1]
        lower_bound[3 * i + 2] = 0
        upper_bound[3 * i + 2] = 0.5

    for i in range(5):
        x0 = centers.flatten()

                        constraints=[{'type': 'ineq', 'fun': constraint_wall},
                                     {'type': 'ineq', 'fun': constraint_overlap},
                                     {'type': 'eq', 'fun': lambda x: np.sum(x[2*n:])}],
                        options={'maxiter': 2000})
        if validate_packing(res.x.reshape(n, 2) / 2, res.x[n:2*n]):
            if np.sum(res.x[n:2*n]) > best_sum:
                best_centers = res.x.reshape(n, 2) / 2
                best_radii = res.x[n:2*n]
                best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
