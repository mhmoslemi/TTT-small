import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_cycles = 50
    eps_stop = 1e-4
    num_trials = 40
    alpha = 0.8
    step = 0.1

    def get_distances(centers):
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = dists[j, i] = np.linalg.norm(centers[i] - centers[j])
        return dists

    def get_boundaries(centers):
        return np.minimum.reduce([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)

    def perturb(center, step):
        noise = np.random.uniform(-step, step, size=2)
        return np.clip(center + noise, 0, 1)

    def solve_lp(centers):
        A_ub, b_ub = [], []
        for i in range(n):
            for j in range(i + 1, n):
                row = np.zeros(n)
                row[i], row[j] = 1, 1
                A_ub.append(row)
                b_ub.append(get_distances(centers)[i, j])
        b_ub = np.array(b_ub)
        bounds = [(0, boundary[i]) for i in range(n) for boundary in [get_boundaries(centers)]]
        c = -np.ones(2 * n)
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not result.success:
            raise Exception(f"Solver does not converge, final sum_radii: {np.sum(-result.fun)}")
        return result.x, result.fun

    best_sum = -np.inf
    best_positions = None
    best_radii = None

    for trial in range(num_trials):
        centers = np.random.rand(n, 2)
        for cycle in range(max_cycles):
            for i in np.random.permutation(n):
                old_center = centers[i]
                new_center = perturb(old_center, step)
                centers[i] = new_center
                radii, sum_radii = solve_lp(centers)
                if sum_radii > best_sum + 1e-9:
                    best_sum = sum_radii
                    best_positions, best_radii = centers.copy(), radii

            step *= alpha
            if step < eps_stop:
                break

    return best_positions, best_radii, -best_sum
