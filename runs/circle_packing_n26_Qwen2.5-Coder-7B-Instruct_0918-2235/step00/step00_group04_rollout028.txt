import numpy as np
from scipy.optimize import linprog, minimize

def lp_solve_and_grad(x):
    n = 26
    A = np.array([
        np.block([[np.eye(n), -np.eye(n)], 
                   [np.eye(n, k=n), np.eye(n, k=n)],
                   [np.eye(n, k=-n), np.eye(n, k=-n)],
                   [np.eye(n, k=n) + np.eye(n, k=-n), -np.eye(n, k=n) + np.eye(n, k=-n)],
                   np.zeros((n, 2*n))])
    B = np.array([
        np.ones(n),
        np.zeros(n),
        np.ones(n),
        np.zeros(n),
        np.zeros(n)
    ])
    c = np.zeros(2*n)
    
    for i in range(n):
        A[4*i+0, i] = 1
        B[4*i+0,] = x[i]
        A[4*i+1, i] = -1
        A[4*i+2, i+n] = 1
        B[4*i+2,] = x[i+n]
        A[4*i+3, i+n] = -1
        B[4*i+3,] = 1 - x[i+n]
    
    for i in range(n):
        for j in range(i+1, n):
            A[4*n+(i*n+(j-i))//2, i] = 1
            A[4*n+(i*n+(j-i))//2, j+n] = 1
            B[4*n+(i*n+(j-i))//2,] = np.sqrt((x[i]-x[j+n])**2 + (x[i+n]-x[j])**2)
    
    res = linprog(c, A_ub=A, b_ub=B, bounds=[(0,1)]*52, method='highs', options={'presolve':True, 'maxiter':200})
    if not res.success:
        return np.inf, None
    
    r = res.x[:n]
    lambda_ = res.dual
    
    gradient_x = np.zeros(2*n)
    gradient_y = np.zeros(2*n)
    
    for i in range(26):
        gradient_x[i] = lambda_[i] - lambda_[26 + i]
        gradient_y[i] = lambda_[26 + i] - lambda_[27 + i]
    
    for i in range(n):
        for j in range(i+1, n):
            if np.dot([x[i] - x[j+n], x[i+n] - x[j]], lambda_[4*n+(i*n+(j-i))//2]) >= 0:
                d = np.sqrt((x[i]-x[j+n])**2 + (x[i+n]-x[j])**2)
                u = (x[i]-x[j+n])/d
                v = (x[i+n]-x[j])/d
                gradient_x[i] += lambda_[4*n+(i*n+(j-i))//2]*u
                gradient_x[j+n] += -lambda_[4*n+(i*n+(j-i))//2]*u
                gradient_y[i] += lambda_[4*n+(i*n+(j-i))//2]*v
                gradient_y[j+n] += -lambda_[4*n+(i*n+(j-i))//2]*v
    
    return -np.sum(r), [gradient_x, gradient_y]

def run_packing():
    np.random.seed(42)
    centers = np.random.rand(26, 2)
    #centers = np.array([[0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.5], [0.5, 0.75], [0.75, 0.5], [0.25, 0.5], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.5], [0.5, 0.75], [0.75, 0.5], [0.25, 0.5], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.5], [0.5, 0.75], [0.75, 0.5], [0.25, 0.5], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.5]])
    
    for _ in range(1000):
        objective, gradient = lp_solve_and_grad(centers.flatten())
        bound_constraints = np.all(centers >= 0) and np.all(centers <= 1)
        if objective >= -2.636 and bound_constraints:
            return centers, np.clip(centers.flatten(), 0, None), objective
        
        if bound_constraints and np.any(objective > -2.636):
            step_size = 1e-3
            best_obj = objective
            best_centers = centers.copy()
            for i in range(52):
                for sign in [-1, 1]:
                    delta = np.zeros(52)
                    delta[i] = step_size * sign
                    new_centers = centers.flatten() + delta
                    new_centers = new_centers.clip(0, 1).reshape(26, 2)
                    new_objective, _ = lp_solve_and_grad(new_centers.flatten())
                    if new_objective > best_obj:
                        best_obj = new_objective
                        best_centers = new_centers
                    
            centers = best_centers
    return centers, np.clip(centers.flatten(), 0, None), objective
