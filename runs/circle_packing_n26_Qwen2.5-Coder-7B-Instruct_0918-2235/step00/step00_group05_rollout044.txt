import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance

def generate_centres(theta, dx, dy, s):
    OFFSETS = [
        [-4, 0], [-3, 0], [-2, 0], [-1, 0], [0, 0], [1, 0], [2, 0], [3, 0], [4, 0],
        [-3, 1], [-2, 1], [-1, 1], [0, 1], [1, 1], [2, 1], [3, 1],
        [-2, 2], [-1, 2], [0, 2], [1, 2], [2, 2],
        [-1, 3], [0, 3], [1, 3],
        [0, 4]
    ]
    b1 = np.array([s, 0])
    b2 = np.array([s/2, np.sqrt(3)*s/2])
    centres = []
    for offset in OFFSETS:
        point1 = np.array(offset) * b1
        center1 = (point1[0] + 0.25 * s, point1[1] + np.sqrt(3)/6 * s)
        if abs(center1[0]) > 1 or abs(center1[1]) > 1: continue
        centres.append(center1)
        vertical_flip = [-1 * point[0] for point in OFFSETS]
        vertical_point1 = np.array(vertical_flip) * b1
        center2 = (vertical_point1[0] + 0.25 * s, vertical_point1[1] + np.sqrt(3)/6 * s)
        if abs(center2[0]) > 1 or abs(center2[1]) > 1: continue
        centres.append(center2)
    for center in centres:
        rotated_center = np.dot(np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]]), center)
        translated_center = (rotated_center[0] + dx, rotated_center[1] + dy)
        if abs(translated_center[0]) > 1 or abs(translated_center[1]) > 1: continue
        centres.append(translated_center)
    return np.array(centres)

def solve_lps(centres):
    n = centres.shape[0]
    c = -np.ones(n)
    A = np.vstack((centres[:, 0], centres[:, 1], 1 - centres[:, 0], 1 - centres[:, 1]))
    b = np.zeros(4 * n)
    for i in range(n):
        for j in range(i + 1, n):
            bi = np.linalg.norm(centres[i] - centres[j])
            A = np.vstack((A, np.zeros(4 * n)))
            A[-1, 4 * i + 0] = centres[j, 0] - centres[i, 0]
            A[-1, 4 * i + 1] = centres[j, 1] - centres[i, 1]
            A[-1, 4 * i + 2] = centres[j, 0] - centres[i, 0]
            A[-1, 4 * i + 3] = centres[j, 1] - centres[i, 1]
            A[-1, 4 * j + 0] = -(centres[j, 0] - centres[i, 0])
            A[-1, 4 * j + 1] = -(centres[j, 1] - centres[i, 1])
            A[-1, 4 * j + 2] = -(centres[j, 0] - centres[i, 0])
            A[-1, 4 * j + 3] = -(centres[j, 1] - centres[i, 1])
            b[-1] = bi
    result = linprog(c, A_ub=A, b_ub=b, method='highs')
    radii = np.maximum(result.x, 0)
    return radii, result.fun

def validate_packing(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
