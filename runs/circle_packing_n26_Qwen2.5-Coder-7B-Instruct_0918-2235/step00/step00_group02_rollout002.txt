```python
import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26

    def objective(radii):
        centers = pack_circles(radii)
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def pack_circles(radii):
        centers = np.zeros((n, 2))
        for i in range(n):
            # Place circle at a reasonable initial position
            centers[i] = np.random.rand(2)
        return centers

    def constraint(radii):
        centers = pack_circles(radii)
        overlaps = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    overlaps += 1
        return overlaps

    x0 = np.ones(n) / 26  # Initial guess for radii
    bounds = [(0, None) for _ in range(n)]  # Radii must be nonnegative

    cons = {'type': 'ineq', 'fun': constraint}
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)

    if not res.success:
        raise ValueError("Optimization failed")

    radii = res.x
    centers = pack_circles(radii)

    return centers, radii, np.sum(radii)
```
