import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_lpx(centers, radii):
    n = 26
    dmat = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    
    A_eq = np.ones((n, n)) * 0.5
    b_eq = np.ones(n)
    A_constraints = []
    b_constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = radii[i] + radii[j]
            row[j] = -1
            A_constraints.append(row)
            b_constraints.append(dmat[i, j])
    
    for i in range(n):
        row = np.zeros(n)
        row[i] = radii[i]
        row[i] = 1
        A_constraints.append(row)
        b_constraints.append(centers[i, 0])
    
    result = linprog(-radii, A_ub=A_constraints, b_ub=b_constraints, A_eq=A_eq, b_eq=b_eq, method='highs', options={"disp": False})
    if result.success:
        return -result.fun
    else:
        return -np.inf

def evaluate(positions):
    if positions.shape[0] != 52:
        return -np.inf
    centers = positions.reshape(26, 2)
    radii = np.sqrt(np.sum(centers ** 2, axis=1))
    return -solve_lpx(centers, radii)

bounds = [(0, 1)] * 52
result = differential_evolution(evaluate, bounds, popsize=20, recombination=0.7, maxiter=100)
centers = result.x.reshape(26, 2)
radii = np.sqrt(np.sum(centers ** 2, axis=1))
sum_radii = np.sum(radii)

for i in range(26):
    if np.isnan(radii[i]) or radii[i] < 0 or (centers[i][0] < 0 or centers[i][0] > 1 or centers[i][1] < 0 or centers[i][1] > 1):
        print("Constraints violated")
        sum_radii = 0

print(f"Sum of radii: {sum_radii:.5f}")
print(f"Centers:\n{centers}")
print(f"Radii:\n{radii}")
