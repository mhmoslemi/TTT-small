import numpy as np
from scipy.optimize import linprog

def get_pairs(n):
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    return pairs

def validate_packing_and_accept(centers, radii, step_size, alpha, epsilon):
    n = centers.shape[0]
    pairs = get_pairs(n)
    
    W, B = np.tri(n, k=1), np.zeros(n)
    A, b = np.zeros((2 * n + 2 * n * n, n)), np.zeros(2 * n + 2 * n * n)

    for i in range(n):
        A[i, i] = 1
        A[n + i, i] = -1
        A[i, i % n] = -2 * step_size
        A[2 * n + i, i] = 2 * step_size
        b[i], b[n + i] = radii[i] + step_size, radii[i] - step_size

    for i, j in pairs:
        A[2 * n * n + i * n + j, i] = 1
        A[2 * n * n + i * n + j, i] = 1
        A[2 * n * n + i * n + j, j] = -1
        A[2 * n * n + i * n + j, j] = -1
        b[2 * n * n + i * n + j] = 1 - min(radii[i] + radii[j] + 2 * step_size, min(W[i], 1 - W[j]))

    for i in range(n):
        A[2 * n * n + n * n + i, i] = 1
        A[2 * n * n + n * n + i, i] = 1
        b[2 * n * n + n * n + i] = W[i]

    result = linprog(c=A.dot(radii), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if result.success:
        new_radii = result.x
        new_sum = np.sum(new_radii)
        if new_sum > np.sum(radii) + epsilon:
            step_size *= alpha
            return new_radii, new_step_size, True
        elif step_size > 1e-4:
            step_size *= alpha
            return radii, new_step_size, False
        return radii, new_step_size, False
    return radii, new_step_size, False

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)  # Ensure reproducibility for debugging
    centres = np.random.rand(n, 2)
    radii = np.zeros(n)
    step_size = 0.1
    alpha = 0.8
    epsilon = 1e-9
    trials = 40
    best_sum = -1
    best_positions = None
    best_radii = None

    for _ in range(trials):
        random_state = np.random.get_state()
        random_positions = np.random.rand(n, 2)
        random_radii = validate_packing_and_accept(centres, radii, step_size, alpha, epsilon)[0]
        new_sum = np.sum(random_radii)
        if new_sum > best_sum:
            best_sum = new_sum
            best_positions = random_positions
            best_radii = random_radii
            np.random.set_state(random_state)

    return best_positions, best_radii, best_sum
