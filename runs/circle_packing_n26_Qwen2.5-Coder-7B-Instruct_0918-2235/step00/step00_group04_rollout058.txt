import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint, Bounds
from scipy.spatial.distance import cdist

# Constants
N = 26
d = 1
lambda_B = 1e6
lambda_O = 1e6

def init_hex_grid():
    n_per_side = 6
    centre_grid = []
    radii_grid = []
    for i in range(n_per_side):
        for j in range(3 if i % 2 == 0 else 4):
            x = (i + j / 2) * d / n_per_side
            y = (i / 2 + j * 2 / 3) * d / n_per_side
            centre_grid.append((x, y))
    centre_grid = np.array(centre_grid)
    centre_grid = centre_grid[np.random.choice(len(centre_grid), N, replace=False)]
    radii_grid = np.full(N, 0.01)
    return centre_grid, radii_grid

def build_variable_vector(centres, radii):
    return np.concatenate((centres.flatten(), radii))

def unpack(z):
    x = z[:52].reshape(-1, 2)
    y = z[52:104].reshape(-1, 2)
    r = z[104:]
    return x, y, r

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    boundary_penalty = np.sum(np.maximum(0, r - x)**2 + 
                              np.maximum(0, r - (1 - x))**2 + 
                              np.maximum(0, r - y)**2 + 
                              np.maximum(0, r - (1 - y))**2)
    overlap_penalty = np.sum(np.maximum(0, r[:, np.newaxis] + r - cdist(x, x))**2)
    return -np.sum(r) + lambda_B * boundary_penalty + lambda_O * overlap_penalty

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    n = len(x)
    grad = np.zeros(78)
    for i in range(n):
        grad[2*i] += -2 * lambda_B * np.maximum(0, r[i] - x[i]) if r[i] > x[i] else 0
        grad[2*i+1] += -2 * lambda_B * np.maximum(0, r[i] - (1 - x[i])) if r[i] > (1 - x[i]) else 0
        grad[2*i] += -2 * lambda_B * np.maximum(0, r[i] - y[i]) if r[i] > y[i] else 0
        grad[2*i+1] += -2 * lambda_B * np.maximum(0, r[i] - (1 - y[i])) if r[i] > (1 - y[i]) else 0
        for j in range(i+1, n):
            dist = np.sqrt(((x[i] - x[j])**2 + (y[i] - y[j])**2))
            if dist > r[i] + r[j]:
                continue
            grad[2*i] += -2 * lambda_O * (r[i] + r[j] - dist) * (x[i] - x[j]) / dist if r[i] + r[j] > dist else 0
            grad[2*i+1] += -2 * lambda_O * (r[i] + r[j] - dist) * (y[i] - y[j]) / dist if r[i] + r[j] > dist else 0
            grad[2*j] += -2 * lambda_O * (r[i] + r[j] - dist) * (x[j] - x[i]) / dist if r[i] + r[j] > dist else 0
            grad[2*j+1] += -2 * lambda_O * (r[i] + r[j] - dist) * (y[j] - y[i]) / dist if r[i] + r[j] > dist else 0
    return grad

def solve_lp(centres):
    n = len(centres)
    A = np.zeros((n, 2 * n))
    b = np.zeros(n)
    for i in range(n):
        A[i, 2*i] = 1
        b[i] = 0.01
        for j in range(i+1, n):
            A[j, 2*i] = 1
            A[j, 2*j] = 1
            d_ij = np.sqrt(((centres[i, 0] - centres[j, 0])**2 + (centres[i, 1] - centres[j, 1])**2))
            b[j] = d_ij - 0.01
    A = np.vstack((A, -np.eye(n * 2)))
    b = np.hstack((b, np.zeros(n)))
    c = np.hstack((np.zeros(n), np.ones(n)))
    result = linprog(c, A_ub=A, b_ub=b, bounds=Bounds(0, None), method='highs')
    if result.success:
        return result.x[n:], result.fun
    else:
        raise ValueError("Failed to solve LP")

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)
    bounds = [(0,1)]*52 + [(0,None)]*26
    res = minimize(compute_penalty_objective, z, jac=compute_penalty_gradient, method='trust-constr', bounds=bounds, options={'gtol':1e-6, 'ftol':1e-8, 'maxiter':3000, 'verbose':0})
    centres_opt, radii_opt = unpack(res.x)
    radii_opt, _ = solve_lp(centres_opt)
    return centres_opt, radii_opt, np.sum(radii_opt)
