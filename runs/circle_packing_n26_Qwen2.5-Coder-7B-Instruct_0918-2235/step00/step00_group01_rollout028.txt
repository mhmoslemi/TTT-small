import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    
    # Initialize population of particles
    P = 60
    N = 26
    v_min = np.zeros(78)
    v_max = np.array([1, 1, 0.5] * N)
    particles = np.random.rand(P, 78) * (v_max - v_min) + v_min
    velocities = np.zeros_like(particles)
    p_best = particles.copy()
    p_best_value = -np.inf * np.ones(P)
    g_best = p_best[np.argmax(p_best_value)]
    g_best_value = -np.inf
    
    lambda_factor = 1000
    G = 200
    v_max_val = v_max[:3]
    
    for g in range(1, G + 1):
        for p in range(P):
            # Update velocity
            velocities[p] = (g_best - particles[p]) * np.random.rand() * 0.157 + particles[p] * 0.7298
            particles[p] += velocities[p]
            
            # Clip position to bounds
            particles[p] = np.clip(particles[p], v_min, v_max)
            
            # Evaluate fitness
            Pen = eval_penalties(particles[p])
            value = np.sum(particles[p][2::3]) - lambda_factor * Pen
            if value > p_best_value[p]:
                p_best[p] = particles[p]
                p_best_value[p] = value

        if p_best_value.max() > g_best_value:
            g_best = p_best[p_best_value.argmax()]
            g_best_value = p_best_value.max()
        
        if g % 20 == 0:
            lambda_factor *= 1.02

    # Post-processing
    best_v = g_best
    centers = best_v[:N * 2].reshape((N, 2))
    radii_tmp = best_v[N * 2:]
    radii_tmp = np.maximum(0, np.sqrt(penalty_l_ij(centers)))
    radii = np.minimum(0.5, radii_tmp)
    
    # Radius refinement
    for _ in range(10):
        for i in range(N):
            x = centers[i, 0]
            y = centers[i, 1]
            r_max = min(x, 1 - x, y, 1 - y)
            for j in range(N):
                if j == i:
                    continue
                r_max = min(r_max, radii[j] + np.linalg.norm(centers[i] - centers[j]) - radii[j])
            if radii[i] > r_max:
                radii[i] = r_max

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def eval_penalties(v):
    N = 26
    penalty = np.zeros(4)
    for i in range(N):
        penalty_lst = []
        penalty_lst.append((v[i] > 1 - v[2] or v[2] < 0) - 0)
        penalty_lst.append((v[2 + i * 3] > 1 - v[-2] or v[-2] < 0) - 0)
        for j in range(i + 1, N):
            penalty_lst.append((penalty_l_ij(v[i:i+3], v[j:j+3]) - v[-2] - v[-1]) - 0)
        penalty[i] = max(penalty_lst)

    lambda_factor = 1000
    return lambda_factor * np.sum(penalty)

def penalty_l_ij(c1, c2):
    return np.sqrt(np.sum((c1[:2] - c2[:2]) ** 2)) - (c1[-1] + c2[-1])
