import numpy as np
from scipy.optimize import minimize

def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])  # lower
        cons.append(1 - xs[i] - rs[i])  # upper
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i in range(n):
        for j in range(i+1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist2 = dx*dx + dy*dy
            cons.append(dist2 - (rs[i] + rs[j])**2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

def generate_initial_guess(n, r0):
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    for i in range(n):
        while True:
            x = np.random.uniform(r0, 1 - r0)
            y = np.random.uniform(r0, 1 - r0)
            valid_position = True
            for circle_j in range(i):
                dx = x - centers[circle_j, 0]
                dy = y - centers[circle_j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[circle_j] + radii[i]:
                    valid_position = False
                    break
            if valid_position:
                centers[i] = [x, y]
                break
    return np.hstack((centers[:, 0], centers[:, 1], radii))

def run_packing():
    n = 26
    r0 = 0.02  # initial radius
    bnds = [(0,1)] * (2 * n) + [(0, 0.5)] * n  # bounds for x, y, and r
    
    x0 = generate_initial_guess(n, r0)
    
    res = minimize(objective, x0,
                   method='SLSQP',
                   bounds=bnds,
                   constraints={'type': 'ineq', 'fun': constraint_all},
                   options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    
    if not res.success:
        raise ValueError("Optimization failed.")
    
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    sum_r = rs.sum()
    
    centers = np.column_stack((xs, ys))
    return centers, rs, sum_r

This solution defines the required `run_packing()` function that generates an initial guess, invokes the optimizer, and returns the centers, radii, and sum of radii. The optimizer is confined to the specified bounds and constraints. If the optimization fails, it raises a `ValueError`.
