import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog

def build_lp(C):
    N = C.shape[0]
    d = distance.cdist(C, C)
    m = N * (N - 1) // 2
    A = np.zeros((m, N))
    b = np.zeros(m)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = d[i, j]
            idx += 1
    for i in range(N):
        A[idx, i] = 1
        b[idx] = C[i, 0]
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - C[i, 0]
        idx += 1
        A[idx, i] = 1
        b[idx] = C[i, 1]
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - C[i, 1]
        idx += 1
    return A, b

def compute_sum_radius(C, radii):
    return np.sum(radii)

def de_obj(C):
    A, b = build_lp(C)
    result = linprog([1] * C.shape[0], 'min', A_ub=A, b_ub=b, bounds=[(0, None)] * C.shape[0], method='highs', options={'reltol': 1e-9})
    if not result.success:
        return 1e6
    return -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    dim = C.shape[1]
    popsize = 15 * dim
    step = 0.02
    best_C = np.zeros((popsize, dim))
    best_r = -np.inf
    best_sum = -np.inf

    for i in range(popsize):
        C = np.random.rand(popsize, dim)
        C += step * np.random.randn(popsize, dim)
        C = np.clip(C, 0, 1)
        for _ in range(10):
            C[i] += step * np.random.randn(dim)
            C[i] = np.clip(C[i], 0, 1)
        best_C[i] = C[i].copy()

    for best_idx in range(3000):
        C = best_C.copy()
        order = np.argsort([-de_obj(C[i:i+1]) for i in range(popsize)])
        best_C = C[order][:popsize]
        for i in range(popsize):
            C[i] += step * np.random.randn(dim)
            C[i] = np.clip(C[i], 0, 1)
        best_C[i] = C[i].copy()
        if np.any(distance.cdist(best_C, best_C) < step):
            best_C = np.clip(best_C + step * np.random.randn(popsize, dim), 0, 1)
        best_C = best_C[order[:popsize]]
        if np.any(distance.cdist(best_C, best_C) < step):
            best_C = np.clip(best_C + step * np.random.randn(popsize, dim), 0, 1)
        if np.any(distance.cdist(best_C, best_C) < step):
            best_C = np.clip(best_C + step * np.random.randn(popsize, dim), 0, 1)
        best_C = best_C[order[:popsize]]
    A, b = build_lp(best_C)
    result = linprog([1] * best_C.shape[0], 'min', A_ub=A, b_ub=b, bounds=[(0, None)] * best_C.shape[0], method='highs', options={'reltol': 1e-9})
    if not result.success:
        return np.zeros((N, dim)), np.zeros(N), 0.0
    radii = result.x
    centers = best_C.copy()
    return centers, radii, np.sum(radii)
