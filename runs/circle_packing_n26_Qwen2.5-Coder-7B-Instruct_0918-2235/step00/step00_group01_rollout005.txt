import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    n = 26
    R_bound_max = 0.01
    penetrating_pairs = []

    # Initial random population
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * R_bound_max

    # Country flag to save compute, no need next step
    min_overlap = 0.5  # Arbitrary minimum overlap runners can set

    # Outer PSO loop
    Accepted = False
    while not Accepted:
        for i in range(n):
            for j in range(i + 1, n):
                distance = np.linalg.norm(centers[i] - centers[j])
                if distance < 2 * min_overlap * np.min(radii[i:j+1]):
                    penetrating_pairs.append((i, j))
                elif pen_distance_check(centers[i], centers[j], np.min(radii[i:j+1])) < min_overlap * np.min(radii[i:j+1]):
                    penetrating_pairs.append((i, j))
                
                if len(penetrating_pairs) > 0:
                    Accepted = True
                    break
            if Accepted:
                break
        # If there's no such pair, break early
        if not Accepted:
            break

    # Fallback if no overlap found
    if not Accepted:
        accepted_centers = centers
        accepted_radii = radii
    else:
        accepted_centers, accepted_radii, _ = generate_feasible_packing(n, R_bound_max)
        penetrating_pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                distance = np.linalg.norm(accepted_centers[i] - accepted_centers[j])
                if pen_distance_check(accepted_centers[i], accepted_centers[j], np.min(accepted_radii[i:j+1])) < min_overlap * np.min(accepted_radii[i:j+1]):
                    penetrating_pairs.append((i, j))
                
                if len(penetrating_pairs) > 0:
                    Accepted = True
                    break
            if Accepted:
                break

    # Local radius refinement
    refined_centers, refined_radii = refine_packing(accepted_centers, accepted_radii)
    return refined_centers, refined_radii, np.sum(refined_radii)

def generate_feasible_packing(n, R_bound_max):
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * R_bound_max
    radii_sum = np.sum(radii)
    while np.any(centers[:,0] + radii > 1 + 1e-12) or np.any(centers[:,1] + radii > 1 + 1e-12) or radii_sum < n:
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n) * (1/2)
        radii_sum = np.sum(radii)
    return centers, radii, radii_sum

def refine_packing(centers, radii):
    while True:
        for i in range(26):
            centers[i], radii[i] = local_radius_max(centers, radii, i)
        if all(radii > radii.sum() / n * (1/3)):  # Threshold to ensure convergence
            return centers, radii

def local_radius_max(centers, radii, ind):
    min_overall = min(centers[ind][0] - 1, centers[ind][1] - 1, 1-centers[ind][0], 1-centers[ind][1])
    for j in range(26):
        if j != ind:
            r = (pen_distance_check(centers[ind], centers[j], np.min(radii[j:ind+1])) - radii[ind] - radii[j]) / 2
            if pen_compare(r, centers[ind], centers[j], np.min(radii[j:ind+1])) and r < min_overall:
                min_overall = r
    return centers[ind], min_overall

def pen_distance_check(center_x, center_y, min_r):
    return min(center_x, 1-center_x, center_y, 1-center_y, min_r)

def pen_compare(r, center_x, center_y, r_min_coor):
    return r > 0 and r < r_min_coor
