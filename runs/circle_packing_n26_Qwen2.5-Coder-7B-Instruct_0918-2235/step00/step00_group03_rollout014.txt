import numpy as np

def refine_radii(centres, radii, eps=1e-15, max_iter=200):
    n = len(centres)
    dist_matrix = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    dist_matrix = np.clip(dist_matrix, 0.0, None)
    
    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            a = centres[i, 0] - radii[i]
            b = 1 - centres[i, 0] - radii[i]
            c = centres[i, 1] - radii[i]
            d = 1 - centres[i, 1] - radii[i]
            
            r = min(a, b, c, d)
            
            dists = dist_matrix[i, :] - radii
            r_min = np.max(dists[dists > eps])
            
            new_r = np.min([r, r_min, 0.5])
            new_radii[i] = np.max([new_r, 0])
            
        if np.sum(np.abs(new_radii - radii)) < eps:
            return new_radii
        radii = new_radii
    return radii

def anneal_one_start(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):
    centres = init_centres.copy()
    radii = np.full(26, 1e-3)
    best_centres, best_radii, best_sum, current_sum = centres.copy(), radii.copy(), np.inf, -np.inf
        
    for _ in range(steps):
        T = 0.25 * (1 - _/steps)
        s = init_step * (1 - _/steps)
    
        for i in range(26):
            origセンター = centres[i].copy()
            newcentre = np.clip(origセンター + rng.uniform(-s, s, size=2), 0, 1)
            
            centres[i] = newcentre
            new_radii = refine_radii(centres, radii)
            
            new_sum = np.sum(new_radii)
            if new_sum > current_sum:
                current_sum, new_radii = new_sum, new_radii.copy()
            else:
                imporance = np.exp((new_sum - current_sum) / T)
                if rng.random_sample() < imporance:
                    current_sum, new_radii = new_sum, new_radii.copy()
        
        if current_sum < best_sum:
            best_centres, best_radii, best_sum = centres.copy(), radii.copy(), current_sum
    
    return best_centres, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    RESTARTS = 30
    centres = np.zeros((26, 2))

    best_results = []

    best_centres, best_radii, best_sum = min(best_results, key=lambda x: -x[2])
