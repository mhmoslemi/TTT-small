import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(v, n, penalty_lambda):
    x = v[:n].reshape((n, 2))
    r = v[n:2*n]
    penalty_overlap = np.sum(np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - np.sqrt(np.sum((x[:, np.newaxis, :] - x[np.newaxis, :, :])**2, axis=2)))**2)
    penalty_boundaries = np.sum(r**2) - np.sum(np.minimum(np.minimum(x[:, 0], x[:, 1]), np.minimum(1 - x[:, 0], 1 - x[:, 1]))**2)
    return -np.sum(r) + penalty_lambda * (penalty_overlap + penalty_boundaries)

def constraint(v, n):
    return np.sum(v[n:2*n])**2 - 4.6**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n)
    bounds += [(0, 0.5)] * n
    penalty_lambda = 1000
    result = differential_evolution(objective, bounds, args=(n, penalty_lambda), popsize=10, maxiter=50, tol=1e-12, disp=False, workers=1, init='sobol')
    best_v = result.x
    best_r = best_v[n:2*n]
    best_sum_radii = np.sum(best_r)
    if best_sum_radii <= 0:
        raise ValueError("Invalid optimal sum radii")
    feasible = False
    while not feasible:
        initial_guess = np.column_stack([np.random.rand(2, n), np.random.rand(n)])
        initial_guess[1, :] = np.maximum(np.minimum(initial_guess[0, :], np.minimum(1 - initial_guess[0, :], np.minimum(initial_guess[0, :], 1 - initial_guess[0, :]))), 0.1)
        initial_guess = initial_guess.flatten()
        optim_result = minimize(lambda v: -constraint(v, n) ** 2, initial_guess, method='BFGS', bounds=bounds, options={'disp': False})
        if optim_result.success:
            feasible = True
            best_sum_radii = 2 * best_sum_radii
    return best_v[:n].reshape((n, 2)), best_r, best_sum_radii
