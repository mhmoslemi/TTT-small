import numpy as np
import cvxpy as cp
import random

# -------------------------------------------------------------
# 1.  Solve the linear programme that gives the best radii for a
#    fixed arrangement of centres.
# -------------------------------------------------------------
def optimise_radii(centers: np.ndarray) -> np.ndarray:
    """Compute the optimal radii for the given centre coordinates."""
    n = centers.shape[0]
    r = cp.Variable(n, nonneg=True)
    x = centers[:, 0]
    y = centers[:, 1]
    constraints = (r <= x) + (r <= 1 - x) + (r <= y) + (r <= 1 - y) + [r[i] + r[j] <= np.linalg.norm(centers[i] - centers[j]) for i in range(n) for j in range(i+1, n)]
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return np.maximum(r.value, 0.0)
