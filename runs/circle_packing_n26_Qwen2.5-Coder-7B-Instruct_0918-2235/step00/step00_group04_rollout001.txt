import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = len(centers)
    c = -np.ones(n)  # Maximize sum

    A = np.zeros((3 * n + n * (n - 1) // 2, n))
    b = np.zeros(3 * n + n * (n - 1) // 2)

    # Position constraints
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0]
        A[i + n, i] = 1
        b[i + n] = 1 - centers[i, 0]
        A[i + 2 * n, i] = 1
        b[i + 2 * n] = centers[i, 1]
        A[i + 3 * n, i] = 1
        b[i + 3 * n] = 1 - centers[i, 1]

    # Pairwise distance constraints
    dist_matrix = np.sqrt(
        np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1)
    )
    dist_matrix -= 1e-8  # Small tolerance to avoid numerical issues

    for i in range(n):
        for j in range(i + 1, n):
            A[3 * n + i * (n - 1) // 2 + (j - i - 1), i] = 1
            A[3 * n + i * (n - 1) // 2 + (j - i - 1), j] = 1
            b[3 * n + i * (n - 1) // 2 + (j - i - 1)] = dist_matrix[i, j]

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')

    if not res.success:
        print("Warning: linprog did not converge.")

    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(0)

    # Generate initial centers on a grid and remove random points to get exactly 26
    grid = np.array([(x, y) for x in range(6) for y in range(7)])
    random_indices = np.random.choice(len(grid), 4, replace=False)
    initial_centers = grid[np.array([i for i in range(len(grid)) if i not in random_indices])]

    # Ensure centers are unique and within the unit square
    centers = np.array(initial_centers)
    np.random.shuffle(centers)
    if not np.allclose(centers, np.unique(centers, axis=0)):
        raise Exception("Initial centers are not unique.")

    current_centers, current_radii, current_sum = centers, np.zeros(n), 0.0
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
    step = 0.1

    MAX_ITERS = 20_000
    K_JUMP = 500
    N_REDUCE = 500
    ε = 1e-8

    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        delta = np.random.uniform(-step, step, size=2)
        new_pos = centers[i] + delta
        new_pos = np.clip(new_pos, 0, 1)
        new_centers = np.vstack((centers[:i], new_pos, centers[i+1:]))

        new_radii, new_sum = compute_lp_solution(new_centers)

        if new_sum > current_sum + ε:
            current_centers, current_radii, current_sum = new_centers, new_radii, new_sum

            if new_sum > best_sum:
                best_centers, best_radii, best_sum = current_centers, current_radii, new_sum

        if iter % K_JUMP == 0:
            new_centers = np.random.rand(n, 2)
            np.clip(new_centers, 0, 1, out=new_centers)
            new_radii, new_sum = compute_lp_solution(new_centers)

            if new_sum > best_sum:
                best_centers, best_radii, best_sum = new_centers, new_radii, new_sum

        if iter % N_REDUCE == 0:
            step *= 0.9

    return best_centers, best_radii, best_sum

# Example usage
if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(centers)
    print(radii)
    print(sum_radii)
