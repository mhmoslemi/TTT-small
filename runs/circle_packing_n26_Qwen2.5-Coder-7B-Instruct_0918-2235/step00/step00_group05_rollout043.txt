import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance

def generate_lattice_points():
    nrows = 3
    base = np.array([[9/2, 0, 1/2, 3/2, 2, 1], [3/2, 3/2, 0, 0, 0, 0]])
    base = np.stack([np.cos(np.pi * base / 3), np.sin(np.pi * base / 3)], axis=2)
    base = base.repeat(nrows, axis=0).reshape(-1, 2)
    shift = np.array([[0, 0], [0, 1], [2, 0], [2, 2]])
    lattice_points = (base + shift) % (4 * nrows)
    lattice_points -= nrows
    return lattice_points

def rotation_matrix(theta):
    return np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]])

def generate_centres(theta, dx, dy, s):
    lattice_points = generate_lattice_points()
    rotation = rotation_matrix(theta)
    centres = np.dot(lattice_points, rotation).T
    centres += np.array([[dx, dy], [dx, dy], [dx, dy]])
    centres *= s * 2
    return centres

def solve_lps(centres):
    n = centres.shape[0]
    bounds = [(0, None) for _ in range(n)]
    c = np.ones(n)
    A = np.vstack([centres, np.ones(n), 1 - centres]).T
    b = 2 * np.ones(n+2)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise ValueError("LP failed")
    return res.x, res.fun
