strategy = """
A 2-parameter "hexagonal-lattice + scale" strategy with LP radius-refinement

The core idea is to replace the high-dimensional, unconstrained search of the previous
approaches with a small, well-structured manifold of candidate packings that
captures the essential geometry of an optimal 26-circle arrangement in a unit square.
The only variables that are optimized are:

1. a rotation angle θ of a regular hexagonal lattice,
2. a small horizontal/vertical offset (dx,dy) of that lattice inside the square.

All circle centers are generated deterministically from (θ,dx,dy).
For each geometry we exactly compute the largest admissible radii by solving
a single linear program (the same LP used in the earlier strategies), which
returns the maximum radius vector for the given fixed center set.
Thus the optimisation problem reduces to a plain two-dimensional search
over (θ,dx,dy), while still guaranteeing that every candidate is radius-optimal
for the chosen centre layout. This approach is fundamentally different from the
earlier 52-dimensional global optimisers: it leverages the strong geometric
regularity of circle packing and dramatically reduces the search space.
"""

def build_voronoi(centers):
    # Builds Voronoi diagram for the given circle centers
    return Voronoi(centers)


def compute_radii(centers):
    # Compute the largest admissible radii using linear programming
    n = centers.shape[0]
    r = np.zeros_like(centers)[:, 2]
    for i in range(n):
        r[i] = np.sum(np.square((centers[:, :2] - centers[i, :2])), axis=1)
        r[i] = min(np.sqrt(r[i]) - centers[i, 2], centers[i, 2])
    return r


def solve_lps(centers):
    # Solves the linear programming problem for the given centers
    n = centers.shape[0]
    r = compute_radii(centers)
    return r, sum(r)
