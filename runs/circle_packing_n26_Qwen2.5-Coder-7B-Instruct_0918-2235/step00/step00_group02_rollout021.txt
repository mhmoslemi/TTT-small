import numpy as np
from scipy.optimize import minimize, bounds as Bounds

def calc_max_radii(C):
    n = C.shape[0]
    r = np.zeros(n)
    for i in range(n):
        r[i] = min(C[i, 0], 1 - C[i, 0], C[i, 1], 1 - C[i, 1])
        for j in range(i + 1, n):
            d = np.linalg.norm(C[i] - C[j])
            r[i] = min(r[i], d - r[j])
            r[j] = min(r[j], d - r[i])
    return r, np.sum(r)

def objective(x):
    C = x.reshape((26, 2))
    r, s = calc_max_radii(C)
    if np.isnan(r).any() or np.isinf(r).any() or np.any(r < 0):
        return -1e9
    if not validate_packing(C, r)[0]:
        return -1e9
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    C0 = np.random.rand(n, 2)  # Initial random placement
    C0 = np.clip(C0, 0.05, 0.95)  # Ensure some space from boundaries
    bounds = Bounds(0, 1, n * 2)
    
    res = minimize(objective, C0.reshape(-1), bounds=bounds, method='L-BFGS-B')
    if res.success and res.fun < -2.636:
        C, r = res.x.reshape((n, 2)), -res.fun
        final_radii = np.zeros(n)
        for i in range(n):
            final_radii[i] = min(C[i, 0], 1 - C[i, 0], C[i, 1], 1 - C[i, 1])
            for j in range(i + 1, n):
                d = np.linalg.norm(C[i] - C[j])
                final_radii[i] = min(final_radii[i], d - final_radii[j])
                final_radii[j] = min(final_radii[j], d - final_radii[i])
        
        # Ensure final radii satisfy constraints
        if np.any(final_radii < 0):
            return np.zeros((n, 2)), np.zeros(n), 0.0

        return np.zeros((n, 2)), np.zeros(n), 0.0
