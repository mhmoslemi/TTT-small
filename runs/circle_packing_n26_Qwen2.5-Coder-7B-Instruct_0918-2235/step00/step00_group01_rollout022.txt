from scipy.optimize import minimize
import numpy as np

def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])
        cons.append(1 - xs[i] - rs[i])
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i in range(n):
        for j in range(i+1,n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist2 = dx*dx + dy*dy
            cons.append(dist2 - (rs[i] + rs[j])**2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    xs = np.zeros(n)
    ys = np.zeros(n)
    rs = np.full(n, 0.02)
    v0 = np.concatenate((xs, ys, rs))

    bnds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    res = minimize(objective, v0, method='SLSQP', bounds=bnds, 
                   constraints={'type':'ineq', 'fun':constraint_all}, 
                   options={'ftol':1e-9, 'maxiter':5000, 'disp':False})
    
    if not res.success:
        raise Exception("Optimization failed")
    
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs, ys))
    sum_r = rs.sum()
    
    return centers, rs, sum_r
