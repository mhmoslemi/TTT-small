import numpy as np
from scipy.optimize import linprog

def solve_radius_lp(centres):
    n = centres.shape[0]
    c = -np.ones(n)
    A = np.zeros((2*n*(n-1)//2, n))
    b = np.zeros(2*n*(n-1)//2)
    count = 0

    for i in range(n):
        A[count, i] = 1
        b[count] = centres[i, 0]
        count += 1
        A[count, i] = 1
        b[count] = 1 - centres[i, 0]
        count += 1
        A[count, i] = 1
        b[count] = centres[i, 1]
        count += 1
        A[count, i] = 1
        b[count] = 1 - centres[i, 1]
        count += 1

        for j in range(i+1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            A[count, i] = 1
            A[count, j] = 1
            b[count] = d
            count += 1

    r = linprog(c, A_ub=A, b_ub=b, method='highs')
    radii = -r.x
    sum_radii = np.sum(radii)
    return radii, sum_radii

n_particles = 30
inertia = 0.729
cognitive = 1.49445
social = 1.49445
max_iter = 3000
vel_clip = 0.1
alpha = 0.999

np.random.seed(42)
centers = np.zeros((n_particles, 26, 2))

def initialize_particles():
    grid = np.arange(0.1, 1, 0.2)
    grid_points = np.array(np.meshgrid(grid, grid)).T.reshape(-1, 2)
    np.random.shuffle(grid_points)
    chosen_indices = np.random.choice(len(grid_points), 26, replace=False)
    centers[0, chosen_indices] = grid_points[chosen_indices] + 0.01 * np.random.randn(26, 2)

initialize_particles()

velocities = np.zeros_like(centers)
pbests = np.copy(centers)
pbest_sums = np.zeros(n_particles)
gbest = np.copy(pbests[np.argmax(pbest_sums), :, :])
gbest_sum = np.max(pbest_sums)

for _ in range(max_iter):
    for p in range(n_particles):
        velocities[p] = inertia * velocities[p] + cognitive * np.random.rand(1, 2) * (pbests[p] - centers[p]) + social * np.random.rand(1, 2) * (gbest - centers[p])
        velocities[p] = np.clip(velocities[p], -vel_clip, vel_clip)
        centers[p] += velocities[p]
        centers[p] = np.clip(centers[p], 0, 1)

        radii, sum_radii = solve_radius_lp(centers[p])
        if sum_radii > pbest_sums[p]:
            pbests[p] = centers[p]
            pbest_sums[p] = sum_radii

        if sum_radii > gbest_sum:
            gbest = centers[p]
            gbest_sum = sum_radii

    if alpha is not None:
        velocities *= alpha

best_centres = gbest
best_radii = solve_radius_lp(gbest)[0]
best_sum = gbest_sum

return best_centres, best_radii, best_sum
