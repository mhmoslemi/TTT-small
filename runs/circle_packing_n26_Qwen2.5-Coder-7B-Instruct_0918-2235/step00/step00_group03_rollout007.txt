import numpy as np

def refine_radii(centers, radii, eps=1e-12, max_iter=200):
    n = centers.shape[0]
    dist_matrix = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=-1)
    np.fill_diagonal(dist_matrix, np.inf)

    for _ in range(max_iter):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            b_i = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
            neighbor_limits = np.maximum(0, dist_matrix[:, i] - radii - eps)
            new_radii[i] = min(b_i, np.min(neighbor_limits))
        if np.allclose(new_radii, radii, atol=eps):
            break
        radii = new_radii
    return radii

def anneal_one_start(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):
    centers = init_centres
    radii = np.full(26, 1e-3)
    best_centers, best_radii, best_sum = centers, radii, 0

    for stepnum in range(steps):
        T = 0.25 * (1 - stepnum / steps)
        step = 0.05 * (1 - stepnum / steps)
        for i in range(26):
            dx, dy = rng.uniform(-step, step, 2)
            new_pos = np.clip(centers[i] + [dx, dy], 0, 1)
            centers[i] = new_pos
            new_radii = refine_radii(centers, radii)
            new_sum = np.sum(new_radii)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii.copy()
            else:
                if np.random.rand() < np.exp((new_sum - np.sum(radii)) / T):
                    radii = new_radii

    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    RESTARTS = 30
    centers = np.random.rand(26, 2)
    best_centers, best_radii, best_sum = centers, np.full(26, 1e-3), 0

    for _ in range(RESTARTS):
        centers = np.random.rand(26, 2)
        new_centers, new_radii, new_sum = anneal_one_start(None, centers)
        if new_sum > best_sum:
            best_centers, best_radii, best_sum = new_centers, new_radii, new_sum

    return best_centers, best_radii, np.sum(best_radii)

This code defines three functions: `refine_radii`, `anneal_one_start`, and `run_packing`. Each function is designed according to the strategy outlined earlier. `refine_radii` refines the radii based on the current centers, `anneal_one_start` performs the simulated annealing to optimize the centers and radii, and `run_packing` orchestrates multiple restarts to find the best solution. The final solution is returned as a tuple containing the centers, radii, and sum of radii.
