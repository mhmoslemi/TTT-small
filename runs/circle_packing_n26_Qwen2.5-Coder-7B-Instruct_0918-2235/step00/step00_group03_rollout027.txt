import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    max_cycles = 50
    step = 0.1
    alpha = 0.8
    eps_stop = 1e-4
    num_trials = 40
    best_sum = -np.inf
    best_positions = np.zeros((n, 2))

    def distance_matrix(coords):
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = dists[j, i] = np.linalg.norm(coords[i] - coords[j])
        return dists

    def inner_lp(coords):
        dists = distance_matrix(coords)
        A_ub = np.zeros((n * (n - 1) // 2, n))
        b_ub = np.zeros(n * (n - 1) // 2)
        ub = np.zeros(n)
        for i in range(n):
            ub[i] = min(coords[i, 0], 1 - coords[i, 0], coords[i, 1], 1 - coords[i, 1])
        A_ub_idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[A_ub_idx, i] = 1
                A_ub[A_ub_idx, j] = 1
                b_ub[A_ub_idx] = dists[i, j]
                A_ub_idx += 1
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
        return res.success, res.x, res.fun

    for _ in range(num_trials):
        coords = np.random.rand(n, 2)
        while not inner_lp(coords)[0]:  # Retry if initial positions are invalid
            coords = np.random.rand(n, 2)
        bounds = [(0, min(x, 1 - x, y, 1 - y)) for x, y in coords]
        current_sum = 0
        for cycle in range(max_cycles):
            for i in np.random.permutation(n):
                old_coords = coords[i].copy()
                delta = np.random.uniform(-step, step, 2)
                coords[i] = np.clip(old_coords + delta, 0, 1)
                success, new_r, new_sum = inner_lp(coords)
                if new_sum > current_sum + 1e-9:
                    current_sum = new_sum
                else:
                    coords[i] = old_coords
            step *= alpha
            if step < eps_stop:
                break
        if current_sum > best_sum:
            best_positions = coords
            best_sum = current_sum
    _, _, best_r = inner_lp(best_positions)
    return best_positions, best_r, best_sum
