import numpy as np
from scipy.optimize import minimize

def constraint_all(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])
        cons.append(1 - xs[i] - rs[i])
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist2 = dx**2 + dy**2
            cons.append(dist2 - (rs[i] + rs[j])**2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    r0 = 0.02
    v0 = np.random.uniform(0.06, 0.94, 78)
    v0[2::3] = r0
    bounds = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints [{'type': 'ineq', 'fun': constraint_all}], options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    if not res.success:
        raise Exception('Optimization failed')
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs, ys))
    radii = rs
    sum_r = rs.sum()
    return centers, radii, sum_r
