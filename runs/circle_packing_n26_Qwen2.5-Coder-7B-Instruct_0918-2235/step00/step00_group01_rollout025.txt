import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def evaluate_centers(centers):
    x = centers[:, 0]
    y = centers[:, 1]
    n = len(x)
    d = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)
    np.fill_diagonal(d, float('inf'))
    r_bounds = [(0, None)] * n
    c = np.ones(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [1] + [0] * (n - j - 1))
            b_ub.append(d[i, j])
    for i in range(n):
        for boundary in [x[i], 1 - x[i], y[i], 1 - y[i]]:
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(boundary)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    return -result.fun if result.success else float('nan')

def run_packing():
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    popsize = 20
    maxiter = 200
    tol = 1e-12
    result = differential_evolution(evaluate_centers, bounds, popsize=popsize, maxiter=maxiter, tol=tol)
    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    centers = np.array(result.x).reshape((26, 2))
    r_bounds = [(0, None)] * 26
    c = np.ones(26)
    A_ub, b_ub = [], []
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [1] + [0] * (26 - j - 1))
            b_ub.append(np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2))
    for i in range(26):
        for boundary in [centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]]:
            A_ub.append([0] * i + [1] + [0] * (26 - i - 1))
            b_ub.append(boundary)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    radii = result.x
    return centers, radii, radii.sum()

centers, radii, sum_radii = run_packing()
print(sum_radii)
