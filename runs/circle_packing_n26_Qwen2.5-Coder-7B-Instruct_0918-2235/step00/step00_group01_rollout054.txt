import numpy as np
from scipy.optimize import minimize

def objective(u, distances):
    n = len(u) // 2
    c = np.ones(n)
    A_ub = np.zeros((n, n))
    b_ub = np.zeros(n)
    r_bounds = [(0, None)] * n
    bounds = [(0, 1)] * n * 2
    for i in range(n):
        A_ub[i, 2 * i: 2 * i + 2] = [1, 0]
        b_ub[i] = distances[i, i] - r_bounds[0][1]
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[n + (i * n + j), 2 * i: 2 * i + 2] = [1, 0]
            A_ub[n + (i * n + j), 2 * j: 2 * j + 2] = [0, 1]
            b_ub[n + (i * n + j)] = distances[i, j] - r_bounds[0][1]
    result = minimize(lambda x: -np.sum(x), u, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': synthesized_constraints(A_ub, b_ub)})
    return -result.fun

def synthetic_constraints(A_ub, b_ub):
    def constraints(x):
        return b_ub - np.dot(A_ub, x)
    return constraints

def run_packing():
    np.random.seed(0)
    n = 26
    centers = np.random.rand(n, 2)
    distances = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    centers = centers.flatten()
    result = minimize(objective, centers, args=(distances,), method='SLSQP', bounds=[(0, 1)] * 52)
    best_centers = result.x.reshape(n, 2)
    best_distances = np.sqrt(np.sum((best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]) ** 2, axis=2))
    radii = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            if best_distances[i, j] > 0:
                radii[i] = best_distances[i, j] - best_distances[i, j] ** 2 / (4 * best_centers[i, 0] * (1 - best_centers[i, 0])) if best_centers[i, 0] > 0 and 1 - best_centers[i, 0] > 0 else 0
                radii[j] = best_distances[i, j] - best_distances[i, j] ** 2 / (4 * best_centers[j, 0] * (1 - best_centers[j, 0])) if best_centers[j, 0] > 0 and 1 - best_centers[j, 0] > 0 else 0
        radii[i] = min(radii[i], best_centers[i, 0], 1 - best_centers[i, 0])
    return best_centers, radii, np.sum(radii)
