import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from scipy.linalg import orth

def calc_max_radii(C):
    n = C.shape[0]
    D = distance_matrix(C, C)
    b = np.minimum(C[:,0], 1-C[:,0])
    b = np.minimum(b, C[:,1])
    b = np.minimum(b, 1-C[:,1])
    
    A = np.maximum(C[:,0][:, np.newaxis] - C[:,0][np.newaxis, :], 0)
    A += np.maximum(C[:,1][:, np.newaxis] - C[:,1][np.newaxis, :], 0)
    A -= 2 * np.eye(n)
    
    # Solving the lp using numpy
    for i in range(n):
        A[i, i] = 0
    c = np.concatenate((b - C[:,0], b - C[:,1]))
    A_ub = np.vstack([A, -A, -np.eye(n), -np.eye(n)])
    b_ub = np.concatenate([b, -np.zeros(n), np.zeros(n)])
    x = minimize(lambda r: -r.dot(b), r[:n], method='highs', bounds=[(0, x) for x in b], constraints={'type': 'ineq', 'fun': lambda r: r - b})
    r = x.x
    return r, r.sum()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    sigma = 0.02
    T0 = 0.1
    alpha = 0.995
    N_max = 50000
    restart_threshold = 1000
    centres = np.random.uniform(0.05, 0.95, size=(n, 2))
    radii = np.random.uniform(0.01, 0.1, size=n)
    best_centres = centres.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for k in range(N_max):
        i = np.random.randint(n)
        old_centre = centres[i]
        new_centre = old_centre + sigma * np.random.normal(size=2)
        new_centre[0] = max(0.05, min(0.95, new_centre[0]))
        new_centre[1] = max(0.05, min(0.95, new_centre[1]))
        centres[i] = new_centre
        r, s = calc_max_radii(centres)
        Delta = s - best_sum
        if Delta >= 0 or np.exp(Delta / T0) > np.random.rand():
            best_centres = centres.copy()
            best_radii = r
            best_sum = s
        T0 *= alpha
        if Delta < 0 and k % restart_threshold == 0:
            centres = np.random.uniform(0.05, 0.95, size=(n, 2))
            radii = np.random.uniform(0.01, 0.1, size=n)
    
    final_r, final_sum = calc_max_radii(best_centres)
    return best_centres, final_r, final_sum
