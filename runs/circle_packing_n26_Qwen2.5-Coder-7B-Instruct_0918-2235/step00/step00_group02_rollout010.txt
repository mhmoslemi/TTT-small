import numpy as np

def run_packing():
    # Initial parameters
    n = 26
    target_sum_radii = 2.636

    # Generate initial circle centers randomly within the unit square, avoiding overlapping too much
    centers = np.random.rand(n, 2) * 0.9 + 0.05  # [0.05, 0.95] range to prevent circles touching boundaries
    radii = np.random.rand(n) * 0.1  # Initial radii are small and evenly distributed

    # Iterative relocation to maximize sum of radii while satisfying constraints
    max_iterations = 1000
    for _ in range(max_iterations):
        # Calculate distances between centers
        distance_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1))
        np.fill_diagonal(distance_matrix, np.inf)  # No distance to self

        # Calculate potential new radii and centers
        new_radii = np.copy(radii)
        for i in range(n):
            for j in range(i + 1, n):
                if distance_matrix[i, j] < new_radii[i] + new_radii[j]:
                    # Adjust radii to avoid overlap
                    new_radii[j] = (distance_matrix[i, j] - new_radii[i]) / 2
                    new_radii[i] = (distance_matrix[i, j] - new_radii[j]) / 2

        # Update centers to maintain symmetry and avoid boundary contact
        avg_center = np.mean(centers, axis=0)
        centers -= avg_center
        centers *= 0.98  # Contract slightly
        centers += avg_center

        # Check and update radii and centers
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                radii[i] = 0
                centers[i] = np.random.rand(2) * 0.9 + 0.05
            else:
                radii = new_radii

        # Calculate sum of radii
        sum_radii = np.sum(radii)

    # Ensure sum of radii is as close as possible to the target
    scale = target_sum_radii / sum_radii
    radii *= scale

    return centers, radii, np.sum(radii)
