"""

run_packingä function that packs 26 circles in a unit square [0,1]×[0,1] 
to maximize the sum of radii.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance_matrix

def generate_centres(theta, dx, dy, s):
    b1 = np.array([s, 0])
    b2 = np.array([s / 2, np.sqrt(3) * s / 2])
    centres = []

    rows = [[i, j] for j in range(8) for i in range(j, 9-j)] + \
            [[i, j] for j in range(9) for i in range(8)] + \
            [[i, j] for j in range(9) for i in range(8-j)]
    for i, j in rows:
        point = np.array([i * b1[0] + j * b2[0], i * b1[1] + j * b2[1]])
        point = np.dot(point, np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]]))
        point += np.array([dx, dy])
        centres.append(point)

    centres = np.array(centres)
    return centres

def solve_lps(centres):
    n = len(centres)
    A_ineq = np.vstack((centres, np.eye(n), -np.eye(n), np.eye(n), -np.eye(n), 
                          distance_matrix(centres, centres) - np.eye(n)))
    b_ineq = np.hstack((centres.flatten(), np.zeros(n), np.zeros(n), np.ones(n), np.zeros(n),
                          np.zeros(n)))
    c = np.ones(n)

    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, method='highs')
    if res.success:
        return res.x, res.fun
    else:
        return np.zeros(n), 0.0

def nelder_mead(seed, tolerance=1e-7):
    s0 = np.array(seed)
    bounds = [(0, np.pi/6), (-0.025, 0.025), (-0.025, 0.025)]
    res = minimize(lambda p: -solve_lps(generate_centres(*p)[1])[1], s0, method='nelder-mead', bounds=bounds)
    return tuple(res.x)

def run_packing():
    best_z, best_sum = None, -np.inf
    N = 51

    # Coarse grid search
    for theta in np.linspace(0, np.pi/6, N):
        for dx in np.linspace(-0.025, 0.025, N):
            for dy in np.linspace(-0.025, 0.025, N):
                centres = generate_centres(theta, dx, dy, 0.08)
                if np.any((centres < 0) | (centres > 1)):
                    continue
                radii, s = solve_lps(centres)
                if s > best_sum:
                    best_sum = s
                    best_z = (centres, radii)

    # Local refinement with Nelder-Mead
    for _ in range(100):
        seed = list(best_z[0].flatten())
        theta, dx, dy = nelder_mead(seed, tolerance=1e-7)
        centres = generate_centres(theta, dx, dy, 0.08)
        radii, s = solve_lps(centres)
        if s > best_sum:
            best_sum = s
            best_z = (centres, radii)

    # Final jiggle
    for _ in range(200):
        theta, dx, dy = np.random.uniform(np.pi/10, 5*np.pi/10), np.random.uniform(-0.01, 0.01), np.random.uniform(-0.01, 0.01)
        centres = generate_centres(theta, dx, dy, 0.08)
        radii, s = solve_lps(centres)
        if s > best_sum:
            best_sum = s
            best_z = (centres, radii)

    best_centres, best_radii = best_z
    return best_centres, best_radii, best_sum
