import numpy as np
from scipy.spatial.distance import cdist

rng = np.random.default_rng()

def refine_radii(centres, radii, eps=1e-9, max_iter=200):
    dist_matrix = cdist(centres, centres, 'euclidean')
    for _ in range(max_iter):
        new_radii = np.copy(radii)
        for i in range(26):
            x_i, y_i = centres[i]
            b_i = min(x_i, 1 - x_i, y_i, 1 - y_i)
            min_dist = np.inf
            for j in range(26):
                if i != j:
                    dist_ij = dist_matrix[i, j]
                    min_dist = min(min_dist, max(0, dist_ij - radii[j] - eps))
            new_radii[i] = max(b_i, min_dist)
        if np.max(np.abs(radii - new_radii)) < eps:
            break
        radii = new_radii
    return new_radii

def anneal_one_start(init_centres, rng, steps=500, init_step=0.07, min_step=0.01):
    centres = init_centres.copy()
    radii = np.full(26, 1e-3)
    best_centres = centres.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    for step in range(steps):
        T = 0.25 * (1 - step / steps)
        s = init_step * (1 - step / steps)
        i = rng.choice(26)
        delta = (rng.random(2) * 2 - 1) * s
        new_centres = centres.copy()
        new_centres[i] += delta
        new_centres = np.clip(new_centres, 0, 1)
        new_radii = refine_radii(new_centres, radii)
        new_sum = np.sum(new_radii)
        if new_sum > best_sum or rng.random() < np.exp((new_sum - best_sum) / T):
            centres = new_centres
            radii = new_radii
        if new_sum > best_sum:
            best_centres, best_radii, best_sum = new_centres, new_radii, new_sum
    return best_centres, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    RESTARTS = 30
    best_centers, best_radii, best_sum = None, None, 0.0
    for _ in range(RESTARTS):
        initial_centers = rng.random((26, 2))
        centres, radii, sum = anneal_one_start(initial_centers, rng)
        if sum > best_sum:
            best_centers, best_radii, best_sum = centres, radii, sum
    return best_centers.reshape(26, 2), best_radii, best_sum
