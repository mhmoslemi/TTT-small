```python
import numpy as np
from scipy.optimize import minimize

def build_variable_vector(centres, radii):
    return np.concatenate((centres.flatten(), radii))

def unpack(z):
    n = len(z) // 3
    centers = z[:2*n].reshape((n, 2))
    radii = z[2*n:]
    return centers, radii

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    nb = lambda_B * np.sum(
        (np.maximum(0, r - x))**2 +
        (np.maximum(0, r - (1 - x))**2) +
        (np.maximum(0, r - y))**2 +
        (np.maximum(0, r - (1 - y))**2)
    )
    no = lambda_O * np.sum(
        np.maximum(0, r + r - np.sqrt((x - x)**2 + (y - y)**2))**2
        for i in range(x.shape[0]) for j in range(i + 1, x.shape[0])
    )
    return -np.sum(r) + nb + no

def compute_penalty_gradient(z, lambda_B, lambda_O):
    x, y, r = unpack(z)
    nb_grad_x = -2 * lambda_B * np.maximum(0, r - x) * (x < r)
    nb_grad_y = -2 * lambda_B * np.maximum(0, r - y) * (y < r)
    nb_grad_r = lambda_B * (
        (x < r).astype(int) + (y < r).astype(int)
    )
    no_grad_x = np.zeros_like(x)
    no_grad_y = np.zeros_like(y)
    no_grad_r = lambda_O * np.sum(
        (x - x)**2 * np.maximum(0, r + r - np.sqrt((x - x)**2 + (y - y)**2))**2
        for i in range(x.shape[0]) for j in range(i + 1, x.shape[0])
    )
    grad = np.concatenate((
        nb_grad_x.flatten(), nb_grad_y.flatten(),
        nb_grad_r + no_grad_r
    ))
    return grad

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial conditions: uniform hexagonal grid with a few random points
    hex_grid = np.array([
        np.array([0.17, 0.17]), np.array([0.53, 0.17]), np.array([0.89, 0.17]),
        np.array([0.33, 0.33]), np.array([0.7, 0.33]), np.array([1.0, 0.33]),
        np.array([0.65, 0.5]), np.array([0.0, 0.65]), np.array([1.0, 0.85]),
        np.array([0.35, 1.0]), np.array([0.75, 0.75]), np.array([0.25, 0.75]),
        np.array([0.6, 0.25]), np.array([0.6, 0.8]), np.array([0.3, 0.2]),
        np.array([0.6, 0.9]), np.array([0.9, 0.2]), np.array([0.6, 0.6])
    ])
    np.random.shuffle(hex_grid)
    starting_points = hex_grid[:26]
    radii = np.ones(26) * 0.01
    z = build_variable_vector(starting_points, radii)

    # Penalty weights
    lambda_B = 1e6
    lambda_O = 1e6

    # Objective & gradient functions
    objective = lambda z: compute_penalty_objective(*unpack(z), lambda_B, lambda_O)
    gradient = lambda z: compute_penalty_gradient(z, lambda_B, lambda_O)

    # Bounds
    bounds = [(0, 1)] * 52 + [(0, None)] * 26

    # Optimize
    result = minimize(objective, z, jac=gradient, method='trust-constr', bounds=bounds,
                      options={'gtol': 1e-6, 'ftol': 1e-8, 'maxiter': 3000})

    if not result.success:
        raise ValueError("Optimization failed")

    # Extract results
    centres, radii = unpack(result.x)
    sum_radii = np.sum(radii)
    return centres, radii, sum_radii
```
