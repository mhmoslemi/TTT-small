import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

N = 26

def build_lp(C):
    d = cdist(C, C)
    A = []
    b = []
    for i in range(N):
        A.append([1] * N)
        b.append(1)
        for j in range(i + 1, N):
            A.append([1, -1] + [0] * (N - 2))
            b.append(d[i, j])
            A.append([-1, 1] + [0] * (N - 2))
            b.append(d[i, j])
    for i in range(N):
        A.append([0, 1] + [0] * N)
        b.append(C[i, 0])
        A.append([0, -1] + [0] * N)
        b.append(1 - C[i, 0])
        A.append([1, 0] + [0] * N)
        b.append(C[i, 1])
        A.append([-1, 0] + [0] * N)
        b.append(1 - C[i, 1])
    return np.array(A), np.array(b)

def compute_sum_radius(C):
    A, b = build_lp(C)
    result = linprog([-np.ones(N)], A_ub=A, b_ub=b, bounds=[(0, None)] * 2 * N, method="highs", tol=1e-9)
    if result.status != 0:
        raise Exception("LP infeasible")
    r = -result.x
    r = np.clip(r, 0, None)
    return np.sum(r)

def de_obj(C):
    return -compute_sum_radius(C)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    C = 0.5 * (np.arange(2*N).reshape(N, 2) % 2) + 0.5
    C += 0.02 * np.random.randn(N, 2)
    C = np.clip(C, 0, 1)
    best_C = C
    best_val = de_obj(C)

    result = differential_evolution(de_obj, bounds=[(0, 1), (0, 1)] * N, strategy="best1bin", mutation=0.9, recombination=0.7, maxfev=3000)

    if result.success:
        best_C = result.x.reshape(N, 2)
        best_val = -result.fun

    try:
        A, b = build_lp(best_C)
        result = linprog([-np.ones(N)], A_ub=A, b_ub=b, bounds=[(0, None)] * 2 * N, method="highs", tol=1e-9)
        if result.status != 0:
            raise Exception("LP infeasible")
        r = -result.x
        r = np.clip(r, 0, None)
        return best_C, r, np.sum(r)
    except Exception as e:
        print(e)
        return np.zeros((N, 2)), np.full(N, 0.1), 2.6

    return best_C, best_val, best_val
