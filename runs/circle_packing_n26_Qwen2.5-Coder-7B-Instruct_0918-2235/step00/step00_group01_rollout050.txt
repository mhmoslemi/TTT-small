import numpy as np

np.random.seed(42)

def dist2(p1, p2):
    return np.sum((p1 - p2) ** 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    MIN_R = 1e-8
    S = 5000
    MAX_ITER = 200000

    # Phase 1 - Random greedy insertion
    order = np.random.permutation(n)
    for k in order:
        sampled = np.random.rand(S, 2)
        valid = dist2(sampled[:, :, np.newaxis], pts[:k, :][:, np.newaxis, :]) .max(axis=1, keepdims=True) >= rad[:k, np.newaxis] ** 2
        valid &= (sampled[:, 0] >= MIN_R) & (sampled[:, 0] <= 1 - MIN_R) & (sampled[:, 1] >= MIN_R) & (sampled[:, 1] <= 1 - MIN_R)
        if valid.any():
            coords = sampled[valid]
            clearance = np.minimum(np.minimum(coords[:, 0], 1 - coords[:, 0]), np.minimum(coords[:, 1], 1 - coords[:, 1]))
            Clearance = clearance.min(axis=1)
            clearance[clearance < MIN_R] = MIN_R
            rad[k] = Clearance.max()
            coords = coords[clearance == Clearance.max()][0]
            pts[k] = coords

    # Phase 2 - Force-field relaxation & simulated annealing
    T = 1.0
    T_final = 1e-4
    cooling_rate = 0.999
    epsilon_w = 1.0
    epsilon_p = 1.0
    VMAX = np.inf

    def compute_potential(pts, rad):
        U = 0.0
        for i in range(n):
            U += 0.5 * epsilon_w * (1 / (rad[i] + min(pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1])) ** 2)
            for j in range(i + 1, n):
                dsq = dist2(pts[i], pts[j])
                if dsq < (rad[i] + rad[j]) ** 2:
                    U += 0.5 * epsilon_p * (1 / (rad[i] + rad[j] - np.sqrt(dsq)) ** 2)
        return U, VMAX

    U Old, VMAX = compute_potential(pts, rad)
    while T > T_final:
        for _ in range(MAX_ITER):
            i = np.random.randint(n)
            dx = 2 * np.random.rand() * T - T
            dy = 2 * np.random.rand() * T - T
            x_prime, y_prime = pts[i] + dx, pts[i] + dy
            x_prime = np.minimum(np.maximum(x_prime, rad[i]), 1 - rad[i])
            y_prime = np.minimum(np.maximum(y_prime, rad[i]), 1 - rad[i])
            Clearance = np.minimum(np.minimum(x_prime, 1 - x_prime), np.minimum(y_prime, 1 - y_prime))
            Clearance[ Clearance < MIN_R ] = MIN_R
            rad_prime = Clearance.max()
            if rad_prime <= rad[i]:
                continue
            U_prime, _ = compute_potential(pts, rad_prime)
            delta_U = U_prime - U_Old
            if delta_U <= 0 or np.random.rand() < np.exp(-delta_U / T):
                pts[i] = x_prime, y_prime
                rad[i] = rad_prime
            U_Old = U_prime

        T *= cooling_rate

    # Post-processing & refinement
    for i in range(n):
        clearance = np.minimum(np.minimum(pts[i, 0], 1 - pts[i, 0]), np.minimum(pts[i, 1], 1 - pts[i, 1]))
        for j in range(n):
            if i == j:
                continue
            clearance = np.minimum(clearance, np.sqrt(dist2(pts[i], pts[j])) - rad[j])
        rad[i] = min(clearance, rad[i])

    return pts, rad, rad.sum()
