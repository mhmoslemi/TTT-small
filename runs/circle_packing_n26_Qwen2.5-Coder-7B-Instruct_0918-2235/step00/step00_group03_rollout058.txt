import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def solve_lp(centers, radii_indices):
    n = centers.shape[0]
    A = np.zeros((n * (n - 1) // 2, n))
    b = np.zeros(n * (n - 1) // 2)
    
    for i in range(n):
        for j in range(i + 1, n):
            idx = i * (n - 1) // 2 + j - i - 1
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j])

    density_constraint_matrix = np.zeros((n, n))
    density_constraint_matrix[np.diag_indices(n)] = 1
    density_constraint_matrix[np.diag_indices_from(density_constraint_matrix, k=1)] = -1
    density_constraint_matrix[np.diag_indices_from(density_constraint_matrix, k=-1)] = -1
    
    bounds = np.zeros(n)
    for i in range(n):
        bounds[i] = np.min([centers[i][0], centers[i][1], 1 - centers[i][0], 1 - centers[i][1]])
    
    res = linprog(-radii_indices, A_ub=A, b_ub=b, A_eq=density_constraint_matrix, b_eq=density_constraint_matrix @ np.ones(n), bounds=(0, bounds))
    
    if not res.success:
        return np.zeros_like(radii_indices)
    
    return -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.zeros(n)
    best_sum = -np.inf
    best_centers = None

    for _ in range(40):
        current_sum = 0
        step_size = 0.1
        alpha = 0.8
        restarts = 0

        while step_size > 1e-4:
            for i, (x, y) in enumerate(centers):
                dx, dy = np.random.uniform(-step_size, step_size, 2)
                x_new = np.clip(x + dx, 0, 1)
                y_new = np.clip(y + dy, 0, 1)
                
                radii_new = solve_lp(np.vstack((centers, np.array([[x_new, y_new]]))), radii)
                sum_new = np.sum(radii_new)
                
                if sum_new > current_sum * 1.00000001:
                    current_sum = sum_new
                    centers[i] = [x_new, y_new]
                    radii[i] = radii_new[i]
                else:
                    restarts += 1
                    
            if restarts == 5:
                step_size *= 0.5

            step_size *= alpha

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum

# Main execution
centers, radii, sum_radii = run_packing()

# Validate the solution
validity, message = validate_packing(centers, radii)
print(f"Validation: {validity} - {message}")
print(f"Sum of radii: {sum_radii}")
