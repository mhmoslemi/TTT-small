import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def calc_max_radii(C):
    n = len(C)
    M = squareform(pdist(C))
    A = np.eye(n) + np.eye(n, k=1) + np.eye(n, k=-1)
    b = np.minimum(C[:, 0], 1 - C[:, 0])
    b = np.minimum(b, np.minimum(C[:, 1], 1 - C[:, 1]))
    b = np.minimum(b, M[:, :] - A[:, :] * M[:, :])
    b[b < 0] = 0
    from scipy.optimize import linprog
    res = linprog(-b.flatten(), A_ub=A, b_ub=b.flatten(), method='highs', bounds=(0, np.inf), options={'tol': 1e-9})
    if not res.success:
        return np.zeros(n), np.nan
    return C, res.x

def objective(params):
    n = params.shape[0] // 2
    C = params[:n].reshape(-1, 2)
    r = params[n:]
    M = squareform(pdist(C))
    if np.any(r <= 0) or np.any(M < r[:, np.newaxis] + r[np.newaxis, :]):
        return -np.inf
    total_radius = np.sum(r)
    return -total_radius

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    a = 0.18
    v = np.sqrt(3) * a / 2
    rows = [6] * 2 + [5] * 2 + [6] * 2 + [5] * 2 + [6] * 2
    x = np.array([a * (i + 1) for i in rows]).cumsum()[:-1] + a
    y = np.array([np.arange(row) * v for row in rows]).flatten()
    C = np.column_stack((x[np.argsort(y)], y[np.argsort(y)]))
    C = C[1:-2, :]  # Remove two circles in the lower-left and upper-right corners

    if C.shape[0] != n:
        C = np.random.rand(n, 2) * 0.8 + 0.1

    x0 = np.concatenate((C.flatten(), np.ones(n) * 0.1055))
    bounds = [(0, None)] * 2 * n
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds)
    if result.success:
        C_new = result.x[:n].reshape(-1, 2)
        r_new = result.x[n:]
        _, r = calc_max_radii(C_new)
        if r_new.shape[0] != n:
            r = np.full(n, 0.1055)
        total_radius = np.sum(r)
        return C_new, r, total_radius
    else:
        raise Exception("Optimization failed")
