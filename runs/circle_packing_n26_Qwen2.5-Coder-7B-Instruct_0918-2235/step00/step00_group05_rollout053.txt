import numpy as np
from scipy.optimize import linprog, differential_evolution

def hex_grid(radius, packing_density):
    return np.array([[radius * (2 * i + j % 2) * np.cos(j * 2 * np.pi / 3),
                       radius * np.sin(j * 2 * np.pi / 3)]
                      for i in range((2 * packing_density + np.sqrt(3) * packing_density * packing_density) // (2 * np.pi))])

def add_circles_around_quadrant(centers, quadrant_radius):
    xc, yc = centers.shape[1] // 2, centers.shape[0] // 2
    index = np.where((centers[:, 0] >= xc - quadrant_radius) & (centers[:, 0] <= xc + quadrant_radius) &
                     (centers[:, 1] >= yc - quadrant_radius) & (centers[:, 1] <= yc + quadrant_radius))
    padding = 1 - quadrant_radius
    delta_x, delta_y = 2 * radius, 2 * radius
    for x, y, ci in zip(centers[index, 0], centers[index, 1], np.arange(len(index[0]))):
        if (xc - padding - x)**2 + (yc - padding - y)**2 <= radius * radius:
            centers[index[0][ci] + 1] = [x + delta_x, y]
            centers[index[0][ci] + 2] = [x - delta_x, y]
            centers[index[0][ci] + 3] = [x, y + delta_y]
            centers[index[0][ci] + 4] = [x, y - delta_y]
            break

def solve_radii_fixed_centres(centres):
    A = np.vstack([centres[:, 0], centres[:, 1], np.full(centres.shape[0], 1),].T)
    b = np.array([1 - c[0] for c in centres] + [c[0] for c in centres] + 
                 [1 - c[1] for c in centres] + [c[1] for c in centres])
    res = linprog([1] * centres.shape[0], A_ub=A, b_ub=b, method='highs')
    return res.x, np.sum(res.x)

def penalty_centres_fixed_radii(centres, radii):
    n = centres.shape[0]
    A = []
    b = []
    for i in range(n):
        for j in range(i + 1, n):
            A.append([centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1], 1])
            b.append(radii[i] + radii[j])
    for i in range(n):
        A.append([1, 0, -radii[i]])
        b.append(1 - centres[i, 0])
        A.append([0, 1, -radii[i]])
        b.append(1 - centres[i, 1])
        A.append([1, 0, radii[i]])
        b.append(centres[i, 0])
        A.append([0, 1, radii[i]])
        b.append(centres[i, 1])
    return np.array(b) - np.array(A.dot(radii)) - 1e-12

def coordinated_ascent(init_centres, max_iter=15):
    current_sumradii = np.inf
    for _ in range(max_iter):
        radii, sum_radii = solve_radii_fixed_centres(init_centres)
        if sum_radii <= current_sumradii:
            break
        current_sumradii = sum_radii
        cons = ({'type': 'eq', 'fun': pen, 'args': (radii,)} for pen in [penalty_centres_fixed_radii])
        result = differential_evolution(lambda c: -sum_radii, (-np.array([0, 0]), np.array([1, 1])), args=(radii,), bounds=[(0, 1)] * init_centres.shape[1], constraints=cons, disp=False)
        init_centres = result.x.reshape((init_centres.shape[0], 2))
    return init_centres, radii, sum_radii

This code defines the functions `hex_grid` to generate a hexagonal lattice, `add_circles_around_quadrant` to add surrounding circles, `solve_radii_fixed_centres` to solve the LP for fixed centres, `penalty_centres_fixed_radii` to compute the penalty for fixed radii, `coordinated_ascent` for the main optimization loop, `local_audit_and_refine` for local refinement, and finally `run_packing` to run the entire packing process. The best solution is selected from multiple seeds to ensure robustness.
