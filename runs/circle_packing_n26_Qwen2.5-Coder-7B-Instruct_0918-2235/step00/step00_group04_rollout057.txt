import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial import cKDTree
from scipy.sparse import lil_matrix

def lp_dual_gradient(c, radii, offset, eps=1e-12):
    n = len(c) // 2
    A_ub = lil_matrix((n*4 + n*n//2, n*2))
    b_ub = np.zeros(n*4 + n*n//2)

    # boundary constraints
    for i in range(n):
        A_ub[i*2, 2*i] = 1
        A_ub[i*2 + 1, 2*i + 1] = 1
        b_ub[i*2] = c[2*i]
        b_ub[i*2 + 1] = 1 - c[2*i + 1]

    # pair constraints
    tree = cKDTree(c[:2*n])
    pairs = tree.query_pairs(max_distance=np.inf, k=None, return_sorted=False)
    for i, j in pairs:
        A_ub[n*4 + i*n + j, 2*i] = 1 / (c[2*i] - c[2*j] + eps)
        A_ub[n*4 + i*n + j, 2*j + 1] = 1 / (c[2*j + 1] - c[2*i + 1] + eps)
        b_ub[n*4 + i*n + j] = c[2*i] + c[2*j]

    res = linprog(c=0, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if res.success:
        dual = res.dual
    else:
        dual = np.zeros(A_ub.shape[0])

    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for k, d in enumerate(dual):
        if k < n*4:
            if k % 2 == 0:
                grad_x[k//2] += d if k//2 == 0 else -d
            else:
                grad_y[k//2] += d if k//2 == 0 else -d
        else:
            i, j = np.unravel_index(k - n*4, (n, n))
            dij = np.sqrt((c[2*i] - c[2*j])**2 + (c[2*i + 1] - c[2*j + 1])**2)
            ui, vi = (c[2*i] - c[2*j]) / (dij + eps), (c[2*i + 1] - c[2*j + 1]) / (dij + eps)
            grad_x[i] += d * ui
            grad_y[i] += d * vi
            grad_x[j] -= d * ui
            grad_y[j] -= d * vi

    return -grad_x, -grad_y

def objective(c):
    n = len(c) // 2
    A_ub = lil_matrix((n*4 + n*n//2, n*2))
    b_ub = np.zeros(n*4 + n*n//2)

    # boundary constraints
    for i in range(n):
        A_ub[i*2, 2*i] = 1
        A_ub[i*2 + 1, 2*i + 1] = 1
        b_ub[i*2] = c[2*i]
        b_ub[i*2 + 1] = 1 - c[2*i + 1]

    # pair constraints
    tree = cKDTree(c[:2*n])
    pairs = tree.query_pairs(max_distance=np.inf, k=None, return_sorted=False)
    for i, j in pairs:
        A_ub[n*4 + i*n + j, 2*i] = 1 / (c[2*i] - c[2*j] + 1e-12)
        A_ub[n*4 + i*n + j, 2*j + 1] = 1 / (c[2*j + 1] - c[2*i + 1] + 1e-12)
        b_ub[n*4 + i*n + j] = c[2*i] + c[2*j]

    res = linprog(c=0, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if not res.success:
        return -np.inf, np.zeros(2*n)
    radii = res.x
    return -np.sum(radii), lp_dual_gradient(c, radii, offset)

def run_packing():
    n = 26
    offset = 5
    init_c = offset * np.array([[np.cos(2*i/offset), np.sin(2*i/offset)] for i in range(n*offset)]).flatten()
    init_c = init_c[:2*n]
    bounds = [(0, 1)] * 2*n

    result = minimize(objective, init_c, bounds=bounds, method='L-BFGS-B', jac=True)
    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0

    radii = result.x[:n]
    centers = result.x[n:].reshape(n, 2)

    return centers, radii, np.sum(radii)
