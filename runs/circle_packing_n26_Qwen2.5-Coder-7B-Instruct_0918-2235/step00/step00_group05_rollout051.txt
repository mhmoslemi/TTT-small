import numpy as np
from scipy.optimize import linprog, nelder_mead

def generate_centres(theta, dx, dy, s):
    b1 = np.array([s, 0])
    b2 = np.array([s * 0.5, np.sqrt(3) * s * 0.5])
    centres = np.array([
        b1 * np.array([-4, -3, -2, -1, 0, 1, 2, 3, 4]),
        b1 * np.array([-4, -3, -2, -1, 0, 1, 2, 3, 4]),
        b1 * np.array([-5, -4, -3, -2, -1, 0, 1, 2, 3])
    ])
    centres = np.array([
        b2 * np.array([-4, -3, -2, -1, 0, 1, 2, 3, 4]),
        b2 * np.array([-3, -2, -1, -1, 0, 0, 1, 2, 3]),
        b2 * np.array([-2, -1, -1, 0, 0, 1, 1, 2, 2])
    ])
    centre_sum = centres[0] + centres[1] + centres[2]
    cols = np.array([np.sum(centre_sum, axis=1), np.sum(centre_sum, axis=0)])
    centres += cols[:, np.newaxis]
    centres = np.dot(np.array([[np.cos(theta), -np.sin(theta)], [np.sin(theta), np.cos(theta)]]), centres.T).T
    centres[:, 0] += dx
    centres[:, 1] += dy
    return centres

def within_bounds(centres):
    return np.all((centres >= 0) & (centres <= 1), axis=1)

def solve_lps(centres):
    r = np.ones(26)
    bounds = [(0.0, None)] * 26
    constraints = []
    for i in range(26):
        constraints.append({'type': 'le', 'fun': lambda r, i=i: r[i]})
        constraints.append({'type': 'le', 'fun': lambda r, i=i: 1.0 - r[i]})
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({'type': 'le', 'fun': lambda r, i=i, j=j: np.hypot(*(centres[i] - centres[j])) - (r[i] + r[j])})
    res = linprog(r, A_eq=None, b_eq=None, A_ub=np.array([constraints[i]['fun'].__globals__['r'] for i in range(len(constraints))]), b_ub=np.array([constraints[i]['fun'].__globals__['bub_funs'][i] for i in range(len(constraints))]), bounds=bounds, method='highs')
    return np.array(res.x), res.fun

def run_packing():
    best_sum = -np.inf
    best_z = None

    theta_range = np.linspace(0, np.pi / 6, 51)
    dx_range = np.linspace(-0.025, 0.025, 51)
    dy_range = np.linspace(-0.025, 0.025, 51)

    for theta in theta_range:
        for dx in dx_range:
            for dy in dy_range:
                centres = generate_centres(theta, dx, dy, 0.09)
                if not np.all(within_bounds(centres)):
                    continue
                r, sum_r = solve_lps(centres)
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_z = (centres, r)

    best_centres, best_r = best_z
    return best_centres, best_r, best_sum
