import numpy as np
from scipy.optimize import differential_evolution

def objective(v, n):
    centers = v[:n*2].reshape((n, 2))
    radii = v[n*2:]
    penalty = 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += max(0, r - x)**2 + max(0, r - (1-x))**2 + max(0, r - y)**2 + max(0, r - (1-y))**2
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            dist = np.sqrt((x - xj)**2 + (y - yj)**2)
            penalty += max(0, r + rj - dist)**2
    return -np.sum(radii) + 1000 * penalty

def run_packing():
    n = 26
    bounds = [(0, 1)] * (n * 2) + [(0, 0.5)] * n
    result = differential_evolution(objective, bounds, args=(n,), popsize=10, maxiter=50, seed=np.random.randint(1, 10000))
    best_v = result.x
    best_centers = best_v[:n*2].reshape((n, 2))
    best_radii = best_v[n*2:]

    # Use an LP solver to get the exact optimal radii for the best centers
    from scipy.optimize import linprog
    A = []
    b = []
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        A.append([x, y, -1])
        b.append(r)
    for i in range(n):
        for j in range(i + 1, n):
            xj, yj = best_centers[j]
            rj = best_radii[j]
            A.append([x - xj, y - yj, -1])
            A.append([xj - x, yj - y, -1])
            b.append(r + rj)
            b.append(rj + r)
    for i in range(n):
        A.append([-1, 0, 0])
        b.append(best_radii[i])
        A.append([0, -1, 0])
        b.append(best_radii[i])
        A.append([0, 0, 1])
        b.append(best_radii[i])
        A.append([0, 0, -1])
        b.append(best_radii[i])
    c = np.zeros(n*2 + n)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * (n * 2 + n), method='highs')
    best_radii_optimal = result.x[n:]
    
    return best_centers, best_radii_optimal, np.sum(best_radii_optimal)

centers, radii, sum_radii = run_packing()
