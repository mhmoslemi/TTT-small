```python
import numpy as np
import scipy.optimize as opt
import cma

# Sensitivity: deviation for range checking, not validation
DEV = 1e-5

# Parameters
num_circles = 26
side = 1.0
tolerance = 1e-9

# Penalties
weight_wall = 1e4
weight_overlap = 1e4

# Verifier boundaries
bounds = [(0, side), (0, side)] * num_circles + [(0, side / 2)] * num_circles

# Initial layout incorporating a hexagonal lattice
def initial_mean():
    a = np.sqrt(3) / 3 * side
    b = 2 * side / 3
    c = side / 2
    jitter = 0.01
    
    centres = []
    for i in range(int(num_circles / 3 + 1)):
        for j in range(i + 1):
            x = (j * a + (i - j // 2) * c + np.random.uniform(-jitter, jitter, size=1)) % side
            y = (i * 2 * b / 3 + np.random.uniform(-jitter, jitter, size=1)) % side
            centres.append((x, y))
    centres += [(side / 2, side / 2)]
    centres += [
        ((3 * side / 4, side / 4), (5 * side / 4, side / 4), (3 * side / 4, 3 * side / 4), (5 * side / 4, 3 * side / 4)),
        ((side / 3, side), (side - side / 3, side)),
        # Add additional patterns as needed
    ]
    centres = np.array(centres).flatten()
    radii = np.ones(num_circles) * (side / 2) / np.sqrt(num_circles) * np.random.uniform(0.9, 1.1)
    return np.hstack((centres, radii))

# Objective function with penalty
def objective(z):
    r = refine_radii(z[:2 * num_circles])
    lp_sum = np.sum(r)
    penalty = 0

    for i in range(num_circles):
        r_i = r[i]
        x_i, y_i = z[i * 2], z[i * 2 + 1]

        # Wall penalty
        penalty += weight_wall * np.sum((np.abs(z[i * 2] - np.array([0, side])) + np.abs(z[i * 2 + 1] - np.array([0, side]))) - r_i) ** 2

        # Overlap penalty
        for j in range(i + 1, num_circles):
            r_j = r[j]
            x_j, y_j = z[j * 2], z[j * 2 + 1]
            dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
            penalty += weight_overlap * max(0, (r_i + r_j) - dist) ** 2

    ls_sum = lp_sum - penalty
    return ls_sum

# Unconstrained program to solve linear program (LP) getting max radii for given centre coordinates
def refine_radii(centres):
    A = []
    b = []

    for i in range(num_circles):
        A.append([-1 if r < 0 else 0 for r in radii])
        b.append(bounding_radius(centres[i * 2], centres[i * 2 + 1]))

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            A.append([1 if r < 0 else 0 for r in radii])
            b.append(max(0, np.sqrt((centres[i * 2] - centres[j * 2]) ** 2 + (centres[i * 2 + 1] - centres[j * 2 + 1]) ** 2) - radii[i] - radii[j]))

    solver = opt.linprog([1] * num_circles, A_ub=A, b_ub=b, method='highs')
    return solver.x

# Calculate max radius for a circle placement taking into account both wall and overlap constraints
def bounding_radius(x, y):
    return min(x, y, side - x, side - y)

# CMA-ES main loop
def cma_search():
    mean = initial_mean()
    sigma = (side / num_circles) / 6
    es = cma.CMAEvolutionStrategy(mean, sigma, bounds=bounds)
    res = es.optimize(objective)
    return res

# Local simulated annealing to further refine the solution
def simulate_annealing(z):
    for _ in range(200):
        shake_indices = np.random.choice(num_circles, 3, replace=False)
        shake_offsets = np.random.uniform(-DEV, DEV, size=3 * 2)
        new_z = z.copy()
        new_z[shake_indices * 2 + 0] += shake_offsets[0:3:2]
        new_z[shake_indices * 2 + 1] += shake_offsets[1:3:2]
        new_r = refine_radii(new_z[:2 * num_circles])
        new_sum = np.sum(new_r)
        if new_sum > np.sum(z[2 * num_circles:]):
            z = new_z
    return z

# Main function to run the packing
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    result = cma_search()
    z = result[0]
    r = refine_radii(z[:2 * num_circles])
    ls_sum = np.sum(r)
    z_after = simulate_annealing(z)
    r_after = refine_radii(z_after[:2 * num_circles])
    ls_sum_after = np.sum(r_after)
    validate = validate_packing(z_after[:2 * num_circles].reshape(-1, 2), r_after[num_circles:])
    if not validate[0]:
        r_after = refine_radii(z_after[:2 * num_circles])
        ls_sum_after = np.sum(r_after)
    return z_after[:2 * num_circles].reshape(-1, 2), r_after[num_circles:], ls_sum_after

# Entry point called by the grading script
def execute():
    return run_packing
```
