import numpy as np

# Parameters for the annealing
s_start = 0.15
s_end = 0.001
T_start = 1.0
T_end = 1e-4
N = 20000

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # For reproducibility
    n = 26
    centers = np.random.uniform(0, 1, size=(n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0

    def inflation_step(centers, radii):
        for _ in range(50):
            new_radii = np.zeros(n)
            for i in range(n):
                x_i, y_i = centers[i]
                r_i_max = min(x_i, 1-x_i, y_i, 1-y_i, np.inf)
                for j in range(n):
                    if i != j:
                        r_j, x_j, y_j = radii[j], centers[j, 0], centers[j, 1]
                        d = np.linalg.norm([x_i - x_j, y_i - y_j])
                        r_i_max = min(r_i_max, d - r_j)
                new_radii[i] = r_i_max
            new_radii.clip(0, None, new_radii)
            if np.allclose(radii, new_radii):
                return new_radii
            radii = new_radii
        return radii
    
    best_conf = centers.copy()
    
    for _ in range(100):  # 100 restarting attempts
        centers = np.random.uniform(0, 1, size=(n, 2))
        sort_index = np.argsort(np.sum(centers**2, axis=1), kind='mergesort')
        centers = centers[sort_index]
        radii = inflation_step(centers, best_radii)
        current_sum = np.sum(radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_radii = radii
            best_conf = centers

    return best_conf, best_radii, best_sum_radii
