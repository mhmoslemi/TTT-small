import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C)
    lhs_ineq = np.eye(2*N)
    rhs_ineq = np.empty(2*N)
    for i in range(N):
        for j in range(i+1, N):
            lhs_ineq[2*i, j] = 1
            lhs_ineq[2*i+1, j] = 1
            rhs_ineq[2*i] = d[i, j]
            rhs_ineq[2*i+1] = d[i, j]
        lhs_ineq[2*i, i] = -1
        rhs_ineq[2*i] = C[i, 0]
        lhs_ineq[2*i+1, i] = -1
        rhs_ineq[2*i+1] = 1 - C[i, 0]
    for i in range(N):
        lhs_ineq[2*N+i, i] = -1
        rhs_ineq[2*N+i] = C[i, 1]
        lhs_ineq[2*N+i+1, i] = -1
        rhs_ineq[2*N+i+1] = 1 - C[i, 1]
    res = linprog(-np.ones(2*N), A_ub=lhs_ineq, b_ub=rhs_ineq, bounds=(0, None), method='highs', verbose=0)
    if res.success and not np.isnan(res.x).any():
        r = res.x
        return r, lhs_ineq, rhs_ineq
    else:
        raise ValueError("LP infeasible")

def compute_sum_radius(Best_C):
    Best_r, _, _ = build_lp(Best_C)
    Best_r = np.clip(Best_r, 0, None)
    sum_r = -np.sum(Best_r)
    return Best_r, sum_r
