import numpy as np
from scipy.optimize import linprog
import random

def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    c = -np.ones(n_circles)

    A_ub = []
    b_ub = []

    for i in range(n_circles):
        A_ub.append([1 if j == i else 0 for j in range(n_circles)])
        b_ub.append(centres_i[i, 0])
        A_ub.append([1 if j == i else 0 for j in range(n_circles)])
        b_ub.append(1 - centres_i[i, 0])
        A_ub.append([1 if j == i else 0 for j in range(n_circles)])
        b_ub.append(centres_i[i, 1])
        A_ub.append([1 if j == i else 0 for j in range(n_circles)])
        b_ub.append(1 - centres_i[i, 1])

    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            d_ij = np.linalg.norm(centres_i[i] - centres_i[j])
            A_ub.append([2 if j == i else 0 if i < j else 2 if j > i else 0 for k in range(n_circles)])
            b_ub.append(d_ij)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))

    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_particles = 30
    n_circles = 26
    max_iter = 3000
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    vel_clip = 0.1
    alpha = 0.999

    np.random.seed(42)

    grid = (np.arange(6) * 0.2).repeat(5)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid = np.column_stack((grid_x.T.flatten(), grid_y.T.flatten()))
    grid = grid[np.random.choice(len(grid), n_circles, replace=False)]
    perturbation = np.random.normal(0, 0.01, (n_circles, 2))
    grid += perturbation

    centres = np.tile(grid, (n_particles, 1, 1)) + 1e-6 * np.random.normal(0, 0.01, (n_particles, n_circles, 2))
    velocities = np.zeros_like(centres)
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres[np.random.choice(n_particles), :, :]
    gbest_sum = np.inf

    for iter in range(max_iter):
        for p in range(n_particles):
            radii, sum_radii = solve_radius_lp(centres[p])
            if sum_radii > pbest_sums[p]:
                pbests[p] = centres[p].copy()
                pbest_sums[p] = sum_radii

            if sum_radii > gbest_sum:
                gbest = centres[p].copy()
                gbest_sum = sum_radii

        for p in range(n_particles):
            for i in range(n_circles):
                r1, r2 = np.random.uniform(0, 1, 2)
                velocities[p, i] *= inertia
                velocities[p, i] += cognitive * r1 * (pbests[p, i] - centres[p, i])
                velocities[p, i] += social * r2 * (gbest[i] - centres[p, i])
                velocities[p, i] = np.clip(velocities[p, i], -vel_clip, vel_clip)

            centres[p] += velocities[p]
            centres[p] = np.clip(centres[p], 0, 1)

        if iter % 500 == 0:
            restart_indices = np.random.choice(n_particles, int(0.1 * n_particles), replace=False)
            centres[restart_indices] = np.tile(grid, (len(restart_indices), 1, 1)) + 1e-6 * np.random.normal(0, 0.01, (len(restart_indices), n_circles, 2))

        alpha *= 0.999

    return gbest, solve_radius_lp(gbest)[0], solve_radius_lp(gbest)[1]
