import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def compute_sum_radii(positions):
    n = 26
    positions = positions.reshape((n, 2))
    radii = np.zeros(n)
    dmat = np.sqrt(np.sum((positions[:, np.newaxis] - positions[np.newaxis, :]) ** 2, axis=2))
    np.fill_diagonal(dmat, np.inf)
    
    A_eq = np.vstack((np.eye(n), -np.eye(n)))
    b_eq = positions[:, 0]
    
    A_ub_1 = np.zeros((n * (n - 1), n))
    np.fill_diagonal(A_ub_1, np.ones(n))
    np.fill_diagonal(A_ub_1[:, 1:n], -np.ones(n-1))
    np.fill_diagonal(A_ub_1[:, ::n], -np.ones(n-1))
    A_ub_2 = np.zeros((n * (n - 1), n))
    np.fill_diagonal(A_ub_2, np.ones(n))
    np.fill_diagonal(A_ub_2[:, ::n], np.ones(n-1))
    np.fill_diagonal(A_ub_2[:, 1:n], -np.ones(n-1))
    A_ub = np.vstack((A_ub_1, A_ub_2))
    
    b_ub = dmat.flatten()
    
    bounds = np.concatenate((np.full((n, 2), [0, 1]), np.maximum(0, dmat - positions[:, 0] - positions[:, 1])))
    constraints = {'type': 'ineq', 'fun': lambda r: np.dot(A_ub, np.clip(r, 0, None)) - b_ub}
    
    res = linprog(np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and not np.any(radii == 0):
        radii = res.x
        return -np.sum(radii)
    else:
        return -np.inf

def run_packing():
    np.random.seed(0)
    centers_log = differential_evolution(lambda c: -compute_sum_radii(c), bounds=[(0, 1)] * 52, maxiter=1000).x
    centers = centers_log.reshape(26, 2)
    return centers, compute_sum_radii(centers), -compute_sum_radii(centers)

centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of Radii:", sum_radii)
