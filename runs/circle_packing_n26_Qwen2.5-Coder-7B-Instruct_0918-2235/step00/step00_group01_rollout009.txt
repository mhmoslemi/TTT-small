import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def objective(u):
    x = u[:26]
    y = u[26:]
    c = np.ones(26)
    dist = np.sqrt((x[:, None] - x) ** 2 + (y[:, None] - y) ** 2)
    A_ub = np.zeros((25, 52))
    for i in range(26):
        A_ub[2 * i, 2 * i] = 1
        A_ub[2 * i, 2 * i + 1] = 1
        A_ub[2 * i + 1, 2 * i] = 1
        A_ub[2 * i + 1, 2 * i + 1] = 1
    b = np.minimum(x, 1 - x)
    mask = np.triu(np.ones(26, dtype=bool), k=1).astype(int)
    A_ub[mask] = dist[mask] - dist.T[mask]
    b[mask] = dist[mask]
    r_bounds = (0, None)
    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=r_bounds, method='highs')
    return -res.fun

bounds = [(0, 1)] * 52
result = differential_evolution(objective, bounds, popsize=20, maxiter=200, tol=1e-12)

centers = (result.x[:26], result.x[26:])
radii = np.zeros(26)
for i in range(26):
    x = centers[0][i]
    y = centers[1][i]
    dist = np.sqrt((centers[0] - x) ** 2 + (centers[1] - y) ** 2)
    A_ub = np.zeros((26, 26))
    A_ub[i, i] = 1
    A_ub[:, i] += np.minimum(x, 1 - x)
    A_ub[i, :] += np.minimum(y, 1 - y)
    b = np.minimum(x, 1 - x)
    mask = np.triu(np.ones(26, dtype=bool), k=1).astype(int)
    A_ub[mask] = dist[mask] - dist.T[mask]
    b[mask] = dist[mask]
    r_bounds = (0, None)
    res = linprog(np.ones(26), A_ub=A_ub, b_ub=b, bounds=r_bounds, method='highs')
    radii[i] = res.x[i]

sum_radii = radii.sum()
return centers, radii, sum_radii
