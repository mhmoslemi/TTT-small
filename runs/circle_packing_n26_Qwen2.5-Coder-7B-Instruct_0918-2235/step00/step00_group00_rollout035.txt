import numpy as np
from scipy.optimize import differential_evolution

def objective(v, n):
    x, y, r = v[:n].reshape(-1, 2), v[n:2*n], v[2*n:]
    overlap = np.sum(np.maximum(0, r[:, np.newaxis] + r - np.sqrt(np.sum((x[:, np.newaxis] - x)**2, 2)))**2)
    left_boundary = np.sum(np.maximum(0, r - x[:, 0])**2)
    right_boundary = np.sum(np.maximum(0, r - (1 - x[:, 0]))**2)
    bottom = np.sum(np.maximum(0, r - y[:, 0])**2)
    top = np.sum(np.maximum(0, r - (1 - y[:, 0]))**2)
    penalty = overlap + left_boundary + right_boundary + bottom + top
    return -np.sum(r) + 1000 * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    result = differential_evolution(
        objective, bounds, args=(n,), maxiter=50, popsize=10, recombination=0.9, seed=np.random.randint(0, 10000),
        fitness_threshold=1e-12, disp=False
    )
    best_v = result.x
    x, y, r = best_v[:n].reshape(-1, 2), best_v[n:2*n], best_v[2*n:]
    sum_radii = -result.fun
    centric_radius = np.linalg.solve(np.eye(n) + np.pad(np.eye(n), ((n, 0), (n, 0)), mode='constant', constant_values=-1), r)
    if sum(centric_radius) > sum_radii:
        r = centric_radius
        sum_radii = sum(r)
    if sum(r) <= sum_radii:
        r = np.maximum(r, centric_radius)
    return np.vstack((x, y)).T, r, np.sum(r)

centers, radii, sum_radii = run_packing()
