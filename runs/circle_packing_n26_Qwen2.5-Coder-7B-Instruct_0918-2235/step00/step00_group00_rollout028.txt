import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def objective(v, n=26, lambda_=1000):
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    n2 = 2*n
    penalty = 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += np.maximum(0, r + radii[i+1] - np.sqrt((x - centers[i+1, 0])**2 + (y - centers[i+1, 1])**2))**2
        for j in range(i + 1, n):
            penalty += np.maximum(0, r + radii[j] - np.sqrt(np.sum((centers[i] - centers[j]) ** 2))**2
    return -np.sum(radii) + lambda_ * penalty

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2 * n)
    bounds.extend([(0, 0.5)] * n)
    lambda_ = 1000
    result = differential_evolution(objective, bounds, workers=1, popsize=10, strategy='best1bin', tol=1e-12)
    best_v = result.x
    best_centers = best_v[:2*n].reshape(n, 2)
    best_radii = best_v[2*n:]
    if result.fun < 0:  # infeasible solution
        return np.ones((n, 2)), np.zeros(n), np.sum(np.zeros(n))
    else:
        A = np.array([[2, 0, -2], [0, 2, -2], [-2, 0, 2], [0, -2, 2]])
        b = np.array([1, 1, 1, 1])
        for i in range(n):
            A = np.vstack([A, [0, 0, np.sqrt((i+1) % n - i)**2 - 4 * best_radii[i]]])
            b = np.append(b, 0)
            # add constraints for each circle being inside the square
            A = np.vstack([A, [0, 0, np.sqrt((i+1) % n - i)**2 - 4 * best_radii[i]]])
            b = np.append(b, 0)
        res = linprog(np.zeros(n), A_ub=A, b_ub=b, bounds=[[0, best_radii[i]] for i in range(n)], method='highs')
        if res.success:
            return best_centers, res.x, np.sum(res.x)
        else:
            raise ValueError("Linear programming failed")

centers, radii, sum_radii = run_packing()
print(f" Centers:\n {centers}")
print(f" Radii:\n {radii}")
print(f" Sum of Radii: {sum_radii}")
