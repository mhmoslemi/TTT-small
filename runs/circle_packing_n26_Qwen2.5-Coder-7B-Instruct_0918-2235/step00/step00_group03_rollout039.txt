import numpy as np
from scipy.optimize import minimize

def penalty(vars, n):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    penalty = 0.0
    overlap = lambda i, j: max(0, r[i] + r[j] - np.hypot(x[i] - x[j], y[i] - y[j]))
    for i in range(n):
        penalty += r[i]**2
        penalty += overlap(i, 0)**2
        penalty += overlap(i, 1)**2
        penalty += overlap(i, 2)**2
        penalty += overlap(i, 3)**2
        for j in range(i + 1, n):
            penalty += overlap(i, j)**2
    return penalty

def energy(vars, n):
    return -np.sum(vars[2*n:]) + vars['lambda'] * penalty(vars, n)

def compute_gradient(vars, n):
    rad_grads = -2 * vars[2*n:]
    radiuses = vars[2*n:]
    for j in range(n):
        for k in range(n):
            d = np.hypot(vars[j] - vars[k], vars[j+n] - vars[k+n])
            dx = (vars[j] - vars[k]) / d * rad_grads[j] + (vars[j] - vars[k]) / d * rad_grads[k]
            dy = (vars[j+n] - vars[k+n]) / d * rad_grads[j] + (vars[j+n] - vars[k+n]) / d * rad_grads[k]
            rad_grads[j] += dx
            rad_grads[k] += dx
            rad_grads[j+n] += dy
            rad_grads[k+n] += dy
    return np.concatenate((vars[:2*n], rad_grads))

def optimize_for_lambda(lambda_val, n, init_center, init_radius):
    vars = np.empty(3*n)
    vars[:n] = np.clip(init_center + np.random.randn(n) * 0.1, 0, 1)
    vars[n:2*n] = np.clip(init_center + np.random.randn(n) * 0.1, 0, 1)
    vars[2*n:] = init_radius + np.random.randn(n) * 1e-4
    vars['lambda'] = lambda_val
    result = minimize(energy, vars, jac=compute_gradient, method='L-BFGS-B', bounds=[(0, 1)] * 2*n + [(0, 0.5)] * n)
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    init_center = np.random.rand(n)
    init_radius = np.full(n, 1e-4)
    K = 20
    R = 50
    best_sum_radii = 0.0
    best_vars = None
    for _ in range(R):
        vars = init_center.copy()
        for k in range(K):
            vars = optimize_for_lambda(2**k, n, init_center, init_radius)
        current_sum_radii = energy(vars, n)
        if current_sum_radii > best_sum_radii:
            best_sum_radii = current_sum_radii
            best_vars = vars
    centers = best_vars[:n]
    radii = best_vars[n:2*n]
    return centers, radii, best_sum_radii
