def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers and radii
    n = 26
    centres = np.zeros((n, 2))
    radii = np.full(n, 0.01)
    z = np.hstack((centres.flatten(), radii))

    # Define the objective function
    def objective(z):
        centres = z[:2*n].reshape((n, 2))
        radii = z[2*n:]
        return -np.sum(radii) + compute_overlap_penalties(centres, radii)

    # Define the Jacobian function
    def jacobian(z):
        centres = z[:2*n].reshape((n, 2))
        radii = z[2*n:]
        d_centres_d_z = np.block([[np.eye(2*n), np.zeros((2*n, n))]], [np.zeros((n, 2*n)), np.diag(radii)])
        d_radii_d_z = np.block([[np.zeros((2*n, 2*n)), np.eye(n)]])
        d_overlap_d_z = compute_overlap_gradients(centres, radii)
        return -(d_centres_d_z + d_radii_d_z) @ (d_overlap_d_z.T)
    
    # Define bounds for the variables
    bounds = [(0, 1)]*52 + [(0, None)]*26
    
    # Run the optimizer
    res = minimize(objective, z, jac=jacobian, method='trust-constr', bounds=bounds, options={'gtol': 1e-6, 'ftol': 1e-8, 'maxiter': 3000})
    
    # Extract the optimal centers and radii
    centres_opt = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:]
    
    # Solve the lp to tighten radii
    radii_opt, _ = solve_lp(centres_opt)
    
    # Sum of radii
    sum_radii = np.sum(radii_opt)
    
    return centres_opt, radii_opt, sum_radii

def compute_overlap_penalties(centres, radii):
    n = len(centres)
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centres[i] - centres[j])
            penalty += np.max(0, radii[i] + radii[j] - dist)**2
    return penalty

    for row in range(num_rows):
        for col in range(num_cols):
            if (row + col) % 2 == 0:

                centers.append([(col + 0.5) / num_cols, (row + 0.5) / num_rows + ((row % 2) * 0.5) * 0.5])

def unpack(z):

def solve_lp(centers):
    n = centers.shape[0]
    c = np.full(1, 1)
    A = np.zeros((2*n, n))
    b = np.zeros((2*n,))
    for i in range(n):
        A[2*i, i] = 2
        A[2*i+1, i] = 2
        b[2*i] = 1 - 2 * centers[i, 0]
        b[2*i+1] = 1 - 2 * centers[i, 1]
    radius_sol = np.linalg.solve(A, b)
    return radius_sol[0], radius_sol
