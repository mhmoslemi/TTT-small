import numpy as np
from scipy.optimize import linprog, minimize
from random import random, uniform

def compute_lp_solution(centers):
    n = len(centers)
    c = -np.ones(n)  # Maximize sum
    A_ub = []        
    b_ub = []
    
    for i in range(n):
        A_ub.append([1] * n)  # r_i <= 1 - x_i
        b_ub.append(1 - centers[i, 0])
        A_ub.append([1] * n)  # r_i <= 1 - y_i
        b_ub.append(1 - centers[i, 1])
        A_ub.append([-1] * n) # r_i <= x_i
        b_ub.append(centers[i, 0])
        A_ub.append([-1] * n) # r_i <= y_i
        b_ub.append(centers[i, 1])
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub.append([1, 0] * i + [1, 0] * j)
            b_ub.append(dist**2)
            A_ub.append([0, 1] * i + [1, 0] * j)
            b_ub.append(dist**2)
            A_ub.append([0, 0] * i + [1, 1] * j)
            b_ub.append(dist**2)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x if res.success else np.ones(n) * 0.1, -res.fun if res.success else 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    random.seed(0)
    n = 26
    best_sum = 0
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    step = 0.1
    
    # Initial configuration: 5x6 grid minus 4 random
    grid = np.array([
        [0, 0.16], [0.16, 0.16], [0.32, 0.16], [0.48, 0.16], [0.64, 0.16], [0.8, 0.16],
        [0, 0.32], [0.16, 0.32], [0.32, 0.32], [0.48, 0.32], [0.64, 0.32], [0.8, 0.32],
        [0, 0.48], [0.16, 0.48], [0.32, 0.48], [0.48, 0.48], [0.64, 0.48], [0.8, 0.48],
        [0, 0.64], [0.16, 0.64], [0.32, 0.64], [0.48, 0.64], [0.64, 0.64], [0.8, 0.64],
        [0, 0.8], [0.16, 0.8], [0.32, 0.8], [0.48, 0.8], [0.64, 0.8], [0.8, 0.8]
    ])
    centers = np.delete(grid, np.random.choice(grid.shape[0], size=4, replace=False), axis=0)
    
    current_centers = centers
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
    
    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        delta_x = uniform(-step, step)
        delta_y = uniform(-step, step)
        new_center = np.clip(current_centers[i] + np.array([delta_x, delta_y]), 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = new_center
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = current_centers, current_radii, new_sum
        
        if iter % N_REDUCE == 0 and iter > step < 1e-4 / 0.05:  # Every 500 iterations or so
            centers = centers * 0.5 + np.random.rand(n, 2) / 2
            centers = np.clip(centers, 0, 1)
            current_centers = centers
            current_radii, current_sum = compute_lp_solution(centers)
        
        if iter % N_REDUCE == 0:
            step *= 0.9
    
    return best_centers, best_radii, best_sum
