import numpy as np
from scipy.optimize import differential_evolution, linprog

def objective(v, λ):
    n = v.shape[0] // 3
    xs = v[:n].reshape(-1, 2)[:, 0]
    ys = v[:n].reshape(-1, 2)[:, 1]
    rs = v[n:2*n]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dij = np.linalg.norm(xs[i] - xs[j] + 1j * (ys[i] - ys[j]))
            penalty += max(0, rs[i] + rs[j] - dij)**2
    penalty += np.sum(1e10 * np.maximum(0, rs - xs)) + np.sum(1e10 * np.maximum(0, rs - (1 - xs))) + np.sum(1e10 * np.maximum(0, rs - ys)) + np.sum(1e10 * np.maximum(0, rs - (1 - ys)))
    return -np.sum(rs) + λ * penalty
