import numpy as np
from scipy.optimize import linprog

def generate_lattice_coords(n_rows, n_cols, offset_x, offset_y, spacing):
    lattice = np.array([(x, y) for x in np.arange(n_cols) for y in np.arange(n_rows)])
    rotation = np.array([[np.cos(np.pi / 3), -np.sin(np.pi / 3)],
                         [np.sin(np.pi / 3), np.cos(np.pi / 3)]])
    lattice = np.dot(lattice, rotation) * spacing
    lattice[:, 0] += offset_x
    lattice[:, 1] += offset_y
    lattice -= lattice.min(axis=0)
    lattice.T[0] /= lattice.T[0].max()
    lattice.T[1] /= lattice.T[1].max()
    return lattice

def solve_lps(centres, initial_radius=0.01):
    objective = -np.ones(centres.shape[0])
    constraints = [
        {'type': 'ineq', 'fun': lambda x, i=i: x[i] - centres[i, 0]},
        {'type': 'ineq', 'fun': lambda x, i=i: 1 - centres[i, 0] - x[i]},
        {'type': 'ineq', 'fun': lambda x, i=i: x[i] - centres[i, 1]},
        {'type': 'ineq', 'fun': lambda x, i=i: 1 - centres[i, 1] - x[i]},
    ]
    for i in range(centres.shape[0]):
        for j in range(i + 1, centres.shape[0]):
            dx, dy = centres[i] - centres[j]
            dist = np.hypot(dx, dy)
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist - x[i] - x[j]})
    res = linprog(objective, A_ub=[c['fun'](None) for c in constraints], bounds=(0, None), method='highs')
    radii = -res.x
    return radii, res.fun

def run_packing():
    best_z, best_sum = None, -np.inf
    s_initial = 0.08
    n_seed = 100

    # Coarse grid search
    for theta in np.linspace(0, np.pi/6, 51):
        for dx in np.linspace(-0.025, 0.025, 51):
            for dy in np.linspace(-0.025, 0.025, 51):
                centres = generate_lattice_coords(15, 2, dx, dy, s_initial)
                if any_outside(centres):
                    continue
                radii, s = solve_lps(centres)
                if s > best_sum:
                    best_sum, best_z = s, (centres, radii)

    # Local refinement with Nelder–Mead
    def nelder_mead(theta, dx, dy, tolerance=1e-7):
        result = minimize(
            lambda x: -solve_lps(generate_lattice_coords(15, 2, x[0], x[1], s_initial))[1],
            [theta, dx, dy],
            method="nelder-mead",
            options={'maxiter': 1000, 'xatol': tolerance},
        )
        return result.x

    for _ in range(n_seed):
        theta, dx, dy = nelder_mead(best_z[0][0], best_z[0][1], best_z[0][2], tolerance=1e-7)
        centres = generate_lattice_coords(15, 2, dx, dy, s_initial)
        radii, s = solve_lps(centres)
        if s > best_sum:
            best_z, best_sum = (centres, radii), s

    # Final jiggle
    def random_perturb(z):
        dx, dy = np.random.uniform(-0.002, 0.002, 2)
        return nelder_mead(z[0] + dx, z[1] + dy, z[2], tolerance=1e-9) if z[1] + dx > -0.002 and z[1] + dx < 0.002 and z[2] + dy > -0.002 and z[2] + dy < 0.002 else z

    for _ in range(200):
        best_z = random_perturb(best_z)
        radii, s = solve_lps(best_z[0])
        if s > best_sum:
            best_z, best_sum = (best_z[0], radii), s

    # Final LP re-solve to guarantee optimal radii
    best_centres = best_z[0]
    best_radii, best_sum = solve_lps(best_centres)

    return best_centres, best_radii, best_sum
