import numpy as np
import cvxpy as cp

def calculate_max_radius(centers, radii):
    x, y = centers
    r = np.maximum(0.0, np.minimum(1 - x, 1 - y, x, y))
    dists = np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1)
    r_max = np.maximum(r, dists - radii[:, np.newaxis] - radii[np.newaxis, :])
    r_max[np.where(dists <= radii[:, np.newaxis] + radii[np.newaxis, :])] = 0
    return np.min(r_max, axis=1)

def run_packing() -> tuple:
    h = 0.05
    epsilon = 1e-6
    n = 26

    # Generate dense grid
    x = np.arange(h / 2, 1, h)
    y = np.arange(h / 2, 1, h)
    candidates = np.array(list(product(x, y)))

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < n:
        # Compute maximum feasible radii for all candidate points
        r_candidates = calculate_max_radius(candidates, placed_radii)

        # Filter out infeasible candidates
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break

        # Choose the candidate with the largest radius
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]

        # Add the new circle to the list
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])

        # Remove all candidates that are too close to the new circle
        dists = np.linalg.norm(candidates - placed_centres[-1], axis=1)
        too_close = dists < (placed_radii[-1] + 1e-9)
        candidates = candidates[~too_close]

    # Convert lists to arrays
    placed_centres = np.array(placed_centres)
    placed_radii = np.array(placed_radii)

    # Local radius refinement
    alpha = 0.02
    n_iter = 5000
    for _ in range(n_iter):
        # Create problem
        r = cp.Variable(n)
        constraints = [
            r >= 0,
            r <= placed_centres[:, 0],
            r <= 1 - placed_centres[:, 0],
            r <= placed_centres[:, 1],
            r <= 1 - placed_centres[:, 1]
        ]
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(r[i] + r[j] <= np.linalg.norm(placed_centres[i] - placed_centres[j]))

        objective = cp.Maximize(sum(r))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, abstol=1e-9)

        if np.sum(r.value) > np.sum(placed_radii):
            placed_radii = r.value

    total_radius_sum = np.sum(placed_radii)
    return placed_centres, placed_radii, total_radius_sum
