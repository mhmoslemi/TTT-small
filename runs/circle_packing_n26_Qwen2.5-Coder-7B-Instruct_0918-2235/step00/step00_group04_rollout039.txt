```python
import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist
import random

# Constants
N_CIRCLES = 26
MAX_ITERS = 20000
K_JUMP = 500
N_REDUCE = 500
STEP = 0.1
BOUNDARY_BUFFER = 1e-6
EPSILON = 1e-12
METHOD = 'highs'

def initial_centers():
    grid = np.array([(i, j) for i in np.linspace(0, 1, 5) for j in np.linspace(0, 1, 6)])
    centers = grid.tolist()
    random.shuffle(centers)
    centers = centers[:N_CIRCLES]
    return np.array(centers)

def solve_LP(centers):
    n = len(centers)
    c = -np.ones(n)
    A_ub = np.empty((n + n*(n-1)//2, n))
    b_ub = np.empty(n + n*(n-1)//2)
    
    # r_i <= x_i
    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = centers[i, 0]
    
    # r_i <= 1-x_i
    for i in range(n):
        A_ub[n+i, i] = 1
        b_ub[n+i] = 1 - centers[i, 0]
    
    # r_i <= y_i
    for i in range(n):
        A_ub[2*n+i, i] = 1
        b_ub[2*n+i] = centers[i, 1]
    
    # r_i <= 1-y_i
    for i in range(n):
        A_ub[3*n+i, i] = 1
        b_ub[3*n+i] = 1 - centers[i, 1]
    
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A_ub[4*n + i*(n-1) + j - i, i] = 1
            A_ub[4*n + i*(n-1) + j - i, j] = 1
            b_ub[4*n + i*(n-1) + j - i] = d
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method=METHOD)
    if res.success:
        return res.x, np.sum(res.x)
    else:
        raise ValueError("Failed to solve LP")

def run_packing():
    random.seed(42)
    centers = initial_centers()
    current_radii, current_sum = solve_LP(centers)
    best_centers, best_radii, best_sum = centers, current_radii, current_sum
    step = STEP
    
    for iter in range(MAX_ITERS):
        i = random.randint(0, N_CIRCLES - 1)
        new_x = np.clip(centers[i, 0] + np.random.uniform(-step, step), 0, 1)
        new_y = np.clip(centers[i, 1] + np.random.uniform(-step, step), 0, 1)
        centers_new = centers.copy()
        centers_new[i,:] = [new_x, new_y]
        new_radii, new_sum = solve_LP(centers_new)
        
        if new_sum > current_sum + EPSILON:
            centers = centers_new
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = centers, current_radii, current_sum
        
        if iter % K_JUMP == 0:
            centers = initial_centers()
            current_radii, current_sum = solve_LP(centers)
        
        if iter % N_REDUCE == 0 and step > 1e-4:
            step *= 0.9
    
    return best_centers, best_radii, best_sum
```
