import numpy as np
from scipy.optimize import linprog

def pairwise_distances(centers):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
    return dists

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    Def max_cycles, num_trials in the strategy.
    max_cycles = 50
    num_trials = 40
    eps_stop = 1e-4
    step_size_start = 0.1
    alpha = 0.8
    n = 26
    step_size = step_size_start
    
    initialize best_sum and counters
    
    for trial in range(num_trials):
        # (A) Random seed generation
        centers = np.random.rand(n, 2)

        # (B1) Inner LP solve @ initial layout
        current_sum = 0
        best_sum = -np.inf
        while step_size > eps_stop:
            # Construct A_ub and bounds
            A_ub = np.zeros((n*(n-1)//2, n))
            b_ub = np.zeros(n*(n-1)//2)
            bounds = np.zeros((n, 2))
            for i in range(n):
                bounds[i] = (0, min(centers[i, 0], 1-centers[i, 0], centers[i, 1], 1-centers[i, 1]))
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    A_ub[i*(n-1)//2 + j-i-1, i] = 1
                    A_ub[i*(n-1)//2 + j-i-1, j] = 1
                    b_ub[i*(n-1)//2 + j-i-1] = d

            # (B2) Solve the LP.inner
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
            
            # Update current sum and check improvment
            if res.success and res.fun < current_sum + 1e-9:
                continue
                
            current_sum = -res.fun
            new_centers = centers.copy()
            for i in range(n):
                d = distances[i, :]
                for j in range(i+1, n):
                    if i==j: continue
                    delta_x = np.random.uniform(-step_size, step_size)
                    delta_y = np.random.uniform(-step_size, step_size)
                    new_centers[i, 0] += delta_x
                    new_centers[i, 1] += delta_y
                    new_centers[i, 0] = np.clip(new_centers[i, 0], 0, 1)
                    new_centers[i, 1] = np.clip(new_centers[i, 1], 0, 1)
                    
                    A_ub_new, b_ub_new = A_ub.copy(), b_ub.copy()
                    A_ub_new[i*(n-1)//2 + j-i-1, i] -= 1
                    A_ub_new[i*(n-1)//2 + j-i-1, j] -= 1
                    A_ub_new[i*(n-1)//2 + j-i-1, i] += 1
                    A_ub_new[i*(n-1)//2 + j-i-1, j] += 1
                    b_ub_new[i*(n-1)//2 + j-i-1] += 2 * np.sum((new_centers[i] - new_centers[j]) ** 2)
                    
                    res_new = linprog(-np.ones(n), A_ub=A_ub_new, b_ub=b_ub_new, bounds=bounds, method='highs', options={'disp': False})
                    
                    if res_new.success and res_new.fun < res.fun:
                        centers = new_centers.copy()

            step_size *= alpha
            centers = centers.copy()
            
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()

    return best_centers, np.maximum(linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False}).x, 0), best_sum
