import numpy as np
from scipy.optimize import differential_evolution

def objective(v, n):
    penalty = 0
    xy_r = v.reshape((n, 3))
    for i in range(n):
        x, y, r = xy_r[i]
        if not (0 <= x <= 1 and 0 <= y <= 1 and 0 <= r <= min(x, 1-x, y, 1-y)):
            return -np.inf
        for j in range(i+1, n):
            dx, dy = xy_r[j] - xy_r[i]
            dist = np.sqrt(dx**2 + dy**2)
            penalty += max(0, r + xy_r[j, 2] - dist)**2
    return -np.sum(xy_r[:, 2]) + 1000 * penalty

def centers_to_circle_params(centers):
    n = centers.shape[0]
    xy_r = centers.copy()
    for i in range(n):
        r_max = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        xy_r[i, 2] = r_max
    return xy_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)
    
    de_result = differential_evolution(objective, bounds, args=(n,), popsize=10, maxiter=50, polish=False)
    de_circles = de_result.x.reshape((n, 3))
    
    best_circle_params = de_circles[de_circles[:, 2].argmax()]
    centers = centers_to_circle_params(best_circle_params).reshape((n, 2))
    radii = best_circle_params.T[2]
    
    return centers, radii, np.sum(radii)
