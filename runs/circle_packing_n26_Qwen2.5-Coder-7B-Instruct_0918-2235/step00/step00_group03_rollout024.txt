import numpy as np
from scipy.optimize import minimize, Bound

# Structure to hold a variable set (x, y, r)
class VariableSet:
    def __init__(self.bounds):
        self.num_variables = 2 * bounds + bounds
        self.bounds = bounds

    def reverse_transform(self, vars):
        xs = vars[:self.bounds]
        ys = vars[self.bounds:2*self.bounds]
        rs = vars[2*self.bounds:]
        return xs, ys, rs

    def transform(self, xs, ys, rs):
        return np.concatenate((xs, ys, rs))

    def gradient(self, vars, penalty_func):
        xs, ys, rs = self.reverse_transform(vars)
        overlap_penalties = penalty_func(xs, ys, rs)
        pen_grad = self.penalty_jacobian(xs, ys, rs, overlap_penalties)
        return -np.sum(rs) + np.dot(vars, pen_grad/2)

    def penalty_jacobian(self, xs, ys, rs, overlap_penalties):
        gradient = np.zeros_like(xs)
        for i in range(len(xs)):
            for j in range(i + 1, len(xs)):
                dx, dy = xs[i]-xs[j], ys[i]-ys[j]
                dist = np.sqrt(dx**2 + dy**2)
                if dist - rs[i] - rs[j] > 0:
                    continue
                outside_dist = dist - rs[i] - rs[j]
                penalty = outside_dist**2
                dsdx, dsdy = dx/dist, dy/dist
                gradient[i] += 2 * penalty * dsdx - 2*rs[i]
                gradient[j] += 2 * penalty * dsdy - 2*rs[j]
        gradient_x = gradient[:self.bounds]
        gradient_y = gradient[self.bounds:2*self.bounds]
        gradient_r = gradient[2*self.bounds:]
        pen_grad = np.concatenate((gradient_x, gradient_y, gradient_r))
        return pen_grad / 2

def overlap_penalty(xs, ys, rs):
    n = len(xs)
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = xs[i]-xs[j], ys[i]-ys[j]
            dist = np.sqrt(dx**2 + dy**2)
            penalty += max(0, dist - rs[i] - rs[j])**2
    return penalty

def optimize_for_lambda(lambda_, init_vars, variable_set):
    res = minimize(
        fun=lambda x: variable_set.gradient(x, overlap_penalty),
        x0=init_vars,
        method='L-BFGS-B',
        jac=True,
        bounds=[Bound(0, 1)] * variable_set.bounds * 2 + [Bound(0, 0.5)] * variable_set.bounds
    )
    if not res.success:
        raise ValueError("Optimization failed: " + res.message)
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = 26
    variable_set = VariableSet(bounds)
    max_radius = 0.5
    num_restarts = 50

    best_sum_r = 0
    best_centers_r = None

    for _ in range(num_restarts):
        init_vars = np.random.rand(variable_set.num_variables) * max_radius
        try:
            vars = optimize_for_lambda(1.0, init_vars, variable_set)
            xs, ys, rs = variable_set.reverse_transform(vars)
            sum_r = np.sum(rs)
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_centers_r = xs, ys
        except ValueError as e:
            print(e)

    if best_centers_r is None:
        return np.zeros((bounds, 2)), np.zeros(bounds), 0

    return best_centers_r, rs, best_sum_r
