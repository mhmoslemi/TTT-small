import numpy as np
from scipy.optimize import minimize

def penalty(vars, n, lambda_):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    penalty_overlap = np.sum(np.maximum(r + r[:, None] - np.hypot(x[:, None] - x, y[:, None] - y), 0)**2)
    penalty_boundary_left = np.sum(np.maximum(x - r, 0)**2)
    penalty_boundary_right = np.sum(np.maximum(1 - x - r, 0)**2)
    penalty_boundary_bottom = np.sum(np.maximum(y - r, 0)**2)
    penalty_boundary_top = np.sum(np.maximum(1 - y - r, 0)**2)
    return -np.sum(r) + lambda_ * (penalty_overlap + penalty_boundary_left + penalty_boundary_right + penalty_boundary_bottom + penalty_boundary_top)

def objective(vars, n):
    return -np.sum(vars[2*n:])

def gradient(vars, n, lambda_):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    grad = np.zeros_like(vars)
    
    # Overlap constraints gradient
    grad_r = np.zeros_like(r)
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.hypot(x[j] - x[i], y[j] - y[i])
            s_ij = d_ij - r[i] - r[j]
            if s_ij < 0:
                d_dij_dx_i = (x[i] - x[j]) / d_ij
                d_dij_dy_i = (y[i] - y[j]) / d_ij
                d_dij_dx_j = -(x[i] - x[j]) / d_ij
                d_dij_dy_j = -(y[i] - y[j]) / d_ij
                grad_r[i] += lambda_ * (-2 * s_ij)
                grad_r[j] += lambda_ * (-2 * s_ij)
                grad_x[i] += lambda_ * 2 * s_ij * d_dij_dx_i
                grad_x[j] += lambda_ * 2 * s_ij * d_dij_dx_j
                grad_y[i] += lambda_ * 2 * s_ij * d_dij_dy_i
                grad_y[j] += lambda_ * 2 * s_ij * d_dij_dy_j
    grad[2*n:] = grad_r
    grad[:n] = grad_x
    grad[n:2*n] = grad_y
    
    # Boundary constraints gradient
    grad_r += lambda_ * 2 * np.maximum(r - 0.01, 0)
    grad_r -= lambda_ * 2 * np.maximum(r - 0.5, 0)
    grad_x += lambda_ * 2 * np.maximum(x - 0.01, 0)
    grad_x -= lambda_ * 2 * np.maximum(x - 1 + 0.01, 0)
    grad_y += lambda_ * 2 * np.maximum(y - 0.01, 0)
    grad_y -= lambda_ * 2 * np.maximum(y - 1 + 0.01, 0)
    
    return grad

def optimize_for_lambda(lambda_, init_vars, n):
    result = minimize(objective, init_vars, args=(n,), method='L-BFGS-B', bounds=[(0.01, 1),]*n + [(0.01, 0.5)]*n , jac=lambda vars: gradient(vars, n, lambda_))
    return result.x

def run_packing() -> tuple(np.ndarray, np.ndarray, float):
    n = 26
    lambda_ = 1.0
    rho = 1.5
    tol = 1e-8
    k_max = 100
    R = 50
    
    best_centers = None
    best_radii = None
    best_sum_radii = -np.inf
    
    for _ in range(R):
        vars = np.random.rand(3*n) * 0.5 + 0.25
        for _ in range(k_max):
            vars = optimize_for_lambda(lambda_, vars, n)
            sum_radii = np.sum(vars[2*n:])
            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = vars[:n]
                best_radii = vars[2*n:]
            lambda_ *= rho
            if np.linalg.norm(gradient(vars, n, lambda_)) < tol:
                break
    
    return best_centers, best_radii, best_sum_radii
