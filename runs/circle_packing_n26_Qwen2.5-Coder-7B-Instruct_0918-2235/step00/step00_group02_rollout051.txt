import numpy as np
from scipy.optimize import linear_sum_assignment, minimize
from scipy.spatial import distance_matrix

def calc_max_radii(C):
    n = len(C)
    D = distance_matrix(C, C)
    r = np.zeros(n)
    A = np.zeros((n*n, n*n))
    b = np.zeros(n*n)
    for i in range(n):
        for j in range(i+1, n):
            idxij = i*n + j
            idxji = j*n + i
            A[idxij, i], A[idxij, j] = 1, 1
            A[idxji, j], A[idxji, i] = 1, 1
            b[idxij], b[idxji] = D[i, j], D[j, i]
        A[i*n+i, i], A[i*n+i, n+i] = 1, 1
        A[(n+1)*i+n+i, i], A[(n+1)*i+n+i, n+i] = 1, 1
        A[(n+2)*i+n+i, i], A[(n+2)*i+n+i, n+i] = 1, 1
        A[(n+3)*i+n+i, i], A[(n+3)*i+n+i, n+i] = 1, 1
        A[(n+4)*i+n+i, i], A[(n+4)*i+n+i, n+i] = 1, 1
    r = np.maximum(0, np.linalg.solve(A, b))
    r = np.minimum(r, np.minimum(C[:, 0], np.minimum(1 - C[:, 0], np.minimum(C[:, 1], 1 - C[:, 1]))))
    return r

def objective(vars):
    C = vars[:26].reshape(-1, 2)
    r = calc_max_radii(C)
    if np.any(r < 0) or np.any(r > 0.5):
        return -np.inf
    return -np.sum(r)

def run_packing():
    n = 26
    a = 0.18
    v = np.sqrt(3) * a / 2
    rows = [6, 5, 6, 5, 6]
    start_positions = np.arange(0, 1 - a, a)
    C0 = np.array([[i * a + start_positions[row] + 0.01, 
                   (j + row // 2) * v + start_positions[row + 1] + 0.01] 
                  for row in range(5) 
                  for j in range(rows[row])]).reshape(-1, 2)[:26]
    
    C0 = np.clip(C0, 0.05, 0.95)
    
    result = minimize(objective, C0.flatten(), method='L-BFGS-B', bounds=[(0.05, 0.95)] * 52, options={'maxiter': 10000})
    if result.fun == -np.inf:
        result = minimize(objective, np.random.rand(26 * 2) * 0.9 + 0.05, method='L-BFGS-B', bounds=[(0.05, 0.95)] * 52, options={'maxiter': 10000})
    
    C_best = result.x[:26].reshape(-1, 2)
    r_best = calc_max_radii(C_best)
    if np.any(r_best < 0) or np.any(r_best > 0.5):
        return np.array([[0.5, 0.5] for _ in range(n)]), np.zeros(n), 0
    
    r_best = np.maximum(r_best, 0)
    radii_sum = np.sum(r_best)
    return C_best, r_best, radii_sum
