import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(0)
    centers = np.random.uniform(size=(n, 2))
    centers = (centers + 0.5) % 1  # Spread points outwards to avoid touching easily

    def inflation_step(radii):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            x, y = centers[i]
            r_i = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if j == i:
                    continue
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[j]:
                    r_i = min(r_i, dist - radii[j])
            new_radii[i] = r_i
        return new_radii

    def acceptance_test(sum_curr, sum_new, temperature):
        if sum_new > sum_curr:
            return True
        else:
            acceptance_prob = np.exp((sum_new - sum_curr) / temperature)
            return np.random.rand() < acceptance_prob

    # Initial parameters
    step_size_start = 0.15
    step_size_end = 0.001
    temperature_start = 1.0
    temperature_end = 1e-4
    N = 20000

    best_sum = 0
    best_centers = None
    best_radii = None

    for k in range(N):
        i = np.random.randint(n)
        delta = 2 * np.random.uniform(size=2) - 1
        delta *= step_size_start * (step_size_end / step_size_start) ** k / N
        delta = np.clip(delta, -1, 1)
        centers[i] += delta
        centers[i] %= 1

        new_radii = inflation_step(radii)
        new_sum = np.sum(new_radii)

        if acceptance_test(best_sum, new_sum, temperature_start * (temperature_end / temperature_start) ** k / N):
            best_centers = centers.copy()
            best_radii = new_radii
            best_sum = new_sum

    final_radii = inflation_step(radii)

    return best_centers, final_radii, best_sum
