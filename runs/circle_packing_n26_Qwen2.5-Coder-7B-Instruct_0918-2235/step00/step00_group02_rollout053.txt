import numpy as np
from scipy.optimize import linprog, differential_evolution

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_sum_radii(positions):
    n = 26
    centers = positions.reshape(n, 2)
    dmat = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    radii = np.linalg.norm(dmat - 1, axis=1) / 2

    A_ub = np.vstack((radii[:, np.newaxis] + radii[np.newaxis, :], 
                        centers[:, 0][:, np.newaxis], 1 - centers[:, 0][:, np.newaxis],
                        centers[:, 1][:, np.newaxis], 1 - centers[:, 1][:, np.newaxis]))
    b_ub = np.hstack((np.ones(n), centers[:, 0], centers[:, 0], centers[:, 1], centers[:, 1]))

    inequalities = np.ones((n, 26 + 4 * n))
    inequalities[:, np.arange(26) + 26] = 1
    inequalities[np.arange(n), 26:] = -1
    inequalities[:, np.arange(n, 4 * n, 4) + 26] = -1
    inequalities[np.arange(n, 4 * n, 4) + 27] = -1
    inequalities[np.arange(n, 4 * n, 4) + 28] = -1
    inequalities[np.arange(n, 4 * n, 4) + 29] = -1

    A_ub = np.vstack((A_ub, inequalities)), np.hstack((b_ub, np.ones(4 * n)))
    c = -np.ones(26)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')

    if result.success:
        return -result.fun
    else:
        return -np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    positions = differential_evolution(compute_sum_radii, bounds=[(0, 1)] * 52, popsize=20, recombination=0.7).x
    centers = positions.reshape(26, 2)
    radii = np.full(26, 0.1125)
    return centers, radii, compute_sum_radii(positions)
