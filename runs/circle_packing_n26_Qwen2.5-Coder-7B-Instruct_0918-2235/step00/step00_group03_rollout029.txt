import numpy as np

def refine_radii(centres, radii, eps=1e-12, max_iter=200):
    n = centres.shape[0]
    dist_matrix = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    for _ in range(max_iter):
        new_radii = np.empty_like(radii)
        for i in range(n):
            b_i = np.minimum(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
            min_delta_ij = np.maximum(0, dist_matrix[i] - radii - eps)
            new_radii[i] = np.minimum(b_i, np.min(min_delta_ij))
            new_radii[i] = np.maximum(0, new_radii[i])
        if np.sum(np.abs(new_radii - radii)) < eps:
            break
        radii = new_radii
    return radii
