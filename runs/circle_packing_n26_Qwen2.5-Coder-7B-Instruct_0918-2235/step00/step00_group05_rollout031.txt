import numpy as np
from scipy.optimize import linprog, minimize, Bounds
from scipy.spatial.distance import directed_hamming

def gradient_wall_penalty(x, r):
    n = len(r)
    g = np.zeros(2*n)
    for i in range(n):
        g[i*n:i*n+2] = 2*(r[i] - min(x[i*n], 1-x[i*n], x[i*n+1], 1-x[i*n+1]))
    return g

def gradient_overlap_penalty(x, r):
    n = len(r)
    g = np.zeros(2*n)
    dists = np.sqrt((x[::2][:, None] - x[::2][None, :])**2 + 
                    (x[1::2][:, None] - x[1::2][None, :])**2)
    for i in range(n):
        for j in range(i+1, n):
            if dists[i, j] < r[i] + r[j]:
                g[i*n:i*n+2] += -2*(dists[i, j] - r[i] - r[j]) * (x[i*n] - x[j*n])
                g[j*n:j*n+2] += 2*(dists[i, j] - r[i] - r[j]) * (x[i*n] - x[j*n])
    return g

def gradient_penalty(x, r):
    return gradient_wall_penalty(x, r) + gradient_overlap_penalty(x, r)

def objective_and_gradient(x, r, lambda_wall, lambda_overlap):
    n = len(r)
    f = np.sum(r) - (lambda_wall * np.sum(gradient_wall_penalty(x, r))) - (lambda_overlap * np.sum(gradient_overlap_penalty(x, r)))
    g = np.sum(np.ones(2*n)) - lambda_wall * gradient_wall_penalty(x, r) - lambda_overlap * gradient_overlap_penalty(x, r)
    return f, g

def optimise_circles(x0, r0, lambda_wall, lambda_overlap):
    bounds = Bounds(0, 1)
    res = minimize(lambda x: objective_and_gradient(x.reshape(2*26, order='F')[::2], x.reshape(2*26, order='F')[1::2], lambda_wall, lambda_overlap), x0, bounds=bounds)
    x_opt = res.x.reshape(2, 26, order='F')
    r_opt = r0
    return x_opt, r_opt

def lp_radii_fixup(x_opt, r0):
    n = 26
    A = np.hstack((np.eye(n), np.eye(n), -np.eye(n), -np.eye(n)))
    B = np.full(4*n, 0.5)
    for i in range(n):
        for j in range(i+1, n):
            A[i*4+2, j] = 1
            A[j*4+1, i] = 1
    c = np.hstack((np.zeros(n), np.ones(n)))
    res = linprog(c, A_ub=A, b_ub=B, bounds=Bounds(0, None))
    r_opt_lp = res.x[n:2*n]
    return r_opt_lp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(2*n)
    r0 = (0.5 * (x0[:n] + x0[n:])) / 2
    lambdas = [1e4, 1e5]
    for i in range(len(lambdas)):
        lambda_wall, lambda_overlap = lambdas[i], lambdas[i]
        x_opt, r_opt = optimise_circles(x0, r0, lambda_wall, lambda_overlap)
        r_opt_lp = lp_radii_fixup(x_opt, r_opt)
        if np.sum(r_opt) > np.sum(r_opt_lp):
            r_opt = r_opt_lp
        if np.sum(r_opt) > 2.636:
            return x_opt, r_opt, np.sum(r_opt)
        x0 = x_opt
    return x_opt, r_opt, np.sum(r_opt)

centers, radii, sum_radii = run_packing()
noise = np.random.normal(0, 0.01, size=radii.shape)
radii += noise
return centers, radii, sum_radii
