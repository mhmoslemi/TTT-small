```python
import numpy as np
from scipy.optimize import minimize

def energy(vars, lambda_):
    x = vars[:26]
    y = vars[26:52]
    r = vars[52:]
    E = -np.sum(r)
    pen_overlap = np.sum(np.maximum(0, r[:, np.newaxis] + r - np.sqrt((x[:, np.newaxis] - x) ** 2 + (y[:, np.newaxis] - y) ** 2)))
    E += lambda_ * pen_overlap
    return E

def optimize_for_lambda(lambda_, init_vars):
    bounds = [0, 1] * 26 + [0, 0.5] * 26
    result = minimize(energy, init_vars, args=(lambda_,), bounds=bounds)
    return result.x

def run_packing():
    n = 26
    K = 30
    R = 50
    max_sum_radii = 0
    best_vars = None

    for _ in range(R):
        init_vars = np.random.rand(2 * n + n) * (0.85 - (0.7 * np.random.rand(2 * n + n))) + (0.1 * np.random.rand(2 * n + n))
        for lambda_ in 10 ** np.linspace(-1, 5, K):
            vars = optimize_for_lambda(lambda_, init_vars)
            sum_radii = np.sum(vars[52:])
            if sum_radii > max_sum_radii:
                max_sum_radii = sum_radii
                best_vars = vars

    x = best_vars[:26]
    y = best_vars[26:52]
    r = best_vars[52:]
    centers = np.stack([x, y], axis=1)
    return centers, r, max_sum_radii
```
This solution implements the described "barrier method" with a continuous-depressed penalty to solve the circle packing problem. It multi-restarts the optimization with increasing penalty parameters and ensures a robust, heuristic solution for maximizing the sum of radii within given constraints.
