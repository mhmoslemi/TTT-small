import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    candidates = np.array([(x, y) for y in np.arange(h, 1 - h, h) for x in np.arange(h, 1 - h, h) if x > epsilon and x < 1 - epsilon and y > epsilon and y < 1 - epsilon])

    placed_centres = []
    placed_radii = []

    def compute_r_max(candidate, placed_centers, placed_radii):
       # Distance squared between two points
        dists = np.sum((candidate - placed_centers)**2, axis=1)
        r_max = np.minimum(dists - placed_radii, np.array([candidate[0], candidate[1], 1 - candidate[0], 1 - candidate[1]]))
        r_max = np.maximum(r_max, 0)
        return np.minimum(r_max, 1)

    while len(placed_centres) < 26:
        r_candidates = np.array([compute_r_max(candidate, np.array(placed_centres), np.array(placed_radii)) for candidate in candidates])

        feasible_idx = np.where(r_candidates > 0)[0]

        if feasible_idx.size == 0:
            break

        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])

        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    placed_centres = np.array(placed_centres)
    placed_radii = np.array(placed_radii)

    # Local optimisation
    n_iter = 5000
    alpha = 0.02
   radii = cp.Variable(26, nonneg=True)

    constraints = [radii <= placed_centres[:, 0], radii <= 1 - placed_centres[:, 0], radii <= placed_centres[:, 1], radii <= 1 - placed_centres[:, 1]]
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append(radii[i] + radii[j] <= cp.norm(placed_centres[i] - placed_centres[j], 2))

    problem = cp.Problem(cp.Maximize(radii.sum()), constraints)
    problem.solve(solver=cp.ECOS, abstol=1e-9)

    if problem.status == cp.OPTIMAL_OR_NEAR_OPTIMAL:
        return placed_centres, radii.value, np.sum(radii.value)
    else:
        return placed_centres, placed_radii, np.sum(placed_radii)
