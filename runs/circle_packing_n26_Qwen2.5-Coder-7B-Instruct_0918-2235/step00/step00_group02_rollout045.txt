import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_sum_radii(positions):
    n = 26
    dmat = np.sqrt(np.sum((positions[:, None, :2] - positions[None, :, :2]) ** 2, axis=2))
    A_ub = np.zeros((n * (n - 1), 2 * n))
    b_ub = np.zeros(n * (n - 1))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * (n - 1) + j - i - 1, 2 * i:2 * i + 2] = [1, 1]
            A_ub[i * (n - 1) + j - i - 1, 2 * j:2 * j + 2] = [-1, -1]
            b_ub[i * (n - 1) + j - i - 1] = dmat[i, j]
    
    A_eq = np.zeros((4 * n, 2 * n))
    for i in range(n):
        A_eq[i, 2 * i:2 * i + 2] = [1, -1]
        A_eq[n + i, 2 * i] = 1
        A_eq[2 * n + i, 2 * i + 1] = 1
        A_eq[3 * n + i, 2 * i + 1] = -1
        A_eq[3 * n + i, 2 * i] = -1

    b_eq = np.zeros(4 * n)
    for i in range(n):
        b_eq[i] = positions[i, 0]
        b_eq[n + i] = positions[i, 0] - 1
        b_eq[2 * n + i] = positions[i, 1]
        b_eq[3 * n + i] = positions[i, 1] - 1
    
    try:
        sol = linprog(-np.ones(2 * n), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
        return -sol.fun, sol.x
    except:
        return -np.inf, np.zeros(2 * n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonally arranged positions
    centers = np.array([
        [0.125, 0.125], [0.5, 0.125], [0.875, 0.125],
        [0.125, 0.5], [0.5, 0.5], [0.875, 0.5],
        [0.125, 0.875], [0.5, 0.875], [0.875, 0.875],
        [0.375, 0.375], [0.625, 0.375], [0.875, 0.625], [0.375, 0.625],
        [0.625, 0.875], [0.125, 0.375], [0.375, 0.625],
        [0.625, 0.375], [0.875, 0.625], [0.375, 0.125],
        [0.625, 0.875], [0.875, 0.375], [0.375, 0.875],
        [0.125, 0.5], [0.375, 0.625], [0.625, 0.375],
        [0.875, 0.5], [0.625, 0.125], [0.5, 0.875]
    ])
    
    n = len(centers)
    radii = np.zeros(n)
    while np.sum(radii) < 2.636:
        sol, _ = compute_sum_radii(centers)
        radii = sol[:n]
        if np.sum(radii) >= 2.636:
            break
    
    if np.sum(radii) < 2.636:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    return centers, radii, np.sum(radii)
