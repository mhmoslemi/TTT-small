import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    
    # Constants
    P = 60  # Population size
    w = 0.7298  # Inertia weight
    c1 = 1.49618  # Cognitive constant
    c2 = 1.49618  # Social constant
    G = 200  # Maximum generations
    λ = 1000  # Initial penalty multiplier
    v_max = np.array([1, 1, 0.5])
    
    # Initial random positions and positions
    population = np.random.rand(P, 78)
    for i in range(P):
        population[i, :52] = population[i, [::2, ::2]] * (np.array([1, 1, 0, 1, 0] * 13))  # Keep bounds
        population[i, 52:] = np.clip(population[i, [1, 3, 5]] * 0.01, 0, 0.5)  # Ensure radius <= 0.5
    
    p_best = np.copy(population)
    g_best = population[np.argmin([np.sum(p[:, 52:]) - λ * penalties(p) for p in population])]
    best_fit = np.sum(g_best[:, 52:]) - λ * penalties(g_best)
    
    for generation in range(G):
        new_population = np.copy(population)
        velocities = np.zeros((P, 78))
        for p in range(P):
            r = np.random.rand() * 2 - 1
            s = np.random.rand() * 2 - 1
            velocities[p] = w * velocities[p] + c1 * r * (p_best[p] - population[p]) + c2 * s * (g_best - population[p])
            velocities[p] = np.clip(velocities[p], -v_max, v_max)
            new_population[p] = population[p] + velocities[p]
            new_population[p] = np.clip(new_population[p], 0, 1)
            new_population[p, :52] = new_population[p, [::2, ::2]] * (np.array([1, 1, 0, 1, 0] * 13))  # Ensure bounds
            new_population[p, 52:] = np.clip(new_population[p, [1, 3, 5]] * 0.01, 0, 0.5)  # Ensure radius <= 0.5
            
            fit = np.sum(new_population[p, 52:]) - λ * penalties(new_population[p])
            if fit > best_fit:
                best_fit = fit
                p_best[p] = new_population[p]
            if fit < λ * penalties(g_best):
                g_best = new_population[p]
        
        population = np.copy(new_population)
        λ *= 1.02
    
    # Post-processing for refinement
    def local_radius_maximization(v guesses):
        centers = v[:52].reshape((26, 2))
        radii = np.clip(v[52:], 0, 0.5)
        for _ in range(100):
            for i in range(26):
                r_min = np.inf
                for j in range(26):
                    if i != j:
                        r_min = min(r_min, (1e-6 + np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[j]))
                    for k in range(2):
                        r_min = min(r_min, centers[i, k], 1 - centers[i, k])
                radii[i] = r_min
            radii = np.clip(radii, 0, 0.5)
        return centers, radii
    
    centers, radii = local_radius_maximization(np.concatenate((g_best[:52], g_best[52:])))
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

def penalties(v):
    centers = v[:52].reshape((26, 2))
    radii = v[52:]
    penalty = 0
    for i in range(26):
        penalty -= np.minimum(0, centers[i, 0] - radii[i])
        penalty -= np.minimum(0, 1 - centers[i, 0] - radii[i])
        penalty -= np.minimum(0, centers[i, 1] - radii[i])
        penalty -= np.minimum(0, 1 - centers[i, 1] - radii[i])
        for j in range(i + 1, 26):
            d = np.sum((centers[i] - centers[j])**2)
            penalty -= np.minimum(0, radii[i] + radii[j] - d)
    return penalty
