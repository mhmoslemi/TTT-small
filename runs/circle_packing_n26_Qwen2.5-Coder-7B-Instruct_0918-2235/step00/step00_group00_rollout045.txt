import numpy as np
import cvxpy as cp

# --------------------------------------------------------------------
# 1.  Solve the linear programme that gives the best radii for a
#    fixed arrangement of centres.
# --------------------------------------------------------------------
def optimise_radii(centers: np.ndarray) -> np.ndarray:
    """
    Compute the optimal radii for the given centre coordinates.

    Parameters
    ----------
    centres : (n,2) ndarray
        Positions of the circle centres.

    Returns
    -------
    radii : (n,) ndarray
        Optimised radii that satisfy all constraints.
    """
    n = centres.shape[0]
    # Decision variables: radii, required to be non‑negative.
    r = cp.Variable(n, nonneg=True)

    # Constraints
    constraints = []

    # Boundary constraints
    x = centres[:, 0]
    y = centres[:, 1]
    constraints += [r <= x,
                    r <= 1 - x,
                    r <= y,
                    r <= 1 - y]

    # Pairwise distance constraints
    for i in range(n):
        ci = centres[i]
        for j in range(i + 1, n):
            cj = centres[j]
            d = np.linalg.norm(ci - cj)
            constraints.append(r[i] + r[j] <= d)

    # Problem definition
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)

    # Solve – we use ECOS because the problem is a linear programme.
    # Falling back to the default solver if ECOS is unavailable.
    prob.solve(solver=cp.ECOS, warm_start=True,
               abstol=1e-9, reltol=1e-9, feastol=1e-9)

    if r.value is None:
        raise RuntimeError("LP failed to find a solution")

    # Numerical safety: clip tiny negative values to zero.
    radii = np.maximum(r.value, 0.0)
    return radii
