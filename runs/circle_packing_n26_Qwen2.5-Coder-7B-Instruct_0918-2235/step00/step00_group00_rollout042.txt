import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(v, λ):
    n = 26
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = v[i], v[i + n]
            xj, yj = v[j], v[j + n]
            dij = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
            penalty += np.maximum(0, r + rj - dij) ** 2
    penalty += np.maximum(0, r - xi) ** 2
    penalty += np.maximum(0, r - (1 - xi)) ** 2
    penalty += np.maximum(0, r - yi) ** 2
    penalty += np.maximum(0, r - (1 - yi)) ** 2
    return -r + λ * penalty

def penalty(v, λ):
    n = 26
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = v[i], v[i + n]
            xj, yj = v[j], v[j + n]
            dij = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
            penalty += np.maximum(0, r + rj - dij) ** 2
    penalty += np.maximum(0, r - xi) ** 2
    penalty += np.maximum(0, r - (1 - xi)) ** 2
    penalty += np.maximum(0, r - yi) ** 2
    penalty += np.maximum(0, r - (1 - yi)) ** 2
    return penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n)
    λ = 1000
    best_solution = None
    best_radius_sum = 0

    for _ in range(15):
        result = differential_evolution(lambda v: objective(v, λ), bounds, popsize=10, maxiter=50)
        penalty_value = penalty(result.x, λ)
        if penalty_value == 0:
            r = result.x[n:]
            if np.sum(r) > best_radius_sum:
                best_solution = result.x
                best_radius_sum = np.sum(r)

    if best_solution is None:
        return np.zeros((n, 2)), np.zeros(n), 0

    x = best_solution[:n]
    y = best_solution[n:]
    initial_radii = np.clip(x, 0, 0.5) / 2
    result = minimize(lambda r: penalty(x, y, r), initial_radii, bounds=[(0, 0.5)] * n)

    return np.column_stack((x, y)), result.x, np.sum(result.x)

centers, radii, sum_radii = run_packing()
