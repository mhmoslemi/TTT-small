import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial centers placed in a close-packed hexagonal pattern
    n = 26
    R = np.sqrt(3) * 0.5
    centers = [(R * (2 * i + j % 2), R * np.sqrt(3) * (j // 2)) for i in range(n) for j in range(n)]
    centers = centers[:n]  # Keep only the first 26 centers
    
    # Bound for each coordinate (0 to 1)
    bounds = [(0, 1)] * 2 * n
    
    def objective(x):
        # Reshape centres from vector to matrix
        centres = x.reshape((n, 2))
        
        # Construct A_ub and b_ub for linprog
        A_ub = []
        b_ub = []
        indices = {}
        def add_boundary_constraint(i, c):
            if i not in indices:
                indices[i] = len(indices)
            A = np.zeros(2*n)
            A[indices[i]] = c
            A[n + indices[i]] = 1
            A_ub.append(A)
            b_ub.append(1)
        
        def add_pair_constraint(i, j, d_ij):
            if (i, j) not in indices:
                indices[(i, j)] = len(indices)
            A = np.zeros(2 * n)
            A[indices[(i, j)]] = (centres[j, 0] - centres[i, 0]) / d_ij
            A[indices[(i, j)] + n] = (centres[j, 1] - centres[i, 1]) / d_ij
            A[-indices[(i, j)] - 1] = (centres[i, 0] - centres[j, 0]) / d_ij
            A[-indices[(i, j)] - 2] = (centres[i, 1] - centres[j, 1]) / d_ij
            A_ub.append(A)
            b_ub.append(d_ij)
        
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(centres[j] - centres[i])
                add_pair_constraint(i, j, d_ij)
            add_boundary_constraint(i, -1)
            add_boundary_constraint(i, 0)
        
        # Solve the linear programme to get the optimal radii
        res = linprog([-sum(x)], A_ub=A_ub, b_ub=b_ub, method='highs', bounds=bounds)
        if not res.success:
            raise ValueError(f"LP optimization failed: {res.message}")
        
        return -res.fun
    
    def gradient(x):
        centres = x.reshape((n, 2))
        radii = -res.fun  # From previous optimization
        d_positions = np.zeros_like(x)
        
        def d_boundary(i, c):
            d_positions[2 * i] += c * (centres[i, 0] if c == -1 else 1)
            d_positions[2 * i + 1] += c * (centres[i, 1] if c == -1 else 1)
        
        def d_pair(i, j, d_ij):
            u = (centres[j, 0] - centres[i, 0]) / d_ij
            v = (centres[j, 1] - centres[i, 1]) / d_ij
            d_positions[2 * i] -= radii[j] * u
            d_positions[2 * i + 1] -= radii[j] * v
            d_positions[2 * j] += radii[i] * u
            d_positions[2 * j + 1] += radii[i] * v
        
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(centres[j] - centres[i])
                d_pair(i, j, d_ij)
            d_boundary(i, -radii[i])
            d_boundary(i, radii[i])
        
        return -d_positions
    
    result = minimize(objective, np.prod(centers), method='L-BFGS-B', bounds=bounds, jac=gradient)
    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")
    
    final_centres = result.x.reshape((n, 2))
    radii = -result.fun
    final_sum_radii = np.sum(radii)
    
    return final_centres, radii, final_sum_radii
