import numpy as np
from scipy.optimize import differential_evolution, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dim = 2 * n

    bounds = [(0, 1)] * (2 * n)

    def penalty(v):
        r = np.reshape(v[n:], n)
        centers = np.reshape(v[:n], (n, 2))

        mem = r + r[:, np.newaxis] - np.sum((centers - centers[:, np.newaxis, :]) ** 2, axis=2)
        mem *= mem > 0
        mem += np.sum((r - centers[:, 0]) ** 2)
        mem += np.sum((r - (1 - centers[:, 0])) ** 2)
        mem += np.sum((r - centers[:, 1]) ** 2)
        mem += np.sum((r - (1 - centers[:, 1])) ** 2)
        return np.sum(mem)

    def objective(v):
        return -np.sum(v[n:]) + 1000 * penalty(v)

    res = differential_evolution(objective, bounds, popsize=10, maxiter=50, bounds=bounds, seed=np.random.randint(0, 10000))

    best_diff = np.inf
    best_r = None
    best_v = None
    for candidate in res.x:
        v = np.reshape(candidate, (dim,))
        r = np.reshape(v[n:], n)
        if np.all(r > 1e-6):
            if best_diff > np.sum(r):
                best_diff = np.sum(r)
                best_r = r
                best_v = v

    centers = np.reshape(best_v[:n], (n, 2))
    radii = best_r

    res = linprog(radii, A_ub=np.column_stack([2 * radii, -2 * centers[:, 0], -2 * centers[:, 1], np.ones(n), 2 * centers[:, 0], 2 * centers[:, 1]]),
                 b_ub=1 - radii, method='highs')

    if res.success:
        radii = res.x
        return centers, radii, np.sum(radii)
    else:
        return centers, radii, np.sum(radii)
