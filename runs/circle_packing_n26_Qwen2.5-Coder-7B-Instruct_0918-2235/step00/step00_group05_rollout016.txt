def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    import numpy as np

    def hessian(f):
        def grad(x):
            return np.gradient(f(x), side='center')
        def hess(x):
            return np.gradient(grad(x), side='center')
        return hess

    def gradient(f, x, epsilon=1e-5):
        return np.array([f(x + np.array([epsilon,] * len(x))) - f(x - np.array([epsilon,] * len(x))) for _ in range(len(x))])

    def lp_radii_fixup(centers: np.ndarray) -> np.ndarray:
        from scipy.optimize import linprog

        n = centers.shape[0]
        A_eq, b_eq, A_in, b_in = [], [], [], []

        for i in range(n):
            for j in range(i + 1, n):
                dist_ij = np.linalg.norm(centers[i] - centers[j])
                A_eq.append(np.r_[1, centers[i], centers[j]])
                b_eq.append(dist_ij)
                A_in.append(np.r_[-1, centers[i], centers[j]])
                b_in.append(dist_ij - 2 * centers[i, 0])

            dist_ii_to_wall = np.r_[centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1]]
            A_in.append(np.r_[-1, centers[i]])
            b_in.append(dist_ii_to_wall)

        bounds = [(0, None) for _ in range(n)] + [(0, 0.5) for _ in range(n)]

        r = np.array([min(np.linalg.norm(centers[i] - x), 1 - 2 * x) for x in centers])
        res = linprog(-r, A_eq=A_eq, b_eq=b_eq, A_ub=A_in, b_ub=b_in, method='highs', bounds=bounds)
        if not res.success:
            raise Exception('LP optimization failed but this should not happen')

        optimal_radii = res.x[n:]
        return optimal_radii

    def objective(x, rk):
        rki = x[n:]
        penalty_wall = np.sum(np.maximum(0, rki - x[:n]).reshape(-1, 2) ** 2)
        penalty_overlap = np.sum(np.maximum(0, (rki + x[n + 2 * n:n + 3 * n]) - distance[i] for i in range(n)).reshape(-1, 2) ** 2)
        return np.sum(rki) - lambda_wall * penalty_wall - lambda_overlap * penalty_overlap, penalty_wall + penalty_overlap

    def distance(x, y):
        return np.linalg.norm(x - y)

    def shake_up(centers, radii):
        for _ in range(200):
            k = np.random.randint(1, len(centers))
            jitter = np.random.uniform(-0.01, 0.01, size=(k, 2))

            new_centers[:k] += jitter
            new_radii = lp_radii_fixup(new_centers)
            if np.sum(new_radii) > np.sum(radii):
                centers, radii = new_centers, new_radii

        return centers, radii

    initial_radii = lp_radii_fixup(initial_centers)

    best_radii = None
    best_sum = 0

        radii = np.minimum(lp_radii_fixup(centers), 0.5)
