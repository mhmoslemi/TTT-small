import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_sum_radii(positions):
    n = 26
    dmat = np.sqrt(np.sum((positions.reshape(n, 2, 1) - positions.reshape(1, 2, n)) ** 2, axis=1))
    np.fill_diagonal(dmat, np.inf)
    A_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([np.concatenate([np.zeros(i * 2 + j * 2), np.array([1, 1, -1, 0, 0]), np.zeros((26 - i - j - 2) * 2)]), 
                          np.concatenate([np.zeros(i * 2 + j * 2), np.array([0, 0, -1, 1, 0]), np.zeros((26 - i - j - 2) * 2)]), 
                          np.concatenate([np.zeros(i * 2 + j * 2), np.array([0, 0, -1, 0, 1]), np.zeros((26 - i - j - 2) * 2)])])
    A_ub = np.array(A_ub)
    b_ub = np.inf * np.ones(A_ub.shape[0])
    for i in range(n):
        A_ub[i, np.arange(i * 2, i * 2 + 2)] += [1, -1]
        b_ub[i] = 2 * positions[i, 0]
    for i in range(n):
        A_ub[i, np.arange(i * 2 + 2, i * 2 + 4)] += [1, -1]
        b_ub[i] = 2 * positions[i, 1]
    for i in range(n):
        A_ub[i, np.arange(i * 2 + 4, i * 2 + 6)] += [-1]
        b_ub[i] = positions[i, 0]
    for i in range(n):
        A_ub[i, np.arange(i * 2 + 6, i * 2 + 8)] += [-1]
        b_ub[i] = 1 - positions[i, 0]
    for i in range(n):
        A_ub[i, np.arange(i * 2 + 8, i * 2 + 10)] += [-1]
        b_ub[i] = positions[i, 1]
    for i in range(n):
        A_ub[i, np.arange(i * 2 + 10, i * 2 + 12)] += [-1]
        b_ub[i] = 1 - positions[i, 1]
    res = linprog(np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        return -np.inf
    return -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5],

        [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6]
    ])
    radii = np.array([0.1] * 26)
    res = differential_evolution(compute_sum_radii, bounds=[(0, 1)] * 52, popsize=20, recombination=0.7, maxiter=100)
    radii = -np.ones(26)
    dmat = np.sqrt(np.sum((res.x.reshape(26, 2, 1) - res.x.reshape(1, 2, 26)) ** 2, axis=1))
    np.fill_diagonal(dmat, np.inf)
    A_ub, b_ub = [], []
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub.append([np.concatenate([np.zeros(i * 2 + j * 2), np.array([1, 1, -1, 0, 0]), np.zeros((26 - i - j - 2) * 2)]),

                          np.concatenate([np.zeros(i * 2 + j * 2), np.array([0, 0, -1, 0, 1]), np.zeros((26 - i - j - 2) * 2)])])
    A_ub = np.array(A_ub)
    b_ub = np.inf * np.ones(A_ub.shape[0])
    for i in range(26):
        A_ub[i, np.arange(i * 2, i * 2 + 2)] += [1, -1]
        b_ub[i] = 2 * res.x[i, 0]
    for i in range(26):
        A_ub[i, np.arange(i * 2 + 2, i * 2 + 4)] += [1, -1]
        b_ub[i] = 2 * res.x[i, 1]
    for i in range(26):
        A_ub[i, np.arange(i * 2 + 4, i * 2 + 6)] += [-1]

        b_ub[i] = 1 - res.x[i, 1]
    res = linprog(np.ones(26), A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        radii = res.x
    return centers, radii, -compute_sum_radii(res.x)
