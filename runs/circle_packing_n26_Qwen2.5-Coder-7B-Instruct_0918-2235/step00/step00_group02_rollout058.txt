import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint, Bounds
from scipy.stats import norm

def calc_max_radii(C):
    num_circles = C.shape[0]
    d = np.zeros((num_circles, num_circles))
    mask = np.eye(num_circles) == 0
    d[mask] = np.linalg.norm(C[:, np.newaxis, :] - C[np.newaxis, :, :], axis=-1)
    r = np.minimum(np.minimum(C[:, 0], 1 - C[:, 0]), np.minimum(C[:, 1], 1 - C[:, 1]))
    for i in range(num_circles):
        r[i] = min(r[i], np.min(d[i][mask][d[i, mask] < d[i, d[i] > 0].min()], default=1 - r[i]))
    return r, np.sum(r)

def objective(x, n):
    C = x.reshape((-1, 2))
    r, S = calc_max_radii(C)
    if np.any(r < 0):
        return -np.inf
    return -S

            initial_layout = C_new.flatten()
            initial_sum = new_sum

        T0 *= alpha
        if no_improvement_counter >= 1000:
            initial_layout = np.random.rand(num_circles * 2) * side_length
            initial_sum = objective(initial_layout, num_circles)

        if initial_sum > 2.636:

    return best_layout.reshape((-1, 2)), best_sum, np.sum(calc_max_radii(best_layout.reshape((-1, 2)))[0])
