def hex_grid(side_length, packing_density):

    Generates a hexagonal grid within the unit square

            break
        triangles += 1
    A = np.zeros((triangles * 3, 2))
    k = 0
    step = side_length / (np.sqrt(3) * (2 * triangles + 2))
    for c in range(-triangles, triangles):
        i = np.arange(0, triangles - abs(c), dtype=int)
        A[k: k + len(i), 0] = np.sin(pi * (i + 0.5) / (2 * triangles)) * (1 + 2 * c)
        r = (i % 2 == 0)
        A[k: k + len(i), 1] = np.cos(pi * (i + 0.5) / (2 * triangles)) * r
        k += len(i)
    A *= step
    A -= A.min(0)
    A /= A.max(0)
    A *= side_length
    return A

def solve_radii_fixed_centres(centers):
    """
    Solve the linear program to find the maximum radii for fixed centers
    """
    n = centers.shape[0]
    A = np.hstack((np.eye(n), -np.eye(n)))
    b = np.hstack((np.ones(n) * (1 - RADIUS_TOLERANCE), RADIUS_TOLERANCE * np.ones(n)))
    c = np.ones(n)
    result = linprog(c, A_ub=A, b_ub=b, method='highs')
    if result.status != 0:
        raise Exception("LP infeasible; drop circles or shrink radii.")
    return result.x, result.fun

def penalty_centres_fixed_radii(centers, radii):
    """
    Penalize overlapping circles and boundary violations
    """
    overlap_penalty = np.sum(np.maximum(0, radii[:, np.newaxis] + radii - np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)))
    boundary_penalty = np.sum(np.maximum(0, radii - np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1], RADIUS_TOLERANCE)))
    return overlap_penalty + boundary_penalty
