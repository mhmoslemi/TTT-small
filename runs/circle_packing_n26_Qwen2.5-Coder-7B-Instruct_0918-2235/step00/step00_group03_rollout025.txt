import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance matrix as distmatrix

def energy(vars, lambda_):
    n = 26
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    overlaps = distmatrix(x, x) + distmatrix(y, y) - r - r - 1
    overlap_penalties = np.where(overlaps < 0, overlaps ** 2, 0).sum()
    total_energy = -n * r.sum() + lambda_ * overlap_penalties
    return total_energy, r, overlaps

def gradient(vars, lambda_):
    n = 26
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    g = np.zeros_like(vars)
    overlaps = distmatrix(x, x) + distmatrix(y, y) - r - r - 1
    overlap_penalties = np.where(overlaps < 0, overlaps ** 2, 0)

    for i in range(n):
        for j in range(i + 1, n):
            dx = x[j] - x[i]
            dy = y[j] - y[i]
            dist = np.hypot(dx, dy)
            gradij = 2 * overlap_penalties[i, j] * 2 * dx / dist if dist > 0 else 0
            g[i] += gradij
            g[j] -= gradij
    
    g[n:] += -lambda_ * overlap_penalties.sum(axis=1)
    g = g + np.where(g == np.inf, 0, g)
    return g

def optimize_for_lambda(lambda_, init_vars):
    result = minimize(energy, init_vars, args=(lambda_,), method='L-BFGS-B', jac=gradient, bounds=[(0, 1)] * n + [(0, 0.5)] * n)
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    R = 50
    K = 30
    lambda_0 = 1.0
    rho = 1.5
    gradient_tolerance = 1e-6
    sum_tolerance = 1e-8

    best_vars = None
    best_sum = 0

    for _ in range(R):
        init_vars = np.random.rand(n*2 + n) * 0.98 + 0.01
        for k in range(K):
            lambda_ = lambda_0 * rho**k
            vars = optimize_for_lambda(lambda_, init_vars)
            _, radii, overlaps = energy(vars, lambda_)
            sum_radii = radii.sum()
            if sum_radii > best_sum:
                best_vars = vars
                best_sum = sum_radii
            if np.linalg.norm(gradient(vars, lambda_)) < gradient_tolerance:
                break

        if k == K - 1 and np.abs(best_vars[2*n:].sum() - n * 0.5) > sum_tolerance:
            return run_packing()

    x = best_vars[:n]
    y = best_vars[n:2*n]
    r = best_vars[2*n:]
    centers = np.vstack((x, y)).T
    return centers, r, best_sum
