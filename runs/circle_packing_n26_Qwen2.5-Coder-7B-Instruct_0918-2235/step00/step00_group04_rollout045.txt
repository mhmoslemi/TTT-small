```python
import numpy as np
from scipy.optimize import linprog

# Constants
MAX_ITERS = 20000
K_JUMP = 500
N_REDUCE = 500
EPSILON = 1e-12
METHOD = "highs"

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # maximize sum

    A_ub = np.empty((4*n + n*(n-1)//2, n))
    b_ub = np.empty(4*n + n*(n-1)//2)

    idx = 0
    for i in range(n):
        A_ub[idx, i] = np.array([1, 0, 0])  # Constraint r_i <= x_i
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = np.array([-1, 0, 0])  # Constraint r_i <= 1 - x_i
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = np.array([0, 1, 0])  # Constraint r_i <= y_i
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = np.array([0, -1, 0])  # Constraint r_i <= 1 - y_i
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method=METHOD, bounds=(0, None))
    if res.success:
        radii = res.x
        sum_radii = -res.fun
        return radii, sum_radii
    else:
        radii = np.zeros(n)
        sum_radii = 0
        return radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)

    # Initial configuration
    grid = np.arange(0, 1 + 0.16, 0.16)
    initial_centers = np.array([(x, y) for x in grid for y in grid])
    np.random.shuffle(initial_centers)
    initial_centers = initial_centers[:26]

    current_centers = initial_centers.copy()
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum

    step = 0.1

    for iter in range(MAX_ITERS):
        i = np.random.randint(26)
        new_x = np.clip(current_centers[i, 0] + np.random.uniform(-step, step), 0, 1)
        new_y = np.clip(current_centers[i, 1] + np.random.uniform(-step, step), 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = (new_x, new_y)
        
        new_radii, new_sum = compute_lp_solution(new_centers)

        if new_sum > current_sum + EPSILON:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = current_centers
                best_radii = current_radii
                best_sum = new_sum

        if iter % K_JUMP == 0:
            random_centers = np.random.uniform(0, 1, (26, 2))
            random_r, _ = compute_lp_solution(random_centers)
            if np.sum(random_r) > best_sum:
                best_centers = random_centers
                best_radii = random_r
                best_sum = np.sum(random_r)

        if iter % N_REDUCE == 0 and step > 1e-4:
            step *= 0.9

    return best_centers, best_radii, best_sum
```
