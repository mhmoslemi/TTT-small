import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    R = 40
    max_iter = 30
    initial_step_size = 0.1
    step_size_decay = 0.8
    tolerance = 1e-9
    
    # Initialize best results
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    for _ in range(R):
        # Initialize random centers
        centers = np.random.rand(n, 2)
        
        for iter in range(max_iter):
            # Update step size
            s = initial_step_size * (step_size_decay ** iter)
            
            # Decompose the vectorized positions into x, y components
            xs = centers[:, 0]
            ys = centers[:, 1]
            
            # Precompute pairwise distance matrix
            dists = squareform(pdist(centers))
            np.fill_diagonal(dists, np.inf)
            
            best_admissible_radius = np.inf * np.ones(n)
            for i in range(n):
                # Compute the maximum feasible radius for center i
                r_i = np.minimum(dists[i] - ys, dists[i] + ys)
                r_i = np.minimum(r_i, dists[i] - xs, dists[i] + xs)
                r_i = np.minimum(r_i, 1 - xs, 1 + xs, 1 - ys, 1 + ys)
                best_admissible_radius[i] = np.minimum(r_i)
            
            # Define the bounds for the LP problem
            bounds = [(0, r) for r in best_admissible_radius]
            
            # Define the coefficients for the objective function
            c = -np.ones(n)
            
            # Define the inequality constraints for the LP problem
            A_ineq = np.zeros((n * (n - 1) // 2, n))
            b_ineq = np.zeros(n * (n - 1) // 2)
            idx = 0
            for i in range(n):
                for j in range(i + 1, n):
                    A_ineq[idx, i] = 1
                    A_ineq[idx, j] = 1
                    b_ineq[idx] = dists[i, j]
                    idx += 1
            
            # Solve the linear programming problem
            res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
            
            # Check if the solution is successful
            if not res.success:
                continue
            
            # If acceptance criterion is satisfied, update current best solution
            new_sum = -res.fun
            if new_sum > best_sum + tolerance:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = -res.x
    
    # Ensure positions are clipped to [0, 1]
    best_centers = np.clip(best_centers, 0, 1)
    
    return best_centers, best_radii, best_sum
