import numpy as np
from scipy.optimize import linprog

def solve_radius_lp(centres):
    n = len(centres)
    A_ineq_parts = []
    b_ineq_parts = []

    for i in range(n):
        A_ineq = np.zeros(n)
        A_ineq[i] = 1
        b_ineq_parts.append(centres[i, 0])
        A_ineq_parts.append(A_ineq.copy())
        b_ineq_parts.append(1 - centres[i, 0])

        A_ineq[i] = 1
        b_ineq_parts.append(centres[i, 1])
        A_ineq_parts.append(A_ineq.copy())
        b_ineq_parts.append(1 - centres[i, 1])

    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt(np.sum((centres[i] - centres[j]) ** 2))
            A_ineq = np.zeros(n)
            A_ineq[i] = 1
            A_ineq[j] = 1
            b_ineq_parts.append(d_ij)
            A_ineq_parts.append(A_ineq.copy())
            b_ineq_parts.append(d_ij)

    A_ineq = np.vstack(A_ineq_parts)
    b_ineq = np.hstack(b_ineq_parts)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=(0, None), method='highs')
    return res.x, -res.fun

def run_packing():
    np.random.seed(42)

    n_particles = 30
    n_circles = 26
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    max_iter = 3000
    vel_clip = 0.1
    alpha = 0.999

    centres = np.random.normal(scale=0.01, size=(n_particles, n_circles, 2))
    centres += np.array([[i // 5, i % 5] for i in range(n_circles)] * n_particles) / 5.0
    np.random.shuffle(centres)
    centres = centres[:n_particles]

    velocities = np.zeros((n_particles, n_circles, 2))
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres[np.argmin(pbest_sums)]
    gbest_sum = 0

    for _ in range(max_iter):
        for p in range(n_particles):
            v = inertia * velocities[p] + \
                cognitive * np.random.rand() * (pbests[p] - centres[p]) + \
                social * np.random.rand() * (gbest - centres[p])
            v = np.clip(v, -vel_clip, vel_clip)
            centres[p] = np.clip(centres[p] + v, 0, 1)
            radii, sum_radii = solve_radius_lp(centres[p])

            if sum_radii > pbest_sums[p]:
                pbests[p] = centres[p]
                pbest_sums[p] = sum_radii

            if sum_radii > gbest_sum:
                gbest = centres[p]
                gbest_sum = sum_radii

        if alpha is not None:
            velocities *= alpha

    return gbest, solve_radius_lp(gbest)[0], gbest_sum
