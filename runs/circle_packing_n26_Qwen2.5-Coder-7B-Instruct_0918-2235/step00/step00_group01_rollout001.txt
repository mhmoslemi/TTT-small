import numpy as np
from scipy.optimize import minimize

def ineqs(v):
    n = 26
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])
        cons.append(1 - xs[i] - rs[i])
        cons.append(ys[i] - rs[i])
        cons.append(1 - ys[i] - rs[i])
    pairs = [(i,j) for i in range(n) for j in range(i+1,n)]
    for i,j in pairs:
        dx = xs[i] - xs[j]
        dy = ys[i] - ys[j]
        dist2 = dx dx + dy dy
        cons.append(dist2 - (rs[i] + rs[j])**2)
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])

n = 26
r0 = 0.02
xs = np.random.uniform(r0, 1-r0, n)
ys = np.random.uniform(r0, 1-r0, n)
rs = np.full(n, r0)
v0 = np.concatenate((xs, ys, rs))
bounds = [(-1e-9, 1)] * 2*n + [(0, 0.5)]

def solve_packing():
    res = minimize(objective, v0,
               method='SLSQP',
               bounds=bounds,
               constraints=[{'type':'ineq','fun':ineqs}],
               options={'ftol':1e-9, 'maxiter':5000})
    if not res.success:
        raise Exception('Optimization failed')
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs,ys))
    radii = rs
    sum_r = rs.sum()
    return centers, radii, sum_r

def run_packing():
    centers, radii, sum_radii = solve_packing()
    return centers, radii, sum_radii
