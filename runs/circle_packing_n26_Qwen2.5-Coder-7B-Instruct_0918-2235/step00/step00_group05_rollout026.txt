import numpy as np
import scipy.optimize as opt
import cma

def initial_mean():
    a = np.sqrt(2 / 3) * (1 + 1 / np.sqrt(12)) / 2
    b = np.sqrt(2 / 3) * (1 - 1 / np.sqrt(12)) / 2
    r = a * (np.sqrt(3) - 1) / 2
    centers = []
    for i in range(26):
        x = a * (i % 3 + 0.5 * (i // 3 % 2))
        y = 1.5 * b * (i // 3 if i < 13 else i // 3 - 6)
        centers.append(np.array([x, y]))
    centers = np.array(centers)
    radii = refine_radii(centers)
    return centers, radii

def refine_radii(centers):
    c = centers.shape[0]
    A = np.zeros((c + c, c))
    b = np.zeros(c + c)
    for i in range(c):
        A[i, i] = 1
        b[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(i + 1, c):
            if np.sum((centers[i] - centers[j]) ** 2) < (radii[i] + radii[j]) ** 2:
                h = np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - radii[i] - radii[j] + 1e-12
                A[c + i, j] = 1
                A[c + j, i] = 1
                b[c + i] += h
    r = opt.linprog(np.ones(c), A_ub=A, b_ub=b, method='highs').x
    return r

def objective(z):
    n = len(z) // 2
    centres = z[:n].reshape(-1, 2)
    radii = refine_radii(centres)
    centres_with_radii = np.hstack((centres, radii[:, np.newaxis]))
    penalties = 0
    for walls in set(sum(radii.reshape(-1, 1), axis=0).tolist()):
        f = 0.5 * (walls - 1) ** 2 / walls
        penalties += np.sum(f[radii < walls])
    non_overlap = ((np.sum((radii.reshape(-1, 1) + radii.reshape(1, -1)) ** 2, axis=0)
                   - np.sum(radii ** 2)) * 0.5 * (radii.reshape(-1, 1) + radii.reshape(1, -1)) / (radii.reshape(-1, 1) + radii.reshape(1, -1)))
    penalties += np.sum(non_overlap.clip(min=0))
    return -np.sum(radii) + penalties

def simulate_annealing(best_z):
    n = len(best_z) // 2
    centres = best_z[:n].reshape(-1, 2)
    radii = refine_radii(centres)
    best_sum_r = np.sum(radii)
    for _ in range(200):
        shake_indices = np.random.choice(n, 3, replace=False)
        shake_centres = centres.copy()
        shake_centres[shake_indices, :] += np.random.uniform(-0.01, 0.01, (3, 2))
        if np.any(shake_centres < 0) or np.any(shake_centres > 1):
            continue
        shake_radii = refine_radii(shake_centres)
        new_sum_r = np.sum(shake_radii)
        if new_sum_r > best_sum_r + 1e-9:
            best_sum_r = new_sum_r
            centres = shake_centres
            best_z[:n] = centres.flatten()
            radii = shake_radii
    return best_z, best_sum_r
