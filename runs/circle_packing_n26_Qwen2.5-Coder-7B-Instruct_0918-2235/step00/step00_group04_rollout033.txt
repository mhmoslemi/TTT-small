import numpy as np
from scipy.optimize import linprog, minimize

def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    d = np.array([[np.linalg.norm(centres_i[i] - centres_i[j]) for j in range(n_circles)] for i in range(n_circles)])
    c = -np.ones(n_circles)
    A_ub = np.vstack([
        np.diag([1] * n_circles),
        np.diag([-1] * n_circles),
        np.diag([1] * n_circles),
        np.diag([-1] * n_circles)
    ])
    b_ub = np.concatenate([centres_i[:, 0], 1 - centres_i[:, 0], np.clip(centres_i[:, 1], 0, 1), 1 - np.clip(centres_i[:, 1], 0, 1)])
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            A_ub = np.vstack([A_ub, np.zeros(n_circles)])
            A_ub[-1, i], A_ub[-1, j] = 1, 1
            b_ub = np.append(b_ub, d[i, j])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[0, None], method='highs')
    return res.x, -res.fun

def objective pos, num_particles, centroids):
    best_pos_list = []
    best_sum_list = []
    for pos_i in range(num_particles):

        best_pos_list.append(best_pos)
        best_sum_list.append(best_sum)
    return -np.array(best_sum_list)

def PSO(num_particles, num_iterations, inertia, cognitive, social, centroid_x, centroid_y, max_radius=1):
    np.random.seed(42)
    n_circles = len(centroid_x)
    positions = np.random.randn(num_particles, n_circles, 2) * max_radius * 0.01
    velocities = np.zeros_like(positions)
    pbests = positions.copy()
    pbest_sums = np.zeros(num_particles)
    gbest = positions[np.argmax(pbest_sums)]
    gbest_sum = np.max(pbest_sums)

    for iteration in range(num_iterations):
        rp1 = np.random.rand(num_particles)
        rp2 = np.random.rand(num_particles)
        velocities = inertia * velocities + cognitive * rp1 * (pbests - positions) + social * rp2 * (gbest - positions)
        positions += velocities
        positions = np.clip(positions, 0, max_radius)
        pbests = positions.copy()

        updated_sums = []
        for particle in range(num_particles):
            currpos = positions[particle,...]
            radii, sum_radii = solve_radius_lp(currpos)
            if sum_radii > pbest_sums[particle]:
                pbests[particle] = currpos
                pbest_sums[particle] = sum_radii

        if np.max(pbest_sums) > gbest_sum:
            gbest_sum = np.max(pbest_sums)
            gbest = pbests[np.argmax(pbest_sums)]

    best_pos_list = []
    best_sum_list = []
    for pos_i in range(num_particles):
        best_pos, best_sum = solve_radius_lp(positions[pos_i,...])
        best_pos_list.append(best_pos)
        best_sum_list.append(best_sum)
        
    return gbest, np.array(best_sum_list)

# Example usage
centroid_x = np.array([0, 1, 2])
centroid_y = np.array([0, 1, 2])

positions, velocities = PSO(n_particles=30, num_iterations=5000, inertia=0.729, cognitive=1.49445, social=1.49445, centroid_x=centroid_x, centroid_y=centroid_y)

best_pos = positions[np.argmax(velocities)]
centers = np.vstack([best_pos[n] for n in range(len(best_pos))])
radii = solve_radius_lp(centers)[0]
sum_radii = np.sum(radii)

print(f"Centers: {centers}, Radii: {radii}, Sum Radii: {sum_radii}")
