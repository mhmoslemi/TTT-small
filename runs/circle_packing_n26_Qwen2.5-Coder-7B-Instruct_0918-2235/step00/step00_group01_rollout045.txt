import numpy as np
from scipy.optimize import minimize

def generate_initial_guess(num_circles):
    min_radius = 0.01
    positions = []
    while len(positions) < num_circles:
        x = np.random.uniform(min_radius, 1 - min_radius)
        y = np.random.uniform(min_radius, 1 - min_radius)
        radius = np.random.uniform(min_radius, 0.5)
        if all((x - pos[0])**2 + (y - pos[1])**2 > (radius + pos[2])**2 for pos in positions):
            positions.append([x, y, radius])
    initial_v = np.array([item for sublist in positions for item in sublist])
    return initial_v

def out_of_bounds(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    return np.any(xs - rs < 0) or np.any(1 - xs - rs < 0) or np.any(ys - rs < 0) or np.any(1 - ys - rs < 0)

def pairwise_non_overlap(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    pairs = [(i, j) for i in range(len(xs)) for j in range(i+1, len(xs))]
    return np.all((xs[:, None] - xs[None, :])**2 + (ys[:, None] - ys[None, :])**2 - (rs[:, None] + rs[None, :])**2 >= 0)

def objective(v):
    return -np.sum(v[2::3])  # Minimise negative sum => Maximise sum

def ineqs(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    boundary_conditions = (xs - rs >= 0) & (1 - xs - rs >= 0) & (ys - rs >= 0) & (1 - ys - rs >= 0)
    non_overlap_conditions = (xs[:, None] - xs[None, :])**2 + (ys[:, None] - ys[None, :])**2 - (rs[:, None] + rs[None, :])**2 >= 0
    return np.hstack((boundary_conditions.reshape(-1,1), non_overlap_conditions.reshape(-1,1))).flatten()

def constraints_func(v):
    bounds = [(0, None), (0, None), (0, 0.5)] * 26
    return [{'type': 'ineq', 'fun': ineqs, 'bounds': bounds}]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    initial_v = generate_initial_guess(num_circles)
    constraints = constraints_func(initial_v)
    res = minimize(objective, initial_v, method='SLSQP', bounds=[(0, 1), (0, 1), (0, 0.5)]*26, 
                  constraints=constraints, options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    if not res.success:
        raise Exception("Optimization failed")
    
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs, ys))
    radii = rs
    sum_r = rs.sum()
    return centers, radii, sum_r
