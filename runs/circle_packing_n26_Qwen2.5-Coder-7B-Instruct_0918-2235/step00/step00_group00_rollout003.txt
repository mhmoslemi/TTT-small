import numpy as np
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)  # Initial positions
    radii = np.zeros(n)  # Initial radii
    delta = 0.15  # Initial step size
    t_start = 1.0
    t_end = 1e-4
    n_iterations = 20000
    
    def cooling_schedule(k, N):
        return t_start * (t_end / t_start) ** (k / N)
    
    def inflation_step(centers, radii):
        new_radii = np.zeros_like(radii)
        for i in range(n):
            x_i, y_i = centers[i]
            new_radii[i] = min(
                x_i, 1 - x_i, y_i, 1 - y_i,
                min((np.linalg.norm(centers[j] - centers[i]) - radii[j]) for j in range(n) if i != j)
            )
        new_radii = np.maximum(new_radii, 0)
        return new_radii
    
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for k in range(n_iterations):
        i = random.randint(0, n - 1)
        c_i = centers[i]
        delta_x, delta_y = np.random.uniform(-delta, delta, 2)
        c_i_new = c_i + np.array([delta_x, delta_y])
        c_i_new = np.clip(c_i_new, 0, 1)
        
        radii_new = inflation_step(np.delete(centers, i, axis=0), np.delete(radii, i))
        sum_new = sum(radii_new)
        
        if sum_new > best_sum:
            best_sum = sum_new
            best_centers = centers.copy()
            best_radii = radii_new.copy()
        else:
            p = np.exp((sum_new - best_sum) / cooling_schedule(k, n_iterations))
            if np.random.rand() < p:
                centers[i] = c_i_new
                radii = radii_new
    
    best_radii_nonzero = np.maximum(best_radii, 1e-12)  # Ensure non-zero radii
    total_radius = sum(best_radii_nonzero)
    
    return best_centers, best_radii_nonzero, total_radius
