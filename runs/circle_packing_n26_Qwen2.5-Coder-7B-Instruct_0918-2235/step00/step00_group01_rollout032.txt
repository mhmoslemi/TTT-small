import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_objective(x, fixed_radii=None):
    n = 26
    x = x.reshape(n, 2)
    d = np.sqrt((x[:, np.newaxis, 0] - x[np.newaxis, :, 0]) ** 2 + 
                 (x[:, np.newaxis, 1] - x[np.newaxis, :, 1]) ** 2)
    c = np.ones(n)
    A = np.zeros((n, n)) - 1
    A = np.triu(A, k=1)
    A[np.diag_indices(n)] = 0
    b = np.max(d) - np.eye(n)
    
    if fixed_radii is not None:
        c[:n] += fixed_radii
        A = np.hstack([A, np.eye(n)])
        b = np.hstack([b, fixed_radii])
        c = np.pad(c, (0, fixed_radii.shape[0]), 'constant')
        A = np.pad(A, ((0, fixed_radii.shape[0]), (0, fixed_radii.shape[0])), 'constant')
        b = np.pad(b, (0, fixed_radii.shape[0]), 'constant')
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=np.zeros((2*n, 2)), method='highs')
    return -res.fun

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    popsize = 20
    maxiter = 200
    tol = 1e-12

    def objective(x):
        centers = x.reshape(26, 2)
        radii = lp_objective(x)
        return -radii.sum()

    result = differential_evolution(objective, bounds, popsize=popsize, maxiter=maxiter, tol=tol)
    x = result.x
    x = x.reshape(26, 2)
    best_centers = x
    best_radii = lp_objective(x)

    opt_result = minimize(lambda x: -lp_objective(x, fixed_radii=best_radii), x.flatten(), method='Nelder-Mead',
                         bounds=bounds, options={'xatol': tol, 'fatol': tol}, initial_simplex=None, bounds=bounds)

    result_x = opt_result.x
    result_x = result_x.reshape(26, 2)
    final_centers = result_x
    final_radii = lp_objective(result_x, fixed_radii=best_radii)

    return final_centers, final_radii, final_radii.sum()
