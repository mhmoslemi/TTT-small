import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    MIN_R = 1e-8

    def dist2(i, j):
        return np.sum((pts[i] - pts[j]) ** 2)

    # Step 2.1 - Random permutation
    order = np.random.permutation(n)

    # Step 2.2 - Iterative placement
    for t in range(n):
        k = order[t]
        S = 5000
        candidates = np.random.rand(S, 2)
        filtered = []

        for x, y in candidates:
            if all(np.sqrt(dist2(k, j)) > rad[j] + MIN_R for j in range(n)) and \
               x < 1 - MIN_R and x > MIN_R and y < 1 - MIN_R and y > MIN_R:
                clearance = min(min(np.sqrt(dist2(k, j)) - rad[j] for j in range(n)), min(x, 1-x, y, 1-y))
                clearance = max(clearance, MIN_R)
                if clearance > rad[k]:
                    filtered.append((x, y, clearance))

        if filtered:
            filtered.sort(key=lambda x: x[2], reverse=True)
            x, y, _ = filtered[0]
            rad[k] = _
            pts[k] = (x, y)
        else:
            rad[k] = MIN_R

    # Shake
    shake = np.random.normal(0, 1e-4, (n, 2))
    shake_radius = 1e-4 * np.random.rand(n)
    pts += shake
    pts = np.clip(pts - shake_radius, MIN_R, 1 - MIN_R)
    pts = np.clip(pts + shake_radius, MIN_R, 1 - MIN_R)

    # Step 3.1 - Parameter schedule
    T_init = 1.0
    T_final = 1e-4
    cooling_rate = 0.999

    # Step 3.2 - Iterative update loop
    T = T_init
    count = 0
    max_it = 200000
    best_sum_radii = rad.sum()
    old_rad = rad.copy()
    best_rad = rad.copy()
    best_pts = pts.copy()
    last_acc = 0
    last_rej = 0

    while T > T_final and count < max_it:
        i = np.random.randint(n)
        Δ = 2 * np.pi * np.random.rand() * shaking_radius
        Δx, Δy = np.cos(Δ), np.sin(Δ)
        x_prime, y_prime = pts[i, 0] + Δx, pts[i, 1] + Δy
        x_prime = np.clip(x_prime, MIN_R, 1 - MIN_R)
        y_prime = np.clip(y_prime, MIN_R, 1 - MIN_R)

        clearance_wall = min(x_prime, 1 - x_prime, y_prime, 1 - y_prime)
        clearance_pair = min([np.sqrt(dist2(i, j)) - rad[j] for j in range(n) if j != i], default=float('inf'))
        clearance_pair = max(clearance_pair, MIN_R)
        clearance_pair = min(clearance_pair, clearance_wall)
        rad_prime = min(clearance_wall, clearance_pair)

        # Assign the temporary state
        pts_temp = pts.copy()
        rad_temp = rad.copy()
        pts_temp[i, :] = (x_prime, y_prime)
        rad_temp[i] = rad_prime

        # Compute ΔU
        old_U = np.sum(np.min([2.0 * np.pi * 0.5 * ((rad[j] + np.min([np.sqrt(dist2(i, j)) - rad[j], clearance_wall, clearance_pair]) if j != i else float('inf')) ** 2), rad ** 2 * clearance_wall ** 2, rad_squared for j, rad_squared in enumerate(rad ** 2)], axis=0))
        new_U = np.sum(np.min([2.0 * np.pi * 0.5 * ((rad_prime + np.min([np.sqrt(dist2(i, j)) - rad_prime, clearance_wall, clearance_pair]) if j != i else float('inf')) ** 2), rad_temp ** 2 * clearance_wall ** 2, rad_squared for j, rad_squared in enumerate(rad_temp ** 2)], axis=0))

        ΔU = new_U - old_U

        # Metropolis criterion
        if ΔU <= 0 or np.random.rand() < np.exp(-ΔU / T):
            rad[i], pts[i] = rad_prime, (x_prime, y_prime)
            if rad[i] > best_sum_radii:
                best_sum_radii = rad[i]
                best_rad = rad.copy()
                best_pts = pts.copy()
        else:
            last_rej += 1

        count += 1
        T *= cooling_rate
        if last_rej > 10000:
            T *= 0.9

    # Feasibility sweep
    for i in range(n):
        clearance_wall = min(pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1])
        clearance_pair = min([np.sqrt(dist2(i, j)) - rad[j] for j in range(n) if j != i], default=float('inf'))
        clearance_pair = max(clearance_pair, MIN_R)
        clearance_pair = min(clearance_pair, clearance_wall)
        rad[i] = min(clearance_wall, clearance_pair)

    for _ in range(10):
        rad_last = rad.copy()
        for i in range(n):
            clearance_wall = min(pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1])
            clearance_pair = min([np.sqrt(dist2(i, j)) - rad[j] for j in range(n) if j != i], default=float('inf'))
            clearance_pair = max(clearance_pair, MIN_R)
            clearance_pair = min(clearance_pair, clearance_wall)
