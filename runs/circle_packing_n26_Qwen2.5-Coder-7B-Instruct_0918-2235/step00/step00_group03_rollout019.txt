import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple:
    n = 26
    max_cycles = 50
    step_size = 0.1
    alpha = 0.8
    eps_stop = 1e-4
    num_trials = 40
    best_sum = -np.inf

    def compute_distances(centers):
        d = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                d[i, j] = np.linalg.norm(centers[i] - centers[j])
                d[j, i] = d[i, j]
        return d

    def solve_inner_lp(centers):
        radii_bounds = np.maximum(0, np.minimum(centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]))
        A_ub = []
        b_ub = []
        for i in range(n):
            for j in range(i + 1, n):
                A_ub.append([1, 1])
                b_ub.append(compute_distances(centers)[i, j])
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, bounds) for bounds in radii_bounds])
        return -res.fun, res.x

    def solve_outer_problem():
        nonlocal best_sum
        nonlocal best_positions
        nonlocal best_radii
        for trial in range(num_trials):
            centers = np.random.rand(n, 2)
            for cycle in range(max_cycles):
                for i in np.random.permutation(n):
                    old_pos = centers[i]
                    delta = np.random.uniform(-step_size, step_size, size=2)
                    centers[i] = np.clip(old_pos + delta, 0, 1)
                    sum_r, _ = solve_inner_lp(centers)
                    if sum_r <= best_sum + 1e-10:
                        centers[i] = old_pos
                    else:
                        best_sum = sum_r
                step_size *= alpha
                if step_size < eps_stop:
                    break
            if best_sum > best_sum:
                best_sum = sum_r
                best_positions = centers.copy()
                best_radii = get_radii(best_positions)
        return best_positions, best_radii, best_sum

    def get_radii(centers):
        _, radii = solve_inner_lp(centers)
        return radii

    best_positions, best_radii, best_sum = solve_outer_problem()
    return best_positions, best_radii, best_sum
