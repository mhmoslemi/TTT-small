import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from cvxpy import Variable, Problem, Minimize, sum_entries, solve
from numpy.linalg import norm

# Function to calculate maximum radii for given centers
def calc_max_radii(C):
    n = C.shape[0]
    R = Variable(n)
    d = pdist(C)
    constraints = [R >= 0, R <= C[:, 0], R <= C[:, 1], R <= 1-C[:, 0], R <= 1-C[:, 1]]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(R[i] + R[j] <= d[i, j])
    prob = Problem(Minimize(sum_entries(R)), constraints)
    prob.solve(qp_solver='highs', verbose=False)
    if prob.status != 'optimal':
        return np.zeros(n), -np.inf
    return R.value, sum(R.value)

        return float('-inf')
    return -S
