import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_radius_sum(centers, radii, n):
    c = np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append(np.zeros(n) +radii[i])
        b_ub.append(centers[i, 0])
        
        A_ub.append(np.zeros(n) +radii[i])
        b_ub.append(1 - centers[i, 0])
        
        A_ub.append(np.zeros(n) +radii[i])
        b_ub.append(centers[i, 1])
        
        A_ub.append(np.zeros(n) +radii[i])
        b_ub.append(1 - centers[i, 1])
    
    for i in range(n):
        for j in range(i + 1, n):
            distance = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub.append(np.zeros(n) +radii[i] + radii[j])
            b_ub.append(distance)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    return -result.fun

def objective(u):
    x = u[:26]
    y = u[26:]
    centers = np.column_stack((x, y))
    return -lp_radius_sum(centers, np.zeros(26), 26)

def run_packing():
    np.random.seed(0)
    bounds = [(0, 1)] * 52
    u0 = np.random.rand(52)
    result = differential_evolution(objective, bounds, maxiter=200, popsize=20, tol=1e-12)
    
    x = result.x[:26]
    y = result.x[26:]
    centers = np.column_stack((x, y))
    radii = np.zeros(26)
    
    for i in range(26):
        for j in range(i + 1, 26):
            if centers[i][0] == centers[j][0] and centers[i][1] == centers[j][1]:
                raise ValueError("Overlapping circles detected")
        
        distance = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        radii[i] = min(distance / 2, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
