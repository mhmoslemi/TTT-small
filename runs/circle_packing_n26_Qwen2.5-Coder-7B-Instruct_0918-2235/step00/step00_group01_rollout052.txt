import numpy as np

def run_packing():
    n = 26
    P = 60
    w = 0.7298
    c1 = 1.49618
    c2 = 1.49618
    G = 200
    v_max = np.array([1, 1, 0.5])
    
    # Initialize the population
    np.random.seed(12345)
    positions = np.random.rand(P, n, 2)
    radii = np.random.rand(P, n) * 0.01
    particles = np.concatenate((positions, radii), axis=2)
    best_particles = np.copy(particles)
    best_fitness = -np.inf
    penalty_multiplier = 1000
    
    # PSO loop
    for g in range(G):
        for p in range(P):
            # Update velocity and position
            velocities = w * particles[p, :, :3] + c1 * np.random.rand(n, 3) * (best_particles[p, :, :3] - particles[p, :, :3]) + c2 * np.random.rand(n, 3) * (best_particles[:, :, :3].mean(axis=0) - particles[p, :, :3])
            velocities = np.clip(velocities, -v_max, v_max)
            particles[p, :, :3] += velocities
            
            # Ensure bounds
            particles[p, :, 0] = np.clip(particles[p, :, 0], 0, 1)
            particles[p, :, 1] = np.clip(particles[p, :, 1], 0, 1)
            particles[p, :, 2] = np.clip(particles[p, :, 2], 0, 0.5)
            
            # Evaluate penalty
            positions = particles[p, :, :2]
            radii = particles[p, :, 2]
            penalties = np.maximum(positions[:, 0] - radii, 0) + np.maximum(1 - positions[:, 0] - radii, 0) + np.maximum(positions[:, 1] - radii, 0) + np.maximum(1 - positions[:, 1] - radii, 0)
            for i in range(n):
                for j in range(i + 1, n):
                    penalties[i] += np.maximum((radii[i] + radii[j]) - np.linalg.norm(positions[i] - positions[j]), 0)
            penalty = penalty_multiplier * np.sum(penalties)
            
            # Evaluate fitness
            fitness = np.sum(radii) - penalty
            if fitness > best_fitness:
                best_fitness = fitness
                best_particles[p] = particles[p]
                
        # Update penalty multiplier
        penalty_multiplier *= 1.02
        
        # Early stopping
        if np.allclose(best_fitness, best_fitness, atol=1e-5) and g >= 15:
            break
    
    # Post-processing
    centers = best_particles[:, :, :2].mean(axis=0)
    radii = best_particles[:, :, 2].mean(axis=0)
    for _ in range(10):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            delta_x, delta_y = np.random.uniform(-1e-4, 1e-4, 2)
            centers[i] += (delta_x, delta_y)
            centers[i][0] = np.clip(centers[i][0], 0, 1)
            centers[i][1] = np.clip(centers[i][1], 0, 1)
            radius = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j]) - radii[j]
                    min_dist = min(min_dist, dist)
            if min_dist > 0:
                radius = min_dist - radii[j]
            radii[i] = radius
            centers[i] -= (delta_x, delta_y)
    
    return centers, radii, np.sum(radii)
