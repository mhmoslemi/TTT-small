import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance

def objective(centers):
    n = centers.shape[0]
    radii = solve_lp(centers).x[::2]
    return -np.sum(radii)

def solve_lp(centers):
    n = centers.shape[0]
    A = np.vstack([
        centers[:, 0], 
        centers[:, 0] - 1, 
        centers[:, 1], 
        centers[:, 1] - 1
    ])
    b = np.ones(4 * n)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = distance.euclidean(centers[i], centers[j])
            A = np.vstack([A, np.array([d_ij, d_ij - 0.001])])
            b = np.append(b, [0.01, d_ij])
    return opt.linprog(
        A_ub=A, 
        b_ub=b, 
        method='highs', 
        bounds=np.full(2 * n, (0, 1))
    )

def gradient(centers):
    n = centers.shape[0]
    theta = solve_lp(centers)
    dual = theta.dual
    x = theta.x[::2]
    y = theta.x[1::2]
    
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = distance.euclidean(centers[i], centers[j])
            u_ij = (centers[i, 0] - centers[j, 0]) / d_ij
            v_ij = (centers[i, 1] - centers[j, 1]) / d_ij
            
            if np.isclose(d_ij, x[i], atol=0.001):
                lambda_ij = dual[4 * n + 2 * (i * n - i // 2 - 1) + (i % 2) + j - (i * (i - 1) // 2)]
            else:
                lambda_ij = 0
    
            grad_x[i] += lambda_ij * u_ij
            grad_x[j] -= lambda_ij * u_ij
            grad_y[i] += lambda_ij * v_ij
            grad_y[j] -= lambda_ij * v_ij
    
    return -np.hstack((grad_x, grad_y))

def run_packing():
    centers = np.array([
        [0.25, 0.05],
        [0.25, 0.35],
        [0.25, 0.65],
        [0.25, 0.95],
        [0.55, 0.25],
        [0.55, 0.55],
        [0.55, 0.85],
        [0.85, 0.55],
        [0.85, 0.85],
        [0.05, 0.25],
        [0.35, 0.25],
        [0.65, 0.25],
        [0.95, 0.25],
        [0.05, 0.55],
        [0.35, 0.55],
        [0.65, 0.55],
        [0.95, 0.55],
        [0.05, 0.85],
        [0.35, 0.85],
        [0.65, 0.85],
        [0.95, 0.85],
        [0.25, 0.45],
        [0.75, 0.45],
        [0.25, 0.55],
        [0.75, 0.55],
        [0.25, 0.65],
        [0.75, 0.65]
    ])
    result = opt.minimize(
        objective, 
        centers, 
        method='L-BFGS-B', 
        jac=gradient, 
        bounds=np.full(2 * n, (0, 1))
    )
    radii = solve_lp(result.x).x[::2]
    return result.x.reshape((-1, 2)), radii, np.sum(radii)
