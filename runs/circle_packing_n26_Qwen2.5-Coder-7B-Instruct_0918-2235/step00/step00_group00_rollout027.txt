import numpy as np
from scipy.optimize import differential_evolution, linprog

def penalty(v):
    n = len(v) // 3
    r = v[n:]
    ri = r.reshape(-1, 1)
    overlap = np.sum(np.maximum(0, r + ri - np.minimum(r[:, np.newaxis], ri)), axis=1) ** 2
    boundaries = np.sum(np.maximum(0, r - np.minimum(v[:n], 1 - v[:n])), axis=1) ** 2
    return np.sum(r) - 1000 * (np.sum(overlap) + np.sum(boundaries))

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    result = differential_evolution(penalty, bounds, popsize=10, maxiter=50, disp=False)
    best_v = result.x
    best_r = best_v[n:]
    best_sum = np.sum(best_r)
    if result.success and best_sum > 0:
        centers = best_v[:n].reshape(-1, 2)
        for _ in range(10):
            r_opt = linprog([1] * n, A_eq=None, A_ub=None, b_ub=None, A_int=None, bounds=[(0, None)] * n, method='highs', x0=best_r)
            if r_opt.success:
                r_optimized = r_opt.x
                if np.sum(r_optimized) > best_sum:
                    best_r = r_optimized
                    best_sum = np.sum(best_r)
        return centers, best_r, best_sum
    else:
        return np.zeros((n, 2)), np.zeros(n), 0

centers, radii, sum_radii = run_packing()
