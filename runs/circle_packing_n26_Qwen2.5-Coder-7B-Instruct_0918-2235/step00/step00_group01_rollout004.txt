import numpy as np
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # Fixed seed for deterministic output
    n = 26
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    
    # Phase 1: Random greedy insertion
    order = np.random.permutation(n)
    for t in range(n):
        k = order[t]
        S = 5000
        candidates = np.random.rand(S, 2)
        feasible = True
        for i in range(S):
            x, y = candidates[i]
            if all(np.sqrt(np.sum((x - pts[j]) ** 2)) >= rad[j] + 1e-8 for j in range(k)) and \
               0 <= x <= 1 and 0 <= y <= 1:
                clearance = min(min(np.sqrt(np.sum((x - pts[j]) ** 2)) - rad[j] for j in range(k)), min(x, 1 - x, y, 1 - y))
                if clearance > 1e-6:
                    rad[k] = clearance
                    pts[k] = (x, y)
                    feasible = False
                    break
        
        if feasible and k == n - 1:
            return pts, rad, np.sum(rad)
        
    if feasible:
        # Optional: Local shake
        step = 1e-4
        for _ in range(n):
            x, y = pts[i]
            dx, dy = np.random.uniform(-step, step, 2)
            new_x, new_y = x + dx, y + dy
            new_x = np.clip(new_x, rad[i], 1 - rad[i])
            new_y = np.clip(new_y, rad[i], 1 - rad[i])
            clearance = min(min(np.sqrt(np.sum((new_x - pts[j]) ** 2)) - rad[j] for j in range(n)), min(new_x, 1 - new_x, new_y, 1 - new_y))
            if clearance > rad[i]:
                rad[i] = clearance
                pts[i] = (new_x, new_y)
            
    # Phase 2: Force-field relaxation & simulated annealing
    T_init, T_final, cooling_rate = 1.0, 1e-4, 0.999
    T = T_init
    epsilon_w, epsilon_p = 1.0, 1.0
    while T > T_final:
        i = np.random.randint(n)
        x, y = pts[i]
        dx, dy = np.random.uniform(-0.01 * T, 0.01 * T, 2)
        new_x, new_y = x + dx, y + dy
        new_x = np.clip(new_x, rad[i], 1 - rad[i])
        new_y = np.clip(new_y, rad[i], 1 - rad[i])
        clearance = min(min(np.sqrt(np.sum((new_x - pts[j]) ** 2)) - rad[j] for j in range(n)), min(new_x, 1 - new_x, new_y, 1 - new_y))
        if clearance >= rad[i]:
            rad[i] = clearance
            pts[i] = (new_x, new_y)
        
        if np.random.rand() < np.exp(-(cdist([pts[i]], [pts[i]])[0][0] - rad[i])**2 / (2 * epsilon_p)):
            T *= cooling_rate
        else:
            rad[i] = 0.5 * (rad[i] + 1e-6)
            pts[i] = (x, y)
    
    # Post-processing & refinement
    for i in range(n):
        pts[i], rad[i] = refine_circle(i, pts, rad)
    
    return pts, rad, np.sum(rad)

def refine_circle(k, pts, rad):
    while True:
        x, y = pts[k]
        dx, dy = np.random.uniform(-1e-4, 1e-4, 2)
        new_x, new_y = x + dx, y + dy
        new_x = np.clip(new_x, rad[k], 1 - rad[k])
        new_y = np.clip(new_y, rad[k], 1 - rad[k])
        clearance = min(min(np.sqrt(np.sum((new_x - pts[j]) ** 2)) - rad[j] for j in range(n) if j != k), min(new_x, 1 - new_x, new_y, 1 - new_y))
        if clearance >= rad[k]:
            rad[k] = clearance
            pts[k] = (new_x, new_y)
            break
    return pts[k], rad[k]
