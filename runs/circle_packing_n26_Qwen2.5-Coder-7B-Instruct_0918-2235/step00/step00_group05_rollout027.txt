import numpy as np
import cma
from scipy.optimize import linprog

def refine_radii(centres):
    n = centres.shape[0]
    bounds = [(0, min(ci, 1-ci)) for ci in centres.flatten()]
    cons = [{'type': 'ineq', 'fun': lambda r: min(ci - r[i], 1-ci - r[i])} for i, ci in enumerate(centres.flatten())]
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centres[i] - centres[j]) ** 2)) - r[i] - r[j]})
    result = linprog([1]*(2*n), A_ub=np.vstack(cons).T, b_ub=[c(n, 2*n)*0.99 for _ in range(len(cons))], bounds=bounds, method='highs', options={'disp': False})
    return result.x[n:]

def objective(z):
    centres = z[:52].reshape((26, 2))
    r_lp = refine_radii(centres)
    return np.sum(r_lp) + 1e-8  # Add tiny constant to avoid negative sum (validator tolerance)

def simulate_annealing(best_z):
    centres = best_z[:52].reshape((26, 2))
    np.random.seed(42)
    for _ in range(200):
        shake = np.random.uniform(-0.01, 0.01, (26, 2))
        shake = np.clip(shake, 0, 1) - 0.5
        proposed = centres + shake
        r_lp = refine_radii(proposed)
        if np.sum(r_lp) > np.sum(best_z[52:]):
            best_z[:52] = proposed.flatten()
            best_z[52:] = r_lp
    return best_z

def run_packing():

        [0.125, 0.125], [0.5, 0.125], [0.875, 0.125],

        [0.375, 0.375], [0.625, 0.375],
        [0.375, 0.625], [0.625, 0.625],
        [0.75, 0.25], [0.75, 0.5], [0.75, 0.75],
        [0.5, 0.25], [0.5, 0.75],
        [0.25, 0.25], [0.25, 0.5], [0.25, 0.75]
    ])
    m0 = np.hstack((centroids, 0.1 * np.ones(26)))  # Initial mean with small jitter
    bounds = ((0, 0.5),) * 52
    es = cma.CMAEvolutionStrategy(m0, 0.03, {'bounds': bounds})
    while not es.stop():
        solutions = es.ask()
        es.tell(solutions, [objective(z) for z in solutions])
    best_z = es.result[0]
    best_z = simulate_annealing(best_z)
