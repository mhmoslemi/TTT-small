import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog, nelder_mead

def generate_hexagon_lattice(theta, dx, dy, s):
    R = 2 * s / np.sqrt(3)
    N1 = np.array([R * np.cos(2 * np.pi * i / 3 + theta) for i in range(9)])
    N2 = np.array([R * np.cos(2 * np.pi * i / 3 + theta + 2 * np.pi / 6) for i in range(8)])
    N3 = np.array([R * np.cos(2 * np.pi * i / 3 + theta + 4 * np.pi / 6) for i in range(9)])
    centres = np.vstack((N1 + N2 + N3, N1 + N2 + N3 + np.array([s, 0]), N1 + N2 + N3 + np.array([s / 2, s * np.sqrt(3) / 2])))
    centres[:, 0] += dx
    centres[:, 1] += dy
    return centres

def in_square(centres):
    return np.all((0 <= centres) & (centres <= 1), axis=1)

def solve_lps(centres):
    radii = np.zeros(26)
    for i in range(26):
        constraint = []
        for j in range(i + 1, 26):
            d = distance.euclidean(centres[i], centres[j])
            constraint.append((centres[i][0], centres[j][0], centres[i][1], centres[j][1], d))
        for x in range(2):
            constraint.append((centres[i][x], 1, 0, 0, centres[i][x]))
            constraint.append((centres[i][x], 0, 1, 0, centres[i][x]))
        A = np.array([[c[0], c[2], 1, 0, 0] for c in constraint] +
                      [[c[1], c[3], 0, 1, 0] for c in constraint] +
                      [[c[0], c[2], 0, 0, -1] for c in constraint] +
                      [[c[1], c[3], 0, 0, -1] for c in constraint] +
                      [[x, 0, 0, 0, 0] for x in centres[:, 0]] +
                      [[0, y, 0, 0, 0] for y in centres[:, 1]])
        b = np.array([c[4] for c in constraint] + centres[:, 0] + centres[:, 1])
        result = linprog(-radii, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        radii[:26] = -result.x
    return radii, np.sum(radii)

def run_packing():
    best_sum = -np.inf
    for x in np.linspace(0, np.pi / 6, 51):
        for y in np.linspace(-0.025, 0.025, 51):
            for z in np.linspace(-0.025, 0.025, 51):
                centres = generate_hexagon_lattice(x, y, z, 0.09)
                if in_square(centres).all():
                    radii, sum_radii = solve_lps(centres)
                    if sum_radii > best_sum:
                        best_sum, best_centers, best_radii = sum_radii, centres, radii
    return best_centers, best_radii, best_sum

centers, radii, sum_radii = run_packing()
