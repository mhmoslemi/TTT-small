import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize


def hex_grid(radius, packing_density):
    cell_size = 2 * radius / np.sqrt(3)
    num_rows = int(np.ceil(2 / cell_size))
    num_cols = int(np.ceil(np.sqrt(3 * packing_density * num_rows * num_rows * cell_size * cell_size * np.pi / 3)))
    grid = np.zeros((num_rows, num_cols))
    
    for i, j in np.ndindex(grid.shape):
        if i % 2 == j % 2:
            grid[i, j] = 1
    
    return grid


def add_circles_around_quadrant(centers, radius, spacing=0.1):
    quarter_points = np.array([
        (0.5, 0.5), (-0.5, 0.5), (0.5, -0.5), (-0.5, -0.5),
        (-0.35, -0.35), (0.35, -0.35), (-0.35, 0.35), (0.35, 0.35)
    ])
    
    new_centers = []
    spacing *= radius
    for point in quarter_points:
        x, y = point
        cx = 0.5 + x * spacing
        cy = 0.5 + y * spacing
        new_centers.append((cx, cy))
    
    for cx, cy in new_centers:
        if 0 <= cx <= 1 and 0 <= cy <= 1:
            centers.append((cx, cy))
    
    return np.array(centers)


def solve_radii_fixed_centres(centers):
    n = len(centers)
    radii_bounds = [(0, min(x, 1 - x, y, 1 - y)) for x, y in centers]
    
    constraints = [
        {'type': 'ineq', 'fun': lambda r: min(x - r for x in r)}
        for _ in range(n)
    ]
    
    for i in range(n):
        for j in range(i + 1, n):
            cj, cy = centers[j]
            constraints.append({
                'type': 'ineq', 'fun': lambda r: r[i] + r[j] - np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2)
            })
    
    result = linprog(
        -np.ones(n),  # Maximize sum, hence minimize negative sum
        bounds=radii_bounds,
        A=constraints,
        method='highs'
    )
    
    if result.success:
        return np.array(-result.x), result.fun
    
    return np.zeros(n), 0


def penalty_centres_fixed_radii(centers, radii):
    n = len(centers)
    penalty = 0
    
    for i in range(n):
        x, y = centers[i]
        penalty += max(0, radii[i] - min(x, 1 - x, y, 1 - y))
    
    for j in range(n):
        for k in range(j + 1, n):
            penalty += max(0, radii[j] + radii[k] - np.sqrt((centers[j][0] - centers[k][0]) ** 2 + (centers[j][1] - centers[k][1]) ** 2))
    
    return penalty


def coordinated_ascent(init_centres, max_iter=15):
    n = len(init_centres)
    best_centres = init_centres.copy()
    best_radii = np.zeros(n)
    best_sum_radii = 0
    
    for _ in range(max_iter):
        radii, sum_radii = solve_radii_fixed_centres(best_centres)
        penalty = penalty_centres_fixed_radii(best_centres, best_radii)
        
        result = differential_evolution(
            lambda c: penalty_centres_fixed_radii(centers=c.reshape((n, 2)), radii=radii),
            bounds=[(0, 1)] * (2 * n),
            constraints={'type': 'eq', 'fun': lambda c: np.sum(radii)}
        )
        
        new_centres = result.x.reshape((n, 2))
        new_sum_radii = sum_radii
        
        if new_sum_radii > best_sum_radii:
            best_centres = new_centres
            best_radii = radii
            best_sum_radii = new_sum_radii
    
    return best_centres, best_radii, best_sum_radii


def local_audit_and_refine(centers, radii):
    n = len(centers)
    penalty = 0
    
    for i in range(n):
        x, y = centers[i]
        penalty += max(0, radii[i] - min(x, 1 - x, y, 1 - y))
    
    for j in range(n):
        for k in range(j + 1, n):
            penalty += max(0, radii[j] + radii[k] - np.sqrt((centers[j][0] - centers[k][0]) ** 2 + (centers[j][1] - centers[k][1]) ** 2))
    
    return penalty


def run_packing() -> tuple:
    max_sum_radii = 0
    best_centres = None
    best_radii = None
    
    for seed in range(10):
        np.random.seed(seed)
        lattice = hex_grid(radius=0.05, packing_density=0.8)
        centers = np.array([np.random.rand(2) + (i, j) * (1 / lattice.shape[0] * 0.9) for i in range(lattice.shape[0]) for j in range(lattice.shape[1]) if lattice[i, j]])
        centers = add_circles_around_quadrant(centers, 0.05, spacing=0.1)
        
        centres, radii, sum_radii = coordinated_ascent(centers)
        if sum_radii > max_sum_radii:
            max_sum_radii = sum_radii
            best_centres = centres
            best_radii = radii
    
    return best_centres, best_radii, sum_radii
