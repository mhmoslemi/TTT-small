def compute_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_dist = np.inf
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            min_dist = min(min_dist, dist - radii[j])
        radii[i] = min(min_dist, centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1])
    return radii
