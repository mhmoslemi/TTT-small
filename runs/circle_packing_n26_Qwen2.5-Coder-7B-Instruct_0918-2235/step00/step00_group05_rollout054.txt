def lp_radii_fixup(centers, radii):
    c = np.ones(n)
    A = squareform(pdist(np.hstack((centers, radii.reshape(-1, 1))), 'euclidean')) - radii.reshape(-1, 1) - radii.reshape(1, -1)
    A = A[A > 0]
    b = np.zeros(A.shape[0])
    res = linprog(c, A_ub=A.T, b_ub=b, method='highs')
    if res.success:
        return res.x
    return radii

def penalty(centers, radii):
    wall_penalty = np.maximum(0, radii - centers[:, 0]) ** 2 + \
                   np.maximum(0, radii - (1 - centers[:, 0])) ** 2 + \
                   np.maximum(0, radii - centers[:, 1]) ** 2 + \
                   np.maximum(0, radii - (1 - centers[:, 1])) ** 2
    overlap_penalty = np.sum(np.maximum(0, radii.reshape(-1, 1) + radii.reshape(1, -1) - squareform(pdist(np.hstack((centers, radii.reshape(-1, 1))), 'euclidean'))))
    return 1e4 * (wall_penalty + 2 * overlap_penalty)

def gradient(centers, radii):
    n = centers.shape[0]
    grad_x = np.zeros_like(centers)
    grad_r = np.zeros(n)
    
    for i in range(n):
        r_i = radii[i]
        node_i = centers[i]
        overlaps = np.where(squareform(pdist(np.hstack((centers, radii.reshape(-1, 1))), 'euclidean')) - radii.reshape(-1, 1) - radii.reshape(1, -1)) > 0
        for j in overlaps[0]:
            if j > i:
                r_j = radii[j]
                node_j = centers[j]
                dist_ij = np.linalg.norm(node_i - node_j)
                if r_i + r_j > dist_ij:
                    grad_r[i] += 2 * (dist_ij - r_i - r_j)
                    grad_x[i] += 2 * (node_i - node_j) / dist_ij
        
        if r_i > min(node_i[0], 1 - node_i[0], node_i[1], 1 - node_i[1]):
            grad_r[i] -= 2 * r_i - min(node_i[0], 1 - node_i[0], node_i[1], 1 - node_i[1])
    
    return np.concatenate([grad_x, grad_r])

def objective(x):

    radii = x[:n, 2:]
    return -np.sum(radii) + penalty(centers, radii)
