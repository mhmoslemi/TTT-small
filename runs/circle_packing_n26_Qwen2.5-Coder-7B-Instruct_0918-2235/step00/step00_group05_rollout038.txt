import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
from scipy.spatial.distance import cdist

def hex_grid(radius, packing_density):
    R = radius
    s = 2 * R / np.sqrt(3)
    grid = np.array([[x, y] for x in np.arange(0, 1, s) for y in np.arange(0, 1, s)])
    centre_counts = np.ones_like(grid[:, 0][:, None], dtype=int)
    while centre_counts.sum() < packing_density * (len(grid) * 3):
        grid = np.append(grid, grid + np.random.uniform(-0.1 * s, 0.1 * s, size=(grid.shape[0], 2)), axis=0)
        centre_counts = np.ones_like(grid[:, 0][:, None], dtype=int)
    return grid

def add_circles_around_quadrant(center, n, radius):
    theta = np.linspace(0, 2 * np.pi, n * 4, endpoint=False)[::4]
    x = center[0] + radius * np.cos(theta)
    y = center[1] + radius * np.sin(theta)
    return np.array([[x_, y_] for x_, y_ in zip(x, y)])

def add_circles(centers, max_radius, total_circles):
    quadrant_radius = max_radius / np.sqrt(7)
   ircles = centers.copy()
    for quadrant in [(0, 0), (0.5, 0), (0.5, 0.5), (0, 0.5)]:
        while circles.shape[0] < total_circles:
            new_circles = add_circles_around_quadrant(quadrant, max_radius // quadrant_radius, quadrant_radius)
            valid = np.logical_and(new_circles[:, 0] >= 0, new_circles[:, 0] <= 1)
            valid &= np.logical_and(new_circles[:, 1] >= 0, new_circles[:, 1] <= 1)
            valid &= np.all(cdist(new_circles, centers[valid], 'euclidean') > 2 * quadrant_radius, axis=1)
            circles = np.append(circles, new_circles[valid], axis=0)
            circles = circles[:total_circles]
    return circles

def solve_radii_fixed_centres(centres):
    n = centres.shape[0]
    radii_ineqs = np.zeros((2*n, 2*n))
    radii_ineqs[:n, np.arange(n)] = 1
    radii_ineqs[n:, np.arange(n)] = -1
    box_ineqs = np.vstack((np.eye(n), -np.eye(n)))
    box_ineqs = np.vstack((box_ineqs, np.tile([1, 1], (n, 1)), np.tile([1, 1], (n, 1)) / (1 - centres[:, 0].reshape(-1, 1)), np.tile([1, 1], (n, 1)) / (1 - centres[:, 1].reshape(-1, 1))))
    constraints = [{'type': 'ineq', 'fun': lambda x, i=i: x[i]}, {'type': 'ineq', 'fun': lambda x, i=i: x[i] - distances[i]} for i in range(n)]
    constraints += [{'type': 'ineq', 'fun': lambda x, i=i: x[i] - box_ineqs[i]} for i in range(2*n)]
    bounds = [(0, None)] * n
    result = minimize(lambda x: -np.sum(x), radii_ineqs @ np.ones(n), method='SLSQP', bounds=bounds, constraints=constraints)
    return radii_ineqs @ result.x, -result.fun

def penalty_centres_fixed_radii(centres, radii):
    distances = cdist(centres, centres)
    penalties = np.maximum((radii[:, None] + radii) - distances, 0)
    boundary_penalties = radii - (np.minimum(centres[:, 0], 1 - centres[:, 0], 0), np.minimum(centres[:, 1], 1 - centres[:, 1], 0)).max(axis=0)
    return penalties.sum() + 1e6 * boundary_penalties.sum()

def coordinated_ascent(init_centres, max_iter=15):
    best_centres = init_centres
    best_sum_radii = -np.inf
    for _ in range(max_iter):
        radii, sum_radii = solve_radii_fixed_centres(init_centres)
        penalty = penalty_centres_fixed_radii(init_centres, radii)
        if sum_radii - penalty > best_sum_radii:
            best_centres = init_centres.copy()
            best_sum_radii = sum_radii - penalty
        init_centres = differential_evolution(lambda centres: penalty_centres_fixed_radii(centres, radii), bounds=[(0, 1), (0, 1)] * 26, args=(radii,), options={'popsize': 25}).x reshape(-1, 2)
    return best_centres, best_sum_radii
