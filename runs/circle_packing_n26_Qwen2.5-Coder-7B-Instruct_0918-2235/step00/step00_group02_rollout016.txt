import numpy as np
from scipy.optimize import linear_program

def compute_sum_radii(positions):
    n = 26
    r = np.zeros(n)
    A_ub = np.zeros((n + n, n + 1))
    b_ub = np.zeros(n + n)
    c = np.zeros(n + 1)
    
    A_ub[:n, 1:] = 1
    b_ub[:n] = 1
    c[1:] = -1
    
    for i in range(n):
        A_ub[n + i, 1:] = [1 if j == i else 0 for j in range(n)]
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt(np.sum((positions[i + n*1:i + n*2] - positions[j + n*1:j + n*2]) ** 2))
            A_ub[n * 2 + i * (n + 1) + j] = 2 * r
            A_ub[n * 2 + i * (n + 1) + j, 0] = 1
            A_ub[n * 2 + i * (n + 1) + j, i + 1] = -1
            A_ub[n * 2 + i * (n + 1) + j, j + 1] = -1
            b_ub[n * 2 + i * (n + 1) + j] = d_ij
            
            A_ub[n * 2 + i * (n + 1) + n * (n - (i + 1)) + j] = 2 * r
            A_ub[n * 2 + i * (n + 1) + n * (n - (i + 1)) + j, 0] = 1
            A_ub[n * 2 + i * (n + 1) + n * (n - (i + 1)) + j, i + 1] = -1
            A_ub[n * 2 + i * (n + 1) + n * (n - (i + 1)) + j, j + 1] = -1
            b_ub[n * 2 + i * (n + 1) + n * (n - (i + 1)) + j] = d_ij
    
    res = linear_program(c, A_ub, b_ub, method='highs')
    return -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    positions = np.linspace(0, 1, 51)[:-1][::2]
    centers = np.array([[x, y] for x in positions for y in positions]).reshape(-1, 2)
    centers = centers[:26]
    sum_radii = compute_sum_radii(centers)
    return centers, np.full(26, sum_radii / 26), sum_radii
