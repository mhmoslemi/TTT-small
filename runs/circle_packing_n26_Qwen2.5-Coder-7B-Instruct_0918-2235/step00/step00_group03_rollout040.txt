import numpy as np
from scipy.optimize import minimize, Bounds

def penalty(vars, n, K):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            penalty += np.max(0, dist_ij - r[i] - r[j])**2
    return K * penalty

def constraint(vars, n):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    n_violations = (x - r) < 0.0
    n_violations += (y - r) < 0.0
    n_violations += (1 - x - r) < 0.0
    n_violations += (1 - y - r) < 0.0
    return n_violations.sum()

    return -vars[2*n:].sum() + penalty(vars, n, K)
