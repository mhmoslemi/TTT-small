from scipy.optimize import minimize, Bounds
import numpy as np

def optimize_for_lambda(lambda_value, init_vars):
    """
    Optimizes the packing problem for a given lambda value.
    
    Args:
        lambda_value: Penalty coefficient for the constraints.
        init_vars: Initial guess for the 78-variable vector.
        
    Returns:
        The optimized variables (shape: 78).
    """
    
    def energy(vars):
        # Extract position and radius variables
        x = vars[:26]
        y = vars[26:52]
        r = vars[52:]
        
        # Calculate the penalty for constraints
        boundary_penalty = lambda_value * np.sum(
            np.maximum(x - r, 0) ** 2 +
            np.maximum(1 - x - r, 0) ** 2 +
            np.maximum(y - r, 0) ** 2 +
            np.maximum(1 - y - r, 0) ** 2
        )
        
        overlap_penalty = lambda_value * np.sum(
            np.maximum(
                np.sqrt((x[:, None] - x) ** 2 + (y[:, None] - y) ** 2) - r[:, None] - r - 1e-12, 0
            ) ** 2
        )
        
        # Total penalty
        penalty = boundary_penalty + overlap_penalty
        
        # Total energy
        total_energy = -np.sum(r) + penalty
        return total_energy
    
    # Set bounds for variables
    pos_bounds = Bounds(0, 1)
    radius_bounds = Bounds(0, 0.5)
    
    # Combine position and radius bounds
    bounds = pos_bounds + pos_bounds + radius_bounds
    
    # Perform the optimization
    result = minimize(
        energy,
        init_vars,
        method='L-BFGS-B',
        jac=None,
        bounds=bounds
    )
    
    return result.x
