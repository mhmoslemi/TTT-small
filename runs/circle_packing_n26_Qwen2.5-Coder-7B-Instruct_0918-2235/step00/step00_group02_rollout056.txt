import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C, 'euclidean')
    A = np.zeros((N * (N - 1) + 4 * N, N))
    b = np.zeros(N * (N - 1) + 4 * N)
    for i in range(N):
        for j in range(i + 1, N):
            A[i * (N - 1) + j - i - 1, i] = 1
            A[i * (N - 1) + j - i - 1, j] = 1
            b[i * (N - 1) + j - i - 1] = d[i, j]
    for i in range(N):
        A[N * (N - 1) + 4 * i, i] = 1
        b[N * (N - 1) + 4 * i] = C[i, 0]
        A[N * (N - 1) + 4 * i + 1, i] = 1
        b[N * (N - 1) + 4 * i + 1] = 1 - C[i, 0]
        A[N * (N - 1) + 4 * i + 2, i] = 1
        b[N * (N - 1) + 4 * i + 2] = C[i, 1]
        A[N * (N - 1) + 4 * i + 3, i] = 1
        b[N * (N - 1) + 4 * i + 3] = 1 - C[i, 1]
    return A, b

def compute_sum_radius(C):
    A, b = build_lp(C)
    if len(A) != 0:
        res = linprog(-np.ones(C.shape[0]), A_ub=A, b_ub=b, method='highs')
        if res.success:
            return -res.fun
        else:
            return 1e6
    else:
        return 1e6

def de_obj(C):
    return compute_sum_radius(C)

def run_packing():
    N = 26
    C = np.random.rand(N, 2)
    popsize = 15 * N
    maxfev = 3000
    result = differential_evolution(de_obj, bounds=[(0, 1) for _ in range(2 * N)], popsize=popsize, maxfev=maxfev, strategy='best1bin', mutation=0.9, recombination=0.7)
    best_C = result.x.reshape(N, 2)
    best_r = np.array([max(0, r) for r in compute_sum_radius(best_C)])
    assert validate_packing(best_C, best_r)[0], "Validation failed"
    return best_C, best_r, np.sum(best_r)
