import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # PSO parameters
    P = 60
    w = 0.7298
    c1 = 1.49618
    c2 = 1.49618
    G = 200
    v_max = np.array([1.0, 1.0, 0.5])

    # Initialisation
    np.random.seed(12345)
    X = np.random.rand(P, 78)
    V = np.zeros((P, 78))
    p_best = X.copy()
    p_bestfit = -np.inf*np.ones(P)
    g_best = p_best[0].copy()
    g_bestfit = -np.inf

    # Early stopping parameters
    bestfit = -np.inf
    bestgen = 0
    patience = 15

    # Penalty multiplier ramp
    λ = 1000

    # PSO iteration
    for g in range(G):
        for p in range(P):
            # Update velocity
            V[p] = w * V[p] + c1 * np.random.rand(78) * (p_best[p] - X[p]) + c2 * np.random.rand(78) * (g_best - X[p])
            # Clip velocities
            V[p] = np.clip(V[p], -v_max, v_max)
            # Update position
            X[p] = X[p] + V[p]
            # Clip positions
            X[p] = np.clip(X[p], np.array([0, 0, 0]), np.array([1, 1, 0.5]))
            # Evaluate fitness
            fit = evaluate_fitness(X[p], λ)
            if fit > p_bestfit[p]:
                p_best[p] = X[p]
                p_bestfit[p] = fit

            if fit > g_bestfit:
                g_best = X[p]
                g_bestfit = fit
                bestfit = fit
                bestgen = g
                patience = 15
            else:
                patience -= 1

        # Adaptive penalty scaling
        λ *= 1.02

        # Early stopping
        if patience == 0:
            break

    # Local refinement
    centers = g_best[:76].reshape(-1, 2)
    radii = g_best[76:] / 2
    refine_radii(centers, radii)
    
    return centers, radii, np.sum(radii)

def evaluate_fitness(v, λ):
    r = v[2::3]
    x = v[0::3]
    y = v[1::3]
    penalty = 0

    # Boundary clearance penalty
    penalty += np.sum(np.clip(-(x - r), 0))
    penalty += np.sum(np.clip(-(1 - x - r), 0))
    penalty += np.sum(np.clip(-(y - r), 0))
    penalty += np.sum(np.clip(-(1 - y - r), 0))

    # Overlap penalty
    for i in range(len(r)):
        for j in range(i + 1, len(r)):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            penalty += np.clip((r[i] + r[j]) - dist, 0)

    return np.sum(r) - λ * penalty

def refine_radii(centers, radii):
    v = np.hstack((centers.flatten(), radii * 2))
    for _ in range(10):
        for i in range(26):
            v[i*3:i*3+3] += np.random.rand(3) * 2e-4 - 1e-4
            v[26*3+i] = max(0, min(1, compute_radius(v[0:i*3+3], v[26*3+i], 26)))
        if evaluate_fitness(v, 1e6) < evaluate_fitness(v, 1e6, update_only=False):
            v[0:78] = v[0:78] * 2
        else:
            v[0:78] = v[0:78] / 2
    
    centers = v[:76].reshape(-1, 2)
    radii = v[76:] / 2

def compute_radius(x, r, n):
    best_r = r
    best_l = 0
    while best_r - best_l > 1e-10:
        mid = 0.5 * (best_r + best_l)
        if all(check_non_overlap(x, r, mid) for _ in range(n)):
            best_l = mid
        else:
            best_r = mid
    return best_r

def check_non_overlap(x, r, cr):
    xi, yi = x[:2]
    ri = x[2]
    dx, dy = xi - ri - cr*np.cos(x[3]), yi - ri - cr*np.sin(x[3])
    return dx**2 + dy**2 > (ri + cr)**2
