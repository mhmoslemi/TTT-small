import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Calculate hexagonal packing parameters
    side_length = np.sqrt(3) / 3
    radius = side_length / (2 * np.sqrt(3))
    
    # Fill the unit square with the hexagonal packing
    row_indices = np.arange(0, n, 2)
    col_indices = np.arange(0, n // 2 + 1, 1)
    
    row_start = 1 / (2 * np.sqrt(3))
    col_start = 0.25
    
    row_offset = row_start
    col_offset = col_start
    
    for i, row_index in enumerate(row_indices):
        for j, col_index in enumerate(col_indices):
            if i % 2 == 0:
                x = col_offset + j * 0.75 * side_length
                y = row_offset + i * side_length
                centers[row_index + i // 2][0] = x
                centers[row_index + i // 2][1] = y
                radii[row_index + i // 2] = radius
            else:
                x = col_offset + j * 0.75 * side_length
                y = row_offset + i * side_length - side_length / 2
                centers[row_index + i // 2][0] = x
                centers[row_index + i // 2][1] = y
                radii[row_index + i // 2] = radius
    
    # Adjust the last row and column to ensure it fits within the unit square
    last_row_index = n - (n // 2 + 1)
    last_col_index = n // 2
    
    last_row_offset = row_start + (n - (n // 2 + 1)) * side_length
    last_col_offset = col_start + (n // 2) * 0.75 * side_length
    
    center_shift = np.array([last_col_offset, last_row_offset])
    centers -= center_shift
    centers *= 0.99  # Scale down slightly to ensure it fits within the unit square
    
    # Calculate the sum of radii
    sum_radii = np.sum(radii)
    
    # Validate the packing
    assert validate_packing(centers, radii)[0], "Invalid packing"
    
    return centers, radii, sum_radii
