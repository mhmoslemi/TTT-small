import numpy as np
from scipy.optimize import minimize

def init_hex_grid():
    grid = np.array([[0.16666667, 0.16666667], [0.16666667, 0.5], [0.16666667, 0.83333333], 
                      [0.5,  0.33333333],          [0.5,  0.66666667], 
                      [0.83333333, 0.16666667], [0.83333333, 0.5], [0.83333333, 0.83333333],
                      [0.33333333, 0.33333333], [0.33333333, 0.66666667], [0.66666667, 0.33333333], 
                      [0.66666667, 0.66666667],
                      [0.33333333, 0.83333333], [0.5,    0.83333333], [0.66666667, 0.83333333]])
    centres = []
    for i in range(5):
        for j in range(i, 5):
            for offset_x in [-0.25, 0.25]:
                for offset_y in [-0.25, 0.25]:
                    centre = grid[i] + np.array([offset_x, offset_y])
                    if not (centre[0] < 0 or centre[0] > 1 or centre[1] < 0 or centre[1] > 1):
                        centres.append(centre)
    centres = np.array(centres)
    np.random.shuffle(centres)
    return centres[:26].reshape(-1, 2), np.array([0.01]*26)

def build_variable_vector(centres, radii):
    return np.concatenate((centres.flatten(), radii))

def unpack(z):
    n = 26
    x = z[:52].reshape(n, 2)
    r = z[52:]
    return x, y, r

def compute_boundary_penalty(x, y, r):
    penalty = 0
    for i in range(n):
        penalty += max(0, r[i] - x[i])**2
        penalty += max(0, r[i] - (1 - x[i]))**2
        penalty += max(0, r[i] - y[i])**2
        penalty += max(0, r[i] - (1 - y[i]))**2
    return penalty

def compute_overlap_penalty(x, y, r):
    n = 26
    penalty = 0
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
            penalty += max(0, r[i] + r[j] - d_ij)**2
    return penalty

def objective(z):
    x, y, r = unpack(z)
    lambda_B = 10**6
    lambda_O = 10**6
    return -np.sum(r) + lambda_B * compute_boundary_penalty(x, y, r) + lambda_O * compute_overlap_penalty(x, y, r)

def jacobian(z):
    n = 26
    x, y, r = unpack(z)
    grad = np.zeros(78)
    for i in range(n):
        grad[i] = -2 * max(0, r[i] - x[i]) * (x[i] - 1)
        grad[i] += -2 * max(0, r[i] - (1 - x[i])) * (1 - x[i])
        grad[i] += -2 * max(0, r[i] - y[i]) * (y[i] - 1)
        grad[i] += -2 * max(0, r[i] - (1 - y[i])) * (1 - y[i])

    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
            grad[i] += -2 * max(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij
            grad[i] += -2 * max(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij
            grad[j] += -2 * max(0, r[i] + r[j] - d_ij) * (x[j] - x[i]) / d_ij
            grad[j] += -2 * max(0, r[i] + r[j] - d_ij) * (y[j] - y[i]) / d_ij

    grad[52:] = -r
    return grad

def solve_lp(centres):
    return np.array([np.min([c, 1-c]) for c in centres.flatten()]), None

def run_packing():
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)
    bounds = [(0,1)]*52 + [(0,None)]*26
    
    result = minimize(objective, z,
                      jac=jacobian,
                      method='trust-constr',
                      bounds=bounds,
                      options={'gtol':1e-6, 'ftol':1e-8, 'maxiter':3000})
    
    centres_opt, radii_opt = unpack(result.x)
    radii_opt, _ = solve_lp(centres_opt)
    
    sumRadii = np.sum(radii_opt)
    return centres_opt, radii_opt, sumRadii
