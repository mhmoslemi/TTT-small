import numpy as np
import scipy.optimize as opt
import scipy.spatial

def calc_max_radii(C):
    n = C.shape[0]
    d = scipy.spatial.distance_matrix(C, C)
    d[d <= 0] = np.inf
    
    # Compute max possible radii if all circles are scaled up to same size
    R = C.max(0) - C.min(0)
    max_radii = np.min(R / 2)
    
    # Construct LP problem
    A = np.zeros((n * 3 + n * (n - 1) // 2, n))
    b = np.zeros(n * 3 + n * (n - 1) // 2)
    
    # Set boundary conditions
    for i in range(n):
        A[i * 3, 2 * i: 2 * i + 2] = [1, 0]
        A[i * 3 + 1, 2 * i: 2 * i + 2] = [0, 1]
        b[i * 3] = C[i, 0]
        b[i * 3 + 1] = C[i, 1]
    
    # Set circle-circle constraints
    counter = 2 * n
    for i in range(n):
        for j in range(i + 1, n):
            A[counter, 2 * i: 2 * i + 2] = [1, 0]
            A[counter, 2 * j: 2 * j + 2] = [-1, 0]
            b[counter] = d[i, j]
            counter += 1
            A[counter, 2 * i: 2 * i + 2] = [0, 1]
            A[counter, 2 * j: 2 * j + 2] = [0, -1]
            b[counter] = d[i, j]
            counter += 1
    
    # Solve LP
    A_ub = A[ :, A.shape[1] // 2 : ]
    b_ub = b[ A.shape[0] // 2 : ]
    A_eq = A[ : A.shape[0] // 2 , : A.shape[1] // 2 ]
    b_eq = b[ : A.shape[0] // 2 ]
    
    lp_res = opt.linprog(
        -max_radii * np.ones(n), 
        A_ub=A_ub, b_ub=b_ub, 
        A_eq=A_eq, b_eq=b_eq, 
        bounds=(0, 1), 
        options={'method': 'highs'}
    )
    
    if lp_res.success:
        r = lp_res.x
        r[r < 0] = 0  # Avoid numerical noise
        return r, r.sum()
    else:
        raise Exception("LP solver failed")

def run_packing():
    n = 26
    initial_radius = 0.1055
    centers = np.zeros((n, 2))
    radii = np.full(n, initial_radius)
    
    # Initial configuration: (hexagonal type) layout
    for i in range(n):
        r = 26 - 5  # Number of circles (r=26, r=15, ...)
        C = np.array([
            0.5,
            0.5 * np.sqrt(3),
            0.0,
            0.5 * np.sqrt(3) + 1,
            -0.5 * np.sqrt(3) - 1
        ])
        C += np.random.randn(5, 2) * 0.1
        C = C[C[:, 0] >= 0, :]
        C = C[C[:, 0] <= 1, :]
        C = C[C[:, 1] >= 0, :]
        C = C[C[:, 1] <= 1, :]
        centers[i%r, 0] = C[i%r, 0]
        centers[i%r, 1] = C[i%r, 1]
    
    # Ensuring radius constraints
    scale_factor = 0.95
    centers *= scale_factor
    centers[:, 0] += 0.025
    centers[:, 1] += 0.025
    centers[:, 0] -= centers[:, 0].min()
    centers[:, 1] -= centers[:, 1].min()
    centers[:, 0] /= centers[:, 0].max()
    centers[:, 1] /= centers[:, 1].max()
    
    bound_radius = 1 - centers.max(axis=0) - centers.min(axis=0)
    bound_radius = bound_radius.min()
    centers = centers * bound_radius / 2
    internals = centers[np.argsort(centers[:, 0] + centers[:, 1])]
    
    best_centers = centers
    best_sum = 0
    
    for _ in range(100):
        centers = internals + np.random.randn(*internals.shape) * 0.1
        centers = np.clip(centers, 0, 1)
        try:
            r, S = calc_max_radii(centers)
            if S > best_sum:
                best_sum = S
                best_centers = centers
        except Exception:
            continue
    
    # Additional fine-tuning
    while best_sum < 2.636:
        mask = np.array([best_centers[:, 0].sum() / best_centers.shape[0], best_centers[:, 1].sum() / best_centers.shape[0]], dtype=int)
        order = np.argsort(mask).tolist() + [26]
        for i in order:
            if best_sum >= 2.636:
                break
            best_centers[i, 0] *= 1.01
            best_centers[i, 1] *= 1.01
            centers = np.clip(best_centers, 0, 1)
            try:
                r, S = calc_max_radii(best_centers)
                if S > best_sum:
                    best_sum = S
                    best_centers = centers
            except Exception:
                continue
    
    return best_centers, r, best_sum
