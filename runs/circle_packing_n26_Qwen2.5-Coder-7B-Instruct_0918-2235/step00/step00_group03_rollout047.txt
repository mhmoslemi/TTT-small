import numpy as np
from scipy.optimize import minimize, Bounds

def penalty(vars, n):
    x = vars[:n].reshape(n, 1)
    y = vars[n:n*2].reshape(n, 1)
    r = vars[n*2:n*3].reshape(n, 1)
    s_overlap = np.sqrt((x - x.T)**2 + (y - y.T)**2) - (r + r.T)
    penalty_overlap = np.sum(np.maximum(0, -s_overlap)**2)
    penalty_boundary = 0
    return penalty_overlap + penalty_boundary

def energy(vars, n, lambda_):
    sum_r = np.sum(vars[n*2:n*3])
    penalty_value = penalty(vars, n)
    return -sum_r + lambda_ * penalty_value

def optimize_for_lambda(lambda_, init_vars):
    n = init_vars.shape[0] // 3
    bounds = Bounds(0, 1)
    result = minimize(energy, init_vars, args=(n, lambda_), method='L-BFGS-B', jac='2-point', bounds=bounds)
    return result.x

def run_packing():
    n = 26
    lambda_0 = 1.0
    lambda_factor = 1.5
    max_lambda = 1e5
    gradient_tolerance = 1e-6
    sum_radii_tolerance = 1e-8
    restarts = 50
    best_result = None
    best_sum_radii = -np.inf
    
    np.random.seed(0)
    for _ in range(restarts):
        init_vars = np.random.random(n * 3)  # X, Y, R
        init_vars *= 0.5  # Scale up initial R slightly
        init_lambda = lambda_0
        current_sum_radii = -np.inf
        for k in range(100):
            current_sum_radii = -energy(init_vars, n, init_lambda)[0]
            if current_sum_radii - best_sum_radii < sum_radii_tolerance:
                break
            init_vars = optimize_for_lambda(init_lambda, init_vars)
            init_lambda *= lambda_factor
            if init_lambda > max_lambda:
                break
        if current_sum_radii > best_sum_radii:
            best_sum_radii = current_sum_radii
            best_result = init_vars
    
    centers = best_result[:n*2].reshape(n, 2)
    radii = best_result[n*2:n*3]
    sum_radii = best_sum_radii
    return centers, radii, sum_radii
