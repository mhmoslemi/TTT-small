def generate_centres(theta, dx, dy, s):
    raw_lattice = np.zeros((3, 9, 2))
    raw_lattice[0, :, 0] = np.arange(-4, 5)
    raw_lattice[0, :, 1] = 0
    raw_lattice[1, :, 0] = np.arange(-3, 5)
    raw_lattice[1, :, 1] = 1
    raw_lattice[2, :, 0] = np.arange(-4, 5)
    raw_lattice[2, :, 1] = 2

    lattice = raw_lattice.reshape(-1, 2) @ [[s / sqrt(3), s * 1 / 2], [0, s * sqrt(3) / 3]]
    lattice = lattice @ [[cos(theta), -sin(theta)], [sin(theta), cos(theta)]]
    lattice += np.array([[3 * s * cos(theta), 3 * s * sin(theta)], 
                          [2 * s * cos(theta), 2 * s * sin(theta)], 
                          [s * cos(theta), s * sin(theta)]]) @ np.array([[dx], [dy]])
    return lattice

def random_perturb(best_z):
    theta, dx, dy = best_z

    dx += np.random.uniform(-0.002, 0.002)
    dy += np.random.uniform(-0.002, 0.002)
    return theta, dx, dy
