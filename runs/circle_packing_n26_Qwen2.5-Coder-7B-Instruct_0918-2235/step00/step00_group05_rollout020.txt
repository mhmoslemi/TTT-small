import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog, minimize
from math import pi, cos, sin, linspace

def check_feasibility(centres, radii):
    for i in range(len(centres)):
        if radii[i] < 0:
            return False
        for j in range(i + 1, len(centres)):
            dist = distance.euclidean(centres[i], centres[j])
            if dist < (radii[i] + radii[j]):
                return False
        x, y = centres[i]
        if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
            return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def generate_hexagonal_grid(s, offset):
        centers = []
        for row in range(3):
            for n1 in range(-4 + row, 5 - row):
                n2 = row - n1
                x = n1 * s * 0.5 + n2 * s
                y = n1 * s * np.sqrt(3) * 0.5
                for dx, dy in [(-1, -1), (-1, 0), (-1, 1), (0, -1), (0, 1)]:
                    centers.append(np.array([x + offset[0] + dx * 0.01, y + offset[1] + dy * 0.01]))
        return np.array(centers)

    def solve_radius_lp(centers):
        n = len(centers)
        A = np.vstack((np.eye(n), -np.eye(n), centers[:, 0], centers[:, 0] - 1, centers[:, 1], centers[:, 1] - 1))
        b = np.hstack((np.ones(n), npzeros(n), centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]))
        res = linprog([], A_ub=A, b_ub=b, method='simplex')
        if not check_feasibility(centers, res.x):
            raise ValueError("Feasibility check failed")
        return res.x, res.success

    def objective(pos):
        offset = pos[:2]
        rotation_angle = pos[2]
        r0 = 0.09
        centre_rotation_matrix = np.array([[cos(rotation_angle), -sin(rotation_angle)],
                                             [sin(rotation_angle), cos(rotation_angle)]])
        centres = centers_base.copy()
        centres = np.dot(centre_rotation_matrix, centres.T).T + offset
        centres = np.clip(centres, 0, 1)
        lp_radii, success = solve_radius_lp(centres)
        if not success:
            raise ValueError("LP optimization failed")
        return 1.0 - np.sum(lp_radii)  # maximize sum of radii

    def random_perturb(pos, seed):
        np.random.seed(seed)
        dx = np.random.uniform(-0.002, 0.002)
        dy = np.random.uniform(-0.002, 0.002)
        angle = np.random.uniform(-np.pi / 6, np.pi / 6)
        return np.array([dx, dy, angle])

    def are_close(pos1, pos2):
        return np.allclose(pos1, pos2, atol=1e-6)

    n_points = 26
    s_initial = 0.09
    centres_base = generate_hexagonal_grid(s_initial, np.array([0.5, 0.5]))
    bounds = ((-0.05, 0.05), (-0.05, 0.05), (0, pi / 6))

    best_objective = np.inf
    best_centers = None
    best_radii = None

    for _ in range(100):
        pos = np.array([np.random.uniform(-0.05, 0.05), np.random.uniform(-0.05, 0.05), np.random.uniform(0, pi / 6)])
        while True:
            pos = minimize(objective, pos, method='Nelder-Mead', bounds=bounds).x
            offset = pos[:2]
            rotation_angle = pos[2]
            centre_rotation_matrix = np.array([[cos(rotation_angle), -sin(rotation_angle)],
                                                 [sin(rotation_angle), cos(rotation_angle)]])
            centres = centres_base.copy()
            centres = np.dot(centre_rotation_matrix, centres.T).T + offset
            centre = np.array([[0.5], [0.5]])
            centre = centre_rotation_matrix.dot(centre) + offset
            distance_center = np.linalg.norm(centres - centre, axis=1)
            radius = np.minimum((0.5 * np.sqrt(2) - distance_center), np.zeros(len(distance_center)))
            if check_feasibility(centres, radius):
                objective_val = -np.sum(radius)
                if objective_val < best_objective:
                    best_objective = objective_val
                    best_centers = centres
                    best_radii = radius
                break
            else:
                pos = random_perturb(pos, _)

    return best_centers, best_radii, -best_objective
