import numpy as np
from scipy.optimize import linprog

def solve_radius_lp(centres_i):
    n = centres_i.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((n + n * (n - 1) // 2, n))
    b_ub = np.zeros(n + n * (n - 1) // 2)
    
    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = centres_i[i, 0]
        A_ub[i + n, i] = 1
        b_ub[i + n] = centres_i[i, 1]
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centres_i[i] - centres_i[j])
            idx = n + n * (i + 1) // 2 + i + j - (i + 2)
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x, -res.fun
    else:
        raise Exception("LP solution failed")

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n_particles = 30
    n_circles = 26
    dx = dy = 1 / 5.0
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    max_iter = 3000
    vel_clip = 0.1
    alpha = 0.999
    
    # Initialize grid points
    grid_x, grid_y = np.meshgrid(np.arange(0.5, 5.5, dx), np.arange(0.5, 6.5, dy))
    grid_flat = np.column_stack((grid_x.flatten(), grid_y.flatten()))
    
    # Select 26 random indices to be the candidate points
    candidate_indices = np.random.choice(grid_flat.shape[0], n_circles, replace=False)
    candidate_points = grid_flat[candidate_indices]
    
    # Initialize particles randomly within a Gaussian perturbation
    particles = candidate_points + 0.01 * np.random.randn(n_particles, n_circles, 2)
    velocities = np.zeros_like(particles)
    pbests = particles.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = particles[0].copy()
    gbest_sum = solve_radius_lp(gbest)[1]
    
    for _ in range(max_iter):
        if np.random.rand() < 0.01:  # Restart with new grid occasionally
            candidate_indices = np.random.choice(grid_flat.shape[0], n_circles, replace=False)
            candidate_points = grid_flat[candidate_indices]
            particles = candidate_points + 0.01 * np.random.randn(n_particles, n_circles, 2)
            velocities = np.zeros_like(particles)
            pbests = particles.copy()
            pbest_sums = np.zeros(n_particles)
            gbest = particles[0].copy()
            gbest_sum = solve_radius_lp(gbest)[1]
        
        for i in range(n_particles):
            for j in range(n_circles):
                r1, r2 = np.random.rand(2)
                velocities[i, j] = inertia * velocities[i, j] + cognitive * r1 * (pbests[i, j] - particles[i, j]) + social * r2 * (gbest[j] - particles[i, j])
                velocities[i, j] = np.clip(velocities[i, j], -vel_clip, vel_clip)
                particles[i, j] += velocities[i, j]
                particles[i, j] = particles[i, j].clip(0, 1)
            
            radii, sum_radii = solve_radius_lp(particles[i])
            if sum_radii > pbest_sums[i]:
                pbests[i] = particles[i].copy()
                pbest_sums[i] = sum_radii
                if sum_radii > gbest_sum:
                    gbest = particles[i].copy()
                    gbest_sum = sum_radii
        
        if alpha is not None:
            velocities *= alpha
    
    best_centres = gbest
    best_radii, best_sum = solve_radius_lp(best_centres)
    
    return best_centres, best_radii, best_sum
