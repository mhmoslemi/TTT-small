```python
import numpy as np
from scipy.optimize import linprog

# PSO parameters
n_particles = 30
inertia = 0.729
cognitive = 1.49445
social = 1.49445
max_iter = 3000
vel_clip = 0.1
alpha = 0.999

# Seed configuration
grid = np.array([
    [0.2, 0.2],
    [0.4, 0.2],
    [0.6, 0.2],
    [0.8, 0.2],
    [0.1, 0.4],
    [0.3, 0.4],
    [0.5, 0.4],
    [0.7, 0.4],
    [0.9, 0.4],
    [0.2, 0.6],
    [0.4, 0.6],
    [0.6, 0.6],
    [0.8, 0.6],
    [0.1, 0.8],
    [0.3, 0.8],
    [0.5, 0.8],
    [0.7, 0.8],
    [0.9, 0.8],
    [0.2, 0.1],
    [0.4, 0.1],
    [0.6, 0.1],
    [0.8, 0.1],
    [0.2, 0.3],
    [0.4, 0.3],
    [0.6, 0.3],
    [0.8, 0.3],
    [0.2, 0.7],
    [0.4, 0.7],
    [0.6, 0.7],
    [0.8, 0.7],
    [0.2, 0.9],
    [0.4, 0.9],
    [0.6, 0.9],
    [0.8, 0.9]
])

# Initialize populations and parameters
centres = np.random.rand(n_particles, 26, 2) * 0.8 + grid * 0.1
velocities = np.zeros((n_particles, 26, 2))
pbests = centres.copy()
pbest_sums = -np.inf * np.ones(n_particles)
gbest = centres[np.argmax(pbest_sums)]
gbest_sum = pbest_sums.max()

# Linear programming wrapper
def solve_radius_lp(centres_i):
    n_circles = len(centres_i)
    c = -np.ones(n_circles)
    A = np.empty((n_circles, n_circles))
    b = np.zeros(n_circles)
    
    for i in range(n_circles):
        A[i, i] = 1
        A[i, i:] = np.clip(centres_i[i]-centres_i[i+1:], 0, None)
        A[i+1:] -= A[i, i:]
        b[i] = centres_i[i, np.clip(centres_i[i]//0.2, 0, n_circles-1)]
    
    res = linprog(c, A_ub=A, b_ub=b, method='highs')
    return res.x, -res.fun

# PSO main loop
for it in range(max_iter):
    r1, r2 = np.random.rand(n_particles, 1, 1), np.random.rand(n_particles, 1, 1)
    
    for i in range(n_particles):
        velocities[i] = inertia * velocities[i] +\
            cognitive * r1 * (pbests[i] - centres[i]) +\
            social * r2 * (gbest - centres[i])
        
        velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)
        centres[i] = np.round(centres[i] + velocities[i], 6)
        centres[i] = np.clip(centres[i], 0, 1)
        
        radii, sum_radii = solve_radius_lp(centres[i])
        
        if sum_radii > pbest_sums[i]:
            pbests[i], pbest_sums[i] = centres[i], sum_radii
        
        if sum_radii > gbest_sum:
            gbest, gbest_sum = centres[i], sum_radii
    
    if it % 500 == 0:
        centres = gbest + (np.random.rand(n_particles, len(centres), 2) - 0.5) * 0.01
        centres = np.round(centres, 6)
        centres = np.clip(centres, 0, 1)
        for i in range(n_particles):
            radii, sum_radii = solve_radius_lp(centres[i])
            if sum_radii > pbest_sums[i]:
                pbests[i], pbest_sums[i] = centres[i], sum_radii
            if sum_radii > gbest_sum:
                gbest, gbest_sum = centres[i], sum_radii

return gbest, solve_radius_lp(gbest)[0], gbest_sum
```
