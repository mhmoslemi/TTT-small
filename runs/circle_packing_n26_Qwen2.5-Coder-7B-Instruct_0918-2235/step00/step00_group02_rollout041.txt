import numpy as np
import scipy.spatial
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize variables
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    sum_radii = 0

    # Define a function to calculate the distance between two circles
    def distance(circle1, circle2):
        return np.linalg.norm(circle1[:2] - circle2[:2])

    # Place a circle at the center of the square
    centers[0] = np.array([0.5, 0.5])
    radii[0] = 0.236
    sum_radii += radii[0]

    # Place circles in a hexagonal packing pattern
    angle = 2 * math.pi / 6
    radius = 0.236
    for i in range(1, n):
        row = i // 6
        col = (i % 6) // 2
        if i % 2 == 0:
            offset = (col + 0.5) * 2 * radius / math.sqrt(3)
        else:
            offset = (col + 1) * 2 * radius / math.sqrt(3)
        centers[i] = np.array([offset / 2 + 0.25, row * radius * math.sqrt(3)])
        centers[i][0] += 0.25 * (1 if row % 2 == 1 else -1)
        centers[i][1] += 0.5 * (row % 2)
        radii[i] = radius * (math.sqrt(2) if (col + row) % 2 == 0 else math.sqrt(2) * 0.5)
        sum_radii += radii[i]

    # Adjust the radii and centers to maximize the sum of radii
    for _ in range(100):
        tree = scipy.spatial.cKDTree(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = tree.query_ball_point(centers[i], radii[i] + radii[j], workers=-1)
                if len(dist) > 1:
                    radii[dist[1]] -= 0.01
                    sum_radii -= 0.01

    return centers, radii, sum_radii
