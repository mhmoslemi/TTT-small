import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist
from scipy.stats import uniform

### Parameters
EPSILON = 1e-3
STEP_SIZE = 0.01

def overlap_penalty(centers, radii):
    dists = cdist(centers, centers)
    overlap_penalty = np.sum((np.maximum(0, radii[:, None] + radii - dists) ** 2))
    return overlap_penalty

def wall_penalty(centers, radii):
    penalty = np.sum(
        np.maximum(np.zeros_like(radii), radii - centers[:, 0]) ** 2) + \
        np.sum(np.maximum(np.zeros_like(radii), radii - (1 - centers[:, 0])) ** 2) + \
        np.sum(np.maximum(np.zeros_like(radii), radii - centers[:, 1]) ** 2) + \
        np.sum(np.maximum(np.zeros_like(radii), radii - (1 - centers[:, 1])) ** 2) 
    return penalty

def total_penalty(centers, radii):
    return EPSILON * overlap_penalty(centers, radii) + wall_penalty(centers, radii)

def sum_radii(centers, radii):
    return -np.sum(radii)

def gradient(centers, radii):
    grad_xi = np.zeros_like(centers[:, 0])
    grad_yi = np.zeros_like(centers[:, 0])
    grad_r = np.zeros_like(radii)
    dists = cdist(centers, centers)
    active = 2 * radii[:, None] + radii >= dists
    grad_xi += -STEP_SIZE * np.sum(active * (radii + radii[:, None] - dists)[:, :, None] * 
                                  (centers[active[:, :, 1] &, centers[active[:, :, 2]])[:, 0, :] - centers[:, 0, None, :]) * 
                                  (1 / dists[active[:, :, 1], active[:, :, 2]]), axis=(0, 1))
    grad_yi += -STEP_SIZE * np.sum(active * (radii + radii[:, None] - dists)[:, :, None] * 
                                  (centers[active[:, :, 1], centers[active[:, :, 2]])[:, 1, :] - centers[:, 1, None, :]) * 
                                  (1 / dists[active[:, :, 1], active[:, :, 2]]), axis=(0, 1))
    grad_r += np.ones_like(radii)
    return np.concatenate((grad_xi, grad_yi, grad_r))

def lp_radii_fixup(centers, radii):
    n = centers.shape[0]
    A_wall = np.vstack([np.eye(n) for _ in range(4)])
    b_wall = np.zeros(4 * n)
    for i in range(n):
        b_wall[i], b_wall[n+i], b_wall[2*n+i], b_wall[3*n+i] = centers[i, 0] - radii[i], centers[i, 1] - radii[i], 1 - centers[i, 0] - radii[i], 1 - centers[i, 1] - radii[i]
    
    A_overlap = np.zeros((n * n, n))
    for i in range(n):
        for j in range(i + 1, n):
            A_overlap[i*n + j], A_overlap[j*n + i] = 1, 1
            A_overlap[i*n + j, j] = -1
            A_overlap[j*n + i, i] = -1
    
    b_overlap = np.zeros(n * n)
    for i in range(n):
        for j in range(i + 1, n):
            b_overlap[i*n + j], b_overlap[j*n + i] = radii[i] + radii[j] - np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
    
    result = linprog(-radii, A_ub=np.vstack((A_wall, A_overlap)), b_ub=np.hstack((b_wall, b_overlap)), bounds=(0, radii.max()))
    if result.success:
        return result.x
    else:
        return radii

def initial_centers(n):
    lattice = np.array([[i % 3 // 1, (i % 3) % 2] for i in range(n)]) / 2
    jitter = np.random.uniform(-STEP_SIZE/2, STEP_SIZE/2, (n, 2))
    centers = np.minimum(np.maximum(lattice + jitter, 0), 1)
    radii = np.min(centers, axis=1)
    return centers, radii
