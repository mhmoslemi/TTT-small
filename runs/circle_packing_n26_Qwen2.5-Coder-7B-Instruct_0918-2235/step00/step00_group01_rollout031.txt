import numpy as np
from scipy.optimize import linprog, differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)

    def objective(u):
        x = u[:26]
        y = u[26:]
        d = np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2)
        boundary = np.minimum(x, 1 - x, y, 1 - y)
        n = len(x)
        c = np.ones(n)
        A_ub = np.zeros((n * (n - 1) + 4 * n, n))
        b_ub = np.zeros(n * (n - 1) + 4 * n)
        row = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[row, i] = 1
                A_ub[row, j] = 1
                b_ub[row] = d[i, j]
                row += 1
        for i in range(n):
            A_ub[row, i] = 1
            b_ub[row] = boundary[i]
            row += 1
            A_ub[row, i] = 1
            b_ub[row] = boundary[i]
            row += 1
            A_ub[row, i] = 1
            b_ub[row] = boundary[i]
            row += 1
            A_ub[row, i] = 1
            b_ub[row] = boundary[i]
            row += 1
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        return -result.fun

    bounds = [(0, 1)] * 52
    result = differential_evolution(objective, bounds, popsize=20, maxiter=200, tol=1e-12)

    best_u = result.x
    best_x = best_u[:26]
    best_y = best_u[26:]
    d = np.sqrt((best_x[:, None] - best_x[None, :])**2 + (best_y[:, None] - best_y[None, :])**2)
    boundary = np.minimum(best_x, 1 - best_x, best_y, 1 - best_y)
    c = np.ones(26)
    A_ub = np.zeros((26 * 26 + 4 * 26, 26))
    b_ub = np.zeros(26 * 26 + 4 * 26)
    row = 0
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub[row] = d[i, j]
            row += 1
    for i in range(26):
        A_ub[row, i] = 1
        b_ub[row] = boundary[i]
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = boundary[i]
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = boundary[i]
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = boundary[i]
        row += 1
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')

    centers = np.column_stack((best_x, best_y))
    radii = result.x
    sum_radii = radii.sum()

    return centers, radii, sum_radii
