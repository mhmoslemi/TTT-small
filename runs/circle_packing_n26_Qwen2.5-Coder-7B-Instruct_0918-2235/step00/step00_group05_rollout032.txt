import numpy as np
from scipy.optimize import minimize as scipy_minimize, linprog as scipy_linprog
from scipy.spatial.distance import pdist, squareform
from scipy.stats import multivariate_normal

def polynomial_interpolation(x, y):
    # Polynomial (degree 2) interpolation for given points (x, y)
    # Co>_2 -> correctness, Co>_3 -> not too complex
    # Assembly of a 3x3 matrix using the points
    A = np.array([[x[0]**2, x[0], 1], 
                  [x[1]**2, x[1], 1],
                  [x[2]**2, x[2], 1]])
    b = np.array([y[0], y[1], y[2]])
    # Solving the system of equations to find coefficients of the quadratic polynomial
    coefficients = np.linalg.solve(A, b)
    return coefficients

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_centers = None
    best_radii = None
    best_sum_radii = 0.0
    max_iterations = 8

    for _ in range(max_iterations):
        # Initialize centers using hexagonal lattice with jitter
        centers = np.zeros((n, 2))
        radius_bound = 0.5
        jitter_radius = radius_bound / 10
        radius_grid_points = np.linspace(0, 1 - 2 * radius_bound, n)

        for i in range(int(np.sqrt(n / 2))):
            angle = 2 * np.pi * i / int(np.sqrt(n / 2))
            x = radius_grid_points[i % int(np.sqrt(n / 2))]
            y = radius_grid_points[i // (int(np.sqrt(n / 2)))]
            centers[2 * i] = [(1 + x * np.cos(angle) + jitter_radius * np.random.normal()) % 1, 
                             (1 + y * np.sin(angle) + jitter_radius * np.random.normal()) % 1]
            centers[2 * i + 1] = [(1 + x * np.cos(angle + np.pi) + jitter_radius * np.random.normal()) % 1, 
                                 (1 + y * np.sin(angle + np.pi) + jitter_radius * np.random.normal()) % 1]

        # Add jitter to centers
        jitter_radius /= 3
        centers += jitter_radius * np.random.choice([1, -1], size=(n, 2)) / 2

        # Run L-BFGS-B to optimize centers and radii
        def objective(params):
            xs = params[:n]
            ys = params[n:n*2]
            rs = params[n*2:]
            walls_violation = np.sum(np.where(xs <= rs + 1, (rs + 1) - xs, 0) ** 2
                                     + np.where(ys <= rs + 1, (rs + 1) - ys, 0) ** 2)
            overlaps_violation = np.sum(np.where(np.any(params[n:].reshape(n, 2)-
                                                         params[n:].reshape(n, 2)[np.newaxis, :] == 0,
                                                         axis=2) & (rs[:, np.newaxis] + rs[np.newaxis, :] < 2),
                                                 (rs[:, np.newaxis] + rs[np.newaxis, :]) - np.sqrt(np.sum((params[n:].reshape(n, 2)-
                                                                 params[n:].reshape(n, 2)[np.newaxis, :] ** 2, axis=2)),
                                                 0) ** 2, 0))
            pen = 1e4 * walls_violation + 1e4 * overlaps_violation
            return np.sum(rs) - pen

        def constraint_center(params):
            return params[n*2:] <= 0.5

        def constraint_radius(params):
            return params[n*2:] >= 0

        res = scipy_minimize(objective, np.concatenate((centers.flatten(), np.ones(n))), 
                             method='L-BFGS-B', 
                             bounds=[(0, 1), (0, 1)] * n + [0, 0.5] * n,
                             constraints=[{'type': 'ineq', 'fun': constraint_radius},
                                          {'type': 'ineq', 'fun': constraint_center}])

        best_res = res
        params = best_res.x
        xs = params[:n]
        ys = params[n:n*2]
        rs = params[n*2:]

        # Apply LP fix-up to optimize radii precisely
        c = -np.ones(n)  # maximize
        A = np.hstack([np.eye(n), np.sqrt(scalar_product_matrix(xs, ys, rs)), 
                        np.sqrt(scalar_product_matrix(xs, ys, rs))])
        b = np.concatenate([np.zeros(n), rs, rs-0.0001])  # r_i <= x_i, 1-x_i, y_i, 1-y_i
        res_lp = scipy_linprog(c, A_ub=A, b_ub=b)
        if res_lp.success:
            rs = res_lp.x[n*2:]

        # Update best packing
        current_sum_radii = np.sum(rs)
        if current_sum_radii > best_sum_radii:
            best_sum_radii = current_sum_radii
            best_centers = np.array([xs, ys]).T
            best_radii = rs

    return best_centers, best_radii, best_sum_radii

def scalar_product_matrix(xs, ys, rs):
    return np.array([[1+2*r*np.sin(np.pi*i/2)*(np.cos(np.pi*j/2)) for j in range(len(rs))] for i in range(len(rs))])

# Call the run_packing function and validate the result
centers, radii, sum_radii = run_packing()
assert validate_packing(centers, radii) == (True, "ok")

print("Centers:", centers)
print("Radii:", radii)
print("Sum of radii:", sum_radii)
