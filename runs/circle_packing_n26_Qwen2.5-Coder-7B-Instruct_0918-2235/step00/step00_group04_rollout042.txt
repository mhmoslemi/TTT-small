import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def objective_and_gradient(c, centers, radius_multipliers, A_ub, b_ub):
    n = centers.shape[0]
    dualments = radius_multipliers

    # Constraints: boundaries and circles touching zero or one edge
    b = np.concatenate((np.ravel(centers[:, :n]), np.ravel(centers[:, n:])))
    A = np.vstack((np.eye(n).repeat(2), 
                  np.eye(n, n, 1).repeat(2),
                  np.eye(n, n, -1).repeat(2),
                  np.eye(n) + np.eye(n, n, 1),
                  np.eye(n) - np.eye(n, n, 1))).reshape(4*n + n*(n-1)//2, 2*n)

    # Ensure the circles fit in the unit square
    radii = A_ub @ c >= b
    penalty = np.sum(0.1 * np.where(radii, 1.0, 0.0))

    # Evaluate objective and gradient
    distances = squareform(pdist(centers[:, :n]))
    pen = np.clip(distances - (A[4*n + n*np.arange(n):, :2*n] @ c).reshape(n, n), 0, None)
    total_radius_sum = np.sum(c[:n] + c[n:])
    objective = -total_radius_sum - penalty
    gradient = -(A @ c - b) / distances.diagonal().repeat(2)
    gradient[:n] += dualments[:-4*n][:, 0]
    gradient[n:] += dualments[:-4*n][:, 1]

    return objective, gradient
