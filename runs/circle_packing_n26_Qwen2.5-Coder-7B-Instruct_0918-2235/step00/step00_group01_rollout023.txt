import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

# Global variables
n = 26
pair_indices = [(i, j) for i in range(n) for j in range(i+1,n)]  # precomputed pair indices

def constraints(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        cons.append(xs[i] - rs[i])  # boundary lower x
        cons.append(1 - xs[i] - rs[i])  # boundary upper x
        cons.append(ys[i] - rs[i])  # boundary lower y
        cons.append(1 - ys[i] - rs[i])  # boundary upper y
    for i, j in pair_indices:
        dx = xs[i] - xs[j]
        dy = ys[i] - ys[j]
        dist2 = dx**2 + dy**2
        cons.append(dist2 - (rs[i] + rs[j])**2)  # pairwise non-overlap
    return np.array(cons)

def objective(v):
    return -np.sum(v[2::3])  # minimise negative sum → maximise sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bnds = [(0, 1), (0, 1), (0, 0.5)] * n  # bounds for each variable
    rs0 = 0.02
    xs = np.random.uniform(rs0, 1-rs0, n)
    ys = np.random.uniform(rs0, 1-rs0, n)
    v0 = np.concatenate([xs, ys, [rs0] * n])
    
    res = minimize(objective, v0,
                   method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'ineq', 'fun': constraints}],
                   options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
    
    if not res.success:
        raise Exception("Optimization failed")
    
    xs = res.x[0::3]
    ys = res.x[1::3]
    rs = res.x[2::3]
    centers = np.column_stack((xs, ys))
    radii = rs
    sum_r = rs.sum()
    
    return centers, radii, sum_r
