import numpy as np
from scipy.optimize import differential_evolution, linprog

def compute_sum_radii(positions):
    n = 26
    dmat = np.sqrt(np.sum((positions.reshape(n, 2, 2)[:, :, 0] - positions.reshape(n, 2, 2)[:, :, 0][:, np.newaxis]) ** 2, axis=2))
    dmat[dmat == 0] = np.inf
    dmat = dmat + dmat.T - np.eye(n)

    A_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append([0] * n + [1, 1, -1, -1])
    A_ub = np.array(A_ub)

    b_ub = np.ones(dmat.shape[0]) * (dmat - 1)

    A_eq = []
    for i in range(n):
        A_eq.append([0] * n + [1])
        A_eq.append([1] * n + [0])
        A_eq.append([0] * n + [0, 1])
        A_eq.append([0] * n + [0, 0, 1])

    A_eq = np.array(A_eq)
    b_eq = np.ones(4 * n) * 1

    c = -np.ones(n)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')

    if not res.success:
        return -np.inf

    return -res.fun

def evaluate_positions(positions):
    n = 26
    if np.any(np.isnan(positions)) or np.any(positions < 0) or np.any(positions > 1):
        return -np.inf
    return compute_sum_radii(positions)

def run_packing():
    x0 = np.zeros(2 * 26)
    bounds = [(0, 1)] * (2 * 26)
    result = differential_evolution(evaluate_positions, bounds, popsize=20, mutation=0.7, maxiter=100)
    positions = result.x
    sum_radii = -result.fun
    centers = positions[:26].reshape(26, 2)
    radii = np.ones(26) * (sum_radii / 26)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
