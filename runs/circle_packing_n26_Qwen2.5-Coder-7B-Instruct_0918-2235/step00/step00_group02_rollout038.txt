import numpy as np
from scipy.optimize import minimize, linprog

def compute_sum_radii(positions):
    n = 26
    positions = positions.reshape(26, 2)
    dmat = np.sqrt(np.sum((positions[:, np.newaxis, :] - positions[np.newaxis, :, :]) ** 2, axis=2))
    diag = np.diag(dmat)
    dmat = dmat - np.diag(diag)
    r_vars = np.arange(n, 2 * n)
    A_ub = np.zeros((2 * n * n, 2 * n))
    b_ub = np.zeros(2 * n * n)

    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * n + j, i] = 1
            A_ub[i * n + j, j] = 1
            A_ub[i * n + j, r_i] = -1
            A_ub[i * n + j, r_j] = -1
            b_ub[i * n + j] = dmat[i, j]

    for i in range(n):
        r_i = i + n
        A_ub[n * n + i, i] = 1
        A_ub[n * n + i, r_i] = -1
        b_ub[n * n + i] = positions[i, 0]

        A_ub[n * n + i + n, i] = 1
        A_ub[n * n + i + n, r_i] = -1
        b_ub[n * n + i + n] = 1 - positions[i, 0]

        A_ub[n * n + i + 2 * n, i] = 1
        A_ub[n * n + i + 2 * n, r_i] = -1
        b_ub[n * n + i + 2 * n] = positions[i, 1]

        A_ub[n * n + i + 3 * n, i] = 1
        A_ub[n * n + i + 3 * n, r_i] = -1
        b_ub[n * n + i + 3 * n] = 1 - positions[i, 1]

    c = -np.ones(2 * n)
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if result.success:
            r_values = result.x[n:]
            radii = np.maximum(r_values, 0)
            radii_sum = np.sum(radii)
            return -radii_sum
        else:
            return np.inf
    except Exception:
        return np.inf

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_positions = np.zeros((52,))
    result = minimize(compute_sum_radii, initial_positions, method='differential_evolution', bounds=[(0, 1)] * 52, popsize=20, recombination=0.7, maxfev=1000)
    positions = result.x.reshape(26, 2)
    radii = compute_radii_for_positions(compute_sum_radii, positions)
    return positions, radii, np.sum(radii)
