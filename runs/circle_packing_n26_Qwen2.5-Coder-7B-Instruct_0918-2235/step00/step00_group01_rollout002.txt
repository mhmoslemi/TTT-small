import numpy as np
from scipy.optimize import linprog, differential_evolution

def lp_for_circles(centers):
    n = len(centers)
    c = np.ones(n)
    A_ub = []
    b_ub = []
    r_bounds = [(0, None) for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2)
            A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [1] + [0] * (n - j - 1))
            b_ub.append(d)
        for axis, boundary in [(0, 1), (1, 1)]:
            A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [0] * (n - j))
            b_ub.append(boundary - centers[i][axis])
    return linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs').x

def run_nelder_mead(centers):
    n = len(centers)
    bounds = [(0, 1)] * 2 * n
    
    def objective(u):
        centers_new = u.reshape(n, 2)
        return -lp_for_circles(centers_new).sum()
    
    result = differential_evolution(objective, bounds=bounds, popsize=20, maxiter=200, tol=1e-12)
    return result.x.reshape(n, 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    u = np.random.rand(52)
    centers = run_nelder_mead(u)
    radii = lp_for_circles(centers)
    return centers, radii, radii.sum()
