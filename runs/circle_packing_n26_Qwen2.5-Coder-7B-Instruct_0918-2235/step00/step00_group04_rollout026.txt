import numpy as np
from scipy.optimize import linprog

GLOB_PARAMS = {
    'n_particles': 30,
    'max_iter': 3000,
    'inertia': 0.729,
    'cognitive': 1.49445,
    'social': 1.49445,
    'vel_clip': 0.1,
    'alpha': 0.999,
    'grid_size': 30,
}

def solve_radius_lp(centres):
    n_circles = centres.shape[0]
    A_ub = np.empty((n_circles, n_circles))
    for i in range(n_circles):
        A_ub[i, i] = 0
        for j in range(i + 1, n_circles):
            d_ij = np.linalg.norm(centres[i] - centres[j])
            A_ub[i, j] = d_ij - 1
            A_ub[j, i] = -d_ij
    
    c = -np.ones(n_circles)
    b_ub = np.empty(n_circles)
    b_ub[:n_circles] = centres[:, 0]
    b_ub[n_circles:] = 1 - centres[:, 0]
    b_lb = np.zeros(n_circles)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=None, b_eq=None, bounds=(b_lb, None))

    if not res.success:
        print('WARNING:', res.message)
    
    radii = -res.x
    sum_radii = np.sum(radii)
    return radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    centres = np.random.rand(GLOB_PARAMS['grid_size'], 2)
    centre_indices = np.arange(GLOB_PARAMS['grid_size'])
    np.random.shuffle(centre_indices)
    centres = centres[centre_indices, :]
    centres = centres[:GLOB_PARAMS['n_particles'], :]
    n_particles, _ = centres.shape
    velocities = np.zeros_like(centres)
    pbests = np.copy(centres)
    pbest_sums = np.zeros(n_particles)
    gbest = np.zeros((GLOB_PARAMS['n_particles'], 2))
    gbest_sum = -np.inf

    for iteration in range(GLOB_PARAMS['max_iter']):
        for p in range(n_particles):
            velocities[p] *= GLOB_PARAMS['alpha']
            r1, r2 = np.random.rand(2)
            for i in range(GLOB_PARAMS['n_particles']):
                if i == p:
                    continue
                velocities[p, i] += (GLOB_PARAMS['inertia'] * velocities[p, i] +
                                      GLOB_PARAMS['cognitive'] * r1 * (pbests[p, i] - centres[p, i]) +
                                      GLOB_PARAMS['social'] * r2 * (gbest[p, i] - centres[p, i]))
                velocities[p, i] = np.clip(velocities[p, i], -GLOB_PARAMS['vel_clip'], GLOB_PARAMS['vel_clip'])
                centres[p, i] += velocities[p, i]
                centres[p, i] = np.clip(centres[p, i], 0, 1)

            radii, sum_radii = solve_radius_lp(centres[p])
            if sum_radii > pbest_sums[p]:
                pbests[p] = np.copy(centres[p])
                pbest_sums[p] = sum_radii
                if sum_radii > gbest_sum:
                    gbest = np.copy(pbests[p])
                    gbest_sum = sum_radii

    return gbest, pbests[:, 0], gbest_sum
