import numpy as np
from scipy.optimize import linprog, differential_evolution

def hex_grid(radius, packing_density):
    """
    Generate a hexagonal grid with a given radius and packing density.
    """
    side_length = 2 * radius / np.sqrt(3)
    grid = []
    x = y = 0
    row_offset = 0.5 * side_length
    while y < 1:
        if x >= 1:
            x = 0
            y += side_length
        grid.append((x + row_offset, y))
        if (int((x + row_offset) / side_length) % 2) == 0:
            x += 0.75 * side_length
        else:
            x += 1.5 * side_length
    return np.array(grid)

def add_circles_around_quadrant(centers, radius, quadrant):
    """
    Add circles around a given quadrant (0,1,2,3) with a specified radius.
    """
    quarter_circle = hex_grid(radius, packing_density=0.9).reshape(-1, 2)
    quarter_circle[:, 1] -= radius * (1 + np.sqrt(3))
    quarter_circle[:, 0] -= radius * 1.5
    if quadrant == 0:
        quarter_circle[:, 0] += radius * 1.5
    elif quadrant == 1:
        quarter_circle[:, 0] += radius * 3
        quarter_circle[:, 1] += radius * (1 + np.sqrt(3))
    elif quadrant == 2:
        quarter_circle[:, 1] += radius * 2 * (1 + np.sqrt(3))
    elif quadrant == 3:
        quarter_circle[:, 1] += radius * 2 * (1 + np.sqrt(3))
        quarter_circle[:, 0] += radius * 1.5
    centers = np.append(centers, quarter_circle, axis=0)
    return centers

def solve_radii_fixed_centres(centres):
    """
    Solve for the radii fixed at given centres.
    """
    n = centres.shape[0]
    A = np.zeros((n, n))
    b = np.zeros(n)
    c = np.ones(n)
    
    for i in range(n):
        A[i, i] = 1
        b[i] = min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            A[i, j] = 1
            A[j, i] = 1
            b[i] = b[i] - d
            b[j] = b[j] - d
    
    result = linprog(c, A_ub=A, b_ub=b, method='highs')
    return result.x, result.fun

def penalty_centres_fixed_radii(centres, radii):
    """
    Penalty function for centres fixed at given radii.
    """
    n = centres.shape[0]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            penalty += max(0, radii[i] + radii[j] - d)
    for i in range(n):
        penalty += np.max(0, radii[i] - min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1]))
    return penalty

def run_packing():
    """
    Run the packaging algorithm.
    """
    best_centers = None
    best_radii = None
    best_sum = 0
    np.random.seed(0)  # Deterministic random seed for reproducibility
