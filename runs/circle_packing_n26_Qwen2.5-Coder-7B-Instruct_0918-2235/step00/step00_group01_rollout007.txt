```python
import numpy as np
from scipy.optimize import minimize

# Precompute the pairwise distance indices
n = 26
pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

# Constraint function
def ineqs(v):
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    n = len(rs)
    cons = []
    for i in range(n):
        cons.extend([xs[i] - rs[i], 1 - xs[i] - rs[i], ys[i] - rs[i], 1 - ys[i] - rs[i]])
    for i, j in pairs:
        cons.append((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2 - (rs[i] + rs[j])**2)
    return np.array(cons)

# Objective function to maximize sum of radii
def objective(v):
    return -np.sum(v[2::3])

# Generate initial guess
def generate_initial_guess(n, r0=0.02):
    xs, ys, rs = [], [], []
    while len(xs) < n:
        x, y = np.random.uniform(r0, 1 - r0, 2)
        r = r0
        is_valid = True
        for cx, cy, cr in zip(xs, ys, rs):
            if (x - cx)**2 + (y - cy)**2 < (r + cr)**2:
                is_valid = False
                break
        if is_valid:
            xs.append(x)
            ys.append(y)
            rs.append(r)
    return np.hstack((np.array(xs), np.array(ys), np.array(rs)))

# Bounds for variables
bounds = [(0, 1), (0, 1), (0, 0.5)] * n

# Initial guess
v0 = generate_initial_guess(n)

# Constraints specification
constraints = [{'type': 'ineq', 'fun': ineqs}]

# Optimization call
res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=constraints,
               options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})

# Extract solution
if not res.success:
    raise ValueError("Optimization failed")

xs = res.x[0::3]
ys = res.x[1::3]
rs = res.x[2::3]
centers = np.column_stack((xs, ys))
radii = rs
sum_r = rs.sum()

# Return the solution
return centers, radii, sum_r
```
