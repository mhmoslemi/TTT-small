import numpy as np
from scipy.optimize import minimize

def init_hex_grid():
    n = 26
    radius = 0.01
    centres = np.array([[0.5, 0.5], [0.3, 0.3], [0.7, 0.7], [0.1, 0.7], [0.9, 0.3]])
    np.random.seed(42)
    for _ in range(n-5):
        i = np.random.randint(5)
        centres = np.delete(centres, i, axis=0)
        for _ in range(5):
            x = 0.3 - 0.1 + np.random.rand() * 0.2
            y = 0.3 - 0.1 + np.random.rand() * 0.2
            centres = np.vstack((centres, [x, y]))
    centres = np.delete(centres, np.where(np.isinf(centres).sum(axis=1))[0], axis=0)
    centres = centres[:n, :]
    return centres, np.full(n, radius)

def build_variable_vector(centres, radii):
    return np.concatenate((centres.flatten(), radii))

def unpack(z):
    n = 26
    x = z[:n].reshape(n//2, 2)
    y = z[n:n*2].reshape(n//2, 2)
    r = z[n*2:]
    return x, y, r

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    penalty_B = np.sum(np.maximum(0, r - x)**2) +
                np.sum(np.maximum(0, r - (1 - x))**2) +
                np.sum(np.maximum(0, r - y)**2) +
                np.sum(np.maximum(0, r - (1 - y))**2)
    penalty_O = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt((x[i, 0] - x[j, 0])**2 + (y[i, 1] - y[j, 1])**2)
            penalty_O += np.maximum(0, r[i] + r[j] - d_ij)**2
    return -np.sum(r) + lambda_B * penalty_B + lambda_O * penalty_O

        grad[i*3 + 2] = -2 * lambda_B * np.maximum(0, r[i] - x[i]) * np.sign(r[i] - x[i]) / 2
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt((x[i, 0] - x[j, 0])**2 + (y[i, 1] - y[j, 1])**2)

            ds = max(0, r[i] + r[j] - d_ij)**2
            grad[3*i] += 4 * lambda_O * ds / d_ij * dx_ij * np.sign(r[i] + r[j] - d_ij) / 2
            grad[3*i + 1] += 4 * lambda_O * ds / d_ij * dy_ij * np.sign(r[i] + r[j] - d_ij) / 2
            grad[3*i + 2] += 4 * lambda_O * ds / d_ij * dx_ij * np.sign(r[i] + r[j] - d_ij) / 2

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)
    bounds = [(0, 1)] * 52 + [(0, None)] * 26
    lambda_B, lambda_O = 1e6, 1e6
    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, lambda_B, lambda_O)
    def jacobian(z):
        x, y, r = unpack(z)
        return compute_penalty_gradient(x, y, r, lambda_B, lambda_O)
    res = minimize(objective, z, jac=jacobian, method='trust-constr', bounds=bounds, options={'gtol': 1e-6, 'ftol': 1e-8, 'maxiter': 3000, 'verbose': 0})
    centres_opt, radii_opt = unpack(res.x)
    sum_r = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_r
