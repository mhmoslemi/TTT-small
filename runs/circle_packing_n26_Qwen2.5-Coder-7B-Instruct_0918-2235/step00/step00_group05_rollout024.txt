import numpy as np
from scipy.optimize import minimize, linprog

def objective(vars):
    n = 26
    x = vars[:2*n]
    r = vars[2*n:]
    x = x.reshape(n, 2)
    P_wall = np.sum(np.maximum(0, r - x[:,0]) ** 2) + \
            np.sum(np.maximum(0, r - (1 - x[:,0])) ** 2) + \
            np.sum(np.maximum(0, r - x[:,1]) ** 2) + \
            np.sum(np.maximum(0, r - (1 - x[:,1])) ** 2)
    P_ovl = 0
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((x[i] - x[j]) ** 2, axis=-1))
            P_ovl += np.maximum(0, (r[i] + r[j]) - dist) ** 2
    return np.sum(r) - 1e4 * P_wall - 1e4 * P_ovl

def jacobian(vars):
    n = 26
    x = vars[:2*n]
    r = vars[2*n:]
    x = x.reshape(n, 2)
    Px_wall = np.zeros(2*n + n)
    Px_ovl = np.zeros(2*n + n)
    for i in range(n):
        Px_wall[i*2:i*2+2] = 1e4 * np.array([-1, -1])
        Px_wall[2*n+i] = 1e4 * 2 * (np.maximum(0, r[i] - x[i,0]) - r[i])
        Px_wall[2*n+i] = 1e4 * 2 * (np.maximum(0, 1-x[i,0] - r[i]) - r[i])
        Px_wall[2*n+i] = 1e4 * 2 * (np.maximum(0, r[i] - x[i,1]) - r[i])
        Px_wall[2*n+i] = 1e4 * 2 * (np.maximum(0, 1-x[i,1] - r[i]) - r[i])

        Px_ovl[i*2] = np.zeros(2*n + n)
        Px_ovl[i*2 + 1] = np.zeros(2*n + n)
        
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((x[i] - x[j]) ** 2, axis=-1))
            overlap = (r[i] + r[j]) - dist
            if overlap > 0:
                Px_ovl[i*2] += 1e4 * (-x[i] + x[j]) / dist * overlap
                Px_ovl[i*2 + 1] += 1e4 * (-x[i] + x[j]) / dist * overlap
                Px_ovl[j*2] += 1e4 * (x[i] - x[j]) / dist * overlap
                Px_ovl[j*2 + 1] += 1e4 * (x[i] - x[j]) / dist * overlap

    return Px_wall + Px_ovl

def lp_radii_fixup(x):
    n = 26
    r = x[:n]
    c = x[n:]
    c = c.reshape(n, 2)
    cons = [{'type': 'ineq', 'fun': lambda r: 1 - 2*r - c[:,0]},
            {'type': 'ineq', 'fun': lambda r: 1 - 2*r - c[:,1]},
            {'type': 'ineq', 'fun': lambda r: c[:,0] - 2*r},
            {'type': 'ineq', 'fun': lambda r: c[:,1] - 2*r}]
    opt = linprog(
        c=np.ones(n), 
        A_ub=np.vstack([1 - 2*c[:,0], 1 - 2*c[:,1], c[:,0] - 2*c[:,0], c[:,1] - 2*c[:,1]]).T, 
        b_ub=r, 
        method='highs'
    )
    return opt.x

def shake_up(vars):
    n = 26
    x = vars[:2*n]
    r = vars[2*n:]
    x = x.reshape(n, 2)
    k = 3
    mask = np.random.choice(n, k, replace=False)
    delta = 0.01
    x[mask] += np.random.uniform(-delta, delta, (k, 2))
    return np.hstack((x.ravel(), r))

def run_packing():
    n = 26
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    bounds = bounds + [(1e-8, 1) for _ in range(n)]  # Minimal separation floor
    centers = np.random.rand(n, 2) * 0.5 + 0.25
    init_radii =(lp_radii_fixup(np.hstack([centers.flatten(), np.ones(n)*0.5]))
    x0 = np.hstack([centers.flatten(), init_radii])

    best_centers = None
    best_radii = None
    best_sum = 0.0

    for _ in range(8):  # Try multiple seed configurations
        result = minimize(objective, x0, method='L-BFGS-B', jac=jacobian, bounds=bounds)        
        if result.success:
            found_centers = result.x[:2*n].reshape(n, 2)
            found_radii = result.x[2*n:]
            found_sum = np.sum(found_radii)
            if found_sum > best_sum:
                best_centers = found_centers
                best_radii = found_radii
                best_sum = found_sum

        if found_sum > 2.636:
            return best_centers, best_radii, best_sum

        x0 = shake_up(result.x)

    return best_centers, best_radii, best_sum
