import numpy as np
import scipy.optimize

def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    d_ij_squared = np.sum((centres_i[:, np.newaxis, :] - centres_i[np.newaxis, :, :]) ** 2, axis=2)
    d_ij = np.sqrt(d_ij_squared)
    np.fill_diagonal(d_ij, np.inf)

    c = -np.ones(n_circles)
    A_ub = np.vstack([
        -centres_i[:, 0], centres_i[:, 0], -centres_i[:, 1], centres_i[:, 1],
        d_ij - np.triu(np.ones(d_ij.shape) * np.inf)
    ]).T
    b_ub = np.hstack([centres_i[:, 0], 1 - centres_i[:, 0], centres_i[:, 1], 1 - centres_i[:, 1]])

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(None, None), method='highs')
    radii = -res.fun * np.ones(n_circles)
    return radii, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_particles = 30
    n_circles = 26
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    max_iter = 3000
    vel_clip = 0.1
    alpha = 0.999

    np.random.seed(42)
    grid = np.mgrid[0:1:1/5., 0:1:1/5.].cells
    grid = grid.reshape(2, -1).T[np.random.permutation(grid.shape[0])]
    grid = grid[:26] + np.random.normal(0, 0.01, grid[:26].shape)
    grid = np.clip(grid, 0, 1)

    centres = np.repeat(grid[np.newaxis, :, :], n_particles, axis=0)
    velocities = np.zeros_like(centres)
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres[np.argmin(pbest_sums)]
    gbest_sum = np.min(pbest_sums)

    alpha = alpha ** (1 / max_iter)

    for _ in range(max_iter):
        for p in range(n_particles):
            velocities[p] *= inertia
            velocities[p] += cognitive * np.random.rand() * (pbests[p] - centres[p])
            velocities[p] += social * np.random.rand() * (gbest - centres[p])
            velocities[p] = np.clip(velocities[p], -vel_clip, vel_clip)
            centres[p] += velocities[p]
            centres[p] = np.clip(centres[p], 0, 1)

        radii, sums = zip(*[solve_radius_lp(c) for c in centres])
        if np.any(np.isnan(radii)) or np.any(np.isnan(sums)):
            raise ValueError("NaN values detected in solution")

        for p in range(n_particles):
            if sums[p] > pbest_sums[p]:
                pbests[p], pbest_sums[p] = centres[p], sums[p]

        if sums[np.argmin(sums)] > gbest_sum:
            gbest, gbest_sum = pbests[np.argmin(sums)], sums[np.argmin(sums)]

        if _ % 500 == 0 and pbest_sums[np.random.randint(n_particles)] < np.min(pbest_sums):
            perturbed_grid = grid + np.random.normal(0, 0.01, grid.shape)
            perturbed_grid = np.clip(perturbed_grid, 0, 1)
            centres[np.random.randint(n_particles, size=int(n_particles * 0.1)), :] = perturbed_grid

    radii, _ = solve_radius_lp(gbest)
    return gbest, radii, np.sum(radii)
