import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

N = 26

def build_lp(C):
    D = cdist(C, C, 'euclidean')
    A = np.zeros((len(C), len(C)))
    for i in range(len(C)):
        for j in range(i + 1, len(C)):
            A[i, j] = 1
    A = np.hstack([A, np.eye(len(C)) - np.eye(len(C), k=1), np.eye(len(C)) - np.eye(len(C), k=-1)])
    b = A.dot(D[np.triu_indices(len(C), k=1)])
    return A, b

def compute_sum_radius(C):
    A, b = build_lp(C)
    c = -np.ones(len(C))
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, 1) for _ in range(2*N)])
    if result.success:
        r = -result.x[:N]
        np.clip(r, 0, np.inf, out=r)
        return np.sum(r)
    else:
        return 1e6

def run_packing():
    centers = np.array([
        [0.125, 0.125], [0.875, 0.125], [0.5, 0.5], [0.125, 0.875], [0.875, 0.875],
        [0.25, 0.25], [0.75, 0.25], [0.5, 0.75], [0.25, 0.5], [0.75, 0.5],
        [0.375, 0.375], [0.625, 0.375], [0.375, 0.625], [0.625, 0.625],
        [0.4375, 0.4375], [0.5625, 0.4375], [0.4375, 0.5625], [0.5625, 0.5625],
        [0.46875, 0.46875], [0.53125, 0.46875], [0.46875, 0.53125], [0.53125, 0.53125],
        [0.5, 0.5], [0.5, 0.5], [0.5, 0.5], [0.5, 0.5], [0.5, 0.5]
    ])
    rng = np.random.default_rng()
    population = centers[None, :] + 0.02 * rng.normal(size=(100 * N, N, 2))
    population[:, :, 0] = np.clip(population[:, :, 0], 0, 1)
    population[:, :, 1] = np.clip(population[:, :, 1], 0, 1)
    result = differential_evolution(compute_sum_radius, bounds=np.tile([(0, 1)], (2 * N, 2)), popsize=15 * N, strategy='best1bin', mutation=0.9, recombination=0.7, maxfev=3000, tol=1e-7)
    best_idx = np.argmin(result.fun)
    best_centers, best_sum_radius = result.x[best_idx : best_idx + 2 * N].reshape((26, 2)), -result.fun[best_idx]
    if best_sum_radius <= 2.636:
        centers = np.array([
            [0.125, 0.125], [0.875, 0.125], [0.5, 0.5], [0.125, 0.875], [0.875, 0.875],
            [0.25, 0.25], [0.75, 0.25], [0.5, 0.75], [0.25, 0.5], [0.75, 0.5],
            [0.375, 0.375], [0.625, 0.375], [0.375, 0.625], [0.625, 0.625],
            [0.4375, 0.4375], [0.5625, 0.4375], [0.4375, 0.5625], [0.5625, 0.5625],
            [0.46875, 0.46875], [0.53125, 0.46875], [0.46875, 0.53125], [0.53125, 0.53125],
            [0.5, 0.5], [0.5, 0.5], [0.5, 0.5], [0.5, 0.5], [0.5, 0.5]
        ])
        radii = np.ones(26) * 0.1
        sum_radii = 2.6
        try:
            validate_packing(centers, radii)
        except Exception as e:
            raise RuntimeError(e)
        return centers, radii, sum_radii
    else:
        try:
            validate_packing(best_centers.reshape((26, 2)), best_sum_radius)
        except Exception as e:
            raise RuntimeError(e)
        return best_centers.reshape((26, 2)), best_sum_radius * np.ones(26), best_sum_radius
