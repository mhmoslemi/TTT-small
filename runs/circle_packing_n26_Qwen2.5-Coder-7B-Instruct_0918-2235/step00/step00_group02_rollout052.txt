import numpy as np
from scipy.optimize import linprog, differential_evolution

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_sum_radii(positions):
    n = 26
    dmat = np.sqrt(np.sum((positions.reshape(n, 2)[:, np.newaxis, :] - positions.reshape(n, 2)[np.newaxis, :, :]) ** 2, axis=2))
    A_ub = np.zeros((n * (n - 1) + 4 * n, 4 * n))
    b_ub = np.zeros(n * (n - 1) + 4 * n)
    
    for i in range(n):
        A_ub[i * (n - 1): (i + 1) * (n - 1), i * 4 + 0] = 1.
        A_ub[i * (n - 1): (i + 1) * (n - 1), i * 4 + 1] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), i * 4 + 2] = 1.
        A_ub[i * (n - 1): (i + 1) * (n - 1), i * 4 + 3] = 0.
        b_ub[i * (n - 1): (i + 1) * (n - 1)] = positions[i, 0]
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 1) * 4 + 0] = -1.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 1) * 4 + 1] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 1) * 4 + 2] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 1) * 4 + 3] = 1.
        b_ub[i * (n - 1): (i + 1) * (n - 1)] = positions[i, 0]
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 2) * 4 + 0] = 1.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 2) * 4 + 1] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 2) * 4 + 2] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 2) * 4 + 3] = 1.
        b_ub[i * (n - 1): (i + 1) * (n - 1)] = positions[i, 1]
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 3) * 4 + 0] = -1.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 3) * 4 + 1] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 3) * 4 + 2] = 0.
        A_ub[i * (n - 1): (i + 1) * (n - 1), (i + 3) * 4 + 3] = 1.
        b_ub[i * (n - 1): (i + 1) * (n - 1)] = positions[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            idx1 = i * 4 + 0
            idx2 = i * 4 + 2
            idx3 = j * 4 + 0
            idx4 = j * 4 + 2
            A_ub[n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j): n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j) + 2, idx1] = 1.
            A_ub[n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j): n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j) +
                  2, idx2] = 1.
            A_ub[n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j): n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j) + 2, idx3] = -1.
            A_ub[n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j): n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j) + 2, idx4] = -1.
            b_ub[n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j): n * (n - 1) + 4 * n - 2 * (i * (n - 1) + j) + 2] = dmat[i, j] - 1e-12
    
    try:
        result = linprog(-np.ones(n * 4), A_ub=A_ub, b_ub=b_ub, method='highs', bounds=[(-1 + xs, 1 - xs) for x, xs in zip(positions.reshape(n, 2), dmat.min(axis=0).reshape(n, 1))])
        if not result.success:
            return -np.inf
        radii = np.array(result.x[:n * 4].reshape(n, 2)[:, 0] + result.x[:n * 4].reshape(n, 2)[:, 1])
        sum_radii = np.sum(radii)
        return sum_radii
    except Exception as e:
        return -np.inf

def run_packing():
    centers = np.array([[0.595, 0.35], [0.61, 0.35], [0.595, 0.3], [0.61, 0.3], [0.595, 0.25], [0.61, 0.25], [0.595, 0.2], [0.61, 0.2], [0.595, 0.15], [0.61, 0.15], [0.595, 0.1], [0.61, 0.1], [0.595, 0.05], [0.61, 0.05], [0.595, 0.0], [0.61, 0.0], [0.595, 0.55], [0.61, 0.55], [0.595, 0.6], [0.61, 0.6], [0.595, 0.65], [0.61, 0.65], [0.595, 0.7], [0.61, 0.7], [0.595, 0.75], [0.61, 0.75], [0.595, 0.8], [0.61, 0.8]])
    radii = np.full(26, 0.1)
    sum_radii = compute_sum_radii(np.array(centers))
    return centers, radii, sum_radii

def objective(x):
    centers = x[:52].reshape(26, 2)
    sum_radii = compute_sum_radii(centers)
    return -sum_radii

centers = np.array([[0.595, 0.35], [0.61, 0.35], [0.595, 0.3], [0.61, 0.3], [0.595, 0.25], [0.61, 0.25], [0.595, 0.2], [0.61, 0.2], [0.595, 0.15], [0.61, 0.15], [0.595, 0.1], [0.61, 0.1], [0.595, 0.05], [0.61, 0.05], [0.595, 0.0], [0.61, 0.0], [0.595, 0.55], [0.61, 0.55], [0.595, 0.6], [0.61, 0.6], [0.595, 0.65], [0.61, 0.65], [0.595, 0.7], [0.61, 0.7], [0.595, 0.75], [0.61, 0.75], [0.595, 0.8], [0.61, 0.8]])
radii = np.full(26, 0.1)
sum_radii = compute_sum_radii(centers)

result = differential_evolution(objective, bounds=[(0, 1)] * 52, popsize=20, maxiter=100, tol=1e-3)
if result.success:
    centers = result.x[:52].reshape(26, 2)
    sum_radii = compute_sum_radii(centers)
else:
    centers, sum_radii = run_packing()

valid, message = validate_packing(centers, sum_radii)
print(valid, message)

return centers, radii, sum_radii
