import numpy as np
from scipy.optimize import differential_evolution, linprog

def objective(v, λ):
    n = v.shape[0] // 3
    x = v[:n].reshape(-1, 2)
    r = v[n:(2 * n)].reshape(-1)
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dij = np.sqrt(np.sum((x[i] - x[j]) ** 2))
            penalty += max(0, r[i] + r[j] - dij) ** 2
    penalty += np.sum(np.maximum(0, r - x[:, 0]) ** 2 + np.maximum(0, r - (1 - x[:, 0]) ** 2) ** 2 +
                      np.maximum(0, r - x[:, 1]) ** 2 + np.maximum(0, r - (1 - x[:, 1]) ** 2) ** 2)
    return -np.sum(r) + λ * penalty

def run_packing():

    bounds = [(0, 1), (0, 1)] * 26 + [(0, 0.5)] * 26
