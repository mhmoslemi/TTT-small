import numpy as np

def refine_radii(centres, radii, eps=1e-12, max_iter=200):
    for _ in range(max_iter):
        old_radii = np.copy(radii)
        for i in range(26):
            r_lim = min(centres[i, 0], 1 - centres[i, 0], centres[i, 1], 1 - centres[i, 1])
            dists = np.sqrt(np.sum((centres - centres[i, :])**2, axis=1)) - radii
            valid_dists = dists[dists > eps]
            min_dist = np.inf if len(valid_dists) == 0 else np.min(valid_dists)
            new_r_i = min(r_lim, min_dist - eps)
            if new_r_i < 0:
                new_r_i = 0
            radii[i] = new_r_i
        if np.max(np.abs(old_radii - radii)) < eps:
            break
    return radii

def anneal_one_start(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):
    centres = init_centres
    radii = np.full(26, 1e-3)
    refine_radii(centres, radii)
    base_sum = np.sum(radii)
    best_sum = base_sum
    best_centres = centres
    best_radii = radii
    
    for step in range(steps):
        T = 0.25 * (1 - step / steps)
        s = init_step * (1 - step / steps)
        
        i = rng.integers(26)
        old_pos = centres[i, :]
        new_pos = old_pos + rng.uniform(-s, s, 2)
        new_pos = np.clip(new_pos, 0, 1)
        centres[i, :] = new_pos
        
        new_radii = refine_radii(centres, radii)
        new_sum = np.sum(new_radii)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_centres = centres
            best_radii = new_radii
        elif new_sum > base_sum:
            base_sum = new_sum
            base_centres = centres
            base_radii = new_radii
    
    return best_centres, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    RESTARTS = 30
    rng = np.random.default_rng()
    best_sum = 0
    best_centers = None
    best_radii = None
    
    for _ in range(RESTARTS):
        init_centres = rng.random((26, 2))
        result = anneal_one_start(rng, init_centres)
        centres, radii, sum = result
        if sum > best_sum:
            best_sum = sum
            best_centers = centres
            best_radii = radii
    
    return best_centers.reshape(26, 2), best_radii, best_sum
