import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    pts = np.zeros((n, 2))
    rad = np.zeros(n)
    MIN_R = 1e-8
    MAX_ITER = 200000
    T_init = 1.0
    T_final = 1e-4
    cooling_rate = 0.999
    epsilon_w = 1.0
    epsilon_p = 1.0
    sigma = 0.01
    accept_advance = True

    def dist2(x, y, u, v):
        return (x - u) ** 2 + (y - v) ** 2

    def update_radius(i):
        clearance = min(
            np.min(np.sqrt(dist2(pts[i, 0], pts[i, 1], pts[:, 0], pts[:, 1])) - rad) - rad[i],
            np.min(np.array([pts[i, 0], 1 - pts[i, 0], pts[i, 1], 1 - pts[i, 1]]))
        )
        rad[i] = max(MIN_R, clearance)
        return rad[i]

    for order in range(10):
        for k in range(n):
            S = 5000
            for _ in range(S):
                x, y = np.random.rand(2)
                valid = True
                for j in range(k):
                    if dist2(x, y, pts[j, 0], pts[j, 1]) < (rad[j] + MIN_R) ** 2:
                        valid = False
                        break
                    if x < MIN_R or x > 1 - MIN_R or y < MIN_R or y > 1 - MIN_R:
                        valid = False
                        break
                if valid:
                    rad[k] = update_radius(k)
                    pts[k, 0] = x
                    pts[k, 1] = y
                    break
            if rad[k] == 0:
                rad[k] = MIN_R
            else:
                break

    for T in np.linspace(T_init, T_final, MAX_ITER):
        for k in range(n):
            xi, yi = pts[k]
            delta = np.random.randn(2) * sigma
            x_new, y_new = xi + delta[0], yi + delta[1]
            x_new = min(max(x_new, rad[k]), 1 - rad[k])
            y_new = min(max(y_new, rad[k]), 1 - rad[k])
            r_new = update_radius(k)
            dU = epsilon_w * ((r_new / (r_new + xi + yi)) / (r_new + xi + yi) - (rad[k] / (rad[k] + xi + yi)) / (rad[k] + xi + yi)) ** 2
            if dU >= 0 and np.random.rand() < np.exp(-dU / T):
                r, xi, yi = r_new, x_new, y_new
        accept_advance = False

    for _ in range(10):
        for k in range(n):
            xi, yi = pts[k]
            delta = np.random.randn(2) * sigma
            x_new, y_new = xi + delta[0], yi + delta[1]
            x_new = min(max(x_new, rad[k]), 1 - rad[k])
            y_new = min(max(y_new, rad[k]), 1 - rad[k])
            r_new = update_radius(k)
            if r_new > rad[k] or np.random.rand() < 1.5 * np.exp(-(r_new - rad[k]) / sigma):
               (rad[k], xi, yi) = (r_new, x_new, y_new)

    for k in range(n):
        r = update_radius(k)
        rad[k] = r

    for k in range(n):
        xi, yi = pts[k]
        delta = np.random.randn(2) * sigma
        x_new, y_new = xi + delta[0], yi + delta[1]
        x_new = min(max(x_new, rad[k]), 1 - rad[k])
        y_new = min(max(y_new, rad[k]), 1 - rad[k])
        r_new = update_radius(k)
        if r_new > rad[k] or np.random.rand() < 2 * np.exp(-(r_new - rad[k]) / sigma):
            (rad[k], xi, yi) = (r_new, x_new, y_new)

    sum_radii = np.sum(rad)
    return pts, rad, sum_radii
