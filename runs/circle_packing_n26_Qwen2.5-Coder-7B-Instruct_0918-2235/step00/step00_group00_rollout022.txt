import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    dists = np.maximum(candidates[:, np.newaxis, :] - placed_centres[np.newaxis, :, :], 0.0)
    dists = np.minimum(dists, 1.0 - dists)
    distances = np.sum(dists ** 2, axis=2) ** 0.5
    r_max = np.minimum(distances - placed_radii, candidates[:, 0]) - 0.001
    r_max = np.minimum(r_max, 1.0 - candidates[:, 0]) - 0.001
    r_max = np.minimum(r_max, candidates[:, 1]) - 0.001
    r_max = np.minimum(r_max, 1.0 - candidates[:, 1]) - 0.001
    return np.maximum(r_max, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    h = 0.05
    epsilon = 1e-6
    n = 21
    x = np.arange(epsilon, 1, h)
    y = np.arange(epsilon, 1, h)
    X, Y = np.meshgrid(x, y)
    candidates = np.stack((X.flatten(), Y.flatten()), axis=1)

    placed_centres = []
    placed_radii = []

    while len(placed_centres) < 26:
        r_candidates = compute_r_max(candidates, np.array(placed_centres), np.array(placed_radii))
        feasible_idx = np.where(r_candidates > 0)[0]
        if feasible_idx.size == 0:
            break
        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    for _ in range(5000):
        idx = np.random.randint(0, len(placed_centres))
        delta = (np.random.rand(2) * 0.04 - 0.02)
        new_centre = placed_centres[idx] + delta
        new_centre = np.clip(new_centre, 0, 1)
        r_candidates = compute_r_max(np.array([new_centre]), np.array(placed_centres), np.array(placed_radii))
        if r_candidates > placed_radii[idx]:
            placed_radii[idx] = r_candidates
            r_candidates = compute_r_max(np.array(placed_centres), np.array(placed_centres), np.array(placed_radii))
            if any(r < 0 for r in r_candidates) or any(r > 1 for r in r_candidates):
                placed_radii[idx] -= 0.001

    radii = np.array([np.maximum(r, 0.0) for r in placed_radii])
    return np.array(placed_centres), radii, np.sum(radii)
