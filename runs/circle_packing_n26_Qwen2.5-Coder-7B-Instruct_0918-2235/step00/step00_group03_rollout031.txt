import numpy as np
from scipy.optimize import minimize

def optimizer_gradient(vars, lambda_):
    n = len(vars) // 3
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, y, r, i, lambda_: 1 - x[i] - r[i] - lambda_})
        constraints.append({'type': 'ineq', 'fun': lambda x, y, r, i, lambda_: x[i] - r[i] - lambda_})
        constraints.append({'type': 'ineq', 'fun': lambda x, y, r, i, lambda_: 1 - y[i] - r[i] - lambda_})
        constraints.append({'type': 'ineq', 'fun': lambda x, y, r, i, lambda_: y[i] - r[i] - lambda_})
        for j in range(i+1, n):
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            constraints.append({'type': 'ineq', 'fun': lambda x, y, r, i, j, lambda_: dist - r[i] - r[j] - lambda_})
    
    def penalty(vars, lambda_):
        penalty = 0
        for i in range(n):
            penalty += max(0, 1 - x[i] - r[i] - lambda_)**2 + max(0, x[i] - r[i] - lambda_)**2 + max(0, 1 - y[i] - r[i] - lambda_)**2 + max(0, y[i] - r[i] - lambda_)**2 + sum(max(0, np.hypot(x[i] - x[j], y[i] - y[j]) - r[i] - r[j] - lambda_)**2 for j in range(i+1, n))
        return penalty
    
    result = minimize(lambda vars: -sum(r) + lambda_ * penalty(vars, lambda_), vars, method='L-BFGS-B', jac='2-point', bounds=[(0, 1), (0, 1)] * n + [(0, 0.5)] * n, constraints=constraints)
    return result.x

def run_packing():
    n = 26
    lambda_ = 1.0
    rho = 1.5
    K = 100
    R = 50
    tol = 1e-8
    max_iter = 5
    best_sum_radii = -1
    best_vars = None
    
    for _ in range(R):
        vars = np.random.rand(n * 3)
        for _ in range(K):
            vars = optimizer_gradient(vars, lambda_)
            lambda_ *= rho
            for _ in range(max_iter):
                if lambda_ > 1e5:
                    break
                vars = optimizer_gradient(vars, lambda_)
                lambda_ *= rho
        
            current_sum_radii = -np.sum(vars[2*n:])
            if current_sum_radii > best_sum_radii - tol:
                best_sum_radii = current_sum_radii
                best_vars = vars
    
    centers = best_vars[:n]
    radii = best_vars[2*n:]
    return centers, radii, best_sum_radii

centers, radii, sum_radii = run_packing()
