import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append([0, 0, -1])  # r_i <= x_i
        b_ub.append(centers[i, 0])
        A_ub.append([0, 0, 1])   # r_i <= 1 - x_i
        b_ub.append(1 - centers[i, 0])
        A_ub.append([0, -1, 0])  # r_i <= y_i
        b_ub.append(centers[i, 1])
        A_ub.append([0, 1, 0])   # r_i <= 1 - y_i
        b_ub.append(1 - centers[i, 1])
    
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            A_ub.append([-1, 0, 0, 0, 0, -1])  # r_i + r_j <= dist_ij
            b_ub.append(dist_ij)
            A_ub.append([0, -1, 0, 0, 0, 1])
            b_ub.append(dist_ij)
            A_ub.append([0, 0, -1, 0, 0, 0])
            b_ub.append(dist_ij)
            A_ub.append([0, 0, 0, -1, 0, 0])
            b_ub.append(dist_ij)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x[:n], -res.fun
    else:
        print("Unable to solve LP.")
        return np.zeros(n), 0.0

def run_packing():
    np.random.seed(42)
    grid = np.array([[0.05 + 0.16 * i, 0.05 + 0.16 * j] for i in range(7) for j in range(7)])
    np.random.shuffle(grid)
    centers = grid[:26]
    
    current_centers = centers
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
    
    step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    ε = 1e-12
    
    for iter in range(MAX_ITERS):
        i = np.random.randint(26)
        new_x = np.clip(centers[i, 0] + np.random.uniform(-step, step), 0, 1)
        new_y = np.clip(centers[i, 1] + np.random.uniform(-step, step), 0, 1)
        new_centers = centers
        new_centers[i] = [new_x, new_y]
        
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum + ε:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = current_centers, current_radii, current_sum
        
        if iter % N_REDUCE == 0 and step > 1e-4:
            step *= 0.9
        
        if iter % K_JUMP == 0:
            new_centers = grid[:26] + np.random.uniform(0, 0.02, size=(26, 2))
            new_centers = np.clip(new_centers, 0, 1)
            new_radii, new_sum = compute_lp_solution(new_centers)
            if new_sum > best_sum:
                best_centers, best_radii, best_sum = new_centers, new_radii, new_sum
    
    return best_centers, best_radii, best_sum
