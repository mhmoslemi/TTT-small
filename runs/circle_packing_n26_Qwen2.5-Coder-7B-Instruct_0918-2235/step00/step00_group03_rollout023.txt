import numpy as np

def refine_radii(centres, radii, eps=1e-12, max_iter=200):
    n = centres.shape[0]
    dist_matrix = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) ** 2, axis=2))
    for i in range(n):
        dij = dist_matrix[i, np.arange(n) != i] - radii[numpy.arange(n) != i]
        max_dij = np.where(dij > 0, dij, 0).min()
        beta = np.minimum(centres[i] - radii[i], 1 - centres[i] - radii[i])
        radii[i] = max(0, beta, max_dij - eps)
    return radii

def anneal_one_start(rng, init_centres, steps=500, init_step=0.07, min_step=0.01):
    n = len(init_centres)
    best_sum = -np.inf
    temperature = 0.25
    for _ in range(steps):
        step_size = init_step * (1 - _ / steps)
        acceptance_ratio = 0
        for _ in range(n):
            i = rng.choice(n)
            dx = (rng.random(2) * 2 - 1) * step_size
            dx = np.clip(dx, -1, 1)
            new_centres = init_centres.copy()
            new_centres[i] += dx
            new_radii = refine_radii(new_centres, np.full_like(range(n), 1e-3))
            new_sum = np.sum(new_radii)
            if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / temperature):
                init_centres[i] = new_centres[i]
                best_sum = new_sum
        temperature *= 0.99
    return init_centres, refine_radii(init_centres, np.full_like(range(n), 1e-3)), best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    restarts = 50
    best_sum = -np.inf
    best_centers = best_radii = None
    rng = np.random.default_rng()
    for _ in range(restarts):
        centres = rng.random((n, 2))
        centres = anneal_one_start(rng, centres)[0]
        radii = refine_radii(centres, np.full(n, 1e-3))
        sum_radii = np.sum(radii)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centres
            best_radii = radii
    return best_centers.reshape(26, 2), best_radii, best_sum
