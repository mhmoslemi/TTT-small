import numpy as np
from scipy.optimize import linprog
from numpy.random import default_rng

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def compute_distances(centers):
        n = centers.shape[0]
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                distances[i, j] = np.linalg.norm(centers[i] - centers[j])
        return distances

    def compute_bounds(centers):
        n = centers.shape[0]
        return np.minimum(centers[:, 0], 1 - centers[:, 0]), \
               np.minimum(centers[:, 1], 1 - centers[:, 1])

    def solve_inner_LP(centers):
        n = centers.shape[0]
        distances = compute_distances(centers)
        bounds = compute_bounds(centers)[0, :] + compute_bounds(centers)[1, :]
        A_ub = np.zeros((n * (n - 1) // 2, 2 * n))
        b_ub = distances[np.triu_indices(n, k=1)][:]
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=np.zeros((2 * n, 2)), method='highs')
        if not res.success:
            raise Exception(res.message)
        return res.x, -res.fun

    def coordinate_descent(centers, max_cycles=50, step=0.1, alpha=0.8, eps_stop=1e-4):
        step_size = step
        max_clicks = max_cycles
        n = centers.shape[0]
        rng = default_rng()
        for cycle in range(max_clicks):
            clicks = 0
            for i in rng.choice(range(n), size=n, replace=False):
                old_pos = centers[i]
                delta = rng.uniform(-step_size, step_size, size=2)
                centers[i] = np.clip(old_pos + delta, 0, 1)
                r, sum_radius = solve_inner_LP(centers)
                if sum_radius >= r[-1] + 1e-9:
                    clicks += 1
                else:
                    centers[i] = old_pos
            step_size *= alpha
            if step_size < eps_stop:
                break
            if clicks == 0:
                break

        r, sum_radius = solve_inner_LP(centers)
        return centers, r, sum_radius

    def multi_start(num_trials=40):
        best_sum = -np.inf
        best_centers = None
        best_r = None
        for _ in range(num_trials):
            centers = np.random.rand(num_trials, 2, 2).squeeze()
            centers_r, sum_radius = coordinate_descent(centers)
            if sum_radius > best_sum:
                best_sum = sum_radius
                best_centers = centers_r
                best_r = r
        return best_centers, best_r, best_sum

    return multi_start()
