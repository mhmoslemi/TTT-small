import numpy as np
from scipy.optimize import linprog, differential_evolution

def hex_grid(radius, packing_density):
    # Calculate the number of circles and unit cell size
    n_circles = int(packing_density * (np.pi / (3 * np.sqrt(3))))
    s = 2 * radius / np.sqrt(3)
    n_cells_x = int(np.ceil(1 / s))
    n_cells_y = int(np.ceil(np.sqrt(3) / s * n_cells_x))
    grid = np.zeros((n_cells_x * n_cells_y, 2))
    
    # Populate the grid
    idx = 0
    for y in range(n_cells_y):
        if y % 2 == 1:
            x_start = s / 2
        else:
            x_start = 0
        for x in range(n_cells_x):
            grid[idx, 0] = x_start + x * s
            grid[idx, 1] = y * s * np.sqrt(3) / 2
            idx += 1
    
    # Trim to unit square
    mask = ((grid[:, 0] >= 0) & (grid[:, 0] <= 1) & (grid[:, 1] >= 0) & (grid[:, 1] <= 1))
    return grid[mask]

def add_circles_around_quadrant(x, y, radius, packing_density):
    quarter_grid = hex_grid(radius, packing_density)
    quarter_grid[:, 0] += 0.25
    quarter_grid[:, 1] += 0.25
    quarter_grid[:, 0] %= 1
    quarter_grid[:, 1] %= 1
    quarter_grid = quarter_grid[np.random.choice(len(quarter_grid), size=0, replace=False)]
    return ((x, y) + quarter_grid).reshape(-1, 2)

def solve_radii_fixed_centres(centres):
    n = len(centres)
    radii = np.where(
        np.min(centres, axis=1, keepdims=True) - np.max(centres, axis=1, keepdims=True) > 0,
        (np.min(centres, axis=1) - np.max(centres, axis=1)) / 2,
        0
    )
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            norm = np.linalg.norm(centres[i] - centres[j])
            constraints.append({'type': 'ineq', 'fun': lambda t, j=j: t + t[j] - norm})
    result = linprog(np.ones(n), A_ub=[*constraints, -radii], b_ub=[*np.inf] * n, method='highs')
    return result.x, result.fun

def penalty_centres_fixed_radii(centres, radii):
    n = len(centres)
    distances = np.ndarray((n, n))
    np.fill_diagonal(distances, np.inf)
    for i in range(n):
        for j in range(i + 1, n):
            distances[i, j] = distances[j, i] = np.linalg.norm(centres[i] - centres[j])
    
    objective = 0
    for i in range(n):
        for j in range(i + 1, n):
            objective += np.maximum(0, radii[i] + radii[j] - distances[i, j])
        objective += np.maximum(0, radii[i] - min(centres[i]) - min(1 - centres[i]))
    
    return objective

def coordinated_ascent(init_centres, max_iter=15):
    n = len(init_centres)
    radii, _ = solve_radii_fixed_centres(init_centres)
    for _ in range(max_iter):
        penalty_result = differential_evolution(penalty_centres_fixed_radii, bounds=[(0, 1)] * (2 * n), args=(radii,), tol=1e-12)
        centres = penalty_result.x.reshape(n, 2)
        radii, _ = solve_radii_fixed_centres(centres)
    return centres, radii

def local_audit_and_refine(centres, radii):
    n = len(centres)
    for _ in range(10):
        i = np.random.randint(n)
        delta = 0.01
        day = centres[i] - radii[i]
        dex = centres[i] + radii[i]
        centre_solutions = []
        for dx in np.linspace(day[0], dex[0], 10):
            for dy in np.linspace(day[1], dex[1], 10):
                centre_solutions.append((dx, dy))
        radii_solutions = []
        for centre_solution in centre_solutions:
            radii_list = [radii[j] for j in range(n) if j != i]
            centre_solution_list = list(centre_solutions)
            centre_solution_list.remove(centre_solution)
            candid_centres = np.array(centre_solution_list)
            candid_centres = np.vstack((candid_centres, centre_solution))
            radii_solution, _ = solve_radii_fixed_centres(candid_centres)
            radii_list[i] = radii_solution[i]
            radii_solutions.append(radii_list)
        best_solution = max(radii_solutions, key=sum)
        radii[i] = best_solution[i]
    return centres, radii

def run_packing():
    n = 26
    best_centres = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0
    for seed in range(10):
        np.random.seed(seed)
        init_centres = hex_grid(0.03, packing_density=0.85)
        if init_centres.shape[0] > n:
            init_centres = init_centres[np.random.choice(init_centres.shape[0], size=n, replace=False)]
        init_centres = init_centres[:, :2]
        init_centres = init_centres + np.random.normal(0, 0.01, size=init_centres.shape)
        centres, radii = coordinated_ascent(init_centres)
        sum_radii = sum(radii)
        if sum_radii > best_sum_radii:
            best_centres, best_radii, best_sum_radii = centres, radii, sum_radii
    return best_centres, best_radii, best_sum_radii
