import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_trials = 40
    step_size = 0.1
    step_decay = 0.8
    min_delta = 1e-9
    min_step_size = 1e-4
    
    def rand_circle():
        return np.random.rand(2)

    def compute_distances(centers):
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        return dists

    def compute_max_radius(center, dists, radii):
        return np.minimum(dists, center[0] + center[1], 1 - center[0] - center[1], 1 - center[1]).reshape(-1) - radii
    
    def solve_lp(centers, radii, dists):
        A_ub = np.zeros((dists.size, n))
        b_ub = dists.flatten()
        c = -np.ones(n)
        bounds = np.array([[0, np.inf] for _ in range(n)])
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            raise ValueError("LP failed")
        return -res.fun, res.x

    def stepify(centers, radii):
        i = np.random.randint(n)
        delta = np.clip(np.random.randn(2), -step_size, step_size)
        new_center = centers[i] + delta
        new_center = np.clip(new_center, 0, 1)
        new_dists = compute_distances(np.concatenate((centers[:i], new_center, centers[i+1:])))
        new_max_radius = compute_max_radius(new_center, new_dists, radii[:i] + 1e-9 + radii[i+1:])
        new_radii = np.minimum(new_max_radius, radii)
        new_radii[:i] += new_radii[:i] >= radii[:i]
        new_radii[i+1:] += new_radii[i+1:] >= radii[i+1:]
        new_sum = np.sum(new_radii)
        return new_centers, new_radii, new_sum
    
    for _ in range(max_trials):
        centers = np.array([rand_circle() for _ in range(n)])
        dists = compute_distances(centers)
        radii = np.zeros(n)
        step_size = step_size
        while step_size >= min_step_size:
            new_centers, new_radii, new_sum = stepify(centers, radii)
            if new_sum > np.sum(radii) + min_delta:
                centers = new_centers
                radii = new_radii
            else:
                step_size /= 2
        sum_radii, optimal_radii = solve_lp(centers, radii, dists)
        if sum_radii > np.sum(radii):
            if sum_radii > np.sum(optimal_radii):
                return centers, optimal_radii, sum_radii
            return centers, radii, sum_radii
        step_size /= 2
    
    return centers, radii, np.sum(radii)

if __name__ == "__main__":
    np.random.seed(42)
    centers, radii, sum_radii = run_packing()
    print("Centers:", centers)
    print("Radii:", radii)
    print("Sum of radii:", sum_radii)
    assert validate_packing(centers, radii)[0], "Packing is invalid"
