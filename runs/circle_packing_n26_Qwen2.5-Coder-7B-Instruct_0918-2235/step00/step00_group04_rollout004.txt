import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Hexagonal lattice initialisation
    hex_lattice = np.zeros((n, 2))
    radius = 1 / (2 * np.sqrt(3))
    row_offset = np.sin(np.pi / 3)
    col_offset = 1 / 2
    for i in range(6):
        for j in range(5):
            if j % 2 == 0:
                hex_lattice[i * n // 6 + j // 2, 0] = col_offset * j + col_offset
            else:
                hex_lattice[i * n // 6 + j // 2, 0] = col_offset * j
            hex_lattice[i * n // 6 + j // 2, 1] = row_offset * i
    hex_lattice = hex_lattice[np.argsort(np.sum(hex_lattice ** 2, axis=1)), :]
    hex_lattice = hex_lattice[:n, :] / np.max(hex_lattice)

    bounds = [(0, 1)] * (2 * n)

        return -np.sum(r)

    def lp_solve(c):
        A_ub, b_ub, pairs = build_constraints(c)
        res = minimize(linear_objective, duals, args=(A_ub, b_ub, pairs),
                          method='L-BFGS-B', bounds=bounds, jac=jacobian, callback=callback)
        if res.success:
            return res.x, res.dual - res.slack
        else:
            raise ValueError("LP solver failed")

    def build_constraints(c):
        A_ub = np.vstack([c[:, 0], c[:, 0] - 1, c[:, 1], c[:, 1] - 1])
        b_ub = np.zeros(4 * n)
        pairs = []
        d_ij, u_ij, v_ij = [], [], []
        for i in range(n):
            if i % 2 == 0:
                j_indices = np.array(range(i + 1, min(n, i + 16)))
            else:
                j_indices = np.array(range(i + 1, min(n, i + 15)))
            for j in j_indices:
                d_ij.append(np.linalg.norm(c[i] - c[j]))
                u_ij.append((c[i, 0] - c[j, 0]) / d_ij[-1])
                v_ij.append((c[i, 1] - c[j, 1]) / d_ij[-1])
        A_ub = np.vstack([A_ub, np.array(u_ij), np.array(v_ij)])
        b_ub = np.append(b_ub, 2 * np.array(d_ij))
        return A_ub, b_ub, pairs

    def linear_objective(duals, A_ub, b_ub, pairs):
        return -np.sum(np.sum(b_ub - np.dot(A_ub, duals.reshape(-1, 1)), axis=0))

    def jacobian(duals, A_ub, b_ub, pairs):
        grad = np.zeros((4 * n,))
        for i in range(n):
            dS_dxi = duals[i] - duals[i + 2*n] + np.sum((duals[j*3 : j*3+2*n].reshape((n, 2)) * ((c[i, 0] - c[j, 0]) / d_ij[j])))
            dS_dyi = duals[i + 2*n] - duals[i + 2*n + n] + np.sum((duals[j*3 +2*n: j*3+4*n].reshape((n, 2)) * ((c[i, 1] - c[j, 1]) / d_ij[j])))
            grad[i] = dS_dxi
            grad[i + 2*n] = dS_dyi
        return -grad

    def callback(res):
        nonlocal best_res
        if res.fun < best_res.fun:
            best_res = res

    best_res = minimize(linear_objective, np.ones(4 * n), args=(A_ub, b_ub, pairs),
                        method='L-BFGS-B', bounds=bounds, jac=jacobian, callback=callback)
    if best_res.success:
        r, duals = lp_solve(best_res.x.reshape((n, 2)))
        c = best_res.x.reshape((n, 2))
        r = np.clip(r, 0, None)
    else:
        raise ValueError("Optimization failed")
    
    return c, r, np.sum(r)
