import numpy as np
import scipy.optimize as opt
import cma

def initial_mean():
    a = np.sqrt(3) / 2
    b = 1.0
    c = 1.0
    jitter = 0.01
    centres = np.zeros((26, 2))
    i, j = 0, 0
    while i + j < 26:
        if i < 4:
            # Corner circles
            centres[i] = [np.random.uniform(-a, a + jitter), np.random.uniform(-b, b + jitter)]
            i += 1
        elif i < 12:
            # Edge circles (midpoint method)
            centres[i] = [np.random.uniform(-a, a + jitter), np.random.uniform(-2 * b + 2 * i * b / 7, 2 * b + 2 * i * b / 7 + jitter)]
            i += 1
        else:
            # Interior circles
            centres[i] = [np.random.uniform(-a, a + jitter), np.random.uniform(-2 * b + 2 * j * b / 7, 2 * b + 2 * j * b / 7 + jitter)]
            i += 1
            j += 1
    r = refine_radii(centres)
    return np.concatenate((centres, r))

def refine_radii(centres):
    bnds = np.vstack((centres, np.minimum(centres[:, 0], 1 - centres[:, 0]), np.minimum(centres[:, 1], 1 - centres[:, 1])))
    return opt.linprog(np.ones(26), Aeq=None, beq=None, A=None, b=None, bounds=bnds, method='highs').x

def objective(z):
    n = 26
    centres = z[:n*2].reshape((n, 2))
    r = refine_radii(centres)
    return np.sum(r) - ((10**4) * np.sum(np.maximum(0, np.minimum(centres[:, 0], 1 - centres[:, 0]) - r)) + (10**4) * np.sum(np.maximum(0, centres[:, 0] + r - 1)) + 2 * (10**4) * np.sum(np.maximum(0, centres[:, 1] + r - 1)) + 2 * (10**4) * np.sum(np.maximum(0, 1 - centres[:, 1] - r) + 1 - centres[:, 1] - r + np.maximum(0, centres[:, 1] - centres[:, 2:].reshape((n, 26 - 1)) - r[:n].reshape((n, 1)) - r[26:].reshape((n, 1)) - r[26:].reshape((n, 1)) + r[:n].reshape((n, 1)) + r[26:].reshape((n, 1)))) + 2 * (10**4) * np.sum(np.maximum(0, centres[:, 1] + centres[:, 2:].reshape((n, 26 - 1)) + r[:n].reshape((n, 1)) + r[26:].reshape((n, 1)) + r[26:].reshape((n, 1)) - r[:n].reshape((n, 1)) - r[26:].reshape((n, 1)) - 1)))

def simulate_annealing(best_z):
    z = best_z
    for _ in range(200):
        indices = np.random.choice(26, 3)
        jitter = 0.01
        shake = np.clip(2 * (np.random.rand(26, 2) - 0.5) * jitter, 0, None)
        shake[indices] *= 1  # Ensure we don't move other circles
        z_new = np.clip(z + shake, 0, 1)
        r_new = refine_radii(z_new)
        v_new = np.sum(r_new)
        if v_new > v_new:
            z = z_new
            v_new = v_new
        else:
            shake = -shake
            z_new = z + shake
            r_new = refine_radii(z_new)
            v_new = np.sum(r_new)
            if v_new > v_new:
                z = z_new
        improvement = v_new - v_new
        if improvement > 1e-9:
            break
    return z

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    m0 = initial_mean()
    es = cma.CMAEvolutionStrategy(m0, 0.1, {'seed': 42, 'popsize': 200, 'bounds': [(0, 1), (0, 1), (0, 0.5), (0, 0.5)]*26})
    while not es.stop():
        solutions = es.ask()
        es.tell(solutions, [objective(sol) for sol in solutions])
    best_z = es.result[0]
    best_x = best_z[~(np.mod(np.arange(52)//2, 2))]
    best_y = best_z[~(np.mod(np.arange(52)//2, 2))]
    best_r = best_z[(np.mod(np.arange(52)//2, 2))]
    for _ in range(200):
        best_z = simulate_annealing(best_z)
    v = 26.36
    return best_x, best_y, best_r, v
