import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def hex_grid(radius, packing_density):
    s = 2 * radius / np.sqrt(3)
    nx, ny = int(np.ceil(1 / s)), int(np.ceil(2 / s))
    cx = np.linspace(0.5, nx * s, nx)
    cy = np.linspace(0.25, ny * s, ny)
    x, y = np.meshgrid(cx, cy, indexing='ij')
    x, y = x.ravel(), y.ravel()
    return x, y

def add_circles_around_quadrant(x, y, radius):
    centers = np.array([
        [x, y],
        [x, y + radius],
        [x + radius, y + radius],
        [x + radius, y],
        [2 * x + radius, 0.5],
        [2 * x, 0.5],
        [2 * x, y],
        [x + radius, 2 * y],
        [x, 2 * y],
        [x, y]
    ])
    return centers

def solve_radii_fixed_centres(centers):
    n = centers.shape[0]
    radii_bounds = [(0, min(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])) for _ in range(n)]
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i, 0]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - (1 - centers[i, 0])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i, 1]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - (1 - centers[i, 1])})
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[i] + r[j] - np.sqrt(np.sum((centers[i] - centers[j]) ** 2))})
    result = linprog([1] * n, bounds=radii_bounds, A_ub=[c['fun'] for c in constraints], b_ub=[c['fun'] for c in constraints], method='highs')
    return result.x, result.fun

def penalty_centres_fixed_radii(centres, radii):
    n = centres.shape[0]
    pen_terms = np.array([
        max(0, r + r_j - np.sqrt(np.sum((c - centres[j]) ** 2))) for j, (r, c) in enumerate(zip(radii, centres))
    ])
    bound_penalties = np.sum([max(0, r - min(x, 1 - x, y, 1 - y)) for x, y, r in zip(centres[:, 0], centres[:, 1], radii)])
    return np.sum(pen_terms) + 0.001 * bound_penalties

def coordinated_ascent(init_centres, max_iter=15):
    for _ in range(max_iter):
        radii, _ = solve_radii_fixed_centres(init_centres)
        centre_bounds = [(0, 1) for _ in range(2 * init_centres.shape[0])]
        def loss(params):
            cf = params.reshape(init_centres.shape)
            return penalty_centres_fixed_radii(cf, radii)
        result = differential_evolution(loss, centre_bounds, strategy='best2bin', popsize=50, mutation=(0, 1.5), recombination=0.7, disp=False)
        init_centres = result.x.reshape(init_centres.shape)
    return init_centres, radii
