import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Build initial centre grid and radii
    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)

    # 2. Define objective and Jacobian
    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, lambda_B, lambda_O)

    def jacobian(z):
        x, y, r = unpack(z)
        return compute_penalty_gradient(x, y, r, lambda_B, lambda_O)

    # 3. Bounds
    bounds = [(0,1)]*52 + [(0,None)]*26

    # 4. Run trust-constr optimiser
    res = minimize(objective, z,
                   jac=jacobian,
                   method='trust-constr',
                   bounds=bounds,
                   options={'gtol':1e-6, 'ftol':1e-8,
                            'maxiter':3000,
                            'verbose':0})

    # 5. Extract best found
    centres_opt, radii_opt = unpack(res.x)

    # 6. Optional LP tightening
    radii_opt, _ = solve_lp(centres_opt)

    sum_r = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_r

def init_hex_grid():
    num_points = 30
    grid = np.array([(i%6+0.5*(i//6%2))/6., i//6/5.] for i in range(num_points)])
    np.random.shuffle(grid)
    centres = grid[:26]
    radii = np.full(26, 0.01)
    return centres, radii

def build_variable_vector(centres, radii):
    return np.concatenate((centres.ravel(), radii))

def unpack(z):
    centres = z[:52].reshape(26, 2)
    radii = z[52:]
    return centres, radii

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    penalty_B = np.sum((np.maximum(0, r - x) ** 2 +
                        np.maximum(0, r - (1 - x)) ** 2 +
                        np.maximum(0, r - y) ** 2 +
                        np.maximum(0, r - (1 - y)) ** 2))
    pen_overlap = 0
    for i in range(len(r)):
        for j in range(i):
            d_ij = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
            pen_overlap += np.maximum(0, r[i] + r[j] - d_ij) ** 2
    return -np.sum(r) + lambda_B * penalty_B + lambda_O * pen_overlap

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    grad = np.zeros(len(x) + len(r))
    for i in range(len(r)):
        grad_x = -2 * lambda_B * np.maximum(0, r - x)
        grad_y = -2 * lambda_B * np.maximum(0, r - y)
        grad_x[i] += lambda_B * (-np.maximum(0, r - (1 - x))[i])
        grad_y[i] += lambda_B * (-np.maximum(0, r - (1 - y))[i])
        for j in range(i):
            d_ij = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
            grad_x[i] += -2 * lambda_O * np.maximum(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij
            grad_y[i] += -2 * lambda_O * np.maximum(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij
            grad_x[j] += -2 * lambda_O * np.maximum(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij
            grad_y[j] += -2 * lambda_O * np.maximum(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij
    grad[52:] = -r.copy()
    return grad

def solve_lp(centres):
    from scipy.optimize import linprog
    n = centres.shape[0]
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        A[i, i] = 2
    for i in range(n):
        for j in range(i+1, n):
            A[i, j] = 2
            A[j, i] = 2
    for i in range(n):
        A[i, n+i] = 2
    for i in range(n):
        b[i] = 2
        b[i+n] = 2
    for i in range(n):
        A[i, 2*n+i] = -2
        A[i, 3*n+i] = -2
        A[i, 4*n+i] = -2
        A[i, 5*n+i] = -2
        b[i+2*n] = centres[i][0] + 2
        b[i+3*n] = 1 - centres[i][0] + 2
        b[i+4*n] = centres[i][1] + 2
        b[i+5*n] = 1 - centres[i][1] + 2
    res = linprog(np.ones(6*n), A_ub=A, b_ub=b, bounds=[(0, None)]*(2*n), method='highs')
    if res.success:
        return res.x[:n], res.fun
    else:
        raise Exception("Linear programming failed to solve")
