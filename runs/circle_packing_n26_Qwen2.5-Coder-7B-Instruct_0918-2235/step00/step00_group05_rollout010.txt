def initial_mean():
    """Generate an initial mean vector covering [0,1]^2 with a hexagonal grid and refine the radii"""
    n = 26
    side = np.sqrt(2) * 0.5 * n
    a = 1.0 / side
    b = np.sqrt(3) / 2 * a
    jitter = 0.01
    centres = []
    for i in range(n):
        row = i // 3
        col = i % 3
        if col % 2 == 0:
            dx = 0
        else:
            dx = b * 0.5
        x = (col * b + dx + jitter * (np.random.rand() - 0.5)) % 1
        if row % 2 == 0:
            y = row * a + a * 0.25 + jitter * (np.random.rand() - 0.5)
        else:
            y = row * a + a * 0.75 + jitter * (np.random.rand() - 0.5)
        centres.append([x, y])
    r = refine_radii(np.array(centres + [0] * n))
    return np.concatenate([np.array(centres), r])


def refine_radii(centres):
    """Refine the radii for a given centre set"""
    r = np.zeros(26)
    for i in range(26):
        r[i] = radii_for_ball(centres, i)
    return r


def radii_for_ball(centres, center_index):
    """Solve the linear program to find the optimal radius for the given centre"""
    n = 26
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        if i != center_index:
            d = np.linalg.norm(centres[center_index] - centres[i])
            A[i, center_index] = 1
            A[i, i] = -1
            b[i] = d
    r = scipy.optimize.linprog(c=np.ones(n) * 0.4, A_eq=A, b_eq=b, bounds=[(0, None)] * n)
    return r.fun


def objective(z):
    """Objective function for optimization"""
    centres = z[:26]
    return np.sum(refine_radii(centres))
