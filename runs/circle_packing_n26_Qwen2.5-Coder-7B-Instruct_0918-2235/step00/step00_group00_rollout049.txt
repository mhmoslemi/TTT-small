import numpy as np
from scipy.optimize import differential_evolution, linprog

def objective(v, n):
    params = v.reshape((n, 3))
    centers, radii = params[:, :2], params[:, 2]
    overlap_penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            overlap_penalty += max(0, radii[i] + radii[j] - dist) ** 2
    boundary_penalty = np.sum(np.maximum(0, radii - centers[:, 0]) ** 2 +
                             np.maximum(0, radii - (1 - centers[:, 0])) ** 2 +
                             np.maximum(0, radii - centers[:, 1]) ** 2 +
                             np.maximum(0, radii - (1 - centers[:, 1])) ** 2)
    return -np.sum(radii) + 1000 * (overlap_penalty + boundary_penalty)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    result = differential_evolution(objective, bounds, args=(n,), popsize=10, maxiter=50)
    best = result.x.reshape((n, 3))
    best_centers, best_radii = best[:, :2], best[:, 2]
    
    # Re-solve with exact solver for optimal radii
    res = linprog([1] * n, A_ub=np.c_[np.ones((n, n)), -2 * best_centers], b_ub=[1] * n,
                    A_eq=None, b_eq=None, bounds={[0, None] for _ in range(n)},
                    method='highs')
    exact_radii = res.x
    return best_centers, exact_radii, np.sum(exact_radii)
