import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    max_radius = 1.0
    centers = np.random.rand(n, 2)
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = 0.0
    step_size = 0.1
    alpha = 0.8
    tolerance = 1e-9
    restarts = 40
    max_cycles = 30

    def build_lp(centers):
        n = centers.shape[0]
        A_ub = np.zeros((n * (n - 1) // 2, n))
        b_ub = np.zeros(n * (n - 1) // 2)
        c = -np.ones(n)
        bounds = np.zeros((n, 2))
        bounds[:, 1] = [np.min([c, 1 - c], axis=0) for c in centers[:, 0] + centers[:, 1]]
        
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[(i * (n - i - 1) // 2 + j - i - 1), i] = 1
                A_ub[(i * (n - i - 1) // 2 + j - i - 1), j] = 1
                b_ub[(i * (n - i - 1) // 2 + j - i - 1)] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        
        return A_ub, b_ub, c, bounds

    for _ in range(restarts):
        centers = np.random.rand(n, 2)
        for _ in range(max_cycles):
            current_sum = 0.0
            for i in range(n):
                c_old = centers[i].copy()
                delta = np.random.uniform(-step_size, step_size, 2)
                centers[i] += delta
                centers[i] = np.clip(centers[i], 0, 1)
                A_ub, b_ub, c, bounds = build_lp(centers)
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
                if res.success:
                    current_sum += res.fun
            if current_sum > best_sum + tolerance:
                best_sum = current_sum
                best_centers = centers.copy()
            else:
                centers = best_centers.copy()
            step_size *= alpha
            if step_size < 1e-4:
                break
    best_coords = best_centers
    external_dist = np.linalg.norm(best_coords, axis=1) - best_radii
    bool_mask = external_dist >= 0
    best_centers = best_centers[bool_mask, :]
    best_radii = best_radii[bool_mask]
    A_ub, b_ub, c, bounds = build_lp(best_centers)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
    return best_centers, res.x, res.fun
