import numpy as np
from scipy.optimize import differential_evolution
import cvxpy as cp

def objective(v, n):
    r = v[n:]
    x = v[:n]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j]) ** 2 + (x[n + i] - x[n + j]) ** 2)
            penalty += max(0, r[i] + r[j] - dist) ** 2
    for i in range(n):
        penalty += max(0, r[i] - x[i]) ** 2
        penalty += max(0, r[i] - (1 - x[i])) ** 2
        penalty += max(0, r[i] - x[n + i]) ** 2
        penalty += max(0, r[i] - (1 - x[n + i])) ** 2
    return -np.sum(r) + 1000 * penalty

def run_packing():
    n = 26
    bounds = [(0, 1)] * n + [(0, 0.5)] * n
    result = differential_evolution(optimizer, bounds, popsize=10, maxiter=50, strategies=['rand/1/exp'])
    v = result.x
    r = v[n:]
    x = v[:n]
    # Ensure r is nonnegative
    r = np.maximum(0, r)
    # Correct radius values if they are too large due to numerical issues
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j]) ** 2 + (x[n + i] - x[n + j]) ** 2)
            if r[i] + r[j] > dist:
                arr = np.array([r[i], r[j]])
                arr = arr * dist / (r[i] + r[j])
                r[i] = arr[0]
                r[j] = arr[1]
    # Solve LP to get exact radii
    problem = cp.Problem()
    problem.addConstraints([cp.sqrt((x[i] - x[j]) ** 2 + (x[n + i] - x[n + j]) ** 2) >= r[i] + r[j] for i in range(n) for j in range(i + 1, n)])
    problem.addConstraints([x[i] >= r[i] for i in range(n)])
    problem.addConstraints([x[i] <= 1 - r[i] for i in range(n)])
    problem.addConstraints([x[n + i] >= r[i] for i in range(n)])
    problem.addConstraints([x[n + i] <= 1 - r[i] for i in range(n)])
    problem.setObjective(cp.sum(r))
    problem.solve()
    r_exact = r.value
    sum_radii = np.sum(r_exact)
    return x[:n].reshape(-1, 2), r_exact, sum_radii

def optimizer(v, n):
    return objective(v, n)

centers, radii, sum_radii = run_packing()
