def calc_max_radii(C):
    n = C.shape[0]
    r = np.zeros(n)
    for i in range(n):
        b = np.zeros(n)
        for j in range(n):
            if j != i:
                b[i] += min(C[i, 0] - C[j, 0], C[i, 1] - C[j, 1], C[j, 0] - C[i, 0], C[j, 1] - C[i, 1])

        l = abs(np.diff(C[i, 0]))
        r[i] = min(b, l)

    return total_radius, r
