To implement the described algorithm, we will create three functions: `refine_radii`, `anneal_one_start`, and `run_packing`. These functions will work together to generate a feasible packing of 26 circles in a unit square and maximize the sum of their radii.

Here is the complete implementation:

```python
import numpy as np

# Initialize the global random number generator
rng = np.random.default_rng()

def refine_radii(centres, radii, eps=1e-12, max_iter=200):
    """
    Refine radii for a fixed set of centres to ensure non-overlap and boundary constraints.
    
    Parameters:
    centres (np.ndarray): array of shape (26,2)
    radii (np.ndarray): array of shape (26,)
    
    Returns:
    np.ndarray: refined radii
    """
    n = len(radii)
    dist_matrix = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    
    while max_iter > 0:
        max_changes = 0
        for i in range(n):
            x, y = centres[i]
            max_radius = min(x, 1 - x, y, 1 - y)
            distances = np.maximum(0, dist_matrix[i] - radii - eps)[:, i]
            note = np.argmin(distances)
            max_radius = min(max_radius, distances[note])
            if max_radius < 0:
                max_radius = 0
            if max_radius > radii[i]:
                radii[i] = max_radius
                max_changes += 1
        if max_changes == 0:
            break
        max_iter -= 1
    
    return radii

def anneal_one_start(initial_centers, rng, steps=500, init_step=0.07, min_step=0.01):
    """
    Perform simulated annealing on circle locations to maximize the sum of radii.
    
    Parameters:
    initial_centers (np.ndarray): initial array of shape (26,2)
    rng (np.random.Generator): random number generator
    steps (int): number of annealing steps
    init_step (float): initial step-size
    min_step (float): minimum step-size
    
    Returns:
    tuple: (best_centres, best_radii, best_sum)
    """
    n = 26
    best_centers = initial_centers.copy()
    best_radii = np.full(n, 1e-3)
    best_sum = np.sum(refine_radii(best_centers, best_radii))
    
    step = init_step
    for stepnum in range(steps):
        T = 0.25 * (1 - stepnum / steps)
        step = 0.07 * (1 - stepnum / steps)
        
        for i in range(n):
            old_pos = best_centers[i]
            delta = (rng.random(2) - 0.5) * 2 * step
            new_pos = old_pos + delta
            new_pos = np.clip(new_pos, 0, 1)
            best_centers[i] = new_pos
            
            new_radii = refine_radii(best_centers, best_radii)
            new_sum = np.sum(new_radii)
            
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = best_centers.copy()
            else:
                metropolis_p = np.exp((new_sum - best_sum) / T)
                if rng.random() < metropolis_p:
                    best_sum = new_sum
                    best_centers = best_centers.copy()
                else:
                    best_centers[i] = old_pos
    
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Run the packing algorithm with multiple restarts.
    
    Returns:
    tuple: (centers, radii, sum_of_radii)
    """
    RESTARTS = 30
    best_total = float('-inf')
    best_centers = None
    best_radii = None
    
    for _ in range(RESTARTS):
        init_centres = rng.random((26, 2))
        centers, radii, total = anneal_one_start(init_centres, rng)
        if total > best_total:
            best_total = total
            best_centers = centers
            best_radii = radii
    
    return best_centers, best_radii, best_total

# Example usage:
```
