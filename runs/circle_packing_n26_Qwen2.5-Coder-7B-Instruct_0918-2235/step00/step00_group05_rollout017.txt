import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist

def lp_radii_fixup(R, centers):
    n = len(centers)
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        r_i = R[i]
        for j in range(i+1, n):
            r_j = R[j]
            A[i,j] = 2 * (centers[j,0] - centers[i,0])
            A[i,j*n+n] = 2 * (centers[j,1] - centers[i,1])
            b[i] += r_j**2
    return linprog(R, A_ub=A, b_ub=b, A_eq=None, b_eq=None, method='highs').x

def gradient(R, centres):
    n = len(centres)
    dR = np.ones(n)
    for i in range(n):
        r_i = R[i]
        for j in range(i):
            r_j = R[j]
            dist = np.linalg.norm(centres[i] - centres[j])
            dR[i] += 2 * (dist - r_i - r_j) * (centres[i] - centres[j]) / dist
    return dR

def objective(R, centres, pen_wall, pen_ovl):
    boundary_purtubation = 1e-10
    centers = np.stack([centres[:, 0] + R,
                          centres[:, 1] + R,
                          np.ones_like(R) - centres[:, 0] - R,
                          np.ones_like(R) - centres[:, 1] - R], axis=-1)
    trunctated_centers = np.where(centers < 0, centers,
                                  np.where(centers > 1, centers, 1e-5 + boundary_purtubation))
    R_wall = np.sum(np.min(trunctated_centers, axis=-1), axis=0)
    R_ovl = cdist(centres, centres)
    R_ovl = np.sum(np.maximum(0, r_i + r_j - R_ovl), axis=1)
    return np.sum(R) - pen_wall * R_wall - pen_ovl * R_ovl

def shake(centers, radii, step_size, n):
    for _ in range(n):
        idx = np.random.randint(0, len(centers), size=min(len(centers), 3))
        centres[idx] += np.random.uniform(-step_size, step_size, (len(idx), centres.shape[-1]))
        radii[idx] = lp_radii_fixup(radii[idx], centres[idx])
    return centres, radii
