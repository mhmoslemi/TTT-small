import numpy as np
from scipy.optimize import linprog, minimize

def generate_centres(theta, dx, dy, s):
    b1 = np.array([s, 0])
    b2 = np.array([s/2, s * np.sqrt(3)/2])
    centre_coords = []
    centre_coords.extend(generate_row(b1, b2, -4, 4, n2=0, s=s))
    centre_coords.extend(generate_row(b1, b2, -3, 4, n2=1, s=s))
    centre_coords.extend(generate_row(b1, b2, -4, 4, n2=2, s=s))
    centres = np.array(centre_coords)
    rotation_matrix = np.array([[np.cos(theta), -np.sin(theta)],
                                [np.sin(theta), np.cos(theta)]])
    centres = np.dot(centres, rotation_matrix) + np.array([dx, dy])
    return centres

def generate_row(b1, b2, n1_start, n1_end, n2, s, max_dist=1.0):
    points = []
    for n1 in range(n1_start, n1_end + 1):
        point = (n1 * b1[0] + n2 * b2[0], n1 * b1[1] + n2 * b2[1])
        if np.linalg.norm(point) <= max_dist:
            points.append(point)
    return points

def solve_lps(centres):
    n = len(centres)
    x = lpsolve(centres, n)
    r = np.array(x['radii'])
    return r, x['sum']

def lpsolve(centres, n):
    c = np.ones(n)
    A = np.zeros((3*n, n))
    A[:n, :] = np.eye(n)
    A[n:2*n, :] = -centres
    a_eq = np.ones(2*n)
    for p in range(n):
        for q in range(p+1, n):
            A[2*n + n*p + q, p] = 1
            A[2*n + n*p + q, q] = 1
            a_eq[2*n + n*p + q] = -np.linalg.norm(centres[p] - centres[q])
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c=c, A_ub=A, b_ub=a_eq, bounds=bounds, method='highs')
    radii = res.x
    sum_r = res.fun
    return {'radii': radii, 'sum': sum_r}

def nelder_mead(seed, tolerance=1e-7, max_iter=1000):
    res = minimize(objective, seed, method='nelder-mead', tol=tolerance,
                   options={'maxiter': max_iter})
    return res.x

def objective(params):
    theta, dx, dy = params
    centres = generate_centres(theta, dx, dy, 0.08)
    if any_outside(centres):
        return -1e308
    r, s = solve_lps(centres)
    return -s

def any_outside(centres):
    return np.any(np.linalg.norm(centres, axis=1) > 1 + 1e-12)

def sog(centres, theta, dx, dy):
    return ((centres @ np.array([np.cos(theta), -np.sin(theta)]) +
            np.array([dx, dy])) ** 2).sum()

def solve(centres, scale):
    centres *= scale
    x = lpsolve(centres, len(centres))
    r = np.array(x['radii'])
    return (centres, r, x['sum'])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centres, radii, sum_radii = solve(centres, 0.08)
    return centres, radii, sum_radii

This code implements the specified strategy, generating the centers using a hexagonal lattice with a rotation and offset, solving the LP problem to determine the optimal radii, and optimizing this entire configuration using a Nelder–Mead algorithm.
