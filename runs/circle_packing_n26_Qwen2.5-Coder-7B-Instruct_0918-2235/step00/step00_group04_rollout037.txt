import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def init_hex_grid():
    # Construct a 6x5 hexagonal lattice with some random points dropped
    hex_lattice = np.array([

    ])
    centers = np.random.permutation(hex_lattice)[:26]
    return centers, np.full(26, 0.01)

def build_variable_vector(centers, radii):
    return np.concatenate((centers.flatten(), radii))

def unpack(z):

def compute_distance_matrix(centers):
    dist = pdist(centers)
    return squareform(dist)

def compute_boundary_penalty(r, centers):
    delta_x = np.clip(r - centers[:, 0], 0, None)
    delta_y = np.clip(r - centers[:, 1], 0, None)
    pen_x = 2 * np.sum(delta_x ** 2)
    pen_y = 2 * np.sum(delta_y ** 2)
    return pen_x + pen_y

def compute_overlap_penalty(r, centers, dist_matrix):
    overlap = np.clip(r[:, None] + r - dist_matrix, 0, None)
    return 2 * np.sum(overlap ** 2)

def objective(z):
    centers = z[:52].reshape((26, 2))
