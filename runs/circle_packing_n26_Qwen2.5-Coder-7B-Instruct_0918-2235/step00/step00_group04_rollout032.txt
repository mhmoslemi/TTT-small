import numpy as np
from scipy.optimize import linprog

def solve_radius_lp(centres_i):
    n_circles = centres_i.shape[0]
    D = np.sqrt(np.sum((centres_i[:, np.newaxis, :] - centres_i[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(D, np.inf)
    c = -np.ones(n_circles)
    A_ub_indices = np.where(np.triu((D > 1) & (0.5 < D), k=1).nonzero())
    A_ub = np.zeros((len(A_ub_indices[0]), n_circles))
    A_ub[:, A_ub_indices[0]] = np.eye(n_circles)[A_ub_indices[0]]
    A_ub[:, A_ub_indices[1]] = np.eye(n_circles)[A_ub_indices[1]]
    b_ub = D[A_ub_indices] - (np.eye(n_circles)[A_ub_indices[0]] + np.eye(n_circles)[A_ub_indices[1]])
    A_ub = np.vstack([np.hstack([np.eye(n_circles), np.zeros((n_circles, n_circles))]),
                       np.hstack([np.zeros((n_circles, n_circles)), np.eye(n_circles)])])
    b_ub = np.hstack([centres_i[:, 0], np.ones(n_circles)])
    b_ub = np.hstack([centres_i[:, 1], np.ones(n_circles)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return -res.x, -res.fun

def run_packing():
    n_particles = 30
    n_circles = 26
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    max_iter = 3000
    vel_clip = 0.1
    alpha = 0.999

    np.random.seed(42)
    grid = np.array(np.meshgrid(np.linspace(0, 1, 6), np.linspace(0, 1, 5)))
    grid = grid.reshape(n_circles * 5, 2)
    np.random.shuffle(grid)
    grid = np.delete(grid, np.random.choice(np.arange(n_circles * 5), 4), axis=0)[:n_circles]
    seed_positions = grid + (np.random.normal(0, 0.01, (n_circles, 2)) - 0.5) * 0.1

    centres = np.tile(seed_positions, (n_particles, 1, 1)) + np.random.normal(0, 0.01, (n_particles, n_circles, 2))
    velocities = np.zeros_like(centres)
    pbests = centres.copy()
    pbest_sums = np.zeros(n_particles)
    gbest = centres.mean(axis=0)
    gbest_sum = solve_radius_lp(gbest)[1]

    for i in range(max_iter):
        for p in range(n_particles):
            for m in range(n_circles):
                velocities[p, m] = inertia * velocities[p, m] + cognitive * np.random.rand() * (pbests[p, m] - centres[p, m]) \
                                   + social * np.random.rand() * (gbest[m] - centres[p, m])
                velocities[p, m, 0] = np.clip(velocities[p, m, 0], -vel_clip, vel_clip)
                velocities[p, m, 1] = np.clip(velocities[p, m, 1], -vel_clip, vel_clip)

            centres[p] += velocities[p]
            centres[p, :, 0] = np.clip(centres[p, :, 0], 0, 1)
            centres[p, :, 1] = np.clip(centres[p, :, 1], 0, 1)

            radii, sum_radii = solve_radius_lp(centres[p])
            if sum_radii > pbest_sums[p]:
                pbests[p] = centres[p].copy()
                pbest_sums[p] = sum_radii
                if sum_radii > gbest_sum:
                    gbest = pbests[p].copy()
                    gbest_sum = pbest_sums[p]
        
        if i % 1000 == 999:
            velocities *= alpha

    return gbest, solve_radius_lp(gbest)[0], gbest_sum
