import numpy as np
from scipy.optimize import linprog
from scipy.optimize.minimize import minimize

# Function to build the linear programming problem
def build_lp(centers):
    n = centers.shape[0]
    A = []
    b = []
    for i in range(n):
        A.append([1, 0])
        b.append(centers[i, 0])
        A.append([0, 1])
        b.append(centers[i, 1])
        A.append([1, 0])
        b.append(1 - centers[i, 0])
        A.append([0, 1])
        b.append(1 - centers[i, 1])
        for j in range(i + 1, n):
            A.append([1, 0, 1, 0])
            b.append(np.linalg.norm(centers[i] - centers[j]))
    A = np.array(A)
    b = np.array(b)
    return A, b

# Gradient computation
def gradient(centers, A, b):
    n = centers.shape[0]
    dual = linprog(np.ones(2*n), A_ub=A, b_ub=b, method='highs').dual
    grad_x = np.zeros(2*n)
    grad_y = np.zeros(2*n)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist > 0:
                dual_ij = dual[2*n + j*n + i]
                grad_x[2*i] += dual_ij * (centers[i, 0] - centers[j, 0]) / dist
                grad_x[2*j] -= dual_ij * (centers[i, 0] - centers[j, 0]) / dist
                grad_y[2*i] += dual_ij * (centers[i, 1] - centers[j, 1]) / dist
                grad_y[2*j] -= dual_ij * (centers[i, 1] - centers[j, 1]) / dist
        grad_x[2*i] += dual[i*2] - dual[(i*2)+1]
        grad_y[2*i] += dual[i*2] - dual[(i*2)+1]
        grad_x[2*(n+i)] += dual[(n*2) + i*2] - dual[(n*2) + (i*2)+1]
        grad_y[2*(n+i)] += dual[(n*2) + i*2] - dual[(n*2) + (i*2)+1]
    return np.array(list(grad_x) + list(grad_y))

# Objective function
def objective(centers, A, b):
    _, radii = linprog(np.ones(2*centers.shape[0]), A_ub=A, b_ub=b, method='highs')
    return -np.sum(radii)

# Main function to run the packing
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([[0.5, 0.5]])
    for _ in range(24):
        centers = np.vstack((centers, centers[-1] + np.array([0.25, 0.25])))
    centers = centers[:26, :]
    centers = centers[np.random.permutation(26)]
    A, b = build_lp(centers)
    result = minimize(objective, centers.flatten(), args=(A, b), jac=gradient, method='L-BFGS-B', bounds=[(0, 1)] * (2 * centers.shape[0]))
    optimized_centers = np.reshape(result.x, (26, 2))
    _, radii = linprog(np.ones(2*optimized_centers.shape[0]), A_ub=A, b_ub=b, method='highs')
    sum_radii = np.sum(radii)
    return optimized_centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print((centers, radii, sum_radii))
