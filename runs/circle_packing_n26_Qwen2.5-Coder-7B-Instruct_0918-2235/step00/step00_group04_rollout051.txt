import numpy as np
from scipy.optimize import minimize

def init_hex_grid():
    n = 30
    x = np.linspace(0.1667, 0.8333, n)
    y = np.array([0.5] * n)
    x[1::2] += 0.5
    y[1::2] += np.full(n//2, (np.sqrt(3)/6))
    centres = np.vstack((x, y)).T
    return centres[:26], np.full(26, 0.01)

def build_variable_vector(centres, radii):
    return np.concatenate([centres.flatten(), radii])

def unpack(z):
    centres = z[:60].reshape(26, 2)
    radii = z[60:]
    return centres, radii

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
   -boundary_penalty = lambda_B * np.sum(np.maximum(np.zeros_like(r), r - x)**2
                                         + np.maximum(np.zeros_like(r), r - (1 - x))**2
                                         + np.maximum(np.zeros_like(r), r - y)**2
                                         + np.maximum(np.zeros_like(r), r - (1 - y))**2)
    overlap_penalty = lambda_O * np.sum([np.maximum(np.zeros_like(r), r[i] + r[j] - d_ij)**2
                                      for i in range(26) for j in range(i+1, 26)
                                      for d_ij in np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)])
    return np.sum(r) - boundary_penalty - overlap_penalty

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    n = 26
    grad = np.zeros(78)
    grad_x = grad + np.zeros(78)
    grad_y = grad + np.zeros(78)
    grad_r = grad + np.zeros(78)

    # Boundary gradient
    for i in range(n):
        grad_x[i*3] += -2 * np.maximum(0, r[i] - x[i]) * (x[i] < r[i])
        grad_y[i*3+1] += -2 * np.maximum(0, r[i] - y[i]) * (y[i] < r[i])
        grad_x[i*3+2] += -2 * np.maximum(0, r[i] - (1 - x[i])) * (1 - x[i] < r[i])
        grad_y[i*3+2] += -2 * np.maximum(0, r[i] - (1 - y[i])) * (1 - y[i] < r[i])

    # Overlap gradient
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
            grad_x[i*3] += -2 * np.maximum(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (r[i] + r[j] - d_ij > 0)
            grad_y[i*3] += -2 * np.maximum(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij * (r[i] + r[j] - d_ij > 0)
            grad_x[j*3] += -2 * np.maximum(0, r[i] + r[j] - d_ij) * (x[i] - x[j]) / d_ij * (r[i] + r[j] - d_ij > 0)
            grad_y[j*3] += -2 * np.maximum(0, r[i] + r[j] - d_ij) * (y[i] - y[j]) / d_ij * (r[i] + r[j] - d_ij > 0)

    return grad[:78]

def run_packing():
    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, 10**6, 10**6)

    def jacobian(z):
        x, y, r = unpack(z)
        return compute_penalty_gradient(x, y, r, 10**6, 10**6)

    centres, radii = init_hex_grid()
    z = build_variable_vector(centres, radii)
    bounds = [(0,1)]*52 + [(0,None)]*26
    res = minimize(objective, z, jac=jacobian, method='trust-constr', bounds=bounds, options={'maxiter':3000, 'gtol':1e-6, 'ftol':1e-8})
    centres_opt, radii_opt = unpack(res.x)
    sum_r = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_r
