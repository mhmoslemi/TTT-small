import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def build_lp(C):
    N = C.shape[0]
    d = cdist(C, C)
    d[np.triu_indices(N, 1)] = np.inf
    inequalities = []
    for i in range(N):
        inequalities.extend([
            [1, 0, 0, -1, d[i, j]] for j in range(N) if j != i
        ])
        inequalities.extend([
            [-1, 0, 0, 1, d[i, j]] for j in range(N) if j != i
        ])
    inequalities.extend([
        [0, 1, 0, 0, -C[i, 0]] for i in range(N)
    ])
    inequalities.extend([
        [0, 0, 1, 0, -C[i, 1]] for i in range(N)
    ])
    inequalities.extend([
        [0, 0, 0, 1, -(1 - C[i, 0])] for i in range(N)
    ])
    inequalities.extend([
        [0, 0, 0, 0, -(1 - C[i, 1])] for i in range(N)
    ])
    return np.array(inequalities), np.zeros(N)

def compute_sum_radius(C):
    C = np.clip(C, 0, 1)
    inequalities, objectives = build_lp(C)
    try:
        result = linprog(objectives, A_ub=inequalities, bounds=[0, None], method="highs", options={"disp": 0})
        return -result.fun, C, result.x
    except Exception as e:
        return 1e6, C, None

def de_obj(C):
    return compute_sum_radius(C)[0]

def initialize_population(N, seed, popsize):
    C = np.repeat(seed[np.newaxis, :, :], popsize, axis=0)
    C += np.random.normal(0, 0.02, C.shape)
    C += np.random.rand(C.shape) * 2 - 1
    C = np.clip(C, 0, 1)
    return C
