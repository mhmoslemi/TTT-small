def coordinated_ascent(init_centres, max_iter=15):
    current_centres = init_centres
    current_radii, current_sum = solve_radii_fixed_centres(current_centres)
    
    for _ in range(max_iter):
        DE_result = differential_evolution(lambda c: -solve_radii_fixed_centres(c.reshape((len(current_centres), 2)))[1], 
                                           bounds=[(0, 1), ] * len(current_centres),
                                           x0=current_centres.reshape((len(current_centres), 2)),
                                           disp=False,
                                           bounds=[(0, 1), ] * len(current_centres),
                                          _ieqcons=[lambda c, r: solve_radii_fixed_centres(c.reshape((len(current_centres), 2)))[0] - r],
                                           maxiter=100)
        current_centres = DE_result.x.reshape((len(current_centres), 2))
        new_radii, new_sum = solve_radii_fixed_centres(current_centres)

        if new_sum > current_sum:
            current_sum = new_sum
            current_radii = new_radii
        else:
            diff = DE_result.fun - -current_sum
            if diff < 1e-6:
                break
            else:
                current_centres = current_centres

    return current_centres, current_radii, current_sum
