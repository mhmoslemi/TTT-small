import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_for_radii(centers):
    radii = 0.1 * np.ones(26)
    n = centers.shape[0]
    c = np.ones(n)
    A_ub = np.zeros((n*(n-1)//2 + n, n))
    b_ub = np.zeros(n*(n-1)//2 + n)
    for i in range(n):
        A_ub[i*(n-1)//2+i, i] = 1.0
        b_ub[i*(n-1)//2+i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i] - centers[j])**2).sum()
            A_ub[i*(n-1)//2+j] = np.hstack((np.zeros(i), 1.0, np.zeros(n-j-1)))
            b_ub[i*(n-1)//2+j] = dist
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    return -res.fun, res.x

def de_objective(u):
    x = u[:26]
    y = u[26:]
    centers = np.vstack((x, y)).T
    sum_radii, radii = lp_for_radii(centers)
    return -sum_radii

np.random.seed(0)
def run_packing():
    bounds = [(0, 1)] * 52
    popsize = 20
    maxiter = 200
    tol = 1e-12
    result = differential_evolution(de_objective, bounds, popsize=popsize, maxiter=maxiter, tol=tol)
    best_centers = np.array(result.x[:26])
    best_radii = np.array(result.x[26:])
    sum_radii, _ = lp_for_radii(best_centers)
    return best_centers, best_radii, sum_radii
