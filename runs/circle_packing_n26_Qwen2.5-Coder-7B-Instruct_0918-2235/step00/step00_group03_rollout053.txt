from scipy.spatial import distance

def compute_all_radii(centers, radii, dist_matrix, eps):
    n = centers.shape[0]
    new_radii = np.zeros(n)
    for i in range(n):
        b_i = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        min_dist = np.inf
        for j in range(n):
            if j != i:
                dist_ij = dist_matrix[i, j] - radii[j] - eps
                if dist_ij > 0:
                    min_dist = min(min_dist, dist_ij)
        r_i = min(b_i, min_dist)
        new_radii[i] = r_i
    return np.maximum(new_radii, 0)

def refine_all(centers, radii, eps=1e-12, max_iter=200):

        if np.allclose(radii, new_radii):
            break
        radii = new_radii
        centers = new_centers
    return radii

    rng = np.random.default_rng()
    RESTARTS = 30
    best_centers = best_radii = None
