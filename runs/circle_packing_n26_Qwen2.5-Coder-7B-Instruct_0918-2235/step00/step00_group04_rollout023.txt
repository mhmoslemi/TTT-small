import numpy as np
from scipy.optimize import linprog

def compute_lp_solution(centers):
    n = centers.shape[0]
    radii_bounds = [(0, None) for _ in range(n)]
    
    c = -np.ones(n)
    
    A_eq = np.zeros((2*n, n))
    b_eq = np.zeros(2*n)
    A_ineq = np.empty((4*n + n*(n-1)//2, n))
    b_ineq = np.empty(4*n + n*(n-1)//2)
    
    for i in range(n):
        A_eq[2*i, i] = 1
        b_eq[2*i] = centers[i, 0]
        A_eq[2*i+1, i] = 1
        b_eq[2*i+1] = centers[i, 1]
    
    row = 2*n
    for i in range(n):
        for j in range(i+1, n):
            delta_x = centers[i, 0] - centers[j, 0]
            delta_y = centers[i, 1] - centers[j, 1]
            dist_squared = delta_x**2 + delta_y**2
            b_ineq[row] = dist_squared
            A_ineq[row, i] = 1
            A_ineq[row, j] = 1
            row += 1
    
    A_ineq = np.vstack((A_ineq, np.eye(n), -np.eye(n), np.tile(np.ones((n, 1)), (2, 1))))
    b_ineq = np.hstack((b_ineq, centers[:, 0], centers[:, 0], centers[:, 1], centers[:, 1]))
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=radii_bounds, method='highs')
    
    if not res.success:
        raise ValueError("Optimization failed")
    
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    grid_size = 5
    centers = np.meshgrid(np.linspace(0, 1, grid_size), np.linspace(0, 1, grid_size))
    centers = np.array([centers[0].flatten(), centers[1].flatten()]).T
    centers = np.random.choice(centers, n, replace=False)
    
    current_centers = centers.copy()
    current_radii, current_sum = compute_lp_solution(current_centers)
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    
    step = 0.1
    MAX_ITERS = 20000
    K_JUMP = 500
    N_REDUCE = 500
    epsilon = 1e-12
    
    for iter in range(MAX_ITERS):
        i = np.random.randint(n)
        new_x = np.clip(current_centers[i, 0] + np.random.uniform(-step, step), 0, 1)
        new_y = np.clip(current_centers[i, 1] + np.random.uniform(-step, step), 0, 1)
        new_centers = current_centers.copy()
        new_centers[i] = [new_x, new_y]
        new_radii, new_sum = compute_lp_solution(new_centers)
        
        if new_sum > current_sum + epsilon:
            current_centers = new_centers.copy()
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_centers = new_centers.copy()
                best_radii = new_radii
                best_sum = new_sum
        
        if iter % N_REDUCE == 0 and iter > 0:
            step *= 0.9
        
        if iter % K_JUMP == 0:
            new_centers = np.random.choice(centers, n, replace=False)
            new_radii, new_sum = compute_lp_solution(new_centers)
            if new_sum > best_sum:
                best_centers = new_centers.copy()
                best_radii = new_radii
                best_sum = new_sum
    
    return best_centers, best_radii, best_sum

# Example usage
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of radii:", sum_radii)
