import numpy as np
from scipy.optimize import linprog, differential_evolution
from math import sqrt

def hex_grid(radius, packing_density=0.5):
    s = 2 * radius / sqrt(3)
    rows = int(packing_density / (sqrt(3) * 0.75))
    col = int(packing_density * 0.5 / 3)
    grid = np.zeros((rows, col))
    for row in range(rows):
        for col in range(col):
            x = (col + 0.5 * (row % 2)) * s
            y = row * s * sqrt(3) / 2
            if x < 1 and y < 1:
                grid[row, col] = 1
    centers = np.argwhere(grid == 1)
    centers = centers[:, ::-1] * s
    return centers

def add_circles_around_quadrant(center, radius):
    return np.array([[center[0] + radius, center[1] + radius],
                     [center[0] + radius, center[1] - radius],
                     [center[0] - radius, center[1] - radius],
                     [center[0] - radius, center[1] + radius]])

def solve_radii_fixed_centres(centers):
    n = centers.shape[0]
    A = np.zeros((2*n, n))
    b = np.zeros(2*n)
    for i in range(n):
        A[i, i] = -1
        A[i+n, i] = 1
        b[i] = centers[i, 0] + centers[i, 1]
        b[i+n] = 1 - centers[i, 0] + centers[i, 1]
    res = linprog(A=A, b_eq=b, bounds=[(0, None)] * n)
    return res.x, sum(res.x)

def penalty_centres_fixed_radii(centres, radii):
    n = centres.shape[0]
    penalty = 0
    for i in range(n):
        penalty += max(0, radii[i] - min(centres[i, 0], 1-centres[i, 0], centres[i, 1], 1-centres[i, 1]))
    for i in range(n):
        for j in range(i + 1, n):
            dist = sqrt(np.sum((centres[i] - centres[j]) ** 2))
            penalty += max(0, radii[i] + radii[j] - dist)
    return penalty
