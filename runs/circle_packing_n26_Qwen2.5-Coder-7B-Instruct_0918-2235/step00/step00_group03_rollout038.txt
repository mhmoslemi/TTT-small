import numpy as np
from scipy.optimize import minimize

def refine_radii(centers, radii):
    n = centers.shape[0]
    dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dist_matrix, 0)
    min_radii = np.min(np.maximum(0, dist_matrix - radii[:, np.newaxis] - radii[np.newaxis, :]), axis=1)
    new_radii = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1]))
    new_radii = np.minimum(new_radii, min_radii)
    return np.maximum(new_radii, 0)

def objective_function(params, centers):
    radii = params.reshape(26,)
    return -np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.any(np.isnan(centers)) or np.any(np.isnan(radii)):
        return False, "NaN values present"
    if not np.all(radii >= 0):
        return False, "Radii must be non-negative"
    if not np.all((centers[:, 0] >= radii) & (centers[:, 0] <= 1 - radii) & (centers[:, 1] >= radii) & (centers[:, 1] <= 1 - radii)):
        return False, "Circles must lie within unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing():
    n = 26
    REPEATS = 30
    centers_best = None
    sum_radii_best = 0
    rng = np.random.default_rng()
    
    for _ in range(REPEATS):
        centers = rng.random((n, 2))
        bounds = ((0, 1), (0, 1))
        constraints = ({'type': 'ineq', 'fun': lambda p: refine_radii(centers, p)}
                       for _ in range(n))
        result = minimize(objective_function, np.full(n, 1e-3), bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 2000})
        radii = result.x
        sum_radii = -result.fun
        if sum_radii > sum_radii_best:
            sum_radii_best = sum_radii
            centers_best = centers
            radii_best = radii
    
    validated, message = validate_packing(centers_best, radii_best)
    if not validated:
        raise ValueError(message)

    return centers_best, radii_best, sum_radii_best
