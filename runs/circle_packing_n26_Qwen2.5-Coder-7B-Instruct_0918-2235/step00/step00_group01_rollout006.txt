import numpy as np

def distance(x1, y1, x2, y2):
    return np.sqrt((x1-x2)**2 + (y1-y2)**2)

def penalty(v, n=26, P=60):
    pen = 0
    for i in range(n):
        x_i = v[2*i]
        y_i = v[2*i+1]
        r_i = v[2*i+2]
        pen -= np.clip(-(x_i - r_i), 0)
        pen -= np.clip(-(1 - x_i - r_i), 0)
        pen -= np.clip(-(y_i - r_i), 0)
        pen -= np.clip(-(1 - y_i - r_i), 0)
    
    for i in range(n):
        for j in range(i + 1, n):
            x_i = v[2*i]
            y_i = v[2*i+1]
            r_i = v[2*i+2]
            x_j = v[2*j]
            y_j = v[2*j+1]
            r_j = v[2*j+2]
            c_i = np.array([x_i, y_i])
            c_j = np.array([x_j, y_j])
            pen -= np.clip((r_i + r_j) - distance(x_i, y_i, x_j, y_j), 0)
    
    return pen

def objective(v, P=60, n=26):
    λ = 1000
    return np.sum(v[2*n:]) - λ * penalty(v)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    P = 60
    G = 200
    w = 0.7298
    c1 = 1.49618
    c2 = 1.49618
    v_max = np.array([1, 1, 0.01]) * 10
    np.random.seed(12345)

    v = np.random.rand(P, n * 3) * v_max
    p_best = v.copy()
    g_best = v[0]

    g_best_val = objective(g_best)

    λ = 1000
    for g in range(G):
        for p in range(P):
            r = v[p, 2*n:]
            dv = w * v[p, :2*n] + c1 * np.random.rand(n * 3) * (p_best[p, :2*n] - v[p, :2*n]) + c2 * np.random.rand(n * 3) * (g_best - v[p, :2*n])
            v[p, :2*n] += dv
            v[p, :2*n] = np.clip(v[p, :2*n], np.zeros(2*n), v_max)
            v[p, 2*n:] = np.clip(v[p, 2*n:], np.zeros(n), v_max)
            Vp = v[p]
            Vp[2*n:] += r
            Vp[2*n:] = np.clip(Vp[2*n:], np.zeros(n), v_max)
            Vp = np.clip(Vp, np.zeros(n * 3), v_max)
            Vp = np.round(Vp, decimals=6)
            Vp[2*n:] -= r
            Vp[2*n:] = np.clip(Vp[2*n:], np.zeros(n), v_max)
            Vp[2*n:] = np.round(Vp[2*n:], decimals=6)
            val = objective(Vp, P, n)
            if val > g_best_val:
                g_best_val = val
                g_best = Vp
            if val > p_best_val[p]:
                p_best[p] = Vp

        λ *= 1.02

    centers = g_best[:2*n].reshape(n, 2)
    radii = np.maximum(g_best[2*n:], 0)
    sum_radii = np.sum(radii)

    centers = centers.astype(np.float32)
    radii = radii.astype(np.float32)

    return centers, radii, sum_radii
