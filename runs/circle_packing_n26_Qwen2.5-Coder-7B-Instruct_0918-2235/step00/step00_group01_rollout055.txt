import numpy as np
from scipy.optimize import linprog, differential_evolution

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    n = 26
    h = 1 / n
    centers = np.random.rand(n, 2)
    bounds = [(0, 1)] * (2 * n)
    center_shape = centers.shape

    def objective_and_grad(u):
        x = u[:n]
        y = u[n:]
        r = compute_radii(x, y)
        objective = r.sum()
        grad = compute_grad(x, y, r)
        return -objective, -grad

    def compute_radii(x, y):
        pour = min(h * 2 / (n - 1), 1 - h * 2 / (n - 1))
        r = pour * np.ones(n)
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - pour})
            for j in range(i + 1, n):
                dist = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - (r[i] + r[j])})
        res = linprog([], A_ub=[c['fun'](r) for c in constraints], b_ub=[c['rhs'] for c in constraints], bounds=[(0, None)] * n, method='highs')
        r = np.maximum(res.x, pour)
        return r

    def compute_grad(x, y, r):
        _, _, grad = linprog([], A_ub=[c['fun'](r) for c in constraints], b_ub=[c['rhs'] for c in constraints], bounds=[(0, None)] * n, method='highs', callback=None)
        return grad

    sol = differential_evolution(objective_and_grad, bounds, disp=False, popsize=20, maxiter=200, tol=1e-12)
    centers = sol['x'][:n].reshape(center_shape)
    radii = compute_radii(centers[:, 0], centers[:, 1])
    sum_radii = radii.sum()
    return centers, radii, sum_radii
