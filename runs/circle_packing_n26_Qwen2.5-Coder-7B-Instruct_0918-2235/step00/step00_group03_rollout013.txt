import numpy as np
from scipy.optimize import linprog

def shell Necklace Annealing(centers, radii, tolerance=1e-9, alpha=0.8):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    
    def compute_dists(centers):
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = dists[j, i] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

    def compute_bounds(centers):
        bounds = np.zeros((n, 2))
        for i in range(n):
            boundingRadii = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            bounds[i, :] = [0, boundingRadii]
        return bounds

    def compute_objective(radii):
        return -np.sum(radii)

    def solve_lp(centers, bounds):
        n = centers.shape[0]
        dists = compute_dists(centers)
        A_ub = []
        b_ub = []
        
        for i in range(n):
            for j in range(i + 1, n):
                A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [-1] + [0] * (n - j - 1))
                b_ub.append(dists[i, j])

        A_eq = []
        b_eq = []

        for i in range(n):
            A_eq.append([0] * i + [1] + [0] * (n - i - 1))
            b_eq.append(1)

        bounds = compute_bounds(centers)
        
        result = linprog(compute_objective, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if result.success:
            return result.x
        else:
            raise Exception("LP solver failed")

    step_size = 0.1
    best_sum_radii = -np.inf
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    while step_size > 0.0001:
        for _ in range(n):
            i = np.random.randint(n)
            delta = np.random.uniform(-step_size, step_size, 2)
            new_centers = centers.copy()
            new_centers[i] += delta
            new_centers[i] = np.clip(new_centers[i], 0, 1)
            
            current_sum_radii = solve_lp(new_centers, radii)
            if current_sum_radii > best_sum_radii + tolerance:
                centers = new_centers
                best_sum_radii = current_sum_radii
                best_centers = centers.copy()
        
        step_size *= alpha
    
    return best_centers, solve_lp(centers, radii), best_sum_radii
