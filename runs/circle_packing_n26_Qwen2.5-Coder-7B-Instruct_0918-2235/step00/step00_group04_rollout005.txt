import numpy as np
from scipy.optimize import linprog, minimize

def compute_distances(centers):
    n = centers.shape[0]
    distances = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            distances[i, j] = dist
            distances[j, i] = dist
    return distances

def objective_and_gradient(centers):
    n = centers.shape[0]
    x = centers.flatten()
    radii = np.ones(n)  # Initial guess for radii, will be recomputed by LP
    distances = compute_distances(centers)
    
    # Define matrix A and vector b for the LP
    A_ub = np.zeros((4*n + n*n//2, n))
    b_ub = np.ones(4*n + n*n//2)
    
    # Add boundary constraints
    for i in range(n):
        A_ub[i, 2*i] = 1  # x_i constraint
        A_ub[i, 2*i+1] = 0  # y_i constraint
        b_ub[i] = centers[i, 0]

        A_ub[i + n, 2*i] = 0  # x_i constraint
        A_ub[i + n, 2*i+1] = 1  # y_i constraint
        b_ub[i + n] = centers[i, 1]
    
    # Add pair constraints
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[2*n + pair_idx, 2*i] = 1
            A_ub[2*n + pair_idx, 2*i+1] = 1
            A_ub[2*n + pair_idx, 2*j] = -1
            A_ub[2*n + pair_idx, 2*j+1] = -1
            pair_idx += 1
    
    res = linprog(-radii, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.success:
        radii = res.x
        duals = res.dual
    else:
        # Fallback if LP fails, perform a simple downhill step
        # This is not ideal but prevents the optimizer from returning negative radii
        centers += 1e-3 * np.random.randn(n, 2)
        centers = np.clip(centers, 0, 1)
        return objective_and_gradient(centers)
    
    # Compute gradient
    def delta_ij(dx, dy):
        return np.sum((radii + radii[:, None] + dx + dy) * (distances - (radii[:, None] + dx + dy)) ** 0.5, axis=1)
    
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            uij = (centers[i, 0] - centers[j, 0]) / distances[i, j]
            vij = (centers[i, 1] - centers[j, 1]) / distances[i, j]
            grad_x[i] += duals[pair_idx] * uij
            grad_x[j] -= duals[pair_idx] * uij
            grad_y[i] += duals[pair_idx] * vij
            grad_y[j] -= duals[pair_idx] * vij
            pair_idx += 1
    
    for i in range(n):
        grad_x[i] += duals[i] if duals[i] > 0 else 0
        grad_x[i] -= duals[i + n] if duals[i + n] > 0 else 0
        grad_y[i] += duals[i] if duals[i] > 0 else 0
        grad_y[i] -= duals[i + n] if duals[i + n] > 0 else 0
    
    # Combine gradients and negate
    grad_total = np.hstack((grad_x, grad_y))
    return -np.sum(radii), grad_total

def run_packing():
    # Initial deterministic grid
    base_x = np.linspace(0.1, 0.9, 6)
    base_y = np.linspace(0.1, 0.9, 5)
    grid = np.meshgrid(base_x, base_y)
    points = np.vstack((grid[0].ravel(), grid[1].ravel())).T
    centres = points[:26]
    
    # Remove 5 points from corners to start close to optimal packing
    idx = np.argsort(np.sum(centres**2, axis=1))[:26]
    centres = centres[idx]
    
    # Run the optimizer
    result = minimize(objective_and_gradient, centres.ravel(), jac=True, bounds=[(0, 1)] * 52, method='L-BFGS-B', options={'maxiter': 200})
    centres = result.x.reshape((-1, 2))
    radii = np.ones(26)  # Initial guess for radii, will be recomputed by LP
    distances = compute_distances(centres)
    
    A_ub = np.zeros((4*n + n*n//2, n))
    b_ub = np.ones(4*n + n*n//2)
    
    for i in range(n):
        A_ub[i, 2*i] = 1
        A_ub[i, 2*i+1] = 0
        b_ub[i] = centres[i, 0]
        A_ub[i + n, 2*i] = 0
        A_ub[i + n, 2*i+1] = 1
        b_ub[i + n] = centres[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[2*n + pair_idx, 2*i] = 1
            A_ub[2*n + pair_idx, 2*i+1] = 1
            A_ub[2*n + pair_idx, 2*j] = -1
            A_ub[2*n + pair_idx, 2*j+1] = -1
            pair_idx += 1
    
    res = linprog(-radii, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.success:
        radii = res.x
        sum_radii = np.sum(radii)
    else:
        sum_radii = 0
    
    return centres, radii, sum_radii
