import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution
from math import sqrt

N = 26
bounds = [(0, 1)] * (2 * N)

def build_lp(C):
    d pairwise = cdist(C, C)
    lhs = np.vstack((d pairwise - C[:, np.newaxis, 0] - C[np.newaxis, :, 0], C[:, np.newaxis, 1] - C[np.newaxis, :, 1], C[:, np.newaxis, 1] - C[np.newaxis, :, 1], C[:, np.newaxis, 1] - C[np.newaxis, :, 1]))
    rhs = d pairwise
    lhs = np.concatenate((lhs.T, np.eye(N) + np.eye(N, k=1, format='csr'), np.eye(N) - np.eye(N, k=1, format='csr'), np.eye(N) + np.eye(N, k=-1, format='csr'), np.eye(N) - np.eye(N, k=-1, format='csr')), axis=1)
    rhs = np.concatenate((rhs, C[:, 0], 1 - C[:, 0], C[:, 1], 1 - C[:, 1]))
    lhs[:, :N] *= 2
    return lhs, rhs

def compute_sum_radius(C):
    lhs, rhs = build_lp(C)
    res = linprog(-np.ones(N), A_ub=lhs, b_ub=rhs, bounds=[0, 1] * N, method='highs')
    if not res.success:
        raise Exception("Infeasible LP")
    return res.fun

def de_obj(C):
    C = C.reshape(N, 2)
    C = np.clip(C, 0, 1)
    return -compute_sum_radius(C)

def run_packing() -> tuple:
    initial_grid = np.array([(i * sqrt(3), (i + 2 * j) * (3 / 4)) for i in range(18) for j in range(27) if (i + j) % 2 == 0]).T / (3 * sqrt(3)) * 2 / 3
    scale = 0.02
    rand_uniform = np.random.uniform(0, 1, (N * 2, 5))
    C = np.clip((initial_grid + scale * rand_uniform[:, 0:2] + 0.5 * rand_uniform[:, 2:4].min(axis=0)) / 2, 0, 1) * 2 - 1
    popsize = 15 * 2
    result = differential_evolution(de_obj, bounds, popsize=popsize, strategy='best1bin', mutation=0.9, recombination=0.7, maxfev=3000, tol=1e-7)
    best_C = result.x.reshape(N, 2)
    best_r = -de_obj(best_C)
    best_C = np.clip(best_C, 0, 1)
    lhs, rhs = build_lp(best_C)
    res = linprog(-np.ones(N), A_ub=lhs, b_ub=rhs, bounds=[0, 1] * N, method='highs')
    if not res.success:
        raise Exception("Infeasible LP")
    best_r = -res.fun
    try:
        validate_packing(best_C, res.x)
    except ValueError as e:
        raise Exception("Validation error: " + str(e))
    return best_C, res.x, best_r
