import numpy as np
from scipy.optimize import minimize

def init_hex_grid():
    # Create a hexagonal lattice of 6x5 centers
    centers = []
    for x in np.arange(0.1667, 1, 0.3333):
        for y in np.arange(0.1667, 1, 0.3333):
            centers.append([x, y])
        if all((y % 2 for y in np.arange(0.1667, 1, 0.3333))):
            centers.append([x, y + 0.1667 / 2])
    centers = np.array(centers)
    # Randomly drop 4 points to have exactly 26 circles
    drop_indices = np.random.choice(30, 4, replace=False)
    centres = np.delete(centers, drop_indices, axis=0)
    return centres, np.full(26, 0.01)

def build_variable_vector(centres, radii):
    n = centres.shape[0]
    return np.hstack([centres.ravel(), radii])

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    terms = -np.sum(r) + lambda_B * np.sum(np.maximum(0, r - x)**2) + \
                      np.sum(np.maximum(0, r - (1 - x))**2) + \
                      np.sum(np.maximum(0, r - y)**2) + \
                      np.sum(np.maximum(0, r - (1 - y))**2) + \
                      lambda_O * np.sum(np.maximum(0, r[:, None] + r[None, :] - np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2))**2)
    return terms

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    n = x.size
    grad = np.zeros(2 * n)
    grad[0::3] = lambda_B * 2 * np.maximum(0, r - x) * np.where(r > x, 1, 0)
    grad[1::3] = lambda_B * 2 * np.maximum(0, r - (1 - x)) * np.where(r > (1 - x), 1, 0)
    grad[2::3] = lambda_B * 2 * np.maximum(0, r - y) * np.where(r > y, 1, 0)
    grad[2::3] = lambda_B * 2 * np.maximum(0, r - (1 - y)) * np.where(r > (1 - y), 1, 0)
    M = np.zeros((n, n))
    D = GradR(r, x, y)
    grad[2::3] += lambda_O * (M - D.T) @ np.maximum(0, r[:, None] + r[None, :] - np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2))
    grad[2::3] += lambda_O * 2 * np.maximum(0, r[:, None] + r[None, :] - np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2)) * np.where(r[:, None] + r[None, :] > np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2), 1, 0)
    return grad

def GradR(r, x, y):
    n = x.size
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            d = np.sqrt(dx**2 + dy**2)
            D[i, j] = 1. / d * (x[i] - x[j])
            D[j, i] = 1. / d * (x[j] - x[i])
            D[i, j] *= np.sign(r[i] + r[j] - d)
            D[j, i] *= np.sign(r[j] + r[i] - d)
    return D

def run_packing():
    # Initial centres and radii
    centres, radii = init_hex_grid()
    n = centres.shape[0]
    z = build_variable_vector(centres, radii)

    # Objective and gradient functions
    def objective(z):
        x, y, r = unpack(z)
        return compute_penalty_objective(x, y, r, 1e6, 1e6)

    def jacobian(z):
        x, y, r = unpack(z)
        return compute_penalty_gradient(x, y, r, 1e6, 1e6)

    # Bounds
    bounds = [(0, 1)] * (2 * n) + [(0, None)]

    # Optimize with trust-constr
    res = minimize(objective, z, jac=jacobian, method='trust-constr', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 3000, 'disp': False})

    # Extract best found solution
    centres_opt, radii_opt = unpack(res.x)
    sum_radii = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_radii

def unpack(z):
    n = len(z) // 3
    x = z[:n]
    y = z[n:n*2]
    r = z[n*2:]
    return x, y, r
