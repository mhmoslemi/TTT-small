import numpy as np
from scipy.optimize import minimize

def penalty(coords, radii):
    penalty = 0
    n = len(radii)
    for i in range(n):
        x, y = coords[i*2:i*2+2]
        if x - radii[i] < 0:
            penalty += (x - radii[i])**2
        if x + radii[i] > 1:
            penalty += (x + radii[i] - 1)**2
        if y - radii[i] < 0:
            penalty += (y - radii[i])**2
        if y + radii[i] > 1:
            penalty += (y + radii[i] - 1)**2
        for j in range(i + 1, n):
            dx, dy = coords[j*2:j*2+2] - coords[i*2:i*2+2]
            dist = np.hypot(dx, dy)
            if dist - (radii[i] + radii[j]) < 0:
                penalty += (dist - (radii[i] + radii[j]))**2
    return penalty

def objective(vars):
    return -np.sum(vars[26:]) + 1e5 * penalty(vars[:52], vars[52:])

def optimize_for_lambda(init_vars, lambda_val):
    result = minimize(objective, init_vars, method='L-BFGS-B', bounds=[(0, 1)] * 52 + [(0, 0.5)] * 26, jac=True)
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum_radii = 0
    best_vars = None
    for _ in range(50):
        init_coords = np.random.rand(2 * 26)
        init_radii = 1e-4 * np.ones(26)
        init_vars = np.concatenate((init_coords, init_radii))
        result = optimize_for_lambda(init_vars, 1)
        radii = result[52:]
        if np.sum(radii) > best_sum_radii:
            best_vars = result
            best_sum_radii = np.sum(radii)
    centers = best_vars[:52].reshape(26, 2)
    radii = best_vars[52:]
    return centers, radii, best_sum_radii
