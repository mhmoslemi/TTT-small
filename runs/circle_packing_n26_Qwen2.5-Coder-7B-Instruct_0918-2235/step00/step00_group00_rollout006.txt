import numpy as np
import cvxpy as cp

def compute_r_max(candidates, placed_centres, placed_radii):
    n_candidates = len(candidates)
    n_circles = len(placed_centres)
    dists = np.linalg.norm(candidates[:, np.newaxis, :] - placed_centres[np.newaxis, :, :], axis=2)
    min_dists = np.min(dists, axis=1) - placed_radii
    r_max = np.minimum(min_dists, np.minimum(1 - candidates[:, 0], candidates[:, 0]))
    r_max = np.minimum(r_max, np.minimum(1 - candidates[:, 1], candidates[:, 1]))
    r_max = np.clip(r_max, 0.0, None)
    return r_max

def greedy_insertion(candidates, n):
    placed_centres = []
    placed_radii = []

    while len(placed_centres) < n:
        r_candidates = compute_r_max(candidates, placed_centres, placed_radii)
        feasible_idx = np.where(r_candidates > 0)[0]
        
        if feasible_idx.size == 0:
            break

        best = feasible_idx[np.argmax(r_candidates[feasible_idx])]
        placed_centres.append(candidates[best])
        placed_radii.append(r_candidates[best])
        
        dists = np.linalg.norm(candidates - candidates[best], axis=1)
        too_close = dists < (r_candidates[best] + 1e-9)
        candidates = candidates[~too_close]

    return np.array(placed_centres), np.array(placed_radii)
