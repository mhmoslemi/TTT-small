import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def init_hex_grid():
    n = 6
    m = 5
    centers = np.zeros((26, 2))
    radii = np.zeros(26) + 0.01
    counter = 0
    for i in range(n):
        for j in range(m):
            if (i + j) % 2 == 0:
                x = (2 * j + 1) * 0.5 / n
                y = (i * 0.5 + 0.25 * (j % 2)) / m
                for _ in range(np.random.randint(2, 5)):
                    if counter >= 26:
                        break
                    centers[counter, 0] = x
                    centers[counter, 1] = y
                    counter += 1
    return centers, radii

def build_variable_vector(centers, radii):
    return np.concatenate((centers.flatten(), radii))

def unpack(z):
    n = 26
    x = z[:52].reshape(26, 2)
    r = z[52:]
    return x, y, r

def compute_penalty_objective(x, y, r, lambda_B, lambda_O):
    penalty_B = np.sum(np.maximum(0, r - x) ** 2) + np.sum(np.maximum(0, r - (1 - x)) ** 2) + np.sum(np.maximum(0, r - y) ** 2) + np.sum(np.maximum(0, r - (1 - y)) ** 2)
    D = distance_matrix(x, x)
    D[np.diag_indices_from(D)] = np.inf
    penalty_O = np.sum(np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - D) ** 2)
    return -np.sum(r) + lambda_B * penalty_B + lambda_O * penalty_O

def compute_penalty_gradient(x, y, r, lambda_B, lambda_O):
    n = 26
    grad = np.zeros(78)
    D = distance_matrix(x, x)
    penalties = np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - D)
    for i in range(n):
        mask_B = (r[i] > x[i]) & (r[i] > y[i]) & (r[i] > 1 - x[i]) & (r[i] > 1 - y[i])
        grad[i] = -2 * np.sum(np.maximum(0, r[i] - x[i]) * np.maximum(0, r[i] - y[i]) * np.maximum(0, r[i] - (1 - x[i])) * np.maximum(0, r[i] - (1 - y[i])) * (1 - mask_B))
        mask_O = penalties[i, :] > 0
        mask_O[i] = False
        grad[52 + i] = np.sum(np.maximum(0, penalties[i, mask_O]) * x[mask_O, 0] / D[i, mask_O] * (1 / r[mask_O]) - np.maximum(0, penalties[i, mask_O]) * x[mask_O, 1] / D[i, mask_O] * (1 / r[mask_O]))
    for j in range(1, n):
        for i in range(j):
            mask = penalties[j, i] > 0
            grad[j] += np.sum(np.maximum(0, penalties[j, i]) * y[i] / D[i, j] * (1 / r[i]) - np.maximum(0, penalties[j, i]) * y[j] / D[i, j] * (1 / r[j]))
            grad[52 + j] += np.sum(np.maximum(0, penalties[j, i]) * x[j] / D[i, j] * (1 / r[i]) - np.maximum(0, penalties[j, i]) * x[i] / D[i, j] * (1 / r[j]))
    return grad

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    centers, radii = init_hex_grid()
    z = build_variable_vector(centers, radii)
    bounds = [(0, 1)] * 52 + [(0, None)] * 26
    lambda_B, lambda_O = 1e6, 1e6
    res = minimize(compute_penalty_objective, z, jac=compute_penalty_gradient, method='trust-constr', bounds=bounds, options={'gtol': 1e-6, 'ftol': 1e-8, 'maxiter': 3000, 'verbose': 0})
    centres_opt, radii_opt = unpack(res.x)
    sum_r = np.sum(radii_opt)
    return centres_opt, radii_opt, sum_r
