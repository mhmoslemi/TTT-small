def compute_optimal_radii(centers):

    n=centers.shape[0]

    # distances

    d = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)

    # constraints: For each pair i<j: r_i + r_j <= d_ij

    # Build arrays A, b

    pair_indices = [(i,j) for i in range(n) for j in range(i+1,n)]

    num_pairs = len(pair_indices)

    m = num_pairs + 4*n

    A = np.zeros((m, n), dtype=np.float64)

    b = np.zeros(m, dtype=np.float64)

    row = 0

    for (i,j) in pair_indices:

        A[row,i] = 1.0

        A[row,j] = 1.0

        b[row] = d[i,j]

        row += 1

    for i in range(n):

        x, y = centers[i]

        mins = [x, 1 - x, y, 1 - y]

        b[row] = min(mins)

        A[row,i] = 1.0

        row += 1

    #bounds: (0, None)

    bounds = [(0, None)] * n

    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if res.success:

        radii = res.x

    else:

        # fallback - compute approximate radii as min distances to others/2 and boundaries

        radii = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]), np.minimum(centers[:,1], 1-centers[:,1]))

        for i in range(n):

            min_d = np.inf

            for j in range(n):

                if i!=j:

                    min_d = min(min_d, np.linalg.norm(centers[i]-centers[j]))

            radii[i] = min(radii[i], min_d/2 if np.isfinite(min_d) else 0)

        # maybe negative? ensure >=0

        radii = np.maximum(radii,0)

    return radii
