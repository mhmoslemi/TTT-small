import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for circle centers and radii
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.1

    # Define the objective function to maximize the sum of radii
    def objective(x):
        radii = x[:26]
        centers = x[26:].reshape(-1, 2)
        return -np.sum(radii)  # Negative because we will use a minimizer

    # Define the constraints
    constraints = []

    # Each circle must have a non-negative radius
    constraints.append({'type': 'ineq', 'fun': lambda x: radii})

    # Each circle must be within the unit square
    constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x[26:][:, 0]})
    constraints.append({'type': 'ineq', 'fun': lambda x: x[26:][:, 0]})
    constraints.append({'type': 'ineq', 'fun': lambda x: 1 - x[26:][:, 1]})
    constraints.append({'type': 'ineq', 'fun': lambda x: x[26:][:, 1]})

    # Distance between circles must be greater than the sum of their radii
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt(np.sum((x[26:][i] - x[26:][j]) ** 2)) - (x[:26][i] + x[:26][j])})

    # Initial guess for the combined variables
    x0 = np.concatenate((radii, centers.flatten()))

    # Minimize the negative sum of radii
    result = minimize(objective, x0, method='SLSQP', constraints=constraints)

    centers = result.x[26:].reshape(-1, 2)
    radii = result.x[:26]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
