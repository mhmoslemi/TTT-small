import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Generate initial hexagonal lattice
    n = 26
    s = 0.312
    points = []
    k_indices = np.arange(n ** 0.5 + 1)
    for kx in k_indices:
        for ky in k_indices:
            x = s + kx * s
            y = s + ky * np.sqrt(3) / 2 * s
            if abs(x - 0.5) >= 0.4 or abs(y - 0.5) >= 0.4:
                continue
            points.append((x, y))
    points = np.array(points[:n])

    # Step 2: Perform simulated annealing
    T0 = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = 1e-3
    T = T0
    best_sum = -np.inf
    
    while T > T_min:
        for _ in range(M):
            # Pick a random circle and move its center
            i = np.random.randint(n)
            delta = np.random.normal(0, sigma_move, 2)
            c_new = np.clip(points[i] + delta, 1e-12, 1 - 1e-12)
            points[i] = c_new
            
            # Compute pairwise distances and solve the LP
            d_ij = squareform(pdist(points))
            A = d_ij - np.eye(n)
            b = np.minimum(points[:, 0], 1 - points[:, 0]) + np.minimum(points[:, 1], 1 - points[:, 1])
            A[-n:] = np.eye(n)
            b[-n:] = np.zeros(n)
            res = linprog(-np.ones(n), A_ub=A, b_ub=b, method='highs')
            
            if res.success and res.fun >= best_sum:
                best_sum = res.fun
                best_centers = points.copy()
                sigma_move = T / 5

        T *= c

    # Step 3: Perform local refinement (optional)
    # Note: Uncommenting this step will significantly increase execution time
    # for i in range(n):
    #     for _ in range(30):
    #         if np.any(best_centers == points[i]):
    #             best_centers[i] += np.random.normal(0, sigma_move, 2)
    #             best_centers[i] = np.clip(best_centers[i], 1e-12, 1 - 1e-12)
    #             continue
    #             res = linprog(-np.ones(n), A_ub=A, b_ub=b, method='highs')
    #             if res.success and res.fun >= best_sum:
    #                 best_sum = res.fun
    #                 best_centers = points.copy()

    # Extract radii from optimal centre layout
    d_ij = squareform(pdist(best_centers))
    A = d_ij - np.eye(n)
    b = np.minimum(best_centers[:, 0], 1 - best_centers[:, 0]) + np.minimum(best_centers[:, 1], 1 - best_centers[:, 1])
    A[-n:] = np.eye(n)
    b[-n:] = np.zeros(n)
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, method='highs')
    radii = -res.x

    # Return the final packing
    return best_centers, radii, best_sum.item()
