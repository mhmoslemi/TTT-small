def objective(x):

    xs = x[:2*N:2]
    ys = x[1:2*N:2]
    rs = x[2*N:]

    p_boundary = (
        np.maximum(0, rs - xs)**2 +
        np.maximum(0, xs + rs - 1)**2 +
        np.maximum(0, rs - ys)**2 +
        np.maximum(0, ys + rs - 1)**2
    ).sum()
    
    # Overlap penalties
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    overlap = rr - dist
    p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
    
    return sum_radii - 1e5 * (p_boundary + p_overlap)
