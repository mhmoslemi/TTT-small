import numpy as np
from scipy.optimize import linprog

def run_packing():
    n = 26
    
    # Initial positions using a fixed heuristic pattern
    centres = np.array([
        [0.05, 0.10], [0.15, 0.10], [0.25, 0.10], [0.35, 0.10], [0.45, 0.10], [0.55, 0.10],
        [0.10, 0.18], [0.20, 0.18], [0.30, 0.18], [0.40, 0.18], [0.50, 0.18],
        [0.15, 0.26], [0.25, 0.26], [0.35, 0.26], [0.45, 0.26], [0.55, 0.26],
        [0.20, 0.34], [0.30, 0.34], [0.40, 0.34], [0.50, 0.34],
        [0.25, 0.42], [0.35, 0.42], [0.45, 0.42],
        [0.30, 0.50]
    ])
    
    best_obj = -np.inf
    best_centres = None
    best_radii = None
    
    for _ in range(5):
        centres += np.random.uniform(-0.002, 0.002, size=centres.shape)  # small jitter
        for _ in range(200):
            dists = np.sqrt(np.sum(np.square(centres[:, np.newaxis] - centres[np.newaxis, :]), axis=2))
            np.fill_diagonal(dists, np.inf)
            
            A = np.zeros((4*n + n*(n-1)//2, n))
            b = np.zeros(4*n + n*(n-1)//2)
            
            idx = 0
            for i in range(n):
                A[idx, i] = 1
                b[idx] = centres[i, 0]
                idx += 1
                A[idx, i] = 1
                b[idx] = 1 - centres[i, 0]
                idx += 1
                A[idx, i] = 1
                b[idx] = centres[i, 1]
                idx += 1
                A[idx, i] = 1
                b[idx] = 1 - centres[i, 1]
                idx += 1
                for j in range(i+1, n):
                    A[idx, i] = 1
                    A[idx, j] = 1
                    b[idx] = dists[i, j]
                    idx += 1
            
            c = -np.ones(n)
            res = linprog(c, A_ub=A, b_ub=b)
            if res.success and res.fun > best_obj:
                best_obj = res.fun
                best_centres = centres.copy()
                best_radii = res.x.copy()
    
    # Shrink radii to avoid numerical issues
    eps = 1e-7
    best_radii = np.maximum(0.0, best_radii - eps)
    
    return best_centres, best_radii, np.sum(best_radii)
