import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalty_boundary(rs, xs, ys):
    p_boundary = (
        np.maximum(0, rs - xs)**2 +
        np.maximum(0, xs + rs - 1)**2 +
        np.maximum(0, rs - ys)**2 +
        np.maximum(0, ys + rs - 1)**2
    ).sum()
    return p_boundary

def penalty_overlap(rs, xs, ys):
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    overlap = rr - dist
    p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
    return p_overlap

def objective(x):
    xs = x[:N]
    ys = x[N:2*N]
    rs = x[2*N:]
    sum_radii = -rs.sum()
    penalty = lambda_penalty * (penalty_boundary(rs, xs, ys) + penalty_overlap(rs, xs, ys))
    return sum_radii - penalty

DE_popsize = 12
DE_maxiter = 400

bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N

    np.random.seed(seed)
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=DE_popsize, maxiter=DE_maxiter, polish=False)
    if best_result is None or result.fun < best_result.fun:
        best_result = result
