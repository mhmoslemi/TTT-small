import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    n = 26
    bounds = [(0, 1)] * n * 2 + [(0, 0.5)] * n
    
    def penalised_objective(z):
        r = z[52:]
        sum_r = -np.sum(r)
        overlap_penalty = 0.0
        for i in range(n):
            x_i, y_i = z[2*i], z[2*i + 1]
            radius_i = z[52 + i]
            overlap_penalty += max(0, 0.1 - x_i - radius_i) + max(0, x_i + radius_i - 1)
            overlap_penalty += max(0, 0.1 - y_i - radius_i) + max(0, y_i + radius_i - 1)
            for j in range(i + 1, n):
                x_j, y_j = z[2*j], z[2*j + 1]
                radius_j = z[52 + j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                overlap_penalty += max(0, radius_i + radius_j - dist)
        return sum_r + 10000 * overlap_penalty
    
    def local_constraints(z):
        c = []
        for i in range(n):
            x_i, y_i = z[2*i], z[2*i + 1]
            c.append(0.1 - x_i - z[52 + i])
            c.append(x_i + z[52 + i] - 1)
            c.append(0.1 - y_i - z[52 + i])
            c.append(y_i + z[52 + i] - 1)
            for j in range(i + 1, n):
                x_j, y_j = z[2*j], z[2*j + 1]
                radius_j = z[52 + j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                c.append(radius_i + radius_j - dist)
        return np.array(c)
    
    def local_refinement(z):
        result = minimize(penalised_objective, z, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': local_constraints}, options={'disp': False})
        return result.x
    
    result = differential_evolution(penalised_objective, bounds, popsize=20, strategy='rand/1/bin', mutation=0.6, recombination=0.7, maxiter=2000, tol=1e-8)
    
    if result.fun < 1e-8:
        z_opt = result.x
    else:
        z_opt = local_refinement(result.x)
    
    centers = z_opt[:52].reshape((26, 2))
    radii = z_opt[52:]
