import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    
    n = 26
    pair_indices = [(i,j) for i in range(n) for j in range(i+1,n)]
    
    radii_initial = np.full(n, 0.04)
    
    def constraint(v):
        x = v[:52].reshape((n,2))
        r = v[52:]
        arr = []
        arr.append(x[:,0] - r)  # x boundaries
        arr.append(1 - x[:,0] - r)
        arr.append(x[:,1] - r)  # y boundaries
        arr.append(1 - x[:,1] - r)
        dx = x[i_indices,0] - x[j_indices,0]
        dy = x[i_indices,1] - x[j_indices,1]
        dist2 = dx**2 + dy**2
        arr.append(dist2 - (r[i_indices] + r[j_indices])**2)
        return np.concatenate(arr)

    x0 = np.empty(78)
    x0[:52] = ((np.arange(n)%5 + 0.5)/5, (np.arange(n)//5 + 0.5)/5).ravel()
    x0[52:] = radii_initial
    
    bonds = [(0.0,1.0) for _ in range(52)] + [(0.0,0.5) for _ in range(26)]

    cons = NonlinearConstraint(constraint, 0, np.inf) # this will enforce the constraints

    result = minimize(lambda v: -np.sum(v[52:]), x0, method='trust-constr', bounds=bonds,
                      constraints=cons, options={'maxiter':5000,'gtol':1e-12,'disp':False})
    
    if result.success:
        centers = result.x[:52].reshape((n,2))
        radii = result.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise Exception("Optimization failed")
