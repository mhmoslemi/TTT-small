import numpy as np
from scipy.optimize import linprog, minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    eps = 1e-12
    Δt = 1e-3
    α = 4.0
    β = 0.8
    maxIter = 3000
    tol = 1e-10
    δr = 1e-4

    centers = np.random.rand(n, 2) * (1 - 2 * eps) + eps
    radii = np.zeros(n) + 1e-6
    velocities = np.zeros((n, 2))
    F = np.zeros((n, 2))

    for step in range(maxIter):
        F.fill(0)
        for i in range(n):
            for j in range(i + 1, n):
                delta = centers[i] - centers[j]
                dist = np.linalg.norm(delta, ord=2)
                if dist < radii[i] + radii[j] + eps:
                    overlap = radii[i] + radii[j] - dist
                    F[i] += α * overlap * delta / dist
                    F[j] -= α * overlap * delta / dist

            if centers[i][0] < radii[i] + eps:
                F[i][0] += α * (radii[i] + eps - centers[i][0])
            if centers[i][0] > 1 - radii[i] - eps:
                F[i][0] -= α * (centers[i][0] - (1 - radii[i] - eps))
            if centers[i][1] < radii[i] + eps:
                F[i][1] += α * (radii[i] + eps - centers[i][1])
            if centers[i][1] > 1 - radii[i] - eps:
                F[i][1] -= α * (centers[i][1] - (1 - radii[i] - eps))

        velocities += Δt * (F / 1 - β * velocities)
        centers += Δt * velocities

        radii += δr
        radii = np.clip(radii, eps, None)
        for i in range(n):
            delta = centers[i] - centers[:i]
            dists = np.linalg.norm(delta, ord=2, axis=1)
            new_r = np.min(np.clip(1 - dists, eps, None))
            radii[i] = new_r * (1 + eps)

        if np.all(np.abs(radii - np.mean(radii)) < tol):
            break

    status, message = validate_packing(centers, radii)

    if not status:
        raise Exception(message)

    return centers, radii, np.sum(radii)
