import numpy as np
from scipy.optimize import minimize

def hex_lattice(s, n):
    """Generate hexagonal lattice points inside the unit square."""
    rows = int(np.ceil(np.sqrt(n * 3 / (4 * np.sqrt(3))))) + 1
    l = np.linspace(s, 1 - s, rows)
    row_indices = np.arange(rows)
    row_indices = np.concatenate([row_indices[1::2], row_indices[0::2]])
    l = l[row_indices]
    grid_x, grid_y = np.meshgrid(l, l, indexing='ij')
    x = grid_x.flatten()
    y = grid_y.flatten()
    points = np.vstack([x, y]).T
    in_square = np.all(points < 1 - s, axis=1) & np.all(points > s, axis=1)
    points = points[in_square]
    return points[:n]

def penalised_obj(z, n, ε, δ, s):
    """Penalised objective function for optimization."""
    x = z[:n].reshape(-1, 2)
    r = z[n:].reshape(-1,)
    dist_sq = np.sum((x[:, np.newaxis, :] - x[np.newaxis, :, :]) ** 2, axis=-1)
    dist_sq = dist_sq - r[:, np.newaxis] ** 2 - r[np.newaxis, :] ** 2 - 2 * ε
    dist_sq[range(n), range(n)] = np.inf
    penalty = np.sum(np.max(np.maximum(dist_sq, 0) - δ, axis=1) * 1e6)
    obj = -np.sum(r)
    return obj + penalty

def solve_stage(z0, δ, ε, s):
    """Solve the optimization stage."""
    n = len(z0) // 2
    result = minimize(penalised_obj, z0, args=(n, ε, δ, s), method='SLSQP',
                      bounds=[(0 + ε, 1 - ε)] * 2 * n + [(0 + ε, None)] * n,
                      tol=1e-10)
    return result.x

def inflate_balls(z, n, ε, s):
    """Inflate balls to maximum possible radius without overlap."""
    x = z[:n].reshape(-1, 2)
    r = z[n:].reshape(-1,)
    for i in range(n):
        dist = np.sqrt(np.sum((x - x[i]) ** 2, axis=1))
        valid_dist = dist > ε
        new_r = np.minimum(dist - ε, 1 - x[i, 0] if i % 2 == 0 else 1 - x[i, 0], 
                            1 - x[i, 1] if i % 2 == 1 else 1 - x[i, 1])
        r[i] = np.minimum(r[i], new_r)
        if np.any(r[i] < b_r and dist < b_r + r[i]) for b_r, b_x in zip(r, x)):
            r[i] += (b_r - r[i]) / (b_r + r[i])
        x[i] = np.minimum(np.maximum(x[i], ε, 1 - ε), 1 - ε)
    return np.concatenate([x.flatten(), r])

def run_packing():
    """Run the packing algorithm."""
    n = 26
    s = 0.05  # Lattice spacing
    z0 = np.concatenate([hex_lattice(s, 2 * n), np.ones(n) * s / 2])
    z = z0.copy()
    ε = 1e-8
    δ = 5e-4
    z = solve_stage(z, δ, ε, s)
    for i in range(2):
        δ /= 2
        z = solve_stage(z, δ, 1e-12, s)
        z = inflate_balls(z, n, 1e-12, s)
    z = solve_stage(z, δ, 1e-12, s)
    for _ in range(5):
        z = z0.copy() + (np.random.rand(3 * n) * 2 - 1) * 1e-3
        z = solve_stage(z, δ, 1e-12, s)
    centers = z[:n].reshape(-1, 2)
    radii = z[n:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
