import numpy as np

def init_circles(variant):
    radii = np.full(26, 0.04)
    centers = np.zeros((26, 2), dtype=float)
    for i in range(5):
        for j in range(5):
            centers[i*5 + j, 0] = 0.2 * (i + 0.5)
            centers[i*5 + j, 1] = 0.2 * (j + 0.5)
    if variant == 'A':
        centers[25, 0] = 0.6
        centers[25, 1] = 0.9
    else:
        centers[25, 0] = 0.4
        centers[25, 1] = 0.9
    return centers, radii

def inflate(circles, Δr, ε):
    max_iter = 5000
    buffer = 1e-6
    for _ in range(max_iter):
        inflation_possible = False
        for i in range(len(circles)):
            for j in range(i + 1, len(circles)):
                dx, dy = centers[i] - centers[j]
                d = np.sqrt(dx**2 + dy**2)
                if d < radii[i] + radii[j] - ε:
                    overlap = radii[i] + radii[j] - d
                    delta = overlap / (2 * d)
                    centers[i] += delta * dx
                    centers[j] -= delta * dx
                    inflation_possible = True
        centers = np.clip(centers, buffer, 1 - buffer)
        radii += Δr
        radii = np.clip(radii, buffer, 1 - buffer)
        if not inflation_possible:
            break
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # Fixed seed for reproducibility
    centers_A, radii_A = init_circles('A')
    centers_B, radii_B = init_circles('B')
    
    centers_A, radii_A = inflate(centers_A, 0.0005, 1e-6)
    centers_B, radii_B = inflate(centers_B, 0.0005, 1e-6)
    
    sum_radii_A = np.sum(radii_A)
    sum_radii_B = np.sum(radii_B)
    
    if sum_radii_A > sum_radii_B:
        return centers_A, radii_A, sum_radii_A
    else:
        return centers_B, radii_B, sum_radii_B
