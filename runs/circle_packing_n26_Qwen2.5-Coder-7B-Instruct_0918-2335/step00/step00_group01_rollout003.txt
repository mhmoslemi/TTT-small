import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog


def generate_initial_centers(n_circles):
    centers = []
    y_coords = np.linspace(0, 1, 6)
    for y, j in zip(y_coords, range(len(y_coords))):
        if j == 0 or j == len(y_coords) - 1:
            x_coords = np.linspace(0, 1, 4)[1:-1]
        elif j % 2 == 1:
            x_coords = np.linspace(0, 1, 5)[1:-1]
        else:
            x_coords = np.linspace(0, 1, 6)[1:-1]
        for x in x_coords:
            centers.append(np.array([x, y]))
    return np.array(centers)[:n_circles]


def solve_lp(centers):
    n = centers.shape[0]
    A_ub = []
    b_ub = []
    for i in range(n):
        dists = distance_matrix(centers[[i]], centers[:, None])[i, 1:]
        A_ub.append([1] + [0] * i + list(dists * 2))
        b_ub.append(2 * (1 - min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]))) - dists.sum())
        for j in range(i + 1, n):
            A_ub.append([0] * i + [1] + [0] * j + [1] + list(2 * dist_matrix(centers[[i, j]], centers.shape[0]))))
            b_ub.append(2 * (2 * np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (centers[i] + centers[j]).min()) - 2 * centers[i, 0] - 2 * centers[j, 0])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
    if not res.success:
        res.x = np.zeros(n)
    return res.x, -res.fun


def update_positions(centers, radii, forces, step_size, centers_boundaries):
    n = centers.shape[0]
    for i in range(n):
        centers[i] += step_size * forces[i]
        centers[i, 0] = max(0, min(1, centers[i, 0]))
        centers[i, 1] = max(0, min(1, centers[i, 1]))
        if np.any(centers[i] == centers_boundaries[i]):
            # If moved to boundary, adjust by a small amount
            centers[i] += 1e-7 * (np.random.rand(2) * 2 - 1)
    return centers


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    centers = generate_initial_centers(n_circles)
    best_centers = centers.copy()
    best_radii = np.zeros(n_circles)
    best_obj = -1.0
    step_size = 0.05
    tolerance = 1e-6
    max_iterations = 200
    max_restarts = 5
    eps = 1e-7

    for _ in range(max_restarts):
        centers = generate_initial_centers(n_circles)
        if np.any((np.abs(centers - np.array([[0, 1], [1, 0], [1, 1], [0, 1]])) < 1e-7).sum(axis=1)):
            continue

        for iter in range(max_iterations):
            radii, obj = solve_lp(centers)
            if obj > best_obj:
                best_obj = obj
                best_centers = centers.copy()
                best_radii = radii.copy()

            forces = np.zeros_like(centers)
            # Pairwise contact forces
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dists = distance_matrix(centers[[i, j]], centers[:, None])
                    dist_i_j = 2 * dists[0, 1]
                    if dist_i_j > 0:
                        forces[i] += radii[i] + radii[j] - dist_i_j

            # Boundary forces
            centers_boundaries = np.clip(centers, 0, 1)
            for i in range(n_circles):
                # Check each boundary
                for j in range(4):
                    if centers[i, j // 2] <= centers_boundaries[i, j // 2]:
                        forces[i, j // 2] += 1

            centers = update_positions(centers, radii, forces, step_size, centers_boundaries)

            if np.abs(obj - best_obj) < tolerance:
                break

        # Optionally add small random jitter to centers
        centers += np.random.rand(*centers.shape) * 1e-7 * (np.random.rand(*centers.shape) * 2 - 1)

    # Shrink radii slightly
    best_radii = np.maximum(0.0, best_radii - eps)
    return best_centers, best_radii, np.sum(best_radii)
