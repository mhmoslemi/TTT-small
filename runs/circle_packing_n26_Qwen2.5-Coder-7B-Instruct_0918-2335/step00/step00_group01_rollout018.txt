```python
import numpy as np
from scipy.optimize import minimize

n = 26
RESTARTS = 8
eps = 1e-7

pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.zeros(n)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[2*n:])

def constraints(v):
    centers = v[:2*n].reshape((n,2))
    radii = v[2*n:]
    boundary_constraints = np.maximum(0, np.concatenate((
        centers[:, 0] - radii,
        1 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1 - centers[:, 1] - radii
    )))
    pairwise_constraints = np.maximum(0, np.sum((centers[pair_idx_i] - centers[pair_idx_j]) ** 2, axis=1) 
                                       - (radii[pair_idx_i] + radii[pair_idx_j]) ** 2)
    return np.concatenate((boundary_constraints, pairwise_constraints))

bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n

best_value = -np.inf
best_v = None
for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun        = objective,
        x0         = v0,
        method     = 'SLSQP',
        constraints= {'type':'ineq', 'fun':constraints},
        bounds     = bounds,
        options    = {
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if not res.success:
        continue
    value = -res.fun  # because we minimised the negative
    if value > best_value:
        best_value = value
        best_v = res.x

centers = best_v[:2*n].reshape((n,2))
radii = np.maximum(0.0, best_v[2*n:] - eps)

return centers, radii, np.sum(radii)
```
