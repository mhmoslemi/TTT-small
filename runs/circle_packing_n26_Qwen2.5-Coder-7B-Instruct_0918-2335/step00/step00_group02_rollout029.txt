import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_guess = np.random.rand(n, 3)  # Centers and radii

    def objective(x):
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        return -np.sum(radii)  # Minimize negative sum of radii for maximization

    def constraints(x):
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        cons = []
        for i in range(n):
            x_i, y_i, r_i = centers[i]
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x_i - r_i})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x_i + r_i})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - y_i - r_i})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: y_i + r_i})
            for j in range(i + 1, n):
                r_j = radii[j]
                x_j, y_j = centers[j]
                dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist - r_i - r_j})
        return cons

    options = {'disp': False, 'maxiter': 10000}
    result = minimize(objective, initial_guess, method='SLSQP', constraints=constraints, options=options)

    centers = result.x[:2*n].reshape((n, 2))
    radii = result.x[2*n:]
    sum_radii = -result.fun

    return centers, radii, sum_radii
