import numpy as np
from scipy.optimize import minimize

def hex_lattice(s):
    n = 26
    x = np.zeros(n)
    y = np.zeros(n)
    j = 0
    for i in range(n):
        if (i % 3) in [0, 2]:
            x[i] = (i // 3 + (i % 3) / 2) * s
            y[i] = (i // 6 + (i % 3) / 6) * s * np.sqrt(3)
            j += 1
        else:
            x[i] = (i // 3 + 0.5) * s
            y[i] = (i // 6 + 1.5 * (i % 3) / 6) * s * np.sqrt(3)
    return x[:j], y[:j]

def penalty(z, n):
    K = 1e6
    pen = 0
    for i in range(n):
        for j in range(i+1, n):
            dx = z[i*n:i*n+2] - z[j*n:j*n+2]
            dist = np.sum(dx**2)**0.5
            pen += K * max(0, - (z[i*n+2] + z[j*n+2] - dist))
    return pen

def objective(z, n, epsilon):
    return -np.sum(z[n**2:n**2+26])

def solve_stage(z0, delta):
    n = int((len(z0) - 26) / 2)
    cons = [
        {'type': 'ineq', 'fun': lambda z: 1 - z[i*n+2] for i in range(n)},
        {'type': 'ineq', 'fun': lambda z: z[i*n+2] for i in range(n)},
        {'type': 'ineq', 'fun': lambda z: 1 - z[i*n] for i in range(n)},
        {'type': 'ineq', 'fun': lambda z: z[i*n] for i in range(n)}
    ]
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda*z, i=i, j=j: delta - np.sum((z[i*n:i*n+2] - z[j*n:j*n+2])**2)**0.5})
    res = minimize(objective, z0, args=(n, epsilon), method='SLSQP', constraints=cons)
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    s = 0.1
    epsilon = 1e-8
    z0, _ = hex_lattice(s)
    z0, _ = hex_lattice(s)
    sum_radii = 0
    for delta in [1e-8, 5e-4, 1e-12]:
        z0 = solve_stage(z0, delta)
    s_new = 0.05
    x_new, y_new = hex_lattice(s_new)
    z0 = np.concatenate((x_new, y_new, s_new * np.ones(26)))
    z0 = solve_stage(z0, 1e-12)
    o = objective(z0, 26, 0)
    sum_radii += -o
    centers = z0[:26] + np.repeat([z0[26]], (26,2), axis=0)
    radii = z0[26:] + np.repeat([z0[26]], (26,), axis=0)
    return centers, radii, sum_radii
