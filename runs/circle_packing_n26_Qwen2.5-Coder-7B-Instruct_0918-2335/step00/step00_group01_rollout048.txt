```python
import numpy as np
from scipy.optimize import minimize

n = 26
pair_idx = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
bounds = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n
eps = 1e-7

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.zeros(n)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[n * 2:])

def constraints(v):
    centers = v[:n * 2].reshape((n, 2))
    radii = v[n * 2:]
    boundary_constraints = np.maximum(centers[:, 0] - radii, 0) + np.maximum(1 - centers[:, 0] - radii, 0) + \
                            np.maximum(centers[:, 1] - radii, 0) + np.maximum(1 - centers[:, 1] - radii, 0)
    pairwise_constraints = np.diagonal(np.corrcoef(centers[:, 0]), k=0)[n:] + \
                           np.diagonal(np.corrcoef(centers[:, 1]), k=0)[n:] - radii - radii[pair_idx[:, 0]] - radii[pair_idx[:, 1]]
    return np.concatenate((boundary_constraints, pairwise_constraints))

def run_packing():
    RESTARTS = 10
    best_value = -np.inf
    best_v = None
    for seed in range(RESTARTS):
        v0 = initial_guess(seed)
        res = minimize(
            fun=objective,
            x0=v0,
            method='SLSQP',
            constraints={'type': 'ineq', 'fun': constraints},
            bounds=bounds,
            options={
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if not res.success:
            continue
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x
    centers = best_v[:n * 2].reshape((n, 2))
    radii = np.maximum(0.0, best_v[n * 2:] - eps)
    return centers, radii, np.sum(radii)
```
