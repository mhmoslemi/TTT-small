import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Corner circles
    centers[0] = [r_c, r_c]
    radii[0] = r_c
    centers[1] = [1 - r_c, r_c]
    radii[1] = r_c
    centers[2] = [r_c, 1 - r_c]
    radii[2] = r_c
    centers[3] = [1 - r_c, 1 - r_c]
    radii[3] = r_c
    
    # Edge circles
    centers[4] = [0.5, r_e]
    radii[4] = r_e
    centers[5] = [r_e, 0.5]
    radii[5] = r_e
    centers[6] = [0.5, 1 - r_e]
    radii[6] = r_e
    centers[7] = [1 - r_e, 0.5]
    radii[7] = r_e
    
    # Inner hexagonal core
    hex_center_positions = np.array([
        [0.5, 0.5],
        [1 - 0.5 * np.sqrt(3), 0.5],
        [0.5 * np.sqrt(3), 0.5],
        [0.5, 0.5 + 0.866],
        [1 - 0.5 * np.sqrt(3), 0.5 + 0.866],
        [0.5 * np.sqrt(3), 0.5 + 0.866],
        [0.5, 0.5 + 2 * 0.866],
        [1 - 0.5 * np.sqrt(3), 0.5 + 2 * 0.866],
        [0.5 * np.sqrt(3), 0.5 + 2 * 0.866],
        [0.5, 0.5 + 3 * 0.866],
        [1 - 0.5 * np.sqrt(3), 0.5 + 3 * 0.866],
        [0.5 * np.sqrt(3), 0.5 + 3 * 0.866]
    ])
    for i, pos in enumerate(hex_center_positions):
        if np.all(pos[0] >= r_i) and np.all(pos[1] >= r_i) and np.all(pos[0] <= 1 - r_i) and np.all(pos[1] <= 1 - r_i):
            centers[8 + i] = pos
            radii[8 + i] = r_i
    
    return centers, radii

def compute_sum_radii(radii):
    return radii.sum()

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def constraint_func(v):
    x = v[:52].reshape(26,2)
    r = v[52:]
    n = x.shape[0]
    cons = []
    for i in range(n):
        cons.append(r[i]**2 - min(x[i,0], 1-x[i,0], x[i,1], 1-x[i,1])**2)
        for j in range(i+1, n):
            dist = np.linalg.norm(x[i] - x[j])
            cons.append(r[i] + r[j] - dist + 1e-12)
    return np.array(cons)

def run_packing():
    bounds = [(0.30, 0.49), (0.10, 0.25), (0.05, 0.15)]
    x0 = np.array([0.45, 0.20, 0.10])
    result_de = differential_evolution(objective, bounds, x0=x0, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)
    
    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = bounds + ([-np.inf, np.inf],) * 20
    cons = [{'type': 'ineq', 'fun': lambda v: constraint_func(v)[2*i]} for i in range(28)]
    res = minimize(lambda v: -compute_sum_radii(v[52:]), x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol':1e-10,'gtol':1e-10,'maxiter':2000})
    centers = res.x[:52].reshape(26,2)
    radii = res.x[52:]
    
    sum_r = compute_sum_radii(radii)
    centers = centers.astype(np.float32)
    radii = radii.astype(np.float32)
    
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")
    
    return centers, radii, sum_r
