import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Define initial center pattern
    centers = np.array([
        [0.05, 0.12], [0.11, 0.12], [0.17, 0.12], [0.23, 0.12], [0.29, 0.12], [0.35, 0.12],
        [0.05, 0.20], [0.11, 0.20], [0.17, 0.20], [0.23, 0.20], [0.29, 0.20], [0.35, 0.20],
        [0.05, 0.28], [0.11, 0.28], [0.17, 0.28], [0.23, 0.28], [0.29, 0.28], [0.35, 0.28],
        [0.05, 0.36], [0.11, 0.36], [0.17, 0.36], [0.23, 0.36], [0.29, 0.36], [0.35, 0.36],
        [0.14, 0.44], [0.22, 0.44], [0.30, 0.44], [0.38, 0.44],
        [0.22, 0.52]
    ])
    
    def compute_distances(center1, center2):
        return np.sqrt(np.sum((center1 - center2) ** 2))
    
    def solve_lp(centers):
        r = np.zeros(n)
        A_ub = np.zeros((n * 5 + n * n, n))
        b_ub = np.zeros(n * 5 + n * n)
        
        for i in range(n):
            for d in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                x, y = centers[i] + 1e-5 * d
                if 0 <= x <= 1 and 0 <= y <= 1:
                    A_ub[i * 5 + np.where(centers[:, 0] == x)[0][0], i] = 1
                    b_ub[i * 5 + np.where(centers[:, 0] == x)[0][0]] = x
                    A_ub[i * 5 + np.where(centers[:, 1] == y)[0][0], i] = 1
                    b_ub[i * 5 + np.where(centers[:, 1] == y)[0][0]] = y
            A_ub[i * 5 + 4, i] = 1
            b_ub[i * 5 + 4] = centers[i, 0]
            A_ub[i * 5 + 5, i] = 1
            b_ub[i * 5 + 5] = 1 - centers[i, 0]
            A_ub[i * 5 + 6, i] = 1
            b_ub[i * 5 + 6] = centers[i, 1]
            A_ub[i * 5 + 7, i] = 1
            b_ub[i * 5 + 7] = 1 - centers[i, 1]
        
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[i * n + j, i] = 1
                A_ub[i * n + j, j] = 1
                b_ub[i * n + j] = compute_distances(centers[i], centers[j])
        
        c = -np.ones(n)
        res = np.linalg.solve(A_ub, b_ub)
        
        r = b_ub - np.dot(A_ub, r)
        return r
    
    def compute_forces(centers, radii):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = compute_distances(centers[i], centers[j])
                if dist > 1e-5:
                    force = radii[i] + radii[j] - dist
                    forces[i] += force * (centers[j] - centers[i]) / dist
                    forces[j] -= force * (centers[j] - centers[i]) / dist
            for k in range(4):
                if k == 0 and centers[i, 0] <= radii[i]:
                    forces[i, 0] += radii[i] - centers[i, 0]
                elif k == 1 and centers[i, 0] >= 1 - radii[i]:
                    forces[i, 0] += centers[i, 0] - (1 - radii[i])
                elif k == 2 and centers[i, 1] <= radii[i]:
                    forces[i, 1] += radii[i] - centers[i, 1]
                elif k == 3 and centers[i, 1] >= 1 - radii[i]:
                    forces[i, 1] += centers[i, 1] - (1 - radii[i])
        
        return forces
    
    max_iter = 200
    tol = 1e-6
    alpha = 0.03
    eps = 1e-7
    
    best_centers, best_radii = None, None
    best_obj = -np.inf
    
    for _ in range(3):
        random_shift = np.random.rand(26, 2) * 0.2 - 0.1
        centers = (centers + random_shift) % 1
        
        for _ in range(max_iter):
            radii = solve_lp(centers)
            obj = np.sum(radii)
            if obj > best_obj:
                best_obj = obj
                best_centers = centers.copy()
                best_radii = radii.copy()
            
            forces = compute_forces(centers, radii)
            centers += alpha * forces
            
            if np.all(np.abs(forces) < tol):
                break
        
        for i in range(n):
            if best_radii[i] < 0:
                best_radii[i] = 0
            if best_centers[i, 0] < 0:
                best_centers[i, 0] = 0
            if best_centers[i, 0] > 1:
                best_centers[i, 0] = 1
            if best_centers[i, 1] < 0:
                best_centers[i, 1] = 0
            if best_centers[i, 1] > 1:
                best_centers[i, 1] = 1
    
    radii = np.maximum(best_radii - eps, 0.0)
    return best_centers, radii, np.sum(radii)
