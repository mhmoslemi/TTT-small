import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalty(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    penalty_overlap = 0
    for i in range(26):
        if x[i] - r[i] < 0:
            penalty_overlap += (r[i] - (x[i] - 0)) ** 2
        if x[i] + r[i] > 1:
            penalty_overlap += (x[i] + r[i] - 1) ** 2
        if y[i] - r[i] < 0:
            penalty_overlap += (r[i] - (y[i] - 0)) ** 2
        if y[i] + r[i] > 1:
            penalty_overlap += (y[i] + r[i] - 1) ** 2
        for j in range(i + 1, 26):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            d = np.sqrt(dx * dx + dy * dy)
            if d < r[i] + r[j]:
                penalty_overlap += (r[i] + r[j] - d) ** 2
    return -np.sum(r) + 10000 * penalty_overlap

def local_constraints(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return np.array([-x[i] + r[i],
                    1 - x[i] - r[i],
                    -y[i] + r[i],
                    1 - y[i] - r[i]] + [-np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2) + (r[i] + r[j]) for i in range(26) for j in range(i + 1, 26)])

def local_refinement(z_best):
    bounds = np.array([[0, 1], [0, 1]] * 26 + [[0, 0.5]] * 26)
    constraints = dict(type='ineq', fun=lambda z : local_constraints(z))
    z_opt = minimize(penalty, z_best, bounds=bounds, method='SLSQP', constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
    return z_opt.x

    if penalty_best > 1e-8:
        z_best = local_refinement(z_best)
