import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalised_objective(z):
    n = len(z) // 3
    r = z[2*n:]
    
    penalty = 0
    for i in range(n):
        x, y = z[i], z[n + i]
        
        # Wall violation penalty
        if x - r[i] < 0:
            penalty += (r[i] - x) ** 2
        if x + r[i] > 1:
            penalty += (x + r[i] - 1) ** 2
        if y - r[i] < 0:
            penalty += (r[i] - y) ** 2
        if y + r[i] > 1:
            penalty += (y + r[i] - 1) ** 2
        
        for j in range(i + 1, n):
            dx, dy = z[i] - z[n + j], z[n + i] - z[n + j]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < r[i] + r[j]:
                penalty += (r[i] + r[j] - dist) ** 2
                
    return -np.sum(r) + 10000 * penalty
