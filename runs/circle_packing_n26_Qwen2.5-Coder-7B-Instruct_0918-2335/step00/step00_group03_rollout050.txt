import numpy as np
from scipy.optimize import minimize

def rsa_seed(n, r_min, iterations):
    circles = []
    while len(circles) < n:
        valid = True
        for _ in range(iterations):
            center = np.random.rand(2)
            radius = r_min
            valid = True
            for cx, cy in circles:
                if np.sqrt((cx - center[0])**2 + (cy - center[1])**2) < radius:
                    valid = False
                    break
            if valid:
                circles.append((center, radius))
                break
    return np.array([c[0] for c in circles]), np.array([c[1] for c in circles])

def physics_relaxation(centers, radii):
    n = len(centers)
    dt = 1e-3
    T = 2000
    for _ in range(T):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(centers[i] - centers[j])
                gap = d_ij - (radii[i] + radii[j])
                if gap < 0:
                    forces[i] += (centers[j] - centers[i]) * gap / (radii[i] + radii[j])
                    forces[j] -= (centers[j] - centers[i]) * gap / (radii[i] + radii[j])
            forces[i] += np.min([1 - centers[i][0], centers[i][0], 1 - centers[i][1], centers[i][1]]) * np.array([1, 1])
        centers += forces * dt
        radii += np.sum(forces * (centers - 0.5), axis=1) * dt
        radii = np.clip(radii, 0, None)
    return centers, radii

def global_scaling(centers, radii):
    n = len(radii)
    s_max = min(min([d / (r1 + r2) for d, r1, r2 in zip(np.linalg.norm(centers[i] - centers[j] for i in range(n) for j in range(i+1, n)), radii, radii)]),
               min((1 - radii) / radii))
    centers = (centers - 0.5) * s_max + 0.5
    radii *= s_max
    return centers, radii

def penalty_function(x, shapes):
    n = shapes.shape[0]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            gap = np.linalg.norm(shapes[i] - shapes[j]) - (x[i * 3] + x[i * 3 + 1] + x[j * 3] + x[j * 3 + 1])
            penalty += max(0, gap) ** 2
        penalty += max(0, x[i * 3] + x[i * 3 + 1] - (1 - max(x[i * 3], x[i * 3 + 1]))) ** 2
        penalty += max(0, x[i * 3] + x[i * 3 + 1] - x[i * 3 + 2]) ** 2
        penalty += max(0, x[i * 3] + x[i * 3 + 1] - (1 - max(x[i * 3], x[i * 3 + 1]))) ** 2
    return penalty

def objective_function(x, shapes):
    return -np.sum(x[:-26])

def gradient_objective_function(x, shapes):
    n = shapes.shape[0]
    grad = np.zeros_like(x)
    grad[-26:] = -1
    for i in range(n):
        for j in range(i + 1, n):
            gap = np.linalg.norm(shapes[i] - shapes[j]) - (x[i * 3] + x[i * 3 + 1] + x[j * 3] + x[j * 3 + 1])
            if gap < 0:
                grad[i * 3:i * 3 + 2] += (x[j * 3:j * 3 + 2] - x[i * 3:i * 3 + 2]) * gap / (x[i * 3] + x[i * 3 + 1] + x[j * 3] + x[j * 3 + 1])
                grad[j * 3:j * 3 + 2] -= (x[j * 3:j * 3 + 2] - x[i * 3:i * 3 + 2]) * gap / (x[i * 3] + x[i * 3 + 1] + x[j * 3] + x[j * 3 + 1])
        grad[i * 3:i * 3 + 1] += np.maximum(0, x[i * 3] + x[i * 3 + 1] - (1 - max(x[i * 3], x[i * 3 + 1])))
        grad[i * 3:i * 3 + 1] += np.maximum(0, x[i * 3] + x[i * 3 + 1] - x[i * 3 + 2])
        grad[i * 3:i * 3 + 1] += np.maximum(0, x[i * 3] + x[i * 3 + 1] - (1 - max(x[i * 3], x[i * 3 + 1])))
    return grad

def run_packing():
    np.random.seed(42)
    centers, radii = rsa_seed(26, 1e-3, 10000)
    shapes = np.hstack((centers, radii[:, np.newaxis]))
    centers, radii = physics_relaxation(centers, radii)
    shapes = np.hstack((centers, radii[:, np.newaxis]))
    centers, radii = global_scaling(centers, radii)
    shapes = np.hstack((centers, radii[:, np.newaxis]))
    bounds = np.vstack((np.zeros(78), np.array([1, 1] * 26 + [0.5] * 26)))
    x0 = np.hstack((centers.flatten(), radii))
    result = minimize(objective_function, x0, method='L-BFGS-B', jac=gradient_objective_function, args=(shapes,), bounds=bounds, options={'maxiter': 2000})
    return result.x[:52].reshape((26, 2)), result.x[52:], -result.fun
