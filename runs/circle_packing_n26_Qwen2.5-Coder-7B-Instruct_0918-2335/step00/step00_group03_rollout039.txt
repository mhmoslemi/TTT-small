def power_diagram_cell(centers, radii, i):
    polygon = [(0, 0), (1, 0), (1, 1), (0, 1)]
    for j in range(len(centers)):
        if j != i:
            power = np.sum((centers[j] - centers[i]) ** 2) - radii[i] ** 2 + radii[j] ** 2
            a = 2 * (centers[j] - centers[i])
            b = power + np.sum(centers[j] ** 2) - np.sum(centers[i] ** 2)
            polygon = sutherland_hodgman(polygon, a[0], b)
    return polygon

def centroid(polygon):
    x, y = 0, 0
    A = 0
    for k in range(len(polygon) - 1):
        xk, yk = polygon[k][0], polygon[k][1]
        xk1, yk1 = polygon[k + 1][0], polygon[k + 1][1]
        A += xk * yk1 - xk1 * yk
        x += (xk + xk1) * (xk * yk1 - xk1 * yk)
        y += (yk + yk1) * (xk * yk1 - xk1 * yk)
    A /= 2
    return x / (6 * A), y / (6 * A)

def distance_to_polygon(centers, radii, i):
    polygon = power_diagram_cell(centers, radii, i)
    x, y = centers[i]
    dist = np.inf
    for j in range(len(polygon)):
        p1, p2 = polygon[j], polygon[(j + 1) % len(polygon)]
        dx, dy = p2[0] - p1[0], p2[1] - p1[1]
        t = (-(dx * (x - p1[0]) + dy * (y - p1[1]))) / (dx ** 2 + dy ** 2)
        t = max(0, min(1, t))
        px, py = p1[0] + t * dx, p1[1] + t * dy
        dist = min(dist, np.sqrt((x - px) ** 2 + (y - py) ** 2) - radii[i])
    x_min, x_max = min(polygon[k][0] for k in range(len(polygon))), max(polygon[k][0] for k in range(len(polygon)))
    y_min, y_max = min(polygon[k][1] for k in range(len(polygon))), max(polygon[k][1] for k in range(len(polygon)))
    dist = min(dist, x - x_min, x_max - x, y - y_min, y_max - y)
    return dist
