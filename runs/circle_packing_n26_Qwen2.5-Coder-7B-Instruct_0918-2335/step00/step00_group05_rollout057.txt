import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # for deterministic initialisation
    n = 26
    epsilon = 1e-12  # wall tolerance
    delta_r = 1e-5  # global radius increment
    dt = 1e-3  # time step for integration
    max_iter = 3000
    tol = 1e-10

    # Initial random centre and tiny radii seeds
    centres = np.random.rand(n, 2) * (1 - 2 * epsilon) + epsilon
    radii = np.full(n, 1e-6, dtype=np.float64)

    # Velocity arrays, setting to zero for simplicity
    velocity_x = np.zeros(n)
    velocity_y = np.zeros(n)
    
    for step in range(max_iter):
        # Compute pairwise forces and adjust velocity
        forces_x = np.zeros(n)
        forces_y = np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                cx, cy = centres[i]
                dx, dy = centres[j]
                distance = np.sqrt((cx - dx) ** 2 + (cy - dy) ** 2)
                overlap = radii[i] + radii[j] - distance
                force = overlap * (cx - dx) / distance * alpha if overlap > 0 else 0
                forces_x[i] += force
                forces_x[j] -= force
                forces_y[i] += force * (cy - dy) / distance
                forces_y[j] -= force * (cy - dy) / distance

        # Compute wall repulsion forces
        for i in range(n):
            cx, cy = centres[i]
            distance_l = cx - radii[i] - epsilon
            distance_r = 1 - cx - radii[i] - epsilon
            distance_b = cy - radii[i] - epsilon
            distance_t = 1 - cy - radii[i] - epsilon

            force_left = -alpha * distance_l if distance_l > 0 else 0
            force_right = alpha * distance_r if distance_r > 0 else 0
            force_bottom = -alpha * distance_b if distance_b > 0 else 0
            force_top = alpha * distance_t if distance_t > 0 else 0

            forces_x[i] += force_left  # assuming symmetry for simplicity
            forces_x[i] -= force_right
            forces_y[i] += force_bottom
            forces_y[i] -= force_top

        # Update velocities and positions using Euler method with damping
        velocity_x += dt * (forces_x / 1.0 - beta * velocity_x)  # mass = 1
        velocity_y += dt * (forces_y / 1.0 - beta * velocity_y)
        centres += dt * (np.column_stack((velocity_x, velocity_y)))

        # Uniform global radius increment
        radii += delta_r

        # Feasibility clip to ensure it doesn’t extend into walls or settle as negative
        radii = np.minimum(radii, np.minimum(
            np.minimum(1 - 2 * epsilon - radii, radii), 
            np.minimum(1 - 2 * epsilon - radii[:, np.newaxis], radii)
        ))
        radii = np.maximum(radii, epsilon)

        # Convergence check
        delta_r = np.max(np.abs(radii - radii[-1]) / radii) if step > 0 else 1
        if delta_r < tol:
            break

    # Finally refine the radii using linear programming
    objective_coeffs = -radii
    inequality_coeffs = np.array([
        np.minimum(centres[:, 0], 1 - centres[:, 0]),  # left, right boundaries
        np.minimum(centres[:, 1], 1 - centres[:, 1]),  # bottom, top boundaries
        np.sum(np.maximum(np.zeros((n, n)), np.subtract.outer(centres[:, np.newaxis, 0], centres[None, :, 0]) + np.subtract.outer(centres[:, np.newaxis, 1], centres[None, :, 1]) - radii[:, np.newaxis] - radii[None, :]), axis=2)
    ])
    inequality_rhs = np.zeros(inequality_coeffs.shape[-1])
    bounds = (epsilon, None)  # ensure non-negative radius

    try:
        result = linprog(objective_coeffs, A_ub=inequality_coeffs, b_ub=inequality_rhs, bounds=bounds)
        if result.success:
            new_radii = result.x
            sum_radii = np.sum(new_radii)
            return centres, new_radii, sum_radii
    except:
        pass

    # If LP fails, return the last computed radii as they are likely near-optimal for those fixed centres
    sum_radii = np.sum(radii)
    return centres, radii, sum_radii
