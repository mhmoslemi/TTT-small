import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define initial circles at the midpoints of the sides of the unit square
    initial_centers = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    radii = [0.25, 0.25, 0.25, 0.25]
    centers = np.array(initial_centers)

    def distance(c1, c2):
        return np.sqrt(np.sum((c1 - c2) ** 2))

    def max_radius(p, obstacles):
        min_dist = np.inf
        for o in obstacles:
            if isinstance(o, tuple):
                # Wall
                dist = min(p[0], 1 - p[0], p[1], 1 - p[1])
            else:
                # Circle
                x, y = o
                dist = distance(p, x, y) - radii[centers == (x, y)].item()
            min_dist = min(min_dist, dist)
        return max(0, min_dist)

    for _ in range(len(centers), 26):
        obstacles = list(zip(centers[:, 0], centers[:, 1])) + [(0, 0), (1, 0), (1, 1), (0, 1)]
        vertices = []
        for i in range(len(obstacles)):
            for j in range(i + 1, len(obstacles)):
                for k in range(j + 1, len(obstacles)):
                    o1, o2, o3 = obstacles[i], obstacles[j], obstacles[k]
                    # Check if the triangle formed by o1, o2, and o3 is valid
                    if distance(o1, o2) > 1e-12 and distance(o2, o3) > 1e-12 and distance(o3, o1) > 1e-12:
                        # Circle tangent to three points (o1, o2, o3)
                        p = calculate_tangent_circle(o1, o2, o3)
                        if any(d < 1e-12 for d in [p[0], p[1], 1 - p[0], 1 - p[1]]):
                            continue
                        r = max_radius(p, obstacles)
                        vertices.append((p, r))

        if not vertices:
            raise ValueError("No valid vertex found")

        p_max_radius, r_max_radius = max(vertices, key=lambda x: x[1])
        centers = np.vstack((centers, p_max_radius))
        radii.append(r_max_radius)

    def calculate_tangent_circle(o1, o2, o3):
        # Transform coordinates to have a wall at x=0 performing the reflection of o3
        if isinstance(o1, tuple):
            x1, y1 = 1 - o1[0], o1[1]
        else:
            x1, y1 = 1 - o1[:, 0], o1[:, 1]
        if isinstance(o2, tuple):
            x2, y2 = 1 - o2[0], o2[1]
        else:
            x2, y2 = 1 - o2[:, 0], o2[:, 1]
        if isinstance(o3, tuple):
            x3, y3 = 1 - o3[0], o3[1]
        else:
            x3, y3 = 1 - o3[:, 0], o3[:, 1]

        # Calculate the circumcenter of the triangle (o1, o2, o3)
        A = 2 * (x2 * (y3 - y1) + x3 * (y1 - y2) + x1 * (y2 - y3))
        B = 2 * (x2 * x2 + y2 * y2 - x3 * x3 - y3 * y3 + x1 * x1 - y1 * y1)
        C = 2 * (y2 * (x3 - x1) + y3 * (x1 - x2) + y1 * (x2 - x3))
        Px, Py = (A * (x2 * x2 + y2 * y2) - B * (x1 * x1 + y1 * y1) - C * (x3 * x3 + y3 * y3)) / (4 * A * A + B * B + C * C),
                 (A * (x2 * y2 + y2 * y2) - B * (x1 * y1 + y1 * y1) - C * (x3 * y3 + y3 * y3)) / (4 * A * A + B * B + C * C)
        return (1 - Px, Py)

    # Convert centers to non-negative coordinates
    centers = np.mod(centers - np.min(centers, axis=0), 1)

    # Ensure each circle fits inside the square
    for i in range(len(radii)):
        r = radii[i]
        while (centers[i, 0] - r) < 0:
            centers[i, 0] += 1
        while (centers[i, 1] - r) < 0:
            centers[i, 1] += 1
        while (centers[i, 0] + r) > 1:
            centers[i, 0] -= 1
        while (centers[i, 1] + r) > 1:
            centers[i, 1] -= 1

    # Validate packing
    valid, _ = validate_packing(centers, radii)
    if not valid:
        raise ValueError("Invalid packing")

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
