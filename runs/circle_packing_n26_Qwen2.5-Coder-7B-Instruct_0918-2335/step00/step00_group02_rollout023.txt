import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def create_constraints(center_indices):
    n = len(center_indices)
    constraints = []
    for i in range(n):
        x, y = center_indices[i]
        constraints.append({'type': 'ineq', 'fun': lambda z, x=x, y=y, i=i: z[x + i * 3] - z[y + i * 3]})
        constraints.append({'type': 'ineq', 'fun': lambda z, x=x, y=y, i=i: z[x + i * 3] + z[y + i * 3] - 1})
        constraints.append({'type': 'ineq', 'fun': lambda z, x=x, y=y, i=i: z[y + i * 3] - z[x + i * 3]})
        constraints.append({'type': 'ineq', 'fun': lambda z, x=x, y=y, i=i: z[y + i * 3] + z[x + i * 3] - 1})
    for i in range(n):
        for j in range(i + 1, n):
            index_i = center_indices[i]
            index_j = center_indices[j]
            constraints.append({'type': 'ineq', 'fun': lambda z, index_i=index_i, index_j=index_j, i=i, j=j: (z[index_i[0] + i * 3] - z[index_j[0] + j * 3]) ** 2 + (z[index_i[1] + i * 3] - z[index_j[1] + j * 3]) ** 2 - (z[index_i[2] + i * 3] + z[index_j[2] + j * 3]) ** 2})
    return constraints

def objective(z):
    return -np.sum(z[2::3])

def gradient(z):
    n = z.shape[0] // 3
    grad = np.zeros(n * 3)
    grad[2::3] = -1
    return grad

def run_packing():
    centers = np.random.uniform(0, 1, (26, 2))
    radii = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]) / 2
    z0 = np.concatenate((centers.flatten(), radii))

    constraints = create_constraints(np.arange(26))

    for _ in range(6):
        res = minimize(objective, z0, method='trust-constr', jac=gradient, constraints=constraints)
        if res.success and validate_packing(res.x[:-26].reshape(26, 2), res.x[-26:])[0]:
            return res.x[:-26].reshape(26, 2), res.x[-26:], np.sum(res.x[-26:])
        constraints = [c for c in constraints if c['fun'](z0) <= 1e-8]
        z0 = np.random.uniform(0, 1, (26, 2))
        radii = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]) / 2
        z0 = np.concatenate((centers.flatten(), radii))
    return np.zeros(26), np.zeros(26), 0
