import numpy as np
from scipy.optimize import differential_evolution

def penalised_objective(z):
    radii = z[52:]
    N = len(radii)
    sum_radii = -np.sum(radii)
    penalty = 0.0
    for i in range(N):
        if z[2*i] - radii[i] < 0:
            penalty += (z[2*i] - radii[i]) ** 2
        if z[2*i] + radii[i] > 1:
            penalty += (z[2*i] + radii[i] - 1) ** 2
        if z[2*i + 1] - radii[i] < 0:
            penalty += (z[2*i + 1] - radii[i]) ** 2
        if z[2*i + 1] + radii[i] > 1:
            penalty += (z[2*i + 1] + radii[i] - 1) ** 2
        for j in range(i + 1, N):
            dist = np.sqrt((z[2*i] - z[2*j]) ** 2 + (z[2*i + 1] - z[2*j + 1]) ** 2)
            penalty += (radii[i] + radii[j] - dist) ** 2
    return sum_radii + 10000 * penalty

def local_constraints(z):
   radii = z[52:]
    N = len(radii)
    constraints = []
    for i in range(N):
        constraints.extend([z[2*i] - radii[i] - 1e-12,
                           1 - z[2*i] - radii[i] - 1e-12,
                           z[2*i + 1] - radii[i] - 1e-12,
                           1 - z[2*i + 1] - radii[i] - 1e-12])
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((z[2*i] - z[2*j]) ** 2 + (z[2*i + 1] - z[2*j + 1]) ** 2)
            constraints.append(dist - (radii[i] + radii[j]) + 1e-12)
    return constraints

def local_refinement(z):
    bounds = [(0, 1)] * 78
    res = differential_evolution(penalised_objective, bounds, constraints=local_constraints, maxiter=2000, popsize=1, tol=1e-12)
    return res.x

def run_packing():
    np.random.seed(0)
    x = np.random.rand(26)
    y = np.random.rand(26)
    r = 0.5 * np.random.rand(26)
    z = np.concatenate((x, y, r))
    z_best = differential_evolution(penalised_objective, [((0, 1), (0, 1), (0, 0.5))] * 26, popsize=20, strategy='rand1bin', maxiter=2000, tol=1e-8).x
    penalty = penalised_objective(z_best)
    if penalty < 1e-9:
        z_opt = z_best
    else:
        z_opt = local_refinement(z_best)
    centers = z_opt[:52].reshape((26, 2))
    radii = z_opt[52:]
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
