import numpy as np
import scipy.spatial
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    min_radius = 1e-4
    grow_rate = 1e-3
    init_speed = 5e-3
    T_max = 50.0
    epsilon = 1e-6

    def random_initial_positions():
        return np.random.rand(n, 2) * 0.95 + 0.025

    def random_initial_velocities():
        velocities = np.random.randn(n, 2)
        velocities /= np.linalg.norm(velocities, axis=1, keepdims=True)
        velocities *= init_speed
        return velocities

    def event_queue(centers, radii, velocities):
        queue = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist > radii[i] + radii[j]:
                    continue
                dt = (dist - radii[i] - radii[j]) / np.dot(velocities[i] - velocities[j], centers[i] - centers[j])
                queue.append((dt, i, j))
            for j, axis in enumerate(['x', 'y']):
                pos = centers[:, j]
                radius = radii + grow_rate
                dt = (radius - pos) / (velocities[:, j] - grow_rate)
                dt = dt[(dt > 0) & (pos + velocities[:, j] * dt <= radius + grow_rate)]
                if len(dt) > 0:
                    dt = np.min(dt)
                    queue.append((dt, j, -1))
        queue = np.array(queue)
        queue = queue[np.argsort(queue[:, 0])]
        return queue

    def resolve_event(queue, centers, radii, velocities):
        dt, i, j = queue[0]
        centers += velocities * dt
        radii += grow_rate * dt
        queue = queue[1:]
        if i >= 0:
            velocities[i] -= 2 * velocities[i, j] / np.sum(velocities[i]) / (radii[i] + radii[j]) * np.dot(velocities[i] - velocities[j], centers[i] - centers[j]) * (centers[i] - centers[j])
        else:
            axis = j
            centers[:, axis] -= grow_rate * dt
        return centers, radii, velocities, queue

    def shrink_and_repair(centers, radii):
        excess = np.any(radii < 0)
        if excess:
            radii -= np.min(radii[~excess], axis=0) - 2 * epsilon
            dists = scipy.spatial.distance.cdist(centers, centers)
            violations = np.triu(dists < radii[:, None] + radii[None, :] - 2 * epsilon, k=1)
            while np.any(violations):
                for i, j in zip(*np.nonzero(violations)):
                    centers[i] += (centers[i] - centers[j]) * epsilon / np.linalg.norm(centers[i] - centers[j])
                    centers[j] -= (centers[i] - centers[j]) * epsilon / np.linalg.norm(centers[i] - centers[j])
                    dists = scipy.spatial.distance.cdist(centers, centers)
                    violations = np.triu(dists < radii[:, None] + radii[None, :] - 2 * epsilon, k=1)
        return centers, radii

    def lp_polish(centers, radii):
        A = np.vstack([radii, -radii]).T
        A[:, :2] = 0
        A[[0, 2, 4, 6], 2] = 1
        A[[1, 3, 5, 7], 3] = 1
        b = np.hstack([np.ones(n) + 2 * epsilon, np.ones(n), np.ones(n), np.ones(n)])
        c = radii
        result = scipy.optimize.linprog(-c, A_ub=A, b_ub=b, method='highs')
        return result.x

    centers = random_initial_positions()
    velocities = random_initial_velocities()

    queue = event_queue(centers, radii, velocities)

    current_time = 0.0
    while len(queue) > 0 and current_time < T_max:
        dt, i, j = queue[0]
        centers += velocities * dt
        radii += grow_rate * dt
        current_time += dt
        centers, radii, velocities, queue = resolve_event(queue, centers, radii, velocities)

    centers, radii = shrink_and_repair(centers, radii)
    radii = lp_polish(centers, radii)
    radii[radii < 0] = 0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
