from numpy.random import seed, uniform

def init_circles(variant):
    grid_step = 0.2
    grid_size = 5
    base_circles = grid_size**2

        for j in range(grid_size):
            x = grid_step * (i + 0.5)
            y = grid_step * (j + 0.5)
            centers[i + j * grid_size] = [x, y]

    # Place an extra circle for variant A or B
    if variant == 'A':
        centers[base_circles] = [0.6, 0.9]
    else:
        centers[base_circles] = [0.4, 0.9]

def inflate(circles, delta_r, epsilon):

                    centers_diff = circles[j] - circles[i]
                    circles[i] += delta * centers_diff / 2
                    circles[j] -= delta * centers_diff / 2

        # Ensure all circles stay within the unit square
        for circle in circles:
            circle[2] = min(circle[2], min(circle[0], 1 - circle[0], circle[1], 1 - circle[1]) - epsilon)

        for circle in circles:
            allowed_radius = min(circle[0], 1 - circle[0], circle[1], 1 - circle[1]) - epsilon
