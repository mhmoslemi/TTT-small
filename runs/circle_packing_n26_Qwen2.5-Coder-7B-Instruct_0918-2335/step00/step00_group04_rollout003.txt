import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(0)
    
    def penalised_objective(z):
        x, y = z[:n], z[n:2*n]
        r = z[2*n:]
        penalisation = np.sum([max(x - r, 0)**2 + max(1 - x - r, 0)**2 + max(y - r, 0)**2 + max(1 - y - r, 0)**2 for x, y, r in zip(x, y, r)])
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                penalisation += max(0, dist - (r[i] + r[j]))**2
        return -np.sum(r) + 10000 * penalisation

    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    sol = differential_evolution(penalised_objective, bounds, popsize=20, tol=1e-8, maxiter=2000)
    
    if sol.fun < 1e-8:
        z_opt = sol.x
    else:
        def local_constraints(z):
            c = np.zeros(3*n)
            c[:n] = z[:n] - z[n:]
            c[n:2*n] = 1 - z[:n] - z[n:]
            c[2*n:] = 1 - z[n:] - z[2*n:]
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt((z[i] - z[i+n])**2 + (z[i+n] - z[j+n])**2)
                    c[n*2 + 3*i + (j-i-1)] = d - (z[2*n + i] + z[2*n + j])
            return c
        
        def local_obj(z):
            r = z[2*n:]
            return -np.sum(r)
        
        sol_local = minimize(local_obj, sol.x, method='SLSQP', bounds=bounds, constraints=local_constraints, options={'maxiter': 2000, 'ftol': 1e-12})
        z_opt = sol_local.x
    
    centers = z_opt[:2*n].reshape((n, 2))
    radii = z_opt[2*n:]
    return centers, radii, np.sum(radii)
