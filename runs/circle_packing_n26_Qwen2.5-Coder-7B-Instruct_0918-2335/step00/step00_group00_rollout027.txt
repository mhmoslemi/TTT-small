import numpy as np

    radii = []

        r_candidate = np.min(np.array([r_candidate, distance_grid - radius]), axis=0)

        centers.append(centre)
        radii.append(radius)
        
        if np.all(r_candidate == -1e9):
            break
        
        max_radius = np.max([max_radius, radius + 1e-9])
    
    for _ in range(N_refine):
        i = np.random.randint(0, 26)
        centre = centers[i]
        radius = radii[i]
        
        delta = np.random.uniform(-δ_center, δ_center, 2)
        new_centre = centre + delta
