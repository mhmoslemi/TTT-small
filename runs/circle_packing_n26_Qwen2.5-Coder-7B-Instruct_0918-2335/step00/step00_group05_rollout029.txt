from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform
import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    θ = np.linspace(np.pi/6, 11*np.pi/6, n)
    
    # Initial placement
    centers = np.array([(0.5 + 0.2*np.cos(t), 0.5 + 0.2*np.sin(t)) for t in θ])
    radii = np.sqrt(0.5 * (1 - 2 * (np.abs(centers[:, 0] - 0.5) + np.abs(centers[:, 1] - 0.5))))
    
    # Physics-based inflation 
    Δt = 1.0 * 1e-3
    α = 4.0
    β = 0.8
    ε = 1.0 * 1e-12
    δr = 1.0 * 1e-4
    maxIter = 3000
    tol = 1.0 * 1e-10
    
    for step in range(maxIter):
        forces = np.zeros_like(centers)
        wall_forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] + ε:
                    overlap = radii[i] + radii[j] - dist
                    direction = (centers[i] - centers[j]) / dist
                    forces[i] += α * overlap * direction
                    forces[j] -= α * overlap * direction
            if centers[i, 0] < radii[i] + ε:
                wall_forces[i, 0] = α * (radii[i] + ε - centers[i, 0])
            if centers[i, 0] > 1 - radii[i] - ε:
                wall_forces[i, 0] = α * (centers[i, 0] - (1 - radii[i] - ε))
            if centers[i, 1] < radii[i] + ε:
                wall_forces[i, 1] = α * (radii[i] + ε - centers[i, 1])
            if centers[i, 1] > 1 - radii[i] - ε:
                wall_forces[i, 1] = α * (centers[i, 1] - (1 - radii[i] - ε))
                
        velocities = β * velocities + (forces + wall_forces) / 1  # Assuming m=1
        centers += Δt * velocities
        
        radii += δr * np.ones_like(radii)
        radii[radii < 0] = 0
        
        # Boundary clipping
        distances_to_wall = np.column_stack((radii[:, np.newaxis], radii[:, np.newaxis]))
        distances_to_wall[:, 0] = np.minimum(distances_to_wall[:, 0], centers[:, 0] - ε)
        centers[:, 0] = np.maximum(centers[:, 0], distances_to_wall[:, 0])
        
        distances_to_wall[:, 1] = np.minimum(distances_to_wall[:, 1], 1 - centers[:, 0] - ε)
        centers[:, 0] = np.minimum(centers[:, 0], 1 - distances_to_wall[:, 1])
        
        distances_to_wall[:, 0] = np.minimum(distances_to_wall[:, 0], centers[:, 1] - ε)
        centers[:, 1] = np.maximum(centers[:, 1], distances_to_wall[:, 0])
        
        distances_to_wall[:, 1] = np.minimum(distances_to_wall[:, 1], 1 - centers[:, 1] - ε)
        centers[:, 1] = np.minimum(centers[:, 1], 1 - distances_to_wall[:, 1])
        
        # Check convergence
        ΔR = np.max(np.abs(radii - np.roll(radii, 1)) / radii)
        if ΔR < tol:
            break
    
    # Linear programming for optimal radius assignment
    A = Squareform(pdist(centers), checks=False)
    A += np.eye(n)
    b = pdist(centers)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b, bounds=(ε, None))
    if res.success:
        radii = res.x
    
    # Optional simulated annealing for fine-tuning
    def objective_function(radii):
        distances = pdist(centers[:, :2])
        return -np.sum(radii) + np.sum(np.minimum(distances - radii[:, np.newaxis] - radii[np.newaxis, :], 0))
    
    def annealing(radii, β):
        β *= 0.995
        if β < 1e-6:
            return radii
        for _ in range(5000):
            i = np.random.randint(n)
            r_i_prime = radii[i] + np.random.normal(0, β)
            if r_i_prime >= 0:
                radii[i] = r_i_prime
                res = linprog(c, A_ub=A, b_ub=b, bounds=(ε, None))
                if res.success:
                    if objective_function(res.x) > objective_function(radii):
                        radii = res.x
                else:
                    radii[i] = r_i_prime
            β *= 0.995
        return radii
    
    radii = annealing(radii, 1e-3)
    
    return centers, radii, np.sum(radii)
