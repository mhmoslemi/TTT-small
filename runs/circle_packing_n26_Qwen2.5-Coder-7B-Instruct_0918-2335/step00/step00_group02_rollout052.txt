from scipy.optimize import minimize
import numpy as np

def objective(z):
    return -np.sum(z[2:])

def linear_constraints(z):
    return [-z[i] for i in range(0, 2*length, 3)]

def boundary_constraints(z):
    x_indices = np.arange(0, 2*length, 3)
    y_indices = np.arange(1, 2*length, 3)
    r_indices = np.arange(2, 2*length, 3)
    return np.concatenate([z[x_indices] - z[r_indices], 
                           z[1:x_indices] - z[r_indices], 
                           1 - z[1:x_indices] - z[r_indices], 
                           z[y_indices] - z[r_indices], 
                           1 - z[y_indices] - z[r_indices]])

def contact_constraints(z):
    x_indices = np.arange(0, 2*length, 3)
    y_indices = np.arange(1, 2*length, 3)
    r_indices = np.arange(2, 2*length, 3)
    constraints = []
    for i in range(length):
        for j in range(i + 1, length):
            x_diff = z[3*i] - z[3*j]
            y_diff = z[3*i + 1] - z[3*j + 1]
            radius_sum = z[3*i + 2] + z[3*j + 2]
            constraints.append(x_diff**2 + y_diff**2 - radius_sum**2)
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    length = 26
    initial_radius = 0.0

    initial_guess = np.random.rand(length * 3) * 0.5
    for i in range(length):
        initial_radius = min(initial_guess[3*i], 1 - initial_guess[3*i])
        initial_guess[3*i + 2] = initial_radius * np.random.uniform(0.3, 0.8)

    bounds = [(0, 1)] * (3 * length)
    constraints = []
    constraints += [{'type': 'ineq', 'fun': linear_constraints}]
    constraints += [{'type': 'ineq', 'fun': boundary_constraints}]
    constraints += [{'type': 'ineq', 'fun': contact_constraints}]
    
    result = minimize(objective, initial_guess, method='trust-constr', bounds=bounds, constraints=constraints, jac='2-point')
    
    if result.success != True:
        return np.zeros((length, 2)), np.zeros(length), 0
    
    centers = result.x[::3].reshape((length, 1))
    radii = result.x[2::3]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
