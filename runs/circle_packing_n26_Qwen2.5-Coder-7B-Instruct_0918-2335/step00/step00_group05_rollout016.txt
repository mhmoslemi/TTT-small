import numpy as np
from scipy.optimize import minimize

def hex_lattice(n):
    s = np.sqrt(2 * (1 - np.tan(np.pi / 6) ** 2) / 3)
    h = s * np.sqrt(3) / 2
    w = s
    x0, y0 = s / 2, s / 2
    levels = np.arange(np.ceil(np.sqrt(n)).astype(int)) + 1
    idx = np.cumsum(levels)[:-1]
    x = np.tile(x0, len(idx)) + np.repeat(np.arange(levels)[:-1] * w, levels[:-1])[:, None]
    y = np.concatenate([y0 + np.arange(level) * h for level in levels] + [y0 + h * np.arange(n).sum() % level for level in levels])
    inside = np.all((0 < x) & (x < 1) & (0 < y) & (y < 1), axis=1)
    x, y = x[inside], y[inside]
    r = np.full_like(x, s / 2)
    return x, y, r

def solve_stage(z0, delta):
    x, y, r = z0[:26].reshape(26, 2), z0[26:2*26].reshape(26, 2), z0[2*26:]
    constraints = [
        {'type': 'ineq', 'fun': lambda z: z[:26].reshape(26, 2)[:, 0] - z[:26].reshape(26, 2)[:, 1]},
        {'type': 'ineq', 'fun': lambda z: 1 - z[:26].reshape(26, 2)[:, 0] - z[:26].reshape(26, 2)[:, 1]},
        {'type': 'ineq', 'fun': lambda z: z[:26].reshape(26, 2)[:, 1] - z[:26].reshape(26, 2)[:, 0]},
        {'type': 'ineq', 'fun': lambda z: 1 - z[:26].reshape(26, 2)[:, 1] - z[:26].reshape(26, 2)[:, 0]},
        {'type': 'ineq', 'fun': lambda z: (z[:26].reshape(26, 2)[:, 0] - z[26:52].reshape(26, 2)[:, 0][:, np.newaxis])**2 + (z[:26].reshape(26, 2)[:, 1] - z[26:52].reshape(26, 2)[:, 1][:, np.newaxis])**2 - (z[:26].reshape(26, 2)[:, 1] + z[26:52].reshape(26, 2)[:, 1][:, np.newaxis])**2 - delta}
    ]
    result = minimize(lambda z: -z[2*26:].sum(), z0, method='SLSQP', bounds=None, constraints=constraints, options={'maxiter': 1000})
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    x, y, r = hex_lattice(26)
    z0 = np.concatenate([x, y, r])
    z = z0
    for epsilon in [1e-8, 5e-4, 1e-12]:
        z = solve_stage(z, epsilon)
        x, y, r = z[:26].reshape(26, 2), z[26:52].reshape(26, 2), z[52:]
        for _ in range(100):
            r_i = np.minimum(r, 1 - x)  # Touch left side
            r_j = np.minimum(r, 1 - y)  # Touch top side
            r_k = np.minimum(r, x)       # Touch right side
            r_l = np.minimum(r, y)       # Touch bottom side
            for i in range(26):
                r[i] = max(r[i], min(r_i[i], r_j[i], r_k[i], r_l[i]))
            d_ij = np.linalg.norm(x[:, np.newaxis] - x[np.newaxis, :], axis=2)
            violations = np.maximum(0, d_ij - r_i[:, np.newaxis] - r[np.newaxis, :] - 1e-12)
            r -= 0.5 * violations / (d_ij + 1e-12)
            r.clip(min=1e-12, out=r)
            x += np.apply_along_axis(lambda v: v[0] / np.linalg.norm(v), 1, x - z0[:52].reshape(26, 2) - np.where(d_ij < 1e-12, 1e-3 * (r_i[:, np.newaxis] + r[np.newaxis, :] - d_ij), 0))
            r.clip(min=1e-12, out=r)
    sum_radii = r.sum()
    return x, y, sum_radii
