import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog
from scipy.stats import multivariate_normal as mvn

# Initialize parameters
num_circles = 26
grid_size = 0.312
lattice = np.array([[i%9+0.156, (i//9)*np.sqrt(3)/3+0.1875] for i in range(num_circles)])
lattice = lattice[lattice[:, 0] < 0.95 and lattice[:, 0] > 0.05 and lattice[:, 1] < 0.95 and lattice[:, 1] > 0.05]
assert len(lattice) == num_circles, "Failed to generate a 26-circle configuration"
dr = pdist(lattice, 'euclidean')  # Pairwise distances
dr = squareform(dr)
dr = np.clip(dr, 1e-8, None)  # Ensure non-zero distances

T0 = 0.12
T_min = 1e-6
cooling_rate = 0.995
moves_per_temp = 200
sigma_move = lambda T: T / 5
eps = 1e-12

# Warm start linear programming solver
def solve_lp(centers):
    r_bounds = np.array([[0, np.inf]] * num_circles)
    equality_constrs = np.vstack([np.eye(num_circles), np.eye(num_circles)[::-1, :]])
    equality_rhs = pdist(centers, 'euclidean').clip(1e-8)
    inequality_constrs = np.hstack([equity_constrs, np.minimum(centers[:, 0], 1 - centers[:, 0]).reshape(-1, 1)[:, None],
                                    np.minimum(centers[:, 1], 1 - centers[:, 1]).reshape(-1, 1)[:, None]])
    inequality_rhs = np.hstack([equality_rhs, centers[:, 0], centers[:, 1]])
    lp_result = linprog(
        -np.ones(num_circles),
        A_ub=inequality_constrs,
        b_ub=inequality_rhs,
        A_eq=equality_constrs,
        b_eq=equality_rhs,
        bounds=r_bounds,
        method='highs'
    )
    if not lp_result.success:
        raise ValueError(f"LP solver failed: {lp_result.message}")
    return lp_result.x, -lp_result.fun

# Simulated Annealing
best_centers = lattice
best_sum = solve_lp(best_centers)[1]
current_centers = lattice.copy()
current_sum = best_sum
temperature = T0
move_gen = lambda: (np.random.choice(num_circles), mvn.rvs(current_centers, cov=sigma_move(temperature)))
eval_fn = lambda centers: solve_lp(centers)[1]
accept_fn = lambda new_sum: new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / temperature)

start_time = time.time()
while time.time() - start_time < 5:
    for _ in range(moves_per_temp):
        i, delta = move_gen()
        current_centers[i] += delta
        current_centers[i] = np.clip(current_centers[i], eps, 1 - eps)
        new_sum = eval_fn(current_centers)
        if accept_fn(new_sum):
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = current_centers.copy()
        current_centers[i] -= delta
        current_centers[i] = np.clip(current_centers[i], eps, 1 - eps)
    temperature *= cooling_rate

# Optional local refinement (simple gradient ascent)
for _ in range(30):
    most_critical = np.argmin(np.clip(dr - (solve_lp(best_centers)[0] + 1e-12), 0, None))
    best_centers[most_critical] += delta
    best_centers[most_critical] = np.clip(best_centers[most_critical], eps, 1 - eps)
    new_sum = eval_fn(best_centers)
    if new_sum > best_sum:
        best_sum = new_sum
    best_centers[most_critical] -= delta
    best_centers[most_critical] = np.clip(best_centers[most_critical], eps, 1 - eps)

# Output results
centers = best_centers
radii = solve_lp(centers)[0]
return centers, radii, best_sum
