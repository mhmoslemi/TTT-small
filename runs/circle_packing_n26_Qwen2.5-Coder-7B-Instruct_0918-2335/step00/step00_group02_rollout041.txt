import numpy as np
from scipy.spatial import distance_matrix

def initial_centers(num_circles):
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        centers[i, 0] = i / (num_circles - 1)
        centers[i, 1] = (i + num_circles // 2) / num_circles
    return centers

def fit_radii(centers, target_sum):
    n = centers.shape[0]
    sum_radii = np.sum(target_sum / n)
    radii = np.full(n, sum_radii)
    while True:
        distances = distance_matrix(centers, centers)
        overlaps = distances < 2 * radii[:, np.newaxis] - 1e-12
        np.fill_diagonal(overlaps, False)
        overlap_count = np.sum(overlaps, axis=1)
        if np.all(overlap_count <= 1):
            break
        for i in range(n):
            if overlap_count[i] > 1:
                for j in range(n):
                    if overlaps[i, j]:
                        radii[i] -= 1e-6
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = initial_centers(26)
    radii = fit_radii(centers, 2.636)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
