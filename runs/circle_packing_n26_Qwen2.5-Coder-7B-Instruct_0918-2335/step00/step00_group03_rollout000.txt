import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = [
        (0.5, 0.5),  # Center
    ]
    r = [r_c] * 4  # Corners
    
    # Mid-edge circles
    r += [r_e] * 4
    
    # Inner hexagonal core
    h = (2 * np.sqrt(3) * r_i)
    grid = np.array([
        (0.5, 0.5 + r_i),
        (0.5 + h / 2, 0.5 - h / 2),
        (0.5 + h / 2, 0.5 + h / 2),
        (0.5, 0.5 - h / 2),
        (0.5, 0.5 + h / 2),
        (0.5 - h / 2, 0.5 + h / 2),
        (0.5 - h / 2, 0.5 - h / 2),
        (0.5, 0.5 - h)
    ])
    for x, y in grid:
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append((x, y))
            r += [r_i]
    return np.array(centers), np.array(r)    

def compute_sum_radii(radii):
    return np.sum(radii)

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def constraint_func(v):
    n = 26
    x = v[:52].reshape(n, 2)
    r = v[52:]
    constraints = []
    for i in range(n):
        constraints.append(r[i] <= min(x[i, 0], 1 - x[i, 0], x[i, 1], 1 - x[i, 1]))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((x[i] - x[j]) ** 2))
            constraints.append(dist - r[i] - r[j] >= 1e-8)
    return constraints

def run_packing():
    bounds = [
        (0.30, 0.49),
        (0.10, 0.25),
        (0.05, 0.15)
    ]
    result_de = differential_evolution(objective, bounds, popsize=10,
                                       tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)
    
    # SLSQP refinement
    rtotal = len(radii)
    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [(x[0] - rtotal, x[0] + rtotal) for x in bounds]
    bounds_slsqp.extend([(0, 0.5)] * rtotal)
    res = minimize(lambda v: -compute_sum_radii(v[52:]), x0, method='SLSQP',
                  bounds=bounds_slsqp, constraints=constraint_func,
                  options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 2000})
    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]
    
    sum_r = compute_sum_radii(radii)
    return centers, radii, sum_r
