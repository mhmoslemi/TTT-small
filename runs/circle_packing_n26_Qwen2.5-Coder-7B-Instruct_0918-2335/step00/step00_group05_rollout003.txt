import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.stats import norm

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dim = n * 3  # Each circle has 2 center coordinates and 1 radius
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) / 100

    x = np.concatenate((centers.flatten(), radii))

    def objective(x):
        centers = x[:dim].reshape((n, 3))
        radii = centers[:, 2]
        overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                center_i, center_j = centers[i, :2], centers[j, :2]
                radius_i, radius_j = radii[i], radii[j]
                dist = np.linalg.norm(center_i - center_j)
                if dist < radius_i + radius_j:
                    overlap += (radius_i + radius_j - dist)
        min_radius = np.max([np.maximum(centers[:, 0] - radii, 1 - centers[:, 0] - radii),
                             np.maximum(centers[:, 1] - radii, 1 - centers[:, 1] - radii)], axis=0)
        overlap += np.sum(radii - min_radius)
        return -np.sum(radii) + overlap

    def constraint(x):
        centers = x[:dim].reshape((n, 3))
        overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                center_i, center_j = centers[i, :2], centers[j, :2]
                radius_i, radius_j = centers[i, 2], centers[j, 2]
                dist = np.linalg.norm(center_i - center_j)
                if dist < radius_i + radius_j:
                    overlap += (radius_i + radius_j - dist)
        min_radius = np.max([np.maximum(centers[:, 0] - centers[:, 2], 1 - centers[:, 0] - centers[:, 2]),
                             np.maximum(centers[:, 1] - centers[:, 2], 1 - centers[:, 1] - centers[:, 2])], axis=0)
        overlap += np.sum(centers[:, 2] - min_radius)
        return overlap

    constraints = {'type': 'ineq', 'fun': constraint}
    bounds = [(0, 1), (0, 1)] * n + [(0, 0.01)] * n
    res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 10000})

    if not res.success:
        return np.nan * np.ones((n, 2)), np.nan * np.ones(n), np.nan

    centers = res.x[:dim].reshape((n, 3))[:, :2]
    radii = res.x[:dim].reshape((n, 3))[:, 2]
    return centers, radii, np.sum(radii)
