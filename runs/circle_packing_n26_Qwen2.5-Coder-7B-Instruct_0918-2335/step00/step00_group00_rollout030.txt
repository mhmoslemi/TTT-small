import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Define pair indices
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    def constraint_func(v):
        x = v[:52].reshape((n, 2))
        r = v[52:]
        arr = []
        # Boundary constraints
        arr.append(x[:, 0] - r)  # >=0
        arr.append(1 - x[:, 0] - r)  # >=0
        arr.append(x[:, 1] - r)  # >=0
        arr.append(1 - x[:, 1] - r)  # >=0
        # Pairwise constraints
        i_indices = np.array([i for i, j in pair_indices], dtype=np.int64)
        j_indices = np.array([j for i, j in pair_indices], dtype=np.int64)
        dx = x[i_indices, 0] - x[j_indices, 0]
        dy = x[i_indices, 1] - x[j_indices, 1]
        dist2 = dx * dx + dy * dy
        pair_vals = dist2 - (r[i_indices] + r[j_indices]) ** 2
        arr.append(pair_vals)
        return np.concatenate(arr)
    
    def objective(v):
        r = v[52:]
        return -np.sum(r)
    
    x0 = np.zeros(78)
    positions = np.array([(j / 5.0, i / 5.0) for i in range(5) for j in range(5)])  # grid positions
    x0[:50] = positions.ravel()
    x0[50:52] = [0.6, 0.9]  # extra circle
    x0[60:] = 0.04  # initial radii
    
    bounds = [(0.0, 1.0)] * 52
    bounds.extend([(0.0, 0.5)] * n)
    
    cons = NonlinearConstraint(constraint_func, 0, np.inf)
    
    res = minimize(objective, x0, bounds=bounds, method="trust-constr", constraints=cons, options=dict(maxiter=2000, gtol=1e-12, xtol=1e-12))
    
    centers = res.x[:52].reshape((n, 2))
    radii = res.x[52:]
    
    sum_radii = float(np.sum(radii))
    
    # Post-processing to ensure constraints are satisfied within tolerance
    epsilon = 1e-9
    for constraint in pair_indices:
        x_i, y_i = centers[constraint[0]]
        x_j, y_j = centers[constraint[1]]
        r_i, r_j = radii[constraint[0]], radii[constraint[1]]
        while np.sum((x_i - x_j) ** 2 + (y_i - y_j) ** 2) < (r_i + r_j) ** 2:
            radii[constraint[0]] *= (1 - epsilon)
            radii[constraint[1]] *= (1 - epsilon)
            x_i, y_i = centers[constraint[0]]
            x_j, y_j = centers[constraint[1]]
            r_i, r_j = radii[constraint[0]], radii[constraint[1]]
    
    return centers, radii, max(0, np.sum(radii))
