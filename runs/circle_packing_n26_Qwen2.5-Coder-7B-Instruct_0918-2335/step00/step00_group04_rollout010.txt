import numpy as np
from scipy.optimize import linprog, minimize

def solve_lp(c):
    n = c.shape[0]
    c = -np.ones(n)  # Maximize sum of radii
    A = np.zeros((n*(n-1), n))
    b = np.zeros(n*(n-1))
    Awallx = np.tile(np.concatenate([np.eye(25), -np.eye(25)]), 3)
    Awaly = np.tile(np.concatenate([-np.eye(25), np.eye(25)]), 3)
    for i in range(n):
        A[i*(n-1):(i+1)*(n-1), i] = 1
        for j in range(i+1, n):
            Aij = np.array([[0, 0, 0, 0], [0, 0, 0, 0]])
            Aij[0, [i, j]] = [1, 1]
            Aij[1:, [i, j]] = [c[i], c[j]]
            A[i*(n-1)+j-i-1] = Aij[0]
            A[i*(n-1)+j-i-1+n-1] = Aij[1]
            b[i*(n-1)+j-i-1] = np.linalg.norm(c[i] - c[j])
            b[i*(n-1)+j-i-1+n-1] = np.linalg.norm(c[i] - c[j])
    A = np.vstack((A, Awallx, Awaly))
    b = np.hstack((b, np.zeros(n*n), np.zeros(n)))
    bounds = np.column_stack((np.zeros(n), np.ones(n) * 0.5))
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return -result.x

def run_packing():
    np.random.seed(42)
    centers = np.zeros((36, 2))
    for i in range(6):
        for j in range(6):
            centers[i*6 + j] = (i * (1/6) + 0.25, j * (1/6) + 0.25)
    for _ in range(10):
        selected_idx = np.argmax(np.apply_along_axis(np.sum, 1, np.abs(centers - centers[:26, :]) / 2))
        centers = np.delete(centers, selected_idx, axis=0)
    centers += 0.001 * np.random.randn(*centers.shape)
    centers = np.clip(centers, 0, 1)
    r = solve_lp(centers)
    order = np.argsort(r)[::-1]
    r = r[order]
    centers = centers[order][:26]
    sum_radii = np.sum(r)
    return centers, r, sum_radii
