import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def solve_lp(centers):
        n = centers.shape[0]
        c = np.ones(n)               # Coefficients for the maximisation problem
        A = []
        b = []
        
        for i in range(n):
            for j in range(n):
                if i != j:
                    A.append([centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]])
                    b.append(0)
                else:
                    for k in zip(*np.meshgrid(np.arange(n), np.arange(n))):
                        if (k !== (i, j)):
                            A.append([centers[i, 0] - centers[k, 0], centers[i, 1] - centers[k, 1]])
                            b.append(0)
        
        for i in range(n):
            A.append([1, 0])
            b.append(1 - centers[i, 0])
            A.append([0, 1])
            b.append(1 - centers[i, 1])
            A.append([-1, 0])
            b.append(centers[i, 0])
            A.append([0, -1])
            b.append(centers[i, 1])

        A = np.array(A)
        b = np.array(b)

        res = linprog(c, A_ub=A, b_ub=b, method='highs')
        return res.x, res.fun
    
    def compute_forces(centers, radii):
        n = centers.shape[0]
        F = np.zeros_like(centers, dtype=float)
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d_ij < radii[i] + radii[j]:
                    F[i] += (centers[i] - centers[j]) * (radii[i] + radii[j] - d_ij) / d_ij
                    F[j] -= (centers[i] - centers[j]) * (radii[i] + radii[j] - d_ij) / d_ij
            F[i] += np.clip(0, 0.5 - radii[i], radii[i]) * np.array([1, 0])
            F[i] += np.clip(0, 0.5 - radii[i], radii[i]) * np.array([0, 1])
            F[i] += np.clip(0, 0.5 - radii[i], radii[i]) * np.array([-1, 0])
            F[i] += np.clip(0, 0.5 - radii[i], radii[i]) * np.array([0, -1])
        return F

    def project(centers):
        return np.clip(centers, 0, 1)

    def will_skip_linear_solver(last_r, last_radii):
        return (last_r == last_radii).all()

    np.random.seed(42)
    centers = (np.arange(6) + 0.5) / 6
    centers = np.meshgrid(centers, centers)
    centers = centers[0].flatten()[:, None], centers[1].flatten()[:, None]
    centers = centers[0] + centers[1] * 50 / 36 - 25 / 36, centers[1] + centers[0] * 50 / 36 - 25 / 36
    centers = centers[0] + centers[1] * 50 / 36 - 25 / 36, centers[1] + centers[0] * 50 / 36 - 25 / 36
    centers[0, 3:] += np.random.normal(0, 0.01, size=(23, 1))
    centers[1, 3:] += np.random.normal(0, 0.01, size=(23, 1))
    goals = np.array([(0, 0) , (0.3, 0.3), (0.6, 0.6), (0.9, 0.9)], dtype=float)

    best_r = None
    best_radii = None
    best_sum = -np.inf
    for _ in range(3):  # Run several initial seeds to avoid degenerate packings
        for _ in range(500):
            r, cur_sum = solve_lp(centers)
            if will_skip_linear_solver(last_r, r):
                break
            last_r = r
            F = compute_forces(centers, r)
            centers = project(centers - 0.01 * F)
            if will_skip_linear_solver(last_r, r):
                break
            
            cur_sum = r.sum()
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_r = r.copy()
                best_radii = centers[2].copy()

    return best_r.reshape(26, 2), best_radii, best_sum
