import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the packing parameters
    n = 26
    target_sum_radii = 2.636
    
    # Initialize centers and radii randomly within the unit square
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n)
    
    # Apply constraints to ensure valid packing
    while radii.sum() > target_sum_radii:
        radii = np.random.rand(n)
    
    # Ensure circles are within the unit square and non-overlap
    while True:
        # Shuffle centers to try new layouts
        np.random.shuffle(centers)
        
        # Update radii randomly
        radii = np.random.rand(n)
        
        # Check non-overlap and boundary constraints
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        
        if valid and radii.sum() <= target_sum_radii:
            break
    
    return centers, radii, radii.sum()
