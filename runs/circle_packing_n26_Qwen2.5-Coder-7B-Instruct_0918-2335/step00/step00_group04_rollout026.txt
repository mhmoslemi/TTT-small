import numpy as np
from scipy.optimize import linprog
from scipy.spatial import distance

def calculate_objective_and_constraints(centers, radii):
    n = centers.shape[0]
    obj = radii
    A = np.zeros((2*n + 4*n, n))
    b = np.zeros(2*n + 4*n)
    
    # Inequality constraints: r_i + r_j <= d_ij for all pairs (i, j)
    for i in range(n):
        for j in range(i+1, n):
            d_ij = distance.euclidean(centers[i], centers[j])
            A[2*(i*n + j)] = np.array([1, 1,:
            A[2*(i*n + j) + 1] = np.array([-1, -1,:
            b[2*(i*n + j)] = -d_ij
            b[2*(i*n + j) + 1] = -d_ij
    
    # Inequality constraints: r_i <= x_i
    for i in range(n):
        A[2*n + 2*i] = np.array([0, 0, 1, -1])
        b[2*n + 2*i] = 0
    
    # Inequality constraints: r_i <= 1 - x_i
    for i in range(n):
        A[2*n + 2*i + 1] = np.array([0, 0, 1, 1])
        b[2*n + 2*i + 1] = 1
    
    # Inequality constraints: r_i <= y_i
    for i in range(n):
        A[2*n + 4*i + 2] = np.array([0, 0, 1, -1])
        b[2*n + 4*i + 2] = 0
    
    # Inequality constraints: r_i <= 1 - y_i
    for i in range(n):
        A[2*n + 4*i + 3] = np.array([0, 0, 1, 1])
        b[2*n + 4*i + 3] = 1
    
    return obj, A, b

def calculate_repulsive_forces(centers, radii):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Repulsive forces between overlapping circles
    for i in range(n):
        for j in range(i+1, n):
            d_ij = distance.euclidean(centers[i], centers[j])
            s_ij = radii[i] + radii[j] - d_ij
            if s_ij > 0:
                forces[i] += (centers[i] - centers[j]) * s_ij / d_ij
                forces[j] -= (centers[i] - centers[j]) * s_ij / d_ij
    
    # Repulsive walls
    for i in range(n):
        r_i = radii[i]
        x, y = centers[i]
        if x - r_i < 0:   forces[i, 0] += 1e-4
        if x + r_i > 1:   forces[i, 0] += 1e-4
        if y - r_i < 0:   forces[i, 1] += 1e-4
        if y + r_i > 1:   forces[i, 1] += 1e-4
    
    return forces

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    np.random.seed(42)
    
    # Phase A: Coarse dense grid
    grid_size = 6
    grid_spacing = 1 / grid_size
    centers = np.array([(i+0.5)*grid_spacing, (j+0.5)*grid_spacing] for i in range(grid_size) for j in range(grid_size)).reshape(grid_size**2, 2)[:n_circles, :]
    
    # Phase B: Reduce to 26
    distances = distance.squareform(distance.pdist(centers))
    to_remove = np.argsort(np.sum(distances, axis=1))[::-1]
    to_remove = to_remove[:36+n_circles-len(centers)]
    reduced_centers = np.delete(centers, to_remove, axis=0)
    
    # Phase C: Random perturbation
    noise = np.random.normal(0, 0.01, reduced_centers.shape)
    noisy_centers = reduced_centers + noise
    noisy_centers = np.clip(noisy_centers, 0, 1)
    
    # Assertions
    assert noisy_centers.shape == (n_circles, 2), f"Expected {n_circles} centers but got {noisy_centers.shape[0]}"
    assert noisy_centers.min() >= 0, "Centers must be non-negative"
    assert noisy_centers.max() <= 1, "Centers must be within [0,1]"
    
    max_iterations = 200
    tau0 = 1e-3
    gamma = 0.95
    tol = 1e-8
    eps = 0.02
    
    for k in range(max_iterations):
        # Inner LP for radii given centres
        obj, A, b = calculate_objective_and_constraints(noisy_centers, np.zeros(n_circles))
        result = linprog.obj = obj, A_eq = None, b_eq = None, A_ub = A, b_ub = b, bounds=[(0, 0.5)]*n_circles, method='highs')
        if result.success:
            radii = result.x
        else:
            raise Exception("LP failed")
        
        # Outer force calculation
        forces = calculate_repulsive_forces(noisy_centers, radii)
        
        # Center update
        tau = tau0 * gamma**k
        noisy_centers += tau * forces
        noisy_centers = np.clip(noisy_centers, 0, 1)
    
    # Run final LP for best configuration
    final_obj, final_A, final_b = calculate_objective_and_constraints(noisy_centers, np.zeros(n_circles))
    final_result = linprog(final_obj, A_eq=None, b_eq=None, A_ub=final_A, b_ub=final_b, bounds=[(0, 0.5)]*n_circles, method='highs')
    if final_result.success:
        final_radii = final_result.x
        final_centers = noisy_centers
    else:
        raise Exception("Final LP failed")
    
    # Sum_radii
    sum_radii = np.round(np.sum(final_radii), 9)
    
    # Validation
    validation, message = validate_packing(final_centers, final_radii)
    if not validation:
        raise Exception(f"Validation failed: {message}")
    
    return final_centers, final_radii, sum_radii
