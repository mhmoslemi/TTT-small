    linear_constraints = x - r - eps >= 0.0
    quadratic_constraints = distances - (r + r[:, np.newaxis]) ** 2 >= eps
    linear_penalties = np.sum(np.maximum(0, -linear_constraints), axis=(1, 2))
    quadratic_penalties = np.sum(np.maximum(0, -quadratic_constraints), axis=(1, 2))
    return -np.sum(r) + K * (np.sum(linear_penalties) + np.sum(quadratic_penalties))

def validate_radius_increases(centers, radii):
    n = len(centers)
    for i in range(n):
        min_distance = min(centers[i][0] - eps, 1 - centers[i][0] - eps, centers[i][1] - eps, 1 - centers[i][1] - eps)
        for j in range(n):
            if i != j:
                min_distance = min(min_distance, np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2) - radii[j] - eps)
        radii[i] = min(min_distance, radii[i])
    return centers, radii

def validate_non_overlap(centers, radii):
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - eps:
                min_distance = radii[i] + radii[j] - dist
                radius_i = radii[i] + min_distance / 2
                radius_j = radii[j] - min_distance / 2
                direction_x = (centers[j][0] - centers[i][0]) / dist
                direction_y = (centers[j][1] - centers[i][1]) / dist
                centers[i][0] -= radius_i * direction_x
                centers[i][1] -= radius_i * direction_y
                centers[j][0] -= radius_j * direction_x
                centers[j][1] -= radius_j * direction_y
    return centers, radii

def validate_boundary_constraints(centers, radii, eps=1e-15):
    for i in range(len(centers)):
        centers[i][0] = max(eps, min(1 - eps, centers[i][0]))
        centers[i][1] = max(eps, min(1 - eps, centers[i][1]))
    return centers, radii
