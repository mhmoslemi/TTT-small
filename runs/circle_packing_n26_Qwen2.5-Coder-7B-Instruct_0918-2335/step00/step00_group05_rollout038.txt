import numpy as np
import scipy.optimize as op
from math import exp, sqrt, pi

# Validate the packing
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

# Inner LP function
def inner_lp(centers):
    n = centers.shape[0]
    d_ij = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1))
    d_ij[np.diag_indices(n)] = np.inf

    A = np.kron(np.eye(n) + np.eye(n, k=1) + np.eye(n, k=-1), np.ones((2, 2)))
    b = np.zeros(2 * n)
    A = A[:n, :n]
    b = b[:n]

    c = np.ones(n)
    A_eq = d_ij.T.reshape(n * (n - 1) // 2, 2 * n)
    b_eq = d_ij.T.reshape(n * (n - 1) // 2)

    bounds = [(0, None) for _ in range(n)]

    res = op.linprog(c, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'warm_start': True})
    if res.success and not np.isnan(res.x).any() and not (res.x < 0).any():
        return res.x.reshape(-1, 2).mean()
    else:
        return -np.inf

# Main packing function
def run_packing():
    np.random.seed(42)
    T0 = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = 0.02
    n = 26

    # Initial hexagonal lattice
    x = np.arange(1 / (12 * 0.4), 1 - 1 / (12 * 0.4), 1 / (12 * 0.4))
    y = np.arange(1 / (24 * 0.4) + 1 / (2 * np.sqrt(3) * 12 * 0.4), 1 - 1 / (24 * 0.4) - 1 / (2 * np.sqrt(3) * 12 * 0.4), 1 / (24 * 0.4))
    xmesh, ymesh = np.meshgrid(x, y)
    center_indices = np.random.choice(np.arange(xmesh.size), n, replace=False)
    initial_centers = np.zeros((n, 2))
    initial_centers[:, 0] = xmesh.flatten()[center_indices]
    initial_centers[:, 1] = ymesh.flatten()[center_indices]

    current_centers = initial_centers.copy()
    best_sum = -np.inf
    current_sum = inner_lp(current_centers)
    best_centers = initial_centers.copy()

    T = T0
    while T > T_min:
        for _ in range(M):
            i = np.random.randint(n)
            delta = sigma_move * np.random.randn(2)
            new_center = current_centers[i].copy() + delta
            new_center[0] = max(0, min(1, new_center[0]))
            new_center[1] = max(0, min(1, new_center[1]))
            new_centers = current_centers.copy()
            new_centers[i] = new_center
            new_sum = inner_lp(new_centers)
            if new_sum > current_sum or np.random.rand() < exp((new_sum - current_sum) / T):
                current_sum = new_sum
                current_centers = new_centers.copy()
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
        T *= c

    return best_centers, np.zeros(n), best_sum

# Run the packing
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
