def objective(x):
    xs, ys, rs = np.split(x, [N, 2 * N])
    penalty_boundary = np.sum(np.maximum(0, rs - xs) ** 2) + np.sum(np.maximum(0, xs + rs - 1) ** 2) + np.sum(np.maximum(0, rs - ys) ** 2) + np.sum(np.maximum(0, ys + rs - 1) ** 2)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]

    penalty_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)]) ** 2)
