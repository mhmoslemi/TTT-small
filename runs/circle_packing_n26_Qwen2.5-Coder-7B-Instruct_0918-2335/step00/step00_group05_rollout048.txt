import numpy as np
from scipy.optimize import minimize

def hex_lattice(n):
    r = 1 / (2 * np.sqrt(3))
    q = np.arange(n)
    r1 = (q[:n//2] + (q[n//2:] - (q[:n//2] * 2) * 0.5) % (n//2)) % n
    l = 0.5 * np.arange(n)
    x = np.sort(np.concatenate([q + (l/r) % n, r1 + (l - n(r1 != 0)) / r % n]))
    y = np.full_like(x, (np.random.rand(x.size) if n < 50 else np.arange(x.size))) * (np.sqrt(3)/2)
    inside = (x < 1) & (x > r) & (y < 1) & (y > r)
    return x[inside][:26], y[inside][:26]

def boundary_adjust(x, y, r):
    return np.clip(r, 0.01, np.minimum(np.minimum(x, 1-x), np.minimum(y, 1-y))

def distance_constraint(z, r):
    n = len(r)
    dists = np.zeros(n*(n-1)//2)
    for i in range(n):
        for j in range(i+1, n):
            dists[i*(n-1)//2 + j - i - 1] = ((z[2*i] - z[2*j])**2 + (z[2*i+1] - z[2*j+1])**2 - (r[i] + r[j])**2)
    return dists

def objective(z, r, eps):
    return -np.sum(r) + eps * np.sum(d for d in distance_constraint(z, r) if d < 0)

def solve_stage(z0, eps):
    bounds = [(0.01, 1-0.01) for _ in range(len(z0))]
    res = minimize(lambda z: objective(z, z[len(z)//2:], eps), z0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda z: distance_constraint(z, z[len(z)//2:])})
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers, radii = hex_lattice(n)
    radii = boundary_adjust(centers[::2], centers[1::2], 0.05)
    z0 = np.concatenate((centers.flatten(), radii))
    eps = 1e-8
    z = solve_stage(z0, eps)
    for stage in [5e-4, 1e-5, 1e-6, 1e-7, 1e-8]:
        z = solve_stage(z, stage)
        r = z[len(z)//2:]
        centers = z[:len(z)//2].reshape(26, 2)

        radii = r

        if res.fun > -np.sum(radii):
            z = res.x

    return centers, radii, np.sum(radii)

def inflate_balls(r, centers):
    n = len(r)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(((centers[i] - centers[j])**2).sum())
            if dist < r[i] + r[j]:
                dr = (r[i] + r[j] - dist) / 2
                centers[j] += (centers[i] - centers[j]) * (dr / dist)
                centers[i] -= (centers[i] - centers[j]) * (dr / dist)
                r[i] -= dr
                r[j] -= dr

    r = np.maximum(r, np.zeros(n))
    return r
