import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def hex_lattice(s, n):
    """Generate the centers of hexagonally packed circles in a unit square."""
    x = np.arange(s/2, 1 - s/2, s)[np.newaxis, :]
    y = np.concatenate((np.arange(s/2, 1 - s/2, s * np.sqrt(3)/2)[:-1], np.arange(s/2, 1 - s/2, s * np.sqrt(3)/2)))
    x = x.repeat(y.shape[0], axis=0)
    y = np.concatenate((y,) * x.shape[0])
    mask = (x > 0) & (x < 1) & (y > 0) & (y < 1)
    x, y = x[mask], y[mask]
    return x[:n], y[:n]

def penalised_obj(z, n, delta):
    """Penalised objective function."""
    positions = z[:n*n].reshape(n, 2)
    radii = z[n*n:].flatten()
    penalty = 0
    for i in range(n):
        penalty += max(0, -(positions[i, 0] - radii[i] - 1e-12)) * added_p
        penalty += max(0, -(1 - positions[i, 0] - radii[i] - 1e-12)) * add
        penalty += max(0, -(positions[i, 1] - radii[i] - 1e-12)) * added
        penalty += max(0, -(1 - positions[i, 1] - radii[i] - 1e-12)) * add
        for j in range(i + 1, n):
            diff = positions[j] - positions[i]
            dist = np.linalg.norm(diff)
            penalty += max(0, dist - (radii[i] + radii[j]) - delta) * added_p
    return -(np.sum(radii) + penalty)

def solve_stage(z0, delta):
    """Solve the optimisation problem for a given delta."""
    n = int(np.sqrt(len(z0) // 2))
    res = minimize(penalised_obj, z0, args=(n, delta), method='SLSQP', bounds=None)
    return res.x

def inflate_balls(z):
    """Inflate balls until they touch boundaries or each other."""
    n = int(np.sqrt(len(z) // 2))
    positions = z[:n*n].reshape(n, 2)
    radii = z[n*n:].flatten()
    for i in range(n):
        for j in range(i + 1, n):
            dist = distance_matrix(positions[[i, j]], positions[[i, j]])
            if dist < radii[i] + radii[j] - 1e-12:
                diff = positions[j] - positions[i]
                factor = (radii[i] - radii[j]) / dist
                positions[j] = positions[i] + diff * factor
        min_x = max(positions[i, 0] - radii[i], 0)
        max_x = min(positions[i, 0] + radii[i], 1)
        min_y = max(positions[i, 1] - radii[i], 0)
        max_y = min(positions[i, 1] + radii[i], 1)
        delta_x = max(0, min_x - positions[i, 0])
        delta_y = max(0, min_y - positions[i, 1])
        if delta_x > 0:
            positions[i, 0] += delta_x
        if delta_y > 0:
            positions[i, 1] += delta_y
        radii[i] -= delta_x
        radii[i] -= delta_y
    return np.concatenate((positions.flatten(), radii))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    s = 0.1
    n = 26
    x0, y0 = hex_lattice(s, n)
    radii0 = np.full(n, s/2)
    z0 = np.concatenate((np.concatenate((x0, y0)), radii0))
    z0 = boundary_greedy_radius_augment(z0, n)
    best_z = z0

    deltas = [1e-8, 5e-4, 1e-12]
    for delta in deltas:
        z = solve_stage(best_z, delta)
        z = inflate_balls(z)
        if penalised_obj(z, n, delta) < penalised_obj(best_z, n, delta):
            best_z = z

    for _ in range(5):
        z_run = best_z + np.random.uniform(-1e-3, 1e-3, 2*n + n)
        z_run = boundary_greedy_radius_augment(z_run, n)
        z_run = solve_stage(z_run, 1e-12)
        z_run = inflate_balls(z_run)
        if penalised_obj(z_run, n, 1e-12) < penalised_obj(best_z, n, 1e-12):
            best_z = z_run

    positions = best_z[:n*n].reshape(n, 2)
    radii = best_z[n*n:].flatten()
    positions = np.clip(positions, 0, 1)

    return positions, radii, np.sum(radii)

def boundary_greedy_radius_augment(z, n):
    positions = z[:n*n].reshape(n, 2)
    radii = z[n*n:].flatten()
    for i in range(n):
        min_x = positions[i, 0] - radii[i]
        min_y = positions[i, 1] - radii[i]
        max_x = 1 - positions[i, 0] - radii[i]
        max_y = 1 - positions[i, 1] - radii[i]
        min_distances = np.minimum(np.minimum(min_x, min_y), np.minimum(max_x, max_y))
        radii[i] = min(radii[i], min_distances)
    return np.concatenate((positions.flatten(), radii))

centers, radii, sum_radii = run_packing()
print(centers.shape, radii.shape, sum_radii)
