import numpy as np
from scipy.optimize import minimize

def rsa_seed(n, r_min, iterations):
    circles = []
    for _ in range(n):
        placed = False
        for _ in range(iterations):
            x, y = np.random.rand(2)
            r = r_min
            valid = True
            for cx, cy, cr in circles:
                if np.sqrt((x - cx)**2 + (y - cy)**2) < r + cr + 1e-12:
                    valid = False
                    break
            if valid:
                circles.append((x, y, r))
                placed = True
                break
        if not placed:
            return rsa_seed(n, r_min, iterations)
    return np.array([circle[:2] for circle in circles]), np.array([circle[2] for circle in circles])

    T = 2000
    Δt = 1e-3
    for _ in range(T):

def global_scaling(centers, radii):
    n = centers.shape[0]
    s_max = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            s_max = min(s_max, d / (radii[i] + radii[j]))
        s_max = min(s_max, (1 - radii[i]) / radii[i])
    s_max = min(s_max, (1 - radii[j]) / radii[j])
    centers = 0.5 + s_max * (centers - 0.5)
    radii *= s_max
    return centers, radii

def exact_penalty_l_bfgs(centers, radii):
    n = centers.shape[0]
    def objective(x):
        r = x[n:]
        return -r.sum()
    
    def penalty(x):
        r = x[n:]
        p_overlap = 0
        p_boundary = 0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(x[:n] + np.arange(2 * n)[2 * i:2 * i + 2] - x[:n] - np.arange(2 * n)[2 * j:2 * j + 2])
                p_overlap += max(0, (r[i] + r[j]) - d)**2
            p_boundary += max(0, r[i] - x[2 * i])**2 + max(0, r[i] - (1 - x[2 * i]))**2
            p_boundary += max(0, r[i] - x[2 * i + 1])**2 + max(0, r[i] - (1 - x[2 * i + 1]))**2
        return p_overlap + p_boundary
    
    def penalty_gradient(x):
        r = x[n:]
        g = np.zeros(2 * n)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(x[:n] + np.arange(2 * n)[2 * i:2 * i + 2] - x[:n] - np.arange(2 * n)[2 * j:2 * j + 2])
                g[:n] += r[i] / d * (x[:n] + np.arange(2 * n)[2 * i:2 * i + 2] - x[:n] - np.arange(2 * n)[2 * j:2 * j + 2]) if d > 0 else np.zeros(2)
                g[:n] -= r[j] / d * (x[:n] + np.arange(2 * n)[2 * j:2 * j + 2] - x[:n] - np.arange(2 * n)[2 * i:2 * i + 2]) if d > 0 else np.zeros(2)
            g[2 * i] += r[i] / (1 - x[2 * i]) if x[2 * i] < 1 else 0
            g[2 * i + 1] += r[i] / (1 - x[2 * i + 1]) if x[2 * i + 1] < 1 else 0
            g[2 * i] += -r[i] / x[2 * i] if x[2 * i] > 0 else 0
            g[2 * i + 1] += -r[i] / x[2 * i + 1] if x[2 * i + 1] > 0 else 0
        return g
    
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(objective, x0, method='L-BFGS-B', jac=penalty_gradient, bounds=[(0, 1)] * 2 * n + [(0, 0.5)] * n)
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]
    return centers, radii

def run_packing():

    centers, radii = rsa_seed(26, 1e-3, 10000)
