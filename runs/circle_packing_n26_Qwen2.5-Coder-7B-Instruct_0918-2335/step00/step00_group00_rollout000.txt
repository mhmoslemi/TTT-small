import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def run_packing():
    n = 26

    def constraint_func(v):
        x = v[:52].reshape((n, 2))
        r = v[52:]
        arr = []
        arr.append(x[:, 0] - r)
        arr.append(1 - x[:, 0] - r)
        arr.append(x[:, 1] - r)
        arr.append(1 - x[:, 1] - r)
        
        i_indices = np.fromiter(range(n), dtype=np.int64)
        j_indices = np.fromiter(range(i_indices.shape[0] - 1), dtype=np.int64) + 1
        dx = x[i_indices, 0] - x[j_indices, 0]
        dy = x[i_indices, 1] - x[j_indices, 1]
        dist2 = dx ** 2 + dy ** 2
        pair_vals = dist2 - (r[i_indices] + r[j_indices]) ** 2
        arr.append(pair_vals)
        
        return np.concatenate(arr)

    bounds = []
    for _ in range(52):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(n):
        bounds.append((0.0, 0.5))

    x0 = np.zeros(78)
    positions = np.array([(j / 5 + 0.5 / 5, i / 5 + 0.5 / 5) for i in range(5) for j in range(5)])
    x0[:positions.shape[0] * 2] = positions.ravel()
    x0[positions.shape[0] * 2:] = 0.04
    res = minimize(lambda v: -np.sum(v[52:]), x0, method='trust-constr', bounds=bounds,
                   constraints=NonlinearConstraint(constraint_func, 0, np.inf),
                   options={'maxiter': 5000, 'gtol': 1e-12, 'xtol': 1e-12})

    centers = res.x[:52].reshape((n, 2))
    radii = res.x[52:]
    
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
