import numpy as np

def init_circles(variant):
    centers = np.zeros((26, 2))
    for i in range(5):
        for j in range(5):
            centers[i*5 + j] = np.array([i / 4, j / 4])
    centers[25] = np.array([0.6, 0.9]) if variant == 'A' else np.array([0.4, 0.9])
    return centers

def inflate(circles, Δr, ε):
    n = circles.shape[0]
    for iteration in range(5000):
        radius_changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist_ij = np.linalg.norm(circles[i] - circles[j])
                if dist_ij < circles[i][2] + circles[j][2] - ε:
                    delta = (circles[i][2] + circles[j][2] - dist_ij) / (2 * dist_ij)
                    circles[i][:2] += delta * (circles[i][:2] - circles[j][:2]) / 2
                    circles[j][:2] -= delta * (circles[i][:2] - circles[j][:2]) / 2
                    radius_changed = True
        for i in range(n):
            circles[i][:2] = np.clip(circles[i][:2], circles[i][2], 1 - circles[i][2])
            circles[i][2] = min(circles[i][2] + Δr, min(circles[i][0], 1 - circles[i][0], circles[i][1], 1 - circles[i][1]) - 1e-6)
        if not radius_changed:
            break
    return circles[:, 2], circles[:, :2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum_radii = 0
    best_centers, best_radii = None, None
    
    for variant in ['A', 'B']:
        circles = init_circles(variant)
        radii, centers = inflate(circles, 0.0005, 1e-6)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers
            best_radii = radii
    
    if best_sum_radii < 2.636:
        # Fallback to hexagon pattern
        side_length = np.sqrt(3) / 2 * (1 / 5)
        radius = side_length / 2
        centers = np.zeros((26, 2))
        row_height = side_length * np.sqrt(3)
        col_width = side_length * 2
        col_offset = col_width * 0.5
        for i in range(5):
            for j in range(5):
                x = col_offset + col_width * i
                y = row_height * j + (2 * np.cos(np.pi / 6) * side_length) * (i + j // 2)
                if i % 2 == 1:
                    y -= row_height * 0.5
                centers[i*5 + j] = np.array([x % 1, y % 1])
        best_centers = centers
        best_radii = np.ones(26) * radius
        best_sum_radii = 5 * (5 * radius + 10 * np.sqrt(3) * radius**2)
    
    return best_centers, best_radii, best_sum_radii
