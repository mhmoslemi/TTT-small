import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    random_seeds = [0, 1, 2, 3]

    def penalty(x):
        pos_range = np.maximum(0, x[N:2*N] - x[2*N:3*N]**2) + \
                    np.maximum(0, x[2*N:3*N]**2 - x[:N]**2) + \
                    np.maximum(0, x[N:2*N] - x[3*N:4*N]**2) + \
                    np.maximum(0, x[3*N:4*N]**2 - x[:N]**2)
        dist = np.abs(x[:N, None] - x[:N]) + np.abs(x[N:2*N, None] - x[N:2*N])**2
        overlap = np.maximum(0, x[N:2*N] + x[2*N:3*N] - dist)
        return np.sum(pos_range) + np.sum(overlap)

    def objective(x):
        return -np.sum(x[N:2*N]) + lambda_penalty * penalty(x)

    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N

    best_x = []
    best_cost = -np.inf

    for seed in random_seeds:
        np.random.seed(seed)
        result = differential_evolution(
            objective,
            bounds,
            popsize=DE_popsize,
            maxiter=DE_maxiter,
            tol=1e-5
        )
        if result.fun > best_cost:
            best_x = result.x
            best_cost = result.fun

    x = best_x
    centres = np.column_stack([x[:N], x[N:2*N]])
    radii = x[2*N:]
    sum_radii = radii.sum()

    return centres, radii, sum_radii
