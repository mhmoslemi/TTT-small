```python
import numpy as np
from scipy.optimize import minimize

n = 26
pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.zeros(n)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[2*n:])

def constraints(v):
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    boundary_constraints = np.clip(centers - radii, 0, 1) - centers
    pairwise_constraints = (np.sum(centers[pair_idx_i, :] - centers[pair_idx_j, :]**2, axis=1) - 
                            (radii[pair_idx_i] + radii[pair_idx_j])**2)
    return np.concatenate((boundary_constraints.flatten(), pairwise_constraints))

bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_value = -np.inf
    best_v = None
    for seed in range(8):
        v0 = initial_guess(seed)
        res = minimize(
            fun        = objective,
            x0         = v0,
            method     = 'SLSQP',
            constraints= {'type':'ineq', 'fun':constraints},
            bounds     = bounds,
            options    = {
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if not res.success:
            continue
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x
    centers = best_v[:2*n].reshape((n, 2))
    eps = 1e-7
    radii = np.maximum(0.0, best_v[2*n:] - eps)
    return centers, radii, np.sum(radii)
