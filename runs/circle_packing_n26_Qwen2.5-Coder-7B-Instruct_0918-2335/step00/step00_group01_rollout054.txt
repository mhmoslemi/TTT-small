import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from multiprocessing import Pool

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def target_function(x):
    n = 26
    centers = x[: n * 2].reshape(n, 2)
    radii = x[n * 2:]
    penalty_bound = np.maximum(0, radii - np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]))
    penalty_overlap = np.where(distance_matrix(centers, centers) < radii[:, np.newaxis] + radii[np.newaxis, :], 1, 0) - np.eye(n)
    penalty_overlapOrElse = penalty_overlap.sum() * (radii[:, np.newaxis] + radii[np.newaxis, :] - distance_matrix(centers, centers))
    return -np.sum(radii) + np.sum(penalty_bound) + np.sum(penalty_overlapOrElse)

def repair(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        if radii[i] > np.minimum(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]):
            radii[i] = np.minimum(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            dist = distance_matrix(centers[i:i+1], centers[j:j+1])[0,0]
            if dist < radii[i] + radii[j]:
                shrink_factor = (radii[i] + radii[j]) / dist
                radii[i] *= shrink_factor
                radii[j] *= shrink_factor
                if radii[i] < 0 or radii[j] < 0:
                    radii[i] = 0
                    radii[j] = 0
                    repair(centers, radii)
    return centers, radii

def eval_is_feasible(x):
    n = x.shape[0] // 2
    centers, radii = x[: n * 2].reshape(n, 2), x[n * 2:]
    centers, radii = repair(centers, radii)
    return validate_packing(centers, radii)[0]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([[0.1, 0.1], [0.5, 0.1], [0.9, 0.1], [0.2, 0.3], [0.6, 0.3], [1.0, 0.3], [0.15, 0.5], [0.55, 0.5], [0.95, 0.5], [0.3, 0.7], [0.7, 0.7], [1.0, 0.7], [0.2, 0.9], [0.6, 0.9], [1.0, 0.9], [0.1, 0.2], [0.5, 0.2], [0.9, 0.2], [0.1, 0.6], [0.5, 0.6], [0.9, 0.6], [0.1, 0.9], [0.5, 0.9], [0.9, 0.9]])
    radii = 0.01 * np.ones(n)
    history = []

    for epoch in range(1000):
        x0 = np.concatenate((centers.flatten(), radii))
        res = minimize(target_function, x0, method='L-BFGS-B', bounds=[(0, 1), (0, 1)] * n + [(0, 1)] * n)
        if res.success:
            centers, radii = res.x[: n * 2].reshape(n, 2), res.x[n * 2:]
            centers, radii = repair(centers, radii)
            if validate_packing(centers, radii)[0]:
                history.append((epoch, res.fun + np.sum(radii)))
            if len(history) > 1 and abs(history[-2][1] - history[-1][1]) < 1e-6:
                break

    final_centers, final_radii = repair(centers, radii)
    return final_centers, final_radii, sum(final_radii)

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Centers:\n{centers}")
    print(f"Radii:\n{radii}")
    print(f"Sum of Radii: {sum_radii}")
    assert validate_packing(centers, radii)[0]

This code defines a genetic algorithm to solve the circle packing problem, using a penalty method to enforce constraints.
