import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.array([
        [r_c, r_c],
        [1 - r_c, r_c],
        [r_c, 1 - r_c],
        [1 - r_c, 1 - r_c],
        [0.5, r_e],
        [0.5, 1 - r_e],
        [r_e, 0.5],
        [1 - r_e, 0.5]
    ])
    radii = np.array([r_c, r_c, r_c, r_c, r_e, r_e, r_e, r_e])
    return centers, radii

def compute_sum_radii(radii):
    return np.sum(radii)

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    sum_r = compute_sum_radii(radii)
    penalty = 0.0
    for i in range(20):
        for j in range(i + 1, 20):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-8:
                penalty += (radii[i] + radii[j] - dist) ** 2
    return -(sum_r - penalty)

def constraint_func(v):
    centers = v[:52].reshape(26, 2)
    radii = v[52:]
    constraints = []
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        constraints.append(r)
        constraints.append(r <= 0.5)
        constraints.append(x - r >= 0)
        constraints.append(x + r <= 1)
        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist - (radii[i] + radii[j]))
    return np.array(constraints)

def run_packing():

        (0.30, 0.49),
        (0.10, 0.25),
        (0.05, 0.15),
    ]
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)
    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [
        (None, None) for _ in range(52) if _ < 52
    ]
    cons = [
        {'type': 'ineq', 'fun': lambda v: v[i] for i in range(52)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 for i in range(26, 52)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i] for i in range(4)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i+1] for i in range(4)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i+2] for i in range(4)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i+3] for i in range(4)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i+4] for i in range(4)},
        {'type': 'ineq', 'fun': lambda v: v[i] - 0.5 - v[2*i+5] for i in range(9)}
    ]
    for i in range(26):
        for j in range(i + 1, 26):
            cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: v[i] - v[j] - (v[52+i] + v[52+j])})
    res = minimize(objective, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 2000})
