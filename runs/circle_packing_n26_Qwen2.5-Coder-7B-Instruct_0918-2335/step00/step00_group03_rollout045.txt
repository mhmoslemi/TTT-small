import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.array([
        [r_c, r_c], [1-r_c, r_c], [r_c, 1-r_c], [1-r_c, 1-r_c],
        [0.5, 0.5 + r_e], [0.5, 0.5 - r_e], [0.5 + r_e, 0.5], [0.5 - r_e, 0.5],
        [0.267 * 2*r_i, 0.25], [0.267 * 2*r_i, 0.75],
        [0.533 * 2*r_i, 0.25], [0.533 * 2*r_i, 0.75],
        [0.10 + 2*r_i, 0.10 + 2*r_i], [0.10 + 2*r_i, 0.60], 
        [0.10 + 2*r_i, 1.10 - 2*r_i],
        [0.90 - 2*r_i, 0.10 + 2*r_i], [0.90 - 2*r_i, 0.60], 
        [0.90 - 2*r_i, 1.10 - 2*r_i],
        [0.3 * 2*r_i, 0.3 * 2*r_i], [0.3 * 2*r_i, 0.7 * 2*r_i], 
        [0.7 * 2*r_i, 0.3 * 2*r_i], [0.7 * 2*r_i, 0.7 * 2*r_i]
    ])

    radii = np.array([r_c] * 4 + [r_e]*4 + [r_i]*20)
    return centers, radii


def compute_sum_radii(radii):
    return radii.sum()


def objective(params):
    r_c, r_e, r_i = params
    centers, _ = make_seed(r_c, r_e, r_i)
    radii = np.full(26, r_i)
    sum_r = compute_sum_radii(radii)
    return -sum_r


def constraint_func(xi, centers, radii):
    n =centers.shape[0]
    vars = [xi[k*3] for k in range(n)] + [xi[k*3+1] for k in range(n)] + [xi[k*3+2] for k in range(n)]
    centers_var = np.array(vars).reshape((n,3))
    distances = np.sqrt(np.sum((centers_var[:-1, :2] - centers_var[1:, :2]) ** 2, axis=1))
    radii_vec = centers_var[:, 2]
    dist_cond = np.array(distances -radii_vec[:-1] -radii_vec[1:]+1e-12 >= 0)
    bound_cond = np.array(centers_var[:, 0] -radii_vec >= 0) & (centers_var[:, 0] +radii_vec <= 1) & (centers_var[:, 1] -radii_vec >= 0) & (centers_var[:, 1] +radii_vec <= 1)
    bind = np.invert(dist_cond) & np.invert(bound_cond)
    bound_cond[bind] = 0
    return np.concatenate((dist_cond, [int(x) for x in bound_cond]))


def run_packing():
    bounds = [(0.30, 0.49), (0.10, 0.25), (0.05, 0.15)]
    result_de = differential_evolution(objective, bounds=bounds, popsize=10, tol=1e-5)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    # Fine.gained local refinement
    x0 = np.hstack([centers.flatten(), radii * 0.8])
    bounds_slsqp = [(0, 1), (0, 1)] * 26 + [(0, 0.5)] * 26
    cons = [{'type': 'ineq', 'fun': lambda xi: constraint_func(xi, centers, radii)}]
    res = minimize(local_obj, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 2000})
    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]

    sum_r = radii.sum()
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")

    return centers, radii, sum_r


def local_obj(x):
    centers = x[:52].reshape((26, 3))
    radii = centers[:, 2]
    sum_r = compute_sum_radii(radii)
    return -sum_r


if __name__ == '__main__':
    centers, radii, sum_r = run_packing()
