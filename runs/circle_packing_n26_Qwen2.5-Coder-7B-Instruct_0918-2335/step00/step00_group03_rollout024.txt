import numpy as np
from numpy.linalg import norm
from scipy.spatial import ConvexHull

def half_plane_clipping(polygon, a, b):
    inside, outside = [], []
    for start, end in zip(polygon, polygon[1:] + polygon[:1]):
        t = (b - a * start[0]) / (a[1] * start[0] - a[0] * start[1])
        if t >= 0 and t <= 1:
            inside.append(start)
            if end[1] * start[0] <= end[0] * start[1]:
                inside.append((start + t * (end - start)).tolist())
        elif end[1] * start[0] <= end[0] * start[1]:
            outside.append((start + t * (end - start)).tolist())
            outside.append(end)
    outside.append(inside[-1])
    return np.array(outside)

def power_diagram_cell(centers, r, indices, x, y):
    half_planes = []
    for i in indices:
        if i == x or i == y: continue
        a = centers[i] - centers[x]
        b = norm(a)**2 + r[x]**2 - r[i]**2
        half_planes.append((a[1], -a[0], b))
    clippped = np.array([[x, y]])
    for a, b, c in half_planes:
        clippped = half_plane_clipping(clippped, [a, -b], c)
    return ConvexHull(clippped).vertices if not np.allclose(clippped, clippped[-2::-1]) else np.array([len(clippped)-1])

def centroid(polygon):
    x, y = zip(*polygon)
    A = 0.5 * np.sum(x[:len(x)-1] * y[1:] - x[1:] * y[:len(y)-1])
    if A == 0:
        return np.array([np.mean(x), np.mean(y)])
    Cx = (1 / (6 * A)) * np.sum((x[:len(x)-1] + x[1:]) * (x[:len(x)-1] * y[1:] - x[1:] * y[:len(y)-1]))
    Cy = (1 / (6 * A)) * np.sum((y[:len(y)-1] + y[1:]) * (x[:len(x)-1] * y[1:] - x[1:] * y[:len(y)-1]))
    return np.array([Cx, Cy])

def distance_to_polygon(p, polygon):
    return min(norm(np.cross(p - q, r - q)) / norm(r - q) for q, r in zip(polygon, polygon[1:] + polygon[:1]))

    centers = np.random.rand(26, 2)  # uniform in [0,1]^2

            polygon = power_diagram_cell(centers, radii, cell_indices, k, nearest_index)
