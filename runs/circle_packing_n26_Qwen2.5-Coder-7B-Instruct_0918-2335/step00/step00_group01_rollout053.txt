import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    M_bound = M_overlap = 10**2
    sigma_pos = sigma_rad = 0.05
    epsilon = 1e-7
    generations = 800
    early_stop_patience = 150
    mutation_prob = 0.6
    crossover_prob = 0.5
    repair_max_iter = 100

    def initial_population():
        centers = []
        radii = []
        row = 0.1 + (0.8 * np.arange(6) / 10)
        for j in range(0, 30, 2):
            x = 0.1 + (j + np.arange(0 if j % 2 == 0 else 1, 6 if j % 2 == 0 else 5) / 10)
            centers.extend(np.column_stack((
                x[:len(row)] * 2 - 0.9,
                np.full(len(row), row[i]) * 2 - 0.9
            ) for i in range((j + len(x)) // 6)))
        centers = np.array(centers[:n])

        for x, y in centers:
            r = min(x, 1 - x, y, 1 - y) - 1e-4
            radii.extend([r] * (n // len(centers)))

        return centers, np.array(radii)

    def penalty(c, r, M):
        penalty_bound = 0
        penalty_overlap = 0
        for i in range(n):
            if r[i] > min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1]):
                penalty_bound += M * (r[i] - min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1]))**2
            for j in range(i + 1, n):
                dist = np.linalg.norm(c[i] - c[j])
                if dist < r[i] + r[j]:
                    penalty_overlap += M * (r[i] + r[j] - dist)**2
        return penalty_bound, penalty_overlap

    def objective(x):
        c = x[:n*2].reshape((n, 2))
        r = x[n*2:]
        penalty_bound, penalty_overlap = penalty(c, r, M_bound)
        return -(r.sum() + penalty_bound + penalty_overlap)

    def gradient(x):
        c = x[:n*2].reshape((n, 2))
        r = x[n*2:]
        grad_c = np.zeros(n*2)
        grad_r = np.zeros(n)

        for i in range(n):
            dc = np.array([0, 0])
            if r[i] > min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1]):
                dc += 4 * M_bound * (r[i] - min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1])) * (-1 if c[i][0] < c[i][1] else -1 if c[i][0] > c[i][1] else 0)
            for j in range(i + 1, n):
                dist = np.linalg.norm(c[i] - c[j])
                if dist < r[i] + r[j]:
                    dc += 4 * M_bound * (r[i] + r[j] - dist) * (-1 if c[i][j] < c[i][j] else 1)
            grad_c[i*2: i*2+2] = -dc

            dr = np.zeros_like(r)
            if r[i] > min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1]):
                dr[i] += 4 * M_bound * (r[i] - min(c[i][0], 1 - c[i][0], c[i][1], 1 - c[i][1]))
            for j in range(i + 1, n):
                dist = np.linalg.norm(c[i] - c[j])
                if dist < r[i] + r[j]:
                    dr[i] += 4 * M_bound * (r[i] + r[j] - dist)
            grad_r[i] = -dr[i]

        return np.concatenate((grad_c, grad_r))

    def repair(c, r):
        c_new = np.array(c, dtype=float)
        r_new = np.array(r, dtype=float)
        for i in range(n):
            if r_new[i] > min(c_new[i][0], 1 - c_new[i][0], c_new[i][1], 1 - c_new[i][1]):
                r_new[i] = min(c_new[i][0], 1 - c_new[i][0], c_new[i][1], 1 - c_new[i][1]) - 1e-6
            for j in range(i + 1, n):
                dist = np.linalg.norm(c_new[i] - c_new[j])
                if dist < r_new[i] + r_new[j]:
                    shrink_factor = (dist - r_new[i] - r_new[j]) / (2 * r_new[i] + 2 * r_new[j])
                    r_new[i] *= shrink_factor
                    r_new[j] *= shrink_factor
                    if r_new[i] < 0:
                        r_new[i] = 0
                        r_new[(i + 1) % n] += r_new[i]
                    if r_new[j] < 0:
                        r_new[j] = 0
                        r_new[(i + 2) % n] += r_new[j]
        return c_new, r_new

    def evolve():
        best_c, best_r, best_value = centers.copy(), radii.copy(), objective(x)
        best_found = 0
        for gen in range(generations):
            for _ in range(P):
                indices = np.random.choice(P, 2, replace=False)
                c1, r1 = centers[indices[0]], radii[indices[0]]
                c2, r2 = centers[indices[1]], radii[indices[1]]
                x1, x2 = repair(c1, r1), repair(c2, r2)
                if np.random.rand() < crossover_prob:
                    c_new = np.where(np.random.rand(n, 2) < 0.5, x1, x2)
                    r_new = 0.5 * x1 + 0.5 * x2
                else:
                    c_new = x1[:]
                    r_new = x2[:]
                c_new = np.clip(c_new, 0, 1)
                r_new = np.maximum(r_new, 0)
                f_new = objective(np.concatenate((c_new.flatten(), r_new)))
                if f_new < objective(np.concatenate((c1.flatten(), r1))):
                    centers[indices[0]], radii[indices[0]] = c_new, r_new
                if f_new < best_value:
                    best_c, best_r, best_value = c_new, r_new, f_new
                    best_found = 0
                else:
                    best_found += 1
                    if best_found > early_stop_patience:
                        return best_c, best_r, best_value + penalty(best_c, best_r, M_bound)[0] + penalty(best_c, best_r, M_overlap)[1]

            # Update multipliers
            if gen % 20 == 0:
                M_bound *= 3
                M_overlap *= 3

            # Adaptive mutation step size decrease
            sigma_pos *= 0.97
            sigma_rad *= 0.97

        return best_c, best_r, best_value + penalty(best_c, best_r, M_bound)[0] + penalty(best_c, best_r, M_overlap)[1]

    centers, radii = initial_population()
    x = np.concatenate((centers.flatten(), radii))
    best_c, best_r, best_value = evolve()

    # Local improvement using LP for fixed centers
    x_fixed = np.concatenate((best_c.flatten(), np.zeros(n)))
    res = minimize(objective, x_fixed, method="SLSQP", jac=gradient, bounds=[(-1e-6, None)] * n * 2 + [(-1e-6, None)] * n)
    best_c = res.x[:n*2].reshape((n, 2))
    best_r = res.x[n*2:]
    best_value = -res.fun

    # Final radius shrink
    best_r -= epsilon

    return best_c, best_r, best_value
