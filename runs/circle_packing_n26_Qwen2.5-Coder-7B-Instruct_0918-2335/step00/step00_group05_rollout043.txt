    n = 26

    beta = 0.8
    epsilon = 1e-12
    delta_r = 1e-4
    max_iter = 3000
    tol = 1e-10

    np.random.seed(42)
    centers = np.random.uniform(epsilon, 1-epsilon, size=(n, 2))

    for step in range(max_iter):
        forces = np.zeros((n, 2))

                if d < radii[i] + radii[j] + epsilon:
                    overlap = radii[i] + radii[j] - d
                    direction = (centers[j] - centers[i]) / d
                    forces[i] -= alpha * overlap * direction
                    forces[j] += alpha * overlap * direction

                forces[i][0] += alpha * (left_wall + radii[i] + epsilon - centers[i][0])

                forces[i][0] -= alpha * (centers[i][0] - right_wall)

                forces[i][1] += alpha * (bottom_wall + radii[i] + epsilon - centers[i][1])
            top_wall = 1 - radii[i] - epsilon
            if centers[i][1] > top_wall:
                forces[i][1] -= alpha * (centers[i][1] - top_wall)

        velocities = np.multiply(forces, 1.0/1.0) - beta * velocities  # m = 1 by convention
        centers += delta_t * velocities

        radii += delta_r

        radii_allowed = np.minimum(1 - centers[:, 0], 1 - centers[:, 1])

        radii = np.minimum(radii, radii_allowed)
