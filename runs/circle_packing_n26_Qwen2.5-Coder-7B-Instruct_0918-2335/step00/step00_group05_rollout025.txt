import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonal lattice configuration
    s = 0.312
    x = np.arange(s, 1 - s, s)
    y = np.arange(s / 2, 1, s / np.sqrt(3))
    x, y = np.meshgrid(x, y)
    x = x.flatten()
    y = y.flatten()
    x = x[x <= 1 - s]
    y = y[y < 1]
    centers = np.column_stack((x, y))
    if len(centers) < 26:
        s -= 1e-3
        x = np.arange(s, 1 - s, s)
        y = np.arange(s / 2, 1, s / np.sqrt(3))
        x, y = np.meshgrid(x, y)
        x = x.flatten()
        y = y.flatten()
        centers = np.column_stack((x, y))
        centers = centers[centers[:, 0] <= 1 - s]
        centers = centers[centers[:, 1] < 1]
    centers = centers[:26]

    # Initialize radii and temperature
    n = centers.shape[0]
    radii = np.zeros(n)
    radii.fill(0.1)
    T = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = T / 5
    best_sum = -np.inf
    best_centers = centers.copy()
    best_radii = radii.copy()

    # SA loop
    while T > T_min:
        for m in range(M):
            # Copy current state
            cur_centers = centers.copy()
            cur_radii = radii.copy()
            # Randomly choose a circle to move
            i = np.random.randint(n)
            delta = np.random.normal(0, sigma_move, 2)
            delta[0] = np.clip(delta[0], -0.5*s + 2*1e-6, 0.5*s - 2*1e-6)
            delta[1] = np.clip(delta[1], -0.5*s + 2*1e-6, 0.5*s - 2*1e-6)
            cur_centers[i] += delta
            # Clamp to boundary and set to fsda0
            for j in range(2):
                if cur_centers[i, j] < 0:
                    cur_centers[i, j] = 0 + 1e-6
                elif cur_centers[i, j] > 1:
                    cur_centers[i, j] = 1 - 1e-6
            # Update distance matrix and solve LP
            D = distance_matrix(cur_centers, cur_centers)
            r_bounds = np.vstack(((np.zeros(n), np.inf), (np.zeros(n), cur_centers[:, 0] - 1e-6), 
                                  (np.zeros(n), cur_centers[:, 1] - 1e-6), (np.zeros(n), cur_centers[:, 0] - 1e-6), 
                                  (np.zeros(n), cur_centers[:, 1] - 1e-6))).T
            A = np.column_stack((np.ones((n, n)), -np.eye(n), -np.eye(n), np.eye(n), -np.eye(n)))
            b = D.reshape(-1) - r_bounds.reshape(-1)
            r_res = linprog(-radii, A_ub=A, b_ub=b, method="highs", bounds=(0, None), x0=radii)
            cur_radii = r_res.x
            cur_sum = np.sum(cur_radii)
            # Acceptance criterion
            if cur_sum > best_sum:
                best_centers = cur_centers
                best_radii = cur_radii
                best_sum = cur_sum
            elif np.random.rand() < np.exp((cur_sum - best_sum) / T):
                best_centers = cur_centers
                best_radii = cur_radii
                best_sum = cur_sum
        # Cool down
        T *= c

    return best_centers, best_radii, best_sum
