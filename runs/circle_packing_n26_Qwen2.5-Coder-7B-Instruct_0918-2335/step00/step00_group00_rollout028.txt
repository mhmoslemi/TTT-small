import numpy as np
from scipy.spatial import distance

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    N = R * R
    centers = np.zeros((0, 2))
    radii = np.zeros(0)
    r_f = np.full(N, 1.0)
    df = lambda x, y: np.array([x, y])
    dfd = lambda x, y: np.array([[0, 0], [0, 0]])
    df2 = lambda x, y: 0.0
    
    def dist(p1, p2):
        return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
    
    def recalc_radius(candidate):
        return min(candidate[0], min(dist(candidate, center) - radius for center, radius in zip(centers, radii)))
    
    grid = np.array([[i / R, j / R] for i in range(R) for j in range(R)])
    r_blacklist = np.zeros(N, dtype=bool)
    
    for _ in range(26):
        indices = np.where(~r_blacklist)[0]
        max_r = np.max(recalc_radius(grid[indices]))
        max_i = np.argmax(recalc_radius(grid[indices]))
        new_center = grid[indices][max_i]
        new_radius = max_r
        
        centers = np.vstack([centers, new_center])
        radii = np.append(radii, new_radius)
        
        dists = np.sqrt(((grid - new_center)**2).sum(axis=1) - new_radius**2)
        r_f = np.minimum(r_f, dists)
        
        r_blacklist[max_i] = True
    
    delta = 1e-3
    n_refine = 200
    
    for _ in range(n_refine):
        for i in range(26):
            c_i = centers[i]
            r_i = radii[i]
            new_center = c_i + (np.random.rand(2) - 0.5) * delta * 2
            
            new_radius = min(min(new_center[0], 1 - new_center[0]), min(new_center[1], 1 - new_center[1]))
            for center, radius in zip(centers, radii):
                if np.sqrt((new_center[0] - center[0])**2 + (new_center[1] - center[1])**2) - radius < new_radius:
                    new_radius = 0
            
            if new_radius > r_i:
                centers[i] = new_center
                radii[i] = new_radius
    
    for i in range(26):
        r_i = radii[i]
        for center, radius in zip(centers, radii):
            if i != np.where((centers == center).all(axis=1))[0][0]:
                r_i = min(r_i, min(np.sqrt((centers[i][0] - center[0])**2 + (centers[i][1] - center[1])**2) - radius, min(center[0], 1 - center[0]), min(center[1], 1 - center[1])))
        radii[i] = r_i
    
    for i in range(26):
        if radii[i] < 0:
            radii[i] = 0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
