from scipy.optimize import linprog

x_bounds = (1e-12, 1 - 1e-12)

α = 4.0
β = 0.8
ε = 1e-12
δr = 1e-4
maxIter = 3000
tol = 1e-10

def compute_forces(x, y, r):
    n = len(r)
    forces = np.zeros((n, n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            r_ij = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
            if r_ij < r[i] + r[j] + ε:
                overlap = r[i] + r[j] - r_ij
                direction = np.array([x[i] - x[j], y[i] - y[j]]) / r_ij
                forces[i, j] = α * overlap * direction
                forces[j, i] = -forces[i, j]
    return forces.sum(axis=0)

def wall_repulsion(x, y, r):
    F = np.zeros_like(x)
    F[x < r + ε] += α * (r + ε - x[x < r + ε]) * np.array([1, 0])
    F[x > 1 - r - ε] += α * (x[x > 1 - r - ε] - (1 - r - ε)) * np.array([-1, 0])
    F[y < r + ε] += α * (r + ε - y[y < r + ε]) * np.array([0, 1])
    F[y > 1 - r - ε] += α * (y[y > 1 - r - ε] - (1 - r - ε)) * np.array([0, -1])
    return F

def solve_lplike(r, x, y):
    A = np.zeros((n, 2 * n))
    b = np.zeros(n)
    for i in range(n):
        A[i, 2 * i] = 1
        A[i, 2 * i + 1] = 1
        b[i] = min(x[i] - r[i], 1 - x[i] - r[i], y[i] - r[i], 1 - y[i] - r[i])
    res = linprog(-r, A_ub=A, b_ub=b, bounds=(r_min, None), method='highs')

        print("LP failed, using current radii")
