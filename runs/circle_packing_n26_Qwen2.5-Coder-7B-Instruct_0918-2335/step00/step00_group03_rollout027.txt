def disjoint_segments(X):
    n = len(X)
    # Simply find any set of overlapping segments from length 2 in 2D
    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(X[i] - X[j]) < 1e-5:
                return True
    return False

def eval_placement(p):
    C = p.reshape(26,2)
    d = np.linalg.norm(C[:,None,:]-C[None,:,:],axis=-1)
    A = np.zeros((d.shape[0]*(d.shape[1]-1)//2, 26))
    b = np.zeros(d.shape[0]*(d.shape[1]-1)//2)

    index = 0
    for i in range(d.shape[0]):
        for j in range(i+1, d.shape[1]):
            A[index, i] = 1
            A[index, j] = 1
            b[index] = d[i,j]
            index += 1

    A += np.eye(26)
    b += 1e-12

    x, y = C.T
    A += np.vstack((-1 * np.eye(26), -1 * np.eye(26), -1 * np.eye(26), -1 * x[:,None] / (x[:,None] - 1 + 1e-12)[:,0], -1 * y[:,None] / (y[:,None] - 1 + 1e-12)[:,0]))
    b += np.hstack((np.zeros(26), np.zeros(26), -1e-12, np.ones(26) - 1e-12, np.ones(26) - 1e-12))
