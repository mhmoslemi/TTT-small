import numpy as np
from scipy.optimize import minimize

def objective_func(params, n, M_bound, M_overlap, boundary_penalties, overlap_penalties):
    x1 = params[:n]
    y1 = params[n:2*n]
    r = params[2*n:]
    boundary_penalty = M_bound * np.sum(np.maximum(0, r - boundary_penalties))
    overlap_penalty = M_overlap * np.sum(np.maximum(0, overlap_penalties - r))
    return -np.sum(r) + boundary_penalty + overlap_penalty

def compute_penalties(x1, y1, r, n):
    boundary_penalties = np.minimum(np.minimum(np.minimum(np.minimum(x1, 1 - x1), y1), 1 - y1), r)
    overlap_penalties = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x1[i] - x1[j])**2 + (y1[i] - y1[j])**2)
            overlap_penalties[i, j] = dist
            overlap_penalties[j, i] = dist
    return boundary_penalties, overlap_penalties

def run_packing():
    n = 26

    x3 = x1 - 0.5

        for j in range(n):
            if radii[j] < 0:
