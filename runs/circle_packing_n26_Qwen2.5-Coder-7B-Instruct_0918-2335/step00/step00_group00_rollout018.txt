```python
from scipy.optimize import minimize
import numpy as np

n = 26
pair_indices = [(i, j) for i in range(n) for j in range(i+1, n)]

def constraint_func(v):
    x = v[:52].reshape((n, 2))
    r = v[52:]
    arr = []
    # Boundary constraints
    arr.append(x[:, 0] - r)  # each >=0
    arr.append(1 - x[:, 0] - r)  # each >=0
    arr.append(x[:, 1] - r)  # each >=0
    arr.append(1 - x[:, 1] - r)  # each >=0
    # Pairwise constraints
    i_indices = np.array([i for i, j in pair_indices], dtype=np.int64)
    j_indices = np.array([j for i, j in pair_indices], dtype=np.int64)
    dx = x[i_indices, 0] - x[j_indices, 0]
    dy = x[i_indices, 1] - x[j_indices, 1]
    dist2 = dx**2 + dy**2
    pair_vals = dist2 - (r[i_indices] + r[j_indices])**2
    arr.append(pair_vals)
    return np.concatenate(arr)

x0 = np.empty(78)
positions = np.array([(j/5 + 0.5) / 5 for j in range(5)] for i in range(5))
positions = positions[:25]
positions = np.append(positions, [[0.6, 0.9]])
x0[:52] = positions.ravel()
x0[52:] = np.full(n, 0.04)
bounds = [(0, 1)] * 52 + [(0, 0.5)] * n

cons = {'type': 'ineq', 'fun': constraint_func}

res = minimize(fun=lambda v: -np.sum(v[52:]), x0=x0, bounds=bounds,
               constraints=[cons], options={'maxiter': 5000, 'gtol': 1e-12, 'disp': False})

radii = res.x[52:]
sum_radii = np.sum(radii)
centers = res.x[:52].reshape((n, 2))

return centers, radii, sum_radii
```
