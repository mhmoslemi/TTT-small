import math
import numpy as np
from scipy.optimize import minimize

def penalty_bound(radius, center):
    x, y = center
    return max(0, radius - min(x, 1 - x, y, 1 - y)) ** 2 * 1e6

def penalty_overlap(circles, centers, radii):
    n = circles.shape[0]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            penalty += max(0, radii[i] + radii[j] - dist) ** 2 * 1e6
    return penalty

def objective_function(positions, radii):
    centers = positions[:, :2]
    n = centers.shape[0]
    penalty = sum(penalty_bound(radii[i], centers[i]) for i in range(n))
    penalty += penalty_overlap(positions, centers, radii)
    return -(radii.sum() + penalty)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    min_radius = 0.1
    max_radius = 0.4
    centers = np.zeros((n, 4))
    for i in range(5):
        for x in np.linspace(0.1, 0.9, 6 if i % 2 == 0 else 5):
            y_base = 0.1 + i * 0.1
            for y in np.linspace(y_base, y_base + 0.1, 6 if i % 2 == 0 else 5):
                centers[i * 6 + int((x - 0.1) // 0.1)] = np.array([x, y, min_radius, max_radius])
    positions = centers[:, :2]
    radii = (centers[:, 2] + centers[:, 3]) / 2
    bounds = [(0.1, 0.9), (0.1, 0.9)] * n + [(min_radius, max_radius)] * n

    result = minimize(objective_function, np.concatenate((positions.flatten(), radii)), bounds=bounds, method="L-BFGS-B")

    if not result.success:
        raise ValueError("Minimization failed.")
