import numpy as np
from scipy.optimize import minimize

def initial_guess(seed):
    np.random.seed(seed)
    xy = np.random.uniform(0.1, 0.9, size=(26, 2))
    r = np.zeros(26)
    return np.concatenate((xy, r))

def objective(v):
    return -np.sum(v[26:])

def constraints(v):
    n = 26
    centers = v[:2*n].reshape((n, 2))
    r = v[26:]
    boundary_constraints = np.vstack([
        centers[:, 0] - r,
        1 - centers[:, 0] - r,
        centers[:, 1] - r,
        1 - centers[:, 1] - r
    ]).T.flatten()
    pairwise_constraints = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=-1))
    pairwise_constraints = np.triu(pairwise_constraints, k=1).flatten() - (r[:, np.newaxis] + r[np.newaxis, :])
    return np.hstack((boundary_constraints, pairwise_constraints))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    RESTARTS = 8
    eps = 1e-7
    best_value = -np.inf
    best_v = None
    
    for seed in range(RESTARTS):
        v0 = initial_guess(seed)
        res = minimize(
            fun=objective,
            x0=v0,
            method='SLSQP',
            constraints={'type': 'ineq', 'fun': constraints},
            bounds=(0, 0.5),
            options={
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if not res.success:
            continue
        value = -res.fun  # because we minimised the negative
        if value > best_value:
            best_value = value
            best_v = res.x
    
    centers = best_v[:2*n].reshape((n, 2))
    radii = np.maximum(0.0, best_v[2*n:] - eps)
    
    return centers, radii, np.sum(radii)
