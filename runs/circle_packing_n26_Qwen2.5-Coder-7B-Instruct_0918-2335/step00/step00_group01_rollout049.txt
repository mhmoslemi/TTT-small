import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial positions of circles in a hexagonal grid pattern
    centers = np.array([
        [0.1, 0.1], [0.7, 0.1],
        [0.4, 0.2], [0.6, 0.2], [0.3, 0.3], [0.7, 0.3],
        [0.2, 0.4], [0.8, 0.4], [0.5, 0.5],
        [0.6, 0.6], [0.4, 0.6], [0.1, 0.7], [0.7, 0.7],
        [0.3, 0.8], [0.7, 0.8], [0.4, 0.9], [0.6, 0.9],
        [0.2, 0.9], [0.3, 1.0], [0.4, 1.0], [0.5, 1.0], [0.6, 1.0], [0.7, 1.0]
    ])

    best_obj = -np.inf
    best_centres, best_radii = None, None

    for restart in range(3):
        if restart > 0:
            # Add small random jitter to centers
            jitter = 0.01 * np.random.randn(n, 2)
            centers += jitter
            # Optionally rotate centers
            angle = restart * np.pi / 2.
            rotation_matrix = np.array([
                [np.cos(angle), -np.sin(angle)],
                [np.sin(angle),  np.cos(angle)]
            ])
            centers = np.dot(centers, rotation_matrix)

        for iter in range(200):
            # Solve LP for current centers
            c = -np.ones(n)
            A_eq, b_eq = [], []
            A_ub = []
            b_ub = []

            # Boundary constraints
            for i in range(n):
                x, y = centers[i]
                A_ub.append([1, x, y])
                b_ub.append(x)
                A_ub.append([1, 1 - x, y])
                b_ub.append(1 - x)
                A_ub.append([1, x, 1 - y])
                b_ub.append(y)
                A_ub.append([1, 1 - x, 1 - y])
                b_ub.append(1 - y)

            # Pairwise constraints
            for i in range(n):
                for j in range(i + 1, n):
                    dist_ij = np.linalg.norm(centers[i] - centers[j])
                    for k in range(2):
                        A_ub.append([1, centers[i, k], centers[j, k]])
                        b_ub.append(dist_ij)

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')

            if not res.success:
                # Re-initialize centers randomly if LP is infeasible
                centers = np.random.rand(n, 2)
                break

            radii = res.x
            obj = -res.fun

            if obj > best_obj:
                best_obj = obj
                best_centres = centers.copy()
                best_radii = radii.copy()

            # Compute forces from duals
            duals = res.dual
            forces = np.zeros((n, 2))
            for k in range(n):
                i, j = np.triu_indices_from(res.x, 1)
                for m in range(n):
                    x_m, y_m = centers[m]
                    if (np.allclose([k, 2*i, 2*m+1], [0, 2*len(i), 2*len(j)]) or
                        np.allclose([k, 2*i, 2*m+1], [2, 2*len(i), 2*len(j)]) or
                        np.allclose([k, 2*j, 2*m+1], [0, 2*len(i), 2*len(j)]) or
                        np.allclose([k, 2*j, 2*m+1], [2, 2*len(i), 2*len(j)])):
                        forces[m] +=radii[j]*j - j

            # Move centers
            step_size = 0.01
            centers += step_size * forces

        # Ensure centers are within the unit square
        centers = np.clip(centers, 0, 1)

    # Shrink radii slightly to ensure they fit within the square
    eps = 1e-7
    radii = np.maximum(0.0, best_radii - eps)

    return best_centres, radii, np.sum(radii)
