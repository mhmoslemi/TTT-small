import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    grid = np.mgrid[0:1:R*1j, 0:1:R*1j].reshape(2,-1)
    d_side = np.min(np.append(grid[0], 1-grid[0]), axis=0)
    d_side = np.append(d_side, np.min(np.append(grid[1], 1-grid[1]), axis=0))
    r_f = np.min(d_side.reshape(R,R), axis=0)
    r_f = r_f.ravel()

    centers = np.zeros((0,2), dtype=float)
    radii = np.zeros(0, dtype=float)

    for _ in range(26):
        idx = np.argmax(r_f)
        cx, cy = grid[...,idx]
        r = r_f[idx] - 1e-9
        centers = np.vstack([centers, [cx, cy]])
        radii = np.append(radii, r)
        dx = cx - grid[0]
        dy = cy - grid[1]
        dists = np.sqrt(dx**2 + dy**2) - r
        r_f = np.minimum(r_f, dists).ravel()
        r_f[idx] = -1e9

    delta = 1e-3
    N_refine = 200
    for _ in range(N_refine):
        i = np.random.randint(26)
        cx_new = centers[i,0] + (-delta/2 + np.random.rand())*delta
        cy_new = centers[i,1] + (-delta/2 + np.random.rand())*delta
        r_new = min(np.min(np.append([cx_new], [1-cx_new])), 
                    np.min(np.append([cy_new], [1-cy_new])), 
                    np.min(np.sqrt((cx_new-centers[:,0])**2 + (cy_new-centers[:,1])**2) - radii))
        if r_new > radii[i]:
            centers[i] = [cx_new, cy_new]
            radii[i] = r_new

    for i in range(26):
        r = min(np.min(np.append([centers[i,0]], [1-centers[i,0]])), 
                np.min(np.append([centers[i,1]], [1-centers[i,1]])), 
                np.min(np.sqrt((centers[i,0]-centers[:,0])**2 + (centers[i,1]-centers[:,1])**2) - radii))
        radii[i] = r

    return centers, radii, np.sum(radii)
