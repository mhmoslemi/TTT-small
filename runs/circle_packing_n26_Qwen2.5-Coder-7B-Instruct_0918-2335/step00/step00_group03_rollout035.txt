import numpy as np
from scipy import optimize

def random_sequential_addition(N, r_min, iterations):
    r = np.full(N, r_min)
    centers = np.zeros((N, 2))
    for i in range(N):
        while True:
            x, y = np.random.rand(2)
            ok = all((x - r[j])**2 + (y - r[j])**2 >= (r[i] + r[j])**2 for j in range(i))
            if ok:
                centers[i] = x, y
                break
    return centers, r

def physics_simulation(centers, radii, T, dt):
    N = len(radii)
    for _ in range(T):
        forces = np.zeros((N, 2))
        for i in range(N):
            for j in range(i + 1, N):
                dx, dy = centers[i] - centers[j]
                d = np.sqrt(dx**2 + dy**2)
                gap = d - (radii[i] + radii[j])
                if gap > 0:
                    f = gap * dt
                    forces[i] -= f * dx / d
                    forces[j] += f * dx / d
                forces[i][1] -= min(d - radii[i], 0) * dt
                forces[j][1] -= min(d - radii[j], 0) * dt
                forces[i][0] -= min(d - radii[i], 0) * dt
                forces[j][0] -= min(d - radii[j], 0) * dt
                
        for i in range(N):
            dx, dy = forces[i]
            radii[i] += dx * (radii[i] + 0.5) * dt
            centers[i, 0] += dx * dt
            centers[i, 1] += dy * dt
            radii[i] = max(0, min(radii[i], 1 - radii[i]))
            centers[i, 0] = max(0, min(centers[i, 0], 1))
            centers[i, 1] = max(0, min(centers[i, 1], 1))
    return centers, radii

def global_scaling(centers, radii):
    s_max = min(min((np.sqrt((centers[i] - centers[j])**2).min() - radii[i] - radii[j]) for j in range(i)), (1 - radii[i]) / radii[i] for i in range(len(radii)))
    centers = centers + s_max * (centers - 0.5)
    radii *= s_max
    return centers, radii

def penalty(x, centers, radii, alpha):
    N = len(radii)
    fx = -radii.sum()
    p_ij = np.array([max(0, radii[i] + radii[j] - np.sqrt((centers[i] - centers[j])**2)).sum()**2 for i in range(N) for j in range(i + 1, N)])
    p_bs = np.array([max(0, r - x[i]) for i, (x_i, y_i, r) in enumerate(zip(x[:N].reshape((N, 2)), x[N:].reshape((N, 1)), radii))]).sum()**2
    return fx + alpha * (p_ij.sum() + p_bs.sum())

def gradient(x, centers, radii, alpha):
    N = len(radii)
    Nxy = 2 * N
    Jr = np.zeros((1, Nxy, N))
    Jxy = np.zeros((Nxy, Nxy, N))
    
    for i in range(N):
        for j in range(i + 1, N):
            dx, dy = centers[i] - centers[j]
            d = np.sqrt(dx**2 + dy**2)
            gap = d - (radii[i] + radii[j])
            if gap > 0:
                f = gap
                H_ij = np.array([[f * dx / d, f * dy / d], [f * dy / d, f * dx / d]])
                Jr[0, 2 * i:2 * (i + 1), j] -= H_ij.T.flatten()
                Jr[0, 2 * j:2 * (j + 1), i] -= H_ij.flatten()
            Jr[0, Nxy + i] -= 1
            Jr[0, Nxy + j] -= 1
            
    p_ij = np.array([max(0, radii[i] + radii[j] - np.sqrt((centers[i] - centers[j])**2)).sum()**2 for i in range(N) for j in range(i + 1, N)])
    p_bs = np.array([max(0, r - x[i]) for i, (x_i, y_i, r) in enumerate(zip(x[:N].reshape((N, 2)), x[N:].reshape((N, 1)), radii))]).sum()**2
    
    dFx = radii.sum()
    dPij = 2 * alpha * p_ij.sum() * np.eye(N'].reshape((1, Nxy))
    dPbs = 2 * alpha * p_bs * np.eye(N')

    return dFx + dPij.sum() + dPbs.sum()

def run_packing():
    np.random.seed(42)
    N = 26
    iterations = 10000
    T = 2000
    dt = 1e-3
    alpha = 1e5
    r_min = 1e-3
    
    centers, radii_initial = random_sequential_addition(N, r_min, iterations)
    centers, radii = physics_simulation(centers, radii_initial, T, dt)
    centers, radii = global_scaling(centers, radii)
    
    x0 = np.hstack((centers.flatten(), radii))
    res = optimize.minimize(penalty, x0, args=(centers, radii, alpha), jac=gradient, constraints=[{"type": "ineq", "fun": lambda x: x[N:]}], method="L-BFGS-B", bounds=[(0, 1)]*2*N + [(0, 0.5)]*N)
    
    if not res.success:
        raise RuntimeError("Optimization failed")
    
    centers = res.x[:N].reshape((N, 2))
    radii = res.x[N:]
    return centers, radii, radii.sum()
