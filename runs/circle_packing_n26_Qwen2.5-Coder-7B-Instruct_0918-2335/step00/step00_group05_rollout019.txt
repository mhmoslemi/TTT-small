import numpy as np
import scipy.spatial
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter_pso = 2000
    max_iter_sa = 20000
    swarm_size = 80
    w_max = 0.9
    w_min = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.05
    T0 = 0.05
    alpha = 0.995
    epsilon = 1e-12

    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) / 4

    best_centers = np.copy(centers)
    best_radii = np.copy(radii)
    best_sum_radii = -np.inf

    for _ in range(max_iter_pso):
        velocities = np.random.rand(swarm_size, n, 3) * (2 * v_max) - v_max
        for p in range(swarm_size):
            for i in range(n):
                velocities[p, i, :] = w_max * velocities[p, i, :] + c1 * random.random() * (best_centers[p, i] - centers[p, i]) + c2 * random.random() * (best_centers.max(axis=0) - centers[p, i])
                velocities[p, i, 0] = np.clip(velocities[p, i, 0], -v_max, v_max)
                velocities[p, i, 1] = np.clip(velocities[p, i, 1], -v_max, v_max)
            new_centers = centers + velocities[:, :2, :]
            new_radii = np.maximum(epsilon, radii + velocities[:, 2, :])
            for j in range(i + 1, n):
                d = np.hypot(new_centers[:, i, 0] - new_centers[:, j, 0], new_centers[:, i, 1] - new_centers[:, j, 1])
                min_dist = new_radii[:, i] + new_radii[:, j]
                if d < min_dist - epsilon:
                    shift = (min_dist - d) / 2.0
                    direction = ((new_centers[:, i, 0] - new_centers[:, j, 0]) / d, (new_centers[:, i, 1] - new_centers[:, j, 1]) / d)
                    new_centers[:, i, 0] += shift * direction[0]
                    new_centers[:, i, 1] += shift * direction[1]
                    new_centers[:, j, 0] -= shift * direction[0]
                    new_centers[:, j, 1] -= shift * direction[1]
            sum_radii = np.sum(new_radii)
            if sum_radii > best_sum_radii:
                best_centers = np.copy(new_centers)
                best_radii = np.copy(new_radii)
                best_sum_radii = sum_radii
            centers = np.copy(new_centers)
            radii = np.copy(new_radii)

    centers = best_centers
    radii = best_radii
    sum_radii = best_sum_radii

    best_solution = np.zeros((n, 2))
    last_best_sum_radii = -np.inf
    temperature = T0

    for _ in range(max_iter_sa):
        for i in range(n):
            if random.uniform(0, 1) < 0.5:
                dx, dy = np.random.normal(0, temperature, 2)
                new_center = centers[i] + (dx, dy)
                new_radius = radii[i] + np.random.normal(0, temperature / 2)
                new_radius = np.maximum(epsilon, np.minimum(1 - np.linalg.norm(new_center[:2]), new_radius))
                new_center = np.clip(new_center, new_radius, 1 - new_radius)
                for j in range(i + 1, n):
                    d = np.hypot(new_center[0] - centers[j, 0], new_center[1] - centers[j, 1])
                    min_dist = new_radius + radii[j]
                    if d < min_dist - epsilon:
                        shift = (min_dist - d) / 2.0
                        direction = ((new_center[0] - centers[j, 0]) / d, (new_center[1] - centers[j, 1]) / d)
                        new_center[0] += shift * direction[0]
                        new_center[1] += shift * direction[1]
                        centers[j, 0] -= shift * direction[0]
                        centers[j, 1] -= shift * direction[1]
                new_centers = np.copy(centers)
                new_centers[i] = new_center
                new_radii = np.copy(radii)
                new_radii[i] = new_radius
                sum_radii = np.sum(new_radii)
                if sum_radii > last_best_sum_radii:
                    last_best_sum_radii = sum_radii
                    best_solution = np.copy(new_centers)
                else:
                    dE = sum_radii - best_sum_radii
                    if np.exp(-dE / temperature) > random.uniform(0, 1):
                        best_centers = np.copy(new_centers)
                        best_radii = np.copy(new_radii)
                        best_sum_radii = sum_radii
                    else:
                        centers[i] = new_center
                        radii[i] = new_radius

        temperature *= alpha

    return best_centers, best_radii, best_sum_radii
