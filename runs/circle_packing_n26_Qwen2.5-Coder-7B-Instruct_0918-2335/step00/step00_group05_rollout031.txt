import numpy as np
from scipy.optimize import minimize

def hex_lattice(s):
    """Generate a hexagonal lattice of points within the unit square."""
    points = []
    for q in range(int(1 / s)):
        for r in range(int((1 - q * s * np.sqrt(3)) / (2 * s))):
            x = q * s + r * s * 0.5
            y = q * s * np.sqrt(3) / 2 + r * s * np.sqrt(3) / 3
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
    return np.array(points)

def objective(z, s, n):
    """Objective function to minimize (negative sum of radii)."""
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    return -np.sum(r)

def constraints(z, s, n):
    """Constraints for the optimizer: boundary and non-overlap."""
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: x[i] - r[i] - 1e-12})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: x[i] + r[i] - (1 - 1e-12)})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: y[i] - r[i] - 1e-12})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: y[i] + r[i] - (1 - 1e-12)})
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: dist - (r[i] + r[j]) + 1e-12})
    return cons

def solve_stage(z0, s, n, delta):
    """Solve one optimization stage with given params."""
    cons = constraints(z0, s, n)
    cons.append({'type': 'ineq', 'fun': lambda z: np.sum(np.clip(z[n:2*n] - delta, 0, None)) - np.sum(z[n:2*n])})
    cons.append({'type': 'ineq', 'fun': lambda z: np.sum(np.clip(z[n:2*n] + delta, 0, None)) - np.sum(z[n:2*n])})
    res = minimize(objective, z0, method='SLSQP', args=(s, n), constraints=cons)
    return res.x, res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    s = 1 / (2 * n**0.5 * 1.05)  # Slightly tighter than hexagonal packing
    points = hex_lattice(s)
    if len(points) < n:
        raise ValueError("Too few points for 26 circles")
    points = points[:n]
    z0 = np.concatenate([points.flatten(), np.zeros(n)])  # Initial guess: circles at points with zero radius
    z = z0.copy()
    delta = 1e-4
    for _ in range(3):
        z, _ = solve_stage(z, s, n, delta)
        delta *= 0.5
    # Inflate balls to minimize difference to max sum_r
    max_radius = 0.5
    for i in range(n):
        r = z[n + i]
        clearance = min(z[:2*n][i*2:i*2+2] - r, (1 - z[:2*n][i*2:i*2+2]) - r)
        inflate_r = min(clearance / 4, max_radius - r)  # Inflate to touch neighboring circle or border
        z[n + i] += inflate_r
        x, y = z[:2*n][i*2:i*2+2]
        for j in range(n):
            if j != i:
                dx, dy = z[:2*n][j*2:j*2+2] - x, z[:2*n][j*2+1:j*2+3] - y
                dist = np.sqrt(dx**2 + dy**2)
                if dist - (z[n + j] + z[n + i]) < 1e-12:
                    z[:2*n][i*2:i*2+2] -= (z[n + i] * dx / dist, z[n + i] * dy / dist)
    # Ensure sum of radii is maximized by minimizing the negative sum (which maximizes the sum)
    res = minimize(lambda z: -np.sum(z[n:]), z[n:], method='BFGS')
    z[n:] = -res.x
    z[n:] = np.clip(z[n:], 0, max_radius)
    sum_radii = np.sum(z[n:])
    return z[:n].reshape((-1, 2)), z[n:], sum_radii
