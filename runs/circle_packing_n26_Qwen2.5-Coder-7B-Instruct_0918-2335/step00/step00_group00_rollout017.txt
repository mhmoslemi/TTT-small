```python
import numpy as np
from scipy.optimize import NonlinearConstraint, minimize

n = 26

# Generate initial positions and constraints
positions = []
for i in range(5):
    for j in range(5):
        positions.append(((j+0.5)/5.0, (i+0.5)/5.0))
        if len(positions) == 25:
            break
    if len(positions) == 25:
        break
positions.append((0.6, 0.9))
positions = np.array(positions)

radii_initial = np.full(n, 0.04)

x0 = np.empty(78)
x0[:52] = positions.ravel()
x0[52:] = radii_initial

# Create pairs index list
pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]

# Constraint function returning all inequalities
def constraint_func(v):
    x = v[:52].reshape((n, 2))
    r = v[52:]
    arr = []

    # Boundary constraints
    arr.append(x[:, 0] - r)
    arr.append(1 - x[:, 0] - r)
    arr.append(x[:, 1] - r)
    arr.append(1 - x[:, 1] - r)

    # Pairwise packing constraints
    i_vec = np.array([i for i, j in pair_indices], dtype=np.int64)
    j_vec = np.array([j for i, j in pair_indices], dtype=np.int64)
    dx = x[i_vec, 0] - x[j_vec, 0]
    dy = x[i_vec, 1] - x[j_vec, 1]
    dist2 = dx ** 2 + dy ** 2
    arr.append(dist2 - (r[i_vec] + r[j_vec]) ** 2)

    return np.concatenate(arr)

# Lower bounds for variables
bounds = []
for _ in range(26*2):  # 52 variables for positions
    bounds.extend([(0.0, 1.0), (0.0, 1.0)])
for _ in range(n):  # 26 radii variables
    bounds.append((0.0, 0.5))

# Define nonlinear constraint
cons = NonlinearConstraint(constraint_func, 0, np.inf)

# Perform optimization
res = minimize(fun=lambda v: -np.sum(v[52:]), x0=x0, bounds=bounds, constraints=cons,
               method='trust-constr', options={'maxiter': 5000, 'gtol': 1e-12, 'disp': False})

# Check if optimization succeeded
if not res.success:
    # Fallback: Use equal radius arrangement
    r_opt = 0.105
    centers = np.array([[i / 5, i / 5] for i in range(26)])
    centers[:5, 1] += 0.15
    centers[-1, 1] += 0.6
    centers[0, 1] += 0.4
    centers[1, 1] += 0.8
    sum_radii = n * r_opt
else:
    centers = res.x[:52].reshape((n, 2))
    radii = res.x[52:]
    sum_radii = float(np.sum(radii))

# Apply shrink factor for any constraint violation within numerical noise
epsilon = 1e-9
for i in range(n):
    for j in range(i + 1, n):
        if np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) < radii[i] + radii[j] - epsilon:
            factor = 1 - (np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (radii[i] + radii[j])) / (radii[i] + radii[j])
            radii[i] *= factor
            radii[j] *= factor

return centers, radii, sum_radii
```
