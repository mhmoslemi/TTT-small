import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.spatial.distance import cdist
from scipy.stats import norm

def validate_packing(centers, radii):
    n = centers.shape[0]

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26
    radius_centering = 1 / (np.sqrt(3) * 26)

    def initial_circle_radii():
        r_i = np.ones(n) * radius_centering
        return r_i

    def initial_circle_centers():
        circle_centers = np.zeros((n, 2))
        theta = np.linspace(0, 2*np.pi, n, endpoint=False)
        r = np.sqrt(np.random.rand(n))
        circle_centers[:, 0] = 0.5 + 0.25 * r * np.cos(theta)
        circle_centers[:, 1] = 0.5 + 0.25 * r * np.sin(theta)
        return circle_centers

    def objective_function(xyz):
        centers = xyz[:2*n].reshape(-1, 2)
        radii = xyz[2*n:]
        overlaps_and_boundaries_penalty = 0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1
                    or y - r < 0 or y + r > 1):
                overlaps_and_boundaries_penalty += np.inf
            for j in range(i + 1, n):
                if np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) < r[i] + r[j]:
                    overlaps_and_boundaries_penalty += np.inf
        return -sum(radii) + overlaps_and_boundaries_penalty

    def projected_center(center, radius):
        x, y = center
        if x - radius < 0:
            center[0] = radius
        elif x + radius > 1:
            center[0] = 1 - radius
        if y - radius < 0:
            center[1] = radius
        elif y + radius > 1:
            center[1] = 1 - radius
        return center

    def constraint(xyz):
        centers = xyz[:2*n].reshape(-1, 2)
        radii = xyz[2*n:]
        return np.array([np.all(projected_center(centers[i], radii[i]) == centers[i])
                           for i in range(n)]).flatten()

    initial = np.concatenate((initial_circle_centers().flatten(), initial_circle_radii()))

    cons = ({'type': 'ineq', 'fun': lambda xyz: constraint(xyz)})

    res = minimize(objective_function, initial, method='SLSQP', constraints=cons, options={'maxiter': 10000})

    if not res.success:
        return np.zeros((26, 2)), np.zeros((26,)), 0

    centers = res.x[:2*n].reshape(-1, 2)
    radii = res.x[2*n:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

# Example usage
centers, radii, sum_radii = run_packing()
print("Centers: ", centers)
print("Radii: ", radii)
print("Sum of Radii: ", sum_radii)
