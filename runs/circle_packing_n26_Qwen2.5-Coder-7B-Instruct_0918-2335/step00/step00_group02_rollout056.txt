import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    side_length = 1 / np.sqrt(n)
    for i in range(n):
        row = i // int(np.sqrt(n))
        col = i % int(np.sqrt(n))
        x = (col + 0.5) / np.sqrt(n)
        y = (row + 0.5) / np.sqrt(n)
        centers[i] = [x, y]
        radii[i] = side_length / 2

    return centers, radii, np.sum(radii)
