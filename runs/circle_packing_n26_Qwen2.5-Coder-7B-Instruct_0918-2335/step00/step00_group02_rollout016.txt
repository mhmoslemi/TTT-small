import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x):
    N = 26
    xs, ys, rs = x[:N], x[N:2*N], x[2*N:]
    sum_r = -rs.sum()
    p_boundary = (np.maximum(0, rs - xs)**2 + 
                 np.maximum(0, xs + rs - 1)**2 + 
                 np.maximum(0, rs - ys)**2 + 
                 np.maximum(0, ys + rs - 1)**2).sum()
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    p_overlap = np.sum(np.maximum(0, rr - dist[np.triu_indices(N, 1)])**2)
    return -sum_r + 1e5 * (p_boundary + p_overlap)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 2 * 26 + [(0, 0.5)] * 26
    res = differential_evolution(objective, bounds, popsize=12, maxiter=400, random_state=0)
    if not res.success:
        res = differential_evolution(objective, bounds, popsize=12, maxiter=400, random_state=1)
        if not res.success:
            res = differential_evolution(objective, bounds, popsize=12, maxiter=400, random_state=2)
            if not res.success:
                res = differential_evolution(objective, bounds, popsize=12, maxiter=400, random_state=3)
    x = res.x
    xs, ys, rs = x[:26], x[26:52], x[52:]
    centres = np.column_stack([xs, ys])
    sum_radii = rs.sum()
    return centres, rs, sum_radii

centers, radii, sum_radii = run_packing()
