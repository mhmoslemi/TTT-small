import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    v_mag = 5e-3
    T_max = 50.0
    eps = 1e-6
    
    # Initialize random positions and velocities
    np.random.seed(1)
    centers = np.random.rand(n, 2)
    velocities = v_mag * np.random.randn(n, 2)
    
    # Event queue
    queue = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = dx**2 + dy**2
            if dist > 4 * r_min**2:
                dt = (2 * r_min - np.sqrt(dist) + 2 * g * np.sqrt(dist)) / (2 * g)
                heapq.heappush(queue, (dt, (i, j)))
    
    for i in range(n):
        dvx = velocities[i, 0]
        dvy = velocities[i, 1]
        if dvx > g:
            dt = (r_min - centers[i, 0]) / (dvx - g)
            heapq.heappush(queue, (dt, (i, 'left')))
        if dvx < -g:
            dt = (r_min - (1 - centers[i, 0])) / (-dvx - g)
            heapq.heappush(queue, (dt, (i, 'right')))
        if dvy > g:
            dt = (r_min - centers[i, 1]) / (dvy - g)
            heapq.heappush(queue, (dt, (i, 'bottom')))
        if dvy < -g:
            dt = (r_min - (1 - centers[i, 1])) / (-dvy - g)
            heapq.heappush(queue, (dt, (i, 'top')))
    
    while queue:
        t, event = heapq.heappop(queue)
        if t > T_max:
            break
        
        if event[1] == 'left':
            i = event[0]
            centers[i, 0] = r_min + g * t
            dvx = velocities[i, 0]
            velocities[i, 0] = -dvx
        elif event[1] == 'right':
            i = event[0]
            centers[i, 0] = 1 - r_min - g * t
            dvx = velocities[i, 0]
            velocities[i, 0] = -dvx
        elif event[1] == 'bottom':
            i = event[0]
            centers[i, 1] = r_min + g * t
            dvy = velocities[i, 1]
            velocities[i, 1] = -dvy
        elif event[1] == 'top':
            i = event[0]
            centers[i, 1] = 1 - r_min - g * t
            dvy = velocities[i, 1]
            velocities[i, 1] = -dvy
        else:
            i, j = event
            dt = t
            centers[i] += velocities[i] * dt
            centers[j] += velocities[j] * dt
            r_i = r_min + g * t
            r_j = r_min + g * t
            
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = dx**2 + dy**2
            
            if dist < 4 * (r_i + r_j)**2:
                dvx = velocities[i, 0]
                dvy = velocities[i, 1]
                dvjx = velocities[j, 0]
                dvjy = velocities[j, 1]
                
                D = np.array([[2 * dx, 2 * dy], [2 * dy, 2 * dx]])
                u = np.array([2 * (r_i + r_j - np.sqrt(dist)) * dx, 2 * (r_i + r_j - np.sqrt(dist)) * dy])
                v = np.linalg.solve(D, u)
                
                velocities[i, 0] = dvx - v[0]
                velocities[i, 1] = dvy - v[1]
                velocities[j, 0] = dvjx + v[0]
                velocities[j, 1] = dvjy + v[1]
    
    # Validation and repair
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = dx**2 + dy**2
            if dist < 4 * (r_min + eps)**2:
                centers[i, 0] += eps * (dx / dist)
                centers[i, 1] += eps * (dy / dist)
                centers[j, 0] -= eps * (dx / dist)
                centers[j, 1] -= eps * (dy / dist)
    
    # Solve LP to maximize radii
    c = [-1] * n
    A = []
    for i in range(n):
        A.append([1] + [0] * i + [0] * (n - i - 1))
    for i in range(n):
        A.append([0] + [1] + [0] * (i - 1) + [0] * (n - i - 2) + [1])
        A.append([0] + [0] * i + [1] + [0] * (n - i - 2) + [0] * (2 - i) + [1])
        A.append([0] + [0] * (i - 1) + [0] * (n - i - 2) + [1] + [0] * (n - i) + [1])
    
    A = np.array(A)
    b = np.ones(n + 3 * n - 3)
    bounds = [(r_min, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if not res.success:
        return np.random.rand(n, 2), np.zeros(n), 0
    
    radii = res.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

# Example usage
centers, radii, sum_radii = run_packing()
print("sum_radii:", sum_radii)
print("centers:\n", centers)
print("radii:\n", radii)
