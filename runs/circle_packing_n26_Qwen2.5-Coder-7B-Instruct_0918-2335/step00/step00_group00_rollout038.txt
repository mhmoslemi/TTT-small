import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_radii = np.full(n, 0.01)
    initial_centers = np.random.rand(n, 2)
    
    def objective(x):
        centers = x[:n*n].reshape(n, 2)
        radii = x[n*n:]
        return -np.sum(radii)  # Negative because we'll use a minimization algorithm
    
    def constraint(x):
        centers = x[:n*n].reshape(n, 2)
        radii = x[n*n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append({'type': 'ineq', 'fun': lambda t, i=i, j=j, dist=dist: t[n*n+i+1] + t[n*n+j+1] - dist + 1e-12})
        return constraints
    
    bounds = [(0, 1)] * (n * 2)  # Bounds for centers and radii
    
    res = minimize(objective, np.hstack((initial_centers.flatten(), initial_radii)), method='SLSQP', bounds=bounds, constraints=constraint(initial_centers.flatten()))
    
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0
    
    centers = res.x[:n*n].reshape(n, 2)
    radii = res.x[n*n:]
    
    return centers, radii, -res.fun  # Negative because we minimized the negative of the sum of radii
