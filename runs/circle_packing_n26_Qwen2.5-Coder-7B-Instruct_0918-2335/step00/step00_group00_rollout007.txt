import numpy as np
import scipy.optimize as opt

def objective(radii, centers):
    return -np.sum(radii)

def constraint(radii, center, pair_centers):
    dist = np.sqrt(np.sum((center - pair_centers) ** 2, axis=1))
    return dist + radii - np.roll(dist, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    tol = 1e-8
    vertices = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
    init_radii = np.ones(26) / np.sqrt(len(vertices) + 2 * len(vertices) - 4) * np.sqrt(2)
    init_centers = np.vstack((vertices, vertices + (1 - init_radii[0]) * np.array([[0.5, 0.5], [0.5, -0.5], [-0.5, 0.5], [-0.5, -0.5]])))
    init_centers += (np.random.rand(2, 26) - 0.5) * 0.2

    constraints = [{'type': 'ineq', 'fun': constraint, 'args': (init_centers[i], init_centers[:])} for i in range(4)]
    bounds = [(0, None) for _ in range(26)]

    result = opt.minimize(objective, init_radii, args=(init_centers,), method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': tol, 'xtol': tol, 'maxiter': 1000})
    
    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0

    centers = init_centers[:, result.x > tol]
    radii = result.x[result.x > tol]
    return centers, radii, np.sum(radii)
