import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Strategy constants
    n = 26
    X = y = np.random.uniform(0.01, 0.99, n)
    r = np.full(n, 1e-6)
    dt = 1e-3
    alpha = 4.0
    beta = 0.8
    epsilon = 1e-12
    delta_r = 1e-4
    max_iter = 3000
    tol = 1e-10

    # Dynamics loop
    for step in range(1, max_iter + 1):
        F = np.zeros((n, 2))
        for i in range(n):
            overlap = 0
            F_rep = np.zeros(2)
            for j in range(n):
                if i == j:
                    continue
                dx = X[j] - X[i]
                dy = y[j] - y[i]
                d = np.sqrt(dx**2 + dy**2)
                if d < r[i] + r[j] + epsilon:
                    overlap = r[i] + r[j] - d
                    F_rep += alpha * overlap * np.array([dx, dy]) / d
            F[i] += F_rep
            # Wall repulsion
            for dir in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
                dist_to_wall = np.dot(np.array(dir), np.array([X[i], y[i]]))
                if dist_to_wall < r[i] + epsilon:
                    repulsion = alpha * (r[i] + epsilon - dist_to_wall) * np.array(dir)
                    F[i] += repulsion
        
        # Update velocities and positions
        v = np.clip(F * dt / 1, -1, 1)
        X += v * dt * beta
        y += v * dt * beta
        r += delta_r
        r = np.clip(r, epsilon, 1)

    # Feasibility check
    for i in range(n):
        if not (epsilon < X[i] + r[i] < 1 - epsilon and epsilon < y[i] + r[i] < 1 - epsilon):
            raise ValueError("Circle out of bounds")
        for j in range(i + 1, n):
            dist = np.sqrt((X[i] - X[j])**2 + (y[i] - y[j])**2)
            if dist < r[i] + r[j] - epsilon:
                raise ValueError("Circles overlap")

    # LP refinement of radii
    from scipy.optimize import linprog
    A = np.zeros((n + n + n, n))
    b = np.zeros(n + n + n)
    for i in range(n):
        A[i, i] = 1
        A[i + n, i] = -1
        A[i + 2 * n, i] = -1
    for i in range(n):
        for j in range(i + 1, n):
            A[n + i, n + j] = 1
            A[n + j, n + i] = 1
            A[n + i + n + n, i] = -1
            A[n + j + n + n, j] = -1
    b[:n] = 1 - X
    b[n:2 * n] = 1 - y
    b[2 * n:] = X + r - 1
    res = linprog(-r, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        raise ValueError("LP refinement failed")
    r = res.x

    return X, y, np.sum(r)
