import numpy as np
import scipy.optimize as opt

def obj(z):
    return -np.sum(z[2::3])

def cons(z):
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[3*i] - z[3*i+1]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[3*i] - z[3*i+1]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[3*i+2] - z[3*i+1]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[3*i+2] - z[3*i+1]})
        for j in range(i+1, 26):

            dy = z[3*i+1] - z[3*j+1]
            r1 = z[3*i+2]
            r2 = z[3*j+2]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: dist - (r1 + r2)})
    return constraints
