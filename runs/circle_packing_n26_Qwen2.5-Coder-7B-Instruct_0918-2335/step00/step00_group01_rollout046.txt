```python
import numpy as np
from scipy.optimize import minimize

n = 26
pair_idx = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])

def initial_guess(seed):
    np.random.seed(seed)
    return np.array(np.random.rand(4 * n), dtype=np.float32)

def objective(v):
    return -np.sum(v[2 * n:])

def constraints(v):
    centers = v[:2 * n].reshape((n, 2))
    r = v[2 * n:]
    boundary_constraints = np.maximum(0.0, 1 - centers - r, centers - r)
    pairwise_constraints = np.maximum(0.0, np.sum((centers[pair_idx[:, 0]] - centers[pair_idx[:, 1]]) ** 2, axis=1) - (r[pair_idx[:, 0]] + r[pair_idx[:, 1]]) ** 2)
    return np.concatenate((boundary_constraints.ravel(), pairwise_constraints))

bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

def run_packing():
    best_value = -np.inf
    best_v = None
    for seed in range(8):
        v0 = initial_guess(seed)
        res = minimize(
            fun=objective,
            x0=v0,
            method='SLSQP',
            constraints={'type': 'ineq', 'fun': constraints},
            bounds=bounds,
            options={
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if not res.success:
            continue
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x
    eps = 1e-7
    centers = best_v[:2 * n].reshape((n, 2))
    radii = np.maximum(0.0, best_v[2 * n:] - eps)
    return centers, radii, np.sum(radii)
```
