import numpy as np
from scipy.optimize import minimize


def objective(z):
    radii = z[26:]
    return -np.sum(radii)

def boundary_constraints(z):
    return z[:26] - z[26:]

def constraint_pairs(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    constraints = []
    for i in range(26):
        for j in range(i+1, 26):
            xi, yi, ri = x[i], y[i], r[i]
            xj, yj, rj = x[j], y[j], r[j]
            distance_constraint = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            constraints.append(distance_constraint)
    return constraints

def validate_packing(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
