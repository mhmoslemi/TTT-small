import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def calculate_forces(c, r):
    n = len(c)
    d = squareform(pdist(c))
    s = np.clip(r + r[:, np.newaxis] - d, 0, None)
    a = s * (c[:, np.newaxis, :] - c[np.newaxis, :, :]) / d[:, :, np.newaxis]
    F = np.sum(a, axis=1) + np.array([max(0, r[i] - min(c[i, 0], c[i, 1], 1 - c[i, 0], 1 - c[i, 1])) for i in range(n)])[:, np.newaxis]
    return F

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    MAX_ITER = 500
    τ0 = 1e-3
    γ = 0.95
    c = np.zeros((36, 2))
    r = np.zeros(36)
    for k in range(6):
        for l in range(6):
            c[k*6 + l] = np.array([(k+0.5)/6, (l+0.5)/6])
    c = c[:26]
    d = pdist(c)
    s = r + r[:, np.newaxis] - d
    valid = s > 0
    idx = np.where(valid)[0]
    while len(idx) > 1:
        i, j = idx[np.argmin(np.min(d[valid], axis=1))]
        c = np.delete(c, j, axis=0)
        r = np.delete(r, j)
        s = r + r[:, np.newaxis] - d
        valid = s > 0
        idx = np.where(valid)[0]
    x, y = np.random.normal(size=(26, 2)), np.random.normal(size=(26, 2))
    c = x * 0.02 + y * 0.98 + np.array([[i/50, j/50] for i in range(50) for j in range(50)]).T[:, :26]
    c = np.clip(c, 0, 1)
    while True:
        rvals = linprog(-r, bounds=[(0, 0.5)] * 26, A_eq=np.delete(squareform(pdist(c)), np.where(~valid)), b_eq=np.delete(d, np.where(~valid)), method='highs').x
        if np.allclose(r, rvals, atol=1e-8):
            break
        r = rvals
    ä = calculate_forces(c, r)
    τ = τ0
    for k in range(1, MAX_ITER+1):
        c += τ * ä
        g = np.clip(c[:, 0], 0, 1) - c[:, 0]
        if np.any(g > 0):
            c[:, 0] += g
        g = np.clip(c[:, 1], 0, 1) - c[:, 1]
        if np.any(g > 0):
            c[:, 1] += g
        g = c[:, 0] - np.clip(c[:, 0], 0, 1)
        if np.any(g > 0):
            c[:, 0] -= g
        g = c[:, 1] - np.clip(c[:, 1], 0, 1)
        if np.any(g > 0):
            c[:, 1] -= g
        if k % 50 == 0:
            c += np.random.normal(size=c.shape) * 0.02
            c = np.clip(c, 0, 1)
        rvals = linprog(-r, bounds=[(0, 0.5)] * 26, A_eq=np.delete(squareform(pdist(c)), np.where(~valid)), b_eq=np.delete(d, np.where(~valid)), method='highs').x
        if np.allclose(r, rvals, atol=1e-8):
            break
        r = rvals
        ä = calculate_forces(c, r)
        τ *= γ
    rvals = linprog(-r, bounds=[(0, 0.5)] * 26, A_eq=np.delete(squareform(pdist(c)), np.where(~valid)), b_eq=np.delete(d, np.where(~valid)), method='highs').x
    return c, rvals, np.sum(rvals)
