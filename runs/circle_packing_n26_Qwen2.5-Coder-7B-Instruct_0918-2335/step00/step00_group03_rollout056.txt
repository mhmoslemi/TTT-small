import numpy as np
import scipy.optimize
from scipy.spatial import distance_matrix

# Generate an initial hexagonal lattice configuration for the centers
def generate_hex_lattice(num_circles):
    radius = np.sqrt(3) / 4
    side = 2 * radius
    n = np.ceil(np.sqrt((num_circles * np.pi) / (np.sqrt(3) * side ** 2))).astype(int)
    centers = []
    for row in range(n):
        for col in range(n):
            x = col * side + side / 2 * (row % 2)
            y = row * radius * 3 / 2
            if x + side / 2 < 1:
                centers.append([x, y])
    if len(centers) > num_circles:
        centers = sorted(centers, key=lambda c: c[0])
        return centers[:num_circles]
    return centers * (num_circles // len(centers)) + centers[:num_circles % len(centers)]

# Evaluate a given center configuration
def evaluate_placement(p):
    C = p.reshape(26, 2)                         # centers
    d = np.linalg.norm(C[:,None,:]-C[None,:,:], axis=-1)  # 26x26 matrix
    n = C.shape[0]
    
    # A_ub and b_ub for the linear programming problem
    A_list = []
    b_list = []
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_pair = np.zeros((2, n))
            A_pair[0, i] = 1
            A_pair[0, j] = 1
            A_pair[1, i] = 1
            A_pair[1, j] = -1
            A_list.append(A_pair)
            b_list.append(d[i, j])
    
    # Boundary constraints (left, right, bottom, top)
    for i in range(n):
        A_wall = np.zeros((4, n))
        A_wall[0, i] = 1
        A_wall[1, i] = 1
        A_wall[2, i] = 1
        A_wall[3, i] = 1
        A_list.append(A_wall)
        b_list.append(C[i, 0])
        A_list.append(A_wall)
        b_list.append(1 - C[i, 0])
        A_list.append(A_wall)
        b_list.append(C[i, 1])
        A_list.append(A_wall)
        b_list.append(1 - C[i, 1])
    
    # Convert lists to numpy arrays
    A_ub = np.vstack(A_list)
    b_ub = np.array(b_list)
    
    # Non-negativity constraints
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    
    # Solve the linear programming problem
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, 1)] * n, method='highs')
    
    if res.success:
        return -res.fun
    else:
        return -np.inf
