    return clipped_polygon

def power_diagram_cell(centers, radii, i):
    polygon = np.array([[0, 0], [0, 1], [1, 1], [1, 0]])
    for j in range(len(centers)):
        if j != i:
            power_line = 2 * (centers[j] - centers[i])
            distance = np.dot(power_line, centers[i]) - (centers[i][0]**2 - centers[j][0]**2 + radii[j]**2 - radii[i]**2)
            clipped_polygon = half_plane_clip(polygon, power_line, distance)
            polygon = clipped_polygon
    return polygon

def centroid(polygon):
    if len(polygon) < 3:
        return np.mean(polygon, axis=0)
    area = 0.5 * np.abs(np.sum(polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1)))
    x = np.sum((polygon[:, 0] + np.roll(polygon[:, 0], 1)) * (polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1))) / (6 * area)
    y = np.sum((polygon[:, 1] + np.roll(polygon[:, 1], 1)) * (polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1))) / (6 * area)
    return np.array([x, y])

def distance_to_polygon(centre, polygon):
    distances = cdist([centre], polygon, 'euclidean').flatten()

    distances = np.where(perpendiculars >= 0, perpendiculars, distances)
    return np.min(distances)
