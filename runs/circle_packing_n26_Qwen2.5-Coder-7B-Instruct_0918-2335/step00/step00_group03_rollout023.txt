import numpy as np

def inside_square(p):
    return 0 <= p[0] <= 1 and 0 <= p[1] <= 1

def half_plane_clip(polygon, half_plane):

        epsilon = 1e-12
        return np.dot(p, half_plane[0]) + epsilon <= half_plane[1]

    clipped_polygon = [polygon[0]]
    for i in range(1, len(polygon) + 1):
        j = i % len(polygon)
        if point_inside(polygon[i-1], half_plane):
            clipped_polygon.append(polygon[i-1])
        if point_inside(polygon[j], half_plane) and not inside_square(polygon[j]):
            intersection = (polygon[j] - polygon[i-1]) * \
                           np.dot(polygon[j] - polygon[i-1], half_plane[0]) / \
                           np.dot(polygon[j] - polygon[i-1], half_plane[0])
            clipped_polygon.append(intersection)
    return np.array(clipped_polygon, dtype=float)

def power_diagram(centers, radii):
    n = centers.shape[0]
    diagram = []
    for i in range(n):
        half_planes = []
        for j in range(n):
            if i != j:
                half_plane = (2 * (centers[j] - centers[i]), np.dot(centers[j], centers[j]) - np.dot(centers[i], centers[i]) + radii[i] ** 2 - radii[j] ** 2)
                half_planes.append(half_plane)
        diagram.append(half_plane_clip(diagram[-1] if diagram else np.array([[0, 0], [0, 1], [1, 0], [1, 1]]), half_planes[-1]))
    return diagram

def centroid(polygon):
    if len(polygon) < 3:
        centroid = np.sum(polygon, axis=0) / len(polygon)
    else:
        area = 0.5 * np.sum(polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1))
        cx = np.sum((polygon[:, 0] + np.roll(polygon[:, 0], 1)) * (polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1))) / (6 * area)
        cy = np.sum((polygon[:, 1] + np.roll(polygon[:, 1], 1)) * (polygon[:, 0] * np.roll(polygon[:, 1], 1) - polygon[:, 1] * np.roll(polygon[:, 0], 1))) / (6 * area)
        centroid = np.array([cx, cy])
    return centroid

def distance_to_polygon(center, polygon):
    min_distance = np.inf
    n = len(polygon)
    for i in range(n):
        j = (i + 1) % n
        edge = polygon[j] - polygon[i]
        center_edge = center - polygon[i]
        distance = np.abs(np.cross(edge, center_edge)) / np.linalg.norm(edge)
        if np.dot(center_edge, edge) <= 0 or np.dot(center_edge - edge, edge) <= 0:
            distance_to_vertex = np.linalg.norm(center - polygon[i])
            if distance_to_vertex < distance:
                distance = distance_to_vertex
        min_distance = min(min_distance, distance)
    return min_distance
