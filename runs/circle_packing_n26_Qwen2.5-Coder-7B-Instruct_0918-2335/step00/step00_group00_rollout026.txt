```python
import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def initial_positions():
    circle_positions = np.array([[0.5, 0.5]])
    for i in range(25):
        x = np.random.rand()
        y = np.random.rand()
        while any(cdist(circle_positions, [[x, y]]) < 2 * np.sqrt(circle_positions.shape[0]) * 0.1):
            x = np.random.rand()
            y = np.random.rand()
        circle_positions = np.vstack((circle_positions, [[x, y]]))
    return circle_positions

def target_function(params):
    centers = params[:50].reshape(-1, 2)
    radii = params[50:]
    dist_matrix = cdist(centers, centers)
    overlap_penalty = np.sum((dist_matrix - (radii[:, None] + radii[None, :])) ** 2)
    boundary_penalty = np.sum(np.maximum(centers - 1, 0) ** 2 + np.maximum(1 - centers, 0) ** 2)
    radii_penalty = np.sum(np.maximum(-radii, 0) ** 2)
    return overlap_penalty + boundary_penalty + radii_penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_pos = initial_positions()
    initial_radii = np.random.rand(26) * 0.05
    initial_params = np.hstack((initial_pos.reshape(-1), initial_radii))
    result = minimize(target_function, initial_params, method='Nelder-Mead')
    if not result.success:
        raise ValueError("Optimization failed")
    centers = result.x[:50].reshape(-1, 2)
    radii = result.x[50:]
    return centers, radii, np.sum(radii)

centers, radii, sum_radii = run_packing()
print(centers)
print(radii)
print(sum_radii)
```
