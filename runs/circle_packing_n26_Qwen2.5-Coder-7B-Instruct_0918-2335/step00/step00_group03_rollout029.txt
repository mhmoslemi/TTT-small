import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def compute_constraints(positions):
    n = positions.shape[0]
    d = np.linalg.norm(positions[:, None, :] - positions[None, :, :], axis=-1)
    A = np.zeros((n * (n - 1) // 2, n * 2))
    b = np.zeros((n * (n - 1) // 2))
    for i in range(n):
        for j in range(i + 1, n):
            idx1, idx2 = 2 * i, 2 * j
            A[idx1, i] = 1
            A[idx1, j] = 1
            b[idx1] = d[i, j]
            
            idx2, idx3 = 2 * i + 1, 2 * j + 1
            A[idx2, i] = 1
            A[idx2, j] = 1
            b[idx2] = d[i, j]
    return A, b

def evaluate_placement(positions):
    n = positions.shape[0]
    d = np.linalg.norm(positions[:, None, :] - positions[None, :, :], axis=-1)
    A, b = compute_constraints(positions)
    c = -np.ones(n)
    res = linprog(c=c, A_ub=A, b_ub=b, method='highs')
    if not res.success:
        return -np.inf, positions, res.x
    return -res.fun, positions, res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
