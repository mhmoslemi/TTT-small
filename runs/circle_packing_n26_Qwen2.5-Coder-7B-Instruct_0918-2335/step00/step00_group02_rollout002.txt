import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    random_seeds = [0, 1, 2, 3]
    polish = True

    def objective(x):
        xs, ys, rs = x[:N], x[N:2*N], x[2*N:]
        boundary_penalty = (
            np.maximum(0, rs - xs)**2 +
            np.maximum(0, xs + rs - 1)**2 +
            np.maximum(0, rs - ys)**2 +
            np.maximum(0, ys + rs - 1)**2
        ).sum()
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist = np.hypot(dx, dy)
        rr = rs[:, None] + rs[None, :]
        overlap_penalty = np.sum(np.maximum(0, rr - dist[np.triu_indices(N, 1)])**2)
        return -rs.sum() + lambda_penalty * (boundary_penalty + overlap_penalty)

    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N

    best_x = None
    best_fun = float('-inf')

    for seed in random_seeds:
        np.random.seed(seed)
        result = differential_evolution(objective, bounds, popsize=DE_popsize, maxiter=DE_maxiter, polish=polish)
        if result.fun > best_fun:
            best_fun = result.fun
            best_x = result.x

    if best_x is None:
        raise ValueError("No valid packing found")

    xs, ys, rs = best_x[:N], best_x[N:2 * N], best_x[2 * N:]
    centres = np.column_stack([xs, ys])
    sum_radii = rs.sum()

    return centres, radii, sum_radii
