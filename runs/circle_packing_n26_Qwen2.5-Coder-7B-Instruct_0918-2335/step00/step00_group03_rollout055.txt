import numpy as np

def sutherland_hodgman(polygon, half_plane):
    input_vertices = np.vstack([polygon, polygon[0]])
    output_vertices = []
    x, y = half_plane
    
    for i in range(len(input_vertices) - 1):
        px1, py1 = input_vertices[i]
        px2, py2 = input_vertices[i + 1]
        dot1 = x * px1 + y * py1 - (x * px2 + y * py2)
        dot2 = x * px2 + y * py2 - (x * px1 + y * py1)
        
        if dot1 >= 0:
            output_vertices.append([px1, py1])
        if dot1 * dot2 < 0:
            output_x = px1 + dot1 * (px2 - px1) / (dot1 - dot2)
            output_y = py1 + dot1 * (py2 - py1) / (dot1 - dot2)
            output_vertices.append([output_x, output_y])
    
    return output_vertices

def power_diagram_cell(centers, radii, i):
    power_diagram = [[0, 0], [0, 1], [1, 0], [1, 1]]
    
    for j in range(len(centers)):
        if i != j:
            a = 2 * (centers[j] - centers[i])
            b = centers[j] @ centers[j] - centers[i] @ centers[i] + radii[i]**2 - radii[j]**2
            power_diagram = sutherland_hodgman(power_diagram, [a[0], a[1], b])
    
    return power_diagram

def centroid(polygon):
    polygon = np.array(polygon)
    area = 0.5 * np.sum(polygon[:-1, 0] * polygon[1:, 1] - polygon[:-1, 1] * polygon[1:, 0])
    cx = np.sum((polygon[:-1, 0] + polygon[1:, 0]) * (polygon[:-1, 0] * polygon[1:, 1] - polygon[:-1, 1] * polygon[1:, 0])) / (6 * area)
    cy = np.sum((polygon[:-1, 1] + polygon[1:, 1]) * (polygon[:-1, 0] * polygon[1:, 1] - polygon[:-1, 1] * polygon[1:, 0])) / (6 * area)
    return np.array([cx, cy])

def distance_to_polygon(point, polygon):
    return min(np.dot(polygon[i] - point, polygon[i] - point) for i in range(len(polygon)))
