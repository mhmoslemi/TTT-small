def project(pos):
    # Exact feasibility projection for PSO directly in the position space
    for i in range(N):
        pos[i*N + 0] = np.clip(pos[i*N + 0], pos[i*N + 2], 1 - pos[i*N + 2])
        pos[i*N + 1] = np.clip(pos[i*N + 1], pos[i*N + 2], 1 - pos[i*N + 2])
        if pos[i*N + 2] < EPS: pos[i*N + 2] = EPS
        
        for j in range(i + 1, N):
            dist = np.hypot(pos[i*N + 0] - pos[j*N + 0], pos[i*N + 1] - pos[j*N + 1])
            min_dist = pos[i*N + 2] + pos[j*N + 2]
            if dist < min_dist - EPS:
                shift = (min_dist – dist) / 2.0
                direction = ((pos[i*N + 0] - pos[j*N + 0]) / dist, (pos[i*N + 1] - pos[j*N + 1]) / dist)
                pos[i*N + 0] += shift * direction[0]
                pos[i*N + 1] += shift * direction[1]
                pos[j*N + 0] -= shift * direction[0]
                pos[j*N + 1] -= shift * direction[1]
                
    return pos
    
def fitness(positions):

    objective = -np.sum(pos[N:])

            return False, objective
        constraints = pos[N:]

            if i != j:
                d_ij = np.sqrt((pos[j*N : j*N+2] - pos[i*N : i*N+2]).dot(pos[j*N : j*N+2] - pos[i*N : i*N+2]))
                if d_ij < positions[i*N + 2] + positions[j*N + 2]:
                    constraint_feasibility = False
                    # Correcting the feasible circle by moving it away
                    shift = (positions[i*N + 2] + positions[j*N + 2] - d_ij) / 2.0
                    direction = (positions[j*N : j*N+2] - pos[i*N : i*N+2]) / d_ij
                    pos[j*N : j*N+2] -= shift * direction
                    
    return constraint_feasibility, objective
