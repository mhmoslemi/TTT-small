import numpy as np
import random

def project_infeasible(z):
    for i in range(26):
        x_i, y_i, r_i = z[i*3], z[i*3+1], z[i*3+2]
        x_i = np.clip(x_i, r_i, 1 - r_i)
        y_i = np.clip(y_i, r_i, 1 - r_i)
        r_i = max(r_i, 1e-12)
        z[i*3], z[i*3+1], z[i*3+2] = x_i, y_i, r_i
        for j in range(26):
            if i != j:
                x_j, y_j, r_j = z[j*3], z[j*3+1], z[j*3+2]
                d = np.hypot(x_i - x_j, y_i - y_j)
                min_dist = r_i + r_j
                if d < min_dist - 1e-12:
                    shift = (min_dist - d) / 2.0
                    direction = (x_i - x_j, y_i - y_j) / d
                    x_i += shift * direction[0]
                    y_i += shift * direction[1]
                    x_j -= shift * direction[0]
                    y_j -= shift * direction[1]
                    z[i*3], z[i*3+1], z[i*3+2], z[j*3], z[j*3+1], z[j*3+2] = x_i, y_i, r_i, x_j, y_j, r_j
    return z

def fitness(z):
    return -np.sum(z[2::3])

def pso(z, S=80, T=2000, w_max=0.9, w_min=0.4, c1=1.5, c2=1.5, v_max=0.05):
    n = len(z) // 3
    best = np.copy(z)
    for _ in range(S):
        rands = np.random.rand(n, 3*2)
        v = np.copy(z) * 0.
        p_best = np.copy(z)
        for k in range(T):
            for i, r in enumerate(rands):
                v[i] = w_max * v[i] + c1 * r[0*n] * (p_best[i] - z[i*3:i*3+3]) + c2 * r[1*n] * (best - z[i*3:i*3+3])
                x = np.clip(z[i*3:i*3+3] + v[i], 1 + v_max, 1 - v_max)
                z_new = np.copy(z)
                z_new[i*3:i*3+3] = x
                project_infeasible(z_new)
                if fitness(z_new) < fitness(best):
                    best = z_new
                    p_best[i] = z_new[i*3:i*3+3]
        yield best

def simulated_annealing(z, T0=0.05, α=0.995):
    n = len(z) // 3
    best = np.copy(z)
    for _ in range(20000):
        i = random.randint(0, n-1)
        σ_center = T0
        σ_radius = T0 / 2
        dx, dy = np.random.normal(scale=σ_center, size=2)
        dr = np.random.normal(scale=σ_radius)
        r_i = max(z[i*3+2] + dr, 0)
        z_new = np.copy(z)
        change = 0
        if r_i != 0:
            x = np.clip(z[i*3] + dx, r_i, 1 - r_i)
            y = np.clip(z[i*3+1] + dy, r_i, 1 - r_i)
            z_new[i*3], z_new[i*3+1], z_new[i*3+2] = x, y, r_i
            project_infeasible(z_new)
            d = fitness(z_new)
            change = d - fitness(z)
        if change < 0 or np.random.rand() < np.exp(-change / T0):
            z = z_new
            if d < fitness(best):
                best = z
        T0 *= α
    return best

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    z = np.random.rand(26*3) * (1 - 2 * 2e-12)
    best = next(pso(z))
    best = simulated_annealing(best)
    project_infeasible(best)
    centers = best[::3]
    radii = best[2::3]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
