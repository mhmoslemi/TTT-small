import numpy as np

def inflate(circles, Δr, ε):
    n = len(circles)
    for _ in range(5000):
        moved = False
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                d_ij = np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
                if d_ij < r1 + r2 - ε:
                    moved = True
                    d_ij_half = 0.5 * d_ij
                    delta_x = (r1 + r2 - d_ij) * (x2 - x1) / (2 * d_ij)
                    delta_y = (r1 + r2 - d_ij) * (y2 - y1) / (2 * d_ij)
                    circles[i] = (x1 + delta_x, y1 + delta_y, r1)
                    circles[j] = (x2 - delta_x, y2 - delta_y, r2)
            x, y, r = circles[i]
            circles[i] = (max(r, min(x, 1 - x)), max(r, min(y, 1 - y)), r)
            circles[i] = (min(circles[i][0] + Δr, min(circles[i][0], 1 - circles[i][0])), 
                         min(circles[i][1] + Δr, min(circles[i][1], 1 - circles[i][1])), 
                         circles[i][2]) if circles[i][2] < min(circles[i][0], 1 - circles[i][0], circles[i][1], 1 - circles[i][1]) - ε else circles[i]
        if not moved:
            break
    return [r for x, y, r in circles], [x for x, y, r in circles], [y for x, y, r in circles]

def init_circles(variant):
    c = 0.2
    circles = []
    for i in range(5):
        for j in range(5):
            circles.append((i * c + c / 2, j * c + c / 2, 0.01))
    if variant == 'A':
        circles.append((0.6, 0.9, 0.01))
    elif variant == 'B':
        circles.append((0.4, 0.9, 0.01))
    return circles

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    radius_step = 0.0005
    overlap_tolerance = 1e-6
    max_iterations = 5000
    seed_options = ['A', 'B']
    best_packing = None
    best_sum = -1

    np.random.seed(42)
    for seed in seed_options:
        circles = init_circles(seed)
        radii, _, _ = inflate(circles, radius_step, overlap_tolerance)
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            centers = [[circles[i][0], circles[i][1]] for i in range(len(circles))]
            best_packing = (centers, radii, best_sum)

    if not best_packing:
        centers = [[0.105 * (np.cos(2 * np.pi * (i + j) / 26) + 0.5), 
                    0.105 * (np.sin(2 * np.pi * (i + j) / 26) + 0.5)] 
                   for j in range(5) for i in range(range(6, 11, 5))]
        centers.append([np.random.uniform(0.04, 0.96), np.random.uniform(0.04, 0.96)])
        radii = [0.105] * 25 + [0.01]
        return np.array(centers), np.array(radii), np.sum(radii)

    return np.array(best_packing[0]), np.array(best_packing[1]), best_packing[2]
