import numpy as np
from scipy.spatial import Delaunay, distance
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for the center of the circles
    initial_centers = np.random.rand(26, 2)
    initial_radii = np.zeros(26)

    # Define the objective function to maximize the sum of radii
    def objective(x):
        centers, radii = x[:52].reshape(26, 2), x[52:].reshape(26,)
        return -np.sum(radii)  # We use negative because we will use a minimization function

    # Define the constraints
    # Non-negativity constraint for radii
    bounds = [(0, None)] * 52

    # Non-overlap constraint
    def non_overlap_constraint(x):
        centers, radii = x[:52].reshape(26, 2), x[52:].reshape(26,)
        return np.sum([ np.maximum(0, distance.cdist(centers[i].reshape(1,2), centers[j].reshape(1,2), 'euclidean')[0,0] - radii[i] - radii[j]) ** 2 for i in range(26) for j in range(i+1, 26) ])

    constraints = [{'type': 'ineq', 'fun': non_overlap_constraint}]

    # Run the optimization
    result = minimize(objective, np.concatenate((initial_centers.reshape(-1), initial_radii)), method='SLSQP', bounds=bounds, constraints=constraints)

    # Extract the results
    centers, radii = result.x[:52].reshape(26, 2), result.x[52:].reshape(26,)

    return centers, radii, -result.fun  # Negate the sum of radii to get the maximum
