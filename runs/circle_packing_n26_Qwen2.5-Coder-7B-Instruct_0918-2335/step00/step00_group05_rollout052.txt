import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonal lattice parameters
    s = 0.312
    grid = []
    k_max = 11  # 10x10 grid that fits 26 circles
    
    for kx in range(-(k_max//2), (k_max//2) + 1):
        for ky in range(-(k_max//2), (k_max//2) + 1):
            x = s + kx * s
            y = s + ky * (np.sqrt(3) / 2 * s)
            if 0 <= x <= 1 and 0 <= y <= 1:
                grid.append((x, y))
    
    # Select 26 random points from the lattice
    np.random.shuffle(grid)
    initial_centers = np.array(grid[:26])
    
    # Compute pairwise distances
    n = len(initial_centers)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(initial_centers[i] - initial_centers[j])
            dists[j, i] = dists[i, j]
    
    # Simulated Annealing parameters
    T = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = T / 5
    
    # Initial best solution
    best_radii = np.zeros(n)
    best_sum = 0
    current_centers = initial_centers.copy()
    
    # SA loop
    while T > T_min:
        for _ in range(M):
            # Choose random circle and move
            i = np.random.randint(n)
            delta = np.random.normal(scale=sigma_move, size=2)
            delta[delta < 0] = -np.abs(delta[delta < 0])
            delta[delta > 0] = np.abs(delta[delta > 0])
            delta /= np.linalg.norm(delta)  # Normalize for uniform movement
            delta *= np.random.uniform(low=0.1, high=5)  # Random step size
            proposed_center = current_centers[i] + delta
            proposed_center = np.clip(proposed_center, 0, 1)  # Clamp to [0,1]
            
            # Evaluate new configuration
            current_centers[i] = proposed_center
            proposed_dists = dists.copy()
            proposed_dists[i,:] = np.linalg.norm(proposed_center - current_centers, axis=1)
            proposed_dists[:,i] = proposed_dists[i,:]
            
            # Solve the LP
            c = np.ones(n)  # Objective is to maximize sum of r_i
            A = -np.column_stack((np.eye(n), np.eye(n)))
            b = dists.flatten()
            A_eq = np.column_stack((current_centers[:,0], current_centers[:,1], -current_centers[:,0], -current_centers[:,1]))
            b_eq = np.concatenate((current_centers[:,0], current_centers[:,1]))
            r = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
            
            if r.success and np.sum(r.x) > best_sum:
                best_radii = r.x
                best_sum = np.sum(r.x)
        
        # Cool down
        T *= c
    
    return current_centers, best_radii, np.sum(best_radii)
