import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial deterministic pattern
    centers = np.array([
        [0.05*i, 0.10] for i in range(6)] +
        [[0.12*i, 0.18] for i in range(5)] +
        [[0.05*i, 0.26] for i in range(6)] +
        [[0.12*i, 0.34] for i in range(5)] +
        [[0.15*i, 0.42] for i in range(4)] +
        [[0.50, 0.50]]
    ])
    
    def solve_lp(centers):
        radii = np.zeros(n)
        dist_ij = np.maximum(0, centers[:, None, :] - centers[None, :, :])
        dist_ij[range(n), range(n)] = 1e6  # diagonal don't count in
        r_i = np.minimum(dist_ij, (1 - centers[:, None, 0])[:, None, :], (centers[:, None, 0], 0), (1 - centers[:, None, 1], 0), (centers[:, None, 1], 0), key=np.min)
        r_i_plus_r_j = r_i + r_i.T
        A_ub = np.empty((4*n + n*(n-1)//2, n))
        b_ub = np.empty(4*n + n*(n-1)//2)
        A_ub[:n] = -r_i.T
        b_ub[:n] = np.zeros(n)
        A_ub[n:2*n] = r_i.T
        b_ub[n:2*n] = dist_ij.flat
        A_ub[2*n:4*n] = np.hstack((-np.eye(n), np.eye(n)))
        b_ub[2*n:4*n] = centers[:, 0]
        A_ub[4*n:] = np.hstack((np.eye(n), -np.eye(n)))
        b_ub[4*n:] = 1 - centers[:, 0]
        A_ub = A_ub.flatten('F').reshape(4*n + n*(n-1)//2, n)
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and not np.isnan(res.x).any():
            radii = res.x
        return radii, -res.fun
    
    def compute_forces(centres, radii, dual):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i):
                dist_ij = np.linalg.norm(centres[i] - centres[j])
                if dual[4*i + (j+4*i)%4*n + 1] > 1e-12:
                    forces[i] -= dual[4*i + (j+4*i)%4*n + 1] * (centres[i] - centres[j]) / dist_ij
                if dual[4*i + (j+4*i)%4*n + 2] > 1e-12:
                    forces[i] -= dual[4*i + (j+4*i)%4*n + 2] * (centers[j] - centres[i]) / dist_ij
                # Boundary pressures
                for k in range(2):
                    bound = 1 if k % 2 else 0
                    pressure = dual[4*i + 2*k*n + 3*k + 3]
                    if pressure > 1e-12:
                        dist = centres[i][k] - bound
                        force = pressure * dist / abs(dist)
                        forces[i, k] += force
        return forces
    
    eps = 1e-7
    best_obj = -float('inf')
    best_centres = None
    best_radii = None
    
    for _ in range(3):
        centres = 0.5 * ((np.random.rand(n, 2) * 2 - 1) + 1)
        for _ in range(200):
            radii, obj = solve_lp(centres)
            forces = compute_forces(centres, radii, linprog(np.zeros(4*n + n*(n-1)//2), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs').dual)
            centres += 0.01 * forces
            centres = np.clip(centres, 0 + eps, 1 - eps)
            if obj > best_obj:
                best_obj = obj
                best_centres = centres.copy()
                best_radii = np.maximum(0.0, radii - eps)
    
    if best_radii is None:
        best_obj = 0.0
        best_centres = np.zeros((n, 2))
        best_radii = np.zeros(n)
    
    return best_centres, best_radii, np.sum(best_radii)
