import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Generate initial positions: 5x5 grid for 25 + one extra
    positions = []
    for i in range(5):
        for j in range(5):
            positions.append(( (j+0.5)/5.0, (i+0.5)/5.0 ))
            if len(positions) == 25:
                break
        if len(positions) == 25:
            break

    # Add extra point at (0.6, 0.9)
    positions.append((0.6, 0.9))
    positions = np.array(positions)  # shape (26,2)

    radii_initial = np.full(n, 0.04)

    x0 = np.empty(78)
    x0[:52] = positions.ravel()
    x0[52:] = radii_initial

    # Bounds: create bounds list
    bounds = []
    for _ in range(2*x0.shape[0]):
        bounds.append((0.0, 1.0))

    # Pair indices
    i_indices = np.fromiter(("".join(map(str, ij)) for ij in
                            product(range(n), range(ij+1,n))),
                            dtype=int).reshape(-1,2)

    # Constraint function using vector
    def constraint_func(v):
        x = v[:52].reshape((n,2))
        r = v[52:]
        arr = []

        for _ in range(4):
            arr.extend(x[:,_] - r)
            arr.extend(1 - x[:,_] - r)

        dx = x[i_indices[:,0],0] - x[i_indices[:,1],0]
        dy = x[i_indices[:,0],1] - x[i_indices[:,1],1]
        dist2 = dx*dx + dy*dy
        pairwise_vals = dist2 - (2*r[i_indices]**2)
        arr.extend(pairwise_vals)

        return np.array(arr)

    # Use minimize
    res = minimize(fun=lambda v: -np.sum(v[52:]), x0=x0, bounds=bounds,
                   constraints={'type':'ineq', 'fun':constraint_func},
                   options={'maxiter':5000,'disp':False})

    # Reconstruct centers and radii
    centers = res.x[:52].reshape((n,2))
    radii = res.x[52:]

    # Ensure validity and compute sum of radii
    are_violations = constraint_func(res.x) <= np.finfo(float).eps
    if np.any(are_violations):
        shrink_factor = np.prod((1 - constraint_func(res.x)/(2*radii))[are_violations])
        radii *= shrink_factor

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
