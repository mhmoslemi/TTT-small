import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial seed for reproducibility
    np.random.seed(42)

    # Function to compute new centers after correcting overlaps
    def correct_displacements(centers, radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                delta = np.linalg.norm(centers[i] - centers[j]) - (radii[i] + radii[j])
                if delta < 0:
                    centers[i] += delta * (centers[i] - centers[j]) / (2 * np.linalg.norm(centers[i] - centers[j]))
                    centers[j] -= delta * (centers[i] - centers[j]) / (2 * np.linalg.norm(centers[i] - centers[j]))
        return centers

    def clip_to_square(centers, radii):
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        return centers

    def inflate_radii(centers, radii):
        for i in range(len(centers)):
            radii[i] = min(radii[i] + 0.0005, min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]))
        return radii

    def init_circles(variant):
