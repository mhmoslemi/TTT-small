import numpy as np
from scipy.spatial import Delaunay, distance_matrix

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the initial centers randomly within the unit square
    centers = np.random.rand(26, 2)
    
    # Define initial radii randomly within the unit square
    radii = np.random.rand(26)
    
    # Normalize radii such that their sum is close to 2.636
    total_radius = np.sum(radii)
    radii *= 2.636 / total_radius
    
    # Ensure non-overlap and boundary constraints are met
    while True:
        # Construct the Delaunay triangulation
        tri = Delaunay(centers)
        
        # Compute the circumcenter and circumradius of each triangle
        circumcenters = np.array([circumcenter(tri.simplices[i], centers) for i in range(tri.nsimplex)])
        circumradii = np.array([circumradius(tri.simplices[i], centers) for i in range(tri.nsimplex)])
        
        # Check for overlaps
        overlap = np.any(circumradii < radii[tri.simplices].sum(axis=1))
        if overlap:
            # Reposition circles to avoid overlaps
            for i, triangle in enumerate(tri.simplices):
                for j in triangle:
                    centers[j] += np.random.randn(2) * 0.1
        
        # Check if all circles are within the unit square
        outside_square = np.any(centers < 0) or np.any(centers > 1)
        if outside_square:
            # Reposition circles to be within the unit square
            centers -= np.min(centers, axis=0)
            centers *= 1.0 / np.max(centers, axis=0)
        
        # If no overlaps and all within the square, break the loop
        if not overlap and not outside_square:
            break
    
    # Return the final centers, radii, and sum of radii
    return centers, radii, np.sum(radii)

def circumcenter(triangle: np.ndarray, vertices: np.ndarray) -> np.ndarray:
    """Compute the circumcenter of a triangle given vertices."""
    A = vertices[triangle[0]]
    B = vertices[triangle[1]]
    C = vertices[triangle[2]]
    
    Bc = B - C
    Ac = A - C
    
    Ab = A - B
    Cb = C - B
    
    D = 2 * (Bc[0] * Ac[1] - Bc[1] * Ac[0])
    Ux = ((A[0]**2 + A[1]**2)*C[1] + (B[0]**2 + B[1]**2)*A[1] +
            (C[0]**2 + C[1]**2)*B[1] - (A[0]**2 + A[1]**2)*B[1] -
            (B[0]**2 + B[1]**2)*C[1] - (C[0]**2 + C[1]**2)*A[1])
    Uy = ((A[0]**2 + A[1]**2)*C[0] + (B[0]**2 + B[1]**2)*A[0] +
            (C[0]**2 + C[1]**2)*B[0] - (A[0]**2 + A[1]**2)*B[0] -
            (B[0]**2 + B[1]**2)*C[0] - (C[0]**2 + C[1]**2)*A[0])
    
    circumcenter = np.array([Ux / D, Uy / D])
    return circumcenter

def circumradius(triangle: np.ndarray, vertices: np.ndarray) -> float:
    """Compute the circumradius of a triangle given vertices."""
    circumcenter = circumcenter(triangle, vertices)
    A = vertices[triangle[0]]
    return distance_matrix(circumcenter.reshape(1, -1), A.reshape(1, -1))[0][0]
