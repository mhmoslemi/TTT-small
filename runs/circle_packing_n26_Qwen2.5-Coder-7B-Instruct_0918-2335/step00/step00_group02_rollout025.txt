import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def objective(z):
    centers = z[:78].reshape((-1, 2))
    radii = z[78:]
    return -np.sum(radii)

def constraints(z):
    centers = z[:78].reshape((-1, 2))
    radii = z[78:]
    n = centers.shape[0]
    cons = []
    
    # Linear constraints: 0 <= x <= 1, 0 <= y <= 1, 0 <= r <= 0.5
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - centers[i, 0] - radii[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: centers[i, 0] - radii[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - centers[i, 1] - radii[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: centers[i, 1] - radii[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: radii[i]})
    
    # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: (centers[i, 0] - centers[j, 0])**2 + 
                                                                 (centers[i, 1] - centers[j, 1])**2 - 
                                                                 (radii[i] + radii[j])**2})
    return cons

def run_packing():
    np.random.seed(123)
    max_iterations = 2000
    ftol = 1e-12
    gtol = 1e-8
    
    def initial_inserter():
        centers = np.random.rand(26, 2)
        radii = np.min(np.min(centers, axis=1)[:, np.newaxis], axis=1) * 1.5 * np.random.uniform(0.3, 0.8, (26,))
        return centers, radii
    
    best_centers = None
    best_radii = None
    best_obj_value = -np.inf
    for _ in range(5):
        centers, radii = initial_inserter()
        z0 = np.hstack((centers.flatten(), radii))
        result = minimize(objective, z0, method='trust-constr', jac=jacobian, hess='2-point', 
                         constraints=constraints(z0), bounds=bounds, options={'maxiter': max_iterations, 
                                                                               'ftol': ftol, 'gtol': gtol})
        if result.success and result.fun > best_obj_value:
            best_obj_value = result.fun
            best_centers = result.x[:78].reshape((-1, 2))
            best_radii = result.x[78:]
    
    sum_radii = -best_obj_value
    return best_centers, best_radii, sum_radii

def jacobian(z):
    centers = z[:78].reshape((-1, 2))
    radii = z[78:]
    n = centers.shape[0]
    jac = np.zeros_like(z)
    
    # Derivative with respect to radii
    for i in range(n):
        jac[i+78] = -1
    
    # Derivative with respect to centers
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
            disabled_radii = np.array([i+78, j+78])
            active_radii_ij_diff = np.delete(radii, disabled_radii) - np.delete(dist, disabled_radii)
            jacobian[i*2+0] = 2 * (centers[i, 0] - centers[j, 0]) / dist * (dist - np.sum(active_radii_ij_diff))
            jacobian[i*2+1] = 2 * (centers[i, 1] - centers[j, 1]) / dist * (dist - np.sum(active_radii_ij_diff))
            jacobian[j*2+0] = -jacobian[i*2+0]
            jacobian[j*2+1] = -jacobian[i*2+1]
    
    # Jacobian for linear constraints (1, -1)
    for i in range(n):
        indices = [2*i, 2*i+1]
        jac[indices] = np.eye(2)
    
    # Jacobian for boundary constraints (1, -1)
    for i in range(n):
        indices = [2*i+2, 2*i+3]
        jac[indices] = np.eye(2)
    
    return jac

bounds = [(0, 1), (0, 1)] * 26 + [(0, 0.5)] * 26
