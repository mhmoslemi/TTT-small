import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.optimize import linprog

def greedy_insertion(centers, radii, walls):
    vertices = []
    for i, c in enumerate(centers):
        for j, d in enumerate(centers):
            if i != j:
                for wall in walls:
                    dist_cw = np.linalg.norm(c - wall)
                    dist_dw = np.linalg.norm(d - wall)
                    if dist_cw >= radii[i] and dist_dw >= radii[j]:
                        pc = (c + d) / 2.0
                        r = dist_cw + dist_dw - radii[i] - radii[j]
                        vertices.append((pc, r))
    if not vertices:
        return None
    best_vertex = max(vertices, key=lambda v: v[1])
    return best_vertex[0], best_vertex[1]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([(0.5, 0.5)])
    radii = np.array([0.0])
    walls = np.array([(0, 0), (1, 0), (1, 1), (0, 1)])

    for _ in range(25):
        new_center, new_radius = greedy_insertion(centers, radii, walls)
        if new_radius <= 0:
            break
        centers = np.vstack((centers, new_center))
        radii = np.append(radii, new_radius)
        walls = np.vstack((walls, new_center))

    # Linear programming to optimize radii
    A = []
