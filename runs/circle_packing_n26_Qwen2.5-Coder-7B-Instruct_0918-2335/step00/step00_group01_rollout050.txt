import numpy as np
from scipy.optimize import linprog

def solve_LP(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_eq = np.zeros((0, n))
    b_eq = []
    A_ub = np.zeros((4*n + 325, n))
    b_ub = np.zeros(4*n + 325)

    # Pairwise constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r_max = np.min([x, 1-x, y, 1-y])
        A_ub[idx, i] = 1
        b_ub[idx] = r_max
        idx += 1

    res = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
    if not res.success:
        raise Exception("LP solving failed")
    return res.x, -res.fun

def update_positions(centers, radii, duals):
    alpha = 0.01
    n = centers.shape[0]
    forces = np.zeros((n, 2))

    # Pairwise forces
    for i in range(n):
        for j in range(i + 1, n):
            if duals[i*n + j] > 1e-12:
                dist = np.linalg.norm(centers[i] - centers[j])
                force = duals[i*n + j] * (centers[i] - centers[j]) / dist
                forces[i] += force
                forces[j] -= force

    # Boundary forces
    for i in range(n):
        x, y = centers[i]
        rx, ry = radii[i], radii[i]
        if x < 2*rx:
            F = duals[n + i] * np.array([1, 0])
            forces[i] -= F
        if y < 2*ry:
            F = duals[n + n + i] * np.array([0, 1])
            forces[i] -= F
        if 1-x < 2*rx:
            F = duals[n + 2*n + i] * np.array([-1, 0])
            forces[i] -= F
        if 1-y < 2*ry:
            F = duals[n + 3*n + i] * np.array([0, -1])
            forces[i] -= F

    centers += alpha * forces / (np.linalg.norm(forces, axis=1, keepdims=True) + 1e-6)
    return np.clip(centers, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_positions = np.array([
        [0.05, 0.15], [0.15, 0.15], [0.25, 0.15], [0.35, 0.15], [0.45, 0.15], [0.55, 0.15],
        [0.05, 0.23], [0.15, 0.23], [0.25, 0.23], [0.35, 0.23], [0.45, 0.23], [0.55, 0.23],
        [0.05, 0.31], [0.15, 0.31], [0.25, 0.31], [0.35, 0.31], [0.45, 0.31], [0.55, 0.31],
        [0.05, 0.39], [0.15, 0.39], [0.25, 0.39], [0.35, 0.39], [0.45, 0.39], [0.55, 0.39],
        [0.05, 0.47], [0.15, 0.47], [0.25, 0.47], [0.35, 0.47], [0.45, 0.47]
    ])
    best_obj = -np.inf
    best_centres, best_radii = None, None

    for restart in range(3):
        centres = initial_positions
        for iteration in range(200):
            radii, obj = solve_LP(centres)
            if obj > best_obj:
                best_obj = obj
                best_centres, best_radii = centres.copy(), radii.copy()
            duals = linprog(np.zeros(n), A_eq=None, b_eq=None, A_ub=A_ub, b_ub=b_ub, bounds=(0, None)).Dual
            if np.any(np.isinf(duals)):
                break
            centres = update_positions(centres, radii, duals)
        
        centres += 0.1 * np.random.rand(*centres.shape) - 0.05

    radii = np.maximum(0.0, best_radii - 1e-7)
    return best_centres, radii, np.sum(radii)
