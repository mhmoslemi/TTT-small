import numpy as np
from scipy.optimize import minimize

def penalty_bound(x, y, r, M_bound=1e2):
    return M_bound * max(0, r - min(x, 1 - x, y, 1 - y))**2

def penalty_overlap(c_i, c_j, r_i, r_j, M_overlap=1e2):
    dist = np.sqrt(np.sum((c_i - c_j)**2))
    return M_overlap * max(0, r_i + r_j - dist)**2

def objective(params, M_bound, M_overlap):
    centers = params[:26*2].reshape((26, 2))
    radii = params[26*2:]
    return -np.sum(radii) + np.sum([penalty_bound(x, y, r, M_bound) for x, y, r in zip(centers[:, 0], centers[:, 1], radii)]) + np.sum([penalty_overlap(c_i, c_j, r_i, r_j, M_overlap) for i in range(26) for j in range(i + 1, 26)])

def penalty_function(params, M_bound, M_overlap):
    return objective(params, M_bound, M_overlap)

def penalty_gradient(params, M_bound, M_overlap):
    centers = params[:26*2].reshape((26, 2))
    radii = params[26*2:]
    grad = np.zeros_like(params)
    for i in range(26):
        for j in range(2):
            grad[j + i * 2] = np.sum([
                -1 / radii[i] * (penalty_bound(centers[i, 0], centers[i, 1], radii[i], M_bound))
                - 2 * (penalty_overlap(centers[i, 0], centers[j, 0], radii[i], radii[j], M_overlap) * (centers[j, j%2] - centers[i, j%2]))
                for j in range(26) if j != i
            ])
        grad[i * 2 + 26] = np.sum([
            -1 + (penalty_bound(centers[i, 0], centers[i, 1], radii[i], M_bound) + 2 * (penalty_overlap(centers[i, 0], centers[j, 0], radii[i], radii[j], M_overlap) * (radii[j] / np.sqrt(np.sum((centers[j] - centers[i])**2)))))
            for j in range(26) if j != i
        ])
    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([[np.random.rand() for _ in range(2)] for _ in range(26)])
    radii = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1])) - 1e-4
    params = np.hstack((centers.ravel(), radii))
    res = minimize(penalty_function, params, jac=penalty_gradient, method='SLSQP', bounds=[(0, 1) for _ in range(26*2)] + [(0, None) for _ in range(26)], options={'maxiter': 1000})
    best_params = res.x
    best_centers = best_params[:26*2].reshape((26, 2))
    best_radii = best_params[26*2:]
    best_sum_radii = -res.fun
    return best_centers, best_radii, best_sum_radii
