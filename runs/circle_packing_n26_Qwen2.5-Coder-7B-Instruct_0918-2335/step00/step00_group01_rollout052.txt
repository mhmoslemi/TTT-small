import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def maximize_sum_of_radii():
    # Hexagonal grid initialization
    n = 26
    grid_points = np.array([
        [(x, y) for x in np.linspace(0.1, 0.9, 6) for y in np.linspace(0.1, 0.875, 5)],
        [(x, y) for x in np.linspace(0.1, 0.9, 6) for y in np.linspace(0.125, 0.95, 5)],
        [(x, y) for x in np.linspace(0.1, 0.9, 5) for y in np.linspace(0.15, 0.90, 4)]
    ])
    grid_points = grid_points.ravel(order='F')[:, np.newaxis, :]
    grid_points = grid_points.reshape(-1, 2, 2)
    grid_points = grid_points.reshape(-1, 2)
    grid_points = grid_points[:n]
    
    # Initial radius guess
    initial_radii = np.tile(np.min(grid_points, axis=1) - 1e-4, n)
    
    # Flatten the initial centers and radii
    initial_state = np.hstack((grid_points, initial_radii[:, np.newaxis]))
    
    # Parameters
    P = 50
    generations = 800
    delta = 0.05
    sigma_pos = 0.05
    sigma_rad = 0.02
    M_bound = 10**2
    M_overlap = 10**2
    M_bound_multiplier = 3
    M_overlap_multiplier = 3
    temperature = 0.97
    convergence_stagnation = 150
    fudge_factor = 1e-7
    
    best_solution = None
    best_fitness = -np.inf
    stagnation_count = 0
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = []
        for p in range(P):
            centers = initial_state[p, :n, :2]
            radii = initial_state[p, n:]
            total_length = 0
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    penalty_overlap = max(0, radii[i] + radii[j] - dist)**2
                    penalty_bound = max(0, (radii[i] - min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]))**2
                    total_length += -radii[i] + M_bound * penalty_bound + M_overlap * penalty_overlap
            fitness.append(total_length)
        
        # Tournament selection
        new_state = np.zeros((P, 2*n))
        for p in range(P):
            i, j, k = np.random.permutation(P)[:3]
            if fitness[i] >= fitness[j] and fitness[i] >= fitness[k]:
                new_state[p] = initial_state[i]
            elif fitness[j] >= fitness[i] and fitness[j] >= fitness[k]:
                new_state[p] = initial_state[j]
            else:
                new_state[p] = initial_state[k]
        
        # Crossover
        for p in range(0, P, 2):
            parent1 = new_state[p, :n], new_state[p, n:]
            parent2 = new_state[p+1, :n], new_state[p+1, n:]
            child1 = [(np.random.choice(parent1[0][i], parent2[0][i]) for i in range(2)), 
                        np.random.choice(parent1[1][i], parent2[1][i]) for i in range(n)])
            child2 = [(np.random.choice(parent2[0][i], parent1[0][i]) for i in range(2)), 
                        np.random.choice(parent2[1][i], parent1[1][i]) for i in range(n)]
            new_state[p, :] = child1
            new_state[p+1, :] = child2
        
        # Mutation
        for p in range(P):
            for i in range(n):
                if np.random.rand() < 0.4:
                    new_state[p, i*2] += np.random.normal(0, sigma_pos)
                    new_state[p, i*2+1] += np.random.normal(0, sigma_pos)
                if np.random.rand() < 0.6:
                    new_state[p, n+i] += np.random.normal(0, sigma_rad)
        
        # Repair
        for p in range(P):
            for i in range(n):
                x, y = new_state[p, i*2, :], new_state[p, i*2+1]
                r = new_state[p, n+i]
                if r > min(x[0], 1-x[0], x[1], 1-x[1]):
                    r = min(x[0], 1-x[0], x[1], 1-x[1])
                new_state[p, n+i] = r
                for j in range(n):
                    if i != j:
                        dist = np.linalg.norm(new_state[p, i*2:i*2+2] - new_state[p, j*2:j*2+2])
                        if dist < new_state[p, n+i] + new_state[p, n+j]:
                            if dist == 0:
                                new_state[p, n+i] = new_state[p, n+j] = 0
                            else:
                                new_state[p, n+i] = new_state[p, n+i] * (new_state[p, n+i] + new_state[p, n+j]) / dist
                                new_state[p, n+j] = new_state[p, n+j] * (new_state[p, n+i] + new_state[p, n+j]) / dist
                                new_state[p, n+i] = max(0, new_state[p, n+i])
                                new_state[p, n+j] = max(0, new_state[p, n+j])
                                new_state[p, n+i] = new_state[p, n+i] - fudge_factor  # Ensure strict feasibility
                                new_state[p, n+j] = new_state[p, n+j] - fudge_factor  # Ensure strict feasibility
        
        # Update parameters
        if gen % 20 == 0:
            M_bound *= M_bound_multiplier
            M_overlap *= M_overlap_multiplier
            sigma_pos *= temperature
            sigma_rad *= temperature
        
        # Update best solution
        current_best_fit = max(fitness)
        if current_best_fit > best_fitness:
            best_fitness = current_best_fit
            best_solution = initial_state[np.argmax(fitness)]
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= convergence_stagnation:
                break
    
    # Post-processing: LP optimization
    final_centers = best_solution[:n, :2]
    final_radii = best_solution[n:]
    M_bound *= 0.001  # Reduce to 0.1 for LP to focus on precision rather than feasibility
    M_overlap *= 0.001
    constrained_radii = np.clip(final_radii, 0, None)
    dists = cdist(final_centers, final_centers)
    overlap_penalties = (dists - constrained_radii[:, np.newaxis] - constrained_radii[np.newaxis, :]) ** 2
    objective = -np.sum(constrained_radii) + M_bound * np.sum(np.maximum(0, overlap_penalties))
    result = minimize(lambda x: -np.sum(x[:n]), constrained_radii, 
                        method="SLSQP", bounds=[(0, None) for _ in range(n)],
                        constraints=[{"type": "ineq", "fun": lambda x, y, i, j: y[i] + y[j] - x[dists[i, j] + i]} 
                                      for i in range(n) for j in range(i+1, n) if dists[i, j] > 0])
    final_radii = result.x[:n]
    
    return final_centers, final_radii, np.sum(final_radii) + M_bound * np.sum(np.maximum(0, dists - final_radii[:, np.newaxis] - final_radii[np.newaxis, :]) ** 2)

centers, radii, sum_radii = maximize_sum_of_radii()

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

is_valid, msg = validate_packing(centers, radii)
print(f"Validation: {is_valid}, {msg}")
print(f"Final sum of radii: {sum_radii}")
