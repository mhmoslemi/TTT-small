import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def obj(v):
        r = v[52:]
        return -np.sum(r)

    def constraints(v):
        pos = v[:52].reshape(26, 2)
        r = v[52:]
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((26, 26), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26
    n_restarts = 15
    best_radius_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(n_restarts):
        init = np.concatenate([np.random.rand(52), 0.02 * np.ones(26)])
        res = minimize(obj, x0=init, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success and np.all(constraints(res.x) >= -1e-10):
           amin = res.fun
            if amin > best_radius_sum:
                best_radius_sum = amin
                best_centers = res.x[:52].reshape(26, 2)
                best_radii = res.x[52:]

    if not best_centers is None and not best_radii is None:
        return best_centers, best_radii, -best_radius_sum
    else:
        raise Exception("Optimization failed to find a valid solution")
