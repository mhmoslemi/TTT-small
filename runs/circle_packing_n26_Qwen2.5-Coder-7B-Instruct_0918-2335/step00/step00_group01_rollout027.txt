import numpy as np
from scipy.optimize import linprog
import heapq

def run_packing():
    n = 26
    min_radius = 1e-4
    growth_rate = 1e-3
    velocity_magnitude = 5e-3
    max_time = 50.0
    shrinkage_epsilon = 1e-6

    # Initialize positions randomly and velocities randomly
    centers = np.random.rand(n, 2)
    radii = np.full(n, min_radius)
    velocities = velocity_magnitude * (np.random.rand(n, 2) * 2 - 1)

    # Priority queue for events
    event_queue = []

    # Function to compute the time until next collision
    def compute_collision_time(centre1, centre2, radius1, radius2, velocity1, velocity2):
        delta_c = centre1 - centre2
        delta_v = velocity1 - velocity2
        delta_r = radius1 + radius2
        if np.dot(delta_v, delta_c) <= 0:
            return None  # No collision
        a = np.dot(delta_v, delta_v)
        b = 2 * np.dot(delta_c, delta_v)
        c = np.dot(delta_c, delta_c) - delta_r**2
       判别式
        if b**2 - 4*a*c >= 0:
            t = (-b + np.sqrt(b**2 - 4*a*c)) / (2*a)
            if t > 0:
                return t
    for i in range(n):
        for j in range(i + 1, n):
            t = compute_collision_time(centers[i], centers[j], radii[i], radii[j], velocities[i], velocities[j])
            if t is not None:
                heapq.heappush(event_queue, (t, 'circle-circle', i, j))
        for d, bx in enumerate([0, 1]):
            x = 2 * radii[i] if centers[i, d] < bx else (1 - centers[i, d]) / velocities[i, d] + radii[i]
            if x > 0 and x < max_time:
                t = bx / velocities[i, d]
                heapq.heappush(event_queue, (t, 'circle-wall', i, d))
    
    # Simulation loop
    current_time = 0
    while event_queue:
        t_e, event_type, *args = heapq.heappop(event_queue)
        delta_t = t_e - current_time
        for i in range(n):
            centers[i] += velocities[i] * delta_t
            radii[i] += growth_rate * delta_t
        
        current_time = t_e
        
        if event_type == 'circle-circle':
            i, j = args
            delta_c = centers[i] - centers[j]
            delta_v = velocities[i] - velocities[j]
            overlap = min_radius
            for k in range(2):
                if delta_c[k] < 0:
                    overlap = min(overlap, radius[j] - x)
                else:
                    overlap = min(overlap, radius[i] - x)
            shrink = velocities[i] + velocities[j]
            shrink = overlap / shrink / 2
            velocities[i] -= shrink
            velocities[j] -= shrink
        elif event_type == 'circle-wall':
            i, d = args
            responsibility = velocities[i, d]
            velocities[i, d] = -responsibility

    # Apply epsilon shrinkage and perform overlap repair
    radii = radii * (1 - shrinkage_epsilon)
    epsilon = 1e-6
    while True:
        has_overlap = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - epsilon:
                    offsets = (radii[j] - dist / 2 + epsilon) * (centers[i] - centers[j]) / dist
                    centers[i] += offsets
                    centers[j] -= offsets
                    has_overlap = True
        if not has_overlap:
            break

    # LP polish for optimal radii
    c = -np.ones(n)  # Maximize sum of radii
    A = np.zeros((3*n, n))
    b = np.zeros(3*n)
    for i in range(n):
        A[3*i, i] = 1
        A[3*i+1, i] = -4 * centers[i, 0]
        A[3*i+2, i] = -4 * centers[i, 1]
        b[3*i+1] = -2 * radii[i]**2
        b[3*i+2] = -2 * radii[i]**2
    result = linprog(c, A_ub=A, b_ub=b, bounds=(min_radius, None))
    optimized_radii = result.x

    return centers, optimized_radii, np.sum(optimized_radii)
