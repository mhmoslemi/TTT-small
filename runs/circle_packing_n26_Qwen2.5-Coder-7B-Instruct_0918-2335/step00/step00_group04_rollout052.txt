import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def generate_initial_grid():
    centers = np.zeros((36, 2))
    for k in range(6):
        for l in range(6):
            centers[6*k+l] = ((k+0.5)/6, (l+0.5)/6)
    return centers

def remove_blocking_circles(centers):
    n = len(centers)
    distances = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(distances, np.inf)
    for i in range(n):
        distances[i, i] = np.inf
    sorted_indices = np.argsort(distances, axis=1)
    removed = set()
    for i in range(n):
        for j in sorted_indices[i]:
            if j in removed:
                continue
            if distances[i, j] == np.min(distances[i, j+1:]):
                removed.add(j + 1)
    remaining_indices = np.array([x for x in range(n) if x not in removed])
    return centers[remaining_indices]

def random_perturbation(centers):
    eps = 0.01
    centers += eps * np.random.randn(len(centers), 2)
    centers = np.clip(centers, 0, 1)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    MAX_ITER = 300
    tau0 = 1e-3
    gamma = 0.95
    tolerance = 1e-8
    np.random.seed(42)
    centers = generate_initial_grid()
    centers = remove_blocking_circles(centers)
    centers = random_perturbation(centers)
    obj = np.zeros(len(centers))
    cons = []
    bounds = [(0, 0.5)] * 2 * len(centers)

    for attempt in range(3):
        for k in range(MAX_ITER):
            r = np.zeros(len(centers))

            for i in range(len(centers)):
                for j in range(i+1, len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    s_ij = dist - (r[i] + r[j])
                    if s_ij > 0:
                        r[i] += s_ij / 2
                        r[j] += s_ij / 2

                for col in range(2):
                    r[col] += min(centers[col], 1-centers[col])

                obj = np.sum(r)

            if k % 50 == 0:
                centers = random_perturbation(centers)

            A = np.zeros((len(centers), len(centers)))
            b = np.zeros(len(centers))

            for i in range(len(centers)):
                for j in range(len(centers)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if i == j:
                        A[i, j] = 1
                    else:
                        A[i, j] = -1 / dist
            b = obj * np.ones(len(centers))

            res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            centers = res.x.reshape((-1, 2))
            obj = res.fun

            if np.allclose(b, res.fun, atol=tolerance):
                break

        radii = np.array([np.sqrt(np.sum((centers[i] - centers[j]) ** 2) - (obj + (np.sum(centers[i] - centers[j]) ** 2)) / (2 * res.x[i])) for i in range(len(centers))])
        result = validate_packing(centers, radii)
        if result[0]:
            return centers, radii, np.sum(radii)

    return np.zeros((26, 2)), np.zeros(26), 0
