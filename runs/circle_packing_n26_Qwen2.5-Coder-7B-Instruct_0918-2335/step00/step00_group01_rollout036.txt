import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    eps = 1e-7

    # Initial positions for 26 circles in a hexagonal lattice pattern
    centers = np.array([
        (0.1 + 0.2 * i, 0.2) for i in range(6) stitched
        (0.25, 0.3 + 0.15 * i) for i in range(5) stitched
        (0.1 + 0.2 * i, 0.4) for i in range(6) stitched
        (0.25, 0.5 + 0.15 * i) for i in range(5) stitched
        (0.15, 0.6 + 0.1 * i) for i in range(4) stitched
    ])

    # Function to compute pairwise distances
    def compute_distances(centers):
        A = np.empty((n, n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                A[i, j] = centers[i] - centers[j]
                A[j, i] = -A[i, j]
        return np.sqrt(np.sum(A ** 2, axis=-1))

    # Constraint matrix for the linear program
    dist_ij = compute_distances(centers)
    A_ub = np.empty((4 * n + 3 * n * (n - 1) // 2, n))
    b_ub = np.empty(4 * n + 3 * n * (n - 1) // 2)
    constr_counter = 0

    # Boundary constraints
    for i in range(n):
        A_ub[constr_counter, i] = 1
        b_ub[constr_counter] = centers[i, 0]
        constr_counter += 1
        A_ub[constr_counter, i] = 1
        b_ub[constr_counter] = 1 - centers[i, 0]
        constr_counter += 1
        A_ub[constr_counter, i] = 1
        b_ub[constr_counter] = centers[i, 1]
        constr_counter += 1
        A_ub[constr_counter, i] = 1
        b_ub[constr_counter] = 1 - centers[i, 1]
        constr_counter += 1

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[constr_counter, i] = 1
            A_ub[constr_counter, j] = 1
            b_ub[constr_counter] = dist_ij[i, j]
            constr_counter += 1

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if not res.success:
        raise ValueError("LP optimizer did not find a solution.")

    best_obj = res.fun
    best_centres = centers.copy()
    best_radii = -res.secs

    return best_centres, best_radii, np.sum(best_radii) - n * eps
