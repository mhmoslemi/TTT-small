import numpy as np
from scipy.optimize import minimize

def hex_lattice(n):
    s = np.sqrt(3) / 3
    rows = np.int(np.ceil(np.sqrt(3 * n / (np.sqrt(3) + 3))))
    cols = np.int(np.ceil((n - rows * (rows - 1) / 2) / (rows + 1)))
    x = np.array([j for i in range(rows) for j in np.linspace(i * s / 2 + s / 4, i * s / 2 + (cols - i * (i + 1) / 2) * s - s / 4, min(rows, cols - i * (i + 1) / 2))])
    y = np.repeat(np.linspace(s / 2, 1 - s / 2, rows), cols)
    return x[:n], y[:n]

def boundary_adjust(centers, radii):
    min_dis = np.sqrt(3) * min(radii)
    m = len(centers)
    offset = (1 - min_dis) / 2
    centers[:, 0] += offset
    centers[:, 1] += offset
    centers[:, 0] -= np.min(centers[:, 0])
    centers[:, 1] -= np.min(centers[:, 1])
    centers *= (1 - 2 * offset) / (1 - offset)
    return centers, radii

def obj(z, n):
    centers = z[:2 * n].reshape(n, 2)
    radii = z[2 * n:]
    penalties = np.sum(np.maximum(0, -centers[:, 0])) + np.sum(np.maximum(0, -1 + centers[:, 0])) + \
                 np.sum(np.maximum(0, -centers[:, 1])) + np.sum(np.maximum(0, -1 + centers[:, 1])) + \
                 np.sum(np.maximum(0, -1 + np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2)))) + \
                 np.sum(np.maximum(0, -radii))
    return -radii.sum() + 1e6 * penalties

def grad(z, n):
    centers = z[:2 * n].reshape(n, 2)
    radii = z[2 * n:]
    obj_grad = -np.ones_like(radii)
    cons_grads = []
    for i in range(n):
        for j in range(i):
            r_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if r_ij > radii[i] + radii[j]:
                continue
            rel_dist = (radii[i] - radii[j]) / r_ij
            cons_grads.append(np.array([-rel_dist * (centers[j] - centers[i]) / np.sqrt((centers[j] - centers[i]) ** 2), 0]))
            cons_grads.append(np.array([0, -rel_dist * (centers[j] - centers[i]) / np.sqrt((centers[j] - centers[i]) ** 2)]))
            cons_grads.append(np.array([ rel_dist * (centers[j] - centers[i]) / np.sqrt((centers[j] - centers[i]) ** 2), 0]))
            cons_grads.append(np.array([0,  rel_dist * (centers[j] - centers[i]) / np.sqrt((centers[j] - centers[i]) ** 2)]))
            cons_grads.append(np.array([-1 / np.sqrt((centers[i, 0] - centers[j, 0]) ** 2), 0]))
            cons_grads.append(np.array([0, -1 / np.sqrt((centers[i, 1] - centers[j, 1]) ** 2)]))
            cons_grads.append(np.array([1 / np.sqrt((centers[i, 0] - centers[j, 0]) ** 2), 0]))
            cons_grads.append(np.array([0, 1 / np.sqrt((centers[i, 1] - centers[j, 1]) ** 2)]))
    return np.concatenate((obj_grad, np.sum(cons_grads, axis=0)))

def solve_stage(z0, delta):
    cons = [
        {'type': 'ineq', 'fun': lambda z: np.minimum(z[2 * n:], 1e-12)},
        {'type': 'ineq', 'fun': lambda z: np.sum(np.maximum(0, 1 - z[2 * n:] * np.sqrt(np.sum((z[:2 * n].reshape(n, 2)[:, np.newaxis, :] - z[:2 * n].reshape(n, 2)[np.newaxis, :, :]) ** 2, axis=2))))}
    ]
    res = minimize(lambda z: obj(z, n) - delta * np.sum(np.maximum(0, -z[2 * n:])), z0, method='SLSQP', jac=grad, constraints=cons, options={'maxiter': 1000, 'disp': False})
    return res.x

def blue_balls(centers, radii):
    n = len(centers)
    eps = 1e-10
    for _ in range(1000):
        delta_radius = np.minimum(np.minimum(np.minimum(1 - centers[:, 0] + radii, centers[:, 0] + radii),
                                             np.minimum(1 - centers[:, 1] + radii, centers[:, 1] + radii)),
                                  np.minimum(np.minimum(np.min([np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2)) for i in range(n) for j in range(i)]), 1000 * eps),
                                           np.diff((radii[:-1] + radii[1:]), prepend=0, append=0)))
        centers += (delta_radius / 2)[:, np.newaxis] * (centers - centers[:, np.newaxis, :]) / np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        centers = np.maximum(np.minimum(centers, 1 - eps), eps)
        delta_r = np.sum(delta_radius)
        if delta_r < eps:
            break
    return centers, radii

def validate(centers, radii):
    for i in range(n):
        for j in range(i):
            r_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if r_ij < radii[i] + radii[j]:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers, radii = hex_lattice(26 * 3)
    centers, radii = boundary_adjust(centers, radii)
    z0 = np.concatenate((centers.flatten(), radii))
    centers, radii = blue_balls(centers, radii)
    delta = 1e-8
    z = solve_stage(z0, delta)
    sum_radii = sum(radii)
    return centers.reshape(-1, 2), radii, sum_radii

centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}\nRadii: {radii}\nSum of Radii: {sum_radii}")
