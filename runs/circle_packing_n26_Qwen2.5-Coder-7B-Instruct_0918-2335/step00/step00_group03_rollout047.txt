import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    center_ids = [
        0,    1, 3, 7,     2, 4, 10, 14,    6, 8, 11, 15, 18, 19, 23, 24,
        33, 35, 37, 41, 38, 40, 46, 50, 44, 48, 52, 54, 57, 58, 62, 63
    ]
    centers = [
        (r_c, r_c), (1-r_c, r_c), (r_c, r_c), (1-r_c, 1-r_c), (r_c, 1-r_c),
        (0.5, r_e), (0.5, 1-r_e), (r_e, 0.5), (1-r_e, 0.5), (r_e, 1),
        (1, r_e), (1, 1-r_e), (1-r_e, 1), (0.5, 0.5+2*r_i), (0.5, 0.5-2*r_i)
    ]
    centers = np.array(centers[:26])
    centers[center_ids, :] += np.array([[2*0.5, 0], [2*(1-r_c), 0], [0, 2*r_e]])
    radii = np.full(26, r_i)
    return centers, radii

def compute_sum_radii(radii):
    return np.sum(radii)

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def local_obj(x):
    centers = x[:52].reshape(26,2)
    radii = x[52:]
    return -compute_sum_radii(radii)

def single_circle_overlap(centers, i, j):
    return np.linalg.norm(centers[i] - centers[j]) - radii[i] - radii[j]

def run_packing():
    bounds = [
        (0.30, 0.49),
        (0.10, 0.25),
        (0.05, 0.15)
    ]
    result_de = differential_evolution(objective, bounds, popsize=10,
                                       tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    constraints = [
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[0]-x[52:54])**2) - (radii[0] + radii[1]) + 1e-8},

        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[3]-x[52:54])**2) - (radii[1] + radii[4]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[4]-x[52:54])**2) - (radii[4] + radii[5]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[5]-x[52:54])**2) - (radii[5] + radii[3]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[6]-x[52:54])**2) - (radii[6] + radii[1]) + 1e-8},

        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[8]-x[52:54])**2) - (radii[6] + radii[8]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[9]-x[52:54])**2) - (radii[4] + radii[9]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[10]-x[52:54])**2) - (radii[14] + radii[10]) + 1e-8},
        {'type': 'ineq', 'fun': lambda x: np.sqrt((x[11]-x[52:54])**2) - (radii[18] + radii[11]) + 1e-8},

    result_slsqp = minimize(local_obj, x0, method='SLSQP',

                            options={'ftol':1e-10,'gtol':1e-10,'maxiter':2000})

    sum_r = radii.sum()
    return centers, radii, sum_r
