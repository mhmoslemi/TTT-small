import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.array([
        [r_c, r_c], [1-r_c, r_c], [r_c, 1-r_c], [1-r_c, 1-r_c],
        [0.5, r_e], [0.5, 1-r_e], [r_e, 0.5], [1-r_e, 0.5],
        [0.5 + 0.5 * np.cos(2 * np.pi * i / 7), 0.5 + 0.5 * np.sin(2 * np.pi * i / 7)] for i in range(1, 7)
    ])
    radii = np.array([0.5] * 4 + [r_e] * 4 + [r_i] * 18)
    return centers, radii

def compute_sum_radii(radii):
    return radii.sum()

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    sum_r = compute_sum_radii(radii)
    return -sum_r  # Negative because we'll use minimization

def constraint_func(v):
    centers, radii = v[:52].reshape(26,2), v[52:]
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        x, y = centers[i]
        if r_i < x-r_i or r_i < y-r_i or r_i < 1-x-r_i or r_i < 1-y-r_i:
            constraints.append((r_i, 0))
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = r_i + r_j - dist
            if overlap < 0:
                constraints.append((overlap, 0))
    return constraints

def run_packing():
    bounds = [
        (0.30, 0.49),
        (0.10, 0.25),
        (0.05, 0.15)
    ]
    result_de = differential_evolution(objective, bounds, popsize=10,
                                       tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [(-1e-3, 1) for _ in range(52)]
    cons = [{'type': 'ineq', 'fun': lambda v: c[0] for c in constraint_func(v)}]
    res = minimize(lambda v: -compute_sum_radii(v[52:]), x0, method='SLSQP',
                   bounds=bounds_slsqp, constraints=cons,
                   options={'ftol':1e-10,'gtol':1e-10,'maxiter':2000})
    centers = res.x[:52].reshape(26,2)
    radii = res.x[52:]

    sum_r = compute_sum_radii(radii)
    return centers, radii, sum_r
