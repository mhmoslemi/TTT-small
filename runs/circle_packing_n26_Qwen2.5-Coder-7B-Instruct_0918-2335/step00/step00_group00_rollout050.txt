import numpy as np

def reflect_to_boundary(x, y, r):
    x = np.maximum(r, np.minimum(x, 1 - r))
    y = np.maximum(r, np.minimum(y, 1 - r))
    return x, y

def inflate(circles, delta_r, epsilon):
    n = circles.shape[0]
    max_iter = 5000
    for _ in range(max_iter):
        convergence = True
        for i in range(n):
            for j in range(i + 1, n):
                dx = circles[i, 0] - circles[j, 0]
                dy = circles[i, 1] - circles[j, 1]
                d = np.sqrt(dx**2 + dy**2)
                if d < circles[i, 2] + circles[j, 2] - epsilon:
                    overlap = circles[i, 2] + circles[j, 2] - d
                    circles[i, 0] += overlap * dx / (2 * d)
                    circles[i, 1] += overlap * dy / (2 * d)
                    circles[j, 0] -= overlap * dx / (2 * d)
                    circles[j, 1] -= overlap * dy / (2 * d)
                    convergence = False
        new_radii = np.minimum(circles[:, 2] + delta_r, np.minimum(circles[:, 0], 1 - circles[:, 0], circles[:, 1], 1 - circles[:, 1]) - epsilon)
        radius_changes = new_radii - circles[:, 2]
        circles[:, 2] = new_radii
        for i in range(n):
            if not np.isclose(radius_changes[i], 0):
                x, y = circles[i, :2]
                circles[i, :2] = reflect_to_boundary(x, y, circles[i, 2])
        if convergence:
            break
    return circles[:, 2], circles[:, :2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    variant_a_centers = np.array([
        [0.0, 0.0], [0.2, 0.0], [0.4, 0.0], [0.6, 0.0], [0.8, 0.0],
        [0.0, 0.2], [0.2, 0.2], [0.4, 0.2], [0.6, 0.2], [0.8, 0.2],
        [0.0, 0.4], [0.2, 0.4], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
        [0.0, 0.6], [0.2, 0.6], [0.4, 0.6], [0.6, 0.6], [0.8, 0.6],
        [0.1, 0.8], [0.5, 0.8], [0.9, 0.8]
    ])
    variant_b_centers = np.array([
        [0.0, 0.0], [0.2, 0.0], [0.4, 0.0], [0.6, 0.0], [0.8, 0.0],
        [0.0, 0.2], [0.2, 0.2], [0.4, 0.2], [0.6, 0.2], [0.8, 0.2],
        [0.0, 0.4], [0.2, 0.4], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
        [0.0, 0.6], [0.2, 0.6], [0.4, 0.6], [0.6, 0.6], [0.8, 0.6],
        [0.3, 0.8], [0.5, 0.8], [0.7, 0.8]
    ])
    initial_radii = np.full((26, 1), 0.01)
    circles_a = np.hstack((variant_a_centers, initial_radii))
    circles_b = np.hstack((variant_b_centers, initial_radii))
    radii_a, centers_a = inflate(circles_a, 0.0005, 1e-6)
    radii_b, centers_b = inflate(circles_b, 0.0005, 1e-6)
    sum_radii_a = np.sum(radii_a)
    sum_radii_b = np.sum(radii_b)
    return centers_a if sum_radii_a > sum_radii_b else centers_b, radii_a if sum_radii_a > sum_radii_b else radii_b, np.max((sum_radii_a, sum_radii_b))
