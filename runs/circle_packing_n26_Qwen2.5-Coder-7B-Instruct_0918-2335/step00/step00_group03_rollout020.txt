import numpy as np
from scipy.optimize import minimize

def pack_circle():
    np.random.seed(42)
    N = 26
    r_min = 1e-3
    iterations = 10000
    
    def rsa_seed(N, r_min, iterations):
        circles = []
        for n in range(N):
            while True:
                x, y = np.random.uniform(0, 1, 2)
                if all((x - r_min - circle[0])**2 + (y - r_min - circle[1])**2 >= 4*r_min**2 for circle in circles):
                    circles.append((x, y, r_min))
                    break
        return np.array(circles)
    
    initial_config = rsa_seed(N, r_min, iterations)
    positions = initial_config[:, :2]
    radii = initial_config[:, 2]
    
    def physics_relaxation(positions, radii):
        for _ in range(2000):
            new_positions = np.copy(positions)
            new_radii = np.copy(radii)
            for i in range(N):
                forces = np.zeros(2)
                for j in range(N):
                    if i != j:
                        d_ij = np.sqrt(np.sum((new_positions[i] - new_positions[j])**2))
                        gap = d_ij - (new_radii[i] + new_radii[j])
                        if gap < 0:
                            repulsive_force = -gap / d_ij * (new_positions[j] - new_positions[i])
                            forces += repulsive_force
                boundary_force = np.array([max(0, r - x - 1e-12, 1e-12 - x) for x, r in zip(new_positions[i], new_radii[i])])
                forces -= boundary_force
                new_positions[i] += forces * 1e-3
                new_radii[i] *= np.sqrt(np.sum(forces**2))
            new_radii = np.clip(new_radii, 0, None)
            positions = new_positions
            radii = new_radii
        return positions, radii
    
    positions, radii = physics_relaxation(positions, radii)
    
    def global_scaling(positions, radii):
        gaps = [(np.sqrt(np.sum((positions[i] - positions[j])**2)) - (radii[i] + radii[j])) for i in range(N) for j in range(i+1, N)]
        scaling_factor_position = min(min([min(np.abs(x_i - x_j - r_i - r_j) for j in range(N) if i != j] for x_i in positions[:, 0]) for x_i in positions[:, 1])]
        scaling_factor_radius = min([min((1 - r_i) / r_i for r_i in radii)] + [min(1 - x_i for x_i in positions[:, i]) for i in [0, 1]])
        scaling_factor = min(scaling_factor_position, scaling_factor_radius)
        if scaling_factor > 1:
            return positions, radii
        else:
            scaled_positions = 0.5 + scaling_factor * (positions - 0.5)
            scaled_radii = scaling_factor * radii
            return scaled_positions, scaled_radii
    
    positions, radii = global_scaling(positions, radii)
    
    def objective(x):
        positions = x[:52].reshape(N, 2)
        radii = x[52:]
        return -np.sum(radii)
    
    def constraint_1(x):
        positions = x[:52].reshape(N, 2)
        radii = x[52:]
        return np.array([-np.sum((positions[i] - positions[j])**2) + (radii[i] + radii[j])**2 for i in range(N) for j in range(i+1, N)])
    
    def constraint_2(x):
        positions = x[:52].reshape(N, 2)
        radii = x[52:]
        return np.array([np.min([r - x_i for x_i, r in zip(positions[i], radii[i])]) for i in range(N)])
    
    constraints = [{'type': 'ineq', 'fun': constraint_1}, {'type': 'ineq', 'fun': constraint_2}]
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * N
    initial_guess = np.hstack((positions.flatten(), radii))
    
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-10)
    centers = result.x[:52].reshape(N, 2)
    radii = result.x[52:]
    
    return centers, radii, np.sum(radii)
