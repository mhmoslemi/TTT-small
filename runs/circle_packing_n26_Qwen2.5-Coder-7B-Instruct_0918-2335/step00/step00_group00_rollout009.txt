import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) / 2
    radii_sum = np.sum(radii)

    def objective(vars):
        centers = vars[:2 * n].reshape((n, 2))
        radii = vars[2 * n:]
        radii_sum = np.sum(radii)
        return -(radii_sum + 1e-6)  # Maximizing the sum of radii

    def constraints(vars):
        centers = vars[:2 * n].reshape((n, 2))
        radii = vars[2 * n:]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = distance_matrix(centers[i:i + 1], centers[j:j + 1])[0, 0]
                cons.append({'type': 'ineq', 'fun': lambda vars, i=i, j=j, dist=dist: dist - radii[i] - radii[j] + 1e-6})
            cons.append({'type': 'ineq', 'fun': lambda vars, i=i: radii[i]})
            cons.append({'type': 'ineq', 'fun': lambda vars, i=i: 1 - radii[i]})
            cons.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[i * 2]})
            cons.append({'type': 'ineq', 'fun': lambda vars, i=i: 1 - vars[i * 2 + 1]})
        return cons

    x0 = np.concatenate((centers.flatten(), radii))

    result = minimize(objective, x0, constraints=constraints, method='SLSQP')
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]
    radii_sum = np.sum(radii)

    return centers, radii, radii_sum
