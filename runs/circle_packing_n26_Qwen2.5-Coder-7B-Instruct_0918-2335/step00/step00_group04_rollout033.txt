import numpy as np
from scipy.optimize import minimize

def objective(v):
    r = v[52:]  # last 26 entries
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26, 2)  # (x,y) pairs
    r = v[52:]  # radii

    # boundary terms – 4*26
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r  # left/right
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r

    # pairwise terms – 26 choose 2
    diffs = pos[:, None, :] - pos[None, :, :]  # (26,26,2)
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n) * 0.95  # random initial guess with small radii

    best_sol = None
    best_val = -np.inf
    for _ in range(20):
        res = minimize(fun=objective, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type': 'ineq',
                                                   'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success:
            if constraints(res.x).max() < 1e-10:  # Check constraint feasibility
                obj_val = -objective(res.x)
                if obj_val > best_val:
                    best_sol = res.x
                    best_val = obj_val

    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    centers = best_sol[:52].reshape(n, 2)
    radii = best_sol[52:]
    return centers, radii, best_val
