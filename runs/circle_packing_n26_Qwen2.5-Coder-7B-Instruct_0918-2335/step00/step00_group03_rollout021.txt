import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    r_min = 1e-3
    iterations = 10000

    def rsa_seed():
        circles = []
        for _ in range(N):
            attempts = 0
            while attempts < iterations:
                x, y = np.random.rand(2)
                r = r_min
                if np.all((x - r) >= 0) and np.all((x + r) <= 1) and np.all((y - r) >= 0) and np.all((y + r) <= 1):
                    if not any(np.linalg.norm(np.array([x, y]) - np.array(circle[:2])) < r + circle[2] for circle in circles):
                        circles.append([x, y, r])
                        break
                attempts += 1
            if attempts == iterations:
                return rsa_seed()
        return circles

    def boundary_force(r_i, x_i):
        return max(0, r_i - x_i, 0, (1 - x_i) - r_i)

    def penalty(circles):
        overlap = 0
        for i in range(N):
            for j in range(i + 1, N):
                d_ij = np.linalg.norm(np.array(circles[i][:2]) - np.array(circles[j][:2]))
                gap = d_ij - (circles[i][2] + circles[j][2])
                overlap += max(0, gap) ** 2
        boundary = 0
        for circle in circles:
            x, y, r = circle
            boundary += boundary_force(r, x) + boundary_force(r, y) + boundary_force(r, 1 - x) + boundary_force(r, 1 - y)
        return overlap + boundary

    def objective(circles):
        r_sum = 0
        for circle in circles:
            r_sum += circle[2]
        return -r_sum

    def gradient(circles):
        grad = np.zeros(3 * N)
        for i in range(N):
            for j in range(i + 1, N):
                d_ij = np.linalg.norm(np.array(circles[i][:2]) - np.array(circles[j][:2]))
                gij = 1 if d_ij >= circles[i][2] + circles[j][2] else -np.linalg.norm(np.array(circles[i][:2]) - np.array(circles[j][:2])) / (circles[i][2] + circles[j][2] - d_ij)
                grad[i * 3] += 0.5 * (grad[j * 3] + gij)
                grad[i * 3 + 1] += 0.5 * (grad[j * 3 + 1] + gij)
                grad[i * 3 + 2] -= gij
                grad[j * 3] += 0.5 * (grad[i * 3] - gij)
                grad[j * 3 + 1] += 0.5 * (grad[i * 3 + 1] - gij)
                grad[j * 3 + 2] -= gij
        for circle in circles:
            x, y, r = circle
            grad[i * 3] += boundary_force(r, x) / (2 * r)
            grad[i * 3 + 1] += boundary_force(r, y) / (2 * r)
            grad[i * 3 + 2] += boundary_force(r, 1 - x) / (2 * r)
            grad[i * 3 + 3] += boundary_force(r, 1 - y) / (2 * r)
        return grad

    circles = rsa_seed()
    circles_vec = np.array([item for sublist in circles for item in sublist])

    res = minimize(objective, circles_vec, jac=gradient, method='SLSQP', bounds=[(0, 1), (0, 1), (0, 0.5)] * N, constraints={'type': 'ineq', 'fun': penalty})

    optim_circles_vec = res.x
    optim_circles = [optim_circles_vec[i:i+3] for i in range(0, len(optim_circles_vec), 3)]

    for circle in optim_circles:
        mean_radius = np.mean([circle[2]])
        circle[2] = mean_radius

    centers = np.array([circle[:2] for circle in optim_circles])
    radii = np.array([circle[2] for circle in optim_circles])

    return centers, radii, radii.sum()
