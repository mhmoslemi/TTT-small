```python
import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def objective(x):
        return -np.sum(x)

    def constraints(x):
        n = 26
        circles = x.reshape((n, 3))
        centers = circles[:, :2]
        radii = circles[:, 2]
        constraints = [
            {'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i, 2]},
            {'type': 'ineq', 'fun': lambda x, i=i: x[i, 2]},
            {'type': 'ineq', 'fun': lambda x, i=i, j=j: x[i, 2] + x[j, 2] - np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2) + 1e-12}
            for i in range(n) for j in range(i+1, n)
        ]
        return constraints

    x0 = np.ones(78)
    bounds = [(0, 1), (0, 1), (0, None)] * 26
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints())

    circles = result.x.reshape((26, 3))
    centers = circles[:, :2]
    radii = circles[:, 2]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
```
