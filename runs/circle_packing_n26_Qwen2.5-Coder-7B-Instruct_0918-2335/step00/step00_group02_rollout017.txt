import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x) -> float:
    N = 26
    xs, ys, rs = x[:2*N].reshape(-1, 2), x[2*N:2*N+N], x[2*N+N:]
    
    # Sum of radii
    sum_radii = -rs.sum()
    
    # Boundary penalties
    p_boundary = (
        np.maximum(0, rs - xs[:, 0])**2 +
        np.maximum(0, xs[:, 0] + rs - 1)**2 +
        np.maximum(0, rs - ys[:, 0])**2 +
        np.maximum(0, ys[:, 0] + rs - 1)**2
    ).sum()
    
    # Overlap penalties
    dx = xs[:, None, 0] - xs[None, :, 0]
    dy = ys[:, None, 0] - ys[None, :, 0]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    overlap = rr - dist
    p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
    
    return sum_radii + 1e5 * (p_boundary + p_overlap)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * (2*26) + [(0, 0.5)] * 26
    
    # Differential Evolution
    result = differential_evolution(objective, bounds, popsize=12, maxiter=400, random_state=0)
    best_x = result.x
    
    # Local refinement with L-BFGS-B
    final_result = minimize(objective, best_x, method='L-BFGS-B', bounds=bounds)
    final_x = final_result.x
    
    # Extract circles
    centres = np.column_stack([final_x[0:26*2:2], final_x[1:26*2:2]])
    radii = final_x[26*2:]
    sum_radii = radii.sum()
    
    return centres, radii, sum_radii
