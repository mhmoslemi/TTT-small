import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialisation
    n = 26
    x bounds = (1e-12, 1 - 1e-12)
    y_bounds = (1e-12, 1 - 1e-12)
    maxIter = 3000
    dt = 1e-3
    alpha = 4.0
    beta = 0.8
    gamma = 1e-4
    epsilon = 1e-12

    # Random initial positions
    centers = np.random.uniform(epsilon + 0.5 * dt, 1 - epsilon - 0.5 * dt, size=(n, 2))
    radii = np.full(n, 1e-6)

    # Dynamics loop
    for step in range(maxIter):
        # Compute pairwise forces
        F = np.zeros((n, n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - epsilon:
                    delta = radii[i] + radii[j] - d
                    F[i, j] = alpha * delta * (centers[i] - centers[j]) / d
                    F[j, i] = -F[i, j]

        # Wall repulsion
        for i in range(n):
            x_min = max(centers[i, 0] - radii[i] - epsilon, x_bounds[0])
            x_max = min(centers[i, 0] + radii[i] + epsilon, x_bounds[1])
            y_min = max(centers[i, 1] - radii[i] - epsilon, y_bounds[0])
            y_max = min(centers[i, 1] + radii[i] + epsilon, y_bounds[1])
            F[i, i][0] += alpha * (x_min - centers[i, 0]) if centers[i, 0] < x_min else alpha * (x_max - centers[i, 0])
            F[i, i][1] += alpha * (y_min - centers[i, 1]) if centers[i, 1] < y_min else alpha * (y_max - centers[i, 1])

        # Update velocities and positions
        velocities = np.zeros((n, 2))
        for i in range(n):
            velocities[i] += dt * np.sum(F[i], axis=0) / n - beta * velocities[i]
            centers[i] += dt * velocities[i]

        # Global uniform radius increment
        radii += gamma

        # Feasibility clip
        for i in range(n):
            r_i_allowed = min(np.min([np.linalg.norm(centers[i] - centers[j]) - radii[j] for j in range(n) if j != i if np.linalg.norm(centers[i] - centers[j]) > radii[i] + radii[j] - epsilon]), 
                             1 - centers[i, 0], centers[i, 0] - epsilon,
                             1 - centers[i, 1], centers[i, 1] - epsilon)
            radii[i] = min(radii[i], r_i_allowed)

        # Convergence check
        if np.all(np.abs(centers - centers[:-1]) < epsilon):
            break

    # LP refinement of radii (fixed centres)
    c = -radii

    beq = np.zeros(n)

        Aeq[i, 4*i+3] = 1.0
        beq[i] = 1.0 - radii[i]
        for j in range(n):
            Aeq[i, 4*j+2] = 1.0
            Aeq[i, 4*j+1] = centers[i, 0] - centers[j, 0]
            Aeq[i, 4*j] = centers[i, 1] - centers[j, 1]
            beq[i] -= radii[j]

    result = linprog(c, A_ub=[-np.eye(n)], b_ub=[np.zeros(n)], A_eq=Aeq, b_eq=beq, bounds=(0, None))
