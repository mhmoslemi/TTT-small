import numpy as np
from scipy.optimize import minimize

def initial_guess(seed):
    np.random.seed(seed)
    positions = np.random.uniform(0.1, 0.9, size=(26, 2))
    radii = np.zeros(26)
    return np.hstack((positions, radii))

def objective(v):
    return -np.sum(v[26:])

def constraints(v):
    n = 26
    centers = v[:2*n].reshape((n, 2))
    radii = v[26:]
    boundary_constraints = np.minimum(centers[:, 0] + radii, 1) - np.maximum(centers[:, 0] - radii, 0)
    boundary_constraints = np.minimum(boundary_constraints, np.minimum(centers[:, 1] + radii, 1) - np.maximum(centers[:, 1] - radii, 0))
    distance_matrix = np.sum((centers[:, np.newaxis] - centers)**2, axis=2)
    distance_matrix -= (radii[:, np.newaxis] + radii)**2
    distance_constraints = 1e-12 - distance_matrix
    return np.hstack((boundary_constraints, distance_constraints.flatten()))

n = 26
RESTARTS = 10
eps = 1e-7

best_value = -np.inf
best_v = None
for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun        = objective,
        x0         = v0,
        method     = 'SLSQP',
        constraints= {'type':'ineq', 'fun':constraints},
        bounds     = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n,
        options    = {
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if not res.success:
        continue
    value = -res.fun  # because we minimised the negative
    if value > best_value:
        best_value = value
        best_v = res.x

centers = best_v[:2*n].reshape((n,2))
radii = np.maximum(0.0, best_v[2*n:] - eps)
sum_radii = np.sum(radii)

return centers, radii, sum_radii
