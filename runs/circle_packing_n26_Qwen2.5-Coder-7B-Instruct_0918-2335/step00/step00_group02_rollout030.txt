import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x, lambda_penalty):
    N = 26
    xs, ys, rs = x[:2*N], x[2*N:4*N], x[4*N:]
    boundaries = (1 - rs) * (xs + rs > 1) + rs * (xs < R) + (1 - rs) * (ys + rs > 1) + rs * (ys < R)
    overlaps = (rs[:, None] + rs[None, :] - np.hypot(xs[:, None] - xs[None, :], ys[:, None] - ys[None, :])) * (rs[:, None] + rs[None, :] > np.hypot(xs[:, None] - xs[None, :], ys[:, None] - ys[None, :]))
    return -rs.sum() + lambda_penalty * (np.sum(boundaries) + np.sum(overlaps))

def get_run_packing():
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    random_seed = range(4)

    best_x = None
    best_fun = -np.inf

    for seed in random_seed:
        np.random.seed(seed)
        result = differential_evolution(objective, bounds=[(0, 1), (0, 1), (0, 0.5)] * N, args=(lambda_penalty,), popsize=DE_popsize, maxiter=DE_maxiter)
        if result.fun > best_fun:
            best_x = result.x
            best_fun = result.fun

    if best_x is None:
        return np.zeros((N, 2)), np.zeros(N), 0.

    bounds = [(0, 1), (0, 1), (0, 0.5)] * N
    result = minimize(lambda x: objective(x, lambda_penalty), best_x, bounds=bounds, method='L-BFGS-B')

    best_x = result.x
    N = 26
    xs, ys, rs = best_x[:2*N], best_x[2*N:4*N], best_x[4*N:]
    return np.column_stack([xs, ys]), rs, -rs.sum()

run_packing = get_run_packing
