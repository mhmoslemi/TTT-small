import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    resolution = 1 / R
    grid = np.mgrid[0:1:resolution, 0:1:resolution].reshape(2, -1).T
    
    r_f = np.minimum(grid[:, 0], grid[:, 1], 1 - grid[:, 0], 1 - grid[:, 1])
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    sum_radii = 0.0
    
    for i in range(26):
        k = np.argmax(r_f)
        x, y = grid[k]
        r = r_f[k] - 1e-9
        centers[i] = [x, y]
        radii[i] = r
        sum_radii += r
        
        dx = grid[:, 0] - x
        dy = grid[:, 1] - y
        dist = np.sqrt(dx**2 + dy**2) - r
        r_f = np.minimum(r_f, dist)
        
        # Zero out used point
        r_f[k] = -1e9
    
    # Local refinement
    Delta_c = 1e-3
    N_refine = 200
    
    for _ in range(N_refine):
        i = np.random.randint(26)
        x, y = centers[i]
        u_x, u_y = np.random.uniform(-Delta_c, Delta_c, 2)
        c_prime = [x + u_x, y + u_y]
        
        dx = grid[:, 0] - c_prime[0]
        dy = grid[:, 1] - c_prime[1]
        dist = np.sqrt(dx**2 + dy**2)
        
        r_prime = np.minimum(r_f, dist)
        if r_prime[i] > radii[i]:
            centers[i] = c_prime
            radii[i] = r_prime[i]
            sum_radii += r_prime[i] - radii[i]
    
    # Final radius cleanup
    centers = centers.astype(np.float32)
    radii = radii.astype(np.float32)
    sum_radii = (np.minimum(np.minimum(radii, np.maximum(radii, r_f - 1e-9)), 
                           -np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)).sum())
    sum_radii = max(0, sum_radii)
    
    return centers, radii, sum_radii
