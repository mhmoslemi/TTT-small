import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    g_tot = 2 * g
    v_mag = 5 * 1e-3
    t_max = 50.0
    eps = 1e-6
    
    # Initialize parameters
    np.random.seed(42)
    centers = np.random.rand(n, 2) * 0.96  # Random positions within a buffer of 2% from the boundary
    centers += np.random.randn(n, 2) * 1e-3  # Add small random jitter
    radii = np.full(n, r_min)
    velocities = np.random.randn(n, 2) * v_mag
    
    # Function to compute collision times
    def compute_collision_times(centers, radii, velocities, g_tot, eps):
        collision_times = np.inf * np.ones(n)
        for i in range(n):
            for j in range(i + 1, n):
                Δc = centers[i] - centers[j]
                Δv = velocities[i] - velocities[j]
                Δr = radii[i] + radii[j]
                if np.linalg.norm(Δv) > 1e-6:
                    discriminant = np.linalg.norm(Δc) ** 2 - (Δr + g_tot * Δt) ** 2
                    if discriminant >= 0:
                        t_cc = (-Δc @ Δv + np.sqrt(discriminant)) / np.linalg.norm(Δv)
                        if t_cc > 0:
                            collision_times[i] = min(collision_times[i], t_cc)
                            collision_times[j] = min(collision_times[j], t_cc)
        
        # Wall collisions
        walls = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
        for wall in walls:
            wall_indices = np.where((centers[:, 0] < wall[Horizontal] or centers[:, 0] > wall[Horizontal] - 2 * radii) &
                                    (centers[:, 1] < wall[Vertical] or centers[:, 1] > wall[Vertical] - 2 * radii))
            for i in wall_indices[0]:
                Δc = np.array(wall) - centers[i]
                Δv = velocities[i]
                Δr = radii[i]
                if np.linalg.norm(Δv) > 1e-6:
                    discriminant = np.linalg.norm(Δc) ** 2 - (Δr - g_tot * Δt) ** 2
                    if discriminant >= 0:
                        t_cw = (Δc @ Δv - np.sqrt(discriminant)) / np.linalg.norm(Δv)
                        if t_cw > 0:
                            collision_times[i] = min(collision_times[i], t_cw)
        
        return collision_times
    
    # Event-driven simulation
    current_time = 0
    event_queue = compute_collision_times(centers, radii, velocities, g_tot, eps)
    while np.min(event_queue) < t_max:
        indices = np.where(event_queue == np.min(event_queue))
        i, j = indices[0][0], indices[1][0]
        t_e = event_queue[min(i, j)]
        
        # Update positions and radii
        Δt = t_e - current_time
        centers[i] += velocities[i] * Δt
        centers[j] += velocities[j] * Δt
        radii[i] += g * Δt
        radii[j] += g * Δt
        current_time = t_e
        
        # Resolve event
        Δc = centers[i] - centers[j]
        Δr = radii[i] + radii[j]
        if np.linalg.norm(Δc) < Δr:
            v_norm = Δc / np.linalg.norm(Δc)
            v_i_norm = velocities[i] / np.linalg.norm(velocities[i])
            v_j_norm = velocities[j] / np.linalg.norm(velocities[j])
            velocities[i] = v_i_norm - (2 / (1 + 1)) * (v_i_norm @ v_norm) * v_norm
            velocities[j] = v_j_norm - (2 / (1 + 1)) * (v_j_norm @ v_norm) * v_norm
        
        event_queue = compute_collision_times(centers, radii, velocities, g_tot, eps)
    
    # Shrink radii to avoid violation and run overlap repair
    radii -= eps
    while True:
        D = distance_matrix(centers, centers)
        overlap_indices = np.where((D > 0) & (D < 2 * radii[np.newaxis, :] + np.minimum(radii[:, np.newaxis], 0)))
        if overlap_indices[0].size > 0:
            i, j = overlap_indices[0][0], overlap_indices[1][0]
            Δc = centers[i] - centers[j]
            Δr = radii[i] + radii[j]
            centers[i] += Δc * (Δr - np.linalg.norm(Δc)) / (2 * np.linalg.norm(Δc))
            radii[i] = Δr / 2 - np.linalg.norm(centers[i] - centers[j]) / 2
            centers[j] -= Δc * (Δr - np.linalg.norm(Δc)) / (2 * np.linalg.norm(Δc))
            radii[j] = Δr / 2 - np.linalg.norm(centers[i] - centers[j]) / 2
        else:
            break
    
    # LP polynomial
    A = np.hstack([np.ones(n)[:, np.newaxis], centers, np.eye(n)])
    b = [0] + [1 - 2 * r for r in radii]
    c = np.concatenate([[-1 for _ in range(n)], [0 for _ in range(2 * n)]])
    bounds = [(0, None)] * n + [(0, None)] * (2 * n)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        radii_optimized = result.x[:n]
        return centers, radii_optimized, np.sum(radii_optimized)
    else:
        return centers, np.zeros(n), np.sum(np.zeros(n))
