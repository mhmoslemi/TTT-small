import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonal lattice configuration
    s = np.sqrt(3) / 3 * 0.312
    hex_grid = np.array([
        (s + k_x * s, s + k_y * s * np.sqrt(3) / 2)
        for k_x in range(8)
        for k_y in range(int(k_x % 2 == 0) + 1, int(k_x % 2 == 0) + 5)
    ])
    centers = hex_grid[:26]
    
    if centers.shape[0] != 26:
        s -= 0.0001
        hex_grid = np.array([
            (s + k_x * s, s + k_y * s * np.sqrt(3) / 2)
            for k_x in range(8)
            for k_y in range(int(k_x % 2 == 0) + 1, int(k_x % 2 == 0) + 5)
        ])
        centers = hex_grid[:26]
    
    def evaluate_sum_radii(centers):
        n = centers.shape[0]
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dists[i, j] = dists[j, i] = np.linalg.norm(centers[i] - centers[j])
        
        A = np.zeros((n + n + n, n))
        b = np.zeros(n + n + n)
        
        for i in range(n):
            radius_var_idx = i
            A[radius_var_idx, i] = 1
            b[radius_var_idx] = 0.5
           A[n + radius_var_idx, :] = 2 * centers[i]
            b[n + radius_var_idx] = s
        
            for j in range(i + 1, n):
                rj_var_idx = j
                A[2 * n + radius_var_idx, rj_var_idx] = 2
                A[2 * n + rj_var_idx, i] = 2
                b[2 * n + radius_var_idx] = dists[i, j]
        
        try:
            result = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': False})
            if result.success:
                return np.sum(-result.x), centers
        except:
            return 0, centers
    
    # Simulated Annealing parameters
    T0 = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = 0.05
    
    n_iter = 0
    temperature = T0
    best_centers = centers
    best_sum_radii = 0
    current_centers = centers
    
    while temperature > T_min and n_iter < 3000:
        for move in range(M):
            i = np.random.randint(0, 26)
            delta_x, delta_y = np.random.normal(0, sigma_move, size=2)
            new_center = np.array([current_centers[i, 0] + delta_x, current_centers[i, 1] + delta_y])
            new_center[0] = np.clip(new_center[0], 0, 1)
            new_center[1] = np.clip(new_center[1], 0, 1)
            
            new_centers = current_centers.copy()
            new_centers[i, :] = new_center
            
            new_sum_radii, _ = evaluate_sum_radii(new_centers)
            if new_sum_radii > best_sum_radii or np.random.rand() < np.exp((new_sum_radii - best_sum_radii) / temperature):
                best_centers = new_centers.copy()
                best_sum_radii = new_sum_radii
                
        best_tmp, current_centers = evaluate_sum_radii(best_centers)
        
        if best_tmp > best_sum_radii:
            best_sum_radii = best_tmp
        
        temperature *= c
        n_iter += 1
    
    best_sum_radii, best_centers = evaluate_sum_radii(best_centers)
    return best_centers, np.zeros(26), best_sum_radii
