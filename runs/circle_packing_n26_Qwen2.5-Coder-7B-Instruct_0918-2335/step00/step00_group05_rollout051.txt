import numpy as np

def run_packing():
    np.random.seed(42)  # For reproducibility

    # Initialize parameters
    n = 26
    w_max, w_min = 0.9, 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.05
    T0, alpha = 0.05, 0.995
    N, T_pso = 80, 2000
    T_sa, sigma_center = 20000, 0.05
    epsilon = 1e-12

    # PSO setup
    positions = np.random.rand(n, 2)
    velocities = np.zeros((n, 2))
    radii = np.random.rand(n) * 0.2 + 0.05
    g_best = float('-inf')
    g_best_pos = positions

    # PSO loop
    for t in range(T_pso):
        for i in range(n):
            p_best = positions[i]
            p_value = np.sum(radii)
            velocities[i] = w_min + (w_max - w_min) * t / T_pso
            velocities[i] *= velocities[i] + c1 * np.random.rand() * (p_best - positions[i]) + c2 * np.random.rand() * (g_best_pos - positions[i])
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            positions[i] += velocities[i]

            center, rad = np.clip(positions[i], rad, 1 - rad), max(rad, epsilon)
            positions[i] = center
            radii[i] = rad
            projections(positions, radii)
            current_value = np.sum(radii)

            if current_value > g_best:
                g_best = current_value
                g_best_pos = positions.copy()

        if np.random.rand() < np.exp(-g_best/N / T0):
            positions = g_best_pos

        T0 *= alpha

    # SA setup
    best_positions, best_radii, best_value = g_best_pos, g_best, g_best
    T = T0
    for _ in range(T_sa):
        i = np.random.randint(n)
        perturbed_pos = positions[i] + np.clip(np.random.randn(2), -sigma_center, sigma_center)
        perturbed_rad = np.clip(radii[i] + np.clip(np.random.randn(), -sigma_center / 2, sigma_center / 2), epsilon, 0.5)
        perturbed_positions[i], perturbed_radii[i] = perturbed_pos, perturbed_rad
        projections(perturbed_positions, perturbed_radii)
        perturbed_value = np.sum(perturbed_radii)

        if perturbed_value > best_value or np.random.rand() < np.exp((best_value - perturbed_value) / T):
            best_positions, best_radii, best_value = perturbed_positions.copy(), perturbed_radii.copy(), perturbed_value

        T *= alpha

    # Post-processing
    for i in range(n):
        center, rad = best_positions[i], best_radii[i]
        projections(best_positions, best_radii)
        _, rad = best_positions[i], best_radii[i]
        best_positions[i] = center
        best_radii[i] = rad

    # Ensure minimal overlap
    for i in range(n):
        for j in range(i+1, n):
            if radii[i] > 0 and radii[j] > 0 and np.hypot(best_positions[i][0]-best_positions[j][0], best_positions[i][1]-best_positions[j][1]) < best_radii[i] + best_radii[j] - epsilon/10:
                best_positions[i] += (radii[i] + radii[j] - np.hypot(best_positions[i][0]-best_positions[j][0], best_positions[i][1]-best_positions[j][1])) / 2 * np.array([np.sin(best_positions[i][0]-best_positions[j][0]), np.cos(best_positions[i][0]-best_positions[j][0])])
                best_positions[j] -= (radii[i] + radii[j] - np.hypot(best_positions[i][0]-best_positions[j][0], best_positions[i][1]-best_positions[j][1])) / 2 * np.array([np.sin(best_positions[i][0]-best_positions[j][0]), np.cos(best_positions[i][0]-best_positions[j][0])])

    return best_positions, best_radii, np.sum(best_radii)

def projections(positions, radii):
    for i in range(len(positions)):
        x, y = positions[i]
        r = radii[i]
        positions[i] = np.clip(positions[i], r, 1 - r)
        for j in range(i+1, len(positions)):
            if r > 0 and radii[j] > 0:
                dist = np.sqrt((x-positions[j][0])**2 + (y-positions[j][1])**2)
                if dist < r + radii[j]:
                    positions[i] += (r + radii[j] - dist) / 2 * (positions[j] - positions[i]) / dist
                    positions[j] -= (r + radii[j] - dist) / 2 * (positions[j] - positions[i]) / dist
