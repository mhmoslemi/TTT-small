import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial parameters
    T0 = 0.12
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = 0.15
    num_points = 26
    
    # Generate initial hexagonal lattice
    s = 0.36
    xs = s + s * (np.arange(num_points) % 12) / 6
    ys = s * np.floor(np.arange(num_points) / 12) + s * ((np.arange(num_points) % 12) / 12) - 0.5 * s
    centers_2D = np.stack([xs, ys], axis=1)
    valid_indices = (xs < 1 - s) & (xs > s) & (ys < 1 - s) & (ys > s)
    centers = centers_2D[valid_indices[:num_points]]
    
    best_centers = centers.copy()
    best_sum = -np.inf
    
    # Simulated Annealing loop
    T = T0
    while T > T_min:
        for _ in range(M):
            # Randomly select a circle and propose a new center
            i = np.random.randint(num_points)
            perturbation = sigma_move * np.random.randn(2)
            centers[i, :] += perturbation
            centers[i, :] = np.clip(centers[i, :], 0, 1 - s)
            
            # Compute pairwise distances
            distances = squareform(pdist(centers))
            
            # Define the linear program
            c = -np.ones(num_points)  # Our objective is to maximize the sum of radii
            A = np.zeros((3 * num_points, num_points))
            b = np.zeros(3 * num_points)
            
            # Constraints
            # Non-overlap constraints
            for i in range(num_points):
                for j in range(i + 1, num_points):
                    A[i * 3 + 0, i] = 1
                    A[i * 3 + 0, j] = 1
                    A[i * 3 + 1, i] = -1
                    A[i * 3 + 1, j] = -1
                    A[i * 3 + 1, j] += 1 / 1e6
                    b[i * 3 + 1] = distances[i, j]
            
            # Boundary constraints
            for i in range(num_points):
                A[i * 3 + 2, i] = 1
                b[i * 3 + 2] = centers[i, 0] - s
            for i in range(num_points):
                A[i * 3 + 3, i] = 1
                b[i * 3 + 3] = 1 - centers[i, 0] - s
            for i in range(num_points):
                A[i * 3 + 4, i] = 1
                b[i * 3 + 4] = centers[i, 1] - s
            for i in range(num_points):
                A[i * 3 + 5, i] = 1
                b[i * 3 + 5] = 1 - centers[i, 1] - s
            
            # Non-negativity constraints
            A[6 * num_points:, np.arange(num_points)] = 1
            b[6 * num_points:] = 0
            
            # Solve the linear program
            try:
                result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'warm_start': True})
            except Exception as e:
                continue
            
            if result.success and result.fun < -1e-6:
                new_centers = centers.copy()
                new_centers[i, :] = result.x
                # Recompute pairwise distances for the new layout
                distances = squareform(pdist(new_centers))
                sum_radii = -result.fun
                # Accept move if current sum is higher or with probability
                if sum_radii > best_sum or np.log(np.random.rand()) < (best_sum - sum_radii) / T:
                    best_centers = new_centers.copy()
                    best_sum = sum_radii
        
        T *= c
    
    # Final validation
    valid, msg = validate_packing(best_centers, -best_sum)
    if not valid:
        print(f"Invalid packing: {msg}")
    
    return best_centers, -best_sum, -best_sum

# Call the function and print the result
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of Radii:", sum_radii)
