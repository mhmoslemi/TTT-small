def valid(p, obstacles):
    for o in obstacles:
        if np.linalg.norm(np.array(p) - np.array(o[:2])) < p[2] + o[2] + 1e-12:
            return False
    return True

def pack_circles() -> tuple[np.ndarray, np.ndarray, float]:
    centers = []
    radii = []
    obstacles = [{'type': 'box', 'x': 0.25, 'y': 0.25, 'f': 1}, {'type': 'box', 'x': 0.75, 'y': 0.25, 'f': 1}, {'type': 'box', 'x': 0.75, 'y': 0.75, 'f': 1}, {'type': 'box', 'x': 0.25, 'y': 0.75, 'f': 1}]

    while len(centers) < 26:
        vertices = get_candidate_vertices(obstacles)
        if not vertices:
            return pack_circles()  # Fallback to equal-radius packing
        p = max(vertices, key=lambda p: p[2])
        centers.append(p[:2])
        radii.append(p[2])
        obstacles.append((p[0], p[1], p[2], 1))

    bounds = [(0, 1), (0, 1), (0, 0.5)]
    A = []
    b = []
    for i in range(len(centers)):
        for j in range(i):
            A.append([-1, -1, 0])
            b.append(centers[i][0] + centers[i][1] - centers[j][0] - centers[j][1])
        A.append([-1, 0, 0])
        b.append(1 - centers[i][0])
        A.append([1, 0, 0])
        b.append(centers[i][0])
        A.append([0, -1, 0])
        b.append(1 - centers[i][1])
        A.append([0, 1, 0])
        b.append(centers[i][1])
        A.append([0, 0, -1])
        b.append(0)
        A.append([0, 0, 1])
        b.append(0.5)

    result = linprog([-1]*len(centers), A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if not result.success:
        return pack_circles()  # Fallback to equal-radius packing

    for i in range(len(centers)):
        centers[i] = (result.x[i*3], result.x[i*3+1])
        radii[i] = result.x[i*3+2]

    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = float(radii.sum())

    return centers, radii, sum_radii
