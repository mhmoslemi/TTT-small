from scipy.optimize import minimize

def hex_lattice(num_circles):
    s = np.sqrt(0.8) / np.sqrt(26)  # Estimate lattice spacing
    n = 0
    points = []
    while n < num_circles:
        R = np.sqrt(3) * s / 2
        x = (np.arange(n, n + int(2 * R / s + 1)) * s + s / 2) % 1
        y = ((np.arange(n, n + int(2 * R / s + 1)) * np.sqrt(3) * s / 2) % 1)
        x = x[(y <= 1) & (x <= 1)]
        y = y[(y <= 1) & (x <= 1)]
        points.extend(zip(x, y))
        n += len(points)
    points = points[:num_circles]
    return np.array(points)

def boundary_adjustment(centers):
    for i in range(centers.shape[0]):
        x, y = centers[i]
        min_clearance = min(x, y, 1 - x, 1 - y)
        centers[i, :] = centers[i, :] + min(0, -0.5 * (min_clearance - 0.5))
    return centers
