import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def solve_lp(c):
        n = len(c)
        A = np.zeros((2 * n, 2 * n))
        b = np.zeros(2 * n)
        c = np.zeros(n)
        
        for i in range(n):
            rmax = min(c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1])
            c[i] = -1
            A[2 * i, i] = 1
            b[2 * i] = rmax
            A[2 * i + 1, i] = -1
            b[2 * i + 1] = 0
            C = c
            for j in range(n):
                if j != i:
                    d_ij = np.sqrt((c[i, 0] - c[j, 0]) ** 2 + (c[i, 1] - c[j, 1]) ** 2)
                    A[2 * n + j, i] = 1
                    A[2 * n + j, j] = -1
                    C[n + i] = 1 - d_ij
        
        result = linprog(C, A_ub=A, b_ub=b, method='highs')
        return -result.x

    def compute_forces(r, c):
        F = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.sqrt((c[i, 0] - c[j, 0]) ** 2 + (c[i, 1] - c[j, 1]) ** 2)
                s_ij = r[i] + r[j] - d_ij
                if s_ij > 0:
                    F[i] += (c[i] - c[j]) / d_ij * s_ij
                    F[j] -= (c[i] - c[j]) / d_ij * s_ij
            for k in [0, 1, 2, 3]:
                g = r[i] - c[i, k] if (k % 2 == 0) else r[i] - (1 - c[i, k])
                if g > 0:
                    F[i, k] += g * (-1 if (k % 2 == 0) else 1)
        
        return F

    def project(c):
        return np.clip(c, 0, 1)

    np.random.seed(42)
    MAX_ITER = 500
    τ0 = 1e-3
    γ = 0.95
    ε = 0.02
    
    # Step A: Initial grid
    x, y = np.meshgrid(np.linspace(0, 1, 6), np.linspace(0, 1, 6))
    centers = np.vstack((x.flatten(), y.flatten())).T
    
    # Step B: Reduce to 26
    distances = distance_matrix(centers, centers)
    keep_indices = np.argsort(distances.ravel())[::-1][25:51]
    centers = centers[keep_indices]

    # Step C: Random perturbation
    centers += np.random.normal(scale=0.01, size=centers.shape)
    centers = project(centers)

    for k in range(MAX_ITER):
        r = solve_lp(centers)
        if np.allclose(r, solve_lp(centers), atol=1e-8):
            break
        F = compute_forces(r, centers)
        τ = τ0 * γ ** (k - 1)
        centers += τ * F
        centers = project(centers)
        if k % 50 == 0:
            centers += ε * np.random.normal(size=centers.shape)
            centers = project(centers)

    r = solve_lp(centers)
    sum_radii = np.sum(r)
    return centers, r, sum_radii

centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii:.9f}")
