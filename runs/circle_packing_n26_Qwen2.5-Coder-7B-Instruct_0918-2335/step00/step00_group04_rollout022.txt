import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalty(z):
    n = 26
    r = z[52:]
    obj = -np.sum(r)
    overlap = 0
    for i in range(n):
        if z[2*i] - r[i] < 0:
            overlap += (z[2*i] - r[i]) ** 2
        if z[2*i] + r[i] > 1:
            overlap += (z[2*i] + r[i] - 1) ** 2
        if z[2*i+1] - r[i] < 0:
            overlap += (z[2*i+1] - r[i]) ** 2
        if z[2*i+1] + r[i] > 1:
            overlap += (z[2*i+1] + r[i] - 1) ** 2
        for j in range(i + 1, n):
            dist = np.sqrt((z[2*i] - z[2*j])**2 + (z[2*i+1] - z[2*j+1])**2)
            overlap += max(0, (r[i] + r[j] - dist) ** 2)
    return obj - 10000 * overlap

def local_constraints(z):
    n = 26
    r = z[52:]
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2*i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[2*i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2*i+1] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[2*i+1] - r[i]})
        for j in range(i + 1, n):
            def fun(z, i=i, j=j):
                dist = np.sqrt((z[2*i] - z[2*j])**2 + (z[2*i+1] - z[2*j+1])**2)
                return dist - (r[i] + r[j])
            cons.append({'type': 'ineq', 'fun': fun})
    return cons

def local_opt refinement(z):
    bounds = [(0,1)] * 52 + [(0,0.5)] * 26
    cons = local_constraints(z)
    res = minimize(penalty, z, method='SLSQP', jac=None, constraints=cons, bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    bounds = [(0,1)] * 52 + [(0,0.5)] * 26
    res = differential_evolution(penalty, bounds, popsize=20, strategy='rand/1/bin', mutation=0.6, recombination=0.7, maxiter=2000, tol=1e-8)
    if res.fun < -1e-9 * 26 * 0.5 * 26:
        z_best = local_optimization(res.x)
    else:
        z_best = res.x
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [z_best[2*i], z_best[2*i+1]]
        radii[i] = z_best[52+i]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
