import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    R = 200
    delta_c = 1e-3
    N_refine = 200
    epsilon = 1e-9

    grid_x, grid_y = np.meshgrid(np.linspace(0, 1, R), np.linspace(0, 1, R))
    grid_points = np.array([grid_x.flatten(), grid_y.flatten()]).T
    d_boundary = np.min(np.vstack((grid_x.flatten(), 1-grid_x.flatten(), grid_y.flatten(), 1-grid_y.flatten())), axis=0)
    r_candidate = d_boundary.copy()
    
    centers = np.zeros((N, 2))
    radii = np.zeros(N)

    for i in range(N):
        k = np.argmax(r_candidate)
        centers[i] = grid_points[k]
        radii[i] = r_candidate[k] - epsilon
        dx = grid_x.flatten() - centers[i, 0]
        dy = grid_y.flatten() - centers[i, 1]
        dist = np.sqrt(dx**2 + dy**2) - radii[i]
        r_candidate = np.minimum(r_candidate, dist)
        r_candidate[k] = -1e9

    r_candidate[r_candidate < 0] = 0
    radii += epsilon

    for _ in range(N_refine):
        for j in range(N):
            i = np.random.randint(0, N)
            c_new = centers[i] + np.random.uniform(-delta_c, delta_c, 2)
            dx = grid_x.flatten() - c_new[0]
            dy = grid_y.flatten() - c_new[1]
            dist = np.sqrt(dx**2 + dy**2) - radii
            r_new = np.minimum(np.minimum(d_boundary - radii[i] - epsilon, np.min(dist[:, j])), 0)
            if r_new > radii[i]:
                centers[i] = c_new
                radii[i] = r_new

    for i in range(N):
        dx = grid_x.flatten() - centers[i, 0]
        dy = grid_y.flatten() - centers[i, 1]
        dist = np.sqrt(dx**2 + dy**2)
        radii[i] = np.minimum(np.minimum(d_boundary - np.min(dist[:, i]), np.min(dist[:, i] - radii)), 0)

    return centers, radii, np.sum(radii)
