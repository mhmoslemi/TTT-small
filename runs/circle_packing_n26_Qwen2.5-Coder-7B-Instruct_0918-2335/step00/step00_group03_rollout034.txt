import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    N = 26
    r_min = 1e-3
    iterations = 10000

    # RSA seed
    centers = np.zeros((N, 2))
    radii = np.full(N, r_min)
    for i in range(N):
        placed = False
        for _ in range(iterations):
            x, y = np.random.rand(2)
            overlaps = False
            for j in range(i):
                if np.sqrt((x - centers[j, 0])**2 + (y - centers[j, 1])**2) < radii[j] + r_min:
                    overlaps = True
                    break
            if not overlaps:
                centers[i, 0], centers[i, 1] = x, y
                placed = True
                break
        if not placed:
            return run_packing()
    
    # Physics-based local relaxation
    def force(i, j):
        d_ij = np.sqrt(np.sum((centers[i] - centers[j])**2))
        gap = d_ij - (radii[i] + radii[j])
        if gap < 0:
            return gap / d_ij * (radii[j] - radii[i])
        return 0

    def boundary_force(x, i):
        if x < radii[i]:
            return radii[i] - x
        elif x > 1 - radii[i]:
            return x - (1 - radii[i])
        return 0

    def energy(reps):
        x, y = reps[:52].reshape((-1, 2)), reps[52:]
        n = 26
        e = 0
        for i in range(n):
            for j in range(i + 1, n):
                e += force(i, j)
            e += boundary_force(x[i, 0], i)
            e += boundary_force(x[i, 1], i)
        return e

    def grad(reps):
        n = 26
        x, y = reps[:52].reshape((-1, 2)), reps[52:]
        grad_e = np.zeros_like(reps)
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.sqrt(np.sum((x[i] - x[j])**2))
                g = force(i, j)
                if g != 0:
                    grad_e[i * 2] += g * (x[j, 0] - x[i, 0]) / d_ij
                    grad_e[i * 2 + 1] += g * (x[j, 1] - x[i, 1]) / d_ij
                    grad_e[j * 2] -= g * (x[j, 0] - x[i, 0]) / d_ij
                    grad_e[j * 2 + 1] -= g * (x[j, 1] - x[i, 1]) / d_ij
            grad_e[i * 2] += boundary_force(x[i, 0], i) if x[i, 0] > 0 else -boundary_force(x[i, 0], i)
            grad_e[i * 2 + 1] += boundary_force(x[i, 1], i) if x[i, 1] > 0 else -boundary_force(x[i, 1], i)
        return grad_e

    res = minimize(energy, np.concatenate((centers.flatten(), radii)), jac=grad, method='L-BFGS-B', bounds=[(0, 1)] * 52 + [(0, 0.5)] * N)
