import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    R = 200
    delta_c = 1e-3
    n_refine = 200
    eps = 1e-9

    # Generate grid and initial radii
    grid_x, grid_y = np.mgrid[:1:R*1j, :1:R*1j]
    dx, dy = grid_x - 0.5, grid_y - 0.5
    dist_to_bound = np.minimum(dx, 1-dx, dy, 1-dy)
    grid_points = np.column_stack((grid_x.ravel(), grid_y.ravel()))
    radii = dist_to_bound.ravel()

    # Function to update radii after placing a new circle
    def update_radii(new_center, radius):
        new_dx = grid_points[:, 0] - new_center[0]
        new_dy = grid_points[:, 1] - new_center[1]
        new_dist = np.hypot(new_dx, new_dy) - radius
        nonlocal radii
        radii = np.minimum(radii, new_dist)

    centers = np.empty((N, 2))
    for i in range(N):
        idx = np.argmax(radii)
        center = grid_points[idx]
        radius = min(radii[idx], dist_to_bound[idx]) - eps
        centers[i] = center
        update_radii(center, radius)
        radii[idx] = -1  # Mark as used

    # Local refinement
    for _ in range(n_refine):
        i = np.random.randint(N)
        old_center, old_radius = centers[i], radii[i]
        new_center = old_center + (np.random.rand(2) - 0.5) * 2 * delta_c
        new_radius = min(min(np.hypot(new_center[0] - centers[:, 0], new_center[1] - centers[:, 1]) - radii),
                           min(np.hypot(new_center[0], 0.5) - new_radius, np.hypot(new_center[0], 1.5) - new_radius,
                               np.hypot(new_center[1], 0.5) - new_radius, np.hypot(new_center[1], 1.5) - new_radius)) - eps
        if new_radius > old_radius:
            centers[i] = new_center
            radii[i] = new_radius

    # Final radius update after refinement
    radii = np.minimum(dist_to_bound, 
                       np.minimum(np.hypot(centers[:, None, 0] - centers[None, :, 0], 
                                          centers[:, None, 1] - centers[None, :, 1]) - radii[:, None],
                                  np.hypot(centers[:, 0], 0.5) - radii, 
                                  np.hypot(centers[:, 0], 1.5) - radii, 
                                  np.hypot(centers[:, 1], 0.5) - radii,
                                  np.hypot(centers[:, 1], 1.5) - radii)) - eps
    radii = np.maximum(radii, 0)

    return centers, radii, np.sum(radii)
