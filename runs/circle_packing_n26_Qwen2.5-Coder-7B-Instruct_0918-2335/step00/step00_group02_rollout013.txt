import numpy as np
import scipy.spatial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * (0.1 / 26)  # Scaling to be small enough for unit square
    
    for _ in range(1000):  # Number of iterations
        tree = scipy.spatial.KDTree(centers)
        for i in range(26):
            dx = 0.01 * np.random.randn()
            dy = 0.01 * np.random.randn()
            new_center = centers[i] + np.array([dx, dy])
            new_center = np.clip(new_center, 0, 1)
            if all(tree.query([new_center], k=2)[0][1] > 2 * radii[i]):
                centers[i] = new_center
                
        radii = np.maximum(0, np.random.rand(26) * (0.1 / 26))  # Re-assuring non-negative radii
        radii = radii / np.sum(radii) * 0.5  # Scaling to sum to a reasonable value
    
    return centers, radii, np.sum(radii)
