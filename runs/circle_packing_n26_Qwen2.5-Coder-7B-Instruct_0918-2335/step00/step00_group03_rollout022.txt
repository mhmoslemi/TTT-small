import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def rsa_seed(N, r_min, iterations):
        circles = []
        while len(circles) < N:
            for _ in range(iterations):
                x, y, r = np.random.rand(3)
                if r >= r_min:
                    if all(np.sum((x - cx) ** 2 + (y - cy) ** 2) >= (r + cr) ** 2 for cx, cy, cr in circles):
                        circles.append([x, y, r])
                        break
        return np.array(circles)

    def physics_relaxation(circles, T, dt):
        N = circles.shape[0]
        x, y, r = circles[:, 0], circles[:, 1], circles[:, 2]
        for _ in range(T):
            dx, dy = np.zeros(N), np.zeros(N)
            for i in range(N):
                for j in range(i + 1, N):
                    dx[i] += (r[j] - r[i]) * (x[j] - x[i]) / ((x[j] - x[i]) ** 2 + (y[j] - y[i]) ** 2)
                    dy[i] += (r[j] - r[i]) * (y[j] - y[i]) / ((x[j] - x[i]) ** 2 + (y[j] - y[i]) ** 2)
                    dx[j] += (r[i] - r[j]) * (x[i] - x[j]) / ((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
                    dy[j] += (r[i] - r[j]) * (y[i] - y[j]) / ((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
                dx[i] += 2 * min(1 - x[i], x[i], y[i], 1 - y[i]) * r[i]
                dy[i] += 2 * min(1 - y[i], y[i], 1 - x[i], x[i]) * r[i]
            x += dx * dt
            y += dy * dt
            r += np.abs(dx) * dt
            r = np.clip(r, 0, 0.5)
        return np.column_stack((x, y, r))

    def expand_globally(circles):
        N = circles.shape[0]
        x, y, r = circles[:, 0], circles[:, 1], circles[:, 2]
        s_max = min(min((x[i] - x[j]) / (r[i] + r[j]) for j in range(i + 1, N)), min((1 - x[i]) / r[i] for i in range(N)), min((y[i] - y[j]) / (r[i] + r[j]) for j in range(i + 1, N)), min((1 - y[i]) / r[i] for i in range(N)))
        x = (1 - s_max) * x - 0.5 * s_max + 0.5
        y = (1 - s_max) * y - 0.5 * s_max + 0.5
        r *= s_max
        return np.column_stack((x, y, r))

    def penalty(x):
        circles = x.reshape(26, 3)
        N = circles.shape[0]
        overlap = 0
        boundary = 0
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((circles[i, :2] - circles[j, :2]) ** 2))
                overlap += max(0, (circles[i, 2] + circles[j, 2]) - dist) ** 2
            overlap += max(0, circles[i, 2] - circles[i, 0]) ** 2
            overlap += max(0, circles[i, 2] - (1 - circles[i, 0])) ** 2
            overlap += max(0, circles[i, 2] - circles[i, 1]) ** 2
            overlap += max(0, circles[i, 2] - (1 - circles[i, 1])) ** 2
        return overlap

    def objective(x):
        circles = x.reshape(26, 3)
        return -np.sum(circles[:, 2])

    circles = rsa_seed(26, 1e-3, 10000)
    circles = physics_relaxation(circles, 2000, 1e-3)
    circles = expand_globally(circles)
    x0 = circles.flatten()
    result = scipy.optimize.minimize(objective, x0, method='L-BFGS-B', jac=lambda x: np.array([1 if i % 3 == 0 else 2 * min(circles[i // 3, i % 3], 0.5 - circles[i // 3, i % 3]) * circles[i // 3, 2] if i % 3 < 2 else 0 for i in range(len(x0))]), bounds=np.hstack((np.tile([[0, 1]], (26, 2)), np.tile([[0, 0.5]], (26, 1)))), options={'maxiter': 2000, 'ftol': 1e-10})
    circles = result.x.reshape(26, 3)
    return circles[:, :2], circles[:, 2], circles[:, 2].sum()
