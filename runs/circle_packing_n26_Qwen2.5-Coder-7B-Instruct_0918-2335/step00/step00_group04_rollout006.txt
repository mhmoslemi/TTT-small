import numpy as np
from scipy.optimize import differential_evolution, NonlinearConstraint, minimize

def penalised_objective(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    obj = -np.sum(r)
    penalty = 0
    for i in range(n):
        if x[i] - r[i] < 0:
            penalty += (x[i] - r[i]) ** 2
        if x[i] + r[i] > 1:
            penalty += (x[i] + r[i] - 1) ** 2
        if y[i] - r[i] < 0:
            penalty += (y[i] - r[i]) ** 2
        if y[i] + r[i] > 1:
            penalty += (y[i] + r[i] - 1) ** 2
        for j in range(i+1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            penalty += max((r[i] + r[j] - dist)**2, 0)
    return obj + 10000 * penalty

def local_constraints():
    n = 26
    def fun(z):
        x = z[:n]
        y = z[n:2*n]
        r = z[2*n:]
        cnstr = []
        for i in range(n):
            cnstr.append(x[i] - r[i])
            cnstr.append(1 - x[i] - r[i])
            cnstr.append(y[i] - r[i])
            cnstr.append(1 - y[i] - r[i])
            for j in range(i+1, n):
                dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                cnstr.append(dist - (r[i] + r[j]))
        return np.array(cnstr)

    jac = None
    hess_inv = None
    return NonlinearConstraint(local_constraints, 0, np.inf, fun=fun, jac=jac, hess_inv=hess_inv)

def local_refinement(z):
    bounds = [(0, 1)] * 78
    def local_obj(z):
        r = z[52:]
        return -np.sum(r)
    nlc = local_constraints()
    res = minimize(local_obj, z, method='SLSQP', bounds=bounds, constraints=[nlc], options={'maxiter': 2000, 'ftol': 1e-12})
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    n = 26
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    popsize = 20
    result = differential_evolution(penalised_objective, bounds, popsize=popsize, maxiter=2000, tol=1e-8)
    z_best = result.x
    penalty = penalised_objective(z_best) - (-np.sum(z_best[52:]))
    if penalty > 1e-9:
        z_opt = local_refinement(z_best)
    else:
        z_opt = z_best
    x = z_opt[:26]
    y = z_opt[26:52]
    r = z_opt[52:]
    return x.reshape(-1, 1), y.reshape(-1, 1), np.sum(r)

centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}\nRadii:\n{radii}\nSum of radii: {sum_radii}")
