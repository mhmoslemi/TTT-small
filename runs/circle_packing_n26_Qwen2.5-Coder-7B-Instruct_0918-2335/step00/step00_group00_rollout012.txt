import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    grid = np.array([(x, y) for y in np.linspace(0, 1, R+1)[1:-1] for x in np.linspace(0, 1, R+1)[1:-1]])
    r_f = np.min(np.column_stack((grid[:, 0], 1 - grid[:, 0])), axis=1)
    r_f = np.min(np.column_stack((grid[:, 1], 1 - grid[:, 1])), axis=1)
    r_f = np.minimum(r_f[:, np.newaxis], r_f[np.newaxis, :])
    centers = np.empty((0, 2))
    radii = np.empty(0)
    while len(centers) < 26:
        k = np.argmax(r_f)
        new_center = grid[k]
        new_radius = r_f[k] - 1e-9
        centers = np.append(centers, [new_center], axis=0)
        radii = np.append(radii, new_radius)
        distances = np.linalg.norm(centers - new_center, axis=1) - new_radius
        r_f = np.min(np.column_stack((distances[:, np.newaxis], r_f[np.newaxis, :])), axis=0)
        r_f[k] = -1e9
    for _ in range(200):
        i = np.random.randint(0, 26)
        delta = (np.random.rand(2) - 0.5) * 1e-3
        new_center = centers[i] + delta
        distances = np.linalg.norm(centers - new_center, axis=1) - radii
        new_radius = np.min(np.column_stack((np.column_stack((np.minimum(new_center[:, 0], 1 - new_center[:, 0]), 1 - np.minimum(new_center[:, 0], 1 - new_center[:, 0])))[:, np.newaxis], distances[:, np.newaxis])), axis=0)
        if new_radius > radii[i]:
            centers[i] = new_center
            radii[i] = new_radius
    radii = np.minimum(np.minimum(np.column_stack((np.minimum(centers[:, 0], 1 - centers[:, 0]), 1 - np.minimum(centers[:, 0], 1 - centers[:, 0])))[:, np.newaxis], np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=-1) - radii[np.newaxis, :]), np.column_stack((np.minimum(centers[:, 1], 1 - centers[:, 1]), 1 - np.minimum(centers[:, 1], 1 - centers[:, 1])))[:, np.newaxis])[:, 0]
    return centers, radii, np.sum(radii)
