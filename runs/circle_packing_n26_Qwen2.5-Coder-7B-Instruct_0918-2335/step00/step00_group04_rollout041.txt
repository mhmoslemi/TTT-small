import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import cdist
from scipy.optimize import linprog
from scipy.spatial import Delaunay

def get_radius(center, obstacles):
    x, y = center
    r = min(
        1 - x,
        x,
        1 - y,
        y,
        min(
            cdist([center], [o][0])[0] - r
            for o in obstacles
            if np.linalg.norm(center - o[0]) < np.linalg.norm(center - o[1])
        ),
    )
    return r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial central positions and assigned radii
    corner_positions = np.array([[0.25, 0.25], [0.75, 0.25], [0.75, 0.75], [0.25, 0.75]])
    radii = np.zeros(4)
    centers = corner_positions
    obstacles = ((np.array([[0, 0], [0, 1]]), 0, 0), (np.array([[0, 0], [1, 0]]), 1, 0), (np.array([[1, 0], [1, 1]]), 1, 1), (np.array([[0, 1], [1, 1]]), 0, 1))

    while len(centers) < 26:
        rho_values = []
        candidates = []

        for c in voronoi(centers + obstacles).vertex_to_region:
            if isinstance(c, list):
                vertices = centers[c]
                if len(vertices) == 2 or len(vertices) == 3:
                    vertex = np.mean(vertices, axis=0)
                    rho_value = get_radius(vertex, centers)
                    if rho_value > 0:
                        rho_values.append(rho_value)
                        candidates.append(vertex)
                    elif len(c) == 3:
                        obstacle = vertices
                        if np.linalg.norm(obstacle[0] - obstacle[1]) < np.linalg.norm(obstacle[0] - obstacle[2]):
                            radius = np.linalg.norm(obstacle[0] - obstacle[2]) / 4
                            if radius > 0:
                                rho_values.append(radius)
                                candidates.append(obstacle[0])
                            else:
                                rho_values.append(0)
                                candidates.append(obstacle[0])
                        else:
                            radius = np.linalg.norm(obstacle[0] - obstacle[1]) / 4
                            if radius > 0:
                                rho_values.append(radius)
                                candidates.append(obstacle[1])
                            else:
                                rho_values.append(0)
                                candidates.append(obstacle[1])
        if rho_values:
            # Choose the candidate with the maximum rho value
            chosen_index = np.argmax(rho_values)
            centers = np.vstack((centers, candidates[chosen_index]))
            radii = np.append(radii, rho_values[chosen_index])

        # Update obstacles with the new circle
        obstacles = tuple(
            (obstacle[:2], obstacle[2] - 1, obstacle[2] - 1)
            for obstacle in obstacles
        ) + ((centers[-1], 0, len(centers) - 2),)

    # Run linear programming to optimize radii
    A = np.array([np.concatenate(([1] + np.zeros(len(centers) - 1), [1]))
                 for i in range(len(centers) - 1)])
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            A[i, j] = -1
            A[i, j + 1] = 1      
    A[:, :len(centers)] = cdist(centers, centers, 'euclidean')
    A[:, len(centers):2*len(centers)] = (centers[:, 0] <= 1) * 2 - 1
    A[:, 2*len(centers):3*len(centers)] = (centers[:, 1] <= 1) * 2 - 1
    A[:, 3*len(centers):4*len(centers)] = (centers[:, 0] >= 0) * 2 - 1
    A[:, 4*len(centers):5*len(centers)] = (centers[:, 1] >= 0) * 2 - 1
    b = np.concatenate(([2] * (len(centers) - 1), np.ones(len(centers))))
    
    res = linprog(b, A_ub=A, bounds=(0, 0.5), method='highs')
    if not res.success:
        raise Exception('Linprog failed')
    radii = np.maximum(radii, res.x[:len(centers)])
