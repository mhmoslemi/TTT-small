import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial placement using a hexagonal lattice pattern
    radius = np.sqrt(2) / 3
    num_columns = int(np.ceil(np.sqrt(3 * n / (2 * np.pi))))
    num_rows = int(np.ceil(n / num_columns + 1))
    
    for i in range(num_rows):
        for j in range(num_columns):
            if (i + j) % 2 == 0:
                x = (i * np.sqrt(3) + j) / num_columns
                y = i / (num_rows - 1)
                index = i * num_columns + j
                if index < n:
                    centers[index, 0] = x
                    centers[index, 1] = y
                    radii[index] = radius
    
    # Ensure all circles fit within the unit square
    for i in range(n):
        while centers[i, 0] - radii[i] < 0:
            centers[i, 0] += 2 * radii[i]
        while centers[i, 0] + radii[i] > 1:
            centers[i, 0] -= 2 * radii[i]
        while centers[i, 1] - radii[i] < 0:
            centers[i, 1] += 2 * radii[i]
        while centers[i, 1] + radii[i] > 1:
            centers[i, 1] -= 2 * radii[i]
    
    # Calculate the sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
