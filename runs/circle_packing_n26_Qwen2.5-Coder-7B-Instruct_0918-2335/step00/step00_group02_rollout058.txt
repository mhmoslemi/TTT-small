import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    sum_radii = 0.0

    for i in range(n):
        x = np.random.rand()
        y = np.random.rand()
        r = np.random.rand() * 0.1
        centers[i] = [x, y]
        radii[i] = r
        sum_radii += r

    return centers, radii, sum_radii
