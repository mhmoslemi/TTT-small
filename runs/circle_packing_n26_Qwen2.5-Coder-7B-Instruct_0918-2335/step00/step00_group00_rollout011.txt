import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    grid_size = 200
    delta = 1e-3
    n_iter = 200
    epsilon = 1e-9
    n_circles = 26

    dx = np.linspace(0, 1, grid_size, endpoint=False)
    dy = np.linspace(0, 1, grid_size, endpoint=False)
    x_grid, y_grid = np.meshgrid(dx, dy, indexing='ij')
    grid_points = np.column_stack((x_grid.ravel(), y_grid.ravel()))

    d_boundary = np.minimum(np.minimum(x_grid, 1 - x_grid).ravel(), np.minimum(y_grid, 1 - y_grid).ravel())
    r_candidate = np.minimum(d_boundary, 0.5 - epsilon)
    centers = []
    radii = []
    for _ in range(n_circles):
        idx = np.argmax(r_candidate)
        r = r_candidate[idx]
        center = grid_points[idx]
        centers.append(center)
        radii.append(r)
        r_candidate -= 0.5 - r[:, np.newaxis]

    r_candidate[r_candidate < 0] = np.nan

    def update_grid(new_center, new_radius):
        dx = x_grid - new_center[0]
        dy = y_grid - new_center[1]
        dist = np.sqrt(dx**2 + dy**2) - new_radius
        r_candidate = np.minimum(r_candidate, dist)

    for _ in range(n_iter):
        idx = np.argmax(r_candidate)
        center, radius = centers[idx], radii[idx]
        proposal = center + (np.random.uniform(-delta, delta, 2))
        if (0 <= proposal[0] < 1) and (0 <= proposal[1] < 1):
            new_radius = np.min((np.min([1 - proposal[0], proposal[0], 1 - proposal[1], proposal[1]]), 
                                 np.min([np.sqrt((proposal - c[0])**2 + (proposal - c[1])**2) - r for c, r in zip(centers, radii) if not np.array_equal(c, proposal)])))
            if new_radius > radius:
                update_grid(proposal, new_radius)
                centers[idx] = proposal
                radii[idx] = new_radius
                break
            else:
                continue

    # Post-processing: Recompute exact radii
    for i, (center, rad) in enumerate(zip(centers, radii)):
        min_dist = np.min([np.sqrt((center - c)**2).sum() - r for c, r in zip(centers, radii) if not np.array_equal(c, center)])
        radii[i] = min(min_dist, 1 - min(center))
        radii[i] = max(0, radii[i])

    return np.array(centers).reshape((26, 2)), np.array(radii), sum(radii)
