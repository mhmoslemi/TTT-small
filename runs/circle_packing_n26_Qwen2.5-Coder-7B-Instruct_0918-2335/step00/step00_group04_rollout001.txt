import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    n = 26
    bounds = [(0, 1)] * n + [(0, 0.5)] * n

    def obj(v):
        r = v[n:]
        return -np.sum(r)

    def constraints(v):
        pos = v[:n].reshape(-1, 2)
        r = v[n:]
        cons = np.empty(104, float)
        cons[:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    best_gap = float('inf')
    best_res = None

    for _ in range(15):
        init = np.random.rand(2 * n)
        init[n:] = 0.02
        res = minimize(obj, init, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success and constraints(res.x) >= -1e-10:
            gap = -res.fun - np.sum(res.x[n:])
            if gap < best_gap:
                best_gap = gap
                best_res = res

    if best_res is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    return best_res.x[:n].reshape(-1, 2), best_res.x[n:], -best_res.fun - np.sum(best_res.x[n:])

# validate the packing
centers, radii, sum_radii = run_packing()
assert validate_packing(centers, radii) == (True, "ok"), "Packing is not valid"
print(sum_radii)
