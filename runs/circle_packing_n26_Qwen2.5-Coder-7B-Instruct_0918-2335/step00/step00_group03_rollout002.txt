import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.empty((26, 2))
    centers[0] = [r_c, r_c]
    centers[1] = [1 - r_c, r_c]
    centers[2] = [r_c, 1 - r_c]
    centers[3] = [1 - r_c, 1 - r_c]
    
    x_mid = 0.5
    y_mid = 0.5
    edge_radius = r_e
    edge_positions = np.array([
        [x_mid, edge_radius],
        [1 - edge_radius, x_mid],
        [x_mid, 1 - edge_radius],
        [1 - edge_radius, y_mid]
    ])
    centers[4:8] = edge_positions
    edge_radius = r_e
    
    # Inner hexagonal grid
    # Calculate the separation 'd' between circles in the hexagonal grid
    d = 2 * np.sqrt(3) * r_i
    
    # All cells that can fit within the remaining space
    grid_y = np.arange(r_c + r_i, 1 - r_c, d)
    grid_x = np.arange(r_c + r_i, 1 - r_c, d)
    
    # There will be 18-20 circles in the inner core, arranged in a hexagonal pattern.
    k = 0
    skip_y = False  # flag to skip every other row which rounds down its number of items
    for y in grid_y:
        if skip_y:
            k += 1
            skip_y = False
        else:
            skip_y = True
        for x in grid_x[k:]:
            if (x - (r_c + r_i))**2 + (y - (r_c + r_i))**2 < d**2 and \
               (x - (1 - r_c - r_i))**2 + (y - (r_c + r_i))**2 < d**2 and \
               (x - (1 - r_c - r_i))**2 + (y - (1 - r_c - r_i))**2 < d**2 and \
               (x - (r_c + r_i))**2 + (y - (1 - r_c - r_i))**2 < d**2:
                centers[8 + k * 3 + 0] = [x, y - r_i]
                centers[8 + k * 3 + 1] = [x + np.sqrt(3) * 0.5 * r_i, y]
                centers[8 + k * 3 + 2] = [x - np.sqrt(3) * 0.5 * r_i, y]
        k += 1
    
    radii = np.array([r_c] * 4 + [r_e] * 4 + [r_i] * (26 - 8))
    return centers, radii

def compute_sum_radii(radii):
    return radii.sum()

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def run_packing():
    bounds = [(0.3, 0.49), (0.1, 0.25), (0.05, 0.15)]
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [(0, 1), (0, 1)] * 26 + [(0, None)]
    cons = [{'type': 'ineq', 'fun': lambda v, i=i: v[26 + i] - min(v[2*i]-v[2*i+1], v[2*i+2]-v[2*i+3], v[2*i+4]-v[2*i+5], v[2*i+6]-v[2*i+7])} for i in range(26)]
    res = minimize(lambda v: -compute_sum_radii(v[26:]), x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol':1e-10, 'gtol':1e-10, 'maxiter':2000})
    centers = res.x[:52].reshape(26,2)
    radii = res.x[52:]

    sum_r = radii.sum()
    return centers, radii, sum_r
