import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog
from scipy.stats import norm

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    side_length = 1.0
    t_max = 0.12
    t_min = 1e-6
    cooling_factor = 0.995
    moves_per_temp = 200
    sigma_move = None

    def generate_initial_centers():
        s = side_length / np.sqrt(7)
        x = np.linspace(s, side_length - s, int(side_length / s))
        y = np.linspace(s, side_length - s, int(side_length / s))
        lattice_points = np.column_stack(np.meshgrid(x, y))
        valid_points = lattice_points[~((lattice_points[:, 0] < s) | (lattice_points[:, 1] < s) | 
                                        (lattice_points[:, 0] > side_length - s) | (lattice_points[:, 1] > side_length - s))]
        np.random.shuffle(valid_points)
        return valid_points[:n]

    def calculate_dists(centers):
        return squareform(pdist(centers))

    def linprog_radii(centers):
        dists = calculate_dists(centers)
        b = dists.flatten() - 1e-12
        bounds = zip(np.zeros(n), side_length / 2 * np.ones(n))
        result = linprog([-1] * n, A_ub=dists.T, b_ub=b, bounds=bounds, method='highs')
        return result.x, -result.fun

    def update_radius_and_center(centers, radius):
        return np.where(np.isclose(radius, 0, atol=1e-12), 0, radius), centers
    
    def accept_probability(current_sum, new_sum, temperature):
        return np.exp((new_sum - current_sum) / temperature)

    def annealing_step(current_centers, current_sum, temperature, center_changes):
        new_centers = np.copy(current_centers)
        for c in center_changes:
            delta = norm.rvs(scale=sigma_move, size=2)
            new_centers[c] = current_centers[c] + delta
            new_centers[c] = np.clip(new_centers[c], 1e-12, 1 - 1e-12)
        new_radii, new_sum = linprog_radii(new_centers)
        new_radii, new_centers = update_radius_and_center(new_radii, new_centers)
        if new_sum > current_sum or np.random.rand() < accept_probability(current_sum, new_sum, temperature):
            return new_centers, new_sum
        else:
            return current_centers, current_sum

    def simulated_annealing():
        current_centers = generate_initial_centers()
        current_sum = 0
        best_centers = current_centers
        best_sum = 0
        t = t_max
        while t > t_min:
            center_changes = np.random.choice(n, size=int(n * 0.1), replace=False)
            current_centers, current_sum = annealing_step(current_centers, current_sum, t, center_changes)
            if current_sum > best_sum:
                best_centers = current_centers
                best_sum = current_sum
            t *= cooling_factor
        return best_centers, best_sum

    sigma_move = t_max / 5
    centers, sum_radii = simulated_annealing()

    radii, sum_radii = linprog_radii(centers)
    radii, centers = update_radius_and_center(radii, centers)

    return centers, radii, sum_radii
