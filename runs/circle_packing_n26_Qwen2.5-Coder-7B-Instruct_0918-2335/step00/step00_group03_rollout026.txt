def half_plane_clipping(polygon, a, b):
    epsilon = 1e-12
    result = []
    n = len(polygon)
    for i in range(n):
        j = (i + 1) % n
        xi, yi = polygon[i]
        xj, yj = polygon[j]
        test_i = a * xi + b + epsilon > 0
        test_j = a * xj + b + epsilon > 0
        if test_i != test_j:
            result.append([(xi * b + xj * (a * yi + b) - yi * (a * xj + b)) / (a * yj - a * yi), (a * (xi * yj - xj * yi) + b * (xj - xi)) / (a * yj - a * yi)])
        if test_i:
            result.append([xi, yi])
    return result

            continue

        rj = radii[j]

    return np.array(polygon)

        t = dot / det
        px_t, py_t = xi + t * (xj - xi), yi + t * (yj - yi)
        dist = np.sqrt((px_t - px) ** 2 + (py_t - py) ** 2)
        if det < 0:
            dist = np.sqrt((xi - px) ** 2 + (yi - py) ** 2)
        min_dist = min(min_dist, dist)
    return min_dist
