import numpy as np
from scipy.optimize import linprog,.Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Hexagonal lattice setup
    s = 0.312  # Spacing for the hexagonal lattice
    hex_coords = np.array([(i + 0.5 * (j % 2)) * s, (np.sqrt(3) / 2) * j * s])
    hex_coords = hex_coords.T % 1  # Wrap around the unit square
    
    # Remove points outside the unit square
    hex_coords = hex_coords[(hex_coords[:, 0] <= 1) & (hex_coords[:, 1] <= 1)]
    
    # Select first 26 points
    centers = hex_coords[:26]
    
    # Step 2: SA parameters
    T = lambda t: 0.12 * (0.995 ** t)
    T_min = 1e-6
    c = 0.995
    M = 200
    sigma_move = lambda T: T / 5
    
    # Step 3: Warm start function for lp
    def warm_start(r):
        return r
    
    # Step 4: Simulated Annealing loop
    best_sum = 0
    best_centers = centers.copy()
    for t in range(int(np.log2(1 / T_min))):
        for _ in range(M):
            # Pick a random circle and move it
            i = np.random.randint(0, 26)
            delta = sigma_move(T(t)) * np.random.randn(2)
            new_center = centers[i] + delta
            new_center = np.clip(new_center, 0, 1)
            
            # Solve the lp problem for the new centers
            indices = np.arange(404).reshape(26, 16, 2)
            A_eq_indices = np.sum((indices - np.expand_dims(np.arange(26), 1)) * 16, axis=2)
            A_eq = np.eye(404)
            A_eq[A_eq_indices] = 1
            
            b_eq = 404 * [1j]  # This will be updated correctly later
            for i in range(26):
                for j in range(i + 1, 26):
                    b_eq[A_eq_indices[i * 16 + j]] = centers[i, :] - centers[j, :]
                b_eq[A_eq_indices[i * 16 + 26 + i]] = new_center[0]
                b_eq[A_eq_indices[i * 16 + 26 + i + 8]] = new_center[1]
                b_eq[A_eq_indices[i * 16 + 52]] = centers[i, 0]
                b_eq[A_eq_indices[i * 16 + 53]] = centers[i, 1]
                b_eq[A_eq_indices[i * 16 + 76]] = 1 - centers[i, 0]
                b_eq[A_eq_indices[i * 16 + 77]] = 1 - centers[i, 1]
            
            # Solve the linear program
            result = linprog(-np.ones(26), A_ub=A_eq, b_ub=b_eq, bounds=Bounds(lb=0, ub=None), method='highs', warm_start=warm_start([0] * 26))
            warm_start = lambda r: result.x
            
            if result.success and result.fun < best_sum:
                new_centers = centers.copy()
                new_centers[i] = new_center
                centers = new_centers
                best_sum = -result.fun
    
    # Step 5: Final linear programming for optimal radii
    A_eq_indices = np.sum((indices - np.expand_dims(np.arange(26), 1)) * 16, axis=2)
    A_eq = np.eye(404)
    b_eq = np.zeros(404)
    for i in range(26):
        for j in range(i + 1, 26):
            b_eq[A_eq_indices[i * 16 + j]] = centers[i, :] - centers[j, :]
        b_eq[A_eq_indices[i * 16 + 26 + i]] = centers[i, 0]
        b_eq[A_eq_indices[i * 16 + 26 + i + 8]] = centers[i, 1]
        b_eq[A_eq_indices[i * 16 + 52]] = centers[i, 0]
        b_eq[A_eq_indices[i * 16 + 53]] = centers[i, 1]
        b_eq[A_eq_indices[i * 16 + 76]] = 1 - centers[i, 0]
        b_eq[A_eq_indices[i * 16 + 77]] = 1 - centers[i, 1]
    
    result = linprog(-np.ones(26), A_ub=A_eq, b_ub=b_eq, bounds=Bounds(lb=0, ub=None), method='highs', warm_start=warm_start([0] * 26))
    
    # Extract centers, radii, and sum
    radii = result.x
    sum_radii = -result.fun
    return centers, radii, sum_radii
