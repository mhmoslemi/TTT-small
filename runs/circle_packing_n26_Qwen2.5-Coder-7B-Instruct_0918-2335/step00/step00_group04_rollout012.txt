import numpy as np
import scipy.spatial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = []
    radii = []

    # Start with four corner circles of radius 0.25
    centers.extend([(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)])
    radii.extend([0.25] * 4)

    while len(centers) < 26:
        obstacles = [(0, 0), (1, 0), (1, 1), (0, 1)]
        for c1, r1 in zip(centers, radii):
            obstacles.append((c1[0], c1[1], r1))

        vor = scipy.spatial.Voronoi(obstacles)
        max_radius_centers = []
        max_radius = 0.0
        for p in vor.vertices:
            radius = np.inf
            for obstacle in obstacles:
                if len(obstacle) == 2:  # wall
                    d = min(abs(p[0]), 1 - abs(p[0]), abs(p[1]), 1 - abs(p[1]))
                else:  # circle
                    d = np.linalg.norm(np.array(p) - np.array(obstacle[:2])) - obstacle[2]
                radius = min(radius, d)
            if radius > max_radius:
                max_radius = radius
                max_radius_centers = [p]
            elif radius == max_radius:
                max_radius_centers.append(p)

        if max_radius < 1e-6:
            break

        c_new = np.mean(max_radius_centers, axis=0)
        if any(c_new < 0) or any(c_new > 1):
            c_new = np.clip(c_new, 0, 1)
        r_new = max_radius

        centers.append(tuple(c_new))
        radii.append(r_new)

    # Final LP relaxation
    C = np.array(centers)
    R = np.array(radii)
    n = len(C)

    A = np.zeros((n*(n-1)//2, 2*n+2))
    b = np.zeros((n*(n-1)//2, 1))
    
    edge_counter = 0
    for i in range(n):
        for j in range(i+1, n):
            A[edge_counter, i] = 1
            A[edge_counter, j] = 1
            A[edge_counter, 2*n] = 1
            edge_counter += 1
    
    edge_counter = 0
    for i in range(n):
        A[edge_counter, i] = -1
        A[edge_counter, n] = 1
        edge_counter += 1
    
    edge_counter = 0
    for i in range(n):
        A[edge_counter, i] = -1
        A[edge_counter, n+1] = 1
        edge_counter += 1
    
    edge_counter = 0
    for i in range(n):
        A[edge_counter, i] = -1
        A[edge_counter, n+2] = 1
        edge_counter += 1
    
    for i in range(n):
        A[edge_counter, 2*i] = 1
        A[edge_counter, 2*i+1] = -1
        A[edge_counter, 2*n+1] = -1
        edge_counter += 1
    
    for i in range(n):
        A[edge_counter, 2*i] = 1
        A[edge_counter, 2*i+1] = -1
        A[edge_counter, 2*n+2] = -1
        edge_counter += 1
        

    A[n*(n-1)//2:, 2*n:] = -np.eye(n)
    b[n*(n-1)//2:, :] = radii

    res = scipy.optimize.linprog(-R, A_ub=A, b_ub=b, bounds=(0, None))
    radii = -res.x[:n]
    sum_radii = radii.sum()

    return np.array(centers), radii, sum_radii
