def sutherland_hodgman(polygon, clipper):
    result = []
    clipper_x1, clipper_y1 = clipper
    clipper_x2, clipper_y2 = clipper
    
    for i in range(len(polygon)):
        x, y = polygon[i]
        x1, y1 = clipper_x2, clipper_y2
        if point_in_halfplane((x, y), clipper):
            if not point_in_halfplane((x1, y1), clipper):
                ix, iy = intersection((x, y), (x1, y1), clipper)
                result.append((ix, iy))
            result.append((x, y))
        else:
            if point_in_halfplane((x1, y1), clipper):
                ix, iy = intersection((x, y), (x1, y1), clipper)
                result.append((ix, iy))
        
        x1, y1 = x, y
    
    return result

def point_in_halfplane(point, clipper):
    px, py = point
    cx1, cy1 = clipper[:2]
    cx2, cy2 = clipper[2:]
    return (cx2 - cx1) * (py - cy1) - (cy2 - cy1) * (px - cx1) >= 0

def intersection(p1, p2, clipper):

    x2, y2 = p2
    cx1, cy1, cx2, cy2 = clipper

    v = -((cx2 - cx1) * (y1 - cy1) - (cy2 - cy1) * (x1 - cx1)) / d
    
    ix = x1 + u * (x2 - x1)
    iy = y1 + u * (y2 - y1)
