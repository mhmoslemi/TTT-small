import numpy as np
from scipy.optimize import minimize, Bounds


def objective(z):
    return -np.sum(z[2::3])


def constraints(z):
    n = len(z) // 3
    cons = []
    for i in range(n):
        x1, y1, r1 = z[i * 3: i * 3 + 3]
        cons.append({
            'type': 'ineq',
            'fun': lambda z, i=i: x1 - r1
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda z, i=i: 1 - x1 - r1
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda z, i=i: y1 - r1
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda z, i=i: 1 - y1 - r1
        })
        for j in range(i + 1, n):
            x2, y2, r2 = z[j * 3: j * 3 + 3]
            cons.append({
                'type': 'ineq',
                'fun': lambda z, i=i, j=j: (x1 - x2)**2 + (y1 - y2)**2 - (r1 + r2)**2
            })
    return cons


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    initial_positions = np.random.rand(n, 2)
    initial_radii = np.min(np.minimum(initial_positions, 1 - initial_positions), axis=1) / 2 * 0.3
    bounds = Bounds(0, 0.5)

    def objective_with_fixed(z):
        return -np.sum(z[::3])

    best_solution = None
