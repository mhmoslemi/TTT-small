```python
import numpy as np
from scipy.optimize import minimize
import random

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (26, 2))
    radii = np.zeros(26)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[2*26:])

def constraints(v):
    centers = v[:2*26].reshape((26, 2))
    radii = v[2*26:]
    n = centers.shape[0]
    pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)
    distances = np.sqrt(np.sum((centers[pair_idx_i] - centers[pair_idx_j]) ** 2, axis=1))
    boundary_constraints = np.column_stack(
        (centers[:, 0] - radii, 1 - centers[:, 0] - radii,
         centers[:, 1] - radii, 1 - centers[:, 1] - radii))
    pairwise_constraints = distances - (radii[pair_idx_i] + radii[pair_idx_j])
    return np.concatenate((boundary_constraints.flatten(), pairwise_constraints))

n = 26
pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)
bounds = [(0.0, 0.5)] * (2 * n)

best_value = float('-inf')
best_v = None
RESTARTS = 10
for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun=objective,
        x0=v0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if not res.success:
        continue
    value = -res.fun
    if value > best_value:
        best_value = value
        best_v = res.x

centers = best_v[:2*n].reshape((n, 2))
radii = np.maximum(0.0, best_v[2*n:] - 1e-7)
sum_radii = np.sum(radii)

return centers, radii, sum_radii
```
