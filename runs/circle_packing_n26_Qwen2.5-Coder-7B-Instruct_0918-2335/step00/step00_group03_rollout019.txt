import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    N = 26
    r_min = 1e-3
    iterations = 10000

    # RSA Seed
    def place_circle():
        for _ in range(iterations):
            x, y = np.random.uniform(0, 1, 2)
            r = r_min
            if all(np.sqrt((x - cx)**2 + (y - cy)**2) > 2 * r for cx, cy in circles):
                circles.append((x, y, r))
                return True
        return False

    circles = []
    while len(circles) < N:
        if not place_circle():
            circles = []

    # Physics-Based Local Relaxation
    T = 2000
    dt = 1e-3
    for _ in range(T):
        forces = np.zeros((N, 2))
        for i in range(N):
            for j in range(i + 1, N):
                dx, dy = circles[i][0] - circles[j][0], circles[i][1] - circles[j][1]
                gap = np.sqrt(dx**2 + dy**2) - (circles[i][2] + circles[j][2])
                if gap < 0:
                    forces[i] += gap * np.array([-dx, -dy]) / abs(gap)
                    forces[j] += gap * np.array([dx, dy]) / abs(gap)
            force = forces[i]
            circles[i] = (max(0, min(1, circles[i][0] + dt * force[0])),
                          max(0, min(1, circles[i][1] + dt * force[1])),
                          max(0, circles[i][2] * (1 + dt * force[0])))
            circles[i] = (max(0, min(1, (circles[i][0] - 0.5) * (1 - circles[i][2] / (1 - circles[i][0])) + 0.5)),
                          max(0, min(1, (circles[i][1] - 0.5) * (1 - circles[i][2] / (1 - circles[i][1])) + 0.5)),
                          max(0, circles[i][2]))

    # Global Uniform Expansion
    def scale_factor():
        s_ij = np.array([[min(np.sqrt((circles[i][0] - circles[j][0])**2 + (circles[i][1] - circles[j][1])**2), circles[i][2] + circles[j][2]) for j in range(i + 1, N)] for i in range(N)]).ravel()
        s_s = np.array([circles[i][2] / (1 - circles[i][2]) for i in range(N)]).ravel()
        return min(np.min(s_ij), np.min(s_s))

    s_max = scale_factor()
    circles = [(x * s_max, y * s_max, r * s_max) for x, y, r in circles]

    # Exact Penalty + L-BFGS Refinement
    positions = np.array([circle[:2] for circle in circles]).ravel()
    radii = np.array([circle[2] for circle in circles])
    decision_variables = np.hstack((positions, radii))
    alpha = 1e5

    def objective(decision_variables):
        positions = decision_variables[:52].reshape(-1, 2)
        radii = decision_variables[52:]
        return -radii.sum()

    def penalty(decision_variables):
        positions = decision_variables[:52].reshape(-1, 2)
        radii = decision_variables[52:]
        penalty_constraint = 0
        for i in range(N):
            for j in range(i + 1, N):
                penalty_constraint += max(0, (radii[i] + radii[j]) - np.sqrt(np.sum((positions[i] - positions[j])**2)))**2
            penalty_constraint += max(0, radii[i] - positions[i][0])**2 + max(0, positions[i][0] - radii[i])**2
            penalty_constraint += max(0, radii[i] - positions[i][1])**2 + max(0, positions[i][1] - radii[i])**2
        return alpha * penalty_constraint

    def constraints(decision_variables):
        return [lambda x: objective(x), lambda x: penalty(x)]

    result = minimize(lambda x: objective(x) + penalty(x), decision_variables, constraints=constraints, method='L-BFGS-B', bounds=[(0, 1)] * 52 + [(0, 0.5)] * N)
    new_radii = result.x[52:]

    centers = result.x[:52].reshape(-1, 2)
    radii = new_radii
    sum_radii = radii.sum()

    return centers, radii, sum_radii
