import numpy as np
from scipy.optimize import minimize

def hex_lattice(spacing, square_size):
    rows = int(square_size / spacing)
    cols = int((square_size / spacing) * np.sqrt(3))
    centers = []
    even_row = False
    for r in range(rows):
        for c in range(cols):
            x = (c + (0.5 if even_row else 0.25)) * spacing
            y = (r * 0.8660254037844386) * spacing
            if 0 <= x <= square_size and 0 <= y <= square_size:
                centers.append([x, y])
        even_row = not even_row
    return np.array(centers)

def distance(c1, c2):
    return np.sqrt(np.sum((c1 - c2) ** 2))

def feasible(c, r, square_size):
    return (0 <= c[0] - r < square_size and 0 <= c[1] - r < square_size and
            0 <= c[0] + r < square_size and 0 <= c[1] + r < square_size)

def penalty(z, spacing, square_size):
    N = len(z) // 3
    obj = 0
    for i in range(N):
        r = z[2 * i + 2]
        if not feasible(z[2 * i:2 * i + 2], r, square_size):
            obj += 1e6 * (1 - r)
            # obj += 1e6 * (r - 0.1)
    for i in range(N):
        for j in range(i + 1, N):
            if not feasible(z[2 * i:2 * i + 2], z[2 * i + 2], square_size) or \
               not feasible(z[2 * j:2 * j + 2], z[2 * j + 2], square_size):
                obj += 1e6 * (1 - distance(z[2 * i:2 * i + 2], z[2 * j:2 * j + 2]))
                # obj += 1e6 * (radius(z[2 * i:2 * i + 2], z[2 * i + 2], z[2 * j:2 * j + 2], z[2 * j + 2]) - 0.1)
    return obj

def solve_stage(z0, delta):
    epsilon = 1e-8
    bounds = [(z0[2 * i + 0] - epsilon, z0[2 * i + 0] + epsilon),
              (z0[2 * i + 1] - epsilon, z0[2 * i + 1] + epsilon),
              (z0[2 * i + 2] - epsilon, z0[2 * i + 2] + epsilon) for i in range(len(z0) // 3)]
    cons = []
    for i in range(len(z0) // 3):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2 * i + 0] - z[2 * i + 2] - epsilon})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2 * i + 1] - z[2 * i + 2] - epsilon})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2 * i + 0] + z[2 * i + 2] - 1 + epsilon})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: z[2 * i + 1] + z[2 * i + 2] - 1 + epsilon})
    for i in range(len(z0) // 3):
        for j in range(i + 1, len(z0) // 3):
            def cons_func(z, i=i, j=j, delta=delta):
                d = distance(z[2 * i:2 * i + 2], z[2 * j:2 * j + 2])
                return d - (z[2 * i + 2] + z[2 * j + 2] + delta)
            cons.append({'type': 'ineq', 'fun': cons_func})
    res = minimize(lambda z: penalty(z, z0[0], z0[1]) + z[2], z0, method='SLSQP', bounds=bounds, constraints=cons)
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    square_size = 1.0
    spacing = square_size / np.sqrt(26)
    centers = hex_lattice(spacing, square_size)
    N = len(centers)
    initial_radii = spacing / 2
    initial_rhos = 1 - initial_radii  # rhos = 1 - rs
    initial_z = np.zeros(3 * N)
    initial_z[::3] = centers[:, 0]
    initial_z[1::3] = centers[:, 1]
    initial_z[2::3] = initial_radii
    results = [initial_z]
    delta = 5e-4
    epsilon = 1e-8
    for _ in range(3):
        results.append(solve_stage(results[-1], delta))
        delta *= 0.2
        epsilon *= 0.2
    best_idx = np.argmin([np.sum(result[2::3]) for result in results])
    result = results[best_idx]
    centers = result[::3].reshape(N, 2)
    radii = result[2::3]
    if np.any(radii < 0):
        print("Error: Negative radius found, this should not happen.")
        radii[radii < 0] = 0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
