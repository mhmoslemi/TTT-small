import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_radius = 1 / np.sqrt(2 * n)
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, max_radius)

    def objective(vars):
        centers = vars[:n * 2].reshape(n, 2)
        radii = vars[n * 2:]
        return -np.sum(radii)

    def constraints(vars):
        centers = vars[:n * 2].reshape(n, 2)
        radii = vars[n * 2:]
        constraints = [
            {'type': 'ineq', 'fun': lambda vars: np.maximum(0, centers[:, 0] - radii)},
            {'type': 'ineq', 'fun': lambda vars: np.maximum(0, centers[:, 1] - radii)},
            {'type': 'ineq', 'fun': lambda vars: np.maximum(0, 1 - centers[:, 0] - radii)},
            {'type': 'ineq', 'fun': lambda vars: np.maximum(0, 1 - centers[:, 1] - radii)}
        ]
        dist_matrix = distance_matrix(centers, centers)
        np.fill_diagonal(dist_matrix, np.inf)
        constraints.extend([
            {'type': 'ineq', 'fun': lambda vars: dist_matrix - (radii[:, None] + radii)}
        ])
        return constraints

    bounds = [(0, None), (0, 1)] * n + n * [(0, None)]
    result = minimize(objective, np.hstack((initial_centers.flatten(), initial_radii)), bounds=bounds, constraints=constraints)
    centers = result.x[:n * 2].reshape(n, 2)
    radii = result.x[n * 2:]
    sum_radii = -result.fun

    return centers, radii, sum_radii
