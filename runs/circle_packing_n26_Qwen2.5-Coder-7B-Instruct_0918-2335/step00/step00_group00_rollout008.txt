import numpy as np
from scipy.spatial.distance import cdist

def pack_circles(n_circles, max_radius, target_sum):
    centers = np.zeros((n_circles, 2))
    radii = np.zeros((n_circles,))
    
    # Initialize circles at the center of the square
    centers[:n_circles//4] = np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]])
    if n_circles % 4 == 0:
        n_circles -= 4
    else:
        n_circles -= 3

    while n_circles > 0:
        # Choose a random point in the square
        new_center = np.random.rand(2)
        # Ensure the new circle does not overlap with existing circles
        while any(cdist([new_center], centers[:n_circles], 'euclidean') < (radii[:n_circles] + max_radius)[:, np.newaxis]):
            new_center = np.random.rand(2)
        centers[n_circles] = new_center
        max_radius -= 0.05
        n_circles -= 1
        # Adjust the sum of radii to stay close to the target
        if sum(radii) < target_sum:
            radii = np.array([max_radius] * (n_circles + 4))

    return centers, radii, sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = pack_circles(26, 0.4, 2.636)
    return centers, radii, sum_radii
