from scipy.optimize import linprog, differential_evolution, minimize

def evaluate(p):
    C = p.reshape(26,2)
    d = np.linalg.norm(C[:,None,:]-C[None,:,:], axis=-1)
    A = []
    b = []
    for i in range(26):
        for j in range(i+1, 26):
            A.append([1, 1])
            b.append(d[i, j])
        A.append([1, 0])
        b.append(C[i, 0])
        A.append([-1, 0])
        b.append(C[i, 0] - 1)
        A.append([1, 0])
        b.append(C[i, 1])
        A.append([-1, 0])
        b.append(C[i, 1] - 1)
    A = np.array(A)
    b = np.array(b)
    res = linprog(c=-np.ones(26), A_ub=A, b_ub=b, method='highs')
    return -res.fun
