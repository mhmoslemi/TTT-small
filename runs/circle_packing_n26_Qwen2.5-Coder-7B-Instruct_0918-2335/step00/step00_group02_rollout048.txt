import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x, N, lambda_penalty):
    xs, ys, rs = x[:N], x[N:2*N], x[2*N:]
    sum_r = -rs.sum()
    p_boundary = np.maximum(0, rs - xs)**2 + np.maximum(0, xs + rs - 1)**2 + np.maximum(0, rs - ys)**2 + np.maximum(0, ys + rs - 1)**2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    overlap = rr - dist
    p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
    return sum_r + lambda_penalty * (p_boundary + p_overlap)

def run_packing():
    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    
    best_result = None
    best_value = -np.inf
    random_seeds = [0, 1, 2, 3]
    
    for seed in random_seeds:
        np.random.seed(seed)
        result = differential_evolution(objective, bounds, args=(N, lambda_penalty), popsize=DE_popsize, maxiter=DE_maxiter, seed=seed, polish=True)
        if result.fun > best_value:
            best_value = result.fun
            best_result = result
    
    if best_result is None:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    
    x = best_result.x
    centres = np.column_stack([x[0:N*2:2], x[1:N*2:2]])
    radii = x[2*N:]
    sum_radii = radii.sum()
    
    return centres, radii, sum_radii
