import numpy as np

def run_packing() -> tuple:
    # Define a checkerboard pattern
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    cell_size = 1 / 5
    for i in range(5):
        for j in range(5):
            if (i + j) % 2 == 0:
                x = i * cell_size + cell_size / 2
                y = j * cell_size + cell_size / 2
                r = 0.3 * min(cell_size / 2, (cell_size / 2 - (x % cell_size) - (x % cell_size - cell_size / 2)) ** 2 + 
                                    (cell_size / 2 - (y % cell_size) - (y % cell_size - cell_size / 2)) ** 2)
                centers[5*i+j] = [x, y]
                radii[5*i+j] = r
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
