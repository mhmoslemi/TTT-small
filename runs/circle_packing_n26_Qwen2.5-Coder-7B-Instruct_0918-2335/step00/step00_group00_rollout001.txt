import numpy as np
from scipy.optimize import trust_constr, NonlinearConstraint

def run_packing():
    n = 26
    centers = []
    
    # Generate initial positions: 5x5 grid for 25 + one extra at (0.6,0.9)
    for i in range(5):
        for j in range(5):
            centers.append(np.array([(j+0.5)/5.0, (i+0.5)/5.0], dtype=np.float64))
            if len(centers) == 25:
                break
        if len(centers) == 25:
            break
    extra_center = np.array([0.6, 0.9], dtype=np.float64)
    centers.append(extra_center)
    centers = np.array(centers, dtype=np.float64)
    
    radii = np.full(n, 0.04, dtype=np.float64)
    x0 = np.hstack([centers.flatten(), radii])
    
    bounds = []
    for i in range(26):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x, y
    for i in range(26):
        bounds.append((0.0, 0.5))
    
    pair_indices = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    def constraint_func(v):
        x = v[:52].reshape((n, 2))
        r = v[52:]
        arr = []
        # Boundary constraints
        arr.append(x[:, 0] - r)  # >=0
        arr.append(1 - x[:, 0] - r)
        arr.append(x[:, 1] - r)
        arr.append(1 - x[:, 1] - r)
        # Pairwise constraints
        dx = x[:, None, 0] - x[None, :, 0]
        dy = x[:, None, 1] - x[None, :, 1]
        dist2 = dx**2 + dy**2
        arr.append(dist2 - (r[:, None] + r[None, :])**2)
        arr = np.concatenate(arr, axis=0)
        return arr
    
    cons = NonlinearConstraint(constraint_func, 0, np.inf)
    
    res = trust_constr(
        res=-radii,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 2000, 'gtol': 1e-12, 'xtol': 1e-12}
    )
    
    if res.success:
        centers = res.x[:52].reshape((n, 2))
        radii = res.x[52:]
    else:
        # Fallback to naive pattern
        centers = np.array([(0.1*i, 0.1*j) for i in range(5) for j in range(5)], dtype=np.float64)
        centers = np.vstack([centers, np.array([0.5, 0.5]), np.array([0.6, 0.9])], dtype=np.float64)
        radii = np.full(n, 0.105, dtype=np.float64)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
