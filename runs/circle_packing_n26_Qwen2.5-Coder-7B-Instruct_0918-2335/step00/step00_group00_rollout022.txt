import numpy as np
from scipy.optimize import minimize

def inflate(circles, Δr, ε):
    while True:
        changed = False
        for i in range(len(circles)):
            for j in range(i + 1, len(circles)):
                d_ij = np.sqrt((circles[i][0] - circles[j][0])**2 + (circles[i][1] - circles[j][1])**2)
                if d_ij < circles[i][2] + circles[j][2] - ε:
                    alpha = (circles[i][2] + circles[j][2] - d_ij) / (2 * d_ij)
                    circles[i][:2] += alpha * ((circles[i][:2] - circles[j][:2]) / 2)
                    circles[j][:2] -= alpha * ((circles[i][:2] - circles[j][:2]) / 2)
                    changed = True
        for i, (x, y, r) in enumerate(circles):
            circles[i][:2] = np.clip(circles[i][:2], r, 1 - r)
            circles[i][2] = min(circles[i][2] + Δr, min(x, 1 - x, y, 1 - y) - ε)
        if not changed:
            break

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers_A = np.array([
        (0.1, 0.1), (0.1, 0.5), (0.1, 0.9), (0.5, 0.1), (0.5, 0.5), (0.5, 0.9),
        (0.9, 0.1), (0.9, 0.5), (0.9, 0.9), (0.3, 0.3), (0.3, 0.7), (0.7, 0.3), (0.7, 0.7),
        (0.45, 0.2), (0.45, 0.6), (0.55, 0.2), (0.55, 0.6), (0.6, 0.15), (0.6, 0.85)
    ])
    centers_B = np.array([
        (0.1, 0.1), (0.1, 0.5), (0.1, 0.9), (0.5, 0.1), (0.5, 0.5), (0.5, 0.9),
        (0.9, 0.1), (0.9, 0.5), (0.9, 0.9), (0.3, 0.3), (0.3, 0.7), (0.7, 0.3), (0.7, 0.7),
        (0.45, 0.2), (0.45, 0.6), (0.55, 0.2), (0.55, 0.6), (0.6, 0.15), (0.6, 0.85)
    ])
    radii_A = np.full(26, 0.01)
    radii_B = np.full(26, 0.01)
    for variant, center, radii in zip(['A', 'B'], [centers_A, centers_B], [radii_A, radii_B]):
        inflate(np.column_stack((center, radii)), Δr=0.0005, ε=1e-6)
        sum_radii = np.sum(radii)
        if sum_radii > 2.636:
            return center, radii, sum_radii

sum_radii_A, sum_radii_B = run_packing()[-1], run_packing()[-1]
if sum_radii_A > sum_radii_B:
    return run_packing()[:-1]
else:
    return run_packing()[:-1]
