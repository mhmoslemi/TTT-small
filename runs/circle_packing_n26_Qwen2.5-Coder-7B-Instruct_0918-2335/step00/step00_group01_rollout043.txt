import numpy as np
from scipy.optimize import linprog

def run_packing():
    def sim_collision_time(c1, v1, r1, c2, v2, r2):
        delta_c0 = c1 - c2
        delta_v = v1 - v2
        delta_r0 = r1 + r2
        g_tot = 2 * 1e-3
        a = np.dot(delta_v, delta_v)
        b = 2 * np.dot(delta_c0, delta_v)
        c = np.dot(delta_c0, delta_c0) - (delta_r0 + g_tot)**2
        solutions = np.roots([a, b, c])
        mask = np.isreal(solutions) & (solutions > 0)
        if np.any(mask):
            return solutions[mask].real.min()
        return np.inf

    def sim_wall_collision_time(c, v, r, wall):
        if wall == 'left':
            x_t = c[0] + v[0] * 5e-2
            if r > x_t:
                return (r - x_t) / (v[0] - 1e-3)
        elif wall == 'right':
            x_t = c[0] + v[0] * 5e-2
            if r > 1 - x_t:
                return (r - (1 - x_t)) / (v[0] - 1e-3)
        elif wall == 'bottom':
            y_t = c[1] + v[1] * 5e-2
            if r > y_t:
                return (r - y_t) / (v[1] - 1e-3)
        elif wall == 'top':
            y_t = c[1] + v[1] * 5e-2
            if r > 1 - y_t:
                return (r - (1 - y_t)) / (v[1] - 1e-3)
        return np.inf

    def check_overlap(centers, radii):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    return True
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 <= x - r <= 1 and 0 <= y - r <= 1 and 0 <= x + r <= 1 and 0 <= y + r <= 1):
                return True
        return False

    def fix_overlap(centers, radii):
        epsilon = 1e-6
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    delta_center = centers[i] - centers[j]
                    overlap = (radii[i] + radii[j] - dist) / 2
                    centers[i] -= delta_center * (overlap / np.linalg.norm(delta_center))
        return centers, radii

    def linear_program(centers, radii):
        n = len(centers)
        c = np.ones(n)
        A = np.zeros((3*n, n))
        b = np.zeros(3*n)
        for i in range(n):
            A[i, i] = 1
            A[n + i, i] = -2
            A[2*n + i, i] = 1
            b[i] = 1e-4
            b[n + i] = centers[i][0] - 1e-4
            b[2*n + i] = 1 - centers[i][1] - 1e-4
        for i in range(n):
            for j in range(i + 1, n):
                x_ji = centers[j] - centers[i]
                r_ij = radii[i] + radii[j]
                A[3*n + i*n + j, i] = 2 * x_ji[0]
                A[3*n + i*n + j, j] = -2 * x_ji[0]
                A[3*n + i*n + j, i + n] = 2 * x_ji[1]
                A[3*n + i*n + j, j + n] = -2 * x_ji[1]
                b[3*n + i*n + j] = r_ij**2 - (1e-4)**2
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
        return res.x

    np.random.seed(42)
    centers = np.random.rand(26, 2) * 0.95 + 0.025
    velocities = (np.random.rand(26, 2) - 0.5) * 5e-3 * np.random.randn(26, 2)
    radii = np.full(26, 1e-4)
    current_time = 0
    T_max = 50.0
    dt = 0

    event_queue = []
    for i in range(26):
        for j in range(i + 1, 26):
            t_cc = sim_collision_time(centers[i], velocities[i], radii[i], centers[j], velocities[j], radii[j])
            if not np.isinf(t_cc):
                heapq.heappush(event_queue, (t_cc, 'cc', i, j))
        for edge in ['left', 'right', 'bottom', 'top']:
            t_cw = sim_wall_collision_time(centers[i], velocities[i], radii[i], edge)
            if not np.isinf(t_cw):
                heapq.heappush(event_queue, (t_cw, 'cw', i, edge))

    while event_queue and current_time < T_max:
        t_e, event_type, i, j_edge = heapq.heappop(event_queue)
        dt = t_e - current_time
        centers[i] += velocities[i] * dt
        r_i = radii[i] + 1e-3 * dt
        if event_type == 'cc':
            j, = j_edge
            centers[j] += velocities[j] * dt
            r_j = radii[j] + 1e-3 * dt
            v1 = velocities[i] + (2 * (1 - 1) / (1 - 1)) * (np.dot(velocities[i] - velocities[j], centers[i] - centers[j]) / np.dot(centers[i] - centers[j], centers[i] - centers[j])) * (centers[i] - centers[j])
            velocities[i] = v1
            v2 = velocities[j] - (2 * (1 - 1) / (1 - 1)) * (np.dot(velocities[j] - velocities[i], centers[j] - centers[i]) / np.dot(centers[j] - centers[i], centers[j])) * (centers[j] - centers[i])
            velocities[j] = v2
            radii[i] = r_i
            radii[j] = r_j
        elif event_type == 'cw':
            x, y = centers[i]
            r = radii[i]
            if j_edge == 'left':
                velocity = np.array([-velocities[i, 0], velocities[i, 1]])
            elif j_edge == 'right':
                velocity = np.array([velocities[i, 0], -velocities[i, 1]])
            elif j_edge == 'bottom':
                velocity = np.array([velocities[i, 0], -velocities[i, 1]])
            elif j_edge == 'top':
                velocity = np.array([velocities[i, 0], velocities[i, 1]])
            dt = (r - x) / (velocity[0] - 1e-3)
            centers[i] += velocity * dt
            r_i = r - 1e-3 * dt
            velocities[i] = velocity
            radii[i] = r_i
        for i in [i]:
            for j in range(26):
                t_cc = sim_collision_time(centers[i], velocities[i], radii[i], centers[j], velocities[j], radii[j])
                if not np.isinf(t_cc):
                    heapq.heappush(event_queue, (t_cc, 'cc', i, j))
            for edge in ['left', 'right', 'bottom', 'top']:
                t_cw = sim_wall_collision_time(centers[i], velocities[i], radii[i], edge)
                if not np.isinf(t_cw):
                    heapq.heappush(event_queue, (t_cw, 'cw', i, edge))
        current_time = t_e

    centers, radii = fix_overlap(centers, radii)
    radii = linear_program(centers, radii)

    return centers, radii, np.sum(radii)
