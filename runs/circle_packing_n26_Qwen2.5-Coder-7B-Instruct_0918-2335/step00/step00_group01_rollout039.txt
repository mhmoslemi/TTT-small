import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_gen = 800
    init_multipliers = 10**2
    multipliers = np.array([init_multipliers, init_multipliers])
    repair_count = np.zeros(2)
    max_radius_clip = 0.49995

    def penalty_bound(r, x, y):
        return max(0, min(x, 1 - x, y, 1 - y) - np.clip(r, 0, None))

    def penalty_overlap(c_i, c_j, r_i, r_j):
        d = np.linalg.norm(c_i - c_j)
        return max(0, (r_i + r_j - d) ** 2)

    def objective_func(params):
        x = params[:n*n].reshape((n, 2))
        r = params[n*n:].reshape((n,))
        return -sum(r) + np.sum([multipliers[0] * penalty_bound(r_i, x_i[0], x_i[1]) + multipliers[1] * penalty_overlap(x[i], x[j], r[i], r[j]) for i in range(n) for j in range(i+1, n)])

    def penalty_grad_bound(i, r, x):
        return mul * (x[i] if r < x[i] else 1 - x[i] if r < 1 - x[i] else 0 if r < y[i] else 1 - y[i])

    def penalty_grad_overlap(i, j, r_i, r_j, d):
        return mul * (r_i if d < r_j + r_i else r_j if d < r_j + r_i else 0)

    def mutation(params, sigma_pos, sigma_rad):
        new_params = params.copy()
        for _ in range(np.random.randint(2, 5)):
            index = np.random.randint(n)
            if np.random.rand() < 0.5:
                new_params[index*n:(index+1)*n] += np.random.normal(0, sigma_pos, size=(2,))
            else:
                new_params[n*n + index] += np.random.normal(0, sigma_rad, size=1)
        return new_params

    def crossover(parents):
        child = np.zeros_like(parents)
        for i in range(n):
            child[i*n:(i+1)*n] = 0.5 * (parents[np.random.choice(2)] * (i+1) + parents[np.random.choice(2)] * (i+2))[:-1]
            child[n*n + i] = 0.5 * (parents[np.random.choice(2)][n*n + i] + parents[np.random.choice(2)][n*n + i])
        return child

    def repair(params):
        x = params[:n*n].reshape((n, 2))
        r = params[n*n:].reshape((n,))
        r_new = np.clip(r, 0, None)
        centers = x.copy()
        for i in range(n):
            for j in range(i+1, n):
                if np.linalg.norm(center[i] - center[j]) < np.abs(r_new[i] + r_new[j]):
                    overlap = r_new[i] + r_new[j] - np.linalg.norm(center[i] - center[j])
                    r_new[i] -= overlap * (r_new[i] / (r_new[i] + r_new[j]))
                    r_new[j] -= overlap * (r_new[j] / (r_new[i] + r_new[j]))
                    r_new = np.clip(r_new, 0, None)
        return centers.flatten(), r_new

    def initialization():
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        radius_step = 1 / (n+n)
        for i in range(n):
            row = i // 5
            col = i % 5
            xoffset = 0.105 * (col % 2) + 0.21 * col + 0.025 * (1 if row % 2 == 0 else 0)
            yoffset = 0.1 * row
            x = xoffset + np.random.rand() * radius_step
            y = yoffset + np.random.rand() * radius_step
            r = min(x, 1 - x, y, 1 - y) - 2e-4
            centers[i] = [x + r, y + r]
            radii[i] = r
        return centers.flatten(), radii

    population = np.array([initialization() for _ in range(50)])

    for gen in range(max_gen):
        feasible_population = [ind for ind in population if penalty_bound(ind[-1], ind[50], ind[51]) == 0 and np.all(penalty_overlap(ind[i*2:i*2+4], ind[j*2:j*2+4], ind[46+i], ind[46+j]) == 0 for i in range(n) for j in range(i+1, n))]
        if not feasible_population:
            center_tightness = [sum(penalty_bound(ind[-1], ind[50], ind[51]) for ind in pop) for pop in population]
            population[np.argmin(center_tightness)] = initialization()
        else:
            para_best_sol = None
            para_best_sol_objvalue = np.inf
            for ind in feasible_population:
                objvalue = objective_func(ind)
                if objvalue < para_best_sol_objvalue:
                    para_best_sol = ind
                    para_best_sol_objvalue = objvalue
            population[np.argmax(center_tightness)] = para_best_sol
        population = np.array([mutation(ind, 0.05, 0.02) if np.random.rand() < 0.7 else crossover(population[np.random.permutation(50)]) for ind in population])
        population = repair(population)
        multipliers = multipliers * 3

    return feasible_population[np.argmax(population[:,46])]
