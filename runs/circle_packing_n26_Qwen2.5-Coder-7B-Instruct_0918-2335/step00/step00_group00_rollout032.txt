import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

pair_indices = [(i, j) for i in range(26) for j in range(i + 1, 26)]

def constraint_func(v):
    x = v[:52].reshape((26, 2))
    r = v[52:]
    arr = []
    arr.append(x[:, 0] - r)  # x_i - r_i >= 0
    arr.append(1 - x[:, 0] - r)  # 1 - x_i - r_i >= 0
    arr.append(x[:, 1] - r)  # y_i - r_i >= 0
    arr.append(1 - x[:, 1] - r)  # 1 - y_i - r_i >= 0
    dx = x[pair_indices[:, 0], 0] - x[pair_indices[:, 1], 0]
    dy = x[pair_indices[:, 0], 1] - x[pair_indices[:, 1], 1]
    dist2 = dx ** 2 + dy ** 2
    pair_vals = dist2 - (r[pair_indices[:, 0]] + r[pair_indices[:, 1]]) ** 2
    arr.append(pair_vals)
    return np.concatenate(arr)

def run_packing():
    n = 26
    positions = np.array([((j+0.5)/5.0, (i+0.5)/5.0) for i in range(5) for j in range(5) if len(positions) < 25])[:, None]
    if positions.shape[0] < 26:
        positions = np.vstack([positions, np.array([(0.6, 0.9)])])
    radii_initial = np.full(n, 0.04)
    x0 = np.concatenate([positions.ravel(), radii_initial])
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))  # x, y bounds
    bounds.extend([(0.0, 0.5)] * n)  # radius bounds
    cons = {'type': 'ineq', 'fun': constraint_func}
    res = minimize(fun=lambda v: -np.sum(v[52:]), x0=x0, bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
    centers = res.x[:52].reshape((n, 2))
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print((centers, radii, sum_radii))
