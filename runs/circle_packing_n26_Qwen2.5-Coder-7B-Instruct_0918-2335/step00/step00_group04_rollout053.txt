def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    def solve_lp(centers):
        n = centers.shape[0]
        c = np.ones(n)
        A = []
        b = []
        for i in range(n):
            x_i = centers[i, 0]
            y_i = centers[i, 1]
            A.append([1, 0, x_i, 0, y_i, 0])
            b.append(1 / np.sqrt(x_i * y_i))
            for j in range(i + 1, n):
                x_j = centers[j, 0]
                y_j = centers[j, 1]
                d_ij = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
                A.append([1, 1, 0, x_i + x_j, 0, y_i + y_j])
                b.append(d_ij)
        A = np.array(A)
        b = np.array(b)
        res = linprog(c, A_ub=-A, b_ub=-b, bounds=(0, None), method='highs')
        return res.x

    def init_grid():
        x = (np.arange(6) + 0.5) / 6.0
        y = (np.arange(6) + 0.5) / 6.0
        x, y = np.meshgrid(x, y)
        centers = np.vstack((x.flatten(), y.flatten())).T
        return centers

    def remove_solid_grid(centers):
        areas = np.sqrt(3) * centers[:, 1] * (1 - centers[:, 1])
        pairs = np.array([[i, j] for i in range(len(centers)) for j in range(i + 1, len(centers))])
        distances = np.sqrt((centers[pairs[:, 0], 0] - centers[pairs[:, 1], 0]) ** 2 + 
                            (centers[pairs[:, 0], 1] - centers[pairs[:, 1], 1]) ** 2)
        area_ratio = areas[pairs[:, 0]] / (areas[pairs[:, 0]] + areas[pairs[:, 1]])
        ratios_diff = np.abs(area_ratio - 0.5)
        to_delete = np.argsort(ratios_diff)[:10]
        return np.delete(centers, to_delete, 0)

    def pos_update(centers, radii, epsilon=0.02, gamma=0.95, delta=1e-3):
        forces = np.zeros_like(centers)
        for i, (ci, ri) in enumerate(zip(centers, radii)):
            for j, (cj, rj) in enumerate(zip(centers, radii)):
                if i != j:
                    d_ij = np.linalg.norm(ci - cj)
                    if d_ij <= ri + rj:
                        a_ij = (ci - cj) * max(0, ri + rj - d_ij)
                        forces[i] += a_ij
                        forces[j] -= a_ij
            for axis in range(2):
                gap = ri - centers[i, axis]
                if gap > 0:
                    forces[i, axis] += gap
        forces /= np.linalg.norm(forces, axis=1, keepdims=True)
        return centers + delta * gamma**(0.01) * forces + epsilon * np.random.randn(*forces.shape)

    def project_to_box(centers):
        return np.clip(centers, 0, 1)

    def run_circular_increment():
        centers = init_grid()
        centers = remove_solid_grid(centers)
        for _ in range(5):
            centers += 0.02 * np.random.randn(*centers.shape)
            centers = project_to_box(centers)
        for _ in range(150):
            radii = solve_lp(centers)
            centers = pos_update(centers, radii)
            centers = project_to_box(centers)
        return centers, solve_lp(centers), np.sum(solve_lp(centers))

    return run_circular_increment()


# example usage

centers, radii, sum_radii = run_packing()
print(f"Final sum of radii: {sum_radii:.5f}")
