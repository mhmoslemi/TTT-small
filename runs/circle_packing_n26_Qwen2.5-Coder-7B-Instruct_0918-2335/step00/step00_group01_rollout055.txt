import numpy as np

def fitness(positions, radii, M_bound, M_overlap):
    sum_radii = np.sum(radii)
    pen_bound = np.sum(M_bound * np.maximum(0, radii - np.minimum(positions[:, 0], 1 - positions[:, 0], positions[:, 1], 1 - positions[:, 1])))
    pen_overlap = 0
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
            pen_overlap += M_overlap * np.maximum(0, radii[i] + radii[j] - dist)
    return -(sum_radii + pen_bound + pen_overlap)

def repair(positions, radii):
    for i in range(len(radii)):
        x, y = positions[i]
        r = radii[i]
        min_r = min(x, 1 - x, y, 1 - y)
        if r > min_r:
            r = min_r
        positions[i] = [x, y]
        radii[i] = r
        for j in range(len(radii)):
            if i != j:
                dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    scale = (radii[i] + radii[j]) / dist
                    new_pos_j = positions[j] + (positions[j] - positions[i]) * (scale - 1)
                    r_j = (radii[j] / scale) * max(0, scale - 1)
                    positions[j] = new_pos_j
                    radii[j] = r_j
    return positions, radii
