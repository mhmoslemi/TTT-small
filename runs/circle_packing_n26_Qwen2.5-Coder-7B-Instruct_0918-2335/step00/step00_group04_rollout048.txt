import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def obj(v):
        r = v[52:]
        return -np.sum(r)

    def constraints(v):
        pos = v[:52].reshape(26, 2)
        r = v[52:]
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((26, 26), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0

    for _ in range(20):
        init = np.random.rand(78)
        init[:52] = init[:52].reshape(n, 2) * 0.99  # padding inside the unit square
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success:
            r = -res.fun
            if r > best_sum_radii:
                best_centers = res.x[:52].reshape(n, 2)
                best_radii = res.x[52:]
                best_sum_radii = r

    return best_centers, best_radii, best_sum_radii
