import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    delta_c = 1e-3
    N_refine = 200
    initial_radius = 0.01
    epsilon = 1e-9

    # Generate grid points
    x_grid = np.linspace(0, 1, R)
    y_grid = np.linspace(0, 1, R)
    grid = np.array([[x, y] for x in x_grid for y in y_grid])
    r_f = np.min(np.vstack((x_grid, y_grid, 1 - x_grid, 1 - y_grid)), axis=0) - epsilon

    centers = []
    radii = []

    for _ in range(26):
        k = np.argmax(r_f)
        center = grid[k]
        r = r_f[k]
        centers.append(center)
        radii.append(r)

        dx = x_grid - center[0]
        dy = y_grid - center[1]
        distances = np.sqrt(dx**2 + dy**2) - r
        r_f = np.minimum(r_f, distances)

    # Avoid using points that have negative radius (which should not happen)
    r_f[r_f < 0] = 0

    centers = np.array(centers)
    radii = np.array(radii)

    # Refinement phase
    for _ in range(N_refine):
        for i in range(len(radii)):
            center = centers[i]
            dx = np.random.uniform(-delta_c, delta_c, 2)
            new_center = center + dx
            a, b = new_center
            if not (0 <= a < 1 and 0 <= b < 1):
                continue
            candidate_radius = min(a, 1-a, b, 1-b)
            for j in range(len(radii)):
                if i == j:
                    continue
                d = np.linalg.norm(new_center - centers[j]) - radii[j]
                candidate_radius = min(candidate_radius, d)
            if candidate_radius > radii[i]:
                centers[i] = new_center
                radii[i] = candidate_radius

    for i in range(len(radii)):
        a, b = centers[i]
        r = radii[i]
        candidate_radius = min(a, 1-a, b, 1-b)
        for j in range(len(radii)):
            if i == j:
                continue
            d = np.linalg.norm(centers[i] - centers[j]) - radii[j]
            candidate_radius = min(candidate_radius, d)
        if candidate_radius > r:
            radii[i] = candidate_radius

    return centers, radii, np.sum(radii)
