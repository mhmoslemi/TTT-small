import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def power_diagram(centers, radii):
        n = centers.shape[0]
        power_diagram = [np.array([[0, 0], [1, 0], [1, 1], [0, 1]])] * n
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                center_diff = centers[j] - centers[i]
                dist_squared = np.sum(center_diff ** 2)
                rad_diff = radii[j] - radii[i]
                a = 2 * center_diff
                b = dist_squared - radii[j] ** 2 + rad_diff ** 2
                power_diagram[i] = clip_polygon(power_diagram[i], a, b)
        return power_diagram

    def clip_polygon(polygon, a, b):
        epsilon = 1e-12
        clipped = []
        n = len(polygon)
        for k in range(n):
            v_k, v_k1 = polygon[k], polygon[(k + 1) % n]
            test_k, test_k1 = a @ v_k <= b - epsilon, a @ v_k1 <= b - epsilon
            if test_k:
                clipped.append(v_k)
            if test_k != test_k1:
                intersection = v_k + (np.sum((v_k1 - v_k) * (b - a @ v_k) / (a @ (v_k1 - v_k))) * (v_k1 - v_k) / np.sum((v_k1 - v_k) ** 2))
                clipped.append(intersection)
        return np.array(clipped)

    def centroid(polygon):
        if polygon.shape[0] < 3:
            return np.mean(polygon, axis=0)
        x = np.sum(polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]) / (2 * np.sum(polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]))
        y = np.sum(polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]) / (2 * np.sum(polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]))
        return np.array([x, y])

    def distance_to_polygon(center, polygon):
        min_dist = np.inf
        n = polygon.shape[0]
        for k in range(n):
            v_k, v_k1 = polygon[k], polygon[(k + 1) % n]
            min_dist = min(min_dist, point_to_segment_distance(center, v_k, v_k1))
        return min_dist

    def point_to_segment_distance(point, v_k, v_k1):
        direction = v_k1 - v_k
        projection = np.dot(point - v_k, direction) / np.dot(direction, direction)
        if projection < 0:
            return np.linalg.norm(point - v_k)
        elif projection > 1:
            return np.linalg.norm(point - v_k1)
        else:
            projection_point = v_k + projection * direction
            return np.linalg.norm(point - projection_point)

    def clip_negative_radii(radii):
        return np.maximum(radii, 1e-8)

    radii = np.full(26, 0.01)
    tolerance = 1e-7
    max_iterations = 200
    mu = 0.5
    lambd = 0.8

        power_diagram = power_diagram(centers, radii)
        new_centers = np.array([centroid(cell) for cell in power_diagram])
        new_radii = np.array([distance_to_polygon(center, cell) for center, cell in zip(centers, power_diagram)])
        centers = (1 - mu) * centers + mu * new_centers
        radii = (1 - lambd) * radii + lambd * new_radii
        radii = clip_negative_radii(radii)
