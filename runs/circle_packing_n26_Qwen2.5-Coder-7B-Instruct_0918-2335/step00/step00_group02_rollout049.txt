```python
import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

# Define the objective function
def objective(x):
    n = x.shape[0]
    radii = np.sqrt(x[:-1])
    sum_radii = np.sum(radii)
    return 2.636 - sum_radii

# Define the constraint that ensures circles do not overlap
def constraint(x):
    n = x.shape[0]
    radii = np.sqrt(x[:-1])
    centers = x[-2*n:].reshape(n, 2)
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    return distances - (radii[:, np.newaxis] + radii[np.newaxis, :])

# Define the non-negativity constraint for radii
def non_negativity(x):
    return np.sqrt(x[:-1])

# Define the boundary constraint for centers
def boundary_constraint(x):
    n = x.shape[0]
    centers = x[-2*n:].reshape(n, 2)
    return np.maximum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                      np.maximum(np.minimum(centers[:, 1], 1 - centers[:, 1]), -1))

# Initial guess for the radii and centers
n = 26
x0 = np.zeros(2 * n + 1)
x0[:26] = 0.05

# Define the problem
con = [{'type': 'ineq', 'fun': constraint},
       {'type': 'ineq', 'fun': non_negativity},
       {'type': 'ineq', 'fun': boundary_constraint}]

# Define the bounds for the radii and centers
bounds = [(0, None)] + [(0, 1), (0, 1)] * n

# Run the optimization
res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=con)

# Extract the centers and radii
radii = np.sqrt(res.x[:-1])
centers = res.x[-2*n:].reshape(n, 2)

# Return the centers, radii, and sum of radii
return centers, radii, np.sum(radii)
```
