import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    N = R * R
    delta_c = 1e-3
    N_refine = 200
    
    grid = np.array([[x / R, y / R] for x in range(R) for y in range(R)])
    
    d_side = np.min(np.stack((grid[:, 0], grid[:, 1], 1 - grid[:, 0], 1 - grid[:, 1]), axis=1), axis=1)
    r_f = d_side.copy()
    
    centers = []
    radii = []
    
    for _ in range(26):
        k = np.argmax(r_f)
        new_center = grid[k]
        new_radius = r_f[k] - 1e-9
        centers.append(new_center)
        radii.append(new_radius)
        
        distances = np.linalg.norm(grid - new_center, axis=1) - new_radius
        r_f = np.minimum(r_f, distances)
        
        r_f[k] = -1e9
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    for _ in range(N_refine):
        i = np.random.randint(len(centers))
        current_center = centers[i]
        current_radius = radii[i]
        
        new_center = current_center + np.random.uniform(-delta_c, delta_c, 2)
        new_radius = np.min(np.stack((np.min(np.stack((new_center[0], new_center[1], 1 - new_center[0], 1 - new_center[1]), 
                                                     axis=0)), 
                                      np.min(np.linalg.norm(grid - current_center, axis=1) - current_radius)), 
                                       axis=0))
        
        if new_radius > current_radius:
            centers[i] = new_center
            radii[i] = new_radius
    
    radii = np.minimum(radii, np.min(np.stack((np.min(np.stack((centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]), 
                                                           axis=0)), 
                                            np.min(np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2) - radii[:, np.newaxis]), 
                                           axis=1), 
                                            axis=0))
    return centers, radii, np.sum(radii)
