import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place circles at grid points with decreasing radii
    grid = np.linspace(0.1, 0.9, int(np.sqrt(n)))
    i = 0
    for x in grid:
        for y in grid:
            if i < n:
                centers[i] = [x, y]
                radii[i] = np.random.uniform(0.01, 0.05)
                i += 1

    # Adjust positions and radii to maximize sum
    for _ in range(100):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    radii[i] -= (radii[i] + radii[j] - dist) / 2
                    radii[j] -= (radii[i] + radii[j] - dist) / 2
                    centers[i] += (dist - radii[i] - radii[j]) * (centers[i] - centers[j]) / (2 * dist)
                    centers[j] -= (dist - radii[i] - radii[j]) * (centers[i] - centers[j]) / (2 * dist)

    # Ensure circles are within the unit square
    centers = np.clip(centers, 0, 1)

    # Ensure sum of radii is positive
    radii = np.maximum(radii, 0)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
