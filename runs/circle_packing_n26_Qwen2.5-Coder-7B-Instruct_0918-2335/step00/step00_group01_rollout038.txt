import numpy as np
from scipy import optimize
from scipy.spatial.distance import pdist, squareform

def penalty_bound(x, r, bound):
    return r - np.minimum(x, 1-x, y, 1-y)

def penalty_overlap(c, r):
    dists = squareform(pdist(c))
    overlaps = np.maximum(dists - (r[:, None] + r), 0)
    return overlaps.flatten() ** 2

def objective(params, centers, m_bound, m_overlap):
    n = len(centers)
    nparams = 2 * n + n
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    sum_radii = np.sum(radii)
    bound_penalty = np.where(radii > penalty_bound(centers[:, 0], radii, bounds), m_bound * (radii - penalty_bound(centers[:, 0], radii, bounds)) ** 2, 0)
    overlap_penalty = np.where(penalty_overlap(centers, radii) > 0, m_overlap * penalty_overlap(centers, radii), 0)
    stack = np.concatenate((radii.flatten(), bound_penalty, overlap_penalty.flatten()))
    return sum_radii - np.sum(stack)

def gradient(params, centers, bounds, m_bound, m_overlap):
    n = len(centers)
    nparams = 2 * n + n
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Gradient for radii
    grad_radii = np.ones_like(radii)
    
    # Gradient for boundary penalty
    for i in range(n):
        if radii[i] > penalty_bound(centers[i, 0], radii[i], bounds[i]):
            grad_radii[i] += 2 * m_bound * (radii[i] - penalty_bound(centers[i, 0], radii[i], bounds[i]))
    
    # Gradient for overlap penalty
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                grad_radii[i] += 2 * m_overlap * (radii[i] + radii[j] - dists[i, j]) * (-1)
                grad_radii[j] += 2 * m_overlap * (radii[i] + radii[j] - dists[i, j]) * (-1)
    
    # Flatten gradients and concatenate
    grad = np.concatenate((grad_radii.flatten(), np.zeros(2*n), np.zeros(n)))
    return grad

def run_packing():
    n = 26
    bounds = np.empty(n)
    for i in range(n):
        x = np.linspace(0.1, 0.9, 6)[i % 5:i % 5 + 2]
        y = np.linspace(0.1, 0.9, 5) * 0.5 * (i // 5 % 2 * 2 - 1) + 0.5
        bounds[i] = np.min([np.min(x), np.max(x), np.min(y), np.max(y)]) - 1e-4
    
    start_pos = np.zeros(2 * n + n)
    for i in range(n):
        x = np.linspace(0.1, 0.9, 6)[i % 5:i % 5 + 2]
        y = np.linspace(0.1, 0.9, 5) * 0.5 * (i // 5 % 2 * 2 - 1) + 0.5
        start_pos[2 * i] = np.mean(x)
        start_pos[2 * i + 1] = np.mean(y)
        start_pos[2 * n + i] = min(bounds[i], 0.4)  # Start with a small radius feasible to clip later
    
    start_pos += np.random.randn(start_pos.size) * 1e-2
    
    m_bound = 1e2
    m_overlap = 1e2
    for gen in range(800):
        result = optimize.minimize(
            fun=lambda p: objective(p, start_pos[:2 * n].reshape(n, 2), bounds, m_bound, m_overlap),
            x0=start_pos,
            method="SLSQP",
            jac=lambda p: gradient(p, start_pos[:2 * n].reshape(n, 2), bounds, m_bound, m_overlap),
            bounds=[(0, b) for b in bounds] + [(0, 1) for _ in range(2 * n)],
            constraints=[{'type': 'ineq', 'fun': lambda p: p[i + 2 * n] - p[2 * m + n]} for m in range(n)]
        )
        
        start_pos = result.x
        
        if gen % 20 == 0:
            m_bound *= 3
            m_overlap *= 3
    
    return start_pos[:2 * n].reshape(n, 2), start_pos[2 * n:], np.sum(start_pos[2 * n:])


centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of radii: {sum_radii}")
