import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog
from scipy.stats import norm

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def solve_lp(centers):
    n = centers.shape[0]
    A = np.zeros((n * n + n * 4, n))
    b = np.zeros(n * n + n * 4)

    for i in range(n):
        row = i * n
        A[row] = linprog.stage.A_nx1_t + linprog.stage.A_nxn_t[i]
        b[row] = linprog.stage.b_nx1_t + linprog.stage.b_nxn_t[i]

    for i in range(n):
        row = n * n + i * 4
        A[row, i] += 1
        b[row] += linprog.stage.b_i_t[i]

    for i in range(n):
        row = n * n + i * 4 + 1
        A[row, i] += 1
        b[row] += 1 - centers[i, 0]

    for i in range(n):
        row = n * n + i * 4 + 2
        A[row, i] += 1
        b[row] += centers[i, 0]

    for i in range(n):
        row = n * n + i * 4 + 3
        A[row, i] += 1
        b[row] += 1 - centers[i, 1]

    for i in range(n):
        row = n * n + i * 4 + 4
        A[row, i] += 1
        b[row] += centers[i, 1]

    res = linprog(A, b, method='highs')
    if not res.success:
        raise ValueError("LP solver failed")

    radii = res.x
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    max_iter = 400
    tau0 = 1e-3
    gamma = 0.95

    # Step A: Coarse dense grid
    grid_size = 6
    x_vals = (np.arange(grid_size) + 0.5) / grid_size
    y_vals = (np.arange(grid_size) + 0.5) / grid_size
    centers = np.array([(x, y) for y in y_vals for x in x_vals[:grid_size]])
    assert centers.shape == (36, 2)

    # Step B: Reduce to 26
    distances = distance.cdist(centers[:26], centers[:26])
    distances = np.triu(distances, k=1)
    removals = np.argsort(distances.ravel())[::-1][:10]
    centers = np.delete(centers, removals, axis=0)
    assert centers.shape == (26, 2)

    # Step C: Random perturbation
    noise = 0.01 * norm.rvs(size=centers.shape)
    centers += noise
    centers = np.clip(centers, 0, 1)
    assert centers.shape == (26, 2)

    best_centers = centers.copy()
    best_r = np.zeros(n)
    best_score = 0

    for seed in range(5):
        np.random.seed(seed)
        centers = centers.copy()

        for k in range(max_iter):
            r = solve_lp(centers)

            forces = np.zeros(centers.shape)
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    s_ij = r[i] + r[j] - dist
                    if s_ij > 0:
                        force = (centers[i] - centers[j]) / dist * s_ij
                        forces[i] += force
                        forces[j] -= force
            for i in range(n):
                wall_width = np.minimum(r[i], np.minimum(1 - centers[i, 0], np.minimum(centers[i, 0], np.minimum(1 - centers[i, 1], centers[i, 1]))))
                wall_force = np.maximum(np.zeros(2), wall_width - r[i])
                forces[i] += wall_force

            tau = tau0 * gamma ** k
            centers += tau * forces
            centers = np.clip(centers, 0, 1)

        _, r, sum_r = validate_packing(centers, r)
        if sum_r > best_score:
            best_score = sum_r
            best_centers = centers.copy()

    assert best_centers.shape == (26, 2)
    best_r = solve_lp(best_centers)
    assert best_r.shape == (26,)
    
    return best_centers, best_r, best_score
