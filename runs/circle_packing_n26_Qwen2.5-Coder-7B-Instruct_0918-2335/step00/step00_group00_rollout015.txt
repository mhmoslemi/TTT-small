```python
import numpy as np
from scipy.optimize import NonlinearConstraint, minimize

n = 26

# Define pair indices (i,j) for all unique pairs (i<j)
pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]

# Define Nonlinear constraint
def constraint_func(v):
    x = v[:52].reshape((n, 2))
    r = v[52:]
    arr = []
    # Boundary constraints
    arr.append(x[:, 0] - r)
    arr.append(1 - x[:, 0] - r)
    arr.append(x[:, 1] - r)
    arr.append(1 - x[:, 1] - r)
    # Pairwise constraints
    for i, j in pair_indices:
        dx = x[i, 0] - x[j, 0]
        dy = x[i, 1] - x[j, 1]
        dist2 = dx ** 2 + dy ** 2
        arr.append(dist2 - (r[i] + r[j]) ** 2)
    return np.concatenate(arr)

# Initial position and radius of each circle
x0 = np.empty(78)
positions = np.array([(j / 5.0, i / 5.0) for i in range(5) for j in range(5) if len(x0) < 50])
positions = np.vstack([positions, [0.6, 0.9]])
x0[:52] = positions.ravel()
x0[52:] = 0.04 * np.ones(n)

# Bounds for each variable
bounds = [(0.0, 1.0), (0.0, 1.0)] * n + [(0.0, 0.5)] * n  # x,y bounds, then r bounds

# Non-Linear Constraint object as per NonlinearConstraint method provided by scipy
cons = NonlinearConstraint(constraint_func, 0, np.inf)

# Minimize function which is negative of sum of radii
def fun(v):
    return -np.sum(v[52:])

# Call minimize
res = minimize(fun, x0, bounds=bounds, constraints=cons, method='trust-constr', options={'maxiter': 2000, 'ftol': 1e-12, 'xtol': 1e-12})

# Extract results
centers = res.x[:52].reshape((n, 2))
radii = res.x[52:]
sum_radii = float(np.sum(radii))

# If optimizer fails, reduce radii proportionally
if not res.success:
    shrink_factor = 1 - 1e-12
    radii *= shrink_factor
    sum_radii *= shrink_factor
    centers = np.clip(centers, shrink_factor, 1 - shrink_factor)

return centers, radii, sum_radii
```
