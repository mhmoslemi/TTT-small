```python
import numpy as np
from scipy.optimize import minimize

n = 26

# Generate initial positions: 5x5 grid for 25 + one extra
positions = []
for i in range(5):
    for j in range(5):
        positions.append(((j + 0.5) / 5.0, (i + 0.5) / 5.0))
        if len(positions) == 25:
            break
    if len(positions) == 25:
        break
# Add extra point at (0.6,0.9)
positions.append((0.6, 0.9))
positions = np.array(positions)  # shape (26,2)

radii_initial = np.full(n, 0.04)

x0 = np.empty(78)
x0[:52] = positions.ravel()
x0[52:] = radii_initial

# Bounds: create bounds list
bounds = []
for i in range(26):
    bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x,y
for i in range(26):
    bounds.append((0.0, 0.5))

# Pair indices
pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]

# Define constraint function using vector
def constraint_func(v):
    x = v[:52].reshape((n, 2))
    r = v[52:]
    arr = []
    # Boundary constraints
    arr.append(x[:, 0] - r)  # >= 0
    arr.append(1 - x[:, 0] - r)  # >= 0
    arr.append(x[:, 1] - r)  # >= 0
    arr.append(1 - x[:, 1] - r)  # >= 0
    # Pairwise
    # Compute squared distances:
    i_indices = np.array([i for i, j in pair_indices], dtype=np.int64)
    j_indices = np.array([j for i, j in pair_indices], dtype=np.int64)
    dx = x[i_indices, 0] - x[j_indices, 0]
    dy = x[i_indices, 1] - x[j_indices, 1]
    dist2 = dx * dx + dy * dy
    arr.append(dist2 - (r[i_indices] + r[j_indices]) ** 2)
    return np.concatenate(arr)

# Use minimize
res = minimize(fun=lambda v: -np.sum(v[52:]), x0=x0, bounds=bounds,
               constraints={'type': 'ineq', 'fun': constraint_func},
               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})

centers = res.x[:52].reshape((n, 2))
radii = res.x[52:]

# Sum of radii
sum_radii = float(np.sum(radii))

# Post-processing to ensure no constraints are violated
epsilon = 1e-9
for _ in range(10):
    violated_constraints = np.where(constraint_func(res.x) < 0)[0]
    if len(violated_constraints) == 0:
        break
    shrink_factor = 1 - epsilon * np.min((constraint_func(res.x) / (-constraint_func(res.x)))[violated_constraints])
    res.x[res.x > 0] *= shrink_factor

return centers, radii, sum_radii
```
