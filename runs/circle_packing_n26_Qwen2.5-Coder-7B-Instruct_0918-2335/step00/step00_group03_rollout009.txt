import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.01)

    def power_diagram(centers, radii):
        def power_distance(center, radius, point):
            return np.sum((point - center) ** 2) - radius ** 2

        def clipped_polygon(vertices, half_plane):
            a, b = half_plane
            clipped = []
            for i, (x, y) in enumerate(vertices):
                j = (i + 1) % len(vertices)
                xi, yi = vertices[i]
                xj, yj = vertices[j]
                det = a * (xj - xi) - b * (yj - yi)
                va = 2 * (a * xi - b * yi) + b * (xj - xi) - a * (yj - yi)
                vb = 2 * (a * xj - b * yj) - b * (xi - xj) + a * (yi - yj)
                if det == 0:
                    p = None if va * vb <= 0 else ((a * b * (xj - xi) - a * (yj - yi) * (x - xi) - b * (xj - xi) * (y - yi)) / (a * a + b * b), (b * (xj - xi) - a * (yj - yi) * (y - yi) + b * (x - xi) * (xj - xi)) / (a * a + b * b))
                else:
                    p = ((va * xj + vb * xi - det * x) / det, (va * yj + vb * yi - det * y) / det)
                if va > 0 and vb > 0 or va == 0 and vb >= 0 or vb == 0 and va >= 0 or p is None:
                    clipped.append((x, y))
                elif p is not None:
                    clipped.append(p)
            return clip_convex_polygon(clipped)

        def clip_convex_polygon(vertices):
            def half_plane_from_edge(p1, p2):
                a = p2[1] - p1[1]
                b = p1[0] - p2[0]
                c = p1[1] * p2[0] - p1[0] * p2[1]
                return a, c

            vertices = np.array(vertices)
            if len(vertices) < 3:
                return []
            polygon = vertices
            for i, (x, y) in enumerate(vertices):
                j = (i + 1) % len(vertices)
                u, v = half_plane_from_edge((x, y), (vertices[j][0], vertices[j][1]))
                polygon = clipped_polygon(polygon, (u, v))
            return polygon

        diagrams = []
        for center, radius in zip(centers, radii):
            half_planes = [(2 * (centers[j] - center) @ (np.array([x, y]) - center), np.sum((center - centers[j]) ** 2) - radius ** 2 - np.sum((centers[j] - centers[j]) ** 2) + np.sum((centers[j] - centers[j]) ** 2)) for j in range(n) if j != np.array(range(n)).tolist().index(center)]
            diagram = np.array([point for po in half_planes for point in clipped_polygon([center], po)])
            diagram = diagram[np.isin(np.unique(diagram.astype(int)), range(n))].reshape(-1, 2)
            diagrams.append(diagram)
        return diagrams

    def polygon_centroid(vertices):
        n = len(vertices)
        area = 0.5 * np.sum(vertices[:-1, 0] * vertices[1:, 1] - vertices[1:, 0] * vertices[:-1, 1])
        cx = 1 / (6 * area) * np.sum((vertices[:-1, 0] + vertices[1:, 0]) * (vertices[:-1, 0] * vertices[1:, 1] - vertices[1:, 0] * vertices[:-1, 1]))
        cy = 1 / (6 * area) * np.sum((vertices[:-1, 1] + vertices[1:, 1]) * (vertices[:-1, 0] * vertices[1:, 1] - vertices[1:, 0] * vertices[:-1, 1]))
        return np.array([cx, cy])

    def point_to_segment_distance(p, v1, v2):
        u = (p - v1) / np.linalg.norm(v2 - v1)
        d = np.linalg.norm(v1 - p) - np.abs(u[0] * v1[1] - u[1] * v1[0])
        return d

    def polygon_distance(center, radius, vertices):
        center = np.array(center)
        distance = float('inf')
        for i, (x, y) in enumerate(vertices):
            j = (i + 1) % len(vertices)
            distance = min(distance, point_to_segment_distance(center, np.array([x, y]), np.array([vertices[j][0], vertices[j][1]])))
        return distance - radius

    mu = 0.5
    lambd = 0.8
    tolerance = 1e-7
    max_iterations = 200
    for _ in range(max_iterations):
        prev_sum_radii = radii.sum()
        diagrams = power_diagram(centers, radii)
        new_centers = np.array([polygon_centroid(diagrams[i]) for i in range(n)])
        new_radii = np.array([min(max(0, polygon_distance(new_centers[i], 0, diagrams[i])), 1) for i in range(n)])
        centers = mu * new_centers + (1 - mu) * centers
        radii = lambd * new_radii + (1 - lambd) * radii
        if np.abs(prev_sum_radii - radii.sum()) < tolerance:
            break
    return centers, radii, radii.sum()
