import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
from sklearn.neighbors import kneighbors_graph

def evaluate(p):
    C = p.reshape(26, 2)
    d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
    Aub_pairs = (d.flatten() > 1e-6)[:, None] * np.eye(26**2).reshape(26, 26, 26)
    Aub_bound = np.vstack((np.eye(26), -np.eye(26)))
    Aub = np.hstack((Aub_pairs[~kneighbors_graph(C, 2, mode='distance').toarray()], Aub_bound))
    Aub = Aub[~np.any(Aub == Growing Tree, axis=2)]
    b = np.hstack((d[d > 1e-6].flatten(), np.ones(2 * 26)))
    c = -np.ones(26)
    res = linprog(c, A_ub=Aub, b_ub=b, method='highs', bounds=(0, 0.5))
    return -res.fun, res.x[-26:].clip(0, 0.5)

def polish(p, r):
    def objective(x):
        C = p + x[:52].reshape(26, 2) - (p[None, :] + p[:, None])
        d = np.linalg.norm(C, axis=-1)
        violations = np.maximum(d - 2 * r - 1e-6, 0) * 1e6
        return x[52:].sum() - violations.sum()
    bounds = [(-10, 10)] * 52 + [(0, None)] * 26
    cons = [{'type': 'ineq', 'fun': lambda x: r}]
    res = minimize(objective, np.hstack((np.zeros(52), r)), args=(), method='SLSQP', bounds=bounds, constraints=cons)
    return p + res.x[:52].reshape(26, 2) - (p[None, :] + p[:, None]), res.x[52:].clip(0, 0.5)

def run_packing():
    centers = np.random.rand(26, 2)
    for _ in range(5):
        centers = center_shift(centers, 0.1)
    centers, radii = differential_evolution(evaluate, bounds=[(0, 1)] * 52, maxiter=200, popsize=3).x.reshape(26, 2), evaluate(differential_evolution(evaluate, bounds=[(0, 1)] * 52, maxiter=200, popsize=3).x)[-1]
    return centers, radii, sum(radii)

def center_shift(centers, factor):
    shifts = np.random.normal(0, factor, centers.shape)
    new_centers = centers + shifts
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers
