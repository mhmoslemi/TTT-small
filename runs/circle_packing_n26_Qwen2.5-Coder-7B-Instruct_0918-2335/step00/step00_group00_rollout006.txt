import numpy as np

def init_circles(variant):
    centers = np.zeros((26, 2))
    if variant == 'A':
        centers[:25] = np.meshgrid(np.arange(0.1, 0.9, 0.2), np.arange(0.1, 0.9, 0.2)).reshape(-1, 2)
        centers[25] = np.array([0.6, 0.9])
    else:
        centers[:25] = np.meshgrid(np.arange(0.1, 0.9, 0.2), np.arange(0.1, 0.9, 0.2)).reshape(-1, 2)
        centers[25] = np.array([0.4, 0.9])
    radii = 0.01 * np.ones(26)
    return centers, radii

def inflate(circles, Δr, ε):
    while True:
        changed = False
        for i in range(25):
            for j in range(i + 1, 26):
                dx, dy = circles[i, 0] - circles[j, 0], circles[i, 1] - circles[j, 1]
                d = np.sqrt(dx ** 2 + dy ** 2)
                if d < circles[i, 2] + circles[j, 2] - ε:
                    δ = (circles[i, 2] + circles[j, 2] - d) / (2 * d)
                    circles[i, 0] += δ * dx
                    circles[j, 0] -= δ * dx
                    circles[i, 1] += δ * dy
                    circles[j, 1] -= δ * dy
                    changed = True
        
        circles[:, 0] = np.clip(circles[:, 0], circles[:, 2], 1 - circles[:, 2])
        circles[:, 1] = np.clip(circles[:, 1], circles[:, 2], 1 - circles[:, 2])
        
        circles[:, 2] += Δr * np.ones(26)
        circles[:, 2] = np.minimum(circles[:, 2], np.minimum(circles[:, 0] - circles[:, 2], circles[:, 1] - circles[:, 2], (1 - circles[:, 0]) - circles[:, 2], (1 - circles[:, 1]) - circles[:, 2]) - ε)
        
        if not changed:
            break
        
        # Final mass-shrink step if necessary
        for i in range(25):
            for j in range(i + 1, 26):
                dx, dy = circles[i, 0] - circles[j, 0], circles[i, 1] - circles[j, 1]
                d = np.sqrt(dx ** 2 + dy ** 2)
                if d < circles[i, 2] + circles[j, 2]:
                    circles[i, 2] -= 0.5 * (circles[i, 2] + circles[j, 2] - d)
                    circles[j, 2] -= 0.5 * (circles[i, 2] + circles[j, 2] - d)
                    
    return circles[:, 2], circles[:, :2]

def run_packing():
    best_sum = 0
    best_centers, best_radii = None, None
    
    np.random.seed(42)
    
    for variant in ['A', 'B']:
        centers, radii = init_circles(variant)
        radii, centers = inflate(centers, Δr=0.0005, ε=1e-6)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers, best_radii = centers, radii
    
    if best_sum < 2.636:
        # Fallback to equal radii arrangement
        centers = np.array([[0.5 + 0.095 * np.cos(2 * np.pi * i / 26), 0.5 + 0.095 * np.sin(2 * np.pi * i / 26)] for i in range(26)])
        radii = np.ones(26) * 0.095
        best_sum = np.sum(radii)
        best_centers, best_radii = centers, radii
    
    return best_centers, best_radii, best_sum
