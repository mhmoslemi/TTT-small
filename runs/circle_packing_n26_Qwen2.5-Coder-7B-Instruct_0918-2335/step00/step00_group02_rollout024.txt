import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def objective(z):
    return -np.sum(z[2::3])  # We negate because we minimize in solver

def constraint_boundary(z):
    idxs = np.arange(0, len(z), 3)
    return np.minimum(np.sum(z[idxs+1:idxs+3], axis=1), 1 - np.sum(z[idxs+1:idxs+3], axis=1))

def constraint_non_overlap(z):
    idxs = np.arange(0, len(z), 3)
    dists = squareform(pdist(np.column_stack((z[idxs+1:idxs+3], z[idxs+2:idxs+5])).reshape(-1, 2)))
    radii = z[idxs+3]
    dists = dists - np.add.outer(radii, radii)
    np.triu(dists, k=1, out=dists)
    return np.prod(dists, axis=0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    init_positions = np.random.rand(n_circles, 2)
    init_radii = np.minimum(init_positions[:, 0], 1 - init_positions[:, 0], init_positions[:, 1], 1 - init_positions[:, 1]) / 2
    init_radii *= np.random.uniform(0.3, 0.8, n_circles)

    bounds = [(0, 1)] * (n_circles * 3)
    constraints = [
        {'type': 'ineq', 'fun': lambda z: 0.5 * np.ones(n_circles * 3) - z},
        {'type': 'ineq', 'fun': lambda z: z.reshape(-1, 3)[:, 0] - z.reshape(-1, 3)[:, 2]},  # left side
        {'type': 'ineq', 'fun': lambda z: 1 - z.reshape(-1, 3)[:, 0] - z.reshape(-1, 3)[:, 2]},  # right side
        {'type': 'ineq', 'fun': lambda z: z.reshape(-1, 3)[:, 1] - z.reshape(-1, 3)[:, 2]},  # bottom side
        {'type': 'ineq', 'fun': lambda z: 1 - z.reshape(-1, 3)[:, 1] - z.reshape(-1, 3)[:, 2]},  # top side
        {'type': 'ineq', 'fun': constraint_non_overlap}
    ]
    initial_guess = np.hstack((init_positions.flatten(), init_radii))
    
    results = []
    for _ in range(6):
        res = minimize(objective, initial_guess, method='trust-constr', bounds=bounds, constraints=constraints, jac='2-point')
        results.append(res.x)
        initial_guess = np.hstack((np.random.rand(n_circles, 2), np.minimum(initial_guess[n_circles*3:], 1 - initial_guess[n_circles*3:]/2 * 0.1)))
    
    best_result = min(results, key=lambda x: -np.sum(x[2::3]))
    centers = best_result[0::3].reshape(-1, 2)
    radii = best_result[2::3]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
