import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize

def eval_placement(p):
    C = p.reshape((26, 2))
    d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
    r = np.zeros((26, 26))
    for i in range(26):
        for j in range(i + 1, 26):
            r[i, j] = -min(d[i, j], C[i, 0] + C[j, 0], 1 - C[i, 0] - C[j, 0], C[i, 1] + C[j, 1], 1 - C[i, 1] - C[j, 1]) / 2
    r = r + r.T + np.diagonal(np.full(26, 1.0))
    res = linprog(c=-np.sum(r, axis=1), A_ub=r, b_ub=np.sum(d, axis=1), bounds=[(0, None) for _ in range(26)], method='highs')
    return -res.fun, res.x
