```python
import numpy as np
import scipy.optimize

def penalty(z):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    overlap = 0
    for i in range(n):
        overlap += np.maximum(-x[i] + r[i], 0)**2
        overlap += np.maximum(x[i] - 1 + r[i], 0)**2
        overlap += np.maximum(-y[i] + r[i], 0)**2
        overlap += np.maximum(y[i] - 1 + r[i], 0)**2
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx**2 + dy**2)
            overlap += np.maximum(dist - (r[i] + r[j]), 0)**2
    return -np.sum(r) + 10000 * overlap

bounds = [(0, 1)] * 78
result = scipy.optimize.differential_evolution(
    penalty, bounds, strategy="rand/1/bin", mutation=0.6,
    recombination=0.7, tol=1e-8, maxiter=2000, popsize=20)
z = result.x

def local_constraints(z):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: x[i] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - x[i] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: y[i] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - y[i] - r[i]})
        for j in range(i+1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: dist - (r[i] + r[j])})
    return constraints

if result.fun > 1e-8:
    result = scipy.optimize.minimize(
        penalty, z, method='SLSQP', bounds=bounds, constraints=local_constraints(z)
    )

centers = result.x[:76].reshape((26, 2))
radii = result.x[76:]
sum_radii = np.sum(radii)
return centers, radii, sum_radii
```
