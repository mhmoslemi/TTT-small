import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    ε = 1.0e-12
    Δt = 1.0e-3
    α = 4.0
    β = 0.8
    δr = 1.0e-4
    maxIter = 3000
    tol = 1.0e-10
    
    def init_circle():
        return (np.random.rand(2, n) * (1 - 2 * ε) + ε, np.random.rand(2, n) * (1 - 2 * ε) + ε, np.full(n, 1.0e-6))
    
    def compute_dynamics(T, r_old):
        c_old = T[0:n, :]
        v = np.zeros((2, n))
        F_ij = np.zeros((2, n, n))
        F_x = np.zeros(2*n)
        F_y = np.zeros(2*n)
        
        for i in range(n):
            for j in range(i):
                d = np.linalg.norm(c_old[:, i] - c_old[:, j])
                if d < r_old[i] + r_old[j] + ε:
                    δ = (r_old[i] + r_old[j] - d) / d
                    F_ij[:, i, j] = -δ * (c_old[:, i] - c_old[:, j])
                    F_ij[:, j, i] = -F_ij[:, i, j]
                F_x[2*i] += F_ij[0, i, j]
                F_x[2*j] += F_ij[0, j, i]
                F_y[2*i] += F_ij[1, i, j]
                F_y[2*j] += F_ij[1, j, i]
        
        F_left = np.where(c_old[0] < r_old + ε, α * (r_old + ε - c_old[0]), 0)
        F_right = np.where(c_old[0] > 1 - r_old - ε, α * (c_old[0] - (1 - r_old - ε)), 0)
        F_bottom = np.where(c_old[1] < r_old + ε, α * (r_old + ε - c_old[1]), 0)
        F_top = np.where(c_old[1] > 1 - r_old - ε, α * (c_old[1] - (1 - r_old - ε)), 0)
        
        v[:, :] += Δt * (np.vstack((F_x, F_y)) / 2 + β * v)  # 2* because each force is applied at both ends
        c_old[:, :] += Δt * v
        r_old += δr
        
        r_new = np.minimum(np.minimum(np.minimum(np.minimum(np.minimum(np.minimum(1 - c_old[0, :], c_old[0, :] - ε), r_old), np.minimum(1 - c_old[1, :], c_old[1, :] - ε)), r_old), np.minimum(np.linalg.norm(c_old - c_old[:, np.newaxis], axis=0) - r_old[:, np.newaxis], r_old))
        r_old = r_new  # min constraint for r > ε
        
        ΔR = np.max(np.abs(r_old - r_old[0]))
        return ΔR, r_old, c_old
    
    def refine_radii(positions, r):
        A = np.zeros((n + 2 * n * n, n))
        b = np.zeros(n + 2 * n * n)
        A[:n, np.arange(n)] = 1
        for i in range(n):
            c1 = np.array([positions[0, i], positions[1, i]])
            for j in range(n):
                r1 = r[i]
                if i != j:
                    c2 = np.array([positions[0, j], positions[1, j]])
                    r2 = r[j]
                    d = np.linalg.norm(c1 - c2)
                    if d > r1 + r2 - ε:
                        A[n + 2 * i * n + j, i] = 1
                        A[n + 2 * i * n + j, j] = -1
                        b[n + 2 * i * n + j] = -d + r1 + r2
                    
                    A[n + 2 * i * n + n + j, i] = 1
                    A[n + 2 * i * n + n + j, j] = -1
                    b[n + 2 * i * n + n + j] = -1
        
        result = linprog(-r, A_ub=A, b_ub=b, method='highs')
        if result.success:
            r_opt = result.x
            return r_opt, result.success
        else:
            return r, result.success
    
    def simulated_annealing(x, s):
        σ = 1.0
        for step in range(s):
            t = T[int(np.random.rand() * s)]
            x_new = x.copy()
            for j in range(len(x)):
                x_new[j] += np.random.normal(0, σ)
            s_new, _ = refine_radii(x_new, s)
            Δ = -26 + np.sum(s_new)
            if Δ > 0 or np.random.rand() < np.exp(-Δ / 0.01):
                x = x_new
        return x, True
    
    r_old, c_old, r_old = init_circle()
    
    for step in range(maxIter):
        ΔR, r_old, c_old = compute_dynamics((c_old, c_old), r_old)
        if ΔR < tol:
            break
    
    r_opt, is_valid = refine_radii(c_old, r_old)
    
    if is_valid:
        r_opt = np.maximum(r_opt, ε)
        return c_old, r_opt, np.sum(r_opt)
    else:
        return np.zeros((n, 2)), np.zeros(n), 0.0
