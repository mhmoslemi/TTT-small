import numpy as np
from scipy.spatial distance import cdist
from scipy.optimize import minimize

def fitness_function(params, n, m, penalty_bound, penalty_overlap, max_iter=1000):
    centers = params[:2*n].reshape(n, -1)
    radii = params[2*n:]
    overlap_penalty = 0.0
    bound_penalty = 0.0
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            if d_ij < radii[i] + radii[j]:
                overlap_penalty += (radii[i] + radii[j] - d_ij) ** 2
    
    for i in range(n):
        if radii[i] > m[i]:
            bound_penalty += ((radii[i] - m[i]) ** 2 * penalty_bound)
    
    return -(np.sum(radii) + overlap_penalty + bound_penalty)

def repair(params, n, m):
    centers = params[:2*n].reshape(n, -1)
    radii = params[2*n:]
    for i in range(n):
        if radii[i] > m[i]:
            radii[i] = m[i]
            
        x, y = centers[i]
        min_radius = min(x, 1-x, y, 1-y)
        if radii[i] > min_radius:
            radii[i] = min_radius
    
    return np.concatenate((centersflatten(), radii))

    n = 26
    m = np.array([min(0.5, np.random.uniform(0.1, 0.4, n)) for _ in range(20, 81)])

    radii = m - 1e-4 + np.random.normal(0, 1e-4, n)
    params = np.concatenate((centersflatten(), radii))
    
    penalty_bound = 10**2
    penalty_overlap = 10**2
    bounds = [(0, 1) for _ in range(2*n)]
    bounds += [(0, None) for _ in range(n)]
    
    for _ in range(200):
        result = minimize(fitness_function, params, method='SLSQP', bounds=bounds, args=(n, m, penalty_bound, penalty_overlap))
        if result.success:

        penalty_bound *= 3
