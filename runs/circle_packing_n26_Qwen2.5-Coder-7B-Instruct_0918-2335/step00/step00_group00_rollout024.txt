import numpy as np

def initial_packing():
    # Start with a simple random packing in the square
    num_circles = 26
    centers = np.random.rand(num_circles, 2)
    radii = np.random.rand(num_circles)
    for i in range(num_circles):
        centers[i] = scale_to_bounds(centers[i], radii[i])
    return centers, radii

def scale_to_bounds(center, radius):
    # Scale the center and clip to ensure the circle is within bounds
    scale_factor = 1 - radius
    scaled_center = center * scale_factor
    return np.clip(scaled_center, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial packing
    centers, radii = initial_packing()
    sum_radii = np.sum(radii)
    
    # Simple refinement: increase radii of circles that are closer to the boundaries
    buffer = 0.05
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x < r + buffer or x > 1 - r - buffer:
            radii[i] += buffer
        if y < r + buffer or y > 1 - r - buffer:
            radii[i] += buffer
    
    # Ensure non-overlap and within bounds
    for i in range(len(centers)):
        centers[i] = scale_to_bounds(centers[i], radii[i])
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
