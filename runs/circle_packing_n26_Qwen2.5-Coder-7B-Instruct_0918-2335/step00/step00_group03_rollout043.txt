import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # predefined hexagonal lattice centers
    centers = np.array([
        (0.5, 0.25), (0.25, 0.5), (0.75, 0.5), (0.375, 0.75), (0.625, 0.75),
        (0.125, 0.875), (0.375, 0.875), (0.625, 0.875), (0.875, 0.875),
        (0.5, 0.75), (0.25, 0.625), (0.75, 0.625), (0.375, 0.5), (0.625, 0.5),
        (0.125, 0.375), (0.375, 0.375), (0.625, 0.375), (0.875, 0.375),
        (0.5, 0.25), (0.25, 0.125), (0.75, 0.125), (0.375, 0.375), (0.625, 0.375),
        (0.125, 0.625), (0.375, 0.625), (0.625, 0.625), (0.875, 0.625)
    ])

    # reshape to remove duplicates
    centers = centers[:26]

    # bounds: [0, 1] for each of the 52 variables
    bounds = [(0, 1)] * 52

    # objective function wrapper
    def eval_placement(p):
        C = p.reshape(26, 2)
        d = np.linalg.norm(C[:,None,:]-C[None,:,:], axis=-1)
        A = np.zeros((52, 26))
        b = np.zeros(52)
        for i in range(26):
            A[i, 2*i] = 1
            A[i, 2*i+1] = 1
        b[:26] = centers[:,0]
        b[26:] = centers[:,1]
        upper = np.ones(26)
        lower = 0
        inequality_matrix = np.zeros((326, 26))
        inequality_matrix[:26, np.arange(26)] = 1
        inequality_matrix[26:52, np.arange(26)] = 1
        inequality_matrix[52:78, np.arange(26)] = 1
        inequality_matrix[78:104, np.arange(26)] = 1
        inequality_matrix[104:326, np.arange(26)] = 1
        inequality_right_hand_side = np.array([centers[:,0], centers[:,0], 
                                             centers[:,1], centers[:,1], centers[:,0], 
                                             centers[:,0], centers[:,1], centers[:,1]]).flatten()
        constraint_matrix = np.vstack((A, inequality_matrix))
        constraint_right_hand_side = np.hstack((b, inequality_right_hand_side))
        res = linprog(c=-np.ones(26), A_ub=constraint_matrix, b_ub=constraint_right_hand_side, method='highs')
        return -res.fun

    # global search with DE
    res = differential_evolution(eval_placement, bounds, maxiter=200, popsize=3, tol=1e-6, mutation=(0.3,0.8), recombination=0.7)

    # polishing with SLSQP
    def polishing_objective(x):
        p = x[:52]
        C = p.reshape(26, 2)
        d = np.linalg.norm(C[:,None,:]-C[None,:,:], axis=-1)
        inequality_matrix = np.zeros((326, 26))
        inequality_matrix[:26, np.arange(26)] = 1
        inequality_matrix[26:52, np.arange(26)] = -1
        inequality_matrix[52:78, np.arange(26)] = 1
        inequality_matrix[78:104, np.arange(26)] = -1
        inequality_matrix[104:326, np.arange(26)] = 1
        inequality_right_hand_side = np.array([centers[:,0] - 1e-6, centers[:,0] + 1e-6, 
                                             centers[:,1] - 1e-6, centers[:,1] + 1e-6, centers[:,0] - 1e-6, 
                                             centers[:,0] + 1e-6, centers[:,1] - 1e-6, centers[:,1] + 1e-6]).flatten()
        A, b = np.vstack((inequality_matrix, -np.eye(26))), -inequality_right_hand_side
        res = linprog(c=-np.ones(26), A_ub=A, b_ub=b, method='highs')
        radii = -res.fun
        penalty = np.maximum(-radii + 0.001, 0).sum()
        return -np.sum(x[52:52+26]) + penalty

    x0 = np.concatenate((res.x, np.zeros(26)))
    res_polish = minimize(polishing_objective, x0, method='SLSQP', bounds=[(0, 1) + (None,)]*52 + [(None,)]*26, ftol=1e-9, maxiter=2000)

    centers = res_polish.x[:52].reshape(26, 2)
    radii = -res_polish.x[52:].clip(0, 0.5)

    return centers, radii, np.sum(radii)
