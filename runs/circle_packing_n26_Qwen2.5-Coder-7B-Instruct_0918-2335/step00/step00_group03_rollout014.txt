import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def evaluate(positions):
    n = positions.shape[0]
    centers = positions[:n].reshape(n, 2)
    radii = positions[n:].reshape(n, 1)
    
    # Pairwise distance matrix
    distances = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=-1)
    
    # Inequalities matrix A and right-hand side b
    A = np.zeros((4 * n + n * n, n))
    b = np.zeros(4 * n + n * n)
    
    r_indices = np.arange(n)
    
    # Boundary constraints
    for i in r_indices:
        A[i, i] = 1
        b[i] = radii[i]
    
    # Left wall constraints
    for i in r_indices:
        A[2*1*n + 4*i, i] = 1
        b[2*1*n + 4*i] = centers[i, 0]
    
    # Right wall constraints
    for i in r_indices:
        A[2*2*n + 4*i, i] = 1
        b[2*2*n + 4*i] = 1 - centers[i, 0]
    
    # Bottom wall constraints
    for i in r_indices:
        A[2*3*n + 4*i, i] = 1
        b[2*3*n + 4*i] = centers[i, 1]
    
    # Top wall constraints
    for i in r_indices:
        A[2*4*n + 4*i, i] = 1
        b[2*4*n + 4*i] = 1 - centers[i, 1]
    
    # Non-negativity constraints
    for i in r_indices:
        A[4*n + i, r_indices] = np.eye(n)[i]
        b[4*n + i] = 0
    
    # Pairwise non-overlap constraints
    pair_indices = np.triu_indices(n, k=1)
    for (i, j) in zip(pair_indices[0], pair_indices[1]):
        A[4*n + n*n + (i * n + j), i] = 1
        A[4*n + n*n + (i * n + j), j] = 1
        b[4*n + n*n + (i * n + j)] = distances[i, j]
    
    # Solve the linear program
    res = linprog(c=-radii.flatten(), A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    return -res.fun  # Negative sum of radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Pre-initialize with centroids of an approximate hexagonal lattice with some jitter
    initial_positions = np.array([
        # Monotonic x, lower y, upper y
        (0.0625, 0.0625, 0.2813), (0.2813, 0.0625, 0.2813), (0.5000, 0.0625, 0.2813), (0.7188, 0.0625, 0.2813), (0.9375, 0.0625, 0.2813), # Row 1
        (0.0625, 0.2813, 0.5000), (0.2813, 0.2813, 0.5000), (0.5000, 0.2813, 0.5000), (0.7188, 0.2813, 0.5000), (0.9375, 0.2813, 0.5000), # Row 2
        (0.0625, 0.5000, 0.7188), (0.2813, 0.5000, 0.7188), (0.5000, 0.5000, 0.7188), (0.7188, 0.5000, 0.7188), (0.9375, 0.5000, 0.7188), # Row 3
        (0.0625, 0.7188, 0.9375), (0.2813, 0.7188, 0.9375), (0.5000, 0.7188, 0.9375), (0.7188, 0.7188, 0.9375), (0.9375, 0.7188, 0.9375)  # Row 4
    ])
    n = initial_positions.shape[0]
    initial_positions = initial_positions[np.random.rand(n) < 0.86, :]
    initial_positions = np.reshape(initial_positions, -1)
    
    # Run differential evolution
    res = differential_evolution(evaluate, bounds=[(0, 1)] * (2 * n), popsize=5, maxiter=200, tol=1e-6, mutation=(0.5, 1.0), recombination=0.7, x0=initial_positions)
    
    best_positions = res.x
    best_radii = -evaluate(best_positions)
    best_centers = best_positions[:n].reshape(n, 2)
    best_radii = best_positions[n:].reshape(n, 1)
    
    # Optional polishing with local optimisation
    solution = minimize(evaluate, initial_positions, bounds=[(0, 1)] * (2 * n), jac=None, constraints={"type": "ineq", "fun": lambda x: -evaluate(x)}, method="SLSQP", options={"maxiter": 2000}, tol=1e-9)
    best_positions = solution.x
    best_centers = best_positions[:n].reshape(n, 2)
    best_radii = best_positions[n:].reshape(n, 1)
    best_radii = np.clip(best_radii, 0, 0.5)  # Optional numerical safeguard
    
    return best_centers, best_radii, float(np.sum(best_radii))
