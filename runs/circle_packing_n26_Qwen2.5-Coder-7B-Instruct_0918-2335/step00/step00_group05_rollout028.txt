import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    nt = 8000
    delta_t = 1e-3
    alpha = 4.0
    beta = 0.8
    eps = 1e-12
    dr = 1e-4
    tol = 1e-10
    max_iter = 3000

    # Initialisation
    x = np.random.uniform(eps, 1 - eps, n)
    y = np.random.uniform(eps, 1 - eps, n)
    r = 1e-6 * np.ones(n)

    # Dynamics loop
    for step in range(max_iter):
        F = np.zeros_like(r)
        for i in range(n):
            for j in range(i):
                dist = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2)
                if dist < r[i] + r[j] + eps:
                    difference = r[i] + r[j] - dist
                    direction = (x[i] - x[j], y[i] - y[j]) / dist
                    F[i] += alpha * difference * direction
                    F[j] -= alpha * difference * direction
            if x[i] < r[i] + eps:
                F[i] += alpha * (r[i] + eps - x[i], 0)
            if x[i] > 1 - r[i] - eps:
                F[i] -= alpha * (x[i] - (1 - r[i] - eps), 0)
            if y[i] < r[i] + eps:
                F[i] += alpha * (0, r[i] + eps - y[i])
            if y[i] > 1 - r[i] - eps:
                F[i] -= alpha * (0, y[i] - (1 - r[i] - eps))
        
        v = beta * -F + dr
        x += delta_t * v
        r += dr

        # Feasibility clip
        for i in range(n):
            r[i] = min(r[i], min(1 - x[i], x[i], 1 - y[i], y[i]) - eps)

    # LP refinement
    A = []
    b = []
    c = -r

    for i in range(n):
        A.append([-1, -1, x[i], y[i], 1])
        b.append(0)
        A.append([1, 0, x[i], y[i], 1])
        b.append(1 - x[i])
        A.append([0, -1, x[i], y[i], 1])
        b.append(0)
        A.append([0, 1, x[i], y[i], 1])
        b.append(1 - y[i])

    for i in range(n):
        for j in range(n):
            if i != j:
                A.append([1, -1, x[i], y[i], x[j], y[j], 1, 1])
                b.append(r[j])
                A.append([-1, 1, x[i], y[i], x[j], y[j], -1, -1])
                b.append(r[i])
                A.append([1, 1, x[i], y[i], x[j], y[j], -1, -1])
                b.append(r[j] + r[i])
                A.append([-1, -1, x[i], y[i], x[j], y[j], 1, -1])
                b.append(r[i] + r[j])

    res = linprog(c, A_ub=A, b_ub=b, bounds=(eps, None), method='highs')
    if res.success:
        r = res.x
        for i in range(n):
            for j in range(n):
                if i != j:
                    if abs(r[i] - r[j]) > eps:
                        r[i] = min(max(r[i], eps), r[j] - eps / 2)
                        r[j] = min(max(r[j], eps), r[i] + eps / 2)

    # Simulated Annealing
    T = 1e-3
    radius_accepted = False
    while T > 1e-6 and not radius_accepted:
        for _ in range(5000):
            i = np.random.randint(n)
            r_prime = r.copy()
            r_prime[i] += np.random.normal(0, T)
            if r_prime[i] <= 0:
                r_prime[i] = eps
            elif r_prime[i] >= 1:
                r_prime[i] = 1 - eps
            feasible = True
            for j in range(n):
                if i != j:
                    if r_prime[i] + r_prime[j] < np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2):
                        feasible = False
                        break
            if feasible:
                objective_improvement = np.sum(r_prime) - np.sum(r)
                if objective_improvement > 0 or np.random.rand() < np.exp(-objective_improvement / T):
                    r = r_prime
                    radius_accepted = True

    return np.vstack((x, y)).T, r, np.sum(r)
