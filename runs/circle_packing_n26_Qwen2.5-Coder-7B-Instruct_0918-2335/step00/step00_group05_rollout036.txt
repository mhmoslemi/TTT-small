import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    centers = np.random.rand(n_circles, 2)
    radii = np.random.rand(n_circles) / 10
    feasible_centers = centers.copy()
    feasible_radii = radii.copy()

        centers = params[:52].reshape(n_circles, 2)
        radii = params[52:].reshape(n_circles,)
        radius_sum = -np.sum(radii)
        return radius_sum, centers, radii

    def feasibility_projection(params):
        centers = params[:52].reshape(n_circles, 2)
        radii = params[52:].reshape(n_circles,)
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            x = np.clip(x, r, 1 - r)
            y = np.clip(y, r, 1 - r)
            centers[i] = (x, y)
            for j in range(i + 1, n_circles):
                dx, dy = centers[j] - centers[i]
                d = np.sqrt(dx * dx + dy * dy)
                if d < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - d
                    dx *= overlap / (2 * d)
                    dy *= overlap / (2 * d)
                    centers[i] += (dx, dy)
                    centers[j] -= (dx, dy)
        return centers.flatten()

    def constraint(params):
        centers = np.random.rand(n_circles, 2)
        radii = np.random.rand(n_circles) / 10
        d = pdist(centers)
        d = squareform(d)
        return radii + radii[:, np.newaxis] - d[:, 1:]

    constraints = {'type': 'ineq', 'fun': constraint}

                      method='SLSQP', callback=optimizer_callback)

    centers, radii = feasibility_projection(result.x).reshape(n_circles, 2)

    for i in range(n_circles):
        while not validate_packing(centers, radii)[0]:
