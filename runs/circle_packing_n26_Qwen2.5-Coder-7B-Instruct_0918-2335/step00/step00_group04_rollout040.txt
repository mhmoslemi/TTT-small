import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    MAX_ITER = 400
    n = 36
    centers = np.zeros((n, 2))
    x, y = np.meshgrid(np.linspace(1/6, 5/6, 6), np.linspace(1/6, 5/6, 6))
    centers = np.stack((x.flatten(), y.flatten()), axis=1)[:n]
    targets = np.argpartition(np.sum(np.max(np.abs(centers[:, np.newaxis] - centers[np.newaxis, :]), axis=-1),
                                    axis=1), -10)[-10:]
    centers = np.delete(centers, targets, axis=0)
    
    for i in range(centers.shape[0]):
        centers[i] += 0.01 * np.random.randn(2)
        centers[i] = np.clip(centers[i], 0, 1)
    
    def objective(r):
        return -np.sum(r)
    
    def loss(r):
        slack_ij = r + r - np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=-1))
        slack_ij[slack_ij < 0] = 0
        return slack_ij.sum()
    
    def lp_obj(r):
        slack_ij = np.concatenate((slack_ij[:, np.newaxis], slack_ij[np.newaxis, :]), axis=0)
        r_ij = np.concatenate((r[:, np.newaxis], r[np.newaxis, :]), axis=1)
        eq = np.hstack((r_ij, slack_ij))
        f = -np.sum(r)
        b = np.zeros(2 * r.shape[0])
        b[:r.shape[0]] = np.inf
        A = np.eye(2 * r.shape[0])
        b_eq = np.inf * np.ones(2 * r.shape[0])
        res = linprog(f, method='highs', bounds=np.hstack((np.zeros(r.shape[0]), np.ones(r.shape[0]))),
                      A_ub=np.vstack((slack_ij, -slack_ij)), b_ub=b,
                      A_eq=A, b_eq=b_eq, x0=np.zeros(r.shape[0]))
        return res.fun, res.x

    def solve_lp(c):
        slack_ij = np.sqrt(np.sum((c[:, np.newaxis] - c[np.newaxis, :]) ** 2, axis=-1))
        slack_ij[slack_ij < 1e-9] = np.inf
        r_eq = np.empty(n)
        for i in range(n):
            row = slack_ij[i, i + 1:]; assert row.size > 0
            objective = row.copy()
            position = i + 1
            objective[position:] = np.inf
            x = linprog(objective, bounds=np.vstack((c[:position, :].T, c[position:, :].T)).reshape(-1, 2).T, method='highs').x
            r_eq[i] = min(x[position:], 1 - min(c[i]))
        return r_eq

    def force(c, r):
        F = np.zeros_like(c)
        for i in range(n):
            for j in range(i + 1, n):
                d_ij = np.linalg.norm(c[i] - c[j])
                s_ij = r[i] + r[j] - d_ij
                if s_ij > 0:
                    F[i] += (c[i] - c[j]) / d_ij * s_ij
                    F[j] -= (c[i] - c[j]) / d_ij * s_ij
            w = np.zeros(2)
            for dimension in (0, 1):
                xi = c[i, dimension]
                ri = r[i]
                gap_i = ri - xi
                if gap_i > 0:
                    w[dimension] += gap_i
            F[i] += w
        return F

    rng = np.random.RandomState(42)
    obj = 0
    rs = np.zeros(n)
    for k in range(MAX_ITER):
        rs = solve_lp(centers)
        F = force(centers, rs)
        obj = np.sum(rs)
        if k > 0 and np.abs(obj - prev_obj) < 1e-8:
            break
        prev_obj = obj
        tau = 1e-3 * 0.95 ** k
        centers += tau * F
        centers = np.clip(centers, 0, 1)
        if k % 50 == 0:
            centers += 0.02 * rng.normal(size=centers.shape)
            centers = np.clip(centers, 0, 1)

    r = solve_lp(centers)[0]
    rs[:26] = r
    obj = np.sum(rs[:26])
    valid, msg = validate_packing(centers[:26], rs[:26])
    if not valid:
        raise ValueError("Invalid packing: " + msg)
    obj = round(obj, 9)
    return centers[:26], rs[:26], obj
