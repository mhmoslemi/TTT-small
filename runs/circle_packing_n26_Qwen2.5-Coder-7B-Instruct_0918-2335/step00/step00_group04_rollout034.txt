import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    
    n = 26
    bounds = [(0, 1)]*2*n + [(0, 0.5)]*n
    
    def penalised_objective(z):
        x, y, r = z[:2*n], z[2*n:4*n], z[4*n:]
        penalty = -np.sum(r)
        for i in range(n):
            if x[i] - r[i] < 0:
                penalty -= (r[i] - x[i])**2
            if x[i] + r[i] > 1:
                penalty -= (x[i] + r[i] - 1)**2
            if y[i] - r[i] < 0:
                penalty -= (r[i] - y[i])**2
            if y[i] + r[i] > 1:
                penalty -= (y[i] + r[i] - 1)**2
            for j in range(i + 1, n):
                dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                if dist < r[i] + r[j]:
                    penalty -= (r[i] + r[j] - dist)**2
        return penalty
    
    result = differential_evolution(penalised_objective, bounds, popsize=20, maxiter=2000, tol=1e-8)
    
    z_opt = result.x
    x, y, r = z_opt[:2*n], z_opt[2*n:4*n], z_opt[4*n:]
    penalty = penalised_objective(z_opt)
    
    if penalty > 1e-12:
        def local_obj(z):
            r = z[52:]
            return -np.sum(r)
        
        def local_constraints(z):
            x, y, r = z[:2*n], z[2*n:4*n], z[4*n:]
            constraints = [x[i] - r[i] for i in range(n)]
            constraints += [1 - x[i] - r[i] for i in range(n)]
            constraints += [y[i] - r[i] for i in range(n)]
            constraints += [1 - y[i] - r[i] for i in range(n)]
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                    constraints += [dist - (r[i] + r[j])]
            return constraints
        
        bounds = [(0, 1)]*2*n + [(0, 0.5)]*n
        result = minimize(local_obj, z_opt, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': local_constraints})
        z_opt = result.x
        x, y, r = z_opt[:2*n], z_opt[2*n:4*n], z_opt[4*n:]
    
    centers = np.array([x[i:i+2] for i in range(0, 2*n, 2)])
    radii = r
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(sum_radii)
