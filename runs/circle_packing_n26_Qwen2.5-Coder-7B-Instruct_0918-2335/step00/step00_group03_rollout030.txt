import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = []
    radii = []
    
    # Corners
    centers.append([r_c, r_c])
    centers.append([1 - r_c, r_c])
    centers.append([r_c, 1 - r_c])
    centers.append([1 - r_c, 1 - r_c])
    radii.extend([r_c] * 4)
    
    # Edges (midpoints)
    centers.append([0.5, r_e])
    centers.append([0.5, 1 - r_e])
    centers.append([r_e, 0.5])
    centers.append([1 - r_e, 0.5])
    radii.extend([r_e] * 4)
    
    # Inner hexagonal core
    r = 2 * np.sqrt(3) * r_i
    offset = np.array([0, 0], dtype=float)
    while offset[1] < 1:
        while offset[0] + r < 1:
            if np.linalg.norm(offset, ord=2) > 1 - r:
                centers.append(offset + np.array([1 - np.sqrt(3) * r_i, r_i + r], dtype=float) / 2)
                centers.append(offset + np.array([np.sqrt(3) * r_i, r_i + r], dtype=float) / 2)
                centers.append(offset + np.array([1 - np.sqrt(3) * r_i, -r_i + r], dtype=float) / 2)
                centers.append(offset + np.array([np.sqrt(3) * r_i, -r_i + r], dtype=float) / 2)
                radii.extend([r_i] * 4)
            offset[0] += r
        offset[1] += r
        offset[0] = 0
    
    return np.array(centers), np.array(radii)

def compute_sum_radii(radii):
    return radii.sum()

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    sum_r = compute_sum_radii(radii)
    return -sum_r  # DE minimizes, so negate the sum to maximize

def constraint_func(v):
    n = 26
    centers = v[:52].reshape(n, 2)
    radii = v[52:]
    cons = []
    for i in range(n):
        if centers[i, 0] < radii[i] or centers[i, 1] < radii[i] or centers[i, 0] > 1 - radii[i] or centers[i, 1] > 1 - radii[i]:
            cons.append(centers[i, 0] + radii[i] - 1)
        for j in range(i):
            dist = np.linalg.norm(centers[i] - centers[j], ord=2)
            if dist < radii[i] + radii[j]:
                cons.append(radii[i] + radii[j] - dist)
    return np.array(cons)

def run_packing():
    bounds = [(0.30, 0.49), (0.10, 0.25), (0.05, 0.15)]
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)
    
    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [(0, 1)] * 52
    cons = [{'type': 'ineq', 'fun': constraint_func}]
    res = minimize(lambda v: -compute_sum_radii(v[52:]), x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 2000})
    
    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]
    
    sum_r = compute_sum_radii(radii)
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")
    
    return centers, radii, sum_r
