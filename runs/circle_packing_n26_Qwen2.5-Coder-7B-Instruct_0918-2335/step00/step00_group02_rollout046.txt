import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    random_seeds = [0, 1, 2, 3]

    def objective(x):
        xs, ys, rs = x[:2*N].reshape((-1, 2)), x[2*N:].reshape((-1, 1))
        sum_radii = -rs.sum()
        p_boundary = ((np.maximum(0, rs - xs)**2 +
                      np.maximum(0, xs + rs - 1)**2 +
                      np.maximum(0, rs - ys)**2 +
                      np.maximum(0, ys + rs - 1)**2).sum())
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist = np.hypot(dx, dy)
        rr = rs[:, None] + rs[None, :]
        overlap = rr - dist
        p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
        return sum_radii + lambda_penalty * (p_boundary + p_overlap)

    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N

    best_solution = None
    best_value = -np.inf

    for seed in random_seeds:
        np.random.seed(seed)
        result = differential_evolution(objective, bounds, popsize=DE_popsize, maxiter=DE_maxiter)
        if result.fun > best_value:
            best_value = result.fun
            best_solution = result.x

    r = minimize(objective, best_solution, method='L-BFGS-B', bounds=bounds)

    centers = np.column_stack([r.x[0:N*2:2], r.x[1:N*2:2]])
    radii = r.x[2*N:]
    sum_radii = radii.sum()

    _, status = validate_packing(centers, radii)
    if status != "ok":
        raise ValueError(f"Validation failed: {status}")

    return centers, radii, sum_radii
