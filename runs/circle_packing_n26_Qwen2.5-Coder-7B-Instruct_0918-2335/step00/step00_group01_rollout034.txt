def solve_lp(centres):
    n = centres.shape[0]
    dist_matrix = np.sqrt(np.sum((centres[:, None, :] - centres[None, :, :])**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    
    c = -np.ones(n)
    A_ub = np.empty((4*n + n*(n-1)//2, n))
    b_ub = np.empty(4*n + n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centres[i, 0]
        idx += 1
        
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centres[i, 0]
        idx += 1
        
        A_ub[idx, i] = 1
        b_ub[idx] = centres[i, 1]
        idx += 1
        
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centres[i, 1]
        idx += 1
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist_matrix[i, j]
            idx += 1
    
    res = linprog(c, A_ub, b_ub, bounds=(0, None), method='simplex')
    
    if not res.success:
        raise ValueError("LP solver failed")
    
    return res.x, -res.fun  # Negate sum for maximization

def compute_forces(centres, radii):
    n = centres.shape[0]
    dist_matrix = np.sqrt(np.sum((centres[:, None, :] - centres[None, :, :])**2, axis=2))

                diff = centres[j] - centres[i]
                force = duals[idx] * diff / dist_matrix[j, i]
                forces[i] += force
                forces[j] -= force
            idx += 1

            idx += 1
    
    return forces
