import numpy as np

def run_packing():
    grid_resolution = 200
    delta_c = 1e-3
    n_refine = 200
    n_circles = 26
    
    # Initialize grid
    grid_x, grid_y = np.meshgrid(np.linspace(0, 1, grid_resolution), np.linspace(0, 1, grid_resolution))
    grid_x, grid_y = grid_x.flatten(), grid_y.flatten()
    grid_points = np.vstack((grid_x, grid_y)).T
    
    # Compute distances to sides
    d_boundary = np.min(np.vstack((grid_x, 1-grid_x, grid_y, 1-grid_y)).T, axis=1)
    
    # Initialize candidate radii
    r_f = 0.01
    r_f[grid_x < r_f] = 0.01
    r_f[grid_y < r_f] = 0.01
    r_f[grid_x > 1-r_f] = 0.01
    r_f[grid_y > 1-r_f] = 0.01
    
    # List to store centers and radii
    centers = []
    radii = []
    
    # Greedy circle insertion
    for _ in range(n_circles):
        # Find best candidate
        idx = np.argmax(r_f)
        center = grid_points[idx]
        radius = r_f[idx] - 1e-9
        
        # Append center and radius
        centers.append(center)
        radii.append(radius)
        
        # Update candidate radii
        dx = grid_x - center[0]
        dy = grid_y - center[1]
        dist = np.sqrt(dx**2 + dy**2)
        r_f = np.min(np.vstack((d_boundary, dist - np.array(radii))).T, axis=1)
        
        # Mark this point as used
        r_f[idx] = -1
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Local radius and centre refinement
    for _ in range(n_refine):
        for i in np.random.permutation(n_circles):
            # Propose new center
            delta = np.random.uniform(-delta_c, delta_c, size=2)
            new_center = centers[i] + delta
            
            # Compute new radius
            dx = grid_x - new_center[0]
            dy = grid_y - new_center[1]
            dist = np.sqrt(dx**2 + dy**2)
            r_new = np.min(np.vstack((d_boundary, dist - np.array(radii))).T, axis=1).min()
            
            # Accept move if it increases sum
            if r_new > radii[i]:
                centers[i] = new_center
                radii[i] = r_new
    
    # Recompute radii precisely
    dx = grid_x[:, np.newaxis] - centers[:, 0]
    dy = grid_y[:, np.newaxis] - centers[:, 1]
    dist = np.sqrt(dx**2 + dy**2)
    radii = np.min(np.vstack((d_boundary, dist - np.array(radii))).T, axis=1)
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0)
    
    # Compute sum of radii
    sum_radii = radii.sum()
    
    return centers, radii, sum_radii
