import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def initialize_centers():
    s = 0.312
    k = np.arange(-10, 17, 1)
    x = np.repeat(k + s, 6).reshape(-1, 2)
    y = np.repeat(k[:17], 3).reshape(-1, 2)
    y[1::2] = y[1::2] + s * np.sqrt(3) / 2
    y[2::2] = y[2::2] + s * np.sqrt(3)
    centers = x[y[:, 1] <= 1-s]
    centers = centers[centers[:, 0] <= 1-s]
    centers = centers[:26]
    return centers

def lp_solver(centers):
    n = centers.shape[0]
    d = squareform(pdist(centers))
    A_eq = np.empty((n*(n-1)//2, n))
    b_eq = np.empty(n*(n-1)//2)
    c = np.empty(n)
    
    for i in range(n):
        c[i] = 1
        for j in range(i+1, n):
            A_eq[(i*(i-1)//2+j-i-1), i] = 1
            A_eq[(i*(i-1)//2+j-i-1), j] = 1
            b_eq[(i*(i-1)//2+j-i-1)] = d[i, j]
    
    for i in range(n):
        A_eq[(n*(n-1)//2+i*n), i] = 1
        b_eq[(n*(n-1)//2+i*n)] = min(centers[i, 0], 1-centers[i, 0], centers[i, 1], 1-centers[i, 1])

    res = linprog(-c, A_le=A_eq, b_le=b_eq, bounds=(0, None), method='highs')
    radii = -res.x
    return centers, radii, np.sum(radii)
