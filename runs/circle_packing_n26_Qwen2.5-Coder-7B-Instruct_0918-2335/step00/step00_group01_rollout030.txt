```python
import numpy as np
from scipy.optimize import minimize

n = 26
pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)
eps = 1e-7

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, size=(n, 2))
    radii = np.zeros(n)
    return np.concatenate([centers, radii])

def objective(v):
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    return -np.sum(radii)

def constraints(v):
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    boundary_constraints = np.maximum(centers[:, 0] - radii, 0) + np.maximum(1 - centers[:, 0] - radii, 0) + np.maximum(centers[:, 1] - radii, 0) + np.maximum(1 - centers[:, 1] - radii, 0)
    pairwise_constraints = (centers[pair_idx_i] - centers[pair_idx_j]) ** 2.sum(axis=1) - (radii[pair_idx_i] + radii[pair_idx_j]) ** 2
    return np.hstack((boundary_constraints, pairwise_constraints))

bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_value = -np.inf
    best_v = None
    for seed in range(8):
        v0 = initial_guess(seed)
        res = minimize(
            fun=objective,
            x0=v0,
            method='SLSQP',
            constraints={'type': 'ineq', 'fun': constraints},
            bounds=bounds,
            options={
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if res.success and res.fun > best_value:
            best_value = res.fun
            best_v = res.x
    centers = best_v[:2*n].reshape((n, 2))
    radii = np.maximum(0.0, best_v[2*n:] - eps)
    return centers, radii, np.sum(radii)
```
