import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    vMagnitude = 5e-3
    T_max = 50.0
    epsilon = 1e-6

    np.random.seed(42)
    initial_positions = np.random.rand(n, 2)
    initial_velocities = (np.random.rand(n, 2) - 0.5) * vMagnitude

    def are_within_bounds(v):
        return np.all(np.abs(v) < 2 * r_min)

    def compute_initial_r():
        return np.zeros(n) + r_min

    def compute_event_times(centers, velocities, radii):
        event_times = []
        for i in range(n):
            for j in range(i + 1, n):
                if are_within_bounds(velocities[i] - velocities[j]):
                    delta_c = centers[i] - centers[j]
                    delta_v = velocities[i] - velocities[j]
                    delta_r = radii[i] + radii[j]
                    a = np.dot(delta_v, delta_v)
                    b = 2 * np.dot(delta_c, delta_v)
                    c = np.dot(delta_c, delta_c) - delta_r ** 2
                    if a > 0:
                        t = (-b + np.sqrt(b ** 2 - 4 * a * c)) / (2 * a)
                        if t > 0:
                            event_times.append((t, i, j))

        for i in range(n):
            if are_within_bounds(velocities[i]):
                for wall in [(0, -1), (1, 1), (0, 1), (1, -1)]:
                    v_component = velocities[i][wall[0]]
                    if v_component < g:
                        t = (wall[1] - centers[i][wall[0]]) / (v_component - g)
                        if t > 0:
                            event_times.append((t, i, "wall"))

        event_times.sort()
        return event_times

    def resolve_event(centers, velocities, radii, event):
        t, i, j = event
        if j == "wall":
            v_new_i = velocities[i].copy()
            v_new_i[event[1]] *= -1
            centers[i] += (velocities[i] * (t - 0.0001)) + (v_new_i * 0.0001)
            velocities[i] = v_new_i
            radii[i] += g * t
        else:
            v_new_i = velocities[i].copy()
            v_new_j = velocities[j].copy()
            delta_c = centers[i] - centers[j]
            dot_v_c = np.dot(velocities[i] - velocities[j], delta_c)
            tnc = max(0, dot_v_c / (dot_v_c * 2) - np.sqrt(((dot_v_c**2) / (2 * delta_c**2)) - 1))
            delta_v_c *= tnc
            v_new_i -= delta_v_c
            v_new_j += delta_v_c
            centers[i] += velocities[i] * (t - 0.0001) + v_new_i * 0.0001
            centers[j] += velocities[j] * (t - 0.0001) + v_new_j * 0.0001
            velocities[i] = v_new_i
            velocities[j] = v_new_j
            radii[i] += g * t
            radii[j] += g * t

    def polish_radii(centers, radii):
        A = np.vstack((np.ones(n), *[(radii[:, None] + radii[i][:, None] + np.sqrt(np.sum((centers - centers[i]) ** 2, axis=1))) for i in range(n)]))[:-1].T
        b = np.ones(n + n * (n - 1) // 2)
        for i in range(n):
            A[i] = A[i] + boundaries(centers[i, 0], centers[i, 1], radii[i]) * -1
        result = linprog(-radii, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        return result.x

    def boundaries(x, y, r):
        return [1 - x - r, x + r, 1 - y - r, y + r]

    def shrink_radii(radii):
        return radii * (1 - epsilon)

    def repair_overlap(centers, radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    overlap = (radii[i] + radii[j] - dist) / 2
                    centers[i] += (centers[j] - centers[i]) / dist * overlap * 0.5
                    centers[j] -= (centers[j] - centers[i]) / dist * overlap * 0.5
                    radii[i] -= overlap * 0.5
                    radii[j] -= overlap * 0.5

    centers = initial_positions
    velocities = initial_velocities
    radii = compute_initial_r()
    event_queue = compute_event_times(centers, velocities, radii)

    current_time = 0.0
    while event_queue and current_time < T_max:
        t_e, i, j = event_queue.pop(0)
        Δt = t_e - current_time
        centers += velocities * Δt
        radii += g * Δt
        current_time = t_e
        resolve_event(centers, velocities, radii, (t_e, i, j))
        event_queue = compute_event_times(centers, velocities, radii)

    radii = shrink_radii(radii)
    repair_overlap(centers, radii)
    final_radii = polish_radii(centers, radii)
    return centers, final_radii, np.sum(final_radii)
