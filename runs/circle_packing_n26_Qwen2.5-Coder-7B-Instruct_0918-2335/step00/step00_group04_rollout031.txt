import numpy as np
from scipy.optimize import minimize

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

bounds = [(0, 1)] * 26
bounds += [(0, 1)] * 26
bounds += [(0, 0.5)] * 26

def run_packing():
    best_sol = None
    best_sum = 0
    num_trials = 15
    
    for _ in range(num_trials):
        init = np.random.rand(78) * 2 - 1
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': constraints}, 
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        
        if res.success and np.all(constraints(res.x) >= -1e-10):
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sol = (res.x[:52].reshape(26, 2), res.x[52:], current_sum)
                best_sum = current_sum
    
    return best_sol

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
