import numpy as np
from scipy.optimize import minimize

def constraint_func(v):
    n = 26
    x = v[:52].reshape((n,2))
    r = v[52:]
    arr = []
    arr.append(x[:,0] - r)
    arr.append(1 - x[:,0] - r)
    arr.append(x[:,1] - r)
    arr.append(1 - x[:,1] - r)
    pair_indices = [(i,j) for i in range(n) for j in range(i+1,n)]
    for i,j in pair_indices:
        dx = x[i,0] - x[j,0]
        dy = x[i,1] - x[j,1]
        dist2 = dx*dx + dy*dy
        arr.append(dist2 - (r[i] + r[j])**2)
    return np.concatenate(arr)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    positions = np.array([( (j+0.5)/5.0, (i+0.5)/5.0 ) for i in range(5) for j in range(5) if len(positions) < 25])
    positions = np.vstack([positions, np.array([0.6, 0.9])])
    radii_initial = np.array([0.04] * n)
    x0 = np.concatenate((positions.flatten(), radii_initial))
    bounds = [(0,1)] * (2*n) + [(0,0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    res = minimize(fun=lambda v: -np.sum(v[n:]), x0=x0, bounds=bounds, constraints=cons, options={'maxiter':2000, 'ftol':1e-12, 'disp':False})
    centers = res.x[:52].reshape((n,2))
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
