import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step_size = 0.03
    max_iterations = 200
    convergence_threshold = 1e-6
    max_restarts = 3
    eps = 1e-7

    # Initial pattern of circle centers
    centres = np.zeros((n, 2))
    row_counter = 0
    for row_size in [6, 5, 6, 5, 4, 1]:
        y_center = 0.1 + row_counter * 0.08
        for i in range(row_size):
            x_center = 0.1 + i * 0.058
            centres[row_counter + i] = [x_center, y_center]
        row_counter += 1

    best_centers = centres.copy()
    best_radii = np.zeros(n)
    best_obj = -1

    for restart in range(max_restarts):
        if restart > 0:
            # Random jitter and rotation
            jitter = np.random.uniform(-0.01, 0.01, (n, 2))
            rotation_angle = np.random.choice([0, np.pi/2, np.pi, 3*np.pi/2])
            rotation_matrix = np.array([[np.cos(rotation_angle), -np.sin(rotation_angle)],
                                        [np.sin(rotation_angle), np.cos(rotation_angle)]])
            centres = np.matmul(np.max([0, 1-centres[:, 0]], axis=0) * 0.98, rotation_matrix) + jitter + centres
            centres = np.clip(centres, 0, 1)

        for iter in range(max_iterations):
            # Solve the linear program for the current centers
            A_ub = np.zeros((4*n + n*n, n))
            b_ub = np.zeros(4*n + n*n)
            for i in range(n):
                # Boundary constraints for circle i
                A_ub[i*4:4*i+4, i] = np.array([1, 1, 1, 1])
                b_ub[i*4+0] = centres[i, 0]
                b_ub[i*4+1] = 1 - centres[i, 0]
                b_ub[i*4+2] = centres[i, 1]
                b_ub[i*4+3] = 1 - centres[i, 1]

            for i in range(n):
                for j in range(i+1, n):
                    dist_ij = np.linalg.norm(centres[i] - centres[j])
                    if dist_ij > 0:
                        k = 4*n + (i * (n-1) + j)
                        A_ub[k, i] = 1
                        A_ub[k, j] = 1
                        b_ub[k] = dist_ij

            c = -np.ones(n)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub)

            if not res.success:
                # If the LP is infeasible, restart with jitter
                continue

            radii = res.x
            obj = -res.fun

            if obj > best_obj:
                best_obj = obj
                best_centers = centres.copy()
                best_radii = radii.copy()

            # Compute forces from duals
            forces = np.zeros((n, 2))
            dual_idx = 0
            for i in range(n):
                for j in range(i+1, n):
                    if A_ub[dual_idx, i] == 1 and A_ub[dual_idx, j] == 1:
                        dual_val = res.dual[5 + 2*i*(n-1) + j*i]  # Direct index access
                        double_dist = np.linalg.norm(centres[i] - centres[j])
                        force_ij = dual_val * (centres[i] - centres[j]) / double_dist
                        forces[i] -= force_ij
                        forces[j] += force_ij
                    dual_idx += 1

                boundary_limit = np.array([np.clip(1 - centres[i, 0], 0.01, 0.98),
                                          np.clip(1 - centres[i, 1], 0.01, 0.98),
                                          np.clip(centres[i, 0], 0.01, 0.98),
                                          np.clip(centres[i, 1], 0.01, 0.98)])
                boundary_forces = np.array([boundary_limit[0], boundary_limit[3],
                                              boundary_limit[2], boundary_limit[1]])
                dual_idx += 4
                forces[i] + boundary_forces / 4

            # Move the centers
            centres += step_size * forces

            # Apply small random jitter to avoid local optimum
            jitter = np.clip(np.random.uniform(-0.01, 0.01, (n, 2)), -0.1, 0.1)
            centres += jitter
            centres = np.clip(centres, 0, 1)

            # Re-solve the linear program for the moved centers to ensure feasibility
            res = linprog(c, A_ub=A_ub, b_ub=b_ub)
            if res.success:
                radii = np.maximum(0.0, res.x - eps)
                if np.sum(radii) < best_obj - convergence_threshold:
                    break

        ifbest_obj - best_obj - convergence_threshold) < convergence_threshold:
            break

    return best_centers, best_radii, np.sum(best_radii)
