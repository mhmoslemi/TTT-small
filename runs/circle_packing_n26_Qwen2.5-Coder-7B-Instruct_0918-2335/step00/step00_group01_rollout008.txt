import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from random import random

def penalty_bound(r, x, y):
    return max(0, r - min(x, 1 - x, y, 1 - y))**2

def penalty_overlap(c1, c2, r1, r2):
    return max(0, r1 + r2 - np.linalg.norm(c1 - c2))**2

def validate(pose, radii):
    valid = True
    for r, x, y in zip(radii, pose[:,0], pose[:,1]):
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            valid = False
            break

    dists = distance_matrix(pose, pose)
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            if dists[i, j] < radii[i] + radii[j] - 1e-12:
                valid = False
                break

    return valid

def objective(pose, radii, M_bound, M_overlap):
    n = len(radii)
    penalty = 0
    for i in range(n):
        penalty += penalty_bound(radii[i], pose[i, 0], pose[i, 1])
        for j in range(i + 1, n):
            penalty += penalty_overlap(pose[i], pose[j], radii[i], radii[j])
    return -np.sum(radii) + M_bound * penalty

def mutate(pose, radii, sigma_pos, sigma_rad, M_bound, M_overlap):
    n = len(radii)
    pose_new = np.copy(pose)
    radii_new = np.copy(radii)
    for i in range(n):
        if random() < 0.5:
            transl = np.random.normal(0, sigma_pos, size=2)
            pose_new[i] += transl
        if random() < 0.5:
            rad_change = np.random.normal(0, sigma_rad)
            radii_new[i] += rad_change
        radii_new[i] = max(0, radii_new[i])
        pose_new[i] = np.clip(pose_new[i], 0, 1)
    return mutate(pose_new, radii_new, sigma_pos, sigma_rad, M_bound, M_overlap) if not validate(pose_new, radii_new) else (pose_new, radii_new, objective(pose_new, radii_new, M_bound, M_overlap))

def crossover(pose1, radii1, pose2, radii2, M_bound, M_overlap):
    n = len(radii1)
    pose_child = np.copy(pose1)
    radii_child = np.copy(radii1)
    for i in range(n):
        if random() < 0.5:
            pose_child[i] = pose2[i]
        if random() < 0.5:
            radii_child[i] = radii2[i]
    return minimize(objective, x0=np.concatenate((pose_child.flatten(), radii_child)), args=(M_bound, M_overlap), method="L-BFGS-B", bounds=[(0, 1), (0, None)] * n)

def main():
    n = 26
    pose_init = np.array([[0.1 * (4*i + 1 + (-1)**i) for i in range(n // 2)], [0.2 * (2 * (i // 2) + 1) for i in range(n)]])
    radii_init = np.array([[min(pose_init[i, 0], 1 - pose_init[i, 0], pose_init[i, 1], 1 - pose_init[i, 1]) for i in range(n)]])
    pose_init = pose_init.T
    pose_init, radii_init, _ = mutate(pose_init, radii_init.flatten(), sigma_pos=0.2, sigma_rad=0.2, M_bound=100, M_overlap=100)
    population = [pose_init, radii_init.flatten()]
    M_bound = 100
    M_overlap = 100
    generation = 0

    for generation in range(800):
        print(f"Generation {generation}")
        population.sort(key=lambda x: x[2])
        population = [population[i] for i in range(50)]  # Retain best 50 individuals
        population = population[:50]
        for i in range(5):
            x, y = crossover(population[i * 2], population[i * 2][2], population[i * 2 + 1], population[i * 2 + 1][2], M_bound, M_overlap)
            population.append((x, y))
        population = [(mutate(pose, radii, sigma_pos=0.2 * (0.97**(generation // 20)), sigma_rad=0.2 * (0.97**(generation // 20)), M_bound, M_overlap)) for pose, radii in population]
        population = [item for sublist in population for item in sublist]  # Flatten list

        M_bound *= 3
        M_overlap *= 3

    population.sort(key=lambda x: x[2])
    best = population[0]
    pose_best, radii_best, _ = mutate(best[0], best[1], sigma_pos=0, sigma_rad=0, M_bound, M_overlap)
    return pose_best, radii_best, np.sum(radii_best)

pose, radii, sum_radii = main()
print(pose)
print(radii)
print(sum_radii)
