import numpy as np

def half_plane_clipping(poly, a, b):
    clipped = []
    n = len(poly)
    for i in range(n):
        k = (i + 1) % n
        if a.dot(poly[i]) <= b:
            clipped.append(poly[i])
        if a.dot(poly[i]) * a.dot(poly[k]) < 0:
            diff = poly[k] - poly[i]
            ratios = (b - a.dot(poly[i])) / a.dot(diff)
            clipped.append(poly[i] + ratios * diff)
    return clipped

def power_diagram_cell(centers, radii, i):
    cell = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
    n = len(centers)
    for j in range(n):
        if j != i:
            vec = centers[j] - centers[i]
            norm = np.linalg.norm(vec)
            a = 2 * vec / norm
            p_i = np.linalg.norm(centers[i], 2)**2
            p_j = np.linalg.norm(centers[j], 2)**2
            b = p_i - p_j + radii[i]**2 - radii[j]**2
            cell = half_plane_clipping(cell, a, b)
    return np.array(cell)

def centroid(polygon):
    n = len(polygon)
    if n < 3:
        return np.mean(polygon, axis=0)
    area = 0.5 * np.sum(polygon[i, 0] * polygon[(i + 1) % n, 1] - polygon[(i + 1) % n, 0] * polygon[i, 1] for i in range(n))
    cx = (1 / (6 * area)) * np.sum((polygon[i, 0] + polygon[(i + 1) % n, 0]) * (polygon[i, 0] * polygon[(i + 1) % n, 1] - polygon[(i + 1) % n, 0] * polygon[i, 1]) for i in range(n))
    cy = (1 / (6 * area)) * np.sum((polygon[i, 1] + polygon[(i + 1) % n, 1]) * (polygon[i, 0] * polygon[(i + 1) % n, 1] - polygon[(i + 1) % n, 0] * polygon[i, 1]) for i in range(n))
    return np.array([cx, cy])

def distance_to_polygon(point, polygon):
    min_dist = np.inf
    n = len(polygon)
    for i in range(n):
        p1 = polygon[i]
        p2 = polygon[(i + 1) % n]
        v1 = p1 - point
        v2 = p2 - point
        length_squared = v1.dot(v1)
        d = v1.dot(v2) / length_squared
        if 0 <= d <= 1:
            projection = point + d * v1
            dist = np.linalg.norm(projection - point)
        else:
            dist = min(np.linalg.norm(v1), np.linalg.norm(v2))
        min_dist = min(min_dist, dist)
    return min_dist

    radii = np.full(26, 0.01)

    mu = 0.5
    lambd = 0.8

    def clip_radii(radii):
        return np.maximum(radii, np.zeros_like(radii))

        centers = mu * centers + (1 - mu) * circumcenters
