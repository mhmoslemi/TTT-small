import scipy.spatial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    target_sum_radii = 2.636

    centers = np.array([[0.5 * np.cos(2 * np.pi * i / num_circles), 0.5 * np.sin(2 * np.pi * i / num_circles)]
                        for i in range(num_circles)])

    # Iterate to optimize the packing
    for _ in range(100):  # Number of iterations
        # Ensure the sum of radii does not exceed the target
        total_radius = np.sum(initial_radii)
        if total_radius > target_sum_radii:
            initial_radii *= target_sum_radii / total_radius

        # Enforce non-overlap and boundary constraints
        tree = scipy.spatial.KDTree(centers)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < initial_radii[i] + initial_radii[j] - 1e-12:
                    centers[i] += (initial_radii[i] + initial_radii[j] - dist) * (centers[j] - centers[i]) / dist

        # Normalize the centers to stay within the unit square
        centers /= np.abs(centers).max()

    total_radius = np.sum(initial_radii)
