import numpy as np
from scipy.optimize import minimize

n = 26
eps = 1e-7

# Precompute pair indices
pair_idx_i = np.repeat(np.arange(n), n - 1)
pair_idx_j = np.arange(1, n)[:, np.newaxis].repeat(n - 1, axisolving=0).flatten()

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.zeros(n)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[2*n:])

    b2 = (1-centers[:, 0]*2) - radii >= 0
    b3 = (centers[:, 1]*2) - radii >= 0
    b4 = (1-centers[:, 1]*2) - radii >= 0
    pairwise = np.sum((centers[pair_idx_i] - centers[pair_idx_j])**2, axis=1) - (radii[pair_idx_i] + radii[pair_idx_j])**2 >= 0
    return np.concatenate((b1, b2, b3, b4, pairwise))

bounds = [(0, 1)] * (n * 2) + [(0, 0.5)] * n

RESTARTS = 10
best_value = -np.inf
best_v = None

for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun=objective,
        x0=v0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if not res.success:
        continue
    value = -res.fun
    if value > best_value:
        best_value = value
        best_v = res.x

centers = best_v[:2*n].reshape((n, 2))
radii = np.maximum(0.0, best_v[2*n:] - eps)
sum_radii = np.sum(radii)

return centers, radii, sum_radii
