import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    x0 = np.random.rand(3 * n)

    def obj(v):
        r = v[2 * n:]
        return -np.sum(r)

    def constraints(v):
        pos = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r

        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=x0, method='SLSQP',
                   bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    
    if not res.success or np.any(constraints(res.x) < -1e-10):
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:2 * n].reshape(n, 2)
    radii = res.x[2 * n:]
    sum_radii = -res.fun
    return centers, radii, sum_radii
