import numpy as np

def run_packing():
    R = 200
    d = 1 / R
    center = np.linspace(d / 2, 1 - d / 2, R)
    X, Y = np.meshgrid(center, center)
    grid_x = X.flatten()
    grid_y = Y.flatten()
    grid = np.vstack((grid_x, grid_y)).T
    
    n_circles = 26
    centers = np.empty((n_circles, 2))
    radii = np.empty(n_circles)
    remainders = np.full(R**2, np.inf)
    
    def update_remainders(new_center, new_radius):
        dx = grid_x - new_center[0]
        dy = grid_y - new_center[1]
        dists = np.sqrt(dx**2 + dy**2)
        remainders = np.minimum(remainders, dists - new_radius)
        remainders[radii < 0] = -np.inf
        return remainders
    
    for i in range(n_circles):
        idx = np.argmax(remainders)
        new_center = grid[idx]
        new_radius = min(np.min([1 - new_center[0], new_center[0], 1 - new_center[1], new_center[1]]), remainders[idx] - 1e-9)
        centers[i] = new_center
        radii[i] = new_radius
        remainders = update_remainders(new_center, new_radius)
        remainders[idx] = -np.inf
    
    def adjust_centers_and_radii(centers, radii, n_iterations=200, delta=1e-3):
        for _ in range(n_iterations):
            accepted = False
            for i in range(n_circles):
                offset = np.random.uniform(-delta, delta, 2)
                new_center = centers[i] + offset
                min_dist_to_boundary = np.min([1 - new_center[0], new_center[0], 1 - new_center[1], new_center[1]])
                new_radius = min(min_dist_to_boundary, remainders[grid_x == np.round(new_center[0], 4) & grid_y == np.round(new_center[1], 4)][0] - 1e-9)
                if new_radius > radii[i]:
                    centers[i] = new_center
                    radii[i] = new_radius
                    accepted = True
                    remainders = update_remainders(new_center, new_radius)
                    break
            if not accepted:
                break
        return centers, radii
    
    centers, radii = adjust_centers_and_radii(centers, radii)
    np.clip(radii, 0, None, out=radii)
    
    return centers, radii, np.sum(radii)
