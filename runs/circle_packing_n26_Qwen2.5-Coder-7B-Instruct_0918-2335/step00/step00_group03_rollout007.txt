import numpy as np

def sutherland_hodgman(polygon, half_plane):
    def inside(p):
        return half_plane[0] * p[0] + half_plane[1] * p[1] - half_plane[2] >= 0
    
    output = []
    n = len(polygon)
    for i in range(n):
        p1 = polygon[i]
        p2 = polygon[(i + 1) % n]
        test1 = inside(p1)
        test2 = inside(p2)
        if test1 and test2:
            output.append(p2)
        elif test1 and not test2:
            intersection = np.array([(half_plane[2] - half_plane[0] * p2[0]) / half_plane[0] - half_plane[1] * p2[1]) / half_plane[1], p2[1]] if half_plane[0] != 0 else [(p2[0] + p1[0]) / 2, (half_plane[2] - half_plane[1] * p2[1]) / half_plane[1]])
            output.append(intersection)
            output.append(p2)
        elif not test1 and test2:
            intersection = np.array([(half_plane[2] - half_plane[0] * p1[0]) / half_plane[0] - half_plane[1] * p1[1]) / half_plane[1], p1[1]] if half_plane[0] != 0 else [(p2[0] + p1[0]) / 2, (half_plane[2] - half_plane[1] * p1[1]) / half_plane[1]])
            output.append(p1)
            output.append(intersection)
    return np.array(output)

def power_diagram_cell(centers, radii, i):
    polygon = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
    for j in range(26):
        if i == j:
            continue
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        r1 = radii[i]
        r2 = radii[j]
        a = 2 * (x2 - x1)
        b = 2 * (y2 - y1)
        c = x1 ** 2 - x2 ** 2 + y1 ** 2 - y2 ** 2 + r1 ** 2 - r2 ** 2
        half_plane = np.array([a, b, c])
        polygon = sutherland_hodgman(polygon, half_plane)
    return polygon

def centroid(polygon):
    n = len(polygon)
    if n < 3:
        return np.mean(polygon, axis=0)
    area = 0.5 * np.sum((polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]))
    cx = (1 / (6 * area)) * np.sum((polygon[:, 0] + polygon[:, 1]) * (polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]))
    cy = (1 / (6 * area)) * np.sum((polygon[:, 1] + polygon[:, 0]) * (polygon[:, 0] * polygon[:, 1] - polygon[:, 1] * polygon[:, 0]))
    return np.array([cx, cy])

def point_to_segment_dist(p, v1, v2):
    x1, y1 = v1
    x2, y2 = v2
    px, py = p
    seg_len_squared = (x2 - x1) ** 2 + (y2 - y1) ** 2
    if seg_len_squared == 0:
        return np.sqrt((px - x1) ** 2 + (py - y1) ** 2)
    t = ((px - x1) * (x2 - x1) + (py - y1) * (y2 - y1)) / seg_len_squared
    if 0 <= t <= 1:
        closest_point = np.array([x1 + t * (x2 - x1), y1 + t * (y2 - y1)])
        return np.sqrt((px - closest_point[0]) ** 2 + (py - closest_point[1]) ** 2)
    else:
        return min(np.sqrt((px - x1) ** 2 + (py - y1) ** 2), np.sqrt((px - x2) ** 2 + (py - y2) ** 2))
