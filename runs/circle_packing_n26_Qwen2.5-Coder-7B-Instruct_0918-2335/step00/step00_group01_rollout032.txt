```python
import numpy as np
from scipy.optimize import minimize

pair_idx_i, pair_idx_j = np.triu_indices(26, 1)
eps = 1e-7
RESTARTS = 10

def initial_guess(seed):
    np.random.seed(seed)
    positions = np.random.uniform(0.1, 0.9, (26, 2))
    radii = np.zeros(26)
    return np.concatenate((positions, radii))

def objective(v):
    return -np.sum(v[2*26:])

def constraints(v):
    centers = v[:2*26].reshape((26, 2))
    radii = v[2*26:]
    boundary = np.where(centers[:26,0] - radii >= 0)[0]
    boundary += np.where(1 - centers[:26,0] - radii >= 0)[0]
    boundary += np.where(centers[:26,1] - radii >= 0)[0]
    boundary += np.where(1 - centers[:26,1] - radii >= 0)[0]
    boundary = np.unique(boundary)
    pairwise = (centers[pair_idx_i] - centers[pair_idx_j])**2
    pairwise += (radii[pair_idx_i, np.newaxis] + radii[pair_idx_j])**2
    return np.concatenate((boundary, pairwise[:,0]))

bounds = [(0, 1)] * 2*26 + [(0, 0.5)] * 26

best_value = -np.inf
best_v = None
for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
            fun        = objective,
            x0         = v0,
            method     = 'SLSQP',
            constraints= {'type':'ineq', 'fun':constraints},
            bounds     = bounds,
            options    = {
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
          )
    if not res.success:
        continue
    value = -res.fun
    if value > best_value:
        best_value = value
        best_v = res.x

final_radii = np.maximum(0.0, best_v[2*26:] - eps)
best_centers = best_v[:2*26].reshape((26, 2))

return best_centers, final_radii, np.sum(final_radii)
```
