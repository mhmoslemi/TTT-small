import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import minimize

def objective_function(x):
    n = int(np.sqrt(len(x) // 2))
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    return -np.sum(radii)

def constraint_function(x):
    n = int(np.sqrt(len(x) // 2))
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    dist_matrix = squareform(pdist(centers))
    return dist_matrix - (radii[:, None] + radii)

def boundary_constraint(x):
    n = int(np.sqrt(len(x) // 2))
    centers = x[:2 * n].reshape(n, 2)
    return np.abs(centers) - (np.ones((n, 2)) * 0.5)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_guess = np.random.rand(52) * 0.75 + 0.125
    x0 = np.concatenate((np.random.rand(2 * n) * 0.75 + 0.125, np.zeros(n) * 0.1))
    
    constraints = [
        {'type': 'ineq', 'fun': lambda x: x[2 * n:]},  # Radii must be nonnegative
        {'type': 'ineq', 'fun': boundary_constraint}
    ]
    constraints += [{'type': 'ineq', 'fun': lambda x: constraint_function(x)[i, j] for i in range(n) for j in range(n) if i < j}]
    
    result = minimize(objective_function, x0, method='SLSQP', constraints=constraints, tol=1e-8)
    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0
    
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    return centers, radii, -result.fun
