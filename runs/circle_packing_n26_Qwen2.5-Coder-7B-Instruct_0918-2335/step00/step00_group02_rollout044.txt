import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial placement of circles in a hexagonal grid
    n = 26
    radius = 0.05
    centers = np.zeros((n, 2))
    radii = np.full(n, radius)
    
    # Coordinate for the centers
    x_offset = np.sqrt(3) * radius / 2
    y_offset = radius * 1.5
    row_count = 1
    for i in range(n):
        row = i % 2
        col = i // 2
        x = col * x_offset + (1 - row % 2) * x_offset / 2
        y = row * y_offset
        if i >= row_count * 26 // 2 - 1:
            row_count += 1
            # Adjust for overflowing
            col = (i - (row_count - 1) * 26 // 2) // (row_count - 1) + 1
            x = (col - 1) * x_offset + ((row_count - 1) % 2) * x_offset / 2
            y = (row_count - 1) * y_offset + 1 + radius
        centers[i] = [x, y]
    
    # Center of the unit square
    center = np.array([0.5, 0.5])
    # Scale down the coordinates to fit within the unit square
    centers = centers / 2 + center
    
    # Calculate the sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
