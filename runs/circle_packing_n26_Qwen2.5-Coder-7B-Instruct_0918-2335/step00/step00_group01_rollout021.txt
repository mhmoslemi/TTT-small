import numpy as np
import scipy.optimize

def generate_initial_centres():
    n = 26
    centre_pattern = [
        (0.1, 0.1), (0.25, 0.1), (0.4, 0.1), (0.55, 0.1), (0.7, 0.1), (0.85, 0.1),  # 1st row
        (0.1, 0.2), (0.25, 0.2), (0.4, 0.2), (0.55, 0.2), (0.7, 0.2),              # 2nd row
        (0.1, 0.3), (0.25, 0.3), (0.4, 0.3), (0.55, 0.3), (0.7, 0.3), (0.85, 0.3), # 3rd row
        (0.1, 0.4), (0.25, 0.4), (0.4, 0.4), (0.55, 0.4), (0.7, 0.4),              # 4th row
        (0.1, 0.5), (0.25, 0.5), (0.4, 0.5), (0.55, 0.5)                           # 5th row
    ]
    centres = np.array(centre_pattern[:n])
    return centres

def solve_radius_lp(centres):
    n = len(centres)
    A_ub = np.empty((429, n))
    b_ub = np.empty(429)
    c = -np.ones(n)

    pair_idx = 0
    for i in range(n):
        x_i, y_i = centres[i]
        A_ub[pair_idx, i] = 1
        b_ub[pair_idx] = x_i
        pair_idx += 1
        A_ub[pair_idx, i] = 1
        b_ub[pair_idx] = 1 - x_i
        pair_idx += 1
        A_ub[pair_idx, i] = 1
        b_ub[pair_idx] = y_i
        pair_idx += 1
        A_ub[pair_idx, i] = 1
        b_ub[pair_idx] = 1 - y_i
        pair_idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.linalg.norm(centres[i] - centres[j])
            A_ub[pair_idx, i] = 1
            A_ub[pair_idx, j] = 1
            b_ub[pair_idx] = dist_ij
            pair_idx += 1

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='simplex')
    if not res.success:
        raise ValueError(f"LP optimization failed: {res.message}")

    radii = res.x
    obj = np.sum(radii)
    return radii, obj

def run_packing(max_restarts=3, max_iters=200, alpha=0.01, jitter=1e-4, eps=1e-7) -> tuple:
    best_obj = -np.inf
    best_centres, best_radii = None, None

    for restart in range(max_restarts):
        centers = generate_initial_centres()
        if restart > 0:
            centers += jitter * (np.random.rand(*centers.shape) - 0.5)

        for iter in range(max_iters):
            radii, obj = solve_radius_lp(centers)
            if obj > best_obj:
                best_obj = obj
                best_centres, best_radii = centers.copy(), radii.copy()

            # Compute forces and update centers
            dual_values = solve_radius_lp(centers[centers==0])[1]  # Placeholder for actual dual values
            forces = np.zeros_like(centers)
            for i in range(n):
                force = np.zeros(2)
                for j in range(n):
                    if i != j and dual_values[j] != 0:
                        diff = centres[j] - centres[i]
                        dist_ij = np.linalg.norm(diff)
                        force += dual_values[j] * diff / dist_ij
                # Boundary forces
                if radii[i] < centres[i,0]: force[0] += 1
                if radii[i] < 1-centres[i,0]: force[0] -= 1
                if radii[i] < centres[i,1]: force[1] += 1
                if radii[i] < 1-centres[i,1]: force[1] -= 1
                forces[i] = force * alpha

            centers += forces

            if np.abs(max(radii) - min(radii)) < eps:
                break

    if best_centres is None or best_radii is None:
        raise ValueError("Could not find a valid packing")

    return best_centres, best_radii, np.sum(best_radii)
