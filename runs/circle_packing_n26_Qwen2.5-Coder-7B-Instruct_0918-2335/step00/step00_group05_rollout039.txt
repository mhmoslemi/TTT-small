import numpy as np
import scipy.spatial
import scipy.optimize
from scipy.spatial.distance import cdist

def compute_distances(centers):
    return cdist(centers, centers)

def solve_fixed_circle_system(centers, distances):
    r = np.zeros_like(centers[:, [0]])
    constraints = []
    
    # Boundary constraints
    for i, (x, y) in enumerate(centers):
        constraints.extend([{'type': 'ineq', 'fun': lambda r, i=i: r[i, 0] - y},
                           {'type': 'ineq', 'fun': lambda r, i=i: x - r[i, 0]},
                           {'type': 'ineq', 'fun': lambda r, i=i: y - r[i, 0]},
                           {'type': 'ineq', 'fun': lambda r, i=i: r[i, 0] - (1 - x)}])
    
    # Non-overlap constraints
    for i, (x1, y1) in enumerate(centers):
        for j, (x2, y2) in enumerate(centers):
            dist = distances[i, j]
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - (r[i, 0] + r[j, 0])})
    
    # Non-negativity constraint
    constraints.extend([{'type': 'ineq', 'fun': lambda r, i=i: r[i, 0]} for i in range(len(centers))])
    
    res = scipy.optimize.minimize(
        lambda r: -np.sum(r),
        r,
        method='SLSQP',
        constraints=constraints
    )
    
    return res.x

def run_packing():
    initial_center_spacing = 0.312
    lattice-spacing_range = (0.26, 0.35)
    initial_layout_found = False
    max_attempts = 20
    
    while not initial_layout_found and max_attempts > 0:
        lattice_spacing = np.random.uniform(lattice-spacing_range[0], lattice-spacing_range[1])
        lattice = []
        
        for k_x in range(-2, 3):
            for k_y in range(int(-k_x * (np.sqrt(3) / 2) * lattice_spacing), int(k_x * (np.sqrt(3) / 2) * lattice_spacing) + 1):
                x = k_x * lattice_spacing + initial_center_spacing
                y = k_y * (np.sqrt(3) / 2) * lattice_spacing - k_x * (np.sqrt(3) / 2) * lattice_spacing / 2 + initial_center_spacing
                
                if x >= 0 and x <= 1 and y >= 0 and y <= 1:
                    lattice.append([x, y])
        
        if len(lattice) >= 26:
            initial_layout_found = True
            break
        max_attempts -= 1
    
    if not initial_layout_found:
        raise ValueError("Failed to generate a valid initial lattice layout after multiple attempts.")
    
    initial_centers = np.array(lattice[:26])
    distances = compute_distances(initial_centers)
    
    solution = solve_fixed_circle_system(initial_centers, distances)
    
    return initial_centers, solution, np.sum(solution)

# Run the function and print the result
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of radii:", sum_radii)
