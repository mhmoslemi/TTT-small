import numpy as np
from scipy import optimize

def objective(z):
    return -np.sum(z[2::3])

def constraint.bounds(z):
    x_bounds = [(0, 1)] * 26
    y_bounds = [(0, 1)] * 26
    r_bounds = [(0, 0.5)] * 26

def constraint.boundary(z):
    b = np.zeros(4 * 26)
    for i in range(26):
        b[4 * i + 0] = z[3 * i + 0] - z[3 * i + 2]  # left side
        b[4 * i + 1] = 1 - z[3 * i + 0] - z[3 * i + 2]  # right side
        b[4 * i + 2] = z[3 * i + 1] - z[3 * i + 2]  # bottom side
        b[4 * i + 3] = 1 - z[3 * i + 1] - z[3 * i + 2]  # top side
    return b

def constraint.pairwise(z):
    n = 26
    g = np.zeros(n * (n - 1) // 2)
    k = 0
    for i in range(n):

            k += 1
    return g
