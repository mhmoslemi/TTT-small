import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    v_mag = 5e-3
    T_max = 50.0
    epsilon = 1e-6
    
    # Random initial positions and velocities
    centers = np.random.rand(n, 2)
    velocities = np.random.randn(n, 2) * v_mag
    
    # Initialize radii and times
    radii = np.ones(n) * r_min
    event_queue = []
    
    def distance(c1, c2):
        return np.linalg.norm(c1 - c2)
    
    def event_time(c1, v1, r1, c2, v2, r2):
        v_rel = v1 - v2
        delta_c = c1 - c2
        delta_r = r1 + r2
        sqrt_v_rel = np.linalg.norm(v_rel)
        if sqrt_v_rel == 0:
            return np.inf
        tcc = (-delta_c @ v_rel + np.sqrt(delta_c @ v_rel**2 - delta_c @ delta_c + 2 * delta_r * sqrt_v_rel)) / (2 * sqrt_v_rel)
        if tcc > 0:
            return tcc
        else:
            return np.inf
    
    def wall_collision_time(c, v, radius, wall):
        if wall == 'left':
            return (radius - c[0]) / (v[0] - g)
        elif wall == 'right':
            return (c[0] - (1 - radius)) / (v[0] - g)
        elif wall == 'bottom':
            return (radius - c[1]) / (v[1] - g)
        elif wall == 'top':
            return (c[1] - (1 - radius)) / (v[1] - g)
    
    def add_event(circle1, circle2):
        tcc = event_time(centers[circle1], velocities[circle1], radii[circle1], centers[circle2], velocities[circle2], radii[circle2])
        if not np.isinf(tcc):
            event_queue.append((tcc, circle1, circle2))
    
    def add_wall_events(circle):
        for wall in ['left', 'right', 'bottom', 'top']:
            tcw = wall_collision_time(centers[circle], velocities[circle], radii[circle], wall)
            if not np.isnan(tcw) and tcw >= 0:
                event_queue.append((tcw, circle, -1, wall))
    
    # Initialize event queue
    for i in range(n):
        for j in range(i + 1, n):
            add_event(i, j)
        add_wall_events(i)
    
    current_time = 0
    while event_queue and current_time < T_max:
        t_e, i, j, wall = np.heappop(event_queue)
        if t_e <= (current_time + 1e-9):
            dt = t_e - current_time
            centers[i] += velocities[i] * dt
            centers[j] += velocities[j] * dt
            radii[i] += g * dt
            radii[j] += g * dt
            if i < 0:
                wall_index = np.abs(i) - 1
                if wall == 'left':
                    centers[wall_index][0] = radii[wall_index]
                elif wall == 'right':
                    centers[wall_index][0] = 1 - radii[wall_index]
                elif wall == 'bottom':
                    centers[wall_index][1] = radii[wall_index]
                elif wall == 'top':
                    centers[wall_index][1] = 1 - radii[wall_index]
                velocities[wall_index][0] *= -1
            else:
                if i != j:
                    v_rel = velocities[i] - velocities[j]
                    delta_c = centers[i] - centers[j]
                    dot = v_rel @ delta_c
                    norm_v_rel = np.linalg.norm(v_rel)
                    norm_delta_c = np.linalg.norm(delta_c)
                    v_rel_proj = (dot / norm_delta_c) * delta_c / norm_v_rel
                    v_i_proj = np.dot(v_rel_proj, velocities[i])
                    v_j_proj = np.dot(v_rel_proj, velocities[j])
                    new_v_i_proj = v_i_proj - 2 * v_j_proj / (1 + 1)
                    new_v_j_proj = v_j_proj - 2 * v_i_proj / (1 + 1)
                    velocities[i] -= 2 * v_j_proj / (1 + 1) * v_rel_proj
                    velocities[j] -= 2 * v_i_proj / (1 + 1) * v_rel_proj
                centers[i] += velocities[i] * (t_e - current_time)
                centers[j] += velocities[j] * (t_e - current_time)
                radii[i] += g * (t_e - current_time)
                radii[j] += g * (t_e - current_time)
            current_time = t_e
            if i >= 0:
                add_event(i, j)
            if i >= 0 and 0 <= i < n:
                add_wall_events(i)
            if j >= 0 and 0 <= j < n:
                add_wall_events(j)
        else:
            break
    
    # Shrink radii and repair overlaps
    radii -= epsilon
    for i in range(n):
        for j in range(i + 1, n):
            if distance(centers[i], centers[j]) < radii[i] + radii[j]:
                overlap = (radii[i] + radii[j]) - distance(centers[i], centers[j])
                centers[i] += overlap / 2 * (centers[j] - centers[i]) / distance(centers[j] - centers[i])
                centers[j] -= overlap / 2 * (centers[j] - centers[i]) / distance(centers[j] - centers[i])
                radii[i] -= overlap / 2
                radii[j] -= overlap / 2
                if radii[i] < r_min:
                    radii[i] = r_min
                if radii[j] < r_min:
                    radii[j] = r_min
    
    # Solve LP to polish
    lp_conversion = np.array([[1 if j == i else 0 for j in range(n)] for i in range(n)])
    constraints = [{'type': 'ineq', 'fun': lambda centers, v=i, lp=lp_conversion: (centers[v][:2] - (np.array([1, 0]) * lp[v] @ radii))}, 
                   {'type': 'ineq', 'fun': lambda centers, v=i, lp=lp_conversion: (centers[v][:2] - (np.array([0, 1]) * lp[v] @ radii))} for i in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda centers, v=i, w=j: (np.sqrt(((centers[v][:2] - centers[w][:2]) @ (centers[v][:2] - centers[w][:2]))) - (radii[v] + radii[w]))})
    result = linprog(-radii, method='highs', constraints=constraints, bounds=[(0, r_min), (0, r_min)] * n)
    final_radii = -result.x if result.success else radii
    
    return centers, final_radii, np.sum(final_radii)
