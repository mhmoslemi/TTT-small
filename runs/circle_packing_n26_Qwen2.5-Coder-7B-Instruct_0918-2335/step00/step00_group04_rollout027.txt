import numpy as np
import scipy.spatial as spatial
from scipy.optimize import linprog

def run_packing():
    # Define the unit square corners as circles with zero radius
    walls = [(0, 0), (1, 0), (1, 1), (0, 1)]
    C = walls
    R = [0.5] * 4
    
    # Define the initial seed circles at the corners
    seed_radii = 0.25
    C += [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    R += [seed_radii] * 4
    
    # Perform greedy Voronoi-based insertion until we have 26 circles
    while len(C) < 26:
        # Compute the Voronoi diagram of the current circles and walls
        vor = spatial.Voronoi(C + walls)
        
        # Collect all Voronoi vertices
        Voronoi_vertices = np.array(vor.vertices)
        
        # Create a set to store candidate insertion points
        candidate_insertions = set()
        
        # Consider each vertex and choose the optimal one
        for vertex in Voronoi_vertices:
            # Compute the radius at this vertex in all directions
            radius_left = np.clip(vertex[0] - 0, 0, vertex[0])
            radius_right = np.clip(1 - vertex[0], 0, 1 - vertex[0])
            radius_top = np.clip(vertex[1] - 0, 0, vertex[1])
            radius_bottom = np.clip(1 - vertex[1], 0, 1 - vertex[1])
            
            # Find the minimum radius for non-overlapping and within the square
            min_radius = min(radius_left, radius_right, radius_top, radius_bottom)
            
            # Find intersections with neighboring obstacles
            intersections = []
            for obstacle in C + walls:
                if obstacle[0] == vertex[0]:
                    intersections.append(np.clip(obstacle[1], 0, 1))
                elif obstacle[1] == vertex[1]:
                    intersections.append(np.clip(obstacle[0], 0, 1))
            
            # Compute the minimum radius avoiding specific obstacles
            if intersections:
                min_radius = min(min_radius, *intersections)
            
            # Store the candidate insertion point
            if min_radius > 0:
                candidate_insertions.add((vertex, min_radius))
        
        # Select the candidate insertion with the maximum radius
        if candidate_insertions:
            best_insertion = max(candidate_insertions, key=lambda x: x[1])
            vertex, radius = best_insertion
            C.append(vertex)
            R.append(radius)
        else:
            break
