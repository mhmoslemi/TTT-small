import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    v_initial magnitude = 5e-3
    T_max = 50.0
    epsilon = 1e-6

    def quadratic_coefficients(circle1, circle2):
        c0 = circle1[:2] - circle2[:2]
        v = circle1[2:] - circle2[2:]
        r0 = circle1[3] + circle2[3]
        gtot = 2 * g
        A = np.dot(v, v)
        B = 2 * np.dot(c0, v)
        C = np.dot(c0, c0) - r0 * r0
        return A, B, C

    def compute_event_time(circle1, circle2):
        A, B, C = quadratic_coefficients(circle1, circle2)
        if A == 0:
            return np.inf
        t = (-B + np.sqrt(B * B - 4 * A * C)) / (2 * A)
        if t < 0:
            return np.inf
        return t

    def compute_wall_event_time(circle, wall):
        ecopy = circle.copy()
        if wall == 'left':
            ecopy[0] = 0
            ecopy[2] *= (ecopy[0] - circle[0]) / (circle[2] - ecopy[0])
        elif wall == 'right':
            ecopy[0] = 1
            ecopy[2] *= (ecopy[0] - circle[0]) / (circle[2] - ecopy[0])
        elif wall == 'bottom':
            ecopy[1] = 0
            ecopy[2] *= (ecopy[1] - circle[1]) / (circle[2] - ecopy[1])
        elif wall == 'top':
            ecopy[1] = 1
            ecopy[2] *= (ecopy[1] - circle[1]) / (circle[2] - ecopy[1])
        t = (ecopy[wall] - circle[wall]) / circle[2]
        if t < 0:
            return np.inf
        return t

    def resolve_collision(circle1, circle2):
        i, j = circle1[4], circle2[4]
        x1, y1, v1x, v1y, r1 = circle1[:5]
        x2, y2, v2x, v2y, r2 = circle2[:5]
        dx = x2 - x1
        dy = y2 - y1
        dvx = v2x - v1x
        dvy = v2y - v1y
        d = np.sqrt(dx * dx + dy * dy)
        d2 = d * d
        tau = (dx * dvx + dy * dvy) / d2
        dvtx = dvx - 2 * (dx * dvx + dy * dvy) / d2 * dx
        dvty = dvy - 2 * (dx * dvx + dy * dvy) / d2 * dy
        v1x, v1y = v1x + 0.5 * (dvtx - v1x), v1y + 0.5 * (dvty - v1y)
        v2x, v2y = v2x + 0.5 * (-dvtx - v2x), v2y + 0.5 * (-dvty - v2y)

        circle1[:5] = [x1 + v1x * g, y1 + v1y * g, v1x * (1 + g), v1y * (1 + g), r1 + g]
        circle2[:5] = [x2 + v2x * g, y2 + v2y * g, v2x * (1 + g), v2y * (1 + g), r2 + g]

    def resolve_wall_collision(circle, wall):

        e = x if wall == 'left' or wall == 'right' else y
        v = vx if wall == 'left' or wall == 'right' else vy

        t = (r - e) / a
        if t < 0:
            return
        circle[:5] = [x + vx * t, y + vy * t, -vx * t, -vy * t, r + g]

    def compute_event_queue(n, centers, radii):
        event_queue = []
        for i in range(n):
            for j in range(i + 1, n):
                tau = compute_event_time(centers[i], centers[j])
                event_queue.append((tau, i, j))
            for wall in ['left', 'right', 'bottom', 'top']:
                tau = compute_wall_event_time(centers[i], wall)
                event_queue.append((tau, i, wall))
        event_queue.sort()
        return event_queue

    def simulate_growth(centers, radii):
        n = centers.shape[0]
        velocities = np.random.normal(scale=v_initial_magnitude, size=(n, 2))
        event_queue = compute_event_queue(n, centers, radii)

        while len(event_queue) > 0 and event_queue[0][0] < T_max:
            tcc, i, j = event_queue[0]
            dt = tcc - 0.0
            centers[i] += velocities[i] * dt
            centers[j] += velocities[j] * dt
            radii[i] += g * dt
            radii[j] += g * dt

            if i < n and j < n:
                resolve_collision(centers[i], centers[j])
            elif i < n:
                resolve_wall_collision(centers[i], j)
            elif j < n:
                resolve_wall_collision(centers[j], i)

            del event_queue[0]
            for sj in range(n):
                if sj != i:
                    event_queue.append((compute_event_time(centers[i], centers[sj]), i, sj))
            for sw in ['left', 'right', 'bottom', 'top']:
                event_queue.append((compute_wall_event_time(centers[i], sw), i, sw))

        return centers

    # Initial setup
    centers = np.zeros((n, 5))
    centers[:, :2] = np.random.rand(n, 2)
    centers[:, 2:] = np.random.normal(scale=v_initial_magnitude, size=(n, 2))
    centers[:, 4] = np.arange(n)

    # Run growth simulation
    centers = simulate_growth(centers, np.ones(n) * r_min)

    # Apply shrinkage and repair
    collision_tolerance = 1e-3
    while True:
        shrinked = False
        for i in range(n):
            for j in range(i + 1, n):
                center_distance = np.linalg.norm(centers[i, :2] - centers[j, :2])
                overlap = (center_distance - centers[i, 3] - centers[j, 3]) * 0.5
                if overlap > collision_tolerance:
                    centers[i, :2] -= overlap * np.array(centers[j, :2] - centers[i, :2]) / center_distance
                    centers[j, :2] -= overlap * np.array(centers[i, :2] - centers[j, :2]) / center_distance
                    centers[i, 3] = r_min + (np.linalg.norm(centers[i, :2] - centers[j, :2]) - r_min)
                    centers[j, 3] = r_min + (np.linalg.norm(centers[i, :2] - centers[j, :2]) - r_min)
                    shrinked = True
        if not shrinked:
            break

    A = []
    b = []

            A.append([0, -1, 0, 0, 0])
            A.append([0, 0, 1, 0, 0])
            A.append([0, 0, 0, 1, 0])
            b.append(0)
            b.append(0)
            b.append(0)
            b.append(0)
        if i < n - 1:
            A.append([0, 0, 0, 0, -1])
            A.append([0, 0, 0, 0, 0])
            A.append([0, 0, 0, 0, 0])
            A.append([0, 0, 0, 0, 0])
            b.append(0)
            b.append(0)
            b.append(0)
            b.append(0)
        A.append([-1, 0, 0, 0, 0])
        A.append([0, -1, 0, 0, 0])
        A.append([0, 0, 1, 0, 0])
        A.append([0, 0, 0, 1, 0])
        b.append(0)
        b.append(0)
        b.append(0)
        b.append(0)
    A = np.array(A)
    b = np.array(b)
    bounds = [(r_min, 1.0) for _ in range(n)]
    result = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if result.success:
        max_radii = -result.fun
        max_centers = result.x.reshape(n, 4)[:, :2]
    else:
        max_radii = np.sum(radii)
        max_centers = centers[:, :2]

    return max_centers, result.x, max_radii
