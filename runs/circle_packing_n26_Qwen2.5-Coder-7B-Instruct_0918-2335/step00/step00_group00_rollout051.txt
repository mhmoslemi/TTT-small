import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def init_circles(variant):
        centers = np.array([
            (0.1, 0.1), (0.9, 0.1), (0.3, 0.3), (0.7, 0.3), (0.1, 0.5),
            (0.9, 0.5), (0.3, 0.7), (0.7, 0.7), (0.1, 0.9), (0.9, 0.9),
            (0.2, 0.2), (0.8, 0.2), (0.4, 0.4), (0.6, 0.4), (0.0, 0.6),
            (1.0, 0.6), (0.2, 0.8), (0.8, 0.8), (0.4, 1.0), (0.6, 1.0),
        ])
        if variant == 'A':
            centers = np.vstack([centers, [0.6, 0.9]])
        elif variant == 'B':
            centers = np.vstack([centers, [0.4, 0.9]])
        radii = np.full(26, 0.01)
        return centers, radii

    def inflate(circles, Δr, ε):
        max_iter = 5000
        for _ in range(max_iter):
            changes = False
            for i in range(26):
                for j in range(i + 1, 26):
                    dx = circles[i, 0] - circles[j, 0]
                    dy = circles[i, 1] - circles[j, 1]
                    d = np.sqrt(dx**2 + dy**2)
                    if d < circles[i, 2] + circles[j, 2] - ε:
                        overlap = circles[i, 2] + circles[j, 2] - d
                        circles[i, 0] += overlap * dx / (2 * d)
                        circles[i, 1] += overlap * dy / (2 * d)
                        circles[j, 0] -= overlap * dx / (2 * d)
                        circles[j, 1] -= overlap * dy / (2 * d)
                        changes = True
            circles[:, 0] = np.clip(circles[:, 0], circles[:, 2], 1 - circles[:, 2])
            circles[:, 1] = np.clip(circles[:, 1], circles[:, 2], 1 - circles[:, 2])
            circles[:, 2] += Δr
            circles[:, 2] = np.clip(circles[:, 2], 0, np.minimum(circles[:, 0], 1 - circles[:, 0], circles[:, 1], 1 - circles[:, 1]))
            if not changes:
                break
        for i in range(26):
            for j in range(i + 1, 26):
                dx = circles[i, 0] - circles[j, 0]
                dy = circles[i, 1] - circles[j, 1]
                d = np.sqrt(dx**2 + dy**2)
                if d < circles[i, 2] + circles[j, 2]:
                    circles[i, 2] -= (circles[i, 2] + circles[j, 2] - d) / 2
                    circles[j, 2] -= (circles[i, 2] + circles[j, 2] - d) / 2
        return circles

    best_centers, best_radii, best_sum = None, None, 0.0
    for variant in ['A', 'B']:
        centers, radii = init_circles(variant)
        circles = np.hstack((centers, radii.reshape(-1, 1)))
        circles = inflate(circles, 0.0005, 1e-6)
        sum_radii = circles[:, 2].sum()
        if sum_radii > best_sum:
            best_centers, best_radii, best_sum = circles[:, :2], circles[:, 2], sum_radii

    return best_centers, best_radii, best_sum
