import numpy as np
from scipy.optimize import least_squares

def hex_lattice(s, n):
    x = np.arange(s/2, 1 - s/2, s)
    y = np.concatenate((np.arange(s/2, 1 - s/2 + s * np.sqrt(3)/2, s * np.sqrt(3)/2),
                       np.arange(s/2 + s/2, 1 - s/2 + s * np.sqrt(3)/2, s * np.sqrt(3)/2)))
    mask = np.ravel_multi_index((np.outer((np.arange(len(x)),)*len(y)), 
                                 np.outer(y, (np.arange(len(x)),))), (len(y), len(x))) < n
    xy = np.column_stack((x[mask], y[mask]))
    r = np.full(xy.shape[0], s / 2)
    return xy, r

def project(x, s):
    return np.clip(x, s / 2, 1 - s / 2)

def constraint_f(z, s, δ):
    x, y, r = z.reshape((-1, 2)), s / 2, s / 2
    xy = np.column_stack((x[:, 0], y[:, 0]))
    prob = least_squares(lambda x: np.concatenate((x[:, 0] - r - s / 2, r - s / 2 + x[:, 0],
                                                   x[:, 1] - r - s / 2, r - s / 2 + x[:, 1],
                                                   (np.sqrt(np.sum((xy[:, None, :] - xy[:, :, None])**2, axis=2)
                                                         - (r[:, None] + r[:, :, None]))**2 - δ**2)),
                                                axis=0),
                          np.concatenate((x[:, 0], x[:, 1], r)), bounds=(s / 2, 1 - s / 2))
    z_new = np.concatenate((prob.x[:len(x)], prob.x[len(x):2*len(x)], prob.x[2*len(x):]))
    return z_new, prob.status

def inflate_balls(z, s, K):
    z_new = np.copy(z)
    x, y, r = z_new.reshape((-1, 2)), s / 2, s / 2
    for i in range(len(r)):
        min_dist = np.inf
        for j in range(len(r)):
            if i != j:
                dist = np.sqrt((x[i, 0] - x[j, 0])**2 + (y[i, 0] - y[j, 0])**2)
                if dist < min_dist:
                    min_dist = dist
        new_r = min(min_dist, 1 - x[i, 0], 1 - x[i, 0], 1 - y[i, 0], y[i, 0])
        if new_r <= r[i]:
            continue
        z_new[i, 2] = new_r
        new_x = z_new[i, :2]
        for j in range(len(r)):
            if i != j:
                dist = np.sqrt((x[i, 0] - x[j, 0])**2 + (y[i, 0] - y[j, 0])**2)
                if dist < r[i] + r[j]:
                    alpha = (new_r - r[j]) / (dist - 1e-12)
                    z_new[i, :2] = new_x + alpha * (x[j] - x[i])
    return z_new

def solve_stage(z0, δ):
    s = np.sqrt(2 * len(z0) / 3)
    z, status = constraint_f(z0, s, δ)
    while status != 4 and z is not None and z.shape[0] == len(z0):
        z = inflate_balls(z0, s, 1e6)
        z, status = constraint_f(z, s, δ)
    return z

def run_packing():
    s = np.sqrt(2 * 26 / 3)
    initialguess, _ = hex_lattice(s, 26)
    z0 = np.concatenate((initialguess.flatten(), np.zeros(26)))
    x, status = solve_stage(z0, 1e-8)
    z0 = x
    for δ in np.logspace(7, -1, 9)[1:]:
        z0 = solve_stage(z0, δ)
    result = inflate_balls(z0, s, 1e6)
    result = optimize(result)
    centers = result.reshape((-1, 2))
    radii = result[2::3]
    sum_radii = radii.sum()
    return centers, radii, sum_radii

def optimize(result):
    bounds = []
    for _ in range(8):
        bounds.append(result)

    def objective(x):
        centers = x.reshape((-1, 2))
        radii = x[2::3]
        return -radii.sum()

    res = least_squares(objective, result, bounds=bounds, gtol=1e-10)
    return res.x

# Run the solution
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
