import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.array([
        [r_c, r_c], [0.5 - r_e, r_c], [0.5 + r_e, r_c],
        [1 - r_c, r_c], [0.5, 1 - r_e], [0.5, 1 + r_e],
        [r_c, 0.5], [1 - r_c, 0.5], [0.5, 0.5]
    ])
    edge_centers = np.array([
        [0.5, r_e], [1 - r_e, 0.5], [0.5, 1 - r_e], [r_e, 0.5]
    ])
    inner_centers = np.array([
        [0.5 + r_i * np.cos(2 * np.pi * k / 6), 0.5 + r_i * np.sin(2 * np.pi * k / 6)]
        for k in range(6, 36)
    ])
    centers = np.vstack((centers, edge_centers, inner_centers))
    radii = np.repeat([r_c, r_e, r_i], [1, 1, 20])
    return centers, radii

def compute_sum_radii(radii):
    return sum(radii)

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)  # Minimize negative sum

def constraint_func(v):
    return np.array([
        # boundary constraints for each circle
        min(0.5, v[i*3+1], 1-v[i*3+1], 0.5, 1-v[i*3+2]) - v[i*3+3] for i in range(26)
    ])

def local_obj(x):
    n = len(radii)
    x_centers = x[:2*n].reshape(n, 2)
    x_radii = x[2*n:]
    centers = np.vstack((x_centers, edge_centers, inner_centers))
    radii = np.repeat([x_radii[:2], x_radii[2:3], x_radii[3:]], [1, 1, 20])
    return -compute_sum_radii(radii)  # Minimize negative sum

def run_packing():
    bounds = [
        (0.3, 0.49),
        (0.1, 0.25),
        (0.05, 0.15)
    ]
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [
        (0, 0.5),
        (0, 0.5)
    ] * 26
    constraints = [
        {'type': 'ineq', 'fun': lambda v, r_i=r_i: np.array([
            min(0.5, v[i*3+1], 1-v[i*3+1], 0.5, 1-v[i*3+2]) - v[i*3+3] for i in range(26)
        ])}
    ]
    if result_de.success:
        res = minimize(local_obj, x0, method='SLSQP', bounds=bounds_slsqp, constraints=constraints, options={'ftol':1e-10, 'gtol':1e-10, 'maxiter':2000})
        centers = res.x[:2*26].reshape(26,2)
        radii = res.x[2*26:]
        return centers, radii, radii.sum()
    else:
        return np.random.rand(26,2), np.random.rand(26), 0.0
