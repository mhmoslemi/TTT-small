import numpy as np
from scipy.optimize import minimize

def generate_initial_centres(variant):
    if variant == 'A':
        centres = np.array([
            (0.1, 0.1), (0.5, 0.1), (0.9, 0.1),
            (0.05, 0.3), (0.45, 0.3), (0.85, 0.3),
            (0.1, 0.5), (0.5, 0.5), (0.9, 0.5),
            (0.05, 0.7), (0.45, 0.7), (0.85, 0.7),
            (0.1, 0.9), (0.5, 0.9), (0.6, 0.6), (0.9, 0.9)
        ])
    elif variant == 'B':
        centres = np.array([
            (0.1, 0.1), (0.5, 0.1), (0.9, 0.1),
            (0.05, 0.3), (0.45, 0.3), (0.85, 0.3),
            (0.1, 0.5), (0.5, 0.5), (0.9, 0.5),
            (0.05, 0.7), (0.45, 0.7), (0.85, 0.7),
            (0.1, 0.9), (0.5, 0.9), (0.4, 0.6), (0.9, 0.9)
        ])
    else:
        raise ValueError("Invalid variant")
    return centres, np.zeros(26)

def total_radius(params):
    centres = params[:52].reshape((26, 2))
    radii = params[52:]
    return -np.sum(radii)  # Minimize negative total radius

def non_overlap_fun(params):
    centres = params[:52].reshape((26, 2))
    radii = params[52:]
    n = len(radii)
    fun = []
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1]
            dij = np.sqrt(dx**2 + dy**2)
            fun.append(radii[i] + radii[j] - dij)
    return fun

def boundary_fun(params):
    centres = params[:52].reshape((26, 2))
    radii = params[52:]
    n = len(radii)
    fun = []
    for i in range(n):
        fun.append( centres[i, 0] - radii[i])  # left boundary
        fun.append( 1 - centres[i, 0] - radii[i])  # right boundary
        fun.append( centres[i, 1] - radii[i])  # bottom boundary
        fun.append( 1 - centres[i, 1] - radii[i])  # top boundary
    return fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for variant in ['A', 'B']:
        centres, radii = generate_initial_centres(variant)
        bounds = [(0.01, 0.49)] * 52  # small radii initially, adjust later

        def objective(params):
            return total_radius(params)

        def constraints(params):
            return [non_overlap_fun(params), boundary_fun(params)]

        init_params = np.hstack((centres.flatten(), radii))

        res = minimize(objective, init_params, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 5000})

        if res.fun < best_sum:
            best_sum = res.fun
            best_centers = res.x[:52].reshape((26, 2))
            best_radii = res.x[52:]

    if best_sum > -np.inf:
        # Post-processing to ensure non-overlap and boundary constraints
        for _ in range(10):
            for i in range(26):
                for j in range(i + 1, 26):
                    dx, dy = best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1]
                    dij = np.sqrt(dx**2 + dy**2)
                    if dij < best_radii[i] + best_radii[j]:
                        overlap = best_radii[i] + best_radii[j] - dij
                        dx /= dij
                        dy /= dij
                        best_centers[i] -= 0.5 * overlap * dx
                        best_centers[i] *= np.where(best_centers[i, 0] >= 0.5, 1, -1)
                        best_centers[j] += 0.5 * overlap * dy
                        best_centers[j] *= np.where(best_centers[j, 0] <= 0.5, 1, -1)

            for i in range(26):
                min_dist = min(best_centers[i, 0] + best_radii[i], 1 - best_centers[i, 0] - best_radii[i], best_centers[i, 1] + best_radii[i], 1 - best_centers[i, 1] - best_radii[i])
                best_radii[i] = min(best_radii[i], min_dist)

        return best_centers, best_radii, -best_sum
    else:
        # Fallback to equal radii arrangement of hex-lattice with known suboptimal sum
        centres = np.array([
            (i*(3/2)/sqrt(3), (3*(1+i//2)) / 2) for i in range(13)
        ]).clip(0, 1)
        centres = np.vstack([centres, (0.5, 0.5), (0.6, 0.6)])  # Add extra in corner
        radii = np.tile(np.sqrt(3)/6, 26)
        return centres, radii, sum(radii)
