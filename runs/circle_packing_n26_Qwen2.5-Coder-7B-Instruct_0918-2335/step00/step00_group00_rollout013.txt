import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    R = 200
    resolution = (R, R)
    delta = 1e-3
    N_refine = 200
    initial_radius = 0.01
    epsilon = 1e-9
    
    # Generate grid of candidate points
    x = np.linspace(0, 1, R + 1)[1:-1]
    y = np.linspace(0, 1, R + 1)[1:-1]
    grid_x, grid_y = np.meshgrid(x, y)
    grid_x, grid_y = grid_x.flatten(), grid_y.flatten()
    grid_points = np.vstack((grid_x, grid_y)).T
    
    # Precompute distances to the sides
    d_boundary = np.min(np.column_stack((grid_x, 1 - grid_x, grid_y, 1 - grid_y)), axis=1)
    
    # Initialize the lists for circle centers and radii
    centers = []
    radii = []
    
    for _ in range(26):
        # Find the index with the largest feasible radius
        r_sorted = d_boundary - epsilon  # Subtract a small epsilon to ensure strict interior
        r_sorted[r_sorted < initial_radius] = initial_radius  # Ensure minimum radius
        idx = np.argmax(r_sorted)
        
        center = grid_points[idx]
        radius = r_sorted[idx]
        
        # Place the circle
        centers.append(center)
        radii.append(radius)
        
        # Update the grid
        distances = np.sqrt(np.sum((grid_points - center) ** 2, axis=1))
        r_candidate = np.minimum(d_boundary - epsilon, d_boundary - distances)
        r_candidate[distances < radius] = -1e9  # Mark used points as invalid
        d_boundary = np.minimum(d_boundary, distances - radius)
        r_boundary = d_boundary - epsilon  # Ensure non-negative radii
    
    # Perform local radius and centre refinement
    sums = np.zeros(N_refine)
    for t in range(N_refine):
        for i in np.random.permutation(26):
            r_i = radii[i]
            center = centers[i]
            
            # Propose new centre
            new_center = center + (np.random.uniform(-delta, delta, 2))
            new_center = np.clip(new_center, 0, 1)  # Clamp to [0, 1]
            
            # Compute new feasible radius
            distances = np.sqrt(np.sum((grid_points - new_center) ** 2, axis=1))
            new_radius = np.minimum(d_boundary - epsilon, d_boundary - distances)
            new_radius[distances < r_i] = -1e9  # Mark used points as invalid
            new_radius[new_radius < r_i] = 0  # Constrain new radius
            new_radius = np.min(new_radius, axis=0)  # Circle touches boundary or another circle
            
            # Accept move if it increases total sum
            if new_radius > r_i:
                centers[i] = new_center
                radii[i] = new_radius
                sums[t] += new_radius
    
    # Recompute and clip radii
    temp = np.min(np.column_stack((d_boundary - epsilon, 
                                     np.sqrt(np.sum([(centers[:, None] - centers[i])[[j] for j in range(26) if j != i]]**2, axis=1)), 
                                     np.sum(np.subtract(d_boundary, radii)[np.newaxis, :], axis=0))), axis=1)
    radii = npclip(temp, a_min=0.0)
    
    sum_radii = np.sum(radii)
    
    # Convert to numpy arrays and expand dimensions
    centers = np.array(centers)
    radii = np.array(radii)
    
    return centers, radii, sum_radii
