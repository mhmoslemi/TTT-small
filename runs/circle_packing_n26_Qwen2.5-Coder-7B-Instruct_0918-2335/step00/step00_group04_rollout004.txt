import numpy as np
from scipy.optimize import differential_evolution, minimize

def penalised_objective(z):
    r = z[52:]
    penalty = np.sum(np.where(r > 0.5, np.inf, 0))
    penalty += np.sum((np.sum((-r + np.maximum(0, 1 - z[::2] - r)), axis=1) ** 2 +
                        np.sum((-r + np.maximum(0, 1 - z[1::2] - r)), axis=1) ** 2) * 10000)
    penalty += np.sum((np.sqrt(np.sum((z[::2][np.newaxis, :] - z[::2][:, np.newaxis]) ** 2, axis=2) +
                               np.sum((z[1::2][np.newaxis, :] - z[1::2][:, np.newaxis]) ** 2, axis=2)) -
                        (r[:, np.newaxis] + r[np.newaxis, :])) ** 2)
    return -np.sum(r) + penalty
