```python
import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog
from math import sqrt

# Parameters and utility functions
T_0 = 0.12
T_min = 1e-6
c = 0.995
M = 200
sigma_move = 0.024  # Changed from one-tenth of initial temperature
epsilon = 1e-12

def generate_hexagonal_lattice():
    s = 0.312
    k = np.indices((10, 10))
    x = s * (k[1] + (k[0] % 2) / 2)
    y = s * sqrt(3) * k[0] / 2
    idxs = ((x >= s) & (x <= 1-s) & (y >= s) & (y <= 1-s)).flatten()
    centers = np.column_stack((x[idxs].reshape(-1, 2)[:, 0], y[idxs].reshape(-1, 2)[:, 1]))
    return centers[:26]

def objective(radii):
    return -np.sum(radii)

def constraints(radii, distances):
    non_overlap = radii[:, None] + radii[None, :] <= distances - epsilon
    boundary = np.minimum(radii, np.minimum(np.minimum(np.minimum(x, 1-x), y), 1-y)) <= radii
    return np.vstack((non_overlap, boundary)).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = generate_hexagonal_lattice()
    n = centers.shape[0]
    distances = distance_matrix(centers, centers)
    
    best_sum = 0
    best_centers = centers
    current_centers = centers
    
    T = T_0
    while T > T_min:
        for _ in range(M):
            proposed_center = current_centers.copy()
            idx = np.random.randint(n)
            dx, dy = np.random.normal(0, sigma_move, 2)
            proposed_center[idx] += [dx, dy]
            proposed_center[idx] = np.clip(proposed_center[idx], epsilon, 1 - epsilon)
            
            proposed_distances = distance_matrix(proposed_center, proposed_center)
            problem = linprog(
                objective, 
                A_ub=constraints(radii, proposed_distances), 
                b_ub=(np.zeros_like(distances.flat) - epsilon)[np.triu_indices(n)],
                method='highs'
            )
            if not np.isnan(problem.fun):
                proposed_sum = -problem.fun
                if proposed_sum > best_sum:
                    best_sum = proposed_sum
                    best_centers = proposed_center
                    current_centers = proposed_center
                
        T *= c
    
    return best_centers, np.minimum(best_centers[:, 0] - best_centers[:, 1], np.minimum(best_centers[:, 0], np.minimum(best_centers[:, 1], 1 - best_centers[:, 0]))) - epsilon, best_sum
```
