```python
import numpy as np
from scipy.optimize import minimize

n = 26
pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)
bounds = np.block([[(0, 1)] * n, [(0, 0.5)] * n])

def initial_guess(seed):
    np.random.seed(seed)
    return np.concatenate((np.random.uniform(0.1, 0.9, 2 * n), np.zeros(n)))

def objective(v):
    return -np.sum(v[2 * n:])

def constraints(v):
    centers = v[:2 * n].reshape((n, 2))
    radii = v[2 * n:]
    boundary = np.concatenate((centers[:, 0] - radii, 1 - centers[:, 0] - radii,
                              centers[:, 1] - radii, 1 - centers[:, 1] - radii))
    pairwise = ((centers[pair_idx_i] - centers[pair_idx_j]) ** 2).sum(axis=1) - (radii[pair_idx_i] + radii[pair_idx_j]) ** 2
    return np.concatenate((boundary, pairwise))

RESTARTS = 10
best_value = -np.inf
best_v = None
eps = 1e-7

for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun=objective,
        x0=v0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if not res.success:
        continue
    value = -res.fun
    if value > best_value:
        best_value = value
        best_v = res.x

centers = best_v[:2 * n].reshape((n, 2))
radii = np.maximum(0.0, best_v[2 * n:] - eps)

return centers, radii, np.sum(radii)
```
