import numpy as np
from scipy.spatial.distance import pdist
from scipy.optimize import linprog
import heapq
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    R_min = 10**-4
    g = 10**-3
    velMag = 5 * 10**-3
    T_max = 50.0
    epsilon = 10**-6

    # Initialize centers randomly
    centers = np.random.rand(n, 2) * (1 - 2 * R_min) + R_min

    # Initialize velocities randomly
    vels = np.random.rand(n, 2)
    vels /= np.linalg.norm(vels, axis=1, keepdims=True)

    radii = np.full(n, R_min)
    for i in range(n):
        centers[i] += vels[i] * velMag

    # Compute events
    def event_times(i, j, vels, radii, T_max):
        dx, dy = vels[i, :] - vels[j, :]
        dr = radii[i] + radii[j]
        B = np.array([[dx[-1], -dy[-1]], [-dy[-1], dx[-1]]])
        b = np.array([-dr**2 + (radii[i] - radii[j])**2 + 2*np.sum(vels[i,:] * vels[j,:], axis=0)])
        tcc = np.linalg.solve(B, b) / (2 * (np.linalg.norm(dx)**2 + np.linalg.norm(dy)**2))
        if tcc[0] < 0 or np.isnan(tcc):
            return (np.inf, np.inf)
        else:
            return (tcc[0], tcc[1])

    events = []
    for i in range(n):
        for j in range(i + 1, n):
            event_i = (i, j, *event_times(i, j, vels, radii, T_max))
            event_j = (j, i, *event_times(j, i, vels, radii, T_max))
            heapq.heappush(events, event_i)
            heapq.heappush(events, event_j)
        wall_collisions = []
        for axis in range(2):
            wall_collisions.append((i, -1, (radii[i] - centers[i, axis]) / (vels[i, axis] - g)))
            wall_collisions.append((i, -2, (1 - (radii[i] + centers[i, axis])) / (vels[i, axis] + g)))
        wall_collisions = [(i, -1, t) for (i, _, t) in wall_collisions if t > 0 and not np.isnan(t)]
        events.extend(wall_collisions)

    # Sort events by time
    events.sort(key=lambda x: x[3])

    # Simulation loop
    current_time = 0.0
    while events and current_time < T_max:
        ti, tj, tc, twx, twy = heapq.heappop(events)
        if ti < 0:
            delta_t = twx
        else:
            delta_t = tc
        current_time += delta_t
        for i in range(n):
            centers[i] += vels[i] * delta_t
            radii[i] += g * delta_t

        if ti >= 0 and tj >= 0:
            x1, y1 = centers[ti]
            r1, r2 = radii[ti], radii[tj]
            d = np.linalg.norm(centers[ti] - centers[tj])
            if d == r1 + r2:
                uij = (centers[ti] - centers[tj]) / d
                v1_new = vels[ti] - 2 * uij * np.dot(vels[ti] - vels[tj], uij)
                vels[ti] = v1_new
                vels[tj] = 2 * uij * np.dot(vels[tj] - vels[ti], uij)
                heapq.heappush(events, event_times(ti, tj, vels, radii, T_max))
                heapq.heappush(events, event_times(tj, ti, vels, radii, T_max))
        elif ti >= 0:
            t_dir = -1 if tj < 0 else 2
            x, y = centers[ti]
            r = radii[ti]
            if t_dir == -1:
                r_new = 1 - min(x, 1 - x)
                vels[ti, t_dir] = -vels[ti, t_dir]
                heapq.heappush(events, (ti, t_dir, (r_new - r) / g))
            else:
                r_new = 1 - min(y, 1 - y)
                vels[ti, t_dir] = -vels[ti, t_dir]
                heapq.heappush(events, (ti, t_dir, (r_new - r) / g))
        elif tj >= 0:
            t_dir = -1 if ti < 0 else 2
            x, y = centers[tj]
            r = radii[tj]
            if t_dir == -1:
                r_new = 1 - min(x, 1 - x)
                vels[tj, t_dir] = -vels[tj, t_dir]
                heapq.heappush(events, (tj, t_dir, (r_new - r) / g))
            else:
                r_new = 1 - min(y, 1 - y)
                vels[tj, t_dir] = -vels[tj, t_dir]
                heapq.heappush(events, (tj, t_dir, (r_new - r) / g))

    # Shrink radii by epsilon
    radii -= epsilon

    # Repair any overlaps
    while pdist(centers, 'euclidean') < np.sum(radii):
        for i in range(n):
            for j in range(i + 1, n):
                if radii[i] > radii[j]:
                    dx, dy = centers[i] - centers[j]
                    d = np.linalg.norm(dx, dy)
                    overlap = (d - radii[i] - radii[j]) * 0.5
                    centers[i, 0] += -overlap * dx / d
                    centers[i, 1] += -overlap * dy / d
                    radii[i] -= overlap
                    radii[j] -= overlap

    # Polishing with LP
    c = -radii
    A = np.vstack([np.eye(n), -np.eye(n), np.ones((1, n)), -np.ones((1, n))])
    b = np.hstack([radii, radii, np.zeros((1, 1)), np.ones((1, 1))])
    Aeq = pdist(centers, 'euclidean')
    beq = np.full(len(Aeq), sum(radii))
    res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=(0, None), method='highs')
    radii = res.x

    return centers, radii, np.sum(radii)
