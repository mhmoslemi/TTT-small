import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def objective(x):
        rs = x[2 * N:]
        sum_r = -rs.sum()
        p_boundary = (np.maximum(0, rs - xs)**2 +
                      np.maximum(0, xs + rs - 1)**2 +
                      np.maximum(0, rs - ys)**2 +
                      np.maximum(0, ys + rs - 1)**2).sum()
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist = np.hypot(dx, dy)
        rr = rs[:, None] + rs[None, :]
        overlap = rr - dist
        p_overlap = np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
        return sum_r + lambda_penalty * (p_boundary + p_overlap)

    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400
    random_seeds = [0, 1, 2, 3]

    best_x = None
    best_fun = float('inf')

    for seed in random_seeds:
        np.random.seed(seed)
        bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N
        result = differential_evolution(objective, bounds, popsize=DE_popsize, maxiter=DE_maxiter, polish=False)
        if result.fun < best_fun:
            best_x = result.x
            best_fun = result.fun

    x = best_x
    xs = x[:N * 2:2]
    ys = x[:N * 2:2]
    radii = x[2 * N:]
    centres = np.column_stack([xs, ys])
    sum_radii = radii.sum()

    return centres, radii, sum_radii
