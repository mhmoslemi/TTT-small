import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def penalties(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    obj = -np.sum(r)
    wall_penalty = 0
    for i in range(n):
        wall_penalty += max(0, r[i] - x[i]) + max(0, 1 - x[i] - r[i]) + max(0, r[i] - y[i]) + max(0, 1 - y[i] - r[i])
    pair_penalty = 0
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            pair_penalty += max(0, r[i] + r[j] - d)**2
    return obj + 10000 * (wall_penalty + pair_penalty)

def constraints(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: x[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - x[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: y[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - y[i] - r[i]})
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: np.sqrt((z[i] - z[j])**2 + (z[n+i] - z[n+j])**2) - (z[2*n+i] + z[2*n+j])})
    return cons

def local_refinement(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    res = minimize(penalties, z, method='SLSQP', bounds=[(0, 1),] * 3*n, constraints=constraints(z), options={'maxiter': 2000, 'ftol': 1e-12})
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    bounds = [(0, 1)] * 3 * 26
    n = 26
    z0 = np.random.rand(2*n + n) * 0.9 + 0.05
    z0[52:78] = 0.12
    res = differential_evolution(penalties, bounds, popsize=20, mutation=(0.6, 1), recombination=0.7, maxiter=2000, tol=1e-8)
    if penalties(res.x) < 1e-8:
        z_opt = res.x
    else:
        z_opt = local_refinement(res.x)
    x_opt = z_opt[:n]
    y_opt = z_opt[n:2*n]
    r_opt = z_opt[2*n:]
    centers = np.vstack((x_opt, y_opt)).T
    radii = r_opt
    return centers, radii, np.sum(radii)
