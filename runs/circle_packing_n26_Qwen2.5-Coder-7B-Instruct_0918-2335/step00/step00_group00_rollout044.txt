import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialization
    R = 200
    grid = np.array([[x, y] for x in np.linspace(0, 1, R) for y in np.linspace(0, 1, R)])
    r_f = np.minimum(grid[:, 0], 1 - grid[:, 0])
    r_f = np.minimum(r_f, np.minimum(grid[:, 1], 1 - grid[:, 1]))
    
    centers = []
    radii = []
    EPS = 1e-9

    # Circle insertion loop
    for _ in range(26):
        k = np.argmax(r_f)
        center = grid[k]
        radius = r_f[k] - EPS
        centers.append(center)
        radii.append(radius)
        dx, dy = center - grid[:, [0, 1]]
        dist = np.sqrt(dx**2 + dy**2) - radius
        r_f = np.minimum(r_f, dist)
        r_f[k] = -1E9

    centers = np.array(centers)
    radii = np.array(radii)

    # Local refinement
    δc = 1e-3
    N_refine = 200
    for _ in range(N_refine):
        i = np.random.randint(26)
        center = centers[i]
        proposal = center + np.random.uniform(-δc, δc, 2)
        dx, dy = proposal - centers
        dist = np.sqrt(dx**2 + dy**2) - radii
        r_p = np.minimum(np.minimum(proposal[0], 1 - proposal[0]),
                         np.minimum(proposal[1], 1 - proposal[1]))
        r_p = np.minimum(r_p, dist)
        if r_p > radii[i]:
            centers[i] = proposal
            radii[i] = r_p

    # Final radius recomputation
    r_f = np.minimum(centers[:, 0], 1 - centers[:, 0])
    r_f = np.minimum(r_f, centers[:, 1])
    r_f = np.minimum(r_f, 1 - centers[:, 1])
    for i in range(26):
        dx, dy = centers[i] - centers
        dist = np.sqrt(dx**2 + dy**2) - radii
        r_f = np.minimum(r_f, dist)

    radii = np.minimum(radii, r_f)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
