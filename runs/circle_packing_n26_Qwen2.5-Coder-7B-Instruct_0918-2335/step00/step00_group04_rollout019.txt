import numpy as np
import scipy.optimize

def run_packing():
    np.random.seed(0)
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    def penalised_objective(z):
        x, y, r = z[:n], z[n:2*n], z[2*n:]
        sum_radii = -np.sum(r)
        penalty = np.sum(0.0 * np.clip(r, 0, None))  # Always zero
        penalty += np.sum(np.maximum(x - r - 0, 0)**2)
        penalty += np.sum(np.maximum(x + r - 1, 0)**2)
        penalty += np.sum(np.maximum(y - r - 0, 0)**2)
        penalty += np.sum(np.maximum(y + r - 1, 0)**2)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                penalty += np.maximum(r[i] + r[j] - d, 0)**2
        return sum_radii + 10000 * penalty

    result = scipy.optimize.differential_evolution(penalised_objective, bounds, popsize=20, strategy='rand1bin', mutation=0.6, recombination=0.7, tol=1e-8, maxiter=2000)
    z_best = result.x
    
    if result.success:
        x, y, r = z_best[:n], z_best[n:2*n], z_best[2*n:]
    else:
        z_opt = scipy.optimize.minimize(penalised_objective, z_best, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda z: -10000 * penalised_objective(z)})
        x, y, r = z_opt.x[:n], z_opt.x[n:2*n], z_opt.x[2*n:]
    
    centers = np.vstack((x, y)).T
    sum_radii = np.sum(r)
    
    return centers, r, sum_radii
