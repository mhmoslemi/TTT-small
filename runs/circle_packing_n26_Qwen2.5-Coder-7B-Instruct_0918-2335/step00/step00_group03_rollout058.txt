import numpy as np
from scipy.optimize import differential_evolution, linprog, SLSQP

def evaluate(pos):

    d = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=-1)

            b_ub.append(d[i,j])
        A_ub.append([np.eye(52)[i]] + [-np.eye(52)[i]] + [0]*26)
        b_ub.append(centers[i,0])
        A_ub.append([np.eye(52)[i]] + [0]*26 + [-np.eye(52)[i]] + [0]*26)
        b_ub.append(1 - centers[i,0])
        A_ub.append([np.eye(52)[i]] + [0]*26 + [0]*26 + [-np.eye(52)[i]] + [0]*26)
        b_ub.append(centers[i,1])
        A_ub.append([np.eye(52)[i]] + [0]*26 + [0]*26 + [0]*26 + [-np.eye(52)[i]] + [0]*26)
        b_ub.append(1 - centers[i,1])
        A_ub.append([-np.eye(52)[i]] + [0]*52)
        b_ub.append(0)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    res = linprog(c=-np.ones(26), A_ub=A_ub, b_ub=b_ub, method='highs')
    return -res.fun

def polish(centers, radii):

    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i]-x[i+26]})
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i+26]-(1-x[i])})

        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i+52]-(1-x[i])})
    result = SLSQP(evaluate, centers.reshape(-1), args=(radii,), bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
