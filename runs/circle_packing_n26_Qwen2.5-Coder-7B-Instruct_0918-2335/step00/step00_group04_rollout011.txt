import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d, Delaunay
from scipy.optimize import linprog, NonlinearConstraint

def compute_rho(p, obstacles):
    distances = np.array([min(p[0], 1-p[0], p[1], 1-p[1])]
                         + [np.sqrt(np.sum((p - o[:2])**2)) for o in obstacles])
    return np.min(distances)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    C = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    R = [0.25] * 4
    
    obstacles = [(0.25, 0.25, 0.25), (0.75, 0.25, 0.25), (0.75, 0.75, 0.25), (0.25, 0.75, 0.25)]
    vertices = np.array([(0.5, 0.5), (0.5, 0.5), (0.5, 0.5), (0.5, 0.5)])
    
    while len(C) < 26:
        rho_values = [compute_rho(v, obstacles) for v in vertices]
        if max(rho_values) < 1e-6:
            break
        max_rho_index = np.argmax(rho_values)
        p_star = vertices[max_rho_index]
        r_star = rho_values[max_rho_index]
        
        C.append(p_star)
        R.append(r_star)
        
        obstacles.append((p_star[0], p_star[1], r_star))
        
        vor = Voronoi(np.array(C + [(0,0), (1,0), (1,1), (0,1)]))
        new_vertices = vor.vertices
        clipping_mask = (new_vertices[:, 0] >= 0) & (new_vertices[:, 0] <= 1) & (new_vertices[:, 1] >= 0) & (new_vertices[:, 1] <= 1)
        vertices = new_vertices[clipping_mask]
    
    A = []
    b = []
    for i, (ci, ri) in enumerate(C):
        for j in range(i+1, len(C)):
            (cj, rj) = C[j]
            A.append([ci[0] - cj[0], ci[1] - cj[1]])
            b.append(ri + rj)
        A.append([ci[0] - 0, ci[1] - 0, -1])
        b.append(ri)
        A.append([ci[0] - 1, ci[1] - 0, 1])
        b.append(ri)
        A.append([ci[0] - 0, ci[1] - 0, 1])
        b.append(ri)
        A.append([ci[0] - 1, ci[1] - 0, -1])
        b.append(ri)
    
    for ci in C:
        for wall in vertices:
            d = np.sqrt((ci[0] - wall[0])**2 + (ci[1] - wall[1])**2)
            A.append([ci[0] - wall[0], ci[1] - wall[1], -1])
            b.append(d - ci[2])
            A.append([ci[0] - wall[0], ci[1] - wall[1], 1])
            b.append(d + ci[2])
    
    result = linprog(-np.array(R), A_ub=A, b_ub=b, bounds=[(0, 0.5)] * len(C), method='highs')
    R_opt = -result.x
    
    centers = np.array(C)
    radii = R_opt
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
