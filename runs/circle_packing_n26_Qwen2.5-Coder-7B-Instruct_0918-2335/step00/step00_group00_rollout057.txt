import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    R = 200
    delta_c = 1e-3
    n_iterations = 200
    n_circles = 26
    
    grid = np.linspace(1/R, 1 - 1/R, R)
    grid_x, grid_y = np.meshgrid(grid, grid)
    X = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
    
    d_boundary = np.min(np.concatenate((X[:, 0], 1 - X[:, 0], X[:, 1], 1 - X[:, 1])), axis=0)
    r_f = d_boundary
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        k = np.argmax(r_f)
        centers[i] = X[k]
        radii[i] = r_f[k] - 1e-9  # Ensure strict interior
        distances = np.linalg.norm(X - centers[i], axis=1)
        r_f = np.minimum(r_f, distances - radii[i])
        r_f[k] = -1e9
    
    for _ in range(n_iterations):
        idx = np.random.randint(n_circles)
        c = centers[idx] + np.random.uniform(-delta_c, delta_c, size=(2,))
        distances = np.linalg.norm(X - c, axis=1)
        r_prop = np.minimum(d_boundary, np.minimum(distances - radii, np.inf))
        if r_prop[idx] > radii[idx]:
            centers[idx] = c
            radii[idx] = r_prop[idx]
    
    for i in range(n_circles):
        distances = np.linalg.norm(X - centers[i], axis=1)
        radii[i] = np.minimum(d_boundary, np.minimum(distances - radii, np.inf))
        radii[i] = max(radii[i], 0)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
