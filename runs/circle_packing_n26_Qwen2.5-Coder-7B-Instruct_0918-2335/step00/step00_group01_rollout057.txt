import numpy as np
from scipy import optimize
from heapq import heappush, heappop, heappushpop

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    init_speed = 5e-3
    T_max = 50.0
    epsilon = 1e-6

    def compute_collision_time(c_i, c_j, r_i, r_j, v_i, v_j):
        delta_c0 = c_i - c_j
        delta_v = v_i - v_j
        delta_r0 = r_i + r_j
        g_tot = 2 * g
        norm_delta_v = np.linalg.norm(delta_v)
        norm_delta_c0 = np.linalg.norm(delta_c0)
        discriminant = norm_delta_v**2 * norm_delta_c0**2 - (norm_delta_c0 - delta_r0)**2 * (norm_delta_c0 + delta_r0)**2
        if discriminant < 0:
            return np.inf
        tCc = (-delta_c0 @ delta_v - np.sqrt(discriminant)) / (norm_delta_v**2 + g_tot**2)
        if tCc < 0:
            return np.inf
        return tCc

    def compute_wall_collision_time(c, v, r, wall):
        if wall == 'x0':
            tCw = (r - c[0]) / (v[0] - g)
        elif wall == 'x1':
            tCw = (1 - (r + c[0])) / (v[0] + g)
        elif wall == 'y0':
            tCw = (r - c[1]) / (v[1] - g)
        elif wall == 'y1':
            tCw = (1 - (r + c[1])) / (v[1] + g)
        else:
            raise ValueError("Invalid wall")
        if tCw < 0:
            return np.inf
        return tCw

    def resolve_collision(c_i, c_j, r_i, r_j, v_i, v_j):
        delta_c0 = c_i - c_j
        delta_v = v_i - v_j
        norm_delta_c0 = np.linalg.norm(delta_c0)
        norm_delta_v = np.linalg.norm(delta_v)
        v_i_prime = v_i - 2 * (delta_c0 @ delta_v / norm_delta_c0**2) * delta_c0
        v_j_prime = v_j + 2 * (delta_c0 @ delta_v / norm_delta_c0**2) * delta_c0
        return v_i_prime, v_j_prime

    def resolve_wall_collision(c, v, r, wall):
        if wall == 'x0':
            v_prime = [-v[0], v[1]]
        elif wall == 'x1':
            v_prime = [-v[0], v[1]]
        elif wall == 'y0':
            v_prime = [v[0], -v[1]]
        elif wall == 'y1':
            v_prime = [v[0], -v[1]]
        else:
            raise ValueError("Invalid wall")
        return v_prime

    def event_queue(circles, velocities):
        queue = []
        for i in range(n):
            for j in range(i + 1, n):
                tCc = compute_collision_time(circles[i], circles[j], r_min, r_min, velocities[i], velocities[j])
                heappush(queue, (tCc, ('cc', i, j)))
            for wall in ['x0', 'x1', 'y0', 'y1']:
                tCw = compute_wall_collision_time(circles[i], velocities[i], r_min, wall)
                heappush(queue, (tCw, ('cw', i, wall)))
        return queue

    radar = np.random.rand(4)
    radar = np.array(sorted(radar, key=lambda x: (x % 1, x // 1)))
    circles = np.zeros((n, 2))
    velocities = np.zeros((n, 2))
    for i in range(n):
        circles[i] = radar[i * 4:(i + 1) * 4][:2] + np.random.randn(2) * 1e-4
        velocities[i] = np.random.randn(2) * init_speed
        if np.sum(circles[i]**2) + r_min**2 > 1:
            circles[i] = np.array([-1, -1])

    event_q = event_queue(circles, velocities)
    current_time = 0.0

    while event_q and current_time < T_max:
        t_e, event = heappop(event_q)
        delta_t = t_e - current_time
        circles += velocities * delta_t
        circles[:, 2] += g * delta_t
        current_time = t_e

        if event[0] == 'cc':
            i, j = event[1]
            velocities[i], velocities[j] = resolve_collision(circles[i], circles[j], r_min, r_min, velocities[i], velocities[j])
            event_q = [e for e in event_q if e != event]
            for k in range(n):
                if k != i and k != j:
                    heappushpop(event_q, (compute_collision_time(circles[i], circles[k], r_min, r_min, velocities[i], velocities[k]), ('cc', i, k)))
                    heappushpop(event_q, (compute_collision_time(circles[j], circles[k], r_min, r_min, velocities[j], velocities[k]), ('cc', j, k)))
            heappushpop(event_q, (compute_collision_time(circles[i], circles[j], r_min, r_min, velocities[i], velocities[j]), ('cc', i, j)))
        else:
            i, wall = event[1]
            velocities[i] = resolve_wall_collision(circles[i], velocities[i], r_min, wall)
            event_q = [e for e in event_q if e != event]
            for j in range(n):
                if i != j:
                    heappushpop(event_q, (compute_collision_time(circles[i], circles[j], r_min, r_min, velocities[i], velocities[j]), ('cc', i, j)))
            heappushpop(event_q, (compute_collision_time(circles[i], circles[j], r_min, r_min, velocities[i], velocities[j]), ('cc', i, j)))

    circles = np.clip(circles, r_min, 1 - r_min)
    r_opt = np.ones(n) * r_min
    cons = [{'type': 'ineq', 'fun': lambda r, c=c: (1 - c[:, 0]**2 - r**2)[0]},
            {'type': 'ineq', 'fun': lambda r, c=c: (1 - c[:, 1]**2 - r**2)[0]},
            {'type': 'ineq', 'fun': lambda r, c=c: (1 - (1 - c[:, 0])**2 - r**2)[0]},
            {'type': 'ineq', 'fun': lambda r, c=c: (1 - (1 - c[:, 1])**2 - r**2)[0]}]
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[i] + r[j] - np.sqrt(np.sum((c[i] - c[j])**2))})

    res = optimize.minimize(lambda r: -np.sum(r), r_opt, method='SLSQP', bounds=[(r_min, None) for _ in range(n)], constraints=cons)

    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    circles = np.clip(circles, r_min, 1 - r_min)
    radii = res.x - epsilon
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(circles[i] - circles[j])
            if dist < radii[i] + radii[j]:
                circles[:, :2] += np.random.randn(2, n) * 1e-4
                radii -= epsilon
                res = optimize.minimize(lambda r: -np.sum(r), r_opt, method='SLSQP', bounds=[(r_min, None) for _ in range(n)], constraints=cons)
                if not res.success:
                    return np.zeros((n, 2)), np.zeros(n), 0.0
                circles = np.clip(circles, r_min, 1 - r_min)
                radii = res.x - epsilon
                i = 0
                j = 0
    radii = np.clip(radii, 0, None)
    return circles, radii, np.sum(radii)
