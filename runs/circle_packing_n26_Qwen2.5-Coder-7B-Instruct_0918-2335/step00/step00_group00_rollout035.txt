# Import necessary libraries
import numpy as np

def init_circles(variant):
    centers = np.zeros((26, 2))
    radii = np.full(26, 1e-4)  # Very small initial radius
    if variant == 'A':
        centers[0, :] = [0.2, 0.2]
        centers[1, :] = [0.2, 0.6]
        centers[2, :] = [0.2, 1.0]
        centers[3, :] = [0.6, 0.2]
        centers[4, :] = [0.6, 0.6]
        centers[5, :] = [0.6, 1.0]
        centers[6, :] = [1.0, 0.2]
        centers[7, :] = [1.0, 0.6]
        centers[8, :] = [1.0, 1.0]
        centers[9:26, :] = np.meshgrid(np.arange(0.4, 0.8, 0.2), np.arange(0.4, 0.8, 0.2))
    elif variant == 'B':
        centers[0, :] = [0.2, 0.2]
        centers[1, :] = [0.2, 0.6]
        centers[2, :] = [0.2, 1.0]
        centers[3, :] = [0.6, 0.2]
        centers[4, :] = [0.6, 0.6]
        centers[5, :] = [0.6, 1.0]
        centers[6, :] = [1.0, 0.2]
        centers[7, :] = [1.0, 0.6]
        centers[8, :] = [1.0, 1.0]
        centers[9:26, :] = np.meshgrid(np.arange(0.2, 0.6, 0.2), np.arange(0.2, 0.6, 0.2))
    else:
        raise ValueError("Invalid variant")
    return centers, radii

def inflate(circles, delta_r, epsilon):
    max_iter = 5000
    for _ in range(max_iter):
        changed = False
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.linalg.norm(circles[i] - circles[j])
                if dist < circles[i, 2] + circles[j, 2] - epsilon:
                    delta = (circles[i, 2] + circles[j, 2] - dist) / (2 * dist)
                    circles[i, :2] -= delta * (circles[i, :2] - circles[j, :2])
                    circles[j, :2] += delta * (circles[i, :2] - circles[j, :2])
                    changed = True
        for i in range(26):
            circles[i, :2] = np.clip(circles[i, :2], circles[i, 2], 1 - circles[i, 2])
            circles[i, 2] = min(circles[i, 2] + delta_r,
                            np.min(circles[i, :2])),
                            np.min([1 - circles[i, :2]]))
        if not changed:
            break
    return circles[:, 2], circles[:, :2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    for variant in ['A', 'B']:
        centers, radii = init_circles(variant)
        radii, centers = inflate(centers, 0.0005, epsilon=1e-6)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers
            best_radii = radii

    # If we can't get enough radius, try random seed
    if best_sum_radii <= 2.636:
        centers, _ = init_circles('B')
        radii = np.full(26, 0.105)  # Equal radii based on hexagonal lattice
        radii, centers = inflate(centers, 0.0005, epsilon=1e-6)
        sum_radii = np.sum(radii)
        if best_sum_radii < sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers
            best_radii = radii

    # Ensure sum_radii is above threshold
    assert best_sum_radii > 2.636, "Failed to produce a valid packing"
    return best_centers, best_radii, best_sum_radii

# Results (should pass validate_packing function)
centers, radii, sum_radii = run_packing()
print(f"Sum of Radii: {sum_radii}")
