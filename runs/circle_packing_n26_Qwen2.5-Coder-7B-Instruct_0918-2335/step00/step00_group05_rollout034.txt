import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize parameters and variables
    S = 80  # Swarm size
    T = 2000  # PSO iterations
    T0 = 0.05  # Initial temperature for SA
    alpha = 0.995  # Cooling rate for SA
    epsilon = 1e-12  # Small buffer
    w_max, w_min = 0.9, 0.4  # Inertia weight bounds
    c1, c2 = 1.5, 1.5  # Cognitive and social coefficients for PSO
    v_max = 0.05  # Maximum velocity for PSO

    # Random initial population
    positions = np.random.uniform(0, 1 - epsilon, (S, 26, 2))
    radii = np.random.uniform(epsilon, 0.05, (S, 26))
    inertia = np.full(S, w_max)
    p_best_positions = positions.copy()
    p_best_radii = radii.copy()
    p_best_sum_radii = -np.inf * np.ones(S)

    for _ in range(T):
        velocity = v_max * np.random.rand(S, 26, 2)
        new_positions = positions + velocity
        new_radii = radii.copy()

        # Perform feasibility projection
        for s in range(S):
            for i in range(26):
                new_positions[s, i, 0] = np.clip(new_positions[s, i, 0], new_radii[s, i], 1 - new_radii[s, i])
                new_positions[s, i, 1] = np.clip(new_positions[s, i, 1], new_radii[s, i], 1 - new_radii[s, i])
                r_i = max(new_radii[s, i], epsilon)
                new_radii[s, i] = r_i
                for j in range(i + 1, 26):
                    d = np.hypot(new_positions[s, i, 0] - new_positions[s, j, 0], new_positions[s, i, 1] - new_positions[s, j, 1])
                    min_dist = new_radii[s, i] + new_radii[s, j]
                    if d < min_dist - epsilon:
                        shift = (min_dist - d) / 2.0
                        direction = ((new_positions[s, i, 0] - new_positions[s, j, 0]) / d, (new_positions[s, i, 1] - new_positions[s, j, 1]) / d)
                        new_positions[s, i, 0] += shift * direction[0]
                        new_positions[s, i, 1] += shift * direction[1]
                        new_positions[s, j, 0] -= shift * direction[0]
                        new_positions[s, j, 1] -= shift * direction[1]

                p_best_sum_radii[s] = np.sum(p_best_radii[s] - epsilon)
                new_sum_radii = np.sum(new_radii[s] - epsilon)
                if new_sum_radii > p_best_sum_radii[s]:
                    p_best_positions[s] = new_positions[s].copy()
                    p_best_radii[s] = new_radii[s].copy()
                    p_best_sum_radii[s] = new_sum_radii

        # PSO update
        for s in range(S):
            w_value = w_max - (w_max - w_min) * _ / T
            v_p = w_value * velocity[s] + c1 * np.random.rand(26, 2) * (p_best_positions[s] - positions[s]) + c2 * np.random.rand(26, 2) * (p_best_positions[s].mean(axis=0, keepdims=True) - positions[s])
            v_p = v_p.clip(-v_max, v_max)
            velocity[s] = v_p
            positions[s] += velocity[s]
            positions[s] = positions[s].clip(epsilon, 1 - epsilon)
            radii[s] = (np.random.uniform(epsilon, 0.05, (26,)) - epsilon).clip(epsilon, None)

    best_solution = p_best_sum_radii.argmax()

    # Perform SA around the best solution
    temperature = T0
    num_moves = 20000
    best = p_best_sum_radii[best_solution]
    current = best
    for _ in range(num_moves):
        if temperature < 1e-7:
            break
        i = np.random.randint(26)
        move_type = np.random.randint(2)
        if move_type == 0:
            delta_x, delta_y = np.random.normal(0, temperature, 2)
            new_position = np.clip(p_best_positions[best_solution, i] + delta_x, epsilon, 1 - epsilon)
            new_radius = p_best_radii[best_solution, i].clip(epsilon, None)
        else:
            delta_r = np.random.normal(0, temperature / 2)
            new_radius = p_best_radii[best_solution, i] + delta_r
            new_position = p_best_positions[best_solution, i].clip(epsilon, 1 - epsilon)
        new_sum_radii = np.sum([new_radius])

        if new_sum_radii > current:
            current = new_sum_radii
            p_best_positions[best_solution, i] = new_position
            p_best_radii[best_solution, i] = new_radius
        else:
            accept_prob = np.exp(-(current - new_sum_radii) / temperature)
            if np.random.rand() < accept_prob:
                current = new_sum_radii
                p_best_positions[best_solution, i] = new_position
                p_best_radii[best_solution, i] = new_radius

        temperature *= alpha

    # Final feasibility and safety checks
    for j in range(26):
        x, y = p_best_positions[best_solution, j]
        r = p_best_radii[best_solution, j]
        if x < r or x > 1 - r or y < r or y > 1 - r:
            p_best_positions[best_solution, j] = [(x + y) / 2, (x + y) / 2]
            p_best_radii[best_solution, j] = r / 2
        for k in range(26):
            if j != k:
                d = np.hypot(p_best_positions[best_solution, j, 0] - p_best_positions[best_solution, k, 0], p_best_positions[best_solution, j, 1] - p_best_positions[best_solution, k, 1])
                min_dist = max(p_best_radii[best_solution, j], p_best_radii[best_solution, k])
                if d < min_dist:
                    dist_factor = (min_dist - d) / d
                    shift = dist_factor * (p_best_positions[best_solution, j] - p_best_positions[best_solution, k])
                    p_best_positions[best_solution, j] -= shift
                    p_best_positions[best_solution, k] += shift

    return p_best_positions[best_solution], p_best_radii[best_solution], np.sum(p_best_radii[best_solution] - epsilon)
