import numpy as np
from scipy.optimize import minimize

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (26, 2))
    radii = np.zeros(26)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[2*26:])

def constraints(v):
    n = 26
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    boundary_constraints = np.maximum(0, centers - radii)
    boundary_constraints += np.maximum(0, 1 - centers - radii)
    
    distance_matrix = np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2)
    non_overlap_constraints = (distance_matrix - (radii[:, np.newaxis] + radii[np.newaxis]) ** 2)[:, :, 1:]
    
    return np.concatenate((boundary_constraints, non_overlap_constraints.flatten()))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    eps = 1e-7
    RESTARTS = 10
    best_v = None
    best_value = -np.inf
    
    for seed in range(RESTARTS):
        v0 = initial_guess(seed)
        bounds = [(0, 1)] * (2 * 26) + [(0, 0.5)] * 26
        res = minimize(
            fun        = objective,
            x0         = v0,
            method     = 'SLSQP',
            constraints= {'type': 'ineq', 'fun': constraints},
            bounds     = bounds,
            options    = {
                'ftol': 1e-9,
                'maxiter': 4000,
                'disp': False
            }
        )
        if not res.success:
            continue
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x
    
    if best_v is None:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    
    centers = best_v[:2*26].reshape((26, 2))
    radii = np.maximum(0.0, best_v[2*26:] - eps)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
