import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dxs = [0.2] * 7 + [0.166] * 7 + [0.266] + [0.133] + [0.333]
    dys = [0.1] * 7 + [0.2] * 7 + [0.0] * 6 + [0.3] * 4 + [0.5]
    centers = np.column_stack([np.repeat(np.arange(7) * dxs[i], dxs[i]) + dxs[i] / 2
                               for i in range(n)])
    centers += np.random.normal(0, 1e-6, size=(n, 2))
    radii = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1 - centers[:, 1])) - 1e-4
    radii += np.random.normal(0, 1e-6, size=n)
    step = 0.5
    M_bound = 10**2
    M_overlap = 10**2

    def penalty(bnds):
        penalty_bound = np.maximum(0, bnds - np.minimum(bnds[:, 0], 1 - bnds[:, 0]))
        penalty_overlap = distance_matrix(bnds[:, :2], bnds[:, :2])
        penalty_overlap = np.sum((penalty_overlap - bnds[:, 2][:, None] - bnds[:, 2][None, :])**2 * (penalty_overlap < (bnds[:, 2][:, None] + bnds[:, 2][None, :]))
        return np.sum(penalty_bound * M_bound) + np.sum(penalty_overlap * M_overlap)

    def objective(bnds):
        return -np.sum(bnds[:, 2]) + penalty(bnds)

    def repair(bnds):
        for i in range(n):
            bnds[i, 2] = min(bnds[i, 2], np.minimum(bnds[i, 0], 1 - bnds[i, 0]))
        for i in range(n):
            for j in range(i + 1, n):
                if distance_matrix([bnds[i, :2]], [bnds[j, :2]]) < (bnds[i, 2] + bnds[j, 2]):
                    shrink = (distance_matrix([bnds[i, :2]], [bnds[j, :2]]) - (bnds[i, 2] + bnds[j, 2])) / 2
                    bnds[i, 2], bnds[j, 2] -= shrink + shrink * 0.01 * (np.random.rand(2) - 0.5)
                    bnds[i, 2] = max(bnds[i, 2], 0)
                    bnds[j, 2] = max(bnds[j, 2], 0)
        return bnds
    
    bnds = np.column_stack((centers, radii))
    bnds[:, 2] = np.maximum(0, bnds[:, 2])
    bounds = [(-1, 1), (-1, 1), (0, 1)] * n
    res = minimize(objective, bnds, method='SLSQP', bounds=bounds)
    bnds = res.x.reshape(-1, 3)
    bnds = repair(bnds)
    obj = objective(bnds)
    bnds = verify(bnds)
    return (centers + 0.5, bnds[:, 2:], obj)

def verify(bnds):
    centers = bnds[:, :2]
    radii = bnds[:, 2]
    bounds = ('bounds', (0, 1), (0, 1))
    if not min([radius <= min(x, 1 - x, y, 1 - y)
                for x, y, radius in zip(centers[:, 0], centers[:, 1], radii)]):
        return
    if not min([distance_matrix(centers[[i]], centers[[j]]) >= radii[i] + radii[j]
                for i in range(len(centers))
                for j in range(i + 1, len(centers))]):
        return
    return bnds
