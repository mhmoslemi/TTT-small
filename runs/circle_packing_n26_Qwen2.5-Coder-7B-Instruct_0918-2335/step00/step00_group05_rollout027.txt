import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    x = np.random.uniform(0.05, 0.95, n)
    y = np.random.uniform(0.05, 0.95, n)
    r = np.full(n, 1e-6)
    
    Δt = 1.0e-3
    α = 4.0
    β = 0.8
    ε = 1.0e-12
    δr = 1.0e-4
    maxIter = 3000
    tol = 1.0e-10

    for step in range(maxIter):
        F = np.zeros_like(x)
        for i in range(n):
            F[i] += α * (np.clip(x[i] + r[i], 0, 1) - x[i])
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                d = np.sqrt(dx**2 + dy**2)
                if d < r[i] + r[j] + ε:
                    F[i] -= (dx / d) * (r[i] + r[j] - d)
                    F[j] += (dx / d) * (r[i] + r[j] - d)
        
        v = np.zeros_like(x)
        for i in range(n):
            v[i] += Δt * (F[i] - β * v[i])
            x[i] += Δt * v[i]
            y[i] += Δt * v[i]
            r[i] = max(r[i], 0)

            min_r = np.inf
            for j in range(n):
                if j != i:
                    min_r = min(min_r, (x[j] - x[i])**2 + (y[j] - y[i])**2)
                min_r = min(min_r, x[i]**2 + y[i]**2, (x[i] - 1)**2 + y[i]**2, x[i]**2 + (y[i] - 1)**2)
                r[i] = min(min_r - ε, r[i])

        if np.max(np.abs(r - r[-1])) / r[-1] < tol:
            break

    r = np.clip(r, ε, None)
    for i in range(n):
        for j in range(n):
            if i != j:
                d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)

    r_opt = np.maximum(r_opt, 0)

        r[i] = r_opt[i]
    
    if np.isnan(r).any() or np.any(r < 0) or np.any(x < 0) or np.any(x > 1) or np.any(y < 0) or np.any(y > 1):
