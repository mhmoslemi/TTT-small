import numpy as np
from scipy.optimize import differential_evolution, minimize

N = 26
dim = 3 * N
lambda_penalty = 1e5
DE_popsize = 12
DE_maxiter = 400

def objective(x):
    xs, ys, rs = x[:N], x[N:2*N], x[2*N:]
    sum_radii = -rs.sum()
    p_boundary = (np.maximum(0, rs - xs)**2 +
                  np.maximum(0, xs + rs - 1)**2 +
                  np.maximum(0, rs - ys)**2 +
                  np.maximum(0, ys + rs - 1)**2).sum()
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    p_overlap = np.sum(np.maximum(0, rr - dist[np.triu_indices(N, 1)])**2)
    return sum_radii + lambda_penalty * (p_boundary + p_overlap)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    result = differential_evolution(objective, bounds, popsize=DE_popsize, maxiter=DE_maxiter, tol=1e-9, seed=0, polish=True)
    best_x = result.x
    xs, ys, rs = best_x[:N], best_x[N:2*N], best_x[2*N:]
