```python
import numpy as np
from scipy.optimize import differential_evolution, linprog, Bounds, minimize

def evaluate(p):
    C = p.reshape(26, 2)
    d = np.linalg.norm(C[:, np.newaxis, :] - C[np.newaxis, :, :], axis=-1)
    
    A = []
    b = []
    
    for i in range(26):
        for j in range(i + 1, 26):
            A.append([1, 1, 0, 0])
            b.append(d[i, j])
            
            A.append([1, 0, 0, 0])
            b.append(C[i, 0])
            
            A.append([1, 0, 0, 1])
            b.append(1 - C[i, 0])
            
            A.append([1, 0, 1, 0])
            b.append(C[i, 1])
            
            A.append([1, 0, 1, 1])
            b.append(1 - C[i, 1])
            
            A.append([-1, 0, 0, 0])
            b.append(-C[i, 0])
            
            A.append([-1, 0, 0, 1])
            b.append(-1 + C[i, 0])
            
            A.append([-1, 0, 1, 0])
            b.append(-C[i, 1])
            
            A.append([-1, 0, 1, 1])
            b.append(-1 + C[i, 1])
    
    A = np.array(A)
    b = np.array(b)
    
    res = linprog(c=-np.ones(26), A_ub=A, b_ub=b, method='highs')
    return -res.fun

def initial_positions():
    # Create a hexagonal lattice and keep the 26 closest circles
    center_positions = np.array([[3 * i / 26, j / 13] for i in range(26) for j in range(14)])
    distances = np.linalg.norm(center_positions[:, np.newaxis, :] - center_positions[np.newaxis, :, :], axis=-1)
    nearest_indices = np.argsort(distances, axis=1)[:, 1:27]
    return center_positions[nearest_indices.flatten()].reshape(26, 2)

bounds = Bounds(0, 1)
centers_best = initial_positions()
positions = differential_evolution(evaluate, bounds, maxiter=200, popsize=3, initial=centers_best).x
radii_best = evaluate(positions).reshape(26, 2)

def penalty(x):
    C, r, p = x[:26], x[26:], positions
    d = np.linalg.norm(C[:, np.newaxis, :] - C[np.newaxis, :, :], axis=-1)
    A = np.zeros((len(d)*2, len(d)))
    
    for i in range(len(d)):
        for j in range(i + 1, len(d)):
            A[2*i, :26] = [1, 1, 0, 0]
            A[2*i + 1, :26] = [1, 0, 0, 0]
            A[2*i, :26][j] = -1
            A[2*i + 1, :26][j] = -1
    
    return np.sum(p * np.maximum(r[:, np.newaxis] + r[np.newaxis, :] - d, 0) ** 2)

objective_slsqp = lambda x: -np.sum(x[26:])
constraints_slsqp = [{'type': 'ineq', 'fun': penalty}]
bounds_slsqp = Bounds(0, 1)

result_slsqp = minimize(objective_slsqp, np.concatenate((positions.reshape(52), radii_best.ravel())), method='SLSQP', constraints=constraints_slsqp, bounds=bounds_slsqp, tol=1e-9)

centers_final = result_slsqp.x[:52].reshape(26, 2)
radii_final = result_slsqp.x[52:]

sum_radii = np.sum(radii_final)

return centers_final, radii_final, sum_radii
```
