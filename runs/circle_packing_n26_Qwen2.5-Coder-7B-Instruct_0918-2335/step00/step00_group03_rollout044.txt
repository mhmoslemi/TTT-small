import numpy as np
from scipy.optimize import linprog, differential_evolution, SLSQP


def eval_placement(p):
    C = p.reshape(26, 2)
    d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
    A = C[:, None, :] - C[None, :, :]
    A = np.where(A != 0, A / np.abs(A), 0)
    A = np.concatenate((A, -A, np.eye(26), np.eye(26), -np.eye(26), -np.eye(26)), axis=0)
    b = np.concatenate((-d.sum(axis=1), -C[:, 0], 1 - C[:, 0], -C[:, 1], 1 - C[:, 1], C[:, 0], C[:, 1]), axis=0)
    res = linprog(c=-np.ones(26), A_ub=A, b_ub=b, method='highs', options={'disp': False})
    if not res.success:
        return -1e9  # Return very small value to avoid choosing bad placements
    return -res.fun


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial seed using a hexagonal lattice configuration
    side_length = 2 * (1 / np.sqrt(3))
    n = int(np.ceil(26 / 7))
    centers = np.zeros((26, 2))
    j = 0
    for i in range(n):
        if i % 2 == 0:
            centers[j:j+7, 0] = (i * side_length) % 1
            centers[j:j+7, 1] = np.arange(7) * 2 * (1 / (2 * np.sqrt(3))) + 1 / np.sqrt(3)
            j += 7
        else:
            centers[j:j+7, 0] = (i * side_length) % 1
            centers[j:j+7, 1] = np.arange(7) * 2 * (1 / (2 * np.sqrt(3)))
            j += 7

    # Remove extra circles and sort by closeness to center
    centers = centers[:26]
    distances = np.linalg.norm(centers, axis=1)
    idx = np.argsort(distances)
    centers = centers[idx][:26]
    distances = distances[idx][:26]
    centers = centers[distances > 0.01 * np.max(distances)]
    centers = centers[:26]

    # Enforce bounds [0, 1]
    centers = np.clip(centers, 0, 1)
    p0 = centers.flatten()
    
    # Differential Evolution to find optimal positions
    res = differential_evolution(eval_placement, bounds=[(0, 1)] * 52, maxiter=200, popsize=3, tol=1e-6,
                                mutation=(0.3, 0.8), recombination=0.7, seed=42, vectorized=False)
    
    # SLSQP to polish the solution
    C_best = res.x.reshape(26, 2)
    d = np.linalg.norm(C_best[:, None, :] - C_best[None, :, :], axis=-1)
    A = C_best[:, None, :] - C_best[None, :, :]
    A = np.where(A != 0, A / np.abs(A), 0)
    A = np.concatenate((A, -A, np.eye(26), np.eye(26), -np.eye(26), -np.eye(26)), axis=0)
    b = np.concatenate((-d.sum(axis=1), -C_best[:, 0], 1 - C_best[:, 0], -C_best[:, 1], 1 - C_best[:, 1], C_best[:, 0], C_best[:, 1]), axis=0)
    
    def objective(x):
        return -np.sum(x[26:])
    
    def constraint(x):
        C = x[:52].reshape(26, 2)
        d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
        return d >= 2 * x[52:] / np.sqrt(d)
    
    bounds = [(0, 1) for _ in range(78)] + [(0, None) for _ in range(26)]
    x0 = np.concatenate((C_best.flatten(), np.min(d, axis=1)))
    result = SLSQP(objective, x0, bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint}], options={'maxiter': 2000, 'ftol': 1e-9})
    
    # Final result
    radii = result.x[52:]
    centers = result.x[:52].reshape(26, 2)
    sum_radii = np.sum(radii)
    
    # Clip to avoid degenerate solutions
    radii = np.clip(radii, 0, 0.5)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
