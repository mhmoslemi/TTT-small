import numpy as np
from scipy.optimize import differential_evolution, minimize

def objective(x, N):
    xs = x[:N]
    ys = x[N:2*N]
    rs = x[2*N:]
    sum_r = -rs.sum()
    p_boundary = -np.sum(np.maximum(0, rs - xs)**2)
    p_boundary -= np.sum(np.maximum(0, xs + rs - 1)**2)
    p_boundary -= np.sum(np.maximum(0, rs - ys)**2)
    p_boundary -= np.sum(np.maximum(0, ys + rs - 1)**2)
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    rr = rs[:, None] + rs[None, :]
    overlap = rr - dist
    p_overlap = -np.sum(np.maximum(0, overlap[np.triu_indices(N, 1)])**2)
    
    return sum_r + 1e5 * (p_boundary + p_overlap)

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    N = 26
    dim = 3 * N
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    
    def wrapper(x):
        return objective(x, N)
    
    result = differential_evolution(wrapper, bounds, maxiter=400, popsize=12, random_state=0)
    best_x = result.x
    
    result = minimize(wrapper, best_x, method='L-BFGS-B', bounds=bounds)
    final_x = result.x
    
    centres = np.column_stack([final_x[0:N*2:2], final_x[1:N*2:2]])
    radii = final_x[2*N:]
    sum_radii = radii.sum()
    
    return centres, radii, sum_radii
