def constraints(z):

    n_circles = 26
    for i in range(n_circles):
        c.append(z[i*3+1] - z[i*3])  # x_i - r_i
        c.append(1 - z[i*3] - z[i*3+1])  # 1 - x_i - r_i
        c.append(z[i*3+2] - z[i*3])  # y_i - r_i
        c.append(1 - z[i*3+2] - z[i*3])  # 1 - y_i - r_i
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            dx, dy = z[i*3] - z[j*3], z[i*3+2] - z[j*3+2]
            r_sum = z[i*3+1] + z[j*3+1]
            c.append(dx**2 + dy**2 - r_sum**2)
    return c

def jacobian(z):

    jac = np.zeros((4*n_circles + n_circles*(n_circles-1)//2, 78))

        jac[4*i, i*3] = -1  # d(x_i - r_i)/dr_i = -1
        jac[4*i+1, i*3+1] = -1  # d(1 - x_i - r_i)/dx_i = -1
        jac[4*i+1, i*3] = -1  # d(1 - x_i - r_i)/dr_i = -1
        jac[4*i+2, i*3+2] = 1  # d(y_i - r_i)/dy_i = 1
        jac[4*i+2, i*3+2] = -1  # d(y_i - r_i)/dr_i = -1
        jac[4*i+3, i*3+2] = -1  # d(1 - y_i - r_i)/dy_i = -1
        jac[4*i+3, i*3+2] = -1  # d(1 - y_i - r_i)/dr_i = -1
    for i in range(n_circles):
