import numpy as np
import scipy.optimize
from scipy.spatial import distance_matrix

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial centers (hexagonal grid)
    side_length = 0.2
    hex_width = np.sqrt(3) * side_length
    hex_height = 2 * side_length
    width_count = 5
    height_count = 5
    centers = np.zeros((n, 2))

    row_idx = 0
    col_idx = 0
    for row in range(height_count):
        offset = 0.5 * side_length if row % 2 != 0 else 0
        for col in range(width_count):
            x = offset + col * hex_width
            y = row * hex_height
            if row_idx >= n:
                break
            centers[row_idx] = [x, y]
            row_idx += 1

    # Initial radii
    r_bounds = (0, 1)
    initial_radii = np.full(n, 0.1)

    # Parameter schedule
    M_bound = 10 ** 2
    M_overlap = 10 ** 2

    # Bounds for the optimization variables (x, y, r)
    bounds = np.vstack((centers, initial_radii)).T
    bounds[:, :2] *= 0.9
    bounds[:, :2] = np.maximum(bounds[:, :2], 0)
    bounds[:, :2] = np.minimum(bounds[:, :2], 1 - side_length)
    bounds[:, 2:] = r_bounds

    # Penalty function
    def penalty(cross):
        M = 10 ** 20
        cr = cross.reshape(n, 3)
        x = cr[:, 0]
        y = cr[:, 1]
        r = cr[:, 2]
        return (
            np.sum(r.clip(None, M_bound * (r - np.minimum(x, 1 - x, y, 1 - y)))) +
            np.sum(np.maximum(0, M_overlap * (r + r - np.sqrt(distance_matrix(cross[:, 0:2], cross[:, 0:2])) - 1e-12)))
        )

    # Augmented Lagrangian objective
    def objective(cr):
        nonlocal M_bound, M_overlap
        L = np.sum(cr[:, 2]) + 0.5 * (M_bound + M_overlap) * penalty(cr)
        M_bound *= 3
        M_overlap *= 3
        return L

    # Constraints
    def boundary_constraint(cr):
        r = cr[:, 2]
        x = cr[:, 0]
        y = cr[:, 1]
        return np.sum(r.clip(None, M_bound * (r - np.minimum(x, 1 - x, y, 1 - y)))) - (1e-12)

    def overlap_constraint(cr):
        r = cr[:, 2]
        return np.sum(np.maximum(0, M_overlap * (r + r - np.sqrt(distance_matrix(cr[:, 0:2], cr[:, 0:2])) - 1e-12))) - (1e-12)

    cons = ({'type': 'ineq', 'fun': boundary_constraint},
            {'type': 'ineq', 'fun': overlap_constraint})

    # Run optimization
    result = scipy.optimize.minimize(objective, bounds.flatten(), method='SLSQP', bounds=bounds, constraints=cons)

    # Extract centers, radii, and sum of radii
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
