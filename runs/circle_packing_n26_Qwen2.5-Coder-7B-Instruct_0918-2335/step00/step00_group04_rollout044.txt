import numpy as np
from scipy.spatial import Voronoi
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def get_neighbors(voronoi, vertex):
    neighbors = []
    for region in voronoi.regions:
        if vertex in region:
            for other_vertex in region:
                if other_vertex != vertex and other_vertex != -1:
                    neighbors.append(other_vertex)
    return list(set(neighbors))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    corners = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    centers = np.array(corners)
    radii = np.array([0.25] * n)

    for i in range(4, n):
        distances = np.array([[np.sqrt((centers[j][0] - corners[i % 4][0]) ** 2 + (centers[j][1] - corners[i % 4][1]) ** 2) for j in range(4)]])
        max_delta = np.min(distances - radii)
        centers = np.append(centers, [[corners[i % 4][0] + max_delta / 2, corners[i % 4][1] + max_delta / 2]], axis=0)
        radii = np.append(radii, max_delta / 2)

    voronoi = Voronoi(centers)
    for vertex in range(len(voronoi.vertices)):
        neighbors = get_neighbors(voronoi, vertex)
        max_radius = 0
        for neighbor in neighbors:
            min_distance = min(voronoi.ridge_vertices[vororonoi.find_simplex(voronoi.vertices[vertex])] + [1e-6])
            if min_distance > radii[neighbor]:
                max_radius = min_distance
        if max_radius > 1e-6:
            centers = np.append(centers, [voronoi.vertices[vertex]], axis=0)
            radii = np.append(radii, max_radius)

    A = np.vstack((np.eye(n), -np.eye(n)))
    b = np.hstack((np.ones(n), np.zeros(n)))
    c = radii

    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, 0.5), method='highs')
    radii = result.x

    return centers, radii, np.sum(radii)
