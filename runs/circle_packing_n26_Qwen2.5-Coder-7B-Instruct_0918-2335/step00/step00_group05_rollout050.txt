import numpy as np
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initialize parameters
    swarm_size = 80
    T = 2000
    T0 = 0.05
    alpha = 0.995
    w_max = 0.9
    w_min = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.05
    epsilon = 1e-12
    
    # Initialize swarm
    positions = np.random.rand(swarm_size, n, 2)  # (x, y) for each particle and circle
    velocities = np.zeros((swarm_size, n, 2))
    fitness_values = np.full(swarm_size, -1e10)
    best_positions = np.zeros((n, 2))
    best_fitness = -1e10
    
    for t in range(T):
        for i in range(swarm_size):
            x = positions[i]
            v = velocities[i]
            r = np.clip(np.linalg.norm(v, axis=1), epsilon, None)  # Clamp radius to be non-negative
            fitness = -np.sum(r)  # Negative to penalize
            
            # Collision detection and overlap resolution
            for j in range(n):
                for k in range(j + 1, n):
                    dx, dy = x[j] - x[k]
                    dr = r[j] + r[k]
                    d = np.hypot(dx, dy)
                    if d < dr - epsilon:
                        shift = (dr - d) / (2 * d)
                        direction = (dx, dy) / d
                        x[j] += shift * direction
                        x[k] -= shift * direction
            
            # Feasibility projection
            x[:, 0] = np.clip(x[:, 0], r, 1 - r)
            x[:, 1] = np.clip(x[:, 1], r, 1 - r)
            
            positions[i] = x
            velocities[i] = v
            fitness_values[i] = fitness
            
            if fitness > best_fitness:
                best_fitness = fitness
                best_positions = x
            
            # Update global best for PSO
            if fitness_values[i] > best_fitness:
                best_fitness = fitness_values[i]
                best_positions = positions[i]
        
        # Update velocities and positions
        r = np.clip(np.linalg.norm(velocities, axis=1), epsilon, None)  # Clamp radius to be non-negative
        for i in range(swarm_size):
            x = positions[i]
            v = velocities[i]
            p_best = best_positions[i]
            g_best = best_positions
            velocities[i] = (w_max + (w_max - w_min) * t / T) * velocities[i] + \
                            c1 * np.random.rand(n, 2) * (p_best - x) + \
                            c2 * np.random.rand(n, 2) * (g_best - x)
            positions[i] = x + velocities[i]
            
            # Clamping velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Projection post-velocity update
            positions[i][:, 0] = np.clip(positions[i][:, 0], r, 1 - r)
            positions[i][:, 1] = np.clip(positions[i][:, 1], r, 1 - r)
    
    # SA refinement
    T = T0
    for _ in range(20000):
        i = np.random.randint(swarm_size)
        j = np.random.randint(n)
        x = positions[i][j]
        r = np.clip(np.linalg.norm(velocities[i][j]), epsilon, None)
        
        # New center shift
        delta_x, delta_y = np.random.normal(size=(2,)) * T
        x_new = x + np.array([delta_x, delta_y])
        
        # New radius change
        delta_r = np.random.normal() * T / 2
        r_new = np.clip(r + delta_r, epsilon, None)
        
        # Projects new position and radius
        if (x_new[0] - r_new < 0) or (x_new[0] + r_new > 1) or (x_new[1] - r_new < 0) or (x_new[1] + r_new > 1):
            continue
        positions[i][j] = x_new
        velocities[i][j] = np.array([delta_x, delta_y]) + np.array([0, 0]) * T / 2
        r[j] = r_new
        
        # Acceptance criterion
        fitness_old = -np.sum(r[:j] + r[j+1:])
        fitness_new = -np.sum(r)
        if fitness_new < fitness_old:
            if np.exp((fitness_old - fitness_new) / T) > random.random():
                positions[i][j] = x_new
                velocities[i][j] = np.array([delta_x, delta_y]) + np.array([0, 0]) * T / 2
                r[j] = r_new
    
        # Cooling
        T *= alpha
    
    # Final feasibility check
    for i in range(n):
        x, y = positions[i][j]
        r = r[j]
        if (x - r < epsilon) or (x + r > 1 - epsilon) or (y - r < epsilon) or (y + r > 1 - epsilon):
            continue
        for j in range(n):
            for k in range(j + 1, n):
                dx, dy = positions[i][j] - positions[i][k]
                dr = r[j] + r[k]
                d = np.hypot(dx, dy)
                if d < dr - epsilon:
                    shift = (dr - d) / (2 * d)
                    direction = (dx, dy) / d
                    positions[i][j] += shift * direction
                    positions[i][k] -= shift * direction
                    x, y = positions[i][j]
                    x_new = x + np.array([delta_x, delta_y])
                    r_new = np.clip(r + delta_r, epsilon, None)
                    if (x_new[0] - r_new < 0) or (x_new[0] + r_new > 1) or (x_new[1] - r_new < 0) or (x_new[1] + r_new > 1):
                        continue
                    positions[i][j] = x_new
                    velocities[i][j] = np.array([delta_x, delta_y]) + np.array([0, 0]) * T / 2
                    r[j] = r_new
    
    # Construct the final result
    centers = positions[positions[:, :, 0].mean(axis=0).argmin()][[j for j in range(n)], :, :]
    radii = np.clip(np.linalg.norm(velocities, axis=1), epsilon, None)
    sum_radii = -np.sum(radii)
    
    return centers, radii, sum_radii
