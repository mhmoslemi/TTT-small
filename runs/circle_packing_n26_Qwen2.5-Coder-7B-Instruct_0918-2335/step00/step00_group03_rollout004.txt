import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    r_min = 1e-3
    iterations = 10000

    def initial_placement():
        center = np.zeros((N, 2))
        radii = np.full(N, r_min)
        for i in range(N):
            while True:
                x, y = np.random.rand(2)
                valid = True
                for j in range(i):
                    if np.sqrt((x - center[j, 0])**2 + (y - center[j, 1])**2) < radii[j] + radii[i]:
                        valid = False
                        break
                if valid:
                    center[i, :] = [x, y]
                    break
        return center, radii

    def rsa_seed():
        for _ in range(10):
            center, radii = initial_placement()
            if np.all(center >= -r_min) and np.all(center <= 1 + r_min):
                return center, radii
        return None, None

    def repulsive_force(d, r):
        return d - r

    def boundary_force(x, r):
        return min(x, 1 - x, r)

    def energy(center, radii):
        E = 0
        for i in range(N):
            for j in range(i + 1, N):
                d = repulsive_force(np.linalg.norm(center[i] - center[j]), radii[i] + radii[j])
                if d < 0:
                    E += np.square(d)
            for s in range(2):
                if s == 0:
                    f = boundary_force(center[i, 0], radii[i])
                else:
                    f = boundary_force(center[i, 1], radii[i])
                if f < 0:
                    E += np.square(f)
        return E

    def gradient_energy(center, radii):
        g = np.zeros(2 * N)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(center[i] - center[j])
                if d > 2 * r_min:
                    f = repulsive_force(d, radii[i] + radii[j])
                    if f < 0:
                        ef = np.array([(center[i] - center[j]) / d, (center[j] - center[i]) / d])
                        g[i*2:(i+1)*2] -= ef * f
                        g[j*2:(j+1)*2] += ef * f
            for s in range(2):
                if s == 0:
                    f = boundary_force(center[i, 0], radii[i])
                else:
                    f = boundary_force(center[i, 1], radii[i])
                if f < 0:
                    df = np.array([-np.sign(center[i, s] - 0.5) if center[i, s] < 0.5 else np.sign(center[i, s] - 0.5)])
                    g[i*2+(1-s)*1:i*2+(1-s)*2+1] -= df * f
        return g

    center, radii = rsa_seed()
    if center is None:
        return np.zeros((N, 2)), np.zeros(N), 0.0

    for _ in range(2000):
        g = gradient_energy(center, radii)
        center += g * 1e-3
        radii -= g[:N] * 1e-3
        if np.any(radii < 0):
            radii[radii < 0] = 0

    s_max = min(np.min(np.linalg.norm(center.reshape(N, 2, 1) - center.reshape(1, N, 2), axis=(1, 2)) / (radii.reshape(N, 1) + radii.reshape(1, N))), 
                np.min((1 - radii) / radii))
    center = 0.5 + s_max * (center - 0.5)
    radii *= s_max

    def objective(x):
        center = x[:52].reshape(N, 2)
        radii = x[52:]
        return -radii.sum() + 1e5 * energy(center, radii)

    def gradient(x):
        center = x[:52].reshape(N, 2)
        radii = x[52:]
        return -np.concatenate((gradient_energy(center, radii), -1 * radii)) + 1e5 * gradient_energy(center, radii)

    bounds = [(0, 1)] * (2 * N)
    result = minimize(objective, np.concatenate((center.flatten(), radii)), method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-10, options={'maxiter': 2000})
    return result.x[:52].reshape(N, 2), result.x[52:], result.fun * -1e-5
