import numpy as np
from scipy.optimize import linprog, differential_evolution, NonlinearConstraint, SLSQP

def evaluate_placement(p):
    C = p.reshape(26, 2)
    d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
    m, n = d.shape
    A = np.vstack([np.zeros((m*n, 26)), np.eye(26), np.eye(26), np.eye(26), np.eye(26)])
    b = np.hstack([d.ravel() - 1e-8, C[:, 0], C[:, 1], C[:, 1], C[:, 0]])
    res = linprog(c=-np.ones(26), A_ub=A, b_ub=b, method='highs')
    return -res.fun

def polish_solution(C, r):
    bounds = [(0, None),] * 52
    def objective(x):
        C_new = x[:52].reshape(26, 2)
        d = np.linalg.norm(C_new[:, None, :] - C_new[None, :, :], axis=-1)
        return -np.sum(np.clip(r + 1e-6 * np.clip(r, None, np.min(d, axis=1)), 0, None))
    
    def constraint(x):
        C_new = x[:52].reshape(26, 2)
        d = np.linalg.norm(C_new[:, None, :] - C_new[None, :, :], axis=-1)
        return np.clip(r + 1e-6 * np.clip(r, None, np.min(d, axis=1)), 0, None) - r
    
    nl = NonlinearConstraint(constraint, 0, np.inf)
    res = SLSQP(fun=objective, x0=np.hstack((C.ravel(), r)), bounds=bounds, constraints=[nl], options={'disp': False})
    C_new = res.x[:52].reshape(26, 2)
    r_new = res.x[52:]
    return C_new, r_new

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_centers = np.zeros((26, 2))
    # Populate initial centers with a deterministic hexagonal lattice
    R = np.sqrt(3) / 2
    for i in range(4):
        for j in range(7 if i < 2 else 6):
            x = (j + (i % 2)) * R / 2
            y = i * np.sqrt(3) / 2
            if np LA.norm(initial_centers - np.array([x, y]), axis=1).min() > 0.5:
                initial_centers = np.vstack((initial_centers, np.array([x, y])))
                if initial_centers.shape[0] == 26:
                    break
        if initial_centers.shape[0] == 26:
            break

    res = differential_evolution(evaluate_placement, bounds=[(0, 1),] * 52, maxiter=200, popsize=3, tol=1e-6, mutation=(0.3, 0.8), recombination=0.7)
    C_best = res.x.reshape(26, 2)
    r_best = -evaluate_placement(res.x)

    C_best, r_best = polish_solution(C_best, r_best)

    return C_best, r_best, np.sum(r_best)
