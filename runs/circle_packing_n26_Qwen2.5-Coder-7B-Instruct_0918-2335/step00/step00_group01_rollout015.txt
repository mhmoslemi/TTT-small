import numpy as np
from scipy.optimize import minimize

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (26, 2))
    radii = np.zeros(26)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[52:])

def constraints(v):
    n = 26
    centers = v[:52].reshape((n, 2))
    radii = v[52:]
    
    boundary_constraints = (centers[:, 0] - radii >= 0) & (centers[:, 0] + radii <= 1) & \
                           (centers[:, 1] - radii >= 0) & (centers[:, 1] + radii <= 1)
    
    distance_matrix = np.subtract.outer(centers[:, 0], centers[:, 0]) ** 2 + \
                      np.subtract.outer(centers[:, 1], centers[:, 1]) ** 2
    overlap_constraints = np.where(distance_matrix > 0, distance_matrix - (radii + radii[:, np.newaxis]) ** 2, np.inf)
    
    return np.concatenate((boundary_constraints.ravel(), overlap_constraints.flatten()))

n = 26
pair_idx_i, pair_idx_j = np.triu_indices(n, k=1)
bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

RESTARTS = 10
best_value = -np.inf
best_v = None
for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun=objective,
        x0=v0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if res.success:
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x

centers = best_v[:52].reshape((n, 2))
radii = np.maximum(0.0, best_v[52:] - 1e-7)

return centers, radii, np.sum(radii)
