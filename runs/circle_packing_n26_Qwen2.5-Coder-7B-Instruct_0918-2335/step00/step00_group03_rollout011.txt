import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def evaluate(p):
    C = p.reshape(26, 2)
    d = np.linalg.norm(C[:, None, :] - C[None, :, :], axis=-1)
    A_ineq = np.zeros((455, 26))
    b_ineq = np.zeros(455)
    
    # Pairwise constraints
    k = 0
    for i in range(26):
        for j in range(i + 1, 26):
            A_ineq[k, i] = 1
            A_ineq[k, j] = 1
            b_ineq[k] = d[i, j]
            k += 1
    
    # Boundary constraints
    for i in range(26):
        A_ineq[k, i] = -1
        b_ineq[k] = -C[i, 0]
        k += 1
        A_ineq[k, i] = -1
        b_ineq[k] = -C[i, 1]
        k += 1
        A_ineq[k, i] = -1
        b_ineq[k] = -C[i, 0] + 1
        k += 1
        A_ineq[k, i] = -1
        b_ineq[k] = -C[i, 1] + 1
        k += 1
    
    # Non-negativity constraints
    for i in range(26):
        A_ineq[k, i] = -1
        b_ineq[k] = 0
        k += 1
    
    res = linprog(c=-np.ones(26), A_ub=A_ineq, b_ub=b_ineq, method='highs')
    return -res.fun, res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: hexagonal lattice scaled down
    hex_lattice = np.array([
        (3, 1.5), (3, 2.5), (3, 3.5), (3, 4.5), (3, 5.5),
        (4.7, 1.0), (4.7, 2.0), (4.7, 3.0), (4.7, 4.0), (4.7, 5.0),
        (4.7, 6.0), (6.4, 0.5), (6.4, 1.5), (6.4, 2.5), (6.4, 3.5),
        (6.4, 4.5), (6.4, 5.5), (8.1, 1.0), (8.1, 2.0), (8.1, 3.0),
        (8.1, 4.0), (8.1, 5.0), (8.1, 6.0), (9.8, 1.5), (9.8, 2.5),
        (9.8, 3.5), (9.8, 4.5), (9.8, 5.5)
    ])
    p0 = hex_lattice.ravel()
    
    # Outer optimization using Differential Evolution
    result = differential_evolution(evaluate, bounds=[(0, 1)] * 52, popsize=3, maxiter=200, tol=1e-6, mutation=(0.3, 0.8), recombination=0.7)
    C_best, r_best = result.x[:52].reshape(26, 2), result.x[52:]
    sum_radii = -result.fun
    
    # Polishing step using SLSQP
    optimizer = lambda x: evaluate(x)[0]
    constraints = [{'type': 'ineq', 'fun': lambda x: d[i, j] - x[i] - x[j]} for i in range(26) for j in range(i+1, 26)] + \
                      [{'type': 'ineq', 'fun': lambda x: x[i] - x[i, 0]} for i in range(26)] + \
                      [{'type': 'ineq', 'fun': lambda x: x[i] - x[i, 1]} for i in range(26)] + \
                      [{'type': 'ineq', 'fun': lambda x: x[i] - 1 - x[i, 0]} for i in range(26)] + \
                      [{'type': 'ineq', 'fun': lambda x: x[i] - 1 - x[i, 1]} for i in range(26)] + \
                      [{'type': 'ineq', 'fun': lambda x: -x[i]} for i in range(26)]
    result_slsqp = minimize(optimizer, C_best.ravel(), method='SLSQP', jac=None, constraints=constraints, bounds=[(0, 1)] * 52, options={'disp': False, 'ftol': 1e-9, 'maxiter': 2000})
    C_best, r_best = result_slsqp.x[:52].reshape(26, 2), result_slsqp.x[52:]
    sum_radii = optimizer(result_slsqp.x)
    
    # Clip radii to [0, 0.5]
    r_best = np.clip(r_best, 0, 0.5)
    
    return C_best, r_best, sum_radii
