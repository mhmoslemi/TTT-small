import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing():
    n = 26
    centers = np.random.rand(n, 2) * 0.8 + 0.1  # Initial centers near the center of the square
    radii = np.random.rand(n) * 0.05 + 0.01     # Initial radii guesses

    def obj(x):
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        initialize_centers = x[:2*n]
        initialize_radii = x[2*n:]
        return -np.sum(radii)

    def constraints(x):
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n] + (1 - x[2*n]) - x[i]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n+2*i] - 0.01})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n+2*(i+1)] - 0.01})
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: x[2*n+2*i] + x[2*n+2*(i+1)] - 0.02})
        return constraints

    bounds = [(0.01, 0.99)] * 2*n + [(0.01, 0.1)] * n

    result = minimize(obj, centers.flatten(), method='SLSQP', bounds=bounds, constraints=constraints)

    centers = result.x[:2*n].reshape((n, 2))
    radii = result.x[2*n:]
    if not all(r >= 0 for r in radii):
        print("Radii contain negative values")
        return None, None, None

    centers, radii, sum_radii = validate_and_enforce_constraints(centers, radii)
    if not centers.any():
        print("Validation failed or all-zero solution")
        return None, None, None

    return centers, radii, sum_radii

def validate_and_enforce_constraints(centers, radii):
    n = centers.shape[0]
    while True:
        valid = True
        for i in range(n):
            if not (radii[i] >= 0 and radii[i] <= min(1, 1 - centers[i, 0], 1 - centers[i, 1], centers[i, 0], centers[i, 1])):
                return np.nan * np.ones((n, 2)), np.nan * np.ones(n)
            for j in range(i + 1, n):
                if np.linalg.norm(centers[i] - centers[j]) < radii[i] + radii[j]:
                    centers[i] += (centers[j] - centers[i]) * min(1, 2 * radii[j] - np.linalg.norm(centers[i] - centers[j]))
                    centers[j] -= (centers[j] - centers[i]) * min(1, 2 * radii[j] - np.linalg.norm(centers[i] - centers[j]))
                    valid = False
        
        if valid:
            sum_radii = np.sum(radii)
            return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
if centers is not None and radii is not None:
    print(f"Centers:\n{centers}")
    print(f"Radii:\n{radii}")
    print(f"Sum of radii: {sum_radii}")
else:
    print("Run failed")
