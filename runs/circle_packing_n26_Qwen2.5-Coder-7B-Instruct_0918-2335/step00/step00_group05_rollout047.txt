import numpy as np
from scipy.optimize import minimize

def hex_lattice(n):
    s = np.sqrt(3) / 3 * (1 - 1 / n)
    hexagon = np.arange(n * n)
    hexagon = hexagon[hexagon % n < np.floor(n * n / n - 0.5)]
    i, j = np.unravel_index(hexagon, (n, n))
    x = i * np.sqrt(3) * s / 2 + s / 4 + j % 2 * s / 2
    y = i * s * np.sqrt(3) / 3 + s / 2
    x = x[(x >= s / 2) & (x < 1 - s / 2)]
    y = y[(y >= s / 2) & (y < 1 - s / 2)]
    return np.column_stack((x, y))

def boundary_radius(x, y, r):
    return np.min([y, 1 - y, x, 1 - x]) - r

def construct_initial(z, r):
    n = z.shape[0] // 3
    x, y = z[:n], z[n:2 * n]
    r = z[-n:]
    r += np.minimum(boundary_radius(x, y, r / 2), 0)
    return x, y, r

def penalised_obj(z, δ=1e-4, ε=1e-12):
    n = z.shape[0] // 3
    x, y = z[:n], z[n:2 * n]
    r = z[-n:]
    obj = - np.sum(r)
    constraints = np.hstack([x - r, x + r, y - r, y + r, ((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2) - (r[:, None] + r[None, :])**2])
    constraints = np.maximum(constraints.flatten(), -ε) * δ
    return obj + np.sum(constraints)

def solve_stage(z0, δ):

    result = minimize(penalised_obj, z0, args=(δ), method='SLSQP', bounds=[d for d in zip([0] * len(z0), [1] * len(z0))])
    z = result.x
    return z

def validate_and_output(z):
    n = z.shape[0] // 3
    x, y, r = construct_initial(z, r)
    sum_radii = np.sum(r)

    inside_square = np.all((x >= 0) & (x <= 1)) and np.all((y >= 0) & (y <= 1))
    non_negative_radii = np.all(r >= 0)
    no_overlap = np.all(np.sqrt(np.sum((x[:, None] - x[None, :])**2, axis=2)) >= r[:, None] + r[None, :] - 1e-12)

    if inside_square and non_negative_radii and no_overlap:
        return x, y, r, sum_radii
    else:
        return None

def run_packing():
    n = 26
    hex_circles = hex_lattice(np.ceil(np.sqrt(n * 2 / np.sqrt(3))).astype(int)).T
    hex_circles -= np.min(hex_circles, axis=0)
    hex_circles /= np.max(hex_circles, axis=0)
    initial_radii = np.full(n, 1 / 2 / np.sqrt(3))
    initial_z = np.hstack((hex_circles.flatten(), initial_radii))
    initial_z = initial_z.cast('float64')
    δ = 1e-8
    z = solve_stage(initial_z, δ)
    best = 0
    best_z = z
    for i in range(5):
        z = [np.random.uniform(0, 1, n * 2) * 0.002 - 0.001, np.random.uniform(0, 1, n) * 0.002 - 0.001, z[-n:] + np.random.uniform(0, 1, n) * 0.002 - 0.001]
        z = solve_stage(z, δ)
        if penalised_obj(z) > best:
            best_z = z
            best = penalised_obj(z)
    result = validate_and_output(best_z)
    if result is None:
        raise Exception("Constraint violation detected during validation")
    else:
        x, y, r, sum_radii = result
        return x, y, sum_radii
