import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.optimize import linprog, minimize

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12) or (y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    C = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    R = [0.25] * 4
    Obstacles = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]

    while len(C) < 26:
        Vor = Voronoi([*Obstacles, *C])
        Candidate Vertices = []
        for VTXindex in Vor.ridge_vertices:
            Triple = VTXindex + (type(VTXindex),)
            if Triple[0] < Triple[1] < Triple[2]:
                R1 = R[Triple[0]] if Triple[0] < 4 else 0
                R2 = R[Triple[1]] if Triple[1] < 4 else 0
                R3 = R[Triple[2]] if Triple[2] < 4 else 0
                d1 = np.min([obst[0] for obst in Obstacles if obst[1] == Triple[0]])
                d2 = np.min([obst[0] - obt[1] for obst in Obstacles if obst[1] == Triple[1]])
                d3 = np.min([obst[1] for obst in Obstacles if obst[1] == Triple[2]])
                d = min(d1, d2, d3)
                Candidate Vertices.append((0.5 * (d1 + d2 + d3), VTXindex))
        if Candidate Vertices:
            p_star, VTXindex = max(Candidate Vertices)
            if p_star > 1e-6:
                newCenter = Vor.vertices[VTXindex[0]] + Vor.vertices[VTXindex[1]] + Vor.vertices[VTXindex[2]].mean(axis=0)
                newCenter /= np.linalg.norm(newCenter)
                newRadius = p_star
                C.append(tuple(newCenter))
                R.append(newRadius)
                Obstacles.append(tuple(newCenter))
            else:
                break
        else:
            break

    R = np.array(R)
    C = np.array(C)
    A = []
    for i in range(len(C)):
        for j in range(i + 1, len(C)):
            A.append([1, 1, 0] if j < 4 else [0, 1, 1] + [0] * (j - 4) + [1, -1, 0] if i < 4 else [0, 0, 1] + [0] * (i - 4) + [1, 0, -1] + [0] * (j - i - 4) if i != j else [0, 0, 0] + [0] * (j - i - 4) + [2, 0, 0, -1])
            A.append([0, 0, 1])
            A.append([1, 0, 0])
            A.append([0, 1, 0])
            A.append([-1, 0, 0])
            A.append([0, -1, 0])
            A.append([0, 0, -1])
    B = [np.linalg.norm(np.array(C[i]) - np.array(C[j])) - R[i] - R[j] for i in range(len(C)) for j in range(i + 1, len(C))]
    B.extend([1, 1, 1, -0.5, -0.5, -0.5] * len(C))
    lin_res = linprog(-R, A_ub=A, b_ub=B, bounds=[(0, 0.5)] * len(R))
    if lin_res.success:
        Return np.array([C, lin_res.x, np.sum(lin_res.x)])
    else:
        Lin_res = linprog(np.zeros(len(R)), A_ub=[[1, 1, 0] * 4 + [2, 0, 0, -1] * 4 + [0, 0, 1] for _ in range(len(R))], b_ub=[0.5 + 0.5 + 0.5, 0.5 + 0.5 + 0.5 for _ in range(len(R))], bounds=[(0.5, 0.5) for _ in range(len(R))])
        Return np.array([C, Lin_res.x, np.sum(Lin_res.x)])
