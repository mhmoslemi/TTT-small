import numpy as np
from scipy.optimize import minimize

def objective(z):
    return -np.sum(z[2::3])

def boundary_constraints(z):
    n = len(z) // 3
    bounds = np.zeros((4 * n, 2))
    bounds[:n, 0] = np.column_stack([z[::3] - z[2::3], z[3::3], np.ones(n)])
    bounds[n:2*n, 0] = np.column_stack([1 - z[::3] - z[2::3], z[3::3], np.ones(n)])
    bounds[2*n:3*n, 0] = np.column_stack([z[1::3] - z[2::3], z[3::3], np.ones(n)])
    bounds[3*n:, 0] = np.column_stack([1 - z[1::3] - z[2::3], z[3::3], np.ones(n)])
    bounds[:, 1] = np.ones(4 * n)
    return bounds

def pairwise_constraints(z):
    n = len(z) // 3
    constraints = np.zeros((n * (n - 1) // 2, 7))
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints[idx, :2] = z[i * 3:i * 3 + 2]
            constraints[idx, 2:4] = z[j * 3:j * 3 + 2]
            constraints[idx, 4:] = np.array([z[i * 3 + 2] + z[j * 3 + 2]])
            idx += 1
    return constraints

def evaluate_constraints(z):
    n, m = len(z) // 3, 5
    constraints = np.zeros((4 * n + m, 7))
    constraints[:4 * n, :] = boundary_constraints(z)
    constraints[4 * n:, :] = pairwise_constraints(z)
    return constraints[:, :7] - constraints[:, 7:]

def jac(z):
    n = len(z) // 3
    column = np.zeros((4 * n + n * (n - 1) // 2))
    column[:4 * n] = np.vstack([-np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), np.ones(n), np.zeros(n), np.zeros(n), np.zeros(n), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), np.ones(n), np.zeros(n), np.zeros(n), np.zeros(n), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), -np.eye(n, n, dtype=int), np.ones(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n), np.zeros(n)])
    return column

def run_packing():
    np.random.seed(42)
    N = 6
    best_solution = None
    best_value = -np.inf
    maxiter = 2000
    gtol = 1e-8
    ftol = 1e-12
    bounds = []
    for _ in range(N):
        initial_center = 0.5 * np.random.rand(2, 26)
        initial_radius = np.min(initial_center, 1 - initial_center) / 2
        initial_radius *= np.random.uniform(0.3, 0.8, 26)
        z0 = np.empty(int(len(initial_center) + len(initial_radius)))
        z0[:len(initial_center)] = initial_center.flatten()
        z0[len(initial_center):] = initial_radius
        bounds += [(0, None) for _ in initial_center.flatten()]
        bounds += [(0, 0.5) for _ in initial_radius]
        result = minimize(objective, z0, method='trust-constr', bounds=bounds, constraints={'type': 'ineq', 'fun': evaluate_constraints}, jac=jac, options={'maxiter': maxiter, 'gtol': gtol, 'ftol': ftol})
        if result.success:
            value = result.fun
            if value > best_value:
                best_value = value
                best_solution = result.x.reshape((-1, 3))
    
    if best_solution is not None:
        centers = best_solution[:, :2]
        radii = best_solution[:, 2]
        sum_radii = -best_value
        return centers, radii, sum_radii
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0
