import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    positions = np.random.rand(n, 2) * (1 - 2e-6) + 1e-6
    radii = np.full(n, 1e-6)
    
    def force_repulsion(pos, rad):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(pos[i] - pos[j])
                overlap = rad[i] + rad[j] - dist
                if overlap > 0:
                    force = overlap * (pos[j] - pos[i]) / dist
                    forces[i] -= force
                    forces[j] += force
        return forces
    
    def wall_forces(pos, rad):
        forces = np.zeros((n, 2))
        for i in range(n):
            if pos[i, 0] - rad[i] < 0:
                forces[i, 0] += rad[i] - pos[i, 0]
            if pos[i, 0] + rad[i] > 1:
                forces[i, 0] += pos[i, 0] + rad[i] - 1
            if pos[i, 1] - rad[i] < 0:
                forces[i, 1] += rad[i] - pos[i, 1]
            if pos[i, 1] + rad[i] > 1:
                forces[i, 1] += pos[i, 1] + rad[i] - 1
        return forces
    
    def update_positions(pos, rad):
        forces = force_repulsion(pos, rad) + wall_forces(pos, rad)
        velocities = np.zeros((n, 2))
        for i in range(n):
            velocities[i] += forces[i] / np.maximum(1, rad[i])
            pos[i] += 0.1 * velocities[i]  # add some damping
        return pos
    
    def update_radii(pos, rad, tol=1e-10):
        for _ in range(1000):
            new_radii = np.zeros(n)
            for i in range(n):
                min_rad = np.min(np.linalg.norm(pos[i] - pos[j]) - rad[j] - 1e-6 for j in range(n) if i != j)
                cur_rad = min(pos[i, 0], 1 - pos[i, 0], pos[i, 1], 1 - pos[i, 1], min_rad)
                new_radii[i] = cur_rad * (1 + 1e-3)  # allow a small increase
            if np.all(np.abs(new_radii - rad) < tol):
                break
            rad = new_radii
        return rad
    
    def validate_radius(pos, rad):
        for _ in range(100):
            new_radii = update_radii(pos, rad)
            if np.all(np.abs(new_radii - rad) < 1e-6):
                return rad
            rad = new_radii
        return rad
    
    for _ in range(3000):
        positions = update_positions(positions, radii)
        radii = validate_radius(positions, radii)
    
    result_radii = linprog(-np.ones(n), A_le=np.zeros((n, n)), b_le=np.zeros(n),
                           bounds=(1e-6, None), method="highs").x
    return positions, result_radii, np.sum(result_radii)
