import numpy as np
from scipy.spatial import ConvexHull, Delaunay

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def generate_initial_packing(n):
        np.random.seed(42)
        initial_points = np.random.rand(n, 2)
        hull = ConvexHull(initial_points)
        initial_points = initial_points[hull.vertices]
        return initial_points

    def pack_circles(points):
        n = points.shape[0]
        radii = np.zeros(n)
        centers = points.copy()

        for _ in range(100):
            for i in range(n):
                min_radius = 0.0
                for j in range(n):
                    if i != j:
                        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        min_radius = max(min_radius, (dist - radii[j]) / 2)
                radii[i] = min_radius

            # Collision detection and resolution
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if dist < radii[i] + radii[j] - 1e-12:
                        dx = centers[i][0] - centers[j][0]
                        dy = centers[i][1] - centers[j][1]
                        overlap = (radii[i] + radii[j]) - dist
                        dx /= dist
                        dy /= dist
                        centers[i] -= overlap / 2 * 0.9 * np.array([dx, dy])
                        centers[j] += overlap / 2 * 0.9 * np.array([dx, dy])

        return centers, radii

    initial_points = generate_initial_packing(26)
    centers, radii = pack_circles(initial_points)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
