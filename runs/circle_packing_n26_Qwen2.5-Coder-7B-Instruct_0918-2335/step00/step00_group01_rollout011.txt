import numpy as np
from scipy import optimize, spatial


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    r_min = 1e-4
    g = 1e-3
    v_mag = 5e-3
    T_max = 50.0
    eps = 1e-6
    lp_tol = 1e-4

    def event_time(c_i, v_i, c_j, v_j, r_i, r_j):
        del_c = c_i - c_j
        del_v = v_i - v_j
        del_r = r_i + r_j
        a = np.dot(del_v, del_v)
        b = 2 * np.dot(del_c, del_v)
        c = np.dot(del_c, del_c) - del_r * del_r
        if a == 0 or b == 0 or c == 0:
            return np.inf
        discriminant = b**2 - 4 * a * c
        if discriminant < 0:
            return np.inf
        t = (-b + np.sqrt(discriminant)) / (2 * a)
        if t > 0:
            return t
        return np.inf

    def circle_wall_time(c, v, r, axis):
        if v[axis] > 0 or v[axis] == 0 and c[axis] < 0.5:
            dist = c[axis] - r - 0.5
        else:
            dist = r + 0.5 - c[axis]
        return dist / v[axis] if v[axis] != 0 else np.inf

    def resolve_event(event):
        if event['type'] == 'circle-circle':
            i, j = event['indices']
            v_i, v_j = v[i], v[j]
            c_i, c_j = c[i], c[j]
            r_i, r_j = r[i], r[j]
            del_c = c_i - c_j
            norm = np.linalg.norm(del_c)
            v_i -= 2 * r_j / (r_i + r_j) * np.dot(v_i - v_j, del_c) / norm**2 * del_c
            v_j += 2 * r_i / (r_i + r_j) * np.dot(v_i - v_j, del_c) / norm**2 * del_c
            v[i], v[j] = v_i, v_j
        else:
            i, axis = event['indices']
            v[i][axis] *= -1

    def update_events(centers, velocities, radii):
        events = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                e = {'type': 'circle-circle', 'indices': (i, j), 'time': event_time(c[i], v[i], c[j], v[j], r[i], r[j])}
                if e['time'] < T_max:
                    events.append(e)
            for axis in range(2):
                e = {'type': 'circle-wall', 'indices': (i, axis), 'time': circle_wall_time(c[i], v[i], r[i], axis)}
                if e['time'] < T_max:
                    events.append(e)
        return np.array(events, dtype=[('time', float), ('type', np.str_), ('indices', int, 2)])

    def simulate(centers, velocities, radii):
        times = np.array([0.0])
        events = update_events(centers, velocities, radii)
        np.random.shuffle(velocities)
        while len(events) > 0 and times[-1] < T_max:
            t = np.min(events['time'])
            times = np.append(times, times[-1] + t)
            delta_t = times[-2] + t - times[-1]
            centers += velocities * delta_t
            radii += g * delta_t
            event_index = np.argmin(events['time'])
            resolve_event(events[event_index])
            del events[event_index]
            for i in events['indices']:
                if i[0] == event_index or i[1] == event_index:
                    events['time'][np.where(events['indices'] == i)] = np.inf
            events = update_events(centers, velocities, radii)
        return centers, velocities, radii

    def repair(centers, radii):
        tree = spatial.KDTree(centers)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                overlap = radii[i] + radii[j] - dist
                if overlap > eps:
                    centers[i] += overlap / (2 * dist) * (centers[j] - centers[i])
                    centers[j] -= overlap / (2 * dist) * (centers[j] - centers[i])
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
        return centers, radii

    c = np.random.rand(n_circles, 2) - 0.5
    v = np.random.randn(n_circles, 2) * v_mag
    r = np.ones(n_circles) * r_min
    c, v, r = simulate(c, v, r)
    c, r = repair(c, r)
    r -= eps
    result = {'sum_radii': np.sum(r)}
    res = optimize.linprog(-r, A_ub=None, b_ub=None, A_eq=None, b_eq=None, method='highs', bounds=(0, None), options={'tol': lp_tol})
    r = -res.x
    result['sum_radii'] += np.sum(r)
    result['centers'] = c
    result['radii'] = r
    return result['centers'], result['radii'], result['sum_radii']
