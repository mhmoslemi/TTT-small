import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import differential_evolution, linprog, minimize

def evaluate(p):
    C = p.reshape(26, 2)
    d = squareform(pdist(C))
    n = len(C)
    A = np.zeros((n * (n - 1) + 4 * n, n))
    b = np.zeros(n * (n - 1) + 4 * n)
    
    def constraints(i, j, a, b):
        A[i * (n - 1) + j, i] = 1
        A[i * (n - 1) + j, j] = 1
        b[i * (n - 1) + j] = d[i, j]
    
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints(i, j, k, k)
            k += 1
    
    for i in range(n):
        constraints(26 + i, i, k, k)
        A[k, i] = 1
        b[k] = C[i, 0]
        k += 1
        constraints(26 + i, i + n, k, k)
        A[k, i] = 1
        b[k] = 1 - C[i, 0]
        k += 1
        constraints(26 + i, i + 2 * n, k, k)
        A[k, i] = 1
        b[k] = C[i, 1]
        k += 1
        constraints(26 + i, i + 3 * n, k, k)
        A[k, i] = 1
        b[k] = 1 - C[i, 1]
        k += 1
    
    for i in range(n):
        A[k + i, i] = -1
        b[k + i] = 0
    
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, method='highs')
    return -res.fun if res.message == 'Optimal' else -1

def run_packing() -> tuple:
    initial = np.array([[i * (1 / 7) + (i % 2) / 14 for i in range(14)], 
                        [(2 * j / (3 * 7)) for j in range(14) for _ in range(2)]]).T[:26]

    def polisher(x):
        C = x[:26 * 2].reshape(26, 2)
        r = x[26 * 2:]
        d = squareform(pdist(C))
        constraints = (r[:, np.newaxis] + r[np.newaxis, :] <= d)
        r[r[r < 0] < -1e-9] = 0
        r[r > d - 1e-9] = d + 1e-9
        return minimize(evaluate, x, bounds=[(0, 1)] * 52 + [(0, None)] * 26, method='SLSQP', constraints={'type': 'in', 'fun': lambda x: constraints.flatten()}).x
    
    bounds = [(0, 1)] * 26 * 2
    res = differential_evolution(evaluate, bounds, maxiter=200, popsize=3, tol=1e-6, mutation=(0.3, 0.8), recombination=0.7)
    x_best = polisher(res.x)
    C_best = x_best[:26 * 2].reshape(26, 2)
    r_best = x_best[26 * 2:]
    return C_best, r_best, r_best.sum()
