def sutherland_hodgman(polygon, half_plane):

    a, b = half_plane
    for p1, p2 in zip(polygon, polygon[1:] + polygon[:1]):

        x2, y2 = p2
        c1 = a * x1 + b * y1 <= b
        c2 = a * x2 + b * y2 <= b
        if c1 and c2:
            result.append(p2)

            result.append((x2 + (b - a * x2 * y1) / (a * y2 - a * y1), y2 + (-a * x2 * x2 + a * x2 * x1 + b - b * y2 * y1) / (a * y2 - a * y1)))
