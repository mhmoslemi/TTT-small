import numpy as np
from scipy.optimize import minimize, check_grad

def objective(z):
    return -np.sum(z[2::3])  # Negative sum radii so we maximize

def boundary_constraints(z):
    constraints = []
    for i in range(26):
        constraints.extend([
            z[3 * i] - z[3 * i + 2],   # x_i - r_i >= 0
            1 - z[3 * i] - z[3 * i + 2],  # 1 - x_i - r_i >= 0
            z[3 * i + 1] - z[3 * i + 2],  # y_i - r_i >= 0
            1 - z[3 * i + 1] - z[3 * i + 2]  # 1 - y_i - r_i >= 0
        ])
    return constraints

def overlap_constraints(z):
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            radius_sum = z[3 * i + 2] + z[3 * j + 2]
            constraint = (z[3 * i] - z[3 * j])**2 + (z[3 * i + 1] - z[3 * j + 1])**2 - radius_sum**2
            constraints.append(constraint)
    return constraints

def constraints(z):
    return boundary_constraints(z) + overlap_constraints(z)

def jacobian(z):

            elif k == 1:  # y_i
                jac[4 * i + k, 3 * i + 1] = 1.0
            elif k == 2:  # r_i
                jac[4 * i + k, 3 * i + 2] = -1.0
            elif k == 3:  # boundary constraints on x bounds
                if (k == 2 and i == 0) or \ (k == 3 and i == 25):
                    jac[4 * i + k, 3 * i] = 1.0

                radius_sum = z[3 * i + 2] + z[3 * j + 2]
