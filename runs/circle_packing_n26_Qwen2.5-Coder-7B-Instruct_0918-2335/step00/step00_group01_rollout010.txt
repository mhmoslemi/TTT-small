# Objective function
def objective(x):
    centers = x[:n*2].reshape(n, 2)
    radii = x[n*2:]
    penalty = 0
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        if r_i > min(x_i, 1-x_i, y_i, 1-y_i):
            penalty += M_bound * (r_i - min(x_i, 1-x_i, y_i, 1-y_i))**2
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_j = radii[j]
            if dist < r_i + r_j:
                penalty += M_overlap * (r_i + r_j - dist)**2
    return -np.sum(radii) + penalty

# Gradients
def gradient(x):
    centers = x[:n*2].reshape(n, 2)
    radii = x[n*2:]
    grad_positions = np.zeros((n * 2, n * 2))
    grad_radii = np.zeros(n * 2)
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        if r_i > min(x_i, 1-x_i, y_i, 1-y_i):
            grad_radii[i] += -2 * M_bound * (r_i - min(x_i, 1-x_i, y_i, 1-y_i))
            grad_positions[i*2] += -2 * M_bound * (x_i - min(x_i, 1-x_i, y_i, 1-y_i)) * np.where(r_i > x_i, 1, -1)
            grad_positions[i*2+1] += -2 * M_bound * (y_i - min(x_i, 1-x_i, y_i, 1-y_i)) * np.where(r_i > y_i, 1, -1)
    overlap_matrix = cdist(centers, centers)
    for i in range(n):
        for j in range(i+1, n):
            dist = overlap_matrix[i, j]
            r_j = radii[j]
            if dist < r_i + r_j:
                grad_radii[i] += -2 * M_overlap * (r_i + r_j - dist)
                grad_radii[j] += -2 * M_overlap * (r_i + r_j - dist)
                grad_positions[i*2] += -2 * M_overlap * (centers[i, 0] - centers[j, 0]) * np.where(dist > r_i + r_j, 1, -1)
                grad_positions[i*2+1] += -2 * M_overlap * (centers[i, 1] - centers[j, 1]) * np.where(dist > r_i + r_j, 1, -1)

                grad_positions[j*2+1] += -2 * M_overlap * (centers[j, 1] - centers[i, 1]) * np.where(dist > r_i + r_j, 1, -1)
    grad = np.concatenate([grad_positions.flatten(), grad_radii])
    return grad
