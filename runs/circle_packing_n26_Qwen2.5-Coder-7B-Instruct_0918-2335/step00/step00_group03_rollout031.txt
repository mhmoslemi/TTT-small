import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    # Define circle centers and radii
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Corner circles
    centers[0] = [r_c, r_c]
    radii[0] = r_c
    centers[1] = [1 - r_c, r_c]
    radii[1] = r_c
    centers[2] = [r_c, 1 - r_c]
    radii[2] = r_c
    centers[3] = [1 - r_c, 1 - r_c]
    radii[3] = r_c
    
    # Center circles
    center = 0.5
    edge1 = center - r_i
    edge2 = center + r_i
    
    grid_x = np.repeat([edge1, center, edge2], 3)[:18]
    grid_y = np.repeat([edge1, edge2, center], 6)[:18]
    
    idx = 4
    for x, y in zip(grid_x, grid_y):
        if (np.sqrt((x - 0.5) ** 2 + (y - 0.5) ** 2) - r_i) > 0:
            centers[idx] = [x, y]
            radii[idx] = r_i
            idx += 1
    
    return centers, radii

def compute_sum_radii(radii):
    return np.sum(radii)

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def local_obj(x):
    centers = x[:52].reshape(26, 2)
    radii = x[52:]
    return -compute_sum_radii(radii)

def generate_bounds(r_c_bounds, r_e_bounds, r_i_bounds):
    bounds = []
    bounds.extend([(r_c_bounds[0], r_c_bounds[1]) for _ in range(3)])
    bounds.extend([(r_e_bounds[0], r_e_bounds[1]) for _ in range(3)])
    bounds.extend([(r_i_bounds[0], r_i_bounds[1]) for _ in range(3)])
    return bounds

def generate_constraints(centers, radii):
    n = centers.shape[0]
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq',
                           'fun': lambda v, i=i, r=radii[i]: v[i*3] - r})
        constraints.append({'type': 'ineq',
                           'fun': lambda v, i=i, r=radii[i]: 1 - v[i*3 + 1] - r})
        constraints.append({'type': 'ineq',
                           'fun': lambda v, i=i, r=radii[i]: v[i*3 + 1] - r})
        constraints.append({'type': 'ineq',
                           'fun': lambda v, i=i, r=radii[i]: 1 - v[i*3 + 2] - r})
        constraints.append({'type': 'ineq',
                           'fun': lambda v, i=i, r=radii[i]: v[i*3 + 2] - r})
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((v[i*3: i*3 + 3] - v[j*3: j*3 + 3]) ** 2))
            constraints.append({'type': 'ineq',
                               'fun': lambda v, i=i, j=j, d=dist: v[i*3 + 3] + v[j*3 + 3] - d})
    return constraints

def run_packing():
    r_c_bounds = (0.30, 0.49)
    r_e_bounds = (0.10, 0.25)
    r_i_bounds = (0.05, 0.15)
    
    bounds = generate_bounds(r_c_bounds, r_e_bounds, r_i_bounds)
    
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)
    
    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = generate_bounds([(-np.inf, np.inf)]*26, [(0.0, np.inf)]*26, [(0.0, np.inf)]*26)
    cons = generate_constraints(centers, radii)
    res = minimize(local_obj, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol':1e-10,'gtol':1e-10,'maxiter':2000})
    
    centers = res.x[:52].reshape(26,2)
    radii = res.x[52:]
    
    sum_r = radii.sum()
    return centers, radii, sum_r
