import numpy as np
from random import random, seed

# Constants
S = 80  # Swarm size
T0 = 0.05  # Initial temperature
alpha = 0.995  # Cooling rate
epsilon = 1e-12  # Minimum radius
x_max = 1 - epsilon  # Maximum allowed x-coordinate
y_max = 1 - epsilon  # Maximum allowed y-coordinate
v_max = 0.05  # Maximum velocity component
T_min = 1e-7  # Minimum temperature
num_moves = 20000

num_precise = num_moves - num_coarse

def initialize_particles(n):
    particles = np.zeros((n, 26, 3))
    for p in range(n):
        for i in range(26):
            particles[p, i, 0] = random() * (x_max - 2*epsilon) + epsilon
            particles[p, i, 1] = random() * (y_max - 2*epsilon) + epsilon
            particles[p, i, 2] = random() * min(particles[p, i, 0], particles[p, i, 1], x_max - particles[p, i, 0], y_max - particles[p, i, 1])
    return particles

def project_position(x, y, r):
    x = np.clip(x, r, x_max - r)
    y = np.clip(y, r, y_max - r)
    return x, y

def project_radius(r):
    return max(r, epsilon)

def check_feasibility(centers, radii):
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - epsilon:
                dx = (dist - radii[i] - radii[j]) / 2.0
                direction = (centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                direction /= dist
                centers[i, 0] += dx * direction[0]
                centers[i, 1] += dx * direction[1]
                centers[j, 0] -= dx * direction[0]
                centers[j, 1] -= dx * direction[1]
    return centers

def evaluate_fitness(centers, radii):
    return -np.sum(radii)

def update_velocity(w, particles, p_best, g_best, c1, c2):
    velocities = np.zeros_like(particles)
    for i in range(S):
        person_best = p_best[i]
        global_best = g_best
        velocities[i] = w * particles[i] + c1 * random() * (person_best - particles[i]) + c2 * random() * (global_best - particles[i])
        velocities[i] = np.clip(velocities[i], -v_max, v_max)
    return velocities
