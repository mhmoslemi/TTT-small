import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * n + [(0,0.5)] * n
    var0 = np.random.rand(2 * n)
    def obj(v):
        r = v[n:]
        return -np.sum(r)
    def constraints(v):
        pos = v[:n].reshape(n, 1)
        r = v[n:].reshape(n, 1)
        dists = np.sqrt((pos - pos.T) ** 2 + (r - r.T) ** 2)
        bounds = np.where(np.triu(np.ones((n, n), dtype=bool), k=1), dists - (r + r.T), np.inf)
        return bounds.ravel() - 1e-9
    res = minimize(obj, var0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    if not res.success or np.any(res.fun < -1e-10):
        raise RuntimeError("Optimization failed or constraints not satisfied")
    centers = res.x[:n].reshape(n, 1)
    radii = -res.fun
    return centers, radii, np.sum(radii)
