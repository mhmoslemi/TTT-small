import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # Ensuring reproducibility

    N = 26
    r_min = 1e-3
    iterations = 10000

    def rsa_seed():
        circles = []
        for _ in range(N):
            placed = False
            for _ in range(iterations):
                x, y = np.random.rand(2)
                r = r_min
                if np.all((x - r >= 0) & (x + r <= 1) & (y - r >= 0) & (y + r <= 1)):
                    if all(np.sqrt((x - cx)**2 + (y - cy)**2) > 2 * r for cx, cy in circles):
                        circles.append((x, y, r))
                        placed = True
                        break
            if not placed:
                return None
        circles = np.array(circles)
        return circles[:, :2], circles[:, 2]

    def generate_objective_and_constraints(centers, radii):
        def objective(x):
            return -np.sum(x[N:])

        def gradient(x):
            grad = np.zeros_like(x)
            grad[N:] = -1
            return grad

        def overlap_penalty(x):
            penalty = 0
            for i in range(N):
                for j in range(i + 1, N):
                    dist = np.sqrt(np.sum((x[i*2:i*2+2] - x[j*2:j*2+2])**2))
                    penalty += max(0, (x[N+i] + x[N+j]) - dist)**2
            return penalty

        def boundary_penalty(x):
            penalty = 0
            for i in range(N):
                penalty += max(0, x[N+i] - x[2*i])**2
                penalty += max(0, x[N+i] - (1 - x[2*i]))**2
                penalty += max(0, x[N+i] - x[2*i+1])**2
                penalty += max(0, x[N+i] - (1 - x[2*i+1]))**2
            return penalty

        def penalty(x):
            return overlap_penalty(x) + boundary_penalty(x)

        def penalty_gradient(x):
            gap = np.maximum(0, np.sqrt(np.sum((x[np.newaxis, 2*i:2*i+2]-x[:, 2*i:2*i+2])**2, axis=1)) - (x[N:N+2*N].reshape(-1, 1) + x[N:N+2*N].reshape(1, -1)))
            gap_penalty = gap * gap
            diff = (x[np.newaxis, 2*i:2*i+2] - x[:, 2*i:2*i+2])
            diff /= np.maximum(x[N:N+2*N].reshape(-1, 1) + x[N:N+2*N].reshape(1, -1), 1e-8)
            grad = np.sum(diff * gap_penalty[:, np.newaxis, :], axis=0)
            grad[N:] = -1
            grad[N::2] += gap_penalty[np.diag_indices(N), :]
            grad[N+1::2] += gap_penalty[np.diag_indices(N), :]
            grad[::2] -= gap_penalty[:, np.arange(N) // 2]
            grad[1::2] -= gap_penalty[:, np.arange(N) // 2]
            grad = np.maximum(grad, 0)
            return grad.flatten()

        constraints = [{'type': 'ineq', 'fun': lambda x: x[N:]},
                      {'type': 'ineq', 'fun': lambda x: 0.5 - x[N:]},
                      {'type': 'ineq', 'fun': lambda x: x[2*i] - x[N+i] for i in range(N)},
                      {'type': 'ineq', 'fun': lambda x: x[2*i+1] - x[N+i] for i in range(N)},
                      {'type': 'ineq', 'fun': lambda x: x[N+i] - (1 - x[2*i]) for i in range(N)},
                      {'type': 'ineq', 'fun': lambda x: x[N+i] - (1 - x[2*i+1]) for i in range(N)}]

        return objective, gradient, penalty, penalty_gradient, constraints

    def global_scaling(centers, radii):
        s_max = min(min(np.sqrt(np.sum((centers[np.newaxis, i] - centers[j])**2)) / (radii[i] + radii[j]) for i in range(N) for j in range(i + 1, N)),
                   min(1 - radii, key=lambda r: r & (radii > 0)))
        centers = 0.5 + s_max * (centers - 0.5)
        radii *= s_max
        return centers, radii

    def optimize(centers, radii):
        x0 = np.concatenate((centers.flatten(), radii))
        objective, gradient, penalty, penalty_gradient, constraints = generate_objective_and_constraints(centers, radii)
        result = minimize(objective, x0, method='L-BFGS-B', jac=gradient, constraints=constraints, bounds=[(0, 1)] * 52 + [(0, 0.5)] * N)
        return result.x[:N*2].reshape((-1, 2)), result.x[N:]

    centers, radii = rsa_seed()
    centers, radii = global_scaling(centers, radii)
    centers, radii = optimize(centers, radii)
