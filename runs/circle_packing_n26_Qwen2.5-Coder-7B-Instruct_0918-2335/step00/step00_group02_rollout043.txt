import numpy as np
from scipy.spatial import ConvexHull, voronoi, voronoi_plot_2d
from scipy.spatial.distance import cdist

def generate_initial_centers(num_circles, bounds):
    return np.random.uniform(bounds[0], bounds[1], (num_circles, 2))

def voronoi_diagram_from_centers(centers, bounds):
    extended_bounds = bounds + (bounds / 2)
    extended_bounds = np.vstack((extended_bounds, -extended_bounds))
    extended_centers = np.vstack((centers, centers - extended_bounds[2:]))
    hull = ConvexHull(extended_centers)
    vor = voronoi(extended_centers[hull.vertices], ensure_filter=False)
    return vor

def reshape_voronoi_diagram(vor, num_circles, bounds):
    centers = vor.points[:num_circles]
    vertices = vor.vertices[vor.point_region]
    return centers, vertices

def filter_non_intersecting_voronoi_cells(centers, vertices, bounds):
    inside_bounds = np.all((centers >= bounds[0]) & (centers <= bounds[1]), axis=1)
    valid_indices = [i for i in range(len(centers)) if inside_bounds[i] and (vertices[i] >= bounds[0]).all() and (vertices[i] <= bounds[1]).all()]
    return centers[valid_indices], vertices[valid_indices]

def assign_radii_to_vertices(vertices, bounds):
    occupied_points = set()
    new_vertices = []
    for vertex in vertices:
        available_space = bounds
        for point in occupied_points:
            distance = np.linalg.norm(vertex - point)
            if distance < 1e-6:
                available_space = np.clip(available_space - 2 * np.array([distance/2, distance/2]), 0, None)
        max_radius = min(available_space[:, 0], available_space[:, 1]) / 2
        new_vertices.append(vertex + max_radius * (2 * np.random.rand(2) - 1))
        occupied_points.add(new_vertices[-1])
    return new_vertices

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    bounds = np.array([0, 1])
    initial_centers = generate_initial_centers(num_circles, bounds)
    vor = voronoi_diagram_from_centers(initial_centers, bounds)
    centers, vertices = reshape_voronoi_diagram(vor, num_circles, bounds)
    centers, vertices = filter_non_intersecting_voronoi_cells(centers, vertices, bounds)
    radii = assign_radii_to_vertices(vertices, bounds)
    
    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
