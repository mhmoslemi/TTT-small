import numpy as np
from scipy.optimize import NonlinearConstraint, minimize

def constraint_func(v):
    n = 26
    x = v[:52].reshape((n, 2))
    r = v[52:]
    arr = []
    for i in range(n):
        arr.append(x[i, 0] - r[i])
        arr.append(1 - x[i, 0] - r[i])
        arr.append(x[i, 1] - r[i])
        arr.append(1 - x[i, 1] - r[i])
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for i, j in pair_indices:
        dx = x[i, 0] - x[j, 0]
        dy = x[i, 1] - x[j, 1]
        dist2 = dx ** 2 + dy ** 2
        arr.append(dist2 - (r[i] + r[j]) ** 2)
    return np.concatenate(arr)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    positions = []
    for i in range(5):
        for j in range(5):
            positions.append(np.array([(j + 0.5) / 5.0, (i + 0.5) / 5.0]))
            if len(positions) == 25:
                break
        if len(positions) == 25:
            break
    positions.append(np.array([0.6, 0.9]))
    positions = np.array(positions)
    radii_initial = np.full(n, 0.04)
    initial_guess = np.concatenate([positions.ravel(), radii_initial])
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * n
    cons = NonlinearConstraint(constraint_func, 0, np.inf)
    result = minimize(fun=lambda v: -np.sum(v[52:]), x0=initial_guess, bounds=bounds, constraints=cons, method='trust-constr', options={'maxiter': 2000, 'gtol': 1e-12, 'disp': False})
    if not result.success:
        centers = np.resize(positions, (26, 2))
        radii = np.full(n, 0.105)
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    centers = result.x[:52].reshape((n, 2))
    radii = result.x[52:]
    sum_radii = np.sum(radii)
    shrink_factor = 0.999
    for _ in range(10):
        pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for i, j in pair_indices:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            sum_r_ij = radii[i] + radii[j]
            if dist < sum_r_ij:
                factor = (sum_r_ij - dist) / sum_r_ij
                radii[i] *= factor
                radii[j] *= factor
        sum_radii = np.sum(radii)
        if sum_radii >= 2.636:
            break
    return centers, radii, sum_radii
