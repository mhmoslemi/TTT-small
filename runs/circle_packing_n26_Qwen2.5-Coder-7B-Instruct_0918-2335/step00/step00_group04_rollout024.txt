import numpy as np
from scipy.optimize import linprog

def setup_initial_grid(n_circles):
    radius = 1 / n_circles ** 0.5
    grid = np.linspace(radius, 1 - radius, int(n_circles ** 0.5))
    centers = np.array([[x, y] for x in grid for y in grid])[:n_circles]
    return centers, radius

def find_closest_pairs(centers, r):
    n = len(centers)
    distances = ((centers[:, None, :] - centers[None, :, :])**2).sum(axis=-1)
    max_radius = r[:, None] + r[None, :]
    max_radius[max_radius < distances] = np.inf
    pairs = np.argwhere((distances < max_radius) & (distances > 1e-5))
    return pairs

def solve_lp(centers, boundary_constraints):
    n = len(centers)
    c = np.ones(n)
    A = np.vstack([centers[:, 0] + centers[:, 1] - 1, centers[:, 0] + boundary_constraints, centers[:, 1] + boundary_constraints]).T
    b = np.ones(n)
    for i in range(n):
        b[i] -= centers[i, 1] + boundary_constraints[i]
        b[n + i] += centers[i, 1] + boundary_constraints[i]
        b[2 * n + i] -= centers[i, 0] + boundary_constraints[i]
        b[3 * n + i] += centers[i, 0] + boundary_constraints[i]
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    max_iter = 200
    tau = 1e-3
    gamma = 0.95
    n_circles = 26

    centers, boundary_constraints = setup_initial_grid(n_circles)
    objective = 0
    for i in range(max_iter):
        r = solve_lp(centers, boundary_constraints)
        if i % 50 == 0:
            centers += 0.02 * np.random.randn(*centers.shape)
            centers = np.maximum(np.minimum(centers, 1), 0)

        pairs = find_closest_pairs(centers, r)
        forces = np.zeros_like(centers)
        for i, j in pairs:
            d_ij = np.linalg.norm(centers[i] - centers[j])
            s_ij = r[i] + r[j] - d_ij
            if s_ij > 0:
                f_ij = (centers[i] - centers[j]) * s_ij / d_ij
                forces[i] += f_ij
                forces[j] -= f_ij

            w_i = np.maximum(0, r[i] - (np.minimum(centers[i, 0], 1, centers[i, 1], 1-centers[i, 0], 1-centers[i, 1])))
            w_i[w_i > 0] *= np.sign(w_i[w_i > 0])
            forces[i] += w_i

        centers -= tau * forces
        centers = np.maximum(np.minimum(centers, 1), 0)

        if np.linalg.norm(np.dot(forces.T, forces), 2) < 1e-8:
            break

        tau *= gamma

    final_r = solve_lp(centers, boundary_constraints)
    objective = np.sum(final_r)
    return centers, final_r, objective
