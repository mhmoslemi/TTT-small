import numpy as np
from scipy.optimize import differential_evolution, minimize

def make_seed(r_c, r_e, r_i):
    centers = np.array([
        [r_c, r_c], [1 - r_c, r_c], [r_c, 1 - r_c], [1 - r_c, 1 - r_c],
        [0.5, r_e], [0.5, 1 - r_e], [r_e, 0.5], [1 - r_e, 0.5],
        [r_i * np.cos(np.pi / 3), r_i * np.sin(np.pi / 3)],
        [1 - r_i * np.cos(np.pi / 3), r_i * np.sin(np.pi / 3)],
        [r_i * np.cos(np.pi / 3), 1 - r_i * np.sin(np.pi / 3)],
        [1 - r_i * np.cos(np.pi / 3), 1 - r_i * np.sin(np.pi / 3)],
        [r_i * np.cos(2 * np.pi / 3), r_i * np.sin(2 * np.pi / 3)],
        [1 - r_i * np.cos(2 * np.pi / 3), r_i * np.sin(2 * np.pi / 3)],
        [r_i * np.cos(2 * np.pi / 3), 1 - r_i * np.sin(2 * np.pi / 3)],
        [1 - r_i * np.cos(2 * np.pi / 3), 1 - r_i * np.sin(2 * np.pi / 3)],
        [r_i * np.cos(3 * np.pi / 3), r_i * np.sin(3 * np.pi / 3)],
        [1 - r_i * np.cos(3 * np.pi / 3), r_i * np.sin(3 * np.pi / 3)],
        [r_i * np.cos(3 * np.pi / 3), 1 - r_i * np.sin(3 * np.pi / 3)],
        [1 - r_i * np.cos(3 * np.pi / 3), 1 - r_i * np.sin(3 * np.pi / 3)],
        [r_i * np.cos(4 * np.pi / 3), r_i * np.sin(4 * np.pi / 3)],
        [1 - r_i * np.cos(4 * np.pi / 3), r_i * np.sin(4 * np.pi / 3)],
        [r_i * np.cos(4 * np.pi / 3), 1 - r_i * np.sin(4 * np.pi / 3)],
        [1 - r_i * np.cos(4 * np.pi / 3), 1 - r_i * np.sin(4 * np.pi / 3)],
        [r_i * np.cos(5 * np.pi / 3), r_i * np.sin(5 * np.pi / 3)],
        [1 - r_i * np.cos(5 * np.pi / 3), r_i * np.sin(5 * np.pi / 3)],
        [r_i * np.cos(5 * np.pi / 3), 1 - r_i * np.sin(5 * np.pi / 3)],
        [1 - r_i * np.cos(5 * np.pi / 3), 1 - r_i * np.sin(5 * np.pi / 3)]
    ])
    radii = np.zeros(26)
    radii[:4] = r_c
    for i in range(4, 18):
        radii[i] = r_i
    return centers, radii

def compute_sum_radii(radii):
    return radii.sum()

def objective(params):
    r_c, r_e, r_i = params
    centers, radii = make_seed(r_c, r_e, r_i)
    return -compute_sum_radii(radii)

def constraint_func(v):
    centers = v[:52].reshape(26, 2)
    radii = v[52:]
    cons = []
    for i in range(4):
        x, y = centers[i]
        r = radii[i]
        cons.append([x + r - 1])
        cons.append([1 - (x - r)])
        cons.append([y + r - 1])
        cons.append([1 - (y - r)])
    for i in range(4, 18):
        x, y = centers[i]
        r = radii[i]
        for j in range(i + 1, 18):
            x2, y2 = centers[j]
            r2 = radii[j]
            dist = np.sqrt((x - x2)**2 + (y - y2)**2)
            cons.append([dist - (r + r2)])
    return np.hstack(cons)

def run_packing():
    bounds = [(0.30, 0.49), (0.10, 0.25), (0.05, 0.15)]
    result_de = differential_evolution(objective, bounds, popsize=10, tol=1e-5, polish=False)
    r_c, r_e, r_i = result_de.x
    centers, radii = make_seed(r_c, r_e, r_i)

    x0 = np.hstack([centers.flatten(), radii])
    bounds_slsqp = [(-np.inf, np.inf)] * 52
    cons = {'type': 'ineq', 'fun': constraint_func}
    res = minimize(lambda v: -objective(v[:3]), x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 2000})
    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]

    sum_r = compute_sum_radii(radii)
    return centers, radii, sum_r

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")
    else:
        print(f"Packing valid, sum of radii: {sum_r}")
