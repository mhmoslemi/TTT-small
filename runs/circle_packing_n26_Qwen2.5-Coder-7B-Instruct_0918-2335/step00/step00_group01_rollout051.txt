import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial centers in a hexagonal pattern
    row_1 = np.array([[i/6 + 1/6, 0.10] for i in range(6)])
    row_2 = np.array([[i/6 + 2/6, 0.18] for i in range(5)])
    row_3 = np.array([[i/6 + 3/6, 0.26] for i in range(6)])
    row_4 = np.array([[i/6 + 4/6, 0.34] for i in range(5)])
    row_5 = np.array([[i/6 + 5/6, 0.42] for i in range(4)])
    centers = np.vstack([row_1, row_2, row_3, row_4, row_5, [[0.5, 0.5]]])
    
    best_obj = -np.inf
    best_centers, best_radii = None, None

    for _ in range(3):
        np.random.shuffle(centers)
        for _ in range(200):
            # Solve the LP to find optimal radii for the current centers
            c = -np.ones(n)
            A_ub = []
            b_ub = []
            for i in range(n):
                for j in range(i+1, n):
                    d_ij = np.linalg.norm(centers[i] - centers[j])
                    A_ub.append([1, 1])
                    b_ub.append(d_ij)
                    
                A_ub.append([1, 0])
                A_ub.append([0, 1])
                A_ub.append([1, 0])
                A_ub.append([0, 1])
                b_ub.append(centers[i, 0])
                b_ub.append(1 - centers[i, 0])
                b_ub.append(centers[i, 1])
                b_ub.append(1 - centers[i, 1])
            
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
            
            if res.success and res.fun > best_obj:
                best_obj = res.fun
                best_centers = centers.copy()
                best_radii = res.x

            # Compute forces from duals
            pressures = res.dual
            forces = np.zeros((n, 2))
            for i in range(n):
                ij = 0
                for j in range(i+1, n):
                    d_ij = np.linalg.norm(centers[i] - centers[j])
                    if pressures[ij] > 0:
                        forces[i] += pressures[ij] * (centers[i] - centers[j]) / d_ij
                    ij += 1

                for lim in range(4):
                    if pressures[ij] > 0:
                        if lim == 0:
                            forces[i] += pressures[ij]
                        elif lim == 1:
                            forces[i] -= pressures[ij]
                        elif lim == 2:
                            forces[i] += pressures[ij]
                        else:
                            forces[i] -= pressures[ij]
                    ij += 1

            # Update centers
            alpha = 0.01
            centers += alpha * forces

            # Adjust step size based on convergence
            if _ % 10 == 0:
                alpha *= 0.9
        
        if _ < 2:
            # Add small random jitter to centres
            centers += 0.02 * (np.random.rand(n, 2) - 0.5)
    
    # Shrink radii slightly
    eps = 1e-7
    best_radii = np.maximum(0.0, best_radii - eps)
    
    return best_centers, best_radii, np.sum(best_radii)
