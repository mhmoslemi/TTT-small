import numpy as np
from scipy.optimize import minimize

def hex_lattice(n, s=1/8):
    """ Generate n circles on a hexagonal lattice inside [0,1]^2 """
    rows = np.arange(np.ceil(np.sqrt(n/3)), dtype=float)
    circles_per_row = int(2 * (rows + 0.5))
    widths = circles_per_row * s
    valid_rows = rows[widths < 1]
    cols = np.arange(circles_per_row).astype(float)
    offsets = np.where(circles_per_row % 2, cols / 2, (cols + 0.5) / 2)
    y = valid_rows[:, None] * (np.sqrt(3) / 2) * s + valid_rows[0] * s
    x = (offsets + offsets[:, 1][:, None] * 0.5) % 1
    x += np.array([[0, 1], [0, 1]]).flatten() [:len(x)] * 0.5 * s
    radius = 0.5 * s * np.ones_like(x)
    mask = (x < 1 - radius) & (y < 1 - radius) & (y >= radius) & (x >= radius)
    x = x[mask]
    y = y[mask]
    radius = radius[mask]
    if len(x) < n:
        return hex_lattice(n, s - 0.001 * s)
    return x, y, radius
