import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    N = 26
    r_min = 1e-3
    iterations = 10000

    def random_sequential_addition():
        centers = []
        for _ in range(N):
            attempts = 0
            while attempts < iterations:
                center = np.random.rand(2)
                radius = r_min
                if all(np.linalg.norm(center - c) > 2 * radius for c in centers):
                    centers.append(center)
                    break
                attempts += 1
            else:
                return random_sequential_addition()
        return np.array(centers), np.full(N, r_min)

    def physics_based_relaxation(centers, radii):
        T = 2000
        dt = 1e-3
        for _ in range(T):
            d = np.linalg.norm((centers[:, np.newaxis] - centers[np.newaxis, :])[:, 1:], axis=2)
            gap = np.maximum(0, d - (radii[:, np.newaxis] + radii[np.newaxis, :]))
            repulsive_force = gap / (d + 1e-6)
            side_force = np.zeros((N, 4))
            side_force[np.arange(N), gap.argmax(axis=1)] = -1 * gap.max(axis=1) / (radii + 1e-6)
            f = 0.5 * gap / (radii[:, np.newaxis] + radii[np.newaxis, :]) * (repulsive_force + side_force)
            centers += dt * f.sum(axis=0)
            radii += dt * f.sum(axis=1)
            radii = np.clip(radii, 0, min(1 - np.linalg.norm(centers, axis=1), 0.5))
        return centers, radii

    def global_uniform_expansion(centers, radii):
        s_max = min(np.minimum(1 - centers[:, 0], 1 - centers[:, 1], centers[:, 0], centers[:, 1]), 
                     np.minimum(1 / (radii + 1e-6)))
        centers = 0.5 + s_max * (centers - 0.5)
        radii *= s_max
        return centers, radii

    def optimize(centers, radii):
        x = centers.flatten()
        r = radii
        def objective(x):
            r = x[-N:]
            return -r.sum()
        def penalty(x):
            r = x[-N:]
            pen = 0
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(centers[i] - centers[j])
                    pen += max(0, (r[i] + r[j]) - d) ** 2
                for s in range(4):
                    d = centers[i][s % 2] + r[i] if s < 2 else 1 - centers[i][s % 2] - r[i]
                    pen += max(0, r[i] - d) ** 2
            return alpha * pen
        def fun(x):
            return objective(x) + penalty(x)
        def grad(x):
            r = x[-N:]
            gr = -np.ones_like(x)
            gr[-N:] = objective(x)
            pen = 0
           pen_grad = np.zeros_like(x)
           pen_grad[-N:] = np.zeros_like(r)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.linalg.norm(centers[i] - centers[j])
                    coeff = max(0, (r[i] + r[j]) - d) ** 2
                    if coeff > 1e-6:
                        disp = centers[i] - centers[j]
                        pen_grad[-N:] += 2 * coeff / d * disp
                        gr[-N:] -= 2 * disp * (coeff / d)
                for s in range(4):
                    d = centers[i][s % 2] + r[i] if s < 2 else 1 - centers[i][s % 2] - r[i]
                    coeff = max(0, r[i] - d) ** 2
                    if coeff > 1e-6:
                        pen_grad[-N:] += 2 * coeff / d * (-1 if s % 2 == 0 else 1)
                        gr[-N:] += 2 * (-1 if s % 2 == 0 else 1) * (coeff / d)
            pen_grad[-N:] *= alpha
            gr[-N:] *= -alpha
            pen = alpha * pen
            return fun(x), gr + pen_grad
        alpha = 1e5
        bounds = np.array([(0, 1)] * 2 * N + [(0, 0.5)] * N)
        result = minimize(fun, np.concatenate((centers.flatten(), radii)),jac=grad, method='L-BFGS-B', bounds=bounds, tol=1e-10, options={'maxiter': 2000})
        return result.x[:-N].reshape(-1, 2), result.x[-N:]

    centers, radii = random_sequential_addition()
    centers, radii = physics_based_relaxation(centers, radii)
    centers, radii = global_uniform_expansion(centers, radii)
    centers, radii = optimize(centers, radii)
    return centers, radii, radii.sum()
