import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x = np.random.uniform(0.01, 0.99, n)
    y = np.random.uniform(0.01, 0.99, n)
    r = np.ones(n) * 1e-6
    maxIter = 3000
    Δt = 1e-3
    α = 4.0
    β = 0.8
    ε = 1e-12
    δr = 1e-4
    tol = 1e-10

    for step in range(maxIter):
        F = np.zeros_like((n, n, 2))
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                overlap = r[i] + r[j] - d
                if overlap > 0:
                    F[i, j] = α * overlap * (x[i] - x[j]) / d
                    F[j, i] = α * overlap * (x[j] - x[i]) / d

        Fx = np.sum(F, axis=0)
        Fy = np.sum(F, axis=0)

        wx = np.array([α / (x[i] + r[i] - ε) if x[i] > r[i] + ε else -np.inf
                       for i in range(n)])
        wx[wx == -np.inf] = 0
        wx = np.minimum(wx, 1 - x - r)

        wy = np.array([α / (y[i] + r[i] - ε) if y[i] > r[i] + ε else -np.inf
                       for i in range(n)])
        wy[wy == -np.inf] = 0
        wy = np.minimum(wy, 1 - y - r)

        for i in range(n):
            wx[i] = min(wx[i], np.sum(r[:(i+1)]))
            wy[i] = min(wy[i], np.sum(r[:(i+1)]))

        Fx += β * np.sum(wx * np.eye(n), axis=1)
        Fy += β * np.sum(wy * np.eye(n), axis=1)

        v = np.zeros(n)
        for i in range(n):
            v += Δt * (F[i] / 1.0 - β * v)

        x += Δt * v[:, 0]
        y += Δt * v[:, 1]

        r += δr

        for i in range(n):
            r[i] = np.min([
                np.inf if x[i] < r[i] + ε else 0,
                1 - x[i] - ε,
                np.inf if y[i] < r[i] + ε else 0,
                1 - y[i] - ε,
                np.inf if np.any(np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                                  < r[i] + r[j] - 1e-12 for j in range(n) if i != j) else np.inf,
                np.min(r)
            ])

        if step > 0 and np.max(np.abs(r / np.roll(r, 1) - 1)) < tol:
            break

    constraints = [{'type': 'ineq', 'fun': lambda z, i=i: r[i] - np.min(r)} for i in range(n)]
    res = linprog(1 - np.array(r), method='highs', constraints=constraints, bounds=(ε, None))
    r_opt = res.x

    if not np.all(np.equal(r_opt, r)):
        r = r_opt

    r[r <= 0] = ε

    return np.array([x, y]).T, r, np.sum(r)
