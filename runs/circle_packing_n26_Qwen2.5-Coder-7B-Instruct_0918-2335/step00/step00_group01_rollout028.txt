def repair_overlap(centers, radii):
    # This is a simple repair function that checks for overlaps
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j]:
                # Move circles apart
                direction = (centers[i] - centers[j]) / dist
                centers[i] += direction * (0.5 * (radii[i] + radii[j]) - dist)
                centers[j] -= direction * (0.5 * (radii[i] + radii[j]) - dist)
    return centers, radii
