import numpy as np

def run_packing():
    seed = 42
    np.random.seed(seed)
    num_circles = 26

    def init_circles(variant):
        step = 0.2
        layout = np.arange(25).reshape(5, 5)
        layout[1::2, 2] = 5
        layout = layout.layout_to_coordinates(step).ravel()
        extra = np.array([(0.6, 0.9), (0.4, 0.9)][variant-1])
        centers = np.vstack((np.hstack((layout[:num_circles-2], extra[variant-1])) / step))
        return centers, np.full(num_circles, 0.01)

    def inflate(circles, Δr, ε):
        Δr_buffer = 1e-6
        max_iter = 5000
        for _ in range(max_iter):
            changed = False
            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    d_ij = np.linalg.norm(circles[i] - circles[j])
                    if d_ij < (radii[i] + radii[j] - ε):
                        delta = (radii[i] + radii[j] - d_ij) / (2 * d_ij)
                        circles[i] += delta * (circles[i] - circles[j]) / 2
                        circles[j] -= delta * (circles[i] - circles[j]) / 2
                        changed = True
            circles = np.clip(circles, radii, 1 - radii)
            radii += Δr * np.minimum(1 - radii, 1 - np.abs(circles - 0.5))
            if not changed:
                break
        return circles, radii

    def post_process(circles, radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                d_ij = np.linalg.norm(circles[i] - circles[j])
                overlap = (radii[i] + radii[j] - d_ij) / 2
                if overlap > 1e-12:
                    proportion = overlap / (radii[i] + radii[j])
                    radii[i] *= 1 - proportion
                    radii[j] *= 1 - proportion
        return circles, radii

    def calculate_sum_radii(circles, radii):
        return np.sum(radii)

    best_packing = None
    best_sum_radii = -np.inf
    for variant in range(2):
        centers, initial_radii = init_circles(variant + 1)
        updated_centres, updated_radii = inflate(centers, 0.0005, 1e-6)
        updated_centres, updated_radii = post_process(updated_centres, updated_radii)
        sum_radii = calculate_sum_radii(updated_centres, updated_radii)
        if sum_radii > best_sum_radii:
            best_packing = updated_centres, updated_radii
            best_sum_radii = sum_radii

    if best_sum_radii < 2.636:
        centers, _ = init_circles(1)
        radii = np.full(num_circles, 0.105)
        centers[5::5, 2] = [0.6, 0.4]
        updated_centres, updated_radii = inflate(centers, 0.001, 1e-6)
        updated_centres, updated_radii = post_process(updated_centres, updated_radii)
        return updated_centres, updated_radii, calculate_sum_radii(updated_centres, updated_radii)
    else:
        return best_packing
