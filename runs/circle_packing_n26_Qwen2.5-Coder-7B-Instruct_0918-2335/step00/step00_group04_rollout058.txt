import numpy as np
from scipy.spatial import Voronoi
from scipy.optimize import linprog

def solve_voronoi_packing():
    C = [(0.25, 0.25), (0.75, 0.25), (0.75, 0.75), (0.25, 0.75)]
    R = [0.25] * 4
    
    while len(C) < 26:
        obstacles = C + [(0, 0), (1, 0), (1, 1), (0, 1)]
        vertices = Voronoi(obstacles).vertices
        new_circles = []
        
        for v in vertices:
            min_dist = np.inf
            for o in obstacles:
                if o in C:
                    min_dist = min(min_dist, np.linalg.norm(np.array(v) - np.array(o)) - R[C.index(o)])
                else:
                    min_dist = min(min_dist, np.linalg.norm(np.array(v) - np.array(o)))
            
            if min_dist > 0:
                new_circles.append((v, min_dist))
        
        if not new_circles:
            break
        
        _, i = max((r, i) for i, (_, r) in enumerate(new_circles))
        new_c, new_r = new_circles[i]
        C.append(new_c)
        R.append(new_r)
    
    indices = np.argsort(R)[::-1]
    C = [C[i] for i in indices]
    R = np.array([R[i] for i in indices])
    
    n = len(R)
    A_l = np.zeros((n, n * 5))
    A_e = np.zeros((n, n * 5))
    b_l = np.zeros(n)
    b_e = np.zeros(n)
    
    for i in range(n):
        x, y = C[i]
        r = R[i]
        
        A_l[i, i * 5] = 1
        A_l[i, i * 5 + 1] = -1
        A_l[i, i * 5 + 2] = -1
        A_l[i, i * 5 + 3] = -1
        A_l[i, i * 5 + 4] = -1
        b_l[i] = 0
        
        A_e[i, i * 5] = 1
        A_e[i, i * 5 + 1] = -2 * (x - 0)
        A_e[i, i * 5 + 2] = -2 * (y - 0)
        A_e[i, i * 5 + 3] = -2 * (x - 1)
        A_e[i, i * 5 + 4] = -2 * (y - 1)
        b_e[i] = r ** 2
        
        for j in range(i):
            dx, dy = C[j][0] - x, C[j][1] - y
            r_j, r_i = R[i], R[j]
            if dx * dx + dy * dy <= (r_i + r_j) ** 2:
                A_e[i, j * 5] = 1
                A_e[i, j * 5 + 1] = -2 * dx
                A_e[i, j * 5 + 2] = -2 * dy
                A_e[i, j * 5 + 3] = -2 * dx
                A_e[i, j * 5 + 4] = -2 * dy
                b_e[i] += 2 * np.sqrt(dx * dx + dy * dy) + r_i + r_j - 1e-12
    
    result = linprog(np.ones(n), A_ub=A_l, b_ub=b_l, A_eq=A_e, b_eq=b_e, bounds=(0, 1))
    if not result.success:
        raise Exception("Linear Programming failed.")
    
    R = result.x
    centers = np.array(C)
    sum_radii = np.sum(R)
    
    return centers, R, sum_radii

centers, radii, sum_radii = solve_voronoi_packing()
np.round(centers, 6, out=centers)
np.round(radii, 6, out=radii)
