import numpy as np
from scipy.optimize import differential_evolution, minimize

def run_packing() -> tuple np.ndarray, np.ndarray, float]:
    N = 26
    dim = 3 * N
    lambda_penalty = 1e5
    DE_popsize = 12
    DE_maxiter = 400

    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N

    def objective(x):
        xs, ys, rs = np.split(x, [2 * N, dim])
        sum_r = -rs.sum()
        p_boundary = (np.maximum(0, rs - xs)**2 +
                      np.maximum(0, xs + rs - 1)**2 +
                      np.maximum(0, rs - ys)**2 +
                      np.maximum(0, ys + rs - 1)**2).sum()
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist = np.hypot(dx, dy)
        rr = rs[:, None] + rs[None, :]
        p_overlap = np.sum(np.maximum(0, rr - dist[np.triu_indices(N, 1)])**2)
        return -sum_r + lambda_penalty * (p_boundary + p_overlap)

    best_result = None
    for seed in range(4):
        np.random.seed(seed)
        result = differential_evolution(
            objective, bounds, popsize=DE_popsize, maxiter=DE_maxiter,
            polish=True
        )
        if best_result is None or result.fun < best_result.fun:
            best_result = result

    best_x = best_result.x
    centres = np.column_stack([best_x[0:N*2:2], best_x[1:N*2:2]])
    radii = best_x[2*N:]
    sum_radii = radii.sum()
