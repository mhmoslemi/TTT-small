import numpy as np
from scipy.optimize import minimize

def initial_guess(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (26, 2))
    radii = np.zeros(26)
    return np.concatenate((centers.flatten(), radii))

def objective(v):
    return -np.sum(v[52:])

def constraints(v):
    n = 26
    bounds = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n
    centers = v[:52].reshape((n, 2))
    radii = v[52:]
    boundary_constraints = np.concatenate([
        centers[:, 0] - radii >= 0,
        1 - centers[:, 0] - radii >= 0,
        centers[:, 1] - radii >= 0,
        1 - centers[:, 1] - radii >= 0
    ])
    distance_matrix = np.sum(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=-1) ** 2
    overlap_constraints = distance_matrix >= (radii[:, np.newaxis] + radii[np.newaxis, :]) ** 2
    return np.concatenate((boundary_constraints, overlap_constraints.flatten()))

RESTARTS = 10
eps = 1e-7

best_v = None
best_value = -np.inf

for seed in range(RESTARTS):
    v0 = initial_guess(seed)
    res = minimize(
        fun=objective,
        x0=v0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=bounds,
        options={
            'ftol': 1e-9,
            'maxiter': 4000,
            'disp': False
        }
    )
    if res.success:
        value = -res.fun
        if value > best_value:
            best_value = value
            best_v = res.x

if best_v is None:
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
else:
    centers = best_v[:52].reshape((26, 2))
    radii = np.maximum(0.0, best_v[52:] - eps)

sum_radii = np.sum(radii)

return centers, radii, sum_radii
