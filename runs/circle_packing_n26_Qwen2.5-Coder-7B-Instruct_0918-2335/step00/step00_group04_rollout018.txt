import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)  # 2 * x, y coordinates + r for each circle
    x0 = np.random.rand(3 * n)  # Initial guess for centers and radii

    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)  # (x,y) pairs
        r = v[52:]  # radii

        # Boundary terms - 4 * n
        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        # Pairwise terms - n choose 2
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")
