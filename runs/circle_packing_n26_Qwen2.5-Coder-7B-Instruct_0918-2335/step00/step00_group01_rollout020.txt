import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centre_pattern = [
        (0.10, 0.10), (0.10, 0.18), (0.10, 0.26), (0.10, 0.34), (0.10, 0.42), (0.10, 0.50),
        (0.18, 0.10), (0.18, 0.18), (0.18, 0.26), (0.18, 0.34), (0.18, 0.42),
        (0.26, 0.10), (0.26, 0.18), (0.26, 0.26), (0.26, 0.34), (0.26, 0.42),
        (0.34, 0.10), (0.34, 0.18), (0.34, 0.26), (0.34, 0.34), (0.34, 0.42),
        (0.42, 0.10), (0.42, 0.18), (0.42, 0.26), (0.42, 0.34),
        (0.50, 0.10), (0.50, 0.18), (0.50, 0.26), (0.50, 0.34), (0.50, 0.42)
    ]
    centre_pattern = np.array(centre_pattern)

    def compute_forces(duals, positions):
        k = 0
        forces = np.zeros_like(positions)
        n = positions.shape[0]
        for i in range(n):
            di = positions[i]
            active_pairwise_constraints = []
            for j in range(i + 1, n):
                dj = positions[j]
                if duals[k] > 0:
                    active_pairwise_constraints.append(j)
                k += 1
            min_dist_to_boundary = np.min((di[0], 1 - di[0], di[1], 1 - di[1]))
            dual_bounds = np.clip(duals[4*n:n+4*n], 0, None)
            for j in range(i + 1, n):
                dist_to_j = np.linalg.norm(di - dj)
                for k in range(4):
                    dual_b_i = dual_bounds[k*n+4*i+k]
                    forces[i] += dual_b_i * (np.array([1, 0]) if k == 0 else np.array([-1, 0]) if k == 1 else np.array([0, 1]) if k == 2 else np.array([0, -1]))
                forces[i] += duals[k] * (di - dj) / dist_to_j
        return forces

    def find_solution(centers):
        A_ub = np.zeros((4*n + n*(n-1)//2, n))
        b_ub = np.zeros(4*n + n*(n-1)//2)
        for i in range(n):
            A_ub[4*i    :4*i+4, i] = np.array([1.0, 1.0, 1.0, 1.0])
            b_ub[4*i    :4*i+4]    = np.array([centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]])
        k = 4*n
        for i in range(n):
            for j in range(i + 1, n):
                distance = np.linalg.norm(centers[i] - centers[j])
                A_ub[k :k+2, i] = np.array([1.0, 1.0])
                A_ub[k :k+2, j] = np.array([1.0, 1.0])
                b_ub[k    :k+2] = np.array([distance, distance])
                k += 2
        c = -np.ones(n)
        res = linprog(c=c, A_ub=A_ub, b_ub=b_ub, bounds=(np.zeros(n), None), method='highs')
        if not res.success:
            return np.zeros(n), -np.inf
        return np.maximum(0.0, res.x - 1e-7), res.fun

    step_size = 0.05
    max_iter = 200
    max_restarts = 5
    best_obj = -np.inf
    best_centres, best_radii = None, None

    for _ in range(max_restarts):
        centres = centre_pattern.copy() + 0.1 * np.random.randn(n, 2)
        for _ in range(max_iter):
            radii, obj = find_solution(centres)
            if obj > best_obj:
                best_obj = obj
                best_centres, best_radii = centres.copy(), radii.copy()

            centres += step_size * forces
            if np.abs(obj - best_obj) < 1e-6:
                break
        centres += 0.1 * np.random.randn(n, 2)

    return best_centres, best_radii, np.sum(best_radii)
