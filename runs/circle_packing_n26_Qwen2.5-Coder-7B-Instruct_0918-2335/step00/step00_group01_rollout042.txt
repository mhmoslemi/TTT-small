import numpy as np
from scipy.optimize import linprog, Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r_min = 1e-4
    g = 1e-3
    vel_magnitude = 5e-3
    T_max = 50.0
    epsilon = 1e-6

    # Initialize random positions and velocities
    centers = np.random.rand(n, 2) + np.random.uniform(-1e-3, 1e-3, (n, 2))
    radii = np.full(n, r_min)
    velocities = (np.random.rand(n, 2) - 0.5) * vel_magnitude

    # Function to compute the earliest collision time
    def compute_collision_time(i, j):
        c1, c2 = centers[i], centers[j]
        v1, v2 = velocities[i], velocities[j]
        dist_sq = np.sum((c1 - c2) ** 2)
        delta_r = r_min + g
        if dist_sq <= delta_r ** 2:
            return 0.0
        a = np.sum((v1 - v2) ** 2)
        b = 2 * np.sum((c1 - c2) * (v1 - v2))
        c = dist_sq - delta_r ** 2
        discriminant = b ** 2 - 4 * a * c
        if discriminant < 0:
            return np.inf
        t = (-b + np.sqrt(discriminant)) / (2 * a)
        return t

    # Function to handle boundary collisions
    def handle_boundary_collision(i):
        cx, cy = centers[i]
        vx, vy = velocities[i]
        t_horizontal = (1 - cx if cx >= 0.5 else cx) / max(vx, -vx)
        t_vertical = (1 - cy if cy >= 0.5 else cy) / max(vy, -vy)
        t = min(t_horizontal, t_vertical)
        if t > 0:
            centers[i] += velocities[i] * t
            velocities[i] = -velocities[i]

    # Event queue
    event_queue = []

    for i in range(n):
        for j in range(i + 1, n):
            t = compute_collision_time(i, j)
            if t < np.inf:
                event_queue.append((t, i, j))
        for axis in [0, 1]:

                if t < np.inf:

        event_queue.sort()

            cx, cy = centers[i]
            vx, vy = velocities[i]
            t_horizontal = (1 - cx if cx >= 0.5 else cx) / max(vx, -vx)
            t_vertical = (1 - cy if cy >= 0.5 else cy) / max(vy, -vy)
            t = min(t_horizontal, t_vertical)
            centers[i] += velocities[i] * t
            velocities[i] = -velocities[i]
        for k in [i, j] if j is not None else [i]:

                        if t < np.inf:
                            event_queue.append((t, k, None))
        event_queue.sort()

    # Repair small overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < (r_min + r_min) * 0.99:
                overlap = (r_min + r_min) - dist
                centers[i] += (centers[i] - centers[j]) / (2 * dist) * overlap
                centers[j] += (centers[j] - centers[i]) / (2 * dist) * overlap

    # Shrink radii and repair overlaps again if necessary
    radii -= epsilon
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < (radii[i] + radii[j] + 1e-6) * 0.99:

                radii -= overlap / 2

    A[2*n:] = np.eye(n)
