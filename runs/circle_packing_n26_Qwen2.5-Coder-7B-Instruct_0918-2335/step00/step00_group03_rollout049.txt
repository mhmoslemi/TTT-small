import numpy as np
from scipy import optimize

def random_sequential_addition(N, r_min, iterations):
    centers = []
    radii = []
    for _ in range(N):
        attempts = 0
        while attempts < iterations:
            center = np.random.uniform(0, 1, 2)
            radius = r_min
            valid = True
            for c, r in zip(centers, radii):
                if np.linalg.norm(center - c) < r + radius:
                    valid = False
                    break
            if valid:
                centers.append(center)
                radii.append(radius)
                break
            attempts += 1
        if attempts == iterations:
            return None, None  # Retry with fresh seed
    return np.array(centers), np.array(radii)

def physics_based_relaxation(centers, radii, Δt, T):
    for _ in range(T):
        n = len(centers)
        forces = np.zeros((n, n, 2))
        for i in range(n):
            forces[i, i] = np.array([0, 0])
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx**2 + dy**2)
                gap = d - (radii[i] + radii[j])
                if gap > 0:
                    f = gap * Δt
                else:
                    f = -gap * gap * Δt
                forces[i, j] = np.array([f * dx / d, f * dy / d])
                forces[j, i] = -forces[i, j]
        for i in range(n):
            fx = np.sum(forces[i, :][:, 0])
            fy = np.sum(forces[i, :][:, 1])
            dr = fx * Δt
            radii[i] += dr
            centers[i] += np.array([fx * Δt / radii[i], fy * Δt / radii[i]])
            radii[i] = max(0, min(radii[i], min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]))))
    return centers, radii

def global_uniform_expansion(centers, radii):
    s_max = min(np.min(np.apply_along_axis(lambda r_i, x_i, y_i: (1 - r_i) / r_i, 1, np.array([x for x, y in centers]), np.array([x for x, y in centers]), np.array([y for x, y in centers]))),
                np.min(np.apply_along_axis(lambda r_i, x_i, y_i: (np.linalg.norm([(x_i, y_i) - (x_j, y_j)], 2) - (r_i + r_j)) for x_j, y_j in centers for x_i, y_i in centers)[::3]))
    centers = np.array([0.5 + s_max * (c - 0.5) for c in centers])
    radii *= s_max
    return centers, radii

def objective_function(x, N):
    radii = x[N:]
    return -np.sum(radii)

def constraints(radii, centers):
    constraints = [{'type': 'ineq', 'fun': lambda x, radii=radii, centers=centers: (radii + radii[:, np.newaxis] - np.linalg.norm(centers[:, None] - centers, axis=2))[:, np.triu_indices_from(np.array([0] * N), k=1)][:, 0] + 1e-12},
                   {'type': 'ineq', 'fun': lambda x, radii=radii, centers=centers: np.maximum(0, radii - np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])) + 1e-12}]
    return constraints

def run_packing():
    N = 26
    r_min = 1e-3
    iterations = 10000
    Δt = 1e-3
    T = 2000
    centers, radii = random_sequential_addition(N, r_min, iterations)
    while radii is None:
        centers, radii = random_sequential_addition(N, r_min, iterations)
    centers, radii = physics_based_relaxation(centers, radii, Δt, T)
    centers, radii = global_uniform_expansion(centers, radii)
    N = centers.shape[0]
    x0 = np.concatenate([centers.flatten(), radii])
    res = optimize.minimize(objective_function, x0, args=(N,), method='L-BFGS-B', bounds=[(0, 1)] * (2 * N) + [(0, 0.5)] * N, constraints=constraints(radii, centers))
    return res.x[:2*N].reshape(-1, 2), res.x[2*N:], -res.fun
