import numpy as np
from scipy.optimize import linprog, minimize
import random

def generate_positions(base_layout):
    if base_layout == 'grid':
        centers = np.array([(x/4 + 0.125, y/4 + 0.125) for y in range(5) for x in range(4)])
        centers[19] = (0.75, 0.75)
    elif base_layout == 'hexagon':
        centers = np.array([(0.25, y * np.sqrt(3)/3 + (x % 2) * np.sqrt(3)/6) for y in range(6) for x in range(y + 1)][:-2])
        centers = centers[centers[:,0] <= 0.75]  # Adjust for compact fit
        centers[centers[:,1] <= 1]  # Adjust for compact fit
    elif base_layout == 'random':
        centers = np.random.rand(26, 2)
    else:
        raise ValueError("Invalid base layout")

    return (centers - 0.5) * 2

def perturb(centers, sigma=0.1):
    return np.clip(centers + np.random.normal(0, sigma, centers.shape), 0, 1)

def lp_optimize_radii(centers):
    n = centers.shape[0]
    dis_matrix = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=-1)
    dr_matrix = dis_matrix - 0.02  # Small buffer to avoid grazing contact for LP violation detection

    c = np.ones(n)
    A = np.sum(dr_matrix * (centers[:, None, :] - centers[None, :, :]) / dis_matrix[:, :, np.newaxis], axis=2).flatten()
    b = np.sum(dr_matrix, axis=-1).flatten()
    A = np.vstack([A, centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]])
    b = np.zeros(n + 4)

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.05, 0.15)]*n, method='highs', options={'disp': False})
    if not res.success:
        return None
    return res.x

def evaluate(solution):
    centers, radii = solution[:52].reshape(26, 2), solution[52:]
    dis = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=-1)
    violations = ((dis - radii[:, None] - radii[None, :]) < 0) & ((dis - radii[:, None] - radii[None, :]) > -1e-6)
    overlap_penalties = np.sum(violations, axis=1).flatten().max() * 1e6
    boundary_penalties = np.sum(np.clip(radii - np.minimum(centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]), 0, None)) * 1e5
    return -np.sum(radii) + overlap_penalties + boundary_penalties

def refine_with_sa(centers0, radii0):
    n = centers0.shape[0]
    centers = centers0.copy()
    radii = radii0.copy()
    energy = evaluate((centers.flatten(), radii.flatten()))
    T = 1e-2
    cool_rate = 0.995
    for _ in range(10000):
        idx = random.randint(0, n-1)
        dr = random.uniform(1e-6, 0.05, size=2) if random.random() < 0.5 else random.uniform(1e-6, 0.05)
        dp = random.uniform(-0.05, 0.05, 2) / 2
        new_r = np.clip(radii[idx] + dr, 1e-6, 0.5)
        new_c = np.clip(centers[idx] + dp, 0, 1)
        new_energy = evaluate((np.concatenate((centers.flatten(), radii.flatten())).copy(), np.concatenate((new_c.flatten(), new_r.flatten())).copy()))
        if random.random() < np.exp((energy - new_energy) / T):
            radii[idx] = new_r
            centers[idx] = new_c
            energy = new_energy
        T *= cool_rate

    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)
    x0 = np.random.rand(3 * n)

    def objective(x):
        return -np.sum(x[52:])

    def constraints(x):
        centers = x[:52].reshape(n, 2)
        radii = x[52:]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                cons.append(radii[i] + radii[j] - np.linalg.norm(centers[i] - centers[j]) + 1e-6)
            cons.extend([radii[i] - min(centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1]), 0])
        return cons

    candidates = []
    for base_layout in ['grid', 'hybrid', 'random']:
        centers0 = generate_positions(base_layout)
        for _ in range(5):
            centers = perturb(centers0, sigma=0.02)
            radii = lp_optimize_radii(centers)
            if radii is None:
                continue

            sum_r = np.sum(radii)
            candidates.append((centers, radii, sum_r))
            refined = refine_with_sa(centers, radii)
            candidates.append(refined)

    best = max(candidates, key=lambda c: c[2])
    centers, radii = best[0], best[1]
    return centers, radii, best[2]
