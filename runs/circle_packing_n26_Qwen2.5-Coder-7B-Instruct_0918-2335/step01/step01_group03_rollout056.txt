import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def simulate_growing_discs(centers, speeds, radius_limit):
    n = len(centers)
    active = set(range(n))
    t_stop = np.full(n, np.nan)
    
    while len(active) > 0:
        min_time = np.inf
        stop_indices = []
        
        for i in range(len(centers)):
            if i not in active:
                continue
            for j in range(i + 1, len(centers)):
                if j not in active:
                    continue
                if np.any(t_stop[active] == 0):  # Avoid division by zero if a disc was stopped immediately
                    continue
                dist_ij = distance.euclidean(centers[i], centers[j])
                t_ij = dist_ij / (speeds[i] + speeds[j])
                if t_ij < min_time:
                    min_time = t_ij
                    stop_indices = []
                if t_ij <= min_time + 1e-12:
                    stop_indices.append((i, j))
        
        for i, j in stop_indices:
            if np.isfinite(t_stop[i]) and np.isfinite(t_stop[j]):
                continue
            if min_time >= np.min(t_stop[active]):
                t_min = min_time
                active.remove(i)
                active.remove(j)
                t_stop[i] = t_min
                t_stop[j] = t_min
                radius = speeds[i] * t_stop[i]
                if radius > radius_limit:
                    radius = radius_limit
                if radius <= 1e-10:
                    radius = 0.0
                return True, (centers[i], radius, centers[j], radius)
    
    final_radii = np.array([speeds[i] * t_stop[i] for i in range(n)])
    return False, (centers, final_radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    layout_solutions = []
    base_speed = np.ones(n)
    
    # Regular grid arrangement
    grid_centers = np.array([(i/4, j/4) for j in range(5) for i in range(5)])
    grid_centers.add(grid_centers[-1])  # Duplicate the last center
    layout_solutions.append((grid_centers, base_speed * 2))
    
    # Hexagonal arrangement
    hex_centers = np.array([
        (0.25, 0.375), (0.5, 0.25), (0.75, 0.375), (1.0, 0.5),
        (0.75, 0.625), (1.0, 0.5), (1.25, 0.625), (1.5, 0.5),
        (1.25, 0.375), (1.5, 0.25), (1.25, 0.125), (1.0, 0.5),
        (0.75, 0.125), (0.625, 0.25), (0.5, 0.125), (0.375, 0.25),
        (0.125, 0.125), (0.25, 0.25), (0.125, 0.375), (0.0, 0.5),
        (0.125, 0.625), (0.25, 0.625), (0.375, 0.625), (0.5, 0.5),
        (0.625, 0.625), (0.75, 0.5), (0.625, 0.375)
    ])
    layout_solutions.append((hex_centers, base_speed * 2.1))
    
    # Hybrid arrangement
    hybrid_centers = np.vstack([
        np.array([[0.05, 0.05], [0.95, 0.05], [0.05, 0.95], [0.95, 0.95]]),
        np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]),
        np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]]),
        np.array([[0.35, 0.35], [0.65, 0.35], [0.35, 0.65], [0.65, 0.65]]),
        np.array([[0.45, 0.45], [0.55, 0.45], [0.45, 0.55], [0.55, 0.55]]),
        np.array([[0.05, 0.15], [0.95, 0.15], [0.05, 0.85], [0.95, 0.85]]),
        np.array([[0.15, 0.05], [0.85, 0.05], [0.15, 0.95], [0.85, 0.95]]),
        np.array([[0.25, 0.05], [0.75, 0.05], [0.25, 0.95], [0.75, 0.95]]),
        np.array([[0.35, 0.05], [0.65, 0.05], [0.35, 0.95], [0.65, 0.95]])
    ])
    layout_solutions.append((hybrid_centers, base_speed * 1.8))
    
    # Random arrangement
    random_centers = np.random.rand(n, 2)
    layout_solutions.append((random_centers, base_speed * 1.7))
    
    best_centers = None
    best_radii = None
    best_sum_radii = -np.inf
    
    for centers, initial_speeds in layout_solutions:
        radius_limit = 0.5
        for t in range(2000):  # Run 2000 trials to increase chances of finding a good packing
            speeds = np.random.lognormal(mean=0., sigma=0.4, size=n)
            speeds *= initial_speeds
            success, result = simulate_growing_discs(centers, speeds, radius_limit)
            if success:
                if result[1] + result[3] > best_sum_radii:
                    best_sum_radii = result[1] + result[3]
                    best_centers = np.vstack([result[0], result[2]])
                    best_radii = np.array([result[1], result[3]])
    
    if best_centers is None or best_radii is None:
        return best_centers, best_radii, -1.0
    
    # Run local refinement using SLSQP
    def objective(x):
        return -np.sum(x[n:2*n])
    
    def constraint(v):
        pos = v[:n]
        r = v[n:]
        constraints = np.full(4 * n, float('inf'))
        for i in range(n):
            x, y = pos[i]
            if not (0 <= x <= 1 and 0 <= y <= 1):
                return False
            for j in range(n):
                if i != j:
                    dx, dy = pos[j] - x, pos[j] - y
                    dist2 = dx**2 + dy**2
                    if dist2 < (r[i] + r[j]) ** 2:
                        return False
        return True
    
    bounds = [(0, 1)] * (2 * n)
    sol = minimize(fun=objective, x0=np.hstack([best_centers.ravel(), best_radii]), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint}, options={'maxiter': 5000, 'ftol': 1e-9})
    
    if sol.success and constraint(sol.x):
        best_centers = sol.x[:n].reshape(-1, 2)
        best_radii = sol.x[n:]
        best_sum_radii = -sol.fun
    
    return best_centers, best_radii, best_sum_radii
