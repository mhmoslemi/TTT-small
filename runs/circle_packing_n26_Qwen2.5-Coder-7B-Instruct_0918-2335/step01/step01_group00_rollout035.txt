import numpy as np
from scipy.optimize import linprog, minimize

def lp_value_and_radii(centres):
    w_limits = np.min(np.column_stack([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]]), axis=1)
    num_circles = len(w_limits)
    D_ij = np.zeros((num_circles, num_circles))
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            D_ij[i, j] = np.linalg.norm(centres[i] - centres[j])
    A_ub = np.zeros((num_circles * (num_circles - 1) // 2 + num_circles, num_circles))
    b_ub = np.zeros(num_circles * (num_circles - 1) // 2 + num_circles)
    idx = 0
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            A_ub[idx, i], A_ub[idx, j] = 1, 1
            b_ub[idx] = D_ij[i, j]
            idx += 1
        A_ub[idx, i] = -1
        b_ub[idx] = w_limits[i]
        idx += 1
    c = -np.ones(num_circles)
    bounds = np.array([(0, w) for w in w_limits])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.fun, result.x
    return -np.inf, np.zeros(num_circles)

def run_packing():
    def objective(centers):
        sum_radii = 0
        for i in range(26):
            r = compute_radii(centers[i])
            sum_radii += r
        return sum_radii

    def compute_radii(center):
        distance_matrix = np.zeros((26, 26))
        for i in range(26):
            for j in range(i + 1, 26):
                distance_matrix[i, j] = np.linalg.norm(center - centers[j])
                distance_matrix[j, i] = distance_matrix[i, j]
            distance_matrix[i, i] = np.inf  # No self-distance

        radii = np.zeros(26)
        for i in range(26):
            max_radius = np.inf
            for j in range(26):
                if i != j:
                    max_radius = min(max_radius, distance_matrix[i, j] / 2 - distances_to_walls(i))
                
            if max_radius >= 0:
                radii[i] = max_radius

        return radii

    def distances_to_walls(index):
        x, y = centers[index]
        distances = np.array([x, 1 - x, y, 1 - y])
        return np.min(distances)
    
    centers = np.random.rand(26, 2)
    
    for restart in range(20):
        centers = np.random.rand(26, 2)
        for _ in range(500):
            new_centers = centers.copy()
            index = np.random.randint(26)
            perturbation = np.random.randn(2) * np.random.rand() * 0.02
            new_centers[index] = np.clip(new_centers[index] + perturbation, 0, 1)
            if objective(new_centers) > objective(centers):
                centers = new_centers
    
    centers = np.random.rand(26, 2)
    result = minimize(objective, centers.flatten(), method='BFGS', bounds=[(0, 1)] * 52)
    centers = result.x.reshape(26, 2)
    radii = np.array([compute_radii(center) for center in centers])
    
    return centers, radii, np.sum(radii)
