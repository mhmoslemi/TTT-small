import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def solve_radii_lp(centers):
        c = np.ones(n)
        A_eq = np.zeros((n, n))
        b_eq = np.zeros(n)
        A_ineq = np.zeros((2 * n, n))
        b_ineq = np.zeros(2 * n)
        
        for i in range(n):
            A_eq[i, i] = 1
            b_eq[i] = 0
            
            A_ineq[i * 4, i] = 1
            b_ineq[i * 4] = centers[i, 0]
            
            A_ineq[i * 4 + 1, i] = 1
            b_ineq[i * 4 + 1] = 1 - centers[i, 0]
            
            A_ineq[i * 4 + 2, i] = 1
            b_ineq[i * 4 + 2] = centers[i, 1]
            
            A_ineq[i * 4 + 3, i] = 1
            b_ineq[i * 4 + 3] = 1 - centers[i, 1]
        
        A_ineq_full = np.zeros((n * (n - 1) // 2, n))
        b_ineq_full = np.zeros(n * (n - 1) // 2)
        
        k = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ineq_full[k, i] = 1
                A_ineq_full[k, j] = 1
                b_ineq_full[k] = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                k += 1
        
        A_ineq = np.vstack((A_ineq, A_ineq_full))
        b_ineq = np.hstack((b_ineq, b_ineq_full))
        
        lp_result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ineq, b_ub=b_ineq, bounds=[(0, max_radius)] * n, method='highs')
        if not lp_result.success:
            return np.zeros(n), 0.0
        
        return lp_result.x, lp_result.success
    
    def compute_active_constraints(centers, radii):
        active_constraints = np.zeros(n, dtype=int)
        for i in range(n):
            dists = np.sqrt((centers - centers[i]) ** 2).sum(axis=1)
            valid_dists = np.where((dists > 1e-9) * (dists <= 2 * max_radius))[0]
            valid_dists = valid_dists[valid_dists != i]  # exclude self
            
            if np.min(centers[i] - centers[valid_dists, :], axis=0).sum() == 0:
                # touching left wall
                active_constraints[i] = 1
            elif np.min(1 - centers[i] - centers[valid_dists, :], axis=0).sum() == 0:
                # touching right wall
                active_constraints[i] = 2
            elif np.min(centers[i] - centers[valid_dists, :], axis=1).sum() == 0:
                # touching bottom wall
                active_constraints[i] = 3
            elif np.min(1 - centers[i] - centers[valid_dists, :], axis=1).sum() == 0:
                # touching top wall
                active_constraints[i] = 4
            else:
                # touching a neighbor
                min_dist_indices = np.argsort(dists[valid_dists])[:2]
                j, k = valid_dists[min_dist_indices]
                active_constraints[i] = 5 + j
        return active_constraints
    
    def move_centers(centers, radii, active_constraints, alpha=0.01):
        d = 0.0
        sensitive = True
        centers_new = np.copy(centers)
        for i in range(n):
            if active_constraints[i] == 0: continue
            if active_constraints[i] < 5:
                # wall constraint
                if active_constraints[i] == 1: dx, dy = alpha, 0
                elif active_constraints[i] == 2: dx, dy = -alpha, 0
                elif active_constraints[i] == 3: dx, dy = 0, alpha
                elif active_constraints[i] == 4: dx, dy = 0, -alpha
                centers_new[i] += np.array([dx, dy])
                d = np.maximum(d, max(np.abs(dy), np.abs(dx)))
            else:
                # neighbor constraint
                j = active_constraints[i] - 5
                dist = np.sqrt((centers[i] - centers[j]) ** 2).sum()
                u = (centers[i] - centers[j]) / dist
                dx, dy = alpha * u
                centers_new[i] += np.array([dx, dy])
                d = np.maximum(d, dist + alpha)
        
        # bidirectional refinement for touching pairs
        for i in range(n):
            if active_constraints[i] > 4:
                j = active_constraints[i] - 5
                dist = np.sqrt((centers[i] - centers[j]) ** 2).sum()
                if dist + alpha > 2 * radii[i] + radii[j]:
                    continue
                u = (centers[i] - centers[j]) / dist
                centers_new[j] += -np.array([alpha * u[0], alpha * u[1]])
        return centers_new, d, sensitive
    
    def initial_centers():
        # Hexagonal grid-based initial centers
        side_length = np.sqrt(3) * 0.1
        num_rows = int(np.ceil(np.sqrt(n * 3 / 4 / np.pi)))
        num_cols = int(np.ceil(n / (2 * num_rows)))
        centers = np.zeros((n, 2))
        k = 0
        row = 0
        col = 0
        while k < n:
            centers[k, 0] = (col + 0.5 * (row % 2)) * side_length * np.sqrt(3)
            centers[k, 1] = row * side_length * 0.5
            k += 1
            col += 1
            if col >= num_cols:
                col = 0
                row += 1
        return centers
    
    def random_centers():
        return np.random.rand(n, 2)
    
    def validate_cut_off():
        return False  # This function is not provided in the original task
    
    def simulate_anneal(centers, rad, alpha, k):
        if k % 50 == 0:
            centers += 0.1 * np.random.randn(n, 2)
        return centers
    
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0.0
    
    for seed_method in [initial_centers, random_centers]:
        for i in range(8):
            if validate_cut_off():
                continue
            
            centers = seed_method()
            alpha = 0.01
            best_rad = 0.0
            num_iterations = 0
            
            for iteration in range(200):
                ad, rad = solve_radii_lp(centers)
                
                if rad:
                    if rad > best_rad:
                        best_rad = rad
                        for i in range(n):
                            best_centers[i] = centers[i]
                    active_constraints = compute_active_constraints(centers, rad)
                    centers, d, sensitive = move_centers(centers, rad, active_constraints, alpha)
                    
                    if not sensitive:
                        alpha /= 2
                    
                    if d < 1e-9:
                        break
                else:
                    break
                
                centers = simulate_anneal(centers, rad, alpha, iteration)
                
                num_iterations += 1
            
            if sum(rad) > best_sum_radii:
                best_sum_radii = sum(rad)
                best_centers = centers
                best_radii = rad
    
    res = minimize(
        fun=lambda x: -np.sum(x[52:]),
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=[(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n,
        constraints [{'type': 'ineq', 'fun': lambda x: x[52:]}],
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    final_centers = res.x[:52].reshape(n, 2)
    final_radii = res.x[52:]
    return final_centers, final_radii, np.sum(final_radii)
