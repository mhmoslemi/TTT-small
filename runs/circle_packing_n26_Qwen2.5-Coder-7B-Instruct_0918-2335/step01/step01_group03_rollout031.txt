import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, minimize
from scipy.stats import norm

def generate_hexagon(n):
    rows = np.arange(n // 2 + 1)
    cols = np.concatenate([np.arange(n - r) for r in rows])
    L = np.sqrt(3)
    width = cols[-1] * 2 + (n - n // 2.) * L
    g = 1 / width
    centers = g * np.array([2. * c + (r & 1), r * L] for r, c in zip(rows, cols))
    centers -= centers.mean(axis=0)
    centers -= centers.max(axis=0) / 2.
    centers[centers < 0] += 1.
    centers[centers > 1] -= 1.
    return centers

def generate_grid(nx, ny):
    centers = np.array([(i / (nx - 1), j / (ny - 1)) for i in range(nx) for j in range(ny)])
    centers -= centers.mean(axis=0)
    centers -= centers.max(axis=0) / 2.
    centers[centers < 0] += 1.
    centers[centers > 1] -= 1.
    return centers

def generate_hybrid():
    edge_centers = np.array([[0.5, 0.15], [0.15, 0.5], [0.5, 0.85], [0.85, 0.5]])
    nx, ny = 5, 3
    inner_grid = np.array([[i / (nx - 1), j / (ny - 1)] for i in range(nx) for j in range(ny)])
    inner_grid -= np.array([0.15, 0.15])
    inner_grid -= inner_grid.max(axis=0) / 2.
    inner_grid[inner_grid < 0] += 1.
    inner_grid[inner_grid > 1] -= 1.
    return np.concatenate([edge_centers, inner_grid])

def generate_random(spacing, n):
    centers = np.random.uniform(size=(n, 2))
    return centers

def lp_optimize_radii(centers):
    n = centers.shape[0]
    radii = np.ones(n) * 0.05
    bounds = [(0, None) for _ in range(n)]
    A = []
    b = []
    G = []
    h = []
    for i in range(n):
        A.append([1 if i == j else 0 for j in range(n)])
        b.append(radii[i] / (centers[i, 0] + 0.5))
        b.append(radii[i] / (centers[i, 1] + 0.5))
        b.append(radii[i] / (1 - centers[i, 0] + 0.5))
        b.append(radii[i] / (1 - centers[i, 1] + 0.5))
        G.append([1 if i == j else 0 for j in range(n)] + [0, 0, 0, 0])
        h.append(radii[i])
    res = linprog(radii, A_ub=np.array(G), b_ub=np.array(h), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def evaluate(x):
    centers = x[:26].reshape(26, 2)
    radii = x[26:]
    if np.any(radii < 0) or not np.all((radii <= centers[:, 0]) * (radii <= (1 - centers[:, 0]))
                                      * (radii <= centers[:, 1]) * (radii <= (1 - centers[:, 1]))):
        return -np.inf
    overlap = np.sum([np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) for i in range(26) for j in range(i+1, 26)])
    for i in range(26):
        if radii[i] >= 0.05:
            overlap -= radii[i] ** 2
    return -np.sum(radii) + 1e8 * overlap

def run_packing():
    np.random.seed(0)
    radii_min, radii_max = 0.05, 0.5
    nx, ny = 5, 3
    centers0s = [
        generate_hexagon(5),
        generate_grid(nx, ny),
        generate_hybrid(),
        generate_random(0.01, 26)
    ]
    best_solution = None
    best_sum = -np.inf
    attempts_per_layout = 3
    temp_init = 1e-1
    cool_rate = 0.995
    sa_iterations = 10000
    for centers0 in centers0s:
        for attempt in range(attempts_per_layout):
            centers = centers0.copy()
            centers += np.random.normal(size=centers.shape) * 0.02
            centers = np.clip(centers, 0, 1)
            radii = lp_optimize_radii(centers)
            if radii is not None:
                x0 = np.concatenate([centers.flatten(), radii])
                x = minimize(evaluate, x0, bounds=[(0, 1) for _ in range(26)] + [(radii_min, radii_max) for _ in range(26)], method='SLSQP', options={'ftol': 1e-8, 'maxiter': 2000})
                if evaluate(x.x) > best_sum:
                    best_sum = evaluate(x.x)
                    best_solution = x
    if best_solution is None:
        raise RuntimeError("No valid solution found"
    return best_solution.x[:26].reshape(26, 2), best_solution.x[26:], -best_solution.fun
