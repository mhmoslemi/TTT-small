import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(pos):
    x, y = pos[:26], pos[26:]
    A_ub = np.zeros((62, 26))
    b_ub = np.zeros(62)
    
    for i in range(26):
        A_ub[i, i] = -1
        b_ub[i] = x[i]
        A_ub[i + 26, i] = -1
        b_ub[i + 26] = 1 - x[i]
        A_ub[i, 26 + i] = -1
        b_ub[i] = y[i]
        A_ub[i + 26, 26 + i] = -1
        b_ub[i + 26] = 1 - y[i]
    
    rows = 52
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub[rows, i] = 1
            A_ub[rows, j] = 1
            b_ub[rows] = np.sqrt((x[i] - x[j]) ** 2 + (y[i] - y[j]) ** 2) + 1e-9
            rows += 1
    
    res = linprog([-np.ones(26)], A_ub=A_ub, b_ub=b_ub, bounds=(-np.inf, np.inf))
    if res.status != 0:
        return None
    return res.x
