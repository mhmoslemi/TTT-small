import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import linprog
from numpy.linalg import norm

def initial_setup(n):
    return np.random.rand(n, 2)

def lp_solve_given_centers(centers):
    n, _ = centers.shape
    radians = np.pi / 2
    M = np.zeros((4 * n, n), dtype=float)
    h = np.zeros(4 * n, dtype=float)
    M[:n, np.arange(n)] = 1 - centers[:, 1]
    M[n:2*n, np.arange(n)] = centers[:, 1]
    M[2*n:3*n, np.arange(n)] = 1 - centers[:, 0]
    M[3*n:4*n, np.arange(n)] = centers[:, 0]
    dists = cdist(centers, centers)
    mask = np.triu(np.ones((n, n), bool), k=1)
    A_rows = centers[mask] - centers[~mask]
    A_cols = dists[mask]
    M[4*n:, np.arange(n)] = np.where(mask, A_rows[:, 0], 0)
    M[4*n:, np.arange(n, 2*n)] = np.where(mask, A_rows[:, 1], 0)
    M[4*n:, np.arange(2*n)] = np.where(mask, A_cols, 0)
    h[4*n:] = np.where(mask, 1 - A_cols, 0)
    result = linprog(np.ones(n), A_ub=M, b_ub=h, bounds=(0, None), method='highs')
    if not result.success:
        raise ValueError("LP solve failed. Method was: highs")
    radii = result.x
    return radii

def push_away(centers, radii):
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        active_constraint = determine_active_constraint(centers, radii, i)
        if active_constraint == "left":
            forces[i, 0] = 1
        elif active_constraint == "right":
            forces[i, 0] = -1
        elif active_constraint == "bottom":
            forces[i, 1] = 1
        elif active_constraint == "top":
            forces[i, 1] = -1
        elif active_constraint == "neighbor":
            center_i, center_j = closest_neighbors(centers, radii, i)
            diff = center_j - center_i
            forces[i, :] += diff / norm(diff)
    updated_centers = centers + forces
    updated_centers = np.clip(updated_centers, 0, 1)
    return updated_centers

def determine_active_constraint(centers, radii, i):
    n, _ = centers.shape
    distances = cdist(centers[[i]], centers[:n], 'euclidean')
    wall_constraint = np.min([1 - centers[i, 0] - radii[i], centers[i, 0] + radii[i], 1 - centers[i, 1] - radii[i], centers[i, 1] + radii[i]])
    neighbor_constraints = np.where(distances[:, 0] == np.min(distances[:, 0]), distances[:, 1], np.inf)
    neighbor_constraint = np.min(neighbor_constraints[neighbor_constraints != np.inf])
    if wall_constraint <= neighbor_constraint:
        return "left" if centers[i, 0] <= radii[i] else "right" if centers[i, 0] >= 1 - radii[i] else "bottom" if centers[i, 1] <= radii[i] else "top"
    else:
        return "neighbor"

def closest_neighbors(centers, radii, i):
    n, _ = centers.shape
    distances = cdist(centers[[i]], centers[:n], 'euclidean')
    neighbor = np.where(distances == np.min(distances))[1][0]
    return centers[i], centers[neighbor]

def run_packing():
    n = 26
    max_iterations = 200
    cooling_interval = 20
    escape_interval = 50
    epsilon = 1e-7
    pause_epochs = 5
    alpha = 0.01
    
    seeds = [initial_setup(n), initial_setup(n), initial_setup(n)]
    best_centers = None
    best_radii = None
    best_sum = 0
    
    for seed in seeds:
        centers, best_sum_seed = seed, 0
        for _ in range(max_iterations):
            try:
                radii = lp_solve_given_centers(centers)
                if np.sum(radii) > best_sum_seed + epsilon:
                    centers = push_away(centers, radii)
                    new_radii = lp_solve_given_centers(centers)
                    if np.sum(new_radii) > best_sum_seed + epsilon:
                        best_sum_seed = np.sum(new_radii)
                if _ % cooling_interval == 0:
                    alpha *= 0.5
                if _ % escape_interval == 0:
                    centers += np.random.normal(0, alpha / 2, centers.shape)
                    centers = np.clip(centers, 0, 1)
                if _ >= pause_epochs and np.sum(radii) == best_sum_seed:
                    break
            except Exception as e:
                print(f"Error in packing: {e}")
                return np.zeros((n, 2)), np.zeros(n), 0.0
            
            if np.sum(radii) > best_sum_seed:
                best_sum_seed = np.sum(radii)
                centers, best_sum_seed = push_away(centers, radii), new_radii
                
            if best_sum_seed > best_sum:
                best_sum, best_centers, best_radii = best_sum_seed, centers, new_radii
                
    if best_centers is not None and best_radii is not None:
        final_centers, final_radii = best_centers, best_radii
        finalradii = final_radii.tolist()
