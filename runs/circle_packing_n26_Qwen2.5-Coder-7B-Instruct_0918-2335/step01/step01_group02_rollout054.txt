import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog, minimize

def lp_for_radii(centers):
    n = centers.shape[0]
    r = np.array([1.0] * n)
    constraint_matrix = np.empty((3 * n, n))
    constraint_matrix[:n, :] = centers[:, 0][:, np.newaxis] - r
    constraint_matrix[n:2 * n, :] = centers[:, 1][:, np.newaxis] - r
    constraint_matrix[2 * n:, :] = 1 - centers[:, 0][:, np.newaxis] - r
    constraint_matrix[3 * n:, :] = 1 - centers[:, 1][:, np.newaxis] - r
    distances = squareform(pdist(centers))
    non_overlap_constraints = np.empty((n * (n - 1) * 2, n))
    for i in range(n):
        for j in range(i + 1, n):
            non_overlap_constraints[i * (n - 1) * 2 + j - (i + 1) * i // 2, :] = 1
            non_overlap_constraints[i * (n - 1) * 2 + j - (i + 1) * i // 2, i] = -(distances[i, j] + 1e-7)
            non_overlap_constraints[i * (n - 1) * 2 + j - (i + 1) * i // 2 + n * (n - 1) * 2, :] = 1
            non_overlap_constraints[i * (n - 1) * 2 + j - (i + 1) * i // 2 + n * (n - 1) * 2, j] = -(distances[i, j] + 1e-7)
    return linprog(-r, A_ub=np.vstack([constraint_matrix, non_overlap_constraints]), b_ub=np.hstack([np.ones(n) - 1e-7, np.ones(n * (n - 1) * 2)]), bounds=(0, 1))

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2) * 2 - 1
    for i in range(n):
        for j in range(i + 1, n):
            while np.linalg.norm(centers[i] - centers[j]) < 2 * lp_for_radii(centers)[0][i] + lp_for_radii(centers)[0][j]:
                centers = np.random.rand(n, 2) * 2 - 1
    initial_radii = lp_for_radii(centers)[0]
    
    def objective_function(pos):
        initial_radii = lp_for_radii(pos.reshape(n, 2))[0]
        total_radius = -np.sum(initial_radii)
        return total_radius
    
    constraints = {'type': 'ineq', 'fun': lambda pos: lp_for_radii(pos.reshape(n, 2))[1]}
    bounds = [(0, 1)] * (2 * n)
    res = minimize(objective_function, centers.flatten(), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    positions = res.x.reshape(n, 2)
    best_radii = lp_for_radii(positions)[0]
    sum_radii = -np.sum(best_radii)
    return positions, best_radii, sum_radii

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
