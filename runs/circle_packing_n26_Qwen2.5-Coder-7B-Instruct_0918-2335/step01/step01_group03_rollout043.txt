import numpy as np
from scipy.optimize import linprog, minimize

def optimise_radii(centres):
    n = centres.shape[0]
    bounds = [(0, None)] * n + [(0, None)] * n
    cons = {'type': 'ineq', 'fun': lambda x: np.array([x[i] for i in range(n, 2 * n)])
                            - (np.linalg.norm(
                                centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=2)).flatten()}
    for i in range(n):
        for j in range(i + 1, n):
            cons['fun'] += np.array([x[i] + x[j + n] - (np.linalg.norm(
                centres[i] - centres[j]))])
    res = linprog(None, A_ub=[-1] * n, b_ub=[0] * n, A_eq=np.array([1] * n),
                   b_eq=np.zeros(n // 2), x0=[0.1] * (2 * n), bounds=bounds, method='highs')
    if res.success:
        radii = res.x[n:]
        constraints_sum = np.sum(radii) - np.dot(np.maximum(np.zeros(n // 2),
                                                             (np.linalg.norm(
                                                                 centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=2) -
                                                              np.array([x[i] + x[j + n] for i in range(n)
                                                                        for j in range(i + 1, n)])))) >= 0
        if constraints_sum >= 0:
            return radii

def random_step(delta_max):
    return np.random.uniform(-delta_max, delta_max, size=2)

def slsqp_refine(centers, radii):
    n = centers.shape[0]
    bounds = ([0, 0], [1, 1]) * n + [(0, 0.5)] * n
    initial_guess = np.hstack((centers.flatten(), radii))
    objective = lambda x: -np.sum(x[n:])
    constraints = [{'type': 'ineq', 'fun': lambda x: 1 - x[i] - x[j + n] +
                                          np.linalg.norm(np.array(x[:n]) - x[i + n:n + 2])}
                   for i in range(n) for j in range(i + 1, n)]
    constraints += [{'type': 'ineq', 'fun': lambda x: x[i] - x[n] - x[j + n] +
                                          np.linalg.norm(np.array(x[:n]) - x[i + n:n + 2])}
                    for i in range(n) for j in range(i + 1, n)]
    constraints += [{'type': 'ineq', 'fun': lambda x: x[j + n] - x[i] - x[n] +
                                          np.linalg.norm(np.array(x[:n]) - x[i + n:n + 2])}
                    for i in range(n) for j in range(i + 1, n)]
    constraints += [{'type': 'ineq', 'fun': lambda x: 1 - x[i] - x[j + n]}
                   for i in range(n) for j in range(i + 1, n)]
    constraints += [{'type': 'ineq', 'fun': lambda x: x[i] - x[n] - x[j + n]}
                   for i in range(n) for j in range(i + 1, n)]
    constraints += [{'type': 'ineq', 'fun': lambda x: x[j + n] - x[i] - x[n]}
                   for i in range(n) for j in range(i + 1, n)]
    x0 = initial_guess
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    if res.success:
        updated_centers, updated_radii = res.x[:n], res.x[n:]
        if validate_packing(updated_centers.reshape(-1, 2), updated_radii)[0]:
            return updated_centers, updated_radii
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    centers = np.array([[0.2, 0.2], [0.2, 0.4], [0.2, 0.6], [0.2, 0.8],
                        [0.25, 0.25], [0.25, 0.75], [0.25, 0.25],
                        [0.25, 0.75], [0.5, 0.5], [0.5, 0.5],
                        [0.5, 0.5], [0.5, 0.5], [0.5, 0.5],
                        [0.75, 0.25], [0.75, 0.75], [0.8, 0.2],
                        [0.8, 0.4], [0.8, 0.6], [0.8, 0.8],
                        [0.85, 0.3], [0.85, 0.5], [0.85, 0.7],
                        [0.9, 0.25], [0.9, 0.75]])
    radii = np.full(26, 0.1)
    best_sum = np.sum(radii)
    best_centres = centers
    best_radii = radii

    delta_max = 0.05
    T0 = 1.0
    alpha = 0.995
    NITER = 50000

    for iter in range(1, NITER + 1):
        i = np.random.randint(0, 26)
        dx, dy = random_step(delta_max)
        if 0 <= centers[i, 0] + dx <= 1 and 0 <= centers[i, 1] + dy <= 1:
            new_centres = centers.copy()
            new_centres[i] += [dx, dy]

            new_radii = optimise_radii(new_centres)
            if new_radii is not None:
                new_sum = np.sum(new_radii)
                ΔS = new_sum - best_sum
                if ΔS > 0:
                    centers, radii, best_sum = new_centres, new_radii, new_sum
                    best_centres, best_radii = new_centres, new_radii
                else:
                    T = T0 * alpha ** iter
                    if np.random.rand() < np.exp(ΔS / T):
                        centers, radii = new_centres, new_radii

    best_centres, best_radii = slsqp_refine(best_centres, best_radii)
    return best_centres, best_radii, np.sum(best_radii)
