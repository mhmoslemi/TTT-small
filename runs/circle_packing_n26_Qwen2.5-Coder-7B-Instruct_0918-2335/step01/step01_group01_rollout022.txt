```python
import numpy as np
from scipy.optimize import minimize
import random

def obj(v):
    r = v[26:]  
    return -np.sum(r)

def constraints(v):
    pos = v[:26].reshape(13,2)
    r   = v[26:]
    cons = np.empty(52, float)
    cons[ 0:13] = pos[:,0] - r    
    cons[13:26] = 1 - pos[:,0] - r
    cons[26:39] = pos[:,1] - r
    cons[39:52] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask  = np.triu(np.ones((13,13), bool), k=1)
    pair_cons = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.concatenate([cons, pair_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0,1)]*26          
    bounds += [(0,0.5)]*26        

    best_center = None
    best_radii = None
    best_sum = 0

    for _ in range(20):
        init = np.random.rand(52)
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type':'ineq',
                                                   'fun': constraints},
                       options={'maxiter':4000, 'ftol':1e-9, 'disp':False})
        
        if res.success and np.all(constraints(res.x) >= -1e-10):
            r = res.x[26:]
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_center = res.x[:26].reshape(13,2)
                best_radii = r
                best_sum = cur_sum

    return best_center, best_radii, best_sum

centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}\nRadii: {radii}\nSum of Radii: {sum_radii}")
```
