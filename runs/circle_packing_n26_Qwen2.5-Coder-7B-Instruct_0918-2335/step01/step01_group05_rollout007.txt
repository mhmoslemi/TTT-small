import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    n = len(centers)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([1, 0, 0, 0])
        b_ub.append(centers[i, 0])
        A_ub.append([-1, 0, 0, 0])
        b_ub.append(1 - centers[i, 0])
        A_ub.append([0, 1, 0, 0])
        b_ub.append(centers[i, 1])
        A_ub.append([0, -1, 0, 0])
        b_ub.append(1 - centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            distance = np.linalg.norm(centers[i] - centers[j])
            A_ub.append([1, 0, 1, 0, 0, 0, -1, 0, 0, 0, 0, 0])
            b_ub.append(distance - centers[i, 2] - centers[j, 2])
            A_ub.append([0, 1, 0, 0, 1, 0, 0, 0, -1, 0, 0, 0])
            b_ub.append(distance - centers[i, 2] - centers[j, 2])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(1e-6, 0.5)] * n
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
    if not res.success:
        raise ValueError("LP problem was not solved successfully.")
    return -res.fun, res.x

def evaluate_fitness(individual):
    centers = individual[:2 * 26].reshape(26, 2)
    individual_radii = individual[2 * 26:]
    individual_radii = np.clip(individual_radii, 0, 0.5)
    sum_radii, radii = solve_radii(centers)
    return sum_radii

def crossover(parent1, parent2):
    start = np.random.randint(0, 2 * 26)
    end = np.random.randint(start + 1, 2 * 26 + 1)
    child = np.copy(parent1)
    child[start:end] = parent2[start:end]
    return child

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, size=individual.shape)
    individual = np.clip(individual, 0, 1)
    return individual

def clamp_to_square(individual):
    individual[:2 * 26] = np.clip(individual[:2 * 26], 0, 1)
    return individual

def run_packing():
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    np.random.seed(42)
    population = np.random.rand(population_size, 2 * 26 + n)
    population[:, :2 * 26] = np.clip(population[:, :2 * 26], 0, 1)
    population[:, :n] = np.clip(population[:, :n], 1e-6, 0.5)

    best_sol = None
    best_sum = -np.inf

    for gen in range(generations):
        fitnesses = np.array([evaluate_fitness(individual) for individual in population])
        individual_surviving_ids = np.argsort(fitnesses)[::-1]
        new_population = np.empty_like(population)
        new_population[:elitism] = population[individual_surviving_ids[:elitism]]

        for i in range(elitism, population_size):
            parent1_id = np.random.choice(individual_surviving_ids, size=tournament_size, replace=False)
            parent2_id = np.random.choice(individual_surviving_ids, size=tournament_size, replace=False)
            parent1 = population[parent1_id[np.argmax(fitnesses[parent1_id])]]
            parent2 = population[parent2_id[np.argmax(fitnesses[parent2_id])]]
            child = crossover(parent1, parent2)
            child = mutate(child, initial_sigma)
            child = clamp_to_square(child)
            new_population[i] = child

        population = new_population
        initial_sigma *= sigma_decay

        best_ind_id = np.argmax(fitnesses)
        if fitnesses[best_ind_id] > best_sum:
            best_sum = fitnesses[best_ind_id]
            best_sol = population[best_ind_id]

    best_centers = best_sol[:2 * 26].reshape(26, 2)
    best_radii = best_sol[2 * 26:]
    best_radii = np.clip(best_radii, 0, 0.5)
    _, best_radii = solve_radii(best_centers)
    best_sum, _ = solve_radii(best_centers)

    result = minimize(fun=lambda centers: -np.sum(solve_radii(centers)[1]), x0=best_centers.flatten(), method='SLSQP', bounds=[(0, 1)] * 2 * n)
    best_centers = result.x.reshape(26, 2)
    best_radii, _ = solve_radii(best_centers)

    return best_centers, best_radii, np.sum(best_radii)
