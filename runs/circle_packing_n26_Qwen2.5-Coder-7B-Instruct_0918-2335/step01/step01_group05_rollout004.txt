import numpy as np
from scipy.optimize import linprog

N = 26
pairs = [(i, j) for i in range(N) for j in range(i + 1, N)]

def evaluate_positions(centers):
    #(4 * N, N)
    A_bound = np.row_stack((np.eye(N), -np.eye(N), np.eye(N), -np.eye(N))).T
    b_bound = np.hstack((centers[:, 0], centers[:, 0], centers[:, 1], centers[:, 1]))
    
    A_pairwise = np.array([[ centers[i] - centers[j], centers[i] - centers[j]].reshape(-1) for i, j in pairs ])
    b_pairwise = np.sqrt(np.sum((centers[pairs[:, 0]] - centers[pairs[:, 1]]) ** 2, axis=1))

    c = -np.ones(N)
    
    res = linprog(c, A_ub=np.vstack((A_bound, A_pairwise)), b_ub=np.hstack((b_bound, b_pairwise)), bounds=(0, 0.5))
    
    if not res.success:
        raise ValueError("LP failed")

    radii = res.x
    return np.sum(radii), radii

def jitter(centers, step=0.02):
    centers += np.random.uniform(-step, step, size=centers.shape)
    centers = np.clip(centers, 0, 1)
    return centers

def generate_patterns():
    grid_centers = np.array([[i / 5., j / 5.] for i in range(1, 10) for j in range(1, 10)])
    hex_centers = np.array([[0.18 * (i % 6), np.sqrt(3)/3 * (i % 6) + np.sqrt(3)/18 * (i // 6)] for i in range(21)])
    offset_corner_centers = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]])
    offset_corner_centers = np.vstack((offset_corner_centers, [[0.15, y] for y in np.linspace(0.15, 0.85, 13)]))
    offset_corner_centers = np.vstack((offset_corner_centers, [[x, 0.15] for x in np.linspace(0.15, 0.85, 13)]))
    offset_corner_centers = np.sort(offset_corner_centers, axis=0)
    return [grid_centers[:26], hex_centers, offset_corner_centers]

def run_packing():
    best_sum = -np.inf
    best_centers = None

    for centers in generate_patterns():
        centers = jitter(centers)
        sum_radii, _ = evaluate_positions(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers

    final_sum, final_radii = evaluate_positions(best_centers)
    return best_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
