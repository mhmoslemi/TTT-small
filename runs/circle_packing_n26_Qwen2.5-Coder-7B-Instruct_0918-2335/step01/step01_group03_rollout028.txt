from scipy.optimize import minimize, linprog

    initial_centres.append((0.5, 0.5))

    T0 = 1.0
    alpha = 0.995
    NITER = 50000
    delta_max = 0.05

            A.append(A_bound)
            b.append(b_bound)
            A_bound = [0, -1]

            A.append(A_bound)
            b.append(b_bound)
            A_bound = [0, 1]
            b_bound = 1 - y
            A.append(A_bound)
            b.append(b_bound)

        # Pairwise non-overlap conditions
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                d = np.sqrt(dx**2 + dy**2)

                b_overlap = d - (radii[i] + radii[j])
                A.append(A_overlap)
                b.append(b_overlap)
                A_overlap = [0, 1, 0, -1, 0, 0, 0, 0]
                b_overlap = d - (radii[i] + radii[j])
                A.append(A_overlap)
                b.append(b_overlap)
                A_overlap = [-1, 0, 0, 0, 1, 0, 0, 0]
                b_overlap = d - (radii[i] + radii[j])
                A.append(A_overlap)
                b.append(b_overlap)
                A_overlap = [0, -1, 0, 0, 0, 1, 0, 0]

        return r_result.x

    best_radii = optimise_radii(best_centres)

def find_contact_pairs(radii):
    n = len(radii)
    constraints = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = radii[j] + radii[i]
            dy = np.sqrt((np.random.rand() - 0.5) ** 2 + (np.random.rand() - 0.5) ** 2)
            constraints[i, j] = dx - dy
            constraints[j, i] = dx - dy
    return constraints[np.triu_indices(n, k=1)].ravel()
