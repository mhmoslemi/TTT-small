import numpy as np
from scipy.optimize import linprog

def eval_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        radii[i] = min(centers[i,0], 1 - centers[i,0], centers[i,1], 1 - centers[i,1], 0.5)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            dist = np.sqrt(dx * dx + dy * dy)
            radii[i] = min(radii[i], dist - radii[j] - 1e-9)
            radii[j] = min(radii[j], dist - radii[i] - 1e-9)
    return np.sum(radii), radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pairs = [(i,j) for i in range(n) for j in range(i+1, n)]
    M = len(pairs)
    
    def solve_lp(centres):
        c = -np.ones(n)
        A_ub = np.empty((n*4 + M, n))
        b_ub = np.empty(n*4 + M)

        for i in range(n):
            A_ub[4*i+0, i] = 1
            b_ub[4*i+0] = centres[i,0]
            A_ub[4*i+1, i] = 1
            b_ub[4*i+1] = 1 - centres[i,0]
            A_ub[4*i+2, i] = 1
            b_ub[4*i+2] = centres[i,1]
            A_ub[4*i+3, i] = 1
            b_ub[4*i+3] = 1 - centres[i,1]

        for k, (i, j) in enumerate(pairs):
            index = 4*n + k
            A_ub[index, i] = 1
            A_ub[index, j] = 1
            b_ub[index] = np.linalg.norm(centres[i] - centres[j])
        
        bounds = [(0, 0.5) for _ in range(n)]
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x, -result.fun
        else:
            raise ValueError("LP solver failed.")
    
    def acceptance_ratio(sum_new, sum_old, T):
        return np.exp((sum_new - sum_old) / T)

    def simulated_annealing(centers, max_iter=5000, sigma=0.05, alpha=0.9995, T=0.1):
        best_centers = centers.copy()
        best_sum, _ = eval_radii(best_centers)
        current_centers = centers.copy()
        for _ in range(max_iter):
            i = np.random.randint(n)
            delta = np.random.normal(0, sigma)
            new_centers = current_centers.copy()
            new_centers[i] = np.clip(current_centers[i] + delta, 0, 1)
            sum_new, _ = eval_radii(new_centers)
            if sum_new > best_sum:
                best_sum = sum_new
                best_centers = new_centers.copy()
            elif np.random.rand() < acceptance_ratio(sum_new, sum_old, T):
                current_centers = new_centers.copy()
                sum_old = sum_new
            sigma *= alpha
            T *= 0.99
        return best_centers, best_sum

    initial_centres = np.random.rand(n, 2)
    initial_radius_sum, _ = eval_radii(initial_centres)

    for pattern in [
        np.array([[0.1, a/2], [0.3, a/2], [0.5, a/2], [0.7, a/2], [0.9, a/2] for a in range(1, 10)]),
        np.array([[0.1+dy, 0.1+dx] for dx in np.linspace(0, 0.2, 5) for dy in np.linspace(0, 0.2, 5)]),
        np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]] + 
                   [0.1+dx, 0.1+dy] for dx in np.linspace(0.02, 0.2, 6) for dy in np.linspace(0.02, 0.2, 6)])
    ]:
        pattern = pattern[:n]
        jittered_pattern = np.clip(pattern + (np.random.rand(n, 2) - 0.5) * 0.02, 0, 1)
        r, _ = eval_radii(jittered_pattern)
        if r > initial_radius_sum:
            initial_centres = jittered_pattern.copy()
            initial_radius_sum = r

    best_centres, best_sum = simulated_annealing(initial_centres)
    _, final_radii = solve_lp(best_centres)

    assert validate_packing(best_centres, final_radii)[0]
    return best_centres, final_radii, best_sum

print(run_packing())
