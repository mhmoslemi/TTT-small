import numpy as np
from scipy.optimize import minimize, linprog

def compute_optimal_radii(pos):
    r = np.zeros(26)
    for i in range(26):
        r[i] = pos[i] / 2.0
    for i in range(26):
        for j in range(i + 1, 26):
            xdiff = pos[i] - pos[j]
            ydiff = pos[j * 26 + i] - pos[j * 26 + j]
            dist = np.sqrt(xdiff**2 + ydiff**2)
            r[i] = max(r[i], dist - r[j])
            r[j] = max(r[j], dist - r[i])
    return r

def objective(pos):
    r = compute_optimal_radii(pos)
    return -np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_center = None
    best_radii = None
    best_sum = 0

    for seed in range(20):
        np.random.seed(seed)
        init_centres = np.random.rand(52)
        res = minimize(fun=objective, x0=init_centres, method='SLSQP',
                       bounds=[(0,1)]*52, options={'maxiter':2000})
        
        if res.success:
            r = compute_optimal_radii(res.x)
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_center = res.x[:52].reshape(26,2)
                best_radii = r
                best_sum = cur_sum

    return best_center, best_radii, best_sum
