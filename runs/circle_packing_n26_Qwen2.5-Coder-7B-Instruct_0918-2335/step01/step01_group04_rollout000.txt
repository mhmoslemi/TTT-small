import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, minimize

def pairwise_dists(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.linalg.norm(centers[i] - centers[j])
    return dists

def solve_radii_lp(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    r_max = 0.5
    A_ineq = []
    b_ineq = []
    
    A_ineq.append(np.concatenate((np.eye(n), -np.eye(n), np.eye(n), -np.eye(n))))
    b_ineq.extend([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]])
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append(np.concatenate(([0] * n, [np.zeros(n)], [0] * n, [0] * n, [1] * n)))
            b_ineq.append(centers[i, 0] + centers[j, 0] - centers[i, 1] - centers[j, 1])
            A_ineq.append(np.concatenate(([0] * n, [0] * n, [0] * n, [np.zeros(n)], [1] * n)))
            b_ineq.append(centers[i, 0] + centers[j, 0] - centers[i, 1] - centers[j, 1])
            A_ineq.append(np.concatenate(([0] * n, [0] * n, [np.zeros(n)], [0] * n, [1] * n)))
            b_ineq.append(centers[i, 0] + centers[j, 0] - centers[i, 1] - centers[j, 1])
            A_ineq.append(np.concatenate(([0] * n, [np.zeros(n)], [0] * n, [0] * n, [1] * n)))
            b_ineq.append(centers[i, 0] + centers[j, 0] - centers[i, 1] - centers[j, 1])
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=[(0, r_max)] * n)
    return res.x

def sa_optimize(initial_centers: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    best_centers = initial_centers.copy()
    best_radii = solve_radii_lp(initial_centers)
    best_sum_radii = np.sum(best_radii)
    
    for _ in range(500):
        i = np.random.randint(n)
        dists = cdist(initial_centers, np.array([[0, 0], [0, 1], [1, 0], [1, 1], [0.5, 0.5]]))
        best_neighbor_idx = np.argmin(dists[i, 2:])
        delta_x, delta_y = np.random.normal(size=2) / 1000
        proposed_center = initial_centers[i] + np.array([delta_x, delta_y])
        if 0 <= proposed_center[0] <= 1 and 0 <= proposed_center[1] <= 1:
            initial_centers[i] = proposed_center
            proposed_radii = solve_radii_lp(initial_centers)
            if np.sum(proposed_radii) > best_sum_radii:
                best_sum_radii = np.sum(proposed_radii)
                best_centers = initial_centers.copy()
                best_radii = proposed_radii.copy()
    
    return best_centers, best_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    initial_centers_1 = np.random.rand(n, 2)
    initial_centers_2 = solve_radii_lp(np.random.rand(n, 2))
    initial_centers_3 = solve_radii_lp(np.random.rand(n, 2))
    
    centers, radii = sa_optimize(initial_centers_1)
    sum_radii = np.sum(radii)
    
    for _ in range(2):
        new_centers, new_radii = sa_optimize(initial_centers_2)
        if np.sum(new_radii) > sum_radii:
            centers, radii = new_centers, new_radii
            sum_radii = np.sum(radii)
    
    for _ in range(2):
        new_centers, new_radii = sa_optimize(initial_centers_3)
        if np.sum(new_radii) > sum_radii:
            centers, radii = new_centers, new_radii
            sum_radii = np.sum(radii)
    
    res = minimize(
        fun=lambda x: -np.sum(x[52:]),
        x0=np.concatenate((centers.flatten(), radii)),
        method='SLSQP',
        bounds=[(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n,
        constraints={'type': 'ineq', 'fun': lambda x: np.concatenate([
            x[:52].reshape(n, 2)[:, 0] - x[52:],  
            1 - x[:52].reshape(n, 2)[:, 0] - x[52:],  
            x[:52].reshape(n, 2)[:, 1] - x[52:],  
            1 - x[:52].reshape(n, 2)[:, 1] - x[52:],  
            cdist(x[:52].reshape(n, 2), x[:52].reshape(n, 2)) - (x[52:] + x[52:])[:, None] - 1e-12
        ])},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    final_centers = res.x[:52].reshape(n, 2)
    final_radii = res.x[52:]
    final_sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum_radii
