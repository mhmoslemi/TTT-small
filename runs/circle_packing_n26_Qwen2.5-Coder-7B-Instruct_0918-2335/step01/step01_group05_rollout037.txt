def evaluate_positions(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    dists = np.sqrt(np.sum((centers[:,None,:] - centers[None,:,:])**2, axis=2))
    for i, (p, q) in enumerate(pairs):
        r = max(r, dists[i] / 2 - 1e-9)
    return np.sum(r), r

        if len(circles) >= 26:
            break
    return circles[:26]
