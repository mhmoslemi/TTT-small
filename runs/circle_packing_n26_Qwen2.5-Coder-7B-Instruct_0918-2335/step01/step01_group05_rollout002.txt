import numpy as np
from scipy.optimize import minimize, differential_evolution

def obj(v):
    r = v[52:]
    return -np.sum(r) - 1e6 * np.sum(np.clip(constraints(v), -np.inf, 0) ** 2)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    
    res_de = differential_evolution(obj, bounds, popsize=15, maxiter=500, tol=1e-10)
    if not res_de.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    initial_guess = np.concatenate([res_de.x[:52], 0.5 * np.ones(n)])
    res_slsqp = minimize(obj, initial_guess, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    
    centers = res_slsqp.x[:52].reshape(n, 2)
    radii = res_slsqp.x[52:]
    return centers, radii, -res_slsqp.fun
