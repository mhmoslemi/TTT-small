import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers) -> np.ndarray:
    n = centers.shape[0]
    dists = np.empty((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dists[i, j] = dists[j, i] = np.sqrt(np.sum(diff ** 2))
    return dists

def solve_radii_lp(centers) -> np.ndarray:
    n = centers.shape[0]
    dists = pairwise_dists(centers)
    A_ineq = np.zeros((4 * n + n * (n - 1) // 2, 2 * n))
    b_ineq = np.zeros(4 * n + n * (n - 1) // 2)
    
    for i in range(n):
        A_ineq[4 * i:4 * i + 4, i * 2 + 0] = [1, 0, 0, 0]
        A_ineq[4 * i:4 * i + 4, i * 2 + 1] = [0, 1, 0, 0]
        A_ineq[4 * i:4 * i + 4, i * 2 + 2] = [0, 0, 1, 0]
        A_ineq[4 * i:4 * i + 4, i * 2 + 3] = [0, 0, 0, 1]
        b_ineq[4 * i:4 * i + 4] = centers[i]
        
        for j in range(i + 1, n):
            A_ineq[4 * n + i * (i - 1) // 2 + j - i - 1, i * 2 + 2] = 1
            A_ineq[4 * n + i * (i - 1) // 2 + j - i - 1, i * 2 + 3] = 1
            A_ineq[4 * n + i * (i - 1) // 2 + j - i - 1, j * 2 + 2] = 1
            A_ineq[4 * n + i * (i - 1) // 2 + j - i - 1, j * 2 + 3] = 1
            b_ineq[4 * n + i * (i - 1) // 2 + j - i - 1] = dists[i, j]
    
    result = linprog(-np.ones(2 * n), A_ub=A_ineq, b_ub=b_ineq, method='highs')
    return -result.x[:n]

def sa_optimize(initial_centers) -> Tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    radii = solve_radii_lp(initial_centers)
    centers = initial_centers.copy()
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum_radii = np.sum(radii)
    
    for _ in range(500):
        accept = 0.9
        for _ in range(n):
            i = np.random.randint(n)
            delta_x, delta_y = np.random.normal(scale=[0.005, 0.005])
            new_centers = centers.copy()
            new_centers[i, 0] += delta_x
            new_centers[i, 1] += delta_y
            
            if any((new_centers[i] < -1e-12) | (new_centers[i] > 1 + 1e-12)):
                continue
            
            new_radii = solve_radii_lp(new_centers)
            new_sum_radii = np.sum(new_radii)
            
            if new_sum_radii >= best_sum_radii or np.random.rand() < accept:
                centers = new_centers
                radii = new_radii
                best_sum_radii = new_sum_radii
                accept *= 0.95
    
    return centers, radii

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_center_patterns = np.array([
        [
            [0.5, 0.5],  # Single circle in center
            [0.25, 0.75],
            [0.25, 0.25],
            [0.75, 0.25],
            [0.75, 0.75],
            [0.5, 0.375],
            [0.5, 0.625],
            [0.375, 0.375],
            [0.375, 0.625],
            [0.625, 0.375],
            [0.625, 0.625]
        ],
        [
            [0.1, 0.1],  # Random grid
            [0.1, 0.9],
            [0.9, 0.1],
            [0.9, 0.9],
            [0.5, 0.1],
            [0.5, 0.5],
            [0.5, 0.9],
            [0.1, 0.5],
            [0.9, 0.5],
            [0.1, 0.8],
            [0.9, 0.8]
        ],
        # Add more patterns if needed
    ])

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0.0
    
    for pattern in initial_center_patterns:
        centers, radii = sa_optimize(pattern)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum_radii:
            best_centers = centers
            best_radii = radii
            best_sum_radii = sum_radii
    
    res = minimize(
        fun=lambda x: -np.sum(x[52:]),
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': lambda x: pairwise_dists(x[:52].reshape(n, 2)) - x[52:]},
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )
    
    if not res.success or np.any(pairwise_dists(res.x[:52].reshape(n, 2)) < res.x[52:]:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
