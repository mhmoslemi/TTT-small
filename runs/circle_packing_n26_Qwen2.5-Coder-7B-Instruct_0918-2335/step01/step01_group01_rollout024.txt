import numpy as np
from scipy.optimize import minimize, linprog
from collections import defaultdict

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v, pos, pairs, dists):
    r = v[52:]
    n = pos.shape[0]
    cons = np.empty(104 + 325, float)
    for i in range(n):
        cons[i] = pos[i, 0] - r[i]
        cons[i + n] = 1 - pos[i, 0] - r[i]
        cons[i + 2 * n] = pos[i, 1] - r[i]
        cons[i + 3 * n] = 1 - pos[i, 1] - r[i]
    for i, j in pairs:
        dist = dists[i, j]
        cons[i + 4 * n + j * (26 - i)] = r[i] + r[j] - dist
    return cons

def apollonius_cavities(num):
    corners = [(0, 0), (1, 0), (0, 1), (1, 1)]
    edges = [0.5 * np.array([(0, 0), (0, 1)]), 0.5 * np.array([(1, 0), (1, 1)]), 
             0.5 * np.array([(0, 0), (1, 0)]), 0.5 * np.array([(0, 1), (1, 1)]), 
             0.5 * np.array([(0, 0), (1, 1)]), 0.5 * np.array([(0, 1), (1, 0)])]
    hexagon = np.array([[1, 0, 0], [0, 1, 0], [-1, 0, 0], [0, -1, 0], [1, 1, np.sqrt(3)/2], 
                        [-1, 0, np.sqrt(3)/2], [-1, -1, np.sqrt(3)/2], [0, 1, np.sqrt(3)/2]])
    bounds = np.array([[0, 0], [1, 0], [1, 1], [0, 1], [0.5, 0], [1.5, 0], [1.5, 1], [0.5, 1], [0.25, 0.5], 
                        [0.75, 0.5], [0.5, 0.25], [0.5, 0.75], [0.25, 0.25], [0.25, 0.75], [0.75, 0.25], 
                        [0.75, 0.75], [0.5, 0.5 + np.sqrt(3)/2], [0.25, 0.5 + np.sqrt(3)/2], 
                        [0.75, 0.5 + np.sqrt(3)/2], [0.5 - np.sqrt(3)/2, 0.5], 
                        [0.5 + np.sqrt(3)/2, 0.5], [0.5, 0.5 - np.sqrt(3)/2]])
    
    cavities = []
    for _ in range(num):
        cavity = np.random.choice([corners, edges, hexagon])
        cavity = cavity + np.random.uniform(-0.1, 0.1, size=cavity.shape)
        cavity = cavity.clip(0, 1)
        cavities.append(cavity)
    return cavities

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 26 + [(0, 1)] * 26 + [(0, 0.5)] * 26

    best_center = None
    best_radii = None
    best_sum = 0

    for _ in range(20):
        init = np.random.rand(26 * 2 + 26 * 2)
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type': 'ineq',
                                                   'fun': lambda v: constraints(v, init[:26 * 2].reshape(26, 2), 
                                                                              init[26 * 2 + 26 * i:26 * 2 + 26 * (i + 1)], 
                                                                              apollonius_cavities(26)[i]) for i in range(26)}),
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        
        if res.success and np.all(constraints(res.x, init[:26 * 2].reshape(26, 2), 
                                              init[26 * 2 + 26 * i:26 * 2 + 26 * (i + 1)], 
                                              apollonius_cavities(26)[i]) >= -1e-10):
            r = res.x[52:]
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_center = init[:26 * 2].reshape(26, 2)
                best_radii = r
                best_sum = cur_sum

    return best_center, best_radii, best_sum
