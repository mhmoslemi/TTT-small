import numpy as np
from scipy.optimize import minimize, linprog, nonlin

def lp_opt(p):
    # Boundary constraints
    A_bound = np.vstack((np.eye(26) * 2,
                          np.eye(26) * -1,
                          np.eye(26),
                          np.eye(26) * -1))
    b_bound = np.hstack((np.ones(26), np.ones(26), np.ones(26), np.zeros(26)))

    # Pairwise constraints
    d_matrix = np.sqrt(np.sum((p[:, np.newaxis, :] - p[np.newaxis, :, :])**2, axis=2))
    np.fill_diagonal(d_matrix, np.inf)
    A_pair = np.subtract(d_matrix, np.eye(26))
    b_pair = np.zeros(300)

    # Objective
    c = -np.ones(26)

    # Solve LP
    res = linprog(c, A_ub=np.vstack((A_bound, A_pair)), b_ub=np.hstack((b_bound, b_pair)), method='highs')
    return -res.fun if res.success else -np.inf, res.success

def local_slsqp(c):
    def objective(r):
        return np.sum(-r)
    
    def cons(r):
        d = np.sqrt(np.sum((c[:, None, :] - c[np.newaxis, :, :])**2, axis=2))
        np.fill_diagonal(d, np.inf)
        return d - (r[:, None] + r[None,:])
    
    cons = ({'type': 'ineq', 'fun': cons})
    
    bounds = [(0, 0.5) for _ in range(26)]
    
    _ = minimize(objective, r, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 25})
    return 

def hex_grid():
    hex_lattice = np.array([[0.5, 0.5]]);
    for r in np.arange(0.5, 1.0, 0.25):
        if 2 * r + 0.5 > 1:
            break
        for theta in np.arange(0, 2 * np.pi, 2 * np.pi / 6):
            hex_lattice = np.vstack((hex_lattice, [0.5 + r * np.cos(theta), 0.5 + r * np.sin(theta)]))
    hex_lattice = hex_lattice[:26, :]
    return hex_lattice + np.random.uniform(-0.01, 0.01, (26, 2))

def random_gen():
    return np.random.uniform(0, 1, (26, 2))

def grid_gen():
    grid = np.array([[0.1, 0.1], [0.1, 0.3], [0.1, 0.5], [0.1, 0.7], [0.1, 0.9],
                      [0.3, 0.1], [0.3, 0.3], [0.3, 0.5], [0.3, 0.7], [0.3, 0.9],
                      [0.5, 0.1], [0.5, 0.3], [0.5, 0.5], [0.5, 0.7], [0.5, 0.9],
                      [0.7, 0.1], [0.7, 0.3], [0.7, 0.5], [0.7, 0.7], [0.7, 0.9],
                      [0.9, 0.1], [0.9, 0.3], [0.9, 0.5], [0.9, 0.7], [0.9, 0.9]])
    return np.vstack((grid, [0.5, 0.5]))

def corner_gen():
    corners = np.array([[0.05, 0.05], [0.05, 0.95], [0.95, 0.05], [0.95, 0.95]])
    inner = np.random.uniform(0, 1, (21, 2))
    corner_indices = np.random.choice(4, 5, replace=False)
    for idx in corner_indices:
        corner = corners[idx]
        inner_core = np.random.uniform(0.05, 0.95, (1, 2))
        inner = np.vstack((inner, corner + 0.9 * inner_core))
    return inner[:26, :]

def tournament_select(population):
    pool = np.random.choice(population, 3, replace=False)
    return max(pool, key=lambda ind: ind[1])[0]

def crossover(p_a, p_b):
    n_cenos = int(len(p_a) * 0.5)
    child = np.copy(p_a)
    child[n_cenos:] = p_b[n_cenos:]
    child += np.random.uniform(-0.005, 0.005, len(child))
    child = np.clip(child, 0, 1)
    return child

def mutate(p, p_mut=0.3, m=3):
    for _ in range(int(m * p_mut * len(p))):
        idx = np.random.randint(len(p))
        p[idx] += np.random.normal(0, 0.02, 2)
    p = np.clip(p, 0, 1)
    return p

def validate_packing_ok(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    POP = 30
    EVAL = 4000
    pbest_position = None
    pbest_solution = 0
    rng = np.random.RandomState(42)

    population = [((rng.choice([hex_grid, random_gen, grid_gen, corner_gen])(), 0) for _ in range(POP)]

    for eval_count in range(EVAL):
        parentA = tournament_select(population)
        parentB = tournament_select(population)
        child = crossover(parentA, parentB)
        child = mutate(child)
        local_slsqp(child)
        r_opt, succeeded = lp_opt(child)
        if succeeded and r_opt > pbest_solution:
            pbest_solution = r_opt
            pbest_position = child

    return pbest_position.reshape(26, 2), np.clip(r_opt, 0, None), pbest_solution
