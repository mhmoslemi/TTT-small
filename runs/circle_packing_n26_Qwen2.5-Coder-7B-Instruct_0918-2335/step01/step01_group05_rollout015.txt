import numpy as np
from scipy.optimize import minimize
from scipy.optimize.differentialevolution import differential_evolution
from scipy.spatial.distance import cdist

def obj(v):
    r = v[52:]
    return -np.sum(r)

def penalty(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(429, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    pair = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.sum(np.maximum(pair, 0)**2, axis=-1) * 1e6

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(429, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    pair = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.concatenate([cons, np.maximum(pair, 0)])

def minimize_with_penalty(fun, x0, bounds):
    def penalty_func(x):
        return fun(x) + penalty(x)
    return minimize(penalty_func, x0, method='SLSQP', bounds=bounds)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[:26] = init[:26] / 2 + 0.5
    init[52:] = np.full(n, 0.02)

    res = differential_evolution(obj, bounds, args=(penalty,), maxiter=500, popsize=15)
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    init = np.concatenate([centers.flatten(), radii])

    res = minimize_with_penalty(obj, init, bounds)
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    return centers, radii, -res.fun
