import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    best_sol = None
    best_sum = -np.inf

    for _ in range(20):
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success and constraints(res.x).min() >= -1e-10:
            current_sum = -obj(res.x)
            if current_sum > best_sum:
                best_sum = current_sum
                best_sol = res.x
    
    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = best_sol[:52].reshape(n, 2)
    radii = best_sol[52:]
    return centers, radii, best_sum
