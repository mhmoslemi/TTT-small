import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(2 * n)

    def lp_optimize_radii(centers):
        pos = centers[:n].reshape(n, 2)
        r_bounds = [(0, None)] * n
        dists = squareform(pdist(pos))
        cons = {'type': 'ineq', 'fun': lambda r: dists.flatten() - r ** 2 - 1e-6}
        result = linprog([-1] * n, A_ub=cons['fun'], b_ub=cons['fv'], bounds=r_bounds)
        if not result.success:
            return None
        return result.x

    def objective_and_constraints(candidate):
        centers, radii = candidate[:52].reshape(n, 2), candidate[52:]
        sum_radii = np.sum(radii)
        diff = squareform(pdist(centers)) - radii[:, None] - radii[None, :] + 1e-9
        overlaps = np.sum(np.maximum(-diff, 0))
        boundary_violation = np.sum(np.maximum(radii - np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]), 0))
        return -sum_radii, overlaps + boundary_violation

    def refine_with_sa(centers, LP_radii):
        sigma = 0.02
        T = 1e-1
        cool_rate = 0.995
        L = 10000
        candidate = np.concatenate([centers.flatten(), LP_radii])
        best_candidate = candidate.copy()
        best_sum_radii = objective_and_constraints((centers, LP_radii))[0]
        for _ in range(L):
            idx = np.random.randint(n * 2 + n)
            if idx < n * 2:
                new_candidate = candidate.copy()
                new_candidate[idx] += sigma * (2 * np.random.rand() - 1)
                new_candidate = np.clip(new_candidate, [0, 0, 0] * n + [0] * n, [1, 1, 1] * n + [0.5] * n)
            else:
                new_candidate = candidate.copy()
                new_candidate[idx - n] += 1e-2 * (np.random.rand() * 0.1 if candidate[idx - n] < 0.5 else -np.random.rand() * 0.1)
                new_candidate[idx - n] = max(0, min(0.5, new_candidate[idx - n]))
                new_candidate = np.clip(new_candidate, [0, 0, 0] * n + [0] * n, [1, 1, 1] * n + [0.5] * n)
            new_sum_radii, _ = objective_and_constraints((new_candidate[:52].reshape(n, 2), new_candidate[52:]))
            if new_sum_radii > best_sum_radii:
                best_sum_radii = new_sum_radii
                best_candidate = new_candidate.copy()
            elif np.exp((best_sum_radii - new_sum_radii) / T) > np.random.rand():
                best_candidate = new_candidate.copy()
        return best_candidate[:52].reshape(n, 2), best_candidate[52:]

    def candidate_generator():
        centers0 = np.random.rand(1 + n * 2)
        centers = np.zeros((n, 2))
        centers[:n] = centers0[1:n + 1]
        centers[:, 0] += centers0[n + 1::2]
        centers[:, 1] += centers0[n + 2::2]
        return centers

    best_candidate = None
    best_sum_radii = -np.inf
    for _ in range(5):
        centers = candidate_generator()
        LP_radii = lp_optimize_radii(centers)
        if LP_radii is None:
            continue
        candidate = np.concatenate([centers.flatten(), LP_radii])
        refined = refine_with_sa(candidate[:52].reshape(n, 2), LP_radii)
        sum_radii, _ = objective_and_constraints(refined)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_candidate = refined

    if best_candidate is None:
        raise RuntimeError("Optimization did not succeed")
    centers, radii = best_candidate[:52].reshape(n, 2), best_candidate[52:]
    return centers, radii, best_sum_radii
