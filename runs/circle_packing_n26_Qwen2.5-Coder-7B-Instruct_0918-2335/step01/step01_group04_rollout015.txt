import numpy as np
from scipy.optimize import linprog, minimize

n = 26
box_size = 1.0
max_radius = box_size / 2

def pairwise_dists(centers):
    diffs = centers[:, None, :] - centers[None, :, :]
    return np.sqrt((diffs ** 2).sum(axis=2))

def solve_radii_lp(centers):
    dists = pairwise_dists(centers)
    A_ineq = np.vstack((np.max(centers - dists, axis=1),  # Boundary constraints
                        np.max(dists - centers, axis=1),  # Pair constraints
                        np.min(centers, axis=0),  # Additional boundary constraints
                        1 - np.max(centers, axis=0),  # Additional boundary constraints
                        np.min(centers, axis=1),  # Additional boundary constraints
                        1 - np.max(centers, axis=1)))  # Additional boundary constraints
    b_ineq = np.hstack(([0] * n, distances, [1] * n, [1] * n, [0] * n, [1] * n))
    res = linprog(-np.ones(n), A_ub=A_ineq, b_ub=b_ineq, bounds=[(0, max_radius)] * n, method='highs')
    return -res.fun, res.x

def sa_optimize(initial_centers):
    best_centers = initial_centers
    best_radii = np.zeros(n)
    best_sum Radii = 0
    temp = 1.0
    alpha = 0.999
    num_iterations = 500
    precision = 1e-8
    
    for i in range(num_iterations):
        for _ in range(n):
            idx = np.random.randint(n)
            delta, epsilon = np.random.normal(0, temp, 2)
            new_center = initial_centers[idx] + [delta, epsilon]
            if 0 <= new_center[0] <= box_size and 0 <= new_center[1] <= box_size:
                new_radii, new_sum_radii = solve_radii_lp(np.vstack((initial_centers[:idx], 
                                                                        new_center, 
                                                                        initial_centers[idx+1:])))
                if new_sum_radii > best_sum_radii + precision:
                    initial_centers[idx] = new_center
                    best_centers[idx] = new_center
                    best_radii = new_radii
                    best_sum_radii = new_sum_radii
                elif np.random.rand() < np.exp((new_sum_radii - best_sum_radii) / temp):
                    initial_centers[idx] = new_center
                    best_centers[idx] = new_center
                    best_radii = new_radii
                    best_sum_radii = new_sum_radii
        temp *= alpha
    
    return best_centers, best_radii, best_sum_radii

def run_packing():
    initial_centers = np.random.rand(n, 2)
    best_centers, best_radii, best_sum_radii = sa_optimize(initial_centers)
    
    res = minimize(
        fun=lambda x: -np.sum(x[-n:]),
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=[(0, box_size)] * (2 * n) + [(0, max_radius)] * n,
        constraints={'type': 'ineq', 'fun': lambda x: np.hstack((np.max(x[:(2*n)], axis=0) - x[52:], 
                                                                        x[52:] - np.min(x[:(2*n)], axis=0)))},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success or np.any(np.hstack((np.max(res.x[:52], axis=0) - res.x[52:], 
                                            res.x[52:] - np.min(res.x[:52], axis=0))) < -1e-10):
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
