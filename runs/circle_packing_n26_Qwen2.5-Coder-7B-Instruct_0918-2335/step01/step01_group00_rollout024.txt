import numpy as np
from scipy.optimize import minimize
from scipy.spatial import ConvexHull

def heuristic_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    initial_positions = (np.random.rand(n) * (1 - 2 * np.sqrt(0.5)) + np.sqrt(0.5))[:n]
    initial_positions = np.atleast_2d(initial_positions).T
    initial_radii = np.zeros(n)
    
    for i in range(n):
        initial_radii[i] = np.min([0.5, 1 - np.min([np.abs(p), 1 - p]) for p in initial_positions[:, i]])
    
    centers = initial_positions
    radii = initial_radii
    sum_radii = np.sum(radii)
    
    while sum_radii < 2.636:
        new_radii = np.zeros(n)
        for i in range(n):
            if np.sum((centers[:, i] - centers[:, :i]) ** 2, axis=1) > radii[i] ** 2 + 1e-6:
                new_radii[i] = np.max([0, 0.5 - np.min([np.abs(p), 1 - p]) for p in centers[:, i]])
        if np.sum(new_radii) == np.sum(radii):
            break
        centers = (centers.T * new_radii).T / np.sum(new_radii) / np.sqrt(0.5 / np.max(new_radii))
        radii = new_radii
        sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def obj(v):
        r = v[52:]                      # last 26 entries
        return -np.sum(r)

    def constraints(v):
        pos = v[:52].reshape(26,2)      # (x,y) pairs
        r   = v[52:]                   # radii

        # boundary terms – 4*26
        cons = np.empty(104, float)
        cons[ 0:26] = pos[:,0] - r     # left/right
        cons[26:52] = 1 - pos[:,0] - r
        cons[52:78] = pos[:,1] - r
        cons[78:104] = 1 - pos[:,1] - r

        # pairwise terms – 26 choose 2
        diffs = pos[:,None,:] - pos[None,:,:]   # (26,26,2)
        dists = np.sqrt((diffs**2).sum(axis=2))
        mask  = np.triu(np.ones((26,26), bool), k=1)
        pair_cons = dists[mask] - (r[:,None] + r[None,:])[mask]
        return np.concatenate([cons, pair_cons])

    bounds = [(0,1)]*26          # x
    bounds += [(0,1)]*26          # y
    bounds += [(0,0.5)]*26        # radii

    nrestarts = 20
    best_r = None
    best_res = None

    for _ in range(nrestarts):
        init = np.random.rand(78)
        init[52:] = 0.05 * np.random.rand(26)
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type':'ineq',
                                                   'fun': constraints},
                       options={'maxiter':4000, 'ftol':1e-9, 'disp':False})
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                best_r = res.x[52:]

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers = best_res.x[:52].reshape(26, 2)
    sum_radii = -best_res.fun
    return centers, best_r, sum_radii

# Using the heuristic packing as a fallback if optimization fails
def run_packing_fallback() -> tuple[np.ndarray, np.ndarray, float]:
    try:
        return run_packing()
    except ValueError:
        return heuristic_packing()

# Call the fallback function
def run() -> tuple[np.ndarray, np.ndarray, float]:
    return run_packing_fallback()
