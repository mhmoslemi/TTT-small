import numpy as np
from scipy.optimize import linprog, minimize

def generate_grid(n):
    m = int(np.sqrt(n))
    padding = (n - m*m) // 2
    centers = []
    for i in range(m):
        for j in range(m):
            centers.append([j/m, i/m])
    for i in range(padding):
        centers.append([0.5, 0.5])
    return np.array(centers)

def generate_hexagon(n):
    m = int(np.sqrt(n))
    padding = (n - m*m) // 2
    centers = []
    d = np.sqrt(3)/2
    offsets = [d/2, d, d]
    last_y = 0
    for level in range(m):
        for offset in offsets[:level+1]:
            y = last_y + d/2 * level + offset
            for x in np.linspace(0, 1-d/2*level, level+1):
                centers.append([x, y])
        last_y = y
    for i in range(padding):
        centers.append([0.5, 0.5])
    return np.array(centers)

def generate_edge(n):
    m = int(np.sqrt(n))
    padding = (n - m*m) // 2
    centers = []
    for i in range(m):
        for j in range(m):
            if i == 0 or i == m-1 or j == 0 or j == m-1:
                centers.append([j/m, i/m])
    for i in range(padding):
        centers.append([0.5, 0.5])
    return np.array(centers)

def perturb(centers, sigma):
    perturbed = centers + np.random.normal(0, sigma, centers.shape)
    perturbed = np.clip(perturbed, 0, 1)
    return perturbed

def lp_optimize_radii(centers):
    n = centers.shape[0]
    A = np.zeros((n, 2*n+2*n*(min(n-1, 5))))
    b = np.zeros(n)
    c = np.zeros(2*n)
    ridx = np.arange(n)
    
    xidx = np.arange(n, 2*n)
    for i in range(n):
        c[xidx[i]] = 1
        A[i, xidx[i]] = 1
        A[i, ridx[i]] = -1
        b[i] = centers[i, 0]
    
    for i in range(n):
        c[xidx[i+n]] = 1
        A[i+n, xidx[i+n]] = 1
        A[i+n, ridx[i]] = -1
        b[i+n] = 1 - centers[i, 0]
    
    i = 2*n
    for yidx in reversed(range(2*n, 2*n+2*(min(n-1, 5)))):
        A[i, yidx] = 1
        A[i, ridx[i-2*n]] = -1
        b[i] = centers[i-2*n, 1]
        i += 1
    
    for yidx in reversed(range(2*n, 2*n+2*(min(n-1, 5)))):
        A[i, yidx] = 1
        A[i, ridx[i-2*n]] = -1
        b[i] = 1 - centers[i-2*n, 1]
        i += 1
    
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            c = -1
            A[i*n+j, ridx[i]] = 1
            A[i*n+j, ridx[j]] = 1
            A[i*n+j, xidx[i]] = -1
            A[i*n+j, xidx[j]] = -1
            A[i*n+j, yidx[i]] = -1
            A[i*n+j, yidx[j]] = -1
            b[i*n+j] = d - 1e-6
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    if not result.success:
        return None
    return result.x[n:2*n]
