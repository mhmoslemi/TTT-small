import numpy as np
from scipy.optimize import linprog, minimize

def lp_radii(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    objective = -np.ones(n)
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i][0]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - centers[i][0] - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i][1]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - centers[i][1] - r[i]})
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, d=distance: r[i] + r[j] - distance})
    
    result = linprog(c=objective, A_ub=np.array([c['fun'] for c in constraints]), b_ub=np.zeros(len(constraints)), method='highs')
    if result.success:
        r = -result.x
    return r

def sa_packing(num_moves=50, num_temps=300, init_temp=0.1, min_temp=1e-5):
    n = 26
    centers = np.random.rand(n, 2)
    best_centers = centers
    best_sum_radii = np.sum(lp_radii(centers))
    
    for k in range(num_temps):
        T = init_temp / np.log(1 + k)
        for _ in range(num_moves):
            i = np.random.randint(n)
            proposal = centers.copy()
            proposal[i] += T * np.random.randn(2)
            proposal[i] = np.clip(proposal[i], 0, 1)
            new_radii = lp_radii(proposal)
            new_sum_radii = np.sum(new_radii)
            if new_sum_radii >= best_sum_radii or np.random.rand() < np.exp((new_sum_radii - best_sum_radii) / T):
                centers = proposal
                best_centers = proposal
                best_sum_radii = new_sum_radii
    
    return best_centers, best_sum_radii

def slsqp_refine(centers, sum_radii):
    n = centers.shape[0]
    x0 = np.hstack((centers.flatten(), lp_radii(centers)))
    bounds = [(0, 1)] * n + [(0, 0.5)] * n
    
    def objective(x):
        return -np.sum(x[-n:])
    
    def constraints(x):
        centers = x[:2*n].reshape(n, 2)
        r = x[-n:]
        cons = []
        
        # Boundary constraints
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i][0]})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - centers[i][0] - r[i]})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - centers[i][1]})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - centers[i][1] - r[i]})
        
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                distance = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, d=distance: r[i] + r[j] - distance})
        
        return cons
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 8000, 'ftol': 1e-9, 'disp': False})
    if result.success:
        centers = result.x[:2*n].reshape(n, 2)
        sum_radii = -result.fun
    return centers, sum_radii

def run_packing():
    best_sol = None
    best_sum = 0
    
    for _ in range(3):
        centers, sum_radii = sa_packing(num_moves=50, num_temps=300, init_temp=0.1, min_temp=1e-5)
        centers, sum_radii = slsqp_refine(centers, sum_radii)
        if sum_radii > best_sum:
            best_sol = (centers, sum_radii)
            best_sum = sum_radii
    
    return best_sol

centers, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
