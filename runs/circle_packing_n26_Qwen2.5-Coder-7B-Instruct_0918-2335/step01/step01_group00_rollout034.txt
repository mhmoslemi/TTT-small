from scipy.spatial import distance_matrix

def make_d_matrix(centres: np.ndarray) -> np.ndarray:
    n = centres.shape[0]
    D = distance_matrix(centres, centres)
    np.fill_diagonal(D, np.inf)
    return D

def compute_radii(centres: np.ndarray) -> (np.ndarray, float):
    n = centres.shape[0]
    D = make_d_matrix(centres)
    w_limits = np.minimum.reduce([centres[:,0], 1-centres[:,0], centres[:,1], 1-centres[:,1]])
    
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append(np.concatenate([np.eye(n)[i], np.zeros(n)]))
        b_ub.append(w_limits[i])
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub.append(np.concatenate([np.eye(n)[i], np.eye(n)[j]]))
            b_ub.append(D[i, j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, x) for x in w_limits], method='highs')
    if not res.success:
        return np.zeros(n), -np.inf
    
    return res.x, -res.fun
