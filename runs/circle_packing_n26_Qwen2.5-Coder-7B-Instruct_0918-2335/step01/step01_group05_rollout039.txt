import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

np.random.seed(42)

def solve_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.empty((4*n + n*(n-1)//2, n))
    b_ub = np.empty(4*n + n*(n-1)//2)
    pairs = []
    
    for i in range(n):
        for j in range(i+1, n):
            pairs.append((i, j))
    
    k = 0
    for i in range(n):
        for j in range(i+1, n):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            A_ub[k, i] = 1
            A_ub[k, j] = 1
            b_ub[k] = dist
            k += 1
    
    for i in range(n):
        x_i, y_i = centers[i]
        A_ub[k, i] = 1
        A_ub[k, i] = 1
        A_ub[k, i] = 1
        A_ub[k, i] = 1
        b_ub[k] = x_i
        b_ub[k] = y_i
        b_ub[k] = 1 - x_i
        b_ub[k] = 1 - y_i
        k += 4
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=((0, 0.5),) * n, method='highs')
    if not result.success or result.fun < -1e-9:
        return None, None
    return -result.fun, result.x

def evaluate_fitness(individual):
    centers = individual.reshape(-1, 2)
    sum_r, radii = solve_radii(centers)
    if sum_r is None:
        return 0.0
    return sum_r

def crossover(parent1, parent2):
    n = parent1.shape[0]
    i1, i2 = np.sort(np.random.choice(n, 2, replace=False))
    child1 = np.copy(parent1)
    child2 = np.copy(parent2)
    child1[i1:i2] = parent2[i1:i2]
    child2[i1:i2] = parent1[i1:i2]
    return child1, child2

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, individual.shape)
    return np.clip(individual, 0, 1)

def clamp_to_square(individual):
    return np.clip(individual, 0, 1)

def run_packing():
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigmas = np.full(generations, initial_sigma) * 0.99 ** np.arange(generations)
    elitism = 5
    tournament_size = 5
    
    def selection(population):
        selected = np.empty(elitism, dtype=int)
        for i in range(elitism):
            candidates = np.random.choice(population_size, tournament_size, replace=False)
            best = candidates[np.argmax([evaluate_fitness(population[c]) for c in candidates])]
            selected[i] = best
        return selected
    
    def fitness(individual):
        return evaluate_fitness(individual)
    
    population = np.random.rand(population_size, 2 * n)
    best_solution = None
    best_sum = -np.inf
    
    for gen in range(generations):
        fitness_values = np.array([fitness(individual) for individual in population])
        selected_indices = selection(population)
        offspring = np.empty_like(population)
        for i in range(0, population_size, 2):
            parent1 = population[selected_indices[np.random.randint(elitism)]]
            parent2 = population[selected_indices[np.random.randint(elitism)]]
            child1, child2 = crossover(parent1, parent2)
            mutation_rate = max(0.01, mutation_rate * 0.99)
            child1 = mutate(child1, sigmas[gen])
            child2 = mutate(child2, sigmas[gen])
            offspring[i] = clamp_to_square(child1)
            offspring[i+1] = clamp_to_square(child2)
        
        population = offspring
        
        for i in range(elitism):
            population[selected_indices[i]] = best_solution
        
        for candidate in population:
            current_sum, _ = solve_radii(candidate.reshape(-1, 2))
            if current_sum > best_sum:
                best_sum = current_sum
                best_solution = candidate
    
    if best_solution is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = best_solution.reshape(-1, 2)
    _, radii = solve_radii(centers)
    return centers, radii, best_sum
