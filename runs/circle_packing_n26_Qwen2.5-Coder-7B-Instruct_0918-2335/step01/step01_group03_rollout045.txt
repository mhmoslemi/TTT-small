import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def lp_optimize_radii(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    A_eq = np.eye(n)
    b_eq = np.zeros(n)
    A = np.zeros((n*(n-1)//2, n))
    b = np.zeros(n*(n-1)//2)

    for i in range(n):
        for j in range(i+1, n):
            A[i*(n-1)//2 + j - i - 1, i] = 1
            A[i*(n-1)//2 + j - i - 1, j] = 1
            b[i*(n-1)//2 + j - i - 1] = dists[i, j] - 1e-6

    res = linprog(-np.ones(n), A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=[(1e-6, None) for _ in range(n)])
    return res.x if res.success else None

def evaluate(solution):
    centers, radii = solution[:n*2].reshape(n, 2), solution[n*2:]
    if not validate_packing(centers, radii)[0]:
        return float('-inf')
    return -radii.sum()

def refine_with_sa(centers, radii, n_iter=10000, sigma=0.02, λ₁=1e6, λ₂=1e6, T₀=1e-1, cool_rate=0.995):
    n = len(radii)
    pos = centers.flatten()
    rad = radii.copy()

    T = T₀
    for _ in range(n_iter):
        i = np.random.randint(n)
        new_rad = np.clip(rad[i] + np.random.normal(0, sigma), 0, min(0.5, pos[2*i] + pos[2*i+1]))

        delta = -(rad[i] - new_rad) + λ₁ * max(0, rad[i] + rad[rad != i] - squareform(pdist(centers))[:, rad != i].flatten() - 1e-6).sum() \
                    + λ₂ * max(0, rad[i] - pos[2*i], rad[i] - (1-pos[2*i]), rad[i] - pos[2*i+1], rad[i] - (1-pos[2*i+1])).sum()
        if np.random.rand() < np.exp(delta / T):
            rad[i] = new_rad

        T *= cool_rate

    return np.array(pos.reshape(n, 2)), rad

def run_packing():
    np.random.seed(42)
    n = 26
    N_ATTEMPTS = 5

    radii_best = np.zeros(n)
    centers_best = np.zeros((n, 2))
    sum_radii_best = 0

    patterns = [
        lambda centers: centers,
        lambda centers: centers + np.random.normal(0, 0.02, centers.shape),
        lambda centers: np.array([[0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8], [0.5, 0.5]]),
        lambda centers: np.array([[0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8], [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6]]),
        lambda centers: np.random.rand(n, 2)
    ]

    for pattern in patterns:
        for attempt in range(N_ATTEMPTS):
            centers = pattern(np.random.rand(n, 2))
            distances = squareform(pdist(centers))
            idx = np.argsort(distances)[np.argsort(-np.sum(distances, axis=1))][1:]
            centers = centers[idx]
            radii = lp_optimize_radii(centers)
            if radii is None:
                continue
            sum_radii = -evaluate((centers.flatten(), radii))
            if sum_radii > sum_radii_best:
                centers_best = centers
                radii_best = radii
                sum_radii_best = sum_radii

    refined_centers, refined_radii = refine_with_sa(centers_best, radii_best)
    res = minimize(fun=evaluate, x0=np.concatenate((refined_centers.flatten(), refined_radii)),
                     method='SLSQP', bounds=[(0, 1) for _ in range(2*n) + [0, 0.5] * n],
                     constraints={'type': 'ineq', 'fun': lambda x: x[2*n:].sum()},
                     options={'maxiter': 2000, 'ftol': 1e-8})
    if res.success:
        centers_best = res.x[:n*2].reshape(n, 2)
        radii_best = res.x[n*2:]
        sum_radii_best = -res.fun
    return centers_best, radii_best, sum_radii_best

centers, radii, sum_radii = run_packing()
