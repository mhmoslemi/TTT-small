import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v):
    r = v[52:]
    return -np.sum(r)

def pair_constraints(pos, r):
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return dists[mask] - (r[:,None] + r[None,:])[mask]

def bounds(v):
    return bounds

def const(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    const = np.empty(429, float)
    const[0:26] = pos[:,0] - r
    const[26:52] = 1 - pos[:,0] - r
    const[52:78] = pos[:,1] - r
    const[78:104] = 1 - pos[:,1] - r
    const['pair'] = pair_constraints(pos, r)
    return const

def penalty(v):
    const = const(v)
    return np.sum(np.clip(const, 0, np.inf)**2) * 1e6

def objective(v):
    return -np.sum(v[52:]) + penalty(v)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)

    de_bounds = [(0, 1)] * (2 * n) + [(0.01, 0.5)] * n  # reduced upper bound for radii to avoid self-overlap
    init_de = np.random.rand(2 * n + n)
    init_de[52:] = np.full(n, 0.05)

    de_res = differential_evolution(objective, de_bounds, init=init_de, popsize=15, maxiter=500, bounds=bounds)
    if not de_res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    de_sol = de_res.x
    slsqp_res = minimize(obj, de_sol, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': const})
    if slsqp_res.success:
        centers = slsqp_res.x[:52].reshape(n, 2)
        radii = slsqp_res.x[52:]
        return centers, radii, -slsqp_res.fun
    else:
        best_pos = de_sol[:52].reshape(n, 2)
        best_r = de_sol[52:]
        return best_pos, best_r, objective(de_sol)
