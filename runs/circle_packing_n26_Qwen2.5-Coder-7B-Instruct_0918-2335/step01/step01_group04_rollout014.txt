import numpy as np
from scipy.optimize import linprog, minimize

def objective(x):
    return -np.sum(x[52:])

def constraints(x):
    pos = x[:52].reshape(26, 2)
    r = x[52:]
    cons = []

    # Bound constraints
    for i in range(26):
        cons.append(pos[i, 0] - r[i])
        cons.append(1 - pos[i, 0] - r[i])
        cons.append(pos[i, 1] - r[i])
        cons.append(1 - pos[i, 1] - r[i])

    # Pairwise constraints
    for i in range(26):
        for j in range(i + 1, 26):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            cons.append(dist - (r[i] + r[j]))
    
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    # Initial guess
    initial_centers = np.random.rand(n, 2) * box_size
    initial_radii = np.full(n, 0.02)
    initial_guess = np.concatenate((initial_centers.flatten(), initial_radii))
    
    # Constraints and bounds
    bounds = [(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n
    
    # Solve the linear program for radii given initial centers
    lp_res = linprog(
        c=np.zeros(52 + n),
        A_ub=[-1] * n + [-1] * n + [-1] * n + [-1] * n + [-1] * (26 * 25),
        b_ub=np.minimum(initial_centers[:, 0], 1 - initial_centers[:, 0]) +
             np.minimum(initial_centers[:, 1], 1 - initial_centers[:, 1]) +
             np.ones(n**2 - n),
        A_eq=None,
        b_eq=None,
        x0=initial_guess,
        method='highs',
        bounds=bounds,
        callback=lambda xk: lp_res Terminates early if all constraints are satisfied
    )
    
    if not lp_res.success:
        raise ValueError("Linprog did not converge correctly.")
    
    initial_radii = 0.5 * lp_res.x[len(initial_centers * 2):]
    
    # Boundary push-away step
    alpha = 0.01
    for _ in range(50):
        for i in range(26):
            r_i = initial_radii[i]
            x_i, y_i = initial_centers[i]
            active_constraint = None
            if x_i == r_i:
                active_constraint = 'left'
            elif x_i == 1 - r_i:
                active_constraint = 'right'
            elif y_i == r_i:
                active_constraint = 'bottom'
            elif y_i == 1 - r_i:
                active_constraint = 'top'
            else:
                for j in range(26):
                    if i != j:
                        r_j = initial_radii[j]
                        x_j, y_j = initial_centers[j]
                        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                        if np.isclose(dist, r_i + r_j, atol=1e-7):
                            active_constraint = f'neighbor:{j}'
                            break
            dx, dy = 0, 0
            if active_constraint == 'left':
                dx = 1
            elif active_constraint == 'right':
                dx = -1
            elif active_constraint == 'bottom':
                dy = 1
            elif active_constraint == 'top':
                dy = -1
            elif 'neighbor' in active_constraint:
                j = int(active_constraint.split(':')[-1])
                dx = (x_j - x_i) / dist
                dy = (y_j - y_i) / dist
            initial_centers[i] += alpha * np.array([dx, dy])
            initial_centers[i] = np.clip(initial_centers[i], 0, box_size)
        
        # Recompute radii
        lp_res = linprog(
            c=np.zeros(52 + n),
            A_ub=[-1] * n + [-1] * n + [-1] * n + [-1] * n + [-1] * (26 * 25),
            b_ub=np.minimum(initial_centers[:, 0], 1 - initial_centers[:, 0]) +
                 np.minimum(initial_centers[:, 1], 1 - initial_centers[:, 1]) +
                 np.ones(n**2 - n),
            x0=initial_guess,
            method='highs',
            bounds=bounds,
            callback=lambda xk: lp_res Terminates early if all constraints are satisfied
        )
        
        if not lp_res.success:
            raise ValueError("Linprog did not converge correctly.")
        
        initial_radii = 0.5 * lp_res.x[len(initial_centers * 2):]
        if np.sum(initial_radii) <= lp_res.fun * (1 - 1e-7):
            break
        
        alpha *= 0.5
    
    final_centers = initial_centers
    final_radii = initial_radii
    
    # Final polish with SLSQP
    optimize_res = minimize(
        fun=objective,
        x0=np.concatenate((final_centers.flatten(), final_radii)),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not optimize_res.success:
        raise ValueError("SLSQP did not converge correctly.")
    
    final_centers = optimize_res.x[:52].reshape(26, 2)
    final_radii = optimize_res.x[52:]
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
