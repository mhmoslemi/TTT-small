import numpy as np
from scipy.optimize import minimize, linprog

def optimise_radii(centers):
    n = centers.shape[0]
    bounds = [(0, centers[i, 0]), (0, centers[i, 1]), (0, 1 - centers[i, 0]), (0, 1 - centers[i, 1])] for i in range(n)]
    bounds = [(min(bounds[i][0], bounds[i][2]), min(bounds[i][1], bounds[i][3])) for i in range(n)]
    constraints = [{'type': 'ineq', 'fun': lambda x, i=i, centers=centers: centers[i, 0] + x[i] - x[n + i]},
                   {'type': 'ineq', 'fun': lambda x, i=i, centers=centers: centers[i, 0] - 1 + x[n + i]},
                   {'type': 'ineq', 'fun': lambda x, i=i, centers=centers: centers[i, 1] + x[i] - x[2 * n + i]},
                   {'type': 'ineq', 'fun': lambda x, i=i, centers=centers: centers[i, 1] - 1 + x[2 * n + i]}]
    for i in range(n):
        for j in range(i + 1, n):
            radius_sum = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j, radius_sum=radius_sum: radius_sum - (x[n + i] + x[2 * n + i])})
    try:
        res = linprog(-np.ones(2 * n), bounds=bounds, A_ub=[c['fun'] for c in constraints], b_ub=[-c['fun'] for c in constraints], method='highs')
        if res.success and np.all(np.array([c['fun'] for c in constraints]) <= -1e-10):
            return res.x[:n]
        else:
            return None
    except:
        return None

def random_step(delta_max):
    return np.random.uniform(-delta_max, delta_max, 2)

def slsqp_refine(centers, radii):
    bounds = [(0, 1), (0, 1)]*26
    bounds.extend([(0, 0.5)]*26)
    constraints = [{'type': 'ineq', 'fun': lambda x: x[52:52 + 26] - np.array([x[i, 0] - 0.5 for i, (x, y) in enumerate(zip(x.reshape(-1, 2)[:26], x.reshape(-1, 2)[26:]))])}],
                   {'type': 'ineq', 'fun': lambda x: x[52:52 + 26] - np.array([1 - (x[i, 0] - 0.5) for i, (x, y) in enumerate(zip(x.reshape(-1, 2)[:26], x.reshape(-1, 2)[26:]))])}],
                   {'type': 'ineq', 'fun': lambda x: x[52:52 + 26] - np.array([x[i, 1] - 0.5 for i, (x, y) in enumerate(zip(x.reshape(-1, 2)[:26], x.reshape(-1, 2)[26:]))])}],
                   {'type': 'ineq', 'fun': lambda x: x[52:52 + 26] - np.array([1 - (x[i, 1] - 0.5) for i, (x, y) in enumerate(zip(x.reshape(-1, 2)[:26], x.reshape(-1, 2)[26:]))])}]
                   ]
    for i in range(26):
        for j in range(26):
            if i != j:
                dist = np.sqrt((x.reshape(-1, 2)[i, 0] - x.reshape(-1, 2)[j, 0])**2 + (x.reshape(-1, 2)[i, 1] - x.reshape(-1, 2)[j, 1])**2)
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j, dist=dist: dist - (x[52 + i] + x[52 + j])})
    result = minimize(lambda x: -np.sum(x[52:]), x0=np.concatenate([centers.ravel(), radii]), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    if result.success and np.all(np.array([c['fun'] for c in constraints]) <= -1e-10):
        return result.x[:26].reshape(-1, 2), result.x[26:]
    else:
        return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, None)] * n  # 2 * x, y coordinates + r for each circle
    x0 = np.random.rand(3 * n)  # Initial guess for centers and radii

    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)  # (x,y) pairs
        r = v[52:]  # radii

        # Boundary terms - 4 * n
        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        # Pairwise terms - n choose 2
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    def initial_state(n):
        centers = np.zeros((n, 2))
        radius = 0.1
        for i in range(5):
            for j in range(i, 5):
                centers[5 * i + j, :] = [(i, j) + radius for i in [1, 3]]
                radius += 0.1
        centers[25, :] = (0.5, 0.5)
        radii = optimise_radii(centers)
        if radii is None:
            centers, radii = initial_state(n)
        return centers, radii

    centers, radii = initial_state(n)
    best_centers = centers
    best_radii = radii
    best_sum = np.sum(radii)
    T0 = 1.0
    alpha = 0.995
    NITER = 50000

    for iter in range(1, NITER + 1):
        i = np.random.randint(0, n)
        deltax, deltay = random_step(0.05)
        new_centres = centers.copy()
        new_centres[i, 0] += deltax
        new_centres[i, 1] += deltay
        if new_centres[i, 0] < 0 or new_centres[i, 0] > 1 or new_centres[i, 1] < 0 or new_centres[i, 1] > 1:
            continue

        temp_radii = optimise_radii(new_centres)
        if temp_radii is None:
            continue

        new_sum = np.sum(temp_radii)
        if new_sum > best_sum:
            centers, radii, best_sum = new_centres, temp_radii, new_sum
            best_centers, best_radii = new_centres, temp_radii
        else:
            T = T0 * alpha**iter
            if np.random.rand() < np.exp((new_sum - best_sum) / T):
                centers, radii = new_centres, temp_radii

    centers, radii = slsqp_refine(centers, radii)
    return centers, radii, np.sum(radii)
