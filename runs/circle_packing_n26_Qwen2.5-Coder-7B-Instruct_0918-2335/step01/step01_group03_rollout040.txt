import numpy as np
from scipy.optimize import minimize

# Pre-defined families of circle centers
grid_centers = np.array([
    [i / 4.0 + 0.125, j / 4.0 + 0.125] for i in range(5) for j in range(5)
])
grid_centers = np.vstack((grid_centers, [0, 0]))

hex_centers = np.array([
    [0.25 + (i%2)*0.5, (i//2)*np.sqrt(3)/3 + 0.3333] for i in range(6)
])
hex_centers = np.vstack((hex_centers, [1, 0], [0.5, 0.5], [0, 1], [1, 1]))

hybrid_centers = np.array([
    [0.125, 0.125], [0.375, 0.125], [0.625, 0.125], [0.875, 0.125],
    [0.125, 0.375], [0.375, 0.375], [0.625, 0.375], [0.875, 0.375], 
    [0.125, 0.625], [0.375, 0.625], [0.625, 0.625], [0.875, 0.625], 
    [0.125, 0.875], [0.375, 0.875], [0.625, 0.875], [0.875, 0.875],
    [0.25, 0.25], [0.45, 0.25], [0.65, 0.25], [0.85, 0.25],
    [0.25, 0.65], [0.45, 0.65], [0.65, 0.65], [0.85, 0.65]
])

random_centers = 0.5 + np.random.rand(26, 2) / 2

# Prioritized radii scaling
scaling_order = [0, 25] + sorted(range(1, 25), key=lambda i: (grid_centers[i, 1], grid_centers[i, 0]))

def speed_sampling(centers):
    n = centers.shape[0]
    speeds = np.empty(n)
    for i in range(n):
        dx = np.abs(centers[i, 0] - 0)
        dy = np.abs(centers[i, 1] - 0)
        d = min(dx, dy, 1 - dx, 1 - dy)
        speeds[i] = 1 + 2 / (d + 1e-6)
    return np.random.lognormal(mean=0., sigma=0.4, size=n) * speeds

def simulate_growth(centers, speeds):
    n = centers.shape[0]
    radii = np.zeros(n)
    active = set(range(n))
    t_stop = [None] * n
    d = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    
    while len(active) > 0:
        min_t = float('inf')
        stop_set = set()
        
        for i in active:
            for j in active:
                if j in stop_set:
                    continue
                d_ij = d[i, j]
                t_ij = d_ij / (speeds[i] + speeds[j])
                if t_ij < min_t:
                    min_t = t_ij
                    stop_set = {i}
                elif t_ij == min_t:
                    stop_set.add(j)
            
            t_edge = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]) / speeds[i]
            if t_edge <= min_t:
                min_t = t_edge
                stop_set = {i}
        
        t_stop = [min_t if i in stop_set else t_stop[i] for i in range(n)]
        active -= stop_set
        radii[stop_set] = speeds[stop_set] * t_stop[stop_set]
    
    return centers, radii

                {'type': 'ineq', 'fun': lambda x: np.array([x[n + i] - 0.5 for i in range(n)] + [1 - x[n + i] for i in range(n)])}]
        bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
        res = minimize(lambda x: -np.sum(x[2 * n:]), x0, method='SLSQP', bounds=bounds, constraints=cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_R = -np.inf
    best_centers = best_radii = None
    for centers in [grid_centers, hex_centers, hybrid_centers, random_centers]:
        for _ in range(5):  # 5 random samples per geometry
            speeds = speed_sampling(centers)
            centers_out, radii_out = simulate_growth(centers, speeds)
            R = validate_and_optimize(centers_out, radii_out)
            if R > best_R:
                best_R = R
                best_centers, best_radii = centers_out, radii_out
    
    return best_centers, best_radii, best_R
