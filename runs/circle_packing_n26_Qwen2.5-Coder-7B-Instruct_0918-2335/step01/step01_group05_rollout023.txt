import numpy as np
from scipy.optimize import linprog, minimize

np.random.seed(42)

def solve_radii(centers):
    n = centers.shape[0]
    M = n * (n - 1) // 2
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    A_ub = np.zeros((4 * n + M, n))
    b_ub = np.zeros(4 * n + M)
    
    for i in range(n):
        A_ub[i, i] = -1
        A_ub[n + i, i] = 1
        A_ub[2 * n + i, i] = -1
        A_ub[3 * n + i, i] = 1
    
    for i, j in pairs:
        idx = 4 * n + pairs.index((i, j))
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
    
    c = -np.ones(n)
    outcome = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=((0, 0.5),) * n, method='highs')
    if not outcome.success:
        return -np.inf, np.zeros(n)
    return -outcome.fun, -outcome.x

def evaluate_fitness(individual):
    centers = individual.reshape(26, 2)
    centers = np.clip(centers, 0, 1)
    sum_r, radii = solve_radii(centers)
    return sum_r

def crossover(parent1, parent2):
    i1, i2 = np.random.choice(range(52), 2, replace=False)
    child1 = np.concatenate((parent1[:i1], parent2[i1:i2], parent1[i2:]))
    child2 = np.concatenate((parent2[:i1], parent1[i1:i2], parent2[i2:]))
    return child1, child2

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, size=individual.size)
    return np.clip(individual, 0, 1)

def clamp_to_square(individual):
    return np.clip(individual, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n
    initial_population = np.random.rand(48, n * 2)
    sigma = 0.05
    crossover_prob = 0.9
    mutation_rate = 0.8
    generations = 180
    elitism = 5
    tournament_size = 5

    for _ in range(generations):
        fitness_vals = np.array([evaluate_fitness(ind) for ind in initial_population])
        elite_indices = np.argsort(fitness_vals)[-elitism:]
        elite_population = initial_population[elite_indices]
        new_population = initial_population[elite_indices]
        
        for _ in range(48 - elitism):
            parent1, parent2 = np.random.choice(initial_population, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child1 = mutate(child1, sigma)
            if np.random.rand() < mutation_rate:
                child2 = mutate(child2, sigma)
            new_population = np.vstack((new_population, child1))
            new_population = np.vstack((new_population, child2))
        
        initial_population = new_population
        sigma *= 0.99
    
    best_idx, _ = np.unravel_index[np.argmax(fitness_vals), :]
    best_centers = initial_population[best_idx].reshape(26, 2)
    best_radii, _ = solve_radii(best_centers)
    centers, _ = solve_radii(best_centers)
    radii = radii[:26]
    radii = np.where(radii < 0, 0, radii)
    radii -= np.min(radii)
    radii /= np.max(radii)
    sum_radii = np.sum(radii)
    centers = np.clip(centers, 0, 1)
    return centers, radii, sum_radii
