import numpy as np
from scipy.optimize import minimize

def compute_optimal_radii(pos):
    r = pos - np.roll(pos, 1)
    dists = np.sqrt(np.sum(r**2, axis=1))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    A_ub = np.where(mask, dists - (r[:, 0] + np.roll(r[:, 0], -1)), 0)
    b_ub = np.where(mask, 0.01, np.inf)
    r = np.linalg.lstsq(A_ub.T, b_ub, rcond=None)[0]
    return r

def objective(pos):
    r = compute_optimal_radii(pos)
    return -np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0,1)] * 52
    best_centres = None
    best_radii = None
    best_sum = 0

    for _ in range(20):
        init_centres = np.random.rand(52)
        res = minimize(fun=objective, x0=init_centres, method='SLSQP',
                       bounds=bounds, options={'maxiter':2000, 'ftol':1e-9, 'disp':False})
        
        if res.success:
            r = compute_optimal_radii(res.x)
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_centres = res.x.reshape(26, 2)
                best_radii = r
                best_sum = cur_sum

    return best_centres, best_radii, best_sum
