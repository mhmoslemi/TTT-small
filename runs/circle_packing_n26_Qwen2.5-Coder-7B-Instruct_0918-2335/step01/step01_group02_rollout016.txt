import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_obj(v, pos):
    r = v
    n = len(r)
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        A_ub.append([r[i], -1, -1, 1, -1, -1])
        b_ub.append(pos[i, 0])
        A_ub.append([r[i], 1, 1, -1, -1, -1])
        b_ub.append(1 - pos[i, 0])
        A_ub.append([r[i], -1, -1, -1, 1, 1])
        b_ub.append(pos[i, 1])
        A_ub.append([r[i], 1, 1, -1, -1, 1])
        b_ub.append(1 - pos[i, 1])
    
    i = 0
    j = i + 1
    while i < j and j < n:
        A_ub.append([r[i], r[j], 0, 0, 0, 0])
        b_ub.append(np.sqrt((pos[i, 0] - pos[j, 0])**2 + (pos[i, 1] - pos[j, 1])**2) - r[i] - r[j])
        i += 1
        j += 1
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    return -res.fun if res.success else res.fun

def fitness(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    return lp_obj(r, pos)

def run_packing():
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    
    result = differential_evolution(fitness, bounds, popsize=50, maxiter=200, seed=42)
    best_sum = -result.fun if result.success else result.fun
    
    pos = result.x[:52].reshape(26, 2)
    r = result.x[52:]
    cons = [
        {'type': 'ineq', 'fun': lambda v: v[52+i] for i in range(26)},
        {'type': 'ineq', 'fun': lambda v: 1 - v[52+i] for i in range(26)},
        {'type': 'ineq', 'fun': lambda v: 1 - v[52+i] for i in range(26)},
        {'type': 'ineq', 'fun': lambda v: v[52+i] for i in range(26)},
        {'type': 'ineq', 'fun': lambda v, x=i, y=j, r=r: np.sqrt((pos[x, 0] - pos[y, 0])**2 + (pos[x, 1] - pos[y, 1])**2) - r[x] - r[y] for i in range(26) for j in range(i + 1, 26)}
    ]
    
    result = minimize(lp_obj, result.x[52:], args=(pos), method='SLSQP', bounds=bounds[52:], constraints=cons, options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
    
    return (pos, np.clip(result.x, 0, None), result.fun if result.success else -result.fun)

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
