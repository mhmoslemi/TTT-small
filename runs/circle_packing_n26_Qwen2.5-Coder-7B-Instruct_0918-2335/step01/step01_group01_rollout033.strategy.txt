import numpy as np
from scipy.optimize import linprog, minimize

def compute_optimal_radii(positions):
    n=26
    pos=positions.reshape(n,2)
    # pair-wise distances
    diff=pos[:,None,:]-pos[None,:,:]
    dists=np.sqrt((diff**2).sum(axis=2))
    # Build constraints
    pairs = np.triu_indices(n,1)
    m_pairs=len(pairs[0])
    num_constraints= m_pairs + 4*n
    A=np.empty((num_constraints,n))
    b=np.empty(num_constraints)
    # pair constraints
    for k,(i,j) in enumerate(zip(pairs[0],pairs[1])):
        A[k,i]=1
        A[k,j]=1
        b[k]=dists[i,j]
    # boundary constraints
    for i in range(n):
        idx = m_pairs + 4*i
        # x
        A[idx,i]=1
        b[idx]=pos[i,0]
        idx+=1
        # 1-x
        A[idx,i]=1
        b[idx]=1-pos[i,0]
        idx+=1
        # y
        A[idx,i]=1
        b[idx]=pos[i,1]
        idx+=1
        # 1-y
        A[idx,i]=1
        b[idx]=1-pos[i,1]
    # objective
    c=-np.ones(n)
    bounds=[(0, None)]*n
    res=linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def cost_fun(pos_flat):
    radii=compute_optimal_radii(pos_flat)
    if radii is None:
        return 1e6
    return -np.sum(radii)
