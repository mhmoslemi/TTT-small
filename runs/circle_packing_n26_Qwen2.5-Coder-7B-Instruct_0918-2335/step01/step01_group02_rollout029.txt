import numpy as np
from scipy.optimize import linprog, minimize

# Predefined contact graphs: 5x5 grid, hexagonal patches, row-shifted grids, and random perturbations
graphs = {
    "5x5_grid": [
        (0, 1), (1, 1), (2, 1), (3, 1), (4, 1),
        (0, 2), (1, 2), (2, 2), (3, 2), (4, 2),
        (0, 3), (1, 3), (2, 3), (3, 3), (4, 3),
        (0, 4), (1, 4), (2, 4), (3, 4), (4, 4)
    ],
    "hex_patch": [
        (0, 0), (0, 1), (1, 0), (1, 1),
        (0, 2), (1, 2)
    ],
    "row_shift_5x5": [
        (0, 1), (1, 1), (2, 1), (3, 1),
        (0, 2), (1, 2), (2, 2), (3, 2),
        (0, 3), (1, 3), (2, 3), (3, 3), (4, 3),
        (0, 4), (1, 4), (2, 4), (3, 4), (4, 4)
    ],
    "random_perturbation": [
        (0, 0), (0, 0.5), (0, 1), (0, 1.5), (0, 2),
        (1, 0), (1, 0.5), (1, 1), (1, 1.5), (1, 2),
        (2, 0), (2, 0.5), (2, 1), (2, 1.5), (2, 2),
        (3, 0), (3, 0.5), (3, 1), (3, 1.5)
    ]
}

# Compute the linear inequalities for a given graph
def get_inequalities(graph, radii.bounds):
    n = len(graph)
    A_ub = np.zeros((2 * n, n))
    b_ub = np.zeros(2 * n)
    
    for i, (u, v) in enumerate(graph):
        diff = radii.bounds[u:u+2] + radii.bounds[v:v+2]
        A_ub[i] = np.array([1, 0, 0, 0]) * diff[0]
        A_ub[i + n] = np.array([0, 1, 0, 0]) * diff[1]
        b_ub[i] = diff[0]
        b_ub[i + n] = diff[1]
    
    # Check wall constraints
    for i, (u, v) in enumerate(graph):
        if u == 0:
            A_ub[2 * n + 2 * i] = np.array([1, 0, 0, 0])
            b_ub[2 * n + 2 * i] = radii.bounds[0][0] + radii.bounds[0][1]
        if u == n - 1:
            A_ub[2 * n + 2 * i + 1] = np.array([1, 0, 0, 0])
            b_ub[2 * n + 2 * i + 1] = 1 - radii.bounds[n - 1][0] - radii.bounds[n - 1][1]
        if v == 0:
            A_ub[2 * n + 2 * i + 2] = np.array([0, 1, 0, 0])
            b_ub[2 * n + 2 * i + 2] = radii.bounds[0][1]
        if v == n - 1:
            A_ub[2 * n + 2 * i + 3] = np.array([0, 1, 0, 0])
            b_ub[2 * n + 2 * i + 3] = 1 - radii.bounds[n - 1][1]
    
    return A_ub, b_ub

# Solve LP for a given contact graph
def solve_lp(graph):
    n = len(graph)
    A = np.eye(n)
    b = np.zeros(n)
    bounds = [(0, 0.5)] * n
    A_ub, b_ub = get_inequalities(graph, bounds)
    
    result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        r = result.x
        return np.sum(r), r
    else:
        return 0, None

# Find the best configuration
def find_best_configuration(graphs):
    best_sum = 0
    best_r = None
    for name, graph in graphs.items():
        sum_radius, r = solve_lp(graph)
        if sum_radius > best_sum:
            best_sum = sum_radius
            best_r = r
    
    return best_r

# Perform final light refinement using SLSQP
def refine_solution(radii):
    n = len(radii)
    A = np.eye(n * 2)
    b = np.zeros(n * 2)
    for i in range(n):
        b[i * 2] = 0
        b[i * 2 + 1] = 0.5
    bounds = [(0, 1)] * (2 * n)
    constraints = {'type': 'ineq', 'fun': lambda x:.get_inequalities(graphs["5x5_grid"], bounds)[1] - x[n:2*n] * np.array([0.5 for _ in range(n)])}
    
    result = minimize(lambda x: -np.sum(x[n:2*n]), x0=np.concatenate((np.ones(n), radii)), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    if result.success:
        return np.sum(result.x[n:2*n]), result.x[n:2*n]
    else:
        return np.sum(radii), radii

centers = np.random.rand(52).reshape(26, 2)

sum_radii, radii = refine_solution(radii)
