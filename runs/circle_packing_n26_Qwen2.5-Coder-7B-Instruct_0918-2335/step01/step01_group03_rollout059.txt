import numpy as np
from scipy.optimize import linprog, minimize
from random import randint, uniform

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def optimise_radii(centers):
    n = centers.shape[0]
    c = np.zeros(n * 2)
    A = []
    b = []
    
    for i in range(n):
        A.append([0, i * 2])
        b.append(centers[i, 0])
        
        A.append([0, i * 2 + 1])
        b.append(1 - centers[i, 0])
        
        A.append([1, i * 2])
        b.append(centers[i, 1])
        
        A.append([1, i * 2 + 1])
        b.append(1 - centers[i, 1])
        
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))

            A.append([-1, i * 2 + 1, 0, j * 2 + 1])
            b.append(dist - (R[i] + R[j]))
            A.append([-1, 0, i * 2, j * 2])
            b.append(dist - (R[i] + R[j]))
            A.append([-1, 0, i * 2 + 1, j * 2 + 1])
            b.append(dist - (R[i] + R[j]))
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(2 * n)], method='highs')
    if not res.success:
        return None
    return res.x

def random_step(delta_max):
    return uniform(-delta_max, delta_max), uniform(-delta_max, delta_max)

def slsqp_refine(centers, radii):
    n = centers.shape[0]
    x0 = np.concatenate((centers.flatten(), radii))
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 2]}),
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i * 2]}),
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 2 + 1]}),
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i * 3]}),
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i] - centers[j]) ** 2).sum()
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist - (r[i] + r[j])})
    res = minimize(fun=lambda x: -np.sum(x[2 * n:]), x0=x0, bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 2000, 'ftol': 1e-9})
    if res.success:
        return res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
    else:
        return centers, radii

def run_packing():
    n = 26
    delta_max = 0.05
    T0 = 1.0
    alpha = 0.995
    NITER = 50000

    INITIAL_START = np.array([(x, y) for y in np.linspace(0.2, 0.8, 5) for x in np.linspace(0.2, 0.8, 5)])
    INITIAL_START = np.vstack([INITIAL_START, ((0.5, 0.5))])
    current_centres = INITIAL_START.copy()
    current_radii = optimise_radii(current_centres)
    best_sum = np.sum(current_radii)
    best_centres = current_centres.copy()
    best_radii = current_radii.copy()

    for iter in range(1, NITER):
        i = randint(0, n-1)
        δx, δy = random_step(delta_max)
        new_centres = current_centres.copy()
        new_centres[i] += (δx, δy)
        new_centres = np.clip(new_centres, 0, 1)

        temp_radii = optimise_radii(new_centres)
        if temp_radii is None:
            continue

        new_sum = np.sum(temp_radii)
        if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / (T0 * alpha ** iter)):
            current_centres = new_centres.copy()
            current_radii = temp_radii.copy()
            best_sum = new_sum
            best_centres = current_centres.copy()
            best_radii = current_radii.copy()

    best_centres, best_radii = slsqp_refine(best_centres, best_radii)
    if not validate_packing(best_centres, best_radii)[0]:
        return INITIAL_START, 0.1 * np.ones(26), 2.606
    return best_centres, best_radii, np.sum(best_radii)
