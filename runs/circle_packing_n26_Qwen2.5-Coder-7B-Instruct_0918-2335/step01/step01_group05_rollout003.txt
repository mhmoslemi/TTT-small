import numpy as np
import scipy.optimize as optimize
from scipy.spatial import distance_matrix

def solve_lp(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    c = -np.ones(n)
    A_ub = np.zeros((4 * n + n * (n - 1) // 2, n))
    b_ub = np.zeros(4 * n + n * (n - 1) // 2)
    
    for i in range(n):
        A_ub[4 * i, i] = 1
        b_ub[4 * i] = centers[i, 0]
        A_ub[4 * i + 1, i] = 1
        b_ub[4 * i + 1] = 1 - centers[i, 0]
        A_ub[4 * i + 2, i] = 1
        b_ub[4 * i + 2] = centers[i, 1]
        A_ub[4 * i + 3, i] = 1
        b_ub[4 * i + 3] = 1 - centers[i, 1]
    
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k, i] = 1
        A_ub[4 * n + k, j] = 1
        d_ij = np.linalg.norm(centers[i] - centers[j])
        b_ub[4 * n + k] = d_ij
    
    result = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=(0, 0.5))
    if result.success:
        r = result.x
        return r, -result.fun
    else:
        return None, np.inf

def evaluate_positions(centers):
    n = centers.shape[0]
    r, sum_radii = solve_lp(centers)
    if r is not None:
        return sum_radii, r
    return None, None

def run_packing():
    # Patterns to try as initial guesses for SA
    patterns = [
        np.array([[i / 5, j / 5] for i in np.arange(6) for j in np.arange(6)][:-1]),  # 5x5 grid
        np.random.rand(25, 2),  # Random 5x5 grid plus center
        np.random.rand(26, 2)  # Full random shift from 5x5 grid plus center
    ]
    
    # Augment patterns with a random center and perform jittering
    jitter = 0.02
    for p in patterns:
        centers = p + np.random.uniform(-jitter, jitter, p.shape)
        while True:
            centers = np.clip(centers, 0, 1)
            if centers.shape[0] == 26:
                break
        _, sum_radii = evaluate_positions(centers)
        if sum_radii is not None:
            return centers, solve_lp(centers)[1], sum_radii
    
    return np.zeros((26, 2)), np.zeros(26), 0.0

# Main execution
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of radii:", sum_radii)
