import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers):
    n = len(centers)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
    dists += dists.T
    return dists

def solve_radii_lp(centers):
    n = len(centers)
    A_ineq = np.zeros((2 * n + n * (n - 1) // 2, 2 * n))
    b_ineq = np.zeros(2 * n + n * (n - 1) // 2)
    
    # Boundary constraints
    for i in range(n):
        A_ineq[i, 2 * i] = 1
        A_ineq[i, 2 * i + 1] = 0
        b_ineq[i] = centers[i, 0]
        
        A_ineq[n + i, 2 * i] = 0
        A_ineq[n + i, 2 * i + 1] = 1
        b_ineq[n + i] = centers[i, 1]
        
        A_ineq[2 * n + i, 2 * i] = 1
        A_ineq[2 * n + i, 2 * i + 1] = 0
        b_ineq[2 * n + i] = 1 - centers[i, 0]
        
        A_ineq[2 * n + n + i, 2 * i] = 0
        A_ineq[2 * n + n + i, 2 * i + 1] = 1
        b_ineq[2 * n + n + i] = 1 - centers[i, 1]
    
    # Pairwise constraints
    mask = np.triu(np.ones((n, n), bool), k=1)
    idxs = np.where(mask)[0] * 2 + np.where(mask)[1] * 2
    A_ineq[2 * n + n * (n - 1) // 2:, 2 * np.arange(n)[mask]][:, idxs] = 1
    b_ineq[2 * n + n * (n - 1) // 2:] = pairwise_dists(centers)[mask].flatten()
    
    c = -np.ones(2 * n)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=(0, None))
    return result.x[n:].reshape(-1)
