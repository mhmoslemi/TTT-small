import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    eps = 1e-12
    min_step = 1e-5

    centers = np.random.rand(n, 2)
    radii = np.zeros(n)
    dg = 0.02
    max_iter = 2000
    jitter_amp = 0.1 * dg

    for _ in range(max_iter):
        centers += jitter_amp * (np.random.rand(n, 2) - 0.5)
        centers = np.clip(centers, 0, 1)
        radii += dg

        while True:
            dists = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
            mask = dists < radii[:, None] + radii[None, :] - eps
            if mask.sum() == 0:
                break

            for i, j in zip(*np.where(mask)):
                if i < j:
                    d = dists[i, j]
                    sep = radii[i] + radii[j] - d + eps
                    disp = 0.5 * sep * (centers[i] - centers[j]) / d
                    centers[i] -= disp
                    centers[j] += disp
                    centers = np.clip(centers, 0, 1)

        if dg <= min_step:
            break

        dg *= 0.995

    for i in range(n):
        for j in range(i + 1, n):
            if (centers[i] - centers[j]).dot(centers[i] - centers[j]) < (radii[i] + radii[j] - eps) ** 2:
                return np.zeros((n, 2)), np.zeros(n), 0.0

    leta_radii = np.full(n, 1.0)
    A = np.empty((n*n, n), float)
    B = np.full(n*n, 1.0 - leta_radii)

    for i in range(n):
        A[i*n+i, :] = leta_radii
        B[i*n+i] = 1.0 - radii[i]

    for i, j in zip(*np.where(dists < eps)):
        if i < j:
            A[i*n+j] = 1.0 - radii[j]
            A[j*n+i] = 1.0 - radii[i]
            B[i*n+j] = 1.0 - radii[j] - radii[i]

    radii = np.linalg.solve(A, B - leta_radii)

    radii = np.clip(radii, 1e-9, 0.5)
    return centers, radii, np.sum(radii)
