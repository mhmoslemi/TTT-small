import numpy as np
from scipy.optimize import linprog, minimize

def lp_radii(centers):
    n = centers.shape[0]
    nlp = linprog(-np.ones(n), A_ub=np.concatenate((centers, 1-centers, centers[:, None] - centers[None, :], 1-(centers[:, None] - centers[None, :]) ** 2, np.ones((n, n)) - np.eye(n)), axis=1),
                  b_ub=np.concatenate((np.zeros(n), np.ones(n), np.zeros(n)), axis=0),
                  bounds=np.concatenate((np.zeros(n), [1]*n, np.zeros(n), np.ones(n)), axis=0).reshape(n, 2),
                  method='highs')
    if nlp.success:
        return nlp.x
    else:
        return np.ones(n) * 1e-9

def objective(centers):
    r = lp_radii(centers)
    return -np.sum(r)

def constraints(centers):
    n = centers.shape[0]
    r = lp_radii(centers)
    cons = np.empty(104, float)
    cons[0:26] = centers[:, 0] - r
    cons[26:52] = 1 - centers[:, 0] - r
    cons[52:78] = centers[:, 1] - r
    cons[78:104] = 1 - centers[:, 1] - r
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((n, n), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

bounds = [(0, 1)] * 52

def run_packing():
    T0 = 0.1
    T = T0
    M = 300
    K = 50
    current_positions = np.random.rand(52)
    current_positions = current_positions.reshape(26, 2)
    best_positions = current_positions.copy()
    best_sum = objective(best_positions)
    patience = K
    accept_count = 0
    
    for _ in range(M):
        for _ in range(K):
            proposal = current_positions.copy()
            proposal[proposal.argmax()] += 0.1 * np.random.randn() * T
            if proposal[proposal.argmax()][0] > 1 or proposal[proposal.argmax()][1] > 1:
                proposal[proposal.argmax()][0] = min(1, proposal[proposal.argmax()][0])
                proposal[proposal.argmax()][1] = min(1, proposal[proposal.argmax()][1])
            acceptance_prob = 1
            if objective(proposal) < objective(current_positions):
                acceptance_prob = np.exp((objective(current_positions) - objective(proposal)) / T)
            if acceptance_prob > np.random.rand():
                current_positions = proposal
                accept_count += 1
        T /= np.log(1 + M)
        current_sum = objective(current_positions)
        if current_sum > best_sum or (current_sum == best_sum and accept_count > K * patience):
            best_sum = current_sum
            best_positions = current_positions.copy()
            patience *= 2
        else:
            patience -= 1
        if patience == 0:
            patience = K
    refined_solution = minimize(fun=objective, x0=best_positions.flatten(), method='SLSQP', bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-9, 'disp': False'})
    return refined_solution.x.reshape(26, 2), lp_radii(refined_solution.x.reshape(26, 2)), -objective(refined_solution.x)

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
