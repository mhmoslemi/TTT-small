import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def lp_maximize_radii(centers):
        n = centers.shape[0]
        A = np.zeros((4 * n + n * (n - 1) // 2, n))
        b = np.zeros(4 * n + n * (n - 1) // 2)
        bounds = [(0, None) for _ in range(n)]
        
        for i in range(n):
            A[4 * i, i] = 1
            A[4 * i + 1, i] = 1
            A[4 * i + 2, i] = 1
            A[4 * i + 3, i] = 1
            b[4 * i] = centers[i, 0]
            b[4 * i + 1] = 1 - centers[i, 0]
            b[4 * i + 2] = centers[i, 1]
            b[4 * i + 3] = 1 - centers[i, 1]
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                A[4 * n + i * (n - 1) // 2 + j - i - 1, i] = 1
                A[4 * n + i * (n - 1) // 2 + j - i - 1, j] = 1
                b[4 * n + i * (n - 1) // 2 + j - i - 1] = dist
        
        result = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not result.success:
            raise ValueError("Failed to solve the linear program")
        return result.x

    def push_away_step(centers, old_radii):
        n = centers.shape[0]
        new_centers = np.copy(centers)
        for i in range(n):
            r_i = old_radii[i]
            if centers[i, 0] == r_i:
                new_centers[i, 0] += 0.01
            elif centers[i, 0] == 1 - r_i:
                new_centers[i, 0] -= 0.01
            elif centers[i, 1] == r_i:
                new_centers[i, 1] += 0.01
            elif centers[i, 1] == 1 - r_i:
                new_centers[i, 1] -= 0.01
            else:
                dists = cdist([centers[i]], centers, 'euclidean')
                max_d = dists[0, np.argmax(dists[0])]
                if max_d > 2 * r_i:
                    continue
                j = np.argmax(dists[0])
                u_ij = (centers[i] - centers[j]) / max_d
                new_centers[i] += 0.01 * u_ij
        return new_centers.clip(0, 1)

    def validate_and_optimize(centers):
        old_radii = lp_maximize_radii(centers)
        while True:
            new_centers = push_away_step(centers, old_radii)
            new_radii = lp_maximize_radii(new_centers)
            if np.sum(new_radii) > np.sum(old_radii) + 1e-7:
                centers, old_radii = new_centers, new_radii
            else:
                break
        return centers, old_radii

    def generate_initial_conditions_method_1():
        def generate_hexagon_ring(center, radius, n):
            theta = np.linspace(0, 2 * np.pi, n + 1)
            x = center[0] + (radius - 0.05) * np.cos(theta)
            y = center[1] + (radius - 0.05) * np.sin(theta)
            return np.vstack((x[:-1], y[:-1])).T
        
        hexagon_radius = 1 / np.sqrt(3)
        side = 1 / np.sqrt(3)
        n = int(1 / side) + 1
        hexagon_grid = [generate_hexagon_ring(np.array([0.5 * (i % (2 * n - 1)), 
                                                       0.5 * (i // (2 * n - 1) + (i % (2 * n - 1)) % 2)]), 
                                               hexagon_radius, n) 
                        for i in range((2 * n - 1) ** 2)]
        centers = np.vstack(hexagon_grid).ravel().reshape(-1, 2)
        centers = centers[:(26, )] + (np.random.rand(26, 2) - 0.5) * 0.05
        return centers

    centers = generate_initial_conditions_method_1()
    center, radii = validate_and_optimize(centers)

    return center.reshape(26, 2), radii, np.sum(radii)
