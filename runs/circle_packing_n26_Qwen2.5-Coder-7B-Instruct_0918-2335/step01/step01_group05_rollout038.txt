import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

np.random.seed(42)

def solve_radii(centres):
    n = centres.shape[0]
    A_ub = np.empty((4 * n + n * (n - 1) // 2, n))
    b_ub = np.empty(4 * n + n * (n - 1) // 2)
    c = -np.ones(n)
    
    for i in range(n):
        x1, y1 = centres[i]
        A_ub[4 * i] = x1 - np.eye(n)[i]
        A_ub[4 * i + 1] = 1 - x1 - np.eye(n)[i]
        A_ub[4 * i + 2] = y1 - np.eye(n)[i]
        A_ub[4 * i + 3] = 1 - y1 - np.eye(n)[i]
        b_ub[4 * i:4 * i + 4] = [x1, 1 - x1, y1, 1 - y1]
    
    pairs = np.triu_indices(n, k=1)
    dists = pdist(centres)
    for k, (i, j) in enumerate(pairs):
        A_ub[4 * n + k] = A_ub[i][:] + A_ub[j][:] - dists[k]
        b_ub[4 * n + k] = 0

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=((0, 0.5),) * n, method='highs')
    if not result.success:
        raise ValueError("Failed to solve linear programming problem.")
    
    return -result.fun, result.x

def evaluate_fitness(individual):
    centre_positions = individual.reshape(26, 2)
    centre_positions = np.clip(centre_positions, 0, 1)
    sum_r, radii = solve_radii(centre_positions)
    return sum_r

def crossover(parent1, parent2):
    i1, i2 = np.random.randint(1, 52, 2)
    child1 = np.where(np.random.rand(52) < 0.5, parent1, parent2)
    child2 = np.where(np.random.rand(52) < 0.5, parent1, parent2)
    return child1, child2

def mutate(individual, sigma):
    noise = np.random.normal(0, sigma, 52)
    individual += noise
    individual = np.clip(individual, 0, 1)
    return individual

def check_centres(centres):
    for i in range(26):
        x, y = centres[i]
        r = solve_radii(centres[[i]])[1][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False
    return True

def run_packing():
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    best_centers = np.zeros((26, 2))

        fitnesses = np.array([evaluate_fitness(individual) for individual in population])
        population_sorted_indices = np.argsort(fitnesses)[::-1]
        population = population[population_sorted_indices]

        new_population = []
        for _ in range(population_size - elitism):
            if np.random.rand() < crossover_prob:
                parent1 = population[np.random.randint(population_size)]
                parent2 = population[np.random.randint(population_size)]
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, initial_sigma * sigma_decay ** gen)
                child2 = mutate(child2, initial_sigma * sigma_decay ** gen)
                new_population.append(child1)
                new_population.append(child2)
            else:
                parent = population[np.random.randint(population_size)]
                child = mutate(parent, initial_sigma * sigma_decay ** gen)
                new_population.append(child)

        elitist_individs = population[:elitism]
        new_population.append(elitist_individs)

        population = np.array(new_population)

        # Periodic SLSQP refinement for the best individual found so far
        if gen % (generations // 10) == 0:
            angled_centers = elitist_individs[:, :52].reshape(26, 2)
            _, radii = solve_radii(angled_centers)
            best_sum_radii = solve_radii(angled_centers)[0]
            best_centers = angled_centers

            cons = []

                cons.append({'type': 'ineq', 'fun': lambda r=radii[i], x=angled_centers[i][0], y=angled_centers[i][1]: x - r})

                cons.append({'type': 'ineq', 'fun': lambda r=radii[i], x=angled_centers[i][0], y=angled_centers[i][1]: y - r})

                cons.append({'type': 'ineq', 'fun': lambda r_i=r_i, r_j=r_j, x_i=x_i, y_i=y_i, x_j=x_j, y_j=y_j: x_i - x_j - r_i - r_j})
                cons.append({'type': 'ineq', 'fun': lambda r_i=r_i, r_j=r_j, x_i=x_i, y_i=y_i, x_j=x_j, y_j=y_j: y_i - y_j - r_i - r_j})

        if best_sum_radii > best_sum:
            best_sum = best_sum_radii

    return best_centers, np.linalg.norm(solve_radii(best_centers)[1]), best_sum
