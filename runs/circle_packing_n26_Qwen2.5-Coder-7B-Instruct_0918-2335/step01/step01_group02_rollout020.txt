import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(pos[i, 0])
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1 - pos[i, 0])
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(pos[i, 1])
        A_ub.append([0]*i + [1] + [0]*(n-i-1))
        b_ub.append(1 - pos[i, 1])

    for i in range(n):
        for j in range(i+1, n):
            dist_sq = ((pos[i, 0] - pos[j, 0])**2 + (pos[i, 1] - pos[j, 1])**2)
            A_ub.append([0]*i + [1] + [0]*(j-i-1) + [1] + [0]*(n-j-1))
            b_ub.append(dist_sq - ((0.5)**2 + (0.5)**2))

    objective = -np.ones(n)
    res = linprog(c=objective, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n)
    r_opt = res.x
    sum_r = r_opt.sum()
    return r_opt, sum_r

def fitness(pos):
    return -solve_radii(pos)[1]

def crossover(parent1, parent2, rate):
    child = np.zeros_like(parent1)
    for i in range(child.shape[0]):
        if np.random.rand() < rate:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    noisy_individual = individual + np.random.normal(0, sigma, size=individual.shape)
    noisy_individual = np.clip(noisy_individual, 0, 1)
    flipped = np.random.rand(*individual.shape) < rate
    return np.where(flipped, noisy_individual, individual)

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(pop.shape[0], k, replace=False)
    selected_indices = indices[np.argmin(fitness_vals[indices])]
    return selected_indices

def ga_best_positions():
    np.random.seed(42)
    popsize = 80
    generations = 250
    elite_size = 8
    crossover_rate = 0.9
    mutation_rate = 0.2
    mutation_sigma = 0.03

    pop = np.random.rand(popsize, 26, 2)
    elite = []
    for gen in range(generations):
        fitness_vals = np.array([fitness(p) for p in pop])
        sorted_indices = np.argsort(fitness_vals)
        elite.append(pop[sorted_indices[0]])
        num_children = popsize - elite_size
        children = []
        for _ in range(num_children):
            parent1_idx = tournament(pop, fitness_vals, k=3)
            parent2_idx = tournament(pop, fitness_vals, k=3)
            child = crossover(pop[parent1_idx], pop[parent2_idx], crossover_rate)
            child = mutate(child, mutation_rate, mutation_sigma)
            children.append(child)
        pop = np.vstack([elite, children])

    best_idx = np.argmin(fitness_vals)
    return pop[best_idx]

def refine_solution(centers, radii):
    opt = minimize(fun=lambda x: -np.sum(x[26:]), x0=np.concatenate([centers.ravel(), radii]), 
                   method='SLSQP', constraints={
                       'type': 'ineq', 'fun': lambda x: np.array([
                           x[i + 26] - x[i],
                           1 - x[i + 26] - x[i],
                           x[i + 26] - x[i + 13],
                           1 - x[i + 26] - x[i + 13],
                           np.sqrt((x[i] - x[i + 13])**2 + (x[i + 12] - x[i + 25])**2) - (x[i + 26] + x[i + 26]),
                           np.sqrt((x[i] - x[i + 13])**2 + (x[i + 12] - x[i + 25])**2) - (x[i + 26] + x[i + 26])
                       ])
                   }, bounds=[(0, 1)] + [(0, 0.5)] * 26, options={
                       'maxiter': 4000,
                       'ftol': 1e-9
                   })
    return opt.x[:26].reshape(26, 2), opt.x[26:]
