import numpy as np
from scipy.optimize import linprog
import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    # Initial guess
    initial_centers = np.random.rand(n, 2)
    initial_guess = initial_centers.flatten()
    
    def get_constraints_and_objective(centers):
        pos = centers.reshape(n, 2)
        r = cp.Variable(n, nonneg=True)
        
        # Wall constraints
        left = pos[:, 0] - r
        right = box_size - pos[:, 0] - r
        bottom = pos[:, 1] - r
        top = box_size - pos[:, 1] - r
        
        # Pairwise constraints
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = cp.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_constraints = dists[mask] - (r[:, None] + r[None, :])[mask]
        
        # Objective
        objective = cp.Minimize(-cp.sum(r))
        
        # Constraints
        constraints = [
            left >= 0,
            right >= 0,
            bottom >= 0,
            top >= 0,
            pair_constraints >= 1e-12
        ]
        
        return objective, constraints
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    
    def compute_active_constraints(centers):
        pos = centers.reshape(n, 2)
        r = np.zeros(n)
        for i in range(n):
            left = pos[i, 0] - r[i]
            right = box_size - pos[i, 0] - r[i]
            bottom = pos[i, 1] - r[i]
            top = box_size - pos[i, 1] - r[i]
            dists = np.sqrt(np.sum((pos - pos[i]) ** 2, axis=1))
            neighbor_constraints = dists - (r + r[i])
            min_constraint = np.min([left, right, bottom, top, neighbor_constraints[neighbor_constraints > 1e-12]])
            if min_constraint == left:
                r[i] = min(box_size - pos[i, 0], r[i])
            elif min_constraint == right:
                r[i] = min(pos[i, 0], r[i])
            elif min_constraint == bottom:
                r[i] = min(box_size - pos[i, 1], r[i])
            elif min_constraint == top:
                r[i] = min(pos[i, 1], r[i])
            else:
                j = np.argmin(neighbor_constraints[neighbor_constraints > 1e-12])
                u_ij = (pos[i] - pos[j]) / dists[j]
                move_distance = min([pos[j, 0] - (pos[i, 0] + u_ij * 0.5), box_size - (pos[i, 0] + u_ij * 0.5), pos[j, 1] - (pos[i, 1] + u_ij * 0.5), box_size - (pos[i, 1] + u_ij * 0.5)])
                r[i] = move_distance / 2
                r[j] = move_distance / 2
        return r
    
    def update_centers centers, direction):
        return centers + direction * 0.01
    
    for _ in range(10):  # Multiple seeds for intra-global exploration
        centers = initial_centers * np.random.rand()  # Farthest-first seed
        for _ in range(200):  # MAX_ITER iterations
            before_radius_sum = np.sum(compute_active_constraints(centers))
            r = compute_active_constraints(centers)
            after_radius_sum = np.sum(r)
            if after_radius_sum > before_radius_sum:
                centers = update_centers(centers, r)
            else:
                break
    
    centers = initial_centers
    best_centers = centers
    best_sum = np.sum(compute_active_constraints(centers))
    
    return best_centers, compute_active_constraints(best_centers), best_sum
