import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

MAX_ITER = 200
COOLING_INTERVAL = 20
ESCAPE_INTERVAL = 50
EPSILON = 1e-7
PAUSE_EPOCHS = 5

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n = 26
    box_size = 1.0
    
    def lp_diagonal():
        return [
            {"type": "ineq", "fun": lambda x: x[0]},  # left wall
            {"type": "ineq", "fun": lambda x: box_size - x[0]},  # right wall
            {"type": "ineq", "fun": lambda x: x[1]},  # bottom wall
            {"type": "ineq", "fun": lambda x: box_size - x[1]}  # top wall
        ]
    
    def update_positions(C, r):
        dists = cdist(C, C, 'euclidean')
        improvements = np.zeros_like(r)
        for i in range(n):
            active = np.argmin(np.minimum(dists[i], box_size - r[i]))
            if active < n:
                if i != active:
                    u_ij = (C[i] - C[active]) / np.linalg.norm(C[i] - C[active])
                    C[i] += EPSILON * u_ij
                else:
                    C[i] += EPSILON * np.where(active == 0, (1, 0), (-1, 0))  # left/right or top/bottom case
                    C[i] = np.clip(C[i], 0, box_size)
            improvements[i] = np.minimum(box_size - C[i, 0], box_size - C[i, 1])
        return C, improvements
    
    def push_away(C, r):
        C, improvements = update_positions(C, r)
        best_sum = np.sum(r) + EPSILON
        for _ in range(MAX_ITER):
            if best_sum >= np.sum(r) + EPSILON:
                break
            r = solve_LP(C)
            current_sum = np.sum(r)
            if current_sum > best_sum:
                best_sum = current_sum
                r, _ = update_positions(C, r)
            else:
                C = update_positions(C, r)[0]
            if _ % COOLING_INTERVAL == 0:
                EPSILON *= 0.5
            if _ % ESCAPE_INTERVAL == 0:
                C += (np.random.randn(n, 2) / 2) * EPSILON
        return C, r, np.sum(r)
    
    def solve_LP(C):
        box_boundaries = cdist(C, np.array([(0, 0), (1, 1)]), 'euclidean')
        pairwise_boundaries = cdist(C, C, 'euclidean')
        constraints = []
        constraints += [{'type': 'ineq', 'fun': lambda x: x[i] - box_boundaries[i, 0] - r[i]} for i in range(n)]
        constraints += [{'type': 'ineq', 'fun': lambda x: box_size - x[i] - box_boundaries[i, 1] - r[i]} for i in range(n)]
        constraints += [{'type': 'ineq', 'fun': lambda x: x[i] + x[j] - pairwise_boundaries[i, j] for j in range(i + 1, n)} for i in range(n)]
        bounds = [(0, 0.5)] * n
        result = linprog(-r, A_ub=np.vstack([np.eye(n), np.ones((len(pairwise_boundaries) - n, n)), -np.ones((len(pairwise_boundaries) - n, n))]), 
                         b_ub=np.hstack([box_boundaries[:, 0] + r, box_size * np.ones(len(pairwise_boundaries) - n), box_size * np.ones(len(pairwise_boundaries) - n)]), 
                         bounds=bounds, method='highs')
        return result.x
    
    def initial_positions():
        return np.random.rand(n, 2) * box_size
    
    seeds = [
        np.array([
            [0.5, 0.5], [0.45, 0.45], [0.55, 0.55],
            [0.3, 0.3], [0.7, 0.7], [0.6, 0.4], [0.5, 0.6],
            [0.85, 0.85], [0.9, 0.9], [0.8, 0.8], [0.75, 0.75],
            [0.15, 0.15], [0.1, 0.2], [0.2, 0.1], [0.1, 0.1],
            [0.95, 0.2], [0.85, 0.3], [0.75, 0.4],
            [0.65, 0.5], [0.55, 0.6], [0.45, 0.7],
            [0.35, 0.8], [0.25, 0.9], [0.15, 0.90]
        ]),
        np.random.rand(n, 2) * box_size,
        np.array([
            [i * 1.0 / 5, j * 1.0 / 5] for i in range(4) for j in range(5)
        ])
    ]
    
    best_sum = 0
    best_C, best_r = None, None
    for seed in seeds:
        C = seed
        r, _, sum_r = push_away(C, np.full(n, 0.021))
        if sum_r > best_sum:
            best_C, best_r, best_sum = C, r, sum_r
    
    if validate_packing(best_C, best_r) != (True, "ok"):
        raise Exception("Validation failed")

    return best_C, best_r, best_sum
