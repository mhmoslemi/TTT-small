import numpy as np
from scipy.optimize import minimize, linprog
import random
import math

def lp_value_and_radii(centres):
    w_limits = np.minimum.reduce([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]])
    d_matrix = np.linalg.norm(centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=-1)
    
    n = len(centres)
    c = -np.ones(n)
    A_ub = np.zeros((n * (n - 1) // 2, 2 * n))
    b_ub = np.zeros(n * (n - 1) // 2)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, 2 * i] = 1
            A_ub[idx, 2 * j] = 1
            b_ub[idx] = d_matrix[i, j]
            idx += 1
    
    bounds = [(0, w_limit) for w_limit in w_limits]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and not np.isnan(res.fun).any():
        radii = res.x
        sum_radii = np.sum(radii)
        return radii, sum_radii
    else:
        return np.zeros(n), -np.inf

def initial_layout(n):
    k = 5
    elements = list(range(n // k)) + [5]
    elements.extend(range(n // k + 1, 2 * k))
    elements.extend(range(2 * k + 1, 3 * k))
    elements.extend(range(3 * k + 1, 4 * k))
    elements.extend(range(4 * k + 1, 5 * k))
    elements.extend(range(5 * k + 1, 26))
    
    np.random.shuffle(elements)
    
    layout = np.zeros((n, 2))
    
    x_center = 0.5
    y_center = 0.5
    spacing = 1 / (k * 2 + 1)
    
    for i, idx in enumerate(elements):
        x = x_center + (i % (k * 2 + 1) - k) * spacing
        y = y_center + math.floor(i / (k * 2 + 1)) * spacing
        layout[idx] = [x, y]
    
    layout += np.random.uniform(-0.05, 0.05, size=(n, 2))
    
    return layout

def run_packing():
    np.random.seed(42)
    n = 26
    num_restarts = 150
    num_iterations = 2000
    temperature_start = 0.1
    temperature_decay = 0.97
    
    best_centers = None
    best_sum = -np.inf
    
    for restart in range(num_restarts):
        centers = initial_layout(n)
        max_movement = 0.2
        temperature = temperature_start
        
        for iteration in range(num_iterations):
            idx = np.random.randint(n)
            dx = np.random.normal() * temperature * max_movement
            dy = np.random.normal() * temperature * max_movement
            new_centers = np.copy(centers)
            new_centers[idx, 0] += dx
            new_centers[idx, 1] += dy
            
            new_centers = np.clip(new_centers, 0, 1)
            
            new_radii, new_sum = lp_value_and_radii(new_centers)
            
            if new_sum > best_sum:
                best_centers = new_centers
                best_sum = new_sum
            
            if new_sum >= best_sum:
                centers = new_centers
            else:
                accept_prob = math.exp((new_sum - best_sum) / temperature)
                if np.random.rand() < accept_prob:
                    centers = new_centers
                    best_sum = new_sum
            
            temperature *= temperature_decay
    
    best_radii, _ = lp_value_and_radii(best_centers)
    
    return best_centers, best_radii, best_sum

# Execute the function
centers, radii, sum_radii = run_packing()

# Function to validate the packing
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

# Validate the packing
is_valid, message = validate_packing(centers, radii)
print(f"Is valid: {is_valid}, Message: {message}")
print(f"Radii sum: {sum_radii}")
