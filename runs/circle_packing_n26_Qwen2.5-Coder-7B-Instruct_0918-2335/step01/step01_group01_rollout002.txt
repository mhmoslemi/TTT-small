import numpy as np
from scipy.optimize import minimize, linprog

def compute_optimal_radii(pos):
    n = len(pos) // 2
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx, dy = pos[2 * i] - pos[2 * j], pos[2 * i + 1] - pos[2 * j + 1]
            dists[i, j] = np.sqrt(dx * dx + dy * dy)
    A_ub = np.concatenate([
        pos.reshape(n, 2)[:, 0] - np.ones(n),
        pos.reshape(n, 2)[:, 0] + np.ones(n) - 2,
        pos.reshape(n, 2)[:, 1] - np.ones(n),
        pos.reshape(n, 2)[:, 1] + np.ones(n) - 2
    ], axis=1)
    A_ub += (dists - np.triu(np.ones((n, n)), k=1))
    b_ub = np.append(np.ones(n) * -1e-9, np.triu(np.ones((n, n)), k=1) - 1e-9).flatten()
    c = np.append(np.zeros(n), -np.ones(n))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, 1)] * n)
    if not res.success or np.isnan(res.x).any():
        return None
    return res.x

def objective(pos_flat):
    pos = pos_flat.reshape(-1, 2)
    radii = compute_optimal_radii(pos)
    if radii is None:
        return np.inf
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    dims = 2 * n
    bounds = [(0, 1)] * dims
    max_iter = 2000
    num_runs = 40
    best_sum = 0
    best_center = None
    best_radii = None

    for _ in range(num_runs):
        init = np.random.rand(dims)
        res = minimize(objective, init, method='SLSQP', bounds=bounds, options={'maxiter': max_iter})
        if res.success:
            radii = compute_optimal_radii(res.x.reshape(-1, 2))
            if radii is not None and np.sum(radii) > best_sum:
                best_sum = np.sum(radii)
                best_center = res.x.reshape(-1, 2)
                best_radii = radii

    return best_center, best_radii, best_sum
