import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii(centers):
    n = centers.shape[0]
    r = np.full(n, 0.02)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([1, 0] * i + [-1, 0] * (n - i - 1) + [0, -1] * i + [-1, 0] * (n - i - 1))
        b_ub.append(1)
        A_ub.append([-1, 0] * i + [1, 0] * (n - i - 1) + [0, 1] * i + [1, 0] * (n - i - 1))
        b_ub.append(1)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for i, j in pairs:
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        A_ub.append([1, 0] * i + [-1, 0] * (n - i - 1) + [0, -1] * i + [-1, 0] * (n - i - 1) + [0, 0])
        b_ub.append(dist)
        A_ub.append([-1, 0] * i + [1, 0] * (n - i - 1) + [0, 1] * i + [1, 0] * (n - i - 1) + [0, 0])
        b_ub.append(dist)
    c = -r
    bounds = [(0, 0.5)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.fun <= -1e-9:
        return -res.fun, res.x
    else:
        return 0, np.zeros(n)

def evaluate_fitness(individual):
    centers = individual.reshape(26, 2)
    centers = np.clip(centers, 0, 1)
    return solve_radii(centers)[0]

def crossover(parent1, parent2):
    i1, i2 = np.random.choice(52, 2, replace=False)
    child1 = np.copy(parent1)
    child1[i1:i2] = parent2[i1:i2]
    child2 = np.copy(parent2)
    child2[i1:i2] = parent1[i1:i2]
    return child1, child2

def mutate(individual, sigma):
    individual += sigma * np.random.randn(52)
    individual = np.clip(individual, 0, 1)
    return individual

def clamp_to_square(individual):
    individual = np.clip(individual, 0, 1)
    return individual

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    pop_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    init_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    population = np.random.rand(pop_size, 52)
    best_population = np.copy(population)
    best_fitness = np.zeros(pop_size)

    for generation in range(generations):
        fitness = np.array([evaluate_fitness(individual) for individual in population])
        best_fitness[generation] = fitness.max()
        sorted_indices = np.argsort(fitness)
        top_individuals = population[sorted_indices[-elitism:]]
        new_population = top_individuals

        for _ in range(pop_size - elitism):
            parent1 = population[np.random.choice(tournament_size, size=2, replace=False)].min(axis=0)
            parent2 = population[np.random.choice(tournament_size, size=2, replace=False)].min(axis=0)
            child1, child2 = crossover(parent1, parent2)
            if np.random.rand() < crossover_prob:
                new_population = np.vstack((new_population, child1, child2))
            else:
                new_population = np.vstack((new_population, parent1, parent2))

            mutation_rate *= mutation_rate
            sigma = init_sigma * sigma_decay**generation

            for i in range(len(new_population)):
                new_population[i] = mutate(new_population[i], sigma)
                new_population[i] = clamp_to_square(new_population[i])

        fitness = np.array([evaluate_fitness(individual) for individual in new_population])
        new_individuals = new_population[np.argsort(fitness)[-elitism:]]
        population = new_individuals

        for individual in population:
            _, radii = solve_radii(individual.reshape(26, 2))
            if np.sum(radii) + 1e-9 > 2.636:
                individual += (np.random.randn(52) * 1e-9) % 1

    best_idx = np.argmax(best_fitness)
    best_individual = best_population[best_idx]
    centers, radii = solve_radii(best_individual.reshape(26, 2))
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
