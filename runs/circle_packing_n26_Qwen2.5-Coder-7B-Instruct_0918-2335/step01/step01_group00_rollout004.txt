import numpy as np
from scipy.optimize import linprog, minimize
from math import sqrt
from random import random, seed
from scipy.spatial.distance import cdist

def make_grid(n):
    """Generate a hexagonal lattice of 'n' points in a unit square with jitter."""
    theta = np.linspace(0, 2*np.pi, n+1)[:-1]
    r = np.linspace(0, 1, int(np.sqrt(n)) + 1)[:-1]
    theta, r = np.meshgrid(theta, r)
    x = r * np.cos(theta) * 0.9 + 0.5 + random() * 0.2 - 0.1
    y = r * np.sin(theta) * 0.9 + 0.5 + random() * 0.2 - 0.1
    x = x.flatten()[:n]
    y = y.flatten()[:n]
    return np.column_stack((x, y))

def lp_value_and_radii(centres):
    """Compute the best radii for a given set of centre coordinates."""
    n = len(centres)
    w_limits = np.minimum.reduce([centres[:,0], 1-centres[:,0], centres[:,1], 1-centres[:,1]])
    bounds = [(0, w_limit) for w_limit in w_limits]
    D_ij = cdist(centres, centres)
    A_ub = np.zeros((n, 2*n))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[i, 2*i] = 1
            A_ub[i, 2*j] = 1
            A_ub[j, 2*i] = 1
            A_ub[j, 2*j] = 1
            A_ub[i, 2*i+1] = -1
            A_ub[j, 2*j+1] = -1
        A_ub[i, 2*i+1] = -1
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=D_ij.flatten(), bounds=bounds, method='highs')
    if not res.success:
        return -np.inf, np.zeros(n)
    return -1 * res.fun, res.x

def run_packing():
    seed(0)
    nrestarts = 150
    best_r = None
    best_res = None
    
    for _ in range(nrestarts):
        centres = make_grid(26)
        res, r = lp_value_and_radii(centres)
        if best_res is None or res > best_res:
            best_res, best_r = res, r
    
    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")
    
    centers = best_res.history['x'][-1].reshape(26, 2)
    radii = best_r
    sum_radii = best_res.fun
    return centers, radii, sum_radii
