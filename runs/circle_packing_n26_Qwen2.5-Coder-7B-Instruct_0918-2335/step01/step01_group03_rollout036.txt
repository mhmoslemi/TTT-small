import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def optimal_speeds(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    speeds = np.ones(n)
    for i in range(n):
        d_i = np.min(np.abs(centers[i] - np.array([[0], [1], [0], [1]])))
        speeds[i] = 1 + 2 / (d_i + 1e-6)
        speeds[i] *= np.random.lognormal(mean=0, sigma=0.4)
        speeds[i] = np.clip(speeds[i], 1, None)
    return speeds

def simulate_inflation(centers, speeds):
    n = centers.shape[0]
    active = np.array(range(n))
    stops = np.zeros(n)
    while active.size > 0:
        dist_matrix = squareform(pdist(centers[active]))
        boundary_times = np.min(np.append(dist_matrix / (speeds[active][:, None] + speeds[active][None, :]), 
                                           np.min(centers[active], axis=1)[:, None] / speeds[active][:, None]), 
          axis=1)
        min_time = np.min(boundary_times)
        deactivate = np.where(boundary_times == min_time)[0]
        if deactivate.size > 0:
            stops[active[deactivate]] = min_time
            active = np.delete(active, deactivate)
    return stops

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    families = {
        'grid': np.array([(i // 5, i % 5) for i in range(25)] + [(12.5, 12.5)]),
        'hexagon': np.array([[i % 5, (i // 5) % 6] for i in range(30)]),
        'hybrid': np.array([[0, 0], [0, 4], [4, 0], [4, 4], [2, 2], [6, 2], [2, 6], [6, 6]] 
                           + [(3.5, i) for i in range(18)]),
        'random': np.random.rand(n, 2)
    }
    best_results = None
    for family_name, centers in families.items():
        best_sum = 0
        for _ in range(3000):
            speeds = optimal_speeds(centers)
            stops = simulate_inflation(centers, speeds)
            final_radii = speeds * stops
            total_radius = np.sum(final_radii)
            if total_radius > best_sum:
                best_sum = total_radius
                best_centers = centers
                best_radii = final_radii
        if best_results is None or total_radius > best_results[2]:
            best_results = (best_centers, best_radii, best_sum)
    centers, radii, sum_radii = best_results
    results = minimize(lambda v: -np.sum(v), x0=radii, method='SLSQP', options={'maxiter': 2000},
                       bounds=np.vstack((np.zeros(n), np.ones(n))))
    if results.success and results.fun < sum_radii:
        centers, radii, sum_radii = best_centers, results.x, -results.fun
    return centers, radii, sum_radii
