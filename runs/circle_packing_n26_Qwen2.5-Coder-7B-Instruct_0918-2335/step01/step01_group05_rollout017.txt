import numpy as np
from scipy.optimize import differential_evolution, minimize

def=obj(v):
    r = v[52:]
    return -np.sum(r) + 1e6 * np.sum(np.maximum(constraints(v), 0))

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(429, float)
    cons[:104] = pos[:,0] - r
    cons[104:208] = 1 - pos[:,0] - r
    cons[208:312] = pos[:,1] - r
    cons[312:416] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    cons[416:] = dists[mask] - (r[:,None] + r[None,:])[mask]
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    best_sol = None
    best_sum = -np.inf

    res_de = differential_evolution(obj, bounds, args=(constraints,), popsize=15, maxiter=500, seed=42)
    if res_de.success:
        best_sol = res_de.x
        best_sum = -res_de.fun

    if best_sol is not None and constraints(best_sol).min() >= -1e-10:
        res_slsqp = minimize(fun=obj, x0=best_sol, method='SLSQP',
                             bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                             options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res_slsqp.success and constraints(res_slsqp.x).min() >= -1e-10:
            best_sol = res_slsqp.x
            best_sum = -res_slsqp.fun
    
    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = best_sol[:52].reshape(n, 2)
    radii = best_sol[52:]
    return centers, radii, best_sum
