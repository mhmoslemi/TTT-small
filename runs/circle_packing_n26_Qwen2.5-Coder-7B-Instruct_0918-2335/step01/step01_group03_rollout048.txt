def lp_optimize_radii(centers):
    n = len(centers)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        A_ineq.append([0, 0, 1])
        b_ineq.append(centers[i, 0])
        A_ineq.append([0, 0, 1])
        b_ineq.append(1 - centers[i, 0])
        A_ineq.append([0, 1, 0])
        b_ineq.append(centers[i, 1])
        A_ineq.append([0, 1, 0])
        b_ineq.append(1 - centers[i, 1])
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            d = np.sqrt(diff[0] ** 2 + diff[1] ** 2)
            A_ineq.append([-1, 0, -1])
            b_ineq.append(d)
    res = linprog([-1] * n, A_ub=A_ineq, b_ub=b_ineq, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x
