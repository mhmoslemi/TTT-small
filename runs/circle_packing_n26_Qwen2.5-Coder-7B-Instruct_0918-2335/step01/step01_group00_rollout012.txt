from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        r = v[52:]                      # last 26 entries
        return -np.sum(r)

    def constraints(v):

        r   = v[52:]                   # radii

    bounds += [(0,0.5)]*26        # radii
