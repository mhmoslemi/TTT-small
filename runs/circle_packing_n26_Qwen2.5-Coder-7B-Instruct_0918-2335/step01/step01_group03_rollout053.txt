import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(0)
    layouts = get_layouts()
    best_packing = None
    best_sum_radii = -np.inf

    for layout in layouts:
        packing = simulate_packing(layout)
        if packing is not None:
            (centers, radii, sum_radii) = packing
            if (sum_radii > best_sum_radii):
                best_packing = packing
                best_sum_radii = sum_radii

    if best_packing is None:
        raise RuntimeError("No valid packing found")

    (centers, radii, sum_radii) = best_packing
    if sum_radii < 2.636:
        final_packing = refine_packing(centers, radii)
        if final_packing is not None:
            (centers, radii, sum_radii) = final_packing

    validate, msg = validate_packing(centers, radii)
    if not validate:
        raise RuntimeError(msg)

    return centers, radii

def get_layouts():
    n = 26
    layouts = []

    # Regular grid
    grid = np.mgrid[0:1:5j, 0:1:5j]
    centers = grid.reshape(2, -1).T
    centers = np.vstack((centers, [(0.5, 0.5)]))
    layouts.append(centers)

    # Hexagonal
    center_rows = np.linspace(0, 1, int(np.sqrt(6 * n)))
    centers = []
    radius = (1 - center_rows) / (2 * center_rows)
    dx = 2 * (x - 0.5) * 1 / x
    for y, r in zip(center_rows, radius):
        if not np.isnan(y) and not np.isnan(r):
            dx = 2 * (x - 0.5) * 1 / x
            if x < 1:
                centers.append((dx, y))
    centers = np.array(centers)
    layouts.append(centers)

    # Hybrid
    corner = np.array([(0, 0), (0, 1), (1, 0), (1, 1)])
    edge = np.array([(0.5, y) for y in center_rows[1:-1]] +
                   [(x, 0.5) for x in center_rows[1:-1]])
    grid = np.mgrid[0:1:4j, 0:1:5j]
    small_grid_centers = grid.reshape(2, -1).T
    centers = np.vstack((corner, edge, small_grid_centers))
    layouts.append(centers)

    # Random
    centers = np.random.rand(n, 2)
    layouts.append(centers)

    return layouts

def simulate_packing(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    t_stop = np.full(n, np.inf)
    active = np.arange(n)
    event_tolerance = 1e-12
    max_time = 1000

    for _ in range(3000):
        event_times = np.zeros(n)
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    return None
                t = dist / (radii[i] + radii[j])
                if abs(t - t_stop[i]) < event_tolerance or \
                   abs(t - t_stop[j]) < event_tolerance:
                    return None
                event_times[i] = max(event_times[i], t)
                event_times[j] = max(event_times[j], t)

            dx = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            if dx < radii[i]:
                return None
            event_times[i] = min(event_times[i], dx / radii[i])

        t_min = np.min(event_times)
        stop_set = np.where(np.abs(event_times - t_min) < event_tolerance)[0]
        for i in stop_set:
            t_stop[i] = t_min
            active = active[active != i]

        if len(active) == 0:
            return centers, radii, np.sum(radii)

        for i in stop_set:
            radii[i] = (t_stop[i] * radii[i]).clip(0, 0.5)

def refine_packing(centers, radii):
    n = centers.shape[0]
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    x0 = np.reshape(np.hstack([centers.flatten(), radii]), -1)

    def objective(x):
        return -np.sum(x[n:2*n])

    def constraints(x):
        x = np.reshape(x, (n, 2))
        r = x[:, 2]
        # Boundary constraints
        lower_bound = (x[:, 0] - r).flatten()
        upper_bound = (1 - x[:, 0] - r).flatten()
        lower_left = (x[:, 1] - r).flatten()
        upper_right = (1 - x[:, 1] - r).flatten()
        con = []
        con.extend([{'type': 'ineq', 'fun': lambda x, lb=lb: (x-lb)} for lb in [lower_bound, upper_bound, lower_left, upper_right]])
        # Pairwise constraints
        for i in range(n):
            for j in range(i+1, n):
                xi, yi = x[i]
                xj, yj = x[j]
                r1, r2 = r[i], r[j]
                dist = np.linalg.norm([xi - xj, yi - yj])
                con.append({'type': 'ineq', 'fun': lambda x, lb=lb: (x - lb)} for lb in [dist - (r1 + r2)])
        return con

    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    if not res.success:
        return None

    refined_centers = res.x[:n*2].reshape((n, 2))
    refined_radii = res.x[n*2:]

    if not validate_packing(refined_centers, refined_radii)[0]:
        return None

    return refined_centers, refined_radii, np.sum(refined_radii)
