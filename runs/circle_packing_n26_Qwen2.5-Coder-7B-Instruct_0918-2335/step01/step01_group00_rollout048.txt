```python
import numpy as np
from scipy.optimize import minimize, basinhopping
from scipy.spatial.distance import squareform, pdist

N = 26
NUM_VARS = 2 * N + N

def obj(v):
    return -np.sum(v[NUM_VARS:])

def constraints(v):
    pos = v[:NUM_VARS].reshape(N, 2)
    r = v[NUM_VARS:]
    
    # boundary constraints
    cons = np.empty(4 * N, float)
    cons[0:N] = pos[:, 0] - r
    cons[N:2 * N] = 1 - pos[:, 0] - r
    cons[2 * N:3 * N] = pos[:, 1] - r
    cons[3 * N:] = 1 - pos[:, 1] - r

    # pairwise constraints in squared form
    diffs = squareform(pdist(pos, 'euclidean'))
    combo_mask = np.triu(np.ones((N, N), bool), k=1)
    pair_cons = diffs[combo_mask] - (r[combo_mask]**2)
    
    return np.concatenate([cons, pair_cons])

def inflate_radii(centers, radii):
    for i in range(N):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(N):
            if i == j: continue
            dist = np.linalg.norm(centers[i] - centers[j])
            max_r = min(max_r, dist - radii[j])
        radii[i] = min(max_r, 0.5)
    return radii

def search(v0=None, popsize=10):
    if v0 is None:
        v0 = np.zeros(NUM_VARS)
        np.random.seed(0)
        v0[:NUM_VARS].reshape(N, 2)[:, 0] = np.random.rand(N)
        v0[:NUM_VARS].reshape(N, 2)[:, 1] = np.random.rand(N)
        v0[NUM_VARS:] = np.random.uniform(0.05, 0.2, N)
    bounds = [(0, 1)] * 26  # x
    bounds += [(0, 1)] * 26  # y
    bounds += [(0, 0.5)] * 26  # radii
    
    res = minimize(
        fun=obj, 
        x0=v0, 
        args=(), 
        method='SLSQP', 
        bounds=bounds, 
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 4000, 'ftol': 1e-12}
    )
    
    if not res.success:
        return res
    else:
        best_v = res.x
        best_r = inflate_radii(best_v[:NUM_VARS].reshape(N, 2), best_v[NUM_VARS:])
        
        res_infl = minimize(
            fun=obj, 
            x0=np.concatenate([best_v[:NUM_VARS], best_r]), 
            args=(), 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'ineq', 'fun': constraints},
            options={'maxiter': 4000, 'ftol': 1e-12}
        )
        
        return res_infl if res_infl.fun < res.fun else res

def run_packing(nrestarts=100):
    best_res = None
    for _ in range(nrestarts):
        res = search()
        if best_res is None or res.fun < best_res.fun:
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                
    if best_res is None:
        raise ValueError("Optimization failed to find a valid solution")

    best_v = best_res.x
    centers = best_v[:NUM_VARS].reshape(N, 2)
    radii = inflate_radii(centers, best_v[NUM_VARS:])
    return centers, radii, -best_res.fun

# Example usage
np.random.seed(42)
centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
print(f"Centers: {centers}")
print(f"Radii: {radii}")
```
