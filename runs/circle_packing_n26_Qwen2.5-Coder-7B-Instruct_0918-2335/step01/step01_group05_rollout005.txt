import numpy as np
from scipy.optimize import linprog

def evaluate_positions(centers):
    n = centers.shape[0]
    r = centers[:, 2]
    radii = r / r.max()  # Ensure sum is maximized
    
    # Get rid of duplicate positions
    unique_indices = np.unique(np.round(centers[:, :2], 2), axis=0, return_inverse=True)[1]
    centers_uniq = centers[unique_indices == np.unique(unique_indices)[0]]
    
    if len(centers_uniq) < n:
        radii *= len(centers_uniq) / n
    
    dists = np.sqrt((centers_uniq[:, None, :2] - centers_uniq[None, :, :2])**2).sum(axis=2)
    return radii, np.sum(radii), centers_uniq

def run_packing():
    n = 26
    step = 0.02
    sigma_0 = 0.05
    n_iter = 5000
    alpha = 0.9995
    T = 0.1
    
    # Initial positions from 5x5 grid, plus one random point
    grid = np.array([[i/5 * step for i in range(1, 6)] for j in range(5)] for k in range(5)])
    grid = np.array([pos for row in grid for pos in row])
    grid = np.array([(i*step + 0.01, j*step + 0.01) for i in range(5) for j in range(5)])
    grid = np.vstack((grid, np.random.rand(1, 2)))
    grid = [(x, y) for x, y in zip(grid[:, 0], grid[:, 1])]
    
    # Add additional grid points for more coverage
    grid = np.array(grid, dtype=float)
    grid += np.random.uniform(-step, step, grid.shape) * 2
    grid = [(x, y) for x, y in zip(grid[:, 0], grid[:, 1])]
    grid = [(x, y) if 0 <= x <= 1 and 0 <= y <= 1 else (x - (x < 0) + (x >= 1), y - (y < 0) + (y >= 1)) for x, y in grid]
    
    # Duplicate points to satisfy 26 points
    grid = np.array(grid)
    grid = np.vstack((grid, grid[0]))
    
    best_sum = 0
    best_centers = None
    best_radii = None
    
    for i in range(1):
        centers = grid[i*26:(i+1)*26]
        for j in range(1000):
            # Propose a new center position
            i = np.random.randint(n)
            dx = np.random.normal(0, sigma_0)
            dy = np.random.normal(0, sigma_0)
            new_center = (np.clip(centers[i, 0] + dx, 0, 1), np.clip(centers[i, 1] + dy, 0, 1))
            
            # Evaluate new positions
            centers_new = centers.copy()
            centers_new[i] = new_center
            radii, sum_radii, _ = evaluate_positions(np.column_stack((centers_new,radii)))
            
            # Acceptance rule
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers_new
                best_radii = radii
            elif np.random.rand() < np.exp((sum_radii - best_sum) / T):
                best_sum = sum_radii
                best_centers = centers_new
                best_radii = radii
            
            # Reduce temperature
            sigma_0 *= alpha
    
    return best_centers[:n, :2], best_radii[:n], best_sum
