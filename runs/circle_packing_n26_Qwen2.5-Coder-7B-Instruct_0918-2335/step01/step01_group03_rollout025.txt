def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def find_best_radius(layout, speeds):
    centers = layout.copy()
    n = 26
    radii = np.zeros(n)
    active = set(range(n))
    t_stop = [None] * n
    d = cKDTree(centers)

    while active:
        t_ij = np.inf
        t_edge = np.inf
        stop_set = set()
        for i in active:
            for j in active:
                if i < j:
                    distance = d.query_ball_point(i, radii + radii[active - {i}])
                    if len(distance) > 0 and speeds[distance] + speeds[i] > 0:
                        t_ij = min(t_ij, Radii[i] + Radii[active - {i}])

            t_edge = min(t_edge, speeds[active] * (1 - centers[active, i]))

        t_min = min(t_ij, t_edge)
        for i in active:
            if t_edge - Radii[i] < t_min + 1e-12:
                stop_set.add(i)

        for i in stop_set:
            t_stop[i] = t_min
            active.remove(i)

    max_radius = np.zeros(n)
    for i in range(n):
        t_stop[i] = max(0, min(t_stop[i], 1 Radii[i] + Radii[i]))
        max_radius[i] = speeds[i] * t_stop[i]
    
    sum_radii = np.sum(max_radius)
    return centers, max_radius, sum_radii
