import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def solve_lradii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append([1] + [0]*(n-1))
        b_ub.append(pos[i, 0])
        A_ub.append([1] + [0]*(n-1))
        b_ub.append(1 - pos[i, 0])
        A_ub.append([0]*(i) + [1] + [0]*(n-1-i))
        b_ub.append(pos[i, 1])
        A_ub.append([0]*(i) + [1] + [0]*(n-1-i))
        b_ub.append(1 - pos[i, 1])

    dists = squareform(pdist(pos))
    for i in range(n):
        for j in range(i+1, n):
            if i != j:
                A_ub.append([1, 1] + [0]*(n-2))
                b_ub.append(dists[i, j] - 1e-12)

    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, 0.5)]*n, method='highs')
    return res.x, -res.fun

def fitness(pos):
    r, s_r = solve_lradii(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = parent1.copy()
    for i in range(len(child)):
        if np.random.rand() < rate:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    for i in range(len(individual)):
        if np.random.rand() < rate:
            individual[i] += np.random.randn() * sigma
        individual[i] = np.clip(individual[i], 0, 1)
    return individual

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(len(pop), k, replace=False)
    best_index = indices[np.argmin([fitness_vals[i] for i in indices])]
    return pop[best_index]

def ga_best_positions():
    POPSIZE = 80
    GENS = 250
    ELITE numer = int(0.1 * POPSIZE)
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.2
    MUTATION_SIGMA = 0.03

    np.random.seed(42)
    pop = np.random.rand(POPSIZE, 52)
    fitness_vals = np.array([fitness(ind) for ind in pop])
    elite = pop[np.argsort(fitness_vals)[:ELITE numer]][:]
    pop[:ELITE numer] = elite[:]

    for gen in range(GENS):
        for i in range(ELITE numer, POPSIZE):
            p1, p2 = tournament(pop, fitness_vals, 3), tournament(pop, fitness_vals, 3)
            ind = crossover(p1, p2, CROSSOVER_RATE)
            ind = mutate(ind, MUTATION_RATE, MUTATION_SIGMA)
            pop[i] = ind
        fitness_vals = np.array([fitness(ind) for ind in pop])
        elite = pop[np.argsort(fitness_vals)[:ELITE numer]][:]
        pop[:ELITE numer] = elite[:]

    return pop[np.argmin(fitness_vals)]

def polish_solution(pos):
    n = pos.shape[0]
    r, _ = solve_lradii(pos)
    initial = np.concatenate((pos.flatten(), r))
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    constraints = [
        {'type': 'ineq', 'fun': lambda v, i=i: v[n + i] - v[i + 2 * n]},
        {'type': 'ineq', 'fun': lambda v, i=i: 1 - v[i + 2 * n] - v[n + i]},
        {'type': 'ineq', 'fun': lambda v, i=i: v[2 * n + i] - v[i]},
        {'type': 'ineq', 'fun': lambda v, i=i: 1 - v[i] - v[2 * n + i]}
    ]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq', 'fun': lambda v, i=i, j=j: np.sqrt((v[i + 2 * n] - v[j + 2 * n]) ** 2 + (v[i] - v[j]) ** 2) - (v[n + i] + v[n + j]) - 1e-12
            })
    res = minimize(fitness, initial, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    centers = res.x[:n*2].reshape(n, 2)
    radii = res.x[n*2+2*n:]
    return centers, radii, -res.fun
