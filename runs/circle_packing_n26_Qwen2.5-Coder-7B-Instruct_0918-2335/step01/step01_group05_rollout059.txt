import numpy as np
from scipy.optimize import minimize

N = 26
eps = 1e-12
min_step = 1e-5
cool = 0.995
jitter_amp = 0.1
max_subiter = 20
max_iter = 2000
gs_init = 0.02

def validate_packing(centers, radii):
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    cps = np.random.rand(N, 2)
    rs = np.zeros(N)
    dg = gs_init
    subiter_count = 0
  
    while dg > min_step:
        rs += dg
        cps += jitter_amp * (np.random.rand(N, 2) - 0.5)
        cps[:,0] = np.clip(cps[:,0], rs, 1 - rs)
        cps[:,1] = np.clip(cps[:,1], rs, 1 - rs)
        overlaps = False
        for subiter in range(max_subiter):
            dists = np.sqrt(((cps[:,None,:] - cps[None,:,:])**2).sum(axis=2))
            mask = (dists < rs[:,None] + rs[None,:] - eps).astype(int)
            np.fill_diagonal(mask, 0)
            if mask.sum() == 0:
                overlaps = True
                break
            for i in range(N):
                for j in range(i+1, N):
                    if mask[i,j] > 0:
                        overlap = (rs[i] + rs[j] - dists[i,j] + eps) * 0.5
                        # move along line connecting centers
                        vec = cps[i] - cps[j]
                        dist = np.linalg.norm(vec)
                        disp = overlap * (vec / dist)
                        cps[i] -= disp
                        cps[j] += disp
                        cps[:,0] = np.clip(cps[:,0], rs, 1 - rs)
                        cps[:,1] = np.clip(cps[:,1], rs, 1 - rs)
            overlap = False
    rs_opt = np.maximum(np.minimum(rs, cps[:,0]), np.minimum(rs, 1 - cps[:,0]))
    rs_opt = np.maximum(np.minimum(rs_opt, cps[:,1]), np.minimum(rs_opt, 1 - cps[:,1]))

    sum_radii = np.sum(rs_opt)
    return cps, rs_opt, sum_radii

centers, radii, sum_radii = run_packing()
