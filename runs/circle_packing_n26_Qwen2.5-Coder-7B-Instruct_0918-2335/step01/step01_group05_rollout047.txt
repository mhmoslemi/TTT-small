import numpy as np
from scipy.optimize import minimize, dual_annealing
from scipy.spatial import distance_matrix

def obj(v):
    r = v[52:]
    return -np.sum(r)

def penalty(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    constraints = np.empty(429, float)
    constraints[:104] = pos[:,0] - r
    constraints[104:208] = 1 - pos[:,0] - r
    constraints[208:312] = pos[:,1] - r
    constraints[312:] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    constraints[104:429] = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.sum(np.maximum(constraints, 0) ** 2)

def cons(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    constraints = np.empty(429, float)
    constraints[:104] = pos[:,0] - r
    constraints[104:208] = 1 - pos[:,0] - r
    constraints[208:312] = pos[:,1] - r
    constraints[312:] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    constraints[104:429] = dists[mask] - (r[:,None] + r[None,:])[mask]
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)

    for _ in range(20):
        res_de = dual_annealing(func=lambda v: penalty(v), bounds=bounds, maxiter=500, initial_temp=1000)
        if res_de.success:
            init = res_de.x
            break

    res_slsqp = minimize(obj, init, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': cons},
                          options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    if res_slsqp.success and cons(res_slsqp.x).min() >= -1e-10:
        centers = res_slsqp.x[:52].reshape(n, 2)
        radii = res_slsqp.x[52:]
        return centers, radii, -res_slsqp.fun

    centers = init[:52].reshape(n, 2)
    radii = init[52:]
    return centers, radii, -penalty(init)
