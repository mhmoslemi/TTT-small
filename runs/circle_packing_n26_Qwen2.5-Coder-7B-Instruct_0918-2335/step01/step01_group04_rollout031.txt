import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.linalg.norm(centers[i] - centers[j])
    return dists

def solve_radii_lp(centers):
    n = centers.shape[0]
    dists = pairwise_dists(centers)
    A_ineq = []
    b_ineq = []
    
    # Boundary constraints
    for i in range(n):
        A_ineq.append([1 if j == i else 0 for j in range(n)])
        b_ineq.append(centers[i, 0])
        A_ineq.append([-1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centers[i, 0])
        A_ineq.append([1 if j == i else 0 for j in range(n)])
        b_ineq.append(centers[i, 1])
        A_ineq.append([-1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centers[i, 1])
    
    # Pair constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append([1 if j == ii else 0 for ii in range(n)])
            A_ineq[-1][i] = -1
            A_ineq[-1][j] = 1
            A_ineq[-2] = [-1 if j == ii else 0 for ii in range(n)]
            A_ineq[-2][i] = 1
            A_ineq[-2][j] = -1
            b_ineq.append(dists[i, j])
            b_ineq.append(dists[i, j])
    
    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    
    objective = -np.ones(n)
    res = linprog(objective, A_ub=A_ineq, b_ub=b_ineq, method='highs')
    return -res.fun, res.x

def sa_optimize(initial_centers, max_iter=500):
    best_centers = initial_centers
    best_r = solve_radii_lp(initial_centers)[1]
    best_sum_r = np.sum(best_r)
    for _ in range(max_iter):
        i = np.random.randint(26)
        delta = np.random.normal(0, 0.01)
        epsilon = np.random.normal(0, 0.01)
        new_centers = initial_centers.copy()
        new_centers[i, 0] += delta
        new_centers[i, 1] += epsilon
        if not (0 <= new_centers[i, 0] <= 1 and 0 <= new_centers[i, 1] <= 1):
            continue
        r, _ = solve_radii_lp(new_centers)
        new_sum_r = np.sum(r)
        if new_sum_r - best_sum_r > 1e-8 or (new_sum_r - best_sum_r > 0 and np.random.rand() < 0.1):
            best_centers = new_centers
            best_r = r
            best_sum_r = new_sum_r
    return best_centers, best_r, best_sum_r

def run_packing():
    np.random.seed(42)
    initial_centers = np.random.rand(26, 2)
    initial_centers = initial_centers[(initial_centers[:, 0] + initial_centers[:, 1]).argsort()]
    centers, r, sum_r = sa_optimize(initial_centers)
    centers, r, sum_r = sa_optimize(centers)
    sol = minimize(
        lambda x: -np.sum(x[52:]), 
        np.concatenate((centers.flatten(), r)), 
        method='SLSQP', 
        bounds=[(0, 1)] * 52 + [(0, 0.498)] * 26, 
        constraints=[{'type': 'ineq', 'fun': lambda x: np.array([x[i*2] - x[i*2+1] for i in range(26)] + [np.sqrt(((x[2*i*2+2:2*i*2+4] - x[2*j*2+2:2*j*2+4])**2).sum()) - x[2*i*2+2] - x[2*j*2+2] for i in range(26) for j in range(i+1, 26)])}]
    )
    centers = sol.x[:52].reshape(26, 2)
    r = sol.x[52:]
    sum_r = np.sum(r)
    return centers, r, sum_r

centers, r, sum_r = run_packing()
print(centers, r, sum_r)
