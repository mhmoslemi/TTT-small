import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(0)
    
    def validate_center(c):
        return 0 <= c[0] <= 1 and 0 <= c[1] <= 1
    
    def validate_radius(r):
        return r >= 0
    
    def check_non_overlap(centers, radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j]:
                    return False
        return True
    
    def event_simulation(speeds):
        active = set(range(n))
        t_stop = [None] * n
        dists = squareform(pdist(centers))
        
        while active:
            t_ij = dists[active][:, active] / (radii[active][:, None] + radii[active][None, :])
            t_edge = np.minimum(centers[active][:, 0], 1 - centers[active][:, 0], centers[active][:, 1], 1 - centers[active][:, 1]) / speeds[active]
            t_min_indices = np.unravel_index(np.argmin(np.concatenate([t_ij, t_edge[:, None]])), (t_ij.shape[0], t_ij.shape[1] + 1))
            t_min = t_ij[t_min_indices]
            
            for i in range(n):
                if i == t_min_indices[0] and (t_min_indices[1] == i or t_edge[i] == t_min) or i == t_min_indices[1] and (t_min_indices[0] == i or t_edge[i] == t_min):
                    t_stop[i] = t_min
                    active.discard(i)
        
        return np.array([speeds[i] * t_stop[i] for i in range(n)])
    
    center_families = [
        np.array([
            (i / 5, j / 5)
            for i in range(6) for j in range(6)
        ]),
        np.array([
            (i / 5.5, j / 5) if i % 2 else (i / 5.5, j / 5 + 0.5 / 5.5)
            for i in range(11) for j in range(6)
        ]),
        np.array([
            (0.1, 0.1), (0.1, 0.5), (0.1, 0.9),
            (0.5, 0.1), (0.5, 0.5), (0.5, 0.9),
            (0.9, 0.1), (0.9, 0.5), (0.9, 0.9),
            (j / 10, 0.2 + i / 10) for i in range(4) for j in range(5)
        ]),
        np.random.rand(n, 2)
    ]
    
    best_r = 0
    best_centers = None
    best_radii = None
    
    for centers in center_families:
        if len(centers) != n:
            continue
        
        for _ in range(4):
            speeds = np.random.lognormal(mean=0., sigma=0.4, size=n)
            speeds[0:4] *= 100  # Extreme values to push corner discs
            speeds[4:12] *= 50   # Extreme values to push edge discs
            
            radii = event_simulation(speeds)
            if check_non_overlap(centers, radii):
                sum_radii = np.sum(radii)
                if sum_radii > best_r:
                    best_r = sum_radii
                    best_centers = centers.copy()
                    best_radii = radii.copy()
    
    if best_centers is None:
        return np.zeros((n, 2)), np.zeros(n), 0
    
    # Local refinement with SLSQP
    def objective(x):
        return -np.sum(x[n:])
    
    def vi(v, i):
        x, y = v[:n], v[n:]
        return 1 - y[i]
    
    def xl(v, i):
        x, y = v[:n], v[n:]
        return x[i]
    
    def xr(v, i):
        x, y = v[:n], v[n:]
        return 1 - x[i]
    
    def yl(v, i):
        x, y = v[:n], v[n:]
        return y[i]
    
    def yr(v, i):
        x, y = v[:n], v[n:]
        return 1 - y[i]
    
    constraints = [
        {'type': 'ineq', 'fun': vi, 'args': (i,)} for i in range(n)
    ] + [
        {'type': 'ineq', 'fun': xl, 'args': (i,)} for i in range(n)
    ] + [
        {'type': 'ineq', 'fun': xr, 'args': (i,)} for i in range(n)
    ] + [
        {'type': 'ineq', 'fun': yl, 'args': (i,)} for i in range(n)
    ] + [
        {'type': 'ineq', 'fun': yr, 'args': (i,)} for i in range(n)
    ]
    
    for i in range(n):
        for j in range(i + 1, n):
            def c2(v, i, j):
                x, y = v[:n], v[n:]
                return np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2) - (y[i] + y[j])
            
            constraints += [{'type': 'ineq', 'fun': c2, 'args': (v, i, j)} for v in [best_centers, best_centers + 1e-8 * np.random.rand(n, 2)]]
    
    res = minimize(objective, np.concatenate([best_centers.flatten(), best_radii.flatten()]), method='SLSQP', bounds=((0, 1/2),) * 2*n + ((0, None),) * n, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    
    final_centers = res.x[:n].reshape(n, 2)
    final_radii = res.x[n:]
    
    if not validate_packing(final_centers, final_radii)[0]:
        return final_centers, final_radii, np.sum(final_radii)
    
    return final_centers, final_radii, np.sum(final_radii)
