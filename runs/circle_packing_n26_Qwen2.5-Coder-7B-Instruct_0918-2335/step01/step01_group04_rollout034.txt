import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    n = centers.shape[0]
    A_eq = []
    b_eq = []
    c = -np.ones(n)
    for i in range(n):
        A_eq.append([-1 if n == 1 else 0, 0] + [-1 if j == n - 1 else 0 for j in range(n)] + [1])
        b_eq.append(centers[i, 0] + centers[i, 1] - 1)
        A_eq.append([0, 0] + [-1 if n == 1 else 0 for j in range(n-1)] + [-1 if j == n - 2 else 0 for j in range(n)] + [1])
        b_eq.append(centers[i, 0] + centers[i, 1] - 1)
        A_eq.append([2 * centers[i, 0] - 1, 0 if n == 1 else 0] + [-1 if j == n - 1 else 0 for j in range(n)])
        b_eq.append(0 if n == 1 else centers[i, 0])
        A_eq.append([0 if n == 1 else 0, 2 * centers[i, 1] - 1] + [-1 if j == n - 1 else 0 for j in range(n)])
        b_eq.append(0 if n == 1 else centers[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            distance = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_eq.append([1, 1] + [0 for _ in range(2 * n - 3)] + [1 if j == 1 else 0, 1 if i == 2 else 0] + [-1, -1, -1])
            b_eq.append(distance + centers[i, 0] + centers[i, 1] - centers[j, 0] - centers[j, 1])
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=[(0, 0.5)] * n)
    radii = res.x
    return radii

def compute_gradient(f, x):
    h = 1e-5
    n = len(x)
    grad = np.zeros(n)
    for i in range(n):
        x_plus = np.array(x)
        x_plus[i] += h
        x_minus = np.array(x)
        x_minus[i] -= h
        grad[i] = (f(x_plus) - f(x_minus)) / (2 * h)
    return grad
