import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping, NonlinearConstraint
import cvxpy as cp

N = 26

# Objective function: maximise sum of radii
def obj(v):
    return -np.sum(v[N:])

# Constraints function: boundary and pairwise Separation
def constraints(v):
    pos = v[:N].reshape(N, 2)
    r = v[N:]
    
    # boundary part
    cons = np.empty(104, dtype=float)
    cons[:26] = pos[:, 0] - r  # x-r >= 0
    cons[26:52] = 1 - pos[:, 0] - r  # 1-(x+r) >= 0
    cons[52:78] = pos[:, 1] - r  # y-r >= 0
    cons[78:104] = 1 - pos[:, 1] - r  # 1-(y+r) >= 0
    
    # pairwise separation part
    diffs = pos[:,None,:] - pos[None,:,:]
    dist2 = np.sum(diffs**2, axis=2)
    mask = np.triu(np.ones((N,N), dtype=bool), k=1)
    pair_cons = dist2[mask] - (r[:,None] + r[None,:])[mask]**2  # distance ² - (r_i+r_j)² >= 0
    
    return np.concatenate([cons, pair_cons])

# Constrain r in [0, 0.5]:
bounds = [(0, 1)] * N + [(0, 0.5)] * N

# Build the constraints in proper form for NonlinearConstraint:
def build_nonlinear_constraints():
    nlcs = []
    for i in range(N*2):
        def nlc(v, i=i):
            return constraints(v)[i]
        nlcs.append(NonlinearConstraint(nlc, 0, 0))
    return nlcs

# Inflate radii helper:
def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
            for j in range(N):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist > 0:
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max(max_r, 0)
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii

# Main run_packing function:
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_r = None
    best_res = None
    
    # Perform global optimisation using Differential Evolution
    search_result = differential_evolution(obj, bounds, 
                                           constraints=build_nonlinear_constraints(), 
                                           popsize=100, 
                                           maxiter=2000, 
                                           callback=lambda x: print(f"Evolution progress: {x.fun}"), 
                                           disp=True)
    if not search_result.success:
        raise ValueError("Differential Evolution failed to find a valid solution")
    else:
        best_res = search_result
        best_r = search_result.x[N:]
    
    # Validation isn't implemented in the provided prompt
    
    # After obtaining the best result from DE, perform local optimisation
    # with adjusted global solution bounds:
    local_res_init = minimize(obj, best_res.x, method='SLSQP', bounds=bounds, 
                              constraints=build_nonlinear_constraints(), 
                              options={'maxiter': 4000, 'disp': True})
    if not local_res_init.success:
        raise ValueError("Local SLSQP.getMin failed to find a valid solution")
        
    local_res = basinhopping(local_res_init, local_res_init.x, take_step=lambda v: np.random.normal(v, 0.01),
                              minimizer_kwargs={'bounds': bounds, 'method': 'SLSQP', 'constraints': build_nonlinear_constraints()},
                              niter=400, stepsize=0.01, disp=True)

    # Check the result constraint feasibility
    if np.any(constraints(local_res.x) < -1e-10):
        raise ValueError("Final result violates constraints")

    # Inflate radii
    inflated_radii = inflate_radii(local_res.x[:N].reshape(N, 2), local_res.x[N:], max_iter=100)
    
    # Final refinement step:
    return refine_radii_local(local_res.x[:N].reshape(N, 2), inflated_radii)

def refine_radii_local(centers, radii):
    local_opt = minimize(obj, np.concatenate((centers.flatten(), radii)), 
                         method='SLSQP', 
                         bounds=bounds,
                         constraints=build_nonlinear_constraints(),
                         options={'maxiter': 4000,
                                  'tol': 1e-9,
                                  'disp': False})
    return local_opt.x[:N].reshape(N, 2), local_opt.x[N:]

print(run_packing())
