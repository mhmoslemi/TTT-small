def make_d_matrix(centers):
    diffs = centers[:, None, :] - centers[None, :, :]
    return np.sqrt((diffs ** 2).sum(axis=2))

def compute_radii_and_value(centers):
    if centers.shape[0] != 26 or centers.shape[1] != 2:
        raise ValueError("Invalid centers shape")
    
    wall_limits = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((n*(n-1)//2, 2*n))
    b_ub = np.zeros(n*(n-1)//2)
    
    count = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[count, i*2] = 1
            A_ub[count, j*2] = 1
            A_ub[count, i*2+1] = 1
            A_ub[count, j*2+1] = 1
            b_ub[count] = wall_limits[i]
            count += 1
    
    bounds = [(0, limit) for limit in wall_limits]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and not np.isnan(res.x).any():
        return res.x, -res.fun
    else:
        return np.zeros(n), -np.inf
