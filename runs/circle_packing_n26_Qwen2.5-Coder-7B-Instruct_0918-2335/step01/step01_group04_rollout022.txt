import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    
    # Generate initial seed points within the box
    seed_points = np.random.rand(n, 2)
    
    # Voronoi tessellation to create initial circle packing
    vor = Voronoi(seed_points)
    
    # Calculate radii based on Voronoi distances
    radii = np.array([min(vor.ridge_distances[idx].max() / 2, box_size / 2) for idx in range(n)])
    
    # Ensure radii are within the boundaries
    radii = np.clip(radii, 0, box_size / 2)
    
    # Adjust center positions to ensure circles fit within the box
    centers = seed_points + radii
    
    # Ensure points stay within the unit square
    centers = np.clip(centers, 0, box_size)
    
    # Calculate the sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
