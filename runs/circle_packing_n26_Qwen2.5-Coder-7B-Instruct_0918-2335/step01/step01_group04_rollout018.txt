import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers) -> np.ndarray:
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    return dists

def solve_radii_lp(centers) -> np.ndarray:
    n = centers.shape[0]
    dis = pairwise_dists(centers)
    A_ineq = np.zeros((130, 26))
    b_ineq = np.zeros(130)
    for i in range(n):
        A_ineq[i, i] = 1
        A_ineq[n + 2 * i, i] = 1
        A_ineq[n + 2 * i + 1, i] = 1
        b_ineq[i] = centers[i, 0]
        b_ineq[n + 2 * i] = 1 - centers[i, 0]
        b_ineq[n + 2 * i + 1] = centers[i, 1]
        for j in range(i + 1, n):
            k = n + 2 * i + 2 * n * (j - i - 1) + 2 * j
            A_ineq[k, i] = 1
            A_ineq[k, j] = -1
            b_ineq[k] = dis[i, j]
    
    res = linprog(-np.ones(26), A_ub=A_ineq, b_ub=b_ineq, bounds=(0, None), method='highs')
    if not res.success:
        raise ValueError("LP failed")
    radii = res.x
    return radii

def sa_optimize(initial_centers):
    n = len(initial_centers)
    dis = pairwise_dists(initial_centers)
    max_radius = 0.5
    bounds = np.array([[0, max_radius] for _ in range(n)])
    
    def objective(x):
        return -np.sum(x)
    
    def constraints(x):
        pos = x.reshape(n, n)
        mult = np.triu(np.ones((n, n)), k=1)
        return dis[mult.T < 1] - (x[mult.T < 1] + x[mult > 1])
    
    x0 = radii
    res = minimize(fun=objective, x0=x0, method='L-BFGS-B', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000})
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    initial_radius = 0.02
    initial_radii = np.full(n, initial_radius)
    initial_guess = np.concatenate((centers.flatten(), initial_radii))
    bounds = [(0, None)] * (2 * n) + [(0, 0.5)] * n
    
    def objective(x):
        return -np.sum(x[2 * n:])
    
    def constraints(x):
        pos = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        mult = np.triu(np.ones((n, n)), k=1)
        return np.concatenate((pos[:, 0] - pos[:, 1] * dis[:, :-1] - r, pos[:, 1] - pos[:, 0] * dis[:, :-1] - r, dis[mult.T < 1] - (r[mult.T < 1] + r[mult > 1])))
    
    res = minimize(fun=objective, x0=initial_guess, method='L-BFGS-B', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 10000})
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:2 * n].reshape(n, 2)
    radii = res.x[2 * n:]
    max_radius = 0.5
    bounds = [(0, max_radius)] * n
    
    def objective(x):
        return -np.sum(radii)
    
    def constraints(x):
        pos = centers
        r = radii
        mult = np.triu(np.ones((n, n)), k=1)
        return dis[mult.T < 1] - (r[mult.T < 1] + r[mult > 1])
    
    fine_x0 = np.concatenate((centers.flatten(), radii))
    fine_res = minimize(fun=objective, x0=fine_x0, method='L-BFGS-B', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 5000})
    if not fine_res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    fine_centers = fine_res.x[:2 * n].reshape(n, 2)
    fine_radii = fine_res.x[2 * n:]
    sum_radii = np.sum(fine_radii)
    return fine_centers, fine_radii, sum_radii
