import numpy as np
from scipy.optimize import linprog, minimize

np.random.seed(42)

def solve_radii(centres):
    n = centres.shape[0]
    c = -np.ones(n)
    A_eq = np.empty((4*n, n))
    b_eq = np.empty(4*n)
    row = 0
    for i in range(n):
        x, y = centres[i]
        A_eq[row, i] = 1
        b_eq[row] = x
        row += 1
        A_eq[row, i] = 1
        b_eq[row] = 1 - x
        row += 1
        A_eq[row, i] = 1
        b_eq[row] = y
        row += 1
        A_eq[row, i] = 1
        b_eq[row] = 1 - y
        row += 1
    
    A_ub = np.empty((n*(n-1)//2, n))
    row = 0
    for i in range(n):
        for j in range(i+1, n):
            dx, dy = centres[i] - centres[j]
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub = np.sqrt(dx**2 + dy**2)
            row += 1
    
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_eq, method='highs')
    return -result.fun, -result.x

def evaluate_fitness(individual):
    centres = individual[:52].reshape(26, 2)
    centres = np.clip(centres, 0, 1)
    sum_r, radii = solve_radii(centres)
    return sum_r

def crossover(parent1, parent2):
    i1 = np.random.randint(1, 52)
    i2 = np.random.randint(i1+1, 52)
    child = np.empty_like(parent1)
    child[:i1] = parent1[:i1]
    child[i1:i2] = parent2[i1:i2]
    child[i2:] = parent1[i2:]
    return child

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, size=individual.shape)
    individual = np.clip(individual, 0, 1)
    return individual

def clamp_to_square(individual):
    return np.clip(individual, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    population = np.random.rand(population_size, 52)
    top_individuals = np.array(sorted(population, key=evaluate_fitness, reverse=True)[:elitism])

    for gen in range(generations):
        sigma = initial_sigma * (sigma_decay ** gen)
        new_population = np.empty((population_size, 52))

        for _ in range(population_size // 2):
            parent1 = top_individuals[np.random.randint(elitism)]
            parent2 = top_individuals[np.random.randint(elitism)]

            if np.random.rand() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = np.random.choice([parent1, parent2])

            child = mutate(child, sigma)
            child = clamp_to_square(child)
            new_population[_] = child
            new_population[_+population_size//2] = child

        new_population[:elitism] = top_individuals

        for i in range(elitism, population_size):
            fitnesses = [evaluate_fitness(new_population[i])] * tournament_size
            best_idx = np.argmax(fitnesses)
            new_population[i] = new_population[i + best_idx]

        population = new_population

    best_individual = max(population, key=evaluate_fitness)
    centres = best_individual[:52].reshape(n, 2)
    sum_r, radii = solve_radii(centres)

    def local_optimization(centers, radii):
        result = minimize(fun=lambda x: -np.sum(x), x0=np.hstack((centers.flatten(), radii)),
                          method='SLSQP', bounds=[(0, 0.5)] * n + [(0, 1)] * n,
                          options={'maxiter': 2000, 'disp': False})
        return result.x[:52].reshape(n, 2), result.x[52:]

    final_centers, final_radii = local_optimization(centres, radii)
    return final_centers, final_radii, -sum(final_radii)
