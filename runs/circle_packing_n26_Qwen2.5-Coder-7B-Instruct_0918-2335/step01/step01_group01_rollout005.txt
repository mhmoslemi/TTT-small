def lp_opt(p):
    p = p.reshape(ncircles, 2)
    A = []
    b = []
    for i, (x, y) in enumerate(p):
        A.append([1, 0, x])
        b.append(x / (x + 1))
        A.append([-1, 0, x])
        b.append(1 - (1 - x) / (x + 1))
        A.append([1, 0, y])
        b.append(y / (y + 1))
        A.append([-1, 0, y])
        b.append(1 - (1 - y) / (y + 1))
    for i in range(ncircles):
        for j in range(i+1, ncircles):
            dist = np.sqrt((p[i,0]-p[j,0])**2 + (p[i,1]-p[j,1])**2)
            A.append([1, 0, p[i,0], 1, 0, p[j,0]])
            b.append(dist - (p[i,0] + p[j,0]))
            A.append([-1, 0, p[i,0], -1, 0, p[j,0]])
            b.append(dist - (1 - p[i,0] - p[j,0]))
            A.append([1, 0, p[i,1], 1, 0, p[j,1]])
            b.append(dist - (p[i,1] + p[j,1]))
            A.append([-1, 0, p[i,1], -1, 0, p[j,1]])
            b.append(dist - (1 - p[i,1] - p[j,1]))
    res = linprog([0, 0] + [-1]*ncircles, A_ub=np.array(A), b_ub=np.array(b), bounds=[(0,0.5)]*ncircles, method="highs")
    return -res.fun if res.success else 0

# Penalty function
def penalty(x):
    p = x.reshape(ncircles, 2)
    pen = 0
    for i, (x1, y1) in enumerate(p):
        if x1 < 0 or x1 > 1:
            pen += C_BX * abs(x1)
        if y1 < 0 or y1 > 1:
            pen += C_BY * abs(y1)
        if i < ncircles-1:
            for j in range(i+1, ncircles):
                dist = np.sqrt((x1-x[j,0])**2 + (y1-x[j,1])**2)
                if dist < 1e-8:
                    pen += C_ITH * dist
                elif dist > x1 + x[j,0]:
                    pen += C_ITX * (dist - (x1 + x[j,0]))
                elif dist > 1 - x1 - x[j,0]:
                    pen += C_ITX2 * (dist - (1 - x1 - x[j,0]))
    return pen

# mutation and local relax
def mutate(C):
    i_change = np.random.choice(range(ncircles), size=m_mut, replace=False)
    C[i_change] += np.random.normal(0, sigma_mut, size=(m_mut, 2))
    C = np.clip(C, 0, 1)
    return C

def regen(C, A, b):
    C = mutate(C)
    for _ in range(3):
        res = minimize(fun=lambda x: penalty(x), x0=C.flatten(), method="SLSQP", bounds=[(0, 1), (0, 1)] * ncircles, constraints={'type': 'ineq', 'fun': lambda x, A=A, b=b: A @ x - b})
        if res.success:
            C = res.x.reshape(ncircles, 2)
            break
    return C
