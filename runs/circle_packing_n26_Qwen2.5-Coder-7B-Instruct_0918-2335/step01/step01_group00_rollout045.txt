```python
import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping
from scipy.spatial.distance import pdist, squareform

N = 26
NUM_VARS = N * 2 + N
BOUND = (0, 1)
RADIUS_BOUND = (0, 0.5)

def build_constraints(v):
    """ Returns list of constraints: boundary + pair-wise """
    pos = v[:NUM_VARS].reshape(N, 2)
    r = v[NUM_VARS:]

    # boundary
    cons = []
    cons.append(pos[:,0] - r)
    cons.append(1 - pos[:,0] - r)
    cons.append(pos[:,1] - r)
    cons.append(1 - pos[:,1] - r)

    # pairwise
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((N, N), bool), k=1)
    cons.append(dists[mask] - (r[:,None] + r[None,:])[mask])

    # flatten list of lists to array
    return np.concatenate(cons)

def obj(v):
    """ Negative sum of radii to be minimised """
    return -v[NUM_VARS:].sum()

def inflate_radii(centers, radii, tol=1e-10):
    """ Inflate radii to as large as possible while respecting constraints """
    while True:
        old_radii = radii.copy()
        for i in range(N):
            min_dist_toWall = np.min([centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1]])
            min_dist_toNeighbor = np.inf
            for j in range(N):
                if i == j: continue
                dist_ij = np.sqrt(np.sum((centers[i]-centers[j]) ** 2))
                min_dist_toNeighbor = min(min_dist_toNeighbor, dist_ij - radii[j])
            max_allowed_radius = min(min_dist_toWall, min_dist_toNeighbor)
            radii[i] = np.maximum(0, radii[i], max_allowed_radius)
        if np.allclose(old_radii, radii, atol=tol):
            break
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """ Find optimal circle packing """
    best_solution = None
    best_value = -np.inf

    nrestarts = 100
    for restart in range(nrestarts):
        v = np.random.rand(NUM_VARS)
        v[N*N:] = 0.1 * np.random.rand(N)  # Initial small radii
        res = minimize(fun=obj, x0=v, method='SLSQP',
                       bounds=[BOUND] * N * 2 + [RADIUS_BOUND] * N,
                       constraints={'type':'ineq',
                                   'fun': build_constraints},
                       options={'maxiter':4000, 'ftol':1e-12, 'disp':False})
        if not res.success:
            continue

        if np.all(build_constraints(res.x) >= -1e-10):
            if res.fun > best_value:
                best_value = res.fun
                best_solution = res.x

    if best_solution is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers = best_solution[:NUM_VARS].reshape(N, 2)
    radii = best_solution[NUM_VARS:]
    inflated_radii = inflate_radii(centers, radii)
    res_final = minimize(fun=obj, x0=np.concatenate((centers, inflated_radii)),
                         method='trust-constr', bounds=[BOUND] * N * 2 + [RADIUS_BOUND] * N,
                         constraints={'type':'ineq',
                                     'fun': build_constraints},
                         options={'maxiter':1000, 'disp':False})
    if not res_final.success:
        raise ValueError("Post-inflation optimisation failed to find a valid solution")

    centers = res_final.x[:NUM_VARS].reshape(N, 2)
    radii = res_final.x[NUM_VARS:]
    
    return centers, radii, -res_final.fun
```
