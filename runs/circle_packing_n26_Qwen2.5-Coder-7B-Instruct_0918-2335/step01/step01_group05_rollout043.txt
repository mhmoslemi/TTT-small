import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    r = np.full(n, 0.02, dtype=float)
    cps = np.random.rand(n, 2)
    max_iter = 2000
    min_step = 1e-5
    jitter_amp = 0.1 * r[0]
    eps = 1e-12

    for k in range(max_iter):
        δ = 0.02 * (1 - k / max_iter)
        if δ < min_step:
            break

        jitter = jitter_amp * (np.random.rand(n, 2) - 0.5)
        cps += jitter
        cps = np.clip(cps, 0, 1)

        for sub_iter in range(20):
            dists = np.sqrt(((cps[:, None, :] - cps[None, :, :])**2).sum(axis=2))
            mask = dists < r[:, None] + r[None, :] - eps

            if mask.sum() == 0:
                break

            for i in range(n):
                for j in range(i + 1, n):
                    if mask[i, j]:
                        diff = cps[i] - cps[j]
                        dist = dists[i, j]
                        req_overlap = r[i] + r[j] - dist + eps
                        disp = 0.5 * req_overlap * (diff / dist)
                        cps[i] -= disp
                        cps[j] += disp

            cps[:,0] = np.clip(cps[:,0], r, 1- rs)
            cps[:,1] = np.clip(cps[:,1], r, 1- rs)

    rs = minimize(lambda x: -np.sum(np.minimum(x, 1-x)), r, method='BFGS', bounds=[(0, 1)]*n)['x']
    centers = cps
    sum_radii = np.sum(rs)

    return centers, rs, sum_radii
