import numpy as np
from scipy.optimize import linprog, minimize, NonlinearConstraint

def lp_value_and_radii(centres):
    n = centres.shape[0]
    D = np.linalg.norm(centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=2)
    w = np.minimum.reduce([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]])
    c = -np.ones(n)
    bounds = [(0, w_i) for w_i in w]
    A = D[np.triu(np.ones((n, n), dtype=bool), k=1)]
    b = np.zeros(n * (n - 1) // 2)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return -np.inf, np.zeros(n)
    else:
        return -res.fun, res.x

def run_packing():
    n = 26
    initial_layout = np.zeros((n, 2))
    for i in range(n):
        radius = 1 / (2 * np.sqrt(3))
        angle = 2 * np.pi * i / n
        initial_layout[i] = [0.5 + radius * np.cos(angle), 0.5 + radius * np.sin(angle)]
        initial_layout[i] = initial_layout[i] + 0.02 * np.random.rand(2) - 0.01
    
    def objective(x):
        centres = np.resize(x, (n, 2))
        value, _ = lp_value_and_radii(centres)
        return value
    
    def constraint(x):
        centres = np.resize(x, (n, 2))
        result = lp_value_and_radii(centres)
        if result[0] < 2.636:
            return -result[0] + 1e6
        else:
            return 0
    
    nrestarts = 150
    best_r = None
    best_res = None
    
    for _ in range(nrestarts):
        init = initial_layout.flatten()
        init = init + 0.02 * np.random.rand(len(init)) - 0.01
        res = minimize(objective, init, method='BFGS', bounds=[(0, 1)] * n * 2,
                       constraints=NonlinearConstraint(constraint, 0, 0),
                       options={'maxiter': 2000, 'disp': False})
        if res.success and (-res.fun > -best_res.fun if best_res is not None else True):
            if lp_value_and_radii(np.resize(res.x, (n, 2)))[0] >= 2.636:
                best_res = res
                best_r = np.resize(res.x, (n, 2)).flatten()
    
    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")
    
    centers = best_r[:n * 2].reshape(n, 2)
    sum_radii = -best_res.fun
    return centers, best_r[n * 2:], sum_radii
