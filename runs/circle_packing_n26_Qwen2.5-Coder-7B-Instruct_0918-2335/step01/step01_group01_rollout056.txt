def lp_oracle(positions):
    n = positions.shape[0]
    radii = positions[:, 2]
    A = np.vstack([np.eye(n), -np.eye(n), np.repeat(np.eye(n), 2, axis=1)])
    b = np.hstack([radii, -radii, np.ones(2*n) * (1 - 1e-9)])
    res = linprog(b, A_ub=A.T, bounds=(0, None))
    if res.success:
        return -res.fun
    else:
        return float('-inf') if np.isnan(radii).any() else float('-inf')
