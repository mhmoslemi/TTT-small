import numpy as np
from scipy.optimize import linprog, minimize


def pairwise_dists(centers):
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    return np.sqrt((diffs ** 2).sum(axis=2))


def solve_radii_lp(centers):
    n = centers.shape[0]
    r_bounds = [(0, None) for _ in range(n)]
    x_bounds = [(0, 1) for _ in range(n)]
    y_bounds = [(0, 1) for _ in range(n)]

    A = np.empty((n + n + int(n * (n - 1) / 2), n))
    b = np.empty(n + n + int(n * (n - 1) / 2))

    # Boundary constraints
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0]
        A[i + n, i] = 1
        b[i + n] = 1 - centers[i, 0]
        A[i + 2 * n, i] = 1
        b[i + 2 * n] = centers[i, 1]
        A[i + 3 * n, i] = 1
        b[i + 3 * n] = 1 - centers[i, 1]

    # Pairwise constraints
    k = n + n
    mask = np.triu(np.ones((n, n), bool), k=1)
    count = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k + count, i] = 1
            A[k + count, j] = 1
            count += 1

    return linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, 1)] * n, method='highs').x


def generate_initial_centers():
    hex_centers = []
    k = 0
    while True:
        x = (2 * k + 1) / 2
        y = np.sqrt(3) * (k / 2)
        if y > 1:
            break
        for dx in [0, 1 / 2] + ([0, 1 / 2] if y > 1 / 2 else []):
            hex_centers.append(np.array([x + dx, y]))
        k += 1
    hex_centers = np.array(hex_centers)[:26]

    random_centers = []
    for _ in range(30):
        x, y = np.random.rand(2)
        random_centers.append(np.array([x, y]))
    random_centers = np.array(random_centers)
    distances = pairwise_dists(random_centers)
    nearest_indices = np.argsort(distances, axis=1)[:, :3]
    closest_centers = random_centers[np.repeat(np.arange(30), 3), nearest_indices.flatten()].reshape(26, 2)

    mixed_centers = (hex_centers + closest_centers) / 2
    return mixed_centers


def sa_optimize(initial_centers):
    best_r = None
    best_centers = None
    best_sum_r = 0

    for _ in range(5):
        centers = initial_centers + 0.1 * (np.random.rand(*initial_centers.shape) * 2 - 1)
        while True:
            r = solve_radii_lp(centers)
            if (np.isnan(r) or np.any(r < 0) or np.any(centers[:, 0] - r < 0) or np.any(centers[:, 0] + r > 1)
                or np.any(centers[:, 1] - r < 0) or np.any(centers[:, 1] + r > 1)):
                centers = initial_centers + 0.1 * (np.random.rand(*initial_centers.shape) * 2 - 1)
            else:
                dists = pairwise_dists(centers)
                mask = np.triu(np.ones((26, 26), bool), k=1)
                if np.all(dists[mask] > r[mask] + 1e-10):
                    if r.sum() > best_sum_r + 1e-8:
                        best_r = r
                        best_centers = centers
                        best_sum_r = r.sum()
                    break
                else:
                    dx = np.random.normal(size=26)
                    dy = np.random.normal(size=26)
                    centers += np.vstack((dx, dy)).T * 0.01
                    if not np.all(np.isfinite(centers)) or np.any(centers[:, 0] - best_r < 0) or np.any(centers[:, 0] + best_r > 1) or np.any(centers[:, 1] - best_r < 0) or np.any(centers[:, 1] + best_r > 1):
                        centers = initial_centers + 0.1 * (np.random.rand(*initial_centers.shape) * 2 - 1)


def run_packing():
    initial_centers = generate_initial_centers()
    best_r = None
    best_centers = None
    best_sum_r = 0

    for _ in range(3):
        centers = sa_optimize(initial_centers)
        r = solve_radii_lp(centers)
        dists = pairwise_dists(centers)
        mask = np.triu(np.ones((26, 26), bool), k=1)
        if np.all(dists[mask] > r[mask] + 1e-10):
            if r.sum() > best_sum_r + 1e-8:
                best_r = r
                best_centers = centers
                best_sum_r = r.sum()

    result = minimize(lambda x: -x[52:].sum(), np.concatenate((best_centers.flatten(), best_r)), method='SLSQP', bounds=[(0, 1), (0, 1)] * 52, constraints={'type': 'ineq', 'fun': lambda x: np.concatenate([x[:52], pairwise_dists(x[:52]).flatten() - (x[52:][:, None] + x[52:][None, :]).flatten() - 2e-9])})

    if not result.success:
        return np.zeros((26, 2)), np.zeros(26), 0.0

    centers = result.x[:52].reshape(26, 2)
    r = result.x[52:]
    return centers, r, r.sum()
