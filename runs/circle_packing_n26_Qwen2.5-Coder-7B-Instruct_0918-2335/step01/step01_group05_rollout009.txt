import numpy as np

from scipy.spatial.distance import pdist, squareform

def solve_radii(centers):
    n = centers.shape[0]
    A_ineq = np.vstack([
        centers[:, 0] - np.eye(n),  # x_i - r_i <= 0
        1 - centers[:, 0] - np.eye(n),  # 1 - x_i - r_i <= 0
        centers[:, 1] - np.eye(n),  # y_i - r_i <= 0
        1 - centers[:, 1] - np.eye(n)  # 1 - y_i - r_i <= 0
    ])
    
    D = squareform(pdist(centers))
    A_ineq = np.vstack([A_ineq, D - np.triu(np.ones((n, n)) * centers[:, 0][:, None] + centers[:, 1][:, None], k=1)])
    b_ineq = np.hstack([-np.ones(n), np.zeros(n*(n-1)//2)])
    
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=(0, 0.5), method='highs')
    if not result.success:
        raise ValueError("Invalid LP solution")
    
    return -result.fun, result.x
