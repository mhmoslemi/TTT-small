import numpy as np
import scipy.spatial as spatial
from scipy.optimize import linprog, minimize

def lp_radii(centers):
    n = len(centers)
    c = np.ones(n)
    A_bounds = np.concatenate((np.eye(n), -np.eye(n), np.eye(n), -np.eye(n)), axis=1)
    b_bounds = np.repeat([1, 1], n)
    A_overlap = np.zeros((n * (n - 1) // 2, n))
    b_overlap = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist_squared = centers[i, 0] - centers[j, 0] + centers[i, 1] - centers[j, 1]
            A_overlap[idx, i] = 2 * centers[j, 0] - centers[i, 0]
            A_overlap[idx, j] = 2 * centers[i, 0] - centers[j, 0]
            A_overlap[idx, i + n] = -np.sqrt(dist_squared)
            A_overlap[idx, j + n] = -np.sqrt(dist_squared)
            b_overlap[idx] = (centers[i, 0] + centers[i, 1] + centers[j, 0] + centers[j, 1] - 2) / np.sqrt(dist_squared)
            idx += 1
    A_constraints = np.concatenate((A_bounds, A_overlap), axis=0)
    b_constraints = np.concatenate((b_bounds, b_overlap))
    result = linprog(c, A_ub=A_constraints, b_ub=b_constraints, bounds=(0, None), method='highs')
    return -result.fun, result.x

def simulate_annealing():
    n = 26
    initial_positions = np.random.uniform(0, 1, (n, 2))
    best_positions = initial_positions
    best_sum = 0
    current_positions = initial_positions
    T0 = 0.1
    alpha = 1
    num_steps = int(T0 / alpha)
    num_moves = 50
    current_sum = 0
    for k in range(num_steps):
        for _ in range(num_moves):
            i = np.random.randint(n)
            dx, dy = np.random.randn(2) * 0.05 * T0
            new_positions = current_positions.copy()
            new_positions[i, 0] += dx
            new_positions[i, 1] += dy
            new_positions[i, 0] = np.clip(new_positions[i, 0], 0, 1)
            new_positions[i, 1] = np.clip(new_positions[i, 1], 0, 1)
            new_sum, _ = lp_radii(new_positions)
            if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / T0):
                current_positions = new_positions
                current_sum = new_sum
        if current_sum > best_sum:
            best_positions = current_positions
            best_sum = current_sum
        T0 *= alpha
    return best_sum, best_positions

def optimize_with_slsqp(best_positions):
    def obj(v):
        centers = v[:52].reshape(26, 2)
        radii = v[52:]
        return -np.sum(radii)

    def constraints(v):
        pos = v[:52].reshape(26, 2)
        r = v[52:]
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((26, 26), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    bounds = [(0, 1)] * 26
    bounds += [(0, 1)] * 26
    bounds += [(0, 0.5)] * 26
    init = np.hstack((best_positions.flatten(), lp_radii(best_positions)[1]))
    res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 8000, 'ftol': 1e-9, 'disp': False})
    return -res.fun, res.x[:52].reshape(26, 2), res.x[52:]

def run_packing():
    best_sum = 0
    best_positions = np.zeros((26, 2))
    for _ in range(5):
        current_sum, current_positions = simulate_annealing()
        if current_sum > best_sum:
            best_sum, best_positions = current_sum, current_positions
    return optimize_with_slsqp(best_positions)

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
