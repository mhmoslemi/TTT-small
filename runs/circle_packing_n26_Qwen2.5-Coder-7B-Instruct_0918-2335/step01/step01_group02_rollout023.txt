import numpy as np
from scipy.optimize import linprog, minimize

def lp_for_radii(centers):
    n = centers.shape[0]
    b = np.ones(n) * (1 - 1e-12)
    A = np.zeros((2 * n + n * (n - 1) // 2, n))
    for i in range(n):
        A[i, i] = -1
        A[n + i, i] = 1
        for j in range(i + 1, n):
            A[2 * n + i * (n - 1) // 2 + j - i - 1, i] =中心点 (26, 2) 的中心在 [0,1]^2 内
            A[2 * n + i * (n - 1) // 2 + j - i - 1, j] = 1
    result = linprog([-np.ones(n)], A_ub=A, b_ub=b, bounds=(1e-6, None), method='highs')
    if result.success and result.status == 0:
        return result.x
    else:
        return np.zeros(n) + 1e-6

def sa_for_positions(bounds, ref_sum, move_stdev, n_moves, n_temps):
    n = bounds.shape[0]
    best_centers = None
    best_sum = -np.inf
    curr_pos = np.random.uniform(size=(n,))
    temp = bounds[:, 1] - bounds[:, 0]
    for temp in np.logspace(0, np.log10(temp[-1]), n_temps):
        move_stdev *= temp
        unique_moves = np.zeros(n)
        for _ in range(n_moves):
            i = np.random.randint(n)
            next_pos = curr_pos.copy()
            next_pos[i] += np.random.normal(scale=move_stdev)
            next_pos[i] = np.clip(next_pos[i], bounds[i, 0], bounds[i, 1])
            next_sum = lp_for_radii(next_pos).sum()
            if next_sum >= ref_sum or np.random.rand() < np.exp((next_sum - ref_sum) / temp):
                curr_pos[i] = next_pos[i]
                ref_sum = next_sum
                unique_moves[i] = 1
                if not best_sum or ref_sum > best_sum:
                    best_sum = ref_sum
                    best_centers = curr_pos
    return best_centers, best_sum
