import numpy as np
from scipy.optimize import minimize, linprog

def lp_oracle(pos):
    n = pos.shape[0] // 2
    r = np.zeros(n)
    ub = np.inf * np.ones(104 + 26)
    A_ub = np.zeros((104 + 26, 52))
    A_eq = []
    b_eq = []
    
    for i in range(n):
        r[i] = 0.5 * (np.minimum(pos[i], 1 - pos[i]))
        A_ub[2*i, 2*i] = 1
        A_ub[2*i+1, 2*i+1] = 1
        A_ub[2*i, i] = -1
        A_ub[2*i, i + n] = -1
        A_ub[2*i+1, i] = -1
        A_ub[2*i+1, i + n] = -1
        ub[2*i + 2] = max(0, 1 - pos[i])
        
        for j in range(i+1, n):
            x_i, y_i = pos[i], pos[i + n]
            x_j, y_j = pos[j], pos[j + n]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            A_eq.append([-1, -1])
            b_eq.append(0)
            A_ub[104 + 2*i + j - i*(i-1)//2 - 1, 2*i] = 1
            A_ub[104 + 2*i + j - i*(i-1)//2 - 1, 2*i + 1] = 1
            A_ub[104 + 2*i + j - i*(i-1)//2 - 1, 2*j] = 1
            A_ub[104 + 2*i + j - i*(i-1)//2 - 1, 2*j + 1] = 1
            ub[104 + 2*i + j - i*(i-1)//2 - 1] = dist
    
    for pair in range(n*(n-1)//2):
        A_eq[pair][pair] = -1
        A_eq[pair][pair + n*(n-1)//2] = -1
        b_eq[pair] = 0
    
    result = linprog(-np.ones(n), A_ub=A_ub, b_ub=ub, A_eq=A_eq, b_eq=b_eq, bounds=(0, 1), method='highs', options={'tol': 1e-9})
    if result.status != 0:
        return -1e6
    
    r[~np.isnan(result.x)] = result.x[~np.isnan(result.x)]
    return np.sum(r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NP = 40
    w = 0.6
    c1 = 1.5
    c2 = 1.5
    T0 = 0.04
    cooling_factor = 0.995
    velocity_clamp = 0.03
    mutation_rate = 0.1
    mutation_std = 0.02
    max_generations = 1000
    inner_lp_tolerance = 1e-9
    inertia = np.full(NP, w)
    cognitive = np.full(NP, c1)
    social = np.full(NP, c2)
    T = T0 * np.ones(NP)
    velocity = np.zeros((NP, 52))
    pbest_pos = pos_best = np.random.random((NP, 52)) * 0.2 + 0.4
    pbest_val = np.inf * np.ones(NP)
    gbest_val = -np.inf
    gbest_pos = None

    for i in range(NP):
        r = lp_oracle(pos_best[i])
        if r > pbest_val[i]:
            pbest_val[i] = r
            pbest_pos[i] = pos_best[i]
            if r > gbest_val:
                gbest_val = r
                gbest_pos = pos_best[i]

    for t in range(max_generations):
        for i in range(NP):
            velocity[i] = inertia[i] * velocity[i] + cognitive[i] * np.random.rand() * (pbest_pos[i] - pos_best[i]) + social[i] * np.random.rand() * (gbest_pos - pos_best[i])
            if T[i] > 0:
                velocity[i] += np.random.randn(*velocity[i].shape) * mutation_std
            pos_best[i] += velocity[i]
            pos_best[i] = np.clip(pos_best[i], 0, 1)
            r = lp_oracle(pos_best[i])
            if r > pbest_val[i]:
                pbest_val[i] = r
                pbest_pos[i] = pos_best[i]
                if r > gbest_val:
                    gbest_val = r
                    gbest_pos = pos_best[i]
        T *= cooling_factor

    pos_end = np.copy(pbest_pos[np.argmax(pbest_val)])
    r_end = lp_oracle(pos_end)
    
    bounds = [(0,1)] * 52
    bounds += [(0, 0.5)] * n
    x0 = np.concatenate((pos_end, np.zeros(n)))
    res = minimize(lambda v: -np.sum(v[52:]), x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda v: np.asarray([lp_oracle(v[:52]) for v in [v] * n]).flatten()}, options={'maxiter': 2000, 'ftol': 1e-9})
    if res.success:
        r_end = res.x[52:]
    
    return pos_end.reshape(-1, 2), r_end, np.sum(r_end)
