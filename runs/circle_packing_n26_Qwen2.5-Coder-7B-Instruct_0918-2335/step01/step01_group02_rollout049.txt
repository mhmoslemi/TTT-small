import numpy as np
import random
from scipy.optimize import linprog, minimize

def solve_radii(pos):
    n = pos.shape[0]
    A_ub, b_ub = [], []
    for i in range(n):
        A_ub.append([1] + [0] * i + [1] + [0] * (n - i - 1))  # r_i <= x_i
        A_ub.append([1] + [0] * i + [-1] + [0] * (n - i - 1))  # r_i <= 1 - x_i
        A_ub.append([0] * i + [1] + [0] * (n - i - 1))  # r_i <= y_i
        A_ub.append([0] * i + [-1] + [0] * (n - i - 1))  # r_i <= 1 - y_i
    for i in range(n):
        for j in range(i + 1, n):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            A_ub.append([1] + [0] * i + [1] + [0] * (n - i - 1) + 2 * [0] + [1] + [0] * (n - j - 2))  # r_i + r_j <= dist_ij
            b_ub.append(d)
    r = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs').x
    sum_r = r.sum()
    return r, sum_r

def fitness(pos):
    r, s_r = solve_radii(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = parent1.copy()
    for i in range(len(child)):
        if random.random() < rate:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    for i in range(len(individual)):
        if random.random() < rate:
            individual[i] += sigma * random.gauss(0, 1)
            individual[i] = max(0, min(1, individual[i]))
    return individual

def tournament(pop, fitness_vals, k):
    best = random.choice(pop)
    best_val = fitness_vals[np.where(np.array(list(pop.keys())) == best)[0][0]]
    for _ in range(1, k):
        candidate = random.choice(pop)
        candidate_val = fitness_vals[np.where(np.array(list(pop.keys())) == candidate)[0][0]]
        if candidate_val > best_val:
            best = candidate
            best_val = candidate_val
    return best
