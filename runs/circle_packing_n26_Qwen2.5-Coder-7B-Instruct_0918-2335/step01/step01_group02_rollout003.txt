import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform
from scipy.stats import norm
import random

np.random.seed(42)

def solve_lradii(pos):
    r = np.empty(26)
    for i in range(26):
        dists = np.sqrt(np.sum((pos[i] - pos[:i, :])**2, axis=1))
        b_constraints = [
            np.hstack(([1], dists - (0.5(r[i] + r[:i]) + 1e-12))),
            np.hstack(([1], dists - (0.5(r[i] + r[:i]) + 1e-12))),
            np.hstack(([1], dists - (0.5(r[i] + r[:i]) + 1e-12))),
            np.hstack(([1], dists - (0.5(r[i] + r[:i]) + 1e-12)))
        ]
        res = linprog(c=-np.ones(26), A_ub=b_constraints, b_ub=np.zeros(26), method='highs')
        r[i] = -res.fun
    return r, r.sum()

def fitness(pos):
    r, s_r = solve_lradii(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = np.empty_like(parent1)
    for i in range(52):
        if np.random.rand() < rate:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    for i in range(52):
        if np.random.rand() < rate:
            individual[i] += np.random.normal(0, sigma)
            individual[i] = max(0, min(1, individual[i]))
    return individual

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(len(pop), k, replace=False)
    best = indices[0]
    for i in indices:
        if fitness_vals[i] < fitness_vals[best]:
            best = i
    return best

def ga_best_positions():
    POPSIZE = 80
    GENS = 250
    ELITE_SIZE = int(0.1 * POPSIZE)
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.2
    MUTATION_SIGMA = 0.03

    pop = np.random.rand(POPSIZE, 52)
    fitness_vals = np.array([fitness(individual) for individual in pop])

    for _ in range(GENS):
        elite_indices = np.argsort(fitness_vals)[:ELITE_SIZE]
        elite_population = pop[elite_indices, :]
        elite_fitness = fitness_vals[elite_indices]

        children_indices = np.arange(POPSIZE)
        np.random.shuffle(children_indices)
        children_fitness = np.empty(POPSIZE)

        for i in range(ELITE_SIZE):
            children_fitness[children_indices[i]] = elite_fitness[i]

        for j in range(ELITE_SIZE, POPSIZE):
            parent1_idx = tournament(pop, fitness_vals, 3)
            parent2_idx = tournament(pop, fitness_vals, 3)
            child = crossover(pop[parent1_idx, :], pop[parent2_idx, :], CROSSOVER_RATE)
            child = mutate(child, MUTATION_RATE, MUTATION_SIGMA)
            children_fitness[children_indices[j]] = fitness(child)

        pop = pop[children_indices, :]
        fitness_vals = children_fitness

    elite_indices = np.argsort(fitness_vals)[:ELITE_SIZE]
    best_pos = pop[elite_indices[0], :]
    return best_pos

def polish_solution(pos):
    x0 = np.hstack((pos, solve_lradii(pos)[0]))
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    res = minimize(fun=fitness, x0=x0, bounds=bounds, method='SLSQP', options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    return res.x[:52].reshape(26, 2), res.x[52:], -res.fun

def run_packing():
    best_pos = ga_best_positions()
    centers, radii, sum_r = polish_solution(best_pos)
    return validate_packing(centers, radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or 
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return centers, radii, transform_to_sum(centers, radii)
