import numpy as np
from scipy.optimize import minimize, linprog

def lp_oracle(centers):
    n = centers.shape[0]
    A_ub = np.empty((4*n + n*(n-1)//2, 4*n))
    b_ub = np.empty(4*n + n*(n-1)//2)
    
    for i in range(n):
        A_ub[i, 4*i:4*(i+1)] = [1, -1, 0, 0]
        b_ub[i] = 0
        
        A_ub[i+n, 4*i:4*(i+1)] = [-1, 1, 0, 0]
        b_ub[i+n] = 1 - centers[i, 0]
        
        A_ub[i+2*n, 4*i:4*(i+1)] = [0, 0, 1, -1]
        b_ub[i+2*n] = 0
        
        A_ub[i+3*n, 4*i:4*(i+1)] = [0, 0, -1, 1]
        b_ub[i+3*n] = 1 - centers[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[4*n + (i*n + j), 4*i:4*(i+1)] = 1
            A_ub[4*n + (i*n + j), 4*j:4*(j+1)] = 1
            A_ub[4*n + (i*n + j), 2*i:2*(i+1)] = -1
            A_ub[4*n + (i*n + j), 2*j:2*(j+1)] = -1
            b_ub[4*n + (i*n + j)] = 0
    
    res = linprog(np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': 1e-9})
    if res.status != 0:
        return np.full(n, -1e6)
    return res.x

def initial_swarm(num_particles):
    seeds = [
        np.array([[0, 0], [1, 0], [1, 1], [0, 1]]),
        np.array([[0.5, 0], [0.5, 1], [0, 0.5], [1, 0.5]]),
        np.array([[0.5, 0.5], 0.229 * np.array([(np.cos(k * np.pi / 3), np.sin(k * np.pi / 3)) for k in range(6)]))),
        np.array([[0.5, 0], [0.5, 1], [0, 0.5], [1, 0.5]]),
        10 * np.random.rand(40, 2) - 0.5,
        np.array([[0.5, 0.5], [0.23, 0.23], 0.2 * np.array([(np.cos(k * np.pi / 3) * 0.5 + 0.5, np.sin(k * np.pi / 3) * 0.5 + 0.5) for k in range(3)])]),
        np.array([[0.5, 0.5], 0.27 * np.array([(np.cos(k * np.pi / 3), np.sin(k * np.pi / 3)) for k in range(6)])]),
        np.array([[0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75]]),
        10 * ((np.random.rand(40, 2) - 0.5) / 2.0)
    ]
    seeds = np.array(seeds).reshape(-1, 26, 2)
    seeds = np.random.randn(num_particles, 2) * 0.01 + seeds
    seeds = np.clip(seeds, 0, 1)
    return seeds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_particles = 40
    swarm = initial_swarm(num_particles)
    global_best_pos = swarm[0]
    global_best_val = np.full(num_particles, -1e6)
    inertia = 0.6
    cognitive = 1.5
    social = 1.5
    
    for gen in range(1000):
        for i in range(num_particles):
            velocity = inertia * swarm[i, 26:] + cognitive * np.random.rand(26) - (1 - cognitive) * swarm[i, 26:]
            velocity = np.clip(velocity, -0.03, 0.03)
            if np.random.rand() < 0.1:
                velocity = np.random.rand(26) * 0.04 - 0.02
            swarm[i, 26:] = velocity
            swarm[i, :26] += velocity
            swarm[i, :26] = np.clip(swarm[i, :26], 0, 1)
            radii = lp_oracle(swarm[i, :26].reshape(26, 2))
            if (radii < 0).any():
                swarm[i, :26] = global_best_pos
                continue
            fitness = np.sum(radii)
            global_best_pos = swarm[i, :26] if fitness > global_best_val[i] else global_best_pos
            global_best_val[i] = fitness if fitness > global_best_val[i] else global_best_val[i]
        
        inertia *= 0.995
        cognitive *= 0.995
        social *= 0.995
    
    swarm = np.argmin(global_best_val)
    positions = global_best_pos[0]
    radii = lp_oracle(positions.reshape(26, 2))
    return positions.reshape(26, 2), radii, np.sum(radii)
