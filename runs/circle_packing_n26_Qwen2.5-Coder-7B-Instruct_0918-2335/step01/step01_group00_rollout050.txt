import numpy as np
from scipy.optimize import minimize, linprog, Bounds

def make_d_matrix(centers):
    return np.sqrt(np.sum((centers[:,None,:] - centers[None,:,:])**2, axis=2))

def lp_value_and_radii(centres):
    w_limits = np.minimum.reduce([centres[:,0], 1-centres[:,0], centres[:,1], 1-centres[:,1]])
    D = make_d_matrix(centres)
    Aub = np.ones((len(D)**2, len(D)))
    np.fill_diagonal(Aub, 0)
    b_ub = D[np.triu_indices(len(D), k=1)]

    result = linprog(-np.ones(len(centres)), A_ub=Aub, b_ub=b_ub, bounds=Bounds(0, w_limits), method='highs')
    
    if result.status < 0:
        return -np.inf
    else:
        return -result.fun

def initial_layout(size=26):
    radius = np.sqrt(2)/4
    distance = np.sqrt(3)*radius
    center_distance = 2*distance
    half_length = center_distance * (size // 2)
    side_length = 2 * half_length + radius
    centers = np.zeros((size, 2))
    
    i = 0
    for r in range(size // 2):
        for c in range(r % 2, size, 2):
            centers[i] = np.array([half_length + c * center_distance, half_length + r * distance])
            if i == size - 1:
                break
            i += 1

    jitter = np.random.rand(size, 2) * radius
    centers[:size] += jitter

    return centers

def run_packing():
    nrestarts = 150
    best_centers = None
    best_sum = None
    
    for restart in range(nrestarts):
        initial_centers = initial_layout()
        current_centers = initial_centers.copy()
        current_sum = 0
        
        temp = 0.1
        for _ in range(2000):
            if temp > 1e-12:
                move_circle = np.random.choice(len(current_centers))
                delta = np.random.randn(2) * temp
                proposed_centers = current_centers.copy()
                proposed_centers[move_circle] += delta
                
                for j in range(len(proposed_centers)):
                    proposed_centers[j, 0] = max(0, min(1, proposed_centers[j, 0]))
                    proposed_centers[j, 1] = max(0, min(1, proposed_centers[j, 1]))
                
                proposed_sum = lp_value_and_radii(proposed_centers)
                
                if proposed_sum >= current_sum or np.random.rand() < np.exp((proposed_sum - current_sum) / temp):
                    current_centers = proposed_centers
                    current_sum = proposed_sum
                    
            temp *= 0.97
        
        if best_sum is None or current_sum > best_sum:
            best_centers = current_centers
            best_sum = current_sum
    
    final_sum = lp_value_and_radii(best_centers)
    result = minimize(lambda x: -lp_value_and_radii(x.reshape(26, 2)), best_centers.flatten(),
                     bounds=Bounds(0, 1), method='BFGS')
    
    final_centers = result.x.reshape(26, 2)
    final_sum = lp_value_and_radii(final_centers)
    
    return final_centers, best_centers.ravel()[:26], final_sum
