import numpy as np
from scipy.optimize import minimize, linprog, Bounds
from scipy.spatial import distance_matrix
from scipy.stats import truncnorm

def compute_d_matrix(centers):
    return distance_matrix(centers, centers)

def lp_value_and_radii(centers):
    n = centers.shape[0]
    upper = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    Aub = np.eye(2*n)
    for i in range(n):
        for j in range(i+1, n):
            Aub[2*i, j*2] = 1
            Aub[2*i, j*2+1] = 1
            Aub[2*i+1, j*2] = 1
            Aub[2*i+1, j*2+1] = 1
    b_ub = distance_matrix(centers, centers).flatten()
    c = -np.ones(26)
    res = linprog(c, A_ub=Aub, b_ub=b_ub, bounds=(0, upper))
    if res.success:
        return res.fun, res.x.reshape(-1, 2)[:, 0]
    else:
        return -np.inf, np.zeros(26)

def generate_initial_layout():
    centers = np.zeros((26, 2))
    radius = 0.2
    centers[0:5] = radius + 0.2 * np.array([[1, 0], [0, 1], [-1, 0], [0, -1], [0, 0]])
    centers[5:10] = radius / 2 + 0.2 * np.array([[1, 0], [1/np.sqrt(2), 1/np.sqrt(2)], [-1/np.sqrt(2), 1/np.sqrt(2)], 
                                               [-1/np.sqrt(2), -1/np.sqrt(2)], [1/np.sqrt(2), -1/np.sqrt(2)], [0, 1], [0, -1]])
    centers[10:15] = radius / 3 + 0.2 * np.array([[1, 0], [1/np.sqrt(3), 1/np.sqrt(3)], [-1/np.sqrt(3), 1/np.sqrt(3)], 
                                               [-1/np.sqrt(3), -1/np.sqrt(3)], [1/np.sqrt(3), -1/np.sqrt(3)], [0, 1], [0, -1]])
    centers[15:26] = radius / 4 + 0.2 * np.random.rand(11, 2)
    return centers + np.random.normal(0, 0.05, (26, 2))

def run_packing():
    nrestarts = 150
    max_iterations = 2000
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(nrestarts):
        centers = generate_initial_layout()
        temp = 0.1
        for it in range(max_iterations):
            j = np.random.randint(n)
            move = np.random.normal(0, temp, 2)
            new_centers = centers.copy()
            new_centers[j] += move
            new_centers[j] = np.clip(new_centers[j], [0, 0], [1, 1])
            value, radii = lp_value_and_radii(new_centers)
            if value > best_sum:
                best_sum = value
                best_centers = new_centers
                best_radii = radii
            else:
                if np.random.rand() < np.exp((value - best_sum) / temp):
                    best_sum = value
                    best_centers = new_centers
                    best_radii = radii
            temp *= 0.97

    final_value = -np.inf
    final_centers = best_centers
    final_radii = best_radii
    for _ in range(2):
        result = minimize(lambda x: -lp_value_and_radii(x.reshape(n, 2))[0], best_centers.flatten(), method='BFGS', bounds=Bounds(np.zeros(2*n), np.ones(2*n)))
        if result.success and -result.fun > final_value:
            final_value = -result.fun
            final_centers = result.x.reshape(n, 2)
            final_radii = lp_value_and_radii(final_centers)[1]

    return final_centers, final_radii, -final_value
