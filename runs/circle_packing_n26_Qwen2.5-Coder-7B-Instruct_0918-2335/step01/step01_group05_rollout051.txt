import numpy as np
from scipy.optimize import linprog, Bounds
import scipy.spatial

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + 325, n))
    b_ub = np.zeros(4*n + 325)
    diag = np.eye(n)
    
    # Boundary constraints
    for i in range(n):
        A_ub[4*i] = diag[i]
        b_ub[4*i] = centers[i, 0]
        A_ub[4*i+1] = diag[i]
        b_ub[4*i+1] = 1 - centers[i, 0]
        A_ub[4*i+2] = diag[i]
        b_ub[4*i+2] = centers[i, 1]
        A_ub[4*i+3] = diag[i]
        b_ub[4*i+3] = 1 - centers[i, 1]
    
    # Pairwise constraints
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    for k, (i, j) in enumerate(pairs):
        A_ub[4*n + k, i] = 1
        A_ub[4*n + k, j] = 1
        d_i_j = np.linalg.norm(centers[i] - centers[j])
        b_ub[4*n + k] = d_i_j
    
    return linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')

def evaluate_positions(centers):
    r_opt, sum_radii = solve_lp(centers)
    if r_opt.success:
        return centers, r_opt.x, sum_radii.fun
    return None, None, -np.inf

def generate_grid_candidates(pairs, jitter_size=0.02):
    gridpoints = np.array([(0.2*i, 0.2*j) for i in range(1, 5) for j in range(1, 5)])
    center = np.array([0.5, 0.5])
    candidates = np.vstack((gridpoints, center))
    np.random.shuffle(candidates)
    jittered = candidates + np.random.uniform(-jitter_size, jitter_size, size=candidates.shape)
    return np.clip(jittered, 0, 1, out=jittered)

def generate_hex Candidates(p, jitter_size=0.02):
    h = 0.18
    v = 0.3 * h
    n_points = 3
    rows = [(n+1) * v for n in range(n_points)]
    centers = np.array([[0.5, row] if n % 2 == 0 else [row / h, 0.5] for row in rows])
    center = np.array([0.5, 0.5])
    candidates = np.vstack((centers, center))
    numpy.shufle(candidates)
    jittered = candidates + np.random.uniform(-jitter_size, jitter_size, size=candidates.shape)
    return np.clip(jittered, 0, 1, out=jittered)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 5000
    sigma_factor = 0.9995
    sigma_start = 0.05
    temperature_start = 0.1

    best_centers = None
    best_sum_radii = -np.inf

    initial_candidates = [
        generate_grid_candidates(pairs, 0.02),
        generate_hex_candidates(pairs, 0.02)
    ]

    for candidates in initial_candidates:
        for centers in candidates:
            centers, radii, sum_radii = evaluate_positions(centers)
            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = centers
                best_radii = radii

    current_centers = best_centers
    current_sum_radii = best_sum_radii
    current_radii = best_radii
    temperature = temperature_start

    for it in range(max_iter):
        sigma = sigma_start * (sigma_factor ** it)

        i = np.random.randint(n)
        dx, dy = np.random.normal(0, sigma, size=2)
        new_centers = np.clip(np.array(current_centers), 0, 1)
        new_centers[i] += np.array([dx, dy])

        new_sum_radii = evaluate_positions(new_centers)[2]

        if new_sum_radii > current_sum_radii:
            current_centers = new_centers
            current_sum_radii = new_sum_radii
            if new_sum_radii > best_sum_radii:
                best_sum_radii = new_sum_radii
                best_centers = new_centers
                best_radii = current_radii
        else:
            if np.random.rand() < np.exp((new_sum_radii - current_sum_radii) / temperature):
                current_centers = new_centers
                current_sum_radii = new_sum_radii

        temperature *= sigma_factor

    return best_centers, best_radii, best_sum_radii
