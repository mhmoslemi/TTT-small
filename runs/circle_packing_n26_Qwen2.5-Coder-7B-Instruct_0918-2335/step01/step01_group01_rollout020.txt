import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

POP = 30
EVAL = 4000

def lp_opt(p):
    n = 26
    u = np.triu(np.ones((n, n)), k=1)
    A = np.vstack((p[:,None] - p[None,:], p[:,None] + p[None,:]))
    A = A[u.flatten() == 1]
    b = np.hstack((np.ones(n) - p[:,0], 1 - p[:,0], np.ones(n) - p[:,1], 1 - p[:,1]))
    objective = np.hstack((np.ones(n), np.zeros(n)))
    
    res = linprog(objective, A_ub=A.T, b_ub=b, bounds=[(0, None), (0, None)] * n, method='highs')
    radii = res.x[:n]
    peaks = np.isclose(res.fun, 0, atol=1e-12)
    if np.any(peaks):
        radii[peaks] = 1e-12
    return radii

def fitness(p):
    radii = lp_opt(p)
    return -np.sum(radii)

def tournament_select(pop):
    indices = np.random.choice(POP, 3, replace=False)
    scores = np.array([ind[1] for ind in pop[indices]])
    winner_index = indices[np.argmax(scores)]
    return pop[winner_index][0]

def crossover(A, B):
    p1 = 0.5 * A + 0.5 * B + np.random.uniform(-0.005, 0.005, A.shape)
    return np.clip(p1, 0, 1)

def mutate(p):
    m = np.random.randint(1, 4)
    idxs = np.random.choice(26, m, replace=False)
    noise = np.random.normal(0, 0.02, m * 2).reshape(m, 2)
    p[idxs] += noise
    p = np.clip(p, 0, 1)
    return p

def local_slsqp(p):
    inequality_constraint = {'type': 'ineq', 'fun': lambda p: lp_opt(p) - fitness(p)}
    bounds = [(0, 1)] * 26
    res = minimize(fitness, p[:26], method='SLSQP', bounds=bounds, constraints=inequality_constraint, options={'maxiter': 25, 'disp': False})
    return res.x

def run_packing():
    np.random.seed(42)
    
    initial_population = []
    initial_population.extend(np.reshape(np.random.rand(4, 52), (4, 26, 2)))
    initial_population.extend(np.reshape(np.random.rand(10, 2), (10, 26, 2)))
    initial_population.append(np.reshape(np.array([[i/6, j/5] for i in range(5) for j in range(5)] + [[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.5, 0.5]]), (6, 26, 2)))
    initial_population.extend(np.reshape(np.random.rand(10, 2), (10, 26, 2)))
    
    for p in initial_population:
        radii = lp_opt(p)
        r_sum = np.sum(radii)
        initial_population.append((p.flatten(), r_sum))
    
    best_ind = max(initial_population, key=lambda ind: ind[1])
    p = best_ind[0]
    fitness_value = best_ind[1]
    
    for _ in range(EVAL):
        A = tournament_select(initial_population)
        B = tournament_select(initial_population)
        C = crossover(A, B)
        mutate(C)
        local_slsqp(C)
        r_sum = lp_opt(C)
        if r_sum > fitness_value:
            fitness_value = r_sum
            p = C
    
    best_radii = lp_opt(p)
    best_sum = np.sum(best_radii)
    return p.reshape(26, 2), best_radii, best_sum
