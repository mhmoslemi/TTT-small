import numpy as np
from scipy.optimize import linprog, minimize

def optimise_radii(centers):
    n = len(centers)
    bounds = [(0, None) for _ in range(n)] + [(0, None) for _ in range(n)]
    lin_bounds = []

    for i in range(n):
        r_i = 2 * np.array([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        lin_bounds.extend(r_i.tolist())

    A_eq = []
    b_eq = []

    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            A_eq.append([2, 2, 0, 0, 0, 0, -1 * 2, -1 * 2])
            b_eq.append(d_ij)

    A_ub = []
    b_ub = []

    for i in range(n):
        r_i = 2 * np.array([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        A_ub.append(r_i.tolist())
        b_ub.append(r_i.max())

    linprog_result = linprog(c=[-1 for _ in centers[:n]] + [-2 for _ in centers[n:]], A_ub=A_ub, b_ub=b_ub,
                             A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if linprog_result.success:
        return np.concatenate((linprog_result.x[:n], linprog_result.x[n:] * 0.5))
    else:
        return None

initial_start = np.array([
    (0.2, 0.2), (0.2, 0.4), (0.2, 0.6), (0.2, 0.8), (0.2, 1),
    (0.4, 0.2), (0.4, 0.4), (0.4, 0.6), (0.4, 0.8), (0.4, 1),
    (0.6, 0.2), (0.6, 0.4), (0.6, 0.6), (0.6, 0.8), (0.6, 1),
    (0.8, 0.2), (0.8, 0.4), (0.8, 0.6), (0.8, 0.8), (0.8, 1),
    (1, 0.2), (1, 0.4), (1, 0.6), (1, 0.8), (1, 1),
    (0.5, 0.5)  # Centre disc
])

delta_max = 0.05
T0 = 1.0
alpha = 0.995
NITER = 50000

best_sum = 0
best_centres = np.zeros((26, 2))
best_radii = np.zeros(26)

current_centres = initial_start
current_radii = optimise_radii(current_centres)

if current_radii is not None:
    best_sum = np.sum(current_radii)
    best_centres, best_radii = current_centres, current_radii

for iter in range(NITER):
    i = np.random.randint(0, 26)
    delta_x, delta_y = np.random.uniform(-delta_max, delta_max, 2)
    new_coors = current_centres[i] + np.array([delta_x, delta_y])

    if (0 <= new_coors[0] <= 1) and (0 <= new_coors[1] <= 1):
        temp_centres = current_centres.copy()
        temp_centres[i] = new_coors

        temp_radii = optimise_radii(temp_centres)

        if temp_radii is not None:
            new_sum = np.sum(temp_radii)
            delta_S = new_sum - best_sum

            if delta_S > 0:
                current_centres, current_radii, best_sum = temp_centres, temp_radii, new_sum
                if new_sum > best_sum:
                    best_centres, best_radii = temp_centres, temp_radii
            else:
                T = T0 * alpha ** iter
                if np.random.rand() < np.exp(delta_S / T):
                    current_centres, current_radii = temp_centres, temp_radii

best_centres, best_radii = minimize(
    lambda x: -np.sum(x[52:]), x0=np.concatenate((best_centres.flatten(), best_radii)), method='SLSQP',
    bounds=[(0, 1)] * (2 * 26) + [(0, 0.5)] * 26,
    constraints={'type': 'ineq', 'fun': lambda x: np.array([np.linalg.norm(x[2 * i:2 * i + 2] - x[2 * j:2 * j + 2]) - (x[52 + i] + x[52 + j]) for i in range(26) for j in range(i + 1, 26)])},
    options={'maxiter': 2000, 'ftol': 1e-9}
)
best_sum = -best_radii.sum()

return best_centres, best_radii, best_sum
