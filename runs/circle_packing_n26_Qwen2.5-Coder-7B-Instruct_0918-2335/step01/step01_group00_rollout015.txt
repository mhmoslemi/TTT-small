import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping, NonlinearConstraint, Bounds
from scipy.spatial.distance import cdist


N = 26
NUM_VARS = 26 * 2 + 26


def obj(v):
    return -np.sum(v[52:])


def build_constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]

    # Boundary constraints
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r

    # Pairwise separation constraints (squared distances)
    diffs = pos[:, None, :] - pos[None, :, :]
    dist2 = np.sum(diffs ** 2, axis=2)
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dist2[mask] - (r[:, None] + r[None, :])[mask] ** 2
    return np.concatenate([cons, pair_cons])


def inflate_radii(centers, radii):
    max_iter = 100
    tol = 1e-8
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(26):
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            for j in range(26):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii


def run_packing(validation_function=None) -> tuple[np.ndarray, np.ndarray, float]:
    if validation_function is None:
        raise ValueError('Validation function must be provided')

    bounds = Bounds(0, 1)
    bounds_radius = Bounds(0, 0.5)

    def initial_pos():
        centers = np.random.rand(N, 2)
        # Ensure minimal distance to boundaries
        centers[:, :1] = np.clip(centers[:, :1], 0.15, 0.85)
        centers[:, 1:2] = np.clip(centers[:, 1:2], 0.15, 0.85)
        return centers

    def initial_radius():
        radii = 0.1 * np.ones(N)
        return radii

    def initial_state():
        pos = initial_pos().reshape(1, -1)
        r = initial_radius().reshape(1, -1)
        return np.concatenate((pos, r), axis=0)

    def penalized_obj(x):
        return obj(x) + 1e-4 * np.sum(np.sum(np.square(build_constraints(x).reshape(-1, 2)), axis=1))

    def penalized_filter(x):
        cons = build_constraints(x)
        if np.any(cons < -1e-10):
            return False
        return True

    def penalized_constraint(x):
        return NonlinearConstraint(build_constraints, lb=-1e-10, ub=None, jac='3-point')

    def penalty_constraint(v):
        return 1e-4 * np.sum(np.square(build_constraints(v).reshape(-1, 2)), axis=1)

    def take_step(v, stepsize):
        stepsize = np.clip(stepsize, 0, 1)
        v += stepsize * np.random.randn(*v.shape)
        return v

    for seed in range(100):
        x0 = initial_state()
        try:
            res = differential_evolution(penalized_obj, bounds, penalty_constraint, maxiter=4000)
        except:
            print('DE failed')
            continue

        if penalized_filter(res.x):
            best_x = res.x

            # Global refinement with basinhopping
            res_final = basinhopping(penalized_obj, best_x, stepsize=0.1, take_step=take_step, minimizer_kwargs=dict(bounds=bounds_radius, constraint=penalized_constraint), minimizer_options={'maxiter': 1000}, niter=10)
            if penalized_filter(res_final.x):
                best_x = res_final.x

            # Inflation radii
            centers = best_x[:52].reshape(N, 2)
            radii = inflate_radii(centers, best_x[52:])
            r = radii
            x = np.concatenate((centers, r[:, np.newaxis]), axis=1)

            # Repeated refinement with SLSQP
            res_local = minimize(obj, x, bounds=bounds_radius, constraints=build_constraints(x),
                options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
            centers = res_local.x[:52].reshape(N, 2)
            radii = inflate_radii(centers, res_local.x[52:])
            if validation_function(centers, radii)[0]:
                return centers.flatten(), radii.flatten(), -res_local.fun

    raise ValueError("Optimization failed to find a valid solution")
