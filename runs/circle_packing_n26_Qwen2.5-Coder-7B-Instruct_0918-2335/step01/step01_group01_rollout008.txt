def compute_voronoi_radius(voronoi, points, centers):
    radii = np.zeros(len(centers))
    for i in range(len(centers)):
        center = centers[i]
        region = voronoi.regions[voronoi.point_region[i]]
        if -1 not in region:
            polygon = voronoi.vertices[region]
            dists = np.linalg.norm(polygon - center, axis=1)
            max_dist = max(dists)
            min_dist = min(dists)
            mid_dist = (max_dist + min_dist) / 2
            candidate_radius = mid_dist - min_dist
            if candidate_radius > 0:
                radii[i] = candidate_radius
    return radii

def lp_for_radii(centers):
    n = centers.shape[0]
    A_ub, b_ub = np.zeros((0, n)), np.zeros(0)
    for i in range(n):
        A_ub = np.vstack((A_ub, -np.ones(n)))
        b_ub = np.append(b_ub, 0)
        A_ub = np.vstack((A_ub, np.eye(n)[i]))
        b_ub = np.append(b_ub, max(centers[i, 0] - 1e-6, centers[i, 1] - 1e-6))
        A_ub = np.vstack((A_ub, -np.eye(n)[i]))
        b_ub = np.append(b_ub, 0)
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub = np.vstack((A_ub, np.eye(n)[i] + np.eye(n)[j]))
            b_ub = np.append(b_ub, dist + 1e-6)
    
    A_eq, b_eq = np.zeros((n, n)), np.zeros(n)
    for i in range(n):
        A_eq[i, i] = 1
    
    result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if result.success:
        return -result.fun, result.x
    else:
        return -1e7, np.zeros(n)

def greedy_seeding(radius):
    success = False
    while not success:
        start = np.random.rand(10, 2)
        success = True
        for i in range(10):
            if not (start[i, 0] - radius < 0 or start[i, 0] + radius > 1 or
                    start[i, 1] - radius < 0 or start[i, 1] + radius > 1):
                close = False
                for j in range(i+1, 10):
                    if np.linalg.norm(start[i] - start[j]) < 2 * radius:
                        close = True
                        break
                if close:
                    success = False
                    break
                    
    return start
