import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers) -> np.ndarray:
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    return np.sqrt((diffs ** 2).sum(axis=2))

def solve_radii_lp(centers) -> np.ndarray:
    n = centers.shape[0]
    A_ineq = np.zeros((2 * n + n * (n - 1) // 2, n))
    b_ineq = np.zeros(A_ineq.shape[0])
    
    for i in range(n):
        A_ineq[i, i] = 1
        A_ineq[i + n, i] = -1
        A_ineq[i + 2 * n, i] = -1
        
        for j in range(i + 1, n):
            dist = pairwise_dists(np.array([centers[i], centers[j]]))[0][1]
            k = n + i * (n - 1) // 2 + j - i - 1
            A_ineq[k, i] = 1
            A_ineq[k, j] = -1
            b_ineq[k] = dist
    
    res = linprog(-np.ones(n), A_ub=A_ineq, b_ub=b_ineq, bounds=(0, None), method='highs')
    if not res.success:
        raise ValueError("Linear programming failed")
    return res.x

def sa_optimize(initial_centers) -> Tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    best_radii = None
    best_sum_radii = -np.inf
    
    for _ in range(500):
        new_centers = initial_centers.copy()
        index = np.random.randint(n)
        delta = np.random.randn(2) * 1e-4
        new_centers[index] += delta
        new_centers = np.clip(new_centers, 0, 1)
        
        radii = solve_radii_lp(new_centers)
        sum_radii = np.sum(radii)
        
        if sum_radii > best_sum_radii or (np.isclose(sum_radii, best_sum_radii) and np.random.rand() < 0.1):
            best_centers = new_centers
            best_radii = radii
            best_sum_radii = sum_radii
    
    return best_centers, best_radii
