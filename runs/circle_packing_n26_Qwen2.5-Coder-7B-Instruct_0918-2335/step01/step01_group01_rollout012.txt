import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def lp_oracle(positions):
    n = len(positions)
    equality_constraints = []
    inequality_constraints = []
    coefficients = np.ones(n)

    for i in range(n):
        inequality_constraints.append({'type': 'ineq', 'fun': lambda p, i=i, r=positions[i, 0]: p[i] - r})
        inequality_constraints.append({'type': 'ineq', 'fun': lambda p, i=i, r=1 - positions[i, 0]: r - p[i]})
        inequality_constraints.append({'type': 'ineq', 'fun': lambda p, i=i, r=positions[i, 1]: p[i] - r})
        inequality_constraints.append({'type': 'ineq', 'fun': lambda p, i=i, r=1 - positions[i, 1]: r - p[i]})

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(positions[i] - positions[j])
            inequality_constraints.append({'type': 'ineq', 'fun': lambda p, i=i, j=j, d=dist: p[i] + p[j] - d})

    result = minimize(
        lambda p: -np.sum(p),
        positions.flatten(),
        method='SLSQP',
        bounds=[(0, 1)] * positions.size,
        constraints=inequality_constraints,
        options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not result.success or result.fun != result.fun or np.isnan(result.fun):
        return 1e6
    
    radii = np.maximum(0, result.x).reshape(n)
    return np.sum(radii)

def pso_sa_optimize(initial_positions):
    NP, max_generations = 40, 1000
    inertia, cognitive, social = 0.6, 1.5, 1.5
    T0, cooling_factor, mutation_rate = 0.04, 0.995, 0.1
    velocity_clamp = 0.03
    mutation_std = 0.02

    positions = initial_positions.copy()
    velocities = np.random.randn(NP, 52)
    pbest_positions = positions.copy()
    pbest_vals = lp_oracle(positions).reshape(NP)
    gbest_position = pbest_positions[np.argmax(pbest_vals)]
    gbest_value = np.max(pbest_vals)

    for t in range(max_generations):
        for i in range(NP):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest_positions[i] - positions[i]) + social * r2 * (gbest_position - positions[i])
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            positions[i] += velocities[i]
            
            if np.random.rand() < gbest_value / pbest_vals[i]:
                velocities[i] *= -1
            
            positions[i] = np.clip(positions[i], 0, 1)
            
            value = lp_oracle(positions[i].reshape(26, 2))
            if value > pbest_vals[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_vals[i] = value
        
        gbest_value = np.max(pbest_vals)
        gbest_position = pbest_positions[np.argmax(pbest_vals)]

        # Annealing step
        T = T0 * cooling_factor ** t
        for i in range(NP):
            if np.random.rand() < np.exp((gbest_value - pbest_vals[i]) / T):
                velocities[i] = np.random.randn(52) * mutation_std  # Gaussian perturbation

        if t % 20 == 0:
            print(f"Generation {t}: Best value {gbest_value}, Current value {pbest_vals[gbest_value.argmax()]}")

    return gbest_position, lp_oracle(gbest_position.reshape(26, 2)), gbest_value

def final_polish(centers):
    radii = lp_oracle(centers).reshape(26)
    return centers, radii, np.sum(radii)

def run_packing():
    initial_positions = generate_initial_positions()
    best_position, best_radii, best_sum = pso_sa_optimize(initial_positions)
    best_position, final_radii, final_sum = final_polish(best_position)
