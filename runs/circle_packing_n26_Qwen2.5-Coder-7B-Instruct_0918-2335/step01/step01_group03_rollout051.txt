import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    bounds = [(0, 1)] * (3 * n)
    centers prototype = np.array([
        (0.5, 0.5),
        (0.1, 0.1),
        (0.9, 0.1),
        (0.5, 0.1),
        (0.9, 0.5),
        (0.1, 0.9),
        (0.5, 0.9),
        (0.9, 0.9),
        (0.1, 0.5),
        (0.3, 0.3),
        (0.7, 0.3),
        (0.9, 0.5),
        (0.7, 0.7),
        (0.3, 0.7),
        (0.1, 0.9),
        (0.5, 0.3),
        (0.9, 0.3),
        (0.5, 0.9),
        (0.1, 0.3),
        (0.3, 0.1),
        (0.7, 0.1),
        (0.9, 0.1),
        (0.1, 0.7),
        (0.3, 0.7),
        (0.7, 0.7),
        (0.9, 0.7),
    ])
    
    def edge_dist(x, y):
        dist_x = min(x, 1-x)
        dist_y = min(y, 1-y)
        return max(dist_x, dist_y)
    
    def speed_assignment(centers):
        corner_radius = np.array([0.2] * 4 + [0.15] * 8 + [0.1] * 14)
        dist_to_edges = np.array([edge_dist(x, y) for x, y in centers])
        speed = 1 / (dist_to_edges + 1e-6)
        speed[0:4] = speed[0:4] * (corner_radius / 0.2)
        speed = np.clip(speed, 1, 100)
        np.random.shuffle(speed)
        return speed
    
    def simulate_event(centers, speeds):
        dist_matrix = cdist(centers, centers)
        active = np.arange(n)
        stops = np.zeros(n)
        
        while len(active) > 0:
            t_pairs = dist_matrix[active][:, active] / (speeds[active][:, None] + speeds[active][None, :])
            t_edges = np.min(np.column_stack((centers[:, 0] - 0.5, centers[:, 0] - 0.5, centers[:, 1] - 0.5, centers[:, 1] - 0.5)), axis=1) / speeds[active]
            
            t_min = np.min(np.minimum(t_pairs[t_pairs != 0], t_edges))
            indices = np.where((t_pairs == t_min) | (t_edges == t_min))[0]
            
            for i in indices:
                stops[i] = t_min
                active = active[active != i]
        
        radii = speeds * stops
        return radii
    
    def simulate_and_refine():
        max_radius = 0.5
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        best_sum_radius = 0
        
        for _ in range(4000):
            speeds = speed_assignment(centers)
            radii = simulate_event(centers, speeds)
            
            for _ in range(2):
                res = minimize(lambda v: -np.sum(v[52:]), np.concatenate([centers.flatten(), radii]), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda v: np.concatenate([0.5 * bounds[i][1] - v[i] for i in range(78)]) - np.concatenate([-0.5 * bounds[i][0] - v[i] for i in range(78)])})
                if res.success:
                    centers_new = res.x[:52].reshape(n, 2)
                    radii_new = res.x[52:]
                    sum_radius = np.sum(radii_new)
                    if sum_radius > best_sum_radius:
                        best_centers = centers_new
                        best_radii = radii_new
                        best_sum_radius = sum_radius
        
        return best_centers, best_radii, best_sum_radius
    
    return simulate_and_refine()
