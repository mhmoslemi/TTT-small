import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    n = centers.shape[0]
    M = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + M, n))
    for i in range(n):
        A_ub[i, i] = -1
        A_ub[n + i, i] = 1
        A_ub[2 * n + i, i] = -1
        A_ub[3 * n + i, i] = 1
    rows = np.arange(n)
    cols = rows[:, None] + 1
    mask = rows < cols
    A_ub[4 * n:4 * n + M, rows[mask]] = -1
    A_ub[4 * n:4 * n + M, cols[mask]] = -1
    b_ub = np.concatenate((np.zeros(n), 2 * np.ones(n),(np.sqrt((centers[rows, 0] - centers[cols, 0]) ** 2 + (centers[rows, 1] - centers[cols, 1]) ** 2)).flatten()))
    r_lower = np.zeros(n)
    r_upper = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1]))
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=list(zip(r_lower, r_upper)), method='highs')
    if not res.success:
        return -np.inf, np.zeros(n)
    return -res.fun, -res.x
