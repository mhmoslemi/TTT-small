import numpy as np
from scipy.optimize import linprog, minimize

# Parameters
pop_size = 48
generations = 180
crossover_prob = 0.9
mutation_rate = 0.8
init_sigma = 0.05
sigma_decay = 0.99
elitism = 5
tournament_size = 5

def solve_radii(centers):
    n = len(centers)
    A_ub = np.empty((4*n, n))
    A_ub[:n, :] = np.eye(n)
    A_ub[n:2*n, :] = -np.eye(n)
    A_ub[2*n:3*n, :] = np.column_stack((np.ones(n), np.zeros(n)))
    A_ub[3*n:4*n, :] = np.column_stack((np.zeros(n), np.ones(n)))
    c = -np.ones(n)
    b = np.empty(4*n)
    b[:n] = centers
    b[n:2*n] = 1 - centers
    b[2*n:3*n] = np.ones(n)
    b[3*n:4*n] = 1 - np.ones(n)
    bounds = [(0, 0.5)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return 0, np.zeros(n)
    return -res.fun, res.x

def evaluate_fitness(individual):
    centers = np.clip(np.array(individual).reshape(26, 2), 0, 1)
    _, radii = solve_radii(centers)
    return np.sum(radii)

def crossover(parent1, parent2):
    point1 = np.random.randint(1, 50)
    point2 = np.random.randint(point1, 51)
    child1 = parent1[point1:point2].tolist() + parent2[:point1].tolist() + parent2[point2:].tolist()
    child2 = parent2[point1:point2].tolist() + parent1[:point1].tolist() + parent1[point2:].tolist()
    return child1, child2

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, 52)
    return np.clip(individual, 0, 1)

def clamp_to_square(individual):
    return np.clip(individual, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    population = [np.random.rand(52) for _ in range(pop_size)]
    best_sol = None
    best_sum = -np.inf

    for gen in range(generations):
        fitnesses = [evaluate_fitness(individual) for individual in population]
        fittest_indices = np.argsort(fitnesses)[-elitism:]
        new_population = list(map(list, map(np.array, population)[fittest_indices]))
        
        for _ in range(pop_size - elitism):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, init_sigma * sigma_decay**gen)
            child2 = mutate(child2, init_sigma * sigma_decay**gen)
            child1 = clamp_to_square(child1)
            child2 = clamp_to_square(child2)
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population
        best_index = np.argmax(fitnesses)
        current_sum = fitnesses[best_index]
        if current_sum > best_sum:
            best_sum = current_sum
            best_sol = population[best_index]
        
        print(f"Generation {gen+1}/{generations}, Best Sum: {current_sum:.4f}")
    
    centers = np.clip(np.array(best_sol).reshape(n, 2), 0, 1)
    _, radii = solve_radii(centers)
    final_centers, final_radii = centers, radii
    final_sum = np.sum(final_radii)
    
    for _ in range(10):  # Local SLSQP refinement
        res = minimize(evaluate_fitness, final_centers.flatten(), method='SLSQP',
                       bounds=[(0, 1)] * 52, constraints={'type': 'ineq', 'fun': lambda x: solve_radii(x.reshape(n, 2))[0]},
                       options={'maxiter': 2000})
        if res.success and constraints(res.x).min() >= -1e-10:
            final_centers = res.x.reshape(n, 2)
            final_radii = solve_radii(final_centers)[1]
            final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
