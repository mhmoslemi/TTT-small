import numpy as np
from scipy.optimize import minimize, differential_evolution, BasinHopping, LinearConstraint

N = 26
NUM_VARS = 26 * 2 + 26  # 78

def obj(v):
    r = v[52:]                      # last 26 entries
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)      # (x,y) pairs
    r   = v[52:]                   # radii

    # boundary terms – 4*(N-1)
    bounds_x = np.column_stack([pos[:,0] - r, 1 - pos[:,0] - r])
    bounds_y = np.column_stack([pos[:,1] - r, 1 - pos[:,1] - r])
    cons = np.vstack([bounds_x.flatten(), bounds_y.flatten()])

    # pairwise terms – N*(N-1)/2
    diffs = pos[:,None,:] - pos[None,:,:]   # (N,N,2)
    dists_squared = np.sum(diffs**2, axis=2)
    cons_pairwise = dists_squared - (r[:,None] + r[None,:])**2
    np.fill_diagonal(cons_pairwise[0, :], 1e10)  # Ignore loss with self
    cons = np.vstack([cons, cons_pairwise.flatten()])

    return cons

def inflate_radii(centers, radii, max_iter=100):
    for _ in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r_i = min(n for n in np.r_[centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1]] if n > 0 and n < 1)
            for j in range(N):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r_i = min(max_r_i, dist - radii[j])
            radii[i] = max_r_i
        if np.allclose(radii, old_radii, atol=1e-8):
            break
    return radii

def run_packing(num_restarts=100, maxiter_slsqp=4000, ftol_slsqp=1e-12, popsize='20*N', search_local=True):
    bounds = [(0,1)]*26          # x
    bounds += [(0,1)]*26          # y
    bounds += [(0,0.5)]*26        # radii

    rough_radii = 0.2 * np.ones(N)
    centers = np.random.rand(N,2)
    init = np.concatenate([centers.flatten(), rough_radii])

    best_r = None
    best_res = None

    for _ in range(num_restarts):
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type':'ineq',
                                                   'fun': constraints},
                       options={'maxiter':maxiter_slsqp, 'ftol':ftol_slsqp, 'disp':False})
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                best_r = res.x[52:]
                best_res.centers = res.x[:52].reshape(26, 2)

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    inflated_radii = inflate_radii(best_res.centers, best_r)
    
    if search_local:
        res_final = minimize(fun=obj, x0=np.concatenate([best_res.centers.flatten(), inflated_radii]),
                             method='SLSQP', bounds=bounds, constraints={'type':'ineq', 'fun': constraints},
                             options={'maxiter':2000, 'disp':False})
        if res_final.success:
            inflated_radii = res_final.x[52:]
            
    return best_res.centers, inflated_radii, -best_res.fun

# Example call to run_packing function
centers, radii, sum_radii = run_packing()
print(f"Sum of radii: {sum_radii}")
