import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    n = centers.shape[0]
    c = [-1.0] * n
    A = []
    b = []
    for i in range(n):
        x, y = centers[i]
        A.append([x - 0.5, y - 0.5, 1.0])
        b.append(x + y)
        A.append([x - 0.5, y + 0.5, 1.0])
        b.append(1 - x + y)
        A.append([x + 0.5, y - 0.5, 1.0])
        b.append(1 + x - y)
        A.append([x + 0.5, y + 0.5, 1.0])
        b.append(1 + x + y)
        for j in range(i + 1, n):
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            A.append([-dx / dist, -dy / dist, 1.0])
            b.append(dist - 2 * np.sqrt(dist - 1e-9))
            A.append([dx / dist, dy / dist, 1.0])
            b.append(dist - 2 * np.sqrt(dist - 1e-9))
    A = np.array(A, dtype=float)
    b = np.array(b, dtype=float)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    return res.x

def choose_random_seeds(n, n_seeds):
    seeds = []
    for _ in range(n_seeds):
        if _ == 0:
            seed = np.array([[0.1, 0.1], [0.9, 0.1], [0.5, 0.6], [0.5, 0.9], [0.15, 0.3], [0.85, 0.3], [0.75, 0.25], [0.25, 0.25], [0.7, 0.7], [0.3, 0.7], [0.5, 0.5]])
        else:
            seed = np.random.rand(n, 2)
        seeds.append(seed)
    return seeds
