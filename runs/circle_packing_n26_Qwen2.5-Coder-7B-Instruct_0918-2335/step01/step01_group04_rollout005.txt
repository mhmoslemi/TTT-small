import numpy as np
from scipy.optimize import linprog, minimize
from numpy.linalg import norm

def solve_radii_lp(centers):
    n = len(centers)
    c = -np.ones(n)
    A = []
    b = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A.append([1, 0])
        b.append(x)
        A.append([0, 1])
        b.append(y)
        A.append([-1, 0])
        b.append(1 - x)
        A.append([0, -1])
        b.append(1 - y)

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = norm(centers[i] - centers[j])
            A.append([1, 1])
            b.append(dist)

    res = linprog(c, A_ub=A, b_ub=b, method="highs")
    return res.x

def run_packing():
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    lattice_points = np.array([(x, y) for x in np.linspace(0, 1, 5) for y in np.linspace(0, 1, 5)])
    lattice_points = np.array([(x + 0.25, y + 0.25) for x, y in lattice_points if not (x == y == 0.75 or x == y == 0.25)])

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0

    for _ in range(8):  # 8 different seeds
        if _ == 0:
            centers = lattice_points[:n]
        else:
            centers = np.random.rand(n, 2)

        alpha = 0.01
        max_iter = 200
        cooling_interval = 20
        accept_rate = 0.7
        epsilon = 1e-7
        patience = 10

        for iter in range(max_iter):
            radii = solve_radii_lp(centers)
            if iter % cooling_interval == 0:
                alpha *= 0.5
            distances = norm(centers[None, :, :] - centers[:, None, :], axis=-1)
            constraints = np.nan_to_num(np.max(distances - radii[:, None] - radii[None, :], axis=1))
            move_accept = constraints < epsilon

            tentative_centers = np.copy(centers)
            for i, (x, y) in enumerate(centers):
                neighbors = np.where(constraints[i] > 0)[0]
                if len(neighbors) > 0:
                    r_i = np.min([np.min(radii[neighbors]), x, 1-x, y, 1-y])
                    dx, dy = 0, 0
                    for j in neighbors:
                        min_move = np.min([np.min(radii[j], x), np.min(radii[j], 1-x), np.min(radii[j], y), np.min(radii[j], 1-y)])
                        dx += alpha * (x - centers[j][:, 0]) / norm(x - centers[j][:, 0]) * (min_move - radii[j])
                        dy += alpha * (y - centers[j][:, 1]) / norm(y - centers[j][:, 1]) * (min_move - radii[j])
                        if move_accept[i] and move_accept[j]:
                            dx -= alpha * 0.2 * (x - centers[j][:, 0]) / norm(x - centers[j][:, 0]) * (min_move - radii[j])
                            dy -= alpha * 0.2 * (y - centers[j][:, 1]) / norm(y - centers[j][:, 1]) * (min_move - radii[j])
                    tentative_centers[i] = np.clip(tentative_centers[i] + np.array([dx, dy]), 0, 1)
                else:
                    if move_accept[i]:
                        direction = np.array([1, 0 if np.random.rand() < 0.5 else 1])
                        tentative_centers[i] = np.clip(tentative_centers[i] + alpha * direction, 0, 1)

            new_radii = solve_radii_lp(tentative_centers)
            new_sum_radii = np.sum(new_radii)
            if new_sum_radii > best_sum_radii + epsilon or all(r < r_new for r, r_new in zip(radii, new_radii)):
                best_sum_radii = new_sum_radii
                best_centers = tentative_centers
                best_radii = new_radii
                continue

            if np.sum(move_accept) / n < accept_rate or np.abs(new_sum_radii - best_sum_radii) < epsilon:
                centers = np.copy(best_centers)
                continue

    centers_final = best_centers
    radii_final = best_radii

    def perturb(centers, step):
        return centers + step * np.random.randn(*centers.shape)

    for _ in range(4):
        step = alpha * 2
        restricted_centers = centers_final.copy()
        restricted_radii = solve_radii_lp(restricted_centers)
        iteration = 0
        while iteration < 50:
            for i in range(n):
                for j in range(n):
                    if norm(restricted_centers[i] - restricted_centers[j]) - restricted_radii[i] - restricted_radii[j] < 1e-9:
                        step_i = perturb(restricted_centers[np.array([i])], -step)
                        step_j = perturb(restricted_centers[np.array([j])], step)
                        restricted_centers[i] = step_i[0]
                        restricted_centers[j] = step_j[0]
                        restricted_radii = solve_radii_lp(restricted_centers)
                        iteration = 0
                        step *= 0.5
                        break
                if iteration == 0:
                    step *= 0.5
            iteration += 1

    result = minimize(
        lambda x: -np.sum(x[52:]),
        np.concatenate((centers_final.flatten(), radii_final)),
        method='SLSQP',
        bounds=[(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n,
        constraints={'type': 'ineq',
                     'fun': lambda x: np.array([-np.sum(x[52:]), norm(x[104:150] - np.array([[1, 1], [1, 0], [0, 1]]), axis=-1) - x[150:]]).flatten()},
        options={'maxiter': 4000, 'ftol': 1e-6, 'disp': False}
    )

    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    centers_final = result.x[:52].reshape(n, 2)
    radii_final = result.x[52:]
    sum_radii_final = np.sum(radii_final)

    return centers_final, radii_final, sum_radii_final
