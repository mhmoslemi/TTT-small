import numpy as np

def apollonius_circle(r1, r2, p1, p2, p3):
    """Compute the largest circle touching two circles and a line."""
    # Transformation to place p3 at the origin
    Cp1 = np.array(p1) - p3
    Cp2 = np.array(p2) - p3
    
    # Parameters of the transformation
    a1, b1 = Cp1
    a2, b2 = Cp2
    
    # Apollonius formula for the center (xa, xb) of the circle
    A = 2 * (a2 + a1)
    B = 2 * (b2 + b1)
    C = a1**2 + (b1 + b2)**2 + a2**2 - a1**2 - b1**2
    D = 4 * (a2 * b1 - a1 * b2)
    
    # Radius of the circle
    ra = 2 * (r1 + r2 - distance(p3, p1) - distance(p3, p2)) / np.sqrt(A**2 + B**2 - D)
    
    # Center of the circle
    xa, xb = (C * A + D * B) / (A**2 + B**2 - D) * 0.5, (C * B - D * A) / (A**2 + B**2 - D) * 0.5
    xc, yc = xa + p3[0], xb + p3[1]
    
    return xc, yc, ra

def lp_oracle(centers):
    """
    Solve the linear programming problem for given centers to find the radii.
    
    :param centers: np.array of shape (26, 2)
    :return: np.array of radii of shape (26,)
    """
    n = centers.shape[0]
    A来做lapreq也就是陈思红所讲的table形式
    A = np.zeros((104 + 26 * 325, 52))
    b = np.zeros(104 + 26 * 325)
    
    # Boundary constraints
    for i in range(n):
        r_i = np.array([1, -1, 0] * n)
        r_i[i * 3 + 0] = 1
        r_i[i * 3 + 1] = x_i
        r_i[i * 3 + 2] = x_i
        A[i * 3] = r_i[:26]
        b[i * 3] = x_i
    
    for i in range(n):
        r_i = np.array([1, -1, 0] * n)
        r_i[i * 3 + 0] = 1
        r_i[i * 3 + 1] = 1 - x_i
        r_i[i * 3 + 2] = 1 - x_i
        A[i * 3 + 1] = r_i[:26]
        b[i * 3 + 1] = 1 - x_i
    
    for i in range(n):
        r_i = np.array([1, -1, 0] * n)
        r_i[i * 3 + 0] = 1
        r_i[i * 3 + 1] = 0
        r_i[i * 3 + 2] = 1
        A[i * 3 + 2] = r_i[:26]
        b[i * 3 + 2] = y_i
    
    for i in range(n):
        r_i = np.array([1, -1, 0] * n)
        r_i[i * 3 + 0] = 1
        r_i[i * 3 + 1] = 0
        r_i[i * 3 + 2] = 1 - y_i
        A[i * 3 + 3] = r_i[:26]
        b[i * 3 + 3] = 1 - y_i
    
    # Pairwise constraints
    index = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            r_i = np.array([1, -1] * n)
            r_i[i * 2] = 1
            r_i[j * 2] = 1
            A[index] = r_i[:26]
            b[index] = d_ij
            index += 1
    
    # Solve the LP
    result = linprog(
        -np.ones(n), 
        A_ub=A, b_ub=b, 
        method='highs', 
        bounds=(0, None)
    )
    
    # Clip negative radii
    result.x[result.x < 0] = 0
    
    return result.x

def fitness(centers):
    """Compute the fitness of the centers by summing the radii."""
    radii = lp_oracle(centers)
    if np.all(radii > 0):
        return np.sum(radii)
    else:
        return -np.inf

def minimize_mixed_integer(centers):
    """Optimize the final centers and radii using SLSQP."""
    bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26
    def objective(x):
        centers = x[:52].reshape(26, 2)
        radii = lp_oracle(centers)
        return -np.sum(radii)
    
    constraints = {'type': 'ineq', 'fun': lambda x: constraints(x)}
    
    result = minimize(objective, centers.flatten(), method='SLSQP', bounds=bounds, constraints=constraints)
    optimized_positions = result.x[:52].reshape(26, 2)
    optimized_radii = lp_oracle(optimized_positions)
    return optimized_positions, optimized_radii, np.sum(optimized_radii)
