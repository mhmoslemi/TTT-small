import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

POP = 30
MAX_ITER = 4000
p_mut = 0.3
m = 3
sigma = 0.02
tol = 1e-9

# Precompute matrices for LP
n = 26

A_left = np.eye(n, n, k=-1)
A_right = np.eye(n, n, k=1)
A_bottom = np.eye(n, n)
A_top = np.eye(n, n, k=2)
A_diag = np.eye(n, n, k=2*n)

# Boundary constraints
A_bounds = np.vstack((A_left, A_right, A_bottom, A_top))
b_bounds = np.ones((4*n, 1))

# Pairwise constraints matrices
D = np.arange(n*n).reshape(n, n)
A_pairs = cdist(D, D)
A_pairs = A_pairsnp.triu(A_pairs, k=1) < n
A_pairs = A_pairs.astype(int)
A_pairs = A_pairs * (1 - A_diag)
A_pairs = A_pairs.reshape(-1, n)
A_pairs = np.vstack((A_pairs[:, np.arange(n)] + A_pairs[:, np.arange(n, 2*n)], 
                      A_pairs[:, np.arange(n)] + A_pairs[:, np.arange(n, 2*n)] - 2*n, 
                      A_pairs[:, np.arange(n)] + A_pairs[:, np.arange(n, 2*n)] + n,
                      A_pairs[:, np.arange(n)] + A_pairs[:, np.arange(n, 2*n)] - n))
A_pairs = A_pairs.astype(int)

B_pairs = np.ones((4*n*n, 1))

def lp_opt(p):
    r = linprog(-np.ones(n), A_ub=np.vstack((A_bounds, A_pairs)), b_ub=np.vstack((b_bounds, B_pairs)), bounds=[(tol, None)]*n, method='highs').x
    r[r < tol] = 0
    return r

def tournament_select(pop):
    a, b = np.random.randint(Pop, size=2)
    return pop[a] if pop[a][1] > pop[b][1] else pop[b]

def crossover(a, b):
    C = 0.5 * a[0] + 0.5 * b[0]
    C = np.clip(C + np.random.normal(0, sigma, size=(26, 2)), 0, 1)
    return C

def mutate(C):

    m = int(m)
    indices = np.random.choice(26, size=m, replace=False)
    for idx in indices:
        C[idx] += np.random.normal(0, sigma)
        C[idx] = np.clip(C[idx], 0, 1)

def local_slsqp(C):
    for _ in range(25):
        try:
            minimize(lambda x: 0, C.flatten(), method='SLSQP', bounds=[(0, 1)]*52, constraints={'type': 'ineq', 'fun': lambda x: np.sum(np.maximum(0, lp_opt(x) - C.flatten()))}).x
        except:
            break

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_center = None
    best_radii = None
    best_sum = 0

    init_pop = []

    # Hex grid
    hex_spacing = 2 * np.tan(np.pi / 6) / 2
    hex_circles = np.array([[np.cos(2 * np.pi * k / 7 + np.pi / 14), np.sin(2 * np.pi * k / 7 + np.pi / 14)] 
                                 for k in range(26) 
                                 if (np.cos(2 * np.pi * k / 7 + np.pi / 14)**2 + np.sin(2 * np.pi * k / 7 + np.pi / 14)**2)**0.5 <= 0.5]).reshape(-1, 2)
    hex_circles += np.random.uniform(-0.01, 0.01, size=(26, 2))
    hex_circles = np.clip(hex_circles, 0, 1)
    hex_radii = lp_opt(hex_circles)
    init_pop.append((hex_circles, np.sum(hex_radii)))

    # Random circles
    random_circles = np.random.uniform(0, 1, size=(26, 2))
    random_radii = lp_opt(random_circles)
    init_pop.append((random_circles, np.sum(random_radii)))

    # Grid circles
    grid_circles = np.array([[i / 5, j / 5] for i in range(5) for j in range(5)])
    grid_circles[-1] = np.array([0.5, 0.5])  # Center circle
    grid_circles[:1, 1:] += 0.5
    grid_circles[1:, -1] += 0.5
    grid_circles[-1, 1:] -= 0.5
    grid_circles[1:, 0] -= 0.5
    grid_circles[:, 0] += 0.5
    grid_circles = np.clip(grid_circles, 0, 1)
    grid_radii = lp_opt(grid_circles)
    init_pop.append((grid_circles, np.sum(grid_radii)))

    # Corner-heavy circles
    corner_circles = np.array([[i/5, 0] for i in range(2)] + 
                              [[i/5, 1] for i in range(2)] + 
                              [[1, i/5] for i in range(2)] + 
                              [[0, i/5] for i in range(2)] + 
                              [np.random.uniform(0, 1, 2) for _ in range(10)])
    corner_circles = np.clip(corner_circles, 0, 1)
    corner_radii = lp_opt(corner_circles)
    init_pop.append((corner_circles, np.sum(corner_radii)))

    np.random.shuffle(init_pop)

    for iter in range(MAX_ITER):
        a = tournament_select(init_pop)
        b = tournament_select(init_pop)
        C = crossover(a[0], b[0])
        mutate(C)
        local_slsqp(C)
        new_radii = lp_opt(C)
        new_sum = np.sum(new_radii)
        if new_sum > best_sum:
            best_center = C
            best_radii = new_radii
            best_sum = new_sum

    p_final = best_center
    r_final = lp_opt(p_final)
    sum_r = np.sum(r_final)
    return p_final, r_final, sum_r
