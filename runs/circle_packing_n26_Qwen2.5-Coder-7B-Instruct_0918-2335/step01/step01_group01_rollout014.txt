# Define the objective function for the outer minimization
def outer_obj(p):
    r = np.sqrt((p**2).sum(axis=1))
    return -np.sum(r)

# Define the constraints for the outer minimization
def outer_constraints(p):
    n = p.shape[0] // 2
    centers = p.reshape(n, 2)
    radii = np.sqrt((p**2).sum(axis=1))
    cons = np.empty(4*n + 325, float)
    # 4 * 26 boundary constraints
    for i in range(n):
        r_i = radii[i]
        x_i, y_i = centers[i]
        cons[4*i] = r_i
        cons[4*i+1] = 1 - r_i
        cons[4*i+2] = r_i
        cons[4*i+3] = 1 - r_i
    # 325 * 26^2 pair constraints
    dists = (centers[:, None] - centers[None, :]).norm(axis=2)
    mask = np.triu(np.ones((n, n), bool), k=1)
    pair_cons = dists[mask] - (radii[:, None] + radii[None, :])[mask]
    cons[4*n:] = pair_cons
    return cons

# Define the LP oracle to get the optimal radii for a given center set
def lp_oracle(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    dist_matrix = np.sqrt((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2)
    for i in range(n):
        left = np.zeros(n)
        right = np.ones(n) * 1.5
        left[i] = 1
        right[i] = 1
        b_upper = dist_matrix[i]
        con = {'A_ub': np.vstack([left.T, right.T]), 'b_ub': np.hstack([b_upper, np.ones(n)]), 'type': 'ineq'}
        result = linprog(radii, G=con['A_ub'], b=con['b_ub'], bounds=(0, None), method='highs')
        if not result.success:
            return np.full(n, 0.0001)
        radii[i] = result.x
    return np.maximum(radii, 0)
