import sympy
from scipy.optimize import linprog, SLSQP
import numpy as np

def contact_constraints(graph, wall_constraints):
    n = len(graph)
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []
    
    # Edge constraints
    for i, j in graph:
        dist = np.sqrt(wall_constraints[i] + wall_constraints[j])
        A_eq.append([1, 0, 0, 0, 0, 1, 0, 0])
        b_eq.append(dist - 1e-12)
        A_eq.append([0, 1, 0, 0, 1, 0, 0, 0])
        b_eq.append(dist - 1e-12)
    
    # Wall constraints
    for i, dist in enumerate(wall_constraints):
        if dist > 0:
            A_eq.append([0, 0, 1, 0, 0, 0, 0, 0])
            b_eq.append(dist - 1e-12)
    
    return np.array(A_eq), np.array(b_eq), A_ub, b_ub

def validate_solution(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if not (0 <= x <= 1 and 0 <= y <= 1 and 0 <= r <= 0.5):
            return False
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    graph_options = [
        # Add your contact graph configurations here
   ]
    
    best_solution = None
    best_sum = 0
    
    for graph in graph_options:
        wall_constraints = [0.5 if i in [0, 25] else 0.25 for i in range(n)]
        A_eq, b_eq, A_ub, b_ub = contact_constraints(graph, wall_constraints)
        r_bounds = [(0, 0.5) for _ in range(n)]
        res = linprog([1] * n, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
        
        if res.success:
            radii = res.x
            center_sum = np.sum_radii(radii)
            if center_sum > best_sum:
                best_sum = center_sum
                centers = np.array([(0, i/25) for i in range(n)])
                best_solution = (centers, radii, best_sum)
    
    if best_solution is None:
        return np.zeros((n, 2)), np.zeros(n), 0
    
    # SLSQP refinement
    def objective(x):
        nonlocal best_solution, best_sum
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        if validate_solution(centers, radii):
            return -np.sum(radii)
        return float('inf')
    
    def constraint(x):
        nonlocal best_solution, best_sum
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        return [-np.sum(radii) + best_sum, -1e-6]
    
    cons = {'type': 'ineq', 'fun': constraint}
    bounds = [(-1, 1),(-1, 1)] * n + [(0, 0.5) for _ in range(n)]
    init = np.concatenate([np.zeros((n, 2)).flatten(), np.random.rand(n) * 0.5])
    res = SLSQP(fun=objective, x0=init, constraints=cons, bounds=bounds, options={'maxiter': 1000})

    if res.success:
        centers = res.x[:n*2].reshape(n, 2)
        radii = res.x[n*2:]
        return centers, radii, -res.fun
    
    return best_solution

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
