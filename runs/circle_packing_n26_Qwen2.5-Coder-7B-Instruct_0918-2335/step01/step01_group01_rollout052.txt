import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform
import math

# Global variables
POP = 30  # Population size
EVAL = 4000  # Total number of fitness evaluations
p_mut = 0.3  # Mutation rate
mutation_steps = 3  # Number of random circles to mutate
mutation_std = 0.02  # Standard deviation for mutation

# Precompute static matrices for the LP
def precompute_matrices(dim):
    linprog_ineq = np.zeros((dim*(dim-1)//2, dim))
    linprog_rhs = np.ones(dim*(dim-1)//2) - 1e-9
    return linprog_ineq, linprog_rhs

# LP engine to find the best radii for a given set of centers
def lp_opt(centers, precomputed):
    dim = len(centers) // 2
    d = squareform(pdist(centers.reshape(-1, 2)))
    inequalities = np.hstack((d - centers[:, np.newaxis] - centers[newaxis, :], -np.eye(dim)))
    linprog_ineq, linprog_rhs = precomputed
    linprog_ineq = np.vstack((linprog_ineq, inequalities))
    linprog_rhs = np.concatenate((linprog_rhs, linprog_rhs[d > centers[:, np.newaxis] + centers[newaxis, :]].flatten()))
    objective = np.ones(dim)
    result = linprog(-objective, A_ub=linprog_ineq, b_ub=linprog_rhs, bounds=(0, None), method='highs')
    if result.success and not np.any(result.x < 0):
        return result.x
    return np.zeros(dim)

# Main function to run the packing algorithm
def run_packing():
    np.random.seed(42)
    n_circles = 26
    dim = 2 * n_circles
    precomputed = precompute_matrices(dim)

    # Initial population
    pop = []
    for _ in range(POP):
        p = np.random.rand(dim)
        while True:
            p[p < 0.01] += 0.01
            p[p > 0.99] -= 0.01
            lengths = np.sqrt(p[::2]**2 + p[1::2]**2)
            r_sum = lp_opt(p, precomputed)
            if np.any(r_sum <= 0):
                p = np.random.rand(dim)
            else:
                break
        pop.append((p, np.sum(r_sum)))

    # Evolutionary loop
    for eval in range(EVAL):
        A = np.array([pop[i][0] for i in 3 * np.random.choice(np.arange(POP), 3)])
        B = np.array([pop[i][0] for i in 3 * np.random.choice(np.arange(POP), 3)])
        C = 0.5 * A.sum(axis=0) + 0.5 * B.sum(axis=0) + 0.005 * np.random.rand(dim)
        C[C < 0] = 0
        C[C > 1] = 1
        if np.random.rand() < p_mut:
            m = min(mutation_steps, np.random.randint(1, n_circles + 1))
            indices = np.random.choice(np.arange(len(C)), m, replace=False)
            C[indices] += mutation_std * np.random.randn(m)
            C[C < 0] = 0
            C[C > 1] = 1
        local_slsqp(C, precomputed)
        r_sum = lp_opt(C, precomputed)
        if sum(r_sum) > min((pop[i][1] for i in range(POP))):
            pop = [x for x in pop if x[1] >= sum(r_sum)]
            pop.append((C, sum(r_sum)))

    # Final polishing using SLSQP
    best_idx = np.argmax([x[1] for x in pop])
    best_p, best_r_sum = pop[best_idx]
    x0 = best_p
    bounds = [(0, 1)] * dim
    constraints = {'type': 'ineq', 'fun': lambda x: x.reshape(-1, 2)[:, np.newaxis, :] - x[newaxis, :].reshape(1, -1, 2)[:, :, :]}
    res = minimize(lambda x: -np.sum(lp_opt(x.reshape(-1, 2), precomputed)), x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-9})
    if res.success:
        optim_r = lp_opt(res.x.reshape(-1, 2), precomputed)
        optim_r[optim_r <= 0] = 0
        return res.x.reshape(-1, 2), optim_r, np.sum(optim_r)
    return best_p.reshape(-1, 2), best_r_sum, best_r_sum

def local_slsqp(p, precomputed):
    dim = len(p) // 2
    n_circles = dim // 2
    bounds = [(0, 1)] * dim
    r = lp_opt(p, precomputed)
    A = np.eye(dim)
    b = -np.sum(p, axis=1)
    constraints = {'type': 'ineq', 'fun': lambda x: x.reshape(-1, 2)[:, 0] - x[:, 0]}
    constraints += {'type': 'ineq', 'fun': lambda x: 1 - x.reshape(-1, 2)[:, 0] - x[:, 0]}
    constraints += {'type': 'ineq', 'fun': lambda x: x.reshape(-1, 2)[:, 1] - x[:, 1]}
    constraints += {'type': 'ineq', 'fun': lambda x: 1 - x.reshape(-1, 2)[:, 1] - x[:, 1]}
    constraints += {'type': 'ineq', 'fun': lambda x: (x.reshape(-1, 2)**2).sum(axis=2) - np.outer(x[:, 0], x[:, 0])}
    constraints += {'type': 'ineq', 'fun': lambda x: (x.reshape(-1, 2)**2).sum(axis=2) - np.outer(x[:, 1], x[:, 1])}
    result = minimize(lambda x: -np.sum(x), p, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 25})
    if result.success:
        p[:dim] = result.x
    return p
