def build_constraints(v):
    pos = v[:52].reshape(N, 2)
    r = v[52:]

    # boundary
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r

    # pairwise
    diffs = squareform(pdist(pos))
    pair_cons = diffs - (r[:, None] + r[None, :])
    return np.concatenate((cons, pair_cons)), (cons, pair_cons)

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
