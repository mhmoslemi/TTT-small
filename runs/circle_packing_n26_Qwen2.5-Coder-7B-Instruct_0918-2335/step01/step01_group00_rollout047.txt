import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping

N = 26
NUM_VARS = 2 * N + N

def build_constraints(v):
    pos = v[:NUM_VARS // 2].reshape(N, 2)
    r = v[NUM_VARS // 2:]
    # Boundary constraints
    boundary = np.empty(104, float)
    boundary[:N] = pos[:, 0] - r
    boundary[N:2 * N] = 1 - pos[:, 0] - r
    boundary[2 * N:3 * N] = pos[:, 1] - r
    boundary[3 * N:] = 1 - pos[:, 1] - r
    # Pairwise constraints
    diffs = pos[:, None, :] - pos[None, :, :]
    d2 = np.sum(diffs ** 2, axis=2)
    mask = np.triu(np.ones((N, N), bool), k=1)
    pairwise = d2[mask] - (r[:, None] + r[None, :])[mask] ** 2
    # Flatten constraints array
    return np.concatenate([boundary, pairwise])

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(N):
                if i == j: 
                    continue
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii

def obj(v):
    return -np.sum(v[NUM_VARS // 2:])

def run_packing(max_iter=4000, ftol=1e-12, nrestarts=100, max_iter_post=1000) -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * NUM_VARS

    def local_search(v):
        res = minimize(fun=obj, x0=v, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': build_constraints},
                       options={'maxiter': max_iter_post, 'ftol': ftol})
        return -res.fun, res.x

    best_r = None
    best_res = None

    def validate_solution(v):
        centers, radii = v[:NUM_VARS // 2].reshape(N, 2), v[NUM_VARS // 2:]
        return validate_packing(centers, radii)[0], centers, radii

    for _ in range(nrestarts):
        init_positions = np.random.rand(N, 2)
        init_radii = np.random.rand(N) * 0.2 + 0.1  # Initial guess with 20% variation around 0.1
        init = np.concatenate([init_positions.ravel(), init_radii])
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': build_constraints},
                       options={'maxiter': max_iter, 'ftol': ftol})
        
        if res.success:
            if not validate_solution(res.x)[0]:
                continue

            if best_res is None or -res.fun >= -best_res.fun:
                best_res = res
                best_r = res.x[NUM_VARS // 2:]

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    # Post-refinement using inflate_radii
    best_values = np.concatenate([best_res.x[:NUM_VARS // 2], best_r])
    inflated_rad = inflate_radii(best_res.x[:NUM_VARS // 2].reshape(N, 2), best_r).ravel()
    inflated_x = best_res.x[:NUM_VARS // 2] + (inflated_rad - best_r)[None,:] * (np.random.rand(N, 2) - 0.5)
    inflated_values = np.concatenate([inflated_x.ravel(), inflated_rad])

    best_res = minimize(fun=obj, x0=inflated_values, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': build_constraints},
                       options={'maxiter': max_iter_post, 'ftol': ftol})
    
    centers, radii = best_res.x[:NUM_VARS // 2].reshape(N, 2), best_res.x[NUM_VARS // 2:]
    sum_radii = -best_res.fun

    # Global refinement using basinhopping after 10 best restarts
    if nrestarts > 10 and validate_solution(best_values)[0]:
        best_x, best_y = validate_solution(best_values)
        def take_step(x_new):
            for i in range(N):
                x_new[i] += (np.random.rand(2) - 0.5) * 0.05  # Random step size
            return x_new

        res_bh = basinhopping(func=local_search, x0=best_values, niter=nrestarts,
                               stepsize=0.05, take_step=take_step,
                               interval=25, minimizer_kwargs={"options": {"disp": False}})
        best_res = res_bh.fun, res_bh.x
        centers, radii = best_res[1][:NUM_VARS // 2].reshape(N, 2), best_res[1][NUM_VARS // 2:]
        sum_radii = -best_res[0]

    return centers, radii, sum_radii
