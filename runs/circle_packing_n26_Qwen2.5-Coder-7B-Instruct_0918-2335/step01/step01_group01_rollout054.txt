import numpy as np
from scipy.optimize import minimize, linprog
import scipy.spatial

# Constants
NP = 40
T0 = 0.05
COG = 1.5
SOC = 1.5
INERTIA = 0.5
SIGMA = 0.02

# Problem setup
N = 26
BOUNDARY_CONSTRAINTS = 4 * N
PAIR_CONSTRAINTS = N * (N - 1) // 2
TOTAL_CONSTRAINTS = BOUNDARY_CONSTRAINTS + PAIR_CONSTRAINTS
M = 35  # Number of potential positions on the hexagonal lattice
lattice = hexagonal_lattice(M, M)

# Seed positions
corner_circles = np.array([
    [0.146, 0.146],
    [0.854, 0.146],
    [0.146, 0.854],
    [0.854, 0.854]
])
edge_circles = np.array([
    [0.179, 0.055],
    [1 - 0.179, 0.055],
    [1 - 0.055, 0.179],
    [1 - 0.055, 1 - 0.179],
    [0.179, 1 - 0.055],
    [0.055, 0.179],
    [0.055, 1 - 0.179],
    [1 - 0.179, 1 - 0.055]
])
hexagon_circles = np.array([
    [0.5 + 0.229 * np.cos(np.linspace(0, 2 * np.pi, 7, endpoint=False)), 
     0.5 + 0.229 * np.sin(np.linspace(0, 2 * np.pi, 7, endpoint=False))]
])
ring_circles = generate_bonded_ring()

# Combine seeds
seeds = np.vstack((corner_circles, edge_circles, hexagon_circles, ring_circles))

# Functions
def distance_matrix(pos):
    D = scipy.spatial.distance_matrix(pos, pos)
    return np.clip(D + 1e-12, 1e-12, None)

def lp_radius(pos, D):
    c = np.zeros(N)
    A_ub = np.vstack([np.eye(N), -np.eye(N), -np.multiply(D, pos[:, np.newaxis, 0]), -np.multiply(D, pos[:, np.newaxis, 1])])
    b_ub = np.hstack([np.ones(N) * (1 - pos[:, 0]), np.zeros(N), np.zeros(N), np.zeros(N)])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'tol': 1e-9})
    if result.status != 0:
        print("Infeasible solution")
        result.x.fill(0)
    return np.clip(result.x, 1e-9, 0.5)

def cmp_sphere(p, q):
    return np.sum((p - q) ** 2) - 1e-12 >= 0

def LLVM(bounds, pos):
    return -np.sum(lp_radius(pos, distance_matrix(pos)))

def run_packing():
    best_center = None
    best_radii = None
    best_sum = 0

    for _ in range(20):
        init = np.repeat(seeds[np.random.choice(10, N)], 2, axis=1) + np.random.rand(52) * 0.1
        init[N:] = lp_radius(init[:52].reshape(N, 2), distance_matrix(init[:52].reshape(N, 2)))
        res = minimize(fun=LLVM, x0=init, bounds=bounds, constraints={'type': 'ineq', 'fun': brief_constraint}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        if res.success and np.all(brief_constraint(res.x) >= -1e-10):
            r = res.x[N:]
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_center = res.x[:52].reshape(N, 2)
                best_radii = r
                best_sum = cur_sum
    return best_center, best_radii, best_sum

# Helper functions
def brief_constraint(x):
    pos = x[:52].reshape(N, 2)
    r = x[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((N, N), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

def hexagonal_lattice(width, height):
    y = np.arange(height) * np.sqrt(3) * 0.5
    x = np.arange(width)
    x[1::2] += 0.5
    X = np.repeat(x[:, np.newaxis], height, axis=1)
    Y = np.repeat(y[np.newaxis, :], width, axis=0)
    return np.column_stack((X.flatten(), Y.flatten()))

def generate_bonded_ring():
    # Generate a bonded ring of circles
    circles = []
    radii = []
    # Code to generate the positions and radii of the circle in the ring
    return circles, radii

# Run the packing function
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}")
print(f"Radii: {radii}")
print(f"Sum of Radii: {sum_radii}")
