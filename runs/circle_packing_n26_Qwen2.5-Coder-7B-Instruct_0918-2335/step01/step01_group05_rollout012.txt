import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    eps = 1e-12
    min_step = 1e-5
    jitter_amp = 0.1
    gs = 0.02
    cool = 0.995
    max_iter = 2000
    
    np.random.seed(42)
    
    cps = np.random.rand(n, 2)
    rs = np.zeros(n)
    
    for t in range(max_iter):
        gs *= cool
        if gs < min_step:
            break
        
        jitter_amp *= gs
        cps += jitter_amp * (np.random.rand(n, 2) - 0.5)
        cps = np.clip(cps, 0, 1)
        
        dists = np.sqrt(((cps[:,None,:] - cps[None,:,:])**2).sum(axis=2))
        mask = dists < rs[:,None] + rs[None,:] - eps
        
        while mask.sum() > 0:
            mask_pairs = np.triu(mask, 1)
            for i in range(n):
                for j in range(i):
                    if mask_pairs[i, j]:
                        d = dists[i, j]
                        sep = rs[i] + rs[j] - d + eps
                        disp = 0.5 * sep * (cps[i] - cps[j]) / d
                        cps[i] -= disp
                        cps[j] += disp
                        cps[i] = np.clip(cps[i], rs[i], 1 - rs[i])
                        cps[j] = np.clip(cps[j], rs[j], 1 - rs[j])
                        

    for i in range(n):
        cps[i] = np.clip(cps[i], rs[i], 1 - rs[i])
        
    dists = np.sqrt(((cps[:,None,:] - cps[None,:,:])**2).sum(axis=2))
    mask = (dists < rs[:,None] + rs[None,:] - eps) & (dists > 0)
    overlap_pairs = np.argwhere(mask)
    while overlap_pairs.size > 0:
        for i, j in overlap_pairs:
            r_i, r_j = rs[i], rs[j]
            d_ij = dists[i, j]
            min_d = (r_i + r_j) - eps
            scale_factor = min_d / d_ij
            rs[i] *= scale_factor
            rs[j] *= scale_factor
            
            while dists[i, j] < (r_i + r_j - eps):
                disp = 0.5 * (r_i + r_j - dists[i, j]) * (cps[i] - cps[j]) / dists[i, j]
                cps[i] -= disp
                cps[j] += disp
                cps[i] = np.clip(cps[i], rs[i], 1 - rs[i])
                cps[j] = np.clip(cps[j], rs[j], 1 - rs[j])
                dists = np.sqrt(((cps[:,None,:] - cps[None,:,:])**2).sum(axis=2))
        
        mask = (dists < rs[:,None] + rs[None,:] - eps) & (dists > 0)
        overlap_pairs = np.argwhere(mask)
    
    dists = np.sqrt(((cps[:,None,:] - cps[None,:,:])**2).sum(axis=2))
    mask = dists < rs[:,None] + rs[None,:] - eps
    if mask.any():
        return run_packing()
    
    rs = np.array([np.minimum(np.maximum(r, 0), 1) for r in rs])
    
    sum_radii = np.sum(rs)
    lp_valid, lp_msg = validate_packing(cps, rs)
    if not lp_valid:
        rs = np.array([max(0, r - 1e-9) for r in rs])
        sum_radii = np.sum(rs)
        lp_valid, lp_msg = validate_packing(cps, rs)
    
    return cps, rs, sum_radii
