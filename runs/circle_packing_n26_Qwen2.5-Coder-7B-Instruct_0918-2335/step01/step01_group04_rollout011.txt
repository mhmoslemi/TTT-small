def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    import numpy as np
    from scipy.optimize import linprog, minimize
    
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def linear_program(centers):
        r = np.maximum(0, np.minimum(centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]))
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        dists_masked = dists[mask]
        r_masked = r[:, np.newaxis] + r[None, :]
        pair_cons = r_masked[mask]
        return -np.sum(r), np.concatenate([np.concatenate([r, r], axis=1), dists_masked, pair_cons])
    
    def push_away(centers, radii):
        for i in range(n):
            circle_radius = radii[i]
            circle_x, circle_y = centers[i]
            wall_distances = np.array([circle_radius, 1 - circle_x - circle_radius, circle_y - circle_radius, 1 - circle_y - circle_radius])
            active_wall = np.argmin(wall_distances)
            if active_wall < 2:
                centers[i][active_wall] = 2 * circle_x if active_wall == 0 else 2 * circle_y - 1
            else:
                target = centers[i + 1:]
                dists = np.sqrt(np.sum((centers[i] - target) ** 2, axis=1))
                closest_circle = np.argmin(dists)
                closest_circle_index = i + closest_circle + 1
                closest_circle_radius = radii[closest_circle_index]
                closest_circle_x, closest_circle_y = centers[closest_circle_index]
                u_ij = (target[closest_circle] - centers[i]) / dists[closest_circle]
                centers[i] += 0.01 * u_ij if np.sum(radii) > np.sum(radii[closest_circle_index:]) else -0.01 * u_ij
            
            centers[i] = np.clip(centers[i], 0, box_size)
        
        return centers
    
    def validate_solution(centers, radii):
        if np.any(radii < 0):
            return False, "Negative radius detected"
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > box_size or y - radii[i] < 0 or y + radii[i] > box_size:
                return False, f"Circle {i} outside the unit square"
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    return False, f"Circles {i} and {j} overlap"
        return True, "Valid solution"
    
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.02)
    initial_guess = np.concatenate((initial_centers.flatten(), initial_radii))
    
    bounds = [(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n
    
    best_centers = initial_centers.copy()
    best_radii = initial_radii.copy()
    best_sum = np.sum(initial_radii)
    
    seeds = [
        np.array([[i % 5 / 5, (i // 5 + (i % 5) / 6) % 1] for i in range(n)]),
        np.array(np.random.rand(n, 2)),
        np.array([[i // 5 / 5, (i % 5 + (i // 5 + 1) / 10) % 1] for i in range(n)]),
        np.array([[i // 5 / 5, (i % 5 + (i // 5 + 2) / 10) % 1] for i in range(n)]),
        np.array([[i // 5 / 5, (i % 5 + (i // 5 + 3) / 10) % 1] for i in range(n)]),
    ]
    
    for seed in seeds:
        current_centers = seed.copy()
        current_radii = np.full(n, 0.02)
        best_current_sum = 0.0
        
        alpha = 0.01
        COOLING_INTERVAL = 20
        ESCAPE_INTERVAL = 50
        eps = 1e-7
        PAUSE_EPOCHS = 5
        
        for iter in range(200):
            current_centers = push_away(current_centers.copy(), current_radii)
            response = linprog(current_radii, linprog.A_ub=None, linprog.b_ub=None, linprog.A_eq=None, linprog.b_eq=None, bounds=bounds, method='simplex')
            current_radii = response.x if response.success else current_radii
            current_sum = -response.fun
            
            if current_sum > best_current_sum + eps:
                best_current_sum = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
            else:
                current_centers = seed.copy()
            
            if iter % COOLING_INTERVAL == 0:
                alpha *= 0.5
            if iter % ESCAPE_INTERVAL == 0:
                current_centers += alpha / 2 * np.random.rand(n, 2)
            
            if (iter + 1 - max(np.where(np.cumsum(current_sum != best_current_sum) >= PAUSE_EPOCHS, iter + 1, n)) > 0):
                break
    
    final_centers = best_centers headquarters
    final_radii = best_radii
    final_sum = np.sum(best_radii)
    
    return final_centers, final_radii, final_sum
