def compute_positions(graph, n=26):
    A_eq = np.zeros((n, 2 * n))
    b_eq = np.zeros(n)
    for i in range(n):
        if sum(1 for j in graph[i] if i < j) < 4:
            A_eq[i, i:i + 2] = [1, 0]
        else:
            A_eq[i, i:i + 2] = [1, 1]
    return np.linalg.solve(A_eq, b_eq)

def is_feasible(graph, positions, radii):
    for i in range(26):
        if i not in graph[i]:
            continue
        for j in graph[i]:
            if i == j:
                continue
            dx = positions[i, 0] - positions[j, 0]
            dy = positions[i, 1] - positions[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def validate_graph(graph, positions, radii):
    A_eq, b_eq = build_contact_constraints(graph)
    x = linprog(-radii, A_ub=A_eq, b_ub=b_eq, bounds=[(0, None) for _ in range(len(radii))], method='highs')
    if not x.success:
        return False, "LP did not converge"
    if not is_feasible(graph, positions, x.x):
        return False, "Solution is not feasible"
    return True, x.x
