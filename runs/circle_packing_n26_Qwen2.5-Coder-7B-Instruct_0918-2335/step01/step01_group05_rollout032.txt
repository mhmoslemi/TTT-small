import numpy as np
from scipy.optimize import minimize, differential_evolution

def objective(v):
    r = v[52:]
    penalty = np.sum(np.maximum(0, -r)) * 1e6
    return -np.sum(r) + penalty

def constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    con = np.empty(104, float)
    con[:26] = pos[:, 0] - r
    con[26:52] = 1 - pos[:, 0] - r
    con[52:78] = pos[:, 1] - r
    con[78:104] = 1 - pos[:, 1] - r
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    return np.concatenate([con, dists[mask] - (r[:, None] + r[None, :])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)

    # Global search with Differential Evolution (DE)
    res_de = differential_evolution(
        objective,
        bounds,
        polish=False,
        popsize=15,
        maxiter=500,
        disp=False
    )

    # Fine-tuning with local SLSQP
    successful_de = res_de.success and constraints(res_de.x).min() >= -1e-10
    if successful_de:
        res_slsqp = minimize(
            fun=constraints,
            x0=res_de.x,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraints},
            options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False}
        )
        if res_slsqp.success:
            centers = res_slsqp.x[:52].reshape(n, 2)
            radii = res_slsqp.x[52:]
            return centers, radii, -res_slsqp.fun
        else:
            return res_de.x[:52].reshape(n, 2), res_de.x[52:], -res_de.fun
    else:
        return res_de.x[:52].reshape(n, 2), res_de.x[52:], -res_de.fun
