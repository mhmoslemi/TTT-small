import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def solve_radii_lp(centers):
    n = centers.shape[0]
    A = np.zeros((2*n + n*(n-1)//2, 2*n))
    b = np.zeros(2*n + n*(n-1)//2)
    
    # Radii constraints
    for i in range(n):
        A[i] = np.zeros(2*n)
        A[i][2*i] = 1
        b[i] = centers[i, 0]
    for i in range(n):
        A[n+i] = np.zeros(2*n)
        A[n+i][2*i] = 1
        b[n+i] = 1 - centers[i, 0]
    for i in range(n):
        A[2*n + i] = np.zeros(2*n)
        A[2*n + i][2*i + 1] = 1
        b[2*n + i] = centers[i, 1]
    for i in range(n):
        A[2*n + 4*n + i] = np.zeros(2*n)
        A[2*n + 4*n + i][2*i + 1] = 1
        b[2*n + 4*n + i] = 1 - centers[i, 1]
    
    # Pairwise constraints
    idx = 2*n + 8*n
    for i in range(n):
        for j in range(i+1, n):
            A[idx] = np.zeros(2*n)
            A[idx][2*i] = 1
            A[idx][2*j] = 1
            b[idx] = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            idx += 1
    
    res = linprog(-np.ones(2*n), A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(2*n)], method='highs')
    if not res.success:
        raise ValueError("Linear programming did not converge.")
    radii = res.x[:n]
    return radii

def find_active_constraints(centers, radii):
    n = centers.shape[0]
    distances = cdist(centers, centers, 'euclidean')
    contact_graph = (distances - (radii[:, np.newaxis] + radii)**2 > -1e-10).astype(int)
    contact_graph[np.diag_indices(n)] = 0
    active_constraints = []
    for i in range(n):
        for j in range(n):
            if contact_graph[i, j] == 1:
                constraint = (distances[i, j] - (radii[i] + radii[j])) / (radii[i] + radii[j]) ** 2
                if constraint < 0:
                    active_constraints.append(i)
                    active_constraints.append(j)
                    break
    return np.array(active_constraints)

def run_packing():
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    num_seeds = 8
    max_iter = 200
    cooling_interval = 20
    escape_interval = 50
    accept_rate = 10
    epsilon = 1e-7
    patience = 50
    
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0.0
    
    for seed in range(num_seeds):
        if seed == 0:
            centers = np.array([[i/5, j/5] for i in range(5) for j in range(6)]).reshape(n, 2)
        else:
            centers = np.random.rand(n, 2)
        
        current_sum_radii = 0.0
        alpha = 0.01
        
        for iteration in range(max_iter):
            radii = solve_radii_lp(centers)
            active_constraints = find_active_constraints(centers, radii)
            
            tentative_centers = np.copy(centers)
            for i in set(active_constraints):
                if active_constraints.count(i) == 1:
                    opposite_constraint = np.where(active_constraints == n - i - 1)[0]
                    if np.any(opposite_constraint == active_constraints):
                        j = opposite_constraint[0]
                    else:
                        j = np.where(active_constraints == i - 1)[0]
                        j = j[0]
                    
                    xi, yi = tentative_centers[i]
                    rj = radii[j]
                    vector_xj = centers[j, 0] - xi
                    vector_yj = centers[j, 1] - yi
                    
                    if vector_xj > 0:
                        delta_x = alpha
                    else:
                        delta_x = -alpha
                    
                    if vector_yj > 0:
                        delta_y = alpha
                    else:
                        delta_y = -alpha
                    
                    if xi + delta_x > 1:
                        delta_x = 1 - xi
                    elif xi + delta_x < 0:
                        delta_x = -xi
                    
                    if yi + delta_y > 1:
                        delta_y = 1 - yi
                    elif yi + delta_y < 0:
                        delta_y = -yi
                    
                    tentative_centers[i] += np.array([delta_x, delta_y])
                    
                    xi, yi = tentative_centers[j]
                    xi -= delta_x
                    yi -= delta_y
                    
                    tentative_centers[j] = np.array([xi, yi])
            
            tentative_radii = solve_radii_lp(tentative_centers)
            tentative_sum_radii = np.sum(tentative_radii)
            
            if tentative_sum_radii > current_sum_radii:
                centers = tentative_centers
                radii = tentative_radii
                current_sum_radii = tentative_sum_radii
            else:
                acceptance_ratio = np.sum((tentative_radii - radii) / abs(tentative_radii + radii))
                if acceptance_ratio > accept_rate:
                    centers = tentative_centers
                    radii = tentative_radii
                    current_sum_radii = tentative_sum_radii
                alpha *= 0.5
            
            if iteration % escape_interval == 0 and seed > 0:
                noise = alpha / 2 * np.random.randn(2*n)
                centers += noise.reshape(n, 2)
            
            if iteration % cooling_interval == 0 and iteration > 0:
                alpha *= 0.5
        
        if current_sum_radii > best_sum_radii:
            best_centers = centers
            best_radii = radii
            best_sum_radii = current_sum_radii
    
    result = linprog(-np.ones(2*n), A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(2*n)], method='SLSQP', options={'maxiter': 2000, 'ftol': 1e-9})
    if result.success:
        best_centers = result.x[:n].reshape(n, 2)
        best_radii = result.x[n:].copy()
    return best_centers, best_radii, np.sum(best_radii)
