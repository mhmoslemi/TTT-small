import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def evaluate(solution):
    centers, radii = solution
    n = centers.shape[0]
    sum_r = np.sum(radii)
    if np.any(np.isnan(centers)) or np.any(np.isnan(radii)):
        return sum_r - 1e10  # Penalize NaN
    for i in range(n):
        if radii[i] < 0:
            return sum_r - 1e10
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
            return sum_r - 1e10
    dists = cdist(centers, centers).ravel()
    mask = np.triu(np.ones((n, n), bool), k=1)
    overlapping = np.any(dists[mask] < radii[mask[:-1]] + radii[mask[1:]] - 1e-6)
    if overlapping:
        return sum_r - 1e10
    return sum_r

def lp_optimize_radii(centers):
    n, _ = centers.shape
    objective = [-1] * n
    bounds = [(0, None)] * n
    A = np.zeros((3 * n, n))
    for i in range(n):
        x, y = centers[i]
        dists = np.sqrt((centers[:,0] - x)**2 + (centers[:,1] - y)**2)
        mask = dists > 0
        A[3 * i, i] = 1
        A[3 * i + 1, i] = 1
        A[3 * i + 2, i] = 1
        A[3 * i + 1, :][mask] += centers[:,0][mask] - x
        A[3 * i + 2, :][mask] += centers[:,1][mask] - y
        A[3 * i, :][mask] += 2 * (centers[:,0][mask] - x)
        A[3 * i + 1, :][mask] += 2 * (centers[:,1][mask] - y)
    b = np.hstack([np.minimum(centers[:, 0] - radii, 1 - centers[:, 0] - radii),
                   np.minimum(centers[:, 1] - radii, 1 - centers[:, 1] - radii),
                   np.ones(n)])
    result = linprog(objective, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return -result.fun
    return None

def refine_with_sa(centers, radii):
    n = centers.shape[0]
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum_r = evaluate((best_centers, best_radii))
    T = 1e-1
    cool_rate = 0.995
    SA_iterations = 15000
    for _ in range(SA_iterations):
        for p in range(26):
            new_radii = radii.copy()
            new_centers = centers.copy()
            new_radii[p] = np.random.uniform(0, 0.45)
            new_centers[p] = new_centers[p] + np.random.uniform(-0.01, 0.01, size=2)
            for m in range(2):
                new_radii = np.maximum(new_radii, 0)
                if new_radii.min() < 0:
                    new_radii -= np.min(new_radii)
                new_centers = np.maximum(np.minimum(new_centers, 1.0), 0.0)
                this_sum_r = evaluate((new_centers, new_radii))
                if this_sum_r > best_sum_r:
                    best_sum_r = this_sum_r
                    best_centers = new_centers
                    best_radii = new_radii
                else:
                    metropolis = np.exp((best_sum_r - this_sum_r) / T)
                    if np.random.uniform() < metropolis:
                        best_sum_r = this_sum_r
                        best_centers = new_centers
                        best_radii = new_radii
        T *= cool_rate
    return best_centers, best_radii
