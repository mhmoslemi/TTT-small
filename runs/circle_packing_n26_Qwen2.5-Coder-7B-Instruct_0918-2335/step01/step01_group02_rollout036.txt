import numpy as np
from scipy.optimize import linprog, minimize

def solve_lradii(pos):
    n = pos.shape[0]
    A_ub = np.empty((4*n + 26*(26-1)//2, n))
    b_ub = np.empty(4*n + 26*(26-1)//2)

    # Boundary constraints
    for i in range(n):
        A_ub[4*i] = np.zeros(n)
        A_ub[4*i, i] = 1
        b_ub[4*i] = pos[i, 0]
        A_ub[4*i+1] = np.zeros(n)
        A_ub[4*i+1, i] = 1
        b_ub[4*i+1] = 1 - pos[i, 0]
        A_ub[4*i+2] = np.zeros(n)
        A_ub[4*i+2, i] = 1
        b_ub[4*i+2] = pos[i, 1]
        A_ub[4*i+3] = np.zeros(n)
        A_ub[4*i+3, i] = 1
        b_ub[4*i+3] = 1 - pos[i, 1]

    # Pairwise constraints
    k = 4 * n
    for i in range(n-1):
        for j in range(i+1, n):
            dist = np.sqrt((pos[i, 0] - pos[j, 0])**2 + (pos[i, 1] - pos[j, 1])**2)
            A_ub[k] = -np.zeros(n)
            A_ub[k, i::2] = [1, 1]
            A_ub[k, j::2] = [1, 1]
            b_ub[k] = - (pos[i, 0] - pos[j, 0])**2 - (pos[i, 1] - pos[j, 1])**2 + 2 * dist
            k += 1

    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, 0.5) for _ in range(n)])

    return res.fun, -res.fun

def fitness(pos):
    r, s_r = solve_lradii(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = np.empty_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < rate:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    child = np.copy(individual)
    for i in range(len(child)):
        if np.random.rand() < rate:
            child[i] += np.random.normal(0, sigma)
            if child[i] < 0:
                child[i] = 0
            elif child[i] > 1:
                child[i] = 1
    return child

def tournament(pop, fitness_vals, k):
    selected = []
    for _ in range(k):
        idx = np.random.randint(0, len(pop))
        if len(selected) == 0 or fitness_vals[selected[-1]] < fitness_vals[idx]:
            selected.append(idx)
    return selected[-1]

    POPSIZE = 80
    GENS = 250
    ELITE_SIZE = int(0.1 * POPSIZE)
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.2
    MUTATION_SIGMA = 0.03

    np.random.seed(42)
    pop = np.random.rand(POPSIZE, 52)
    fitness_vals = np.array([fitness(pos) for pos in pop])

    for gen in range(GENS):
        new_pop = []
        elite = np.argsort(fitness_vals)[-ELITE_SIZE:]
        new_pop.extend(pop[elite])

        for _ in range(POPSIZE - ELITE_SIZE):
            parent1_idx = tournament(pop, fitness_vals, k=3)
            parent2_idx = tournament(pop, fitness_vals, k=3)
            child = crossover(pop[parent1_idx], pop[parent2_idx], CROSSOVER_RATE)
            child = mutate(child, MUTATION_RATE, MUTATION_SIGMA)
            new_pop.append(child)

        pop = np.array(new_pop)
        fitness_vals = np.array([fitness(pos) for pos in pop])

    best_idx = np.argmax(fitness_vals)
    return pop[best_idx]
