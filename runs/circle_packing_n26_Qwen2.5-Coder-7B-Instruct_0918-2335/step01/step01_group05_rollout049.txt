import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist
from scipy.stats import uniform

def evaluate_positions(centers):
    n = centers.shape[0]
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + len(pairs), n))
    b_ub = np.zeros(4*n + len(pairs))
    
    for i in range(n):
        A_ub[4*i, i] = 1
        b_ub[4*i] = centers[i, 0]
        A_ub[4*i+1, i] = 1
        b_ub[4*i+1] = 1 - centers[i, 0]
        A_ub[4*i+2, i] = 1
        b_ub[4*i+2] = centers[i, 1]
        A_ub[4*i+3, i] = 1
        b_ub[4*i+3] = 1 - centers[i, 1]
    
    dists = cdist(centers, centers)
    for (i, j), d_ij in zip(pairs, dists[pairs[:, 0], pairs[:, 1]]):
        A_ub[4*n + pairs.index((i, j)), i] = 1
        A_ub[4*n + pairs.index((i, j)), j] = 1
        b_ub[4*n + pairs.index((i, j))] = d_ij
    
    result = linprog(c, A_ub, b_ub, bounds=[(0, 0.5)]*n, method='highs')
    if result.success:
        radii = result.x
        return np.sum(radii), radii
    else:
        return 0, np.zeros(n)

def jitter_centres(centers, jitter_step):
    n = centers.shape[0]
    jittered_centers = centers.copy()
    for i in range(n):
        dx, dy = uniform.rvs(low=-jitter_step, high=jitter_step, size=2)
        jittered_centers[i] = np.clip(jittered_centers[i] + [dx, dy], 0, 1)
    return jittered_centers

def run_packing():
    n = 26
    jitter_step = 0.02
    num_trials = 3
    max_iter = 5000
    sigma_0 = 0.05
    alpha = 0.9995
    temperature = 0.1

    best_centers = None
    best_sum_radii = 0
    best_radii = None
    
    for _ in range(num_trials):
        # Generate initial candidate positions
        grid_centers = np.array([
            [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
            [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
            [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
            [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
            [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
        ])
        hex_centers = []
        h = 0.18
        v = np.sqrt(3)/2 * h
        nrows = 0
        row_offset = 0
        while True:
            row_points = np.array([
                [0.5 + h*i, j*h + row_offset]
                for i in range(nrows)
                for j in range(nrows)
            ])
            hex_centers.extend(row_points)
            if len(hex_centers) >= 26:
                break
            nrows += 1
        offset_centers = np.array([
            [0.25, 0.25], [0.75, 0.25], [0.5, 0.75],
            [0.25, 0.75], [0.75, 0.75]
        ])
        
        initial_centers = np.vstack((grid_centers, hex_centers, offset_centers))
        initial_centers = jitter_centres(initial_centers, jitter_step)
        initial_centers = initial_centers[:26]
        
        # Simulated Annealing
        current_centers = initial_centers
        current_sum_radii = 0
        current_radii = None
        sigma = sigma_0
        for _ in range(max_iter):
            i = np.random.randint(n)
            dx, dy = uniform.rvs(low=-sigma, high=sigma, size=2)
            new_centers = current_centers.copy()
            new_centers[i] = np.clip(new_centers[i] + [dx, dy], 0, 1)
            
            sum_radii, radii = evaluate_positions(new_centers)
            if sum_radii >= current_sum_radii or uniform.rvs() < np.exp((sum_radii - current_sum_radii) / temperature):
                current_sum_radii = sum_radii
                current_centers = new_centers
                current_radii = radii
            
            temperature *= alpha
            sigma *= alpha
        
        if current_sum_radii > best_sum_radii:
            best_sum_radii = current_sum_radii
            best_centers = current_centers
            best_radii = current_radii
    
    final_sum_radii, final_radii = evaluate_positions(best_centers)
    return best_centers, final_radii, final_sum_radii
