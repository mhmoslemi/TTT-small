```python
import numpy as np
from scipy.optimize import minimize

def compute_optimal_radii(pos):
    r = np.zeros(26)
    for i in range(26):
        r[i] = np.min([pos[i, 0], 1 - pos[i, 0], pos[i, 1], 1 - pos[i, 1]])
        for j in range(i + 1, 26):
            d = np.linalg.norm(pos[i] - pos[j])
            if d > 1e-5:
                r[i] = min(r[i], d / 2 - 1e-5)
    return r

def objective(pos_flat):
    pos = pos_flat.reshape(26, 2)
    radii = compute_optimal_radii(pos)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_center = None
    best_radii = None
    best_sum = 0

    for _ in range(30):
        init = np.random.rand(52)
        res = minimize(fun=objective, x0=init, method='SLSQP',
                       bounds=[(0, 1)] * 52, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if res.success and np.all(compute_optimal_radii(res.x.reshape(26, 2)) >= 0):
            r = compute_optimal_radii(res.x.reshape(26, 2))
            cur_sum = np.sum(r)
            if cur_sum > best_sum:
                best_center = res.x.reshape(26, 2)
                best_radii = r
                best_sum = cur_sum

    return best_center, best_radii, best_sum
```
