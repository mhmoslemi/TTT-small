import numpy as np
import scipy.optimize
from scipy.spatial.distance import cdist

N = 26
NUM_VARS = N * 2 + N 

def build_constraints(v):
    pos = v[:NUM_VARS//2].reshape(N, 2)
    r = v[NUM_VARS//2:]
    
    # boundary constraints
    cons = np.empty(4*N, float)
    cons[0:N] = pos[:,0] - r
    cons[N:2*N] = 1 - pos[:,0] - r
    cons[2*N:3*N] = pos[:,1] - r
    cons[3*N:4*N] = 1 - pos[:,1] - r
    
    # pairwise constraints (squared distances)
    dists_squared = cdist(pos, pos)**2
    np.fill_diagonal(dists_squared, np.inf)
    pair_cons = dists_squared - (r[:, np.newaxis] + r[np.newaxis, :])**2
    return np.concatenate((cons, pair_cons.flatten()))

def obj(v):
    return -np.sum(v[NUM_VARS//2:])

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for _ in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(N):
                if i != j:
                    index_ij = i * N + j
                    if index_ij >= NUM_VARS//2 - N:
                        index_ij = index_ij % N
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist - radii[j] > 0:
                        max_r = min(max_r, dist - radii[j])
            radii[i] = max(0.0, min(max_r, 0.5))
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii

def run_packing(nrestarts=100, tolerance=1e-12, verbose=False):
    bounds = [(0, 1.0)] * NUM_VARS
    
    best_res = None
    best_centers = None
    best_radii = None
    
    for _ in range(nrestarts):
        pos = np.random.rand(N, 2)
        r = 0.5 * np.random.rand(N)
        v = np.concatenate((pos.flatten(), r))
        try:
            res = scipy.optimize.minimize(obj, v, bounds=bounds, constraints={'type':'ineq','fun': build_constraints}, options={'maxiter': 2000, 'disp': verbose, 'ftol': tolerance})
        except Exception as e:
            print(f"Re-run {res['nfev']}:\n{e}\n")
            continue
        
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(build_constraints(res.x) >= -1e-10):
                best_res = res
                best_centers = best_res.x[:NUM_VARS//2].reshape(N, 2)
                best_radii = best_res.x[NUM_VARS//2:]

    if best_res is None or best_centers is None or best_radii is None:
        raise ValueError("Optimization failed to find a valid solution")
    
    inflate_radii(best_centers, best_radii)
    
    return best_centers, best_radii, -best_res.fun
