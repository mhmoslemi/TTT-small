import numpy as np
from scipy.optimize import linprog, minimize

def lp_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ineq = []
    b_ineq = []
    
    for i in range(n):
        A_ineq.append([0] * n + [1 if j == i else 0 for j in range(n)])
        b_ineq.append(centers[i, 0])
        A_ineq.append([0] * n + [1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centers[i, 0])
        A_ineq.append([0] * n + [1 if j == i else 0 for j in range(n)])
        b_ineq.append(centers[i, 1])
        A_ineq.append([0] * n + [1 if j == i else 0 for j in range(n)])
        b_ineq.append(1 - centers[i, 1])
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append([-1 for k in range(n)] + [-1 if j == k else 0 for k in range(n)])
            b_ineq.append(-(centers[i] - centers[j]).dot(np.ones(4)))
    
    A_ineq=np.array(A_ineq)
    b_ineq=np.array(b_ineq)
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, method='highs')
    return res.x

def obj(v):
    n = len(v) // 2
    centers = v[:n].reshape(-1, 2)
    radii = lp_radii(centers)
    return -np.sum(radii)

def constraints(v):
    n = len(v) // 2
    centers = v[:n].reshape(-1, 2)
    r = lp_radii(centers)
    
    cons = np.empty(104, float)
    cons[0:26] = centers[:, 0] - r
    cons[26:52] = 1 - centers[:, 0] - r
    cons[52:78] = centers[:, 1] - r
    cons[78:104] = 1 - centers[:, 1] - r
    
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

bounds = [(0, 1)] * 52

def run_packing():
    num_trials = 15
    best_sol = None
    best_sum = 0
    
    for _ in range(num_trials):
        init = np.random.rand(52) * 2 - 1
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        
        if res.success and np.all(constraints(res.x) >= -1e-10):
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sol = (res.x[:n].reshape(26, 2), lp_radii(res.x[:n].reshape(26, 2)), current_sum)
                best_sum = current_sum
    
    return best_sol

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
