    np.random.seed(42)
    cps = np.random.rand(n, 2)
    rs = np.zeros(n)
    eps = 1e-12
    min_step = 1e-5
    jitter_amp = 0.1
    max_passes = 20
    gs = 0.02
    cool = 0.995

        dists = np.sqrt(((cps[:, None, :] - cps[None, :, :])**2).sum(axis=2))

                    cps[j] += disp
