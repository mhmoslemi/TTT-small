import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import Delaunay, distance_matrix
from scipy.stats import randint

def linprog_oracle(pos):
    n = pos.shape[0]
    r = np.ones(n)
    A = np.zeros((104 + 325, n))
    b = np.zeros(104 + 325)
    A[:104, :n:2] = pos[:, np.newaxis]
    A[:104, 1:n:2] = np.arange(n).reshape(-1, 1) % 2
    A[:104, :n:2] -= np.arange(n).reshape(-1, 1) % 2
    A[:104, 1:n:2] -= 1
    A[:104, :n:2] -= pos[:, np.newaxis, np.newaxis]
    A[:104, 1:n:2] += pos[:, np.newaxis, np.newaxis]
    A[:104, :n:2] -= r[:, np.newaxis]
    A[:104, 1:n:2] += 1 - pos[:, np.newaxis]
    A[:104, :n:2] -= r[:, np.newaxis]
    A[:104, 1:n:2] += pos[:, np.newaxis, np.newaxis]
    A[:104, :n:2] -= r[:, np.newaxis]
    pairs = np.triu_indices(n, k=1)
    A[104:104+len(pairs), :n:2] = pos[pairs[0], np.newaxis, np.newaxis]
    A[104:104+len(pairs), 1:n:2] = pos[pairs[1], np.newaxis, np.newaxis]
    A[104:104+len(pairs), :n:2] -= r[pairs[0], np.newaxis] + r[pairs[1], np.newaxis]
    A[104:104+len(pairs), :n:2] += 1e-9
    res = linprog(-r, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return -10, r
    return -res.fun, r

def swarm_optimization():
    n_particles = 40
    n_dims = 52
    inertia = 0.5
    cognitive = 1.5
    social = 1.5
    T0 = 0.05
    cooling_schedule = 0.99
    max_iter = 800
    NP = n_particles
    dim = n_dims
    pbest_val = -np.inf
    all_pbests = np.empty((NP, dim))
    all_gbests = np.empty((NP, dim))
    all_vels = np.empty((NP, dim))
    pbest_vals = np.full(NP, -np.inf)
    
    seeds = np.array([
        [0.146, 0.146],
        [1 - 0.146, 0.146],
        [1 - 0.146, 1 - 0.146],
        [0.146, 1 - 0.146],
        [0.379, 0.379],
        [1 - 0.379, 0.379],
        [1 - 0.379, 1 - 0.379],
        [0.379, 1 - 0.379],
        [0.5, 0.5],
        [(0.5 - 2 * 0.229) / 2, (0.5 - 2 * 0.229) / 2],
        [(0.5 + 2 * 0.229) / 2, (0.5 - 2 * 0.229) / 2],
        [(0.5 + 2 * 0.229) / 2, (0.5 + 2 * 0.229) / 2],
        [(0.5 - 2 * 0.229) / 2, (0.5 + 2 * 0.229) / 2]
    ])
    
    R = np.array([i * 0.1 for i in range(2, 9)])
    R = np.tile(R, (NP, 1)).reshape(-1, 26)
    
    for _ in range(NP):
        seed_idx = randint(0, len(seeds), 1).item()
        init_pos = np.tile(seeds[seed_idx], (26, 1))
        init_pos += np.random.rand(26, 2) * 0.01
        all_pbests[_] = init_pos.copy()
        pbest_vals[_] = -np.inf
        all_gbests[_] = init_pos.copy()
        all_vels[_] = np.zeros(dim)
    
    for iter in range(max_iter):
        T = T0 * cooling_schedule ** iter
        for i in range(NP):
            r1, r2 = np.random.rand(2)
            new_vel = inertia * all_vels[i] + cognitive * r1 * (all_pbests[i] - all_pbests[i]) + social * r2 * (all_gbests[i] - all_pbests[i])
            anneal_prob = np.minimum(1.0, T / T0)
            if np.random.rand() < anneal_prob:
                new_vel = np.random.randn(dim) * 0.02
            new_pos = all_pbests[i] + new_vel
            new_pos[np.where(new_pos < 1e-6)] = 1e-6
            new_pos[np.where(new_pos > 1 - 1e-6)] = 1 - 1e-6
            new_dists = distance_matrix(new_pos, new_pos)
            valid = np.all(new_dists > (R[i][None, :] + R[i][:, None]) - 1e-9)
            if valid:
                val, _ = linprog_oracle(new_pos)
                if val > pbest_vals[i]:
                    pbest_vals[i] = val
                    all_pbests[i] = new_pos.copy()
            if pbest_vals[i] > pbest_val:
                pbest_val = pbest_vals[i]
                all_gbests[i] = all_pbests[i].copy()
    
    best_idx = np.argmax(pbest_vals)
    best_pos = all_gbests[best_idx]
    best_val, best_r = linprog_oracle(best_pos)
    
    return best_pos, best_r, best_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_center = None
    best_radii = None
    best_sum = 0
    
    for _ in range(20):
        res = swarm_optimization()
        if res[2] > best_sum:
            best_center = res[0].reshape(26, 2)
            best_radii = res[1]
            best_sum = res[2]
    
    return best_center, best_radii, best_sum
