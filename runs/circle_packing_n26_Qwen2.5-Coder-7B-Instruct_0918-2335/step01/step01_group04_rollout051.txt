import numpy as np
from scipy.optimize import linprog, minimize
import random
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    tolerances = 1e-9, 1e-12, 1e-7
    
    def solve_radii_lp(centers):
        bounds = [(0, max_radius) for _ in range(n)]
        A = np.zeros((354, n))
        b = np.zeros(354)
        for i in range(n):
            A[i, i] = 1
            b[i] = box_size
        dual_index = 0
        for i in range(n):
            A[354 + 2 * i, i] = 1
            b[354 + 2 * i] = box_size
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                A[354 + 2 * i + 1, i] = 1
                A[354 + 2 * i + 1, j] = -1
                b[354 + 2 * i + 1] = d
        result = linprog([1] * n, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not result.success:
            raise ValueError("LP failed")
        return result.x
    
    def is_within_unit_square(pos):
        return np.all(0 <= pos) & np.all(pos <= 1)
    
    def compute_center_distances(centers):
        return cdist(centers, centers)
    
    def update_positions(centers, constraints, step_size):
        for i in range(n):
            x, y = centers[i]
            
            dx, dy = 0, 0
            
            if constraints[i]['boundary'] == 'left':
                dx += step_size
            elif constraints[i]['boundary'] == 'right':
                dx -= step_size
            elif constraints[i]['boundary'] == 'bottom':
                dy += step_size
            elif constraints[i]['boundary'] == 'top':
                dy -= step_size

            if constraints[i]['neighbour']:
                j = constraints[i]['neighbour']
                d_ij = np.linalg.norm(centers[i] - centers[j])
                u_ji = (centers[j] - centers[i]) / d_ij
                dx += step_size * u_ji[0]
                dy += step_size * u_ji[1]

            dx, dy = np.clip(dx, a_min=-min(step_size, x), a_max=min(step_size, 1-x))
            dy = np.clip(dy, a_min=-min(step_size, y), a_max=min(step_size, 1-y))
            
            centers[i] += [dx, dy]
        return centers
    
    def get_constraint_info(centers, radii):
        constraints = []
        for i in range(n):
            r_i = radii[i]
            neighbors, boundaries = 0, 0
            for j in range(n):
                if i == j: continue
                r_j = radii[j]
                d_ij = np.linalg.norm(centers[i] - centers[j])
                if d_ij <= r_i + r_j:
                    neighbors += 1
            if 2 <= neighbors:
                pass
            elif neighbors == 0:
                boundaries += 1
            if boundaries == 1:
                if centers[i][0] - r_i <= 0:
                    constraints.append({'boundary': 'left', 'neighbour': None})
                elif 1 - centers[i][0] - r_i <= 0:
                    constraints.append({'boundary': 'right', 'neighbour': None})
                elif centers[i][1] - r_i <= 0:
                    constraints.append({'boundary': 'bottom', 'neighbour': None})
                elif 1 - centers[i][1] - r_i <= 0:
                    constraints.append({'boundary': 'top', 'neighbour': None})
            elif neighbors == 1:
                for j in range(n):
                    if i == j: continue
                    r_j = radii[j]
                    d_ij = np.linalg.norm(centers[i] - centers[j])
                    if d_ij <= r_i + r_j:
                        constraints.append({'boundary': None, 'neighbour': j})
        return constraints
    
    def get_new_centers(centers):
        new_centers = centers.copy()
        for _ in range(200):
            new_centers = update_positions(new_centers, get_constraint_info(new_centers, solve_radii_lp(new_centers)), step_size=0.01)
        return new_centers
    
    def refine_with_slsqp(centers):
        def objective(x):
            radii = x[-n:]
            centers = x[:-n].reshape(-1, 2)
            return -np.sum(radii)
        
        bounds = []
        for i in range(n]:
            bounds.append((0, 1))
            bounds.append((0, 1))
            bounds.append((0, box_size))
        
        cons = []
        for i in range(n):
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i: box_size - x[-n][i]
            })
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[i*2] - x[-n][i]
            })
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[i*2+1] - x[-n][i]
            })
            for j in range(i + 1, n):
                dist = np.linalg.norm(x[i*2:i*2+2] - x[j*2:j*2+2])
                cons.append({
                    'type': 'ineq',
                    'fun': lambda x, i=i, j=j, dist=dist: dist - x[-n][i] - x[-n][j]
                })
        
        res = minimize(objective, np.concatenate((centers.flatten(), solve_radii_lp(centers))), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 5000, 'ftol': 1e-9})
        if not res.success:
            raise ValueError("SLSQP failed")
        return res.x[:-n].reshape(-1, 2), res.x[-n:]
    
    best_centers, best_radii, max_sum_radii = np.zeros((n, 2)), np.zeros(n), 0.0
    
    for _ in range(8):
        initial_centers = np.array([[i, j] for i in np.linspace(0, 1, 7) for j in np.linspace(0, 1, 7)])
        if len(initial_centers) > n:
            initial_centers = initial_centers[:n]
        
        while not is_within_unit_square(initial_centers).all() or np.isnan(initial_centers).any():
            initial_centers = np.random.rand(n, 2)
        
        constraints = get_constraint_info(initial_centers, solve_radii_lp(initial_centers))
        best_new_centers = get_new_centers(initial_centers)
        current_sum_radii = np.sum(solve_radii_lp(best_new_centers))
        count = 0
        
        escape_interval = 50
        cooling_interval = 20
        patience = 50
        accept_rate = 0.5
        step_size = 0.01
        
        while count <= patience:
            current_new_centers = get_new_centers(best_new_centers)
            current_sum_radii = np.sum(solve_radii_lp(current_new_centers))
            
            if current_sum_radii > max_sum_radii:
                max_sum_radii = current_sum_radii
                best_centers, best_radii = current_new_centers, solve_radii_lp(best_new_centers)
                count = 0
            else:
                count += 1
                
            step_size /= 2
            if step_size < tolerances[2]:
                step_size = tolerances[2]
            
            if count % escape_interval == 0:
                noise = np.random.randn(n * 2, 1) * (step_size * 2)
                best_new_centers += noise
                
                if not is_within_unit_square(best_new_centers).all():
                    best_new_centers = initial_centers
                    
                constraints = get_constraint_info(best_new_centers, solve_radii_lp(best_new_centers))
                
            if step_size < tolerances[2]:
                constraints = get_constraint_info(best_new_centers, solve_radii_lp(best_new_centers))
                best_new_centers = get_new_centers(best_new_centers)
                if count > patience:
                    break
                
            if len(current_public_centers) == 0:
                break
                
    refined_centers, refined_radii = refine_with_slsqp(best_centers)
    final_radii = solve_radii_lp(refined_centers)
    final_sum_radii = np.sum(final_radii)
    
    return refined_centers, final_radii, final_sum_radii
