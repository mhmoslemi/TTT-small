import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, minimize

def pairwise_dists(centers):
    return cdist(centers, centers)

def solve_radii_lp(centers):
    n = centers.shape[0]
    bounds = [(0, 0.5)] * n
    c = [-1] * n
    eq = None
    A = np.empty((3*n*(n-1)//2 + n*2, n))
    b = np.empty(3*n*(n-1)//2 + n*2)
    
    # Boundary constraints
    k = 0
    for i in range(n):
        A[k, i] = 1
        b[k] = centers[i, 0]
        k += 1
        A[k, i] = 1
        b[k] = 1 - centers[i, 0]
        k += 1
        A[k, i] = 1
        b[k] = centers[i, 1]
        k += 1
        A[k, i] = 1
        b[k] = 1 - centers[i, 1]
        k += 1
    
    # Pairwise constraints
    dists = pairwise_dists(centers)
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1
            A[k, j] = 1
            b[k] = dists[i, j] - 2e-10
            k += 1
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return -res.fun, res.x

def sa_optimize(initial_centers):
    n = len(initial_centers)

    max_iterations = 500
    
    best_sum_radii = -1
    best_centers = initial_centers

        idx = np.random.randint(n)

        epsilon = np.random.normal(0, 100) * 0.001
        
        new_center = initial_centers.copy()
        new_center[idx] += np.array([delta, epsilon])
        new_center[idx] = np.clip(new_center[idx], 0, 1)

            best_sum_radii = new_sum_radii

def run_packing():
    k = int(np.floor(np.sqrt(26)))  # Largest square lattice that fits

        if new_sum_radii - 1e-7 > best_sum_radii:
            best_sum_radii = new_sum_radii
            best_centers = new_centers
