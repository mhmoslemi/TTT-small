import numpy as np
from scipy.optimize import minimize, differential_evolution

def obj_and_constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r  # x - r > 0
    cons[26:52] = 1 - pos[:,0] - r  # 1 - x - r > 0
    cons[52:78] = pos[:,1] - r  # y - r > 0
    cons[78:104] = 1 - pos[:,1] - r  # 1 - y - r > 0
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    diffs_no_diag = np.triu(dists, k=1)
    const_pairs = diffs_no_diag - (r[:,None] + r[None,:])
    penalty = np.sum(np.where(const_pairs < 0, const_pairs**2, 0))
    return -np.sum(r) + 1e6 * penalty, cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    
    res_de = differential_evolution(
        fun=lambda v: obj_and_constraints(v)[0],
        bounds=bounds,
        args=(obj_and_constraints,),
        popsize=15,
        maxiter=500,
        mutation=(0.5, 1.0),
        atol=1e-9,
        pop schwefel=None
    )
    
    if not res_de.success:
        print("Warning: Differential Evolution did not converge to a feasible solution. Returning the best found solution.")
    
    v_best = res_de.x
    res_slsqp = minimize(
        fun=obj_and_constraints,
        x0=v_best,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': lambda v: obj_and_constraints(v)[1]},
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res_slsqp.success:
        print("Warning: SLSQP did not converge successfully. Returning the best found solution.")
    
    centers = res_slsqp.x[:52].reshape(n, 2)
    radii = res_slsqp.x[52:]
    return centers, radii, -obj_and_constraints(res_slsqp.x)[0]
