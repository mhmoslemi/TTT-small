import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.zeros(n)
    epsilon = 1e-12
    min_step = 1e-5
    jitter_amp = 0.1
    max_sub_iter = 20
    gs = 0.02
    cool = 0.995
    
    while gs > min_step:
        np.random.seed(42)
        
        delta = gs * jitter_amp * np.random.rand(1)[0]
        centers += delta * np.random.randn(n, 2)
        centers = np.clip(centers, 0, 1)
        
        for _ in range(max_sub_iter):
            dists = np.sqrt(((centers[:,None,:] - centers[None,:,:])**2).sum(axis=2))
            mask  = dists < radii[:,None] + radii[None,:] - epsilon
            if np.sum(mask) == 0:
                break
            
            for i in range(n):
                for j in range(i + 1, n):
                    if mask[i, j]:
                        d = centers[i] - centers[j]
                        dist = dists[i, j]
                        sep = radii[i] + radii[j] - dist + epsilon
                        disp = 0.5 * sep * (d / dist)
                        centers[i] -= disp
                        centers[j] += disp
            
            centers = np.clip(centers, 0, 1)
        
        gs *= cool
    
   objective = -np.sum(radii)
    constraint = {
        'type': 'ineq',
        'fun': lambda v: -(v[:n] - v[n:] - epsilon)
    }
    
    result = minimize(fun=objective, x0=np.concatenate([centers.flatten(), radii]),
                      method='SLSQP', bounds=[(0, 1)] * (2 * n) + [(0, 1)] * n, options={'ftol': 1e-9, 'disp': False})
    
    final_centers = result.x[:n].reshape(n, 2)
    final_radii = result.x[n:]
    
    return final_centers, final_radii, -result.fun
