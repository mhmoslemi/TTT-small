import numpy as np
from scipy.optimize import linprog, minimize

np.random.seed(42)

def solve_radius(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    # Boundary constraints
    for i in range(n):
        A_ub.append([1, 0] * (n - 1 + i))  # r_i <= x_i
        A_ub.append([1, 0] * (n - 1 + i))  # r_i <= 1−x_i
        A_ub.append([0, 1] * (n - 1 + i))  # r_i <= y_i
        A_ub.append([0, 1] * (n - 1 + i))  # r_i <= 1−y_i
        b_ub.append(pos[i, 0])
        b_ub.append(1 - pos[i, 0])
        b_ub.append(pos[i, 1])
        b_ub.append(1 - pos[i, 1])

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sum((pos[i] - pos[j]) ** 2)
            A_ub.append([1, 0] * (n - 1 + i) + [1, 0] * (n - 1 + j))
            b_ub.append(d)
    
    c = -np.ones(2 * n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
    r_opt = result.x[:n]
    sum_r = -result.fun
    return r_opt, sum_r

def fitness(pos):
    r, s_r = solve_radius(pos)          # LP
    return -s_r                         # GA minimises, so negative of sum

def crossover(parent1, parent2, rate):
    child = np.copy(parent1)
    for i in range(parent1.shape[0]):
        if np.random.rand() < rate:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    child = np.copy(individual)
    for i in range(individual.shape[0]):
        if np.random.rand() < rate:
            child[i] += np.random.normal(0, sigma)
            child[i] = np.clip(child[i], 0, 1)
    return child

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(pop.shape[0], k, replace=False)
    best_idx = indices[np.argmax(fitness_vals[indices])]
    return best_idx

    pop = np.random.rand(popsize, 52)  # 26 positions of 2D coordinates
    
    for gen in range(generations):
        fitness_vals = np.empty(popsize)
        for i in range(popsize):
            fitness_vals[i] = fitness(pop[i, :].reshape(26, 2))
        
        elite_indices = np.argsort(fitness_vals)[-elite_size:]
        elite = pop[elite_indices, :]
        
        offspring = np.empty((popsize-elite_size, 52))
        for i in range((popsize-elite_size)//2):
            parent1 = pop[tournament(pop, fitness_vals, 3), :]
            parent2 = pop[tournament(pop, fitness_vals, 3), :]
            child1 = crossover(parent1, parent2, crossover_rate)
            child2 = crossover(parent2, parent1, crossover_rate)
            child1 = mutate(child1, mutation_rate, mutation_sigma)
            child2 = mutate(child2, mutation_rate, mutation_sigma)
            child1 = mutate(child1, mutation_rate, mutation_sigma)
            child2 = mutate(child2, mutation_rate, mutation_sigma)
            offspring[i*2] = child1
            offspring[i*2+1] = child2
        
        pop = np.vstack((elite, offspring[:popsize-len(elite)]))
    
    best_idx = np.argmax([fitness(pop[i, :].reshape(26, 2)) for i in range(popsize)])
    best_pos = pop[best_idx, :].reshape(26, 2)
    return best_pos
