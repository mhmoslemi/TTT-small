import numpy as np
from scipy.optimize import linprog

def contact_graph_constraints(n, graph, d_bounds):
    A_ineq = []
    b_ineq = []
    r_var = np.arange(n)
    
    # Non-overlap constraints
    for (i, j), _ in np.ndenumerate(graph):
        if i != j:
            if graph[i, j] > 0:
                A_ineq.append([1, 1, -1, -1])
                b_ineq.append(d_bounds[i, j] + 1e-12)
            else:
                A_ineq.append([1, 1, -1, -1])
                b_ineq.append(d_bounds[i, j] - 1e-12)
    
    # Wall constraints (for simplicity, assume left wall only here)
    for i, d in enumerate(d_bounds[:, 1]):
        if d < np.inf:
            A_ineq.append([1, -d_var[0]])
            b_ineq.append(d)
    
    # Non-negativity constraints
    for i in range(n):
        A_ineq.append([1, 0])
        b_ineq.append(0)
    
    return np.array(A_ineq), np.array(b_ineq)

def run_packing():
    np.random.seed(42)
    n = 26
    vertices = np.arange(n)
    
    # Generate a central 5x5 grid graph with an extra circle
    central_grid = np.zeros((n, n), dtype=int)
    x, y = np.ogrid[:5, :5]
    central_grid[x + 2, y] = 1
    central_grid = np.pad(central_grid, ((0, 1), (0, 1)), mode='constant')
    central_grid[5, 1] = 2
    
    # Generate additional shifted and perturbed graph variants
    graph_variants = [central_grid]
    for k in range(1, n):
        graph_variants.append(np.roll(central_grid, k, axis=1))
    
    for i, d in enumerate(central_grid[:, 1]):
        if d < np.inf:
            central_grid[i, 1] = d
    
    # Calculate distance bounds for each edge
    d_bounds = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            if np.any(central_grid[i, :] == central_grid[j, :]):
                shared_1d_coords = np.where(central_grid[i, :] == central_grid[j, :])[0]
                d_bounds[i, j] = 2 * np.dot([np.abs(central_grid[i, k] - central_grid[i, k + 1]) for k in shared_1d_coords], [1, 1])
            else:
                d_bounds[i, j] = np.inf
    
    # Enumerate and evaluate each graph
    best_centers = None
    best_radii = None
    best_sum = 0
    
    for graph in graph_variants:
        A_ineq, b_ineq = contact_graph_constraints(n, graph, d_bounds)
        res = linprog([-1] * n, A_ub=A_ineq, b_ub=b_ineq, method='highs')
        if not res.success:
            continue
        r = res.x
        if r.sum() > best_sum:
            best_sum = r.sum()
            best_radii = r
            best_centers = np.cumsum(r[np.newaxis, :] * np.diff(vertices[:, np.newaxis]), axis=1)
    
    return best_centers, best_radii, best_sum
