import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def solve_radii(centers):
    n = centers.shape[0]
    A = np.vstack([
        centers[:, 0] - centers,
        1 - centers[:, 0] - centers,
        centers[:, 1] - centers,
        1 - centers[:, 1] - centers
    ])
    A = A.T.reshape(-1, 2 * n)
    A = np.hstack([A, np.eye(n)])
    b = np.full(4 * n + n, 1)
    c = -np.ones(n)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, 0.5)] * n)
    if result.success:
        return -result.fun, result.x[-n:]
    else:
        raise ValueError("LP did not converge")

def evaluate_fitness(individuals):
    centers = individuals.reshape(-1, 2)
    sum_radii, radii = solve_radii(centers)
    return sum_radii

def crossover(parent1, parent2):
    i1, i2 = np.random.choice(np.arange(52), 2, replace=False)
    if i1 > i2:
        i1, i2 = i2, i1
    child = np.copy(parent1)
    child[i1 + 1:i2] = parent2[i1 + 1:i2]
    return child

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, size=individual.shape)
    individual = np.clip(individual, 0, 1)
    return individual

def clamp_to_square(individual):
    individual = np.clip(individual, 0, 1)
    return individual

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    def selection(population):
        parent_indices = np.empty((population_size, 2), dtype=int)
        for i in range(population_size):
            contestants = np.random.choice(population_size, tournament_size, replace=False)
            parent_indices[i] = contestants[np.argmax(evaluate_fitness(population[contestants]))]
        return population[parent_indices]

    population = np.random.rand(population_size, 2 * n)
    population = np.apply_along_axis(clamp_to_square, 1, population)
    best_sum = -np.inf
    best_centers = None

    for generation in range(generations):
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = selection(population)[:2]
            if np.random.rand() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            child = mutate(child, initial_sigma)
            child = clamp_to_square(child)
            new_population.append(child)
            weight_this = evaluate_fitness(np.array(child))
            if weight_this > best_sum:
                best_sum = weight_this
                best_centers = child.reshape(-1, 2)
        sorted_indices = np.argsort(evaluate_fitness(population))
        population = np.concatenate([population[sorted_indices[-elitism:]], new_population])
        initial_sigma *= sigma_decay

    centers = best_centers.reshape(-1, 2)
    sum_radii, radii = solve_radii(centers)

    def final_refinement(centers, radii):
        bounds = [(0, 0.5)] * n + [(0, 0.5)] * n
        x0 = np.concatenate([centers, radii])
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - x[n + i]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i] - x[n + i]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[n + i] - x[i]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[n + i] - x[i]})
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: x[n + i] + x[n + j] - (x[i] - x[j])**2 - (x[i] - x[j])**2})
        res = minimize(fun=lambda x: -np.sum(x[:n]), x0=x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'disp': False})
        if res.success:
            final_centers = res.x[:n].reshape(-1, 2)
            final_radii = res.x[n:]
            return final_centers, final_radii, -res.fun
        else:
            return centers, radii, sum_radii

    final_centers, final_radii, final_sum_radii = final_refinement(centers, radii)
    return final_centers, final_radii, final_sum_radii
