import numpy as np
from scipy.optimize import minimize, linprog, Bounds

# Constants
POPSIZE = 80
GENS = 250
ELITE_SIZE = int(POPSIZE * 0.1)
CROSSOVER_RATE = 0.9
MUTATION_RATE = 0.2
MUTATION_SIGMA = 0.03

# Function: Solve for optimal radii given positions using LP
def solve_lradii(pos):
    A_ub = []
    b_ub = []
    n = pos.shape[0]
    for i in range(n):
        A_ub.append([1 if j == i else 0 for j in range(n)])  # r_i <= x_i
        A_ub.append([1 if j == i else 0 for j in range(n)])  # r_i <= 1-x_i
        A_ub.append([1 if j == i else 0 for j in range(n)])  # r_i <= y_i
        A_ub.append([1 if j == i else 0 for j in range(n)])  # r_i <= 1-y_i
        for j in range(i + 1, n):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            d = np.sqrt(dx**2 + dy**2)
            A_ub.append([1 if k == i else 0 for k in range(n)])  # r_i + r_j <= dist_ij
            A_ub.append([1 if k == j else 0 for k in range(n)])  # r_i + r_j <= dist_ij
            b_ub.append(d - 1e-12)
    A_ub = np.array(A_ub, dtype=float)
    b_ub = np.array(b_ub, dtype=float)
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=(0, 0.5))
    r_opt = result.x
    sum_r = result.fun[-1]
    return r_opt, -sum_r

# Function: Evaluate fitness (negative sum of radii)
def fitness(pos):
    r_opt, sum_r = solve_lradii(pos)
    return -sum_r

# Function: Gaussian mutation
def mutate(individual, rate, sigma):
    mask = np.random.rand(len(individual)) < rate
    individual[mask] += np.random.normal(0, sigma, size=individual[mask].shape)
    individual = np.clip(individual, 0, 1)
    return individual

# Function: Uniform crossover
def crossover(parent1, parent2, rate):
    child = np.where(np.random.rand(len(parent1)) < rate, parent2, parent1)
    return child

# Function: Tournament selection
def tournament(pop, fitness_vals, k):
    selected = []
    for _ in range(k):
        candidates = np.random.choice(np.arange(POPSIZE), size=3, replace=False)
        indices = np.argsort(fitness_vals[candidates])[:2]
        winner = candidates[indices[1]]
        selected.append(winner)
    chosen_idx = np.random.choice(selected, size=1, replace=False)
    return pop[chosen_idx]

# Function: Run GA
def ga_best_positions():
    rng = np.random.RandomState(42)
    pop = rng.uniform(0, 1, size=(POPSIZE, 26, 2))
    fitness_vals = np.array([fitness(pop[i]) for i in range(POPSIZE)])

    elite_idx = np.argsort(fitness_vals)[-ELITE_SIZE:]  # Indices of the elite
    elite = pop[elite_idx]

    for g in range(GENS):
        offspring = elite.copy()
        for i in range(ELITE_SIZE, POPSIZE):
            parent_i = tournament(pop, fitness_vals, 3)
            parent_j = tournament(pop, fitness_vals, 3)
            child = crossover(parent_i, parent_j, CROSSOVER_RATE)
            child = mutate(child, MUTATION_RATE, MUTATION_SIGMA)
            offspring[i] = child
        pop = offspring
        fitness_vals = np.array([fitness(pop[i]) for i in range(POPSIZE)])

        elite_idx = np.argsort(fitness_vals)[-ELITE_SIZE:]
        elite = pop[elite_idx]

    return pop[elite_idx[0]]

# Function: Improve solution with local refinement (SLSQP)
def polish_solution(pos):
    r, sum_r = solve_lradii(pos)
    bounds = Bounds(0, 1, keep_feasible=True)
    objective = lambda v: -np.sum(v[:26])
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i] - r[i] - 1e-12})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[i] - r[i] - 1e-12})
    for i in range(26):
        for j in range(i + 1, 26):
            dx = v[:26][i] - v[:26][j]
            dy = v[26:][i] - v[26:][j]
            d = np.sqrt(dx ** 2 + dy ** 2)
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: d - (r[i] + r[j]) - 1e-12})
    options = {
        'maxiter': 10000,
        'ftol': 1e-12,
        'disp': False
    }
    initial_guess = np.concatenate([pos.flatten(), r])
    result_slsqp = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
    best_centers = result_slsqp.x[:26].reshape(26, 2)
    best_radii = result_slsqp.x[26:]
    sum_radii = -result_slsqp.fun
    return best_centers, best_radii, sum_radii
