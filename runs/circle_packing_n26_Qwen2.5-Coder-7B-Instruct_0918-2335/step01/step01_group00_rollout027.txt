import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def obj(v):
        r = v[27:]                      # last 27 entries
        return -np.sum(r)

    def constraints(v):
        pos = v[:27].reshape(9, 3)      # (x,y,r) triplets
        r = pos[:, 2]

        # boundary terms – 3*9
        cons = np.empty(27, float)
        cons[0:9] = pos[:, 0] - r        # left
        cons[9:18] = 1 - pos[:, 0] - r   # right
        cons[18:27] = pos[:, 1] - r      # bottom

        # pairwise terms – 9 choose 2
        diffs = pos[:, None, :2] - pos[None, :, :2]   # (9,9,2)
        dists = np.sqrt((diffs**2).sum(axis=2))
        mask = np.triu(np.ones((9, 9), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    bounds = [(0, 1)] * 9             # x
    bounds += [(0, 1)] * 9             # y
    bounds += [(0, 0.15)] * 9           # radii

    nrestarts = 20
    best_r = None
    best_res = None

    for _ in range(nrestarts):
        init = np.random.rand(27)
        init[27:] = 0.05 * np.random.rand(9)
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type': 'ineq',
                                                   'fun': constraints},
                       options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                best_r = res.x[27:]

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers = best_res.x[:27].reshape(9, 3)
    radii = best_res.x[27:]
    sum_radii = -best_res.fun
    return centers[:, :2], radii, sum_radii
