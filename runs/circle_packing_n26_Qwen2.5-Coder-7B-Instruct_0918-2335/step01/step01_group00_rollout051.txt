import numpy as np
from scipy.optimize import linprog, minimize

def compute_radii_and_value(centres):
    n = centres.shape[0]
    w_limits = np.minimum.reduce([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]])
    d_ij = np.sqrt(np.sum((centres[:, None, :] - centres[:, :, None])**2, axis=2))
    A_ub = np.zeros((n * (n - 1) // 2, n + 1))
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[i * (n - 1) // 2 + j - i - 1, i] = 1
            A_ub[i * (n - 1) // 2 + j - i - 1, j] = 1
    A_ub[:, n] = d_ij[np.triu_indices(n, 1)]
    c = -np.ones(n)
    b_ub = d_ij[np.triu_indices(n, 1)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=tuple([(0, lim) for lim in w_limits]))
    if not res.success or res.fun == -np.inf:
        return None, -np.inf
    values = res.x
    value = -res.fun
    return values, value

def run_packing():
    np.random.seed(42)
    nrestarts = 150
    budget = 2000
    temp0 = 0.1
    gamma = 0.97

    best_value = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(nrestarts):
        centres = np.random.rand(52).reshape(-1, 2) * 0.8 + 0.1  # jittered hexagonal lattice
        optimize_args = {'method': 'BFGS', 'bounds': np.vstack((np.zeros((52, 2)), np.ones((52, 2)))).T}
        def objective(x):
            values, value = compute_radii_and_value(x.reshape(-1, 2))
            if values is None:
                return 1e6
            return -value
        result = minimize(objective, x0=centres.flatten(), **optimize_args)
        values, value = compute_radii_and_value(result.x.reshape(-1, 2))
        if value > best_value:
            best_value = value
            best_centers = result.x.reshape(-1, 2)
            best_radii = values

    if best_value < 2.636:
        raise ValueError("Failed to find a high enough sum of radii")

    return best_centers, best_radii, best_value
