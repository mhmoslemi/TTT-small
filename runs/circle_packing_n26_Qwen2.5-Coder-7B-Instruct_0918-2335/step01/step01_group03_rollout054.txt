import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def generate_positions(family):
    if family == 'grid':
        x, y = np.meshgrid(np.linspace(0, 1, 5), np.linspace(0, 1, 5))
        centers = np.vstack((x.ravel(), y.ravel())).T
        # Add one circle at the center of the grid
        centers = np.vstack((centers, np.array([0.5, 0.5])))
    elif family == 'hex':
        angles = np.arange(0, 2*np.pi, 2*np.pi/5)
        centers = np.array([[np.cos(a), np.sin(a)] for a in angles])
        centers = np.vstack((centers, np.array([0.5, 0.5])))
        centers = np.vstack((centers, centers + np.array([0, 1])*0.5))
        centers = np.vstack((centers, centers + np.array([1, 0])*0.5))
        centers = np.vstack((centers, centers + np.array([0.5, 00.5])*0.5))
    elif family == 'hybrid':
        centers = np.array([[0, 0], [0, 1], [1, 0], [1, 1], [0.5, 0.5]])
        centers = np.vstack((centers, np.array([[0.2, 0.2], [0.2, 0.8], [0.8, 0.2], [0.8, 0.8]])))
        centers = np.vstack((centers, centers + np.array([0, 0.25])*0.5))
        centers = np.vstack((centers, centers + np.array([0.25, 0])*0.5))
        centers = np.vstack((centers, centers + np.array([0, -0.25])*0.5))
        centers = np.vstack((centers, centers + np.array([-0.25, 0])*0.5))
    elif family == 'random':
        centers = np.random.rand(26, 2)
    else:
        raise ValueError("Unknown family")
    # Ensure points are within the unit square
    centers = np.clip(centers, 0, 1)
    return centers

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26
    bounds = [(0, 1)] * (2 * n)  # 2 * x, y coordinates for each circle
    x0 = np.random.rand(2 * n)  # Initial guess for centers
    
    def obj(v):
        return -np.sum(v[:2*n])
    
    def constraints(v):
        pos = v[:2*n].reshape(n, 2)  # (x,y) pairs
        # Boundary terms
        boundary_cons = np.empty(n, float)
        boundary_cons[0:n] = pos[:, 0] - v[2*n].clip(0, None)
        boundary_cons[n:2*n] = 1 - pos[:, 0] - v[2*n].clip(0, None)
        boundary_cons[2*n:3*n] = pos[:, 1] - v[2*n].clip(0, None)
        boundary_cons[3*n:4*n] = 1 - pos[:, 1] - v[2*n].clip(0, None)
        # Pairwise terms
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        distances = np.maximum(v[2*n] + v[2*n+1:], dists) * (dists > 1e-6)
        return np.concatenate([boundary_cons, distances.ravel()])
    
    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9})
    
    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:2*n].reshape(n, 2)
        radii = np.array([np.clip(res.x[2*n], 0, None), np.clip(res.x[2*n+1], 0, None)], dtype=float)
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")
