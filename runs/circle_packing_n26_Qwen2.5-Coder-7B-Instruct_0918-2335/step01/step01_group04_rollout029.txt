import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def lp_radii(centers):
        A = np.zeros((2 * n, n))
        b = np.zeros(2 * n)
        for i in range(n):
            A[i, i] = -1
            b[i] = centers[i, 0]
            A[n + i, i] = -1
            b[n + i] = box_size - centers[i, 0]
        dists = pdist(centers)
        mask = dists < 2 * max_radius
        A = np.vstack([A, -np.eye(n)[mask] + np.eye(n)[np.argsort(mask)[-n + 1:][::-1]]])
        b = np.hstack([b, [max_radius] * (n - 1)])
        result = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, max_radius)] * n, method='highs')
        return -result.fun, result.x
    
    def push_away(centers, radii):
        new_centers = np.array(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            active_constraints = []
            for j in range(n):
                if i != j:
                    dx, dy = x - centers[j][0], y - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist <= r + radii[j]:
                        active_constraints.append((j, (dist - r - radii[j], dx, dy)))
            if active_constraints:
                dx_sum, dy_sum = 0, 0
                for _, (delta, dx, dy) in active_constraints:
                    if abs(dx) >= abs(dy):
                        if dx > 0:
                            new_centers[i, 0] += delta / abs(dx)
                        else:
                            new_centers[i, 0] -= delta / abs(dx)
                    else:
                        if dy > 0:
                            new_centers[i, 1] += delta / abs(dy)
                        else:
                            new_centers[i, 1] -= delta / abs(dy)
            else:
                if x == 0:
                    new_centers[i, 0] += r
                elif x == box_size:
                    new_centers[i, 0] -= r
                if y == 0:
                    new_centers[i, 1] += r
                elif y == box_size:
                    new_centers[i, 1] -= r
            new_centers[i] = np.clip(new_centers[i], 0, box_size)
        return new_centers
    
    def objective_and_constraints(x):
        centers = x.reshape(n, 2)
        radii, wall_cons, pair_cons = lp_radii(centers)
        cons = np.concatenate([wall_cons, pair_cons])
        return (-radii, cons)
    
    initial_centers = np.random.rand(n, 2)
    initial_guess = initial_centers.flatten()
    
    max_iter = 200
    cool_interval = 20
    escape_interval = 50
    eps = 1e-7
    pause_epochs = 5
    alpha = 0.01
    
    best_centers, best_radii, best_sum = None, None, 0.0
    
    for _ in range(3):
        centers = np.random.rand(n, 2) if _ > 0 else np.array([
            (i % 5 + (i // 5 % 2) / 2, (i // 5) / 4 + 0.25) for i in range(n)
        ])
        for iter in range(max_iter):
            _, new_radii = lp_radii(centers)
            new_centers = push_away(centers, new_radii)
            _, wall_cons, pair_cons = lp_radii(new_centers)
            cons = np.concatenate([wall_cons, pair_cons])
            if cons[-1] > best_sum + eps:
                best_centers, best_radii, best_sum = new_centers, new_radii, np.sum(new_radii)
            if iter % cool_interval == 0:
                alpha *= 0.5
            if iter % escape_interval == 0:
                centers = new_centers + (np.random.rand(n, 2) - 0.5) * alpha / 2
                centers = np.clip(centers, 0, box_size)
            if new_centers == centers:
                break
    
    res = minimize(
        fun=lambda x: -objective_and_constraints(x)[0],
        x0=best_centers.flatten(),
        method='SLSQP',
        bounds=[(0, box_size)] * 2 * n + [(0, max_radius)] * n,
        constraints={'type': 'ineq', 'fun': lambda x: objective_and_constraints(x)[1]},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:2 * n].reshape(n, 2)
    radii = res.x[2 * n:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
