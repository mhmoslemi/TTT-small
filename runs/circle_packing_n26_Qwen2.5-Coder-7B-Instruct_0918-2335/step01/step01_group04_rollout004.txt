import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    n = len(centers)
    A = np.vstack((np.eye(n), np.eye(n), -(np.ones((n, 2))).T * 2, 
                    np.tile(np.eye(n) - np.eye(n, k=1), 2), 
                    np.tile(np.eye(n) - np.eye(n, k=-1), 2)))
    b = np.concatenate((np.ones(n), 
                       np.ones(n), 
                       np.zeros(2*n), 
                       np.full(n, 1), 
                       np.full(n, 1)))
    c = np.zeros(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    return np.where(res.success, res.x, 0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    n = 26
    max_radius = 0.5
    random_seeds = 8
    max_iter = 200
    escape_interval = 50
    cooling_interval = 20
    accept_rate = 0.5
    patience = 5
    step_size = 0.01
    tolerance = 1e-9
    
    best_centers = None
    best_radii = None
    best_sum_radii = 0

    for _ in range(random_seeds):
        centers = np.random.rand(n, 2)
        for j in range(5):
            y = 0.577 * (2*j + 1) * np.sqrt((3 * np.log(2)) / np.pi)
            x = np.sqrt(3 * (np.log(2) + np.log(2) * y * y))
            if x <= 1 and y <= 1:
                centers[j*3 + 0] = [x, y]
                centers[j*3 + 1] = [1 - x, y]
                centers[j*3 + 2] = [x, 1 - y]
                
        for j in range(7):
            offset = 4 * np.sqrt(np.log(2) / (np.pi * n)) * np.sqrt(j)
            centers[j*3 + 0] = [0.5, 0.5 + offset / 2]
            centers[j*3 + 1] = [0.5 + offset / 2, 0.5]
            centers[j*3 + 2] = [0.5, 0.5 - offset / 2]
            centers[j*3 + 3] = [0.5 - offset / 2, 0.5]
        
        for iteration in range(max_iter):
            radii = solve_radii_lp(centers)
            sum_radii = np.sum(radii)

            if sum_radii > best_sum_radii:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum_radii = sum_radii

            # Identify active constraints
            A = np.vstack((np.diag(np.full(n, -1)), np.diag(np.full(n, 1)), 
                           np.tile(np.eye(n) - np.eye(n, k=1), 2), 
                           np.tile(np.eye(n) - np.eye(n, k=-1), 2)))
            b = np.concatenate((-radii, radii, np.zeros(2*n), np.zeros(n), np.zeros(n)))
            res = linprog(radii, A_ub=A, b_ub=b, bounds=(0.02, None), method="highs")
            new_radii = res.x
            new_sum_radii = np.sum(new_radii)

            if new_sum_radii > sum_radii:
                centers[:, 0] += (new_radii - radii) * (centers[:, 0] - best_centers[:, 0])
                centers[:, 1] += (new_radii - radii) * (centers[:, 1] - best_centers[:, 1])
            else:
                centers[:, 0] -= step_size * (best_centers[:, 0] - centers[:, 0])
                centers[:, 1] -= step_size * (best_centers[:, 1] - centers[:, 1])

            step_size *= 0.5

            if iteration % cooling_interval == 0:
                step_size = 0.01

            if iteration % escape_interval == 0:
                perturbation = step_size * np.random.normal(size=centers.shape)
                centers = np.clip(centers + perturbation, 0, 1)

            if iteration >= max_iter / 2 and new_sum_radii <= best_sum_radii:
                if iteration == max_iter / 2:
                    iteration = 0
                step_size *= 0.1

            if np.allclose(radii, new_radii, atol=1e-9):
                break

    if best_centers is None or best_radii is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    # Fine-tuning with SLSQP
    res = minimize(lambda x: -np.sum(x[52:]), np.concatenate((best_centers.flatten(), best_radii)),
                  constraints={'type': 'ineq', 'fun': lambda x: constraints(x[:52], x[52:])},
                  bounds=[[0, 1]] * n * 2 + [(0, max_radius)] * n,
                  options={'maxiter': 2000, 'ftol': 1e-9})
    
    if res.success:
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        return best_centers, best_radii, best_sum_radii

def constraints(centers, radii):
    n = len(centers)
    pairs = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    center_differences = centers[pairs[:, 0]] - centers[pairs[:, 1]]
    distances = np.sqrt(np.sum(center_differences ** 2, axis=1))
    return distances - radii[pairs[:, 0]] - radii[pairs[:, 1]]

n = 26
centers, radii, sum_radii = run_packing()
print(sum_radii)
