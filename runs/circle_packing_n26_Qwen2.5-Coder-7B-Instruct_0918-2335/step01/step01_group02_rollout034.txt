import numpy as np
from scipy.optimize import linprog, minimize
from cma import CMAEvolutionStrategy

def solve_radius(positions):
    n = positions.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append([1] + [0] * i + [0] * (n - i - 1))
        b_ub.append(positions[i, 0])
        A_ub.append([1] + [0] * i + [0] * (n - i - 1))
        b_ub.append(1 - positions[i, 0])
        A_ub.append([1] + [0] * i + [0] * (n - i - 1))
        b_ub.append(positions[i, 1])
        A_ub.append([1] + [0] * i + [0] * (n - i - 1))
        b_ub.append(1 - positions[i, 1])

    for i in range(n):
        for j in range(i + 1, n):
            dx = positions[i, 0] - positions[j, 0]
            dy = positions[i, 1] - positions[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            A_ub.append([1] + [1] + [0] * (n - 2))
            A_ub[-1][i + 1] += 1
            A_ub[-1][-1] -= dist
            b_ub.append(0)

    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    radii = result.x
    sum_r = result.fun * -1
    return radii, sum_r

def fitness(positions):
    radii, sum_r = solve_radius(positions)
    return sum_r

def crossover(parent1, parent2, rate):
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < rate:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(individual, rate, sigma):
    mutated = np.copy(individual)
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, sigma)
            mutated[i] = max(0, min(mutated[i], 1))
    return mutated

def tournament(pop, fitness_vals, k):
    selected = []
    for _ in range(k):
        indices = np.random.choice(len(pop), 3, replace=False)
        index = max(zip(fitness_vals[indices], indices))[-1]
        selected.append(index)
    return np.random.choice(selected)

def ga_best_positions():
    n = 26
    popsize = 80
    gens = 250
    elite_size = int(0.1 * popsize)
    crossover_rate = 0.9
    mutation_rate = 0.2
    mutation_sigma = 0.03

    pop = np.random.rand(popsize, 2 * n)
    fitness_vals = np.array([fitness(p) for p in pop])

    for gen in range(gens):
        pop Elite = pop[fitness_vals.argsort()[-elite_size:]]
        offspring = []

        for _ in range(popsize - elite_size):
            parent1 = pop[tournament(pop, fitness_vals, 3)]
            parent2 = pop[tournament(pop, fitness_vals, 3)]
            child = crossover(parent1, parent2, crossover_rate)
            child = mutate(child, mutation_rate, mutation_sigma)
            offspring.append(child)

        pop = np.vstack((pop Elite, offspring))
        fitness_vals = np.array([fitness(p) for p in pop])

    best_pos_idx = np.argmax(fitness_vals)
    best_pos = pop[best_pos_idx]
    return best_pos

    dim = 2 * n
    x0 = np.vstack((positions, np.zeros(n))).flatten()
    bounds = [(0, 1)] * dim

        'maxiter': 4000,
        'ftol': 1e-12,
        'disp': False
    }

    def objective(v):
        pos = v[:dim//2].reshape((-1, 2))
        r = v[dim//2:]
        return -np.sum(r) - 1e-9 * np.sum(np.sqrt((pos[:, None, :] - pos[None, :, :])**2) - r[:, None] - r[None, :] - 1e-9)

    def constraint(v):
        pos = v[:dim//2].reshape((-1, 2))
        r = v[dim//2:]
        cons = []
        for i in range(n):
            cons.append(pos[i, 0] - r[i])
            cons.append(1 - pos[i, 0] - r[i])
            cons.append(pos[i, 1] - r[i])
            cons.append(1 - pos[i, 1] - r[i])
        for i in range(n):
            for j in range(i + 1, n):
                cons.append(np.sqrt((pos[i, 0] - pos[j, 0])**2 + (pos[i, 1] - pos[j, 1])**2) - (r[i] + r[j]) - 1e-9)
        return cons

    constraints = {'type': 'ineq', 'fun': constraint}

    result = minimize(fun=objective, x0=x0, bounds=bounds, method='SLSQP', constraints=constraints, options=options)
    positions = result.x[:dim//2].reshape((-1, 2))
    radii = result.x[dim//2:]
    sum_r = -result.fun
    return positions, radii, sum_r
