```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

# Precompute adjacency matrix and distance matrix
# Distances are normalized by subtracting twice the radius from them, to enforce the radii sum constraint
adj = np.ones((26, 26)) - np.eye(26)
dist = squareform(pdist(np.eye(26)))
dist /= (1 + 1e-12)
for i in range(26):
    for j in range(i + 1, 26):
        adj[i, j] = 0 if dist[i, j] < 1e-12 else 1

# Convert to sparse matrices to save memory
from scipy.sparse import csr_matrix
A_ub = csr_matrix((adj.ravel(), (np.arange(26 * 26), np.kron(range(26), range(26)))))
b_ub = np.ones(26 * 26)

# Initialize the population
POP = 30
EVAL = 4000
p = []
r_sum = []
for strategy in ['hex', 'rand', 'grid', 'corner']:
    p.extend([strategy] * (10 if strategy == 'corner' else 4))
    for _ in range(4 if strategy == 'corner' else 10):
        if strategy == 'hex':
            centers = np.array([[i, j] for i in np.linspace(-0.05, 0.95, 5) for j in np.linspace(-0.05, 0.95, 5)])
            centers = centers[:26] + np.random.uniform(-0.01, 0.01, size=(26, 2))
        elif strategy == 'rand':
            centers = np.random.rand(26, 2)
        elif strategy == 'grid':
            centers = np.zeros((25, 2))
            centers += np.array([[i / 4 + 1 / 8, j / 4 + 1 / 8] for i in range(4) for j in range(4)])
            centers[12, :] += [0.25, 0.25]
            centers[-1, :] += [0.25, 0.25] * 4
        elif strategy == 'corner':
            corners = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
            centers = corners + np.random.uniform(0, 0.25 - 0.01, (4, 2))
            centers = np.concatenate((centers, np.random.rand(22, 2)))
        p[-1].append(centers)
        r_sum.append(0)  # Placeholder for radii sum

# Precompute LP solution
def lp_opt(p):
    A = np.hstack([csr_matrix((np.ones(26 * 26), (np.arange(26 * 26), np.kron(range(26), range(26)))))], A_ub.toarray()])
    b = np.hstack([np.ones(26 * 26), b_ub])
    res = linprog(-np.ones(26), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    r = -res.fun * np.ones(26)
    for i in range(26):
        for j in range(26):
            if np.isclose(A[i * 26 + j, 26:], 1):
                r[i] += A[i * 26 + j, 26:].dot(res.x[26:])
    r[r < 0] = 0
    return r

# Initialize the population with radii
for i in range(POP):
    p[i].append(lp_opt(p[i][1]))

# Evolutionary loop
for _ in range(EVAL):
    for i in range(POP):
        p[i][1] = 0  # Reset radii
    # Selection
    best = np.argsort(r_sum)[-4:]
    parents = [p[i][1] for i in best]
    # Crossover and mutation
    for i in range(POP):
        if i in best:
            p[i][1] = parents[i]
        else:
            A = parents[np.random.randint(4)]
            B = parents[np.random.randint(4)]
            C = (A + B) / 2 + np.random.uniform(-5e-3, 5e-3, size=(26,))
            C = np.clip(C, 0, 1)
            if np.random.rand() < 0.3:
                m = np.random.randint(1, 6)
                C[np.random.choice(26, m, replace=False)] += np.random.normal(0, 0.02, size=m)
                C = np.clip(C, 0, 1)
            # Local relaxation
            res = minimize(lambda x: -np.sum(lp_opt([x])), p[i][0].flatten(), method='SLSQP', bounds=[(0, 1)] * 52, constraints=('eq', A_ub.toarray()), options={'maxiter': 25})
            p[i][0] = res.x.reshape(26, 2)
            p[i][1] = lp_opt(p[i][0])
    # Update global best
    r_sum = [np.sum(p[i][1]) for i in range(POP)]
    best_index = np.argmax(r_sum)
    if r_sum[best_index] > np.max(r_sum):
        p[best_index][1] = lp_opt(p[best_index][0])

# Final polishing
best_p = p[np.argmax(r_sum)][0]
best_r_sum = np.sum(p[np.argmax(r_sum)][1])
res = minimize(lambda x: -np.sum(lp_opt([x])), best_p.flatten(), method='SLSQP', bounds=[(0, 1)] * 52, constraints=('eq', A_ub.toarray()), options={'maxiter': 4000})
best_p = res.x.reshape(26, 2)
best_r_sum = -res.fun
best_r = lp_opt(best_p)

# Dealing with tiny negative radii
best_r[best_r < 1e-6] = 0

# Return the result
centers = best_p
radii = best_r
sum_radii = best_r_sum

# Validation output
assert np.allclose(sum_radii, 2.636, atol=1e-4), f"Sum of radii is {sum_radii}, expected 2.636"
return centers, radii, sum_radii

# Validation check function
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
