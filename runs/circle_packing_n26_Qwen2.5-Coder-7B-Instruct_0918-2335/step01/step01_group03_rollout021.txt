import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def inflate_discs(centers, speeds):
    n = centers.shape[0]
    active = set(range(n))
    t_stop = np.full(n, None)

    while active:
        coord_dists = distance_matrix(centers[active], centers[active])
        pair_t = coord_dists / (speeds[:, None] + speeds[None, :])
        edge_t_min = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]) / speeds[active]
        min_t = np.min(np.hstack([pair_t[np.triu_indices_from(pair_t, k=1)], edge_t_min]))
        stop_indices = np.where(np.isclose(pair_t, min_t, atol=1e-12))[1]
        stop_indices = set(stop_indices.tolist() + list(active.difference(stop_indices)))
        for i in stop_indices:
            t_stop[i] = min_t
            active.remove(i)
    
    radii = speeds * t_stop
    return radii, t_stop

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)  # 2 * x, y coordinates + r for each circle
    x0 = np.random.rand(3 * n)  # Initial guess for centers and radii

    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)  # (x,y) pairs
        r = v[52:]  # radii

        # Boundary terms - 4 * n
        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        # Pairwise terms - n choose 2
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")

# Generating mocked centers for speed testing
def generate_centres():
    np.random.seed(42)
    centres = np.random.rand(26, 2)
    return centres

def generate_speeds(centres):
    d1 = np.minimum(centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1])
    return 1 + 2 / (d1 + 1e-6) * np.random.lognormal(0, 0.4, size=26)

# Running optimized simulation
def main():
    centre_family = generate_centres()
    speed_family = generate_speeds(centre_family)
    radii_family = inflate_discs(centre_family, speed_family)
    sum_radii_family = np.sum(radii_family)
    print(f"Final sum of radii: {sum_radii_family}")
    return centre_family, radii_family, sum_radii_family
