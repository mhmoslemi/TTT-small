import numpy as np
from scipy.optimize import differential_evolution, minimize

def lp_rads(fixed_positions):
    positions = fixed_positions.reshape(26, 2)
    A_ub = np.empty((104, 26), float)
    A_ub[0:26, np.arange(26)] = 1.0
    A_ub[26:52, np.arange(26)] = -1.0
    A_ub[52:78, np.arange(26)] = 1.0
    A_ub[78:104, np.arange(26)] = -1.0
    b_ub = np.empty((104,), float)
    b_ub[0:26] = positions[:, 0]
    b_ub[26:52] = 1 - positions[:, 0]
    b_ub[52:78] = positions[:, 1]
    b_ub[78:104] = 1 - positions[:, 1]
    diffs = positions[:, None, :] - positions[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - np.clip(dists[mask], 0.0, None)
    A_ub[104:, :] = pair_cons
    b_ub[104:] = 0.0
    res = minimize(fun=lambda r: -np.sum(r), x0=[0.5] * 26,
                   method='highs',
                   constraints={'type': 'ineq', 'fun': lambda r: np.dot(A_ub, r) - b_ub},
                   options={'maxiter': 1000, 'disp': False})
    if not res.success:
        return np.zeros(26)
    return res.x

def fitness(positions):
    r = lp_rads(positions)
    return -np.sum(r)

def run_packing():
    bounds = [(0, 1)] * 52
    result = differential_evolution(fitness, bounds,
                                     popsize=50,
                                     maxiter=200,
                                     seed=42,
                                     tol=1e-12)
    centers = result.x.reshape(26, 2)
    radii = lp_rads(result.x)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
