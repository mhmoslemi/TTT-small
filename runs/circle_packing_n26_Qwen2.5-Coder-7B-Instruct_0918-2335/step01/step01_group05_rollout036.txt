import numpy as np
from scipy.optimize import linprog
import random

def hex_grid_points(n):
    h =])*0.18
    v = np.sqrt(3)/2 * h
    x = np.zeros(n)
    y = np.zeros(n)
    j = 0
    for i in range(n):
        if i % 2 == 0:
            x[i] = (i // 2) * h
            y[i] = j * v
        else:
            x[i] = (i // 2) * h + h/2
            y[i] = j * v + v/2
        if x[i] + h/2 > 1 or y[i] + v/2 > 1:
            j += 1
            i -= 1
        else:
            j += 1
    return x[:n], y[:n]

def grid_points(n):
    step = 0.15
    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)
    points = []
    for i in range(len(x)):
        for j in range(len(y)):
            points.append((x[i], y[j]))
    if len(points) < n:
        while len(points) < n:
            x_new, y_new = random.uniform(0, 1), random.uniform(0, 1)
            if all(np.sqrt((x_new - x)**2 + (y_new - y)**2) >= step for x, y in points):
                points.append((x_new, y_new))
    else:
        points = points[:n]
    return np.array(points), np.array(points)

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_ub = np.zeros((4*n + len(pairs), n))
    b_ub = np.zeros(4*n + len(pairs))
    for i in range(n):
        A_ub[4*i, i] = 1
        b_ub[4*i] = centers[i, 0]
        A_ub[4*i+1, i] = 1
        b_ub[4*i+1] = 1 - centers[i, 0]
        A_ub[4*i+2, i] = 1
        b_ub[4*i+2] = centers[i, 1]
        A_ub[4*i+3, i] = 1
        b_ub[4*i+3] = 1 - centers[i, 1]
    for k, (i, j) in enumerate(pairs):
        index = 4*n + k
        A_ub[index, i] = 1
        A_ub[index, j] = 1
        b_ub[index] = np.linalg.norm(centers[i] - centers[j])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
    return result.x, -result.fun

def run_packing():
    n = 26
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    patterns = [
        hex_grid_points(n),
        grid_points(n)
    ]
    best_centers = None
    best_sum = -np.inf
    for centers, jitter in patterns:
        jitter = np.clip(centers + np.random.uniform(-0.02, 0.02, (n, 2)), 0, 1)
        sum_radii, radii = solve_lp(jitter)
        if sum_radii > best_sum:
            best_centers, best_sum, best_radii = jitter, sum_radii, radii
    
    sigma = 0.05
    T = 0.1
    for _ in range(5000):
        i = random.randint(0, n - 1)
        dx, dy = np.random.normal(0, sigma)
        centers[i, 0] += dx
        centers[i, 1] += dy
        centers[i, 0] = max(0, min(1, centers[i, 0]))
        centers[i, 1] = max(0, min(1, centers[i, 1]))
        sum_radii, radii = solve_lp(centers)
        if sum_radii > best_sum or random.random() < np.exp((sum_radii - best_sum) / T):
            best_centers, best_sum, best_radii = centers.copy(), sum_radii, radii
        sigma *= 0.9995
        T *= 0.9995
    
    sum_radii, radii = solve_lp(best_centers)
    return best_centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

# Run the packing function and validate the results
centers, radii, sum_radii = run_packing()
is_valid, message = validate_packing(centers, radii)
print(f"Is the packing valid? {is_valid}")
print(f"Sum of radii: {sum_radii}")
