import numpy as np
from scipy.optimize import linprog, minimize, SLSQP

def solve_radii(centers):
    n = len(centers) // 2
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    A_ub = np.zeros((4 * n + len(pairs), n))
    for i in range(n):
        A_ub[i, i] = 1
        A_ub[i + n, i] = -1
        A_ub[i + 2 * n, i] = 1
        A_ub[i + 3 * n, i] = -1
    for k, (i, j) in enumerate(pairs):
        x_i, y_i = centers[2 * i], centers[2 * i + 1]
        x_j, y_j = centers[2 * j], centers[2 * j + 1]
        A_ub[4 * n + k, i] = 1
        A_ub[4 * n + k, j] = 1
        A_ub[4 * n + k, n + i] = -x_i - x_j
        A_ub[4 * n + k, n + j] = x_i + x_j
        A_ub[4 * n + k, 2 * n + i] = -y_i - y_j
        A_ub[4 * n + k, 2 * n + j] = y_i + y_j
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, bounds=[(0, 0.5)] * n)
    if not res.success:
        return np.nan, np.zeros(n)
    radii = -res.x
    return np.sum(radii), radii

def evaluate_fitness(individual):
    centers = np.clip(individual[:52].reshape(26, 2), 0, 1)
    _, radii = solve_radii(individual[:52].reshape(26, 2))
    return np.sum(radii)

def crossover(parent1, parent2):
    i1, i2 = np.random.randint(1, 51, size=2)
    if i1 > i2:
        i1, i2 = i2, i1
    kid1 = np.concatenate((parent1[:i1], parent2[i1:i2], parent1[i2:]))
    kid2 = np.concatenate((parent2[:i1], parent1[i1:i2], parent2[i2:]))
    return kid1, kid2

def mutate(individual, sigma):
    mutated = individual + np.random.normal(0, sigma, size=individual.shape)
    mutated = np.clip(mutated, 0, 1)
    return mutated

def clamp_to_square(individual):
    individual[:52] = np.clip(individual[:52], 0, 1)
    return individual

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    best_sol = None
    best_sum = -np.inf

    for _ in range(10):  # Multiple restarts for robustness
        population = np.random.rand(population_size, 52)
        sigma = initial_sigma

        for gen in range(generations):
            fitness = np.array([evaluate_fitness(individual) for individual in population])
            if fitness.max() > best_sum:
                best_sum = fitness.max()
                best_sol = population[np.argmax(fitness)]

            new_population = np.empty((population_size, 52))
            for _ in range(population_size // 2):
                parent1, parent2 = np.random.choice(population, size=2, replace=False)
                kid1, kid2 = crossover(parent1, parent2)
                kid1 = mutate(kid1, sigma)
                kid2 = mutate(kid2, sigma)
                kid1 = clamp_to_square(kid1)
                kid2 = clamp_to_square(kid2)
                kid1 = evaluate_fitness(kid1)
                kid2 = evaluate_fitness(kid2)
                new_population[_*2] = kid1
                new_population[_*2+1] = kid2

            population = new_population.copy()

            # Elitism
            top_individuals = population[np.argsort(fitness)[-elitism:]]
            population[:elitism] = top_individuals
            sigma *= sigma_decay

        # Final SLSQP refinement
        if best_sol is not None:
            centers = best_sol[:52].reshape(26, 2)
            _, initial_radii = solve_radii(centers)
            res = minimize(fun=evaluate_fitness, x0=np.concatenate((centers.flatten(), initial_radii)), method='SLSQP',
                           bounds=[(0, 1)] * 2 * n + [(0, 0.5)] * n, options={'maxiter': 2000})
            if res.success:
                best_sol = np.concatenate((res.x[:52], res.x[52:]))

    centers = best_sol[:52].reshape(26, 2)
    _, radii = solve_radii(centers)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
