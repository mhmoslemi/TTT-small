import numpy as np
from scipy.spatial import distance
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    sizes = ['grid', 'hexagonal', 'hybrid', 'random']
    r_max = 0.5
    best_centers = None
    best_radii = None
    best_sum_radii = 0
    np.random.seed(0)  # For reproducibility

    for size in sizes:
        if size == 'grid':
            centers = np.array(np.meshgrid(np.linspace(0.05, 0.95, 5), np.linspace(0.05, 0.95, 5))).reshape(2, -1).T
        elif size == 'hexagonal':
            x = np.arange(0, 1.01, 0.38)
            y = np.arange(0, 1.01, 0.33)
            centers = np.array([[a + b % 0.76, c] for a in x for b in [-0.33, 0.33] for c in y if 0 <= c < 1 and 0 <= c + a % 0.76 < 1])
        elif size == 'hybrid':
            centers = np.array([
                [0.1, 0.1], [0.3, 0.1], [0.5, 0.1],
                [0.7, 0.1], [0.9, 0.1],
                [0.9, 0.3],
                [0.9, 0.5],
                [0.7, 0.7],
                [0.5, 0.7],
                [0.3, 0.7],
                [0.1, 0.7],
                [0.1, 0.5],
                [0.1, 0.3],
                [0.1, 0.1],
                [0.3, 0.9],
                [0.5, 0.9],
                [0.7, 0.9],
                [0.7, 0.7],
                [0.5, 0.5],
                [0.5, 0.3],
                [0.3, 0.5],
                [0.1, 0.5],
                [0.1, 0.3],
                [0.1, 0.1],
                [0.3, 0.1]
            ])
        elif size == 'random':
            centers = np.random.rand(n, 2)

        for _ in range(5):  # Sampling multiple trials per size
            speeds = np.random.lognormal(mean=0., sigma=0.4, size=n) * (1 + 2 / (distance.cdist(centers, centers) + 10**-6))
            active = np.arange(n)
            r = np.zeros(n)
            t_stop = [None] * n
            while len(active):
                t_ij = np.min(np.maximum(distance.cdist(centers[active][:, np.newaxis], centers[np.newaxis, :], 'euclidean') / (speeds[active][:, np.newaxis] + speeds[np.newaxis, :]), 0), axis=1)
                t_edge = np.minimum(centers[active][:, np.newaxis, 0] - r[active][:, np.newaxis], r[active][:, np.newaxis] + 1 - centers[active][:, np.newaxis, 0]
                                    , centers[active][:, np.newaxis, 1] - r[active][:, np.newaxis], r[active][:, np.newaxis] + 1 - centers[active][:, np.newaxis, 1])
                t_min = min(np.min(t_edge), np.min(t_ij))
                stop_set = np.where((t_edge == t_min) | ((t_ij == t_min) & (np.diff(t_ij[t_ij == t_min], prepend=np.inf) != 0)).any(axis=1))[0]
                for i in stop_set:
                    active = active[active != i]
                    t_stop[i] = t_min
                    r[i] = speeds[i] * t_min

            for i in range(n):
                radii = np.maximum(r min(r_max - np.minimum(centers[:, 1], 1 - centers[:, 1]), r_max - np.minimum(centers[:, 0], 1 - centers[:, 0]))
                sum_radii = np.sum(radii)
                if validate_packing(centers, radii)[0]:
                    if sum_radii > best_sum_radii:
                        best_centers = centers
                        best_radii = radii
                        best_sum_radii = sum_radii

    if best_sum_radii == 0:
        return np.zeros((26, 2)), np.zeros(26), 0

    def obj(x):
        return -np.sum(x[52:])

    def constraints(x):
        pos = x[:52].reshape(n, 2)
        r = x[52:]
        cons = np.empty(4 * n)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=np.concatenate([best_centers.ravel(), best_radii]), method='SLSQP', bounds=[(0, 1)] * (3 * n),
                   constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9})

    if res.success:
        return best_centers, best_radii, best_sum_radii
    else:
        return np.zeros((26, 2)), np.zeros(26), 0
