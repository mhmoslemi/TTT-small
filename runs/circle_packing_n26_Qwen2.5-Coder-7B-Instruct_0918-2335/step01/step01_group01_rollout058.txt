import numpy as np
from scipy.optimize import minimize, linprog

def obj(x):
    return -np.sum(x[52:])

def constraints(x):
    pos = x[:52].reshape(26, 2)
    r = x[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask] - 1e-12
    return np.concatenate([cons, pair_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NP = 40
    w = 0.6
    c1 = 1.5
    c2 = 1.5
    T0 = 0.04
    cooling_factor = 0.995
    velocity_clamp = 0.03
    mutation_rate = 0.1
    mutation_std = 0.02
    max_generations = 1000
    inner_lp_tolerance = 1e-9
    sigma = 0.01

    def lp_oracle(pos):
        n = pos.shape[0] // 2
        A = np.empty((n * 4 + n * (n - 1) // 2, 2 * n))
        b = np.empty(n * 4 + n * (n - 1) // 2)
        A[:n * 4, :n] = np.eye(n, 2 * n)
        A[:n * 4, n:] = -np.eye(n, 2 * n)
        for i in range(n):
            b[i] = pos[i]
            b[i + n] = 1 - pos[i]
        A[n * 4:, :n] = np.eye(n, 2 * n)
        A[n * 4:, n:] = -np.eye(n, 2 * n)
        for i in range(n):
            b[i + 2 * n] = pos[i + n]
            b[i + 2 * n + 1] = 1 - pos[i + n]
        D = np.sqrt(np.sum((pos[:n * 2].reshape(n, 2)[:, None, :] - pos[:n * 2].reshape(n, 2)[None, :, :]) ** 2, axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_A = D[mask] - (pos[2 * n:] + pos[2 * n:])[mask]
        A = np.concatenate([A, np.zeros((len(pair_A), 2 * n))], axis=1)
        b = np.concatenate([b, -pair_A])
        result = linprog(pos[2 * n:], A_ub=A, b_ub=b, method='highs', bounds=np.hstack((0.01 * np.ones((n, 2)), 0.5 * np.ones(n))), options={'tol': inner_lp_tolerance})
        if result.status != 0:
            return np.full(n, 1e-6)
        return np.clip(result.x, 1e-6, None)

   (seeds, values) = zip(*sorted(zip([lp_oracle(np.random.rand(78)) for _ in range(4)], [np.random.rand(78) for _ in range(4)]))
    seeds = [s + sigma * np.random.randn(78) for s in seeds]
    seeds = np.concatenate(seeds, axis=0)
    seeds = np.clip(seeds, 0, 1)

    pbest_pos = np.empty((NP, 78))
    pbest_val = np.full(NP, -np.inf)
    gbest_pos = np.copy(seeds[np.argmax(values)[:NP]])
    gbest_val = np.copy(values[np.argmax(values)[:NP]])

    for t in range(max_generations):
        T = T0 * cooling_factor ** t

        for i in range(NP):
            v = w * pbest_pos[i] + c1 * np.random.rand(78) * (pbest_pos[i] - pbest_pos[i]) + c2 * np.random.rand(78) * (gbest_pos - pbest_pos[i])
            if np.random.rand() < T / T0:
                v = sigma * np.random.randn(78)
            if np.random.rand() < mutation_rate:
                k = np.random.randint(1, 3)
                pbest_pos[i][:k * 2] += mutation_std * np.random.randn(k * 2)
            pbest_pos[i] += v
            pbest_pos[i] = np.clip(pbest_pos[i], 0, 1)

            pbest_val[i] = lp_oracle(pbest_pos[i])
            if pbest_val[i] > gbest_val[i]:
                gbest_val[i] = pbest_val[i]
                gbest_pos[i] = np.copy(pbest_pos[i])

        for i in range(NP):
            if pbest_val[i] > gbest_val[i]:
                gbest_val[i] = pbest_val[i]
                gbest_pos[i] = np.copy(pbest_pos[i])

    pos_best = gbest_pos
    r_best = lp_oracle(pos_best)
    r_best = np.clip(r_best, 0, 0.5)

    result = minimize(obj, np.concatenate((pos_best, r_best)), method='SLSQP', bounds=np.hstack(([(-1e-6, 1 + 1e-6)] * 52, [(0, 0.5)] * 26)), constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 4000, 'ftol': 1e-9})
    if not result.success:
        return np.empty((26, 2)), np.empty(26), 0
    radii = np.clip(result.x[52:], 0, 0.5)
    return pos_best[:52].reshape(26, 2), radii, np.sum(radii)
