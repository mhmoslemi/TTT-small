import numpy as np
from scipy.optimize import minimize

def obj(v):
    r = v[52:]                      # last 26 entries
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)      # (x,y) pairs
    r   = v[52:]                   # radii

    # boundary terms – 4*26
    cons = np.empty(104, float)
    cons[ 0:26] = pos[:,0] - r     # left/right
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r

    # pairwise terms – 26 choose 2
    diffs = pos[:,None,:] - pos[None,:,:]   # (26,26,2)
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask  = np.triu(np.ones((26,26), bool), k=1)
    pair_cons = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.concatenate([cons, pair_cons])

def heuristic_placement():
    # Create a heuristic initial placement for circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Place circles in a dense hexagonal pattern within the unit square
    radius = 0.04
    gaps = 0.4
    circle_index = 0
    
    row = 0
    col = 0
    while row < 1 and col < 1:
        x = col * (np.sqrt(3) * radius) + (radius if row % 2 == 1 else 0)
        y = row * (2 * radius) + radius
        if x < 1 and y < 1:
            centers[circle_index, :] = [x, y]
            radii[circle_index] = radius
            circle_index += 1
        col += 1
        if col >= 2 * (circle_index + 1) / (np.sqrt(3) * radius):
            row += 1
            col = 0
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    init_centers, init_radii = heuristic_placement()
    init = np.empty(78, float)
    init[:52] = init_centers.flatten()
    init[52:] = init_radii
    
    bounds = [(0,1)]*26          # x
    bounds += [(0,1)]*26          # y
    bounds += [(0,0.5)]*26        # radii

    nrestarts = 20
    best_r = None
    best_res = None

    for _ in range(nrestarts):
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type':'ineq',
                                                   'fun': constraints},
                       options={'maxiter':4000, 'ftol':1e-9, 'disp':False})
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                best_r = res.x[52:]

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers = best_res.x[:52].reshape(26, 2)
    sum_radii = -best_res.fun
    return centers, best_r, sum_radii
