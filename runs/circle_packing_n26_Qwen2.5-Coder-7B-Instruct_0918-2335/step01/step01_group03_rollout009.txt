```python
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from time import time

# Define constants
N = 26
side = 1.0

# Generate center families
def generate_centers(kind):
    if kind == 'grid':
        return np.array([(i/4, j/4) for i in range(5) for j in range(5)] + [(2.5, 2.5)])
    elif kind == 'hexagonal':
        centers = np.array([(i*(side/4), j*(side/4)) for i in range(6) for j in range(5)]).clip(0, side)
        centers[5,1] -= 0.125
        centers[5,2] -= 0.125
        centers[5,4] += 0.125
        centers[5,5] += 0.125
        return centers
    elif kind == 'hybrid':
        centers = np.array([(i*(side/4), j*(side/4)) for i in range(4) for j in range(5)]).clip(0, side)
        center_x, center_y = side/2, side/2
        small_radius = min(center_x, center_y, side - center_x, side - center_y) / 2.5
        centers = np.vstack((centers, np.array([center_x, center_y]) + [0, small_radius]))
        return centers
    elif kind == 'random':
        return np.random.rand(N, 2)
    else:
        raise ValueError("Invalid center family")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(speeds):
    centers = generate_centers('hybrid')
    radii = (1 + 2 / (cdist(centers, centers) + 1e-6) * np.random.lognormal(mean=0., sigma=0.4)) * speeds
    valid, msg = validate_packing(centers, radii)
    if not valid:
        return -1e-6
    return -np.sum(radii)

def run_packing(max_time=10):
    start_time = time()
    best_centers = None
    best_radii = None
    best_radius_sum = -1e-6
    family_counter = 0

    while True:
        family_type = ['grid', 'hexagonal', 'hybrid', 'random'][family_counter % 4]
        centers = generate_centers(family_type)
        speeds = 1 + 2 / (cdist(centers, centers) + 1e-6) * np.random.lognormal(mean=0., sigma=0.4)

        # Event simulation
        active = np.arange(N)
        t_stop = [None] * N
        distances = cdist(centers, centers)

        while len(active) > 0:
            t_ij = distances[active][:, :, None] + distances[None, :, active] - 2 * centers[active][:, :, None, :]
            t_ij = np.clip(t_ij, 0, None) / (speeds[active] + speeds[:, None, active])
            t_edge = np.minimum.accumulate(np.minimum(centers[:, 0], centers[:, 1],
                                                   side - centers[:, 0], side - centers[:, 1])[:, None])[:, 0,
                                                   np.array(active)]
            t_stop_active = np.min(np.concatenate([t_ij, t_edge[:, None]], axis=1), axis=1)
            stop_set = np.where((t_edge == t_stop_active) | (t_ij == t_stop_active[:, None]), np.arange(N), -1)
            stop_set = np.unique(stop_set[stop_set >= 0])
            t_stop[stop_set] = t_stop_active[stop_set]
            active = np.delete(active, stop_set)

        radii = speeds * t_stop
        valid, _ = validate_packing(centers, radii)
        if valid and np.sum(radii) > best_radius_sum:
            best_centers = centers
            best_radii = radii
            best_radius_sum = np.sum(radii)

        if time() - start_time > max_time or family_counter >= 4000:
            break

        family_counter += 1

    if best_radius_sum <= -1e-6:
        raise RuntimeError("No valid packing found")

    # Local refinement with SLSQP
    x0 = np.concatenate((best_centers.flatten(), best_radii))
    bounds = [(0, side)] * 2 + [(0, 0.5)] * N

    def refine_objective(x):
        centers = x[0::2].reshape(N, 2)
        radii = x[1::2]
        return -np.sum(radii)

    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: side - x[2*i] - x[2*N+i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*N+i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: side - x[2*i+1] - x[2*N+i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - x[2*N+i]})
        for j in range(N):
            if j != i:
                d = np.sum((centers[i] - centers[j]) ** 2) - (x[2*N+i] + x[2*N+j])
                constraints.append({'type': 'ineq', 'fun': lambda x, d=d, i=i, j=j: d - (x[2*N+i] + x[2*N+j])})

    res = minimize(refine_objective, x0, bounds=bounds, constraints=constraints, options={'maxiter': 2000})

    if res.success:
        final_centers = res.x[0::2].reshape(N, 2)
        final_radii = res.x[1::2]
        valid, msg = validate_packing(final_centers, final_radii)
        if valid:
            return final_centers, final_radii, np.sum(final_radii)

    return best_centers, best_radii, best_radius_sum
```
