import numpy as np
from scipy.optimize import linprog, minimize

def lp_oracle(c):
    r = c[52:]
    A = np.zeros((104, 52))
    b = np.zeros(104)
    for i in range(26):
        A[i, 2*i] = 1
        A[i, 2*i+1] = -1
        b[i] = 1
        A[i+26, 2*i] = 1
        A[i+26, 2*i+1] = 1
        b[i+26] = 0
        A[i+52, 2*i] = -1
        A[i+52, 2*i+1] = -1
        b[i+52] = -1
        A[i+78, 2*i] = -1
        A[i+78, 2*i+1] = 1
        b[i+78] = -1
        A[i+102, 2*i] = 1
        A[i+102, 2*i+1] = 1
        b[i+102] = 1
        A[103 + i*(26-i), 2*i] = 1
        A[103 + i*(26-i), 2*i+1] = 1
        A[104 + i*(26-i), 2*i] = 1
        A[104 + i*(26-i), 2*i+1] = -1
        for j in range(i+1, 26):
            A[105 + (i*(26-i)//2) + (j-i-1), 2*i] = 1
            A[105 + (i*(26-i)//2) + (j-i-1), 2*i+1] = 1
            A[105 + (i*(26-i)//2) + (j-i-1), 2*j] = 1
            A[105 + (i*(26-i)//2) + (j-i-1), 2*j+1] = 1
            b[105 + (i*(26-i)//2) + (j-i-1)] = np.sqrt(np.sum((c[2*i:2*i+2] - c[2*j:2*j+2])**2))
    res = linprog([0] + [-1]*26, A_ub=A, b_ub=b, bounds=[(0, 1)]*26 + [(0, 0.5)]*26, method='highs', options={'tol': 1e-9})
    if not res.success:
        return 1e6
    r_opt = res.x[1:]
    mask = r_opt < 1e-12
    r_opt[mask] = 0
    return -np.sum(r_opt)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    positions = np.array([
        [0.155, 0.155], [0.155, 0.845], [0.845, 0.155], [0.845, 0.845],
        [0.5, 0.292], [0.5, 0.708], [0.292, 0.5], [0.708, 0.5],
        [0.314, 0.486], [0.486, 0.314], [0.586, 0.486], [0.486, 0.586],
        [0.424, 0.424], [0.576, 0.576], [0.424, 0.576], [0.576, 0.424],
        [0.5, 0.301], [0.5, 0.699], [0.301, 0.5], [0.699, 0.5],
        [0.438, 0.462], [0.462, 0.438], [0.538, 0.462], [0.462, 0.538]
    ])
    radii = np.array([
        0.146, 0.146, 0.146, 0.146,
        0.179, 0.179, 0.179, 0.179,
        0.229, 0.229, 0.229, 0.229,
        0.285, 0.23, 0.24, 0.24,
        0.23, 0.285, 0.24, 0.24,
        0.12, 0.12, 0.12, 0.12,
        0.105, 0.105, 0.105, 0.105,
        0.115, 0.115, 0.115, 0.115,
        0.135, 0.135, 0.135, 0.135
    ])
    N = 40
    NP = 40
    inertia = 0.6
    cognitive = 1.5
    social = 1.5
    T0 = 0.04
    cooling_factor = 0.995
    velocity_clamp = 0.03
    mutation_rate = 0.1
    mutation_std = 0.02
    max_generations = 1000
    inner_lp_tolerance = 1e-9

    def update_velocity(v, pbest, gbest, pos, best_val, val):
        v_new = inertia * v + cognitive * np.random.rand() * (pbest - pos) + social * np.random.rand() * (gbest - pos)
        return np.clip(v_new, -velocity_clamp, velocity_clamp)

    def annealing(v, T):
        if np.random.rand() < min(1, T/T0):
            v_new = np.random.randn(*v.shape) * mutation_std
            return np.clip(v_new, -velocity_clamp, velocity_clamp)
        else:
            return v

    def mutate(pos):
        for _ in range(int(mutation_rate * 26)):
            i, j = np.random.randint(0, 26, 2)
            pos[2*i:2*i+2] += np.random.randn(2) * mutation_std
            pos[2*j:2*j+2] += np.random.randn(2) * mutation_std
            pos[2:i*2] = np.clip(pos[2:i*2], 1e-6, 1-1e-6)
            pos[2*j*2:] = np.clip(pos[2*j*2:], 1e-6, 1-1e-6)
        return pos

    best_val = -np.inf
    while max_generations > 0:
        swarm = np.concatenate([np.meshgrid(np.linspace(0.025, 0.975, NP), np.linspace(0.025, 0.975, NP)).reshape(2, -1, order='F').transpose(), positions[:, np.newaxis].repeat(NP, axis=1)], axis=1)
        swarm = np.transpose(swarm.reshape(52, NP), (1, 0))
        velocities = np.random.randn(NP, 52)
        personal_best = [None] * NP
        personal_val = [-np.inf] * NP
        global_best_pos = swarm[:, 0]
        global_best_val = personal_val[0]

        for t in range(max_generations):
            for i in range(NP):
                v = update_velocity(velocities[i], personal_best[i], global_best_pos, swarm[i], personal_val[i], lp_oracle(swarm[i]))
                v = annealing(v, T0)
                swarm[i] += v
                swarm[i] = np.clip(swarm[i], 1e-6, 1-1e-6)
                current_val = lp_oracle(swarm[i])
                if current_val > personal_val[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_val[i] = current_val
                    if current_val > global_best_val:
                        global_best_pos = swarm[i].copy()
                        global_best_val = current_val

            T0 *= cooling_factor

        if personal_val[np.argmax(personal_val)] > best_val:
            best_val = personal_val[np.argmax(personal_val)]
            positions = personal_best[np.argmax(personal_val)][:52].reshape(26, 2)
            radii = personal_best[np.argmax(personal_val)][52:]
        else:
            for i in range(int(0.1 * NP)):
                j = np.random.randint(0, NP)
                swarm[j] = positions[np.random.randint(0, 26)].copy() + np.random.randn(52) * mutation_std
                swarm[j] = np.clip(swarm[j], 1e-6, 1-1e-6)
                velocities[j] = np.random.randn(52)
                personal_best[j] = swarm[j].copy()
                personal_val[j] = lp_oracle(swarm[j])
                if personal_val[j] > global_best_val:
                    global_best_pos = swarm[j].copy()
                    global_best_val = personal_val[j]

    res = minimize(lp_oracle, np.concatenate([positions.flatten(), radii]), method='SLSQP', bounds=np.concatenate([[(0, 1)]*52, [(0, 0.5)]*26]), options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    if res.success:
        best_sum = -res.fun
        best_radii = res.x[52:]
    else:
        best_sum = best_val
        best_radii = radii

    return positions, best_radii, best_sum
