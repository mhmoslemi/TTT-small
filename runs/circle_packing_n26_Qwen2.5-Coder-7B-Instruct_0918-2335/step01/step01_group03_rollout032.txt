import numpy as np
from scipy import optimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)  # (x,y) coordinates + r for each circle

    def generate_positions(pattern, centers0):
        centers = np.tile(centers0, (4, 1))
        if pattern == 'hexagon':
            d = np.sqrt(3) / 2 * 0.4
            centers[:, 0] += np.array([0, d, d, d, d, -d, -d, -d, -d])
            centers[:, 1] += np.array([0, -d, 0, d, 2*d, -d, 0, d, 2*d])
            centers = centers[:n]
        elif pattern == '5x5-grid':
            centers = np.array([(i / 4, j / 4) for i in range(4) for j in range(4) if i + j < 4])
            centers = np.repeat(centers, 6, axis=0)[:n]
        elif pattern == 'hybrid':
            corners = np.array([(0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9)])
            edge_centers = np.array([(0.5, 0.1), (0.1, 0.5), (0.5, 0.9), (0.9, 0.5)])
            inner_grid = np.array([(0.15, 0.15), (0.35, 0.15), (0.55, 0.15), 
                                    (0.75, 0.15), (0.15, 0.35), (0.35, 0.35), 
                                    (0.55, 0.35), (0.75, 0.35), (0.15, 0.55), 
                                    (0.35, 0.55), (0.55, 0.55), (0.75, 0.55), 
                                    (0.15, 0.75), (0.35, 0.75), (0.55, 0.75), 
                                    (0.75, 0.75)])
            centers = np.concatenate([corners, edge_centers, inner_grid])
            centers = centers[:n]
        elif pattern == 'random':
            centers = np.empty((n, 2))
            for i in range(n):
                while True:
                    x, y = np.random.rand(2)
                    if validate_packing(np.array([[x, y]]), np.array([0.1]) * (i < 5))[0]: break
                centers[i] = [x, y]
        return centers

    def lp_optimize_radii(centers):
        n = centers.shape[0]
        dist_matrix = scipy.spatial.distance.cdist(centers, centers)
        A = np.zeros((n * (n - 1), n))
        for k in range(n - 1):
            A[k * (n - 1) + np.arange(n - 1 - k), np.arange(1, n - k)] = 1
            A[k * (n - 1) + np.arange(n - 1 - k), np.arange(n - k, 0, -1)] = -1
            A[k * (n - 1) + np.arange(n - k), n - 1] = -dist_matrix[k + 1, 1:]
        b = -dist_matrix.flatten()[n:n * (n - 1)]

        c = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            c[i] = 1 - min(x, 1 - x, y, 1 - y)

        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)])
        if res.success:
            return res.x
        else:
            return None

    def evaluate(solution):
        centers, radii = solution[:52].reshape(n, 2), solution[52:]
        if validate_packing(centers, radii)[0]:
            return radii.sum()
        else:
            return -np.inf

    def refine_with_sa(centers, radii):
        n = centers.shape[0]
        x0 = np.concatenate((centers.flatten(), radii))
        def penalty(centers, radii):
            cost = 0
            for i in range(n):
                cost -= radii[i]
                for j in range(i + 1, n):
                    cost += max(0, radii[i] + radii[j] - distance(centers[i], centers[j]))
            return cost

        def distance(p1, p2):
            return np.sqrt(np.sum((p1 - p2) ** 2))

        def neighbor(x, y, sigma=0.02):
            x_new = x + sigma * (np.random.rand() - 0.5)
            y_new = y + sigma * (np.random.rand() - 0.5)
            return np.clip([x_new, y_new], 0, 1)

        def radius_new(r):
            return np.clip(r + 0.01 * (np.random.rand() - 0.5), 0, 1)

        schedule = lambda t: 10000 / (1.0 + t)
        for T in range(100000):
            i, j = np.random.randint(n, size=2)
            cost_i, cost_j = penalty(centers[i], radii[i]), penalty(centers[j], radii[j])
            delta_cost = evaluate(np.concatenate((np.concatenate((centers, radii)), x0[:52].reshape(n, 2)
                .at[i, :].set(neighbor(centers[i, 0], centers[i, 1])))).flatten()) + \
                evaluate(np.concatenate((np.concatenate((centers, radii)), x0[:52].reshape(n, 2)
                .at[j, :].set(neighbor(centers[j, 0], centers[j, 1])))).flatten()) - \
                evaluate(x0) - cost_i - cost_j

            if delta_cost > 0 or np.random.rand() < np.exp(delta_cost / schedule(T)):
                if i < n: centers[i] = neighbor(centers[i, 0], centers[i, 1])
                if j < n: centers[j] = neighbor(centers[j, 0], centers[j, 1])
                radii[i], radii[j] = radius_new(radii[i]), radius_new(radii[j])

        return centers, radii

    def optimize_sol(centers, radii):
        x0 = np.concatenate((centers.flatten(), radii))
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: x[2 * n + i] + x[2 * n + j] - distance(x[:2 * n].reshape(n, 2)[i], x[:2 * n].reshape(n, 2)[j])})
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        res = optimize.minimize(evaluate, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-8, 'disp': False})
        
        if res.success:
            centers, radii = res.x[:2 * n].reshape(n, 2), res.x[2 * n:]
            if validate_packing(centers, radii)[0]:
                return centers, radii
        return None, None

    min_radii = 0.05
    max_radii = 0.15
    candidates = []
    
    patterns = ['hexagon', '5x5-grid', 'hybrid', 'random']
    for pattern in patterns:
        for _ in range(5):
            centers = generate_positions(pattern, np.array([0.5, 0.5]))
            radii = min_radii + (max_radii - min_radii) * np.random.rand(n)
            centers, radii = optimize_sol(centers, radii)
            if centers is not None:
                candidates.append((centers, radii))

    best = max(candidates, key=lambda c: evaluate(c))
    return best[0], best[1], best[0].shape[0]
