import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    n = centers.shape[0]
    c = np.ones(n)  # Objective: maximize sum of radii
    A = np.vstack([-np.eye(n), np.eye(n)])  # Constraints: r >= 0
    b = np.zeros(2 * n)
    for i in range(n):
        A2 = np.zeros((n, n))
        A2[i, i] = 1
        A = np.vstack([A, A2, -A2])
        b = np.append(b, [1, centers[i, 0], centers[i, 1]])
        for j in range(i + 1, n):
            A3 = np.zeros((n, n))
            A3[j, j] = 1
            A = np.vstack([A, A3 + A2, -A3 - A2])
            b = np.append(b, [1, np.linalg.norm(centers[i, :] - centers[j, :]) - 1e-7])
    return linprog(c, A_eq=None, A_ub=A, b_ub=b, bounds=(0, None), method='highs').x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    initial_centers = None
    
    # Create a lattice-based seed
    x = np.linspace(0, box_size, int(np.sqrt(n)) + 1)
    y = np.linspace(0, box_size, int(np.sqrt(n)) + 1)
    x = np.tile(x, len(y))
    y = np.repeat(y, len(x))
    initial_centers = np.column_stack((x, y))
    initial_centers = initial_centers[:n, :]
    
    # Initial guess for radii
    initial_radii = np.ones(n) * 0.05
    
    # Outer loop for multiple restarts
    best_sum_radii = -np.inf
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for seed in range(8):
        centers = initial_centers.copy()
        random_displacement = np.random.rand(n, 2) * 0.1
        centers += random_displacement
        iter_count = 0
        alpha = 0.01
        cooling_interval = 20
        accept_rate = 0.5
        epsilon = 1e-7
        patience = 20
        
        while iter_count < 200:
            old_sum_radii = np.sum(initial_radii)
            radii = solve_radii_lp(centers)
            constraints = np.vstack((-np.eye(n), np.eye(n)))  # r >= 0
            constraints = np.vstack([constraints, radii[:, np.newaxis] - radii])
            constraints = np.vstack([constraints, centers - box_size / 2])
            constraints = np.vstack([constraints, box_size / 2 - centers])
            constraints = np.hstack([constraints, np.zeros((6 * n, 1))])
            constraints = np.hstack([constraints, -1e-10 * np.ones((6 * n, 1))])
            bounds = [(-box_size / 2, box_size / 2)] * (2 * n)
            bounds += [(0, None)] * n
            res = linprog(np.ones(n), A_ub=constraints, b_ub=np.zeros(6 * n), bounds=bounds, method='highs')
            if res.success:
                initial_radii = res.x
                active_constraints = np.abs(constraints[:, :n] - initial_radii[:, np.newaxis]) < 1e-7
                for i in range(n):
                    if np.sum(active_constraints[i, :]) == 1:
                        if active_constraints[i, :].argmax() < n:  # Wall constraint
                            if active_constraints[i, :active_constraints[i, :].argmax()].any():
                                centers[i, 0] += alpha * ( centers[i, 0] < box_size / 2 and 1 or -1 )
                            else:
                                centers[i, 1] += alpha * ( centers[i, 1] < box_size / 2 and 1 or -1 )
                        else:  # Neighbor constraint
                            j = active_constraints[i, active_constraints[i, :].argmax()].argmax()
                            dist_ij = np.linalg.norm(centers[i, :] - centers[j, :])
                            unit_vector = (centers[i, :] - centers[j, :]) / dist_ij
                            centers[i, :] += alpha * unit_vector
        
            if np.sum(initial_radii) > old_sum_radii:
                iter_count += 1
                alpha *= 0.5
                if iter_count % cooling_interval == 0:
                    alpha = 0.01
                if iter_count % 2 == 0:
                    centers += np.random.randn(n, 2) * alpha * centers / np.mean(np.linalg.norm(centers, axis=1))
                if iter_count % 10 == 0:
                    alpha /= 2
            else:
                break
        
        if np.sum(initial_radii) > best_sum_radii:
            best_sum_radii = np.sum(initial_radii)
            best_centers = centers.copy()
            best_radii = initial_radii.copy()
    
    sum_radii = best_sum_radii
    centers = best_centers
    radii = best_radii
    
    # Fine-tuning with SLSQP
    def objective(x):
        n = x.shape[0] // 2
        return -np.sum(x[n:])
    
    def constraints(x):
        n = x.shape[0] // 2
        c = x[:n]
        r = x[n:]
        A = np.vstack([-np.eye(n), np.eye(n)])  # Constraints: r >= 0
        b = np.zeros(2 * n)
        for i in range(n):
            A2 = np.zeros((n, n))
            A2[i, i] = 1
            A = np.vstack([A, A2, -A2])
            b = np.append(b, [1, c[i, 0], c[i, 1]])
            for j in range(i + 1, n):
                A3 = np.zeros((n, n))
                A3[j, j] = 1
                A = np.vstack([A, A3 + A2, -A3 - A2])
                b = np.append(b, [1, np.linalg.norm(c[i, :] - c[j, :]) - 1e-7])
        return np.vstack([A, np.vstack((-np.eye(n), np.eye(n))) - np.eye(n*n, k=n), np.vstack((np.eye(n), -np.eye(n))) - np.eye(n*n, k=n)])
    
    def bounds(x):
        n = x.shape[0] // 2
        return np.hstack([-np.inf * np.ones(2 * n), np.zeros(n), np.inf * np.ones(n)])
    
    x0 = np.concatenate((centers.flatten(), radii))
    bounds = bounds(x0)
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    
    # Final check and return result
    if res.success and np.all(res.x[n:] >= 0) and np.all(res.x[:n, 0] >= 0) and np.all(res.x[:n, 1] >= 0) and np.all(res.x[:n, 0] <= 1) and np.all(res.x[:n, 1] <= 1):
        centers = res.x[:n].reshape((n, 2))
        radii = res.x[n:]
        sum_radii = np.sum(radii)
    else:
        raise ValueError("Validation failed: invalid solution found.")
    
    return centers, radii, sum_radii
