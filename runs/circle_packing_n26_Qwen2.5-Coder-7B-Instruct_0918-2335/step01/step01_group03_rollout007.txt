# Helper function to run an event simulation and return the sum of radii if feasible
def simulate_and_validate(centers, radii):
    n = centers.shape[0]
    t_stop = [None] * n
    t_ij = np.empty((n, n))

    for i in range(n):
        t_ij[:, i] = np.sqrt(np.sum((centers - centers[i]) ** 2, axis=1))
        t_edge = np.min(centers[i], 1 - centers[i])
        t_min = np.min(np.concatenate([t_ij[:, i], t_edge]))
        t_ij_min_indices = np.argwhere(t_ij[:, i] == t_min)
        t_edge_min_indices = np.argwhere(t_edge == t_min)

        if len(t_ij_min_indices) == 1 and len(t_edge_min_indices) == 1:
            active = np.arange(n)
            active = active[active != int(t_ij_min_indices[0])]
            t_stop[int(t_ij_min_indices[0])] = t_min
        elif len(t_ij_min_indices) == 1 and len(t_edge_min_indices) > 1:
            active = np.arange(n)
            active = active[active != int(t_ij_min_indices[0])]
            t_stop[int(t_ij_min_indices[0])] = t_min
        elif len(t_ij_min_indices) > 1 and len(t_edge_min_indices) == 1:
            active = np.arange(n)
            active = active[active != int(t_edge_min_indices[0])]
            t_stop[int(t_edge_min_indices[0])] = t_min
        else:
            active = np.arange(n)
            t_stop[t_edge_min_indices] = t_min

    r = np.array(radii).astype(float)
    r[np.argwhere(t_stop == None)] = 0
    for i in range(n):
        x, y = centers[i]
        r_i = min(x, 1 - x, y, 1 - y)
        r[i] = max(r[i], r_i)

    if validate_packing(centers, r)[0]:
        return np.sum(r)
    else:
        return None

# Helper function to create a regular grid of centers
def create_regular_grid():
    step = 1 / 5
    x, y = np.mgrid[0:1:step, 0:1:step]
    centers = np.empty((26, 2))
    centers[12:17] = x[1:-1, 1:-1].flatten()[:, None]
    centers[17:] = x[1:-1, 2:-1].flatten()[:, None]
    centers[9:12] = y[1:-1, 1:-1].flatten()[:, None] * (x[1:-1, 1:-1].flatten()[:, None] > 0.5)
    centers[2:5] = y[1:-1, 2:-1].flatten()[:, None] * (x[1:-1, 2:-1].flatten()[:, None] > 0.5)
    centers[3:17] = y[2:-1, 1:-1].flatten()[:, None]
    centers[0:3] = y[1:-1, 0:1].flatten()[:, None]

    return centers

# Helper function to create a hexagonal grid of centers
def create_hexagonal_grid():
    step = 1 / (2 * np.sqrt(3))
    centers = []
    for k in range(5):
        if k % 2 == 0:
            centers.append(np.array([[0] * 2] * (k+1)))
        else:
            centers.append(np.array([[i/(2*k) * np.sqrt(3)] * (2*k-1) for i in range(0, k+1)]))

    return np.concatenate(centers) + [0.5, 0.5]

# Helper function to create a hybrid grid of centers
def create_hybrid_grid():
    step = np.array([1 / 3, 1 / 5, 1 / 3])

    x1, y1 = np.mgrid[0:1:step[0], 0:1:step[1]]
    x2, y2 = np.mgrid[0.45:1:step[1], 0.45:1:step[1]]
    x3, y3 = np.mgrid[0.25:1:step[1], 0.25:1:step[1]]

    centers = np.array([
        np.concatenate([x1.flatten()[:, None], y1.flatten()[:, None]]),
        np.concatenate([x2.flatten()[:, None], y2.flatten()[:, None]]),
        np.concatenate([x3.flatten()[:, None], y3.flatten()[:, None]]),
    ])
    return centers

# Helper function to create a random grid of centers
def create_random_grid():
    centers = np.random.rand(26, 2)
    return centers

# Main function to run the packing simulation and refinement
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)
    N = 26
    T = 3000
    S = 10
    best_r_sum = 0
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)

    geoms = [
        create_regular_grid(),
        create_hexagonal_grid(),
        create_hybrid_grid(),
        create_random_grid()
    ]

    for geom in geoms:
        for _ in range(S):
            speeds = (np.random.lognormal(mean=0., sigma=0.4, size=N) 
                      * (1 + 2 / np.minimum(np.minimum(geom[:, 0], geom[:, 1]), np.minimum(1 - geom[:, 0], 1 - geom[:, 1]))[:, None]))
            for r in [speeds]:
                for __ in range(T):
                    sim_r_sum = simulate_and_validate(geom, r)
                    if sim_r_sum is not None and sim_r_sum > best_r_sum:
                        best_r_sum = sim_r_sum
                        best_centers = geom
                        best_radii = r
    return best_centers, best_radii, best_r_sum

# Run the packing and return the result
centers, radii, sum_radii = run_packing()
