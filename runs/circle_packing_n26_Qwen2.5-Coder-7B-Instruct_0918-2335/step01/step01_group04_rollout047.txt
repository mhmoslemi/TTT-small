import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    return np.sqrt((diffs ** 2).sum(axis=2))

def solve_radii_lp(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    box_size = 1.0
    A_ineq = np.zeros((n * 2 + n * (n - 1), n))
    b_ineq = np.zeros(n * 2 + n * (n - 1))
    
    for i in range(n):
        A_ineq[i, 2 * i] = 1
        A_ineq[i, 2 * i + 1] = 1
        A_ineq[n + i, 2 * i] = -1
        A_ineq[n + i, 2 * i + 1] = -1
        b_ineq[i] = centers[i, 0]
        b_ineq[n + i] = 1 - centers[i, 0]
        b_ineq[n + n + i] = centers[i, 1]
        b_ineq[n + n + n + i] = 1 - centers[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            mask = 2 * n + i * (n - 1) + j - i * (i + 1) // 2
            A_ineq[mask, 2 * i] = 1
            A_ineq[mask, 2 * j] = 1
            A_ineq[mask, 2 * i + 1] = 1
            A_ineq[mask, 2 * j + 1] = 1
            dist = pairwise_dists(np.array([centers[i], centers[j]]))[0, 1]
            b_ineq[mask] = dist
    
    A_ineq *= 1.0  # Ensure that A_ineq is of type float
    res = linprog(c=np.zeros(n), A_ub=A_ineq, b_ub=b_ineq, bounds=[(0, None) for _ in range(n)], method='highs')
    return -res.fun if res.success else np.zeros(n)

def sa_optimize(initial_centers: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    best_centers = initial_centers.copy()
    best_radii = solve_radii_lp(best_centers)
    current_centers = initial_centers.copy()
    current_radii = best_radii.copy()
    best_sum_radii = np.sum(best_radii)
    
    for _ in range(500):
        i = np.random.randint(n)
        delta_x, delta_y = np.random.randn(2) * 0.01
        current_centers[i, 0] += delta_x
        current_centers[i, 1] += delta_y
        if 0 <= current_centers[i, 0] <= 1 and 0 <= current_centers[i, 1] <= 1:
            current_radii = solve_radii_lp(current_centers)
            current_sum_radii = np.sum(current_radii)
            if current_sum_radii > best_sum_radii + 1e-8 or (np.isclose(best_sum_radii, current_sum_radii) and np.random.rand() < 0.1):
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
                best_sum_radii = current_sum_radii
                
    return best_centers, best_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    # Generate initial candidate center lists
    initial_centers_hex = []
    for k in range(1, 100):
        hex_grid = np.array([np.cos(2 * np.pi * j / 3 + np.pi / 3) * k * np.sqrt(3) for j in range(k * k)] +
                           [np.cos(2 * np.pi * j / 3) * k * np.sqrt(3) for j in range(k * k)]).reshape(k * k, 2)
        hex_grid -= np.min(hex_grid, axis=0)
        hex_grid /= np.max(hex_grid, axis=0)
        hex_grid = hex_grid[:k * k]
        if hex_grid.shape[0] >= n and np.max(pairwise_dists(hex_grid)) < box_size:
            initial_centers_hex.append(hex_grid[:n])
            break
    
    initial_centers_random = np.random.rand(n, 2)
    initial_centers_mixed = np.sort(np.vstack((initial_centers_hex[-1][:n//2], initial_centers_random[n//2:])), axis=0)
    
    initial_centers_list = [initial_centers_hex[-1], initial_centers_random, initial_centers_mixed]
    
    # Run SA and refine with SLSQP
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum_radii = 0.0
    
    for initial_centers in initial_centers_list:
        centers, radii = sa_optimize(initial_centers)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum_radii:
            best_centers = centers
            best_radii = radii
            best_sum_radii = sum_radii
    
    bounds = [(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n
    
    def objective(x):
        return -np.sum(x[52:])
    
    def constraints(x):
        pos = x[:52].reshape(n, 2)
        r = x[52:]
        
        # Boundary constraints
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r  # left
        cons[26:52] = 1 - pos[:, 0] - r  # right
        cons[52:78] = pos[:, 1] - r  # bottom
        cons[78:104] = 1 - pos[:, 1] - r  # top
        
        # Pairwise constraints
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])
    
    res = minimize(
        fun=objective,
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success or np.any(constraints(res.x) < -1e-10):
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

# Call the function
centers, radii, sum_radii = run_packing()
print(f"Centers:\n{centers}\nRadii:\n{radii}\nSum of Radii: {sum_radii}")
