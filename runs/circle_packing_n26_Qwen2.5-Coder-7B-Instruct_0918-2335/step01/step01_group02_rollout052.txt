import numpy as np
from scipy.optimize import minimize, linprog

# Seed the random number generator for reproducibility
np.random.seed(42)

def solve_radii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []
    for i in range(n):
        # Boundary constraints
        A_ub.append([1.0 if i == idx else 0.0 for idx in range(n)])
        b_ub.append(pos[i, 0])
        A_ub.append([1.0 if i == idx else 0.0 for idx in range(n)])
        b_ub.append(1 - pos[i, 0])
        A_ub.append([1.0 if i == idx else 0.0 for idx in range(n)])
        b_ub.append(pos[i, 1])
        A_ub.append([1.0 if i == idx else 0.0 for idx in range(n)])
        b_ub.append(1 - pos[i, 1])
    for i in range(n):
        for j in range(i + 1, n):
            # Pairwise constraints
            dist = np.sqrt((pos[i, 0] - pos[j, 0]) ** 2 + (pos[i, 1] - pos[j, 1]) ** 2)
            A_ub.append([1.0 if i == idx else 0.0 for idx in range(n)])
            A_ub[-1].append(1.0 if i == idx else 0.0)
            b_ub.append(dist - (0.5 + 0.5))
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success and np.allclose(result.x, np.zeros(n), atol=1e-6):
        result.x.fill(0)
        result.success = True
        result.message = "LP solved successfully, but radii too small"
    return result.x, -result.fun

def fitness(pos):
    r, s_r = solve_radii(pos)
    return s_r

def crossover(parent1, parent2, rate):
    child = np.empty_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < rate:
            child[i] = np.random.rand()
        else:
            child[i] = parent1[i]
    return child

def mutate(individual, rate, sigma):
    for i in range(len(individual)):
        if np.random.rand() < rate:
            individual[i] += np.random.normal(0, sigma)
            individual[i] = np.clip(individual[i], 0, 1)
    return individual

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(len(pop), k, replace=False)
    return indices[np.argmax(fitness_vals[indices])]

def ga_best_positions(popsize, generations, elite_size, crossover_rate, mutation_rate, mutation_sigma):
    n = 26
    pop = np.random.rand(popsize, 2 * n)
    best_pos = pop[0, :n]  # Just to initialize, will be overridden
    best_sum_r = -np.inf
    for generation in range(generations):
        fitness_vals = np.array([fitness(pop[i, :n]) for i in range(popsize)])
        keep = np.argsort(fitness_vals)[-elite_size:]
        new_pop = np.zeros_like(pop)
        new_pop[:elite_size] = pop[keep]
        for i in range(elite_size, popsize, 2):
            parent1 = pop[tournament(pop, fitness_vals, k=3)]
            parent2 = pop[tournament(pop, fitness_vals, k=3)]
            child1 = crossover(parent1, parent2, crossover_rate)
            child2 = crossover(parent1, parent2, crossover_rate)
            child1 = mutate(child1, mutation_rate, mutation_sigma)
            child2 = mutate(child2, mutation_rate, mutation_sigma)
            new_pop[i] = child1
            new_pop[i + 1] = child2
        pop = new_pop
        current_best_pos = pop[np.argmax(fitness_vals)][:n]
        current_sum_r = fitness(current_best_pos)
        if current_sum_r > best_sum_r:
            best_pos = current_best_pos
            best_sum_r = current_sum_r
    return best_pos

def refine_solution(centers, radii):
    bounds = []
    for _ in range(2 * 26):
        bounds.append((0, 1))
    A = []
    b = []
    for i in range(26):
        A.append([1.0 if i == idx else 0.0 for idx in range(26)])
        b.append(centers[i, 0] - radii[i])
        A.append([1.0 if i == idx else 0.0 for idx in range(26)])
        b.append(1 - centers[i, 0] - radii[i])
        A.append([1.0 if i == idx else 0.0 for idx in range(26)])
        b.append(centers[i, 1] - radii[i])
        A.append([1.0 if i == idx else 0.0 for idx in range(26)])
        b.append(1 - centers[i, 1] - radii[i])
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx ** 2 + dy ** 2)
            A.append([1.0 for _ in range(26)] + [1.0 for _ in range(26)])
            b.append(d - (radii[i] + radii[j]))
    options = {
        'maxiter': 4000,
        'ftol': 1e-9,
        'disp': False,
    }
    initial_guess = list(centers.flatten()) + list(radii)
    result = minimize(lambda v: -v[:26].sum(), initial_guess, method='SLSQP', bounds=bounds,
                      constraints={"type": "ineq", "fun": lambda v: np.array(A) @ v - np.array(b)},
                      options=options)
    return result.x[:n].reshape(n, 2), result.x[n:]

def run_packing():
    popsize = 80

    crossover_rate = 0.9

    mutation_sigma = 0.03
    best_pos = ga_best_positions(popsize, generations, elite_size, crossover_rate, mutation_rate, mutation_sigma)
    centers, radii = refine_solution(best_pos, np.zeros(26))
    sum_radii = radii.sum()
