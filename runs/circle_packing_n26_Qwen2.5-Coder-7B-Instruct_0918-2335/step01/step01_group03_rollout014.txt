import numpy as np
from scipy.optimize import linprog, minimize

def optimise_radii(circles, bounds=((0, 0.5),)*26):
    n = len(circles)
    x = np.concatenate([circles.flatten(), np.zeros(n)])
    c = np.ones(2*n)
    A = []
    b = []
    
    for i in range(n):
        x0, y0 = circles[i]
        A.append([x0, y0, 1])
        b.append(x0)
        A.append([x0, y0, -1])
        b.append(-x0)
        A.append([y0, x0, 1])
        b.append(y0)
        A.append([y0, x0, -1])
        b.append(-y0)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = circles[j][0] - circles[i][0]
            dy = circles[j][1] - circles[i][1]
            d = np.sqrt(dx**2 + dy**2)
            A.append([dx / d, dy / d, -1])
            b.append(d)
            A.append([dx / d, dy / d, 1])
            b.append(-d)

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds)
    if result.success:
        return result.x[n:]
    else:
        return None

def run_packing():
    n = 26
    delta_max = 0.05
    T0 = 1.0
    alpha = 0.995
    NITER = 50000
    
    # Initial configuration: 5x5 grid + one centre circle
    centers = np.array([[0.2, 0.2], [0.2, 0.4], [0.2, 0.6], [0.2, 0.8],
                         [0.4, 0.2], [0.4, 0.4], [0.4, 0.6], [0.4, 0.8],
                         [0.6, 0.2], [0.6, 0.4], [0.6, 0.6], [0.6, 0.8],
                         [0.8, 0.2], [0.8, 0.4], [0.8, 0.6], [0.8, 0.8],
                         [0.5, 0.5]])
    current_radii = np.ones(n) * 0.1
    best_sum = np.sum(current_radii)
    best_centres = centers
    best_radii = current_radii
    
    for iter in range(1, NITER + 1):
        i = np.random.randint(0, n)
        dx, dy = np.random.uniform(-delta_max, delta_max, 2)
        new_coors = centers[i] + (dx, dy)
        if 0 <= new_coors[0] <= 1 and 0 <= new_coors[1] <= 1:
            temp_centers = centers.copy()
            temp_centers[i] = new_coors
            
            temp_radii = optimise_radii(temp_centers)
            if temp_radii is not None:
                new_sum = np.sum(temp_radii)
                ΔS = new_sum - best_sum
                if ΔS > 0:
                    current_centres, current_radii, best_sum = temp_centers, temp_radii, new_sum
                    if new_sum > best_sum:
                        best_centres, best_radii = temp_centers, temp_radii
                else:
                    T = T0 * alpha**iter
                    if np.random.rand() < np.exp(ΔS / T):
                        current_centres, current_radii = temp_centers, temp_radii
    
    # Final refinement
    result = minimize(lambda v: -np.sum(v[52:]), np.concatenate((current_centres.flatten(), current_radii)), 
                     method='SLSQP', bounds=((0, 1),)*25 + ((0, 0.5),)*26, options={'maxiter': 2000, 'ftol': 1e-9})
    if result.success:
        best_centres = result.x[:52].reshape(n, 2)
        best_radii = result.x[52:]
    
    return best_centres, best_radii, np.sum(best_radii)

centers, radii, sum_radii = run_packing()
print("Centers:\n", centers)
print("Radii:\n", radii)
print("Sum of radii:", sum_radii)
