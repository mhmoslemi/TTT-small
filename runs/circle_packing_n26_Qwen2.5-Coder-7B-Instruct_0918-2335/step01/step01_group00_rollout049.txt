import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist
from scipy.stats import norm
from numpy.random import default_rng

def compute_radii_and_value(centres):
    n = centres.shape[0]
    w_limits = np.min(np.stack([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]]), axis=0)
    A_ub = np.block([[np.eye(n), -np.eye(n)], 
                     np.array([np.ones((n, n)), 0])])
    D = cdist(centres, centres)
    b_ub = np.concatenate([D.flat, w_limits.repeat(n)])
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, w) for w in w_limits], method='highs')
    if res.success:
        return -res.fun, res.x[:n]
    else:
        return -np.inf, np.zeros(n)

def run_packing():
    n = 26
    initial_layout = make_initial_layout(n)
    best_objective = -np.inf
    best_radii = np.zeros(n)
    best_centers = np.zeros((n, 2))

    for _ in range(150):
        centers = propose_new_centers(initial_layout)
        objective, radii = compute_radii_and_value(centers)
        if objective > best_objective:
            best_objective = objective
            best_radii = radii
            best_centers = centers

        initial_layout = centers

    initial_layout = best_centers
    objective, radii = compute_radii_and_value(initial_layout)
    result = minimize(lambda x: -compute_radii_and_value(x.reshape((n, 2)))[0], initial_layout.flatten(), method='BFGS', bounds=[(0, 1)]*52)
    final_objective = -result.fun
    final_radii = compute_radii_and_value(result.x.reshape((n, 2)))[1]
    final_centers = result.x.reshape((n, 2))

    return final_centers, final_radii, final_objective

def make_initial_layout(n):
    if n <= 4:
        initial_layout = np.random.rand(n, 2) * 0.9  # Jitter away from boundaries
    else:
        hex_points = hexagonal_grid_layout(n)
        initial_layout = hex_points
    return initial_layout + 0.02 * np.random.randn(*initial_layout.shape)

def hexagonal_grid_layout(n):
    hex_radius = np.sqrt(3) / 3
    hex_height = 2 * hex_radius
    hex_width = 1.5 * hex_radius

    num_rows = np.ceil(np.sqrt(3 * n)).astype(int)
    num_cols = num_rows + 1 if num_rows % 2 != 0 else num_rows
    total_points = num_rows * num_cols

    if total_points < n:
        raise ValueError("Not enough space for all circles in a hexagonal grid")

    grid = np.zeros((total_points, 2))
    idx = 0
    for r in range(num_rows):
        cx = (r % 2) * hex_width / 2.0
        for c in range(num_cols):
            cy = (r * hex_height - hex_height / 2.0) * np.cos(np.pi / 6.0)
            px = cx + c * hex_width / 2.0
            py = cy
            grid[idx] = [px, py]
            idx += 1

    return grid[:n]

def propose_new_centers(centers):
    index = np.random.randint(centers.shape[0])
    dx, dy = norm.rvs(loc=0, scale=1/50)
    new_center = centers[index] + np.array([dx, dy])
    new_center = np.clip(new_center, 0, 1)
    centers[index] = new_center
    return centers
