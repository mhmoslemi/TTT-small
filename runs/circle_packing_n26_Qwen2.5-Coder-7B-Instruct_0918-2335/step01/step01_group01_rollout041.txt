import numpy as np
from scipy.optimize import minimize, linprog

# LP oracle to compute exact radii for a given centroid set
def lp_oracle(pos):
    n = pos.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, 2 * n))
    b_ub = np.zeros(A_ub.shape[0])
    
    R_idx = 0
    for i in range(n):
        A_ub[R_idx, 2 * i] = 1
        A_ub[R_idx, 2 * i + 1] = -1
        b_ub[R_idx] = pos[i, 0]
        R_idx += 1
    
    for i in range(n):
        A_ub[R_idx, 2 * i] = 1
        A_ub[R_idx, 2 * i + 1] = -1
        b_ub[R_idx] = 1 - pos[i, 0]
        R_idx += 1
    
    for i in range(n):
        A_ub[R_idx, 2 * i] = 1
        A_ub[R_idx, 2 * i + 1] = -1
        b_ub[R_idx] = pos[i, 1]
        R_idx += 1
    
    for i in range(n):
        A_ub[R_idx, 2 * i] = 1
        A_ub[R_idx, 2 * i + 1] = -1
        b_ub[R_idx] = 1 - pos[i, 1]
        R_idx += 1
    
    D = np.linalg.norm(pos[:, np.newaxis, :] - pos[np.newaxis, :, :], axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            if i != j:
                A_ub[R_idx, 2 * i] = 1
                A_ub[R_idx, 2 * i + 1] = 1
                A_ub[R_idx, 2 * j] = 1
                A_ub[R_idx, 2 * j + 1] = 1
                b_ub[R_idx] = D[i, j]
                R_idx += 1
    
    result = linprog([-1] * n,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=[(0, 0.5)] * n,
                    method='highs',
                    tol=1e-9)
    
    if result.status != 0:
        return -10
    
    return -result.fun

# Constructing a high-quality seed swarm
def generate_swarm():
    seeds = [
        np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]]),
        np.array([[0.25, 0.5], [0.75, 0.5], [0.5, 0.25], [0.5, 0.75]]),
        np.array([[0.5, 0.5] + [0.5 * np.cos(k * np.pi / 3), 0.5 * np.sin(k * np.pi / 3)] for k in range(6)]),
        np.array([[0.4, 0.4], [0.6, 0.4], [0.6, 0.6], [0.4, 0.6], [0.4 + 0.2, 0.4 + 0.2], [0.4 - 0.2, 0.4 + 0.2], [0.4 + 0.2, 0.4 - 0.2], [0.4 - 0.2, 0.4 - 0.2]]),
        np.array([[0.5, 0.25], [0.5, 0.75], [0.75, 0.5], [0.4, 0.4], [0.6, 0.4], [0.6, 0.6], [0.4, 0.6]]),
        np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.4, 0.4], [0.6, 0.4], [0.6, 0.6], [0.4, 0.6]]),
        np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.25 + 0.12, 0.25], [0.75 - 0.12, 0.25], [0.25 + 0.12, 0.75], [0.75 - 0.12, 0.75]]),
        np.array(np.random.rand(40, 2) * 0.8 + 0.1),
        np.array([[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]] + [[0.05 + 0.9 * np.cos(k * np.pi / 3), 0.05 + 0.9 * np.sin(k * np.pi / 3)] for k in range(6)])
    ]
    swarm = np.concatenate(seeds, axis=0)
    swarm += np.random.normal(0, 0.01, size=swarm.shape)
    swarm = np.clip(swarm, 0, 1)
    return swarm

# Particle-Swarm + Simulated Annealing (PSO-SA) loop
def pso_sa(swarm, max_genre=1000, T0=0.04, cooling_factor=0.995, mutation_rate=0.1, mutation_std=0.02):
    NP = swarm.shape[0]
    D = np.empty((NP, 52 * 2))
    V = np.empty((NP, 52 * 2))
    P_best = np.empty((NP, 52 * 2))
    V_best = np.empty((NP, 52 * 2))
    
    g_best_val = -np.inf
    g_best_pos = None
    
    for p in range(NP):
        V[p] = np.random.randn(52 * 2) * 0.03
        pos = swarm[p].reshape(26, 2)
        dists = np.linalg.norm(pos[:, np.newaxis, :] - pos[np.newaxis, :, :], axis=2)
        constraints = []
        for i in range(26):
            constraints.append(pos[i, 0] - 0.5)
            constraints.append(0.5 - pos[i, 0])
            constraints.append(pos[i, 1] - 0.5)
            constraints.append(0.5 - pos[i, 1])
            for j in range(i + 1, 26):
                constraints.append(dists[i, j] - 0.5)
        
        problem = {'c': constraints, 'A': None, 'b': None}
        opts = {'disp': False, 'maxiter': 4000}
        result = minimize(lp_oracle, swarm[p], constraints=problem, method='SLSQP', options=opts)
        if result.success and np.sum(result.x) > 0:
            D[p] = result.x
            V_best[p] = V[p]
            P_best[p] = swarm[p]
            if sum(D[p]) > g_best_val:
                g_best_val = sum(D[p])
                g_best_pos = swarm[p]
                V_best = V
                P_best = swarm
    
    for t in range(max_genre):
        T = T0 * cooling_factor**t
        infeasible = False
        for i in range(NP):
            if g_best_val == -np.inf:
                print("Restarting swarm...")
                swarm[i] = generate_swarm()[i]
                continue
            V[i] = 0.6 * V[i] + 1.5 * np.random.rand(52 * 2) * (P_best[i] - swarm[i]) + 1.5 * np.random.rand(52 * 2) * (g_best_pos - swarm[i])
            V[i] = np.clip(V[i], -0.03, 0.03)
            r1, r2 = np.random.rand(2)
            swarm[i] += V[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            if np.random.rand() < min(1, T / T0):
                V[i] = np.random.randn(52 * 2) * mutation_std
            dists = np.linalg.norm(swarm[i, np.newaxis, :] - swarm[i, np.newaxis, :], axis=2)
            constraints = []
            for j in range(26):
                constraints.append(swarm[i, j, 0] - 0.5)
                constraints.append(0.5 - swarm[i, j, 0])
                constraints.append(swarm[i, j, 1] - 0.5)
                constraints.append(0.5 - swarm[i, j, 1])
                for k in range(j + 1, 26):
                    constraints.append(dists[j, k] - 0.5)
            
            problem = {'c': constraints, 'A': None, 'b': None}
            opts = {'disp': False, 'maxiter': 4000}
            result = minimize(lp_oracle, swarm[i], constraints=problem, method='SLSQP', options=opts)
            if result.success and sum(result.x) > 0:
                if sum(result.x) > np.sum(D[i]):
                    D[i] = result.x
                    V_best[i] = V[i]
                    P_best[i] = swarm[i]
                    if sum(result.x) > g_best_val:
                        g_best_val = sum(result.x)
                        g_best_pos = swarm[i]
                        V_best = V
                        P_best = swarm
            else:
                infeasible = True
        if infeasible and np.count_nonzero(D < 0) > 20:
            swarm = generate_swarm()
            i = 0
    
    return g_best_pos, D[0], g_best_val

# Final deterministic polish
def polish_solution(pos, maxiter=4000):
    radii = lp_oracle(pos.reshape(26, 2))
    if radii < 0:
        print("LP oracle returned infeasible solution.")
        return pos.reshape(26, 2), radii
    
    res = minimize(lambda x: -np.sum(x[52:]),
                   np.concatenate([pos, radii.reshape(26,)]),
                   method='SLSQP',
                   bounds=[(0, 1)]*52 + [(0, 0.5)]*(len(radii)),
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    
    if res.success:
        return res.x[:52].reshape(26, 2), res.x[52:]
    else:
        return pos, radii
