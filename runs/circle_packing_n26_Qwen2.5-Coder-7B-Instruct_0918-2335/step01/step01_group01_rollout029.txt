import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Delaunay

def lp_oracle(pos):
    n = 26
    A = np.block([
        np.eye(n), -np.eye(n),
        np.eye(n), -np.multiply(pos[:, np.newaxis, 0], np.ones((n, n))),
        np.eye(n), -np.multiply(pos[:, np.newaxis, 1], np.ones((n, n)))
    ])
    b = np.concatenate([pos[:, 0], np.ones(n), pos[:, 1], np.ones(n)])
    G = np.zeros((325, 52))
    for i in range(n):
        for j in range(i + 1, n):
            idx1 = 2 * i
            idx2 = 2 * j
            G[325 - (n - 1) * i + j - 1, idx1] = 1
            G[325 - (n - 1) * i + j - 1, idx1 + 1] = -2 * pos[i, 0]
            G[325 - (n - 1) * i + j - 1, idx2] = 1
            G[325 - (n - 1) * i + j - 1, idx2 + 1] = -2 * pos[j, 0]
            G[325 - (n - 1) * i + j - 1, idx1 + 1] += 4 * np.sqrt(sum((pos[i] - pos[j]) ** 2))
            G[325 - (n - 1) * i + j - 1, idx2 + 1] += 4 * np.sqrt(sum((pos[i] - pos[j]) ** 2))
    h = np.zeros(325)
    res = minimize(lambda r: -np.sum(r), np.ones(n), method='highs', bounds=[(0, 0.5)] * n, constraints={'type': 'ineq', 'fun': lambda r: b - A @ r, 'jac': lambda r: -A}, options={'disp': False})
    if res.success:
        for i, r in enumerate(res.x):
            if r < 1e-12:
                res.x[i] = 0
        return res.x
    else:
        return np.full(n, 1e-6)

def apollonius(gap):
    a, b, c = gap
    ad = np.sqrt(a ** 2 + (c / 2) ** 2)
    bd = np.sqrt(b ** 2 + (c / 2) ** 2)
    return (bd * ad - 2 * np.sqrt(bd * ad * c * (a + b + c))) / (2 * (a + b + c))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    swarm_size = 40
    inertia = 0.6
    cognitive = 1.5
    social = 1.5
    annealing_start = 0.04
    cooling_factor = 0.995
    velocity_clamp = 0.03
    mutation_rate = 0.1
    mutation_std = 0.02
    max_generations = 1000
    inner_lp_tolerance = 1e-9

    def particle_swarm_annealing():
        particles = np.random.rand(swarm_size, 52) * 0.95 + 0.025
        velocities = np.zeros_like(particles)
        pbest_pos = particles
        pbest_val = -np.inf * np.ones(swarm_size)
        gbest_pos = pbest_pos[np.argmax(pbest_val)]
        gbest_val = np.max(pbest_val)
        temperatures = annealing_start * np.ones(swarm_size)

        for _ in range(max_generations):
            for i in range(swarm_size):
                r1, r2 = np.random.rand(2)
                v_new = inertia * velocities[i] + cognitive * r1 * (pbest_pos[i] - particles[i]) + social * r2 * (gbest_pos - particles[i])
                v_new = np.clip(v_new, -velocity_clamp, velocity_clamp)
                mutation = np.random.randn() * mutation_std if np.random.rand() < mutation_rate else 0
                v_new[0::2] += mutation
                v_new[1::2] += mutation
                v_new[0::2] = np.clip(v_new[0::2], 1e-6, 0.99)
                v_new[1::2] = np.clip(v_new[1::2], 1e-6, 0.99)
                new_pos = particles[i] + v_new
                new_pos = np.clip(new_pos, 1e-6, 0.99)
                radii = lp_oracle(new_pos)
                fitness = np.sum(radii)
                if fitness > pbest_val[i]:
                    pbest_pos[i] = new_pos
                    pbest_val[i] = fitness
                if fitness > gbest_val:
                    gbest_pos = new_pos
                    gbest_val = fitness
                temperatures[i] *= cooling_factor

            for i in range(swarm_size):
                if np.random.rand() < min(1, temperatures[i] / annealing_start):
                    v = np.random.randn(52) * mutation_std
                    particles[i] += v

        return gbest_pos, gbest_val

    def final_polish(pos):
        radii = lp_oracle(pos)
        n = 26
        bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26
        res = minimize(lambda v: -np.sum(v[26:]), np.concatenate([pos, radii]), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda v: np.concatenate([pos[:, 0] - v[26:], 1 - pos[:, 0] - v[26:], pos[:, 1] - v[26:], 1 - pos[:, 1] - v[26:]]), 'jac': lambda v: np.block([np.eye(n), -np.eye(n)])}, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
        if res.success and np.all(res.x[26:] >= 0):
            return res.x[:26], res.x[26:]
        else:
            return pos, radii

    best_pos, best_val = particle_swarm_annealing()
    centers, radii = final_polish(best_pos)
    return centers, radii, np.sum(radii)
