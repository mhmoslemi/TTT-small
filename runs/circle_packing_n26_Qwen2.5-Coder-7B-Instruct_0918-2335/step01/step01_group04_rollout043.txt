import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def lp problem(c, A_eq, b_eq, A_ub, b_ub):
        res = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=(0, None))
        if not res.success or res.status != 0:
            raise RuntimeError("LP problem is infeasible or optimal solution not found")
        return res.x

    def compute_active_constraints(centers, radii):
        n = len(centers)
        active_constraints = np.zeros(n, dtype=int)
        for i in range(n):
            wall_distance_left = centers[i, 0] - radii[i]
            wall_distance_right = 1 - centers[i, 0] - radii[i]
            wall_distance_bottom = centers[i, 1] - radii[i]
            wall_distance_top = 1 - centers[i, 1] - radii[i]
            distances = np.linalg.norm(centers - centers[i], axis=1)
            neighbor_distances = distances[distances != 0]
            radii_neighbor = radii[distances != 0]
            neighbor_constraints = neighbor_distances - (radii[i] + radii_neighbor)
            if wall_distance_left <= 0:
                active_constraints[i] = 1
            elif wall_distance_right <= 0:
                active_constraints[i] = 2
            elif wall_distance_bottom <= 0:
                active_constraints[i] = 3
            elif wall_distance_top <= 0:
                active_constraints[i] = 4
            elif np.any(neighbor_constraints <= 1e-6):
                active_constraints[i] = 5
        return active_constraints

    def push_away(centers, radii, step_size):
        n = len(centers)
        for i in range(n):
            active = active_constraints[i]
            if active == 1:
                centers[i, 0] += step_size
            elif active == 2:
                centers[i, 0] -= step_size
            elif active == 3:
                centers[i, 1] += step_size
            elif active == 4:
                centers[i, 1] -= step_size
            elif active == 5:
                distances = np.linalg.norm(centers - centers[i], axis=1)
                distances[distances == 0] = np.inf
                neighbor_index = np.where((distances - radii[i] - radii) < 1e-6)[0]
                if len(neighbor_index) > 0:
                    neighbor_center = centers[neighbor_index[0]]
                    direction = (centers[i] - neighbor_center) / np.linalg.norm(centers[i] - neighbor_center)
                    centers[i] += direction * step_size
        np.clip(centers, 0, 1, centers)
        return centers

    def compute_sum_radii(radii):
        return np.sum(radii)

    # Initialize centers in a hexagonal lattice seed
    hex_grid = np.array([[(i % 2) * 1.0 / 4 + (0.5 * (i // 2) % 2), 0.5 * np.sqrt(3) * (i // 2) / 4] for i in range(n)])
    centers = hex_grid[:n]
    centers += np.random.rand(n, 2) * 0.05  # Add some randomness

    # Optimizer parameters
    MAX_ITER = 200
    COOLING_INTERVAL = 20
    ESCAPE_INTERVAL = 50
    step_size = 0.01
    eps = 1e-7
    PAUSE_EPOCHS = 5

    best_centers = centers.copy()
    best_r = np.zeros(n)
    best_sum = compute_sum_radii(best_r)

    for iter in range(MAX_ITER):
        active_constraints = compute_active_constraints(centers, radii)
        centers = push_away(centers.copy(), radii, step_size)

        new_r = lp_problem(c=-np.ones(n), A_eq=None, b_eq=None, A_ub=np.zeros((0, n)), b_ub=None)
        new_sum = compute_sum_radii(new_r)

        if new_sum > best_sum + eps:
            best_centers = centers.copy()
            best_r = new_r.copy()
            best_sum = new_sum
        elif new_sum < best_sum:
            step_size /= 2

        if iter % COOLING_INTERVAL == 0 and step_size > 1e-10:
            step_size /= 2

        if iter % ESCAPE_INTERVAL == 0:
            centers += np.random.randn(n, 2) * step_size / 2

        if iter % PAUSE_EPOCHS == 0:
            if (iter - 1) > PAUSE_EPOCHS and np.abs(new_sum - best_sum) < 1e-6:
                break

    # Final polishing with SLSQP
    bounds = [(0, box_size)] * 2 * n + [(0, max_radius)] * n
    def objective(x):
        return -np.sum(x[52:])
    def constraints(x):
        pos = x[:52].reshape(n, 2)
        r = x[52:]
        cons = np.empty(104, float)
        cons[0:26] = pos[:, 0] - r
        cons[26:52] = 1 - pos[:, 0] - r
        cons[52:78] = pos[:, 1] - r
        cons[78:104] = 1 - pos[:, 1] - r
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])
    
    res = minimize(fun=objective, x0=np.concatenate((best_centers.flatten(), best_r)), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    if not res.success or np.any(constraints(res.x) < -1e-10):
        raise RuntimeError("Optimization failed")

    final_centers = res.x[:52].reshape(n, 2)
    final_r = res.x[52:]
    return final_centers, final_r, np.sum(final_r)
