import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 3
    
    # Create a Delaunay triangulation of a hexagonal lattice within the square
    hex_lattice = np.array([
        [0.5 + 0.5 * np.cos(2 * np.pi * i / 6), 0.5 + 0.5 * np.sin(2 * np.pi * i / 6)]
        for i in range(6)
    ])
    hex_centers = np.tile(hex_lattice, (3, 3))
    hex_centers = hex_centers[1:19, :]
    
    # Ensure points are within the box and calculate target radii
    hex_centers = hex_centers[np.all(hex_centers >= 0, axis=1) & np.all(hex_centers <= 1, axis=1), :]
    target_radii = box_size / 6 * np.ones(hex_centers.shape[0])
    if hex_centers.shape[0] < n:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    # Randomize the radii slightly to avoid symmetry
    np.random.shuffle(target_radii)
    target_radii = target_radii[:n]
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints
    bounds = [(0, box_size)] * n + [(0, max_radius)] * n
    
    def constraints(x):
        pos = x[:n].reshape(n, 2)
        r = x[n:]
        
        # Boundary constraints
        cons = np.empty(n, float)
        cons[0:n] = pos[:, 0] - r  # left
        cons[0:n] = 1 - pos[:, 0] - r  # right
        cons[0:n] = pos[:, 1] - r  # bottom
        cons[0:n] = 1 - pos[:, 1] - r  # top
        
        # Pairwise constraints
        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])
    
    # Initial guess
    initial_guess = np.concatenate((hex_centers.flatten(), target_radii))
    
    # Optimization
    res = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    # Extract results
    centers = res.x[:n].reshape(n, 2)
    radii = res.x[n:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
