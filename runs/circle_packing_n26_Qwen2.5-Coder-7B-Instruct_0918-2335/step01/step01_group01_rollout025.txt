import numpy as np
from scipy.optimize import minimize, linprog

# Constants
G = 20
NP = 40
T0 = 0.05
CRITICAL_RADIUS = 0.5

# Helper functions
def distance_matrix(centers):
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=-1))
    np.fill_diagonal(dists, np.inf)
    return dists

def lp_oracle(centers):
    r = (1 - centers[:, :2] - centers[:, 2:]) / 2
    r = np.maximum(r, 1e-6)
    r = np.minimum(r, CRITICAL_RADIUS - np.sum(r[:, None], axis=-1))
    r = np.maximum(r, 1e-6)
    A_ub = np.vstack([centers[:, 2:], centers[:, :2], centers[:, 2:], centers[:, :2]])
    b_ub = np.hstack([np.ones_like(centers[:, 2:]), 1 - centers[:, 2:], 1 - centers[:, 2:], 1 - centers[:, 2:]])
    A_eq = np.zeros((0, centers.shape[1]))
    b_eq = []
    result = linprog(-np.sum(r), A_bounds=A_eq, b_bounds=b_eq, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-9)
    if result.success:
        r = np.maximum(r, 1e-6)
        return np.sum(result.x)
    else:
        return -np.inf

def greedy seeding():
    sizes = np.array([0.16, 0.28, 0.28, 0.28, 0.35] + [0.35] * 22)
    centers = np.array([[0, 0],
                        [1, 0],
                        [0.5, 0],
                        [0, 1],
                        [1, 1],
                        [0.5, 1],
                        [0, 0.5],
                        [1, 0.5],
                        [0.25, 0.25],
                        [0.75, 0.25],
                        [0.25, 0.75],
                        [0.75, 0.75],
                        [0.375, 0.375],
                        [0.625, 0.375],
                        [0.375, 0.625],
                        [0.625, 0.625],
                        [0.125, 0.125],
                        [0.375, 0.125],
                        [0.625, 0.125],
                        [0.875, 0.125],
                        [0.125, 0.375],
                        [0.375, 0.375],
                        [0.625, 0.375],
                        [0.875, 0.375],
                        [0.125, 0.625],
                        [0.375, 0.625],
                        [0.625, 0.625],
                        [0.875, 0.625],
                        [0.125, 0.875],
                        [0.375, 0.875],
                        [0.625, 0.875],
                        [0.875, 0.875]])
    return centers, sizes

def swarm_search(seed_centers, seed_sizes):
    centers = seed_centers.copy()
    sizes = seed_sizes.copy()
    pbest = np.inf * np.ones((NP, 52))
    pbest_vals = np.full(NP, -np.inf)
    gbest = np.inf * np.ones(52)
    gbest_val = -np.inf
    T = T0
    inertia = 0.5
    cognitive = 1.5
    social = 1.5

    for _ in range(800):
        for i in range(NP):
            jitter = np.random.rand(52) * 1e-3
            pos = seed_centers[ i % 10] + jitter
            pos = np.clip(pos, 1e-6, 1-1e-6)
            sum_r = lp_oracle(pos)
            if sum_r > pbest_vals[i]:
                pbest[i] = pos.copy()
                pbest_vals[i] = sum_r
            if sum_r > gbest_val:
                gbest = pos.copy()
                gbest_val = sum_r
            r1, r2 = np.random.rand(2)
            v = inertia * (np.random.rand(52) * 2 - 1) + cognitive * r1 * (pbest[i] - pos) + social * r2 * (gbest - pos)
            if np.random.rand() < T / T0:
                v = np.random.randn(52) * 0.02
            pos += v
            pos = np.clip(pos, 1e-6, 1-1e-6)
            sum_r = lp_oracle(pos)
            if sum_r > pbest_vals[i]:
                pbest[i] = pos.copy()
                pbest_vals[i] = sum_r
            if sum_r > gbest_val:
                gbest = pos.copy()
                gbest_val = sum_r
        T *= 0.99

    return gbest, gbest_val

def refine_with_slsqp(gbest, gbest_val):
    def objective(x):
        return -np.sum(x[26:])
    
    def constraint(x):
        pos = x[:52].reshape(26, 2)
        radii = x[52:]
        dists = distance_matrix(pos)
        constraints = np.hstack([pos[:, 0] - radii, 1 - pos[:, 0] - radii, pos[:, 1] - radii, 1 - pos[:, 1] - radii, radii[:, None] + radii[None, :] - dists])
        return constraints

    x0 = np.concatenate((gbest, np.random.rand(26) * CRITICAL_RADIUS))
    cons = {'type': 'ineq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(52)] + [(0, CRITICAL_RADIUS) for _ in range(26)]
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    return result.x[:52].reshape(26, 2), result.x[52:], result.fun

def run_packing():
    seed_centers, seed_sizes = greedy_seeding()
    gbest, gbest_val = swarm_search(seed_centers, seed_sizes)
    centers, radii, final_val = refine_with_slsqp(gbest, gbest_val)
    return centers, radii, final_val

centers, radii, sum_radii = run_packing()
