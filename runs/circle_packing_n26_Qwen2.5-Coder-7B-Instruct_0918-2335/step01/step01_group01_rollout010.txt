import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial import Voronoi

def greedy_seeding(n, side_length):
    centers = []
    radius = (1 - np.sqrt(2)) / (np.sqrt(2) + 1)
    for i in range(4):
        x, y = (i & 1) * radius, (i >> 1) * radius
        if x + radius <= side_length and y + radius <= side_length:
            centers.append((x, y))
    radius -= 1e-6
    return np.array(centers), radius

def compute_distance_matrix(centers):
    n = centers.shape[0]
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist_matrix[i, j] = np.linalg.norm(centers[i] - centers[j])
    for i in range(n):
        dist_matrix[i, i] = np.inf
    return dist_matrix

def constraints_from_centers_and_radius(centers, radius):
    n = centers.shape[0]
    A_ineq = np.zeros((2 * n, n))
    b_ineq = np.zeros(2 * n)
    
    # Boundary constraints
    for i in range(n):
        for j in range(2):
            A_ineq[i * 2 + j, i] = 1 if j == 0 else -1
            b_ineq[i * 2 + j] = radius[i] if j == 0 else 1 - radius[i]
    
    # Contact constraints
    pairs = np.triu_indices(n, k=1)
    for i, j in pairs:
        A_ineq[2 *n + i * n + j, i] = 1
        A_ineq[2 *n + i * n + j, j] = 1
        b_ineq[2 *n + i * n + j] = radius[i] + radius[j]
    
    return A_ineq, b_ineq

def obj(p):
    r = p[52:]
    return -np.sum(r)

def constraints(p):
    pos = p[:52].reshape(26, 2)
    r = p[52:]
    A_ineq, b_ineq = constraints_from_centers_and_radius(pos, r)
    return np.concatenate([A_ineq, b_ineq])

def essentially_zero(x):
    return x < 1e-6

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NP = 40
    inertia = 0.5
    cognitive = 1.5
    social = 1.5
    T0 = 0.05
    cooling_rate = 0.99
    max_iterations = 800
    seed_positions, seed_radius = greedy_seeding(26, 1)
    
    best_center = None
    best_radii = None
    best_sum = 0
    
    np.random.seed(42)
    for _ in range(20):
        optionally_weighted_positions = seed_positions + 0.001 * np.random.randn(NP * 52).reshape(NP, 26, 2)
        velocities = np.random.randn(NP, 52)
        particles = [(pos, _sum) for (pos, _sum) in zip(optionally_weighted_positions, [0] * NP)]
        r_best_global = 0
        pbest_global_center = None
        
        for _ in range(max_iterations):
            gbest = max(particles, key=lambda x: x[1])[0][:52]
            for i in range(NP):
                r_i = particles[i][1]
                pos_i = particles[i][0][:52]
                c1, c2 = np.random.rand() * cognitive, np.random.rand() * social
                r1, r2 = np.random.rand() * 2 - 1, np.random.rand() * 2 - 1
                v_i = inertia * velocities[i] + c1 * r1 * (pbest_global_center[:52] - pos_i) + c2 * r2 * (gbest - pos_i)
                if np.random.rand() < T0 / T0:
                    v_i = np.random.randn(52) * 0.02
                pos_i += v_i
                pos_i = np.clip(pos_i, 1e-6, 1 - 1e-6)
                r = linprog([-1] * 26, linprog.A_ub=constraints_from_centers_and_radius(pos_i, np.ones(26))[0], linprog.b_ub=constraints_from_centers_and_radius(pos_i, np.ones(26))[1], bounds=[(0, None) for _ in range(26 * 2 + 26)], method='highs').x
                sum_r = -linprog([-1] * 26, linprog.A_ub=constraints_from_centers_and_radius(pos_i, r)[0], linprog.b_ub=constraints_from_centers_and_radius(pos_i, r)[1], bounds=[(0, None) for _ in range(26 * 2 + 26)], method='highs').fun
                if sum_r > r_best_global:
                    r_best_global = sum_r
                    pbest_global_center = pos_i
                if sum_r > particles[i][1]:
                    particles[i] = (pos_i, sum_r)
            T0 *= cooling_rate
        
        current_best = max(particles, key=lambda x: x[1])
        if current_best[1] > best_sum:
            best_center = current_best[0][:52].reshape(26, 2)
            best_radii = linprog([-1] * 26, linprog.A_ub=constraints_from_centers_and_radius(best_center, np.ones(26))[0], linprog.b_ub=constraints_from_centers_and_radius(best_center, np.ones(26))[1], bounds=[(0, None) for _ in range(26 * 2 + 26)], method='highs').x
            best_sum = current_best[1]
    
    return best_center[best_center[:, 0] >= 0], best_radii[best_radii >= 0], best_sum
