import numpy as np
from scipy.optimize import minimize, linprog

def compute_optimal_radii(pos):
    n = 26
    indices = np.triu(np.ones((n, n), bool), k=1)
    d = np.sqrt((pos[:, None] - pos[None, :])**2).sum(axis=2)
    A_ub = np.vstack((pos - np.ones(n) * 0.5, 1 - pos - np.ones(n) * 0.5, d[indices] - (d[indices] > 0) * 1e-9, -d[indices] + 1e-9)).T
    b_ub = np.hstack((np.ones(n) * 0.5, np.ones(n) * 0.5, d[indices].flat, np.ones(d[indices].size)))
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None))
    if not res.success:
        return None
    return res.x

def objective(pos_flat):
    radii = compute_optimal_radii(pos_flat)
    if radii is None:
        return 1e9
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    num_runs = 30
    dim = 52
    best_sum = -np.inf
    best-centre = np.zeros(dim)
    best_rad = np.zeros(26)

    for _ in range(num_runs):
        init_centres = np.random.rand(dim)
        res = minimize(objective, init_centres,
                       method='SLSQP', bounds=[(0, 1)]*dim,
                       options={'maxiter': 2000})
        if res.success:
            radii = compute_optimal_radii(res.x)
            if radii is not None:
                cur_sum = np.sum(radii)
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centre = res.x.reshape(26, 2)
                    best_rad = radii

    return best_centre, best_rad, best_sum
