import numpy as np
from scipy.optimize import minimize
from scipy.optimize.differentialevolution import differential_evolution

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraint(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    const = np.empty(104, float)
    const[0:26] = pos[:,0] - r
    const[26:52] = 1 - pos[:,0] - r
    const[52:78] = pos[:,1] - r
    const[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    const['pair'] = dists[mask] - (r[:,None] + r[None,:])[mask]
    return const

def global_search():
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    penalty = 1e6
    cons = {'type': 'ineq', 'fun': constraint}
    
    result = differential_evolution(obj, bounds=bounds, constraints=cons, popsize=15, maxiter=500, strategy='current_tournament', tol=1e-5)
    return result

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    result = global_search()
    v = result.x
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = {'type': 'ineq', 'fun': constraint}
    sol = minimize(obj, v, method='SLSQP', bounds=[(0,1)] * 2 * 26 + [(0, 0.5)] * 26, constraints=cons, tol=1e-9)
    if sol.success:
        v = sol.x
        pos = v[:52].reshape(26,2)
        r = v[52:]
    else:
        pos = v[:52].reshape(26,2)
        r = v[52:]
    return pos, r, np.sum(r)
