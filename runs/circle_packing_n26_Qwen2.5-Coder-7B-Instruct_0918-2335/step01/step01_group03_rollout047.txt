import numpy as np
import networkx as nx
from scipy.spatial import distance_matrix
from scipy.optimize import linprog, minimize

def distances_matrix(centers):
    return distance_matrix(centers, centers)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def linprog_optimize_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_eq = -np.transpose(centers)
    b_eq = -np.ones(n)
    A_ineq = np.vstack((np.eye(n), -np.eye(n), -centers[:, 0], centers[:, 0] - 1, -centers[:, 1], centers[:, 1] - 1))
    b_ineq = np.zeros(6 * n)
    A_ineq[:, n:] += 1
    constraints = ({'type': 'eq', 'fun': lambda r: A_eq @ r + b_eq},
                   {'type': 'ineq', 'fun': lambda r: A_ineq @ r + b_ineq})
    bounds = [(0, 1)] * n * 2
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
    
    if not result.success or np.isnan(result.x).any() or np.isinf(result.x).any():
        return None
    return result.x[:n]

def refine_with_sa(centers, radii):
    n = centers.shape[0]
    T0 = 1e-1
    T = T0
    alpha = 0.995
    lambda1 = 5e6
    lambda2 = 5e6
    energy = -np.sum(radii) + lambda1 * np.sum(np.maximum(0, radii[:, np.newaxis] + radii - distances_matrix(centers) - 1e-6)) + lambda2 * np.sum(np.maximum(0, radii - distances_matrix(np.array([np.linspace(0, 1, n), np.linspace(0, 1, n)]).T, np.array([0, 0]).reshape(1, 2)) - 1e-6))
    
    for i in range(3000):
        energy_prev = energy
        for j in range(n):
            dx, dy = np.random.normal(0, 0.02, 2)
            new_center = centers[j] + np.array([dx, dy])
            new_center = np.clip(new_center, 0, 1)
            new_radii = linprog_optimize_radii(np.vstack((centers[:j], new_center, centers[j+1:])))
            if new_radii is not None:
                new_energy = -np.sum(new_radii) + lambda1 * np.sum(np.maximum(0, new_radii[:, np.newaxis] + new_radii - distances_matrix(np.vstack((centers[:j], new_center, centers[j+1:]))) - 1e-6)) + lambda2 * np.sum(np.maximum(0, new_radii - distances_matrix(np.array([np.linspace(0, 1, n), np.linspace(0, 1, n)]).T, np.array([0, 0]).reshape(1, 2)) - 1e-6))
                if np.random.rand() < np.exp((energy_prev - new_energy) / T):
                    centers[j] = new_center
                    radii = new_radii
                    energy = new_energy
        T *= alpha
    return centers, radii
