import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def lp(ranges):
        c = np.ones(n)
        A = np.zeros((4*n, n))
        b = np.zeros(4*n)
        for i in range(n):
            A[i, i] = 1
            b[i] = ranges[i][1]
            A[i+n, i] = -1
            b[i+n] = -ranges[i][0]
        return linprog(c, A_ub=A, b_ub=b, method='highs').x
    
    def push_away(centers, radii, step_size):
        new_centers = centers.copy()
        for i in range(n):
            dists = np.sqrt(np.sum((new_centers[i] - centers)**2, axis=1))
            active_idx = np.argmin(dists)
            d = dists[active_idx] - radii[active_idx] - radii[i]
            if d > 0:
                continue
            u = (centers[active_idx] - centers[i]) / d
            new_centers[i] += step_size * u
            new_centers[i] = np.clip(new_centers[i], 0, box_size)
        return new_centers
    
    def accept_move(new_radii, old_radii, best_sum, eps):
        if np.sum(new_radii) > best_sum + eps:
            return True, new_radii
        else:
            return False, old_radii

    best_sum = 0.0
    eps = 1e-7
    alpha = 0.01
    cooling_interval = 20
    escape_interval = 50
    max_iter = 200
    pause_epochs = 5
