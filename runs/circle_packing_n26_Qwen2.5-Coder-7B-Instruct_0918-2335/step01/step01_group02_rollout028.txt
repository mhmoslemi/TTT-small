import numpy as np
from scipy.optimize import linprog, minimize

def rectilinear_contact_graph(num_circles):
    vertices = np.arange(num_circles).reshape(5, 5)
    top_row = vertices[0]
    bottom_row = vertices[-1]
    left_col = vertices[:, 0]
    right_col = vertices[:, -1]

    edges = []
    for i in range(4):
        for j in range(4):
            edges.append((top_row[i], top_row[i + 1]))
            edges.append((bottom_row[i], bottom_row[i + 1]))
            edges.append((left_col[i], left_col[i + 1]))
            edges.append((right_col[i], right_col[i + 1]))

    return edges

def hexagonal_contact_graph(num_circles):
    vertices = np.arange(num_circles).reshape(3, 3)
    center = num_circles // 2

    edges = []
    for i in range(3):
        for j in range(3):
            if i != 2:
                edges.append((vertices[i, j], vertices[i + 1, j]))
            if j != 2:
                edges.append((vertices[i, j], vertices[i, j + 1]))
            if i != 2 and j != 2:
                edges.append((vertices[i, j], vertices[i + 1, j + 1]))

    return edges

def expand_graph(graph, add_vertex, add_edge):
    new_graph = list(graph)
    new_graph.append(add_vertex)

    if add_vertex > 0:
        new_graph.append((0, add_vertex))
    if add_vertex < 25:
        new_graph.append((25, add_vertex))
    if add_vertex % 5 != 0:
        new_graph.append((add_vertex - 1, add_vertex))
    if add_vertex % 5 != 4:
        new_graph.append((add_vertex + 1, add_vertex))

    for node, _ in graph:
        if abs(node - add_vertex) == 1:
            new_graph.append((node, add_vertex))
        elif abs(node % 5 - add_vertex % 5) == 1:
            new_graph.append((node, add_vertex))

    if add_vertex not in [0, 5, 10, 15, 20, 25]:
        if (add_vertex - 5, add_vertex) not in new_graph:
            new_graph.append((add_vertex - 5, add_vertex))
        if (add_vertex + 5, add_vertex) not in new_graph:
            new_graph.append((add_vertex + 5, add_vertex))
        if (add_vertex - 1, add_vertex) not in new_graph:
            new_graph.append((add_vertex - 1, add_vertex))
        if (add_vertex + 1, add_vertex) not in new_graph:
            new_graph.append((add_vertex + 1, add_vertex))

    return new_graph

def build_constraints(edges, num_circles):

def solve_lp(num_circles, edges):
    A_ub, b_ub = build_constraints(edges, num_circles)
    res = linprog(np.ones(num_circles) * -1, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
    
    if not res.success:
        return None, None, None
    
    radii = res.x
    max_radius = np.max(radii)
    
    return radii, max_radius, np.sum(radii)

def run_packing():
    num_circles = 26
    best_sum = 0
    best_positions = None
    best_radii = None

    # Rectilinear graphs
    for i in range(25):
        graph = rectilinear_contact_graph(num_circles)
        graph = expand_graph(graph, i, None)
        radii, max_radius, sum_radii = solve_lp(num_circles, graph)
        if radii is not None and sum_radii > best_sum:
            best_sum = sum_radii
            best_positions = None
            best_radii = radii

    # Hexagonal graphs
    for i in range(3):
        for j in range(3):
            graph = hexagonal_contact_graph(num_circles)
            graph = expand_graph(graph, i * 3 + j, None)
            radii, max_radius, sum_radii = solve_lp(num_circles, graph)
            if radii is not None and sum_radii > best_sum:
                best_sum = sum_radii
                best_positions = None
                best_radii = radii

    # Refine solution with SLSQP
    if best_radii is not None:
        def obj(v):
            r = v[52:]
            return -np.sum(r)

        def constraints(v):
            pos = v[:52].reshape(26, 2)
            r = v[52:]
            
            cons = np.empty(104, float)
            cons[0:26] = pos[:, 0] - r
            cons[26:52] = 1 - pos[:, 0] - r
            cons[52:78] = pos[:, 1] - r
            cons[78:104] = 1 - pos[:, 1] - r
            
            diffs = pos[:, None, :] - pos[None, :, :]
            dists = np.sqrt((diffs ** 2).sum(axis=2))
            mask = np.triu(np.ones((26, 26), bool), k=1)
            pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
            return np.concatenate([cons, pair_cons])

        bounds = [(0, 1)] * 26
        bounds += [(0, 1)] * 26
        bounds += [(0, 0.5)] * 26

        refined_res = minimize(fun=obj, x0=np.concatenate([best_positions.reshape(52), best_radii]), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if refined_res.success:
            best_radii = refined_res.x[52:]
            best_sum = -refined_res.fun

    return best_positions.reshape(26, 2) if best_positions is not None else np.zeros((26, 2)), best_radii if best_radii is not None else np.zeros(26), best_sum

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
