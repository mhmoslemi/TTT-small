import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance

POPSIZE = 80
GENERATIONS = 250
ELITE_SIZE = int(POPSIZE * 0.1)
CROSSOVER_RATE = 0.9
MUTATION_RATE = 0.2
MUTATION_SIGMA = 0.03

# Fitness function: evaluate sum of radii using given positions
def fitness(pos):
    r, s_r = solve_lradii(pos)
    return -s_r

# Solve LP to find optimal radii given positions
def solve_lradii(pos):
    n = len(pos)
    A_ub = []
    b_ub = []

    for i in range(n):
        for j in [0, 1]:
            A_ub.append(np.zeros(n))
            b_ub.append(0)
            A_ub[-1][i] = 1.0 if j == 0 else -1.0
            b_ub[-1] = (pos[i][j] if j == 0 else 1 - pos[i][j]) - 0.5

    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(pos[i] - pos[j])
            A_ub.append(np.zeros(n))
            b_ub.append(0)
            A_ub[-1][i] = 1.0
            A_ub[-1][j] = 1.0
            b_ub[-1] = d - (0.5 + 0.5)

    r = opt.linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs').x
    sum_r = r.sum()
    return r, sum_r

# Crossover operator with uniform probability
def crossover(parent1, parent2, rate=0.9):
    child = np.where(np.random.rand(len(parent1)) < rate, parent1, parent2)
    return child

# Mutation operator
def mutate(individual, rate=0.2, sigma=0.03):
    inds = np.where(np.random.rand(len(individual)) < rate)
    individual[inds] = np.clip(individual[inds] + sigma * np.random.randn(len(inds)), 0, 1)
    return individual

# Tournament selection
def tournament(pop, fitness_vals, k=3):
    indices = np.random.choice(np.arange(len(pop)), k)
    return indices[np.argmax(fitness_vals[indices])]

# Main genetic algorithm loop
def ga_best_positions():
    random_state = np.random.RandomState(42)
    pop = random_state.rand(POPSIZE, 52)
    best = None

    for generation in range(GENERATIONS):
        fitness_vals = np.array([fitness(p) for p in pop])
        population = []

        # Elitism
        for i in range(ELITE_SIZE):
            population.append(pop[np.argmax(fitness_vals)])

        # Crossover and mutation
        for _ in range(POPSIZE - ELITE_SIZE):
            parent1 = pop[tournament(pop, fitness_vals)]
            parent2 = pop[tournament(pop, fitness_vals)]
            child = crossover(parent1, parent2, CROSSOVER_RATE)
            child = mutate(child, MUTATION_RATE, MUTATION_SIGMA)
            population.append(child)

        pop = np.array(population)
        best_idx = np.argmax(fitness_vals)
        if best_idx < ELITE_SIZE or fitness_vals[best_idx] > (best[2] if best else -np.inf):
            best = pop[best_idx], fitness_vals[best_idx]
        elif generation % (GENERATIONS // 10) == 0:
            print(f'Generations {generation}: Best sum radii = {best[2]}')

    return best[0]

# Local polish using SLSQP after GA
def polish_solution(pos):
    r, _ = solve_lradii(pos)
    initial_guess = np.concatenate((pos.flatten(), r))
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    constraints = []
    for i in range(52):
        constraints.append({'type': 'ineq', 'fun': bound_constraint(i)})
        constraints.append({'type': 'ineq', 'fun': lower_bound_constraint(i, pos)})
        constraints.append({'type': 'ineq', 'fun': upper_bound_constraint(i, pos)})
    
    for i in range(52):
        for j in range(i + 1, 52):
            constraints.append({'type': 'ineq', 'fun': no_overlap_constraint(i, j, pos)})

    res = opt.minimize(fitness, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    if res.success:
        centers = res.x[:52].reshape(26, 2)
        radii = res.x[52:]
        sum_radii = -res.fun
        return centers, radii, sum_radii
    else:
        raise ValueError('SLSQP failed to converge')

# Constraint functions (negative)
def bound_constraint(i):
    def inner(x):
        idx = i // 2 if i % 2 == 0 else (i // 2) * 2 + 52
        return x[idx] - x[i] - 0.5
    return inner

def lower_bound_constraint(i, pos):
    def inner(x):
        idx = i // 2 if i % 2 == 0 else (i // 2) * 2 + 52
        return -x[idx] + pos[i // 2] + 0.5
    return inner

def upper_bound_constraint(i, pos):
    def inner(x):
        idx = i // 2 if i % 2 == 0 else (i // 2) * 2 + 52
        return -x[idx] + 1 - pos[i // 2] + 0.5
    return inner

def no_overlap_constraint(i, j, pos):
    def inner(x):
        dx = pos[i // 2][0] - pos[j // 2][0]
        dy = pos[i // 2][1] - pos[j // 2][1]
        rsum = 1.0
        return -np.sqrt(dx**2 + dy**2) + rsum
    return inner

# Main function to run the packing algorithm
def run_packing():
    best_pos = ga_best_positions()
    centers, radii, sum_radii = polish_solution(best_pos)
    assert validate_packing(centers, radii)[0], "Invalid packing"
    return centers, radii, sum_radii

# Entry point
if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Total Radius Sum: {sum_radii}")
