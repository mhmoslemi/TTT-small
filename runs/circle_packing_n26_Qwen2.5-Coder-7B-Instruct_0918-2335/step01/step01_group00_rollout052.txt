def lp_value_and_radii(centres):
    n = centres.shape[0]
    w_limits = np.minimum.reduce([centres[:,0], 1-centres[:,0], centres[:,1], 1-centres[:,1]])
    A_ub = np.zeros((n*(n-1)//2, 2*n))
    for i in range(n):
        for j in range(i+1, n):
            A_ub[j-1+i*(n-1)//2, i] = 1
            A_ub[j-1+i*(n-1)//2, n+j] = 1
    D = squareform(pdist(centres))
    b_ub = D + 1e-9  # small offset to include touching circles
    c = -np.ones(n)
