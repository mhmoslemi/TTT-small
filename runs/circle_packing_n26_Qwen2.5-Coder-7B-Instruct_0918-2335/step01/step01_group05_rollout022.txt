import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    n = centers.shape[0]
    M = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + M, n))
    b_ub = np.ones(4 * n + M)
    
    for i in range(n):
        A_ub[i, i] = A_ub[2*n + i, i] = 1
        A_ub[2*n + i + 1, i] = A_ub[3*n + i, i] = -1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[4*n + M*(i-1)//2 + (j-i-1), i] = 1
            A_ub[4*n + M*(i-1)//2 + (j-i-1), j] = 1
            b_ub[4*n + M*(i-1)//2 + (j-i-1)] = dist
    
    result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, method='highs')
    radii = result.x
    return -result.fun, radii

def evaluate_fitness(individual):
    centers = np.clip(np.reshape(individual, (26, 2)), 0, 1)
    sum_r, radii = solve_radii(centers)
    return sum_r

def crossover(parent1, parent2):
    point = np.random.randint(1, 51)
    child1 = np.concatenate((parent1[:point], parent2[point:]))
    child2 = np.concatenate((parent2[:point], parent1[point:]))
    return child1, child2

def mutate(individual, sigma):
    individual += np.random.normal(0, sigma, size=individual.shape)
    return np.clip(individual, 0, 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5
    
    np.random.seed(42)
    population = np.random.rand(population_size, 52)
    
    for _ in range(generations):
        fitnesses = np.array([evaluate_fitness(ind) for ind in population])
        
        new_population = population[np.argsort(fitnesses)[-elitism:]]
        
        while len(new_population) < population_size:
            parent1, parent2 = selection(population, tournament_size)
            children = crossover(parent1, parent2)
            for child in children:
                if np.random.rand() < mutation_rate:
                    child = mutate(child, initial_sigma)
                child = clamp_to_square(child)
                fitness = evaluate_fitness(child)
                if fitness > np.min(fitnesses):
                    new_population = np.vstack((new_population, child))
                    fitnesses = np.append(fitnesses, fitness)
                    if len(new_population) >= population_size:
                        break
        
        population = new_population
        initial_sigma *= sigma_decay
    
    best_index = np.argmax(fitnesses)
    best_individual = population[best_index]
    centers = np.clip(np.reshape(best_individual, (26, 2)), 0, 1)
    sum_r, radii = solve_radii(centers)
    
    result = minimize(fun=lambda x: -np.sum(x), x0=radii,
                      method='SLSQP', bounds=[(0, 0.5)] * n,
                      constraints={'type': 'ineq', 'fun': lambda x: x - radii},
                      options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    radii = result.x
    sum_r = -result.fun
    
    for i in range(n):
        if not (0 <= centers[i, 0] <= 1 and 0 <= centers[i, 1] <= 1):
            sum_r -= 1e-9
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                sum_r -= 1e-9
    
    return centers, radii, sum_r

def selection(population, tournament_size):
    return population[np.random.choice(range(population.shape[0]), size=tournament_size, replace=False)].min(axis=0)

def clamp_to_square(individual):
    return np.clip(individual, 0, 1)
