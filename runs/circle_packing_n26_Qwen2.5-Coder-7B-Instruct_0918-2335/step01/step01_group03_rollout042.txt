import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def optimise_radii(centers):
    n = centers.shape[0]
    r = np.array([0.5] * n)
    A = np.zeros((n * (n - 1), 2 * n))
    b = np.zeros(n * (n - 1))

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A[i * (n - 1) + j - i - 1, 2 * i] = 1
            A[i * (n - 1) + j - i - 1, 2 * i + 1] = -1
            A[i * (n - 1) + j - i - 1, 2 * j] = -1
            A[i * (n - 1) + j - i - 1, 2 * j + 1] = 1
            b[i * (n - 1) + j - i - 1] = dist

    A = np.vstack((A, np.eye(2 * n)))
    b = np.concatenate((b, np.zeros(2 * n)))
    A_eq = np.zeros((4 * n, 2 * n))
    b_eq = np.zeros(4 * n)

    for i in range(n):
        A_eq[4 * i, 2 * i] = 1
        A_eq[4 * i + 1, 2 * i + 1] = 1
        A_eq[4 * i + 2, 2 * i] = -1
        A_eq[4 * i + 3, 2 * i + 1] = -1
        b_eq[4 * i] = 0.5
        b_eq[4 * i + 1] = 0.5
        b_eq[4 * i + 2] = 0
        b_eq[4 * i + 3] = 0

    res = linprog(-r, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=([0, None] * n), method='highs')
    if res.success and np.allclose(np.dot(A, res.x), b, atol=1e-8):
        return res.x[n:]
    return None

def run_packing():
    n = 26
    centers = np.array([[0.2, 0.2], [0.2, 0.4], [0.2, 0.6], [0.2, 0.8],
                         [0.4, 0.2], [0.4, 0.4], [0.4, 0.6], [0.4, 0.8],
                         [0.6, 0.2], [0.6, 0.4], [0.6, 0.6], [0.6, 0.8],
                         [0.8, 0.2], [0.8, 0.4], [0.8, 0.6], [0.8, 0.8],
                         [0.5, 0.5]])
    centers = np.vstack((centers, [0.5, 0.5]))
    current_radii = optimise_radii(centers)
    best_sum = np.sum(current_radii)
    best_centres = centers
    best_radii = current_radii
    T0, alpha = 1.0, 0.995
    NITER, delta_max = 50000, 0.05

    for iter in range(NITER):
        i = np.random.randint(n)
        dx, dy = np.random.uniform(-delta_max, delta_max, 2)
        new_coors = centers[i] + (dx, dy)
        if np.any(new_coors < 0) or np.any(new_coors > 1):
            continue

        temp_centres = centers.copy()
        temp_centres[i] = new_coors

        temp_radii = optimise_radii(temp_centres)
        if temp_radii is None:
            continue

        new_sum = np.sum(temp_radii)
        ΔS = new_sum - best_sum
        if ΔS > 0:
            current_centres, current_radii, best_sum = temp_centres, temp_radii, new_sum
            if new_sum > best_sum:
                best_centres, best_radii = temp_centres, temp_radii
        else:
            T = T0 * alpha ** iter
            if np.random.rand() < np.exp(ΔS / T):
                current_centres, current_radii = temp_centres, temp_radii

    def slsqp_refine(centers, radii):
        n = centers.shape[0]
        x = centers.flatten()
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

        def obj(v):
            return -np.sum(v[n:])

        def constraints(v):
            pos = v[:2 * n].reshape(n, 2)
            r = v[2 * n:]
            cons = np.empty(4 * n, float)
            cons[0:n] = pos[:, 0] - r
            cons[n:2 * n] = 1 - pos[:, 0] - r
            cons[2 * n:3 * n] = pos[:, 1] - r
            cons[3 * n:4 * n] = 1 - pos[:, 1] - r
            diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
            dists = np.sqrt((diffs ** 2).sum(axis=2))
            mask = np.triu(np.ones((n, n), bool), k=1)
            pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
            return np.concatenate([cons, pair_cons])

        res = minimize(fun=obj, x0=x, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 2000, 'ftol': 1e-9})
        if res.success and np.all(constraints(res.x) >= -1e-10):
            new_centers = res.x[:2 * n].reshape(n, 2)
            new_radii = res.x[2 * n:]
            return new_centers, new_radii
        return centers, radii

    best_centres, best_radii = slsqp_refine(best_centres, best_radii)
    return best_centres, best_radii, np.sum(best_radii)
