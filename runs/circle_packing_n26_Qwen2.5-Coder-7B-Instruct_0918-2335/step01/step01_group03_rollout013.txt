# Simulated Annealing step
def simulated_annealing_step(current_centres, current_radii, T):
    i = np.random.randint(n)
    δx, δy = np.random.uniform(-delta_max, delta_max, 2)
    new_coords = current_centres[i].flatten() + [δx, δy]
    new_centres = current_centres.copy()
    new_centres[i] = new_coords

    new_radii = optimise_radii(new_centres)
    if new_radii is None:
        return current_centres, current_radii

    new_sum = np.sum(new_radii)
    ΔS = new_sum - np.sum(current_radii)
    if ΔS > 0:
        return new_centres, new_radii
    elif np.random.rand() < np.exp(ΔS / T):
        return new_centres, new_radii
    else:
        return current_centres, current_radii

# Refinement with SLSQP
def slsqp_refine(best_centres, best_radii):
    def slsqp_objective(x):
        return -np.sum(x[2*n:3*n])
    def slsqp_bounds():
        return [(0,1)] * (2*n) + [(0,0.5)] * n
    def slsqp_constraints(x):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n+i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[n+2*i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n+i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*n+1+2*i]})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((x[2*n:3*n] - x[2*n+2*j:3*n+2*j+2])**2))
                cons.append({'type': 'ineq', 'fun': lambda x, dist, i=i, j=j: dist - x[2*n+i] - x[2*n+j]})
        return cons

    slsqp_result = minimize(slsqp_objective, np.hstack((best_centres.flatten(), best_radii)), method='SLSQP', bounds=slsqp_bounds(), constraints=slsqp_constraints(), options={'maxiter': 2000, 'ftol': 1e-9})
    if slsqp_result.success:
        return slsqp_result.x.reshape(n, 2), slsqp_result.x[2*n:]
    else:
        return best_centres, best_radii
