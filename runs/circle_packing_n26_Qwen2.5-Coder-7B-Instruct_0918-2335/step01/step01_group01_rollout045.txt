import numpy as np
from scipy.optimize import minimize

def compute_optimal_radii(pos):
    r_i = pos[:26]
    r_j = pos[26:]
    pos = pos.reshape(26, 2)
    A_rows = []
    b_rows = []
    indices = np.triu_indices(26, k=1)
    
    for i, j in indices:
        distances = np.sqrt((pos[i, 0] - pos[j, 0])**2 + (pos[i, 1] - pos[j, 1])**2)
        A_rows.append([1, 1, -1, -1, 0, 0] + [0]*i + [0]*j + [-distances, distances] + [0]*29)
        b_rows.append(r_i[i] + r_j[j])
    
    A = np.array(A_rows)
    b = np.array([-np.inf]*2 + [-np.inf]*29 + b_rows)
    res = minimize(lambda r: -np.sum(r), np.zeros(26), method='SLSQP', bounds=[(0, 0.5)]*26, constraints={'type': 'ineq', 'fun': lambda r: b[:26] + np.dot(A[:, 26:], r)})
    if not res.success:
        return None
    return r_i + r_j

def objective(pos_flat):
    radii = compute_optimal_radii(pos_flat)
    if radii is None:
        return np.inf
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_center = None
    best_radii = None
    best_sum = 0

    for _ in range(50):
        init = np.random.rand(52) * 0.8 + 0.1  # Random but within [0,1]^2 with some margin
        res = minimize(fun=objective, x0=init, method='SLSQP',
                       bounds=[(0, 1)]*52, constraints={'type': 'ineq', 'fun': lambda p: objective(p)},
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        if res.success and res.fun < best_sum:
            best_sum = -res.fun
            best_center = res.x.reshape(26, 2)
            best_radii = compute_optimal_radii(best_center)
    
    return best_center, best_radii, best_sum
