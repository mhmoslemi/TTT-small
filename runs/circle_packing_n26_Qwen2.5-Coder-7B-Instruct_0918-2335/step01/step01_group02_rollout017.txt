import numpy as np
from scipy.optimize import linprog, differential_evolution, NonlinearConstraint

def objective(positions):
    radii = solve_radii(positions)
    return -np.sum(radii)

def solve_radii(positions):
    n = positions.shape[0] // 2
    pos = positions.reshape(n, 2)
    radii = np.zeros(n)
    
    for i in range(n):
        r_bound = [0, min(pos[i, 0], 1 - pos[i, 0], pos[i, 1], 1 - pos[i, 1])]
        r, _ = linprog(lambda r: -np.sum(r), A_ub=[], b_ub=[], bounds=[r_bound] * n)
        radii[i] = -r[0]
    
    dists = np.sqrt(np.sum((pos[:, None, :] - pos[None, :, :]) ** 2, axis=2))
    mask = np.triu(np.ones((n, n), bool), k=1)
    pair_cons = dists[mask] - (radii[:, None] + radii[None, :])[mask]
    
    while np.any(pair_cons < 0):
        min_overlap_idx = np.argmin(pair_cons)
        i, j = np.unravel_index(min_overlap_idx, dists.shape)
        diff = pos[i] - pos[j]
        overlap = radii[i] + radii[j] - dists[i, j]
        adjustment = overlap / (np.linalg.norm(diff) * 2)
        pos[i] -= diff * adjustment / 2
        pos[j] += diff * adjustment / 2
        dists = np.sqrt(np.sum((pos[:, None, :] - pos[None, :, :]) ** 2, axis=2))
        pair_cons = dists[mask] - (radii[:, None] + radii[None, :])[mask]
    
    for i in range(n):
        r_bound = [0, min(pos[i, 0], 1 - pos[i, 0], pos[i, 1], 1 - pos[i, 1])]
        r, _ = linprog(lambda r: -np.sum(r), A_ub=[], b_ub=[], bounds=[r_bound] * n)
        radii[i] = -r[0]
    
    return radii

def run_packing():
    n = 26
    bounds = [(0, 1)] * 2 * n
    bounds += [(0, 0.5)] * n
    
    result = differential_evolution(objective, bounds, popsize=50, maxiter=200, seed=42)
    positions = result.x[:2*n]
    radii = solve_radii(positions)
    
    nlp_constraints = []
    for i in range(n):
        nlp_constraints.append(NonlinearConstraint(lambda x: x[i*2]**2 + x[i*2+1]**2 - x[i*2+2**2], 0, float('inf')))
        nlp_constraints.append(NonlinearConstraint(lambda x: (x[i*2] - 1)**2 + x[i*2+1]**2 - x[i*2+2**2], 0, float('inf')))
        nlp_constraints.append(NonlinearConstraint(lambda x: x[i*2]**2 + (x[i*2+1] - 1)**2 - x[i*2+2**2], 0, float('inf')))
        nlp_constraints.append(NonlinearConstraint(lambda x: (x[i*2] - 1)**2 + (x[i*2+1] - 1)**2 - x[i*2+2**2], 0, float('inf')))
    
    for i in range(n):
        for j in range(i + 1, n):
            dist_var = n * 2 + n * 2 + 2 * (i * n + j)
            nlp_constraints.append(NonlinearConstraint(lambda x: x[i*2] - x[j*2] - x[dist_var] + x[i*2+1] - x[j*2+1] - x[dist_var+1] - x[i*2+2] - x[j*2+2], 0, float('inf')))
    
    res = minimize(lambda x: -np.sum(x[n*2:n*2+n]), x0=np.hstack([positions, radii]), method='SLSQP', bounds=bounds, constraints=nlp_constraints, options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
    
    centers = res.x[:n*2].reshape(n, 2)
    values = res.x[n*2:]
    
    if any(values < 0):
        print('Radii are negative')
    
    return centers, values, -np.sum(values)
