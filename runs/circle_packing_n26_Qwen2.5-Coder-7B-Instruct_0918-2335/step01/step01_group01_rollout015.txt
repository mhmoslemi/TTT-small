import numpy as np
from scipy.optimize import minimize

def objective(pos_flat):
    pos = pos_flat.reshape(26, 2)
    r = compute_optimal_radii(pos)
    if r is None:
        return np.inf
    return -np.sum(r)

def compute_optimal_radii(pos):
    n = pos.shape[0]
    r = np.ones(n)
    while True:
        r_opt = np.maximum(r, np.minimum(pos[:, 0], 1 - pos[:, 0]))
        r_opt = np.maximum(r_opt, np.minimum(pos[:, 1], 1 - pos[:, 1]))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((pos[i] - pos[j]) ** 2))
                r_opt[i] = np.minimum(r_opt[i], dist - r[j])
                r_opt[j] = np.minimum(r_opt[j], dist - r[i])
        if np.allclose(r, r_opt):
            return r_opt
        r = 0.5 * (r + r_opt)

def run_packing():
    best_sum = -np.inf
    for _ in range(30):
        init_centres = np.random.rand(26, 2)
        res = minimize(fun=objective, x0=init_centres.flatten(), method='SLSQP',
                       bounds=[(0, 1)] * 52, options={'maxiter': 2000})
        if res.success:
            radii = compute_optimal_radii(res.x.reshape(26, 2))
            if radii is not None and np.sum(radii) > best_sum:
                best_sum = np.sum(radii)
                best_centre = res.x.reshape(26, 2)
                best_rad = radii
    return best_centre, best_rad, best_sum
