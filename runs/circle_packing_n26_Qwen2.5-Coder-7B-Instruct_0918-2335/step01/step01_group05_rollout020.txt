import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def evaluate_positions(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    r = centers[:, 0]  # radius <= x
    b = np.ones(n)
    A_ij = squareform(pdist(centers))
    constraints = [{'type': 'ineq', 'fun': lambda x, i=i, j=j: x[i] + x[j] - A_ij[i, j]}
                   for i in range(n) for j in range(i+1, n)]
    result = linprog(c, A_ub=A_ij, b_ub=2 * r, bounds=(0, 1), method='highs')
    if result.success:
        radii = result.x
        return np.sum(radii), radii
    else:
        raise ValueError("LP solver failed")

def run_packing():
    n = 26
    np.random.seed(42)  # for reproducibility
    patterns = [
        # 5x5 grid
        np.array([[i * 0.2, j * 0.2] for i in range(5) for j in range(5)]).flatten(),
        # Offset corner pattern
        np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]] +
                 [[i * 0.2, j * 0.2] for i in range(5) for j in range(5) if 0 not in [i, j]]).flatten(),
        # Hexagonal packing (approximation)
        np.array([[i * 0.24 + (j % 2) * 0.12, j * np.sqrt(3) * 0.12] for j in range(7) for i in range(7)]).flatten()
    ]

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for pattern in patterns:
        centers = pattern.reshape(26, 2)
        jitter = 0.02
        centers += np.clip(np.random.uniform(-jitter, jitter, centers.shape), 0, 1)
        sum_radii, radii = evaluate_positions(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = radii

    # Refinement with simulated annealing
    sigma = 0.05
    n_iter = 5000
    alpha = 0.9995
    T = 0.1
    current_centers = best_centers.copy()
    current_sum_radii = best_sum
    current_radii = best_radii.copy()

    for _ in range(n_iter):
        i = np.random.randint(n)
        new_centers = current_centers.copy()
        new_centers[i] += np.clip(np.random.normal(0, sigma), -0.5, 0.5)
        new_sum_radii, new_radii = evaluate_positions(new_centers)
        if new_sum_radii > current_sum_radii or np.random.rand() < np.exp((new_sum_radii - current_sum_radii) / T):
            current_centers = new_centers
            current_sum_radii = new_sum_radii
            current_radii = new_radii
            if new_sum_radii > best_sum:
                best_sum = new_sum_radii
                best_centers = new_centers
                best_radii = new_radii
        sigma *= alpha
        T *= alpha

    return best_centers, best_radii, best_sum
