import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog
from scipy.spatial.distance import pdist, squareform

def max_radius_sum_fixed_positions(positions):
    num_circles = len(positions) // 2
    bounds = [(0, 0.5)] * num_circles
    A_ub, b_ub = [], []

    for i in range(num_circles):
        x, y = positions[2*i], positions[2*i+1]
        A_ub.append([x, y, -1, -1, 0, 0, 0])
        b_ub.append(min(x, y, 1-x, 1-y))

    for i in range(num_circles):
        for j in range(i+1, num_circles):
            A_ub.append([-1, -1, -1, -1, 1, 1, 0])
            b_ub.append(np.sqrt((positions[2*i] - positions[2*j]) ** 2 +
                               (positions[2*i+1] - positions[2*j+1]) ** 2))

    res = linprog([-1 for _ in range(num_circles)], A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return -res.fun

def fitness_function(x):
    num_circles = len(x) // 2
    positions = x[:2*num_circles].reshape(-1, 2)
    A_ub, b_ub = [], []

    for i in range(num_circles):
        x, y = positions[i]
        A_ub.append([x, y, -1, -1, 0, 0, 0])
        b_ub.append(min(x, y, 1-x, 1-y))

    for i in range(num_circles):
        for j in range(i+1, num_circles):
            A_ub.append([-1, -1, -1, -1, 1, 1, 0])
            b_ub.append(np.sqrt((positions[i][0] - positions[j][0]) ** 2 +
                               (positions[i][1] - positions[j][1]) ** 2))

    res = linprog([-1 for _ in range(num_circles)], A_ub=A_ub, b_ub=b_ub, bounds=[(0, 0.5)] * num_circles, method='highs')
    return -res.fun

def run_packing():
    num_circles = 26
    bounds = [(0, 1)] * (2 * num_circles)

    sol = differential_evolution(fitness_function, bounds, popsize=50, maxiter=200, tol=1e-12, seed=42)
    positions = sol.x[:2 * num_circles].reshape(-1, 2)
    A_ub, b_ub = [], []

    for i in range(num_circles):
        x, y = positions[i]
        A_ub.append([x, y, -1, -1, 0, 0, 0])

    res = linprog([1 for _ in range(num_circles)], A_ub=A_ub, b_ub=b_ub, bounds=[(0, 0.5)] * num_circles, method='highs')
