from scipy.optimize import minimize, differential_evolution, linprog

def lp_obj(radii, distances):
    return -np.sum(radii)

def lp_constraints(radii, positions, distances):
    cons = []
    for r, pos in zip(radii, positions):
        cons.append({'type': 'ineq', 'fun': lambda x, r=r, pos=pos: r - pos[0]})
        cons.append({'type': 'ineq', 'fun': lambda x, r=r, pos=pos: r - 1 + pos[0]})
        cons.append({'type': 'ineq', 'fun': lambda x, r=r, pos=pos: r - pos[1]})
        cons.append({'type': 'ineq', 'fun': lambda x, r=r, pos=pos: r - 1 + pos[1]})
    for i in range(len(radii)):

            dx, dy = positions[i] - positions[j]
            dist = np.sqrt(dx**2 + dy**2)
            cons.append({'type': 'ineq', 'fun': lambda x, r1=radii[i], r2=radii[j], dist=dist: r1 + r2 - dist})
    return cons
