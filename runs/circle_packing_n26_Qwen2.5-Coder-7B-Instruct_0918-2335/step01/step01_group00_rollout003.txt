import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial import distance_matrix

def lp_value_and_radii(centers):
    n = centers.shape[0]
    w_limits = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    D = distance_matrix(centers, centers)
    A = np.zeros((n, 2*n))
    np.fill_diagonal(A[:n], 1)
    for i in range(n):
        for j in range(i+1, n):
            A[i, j] = -1
            A[j, i] = -1
    b = -D.ravel()
    bounds = np.zeros(2*n)
    bounds[:n] = [0, w_limits]

    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return -np.inf, np.zeros(n)
    return -res.fun, res.x
