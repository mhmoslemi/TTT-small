def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

    A_ub = np.r_[np.eye(n), -np.eye(n)]
    b_ub = np.r_[centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]
    A_eq = np.zeros((n, 2 * n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])

            row[i * 2] = 1
            row[i * 2 + 1] = 1
            row[j * 2] = 1
            row[j * 2 + 1] = 1
            A_eq = np.vstack((A_eq, row))
            b_eq = np.append(b_eq, dist - (radii[i] + radii[j]))
    result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
