import numpy as np
import scipy.optimize as opt

def solve_lp(centres):
    n = len(centres)
    r = np.zeros(n)
    direction = np.full(n, -1)  # Minimize negative of radii to maximize sum
    constraints = []
    
    # Boundary constraints
    for i, center in enumerate(centres):
        x, y = center
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: x - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - x - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: y - r[i]})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - y - r[i]})
    
    # Pairwise constraints
    for i, j in [(i, j) for i in range(n) for j in range(i+1, n)]:
        diff = centres[i] - centres[j]
        dist = np.sqrt(np.sum(diff ** 2))
        constraints.append({'type': 'ineq', 'fun': lambda r, dist=dist, i=i, j=j: r[i] + r[j] - dist})
    
    result = opt.minimize(
        fun=lambda r: direction @ r,
        x0=r,
        constraints=constraints,
        bounds=[(0, 0.5) for _ in range(n)]
    )
    if result.success:
        optimal_radii = result.x
        return optimal_radii, np.sum(optimal_radii)
    else:
        return np.zeros(n), 0.0

def run_packing():
    n = 26
    centres = np.zeros((n, 2))
    
    # Define grid points for initial centers
    grid_points = np.linspace(0.1, 0.9, 5)
    grid_x = np.repeat(grid_points, 5)
    grid_y = np.tile(grid_points, 5)
    grid_x = np.append(grid_x, 0.5)
    grid_y = np.append(grid_y, 0.5)
    shuffle_indices = np.random.permutation(len(grid_x))
    grid_x = grid_x[shuffle_indices]
    grid_y = grid_y[shuffle_indices]
    
    centres = np.column_stack((grid_x, grid_y)).reshape(n, 2)
    
    # Apply small random jitter
    jitter = 0.02
    centres += np.random.uniform(-jitter, jitter, size=(n, 2))
    centres = np.clip(centres, 0, 1)
    
    # Solve LP with jittered centers
    best_centres, best_sum = None, 0.0
    
    for _ in range(5000):
        centres_copy = np.array(centres)
        r, sum_radii = solve_lp(centres_copy)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centres = centres_copy
            jitter = np.clip(jitter * 0.99, 0.001, 0.02)  # Decrease jitter each time
        centres += np.random.uniform(-jitter, jitter, size=(n, 2))
        centres = np.clip(centres, 0, 1)
    
    # Final LP solve with best centers
    final_radii, final_sum = solve_lp(best_centres)
    
    return best_centres, final_radii, final_sum
