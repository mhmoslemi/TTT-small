import numpy as np
from scipy.optimize import linprog, minimize

def radius_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_eq = []
    b_eq = []
    A_ub = []
    b_ub = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub.append([1, 0, -1, 0, -x, y, -x**2, -x*y, -y**2, -x])
        b_ub.append(x)
        A_ub.append([0, 1, 0, -1, -x**2, -y, x*y, -x*y**2, -y**2, y])
        b_ub.append(y)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            A_ub.append([1, 0, 1, 0, -x1, y1, -x1**2, -x1*y1, -y1**2, -x1,
                          -2, 0, 0, 0, -x2, y2, -x2**2, -x2*y2, -y2**2, -x2])
            b_ub.append(-np.sqrt((x1 - x2)**2 + (y1 - y2)**2) + 1e-10 ** 2)
            A_ub.append([0, 1, 0, 1, -x2**2, -y2, x2*y, -x2*y**2, -y2**2, y2,
                          0, -2, 0, 0, -x1, y1, -x1**2, -x1*y1, -y1**2, x1])
            b_ub.append(-np.sqrt((x1 - x2)**2 + (y1 - y2)**2) + 1e-10 ** 2)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
    return res.x

def sa_step(current_centers):
    n = current_centers.shape[0]
    move_size = 0.05 * (1 + (1 - (np.random.rand() * 100)) / 100)
    proposal = current_centers + move_size * (np.random.rand(n, 2) * 2 - 1)
    proposal[:, 0] = np.clip(proposal[:, 0], 0, 1)
    proposal[:, 1] = np.clip(proposal[:, 1], 0, 1)
    return proposal

def run_packing():
    n = 26
    current_centers = (np.random.rand(n, 2) * 0.6 + 0.2)
    best_centers = current_centers
    best_sum_radii = 0
    temperature = 0.1
    num_moves = 50
    num_temps = 300

    for temp_step in range(num_temps):
        for _ in range(num_moves):
            proposal = sa_step(current_centers)
            new_radii = radius_lp(proposal)
            new_sum_radii = np.sum(new_radii)
            old_radii = radius_lp(current_centers)
            old_sum_radii = np.sum(old_radii)

            if new_sum_radii > old_sum_radii or np.random.rand() < np.exp((new_sum_radii - old_sum_radii) / temperature):
                current_centers = proposal
                current_radii = new_radii
                current_sum_radii = new_sum_radii

            if current_sum_radii > best_sum_radii:
                best_centers = current_centers
                best_sum_radii = current_sum_radii

        temperature *= 1 / np.log(1 + temp_step)

    polished_centers = current_centers
    polished_radii = radius_dp(polished_centers)
    polished_sum_radii = np.sum(polished_radii)
    return polished_centers, polished_radii, polished_sum_radii

def radius_dp(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r[i] = min(r[i], np.sqrt((x1 + 10e-7 - x2) ** 2 + (y1 - y2) ** 2) - x1 ** 2)
            r[j] = min(r[j], np.sqrt((x1 - x2) ** 2 + (y1 + 10e-7 - y2) ** 2) - x2 ** 2)
            r[i] = min(r[i], np.sqrt((x1 - x2) ** 2 + (y1 - y2 + 10e-7) ** 2) - x1 ** 2)
            r[j] = min(r[j], np.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2) - x2 ** 2)
    return r

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
