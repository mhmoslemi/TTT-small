import numpy as np
from scipy.optimize import linprog, minimize_slsqp

def lp_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n, dtype=np.float64)  # Minimize -sum(r_i)
    A = np.zeros((n+n+25, n), dtype=np.float64)
    b = np.zeros(n+n+25, dtype=np.float64)
    
    for i in range(n):
        A[i, i] = 1.0
        b[i] = centers[i, 0]  # Left boundary
    
    for i in range(n):
        A[n+i, i] = 1.0
        b[n+i] = 1.0 - centers[i, 0]  # Right boundary
    
    for i in range(n):
        A[2*n+i, i] = 1.0
        b[2*n+i] = centers[i, 1]  # Bottom boundary
    
    for i in range(n):
        A[3*n+i, i] = 1.0
        b[3*n+i] = 1.0 - centers[i, 1]  # Top boundary
    
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i]-centers[j])
            A[2*n+25+(i*n)+j, i] = 1.0
            A[2*n+25+(i*n)+j, j] = 1.0
            b[2*n+25+(i*n)+j] = dist
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)])
    if result.success:
        return result.x
    else:
        raise Exception("LP failed")

def simulated_annealing():
    n = 26
    centers = np.random.rand(n, 2)
    max_sum = -np.inf
    best_centers = None
    T0 = 0.1
    T = T0
    moves_per_T = 50
    num_T = 300
    
    for k in range(num_T):
        for _ in range(moves_per_T):
            i = np.random.randint(n)
            dx, dy = np.random.randn(2)
            new_center = centers[i] + dx * T * 0.05
            new_center = np.clip(new_center, 0, 1)
            
            new_centers = np.copy(centers)
            new_centers[i] = new_center
            
            new_radii = lp_radii(new_centers)
            new_sum = np.sum(new_radii)
            
            if new_sum >= max_sum:
                max_sum = new_sum
                if best_centers is None:
                    best_centers = np.copy(new_centers)
                elif np.sum(new_radii) > np.sum(best_centers):
                    best_centers = np.copy(new_centers)
            else:
                acceptance_prob = min(1, np.exp((new_sum - max_sum) / T))
                if np.random.rand() < acceptance_prob:
                    max_sum = new_sum
                    best_centers = np.copy(new_centers)
            
        T /= np.log(1 + k + 1)
    
    return best_centers, max_sum

def slsqp_optimization(centers, sum_radii):
    n = centers.shape[0]
    r = np.sqrt(sum_radii / n)
    x0 = np.concatenate((centers.flatten(), r * np.ones(n)))
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    constraints = ({'type': 'ineq', 'fun': lambda v: v[2 * n :]},
                   {'type': 'ineq', 'fun': lambda v: 1 - v[2 * n :]},
                   {'type': 'ineq', 'fun': lambda v: v[: 2 * n] - radii})
    
    for i in range(n):
        for j in range(i + 1, n):
            def con_fun(v):
                dist = np.linalg.norm(v[2 * i : 2 * (i + 1)] - v[2 * j : 2 * (j + 1)])
                return dist - v[2 * n + (i * n) + j]
            
            constraints.append({'type': 'ineq', 'fun': con_fun})
    
    result = minimize_slsqp(
        lambda v: -np.sum(v[2 * n :]),
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False}
    )
    
    if result.success:
        best_centers = result.x[: 2 * n].reshape(n, 2)
        best_r = result.x[2 * n :]
        return best_centers, np.sum(best_r)
    else:
        raise Exception("SLSQP failed")

def run_packing():
    best_centers, best_sum = simulated_annealing()
    centers, sum_radii = slsqp_optimization(best_centers, best_sum)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
