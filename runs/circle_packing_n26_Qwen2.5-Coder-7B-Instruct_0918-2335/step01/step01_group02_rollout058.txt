```python
import numpy as np
from scipy.optimize import linprog, minimize

# Define the target number of circles and allowed trials
num_circles = 26
num_trials = 50

# Define the bounds for the radii (0 to 0.5)
bounds = [(0, 0.5)] * num_circles

# Function to generate a contact graph matrix for a given pattern (5x5 grid with one extra circle)
def create_contact_pattern(pattern):
    G = np.zeros((num_circles, num_circles), dtype=int)
    for i, j in pattern:
        G[i, j] = G[j, i] = 1
    return G

# Define theOBJset of 10 promising contact graphs
contact_patterns = [
    # 5x5 grid with one extra circle at the center
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (1, 2), (2, 1), (0, 2), (2, 0), (2, 2), (2, 3), (3, 2), (0, 3), (3, 0), (0, 4), (4, 0), (1, 3), (3, 1), (3, 3), (3, 4), (4, 3), (4, 0), (4, 4)]),
    # 2x2 hexagonally packed clusters with one extra circle in the center
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (1, 2), (2, 1), (0, 2), (2, 0), (2, 2), (0, 3), (3, 0), (0, 4), (4, 0), (1, 3), (3, 1), (3, 3), (3, 4), (4, 3), (4, 0), (4, 4)]),
    # Row-shifted grids
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (2, 1), (1, 2), (0, 2), (2, 0), (0, 3), (1, 3), (2, 3), (0, 4), (1, 4), (2, 4)]),
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (1, 2), (2, 1), (0, 2), (2, 0), (2, 2), (3, 1), (1, 3), (0, 4), (1, 4), (2, 4), (3, 4)]),
    # Randomised perturbations of the above patterns
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (1, 2), (2, 1), (0, 2), (2, 0), (2, 2), (2, 3), (3, 2), (3, 1), (3, 0), (0, 4), (4, 0), (4, 2), (4, 4)]),
    create_contact_pattern([(0, 1), (1, 0), (1, 1), (1, 2), (2, 1), (0, 2), (2, 0), (2, 2), (3, 1), (1, 3), (0, 4), (1, 4), (2, 4)]),
]

# Function to create linear inequalities and objective function coefficients for a given graph
def constructLP(G, wall_contacts):
    n = G.shape[0]
    A = []
    b = []
    c = -np.ones(n)  # We aim to minimize the negative sum of radii

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            if G[i, j] == 0:
                A.append([1, 1])
                b.append(wall_contacts[i] + wall_contacts[j])
            else:
                A.append([-1, 1])
                b.append(wall_contacts[i])
                A.append([1, -1])
                b.append(wall_contacts[j])

    # Boundary constraints for non-boundary vertices
    for i in range(n):
        if not wall_contacts[i]:
            A.append([0, 0, 1])
            b.append(1)

    return A, b, c

# Function to compute optimal radii and positions for a given graph
def optimizeGraph(G, wall_contacts):
    A, b, c = constructLP(G, wall_contacts)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    assert res.success, "Adjust wall_contacts to render the graph feasible"
    return res.x, -res.fun  # Negate again to convert back to sum of radii

# Main function to find the best packing
def run_packing():
    best_centers = None
    best_radians = None
    best_sum = 0

    for pattern in contact_patterns:
        # Initialize wall contacts but ensure all vertices are in contact
        wall_contacts = np.ones(num_circles) * 0.25  # Fits 5x5 grid well
        wall_contacts[pattern[0]] = 0.0  # Replace one with 0 for centered extra circle

        radii, sum_radii = optimizeGraph(pattern, wall_contacts)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = np.array(pattern) * 0.25  # Convert row-column indices to coordinates
            best_radians = radii

    # Refinement step using SLSQP
    initial_guess = np.concatenate([best_centers.flatten(), best_radians])
    cons = [{'type': 'ineq', 'fun': lambda x: x[num_circles:] + (np.sqrt(np.sum((x[:num_circles].reshape(-1, 2)[:, :, None] - x[num_circles:].reshape(-1, 2, 1))**2, axis=2)) - (2 * best_radians[..., None]))},
           {'type': 'ineq', 'fun': lambda x: 1 - x[num_circles:]}]
    for i in range(num_circles):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: -x[i],
                      'args': (i,)})
        if best_centers[i, 0] != 0 and best_centers[i, 0] + best_radians[i] < 1:
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: -best_centers[i, 0],
                          'args': (i,)})
        if best_centers[i, 1] != 0 and best_centers[i, 1] + best_radians[i] < 1:
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: -best_centers[i, 1],
                          'args': (i,)})

    res = minimize(fun=lambda x: -np.sum(x[num_circles:]), x0=initial_guess, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9})
    if res.success:
        return res.x[:num_circles].reshape(-1, 2), res.x[num_circles:], -res.fun
    else:
        return best_centers, best_radians, best_sum

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
```
