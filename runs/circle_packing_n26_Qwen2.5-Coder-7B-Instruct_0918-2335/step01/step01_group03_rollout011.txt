import numpy as np
from scipy.optimize import linprog, minimize as SLSQP

# Constants
N = 26
delta_max = 0.05
T0 = 1.0
alpha = 0.995
NITER = 50000

# Initial configuration: 5x5 grid and one center
centers_grid = np.array([
    [(0.2 + i * 0.2, 0.2 + j * 0.2) for j in range(5)] for i in range(5)]).reshape(-1, 2)
centers_center = np.array([[0.5, 0.5]])
centers_initial = np.vstack([centers_grid, centers_center]).reshape(-1, 2)
radii_initial = np.array([0.25] * 26)

# LP to find optimal radii for given centers
def optimise_radii(centers):
    n = centers.shape[0]
    c = np.ones(n)
    A = np.zeros((n + n, n))
    b = np.zeros(n + n)
    A[:n, :n] = -np.eye(n)
    A[n:, :n] = np.eye(n)
    A[n:, n:] = -(centers[:, None, :] - centers[None, :, :]).sum(axis=2)
    b[:n] = -np.max(centers, axis=1)
    b[n:] = np.sqrt(( centers[:, None, :] - centers[None, :, :] )**2).sum(axis=2) / 2
    result = linprog(c, A_ub=A, b_ub=b, bounds=np.array([[-np.inf, np.inf]] * n + [[0, 0.5]] * n))
    return result.x if result.success else None

# Acceptance probability for SA
def accept_probability(delta_S, T):
    return np.exp(delta_S / T) if delta_S > 0 else 1.0

# Simulated Annealing loop
current_centres = centers_initial.copy()
current_radii = radii_initial.copy()
best_sum = np.sum(current_radii)
best_centres = current_centres.copy()
best_radii = current_radii.copy()

for iter in range(NITER):
    i = np.random.randint(0, N)
    dx, dy = np.random.uniform(-delta_max, delta_max, 2)
    new_coords = current_centres[i] + (dx, dy)
    if not (0 <= new_coords[0] <= 1 and 0 <= new_coords[1] <= 1):
        continue

    temp_centres = current_centres.copy()
    temp_centres[i] = new_coords
    temp_radii = optimise_radii(temp_centres)
    if temp_radii is None:
        continue

    new_sum = np.sum(temp_radii)
    ΔS = new_sum - best_sum
    T = T0 * alpha**iter
    if ΔS > 0:
        current_centres, current_radii, best_sum = temp_centres, temp_radii, new_sum
        if new_sum > best_sum:
            best_centres, best_radii = temp_centres, temp_radii
    else:
        if np.random.uniform() < accept_probability(ΔS, T):
            current_centres, current_radii = temp_centres, temp_radii

# Final refinement using SLSQP
solution = SLSQP(run_obj, run_constraints, bounds=run_bounds, options={'disp': False})
best_centres, best_radii = solution.x[:N], solution.x[N:]
best_sum = np.sum(best_radii)

def run_obj(x):
    return - np.sum(x[52:])

def run_constraints(x):
    centers = x[:52].reshape(26, 2)
    radii = x[52:]
    cons = np.empty(4 * 26, float)
    cons[:26] = centers[:, 0] - radii
    cons[26:52] = 1 - centers[:, 0] - radii
    cons[52:78] = centers[:, 1] - radii
    cons[78:104] = 1 - centers[:, 1] - radii
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (radii[:, None] + radii[None, :])[mask]
    return np.concatenate([cons, pair_cons])
