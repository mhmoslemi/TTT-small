import numpy as np
from scipy.optimize import linprog, minimize

def build_contact_matrix(n):
    contacts = []
    for i in range(n):
        for j in range(i+1, n):
            contacts.append((i, j))
    return contacts

def build_constraint_matrix(graph, n):
    A_ub = []
    b_ub = []
    for i, j in graph:
        A_ub.append([-1, 0, -1, 0])
        A_ub.append([0, -1, 0, -1])
        b_ub.append(0)
        b_ub.append(0)
    
    wall_vertices = {0, 1, 2, 3, 6, 7, 8, 11, 12, 13, 17, 18, 19, 22, 23, 24}
    for i in range(n):
        if i not in wall_vertices:
            A_ub.append([-1, 0, 0, 0])
            A_ub.append([0, -1, 0, 0])
            b_ub.append(0.5)
            b_ub.append(0.5)
    
    return np.array(A_ub), np.array(b_ub)
