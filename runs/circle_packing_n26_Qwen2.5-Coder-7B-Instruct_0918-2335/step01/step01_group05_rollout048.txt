import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    popsize = 15
    maxiter = 500
    penalty = 1e6

    result_de = differential_evolution(fun=obj, bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'popsize': popsize, 'maxiter': maxiter})

    if not result_de.success and result_de.message != 'Maximize failed':
        return np.zeros((n, 2)), np.zeros(n), 0.0

    best_DE = result_de.x

    def obj_wrapped(v):
        return -np.sum(v[52:])

    def constraints_wrapped(v):
        pos = v[:52].reshape(26, 2)
        r = v[52:]
        cons = np.empty(104, float)

        return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

    result_slsqp = minimize(fun=obj_wrapped, x0=best_DE, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints_wrapped}, options={'maxiter': 1000, 'ftol': 1e-9})

    if not result_slsqp.success:
