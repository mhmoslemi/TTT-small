import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_escape(positions, max_radius=1.0):
    n = len(positions)
    r = np.zeros(n)
    
    A_ub = np.zeros((2*n + n*(n-1)//2, 2*n))
    b_ub = np.zeros(2*n + n*(n-1)//2)
    
    # Constraints for feasibility within the square [0, 1]^2
    for i in range(n):
        A_ub[i, 2*i    ] = 1
        A_ub[i, 2*i + 1] = 1
        b_ub[i] = max_radius
    
    # Constraints for non-overlap
    index = 2*n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[index, 2*i    ] = 1
            A_ub[index, 2*i + 1] = 1
            A_ub[index, 2*j    ] = 1
            A_ub[index, 2*j + 1] = 1
            A_ub[index, 2*n + 2*i    ] = -1
            A_ub[index, 2*n + 2*i + 1] = -1
            b_ub[index] = positions[j] - positions[i]
            index += 1
    
    res = linprog(-r.sum(), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return -res.fun, res.x[:n]

def fitness_function(positions):
    n = len(positions) // 2
    distances = np.zeros(n*(n-1)//2)
    index = 0
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((positions[2*i] - positions[2*j])**2 + (positions[2*i + 1] - positions[2*j + 1])**2)
            distances[index] = dist
            index += 1
    
    max_radiuses = np.zeros(n)
    for i in range(n):
        max_radiuses[i] = minDist(positions[i], positions[:i] + positions[i+1:])
    
    opt_r, max_r = lp_escape(positions[:n])
    return opt_r

def run_packing():

    bounds = [(-1, 1)] * n + [(-1, 0.5)] * n
    
    result = differential_evolution(fitness_function, bounds, popsize=50, maxiter=200, seed=42)
    
    init_positions = result.x[:n]
    lp_r = lp_escape(init_positions)[1]

    bounds += [(0, 0.5)] * n

    final_positions = result_slsqp['x'][:n]
    final_r = result_slsqp['x'][n:]
    
    sum_radii = sum(final_r)
    
    return final_positions.reshape(n, 2), final_r, sum_radii

def minDist(p, others):
    return min(np.sqrt((p[0] - o[0])**2 + (p[1] - o[1])**2) - p[2] - o[2] for o in others)
