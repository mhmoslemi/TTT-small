import numpy as np
from scipy.optimize import linprog, minimize

delta_max = 0.05
NITER = 50000
T0 = 1.0
alpha = 0.995
TOL = 1e-9

def random_step(max_delta):
    return max_delta * (2 * np.random.rand(2) - 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n)  # x, y coordinates for each circle
    r_bounds = [(0, 0.5)] * n  # radius bounds for each circle (practical maximum in unit square)
    initial = np.zeros(3 * n)
    initial[::3] = 0.25 + np.arange(n) * 0.125  # 5x5 grid base
    initial[1::3] = 0.25 + np.arange(n) * 0.125
    initial[52:] = 0.1  # initial guess for radii

    def slsqp_refine(x, ftol=1e-9):
        def obj(v):
            return -np.sum(v[52:])
        res = minimize(fun=obj, x0=v, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 2000, 'ftol': ftol})
        return res.x

    def constraints(v):
        pos = v[:52].reshape(n, 2)
        r = v[52:]
        cons = np.empty(100, float)
        for i in range(n):
            cons[4 * i] = pos[i, 0] - r[i]
            cons[4 * i + 1] = 1 - pos[i, 0] - r[i]
            cons[4 * i + 2] = pos[i, 1] - r[i]
            cons[4 * i + 3] = 1 - pos[i, 1] - r[i]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((pos[i] - pos[j]) ** 2))
                cons[100 + 2 * (i * n + j)] = dist - r[i] - r[j]
                cons[100 + 2 * (i * n + j) + 1] = -(dist - r[i] - r[j])
        return cons

    def optimise_radii(c):
        r = np.minimum(0.5, c[52:])
        m = linprog([0] * n, bounds=r_bounds, A_eq=c.reshape(1, 2 * n), b_eq=np.zeros(1), bounds=(0, None))
        if not m.success:
            return None
        return m.x

    for iter in range(NITER):
        i = np.random.randint(n)
        new_coors = initial[3*i:3*(i+1)] + random_step(delta_max)
        if np.any(new_coors < 0) or np.any(new_coors > 1):
            continue
        new_c = initial.copy()
        new_c[3*i:3*(i+1)] = new_coors
        new_r = optimise_radii(new_c)
        if new_r is None:
            continue
        new_sum = np.sum(new_r)
        if new_sum > np.sum(initial[52:]):
            initial = new_c
            initial[52:] = new_r
        else:
            T = T0 * alpha ** iter
            if np.random.rand() < np.exp((new_sum - np.sum(initial[52:])) / T):
                initial = new_c
                initial[52:] = new_r

    initial = slsqp_refine(initial)
    return initial[:52].reshape(-1, 2), initial[52:], np.sum(initial[52:])
