import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def evaluate_positions(centers):
    n = centers.shape[0]
    pairs = list(pdist(centers))
    c = -np.ones(n)
    A_ub = np.vstack([
        np.column_stack((np.eye(n), -np.eye(n), np.eye(n), -np.eye(n))),
        np.column_stack((-np.eye(n), np.eye(n), np.eye(n), np.full((n, n), -1 * (centers[:, 0] + centers[:, 1]), -1, 0))),
        np.column_stack((-np.eye(n), np.eye(n), np.full((n, n), -1 * (centers[:, 0] + 1 - centers[:, 1]), -1, 0))),
        np.column_stack((-np.eye(n), np.full((n, n), -1 * (centers[:, 0] + centers[:, 1]), -1, 0), np.eye(n), -np.eye(n))),
        np.column_stack((-np.eye(n), np.full((n, n), -1 * (centers[:, 0] + 1 - centers[:, 1]), -1, 0), np.eye(n), -np.eye(n))),
        np.column_stack((np.eye(n), np.full((n, n), -1 * (centers[:, 0] + centers[:, 1]), -1, 0), -np.eye(n), np.eye(n)))
    ])
    b_ub = np.hstack([np.clip(centers[:, 0], 0, 1), np.clip(1 - centers[:, 0], 0, 1), np.clip(centers[:, 1], 0, 1), np.clip(1 - centers[:, 1], 0, 1)])
    r = []
    dists = np.full(n * (n - 1) // 2, -1)
    for i, (ci, cj) in enumerate(pairs):
        dists[i] = centers[ci, 0] - centers[cj, 0]
        r.append(dists[i] - centers[ci, 1] - centers[cj, 1])
    A_ub = np.vstack((A_ub, dists[:, None] + r[:, None]))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if not res.success:
        raise Exception(f"LP solver failed: {res.message}")
    return -res.fun, res.x[:n]

def generate_candidates(n=26):
    patterns = []
    # 5x5 grid
    grid = np.array([[i * 0.2, j * 0.2] for i in range(5) for j in range(5)])
    centers = np.concatenate([grid, [[0.5, 0.5]]])
    np.random.shuffle(centers)
    patterns.append(centers[:n])
    
    # Hexagonal pattern
    h = 0.18
    v = np.sqrt(3)/2 * h
    rows = int(1 / (h/2))
    centers = np.zeros((rows * 2 - 1, 2))
    coeff_x = np.zeros((rows * 2 - 1, rows * 2 - 1))
    coeff_y = np.zeros((rows * 2 - 1, rows * 2 - 1))
    count = 0
    for i in range(rows * 2 - 1):
        for j in range(rows * 2 - 1):
            if i % 2 == 0:
                coeff_x[i, j] = 0
                coeff_y[i, j] = j * h
            else:
                coeff_x[i, j] = (j - (j + 1) % 2) * h
                coeff_y[i, j] = (i + j) * v / 2
            if coeff_x[i, j] * coeff_x[i, j] + coeff_y[i, j] * coeff_y[i, j] <= 1:
                centers[count] = [coeff_x[i, j], coeff_y[i, j]]
                count += 1
    
    // Truncate to exactly 26 points and jitter
    centers = centers[:n]
    jitter = 0.02
    jittered_centers = centers + jitter * np.random.uniform(-1, 1, (n, 2))
    jittered_centers = np.clip(jittered_centers, 0, 1)
    patterns.append(jittered_centers)
    
    # Offset corner pattern
    corner_pattern = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]])
    grid = np.array([[i * 0.2, j * 0.2] for i in range(5) for j in range(5)])
    centers = np.concatenate([corner_pattern, grid[1:, 1:]])
    np.random.shuffle(centers)
    patterns.append(centers[:n])
    
    # Jitter and select best from patterns
    best_sum = -np.inf
    best_centers = None
    for centers in patterns:
        jittered_centers = centers + jitter * np.random.uniform(-1, 1, (n, 2))
        jittered_centers = np.clip(jittered_centers, 0, 1)
        sum_radii, _ = evaluate_positions(jittered_centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = jittered_centers
    return best_centers

def simulated_annealing(centers, n_iter=5000, sigma_0=0.05, alpha=0.9995, T=0.1, jitter=0.02):

    best_centers = centers
    best_sum, _ = evaluate_positions(centers)
    sigma = sigma_0
    
    for i in range(n_iter):
        # Pick random circle index
        i = np.random.randint(n)

        center_i = centers[i, 0], centers[i, 1]

        new_centers = centers.copy()
        new_centers[i] = new_center
        
        # Compute new sum and radii
        sum_radii, _ = evaluate_positions(new_centers)
        
        # Decision rule
        if sum_radii >= best_sum or np.log(np.random.rand() < np.exp((sum_radii - best_sum) / T)):

            best_sum = sum_radii
        
        # Acceptance temperature decay
        sigma *= alpha
    
    return centers, best_sum

def run_packing():
    centers = generate_candidates()
    # Initial random jitter
    jitter = 0.02
    jittered_centers = centers + jitter * np.random.uniform(-1, 1, centers.shape)
    jittered_centers = np.clip(jittered_centers, 0, 1)
    
    # Simulated Annealing
    final_centers, best_sum = simulated_annealing(jittered_centers)
    
    # Recompute radii and sum for the best found centers
    final_centers, _ = evaluate_positions(final_centers)
    
    return final_centers, _[1], best_sum
