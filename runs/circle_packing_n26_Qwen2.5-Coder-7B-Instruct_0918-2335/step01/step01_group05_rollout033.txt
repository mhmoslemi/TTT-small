import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = 0.5 * np.random.rand(n)

    res_de = differential_evolution(fun=objpen, bounds=bounds, ngen=500, popsize=15,
                                   strategy='best2bin', atol=1e-10)
    if not res_de.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    res_slsqp = minimize(fun=obj, x0=res_de.x, method='SLSQP',
                         bounds=bounds, constraints={'type': 'ineq', 'fun': constraints},
                         options={'maxiter': 4000, 'ftol': 1e-9})
    if not res_slsqp.success or not constraints(res_slsqp.x).min() >= -1e-10:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    centers = res_slsqp.x[:52].reshape(n, 2)
    radii = res_slsqp.x[52:]
    return centers, radii, -obj(res_slsqp.x)

def objpen(v):
    r = v[52:]
    cons = constraints(v)
    penalty = np.sum(np.maximum(cons, 0) ** 2) * 1e6
    return -np.sum(r) + penalty
