import numpy as np
from scipy.optimize import differential_evolution, LinprogResult, minimize

def lp_radii(positions):
    n = 26
    r = np.empty(n, float)
    A = []
    b = []
    
    for i, (x, y) in enumerate(positions):
        A.append([x, y])
        b.append(x)
        A.append([x, -y])
        b.append(x)
        A.append([-x, y])
        b.append(x)
        A.append([-x, -y])
        b.append(x)
    
    for i, (x, y) in enumerate(positions):
        A.append([y, x])
        b.append(y)
        A.append([y, -x])
        b.append(y)
        A.append([-y, x])
        b.append(y)
        A.append([-y, -x])
        b.append(y)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = positions[i][0] - positions[j][0]
            dy = positions[i][1] - positions[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            A.append([0, 0])
            b.append(dist - (positions[i][2] + positions[j][2]))
            
    res = LinprogResult(A_ub=np.array(A), b_ub=np.array(b), message='Optimal')
    return -res.fun

    positions = x.reshape(26, 2)
    radii = lp_radii(positions)
    return -radii
