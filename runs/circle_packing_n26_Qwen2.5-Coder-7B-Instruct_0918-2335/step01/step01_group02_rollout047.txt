def obj_DE(positions):
    r = positions[52:]
    return -np.sum(r)

def constraints_LP(v):

    cons[0:26] = pos[:, 0] - r

    cons[52:78] = pos[:, 1] - r

    return np.concatenate([cons, pair_cons])
