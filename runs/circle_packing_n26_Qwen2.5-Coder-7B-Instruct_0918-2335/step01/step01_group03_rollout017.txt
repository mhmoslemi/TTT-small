import numpy as np
import networkx as nx
from itertools import combinations
from scipy.optimize import linprog, minimize

N_CIRCLES = 26
RANGE = 0.45
NODE_DISTR = "hexagon"

def generate_positions(pattern, n=N_CIRCLES, r=RANGE):
    if pattern == "hexagon":
        G = nx.hexagonal_lattice_graph(int(np.sqrt(n) + 1))
        pos = nx.spring_layout(G)
        if NODE_DISTR == "weights":
            weights = np.random.rand(len(G.nodes)) * 0.5 + 0.25
            pos = {node: [pos[node][0]*rad, pos[node][1]*rad] for node, rad in zip(G.nodes, weights)}
        else:
            pos = {node: [pos[node][0], pos[node][1]] for node in G.nodes}
        centers = np.array([pos[node] for node in sorted(pos)])
        radii = np.mean([pos[node] for node in sorted(pos)], axis=0) * r
        return centers, radii
    elif pattern == "grid":
        grid_size = int(np.ceil(np.sqrt(n)))
        centers = np.mgrid[0:1:grid_size*1j, 0:1:grid_size*1j].reshape(2,-1).T[:,None]
        radii = r * np.ones_like(centers[:,0])
        return centers[:N_CIRCLES], radii[:N_CIRCLES]
    elif pattern == "hybrid":
        corners = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
        edge_centers = np.array([[0.5, 0], [0.5, 1], [0, 0.5], [1, 0.5]])
        grid_size = 4
        grid_scale = range(1, grid_size + 1)
        sub_grid = np.array(list(product(grid_scale, repeat=2))) / grid_size
        sub_grid[:, 0] *= 1 / grid_size * 0.95
        sub_grid[:, 1] *= 1 / grid_size * 0.95
        centers = np.vstack([corners, edge_centers, [0.2, 0.5], [0.5, 0.2], np.hstack([sub_grid, np.array([[0, 1], [1, 0]]) / 2]))])
        radii = np.array([0.2] * 4 + [0.15] * 4 + [0.1] * 18)
        return centers[:N_CIRCLES], radii[:N_CIRCLES]
    elif pattern == "random":
        centers = np.random.rand(N_CIRCLES, 2)
        radii = r * np.ones(N_CIRCLES)
        return centers, radii
    else:
        raise ValueError(f"Unknown pattern: {pattern}")

def apply_perturbation(centers, r=RANGE, sigma=0.02):
    centers += np.random.randn(N_CIRCLES, 2) * sigma
    centers = np.clip(centers, 0, 1)
    radii = r * np.ones(N_CIRCLES)
    return centers, radii

def lp_solve_radii(centers):
    distances = np.array([[np.linalg.norm(centers[i] - centers[j]) - 2*r for j in range(1, N_CIRCLES)] for i in range(N_CIRCLES)])
    A = np.vstack((centers[:, 0], centers[:, 1], 1-centers[:, 0], 1-centers[:, 1], -distances))
    b = np.zeros(4*N_CIRCLES + (N_CIRCLES**2 - N_CIRCLES) // 2)
    C = -np.ones(N_CIRCLES)
    res = linprog(C, A_ub=A, b_ub=b, bounds=(0, None))
    if res.success:
        return res.x
    else:
        return None

def compute_energy(centers, radii):
    overlaps = 0
    for i, j in combinations(range(N_CIRCLES), 2):
        dist = np.linalg.norm(centers[i] - centers[j])
        if dist < radii[i] + radii[j]:
            overlaps += (radii[i] + radii[j] - dist)
    edge_penalties = np.clip(radii - centers[:, 0], 0, None).sum() + np.clip(radii - (1 - centers[:, 0]), 0, None).sum()
    edge_penalties += np.clip(radii - centers[:, 1], 0, None).sum() + np.clip(radii - (1 - centers[:, 1]), 0, None).sum()
    return -radii.sum() + 1e6 * (overlaps + edge_penalties)

def sa_solve(centers, radii, max_iter=10000, init_temp=1e-1):
    T = init_temp
    while T > 1e-8:
        for _ in range(max_iter):
            i = np.random.randint(N_CIRCLES)
            center_plus_radius_change = centers[i] + np.random.randn(2) * 0.01
            center_plus_radius_change = np.clip(center_plus_radius_change, 0, 1)
            radius_change = np.random.uniform(-0.01, 0.01)
            radius_plus_change = np.clip(radii[i] + radius_change, 0, 0.5)
            new_energy = compute_energy(np.vstack((centers[:i], center_plus_radius_change, centers[i+1:])), np.hstack((radii[:i], radius_plus_change, radii[i+1:])))
            current_energy = compute_energy(centers, radii)
            delta_energy = new_energy - current_energy
            if delta_energy < 0 or np.random.rand() < np.exp(delta_energy / T):
                centers[i] = center_plus_radius_change
                radii[i] = radius_plus_change
        T *= 0.995
    return centers, radii
