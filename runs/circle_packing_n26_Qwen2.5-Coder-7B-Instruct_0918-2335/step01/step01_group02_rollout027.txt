import numpy as np
import itertools
from scipy.optimize import linprog, minimization

def create_contact_graphs():
    graphs = []
    
    # Rectangular lattice
    centers = np.array([[i/4, j/4] for i in range(5) for j in range(5)])
    graph = centers[:-1].tolist() + [(4, j/4) for j in range(5)] + [(i/4, 4) for i in range(5)]
    graph = list(set((min(a, b), max(a, b)) for a, b in graph + [(c, d) for c, d in itertools.combinations(graph, 2)]))
    graphs.append(np.array(graph))
    
    # Lattice patches
    for i, j in [(0, 0), (0, 1), (1, 0)]:
        patch = np.array([[i + x/2, j + y/2] for x, y in itertools.product([-1, 0, 1], repeat=2) if (x, y) != (0, 0)])
        centers = patch[:-1].tolist() + [(patch[-1, 0], j + 1), (i + 1, patch[-1, 1])]
        graph = list(set((min(a, b), max(a, b)) for a, b in centers + [(c, d) for c, d in itertools.combinations(graph, 2)]))
        graphs.append(np.array(graph))
    
    # Row-shifted grids
    for shift in [0.25, -0.25]:
        for i in range(5):
            shift_row = np.array([[5, i] if j == 0 else [i, j] for j in range(5)] + [[5, i + shift]])
            graph = list(set((min(a, b), max(a, b)) for a, b in shift_row + [(c, d) for c, d in itertools.combinations(graph, 2)]))
            graphs.append(np.array(graph))
    
    return graphs

def build_constraints_coo(graph, radius_var, center_dict):
    A = []
    b = []
    I, J, V = [], [], []
    
    # Distance constraints
    for (i, j), (x_i, x_j) in graph:
        row, col = radius_var[i], radius_var[j]
        A.append([1, 1, -1])
        b.append( center_dict[(i, j)][0] - center_dict[(i, j)][1] )
        I.append(row)
        J.append(col)
        V.append(1)
        I.append(col)
        J.append(row)
        V.append(-1)
    
    # Non-overlap constraints
    for (i, j), (x_i, x_j) in graph:
        row, col = radius_var[i], radius_var[j]
        A.append([1, 1, -1])
        b.append(center_dict[(i, j)][0] - center_dict[(i, j)][1] + 1e-12)
        I.append(row)
        J.append(col)
        V.append(-1)
        I.append(row)
        J.append(col)
        V.append(1)
    
    return np.array(I), np.array(J), np.array(V), np.array(b)

def run_packing():
    graphs = create_contact_graphs()
    best_solution = None
    best_sum_radii = 0
    
    for graph in graphs:
        radius_var = np.arange(26)
        center_dict = {}
        
        for i, j in graph:
            center_dict[(i, j)] = (np.sqrt((i + j) / 2) - 0.5, np.sqrt((i + j) / 2) + 0.5)
        
        I, J, V, b = build_constraints_coo(graph, radius_var, center_dict)
        c = -np.ones(26)
        
        for k in range(10):
            res = linprog(c, A_ub=I.reshape((len(I),)), b_ub=b, method='highs', bounds=(0, 1))
            if res.success and sum(res.x) > best_sum_radii:
                best_sum_radii = sum(res.x)
                best_solution = (graph, res.x)
    
    if best_solution:
        graph, radii = best_solution
        return np.repeat(np.arange(5), 5).reshape(-1, 2), radii, best_sum_radii

centers, radii, sum_radii = run_packing()
np.savetxt('centers.txt', centers)
np.savetxt('radii.txt', radii)
print(f"Total Radius Sum: {sum_radii}")
