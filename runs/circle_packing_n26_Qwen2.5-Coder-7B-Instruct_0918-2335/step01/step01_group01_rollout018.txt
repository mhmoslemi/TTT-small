```python
import numpy as np
from scipy.optimize import minimize

def compute_optimal_radii(pos):
    r = np.sqrt-pos) / 4
    return r

def objective(pos_flat):
    radii = compute_optimal_radii(pos_flat)
    if np.any(radii <= 0):
        return np.inf
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0,1)]*52
    init_centres = np.random.rand(52)
    res = minimize(objective, init_centres, method='SLSQP', bounds=bounds, options={'maxiter':2000})
    if res.success:
        radii = compute_optimal_radii(res.x)
        return res.x.reshape(26,2), radii, np.sum(radii)
    return np.zeros((26,2)), np.zeros(26), 0.0
```
