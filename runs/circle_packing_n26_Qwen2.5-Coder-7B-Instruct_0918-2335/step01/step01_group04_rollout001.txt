import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers) -> np.ndarray:
    n = centers.shape[0]
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    return dists

def solve_radii_lp(centers) -> np.ndarray:
    n = centers.shape[0]
    r_min = np.maximum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1 - centers[:, 1]))
    c = -np.ones(n)
    A = np.vstack([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]).T
    b = np.concatenate([r_min, r_min])
    A_eq, b_eq = None, None
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return -res.x

def sa_optimize(initial_centers) -> Tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    best_centers = initial_centers.copy()
    best_radii = solve_radii_lp(initial_centers)
    max_iter = 500
    temp = 1.0
    alpha = 0.999
    best_sum = np.sum(best_radii)
    
    for _ in range(max_iter):
        i = np.random.randint(n)
        dx, dy = np.random.normal(size=2) * temp
        new_center = initial_centers[i] + (dx, dy)
        if 0 <= new_center[0] <= 1 and 0 <= new_center[1] <= 1:
            initial_centers[i] = new_center
            new_radii = solve_radii_lp(initial_centers)
            new_sum = np.sum(new_radii)
            if new_sum - best_sum >= 1e-8 or np.random.rand() < np.exp((new_sum - best_sum) / temp):
                best_centers = initial_centers.copy()
                best_radii = new_radii.copy()
                best_sum = new_sum
        temp *= alpha
    
    return best_centers, best_radii

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.02)
    initial_guess = np.concatenate((initial_centers.flatten(), initial_radii))
    
    result = []
    for _ in range(4):
        centers, _ = sa_optimize(np.random.rand(n, 2))
        result.append(centers)
    
    best_centers = best_radii = None
    best_sum = 0.0
    
    for centers in result:
        radii = solve_radii_lp(centers)
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_centers = centers
            best_radii = radii
            best_sum = current_sum
    
    optim_result = minimize(
        fun=lambda x: -np.sum(x[52:]),
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=[(0, 1)] * n * 2 + [(0, np.inf)] * n,
        constraints={'type': 'eq', 'fun': lambda x: x[52:] - x[:-52] - pairwise_dists(x.reshape(n, 2))[np.triu_indices(n, 1)]},
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )

    if not optim_result.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    best_centers = optim_result.x[:52].reshape(n, 2)
    best_radii = optim_result.x[52:]
    sum_radii = np.sum(best_radii)

    return best_centers, best_radii, sum_radii
