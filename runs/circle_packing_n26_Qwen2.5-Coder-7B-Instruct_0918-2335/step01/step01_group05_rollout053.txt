def solve_radii(centres):

        A_ub[i, i] = 1
        A_ub[n+i, i] = 1
        A_ub[2*n+i, i] = 1
        A_ub[3*n+i, i] = 1
        b_ub[i] = 1 - centres[i, 0]
        b_ub[n+i] = centres[i, 0]
        b_ub[2*n+i] = 1 - centres[i, 1]
        b_ub[3*n+i] = centres[i, 1]
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    for i in range(n):
        for j in range(i+1, n):
            diff = centres[i] - centres[j]
            dist = np.sqrt(np.sum(diff**2))
            A_eq[i*(n-1)+j-i, j] = 1
            A_eq[i*(n-1)+j-i, i] = 1
            b_eq[i*(n-1)+j-i] = dist
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs', bounds=[(0, 0.5)]*n)
    if not result.success:
        raise ValueError("LP solve failed")
    return -result.fun, result.x
