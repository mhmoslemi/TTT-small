import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(positions):
    A_ub = []
    b_ub = []
    for i in range(26):
        A_ub.append([1] * 2 + [0] * 50)
        b_ub.append(1)
        A_ub.append([-1] * 2 + [0] * 50)
        b_ub.append(-1 * positions[i, 0])
        A_ub.append([0] * 2 + [1] * 2 + [0] * 48)
        b_ub.append(1)
        A_ub.append([0] * 2 + [-1] * 2 + [0] * 48)
        b_ub.append(-1 * positions[i, 1])
    for i in range(26):
        for j in range(i + 1, 26):
            distance = np.linalg.norm(positions[i] - positions[j])
            A_ub.append([1, 1, -1] + [0] * (50 - 3))
            b_ub.append(distance - (positions[i, 2] + positions[j, 2]))
    c = -np.ones(26)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * 26, method='highs')
    return result.x, result.fun * -1

def crossover(parent1, parent2, rate):
    child = [parent1[i] if np.random.rand() > rate else parent2[i] for i in range(52)]
    return child

def mutate(individual, rate, sigma):
    individual = [ind + np.random.normal(0, sigma) if np.random.rand() < rate else ind for ind in individual]
    individual = np.clip(individual, 0, 1)
    return individual

def tournament(pop, fitness_vals, k):
    candidates = np.random.choice(len(pop), size=k, replace=False)
    return candidates[np.argmax([fitness_vals[can] for can in candidates])]

def ga_best_positions():
    POPSIZE = 80
    GENS = 250
    ELITE_SIZE = 8
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.2
    MUTATION_SIGMA = 0.03
    np.random.seed(42)
    pop = np.random.rand(POPSIZE, 52)
    fitness_vals = np.array([fitness(individual) for individual in pop])
    best_idx = fitness_vals.argsort()[0]
    best_pos = pop[best_idx]
    for gen in range(GENS):
        new_pop = np.zeros((POPSIZE, 52))
        parents = [best_pos] * ELITE_SIZE
        for i in range(ELITE_SIZE, POPSIZE):
            parent1 = pop[tournament(pop, fitness_vals, 3)]
            parent2 = pop[tournament(pop, fitness_vals, 3)]
            new_pop[i] = crossover(parent1, parent2, CROSSOVER_RATE)
            new_pop[i] = mutate(new_pop[i], MUTATION_RATE, MUTATION_SIGMA)
        pop = new_pop
        fitness_vals = np.array([fitness(individual) for individual in pop])
        best_idx = fitness_vals.argsort()[0]
        best_pos = pop[best_idx]
    return best_pos

def polish_solution(pos):
    initial_guess = np.concatenate((pos, solve_radii(pos)[0]))
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    constraints = []
    for i in range(52):
        if i % 2 == 0:
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - x[i+1]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i] - x[i+1]})
        else:
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - x[i+1]})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i] - x[i+1]})
    for i in range(26):
        for j in range(i + 1, 26):
            distance = np.linalg.norm(pos[i, :2] - pos[j, :2])
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: distance - (x[i+52] + x[j+52])})
    options = {
        'maxiter': 4000,
        'ftol': 1e-9,
        'disp': False
    }
    result = minimize(fun=lambda x: objective(x), x0=initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
    return result.x[:52].reshape(26, 2), result.x[52:], -result.fun

def objective(positions):

def run_packing():
    best_pos = ga_best_positions()
    centers, radii, sum_r = polish_solution(best_pos)
    assert validate_packing(centers, radii)[0], "Invalid packing"
    return centers, radii, sum_r
