import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers) -> np.ndarray:
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists[j, i] = dists[i, j]
    return dists

def solve_radii_lp(centers) -> np.ndarray:
    n = centers.shape[0]
    A_eq = np.empty((n, n))
    b_eq = np.zeros(n)
    A_ineq = []
    b_ineq = []
    for i in range(n):
        x_i, y_i = centers[i]
        A_eq[i, i] = 1
        b_eq[i] = min(x_i, 1 - x_i, y_i, 1 - y_i)
        for j in range(i + 1, n):
            A_ineq.append([ centers[i][0] + centers[i][1], -2*centers[i][0] - 2*centers[j][0], centers[i][1] + centers[j][1] - 2*centers[i][1], 1 - centers[i][0] - centers[i][1], centers[i][0] + centers[i][1] - 2*centers[j][1], 1 - centers[j][0] - centers[j][1], -centers[i][0] - centers[j][0], -centers[i][1] - centers[j][1] ])
            b_ineq.append(centers[i][0] + centers[i][1] + centers[j][0] + centers[j][1] - 1 + centers[i][1] - centers[j][1] - centers[j][0] + centers[j][1] + centers[i][0] + centers[i][1] - centers[j][0] - centers[j][0] + centers[i][1] - centers[j][1] + centers[j][0] + centers[j][1] - 1 + centers[j][0] + centers[j][1] - centers[i][0] - centers[i][0] + centers[j][1] - centers[i][1] + centers[i][0] + centers[j][0] - 1)
    res = linprog(-np.ones(n), A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=[(0, None) for _ in range(n)], method='highs')
    return res.x

def sa_optimize(initial_centers) -> Tuple[np.ndarray, np.ndarray]:
    n = initial_centers.shape[0]
    best_centers = initial_centers.copy()
    best_radii = solve_radii_lp(best_centers)
    best_sum_radii = np.sum(best_radii)
    std_dev = 0.05
    for _ in range(500):
        for i in range(n):
            delta = np.random.normal(0, std_dev)
            epsilon = np.random.normal(0, std_dev)
            proposed_center = initial_centers[i].copy()
            proposed_center[0] += delta
            proposed_center[1] += epsilon
            if 0 <= proposed_center[0] <= 1 and 0 <= proposed_center[1] <= 1:
                initial_centers[i] = proposed_center
                new_radii = solve_radii_lp(initial_centers)
                new_sum_radii = np.sum(new_radii)
                if new_sum_radii > best_sum_radii - 1e-8:
                    best_centers = initial_centers.copy()
                    best_radii = new_radii
                    best_sum_radii = new_sum_radii
        std_dev *= 0.95
    return best_centers, best_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    initial_patterns = []
    initial_patterns.append(np.array([[j / 3, i / 3] for i in range(1, 5) for j in range(1, 5) if i % 2 != j % 2]))
    initial_patterns.append(np.array([np.random.rand(n, 2) for _ in range(3)]))
    initial_patterns.append(np.array([np.array([[0.25, 0.25], [0.75, 0.25], [0.5, 0.75], [0.25, 0.5], [0.75, 0.5], [0.5, 0.25], [0.5, 0.75], [0.5, 0.5], [0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.25], [0.5, 0.75], [0.5, 0.5]]) for _ in range(2)]))
    initial_patterns.append(np.array([np.array([[0.25, 0.25], [0.75, 0.75], [0.25, 0.75], [0.75, 0.25]]) for _ in range(3)]))
    best_centers = None
    best_radii = None
    best_sum_radii = 0.0
    for patterns in initial_patterns:
        for initial_centers in patterns:
            centers, radii = sa_optimize(initial_centers)
            sum_radii = np.sum(radii)
            if sum_radii > best_sum_radii - 1e-8:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum_radii = sum_radii
    res = minimize(
        fun=lambda x: -np.sum(x[52:]),
        x0=np.concatenate((best_centers.flatten(), best_radii)),
        method='SLSQP',
        bounds=[(0, None) for _ in range(52)] + [(0, max_radius) for _ in range(n)],
        constraints={'type': 'ineq', 'fun': lambda x: x[52:]},
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    if not res.success or np.any(constraints(res.x) < -1e-10):
        return np.zeros((n, 2)), np.zeros(n), 0.0
    centers = res.x[:52].reshape(n, 2)
    radii = res.x[52:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
