import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26

    def init_centers():
        center_families = {
            'grid': [(0.25, 0.25), (0.25, 0.5), (0.25, 0.75),
                      (0.5, 0.25), (0.5, 0.5), (0.5, 0.75),
                      (0.75, 0.25), (0.75, 0.5), (0.75, 0.75),
                      (0.375, 0.375), (0.625, 0.375), (0.375, 0.625), (0.625, 0.625)],
            'hexagons': [(0.25, 0.25), (0.25, 0.5), (0.25, 0.75),
                        (0.5, 0.2), (0.5, 0.4), (0.5, 0.6), (0.5, 0.8),
                        (0.75, 0.15), (0.75, 0.35), (0.75, 0.55), (0.75, 0.75),
                        (0.875, 0.25), (0.625, 0.25), (0.5, 0.25), (0.375, 0.25)],
            'hybrid': [(0.25, 0.25), (0.25, 0.75), (0.5, 0.25), (0.5, 0.75), (0.75, 0.25), (0.75, 0.75),
                       (0.125, 0.125), (0.375, 0.125), (0.625, 0.125), (0.875, 0.125),
                       (0.125, 0.375), (0.375, 0.375), (0.625, 0.375), (0.875, 0.375),
                       (0.125, 0.625), (0.375, 0.625), (0.625, 0.625), (0.875, 0.625),
                       (0.125, 0.875), (0.375, 0.875), (0.625, 0.875), (0.875, 0.4375)],
            'random': np.random.rand(26, 2)
        }
        return np.array(center_families[np.random.choice(list(center_families))])

    def simulate_speeds(centers, T=2000):
        births = np.logspace(0, 1, 30)
        speeds = np.zeros((T, n))
        for t in range(T):
            dirs = np.random.randn(n, 2)
            dirs /= np.linalg.norm(dirs, axis=1, keepdims=True)
            r = np.sort(np.random.choice(births, size=3 * n))
            speeds[t] = r[:n]
            speeds[t] /= speeds[t]
        return np.median(speeds, 0)

    def inflate(centers, speeds):
        N = len(centers)
        r = np.zeros(N)
        active = set(range(N))
        t_min = np.inf
        while active:
            distances = cdist(centers[active], centers[active])
            edge_times = np.minimum(centers[active, 0], 1 - centers[active, 0], centers[active, 1], 1 - centers[active, 1]) / speeds[active]
            pair_times = np.nan_to_num(distances / speeds[active, None] + speeds[active, None])
            pair_min = np.min(pair_times)
            edge_min = np.min(edge_times)
            t_min = min(t_min, pair_min, edge_min)
            if t_min >= np.inf:
                break
            stops = np.isclose(pair_times, t_min) | (edge_times == t_min)
            stop_indices = np.where(stops)[0]
            for i in stop_indices:
                j = np.where(stops[i, :])[0]
                if i != j:
                    r[i] = speeds[i] * t_min
                    r[j] = speeds[j] * t_min
                    active.discard(i)
                    active.discard(j)
                else:
                    r[i] = speeds[i] * t_min
                    active.discard(i)
        return centers, r

    def objective(x):
        centers = x[:52].reshape(n, 2)
        radii = x[52:]
        return -np.sum(radii)

    def constraints(x):
        centers = x[:52].reshape(n, 2)
        radii = x[52:]
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda center=x[:52], rad=x[52:], i=i: center[2 * i + 0] - rad[i]})
            constraints.append({'type': 'ineq', 'fun': lambda center=x[:52], rad=x[52:], i=i: 1 - center[2 * i + 0] - rad[i]})
            constraints.append({'type': 'ineq', 'fun': lambda center=x[:52], rad=x[52:], i=i: center[2 * i + 1] - rad[i]})
            constraints.append({'type': 'ineq', 'fun': lambda center=x[:52], rad=x[52:], i=i: 1 - center[2 * i + 1] - rad[i]})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append({'type': 'ineq', 'fun': lambda center=x[:52], rad=x[52:], i=i, j=j: dist - (rad[i] + rad[j])})
        return constraints

    centers = init_centers()
    speeds = simulate_speeds(centers)

    best_centers = centers
    best_radii = np.zeros(n)
    best_sum_radii = 0

    for _ in range(10):
        centers, radii = inflate(centers, speeds)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum_radii:
            best_centers = centers
            best_radii = radii
            best_sum_radii = sum_radii

    return best_centers, best_radii, best_sum_radii
