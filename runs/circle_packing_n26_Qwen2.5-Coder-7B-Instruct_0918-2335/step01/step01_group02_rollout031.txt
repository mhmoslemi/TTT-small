import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_objective(r):
    return -np.sum(r)

def lp_constraints(r, pos, d):
    cons = []
    for i in range(26):
        for j in range(i + 1, 26):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: d[i][j] - (r[i] + r[j])})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: pos[i, 0] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - pos[i, 0] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: pos[i, 1] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - pos[i, 1] - r[i]})
    return cons

def fitness(positions):
    pos = positions.reshape(26, 2)
    d = np.sqrt(np.sum((pos[:, None, :] - pos[None, :, :]) ** 2, axis=2))
    np.fill_diagonal(d, np.inf)
    r = np.zeros(26)
    res = linprog(lp_objective, A_ub=np.vstack(lp_constraints(r, pos, d)), b_ub=np.zeros(52), bounds=[(0, None) for _ in range(26)], method='highs')
    r = -res.fun
    return -res.fun

bounds = [(0, 1)] * 52

def run_packing():
    result = differential_evolution(fitness, bounds, popsize=50, maxiter=200, seed=42)
    best_positions = result.x.reshape(26, 2)
    d = np.sqrt(np.sum((best_positions[:, None, :] - best_positions[None, :, :]) ** 2, axis=2))
    np.fill_diagonal(d, np.inf)
    r = np.zeros(26)
    res = linprog(lp_objective, A_ub=np.vstack(lp_constraints(r, best_positions, d)), b_ub=np.zeros(52), bounds=[(0, None) for _ in range(26)], method='highs')
    r = -res.fun
    constrained_res = minimize(lambda x: -np.sum(x[52:]), x0=np.concatenate([best_positions.ravel(), r]), method='SLSQP', bounds=bounds + [(0, 0.5) for _ in range(26)], constraints=lp_constraints(r, best_positions, d), options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
    centers = constrained_res.x[:52].reshape(26, 2)
    radii = constrained_res.x[52:]
    return centers, radii, np.sum(radii)

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
