import numpy as np
from scipy.optimize import minimize, linprog

def compute_optimal_radii(pos):
    x = pos[:26]
    y = pos[26:]
    r = np.zeros(26)
    A = np.full((4*26, 26), np.inf)
    b = np.ones(4*26) * 1e-9
    A[:26, range(26)] = 1
    A[26:52, range(26)] = -1
    A[52:78, range(26)] = x
    A[78:104, range(26)] = -x
    A[104:128, range(26)] = y
    A[128:152, range(26)] = -y
    for i in range(26):
        for j in range(i+1, 26):
            d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            A[152+2*(i*26+j)], A[154+2*(i*26+j)] = 1, 1
            b[152+2*(i*26+j)], b[153+2*(i*26+j)] = d - 1e-9
    res = linprog(-r, A_ub=A, b_ub=b, bounds=[(0, 0.5)]*26, method='highs')
    if not res.success:
        return None
    return -res.fun * 1.5

def objective(pos_flat):
    radii = compute_optimal_radii(pos_flat)
    if radii is None:
        return 1e9
    return -np.sum(radii)

def run_packing():
    np.random.seed(42)
    best_sum = -np.inf
    best_center = None
    best_radii = None
    for _ in range(50):
        init_pos = np.random.rand(52)
        if _ % 10 == 0:
            # Start with a deterministic hexagonal lattice
            side = np.sqrt(2 * (1/26))
            offset = side / 2
            x = np.hstack((np.arange(0, 1, side)[:-1], 0.5, np.arange(0, 1, side)[:-1]))
            y = np.hstack((np.zeros(25), side, 2 * side, 3 * side, np.arange(0, 1, side)[:-1]))
            pos_flat = np.repeat(x, 2)
            pos_flat = np.insert(pos_flat, 47, np.repeat(y, 2))
            init_pos = pos_flat
        res = minimize(objective, init_pos, method='SLSQP', bounds=[(0, 1)]*52, options={'maxiter': 2000})
        if res.success:
            radii = compute_optimal_radii(res.x)
            if radii is not None and np.sum(radii) > best_sum:
                best_sum = np.sum(radii)
                best_center = res.x[:52].reshape(26, 2)
                best_radii = radii
    return best_center, best_radii, best_sum
