import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    max_iter = 2000
    min_step = 1e-5
    jitter_amp = 0.1
    gs = 0.02  # initial step size
    cool = 0.995  # decrease factor per full pass

    coords = np.random.rand(n, 2)
    radii = np.zeros(n)
    eps = 1e-12

    for _ in range(max_iter):
        step_size = gs * (cool ** _)
        if step_size < min_step:
            break

        coords += step_size * (np.random.rand(n, 2) - 0.5)
        coords[:, 0] = np.clip(coords[:, 0], radii, 1 - radii)
        coords[:, 1] = np.clip(coords[:, 1], radii, 1 - radii)

        dists = np.sqrt(((coords[:, None, :] - coords[None, :, :]) ** 2).sum(axis=2))
        mask = dists < radii[:, None] + radii[None, :] - eps

        for i in range(n):
            for j in range(i + 1, n):
                if mask[i, j]:
                    d = coords[i] - coords[j]
                    dist = dists[i, j]
                    sep = radii[i] + radii[j] - dist + eps
                    disp = 0.5 * sep * (d / dist)
                    coords[i] -= disp
                    coords[j] += disp

        dists = np.sqrt(((coords[:, None, :] - coords[None, :, :]) ** 2).sum(axis=2))
        mask = dists < radii[:, None] + radii[None, :] - eps

        if not np.any(mask):
            break

    dists = np.sqrt(((coords[:, None, :] - coords[None, :, :]) ** 2).sum(axis=2))
    mask = dists < radii[:, None] + radii[None, :] - eps

    def obj(v):
        return -np.sum(v)

    def cons(v):
        r = v.reshape(n,)
        cons = np.empty(n, float)
        cons[:n] = coords[:,0] - r
        cons[n:] = 1 - coords[:,0] - r
        return cons

    centers = coords
    bounds = [(0, 1)] * n + [(0, 1)] * n
    init = np.concatenate([radii, radii])

    res = minimize(fun=obj, x0=init, method='SLSQP',
                   bounds=bounds, constraints={'type': 'eq', 'fun': cons},
                   options={'maxiter': 1000, 'ftol': 1e-9})
    radii = -res.x

    radii = np.clip(radii, 0, 1 - radii[:, None] @ (radii[None, :] * (radii[None, :] > 1e-9)).T)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
