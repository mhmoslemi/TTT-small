import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import cdist
from numpy.random import random, randn, choice

# Constants
NP = 40
steps = 800
T0 = 0.05
cooling_rate = 0.99
d_shift = 2.0
d_vel = 0.02
tol = 1e-9

def lp(radii, center):
    rv = np.concatenate((center, radii))
    A = np.block([
        np.eye(1, 52, 0),
        -np.eye(1, 52, 0),
        np.eye(1, 52, 0),
        -np.eye(1, 52, 0),
        np.eye(52) * (-1)
    ])
    b = np.hstack([np.ones(26), np.zeros(26), np.ones(26), np.zeros(26), -(radii + radii[:, np.newaxis] + 1e-9)])
    Apairs = ((center[:, np.newaxis, :] - center[np.newaxis, :, :])**2).sum(axis=2)**0.5 <= (radii[:, np.newaxis] + radii[np.newaxis, :] + 1e-9)
    Apairs = np.triu(Apairs, 1)
    Apairdata = np.zeros([Apairs.sum(), 2])
    ct = 0
    for i in range(Apairs.shape[0]):
        for j in range(Apairs.shape[1]):
            if Apairs[i, j]:
                Apairdata[ct, :] = [i, j]
                ct += 1
    Apairs = Apairdata
    for pair in Apairs:
        i, j = pair.tolist()
        A = np.block([
            A,
            np.zeros((3, 52)),
            np.eye(2),
            -np.eye(2),
            -np.eye(2),
            -np.eye(2)
        ])
        b = np.hstack([b, np.zeros(6)])
    res = linprog(-radii, A_ub=A, b_ub=b, bounds=(0, None), method='highs', tol=tol)
    if not res.success:
        return -1337
    return res.x

def greedy_seed():
    centers = []
    # Corner circles
    for i in range(4):
        centers.append([0.5 * (random() * 2 - 1), 0.5 * (random() * 2 - 1)])
    # Edge circles
    for i in range(4):
        x = random() * 0.7
        centers.append([x + 0.179, 0.5 * (random() * 2 - 1)])
    centers.append([0.5, 0.5 * (random() * 2 - 1)])
    for i in range(4):
        y = random() * 0.7
        centers.append([0.5 * (random() * 2 - 1), y + 0.179])
    for i in range(2):
        x = random() * 0.7
        centers.append([0.5 + (random() * 2 - 1) * 0.285, 0.5])
    for i in range(2):
        y = random() * 0.7
        centers.append([0.5, 0.5 + (random() * 2 - 1) * 0.285])
    # Add random jitter to avoid overlap
    centers = np.array(centers) * 0.5 + 0.25  # Shift to center due to greedy placement
    centers = np.clip(centers, 0, 1)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_centers = np.zeros((NP, 52))
    best_radii = np.zeros((NP, 26))
    best_sum = 0
    for _ in range(20):  # Number of restarts
        centers = greedy_seed()
        best_val = -1338
        for _ in range(NP):  # Number of particles
            pos = centers + 0.01 * randn(52)
            pos = np.clip(pos, 0, 1)
            r = lp(center=pos[:52], radii=pos[52:])
            if r > best_val:
                best_val = r
                best_centers[_] = pos[:52]
                best_radii[_] = pos[52:]
        T = T0
        for _ in range(steps):  # Number of iterations
            for i in range(NP):
                # Velocity update
                r1, r2 = random(2)
                v = 0.5 * best_centers[i] + 1.5 * (best_centers[np.argmax(best_centers[:, 0])] - best_centers[i]) + 1.5 * (best_centers[-1] - best_centers[i])
                v = v + (randn(52 if _ % 2 == 0 else 2) * d_shift) * min(1, T / T0)
                v = np.clip(v, -1, 1) * d_vel
                # Position update
                pos = best_centers[i] + v
                pos = np.clip(pos, 0, 1)
                # Fitness evaluation
                r = lp(center=pos[:52], radii=pos[52:])
                if r > best_val:
                    best_val = r
                    best_centers[i] = pos[:52]
                    best_radii[i] = pos[52:]
            T *= cooling_rate
    return best_centers[np.argmax(best_radii[:, 0]), :52].reshape(26, 2), best_centers[np.argmax(best_radii[:, 0]), 52:], best_radii[np.argmax(best_radii[:, 0]), 0:].sum()

centers, radii, total_radius = run_packing()
print(f"Centers:\n{centers}\nRadii:\n{radii}\nTotal Radius Sum: {total_radius}")
