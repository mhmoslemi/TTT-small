import numpy as np
from scipy.optimize import minimize, linprog

def compute_pairwise_distances(centers):
    n = centers.shape[0]
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            D[i, j] = D[j, i] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
    return D

def compute_wall_limits(centers):
    return np.minimum.reduce([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]])

def lp_value_and_radii(centers):
    D = compute_pairwise_distances(centers)
    w_limits = compute_wall_limits(centers)
    A_ub = np.empty((26 * (26 - 1) // 2, 26))
    b_ub = np.empty(26 * (26 - 1) // 2)
    k = 0
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub[k, i] = 1
            A_ub[k, j] = 1
            b_ub[k] = D[i, j]
            k += 1
    bounds = [(0, w) for w in w_limits]
    res = linprog(-np.ones(26), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    else:
        return -np.inf, np.zeros(26)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)  # For reproducibility
    n_restart = 150
    n_steps = 2000
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    initial_layouts = np.random.rand(150, 26, 2) * 0.95 + 0.05  # Jittered hexagonal lattice

    for centers in initial_layouts:
        for _ in range(n_steps):
            k = np.random.randint(26)
            delta = np.random.normal(scale=0.02)
            centers[k] += [delta, delta]
            centers[k] = np.clip(centers[k], 0, 1)  # Ensure within bounds

            value, radii = lp_value_and_radii(centers)
            if value > best_sum:
                best_sum = value
                best_centers = centers
                best_radii = radii

            if value < best_sum:
                t = 1.0
                if np.random.rand() < np.exp((value - best_sum) / t):
                    best_sum = value
                    best_centers = centers
                    best_radii = radii
                else:
                    centers[k] -= 2 * [delta, delta]

    # Local refinement
    def objective(x):
        return -lp_value_and_radii(x.reshape(26, 2))[0]

    bounds = [(0, 1)] * 52
    result = minimize(objective, best_centers.flatten(), method='BFGS', bounds=bounds)
    if result.success:
        best_centers = result.x.reshape(26, 2)
        best_radii, _ = lp_value_and_radii(best_centers)

    # Ensure non-overlap and boundary constraints
    for i in range(26):
        if best_radii[i] < 0:
            return np.zeros((26, 2)), np.zeros(26), 0.0

    for i in range(26):
        x, y = best_centers[i]
        r = best_radii[i]
        if not (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
            return np.zeros((26, 2)), np.zeros(26), 0.0

    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
            if dist < best_radii[i] + best_radii[j]:
                return np.zeros((26, 2)), np.zeros(26), 0.0

    return best_centers, best_radii, best_sum
