import numpy as np
from scipy.optimize import linprog, minimize


def optimise_radii(centers):
    n = centers.shape[0]
    bounds = [(0, min(c[0], 1 - c[0])) for c in centers] + [(0, None) for _ in range(n)]
    cons = []

    def make_dist_matrix(centers):
        diffs = centers[:, None, :] - centers[None, :, :]
        dist_matrix = np.sqrt((diffs ** 2).sum(axis=2))
        return dist_matrix

    dist_matrix = make_dist_matrix(centers)

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - min(dist_matrix[i])})

    res = linprog([1] * n, A_ub=dist_matrix.transpose()[:, 1:], b_ub=np.inf * np.ones(n) - centers[:, 0] - centers[:, 1], bounds=bounds, constraints=cons)
    
    return res.x if res.success else None


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    delta_max = 0.05
    T0 = 1.0
    alpha = 0.995
    NITER = 50000

    def random_step(delta_max):
        return np.random.uniform(-delta_max, delta_max, 2)

    centers = 0.2 * np.ones((n, 2)) + 0.1 * (2 * np.random.rand(n, 2) - 1)  # 5x5 grid
    centers[25] = np.array([0.5, 0.5])  # Central disc

    current_centres, current_radii = centers, optimise_radii(centers)
    best_sum, best_centres, best_radii = sum(current_radii), current_centres, current_radii

    for iter in range(NITER):
        i, δx, δy = np.random.randint(0, n), random_step(delta_max)
        new_coors = current_centres[i] + (δx, δy)
        
        if 0 <= new_coors[0] <= 1 and 0 <= new_coors[1] <= 1:
            temp_centres = current_centres.copy()
            temp_centres[i] = new_coors
            temp_radii = optimise_radii(temp_centres)
            
            if temp_radii is not None:
                new_sum = sum(temp_radii)
                ΔS = new_sum - best_sum
                
                if ΔS > 0 or np.random.rand() < np.exp(ΔS / (T0 * alpha ** iter)):
                    current_centres, current_radii, best_sum = temp_centres, temp_radii, new_sum
                    if new_sum > best_sum:
                        best_centres, best_radii = temp_centres, temp_radii

    current_centres, current_radii = best_centres, best_radii
    result = minimize(lambda v: -np.sum(v[52:]), current_centres.ravel(), method='SLSQP', bounds=[(0, 1)] * 52 + [(0, 0.5)] * 26, constraints={'type': 'ineq', 'fun': lambda v: np.array([v[i] - (v[i + 25] + v[i + 50]) for i in range(0, 52, 2)])}, options={'maxiter': 2000, 'ftol': 1e-9})
    return best_centres, best_radii, sum(best_radii)
