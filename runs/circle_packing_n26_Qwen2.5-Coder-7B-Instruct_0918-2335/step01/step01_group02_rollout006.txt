import numpy as np

# Constants
POPSIZE = 80

def solve_radii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append(np.eye(2, n)[i])
        b_ub.append(pos[i, 0])  # x_i <= r_i
        A_ub.append(np.eye(2, n)[i])
        b_ub.append(1 - pos[i, 0])  # 1 - x_i <= r_i
        A_ub.append(np.eye(2, n)[i])
        b_ub.append(pos[i, 1])  # y_i <= r_i
        A_ub.append(np.eye(2, n)[i])
        b_ub.append(1 - pos[i, 1])  # 1 - y_i <= r_i

    for i in range(n):
        for j in range(i + 1, n):
            dist2 = pos[i, 0] ** 2 + pos[i, 1] ** 2 + pos[j, 0] ** 2 + pos[j, 1] ** 2 - 2 * (pos[i, 0] * pos[j, 0] + pos[i, 1] * pos[j, 1])
            A_ub.append(np.hstack([np.eye(2, n)[i], np.eye(2, n)[j]]))
            b_ub.append(np.sqrt(dist2) - (pos[i, 2] + pos[j, 2]))

    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    return res.x, res.fun * -1
