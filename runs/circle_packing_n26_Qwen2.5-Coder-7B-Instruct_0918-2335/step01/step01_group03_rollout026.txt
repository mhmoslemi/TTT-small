import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n)  # 2 * x, y coordinates for each circle

    def optimise_radii(centers):
        r = np.zeros(n)
        A_ineq = np.array([])
        b_ineq = np.array([])
        A_eq = np.array([])
        b_eq = np.array([])

        for i in range(n):
            r[i] = max(0, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
                A_ineq = np.vstack((A_ineq, np.array([1, 1, 0, 0 introduces error preventive checks. 0, 0, 1, 1]), np.array([-1, -1, 0, 0, 0, 0, 1, 1])))
                b_ineq = np.append(b_ineq, np.array([dist - r[i] - r[j], r[i] + r[j]]))
        
        A_eq = np.vstack((A_eq, np.array([1 for _ in range(n)])))
        b_eq = np.array([1])

        linprog_result = linprog(-r, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=(0, None))
        if linprog_result.success:
            return -linprog_result.fun  # We minimize negative radii for maximization
        else:
            return None  # Return None if LP is infeasible

    INITIAL_START = np.array([[i // 5 * 0.2 + 0.1 for i in range(n)] for n in range(25)] + [[0.5, 0.5]])
    current_centres = INITIAL_START
    current_radii = optimise_radii(current_centres)
    best_sum = sum(current_radii)
    best_centres = current_centres
    best_radii = current_radii

    T0 = 1.0
    alpha = 0.995
    delta_max = 0.05
    NITER = 50000

    def random_step(delta_max):
        return np.array([np.random.uniform(-delta_max, delta_max), np.random.uniform(-delta_max, delta_max)])

    for iter in range(1, NITER + 1):
        i = np.random.randint(n)
        δx, δy = random_step(delta_max)
        new_centres = current_centres.copy()
        new_centres[i] += np.array([δx, δy])

        if np.all(new_centres >= 0) and np.all(new_centres <= 1):
            new_radii = optimise_radii(new_centres)
            if new_radii is not None:
                new_sum = sum(new_radii)
                ΔS = new_sum - best_sum
                if ΔS > 0:
                    current_sum = new_sum
                    current_centres = new_centres
                    current_radii = new_radii
                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_centres = new_centres
                        best_radii = new_radii
                else:
                    T = T0 * alpha ** iter
                    if np.random.rand() < np.exp(ΔS / T):
                        current_sum = new_sum
                        current_centres = new_centres
                        current_radii = new_radii

    def slsqp_refine(centers, radii):
        x0 = np.concatenate((centers.flatten(), radii))
        def objective(v):
            return -np.sum(v[n:])
        def boundary(v):
            return np.array([v[i] for i in range(n)] + [1 - v[i] for i in range(n)] + [v[n + i] - 0.5 for i in range(n)])
        def overlap(v):
            v = v.reshape((2 * n + n, 1))
            x = v[:n]
            r = v[n:]
            constraints = np.array([])
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sum((x[i] - x[j]) ** 2)
                    constraints = np.append(constraints, dist - r[i] - r[j])
            return constraints
        sol = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': boundary}, options={'maxiter': 2000, 'ftol': 1e-9})
        if sol.success:
            return sol.x[:n].reshape((n, 2)), sol.x[n:]
        else:
            return centers, radii

    best_centres, best_radii = slsqp_refine(best_centres, best_radii)
    return best_centres, best_radii, sum(best_radii)
