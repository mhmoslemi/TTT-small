import numpy as np
from scipy.optimize import linprog, minimize

def hexagonal_grid(n):
    m = np.sqrt(n)
    k = np.ceil((m + 1) / 2)
    r = (1 / (2 * np.tan(np.pi / 6))) / 2
    centers = np.zeros((n, 2))
    k = int(k)
    c = 0
    for i in range(k):
        row_off = 0 if i % 2 == 0 else r * np.sqrt(3)
        for j in range(n):
            if i * 2 + j < n:
                centers[c, 0] = row_off + (j % 2) * r * np.sqrt(3) + (i % 2) * r
                centers[c, 1] = i * r * np.sqrt(3)
                c += 1
    centers -= centers.min(0)
    centers /= centers.max(0)
    centers[(centers >= 1) | (centers < 0)] = 0
    centers[centers > 1] = 1
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)
    return centers + np.random.uniform(-0.01, 0.01, centers.shape)

def random_centers(n):
    return np.random.rand(n, 2)

def grid_centers(n):
    side_length = np.sqrt(n)
    side = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = (i % side) / side_length
        centers[i, 1] = (i // side) / side_length
    centers[(centers >= 1) | (centers < 0)] = 0
    centers[centers > 1] = 1
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)
    return centers + np.random.uniform(-0.01, 0.01, centers.shape)

def corner_heavy_centers(n):
    centers = np.zeros((n, 2))
    corners = np.array([[0.1667, 0.1667], [0.1667, 0.8333], [0.8333, 0.1667], [0.8333, 0.8333]])
    bits = np.random.choice(corners, n, replace=False)
    centers[:4] = bits[:4] + np.random.uniform(-0.02, 0.02, bits[:4].shape)
    centers[4:] = np.random.uniform(-0.01, 0.01, centers[4:].shape) + (centers[4:] / centers[4:].max())
    return centers

def lp_opt(centers):
    n = centers.shape[0]
    A = np.empty((2 * n + (n * (n - 1)) // 2, 2 * n))
    b = np.empty(2 * n + (n * (n - 1)) // 2)
    
    for i in range(n):
        A[2 * i, 2 * i] = 1
        A[2 * i + 1, 2 * i + 1] = 1
        b[2 * i] = 1
        b[2 * i + 1] = 1
    
    for i in range(n):
        for j in range(i + 1, n):
            A[2 * n + (i * (i + 1)) // 2 + (j - i - 1), 2 * i] = 1
            A[2 * n + (i * (i + 1)) // 2 + (j - i - 1), 2 * j] = 1
            A[2 * n + (i * (i + 1)) // 2 + (j - i - 1), 2 * i + 1] = 1
            A[2 * n + (i * (i + 1)) // 2 + (j - i - 1), 2 * j + 1] = 1
            b[2 * n + (i * (i + 1)) // 2 + (j - i - 1)] = np.linalg.norm(centers[i] - centers[j])
    
    res = linprog(-np.ones(2 * n), A_ub=A, b_ub=b, bounds=[(0, None)] * 2 * n, method='highs')
    
    if not res.success:
        return np.zeros(n)
    
    return -res.fun

def local_slsqp(p):
    n = p.shape[0]
    r = np.sqrt(np.sum((p.reshape(n, 2) - p.reshape(n, 2, 1)) ** 2, axis=2))
    
    def f(x):
        return np.sum(x)
    
    def c1(x):
        return x
    
    def c2(x):
        return -x
    
    def c3(x):
        return np.sum((x.reshape(n, 2) - x.reshape(n, 2, 1)) ** 2, axis=2) + 1e-6 - r
    
    def c4(x):
        return 1 - (x.reshape(n, 2) - x.reshape(n, 2, 1)) ** 2, axis=2) + 1e-6 - r
    
    def eq(x):
        return (r - (x.reshape(n, 2) - x.reshape(n, 2, 1)) ** 2, axis=2) * (x.reshape(n, 2) - x.reshape(n, 2, 1)) ** 2, axis=2) - 1e-6
    
    def local_objective(x):
        return f(x)
    
    cons = [{'type': 'ineq', 'fun': c1},
            {'type': 'ineq', 'fun': c2},
            {'type': 'ineq', 'fun': c3},
            {'type': 'ineq', 'fun': c4}
           ]
    
    sol = minimize(local_objective, p, constraints=cons)
    
    if not sol.success:
        return p
    
    return sol.x
