import numpy as np
from scipy.spatial import distance

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = distance.euclidean(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define a hybrid geometry
    centers[0] = [0.25, 0.25]  # corner circle
    centers[1] = [0.75, 0.25]
    centers[2] = [0.25, 0.75]
    centers[3] = [0.75, 0.75]
    centers[4] = [0.5, 0.25]
    centers[5] = [0.25, 0.5]
    centers[6] = [0.75, 0.5]
    centers[7] = [0.5, 0.75]
    centers[8:12] = [(0.3, 0.3), (0.3, 0.8), (0.8, 0.8), (0.75, 0.3), (0.8, 0.3), (0.75, 0.8)]
    centers[12:14] = [(0.4, 0.6), (0.6, 0.6)]
    centers[14:16] = [(0.45, 0.35), (0.55, 0.65)]
    centers[16:18] = [(0.65, 0.4), (0.7, 0.5)]
    centers[18:20] = [(0.6, 0.5), (0.65, 0.55)]
    centers[20:22] = [(0.6, 0.7), (0.65, 0.75)]
    centers[22:24] = [(0.5, 0.6), (0.55, 0.7)]
    centers[24] = [0.4, 0.7]
    centers[25] = [0.6, 0.3]

    # Apply anisotropic speed scaling
    def speed_scale(x, y):
        return 1 + 2 / min(x, 1 - x, y, 1 - y) + np.random.lognormal(0, 0.4)

    radii[:4] = speed_scale(centers[:4, 0], centers[:4, 1]) * 0.2
    radii[4:8] = speed_scale(centers[4:8, 0], centers[4:8, 1]) * 0.1
    radii[8:12] = speed_scale(centers[8:12, 0], centers[8:12, 1]) * 0.1
    radii[12:14] = speed_scale(centers[12:14, 0], centers[12:14, 1]) * 0.05
    radii[14:16] = speed_scale(centers[14:16, 0], centers[14:16, 1]) * 0.05
    radii[16:18] = speed_scale(centers[16:18, 0], centers[16:18, 1]) * 0.05
    radii[18:20] = speed_scale(centers[18:20, 0], centers[18:20, 1]) * 0.05
    radii[20:22] = speed_scale(centers[20:22, 0], centers[20:22, 1]) * 0.05
    radii[22:24] = speed_scale(centers[22:24, 0], centers[22:24, 1]) * 0.05
    radii[24:26] = speed_scale(centers[24:26, 0], centers[24:26, 1]) * 0.05

    # Inflating-disc simulation
    for _ in range(500):
        colliding = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = distance.euclidean(centers[i], centers[j])
                if dist < radii[i] + radii[j] - 1e-12:
                    colliding = True
                    break
            if colliding:
                break
        if not colliding:
            break

    # Refine with SLSQP
    from scipy.optimize import minimize
    def objective(x):
        radii = x[:n]
        return -np.sum(radii)
    
    def constraint(x):
        radii = x[:n]
        cons = np.empty(n, float)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons[i] = (x - r) * (x + r) - 1 * 1e-12
        return cons

    bounds = [(0, 0.5) for _ in range(n)]
    constraints = {'type': 'ineq', 'fun': constraint}

    res = minimize(fun=objective, x0=radii, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
    if not res.success:
        raise RuntimeError("Optimization did not succeed")
    
    return centers, res.x, np.sum(res.x)
