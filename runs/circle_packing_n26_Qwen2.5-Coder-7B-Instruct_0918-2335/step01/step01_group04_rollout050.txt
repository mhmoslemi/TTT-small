import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    n = len(centers)
    c = np.ones(n, dtype=float)
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n, dtype=float)
    A_ineq = np.zeros((3*n, n))
    b_ineq = np.zeros(3*n, dtype=float)

    for i in range(n):
        A_ineq[i][i] = 1
        A_ineq[2*n+i][i] = -1
        A_eq[i][i] = 2
        b_eq[i] = centers[i][0]

    for i in range(n):
        A_ineq[3*n+i][i] = 1
        A_ineq[5*n+i][i] = -1
        A_eq[i+n][i] = 2
        b_eq[i+n] = centers[i][1]

    for i in range(n):
        for j in range(i + 1, n):
            A_ineq[6*n+i][j] = 1
            A_ineq[7*n+i][i] = 1
            A_ineq[6*n+j][i] = 1
            A_ineq[7*n+j][j] = 1
            A_ineq[i*3*n+2][n+i*j] = 2
            A_ineq[i*3*n+2][n+j*i] = 2
            b_ineq[i*3*n+2] = (centers[i] - centers[j]).dot(centers[i] - centers[j])

    # Solve the linear programming problem
    lp_result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ineq, b_ub=b_ineq, bounds=(0, 1))
    return lp_result.x

def move_center(centers, i, d_min, active_constraint, alpha):
    x_i, y_i = centers[i]
    dx, dy = 0, 0

    if active_constraint:
        x_j, y_j = active_constraint
        dx, dy = alpha * (x_i - x_j) / np.linalg.norm(x_i - x_j)
    else:
        direction = direction_map[active_constraint]
        dx, dy = alpha * direction

    x_i += dx
    y_i += dy

    return centers[:i] + [(x_i, y_i)] + centers[i+1:]

def direction_map:
    return {
        'left': (1, 0),
        'right': (-1, 0),
        'bottom': (0, 1),
        'top': (0, -1)
    }
