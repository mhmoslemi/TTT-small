import numpy as np
from scipy.optimize import minimize, differential_evolution, shgo, basinhopping, NonlinearConstraint, Bounds, SLSQP

N = 26
NUM_VARS = 2 * N + N

def obj(v):
    return -np.sum(v[NUM_VARS:])

def constraints(v):
    pos = v[:NUM_VARS:2].reshape((N, 2))
    r = v[NUM_VARS:].reshape((N,))
    
    cons = []
    
    # Boundary constraints
    cons += [pos[:, 0] - r]
    cons += [1 - pos[:, 0] - r]
    cons += [pos[:, 1] - r]
    cons += [1 - pos[:, 1] - r]
    
    # Separation constraints
    diffs = pos[:, None, :] - pos[None, :, :]
    dists_squared = (diffs**2).sum(axis=2)
    cons += [dists_squared - (r[:, None] + r[None, :])**2]
    
    return np.concatenate(cons)

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(N):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist - 2 * radii[i] < -1e-12:
                return False, f"Circles {i} and {j} overlap"
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = Bounds(0, 1, [0, 1])[:2 * N + N]
    best_r = None
    best_res = None
    
    def local_minimizer(x, minimize_args):
        local_res = minimize(fun=obj, x0=x, method='SLSQP', bounds=bounds, constraints=shgo_constr, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
        return local_res.fun, local_res.x
    
    maxiter = 1000
    penalty_factor = 1e4
    
    def penalty_cons(x):
        r = x[NUM_VARS:]
        return r
    
    shgo_constr = [
        NonlinearConstraint(constraints, -np.inf, np.inf),
        NonlinearConstraint(penalty_cons, 0, 1e-6)
    ]
    
    for _ in range(100):
        init = np.random.rand(NUM_VARS) * 0.1 + np.random.rand(NUM_VARS) * 0.45
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds, constraints=shgo_constr, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
        if res.success and (best_res is None or res.fun < best_res.fun and constraints(res.x).min() > -1e-10): 
            best_res = res
            best_r = res.x[NUM_VARS:]
    
    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")
    
    inflated_r = inflate_radii(best_res.x[:NUM_VARS:2].reshape((N, 2)), best_r)
    final_res = minimize(fun=obj, x0=np.concatenate([best_res.x[:NUM_VARS], inflated_r]), method='SLSQP', bounds=bounds, constraints=shgo_constr, options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False})
    centers = final_res.x[:NUM_VARS:2].reshape((N, 2))
    sum_radii = -final_res.fun
    
    assert validate_packing(centers, inflated_r), "Solution is invalid"
    
    return centers, inflated_r, sum_radii
