import numpy as np
from scipy.optimize import linprog, minimize

def pairwise_dists(centers):
    n = len(centers)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists[j, i] = dists[i, j]
    return dists

def solve_radii_lp(centers, max_radius):
    n = len(centers)
    A = np.zeros((2 * n + n * (n - 1), n))
    b = np.zeros(2 * n + n * (n - 1))
    
    # Boundary constraints
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i][0]
        A[n + i, i] = -1
        b[n + i] = 1 - centers[i][0]
        A[2 * n + i, i] = 1
        b[2 * n + i] = centers[i][1]
        A[2 * n + n + i, i] = -1
        b[2 * n + n + i] = 1 - centers[i][1]
    
    # Pairwise constraints
    idx = 2 * n + 2 * n + n
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = -pairwise_dists(centers)[i, j] + max_radius
            A[idx + 1, i] = -1
            A[idx + 1, j] = -1
            b[idx + 1] = pairwise_dists(centers)[i, j]
            idx += 2
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, max_radius), method='highs')
    return np.maximum(res.x, 0)

def sa_optimize(initial_centers, max_radius, max_iter=500, alpha=0.99):
    n = len(initial_centers)
    best_centers = initial_centers.copy()
    best_radii = solve_radii_lp(best_centers, max_radius).copy()
    best_sum_radii = np.sum(best_radii)
    current_centers = best_centers.copy()
    current_radii = best_radii.copy()
    current_sum_radii = best_sum_radii
    
    for _ in range(max_iter):
        i = np.random.randint(n)
        delta = np.random.normal(0, 0.01)
        epsilon = np.random.normal(0, 0.01)
        new_centers = current_centers.copy()
        new_centers[i] += np.array([delta, epsilon])
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        new_radii = solve_radii_lp(new_centers, max_radius).copy()
        new_sum_radii = np.sum(new_radii)
        
        if new_sum_radii - current_sum_radii >= 1e-8:
            current_centers = new_centers.copy()
            current_radii = new_radii.copy()
            current_sum_radii = new_sum_radii
        elif np.random.rand() < alpha:
            current_centers = new_centers.copy()
            current_radii = new_radii.copy()
            current_sum_radii = new_sum_radii
    
    if current_sum_radii > best_sum_radii:
        best_centers = current_centers
        best_radii = current_radii
        best_sum_radii = current_sum_radii
    
    return best_centers, best_radii
