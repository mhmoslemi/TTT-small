import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers_families = [
        np.array([[0.5, 0.5], [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.5, 0.25], [0.5, 0.75]], dtype=float),
        np.array([[0.5, 0.5], [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.3125, 0.3125], [0.3125, 0.6875], [0.6875, 0.3125], [0.6875, 0.6875]], dtype=float),
        np.array([[0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75], [0.5, 0.5], [0.5, 0.6875], [0.5, 0.3125], [0.3125, 0.5], [0.6875, 0.5], [0.6875, 0.687], [0.3125, 0.6875], [0.3125, 0.3125], [0.6875, 0.3125], [0.6875, 0.6875], [0.875, 0.875], [0.125, 0.125], [0.875, 0.125], [0.125, 0.875], [0.3125, 0.375], [0.6875, 0.625], [0.625, 0.375], [0.375, 0.625]], dtype=float),
        np.random.rand(n, 2)
    ]

    def sim radii(speeds):
        t_stop = np.zeros(n, dtype=float)
        active = np.arange(n)
        while len(active):
            dist = cdist(centers[active], centers[active], 'euclidean')
            t_ij = dist / (speeds[active, None] + speeds[active[:, None]])
            t_ij[np.repeat(np.arange(n), n).reshape(n, n) >= np.arange(n)[:, None] * n + np.arange(n)] = np.inf
            t_boundary = np.min(np.stack([centers[active, 0], 1 - centers[active, 0], centers[active, 1], 1 - centers[active, 1]]), axis=0) / speeds[active]
            t_min = np.min([t_ij.min(axis=1), t_boundary.min(axis=0)])
            t_stop[active[t_ij.argmin()] = t_min
            active = active[t_ij.argmax() != np.argmin(t_ij)]
        radii = speeds * t_stop
        return radii.sum()

    def objective(speeds):
        return -sim_ridges(speeds)

    def bounds():
        speeds = np.empty((n,), dtype=float)
        for i in range(n):
            d = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            speeds[i] = 1 + 2 / (d + 1e-6)
        m = np.random.lognormal(0, 0.4, n)
        return (1, 10) * m * speeds

    def init_speeds():
        return bounds()

    def local_refinement(speeds, centers):
        def obj(x):
            nonlocal speeds, centers
            centers = x[:n * 2].reshape(n, 2)
            radii = speeds * np.min(np.stack([centers, 1 - centers]), axis=0)
            return -np.sum(radii)

        def const_ineq(x):
            nonlocal speeds, centers
            centers = x[:n * 2].reshape(n, 2)
            radii = speeds * np.min(np.stack([centers, 1 - centers]), axis=0)
            dist = cdist(centers, centers, 'euclidean')
            return dist - np.stack([radii, radii], axis=2).sum(axis=2)

        const_ineq_grad = lambda x: grad_const_ineq(x, speeds, centers)

        res = minimize(obj, centers.flatten(), method='SLSQP', jac=None, bounds=bounds(), constraints={'type': 'ineq', 'fun': const_ineq, 'jac': const_ineq_grad}, options={'maxiter': 2000, 'ftol': 1e-9})

        if res.success:
            centers = res.x[:n * 2].reshape(n, 2)
            return centers, speeds * np.min(np.stack([centers, 1 - centers]), axis=0)
        else:
            raise RuntimeError("Local refinement did not succeed")

    def grad_const_ineq(x, speeds, centers):
        m = x.shape[0] // 2
        centers = x[:m * 2].reshape(m, 2)
        radii = speeds * np.min(np.stack([centers, 1 - centers]), axis=0)
        dist = cdist(centers, centers, 'euclidean')
        const = dist - np.stack([radii, radii], axis=2).sum(axis=2)

        grad = np.zeros_like(x)
        for i in range(m):
            for j in range(m):
                if i != j:
                    delta_r = (radii[i] + radii[j] - dist[i, j])
                    grad[m * 2 + 2 * i] += -delta_r / np.sqrt(dist[i, j])
                    grad[m * 2 + 2 * j] += -delta_r / np.sqrt(dist[i, j])
                else:
                    grad[m * 2 + 2 * i] += 0.
                    grad[m * 2 + 2 * j] += 0.
        return grad

    best_radius_sum = float('-inf')
    best_shift_and_scale = None
    for family in centers_families:
        for _ in range(3000):
            speeds = init_speeds()
            radii = sim_ridges(speeds)
            if radii.sum() > best_radius_sum:
                best_radius_sum = radii.sum()
                best_shift_and_scale = (speeds, radii)

    centers, radii = best_shift_and_scale

    if best_radius_sum > 2.636:
        centers, radii = local_refinement(speeds, centers)

    if all(np.isnan(centers).any(), np.isnan(radii).any()):
        raise RuntimeError("NaN values present")

    for i, (x, y) in enumerate(centers):
        r = radii[i]
        if not (0 <= x - r <= 1 and 0 <= x + r <= 1 and 0 <= y - r <= 1 and 0 <= y + r <= 1):
            raise RuntimeError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                raise RuntimeError(f"Circles {i} and {j} overlap")

    return centers, radii, best_radius_sum
