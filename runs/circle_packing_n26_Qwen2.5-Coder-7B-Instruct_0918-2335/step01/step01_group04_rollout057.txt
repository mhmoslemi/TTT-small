import numpy as np
from scipy.optimize import LinprogResult, minimize
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    def hexagonal_lattice_seed():
        rows = 5
        cols = 5
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i % 2 == 0:
                    centers.append([i * box_size / (rows - 1), j * box_size / (cols - 1)])
                else:
                    centers.append([(i * box_size / (rows - 1)) + 0.5, j * box_size / (cols - 1)])
        centers.append([(rows - 2) * box_size / (rows - 1), (cols - 2) * box_size / (cols - 1)])
        centers.append([(rows - 3) * box_size / (rows - 1), (cols - 2) * box_size / (cols - 1)])
        centers.append([(rows - 1) * box_size / (rows - 1), (cols - 2) * box_size / (cols - 1)])
        return np.array(centers[:n])

    def linear_program(centers, radii):
        A = np.zeros((3 * n, n))
        b = np.zeros(3 * n)
        u = np.zeros(n)
        
        # Wall constraints
        for i in range(n):
            A[i, i] = 1
            b[i] = box_size
            u[i] = box_size
        
        for i in range(n):
            A[n + i, i] = -1
            b[n + i] = 0
        
        # Pairwise constraints
        dists = squareform(pdist(centers))
        idx = np.triu_indices(n, k=1)
        A[2 * n:, idx[0]] = 1
        A[2 * n:, idx[1]] = 1
        A[2 * n:, idx[0]] -= radii[idx[0]]
        A[2 * n:, idx[1]] -= radii[idx[1]]
        b[2 * n:] = dists[idx] - 1e-5
        
        result = linprog(-radii, A_ub=A, b_ub=b, bounds=(0, max_radius), method='highs')
        return result.x

    def push_away(centers, radii):
        new_centers = centers.copy()
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            pos_constr = np.array([x - r, 1 - x - r, y - r, 1 - y - r])
            neg_constr = pos_constr < 0
            if neg_constr.sum() == 0:
                continue
            
            possible_moves = []
            if neg_constr[0]:
                possible_moves.append((1, 0))
            if neg_constr[1]:
                possible_moves.append((-1, 0))
            if neg_constr[2]:
                possible_moves.append((0, 1))
            if neg_constr[3]:
                possible_moves.append((0, -1))
            
            if len(possible_moves) == 1:
                move = possible_moves[0]
                new_x, new_y = x + move[0], y + move[1]
            else:
                min_dist = float('inf')
                for move in possible_moves:
                    new_x, new_y = x + move[0], y + move[1]
                    dists = squareform(pdist(np.vstack((centers, np.array([new_x, new_y]).reshape(1, 2)))))
                    idx = np.triu_indices(n + 1, k=1, m=n + 1)
                    dists[idx[0], idx[1]] = dists[idx[1], idx[0]]
                    dists[idx[1], n], dists[n, idx[1]] = dists[idx[1], idx[1]], dists[idx[1], idx[1]]
                    dists[idx[0], n], dists[n, idx[0]] = dists[idx[0], idx[0]], dists[idx[0], idx[0]]
                    if dists.min() > min_dist:
                        continue
                    min_dist = dists.min()
                    new_x, new_y = x + move[0], y + move[1]
            
            new_centers[i] = [new_x, new_y]
        
        return new_centers

    def initial_centers():
        return hexagonal_lattice_seed()

    def objective(x):
        return -np.sum(x[3 * n:])
    
    def constraints(x):
        pos = x[:3 * n].reshape(n, 3)
        r = x[3 * n:]
        cents = pos[:, :2]
        radious = r
        
        cons = np.empty(2 * n, float)
        cons[0:n] = cents[:, 0] - radious  # left
        cons[n:2 * n] = 1 - cents[:, 0] - radious  # right
        
        diffs = cents[:, None, :] - cents[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (radious[:, None] + radious[None, :])[mask]
        
        return np.concatenate([cons, pair_cons])
    
    best_radii = 0
    best_centers = np.zeros((n, 2))
    
    seeds = [hexagonal_lattice_seed(), hexagonal_lattice_seed(), hexagonal_lattice_seed()]
    
    for seed in seeds:
        centers = seed
        for _ in range(200):
            radii = linear_program(centers, np.full(n, 0.5))
            if np.isnan(radii).any():
                break
            if np.any(radii < 0):
                break
            
            new_centers = push_away(centers, radii)
            new_radii = linear_program(new_centers, np.full(n, 0.5))
            if np.isnan(new_radii).any():
                continue
            if np.any(new_radii < 0):
                continue
            
            if np.sum(new_radii) > np.sum(radii):
                centers, radii = new_centers, new_radii
        
        res = minimize(objective, np.concatenate((centers.flatten(), radii)), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        if res.success and np.sum(res.x[3 * n:]) > best_radii:
            best_radii = np.sum(res.x[3 * n:])
            best_centers = res.x[:3 * n].reshape(n, 2)

    return best_centers, res.x[3 * n:], best_radii
