import numpy as np
from scipy.optimize import linprog, minimize

def compute_optimal_radii(pos):
    n = 26
    A_ub = np.empty((4*n + 350, n))
    b_ub = np.empty(4*n + 350)
    
    for i in range(n):
        A_ub[i, i] = 1
        A_ub[n + i, i] = 1
        A_ub[n*2 + i, i] = -1
        A_ub[n*3 + i, i] = -1
        b_ub[i] = pos[i*2]
        b_ub[n + i] = 1 - pos[i*2]
        b_ub[n*2 + i] = pos[i*2 + 1]
        b_ub[n*3 + i] = 1 - pos[i*2 + 1]
    
    diffs = pos[:, None] - pos[None, :]
    dists = np.sqrt((diffs[:, :, 0]**2 + diffs[:, :, 1]**2))
    mask = np.triu(np.ones((n, n), bool), k=1)
    indices = np.stack(np.triu_indices(n, k=1))
    A_ub[4*n + indices[0], indices[1]] = 1
    A_ub[4*n + indices[0], indices[1]] += 1
    A_ub[4*n + indices[0], indices[1]] -=1
    A_ub[4*n + indices[0], indices[1]] -=1
    b_ub[4*n + indices[0]] = dists[mask] - 1e-9
    
    res = linprog(target=np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5))
    return -res.fun if res.success else None

def objective(pos_flat):
    pos = pos_flat.reshape(26, 2)
    radii = compute_optimal_radii(pos)
    if radii is None:
        return -1e9
    return -np.sum(radii)

def run_packing():
    best_sum = -np.inf
    for seed in range(50):
        init_centres = np.random.rand(52)
        res = minimize(objective, init_centres, method='SLSQP', bounds=[(0, 1)]*52, options={'maxiter': 2000})
        if res.success:
            pos = res.x.reshape(26, 2)
            radii = compute_optimal_radii(pos)
            if radii is not None and np.sum(radii) > best_sum:
                best_sum = np.sum(radii)
                best_center = res.x.reshape(26, 2)
                best_rad = radii
    return best_center, best_rad, best_sum
