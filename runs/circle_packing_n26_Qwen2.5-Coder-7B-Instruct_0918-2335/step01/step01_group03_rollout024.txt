import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import cvxpy as cp

def check_feasibility(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False
        if (centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1 or
            centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1):
            return False
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                return False
    return True

def optimize_radii(centers, initial_radii):
    n = centers.shape[0]
    r = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(sum(r))
    constraints = [r <= initial_radii]
    for i in range(n):
        constraints.append(r[i] <= 1 - centers[i, 0] if centers[i, 1] >= 0.5 else centers[i, 1])
        constraints.append(r[i] <= 1 - centers[i, 1] if centers[i, 0] >= 0.5 else centers[i, 0])
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(r[i] + r[j] <= np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)
    return r.value

def simulated_annealing(initial_speeds):
    best_sum = 0
    best_centers = None
    best_radii = None
    for _ in range(5):
        speeds = initial_speeds * np.lognormal(mean=0., sigma=0.4, size=26)
        edges = []
        for i in range(26):
            x, y = speeds[i] * np.random.rand(2)
            for j in range(i):
                edges.append((i, j, np.sqrt((speeds[i, 0] - speeds[j, 0]) ** 2 + (speeds[i, 1] - speeds[j, 1]) ** 2)))
        edges.sort(key=lambda x: x[2], reverse=True)
        centers = np.zeros((26, 2))
        used = np.zeros(26, dtype=bool)
        radii = np.zeros(26)
        for i, edge in enumerate(edges):
            u, v, _ = edge
            if used[u] or used[v]:
                continue
            used[u] = used[v] = True
            dx = (np.random.rand() - 0.5) * 0.01
            dy = (np.random.rand() - 0.5) * 0.01
            centers[u] = speeds[u] * np.array([dx, dy])
            centers[v] = speeds[v] * np.array([1 - dx, 1 - dy])
            radii[u] = speeds[u] * min(np.min(centers[u]), np.min(1 - centers[u]))
            radii[v] = speeds[v] * min(np.min(centers[v]), np.min(1 - centers[v]))
        if check_feasibility(centers, radii):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers
                best_radii = radii
    if best_radii is None:
        return np.zeros(26), np.zeros(26)
    final_radii = optimize_radii(best_centers, best_radii)
    return best_centers, final_radii
