import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (3 * n)
    x0 = np.random.rand(3 * n)

    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)
        r = v[52:]

        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")

def generate_center_families(n=26):
    grid_centers = np.array([
        (0.5, 0.5),
        *np.meshgrid(np.linspace(0.1, 0.9, 4), np.linspace(0.1, 0.9, 4)),
        (0.1, 0.1),
        (0.9, 0.1),
        (0.1, 0.9),
        (0.9, 0.9)
    ])
    hex_centers = generate_hexagonal_centers(n)
    hybrid_centers = generate_hybrid_centers(n)
    random_centers = np.random.rand(n, 2)
    return grid_centers, hex_centers, hybrid_centers, random_centers

def generate_hexagonal_centers(n=26):
    radius = 0.1
    rows = 13
    cols = 6
    centers = []
    y = 0.5 - radius
    for i in range(rows):
        row = (np.arange(cols) * 2 * radius + radius) * np.cos(np.pi / 6)
        centers.extend((row + 0.5 - radius, y + radius).T.tolist())
        y += np.sin(np.pi / 6)
    return np.array(centers[:n])

def generate_hybrid_centers(n=26):
    centers = np.array([
        (0.5, 0.5)
    ])
    for i in range(4):
        centers = np.vstack((centers, np.array([(0.25, 0.25 + i * 0.25)])))
    for i in range(4):
        centers = np.vstack((centers, np.array([(0.75, 0.25 + i * 0.25)])))
    for i in range(18):
        centers = np.vstack((centers, np.array([np.random.rand(2)] * 2)))
    return centers[:n]

def simulate_inflation(centers, speeds):
    n = centers.shape[0]
    active = np.arange(n)
    stops = np.zeros(n)
    while active.size > 0:
        diffs = centers[:, None, :] - centers[active][:, np.newaxis, :]
        dists = np.sqrt((diffs ** 2).sum(axis=-1))
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        t_pairs = dists[mask] / (speeds[active][:, None] + speeds[active][None, :])[mask]
        t_boundary = np.min(np.column_stack((centers[active, 0], 1 - centers[active, 0], 
                                            centers[active, 1], 1 - centers[active, 1])) / speeds[active], axis=1)
        t_min = np.min(np.concatenate((t_pairs.ravel(), t_boundary)), axis=0)
        stop_indices = np.where((t_pairs == t_min.reshape(-1, 1, 1) + 1e-12)[0, :, 0] | (t_boundary == t_min))
        stops[stop_indices] = t_min
        active = np.setdiff1d(active, stop_indices)
    radii = speeds * stops[:, np.newaxis]
    return centers, radii, np.sum(radii)

def speed_assignment(centers):
    dist_to_boundary = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    speeds = 1 + 2 / (dist_to_boundary + 1e-6)
    random_mu, random_sigma = 0, 0.4
    random_multipliers = np.random.lognormal(random_mu, random_sigma, centers.shape[0])
    return speeds * random_multipliers

def refine_packing(centers, radii, bounds):
    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)
        r = v[52:]

        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        diffs = pos[:, None, :] - pos[None, :, :]
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=np.vstack((centers.ravel(), radii.ravel())), method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 2000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        refined_centers = res.x[:52].reshape(n, 2)
        refined_radii = res.x[52:]
        return refined_centers, refined_radii
    else:
        return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers_families = generate_center_families()
    best_centers, best_radii, best_sum_radii = None, None, 0

    for centers in centers_families:
        speeds = speed_assignment(centers)
        centers, radii, sum_radii = simulate_inflation(centers, speeds)
        refined_centers, refined_radii = refine_packing(centers, radii, [(0, 1)] * (3 * n))
        if sum_radii > best_sum_radii:
            best_centers, best_radii, best_sum_radii = refined_centers, refined_radii, sum_radii

    if not best_centers is None and not best_radii is None:
        valid, message = validate_packing(best_centers, best_radii)
        if not valid:
            raise RuntimeError(message)

    return best_centers, best_radii, best_sum_radii

# Call the function and print the results (for validation)
centers, radii, sum_radii = run_packing()
print(f"Centers: {centers}, Radii: {radii}, Sum of Radii: {sum_radii}")
