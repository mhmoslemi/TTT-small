import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def simulate(centers, speeds, time):
        n = len(centers)
        radii = np.zeros(n)
        active = set(range(n))
        t_stop = [None] * n
        distances = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :]) ** 2, axis=2))

        while active:
            t_ij = distances[active][:, active] / (speeds[active] + speeds[active][:, None])
            t_edge = np.minimum(centers[active, 0], 1 - centers[active, 0], centers[active, 1], 1 - centers[active, 1]) / speeds[active]
            t_min_indices = np.unravel_index(np.argmin(np.minimum(t_ij, t_edge[:, None])), (len(active), len(active)))
            t_min = np.min(np.minimum(t_ij, t_edge[:, None]))
            stop_set = set()
            for i, j in zip(t_min_indices[0], t_min_indices[1]):
                if np.isclose(t_edge[active[i]], t_min) or np.isclose(t_ij[active[i], active[j]], t_min, atol=1e-12):
                    stop_set.add(active[i])

            for i in stop_set:
                t_stop[i] = t_min
                active.remove(i)

            radii[stop_set] = speeds[stop_set] * t_stop[stop_set]

        return centers, radii

    def evaluate(centers, speeds):
        n = len(centers)
        radii = np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                radii[i] = radii[j] = 0.5 * (dist - (radii[i] + radii[j]))
        return np.sum(radii)

    def run_simulation(family):
        n = 26
        bounds = [(0, 1)] * (3 * n)
        np.random.seed(0)
        centers = family
        radii = np.random.rand(n)
        speeds = np.zeros(n)
        for i in range(n):
            d = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            speeds[i] = 1 + 2 / (d + 1e-6)
        speeds = np.random.lognormal(mean=0., sigma=0.4, size=n) * speeds
        centers, radii = simulate(centers, speeds, 1.0)

        return centers, radii, evaluate(centers, speeds)

    families = [
        np.array([[i / 5, j / 5] for i in range(6) for j in range(6)] + [[1/2, 1/2]]),
        np.array([[i / 5, j / 5] for i in range(5) for j in range(5) if i % 2 == j % 2] + [[1/2, 1/2]] + [[i / 5, j / 5] for i in range(5) for j in range(5) if i % 2 != j % 2]),
        0.1 * np.random.rand(26, 2) + 0.5,
        np.array([[i / 4, j / 5] for i in range(4) for j in range(5)] + [[i / 5, j / 4] for i in range(5) for j in range(4)] + [[1/2, 1/2]] + [[0.25, 0.25], [0.75, 0.75], [0.25, 0.75], [0.75, 0.25]]),
    ]

    best_centers, best_rgedi, best_radius = None, 0, 0
    for family in families:
        centers, radii, radius = run_simulation(family)
        if radius > best_radius:
            best_centers, best_rgedi, best_radius = centers, radii, radius

    x0 = np.concatenate([best_centers.flatten(), best_rgedi])
    bounds = [(0, 1)] * (3 * n)
    cons = ({'type': 'ineq', 'fun': lambda v, c=best_centers, r=best_rgedi: (np.sqrt(np.sum((c - c[None, :]) ** 2, axis=2)) - (r + r[:, None]) + 1e-12)})
    res = minimize(lambda v tej: -np.sum(v[n:]), x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9})
    
    if not res.success or res.fun > 0:
        return best_centers, best_rgedi, best_radius

    best_centers, best_rgedi, best_radius = res.x[:n*2].reshape(n, 2), res.x[n:], -res.fun
    
    return best_centers, best_rgedi, best_radius
