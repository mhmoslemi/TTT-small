import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v):
    r = v[52:]
    return -(np.sum(r[:-1]) + r[-1])

def constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    return np.concatenate([cons, dists[mask].flatten() - (r[:, None] + r[None, :]).flatten()])

def penalty(v):
    penalties = np.maximum(constraints(v), 0)
    return 1e6 * np.sum(penalties ** 2)

def obj_with_penalty(v):
    return -(np.sum(v[52:]) + penalty(v))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * (n + 1)
    init = np.random.rand(2 * n + n + 1)
    init[52:] = np.full(n + 1, 0.02)

    best_sol = None
    best_sum = -np.inf

    for _ in range(3):
        res_de = differential_evolution(obj_with_penalty, bounds, popsize=15, maxiter=500, tol=1e-6, strategy='best1bin', disp=False)
        if res_de.success and constraints(res_de.x).min() >= -1e-10:
            current_sum = -res_de.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_sol = res_de.x

    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    dists = np.sqrt(np.sum((best_sol[:52].reshape(n, 1) - best_sol[:52]) ** 2, axis=1))
    mask = np.logical_and(dists < 1e-12, np.triu(np.ones((n, n), bool), k=1))
    overlap_idx = np.where(mask)[0]
    if len(overlap_idx) > 0:
        best_sol[52 + overlap_idx[0]] -= (dists[mask][0] - 1e-12) / 2

    res_slsqp = minimize(fun=obj, x0=best_sol[52:], method='SLSQP',
                         bounds=[(0, 0.5)] * (n + 1), constraints={'type': 'ineq', 'fun': constraints},
                         options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})

    if res_slsqp.success and constraints(best_sol).min() >= -1e-10:
        centers = best_sol[:52].reshape(26, 2)
        radii = np.array(list(res_slsqp.x)[:-1] + [res_slsqp.x[-1]])
        return centers, radii, -res_slsqp.fun

    centers = best_sol[:52].reshape(26, 2)
    radii = np.array(list(best_sol[52:])[:-1] + [best_sol[52][-1]])
    return centers, radii, best_sum
