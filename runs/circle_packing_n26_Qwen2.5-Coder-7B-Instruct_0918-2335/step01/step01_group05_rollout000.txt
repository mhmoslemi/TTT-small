import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v, penalty):
    r = v[52:]
    penalty_val = np.sum(np.maximum(0, penalty(v)[2:]))
    return -np.sum(r) - penalty_factor * penalty_val

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(78, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    return cons

def dist_constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt(np.sum((diffs**2).sum(axis=2), axis=0))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return dists[mask] - (r[:,None] + r[None,:])[mask]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    best_sol = None
    best_sum = -np.inf

    penalty_factor = 1e6
    
    for _ in range(20):
        def penalty(v):
            return constraints(v), dist_constraints(v)
        
        res = differential_evolution(fun=obj, bounds=bounds, args=(penalty,), popsize=15, maxiter=500, disp=False)
        if res.success:
            current_sum = -obj(res.x, penalty)
            if current_sum > best_sum:
                best_sum = current_sum
                best_sol = res.x
    
    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = best_sol[:52].reshape(n, 2)
    radii = best_sol[52:]
    return centers, radii, best_sum
