import numpy as np
from scipy.optimize import differential_evolution, minimize

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n)
    init[52:] = np.full(n, 0.02)
    best_sol = None
    best_sum = -np.inf

    for _ in range(20):
        popsize = 15
        maxiter = 500
        pen = 1e6
        penalty = lambda v: pen * np.sum(np.maximum(0, -constraints(v)))
        bounds = bounds
        result_DE = differential_evolution(func=obj, bounds=bounds, popsize=popsize, 
                                          strategy='best2bin', maxiter=maxiter, 
                                          args=(penalty,), callback=None)
        if result_DE.success:
            radius_sum_DE = -result_DE.fun
            if radius_sum_DE > best_sum:
                best_sum = radius_sum_DE
                init = result_DE.x
        
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 4000, 
                                                                                    'ftol': 1e-9, 
                                                                                    'disp': True})
        if res.success and constraints(res.x).min() >= -1e-10:
            current_sum = -obj(res.x)
            if current_sum > best_sum:
                best_sum = current_sum
                best_sol = res.x
    
    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
    
    centers = best_sol[:52].reshape(n, 2)
    radii = best_sol[52:]
    return centers, radii, best_sum

result_centers, result_radii, result_sum = run_packing()
print("Centers:")
print(result_centers)
print("Radii:")
print(result_radii)
print("Total Radius Sum:", result_sum)
