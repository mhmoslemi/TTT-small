import numpy as np
from scipy.optimize import linprog, differential_evolution, SLSQP
from scipy.spatial.distance import pdist, squareform

def linear_program(r, pos):
    n = len(r)
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - pos[i, 0] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - pos[i, 1] - r[i]})
    flat_cons = [con['fun'] for con in cons for _ in range(n)]
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(pos[i] - pos[j])
            A_ub.append([-1, -1])
            b_ub.append(dist - (r[i] + r[j]))
    return linprog(flat_cons, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * len(flat_cons), method='highs').fun

def fitness_function(x):
    pos = x[:26].reshape(26, 2)
    r = linear_program(pos, pos)
    return -r

def run_packing():
    bounds_pos = [(0, 1)] * 52
    bounds_radii = [(0, 0.5)] * 26
    bounds = bounds_pos + bounds_radii
    
    low_bound = np.zeros(78)
    high_bound = np.full(78, 1.5)
    init_bounds = bounds(low_bound, high_bound)
    
    result = differential_evolution(fitness_function, init_bounds, popsize=50, maxiter=200, seed=42)
    
    best_pos = result.x[:52].reshape(26, 2)
    best_r = -result.fun
    
    cons = []
    for i in range(26):
        cons.append({'type': 'ineq', 'fun': lambda best_r, i=i: best_r[i]})
        cons.append({'type': 'ineq', 'fun': lambda best_r, i=i: 1 - best_pos[i, 0] - best_r[i]})
        cons.append({'type': 'ineq', 'fun': lambda best_r, i=i: best_r[i]})
        cons.append({'type': 'ineq', 'fun': lambda best_r, i=i: 1 - best_pos[i, 1] - best_r[i]})
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.linalg.norm(best_pos[i] - best_pos[j])
            cons.append({'type': 'ineq', 'fun': lambda best_r, i=i, j=j: dist - (best_r[i] + best_r[j])})
    
    optimized = SLSQP(linear_program, [np.concatenate([best_pos.ravel(), best_r])], args=(best_pos,), bounds=bounds, constraints=cons, options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
    centers = optimized.x[:52].reshape(26, 2)
    radii = optimized.x[52:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
