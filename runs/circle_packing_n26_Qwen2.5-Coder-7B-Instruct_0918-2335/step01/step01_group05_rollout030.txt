import numpy as np
from scipy.optimize import minimize
from scipy.optimize.differential_evolution import differential_evolution

def obj(v):
    r = v[52:]
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:,0] - r
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r
    diffs = pos[:,None,:] - pos[None,:,:]
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask = np.triu(np.ones((26,26), bool), k=1)
    return np.concatenate([cons, dists[mask] - (r[:,None] + r[None,:])[mask]])

def penalty_fun(c, v):
    return 1e6 * np.sum(np.maximum(0, -c))

def penalty_obj(v):
    c = constraints(v)
    return -np.sum(v[52:]) + penalty_fun(c, v)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0,1)] * 2 * n + [(0, 0.5)] * n
    init = np.random.rand(2 * n + n) * 0.5
    init[52:] = np.full(n, 0.02)
    
    de_res = differential_evolution(penalty_obj, bounds, popsize=15, maxiter=500)
    best_sol = de_res.x
    
    res = minimize.fun(obj, best_sol, method='SLSQP', bounds=bounds, 
                      constraints={'type': 'ineq', 'fun': constraints},
                      options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})
    
    if res.success and constraints(res.x).min() >= -1e-10:
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        return centers, radii, -obj(res.x)
    else:
        return de_res.x[:52].reshape(n, 2), de_res.x[52:], -penalty_fun(constraints(de_res.x), de_res.x)
