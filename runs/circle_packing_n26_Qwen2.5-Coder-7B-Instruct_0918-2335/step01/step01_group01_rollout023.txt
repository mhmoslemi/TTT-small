import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial import cKDTree

def lp_solver(positions):
    n = positions.shape[0]
    r = positions[:, 2:]

    A = np.zeros((104 + 325, 26))
    b = np.zeros(104 + 325)

    # Boundary constraints
    for i in range(n):
        A[2*i] = np.hstack([np.zeros(i), 1, np.zeros(n-1-i)])
        A[2*i+1] = np.hstack([np.zeros(i), -1, np.zeros(n-1-i)])
        b[2*i] = positions[i, 0]
        b[2*i+1] = 1 - positions[i, 0]

    # Contact constraints
    pairs = np.array(list(cKDTree(positions[:, :2]).query_pairs(r=2)))
    for pair in pairs:
        i, j = pair
        A[104 + pairs.shape[0] * 2 * (i + j * n)] = np.hstack([np.zeros(i), 1, np.zeros(j-1), -1, np.zeros(n-j-1)])
        b[104 + pairs.shape[0] * 2 * (i + j * n)] = np.linalg.norm(positions[i, :2] - positions[j, :2]) - positions[i, 2] - positions[j, 2]

    result = linprog(np.sum(r, axis=1), A_ub=A, b_ub=b, bounds=[(0, None)] * 26, method='highs', tol=1e-9)
    if result.success:
        r_opt = result.x
    else:
        r_opt = np.zeros(26)

    return np.sum(r_opt)

def cost_function(positions):
    infeasible = np.any((positions[:, :2] < 0) | (positions[:, :2] > 1)) or np.any(positions[:, 2:] < 0)
    if infeasible:
        return -1e10
    else:
        return -lp_solver(positions)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NP = 40
    inertia = 0.5
    cognitive = 1.5
    social = 1.5
    T0 = 0.05
    cooling_schedule = lambda T, it: T * 0.99 ** it
    T = T0
    max_iterations = 800
    sigma = 0.02
    p_anneal = 0.2
    max_radius = 0.5

    np.random.seed(42)
    initial seed_values = [
        np.array([[0.5, 0.5, 0.16], [0.14, 0.14, 0.14], [0.86, 0.14, 0.14], [0.86, 0.86, 0.14], [0.14, 0.86, 0.14], [0.5, 0.86, 0.22],
                   [0.86, 0.5, 0.22], [0.14, 0.5, 0.22], [0.5, 0.14, 0.28], [0.86, 0.14, 0.28], [0.86, 0.86, 0.28], [0.14, 0.86, 0.28], [0.14, 0.14, 0.28],
                   [0.14, 0.14 + max_radius * (1 - 0.14) - max_radius / 2, 0.14 + max_radius * (1 - 0.14) - max_radius / 2],
                   [0.14 + 2 * max_radius, 0.14, 0.14], [0.86 - 2 * max_radius, 0.14, 0.14], [1 - max_radius * (1 - 0.14), 0.14, 0.14],
                   [0.86, 0.14 + max_radius * (1 - 0.14) - max_radius / 2, 0.14 + max_radius * (1 - 0.14) - max_radius / 2],
                   [0.86, 0.14 + max_radius * (1 - 0.14) - max_radius / 2, 0.14 + max_radius * (1 - 0.14) + max_radius / 2],
                   [0.86, 0.14 + 2 * max_radius, 0.14],
                   [0.86 - max_radius * (1 - 0.14), 0.14 + 2 * max_radius], [0.14, 0.14 + 2 * max_radius], [0.14, 0.14 + max_radius * (1 - 0.14) + max_radius / 2]]),
        ... (other seed patterns)
    ]

    particles = [seed + sigma * np.random.normal(size=(26, 2)) for seed in initial_seed_values]
    pbest = particles.copy()
    pbest_vals = [-np.inf] * NP
    gpbest = min(pbest_vals)
    gpbest_pos = None
    for _ in range(max_iterations):
        for i in range(NP):
            rp, rc = np.random.rand(2)
            if rp < p_anneal:
                velocity = sigma * np.random.normal(size=(26, 2))
            else:
                velocity = inertia * particles[i][30:] + cognitive * rp * (pbest[i] - particles[i]) + social * rc * (gpbest_pos - particles[i])
            new_pos = particles[i][:30] + velocity
            new_pos[:, :2] = np.clip(new_pos[:, :2], 0, 1)
            particles[i][:30], particles[i][30:] = new_pos, new_pos[:, 2:]
            fitness = cost_function(particles[i])
            if fitness > pbest_vals[i]:
                pbest[i] = new_pos
                pbest_vals[i] = fitness
            if fitness > gpbest:
                gpbest = fitness
                gpbest_pos = new_pos

        T = cooling_schedule(T0, _)
    best_radius_sum = -gpbest
    return gpbest_pos[:, :2], gpbest_pos[:, 2:], best_radius_sum
