import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import pdist, squareform

def lp_radius(v, centers):
    r = v[26:]
    n = centers.shape[0]
    A_ub = np.vstack([r, -r, 
                        centers[:, 0] - r, 
                        1 - centers[:, 0] - r, 
                        centers[:, 1] - r, 
                        1 - centers[:, 1] - r]).T
    b_ub = np.hstack([np.zeros(n), 
                       centers[:, 0], 
                       1 - centers[:, 0], 
                       centers[:, 1], 
                       1 - centers[:, 1]])
    dists = squareform(pdist(centers))
    np.fill_diagonal(dists, np.inf)
    A_ub = np.vstack([A_ub, dists - (r[:, None] + r[None, :])]).T
    b_ub = np.hstack([b_ub, np.zeros(n * (n - 1) // 2)])
    res = minimize(lambda r: -np.sum(r), r, method='highs', bounds=[(0, 0.5)] * n, constraints={'type': 'ineq', 'fun': lambda r: A_ub @ r - b_ub})
    return -np.sum(res.x)

def diff_eval(v):
    centers = v[:52].reshape(26, 2)
    return -lp_radius(v, centers)

def run_packing():
    np.random.seed(42)
    bounds = [(0, 1)] * 52
    bounds += [(0, 0.5)] * 26
    result = differential_evolution(diff_eval, bounds, popsize=50, maxiter=200)
    best_centers = result.x[:52].reshape(26, 2)
    v = np.hstack([best_centers.flatten(), best_centers[:, 0]])
    opt = minimize(lambda v: -lp_radius(v, v[:52].reshape(26, 2)), v, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda v: v[:52].reshape(26, 2), 'type': 'ineq', 'fun': lambda v: v[52:]}, options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
    centers = opt.x[:52].reshape(26, 2)
    radii = opt.x[52:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
