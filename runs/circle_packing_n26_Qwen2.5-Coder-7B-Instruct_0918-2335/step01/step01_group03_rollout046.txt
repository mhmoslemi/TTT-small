import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution, constraints as con
from scipy.spatial.distance import euclidean

def generate_positions(layout):
    if layout == 'hex':
        r = 0.2
        center = (np.arange(3) - 1) / 4 + 0.25
        centers = np.array([[x + y * np.sqrt(3) * 0.5, y] for x in center for y in center if x + y <= 2])
        centers = centers[:26]
    elif layout == '5x5':
        center = np.arange(5) / 4 + 0.25
        centers = np.array([[x, y] for x in center for y in center])
    elif layout == 'hybrid':
        corners = np.array([[0.2, 0.2], [0.2, 0.8], [0.8, 0.2], [0.8, 0.8]])
        edges = np.array([[0.5, 0.2], [0.2, 0.5], [0.5, 0.8], [0.8, 0.5]])
        inner = np.array([[x, y] for x in np.arange(4) / 8 + 0.125 for y in np.arange(3) / 8 + 0.125])
        centers = np.vstack((corners, edges, inner))
    elif layout == 'random':
        centers = np.random.rand(26, 2)
    return centers

def lp_optimize_radii(centers):
    n = centers.shape[0]
    A = np.zeros((n * 4 + n * n * 2, n))
    b = np.zeros(n * 4 + n * n * 2)
    
    for i in range(n):
        x, y = centers[i]
        A[i * 4 + 0, i] = 1
        A[i * 4 + 1, i] = -1 * x
        b[i * 4 + 1] = x
        A[i * 4 + 2, i] = -1 * y
        b[i * 4 + 2] = y
        A[i * 4 + 3, i] = -1 * r
        b[i * 4 + 3] = r
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = euclidean(centers[i], centers[j])
            A[n * 4 + i * n + j, i] = 1
            A[n * 4 + i * n + j, j] = -1
            b[n * 4 + i * n + j] = dist - r * 2
    
    result = linprog([-1] * n, A_eq=A, b_eq=b, bounds=(0, None), method='highs')
    if result.success and result.status == 0:
        return result.x
    return None

def perturb(centers, sigma):
    return np.clip(centers + np.random.normal(0, sigma, centers.shape), 0, 1)

def evaluate_solution(sol, radii, centers):
    sum_r = radii.sum()
    for i in range(radii.shape[0]):
        x, y = centers[i]
        if x + radii[i] > 1 or x - radii[i] < 0 or y + radii[i] > 1 or y - radii[i] < 0:
            sum_r -= 1e-6 * radii[i]
        for j in range(i + 1, radii.shape[0]):
            if dist(centers[i], centers[j]) < radii[i] + radii[j]:
                sum_r -= 1e-6 * (radii[i] + radii[j] - dist(centers[i], centers[j]))
    return sum_r

def refine_with_sa(centers, radii):
    n = centers.shape[0]
    objective = lambda x: -sum(x[n:]) + 1e6 * np.sum([max(0, x[i + n] + x[j + n] - dist(centers[i], centers[j]) - 1e-6) for i in range(n) for j in range(i + 1, n)]) + 1e6 * np.sum([max(0, x[i + n] - 0, x[i + n] - 1, x[i + n] - 0, x[i + n] - 1) for i in range(n)])
    bounds = [(0, None)] * n + [(0, 0.5)] * n
    x0 = np.concatenate((centers.flatten(), radii))
    res = differential_evolution(objective, bounds=bounds, strategy='best2bin', popsize=10, tol=1e-8, maxiter=10000, callback=lambda _: KeyboardInterrupt("optimal solution found", res))
    if res.message == "maxiter":
        return res.x[:n].reshape(-1, 2), res.x[n:]
    return x0[:n].reshape(-1, 2), x0[n:]

def dist(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))

def run_packing():
    n = 26
    layouts = ['hex', '5x5', 'hybrid', 'random']
    N_ATTEMPTS = 5
    sigma_perturb = 0.03
    SA_iterations = 10000
    lambda1 = 1e6
    lambda2 = 1e6
    T0 = 1e-1
    cool_rate = 0.995

    best_sum_r = 0
    best_centers = []
    best_radii = []

    for layout in layouts:
        for attempt in range(N_ATTEMPTS):
            centers0 = generate_positions(layout)
            centers = perturb(centers0, sigma_perturb)

            radii = lp_optimize_radii(centers)
            if radii is None:
                continue

            sum_r = radii.sum()
            objective = lambda x: -sum(x[n:]) + lambda1 * np.sum([max(0, x[i + n] + x[j + n] - dist(centers[i], centers[j]) - 1e-6) for i in range(n) for j in range(i + 1, n)]) + lambda2 * np.sum([max(0, x[i + n] - 0, x[i + n] - 1, x[i + n] - 0, x[i + n] - 1) for i in range(n)])
            bounds = [(0, None)] * n + [(0, 0.5)] * n
            x0 = np.concatenate((centers.flatten(), radii))
            res = differential_evolution(objective, bounds=bounds, strategy='best2bin', popsize=10, tol=1e-8, maxiter=SA_iterations)
            if res.message == "maxiter":
                sum_r_refuned = evaluate_solution(res.x[:n].reshape(-1, 2), res.x[n:], res.x[:n].reshape(-1, 2))
                if sum_r_refuned > best_sum_r:
                    best_sum_r = sum_r_refuned
                    centers_optimal = res.x[:n].reshape(-1, 2)
                    radii_optimal = res.x[n:]
            else:
                sum_r_refuned = evaluate_solution(x0[:n].reshape(-1, 2), x0[n:], centers0)
                if sum_r_refuned > best_sum_r:
                    best_sum_r = sum_r_refuned
                    centers_optimal = centers0
                    radii_optimal = radii

    return centers_optimal, radii_optimal, best_sum_r
