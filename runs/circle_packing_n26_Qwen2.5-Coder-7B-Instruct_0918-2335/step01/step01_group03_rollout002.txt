import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
from scipy.spatial.distance import cdist

N_CIRCLES = 26
bounds = [(0, 1)] * (3 * N_CIRCLES)
penalty_weight = 1e6

def lp_optimize_radii(centers):
    n = len(centers)
    c = np.ones(n)
    A = np.block([
        np.eye(n),
        -np.eye(n),
        -np.eye(n),
        -np.eye(n),
        np.zeros((n, 2))
    ])
    b = np.zeros(4 * n)
    A_eq = cdist(centers, centers, 'euclidean').flatten().reshape(n, n) - 2 * np.eye(n)
    b_eq = np.zeros(n)
    bounds_r = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds_r, method='highs')
    if res.success:
        return np.maximum(0, np.minimum(res.x[:n], np.minimum(centers[:, 0], centers[:, 1])[:, None]))
    else:
        return None

def evaluate(solution):
    centers = solution[:N_CIRCLES].reshape(-1, 2)
    radii = solution[N_CIRCLES:] / 2 ** np.arange(N_CIRCLES)
    n = len(centers)
    overlaps = np.sum((cdist(centers, centers, 'euclidean') - 2 * radii) < 0) / 2
    boundary = np.sum(np.minimum(radii, np.minimum(centers[:, 0], centers[:, 1])) < min(radii))
    return -np.sum(radii) + penalty_weight * (overlaps + boundary), overlaps, boundary

def run_packing():
    best_solution = None
    best_value = -np.inf
    position_generators = [
        lambda _: np.random.rand(N_CIRCLES, 3),
        lambda _: 0.5 * np.hstack([np.ones((N_CIRCLES, 1)), np.random.rand(N_CIRCLES, 2)]),
        lambda _: 0.25 * np.hstack([np.ones((N_CIRCLES, 1)), np.random.rand(N_CIRCLES, 2)])
    ]
    for generator in position_generators:
        for _ in range(5):
            centers = generator()[1:N_CIRCLES+1, :2]
            radii = lp_optimize_radii(centers)
            if radii is None:
                continue
            sum_radii = np.sum(radii)
            if sum_radii > best_value:
                best_value = sum_radii
                best_solution = centers, radii
            sol = np.hstack((centers.flatten(), radii))
            res = differential_evolution(evaluate, bounds, args=(sol,), strategy='best1bin', popsize=20, maxiter=20000, tol=1e-8, seed=42)
            sum_radii = -res.fun[0]
            if sum_radii > best_value:
                best_value = sum_radii
                best_solution = res.x[:N_CIRCLES].reshape(-1, 2), 2 * res.x[N_CIRCLES:]
    return best_solution
