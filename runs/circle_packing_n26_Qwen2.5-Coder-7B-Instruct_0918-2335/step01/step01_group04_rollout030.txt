def sa_optimize(initial_centers):

    max_iter = 500
    cool_rate = 0.95
    min_step = 1e-5
    best_centers = initial_centers.copy()

    best_obj = -best_radii.sum()

        epsilon = np.random.normal(scale=0.1)
