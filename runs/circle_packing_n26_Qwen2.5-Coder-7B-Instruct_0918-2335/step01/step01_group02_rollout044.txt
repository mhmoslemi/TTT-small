import numpy as np
from scipy.optimize import linprog, NonlinearConstraint, minimize
from scipy.spatial.distance import cdist

# Define potential contact graph configurations
graph_configs = []
for i in range(1, 6):
    for j in range(1, 6):
        rectangles = [(x * 0.25, y * 0.25) for x in range(i) for y in range(j)]
        num_rectangles = len(rectangles)
        if num_rectangles > 24:
            rectangles[-1] = (i * 0.25 - 0.1, j * 0.25 - 0.1)
        graph_configs.append(rectangles)

# Function to get the contact graph matrix for a set of circles
def get_contact_graph(circles):
    n = len(circles)
    contact_graph = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(circles[i] - circles[j])
            rsum = circles[i][2] + circles[j][2]
            if dist - rsum < 1e-6:
                contact_graph[i, j] = 1
                contact_graph[j, i] = 1
    return contact_graph

# Function to check if a graph is valid
def is_valid_graph(graph, circles):
    n = len(circles)
    for i in range(n):
        if np.sum(graph[i, :]) != 1:
            return False
        circle = circles[i]
        if circle[0] - circle[2] < 0 or circle[0] + circle[2] > 1:
            return False
        if circle[1] - circle[2] < 0 or circle[1] + circle[2] > 1:
            return False
    return True

# Function to get the wall constraint matrix for a set of circles
def get_wall_constraints(circles):
    n = len(circles)
    wall_constraints = []
    for i in range(n):
        circle = circles[i]
        r = circle[2]
        if circle[0] < r:
            wall_constraints.append([0, 0, -1, 1, 0])
        elif circle[0] > 1 - r:
            wall_constraints.append([0, 1, -1, 0, 0])
        if circle[1] < r:
            wall_constraints.append([1, 0, -1, 1, 0])
        elif circle[1] > 1 - r:
            wall_constraints.append([0, 0, 1, -1, 0])
    return np.array(wall_constraints), np.ones(len(wall_constraints))

# Function to solve LP for a given graph
def solve_lp(contact_graph, wall_constraints):
    n = contact_graph.shape[0]
    c = np.ones(n)
    A_eq = np.hstack((contact_graph, -np.eye(n), wall_constraints[:, :2]))
    b_eq = np.hstack((np.ones(n), -contact_graph.sum(axis=1), wall_constraints[:, 2]))
    A_ub = np.vstack((np.eye(n), -np.eye(n)))
    b_ub = np.array([0.5] * n + [0])
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        circles = np.vstack((np.cumsum(np.insert(res.x, 0, 0)).reshape(n, 1),
                             np.zeros((n, 1)),
                             np.ones((n, 1)) * res.x)).T
        circles[0, 1] = 0
        circles[1, 0] = 0
        circles[1, 1] = 1
        circles[2, 0] = 1
        circles[2, 1] = 0
        circles[3, 0] = 0
        circles[3, 1] = 1
        circles[4, 0] = 0.5
        circles[4, 1] = 0.5
        circles = circles[circles[:, 2].argsort()[::-1]]
        circles = np.vstack((np.zeros((25, 1)), circles[1:, 1:].copy())).T
        for i in range(n):
            circles[i, 2] = min(circles[i, 2], circles[i, 0] + circles[i, 1] < 1)
        if is_valid_graph(get_contact_graph(circles), circles):
            return circles[:, 2], circles.sum(axis=1).astype(float)[0]
    return None, None

# Function to find the best packing
def run_packing():
    best_sum = 0
    best_circles = None
    best_r = None
    for graph in graph_configs:
        contact_graph = get_contact_graph(graph)
        circles, sum_r = solve_lp(contact_graph, None)
        if circles is not None and sum_r > best_sum:
            best_sum = sum_r
            best_circles = circles
            best_r = np.array(best_sum - sum_r)
    if best_circles is None:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    circles = np.vstack((best_circles, best_circles * 2))
    circles = circles[circles[:, 2].argsort()[::-1]]
    circles = circles[circles[:, 0] < 1]
    circles = circles[circles[:, 1] < 1]
    circles = circles[circles[:, 0] > 0]
    circles = circles[circles[:, 1] > 0]
    circles = circles[:26]
    circles[:, 2] = np.minimum(circles[:, 2], 0.5)
    circles[:, 2] = np.maximum(circles[:, 2], 0)
    circles = circles[circles[:, 0] < 1]
    circles = circles[circles[:, 1] < 1]
    circles = circles[circles[:, 0] > 0]
    circles = circles[circles[:, 1] > 0]
    return circles[:, :2], circles[:, 2], circles.sum(axis=1).astype(float)[0]

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
