from scipy.optimize import linprog, minimize

def contact_graph_constraints(contact_matrix, radii):
    n = len(radii)
    A_leaf = contact_matrix @ np.eye(n)
    A_pair = 2 * (contact_matrix - np.eye(n))
    b_pair = np.zeros(n * n - n)
    x_bounds = [(0, 0.5)] * n
    return A_leaf, A_pair, b_pair, x_bounds

def evaluate_graph(contact_matrix, parent_indices):
    n = len(contact_matrix)
    while np.all(parent_indices == -1):
        parent_indices = np.argmax(contact_matrix[:, parent_indices], axis=1)
        parent_indices[np.where(contact_matrix[:, parent_indices] == parent_indices[:, np.newaxis])] = -1
    
    radii = np.empty(n)
    for i in range(n):
        path_radii = radii[parent_indices.tolist()]
        x_i = np.sum(path_radii * np.cos(np.linspace(0, 2 * np.pi * (np.arange(path_radii.shape[0]) / path_radii.shape[0]), path_radii.shape[0])))
        y_i = np.sum(path_radii * np.sin(np.linspace(0, 2 * np.pi * (np.arange(path_radii.shape[0]) / path_radii.shape[0]), path_radii.shape[0])))
        radii[i] = np.linalg.norm(np.array([x_i, y_i]))
    
    A_leaf, A_pair, b_pair, x_bounds = contact_graph_constraints(contact_matrix, radii)
    result = linprog(-radii, A_ub=np.vstack([A_leaf, A_pair]), b_ub=np.hstack([np.zeros(n), b_pair]), bounds=x_bounds, method='highs')
    if result.success:
        return sum(abs(result.x)), radii, result.x, A_leaf, A_pair, b_pair
    else:
        return 0, None, None, None, None, None
