import numpy as np
from scipy.optimize import minimize, basinhopping

def build_constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]

    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r

    diffs = pos[:, None, :] - pos[None, :, :]
    d2 = np.sum(diffs ** 2, axis=2)
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = d2[mask] - (r[:, None] + r[None, :])[mask] ** 2
    return np.concatenate([cons, pair_cons])

def obj(v):
    return -np.sum(v[52:])

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(26):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(26):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break

def run_packing(max_iter=4000, ftol=1e-12, popsize=10, nrestarts=100):
    N = 26
    NUM_VARS = 26 * 2 + 26

    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26

    def obj_local(v, best_idx, best_v):
        v[best_idx] = best_v[best_idx]
        return obj(v)

    def take_step(x):
        return x + 0.05 * (np.random.rand(NUM_VARS) - 0.5)

    def single_optimization(init):
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq',
                                   'fun': build_constraints},
                       options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})

        if not res.success or np.any(build_constraints(res.x) < -1e-10):
            return None, -np.inf

        return res.x, res.fun

    best_res = None

    for _ in range(nrestarts):
        init = np.random.rand(NUM_VARS)
        init[:52] = 0.5 * (np.random.rand(52) + 0.5)
        best_idx = np.random.randint(0, NUM_VARS)
        best_res__, best_fun__ = single_optimization(init)

        if best_res__ is not None:
            curr_v = best_res__.copy()
            curr_v[best_idx] = 0
            curr_res__, curr_fun__ = single_optimization(curr_v)
            if curr_fun__ > best_fun__:
                best_res__, best_fun__ = curr_res__, curr_fun__

            inflate_radii(best_res__[:52].reshape(26, 2), best_res__[52:], max_iter=100)
            res_final = minimize(fun=obj_local, x0=best_res__, args=(best_idx, best_res__),
                                 method='SLSQP', bounds=bounds,
                                 constraints={'type': 'ineq',
                                             'fun': build_constraints},
                                 options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})

            if res_final.success and res_final.fun > best_fun__:
                best_res = res_final
                best_fun = res_final.fun

    if best_res is None or np.isinf(best_fun):
        raise ValueError("Optimization did not find a valid solution")

    return best_res.x[:52].reshape(26, 2), best_res.x[52:], -best_fun
