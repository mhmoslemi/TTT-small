import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def lp_radii(pos, dists):
    n = pos.shape[0]
    A_eq = np.concatenate([pos[:, np.newaxis, :] * 2 - np.eye(n), -np.eye(n)], axis=-1)
    b_eq = pos[:, 0] + pos[:, 1]
    A_ub = np.empty((2 * n + n * (n - 1), 2 * n + n), float)
    b_ub = np.empty(2 * n + n * (n - 1), float)
    start = 2 * n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[start + 2 * (i * (i + 1) // 2 + j - i - 1), i] = 1
            A_ub[start + 2 * (i * (i + 1) // 2 + j - i - 1), j] = 1
            A_ub[start + 2 * (i * (i + 1) // 2 + j - i - 1), start + i] = -1
            A_ub[start + 2 * (i * (i + 1) // 2 + j - i - 1), start + j] = -1
            b_ub[start + 2 * (i * (i + 1) // 2 + j - i - 1)] = dists[i, j]
            start += 1
    return linprog(-np.ones(n), A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, method='highs')

def fitness(pos):
    n = pos.shape[0]
    dists = np.sqrt(np.sum((pos[:, None, :] - pos[None, :, :]) ** 2, axis=-1))
    bounds = np.concatenate([pos[:, 0] - dists, 1 - pos[:, 0] - dists,
                            pos[:, 1] - dists, 1 - pos[:, 1] - dists, np.zeros(n)], axis=0)
    res = lp_radii(pos, dists)
    return -res.fun

def refine_solution(pos):
    bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26
    initial = np.concatenate([pos.ravel(), lp_radii(pos, np.sqrt(np.sum((pos[:, None, :] - pos[None, :, :]) ** 2, axis=-1))).x])
    res = minimize(lambda v: -np.sum(v[52:]), x0=initial, bounds=bounds,
                    constraints={'type': 'ineq', 'fun': constraints}, method='SLSQP')
    return res.x[:52].reshape(26, 2), res.x[52:]

def constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 26 + [(0, 1)] * 26 + [(0, 0.5)] * 26
    res = differential_evolution(fitness, bounds=bounds, popsize=50, maxiter=200, seed=42)
    final_pos = res.x[:52].reshape(26, 2)
    final_radii = lp_radii(final_pos, np.sqrt(np.sum((final_pos[:, None, :] - final_pos[None, :, :]) ** 2, axis=-1))).x
    refined_pos, refined_radii = refine_solution(final_pos)
    return refined_pos, refined_radii, np.sum(refined_radii)

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
