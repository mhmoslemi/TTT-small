import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize

def obj(positions):
    pos = positions.reshape(26, 2)
    r = np.maximum(0, x - r for x in pos[:, 0]) + np.maximum(0, 1 - pos[:, 0] - r) + np.maximum(0, y - r for y in pos[:, 1]) + np.maximum(0, 1 - pos[:, 1] - r)
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs[:, :, 0] ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_constraints = dists[mask] - (r[:, None] + r[None, :])[mask]
    return -np.sum(np.hstack((np.maximum(np.full_like(r, -np.inf), r), np.maximum(0, pair_constraints))))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 52
    bounds += [(0, 0.5)] * 26

    result = differential_evolution(func=obj, bounds=bounds, popsize=50, maxiter=200, seed=42)
    best_positions = result.x[:52].reshape(26, 2)
    best_r = result.x[52:]

    constraints_list = []
    for i in range(26):
        constraints_list.append({'type': 'ineq', 'fun': lambda positions, i=i: 1 - positions[i, 0] - best_r[i]})
        constraints_list.append({'type': 'ineq', 'fun': lambda positions, i=i: 1 - positions[i, 1] - best_r[i]})
        constraints_list.append({'type': 'ineq', 'fun': lambda positions, i=i: positions[i, 0] - best_r[i]})
        constraints_list.append({'type': 'ineq', 'fun': lambda positions, i=i: positions[i, 1] - best_r[i]})
        for j in range(i + 1, 26):
            def dist_constraint(positions, i=i, j=j):
                return np.sqrt((positions[i, 0] - positions[j, 0]) ** 2 + (positions[i, 1] - positions[j, 1]) ** 2) - (best_r[i] + best_r[j])
            constraints_list.append({'type': 'ineq', 'fun': dist_constraint})

    local_result = minimize(fun=lambda x: -np.sum(x[52:]), x0=np.concatenate((best_positions.flatten(), best_r)), bounds=bounds, constraints=constraints_list, options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})

    centers = local_result.x[:52].reshape(26, 2)
