import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution, SLSQP

np.random.seed(42)

def solve_radii(centers):
    n = centers.shape[0]
    M = n * (n - 1) // 2
    A_ub = np.empty((4 * n + M, n))
    lb = np.zeros(n)
    ub = np.full(n, 0.5)
    c = -np.ones(n)
    
    A_ub[:n] = -centers[:, 0]
    A_ub[n:2 * n] = centers[:, 0] - 1
    A_ub[2 * n:3 * n] = -centers[:, 1]
    A_ub[3 * n:4 * n] = centers[:, 1] - 1
    
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    row = 4 * n
    for i, j in pairs:
        A_ub[row][:] = centers[i] - centers[j]
        A_ub[row][i] = 1
        A_ub[row][j] = 1
        row += 1
    
    result = linprog(c, A_ub=A_ub, bounds=(lb, ub), method='highs')
    if not result.success:
        raise Exception("LP infeasible")
    return -result.fun, result.x

def evaluate_fitness(individual):
    rearranged = np.clip(individual.reshape(26, 2), 0, 1)
    sum_r, radii = solve_radii(rearranged)
    return sum_r

def crossover(parent1, parent2):
    p = np.random.rand()
    if p < 0.9:
        i1 = np.random.randint(1, 51)
        i2 = np.random.randint(i1 + 1, 52)
        offspring = np.copy(parent1)
        offspring[i1:i2] = parent2[i1:i2]
    else:
        offspring = parent1
    return offspring

def mutate(individual, sigma):
    individual += np.random.normal(scale=sigma, size=individual.shape)
    individual = np.clip(individual, 0, 1)
    return individual

def clamp_to_square(individual):
    individual = np.clip(individual, 0, 1)
    return individual

def run_packing():
    n = 26
    population_size = 48
    generations = 180
    crossover_prob = 0.9
    mutation_rate = 0.8
    initial_sigma = 0.05
    sigma_decay = 0.99
    elitism = 5
    tournament_size = 5

    population = np.random.rand(population_size, 2 * n)
    best_sum = -np.inf
    best_centers = None

    for gen in range(generations):
        fitnesses = [evaluate_fitness(individual) for individual in population]
        top_indices = np.argsort(fitnesses)[-elitism:]
        elitist_individuals = population[top_indices]
        remaining_individuals = 48 - elitism
        parents = np.array(population[np.random.choice(np.argsort(fitnesses)[:remaining_individuals], p=fitnesses[:remaining_individuals]/np.sum(fitnesses[:remaining_individuals]), size=(remaining_individuals, 2))])

        new_population = elitist_individuals
        for _ in range((48 - elitism) // 2):
            parent1, parent2 = parents[np.random.choice(remaining_individuals, 2, replace=False)]
            offspring = crossover(parent1, parent2)
            offspring = mutate(offspring, initial_sigma)
            offspring = clamp_to_square(offspring)
            new_population = np.vstack((new_population, offspring))

        population = new_population
        sigma = initial_sigma * (sigma_decay ** gen)

    best_sum, centers = solve_radii(population[np.argmax([evaluate_fitness(ind) for ind in population])].reshape(26, 2))
    centers = SLSQP(func=evaluate_fitness, bounds=[(0, 1),]*(2*n), jac=None, args=(), method='SLSQP', x0=centers.flatten(), options={'maxiter': 2000}).x.reshape(26, 2)
    return centers, solve_radii(centers)[1], solve_radii(centers)[0]
