import numpy as np
from scipy.optimize import linprog, minimize

# Constants
POPSIZE = 200
GENERATIONS = 100
ELITE_SIZE = 20
CROSSOVER_RATE = 0.9
MUTATION_RATE = 0.5
MUTATION_SIGMA = 0.1
MAX_ITER = 5000

# Gaussian mutate helper
def mutate(individual, rate, sigma):
    perturb = np.random.normal(0, sigma, individual.shape)
    mutated = individual + perturb
    return np.clip(mutated, 0, 1)

# Uniform crossover helper
def crossover(parent1, parent2, rate):
    if np.random.random() < rate:
        cutoff = np.random.randint(1, len(parent1))
        return np.concatenate((parent1[:cutoff], parent2[cutoff:]))
    else:
        return parent1

# Evaluate fitness 
def fitness(pos):
    r_opt, sum_r = solve_radii(pos)
    return -sum_r

# GA helpers
def tournament(pop, fitness_vals, k):
    selected_indices = np.random.choice(len(pop), size=k)
    best_index = selected_indices[np.argmax(fitness_vals[selected_indices])]
    return selected_indices[best_index]

# GA main loop    
def ga_best_positions():
   rng = np.random.default_rng(42)
   pop = rng.random((POPSIZE, 52))

   elite_indices = []
   best_fitness = -np.inf
   best_gen_pos = None

   for generation in range(GENERATIONS):
       fitness_vals = np.array([fitness(ind) for ind in pop])
       elite_indices = np.argsort(fitness_vals)[-ELITE_SIZE:]
       
       best_index = np.argmax(fitness_vals)
       if fitness_vals[best_index] > best_fitness:
           best_fitness = fitness_vals[best_index]
           best_gen_pos = pop[best_index]

       children = pop[elite_indices]
       while len(children) < POPSIZE:
           parent1_idx = tournament(pop, fitness_vals, 3)
           parent2_idx = tournament(pop, fitness_vals, 3)
           child = crossover(pop[parent1_idx], pop[parent2_idx], CROSSOVER_RATE)
           if np.random.random() < MUTATION_RATE:
               child = mutate(child, MUTATION_RATE, MUTATION_SIGMA)
           children.append(child)

       pop = np.array(children)

   return best_gen_pos

# Solve linear program for radii given positions
def solve_radii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    for i in range(n):
        A_ub.append([0]*i + [1] + [0]*(n-i)) #[0] ... , r_i, 0 ...
        b_ub.append(pos[i,0] + 1e-6) #[x_i[1] + 1e-6]

        A_ub.append([0]*i + [1] + [0]*(n-i))
        b_ub.append(1 - pos[i,0] - 1e-6)

        A_ub.append([0]*i + [1] + [0]*(n-i))
        b_ub.append(pos[i,1] + 1e-6)

        A_ub.append([0]*i + [1] + [0]*(n-i))
        b_ub.append(1 - pos[i,1] - 1e-6) 

    for i in range(n):
        for j in range(i+1,n):
            dist = np.sqrt(np.sum((pos[i] - pos[j]) ** 2)) - pos[i][2] - pos[j][2] + 1e-6
            A_ub.append([0]*i + [1] + [0]*j + [1]+ [0]*(n-j-2))
            b_ub.append(dist)
            A_ub.append([0]*i + [1] + [0]*j + [1]+ [0]*(n-j-2))
            b_ub.append(dist)

    A_eq = np.eye(n)
    b_eq = np.zeros(n)

    c = -np.ones(26)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')

    return res.x[:26], res.fun

# Function to refine solution using local optimization
def refine_solution(centers, radii):
    
    res = minimize(lambda v: -np.sum(v[-26:]), 
                   x0=np.concatenate((centers.ravel(), radii)), 
                   bounds=[(0,1)]*26 + [(0,0.5)]*26, 
                   constraints=({'type': 'ineq', 'fun': lambda v: v[26:]}),
                   method='SLSQP', options={'maxiter':MAX_ITER, 'ftol':1e-12, 'disp':False})
    
    return v[:52].reshape(26, 2), v[52:], -res.fun

# Main function
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
   pos = ga_best_positions()
   radii, sum_radii = solve_radii(pos)
   centers, _, sum_radii = refine_solution(pos, radii)
