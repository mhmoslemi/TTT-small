import numpy as np
from scipy.optimize import minimize, linprog, bounds
from scipy.spatial.distance import pdist, squareform

def lp_optimize(p):
    n = len(p) // 2
    I = np.ones((n, n))
    P = (p[:n**2].reshape(n, n) + p[n**2:].reshape(n, n)) / 2
    A = I - P
    B = pdist(p[:n].reshape((-1, 2)), 'euclidean')
    A = np.vstack([A.flatten(), B])
    b = np.hstack([np.zeros(n), B])
    A_ub = -np.eye(2*n)
    b_ub = np.zeros(2*n)
    res = linprog(p[:n**2].reshape(n,n).flatten() + p[n**2:].reshape(n,n).flatten(), A_ub=A_ub, b_ub=b_ub, A_eq=A, b_eq=b, method='highs', tol=1e-9)
    r = res.x[:n**2].reshape(n,n) + res.x[n**2:].reshape(n,n)
    r[r < 1e-12] = 0
    return r.flatten()

def generate_initial_population(n_circles, pop_size):
    hexagonal_radii = np.sqrt(0.25)
    grid_radii = 1 / np.sqrt(2)

    hex_centers = hexagonal_lattice(n_circles, hexagonal_radii)
    grid_centers = grid_lattice(n_circles, grid_radii)

    corner_centers = np.zeros((n_circles, 2))
    for i in range(4):
        corner_centers[i*6:(i+1)*6, i % 2] = 1 / 6 + i / 6
    
    population = []
    for p in [hex_centers, grid_centers, random_centers, corner_centers]:
        r = lp_optimize(np.concatenate([p.flatten(), random_radii]))
        population.append((p, np.sum(r)))
    
    total_population = []
    while len(total_population) < pop_size:
        parent1, parent2 = np.random.choice(population, size=2)
        child, child_r_sum = crossover(parent1, parent2)
        child = mutate(child)

        total_population.append((child, child_r_sum))
    
    return total_population

def hexagonal_lattice(n_circles, radius):
    n = int(np.ceil(np.sqrt(4 * n_circles / np.pi)))
    centers = []
    for i in range(n):
        for j in range(n if i % 2 == 0 else n - 1):
            x = j * np.sqrt(3) / 2 + (i % 2) * np.sqrt(3) / 2
            y = i / 2
            if x >= 1 - radius and y >= 1 - radius:
                continue
            centers.append([x, y])
    centers = np.array(centers)
    return np.tile(centers, (int(n_circles / len(centers)) + 1, 1))[:n_circles] * 2 - 1

def grid_lattice(n_circles, radius):
    n = int(np.ceil(np.sqrt(n_circles / radius**2)))
    centers = np.array([[i/n, j/n] for i in range(n) for j in range(n)]) * 2 - 1
    return centers[:n_circles]

def crossover(parent1, parent2):
    child_shape = (parent1[0].shape[0] * 2, 2)
    p = np.zeros(child_shape)
    p[:len(parent1[0])] = parent1[0] + (parent2[0] - parent1[0]) * np.random.rand(len(parent1[0]))
    p[len(parent1[0]):] = parent1[1] + (parent2[1] - parent1[1]) * np.random.rand(len(parent1[1]))
    child = (p[:len(parent1[0])], np.sum(lp_optimize(p)))
    return child

def mutate(child):
    n = child[0].shape[0]
    m = np.random.randint(1, n)
    mask = np.random.choice(n, m, replace=False)
    child[0][mask] += (np.random.randn(m, 2) * 0.02).clip(-1, 1)
    child[0][child[0] < -1] = -1
    child[0][child[0] > 1] = 1
    return child
