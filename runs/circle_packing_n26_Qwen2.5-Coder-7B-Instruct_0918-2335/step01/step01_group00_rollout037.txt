import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def compute_radii_and_value(centres: np.ndarray) -> (np.ndarray, float):
    wall_limits = np.min(np.column_stack([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]]), axis=-1)
    diss = pdist(centres)
    diss_matrix = squareform(diss)
    n, _ = centres.shape
    A_ub = np.zeros((n * (n - 1) // 2, n))
    for i in range(n):
        A_ub[i * (n - 1) // 2:(i + 1) * (n - 1) // 2, i] = 1
    for i in range(n):
        A_ub[i * (n - 1) // 2:(i + 1) * (n - 1) // 2, i + 1:i + n - 1] = 1
    b_ub = diss_matrix.ravel()
    c = -np.ones((n,))

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, wall_limits), method='highs')

    if res.success and res.status == 0:
        return res.x, -res.fun
    return np.zeros((n,)), -np.inf

def run_packing():
    np.random.seed(42)
    n_circles = 26
    n_restart = 150
    max_iter_sa = 2000
    temp_init = 0.1
    temp_dec = 0.97

    def sa(generation):
        x = generation[:52]
        centers = x.reshape(n_circles, 2)
        max_radius = np.min(np.column_stack([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]]), axis=-1)
        temperature = temp_init
        for _ in range(max_iter_sa):
            idx = np.random.randint(0, n_circles)
            dx, dy = np.random.normal(0, temperature, size=2)
            x_new = x.copy()
            x_new[idx : idx + 2] = [centers[idx, 0] + dx, centers[idx, 1] + dy]
            if (x_new[idx] >= 0 and x_new[idx] <= 1 and
                    x_new[idx + 1] >= 0 and x_new[idx + 1] <= 1):
                value_new, _ = compute_radii_and_value(x_new.reshape(n_circles, 2))
                if value_new < generation[52:] + np.random.exponential(scale=temperature):
                    x = x_new
        return x

    initial_layout = np.random.rand(n_circles * 2)
    initial_layout = initial_layout.reshape(n_circles, 2)
    best_val = -np.inf
    best_centres = np.zeros((26, 2))

    for _ in range(n_restart):
        centers = sa(initial_layout)
        value, _ = compute_radii_and_value(centers)
        if value > best_val:
            best_val = value
            best_centres = centers

    value, radii = compute_radii_and_value(best_centres)
    return best_centres, radii, value
