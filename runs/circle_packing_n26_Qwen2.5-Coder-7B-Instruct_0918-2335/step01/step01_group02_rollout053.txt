import numpy as np
from scipy.optimize import linprog, minimize

def lp_get_radii(centers):
    n = centers.shape[0]
    obj = [-1] * n
    A = []
    b = []
    
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        A.append([x, 0, 1, 0])
        b.append(x)
        A.append([0, y, 0, 1])
        b.append(y)
    
    # non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            dx = x1 - x2
            dy = y1 - y2
            A.append([x1, 0, -x2, -1, 0])
            b.append(dx + 1e-12)
            A.append([0, y1, 0, x1, -1])
            b.append(dy + 1e-12)
            A.append([0, y1, 0, x2, -1])
            b.append(dy + 1e-12)
            A.append([x1, 0, -x2, -1])
            b.append(-dx + 1e-12)
            A.append([0, y1, 0, x2, -1])
            b.append(-dy + 1e-12)
            A.append([x2, 0, -x1, -1, 0])
            b.append(-dx + 1e-12)
            A.append([0, y2, 0, x1, -1])
            b.append(-dy + 1e-12)
            A.append([0, y2, 0, x2, -1])
            b.append(dy + 1e-12)
    
    A = np.array(A)
    b = np.array(b)
    
    result = linprog(c=obj, A_ub=A, b_ub=b, method='highs')
    if not result.success:
        b += 1e-12 * np.ones(len(b))
        result = linprog(c=obj, A_ub=A, b_ub=b, method='highs')
    
    return result.x

def sa_step(curr_centers):
    n = curr_centers.shape[0]
    i = np.random.randint(n)
    dx, dy = np.random.normal(size=2) * curr_centers[i] * 0.1
    new_centers = curr_centers.copy()
    new_centers[i] += dx, dy
    new_centers[i] = np.clip(new_centers[i], 0, 1)
    return new_centers

def run_packing():
    n = 26
    initial_centers = np.random.rand(n, 2)
    best_centers, best_r = np.zeros((n, 2)), 0
    T0, alpha, moves_per_temp = 0.1, 1.5, 50
    current_centers = initial_centers.copy()
    
    for _ in range(300):
        T = T0 / np.log(1 + alpha)
        for _ in range(moves_per_temp):
            new_centers = sa_step(current_centers)
            new_r = lp_get_radii(new_centers)
            new_sum = new_r.sum()
            if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
                current_centers = new_centers
                best_centers = new_centers
                best_r = new_r
                best_sum = new_sum
        
            T /= alpha    
    
    x0 = np.concatenate((current_centers.flatten(), best_r))
    bounds = [(0, 1)] * n * 2 + [(0, 0.5)] * n
    constraints = [{'type': 'ineq', 'fun': lambda v: -c for c in (lambda v: v[n*4:])}(*split(v)) for v in [(x0, bounds)]]
    
    def objectivo(v):
        n = len(v) // (n * 4 + n)
        centers = v[:n * 4].reshape(n, 2)
        radii = v[n * 4:n * 4 + n]
        return -radii.sum()
    
    result = minimize(objectivo, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 4000, 'ftol': 1e-9, 'disp': False})

    return best_centers, best_r, result.fun

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
