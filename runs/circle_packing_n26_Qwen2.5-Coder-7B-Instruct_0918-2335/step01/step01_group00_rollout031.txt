import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping

N = 26
NUM_VARS = N * 2 + N

def obj(v):
    r = v[N:]  # last 26 entries
    return -np.sum(r)

def constraints(v):
    pos = v[:N].reshape(N, 2)  # (x,y) pairs
    r = v[N:]  # radii

    # boundary terms – 4*N
    cons = np.empty(2 * N, float)
    cons[0:N] = pos[:, 0] - r  # left
    cons[N:2 * N] = 1 - pos[:, 0] - r  # right

    # pairwise terms – N choose 2
    diffs = pos[:, None, :] - pos[None, :, :]  # (N,N,2)
    d2 = np.sum(diffs ** 2, axis=2)
    mask = np.triu(np.ones((N, N), bool), k=1)
    pair_cons = d2[mask] - (r[:, None] + r[None, :])[mask] ** 2
    return np.concatenate([cons, pair_cons])

def inflate_radii(centers, radii, max_iter=100, tol=1e-8):
    for it in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(N):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break
    return radii

def run_packing(max_iter=4000, ftol=1e-12, popsize=10, nrestarts=100) -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 2 * N + [(0, 0.5)] * N

    for _ in range(nrestarts):
        init = np.random.rand(NUM_VARS)
        init[N:] = 0.05 * np.random.rand(N)
        res = minimize(fun=obj, x0=init, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        if res.success and np.all(constraints(res.x) >= -1e-10):
            best_sol = res.x
            break
    else:
        raise ValueError("Optimization failed to find a valid solution")

    inflated_radii = inflate_radii(best_sol[:N].reshape(N, 2), best_sol[N:], max_iter=1000)
    best_sol[N:] = inflated_radii

    final_res = minimize(fun=obj, x0=best_sol, method='SLSQP', bounds=bounds,
                         constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    if not final_res.success or np.any(constraints(final_res.x) > 1e-10):
        raise ValueError("Final optimization did not find a valid solution")

    centers = final_res.x[:N].reshape(N, 2)
    radii = final_res.x[N:]
    sum_radii = -final_res.fun

    return centers, radii, sum_radii
