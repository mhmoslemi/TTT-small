import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance_matrix
from random import seed, normalvariate

# Problem const
POP_SIZE = 80
GENS = 250
ELITE_SIZE = int(0.1 * POP_SIZE)
CROSSOVER_RATE = 0.9
MUTATION_RATE = 0.2
MUTATION_SIGMA = 0.03
SEED = 42

def solve_radius(x):
    n = x.shape[0] // 2
    r = np.zeros(n)
    A_ub = []
    b_ub = []
    for i in range(n):
        A_ub.append([0] * i + [1] + [0] * (n - i - 1))
        b_ub.append(1)
        if x[2 * i] < 0.5:
            A_ub.append([0] * i + [1] + [0] * (n - i - 1))
            b_ub.append(x[2 * i])
        if x[2 * i + 1] < 0.5:
            A_ub.append([0] * i + [0] + [1] + [0] * (n - i - 1))
            b_ub.append(x[2 * i + 1])
        for j in range(i + 1, n):
            r_ij = distance_matrix(np.array([[x[2 * i], x[2 * i + 1]]]), np.array([[x[2 * j], x[2 * j + 1]]]))[0][0] - (r[i] + r[j])
            A_ub.append([0] * i + [1] + [0] * (j - i - 1) + [1] + [0] * (n - j - 1))
            b_ub.append(r_ij)
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    return result.x, result.fun

def fitness(pos):
    r, s_r = solve_radius(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = []
    for p1, p2 in zip(parent1, parent2):
        if np.random.rand() < rate:
            child.append(p1)
        else:
            child.append(p2)
    return np.array(child)

def mutate(individual, rate, sigma):
    child = individual.copy()
    for i in range(len(individual)):
        if np.random.rand() < rate:
            child[i] += sigma * normalvariate(0, 1)
            child[i] = max(0, min(1, child[i]))
    return np.array(child)

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(range(POP_SIZE), k, replace=False)
    returns_idx = -1
    for idx in indices:
        if returns_idx == -1 or fitness_vals[idx] < fitness_vals[returns_idx]:
            returns_idx = idx
    return returns_idx

def ga_best_positions():
    seed(SEED)
    pop = np.random.rand(POP_SIZE * 2)
    fitness_vals = [fitness(individual) for individual in pop]
    log = [(np.min(fitness_vals), np.mean(fitness_vals), np.max(fitness_vals))]
    for g in range(GENS):
        elites = np.argsort(fitness_vals)[:ELITE_SIZE]
        if not elites.size:
            break
        offspring = pop[elites]
        for _ in range(POP_SIZE - ELITE_SIZE):
            parent1, parent2 = pop[tournament(pop, fitness_vals, 3)], pop[tournament(pop, fitness_vals, 3)]
            offspring = np.vstack((offspring, crossover(parent1, parent2, CROSSOVER_RATE)))
        for i in range(POP_SIZE - ELITE_SIZE):
            offspring[i] = mutate(offspring[i], MUTATION_RATE, MUTATION_SIGMA)
        pop = offspring
        fitness_vals = [fitness(individual) for individual in pop]
        log.append((np.min(fitness_vals), np.mean(fitness_vals), np.max(fitness_vals)))
    idx = np.argmin(fitness_vals)
    return pop[idx * 2: (idx + 1) * 2]

def refine_solution(centers, radii):
    r = np.copy(radii)
    pos = centers.copy()
    while True:
        result = minimize(fitness, pos, method='SLSQP', bounds=[[0, 1], [0, 1]] * centers.shape[0], constraints=[{'type': 'ineq', 'fun': lambda x: x[2 * i] - x[2 * i + 1] - r[i]} for i in range(centers.shape[0])] + [{'type': 'ineq', 'fun': lambda x: 1 - x[2 * i] - r[i]} for i in range(centers.shape[0])] + [{'type': 'ineq', 'fun': lambda x: 1 - x[2 * i + 1] - r[i]} for i in range(centers.shape[0])] + [{'type': 'ineq', 'fun': lambda x, i=i: distance_matrix(np.array([x[2 * i: 2 * i + 2]]), np.array([x[2 * j: 2 * j + 2]])) - (r[i] + r[j])) for i in range(centers.shape[0]) for j in range(i + 1, centers.shape[0])])
        if result.success:
            best_radii = result.x[2:]
            if np.all(best_radii <= 0.5):
                return centers, best_radii, -result.fun
            pos, r = result.x.reshape(26, 2), best_radii
        else:
            break

def run_packing():
    centers = ga_best_positions()
    _, _, sum_r = refine_solution(centers, np.zeros(26))
    validate, message = validate_packing(centers.reshape(26, 2), np.clip(centers, 0, 0.5))
    assert validate, message
    return centers.reshape(26, 2), sum_r, -sum_r
