import numpy as np
from scipy.optimize import linprog, minimize

def build_contact_constraints(G, n):
    m = len(G.edges)
    A_ub = np.zeros((m + 4 * n, n))
    b_ub = np.zeros(m + 4 * n)

    # Edge constraints
    for i, j in G.edges:
        x_i, y_i = G.nodes[i]['pos']
        x_j, y_j = G.nodes[j]['pos']
        A_ub[i, i] = 1
        A_ub[i, j] = 1
        b_ub[i] = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)

    # Wall constraints
    for i in range(n):
        x_i, y_i = G.nodes[i]['pos']
        wall_distance = min(x_i, 1 - x_i, y_i, 1 - y_i)
        wallId = i + n
        A_ub[wallId, i] = 1
        b_ub[wallId] = wall_distance

    # Contact constraints for newly added vertices
    for i in range(n, 2*n):
        u, v = i, i - n
        x_u, y_u = G.nodes[u]['pos']
        x_v, y_v = G.nodes[v]['pos']
        A_ub[i, u] = 1
        A_ub[i, v] = 1
        b_ub[i] = np.sqrt((x_u - x_v)**2 + (y_u - y_v)**2)

    # Constraints for infeasible positions
    for i in range(2*n, 3*n):
        x_i = np.random.uniform(0, 1)
        y_i = np.random.uniform(0, 1)
        A_ub[i, i] = -1
        A_ub[i, n + i % n] = -1
        b_ub[i] = -1e9

    return A_ub, b_ub

def solve_LP(G, n):
    A_ub, b_ub = build_contact_constraints(G, n)
    c = -np.ones(n)
    bounds = [(0, 1) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x

        dists = np.sqrt((x - x_i)**2 + (y - y_i)**2)

            x_j, y_j = x[j], y[j]
            dist = dists[j] - dists[i]
            bounds[52 + i * 2] = (0, dist)
            bounds[52 + i * 2 + 1] = (0, dist)
    constraints = []
    for i in range(26):
        x_i, y_i = x[i], y[i]
        if x_i + radii[i] > 1:
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, x=x_i, y=y_i: r[i] - (1 - x_i)})
        if x_i - radii[i] < 0:
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, x=x_i, y=y_i: -(r[i] - x_i)})
        if y_i + radii[i] > 1:
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, x=x_i, y=y_i: r[i] - (1 - y_i)})
        if y_i - radii[i] < 0:
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, x=x_i, y=y_i: -(r[i] - y_i)})
    res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    return res.x
