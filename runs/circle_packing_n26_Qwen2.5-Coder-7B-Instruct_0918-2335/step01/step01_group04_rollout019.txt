def solve_radii_lp(centers):

        np.column_stack([np.zeros(n), np.eye(n), -np.eye(n), np.zeros(n)]),
        np.column_stack([-np.eye(n), np.zeros(n), np.zeros(n), -np.eye(n)]),
        np.column_stack([np.eye(n), np.zeros(n), np.zeros(n), np.eye(n)]),
        [*([2] * (n * n - n),) + ([1] * n)]
    ])
    b = np.hstack([np.ones(n), np.zeros(n), np.ones(n), 2 * np.ones(n - n * n // 2)])

    return res.x
