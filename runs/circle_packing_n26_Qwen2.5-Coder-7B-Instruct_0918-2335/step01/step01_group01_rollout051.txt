import numpy as np
from scipy.optimize import linprog, minimize
import random

POP = 30
EVAL = 4000
sigma = 0.02
p_mut = 0.3

# Linear_program engine -------------------------------------------
def boundary_constraints(p):
    n = len(p) // 2
    A_eq = np.zeros((4*n, 2*n))
    b_eq = np.zeros(4*n)
    for i in range(n):
        A_eq[4*i, 2*i] = 1.0
        A_eq[4*i, 2*i+1] = 0.0
        A_eq[4*i+1, 2*i] = -1.0
        A_eq[4*i+1, 2*i+1] = 0.0
        A_eq[4*i+2, 2*i] = 0.0
        A_eq[4*i+2, 2*i+1] = 1.0
        A_eq[4*i+3, 2*i] = 0.0
        A_eq[4*i+3, 2*i+1] = -1.0
        b_eq[4*i] = p[2*i]
        b_eq[4*i+1] = 1.0 - p[2*i]
        b_eq[4*i+2] = p[2*i+1]
        b_eq[4*i+3] = 1.0 - p[2*i+1]
    return A_eq, b_eq

def pairwise_constraints(p, D):
    n = len(p) // 2
    A_ineq = np.zeros((n*n, 2*n))
    b_ineq = np.zeros(n*n)
    for i in range(n):
        for j in range(i+1, n):
            A_ineq[i*n+j, 2*i] += 1.0
            A_ineq[i*n+j, 2*j] += 1.0
            A_ineq[i*n+j, 2*i+1] += 1.0
            A_ineq[i*n+j, 2*j+1] += 1.0
            b_ineq[i*n+j] = D[i,j]
    return A_ineq, b_ineq

def lp_optimize(r0, A_ineq, b_ineq, tol=1e-9):
    n = len(r0)
    A = np.vstack([A_ineq, -np.eye(n)])
    b = np.hstack([b_ineq, np.zeros(n)])
    c = [-1.0] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', tol=tol)
    r_opt = -res.fun * np.ones(n)
    r_opt[r_opt < tol] = 0.0  # Guard against tiny negatives
    return r_opt

# Center initialisation -----------------------------------------------
def hex_grid():
    n = 26
    r = np.sin(np.pi / (2 * n)) / (2 * np.cos(np.pi / (4 * n)))
    theta = np.arange(n+1) * (np.pi / n)
    x = r * np.cos(theta[:-1])
    y = r * np.sin(theta[:-1])
    dx = np.array([0, 1/2, 1/2, 0, -1/2, -1/2])
    dy = np.array([np.sqrt(3)/2, np.sqrt(3)/2, 0, -np.sqrt(3)/2, -np.sqrt(3)/2, 0])
    p = np.concatenate([np.tile(x, (3, 1)), np.tile(y, (3, 1))]).flatten()
    p += np.random.uniform(-0.01, 0.01, 2*n)
    p[p < 0] = 0
    p[p > 1] = 1
    return p

def random_centers():
    return np.random.uniform(0, 1, 52)

def grid_centers():
    b = np.zeros((5, 5))
    x = np.linspace(0, 1, 6)
    y = np.linspace(0, 1, 6)
    b[1, 1] = 1/6.0
    b[0, 2] = 2/6.0
    b[1, 2] = 3/6.0
    b[2, 2] = 4/6.0
    b[3, 2] = 5/6.0
    b[4, 2] = 6/6.0
    b[2, 1] = b[2, 2] + 1/2
    b[2, 3] = b[2, 2] + 1/2
    p = np.array([x[np.random.randint(0, 6)] for _ in range(25)]) % 1
    p = np.append(p, y[np.random.randint(0, 6)] for _ in range(25)]) % 1
    p = p + np.random.uniform(-0.01, 0.01, 52)
    p[p < 0] = 0
    p[p > 1] = 1
    return p

def corner_centers():
    p = np.zeros(52)
    x = np.sort(np.random.choice(np.arange(1, 4), size=8, replace=False))
    y = np.array([1] * 8 + [2] * 8 + [3] * 8 + [4] * 8 + [5] * 8 + [6] * 8) / 6.0
    idx = np.argsort(x + y)
    p = np.append(x[idx], y[idx]) % 1
    p[p < 0] = 0
    p[p > 1] = 1
    p += np.random.uniform(-0.01, 0.01, 52)
    p[p < 0] = 0
    p[p > 1] = 1
    return p

def initial_population():
    pop = []
    centers = [hex_grid(), random_centers(), grid_centers(), corner_centers()]
    n_members = [4, 10, 6, 10]
    for i, center in enumerate(centers):
        for _ in range(n_members[i]):
            n = len(center)
            p = np.reshape(center, (n//2, 2)) + np.random.uniform(-0.01, 0.01, (n//2, 2))
            p = p.flatten()
            p[p < 0] = 0
            p[p > 1] = 1
            d = np.sqrt(((p.reshape(-1, 2)[:, None, :] - p.reshape(-1, 2)[None, :, :])**2).sum(axis=2))
            n_rad = np.sum(lp_optimize(np.ones(n//2), boundary_constraints(p), pairwise_constraints(p, d)))
            pop.append((p, n_rad))
    pop.sort(key = lambda x: x[1], reverse=True)
    return pop[:POP]

# Evolutionary loop -------------------------------------------------------
def local_slsqp(C):
    x0 = C
    def obj(C):
        r_sum = lp_optimize(C[(len(C)//2):], boundary_constraints(C), pairwise_constraints(C, D))
        return -np.sum(r_sum)

    constraints = [
            {'type': 'ineq', 'fun': lambda C: C[:(len(C)//2)] - C[(len(C)//2):]},
            {'type': 'ineq', 'fun': lambda C: C[(len(C)//2):] - C[:(len(C)//2)]},
            {'type': 'ineq', 'fun': lambda C: 1 - C[:(len(C)//2)]},
            {'type': 'ineq', 'fun': lambda C: C[:(len(C)//2)]},
            {'type': 'ineq', 'fun': lambda C: 1 - C[(len(C)//2):]},
            {'type': 'ineq', 'fun': lambda C: C[(len(C)//2):]}
    ]

    bounds = [(0, 1)] * (len(C) // 2) + [(0, 0.5)] * (len(C) // 2)
    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 25})
    C[:(len(C)//2)] = res.x
    return C

def tournament_select(pop):
    i1 = random.randint(0, len(pop)-1)
    i2 = random.randint(0, len(pop)-1)
    return pop[min(i1,i2)][0] if pop[i1][1] > pop[i2][1] else pop[max(i1,i2)][0]

def mutate(C):
    if random.random() < p_mut:
        m = int(random.normal(3, 1)) 
        if m < 1:
            m = 1
        elif m > 26:
            m = 26
        for i in range(m):
            r = random.randint(0, len(C)//2 - 1)
            x = random.gauss(C[2*r], sigma)
            y = random.gauss(C[2*r+1], sigma)
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            C[2*r] = x
            C[2*r+1] = y

def replace_worst(C, r_sum, pop):
    pop.remove(min(pop, key=lambda x: x[1]))
    pop.append((C, r_sum))

# Main function ---------------------------------------------------------
def run_packing():
    centers = initial_population()
    best_centers, best_sum = centers[0][0], centers[0][1]
    for _ in range(EVAL):
        A_ineq, b_ineq = pairwise_constraints(best_centers, D)
        r_sum = lp_optimize(np.ones(len(best_centers)//2), A_ineq, b_ineq)
        r_sum[r_sum < 1e-9] = 0.0 
        best_sum = np.sum(r_sum)
        if best_sum > centers[0][1]:
            best_centers = best_centers.reshape(-1)
            best_centers[:(len(best_centers)//2)] += A_ineq.dot(r_sum)
            best_centers[(len(best_centers)//2):] -= r_sum
            best_centers[:(len(best_centers)//2)] = np.clip(best_centers[:(len(best_centers)//2)], 0, 1)
            best_centers[(len(best_centers)//2):] = np.clip(best_centers[(len(best_centers)//2):], 0, 0.5)
    centers = best_centers.reshape(-1)
    r_sum = lp_optimize(np.ones(len(centers)//2), A_ineq, b_ineq)
    centers = np.reshape(centers, (26, 2))
    r_sum[r_sum < 1e-9] = 0
    return centers, r_sum, np.sum(r_sum)
