import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping

N = 26
NUM_VARS = 2 * N + N

def obj(v):
    r = v[N:]
    return -np.sum(r)

def pair_constraints(v):
    pos = v[:N].reshape(N, 2)
    r = v[N:]
    diffs = pos[:, None, :] - pos[None, :, :]
    d2 = np.sum(diffs**2, axis=-1)
    mask = np.triu(np.ones((N, N), bool), k=1)
    cons = d2[mask] - (r[:, None] + r[None, :])[mask]**2
    return cons.flatten()

def constraints_func(v):
    # Boundary constraints
    pos = np.reshape(v[:N], (N, 2))
    cons_b = np.hstack([pos[:, 0] - v[N:], 1 - pos[:, 0] - v[N:], 
                         pos[:, 1] - v[N:], 1 - pos[:, 1] - v[N:]])
    # Pairwise constraints
    cons_p = pair_constraints(v)
    return np.hstack([cons_b, cons_p])

def inflation_radii(centers, radii, max_iter=100, tol=1e-8):
    for _ in range(max_iter):
        old_radii = radii.copy()
        for i in range(N):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(N):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max_r
        if np.allclose(radii, old_radii, atol=tol):
            break

def local_optimize(centers, radii, maxiter=1000):
    v = np.concatenate([centers.flatten(), radii])
    res = minimize(fun=obj, x0=v, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraints_func},
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if not res.success:
        raise ValueError("Local optimization failed to find a valid solution.")
    return res.x[:N].reshape(-1, 2), res.x[N:]

def run_packing(nrestarts=100) -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * N + [(0, 0.5)] * N
    
    best_obj = -np.inf
    best_v = None
    
    for _ in range(nrestarts):
        init_centers = 0.5 * np.random.rand(N, 2)
        init_radii = 0.5 * np.random.rand(N)
        init_v = np.concatenate([init_centers.flatten(), init_radii])
        
        with np.errstate(over='ignore'):
            res = minimize(fun=obj, x0=init_v, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints_func},
                           options={'maxiter': 4000, 'ftol': 1e-12})
        
        if res.success:
            current_obj = -res.fun
            current_v = res.x
            if current_obj > best_obj:
                best_obj = current_obj
                best_v = current_v
    
    if best_v is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers, radii = best_v[:N].reshape(26, 2), best_v[N:]
    inflation_radii(centers, radii)
    centers, radii = local_optimize(centers, radii)
    
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Optimization produced an invalid packing")
    
    return centers, radii, -best_obj
