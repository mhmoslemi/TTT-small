import numpy as np
from scipy.optimize import linprog, minimize

def compute_radii(centers):
    n = centers.shape[0]
    A = np.zeros((84, n))
    b = np.zeros(84)
    
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0]
        b[n + i] = 1 - centers[i, 0]
        b[2 * n + i] = centers[i, 1]
        b[3 * n + i] = 1 - centers[i, 1]
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            A[4 * n + (j - i - 1) * n + i, i] = 1
            A[4 * n + (j - i - 1) * n + i, j] = -1
            b[4 * n + (j - i - 1) * n + i] = dist - (radii[i] + radii[j])
    
    result = linprog(-radii, bounds=np.zeros_like(radii), A_eq=A, b_eq=b, method='highs')
    return -result.fun, result.x

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    best_centers = centers.copy()
    best_r = np.zeros(n)
    best_sum = 0
