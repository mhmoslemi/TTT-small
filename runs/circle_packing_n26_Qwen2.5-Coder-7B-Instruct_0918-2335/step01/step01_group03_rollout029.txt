import numpy as np
from scipy.optimize import linprog, minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def optimise_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n) # objective: max sum(radii)
    A = np.vstack((np.eye(n), -np.eye(n), 2 * np.eye(n), -2 * np.eye(n))) # boundary and distance constraints
    b = np.hstack((np.ones(n), np.zeros(n), centers[:, 0], centers[:, 0])) # boundary and distance constraints
    Aeq = np.vstack((np.eye(n), -np.eye(n))) # sum of radii constraint
    beq = np.array([1] * n) # sum of radii constraint
    x0 = np.ones(n) # initial guess for r
    c_bounds = [(0, None) for _ in range(n)] # nonnegative radii

    res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=c_bounds, method='highs')
    if res.success and res.x.min() >= 0:
        return res.x
    else:
        return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    NITER = 50000
    INITIAL_START = np.array([
        [0.2, 0.2], [0.2, 0.4], [0.2, 0.6], [0.2, 0.8],
        [0.4, 0.2], [0.4, 0.4], [0.4, 0.6], [0.4, 0.8],
        [0.6, 0.2], [0.6, 0.4], [0.6, 0.6], [0.6, 0.8],
        [0.8, 0.2], [0.8, 0.4], [0.8, 0.6], [0.8, 0.8],
        [0.5, 0.5] # central disc
    ])
    current_centres = INITIAL_START
    current_radii = optimise_radii(current_centres)
    best_sum = np.sum(current_radii)
    best_centres = current_centres
    best_radii = current_radii
    T0 = 1.0
    alpha = 0.995
    delta_max = 0.05

    for iter in range(NITER):
        i = np.random.randint(0, 26)
        delta_x, delta_y = np.random.uniform(-delta_max, delta_max, 2)
        new_centre = current_centres[i] + (delta_x, delta_y)
        if not (0 <= new_centre[0] <= 1 and 0 <= new_centre[1] <= 1):
            continue
        temp_centres = current_centres.copy()
        temp_centres[i] = new_centre
        temp_radii = optimise_radii(temp_centres)
        if temp_radii is not None:
            new_sum = np.sum(temp_radii)
            ΔS = new_sum - best_sum
            if ΔS > 0 or np.random.rand() < np.exp(ΔS / (T0 * alpha**iter)):
                current_centres, current_radii, best_sum = temp_centres, temp_radii, new_sum
                if new_sum > best_sum:
                    best_centres, best_radii = temp_centres, temp_radii

    # SLSQP refinement
    bounds = np.array([[0,1]] * 52)
    bounds[52:] = [(0,0.5)] * 26
    res = minimize(lambda v: -np.sum(v[52:]), 
                    x0=current_centres.flatten(), 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints={'type': 'ineq', 'fun': lambda v: np.hstack((v[:52] - v[52:] - 1e-12,
                                                                            1 - v[:52] - v[52:] - 1e-12,
                                                                            v[52:] - np.abs(v[52:]) + 1e-12))},
                    options={'maxiter': 4000, 'ftol': 1e-12})

    if res.success:
        refined_centres = res.x[:52].reshape(26, 2)
        refined_radii = res.x[52:]
        final_sum = np.sum(refined_radii)
        final_centres, final_radii, final_sum = refined_centres, refined_radii, final_sum
        return final_centres, final_radii, final_sum
    else:
        raise RuntimeError("Optimization did not succeed")
