import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from enum import Enum


def solve_lp(centres):
    n = len(centres)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    A_ub = np.zeros((4 * n + len(pairs), n))
    b_ub = np.zeros(4 * n + len(pairs))

    def build_row(A, b, i, constraint):
        A[i] = np.zeros(n)
        A[i, i] = 1
        b[i] = constraint
        return A, b

    A_ub, b_ub = build_row(A_ub, b_ub, range(n), centres[:, 0])
    A_ub, b_ub = build_row(A_ub, b_ub, range(n, 2 * n), centres[:, 1])
    A_ub, b_ub = build_row(A_ub, b_ub, range(2 * n, 3 * n), 1 - centres[:, 0])
    A_ub, b_ub = build_row(A_ub, b_ub, range(3 * n, 4 * n), 1 - centres[:, 1])
    for (i, j), idx in zip(pairs, range(4 * n, 4 * n + len(pairs))):
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = np.linalg.norm(centres[i] - centres[j])

    c = -np.ones(n)
    result = linprog(c, A_ub, b_ub, bounds=(0, None), method='highs')
    return result.fun[0] * -1, result.x


def evaluate_positions(centers):
    sum_radii, radii = solve_lp(centers)
    return sum_radii, radii


def generate_hexagon_points(side):
    n = 1
    while n * (n + 1) * 3 / 2 < 26:
        n += 1
    nv = n * (n + 1) * 3 // 2
    points = np.empty((nv, 2))
    x = np.linspace(0.5 - side / 2, 0.5 + side / 2, max(2, n))
    y = np.linspace(0.5 - side / 2, 0.5 + side / 2, max(2, n))
    offset = 0.5 * np.sqrt(3)
    for i, yi in enumerate(y):
        for j, xi in enumerate(x):
            if (i + j) % 2 == 0:
                points[i * n + j] = [xi, yi]
            else:
                points[i * n + j] = [xi - offset / 2, yi - offset / 2 * 0.5]
    return points


def random_jitter(centers, jitter_step):
    jittered = centers + jitter_step * np.random.rand(centers.shape[0], centers.shape[1]) - jitter_step * 0.5
    jittered = np.clip(jittered, 0, 1)
    return jittered


def main():
    min_radius = 0.01
    jitter_step = 0.02
    n_grid_points = 5
    grid_space = np.linspace(min_radius, 1 - min_radius, n_grid_points)
    grid_points = np.concatenate([np.meshgrid(grid_space, grid_space) for _ in range(2)]).T
    points = np.vstack([grid_points for _ in range(2)])
    points = np.vstack([points, np.array([[0.5, 0.5]])])
    points = random_jitter(points, jitter_step)
    points = points[:26]  # Remove duplicates and reduce to 26 points
    points = np.unique(points, axis=0)
    while len(points) < 26:
        point = np.array([np.random.rand(), np.random.rand()])
        if np.allclose(np.linalg.norm(point - points, axis=1), point, atol=min_radius).any():
            continue
        points = np.vstack([points, point])
    points = random_jitter(points, jitter_step)
    points = points[:26]  # Ensure we have exactly 26 points

    best_sum_radii = -np.inf
    best_centers = None

    for start_idx in range(1000):
        centres = random_jitter(points, jitter_step)
        sum_radii, radii = evaluate_positions(centres)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centres

    final_sum_radii, final_radii = solve_lp(best_centers)

    return best_centers, final_radii, final_sum_radii


if __name__ == "__main__":
    centers, radii, sum_radii = main()
    print("Final centers:", centers)
    print("Final radii:", radii)
    print("Final sum of radii:", sum_radii)
