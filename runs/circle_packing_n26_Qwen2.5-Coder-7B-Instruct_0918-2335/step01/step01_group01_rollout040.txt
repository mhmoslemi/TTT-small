import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import cKDTree

def distance_matrix(centers):
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=-1))
    return dists + np.eye(len(centers)) * 1e-9

def sum_radii(radii):
    return np.sum(radii)

def objective_and_constraints(centers):
    dists = distance_matrix(centers)
    n = centers.shape[0]
    A_ub = np.zeros((n + (n**2 - n) // 2, n))
    b_ub = np.zeros(n + (n**2 - n) // 2)
    
    A_ub[:n, :n] = np.eye(n)
    b_ub[:n] = 0.5
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[n + idx, i] = 1
            A_ub[n + idx, j] = 1
            b_ub[n + idx] = dists[i, j]
            idx += 1
    
    return lambda x: -sum_radii(x), (A_ub, b_ub)

def run_packing():
    n = 26
    m = 35
    grid_size = np.linspace(0, 1, m)

            ])
        ]
        initial_positions = np.vstack(seed_patterns)
        return initial_positions[np.random.choice(initial_positions.shape[0], n, replace=False)]
    
    def simulate_annealing(centers):
        T = 0.05
        while T > 1e-5:
            for i in range(centers.shape[0]):
                r = np.random.normal(0, T)
                new_centers = centers.copy()
                new_centers[i] += r * np.array([np.random.choice([-1, 1]), np.random.choice([-1, 1])])
                new_centers = np.clip(new_centers, 1e-6, 1 - 1e-6)
                dists = distance_matrix(new_centers)
                new_r = np.sum(new_centers, axis=0)
                old_r = np.sum(centers, axis=0)
                if new_r.sum() > old_r.sum() and new_r.sum() - old_r.sum() < T:
                    centers = new_centers
            T *= 0.99
        return centers
    
    def solve_lp(centers):
        objective, cons = objective_and_constraints(centers)

   _lp_result = minimize(fun=objective, x0=radii, method='SLSQP', bounds=[(0, 0.5)] * n)
    
    return centers, _lp_result.x, -_lp_result.fun
