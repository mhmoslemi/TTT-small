import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    n = centers.shape[0]
    r = np.array([1.0 / 2 * (1 - np.linalg.norm(centers[i] - centers[j])) for i in range(n) for j in range(i+1, n) if np.linalg.norm(centers[i] - centers[j]) > 0])
    r = r[r < 0.5]
    A = np.array([[1, 1] for i in range(len(r))])
    b = -r
    result = linprog(np.ones(len(r)), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_radius = 0.5
    random_seeds = 8
    patience = 10
    cooling_interval = 20
    accept_rate = 0.85
    max_iter = 200
    escape_interval = 50
    alpha = 0.01

    best_centers = None
    best_sum_radii = 0.0
    best_alpha = alpha

    for _ in range(random_seeds):
        # Generate initial lattice-based seed
        getchar = lambda: chr(ord('a')) if np.random.rand() < 0.5 else chr(ord('A'))
        lattice_centers = np.array([
            (np.random.rand() * (1 + 4 * np.random.rand() - 2), np.random.rand() * (1 + 4 * np.random.rand() - 2))
            for _ in range(25)
        ])
        add_random_points = lambda x: np.random.rand(n, 2)
        centers = np.vstack([lattice_centers[:6], lattice_centers[6:12], lattice_centers[12:18], lattice_centers[18:24], add_random_points(lattice_centers[24])])
        
        current_centers = centers.copy()
        step_size = alpha
        for iteration in range(max_iter):
            r = solve_radii_lp(current_centers)
            active_constraints = [np.argmin(np.array([
                current_centers[i, 0], 1 - current_centers[i, 0], current_centers[i, 1], 1 - current_centers[i, 1]
            ] + [np.sum((current_centers[i] - current_centers[j]) ** 2) - r[j] ** 2 for j in range(n) if i != j and r[j] is not None])) for i in range(n)
            ]
            for i in range(n):
                if active_constraints[i] < 4:
                    delta_x = [1 if x == current_centers[i, 0] else -1 if x == 1 - current_centers[i, 0] else 0 for x in [current_centers[i, 0], 1 - current_centers[i, 0]]]
                    delta_y = [1 if y == current_centers[i, 1] else -1 if y == 1 - current_centers[i, 1] else 0 for y in [current_centers[i, 1], 1 - current_centers[i, 1]]]
                    best_delta = None
                    best_val = float('-inf')
                    for dx, dy in zip(delta_x, delta_y):
                        new_centers = current_centers.copy()
                        new_centers[i, 0] += step_size * dx
                        new_centers[i, 1] += step_size * dy
                        new_centers = np.clip(new_centers, 0, 1)
                        new_r = solve_radii_lp(new_centers)
                        val = np.sum(new_r)
                        if val >= best_val:
                            best_val = val
                            best_delta = np.array([dx, dy])
                    current_centers[i, :] += step_size * best_delta
                else:
                    a = active_constraints[i] - 4
                    neighbor = a // n
                    delta_x = (current_centers[i, 0] - current_centers[neighbor, 0]) / np.sum((current_centers[i] - current_centers[neighbor]) ** 2)
                    delta_y = (current_centers[i, 1] - current_centers[neighbor, 1]) / np.sum((current_centers[i] - current_centers[neighbor]) ** 2)
                    current_centers[i, 0] += step_size * delta_x
                    current_centers[i, 1] += step_size * delta_y

            if iteration % escape_interval == 0:
                current_centers += np.random.normal(0, best_alpha / 2, size=(n, 2))
                current_centers = np.clip(current_centers, 0, 1)

            new_r = solve_radii_lp(current_centers)
            new_sum_radii = np.sum(new_r)
            if new_sum_radii > best_sum_radii:
                best_sum_radii = new_sum_radii
                best_centers = current_centers.copy()
                best_alpha = best_alpha.copy()
                step_size = alpha
            elif new_sum_radii == best_sum_radii and step_size > 1e-5:
                step_size *= 0.6

            if np.sum(np.abs(solve_radii_lp(current_centers) - r)) < 1e-7:
                break

            if iteration % cooling_interval == 0:
                step_size *= 0.6

    centers = best_centers
    radii = solve_radii_lp(centers)
    sum_radii = np.sum(radii)
    
    result = minimize(
        fun=lambda x: -np.sum(x[n:]), 
        x0=np.hstack((centers.flatten(), radii)), 
        method='SLSQP', 
        bounds=[(0, 1)] * n + [(0, max_radius)] * n,
        constraints={'type': 'ineq', 'fun': lambda x: np.array([
            ((np.linalg.norm(x[i*n:(i+1)*n] - x[j*n:(j+1)*n]) - x[j*n + n] - x[i*n + n]) >= 1e-12)
        for i in range(n) for j in range(i + 1, n)])
        },
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )

    if result.success:
        radii = result.x[n:]
        sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
