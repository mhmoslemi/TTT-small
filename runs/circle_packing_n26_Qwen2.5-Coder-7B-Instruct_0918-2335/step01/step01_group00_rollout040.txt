import numpy as np
from scipy.optimize import minimize

def obj(v):
    r = v[52:]                      # last 26 entries
    return -np.sum(r)

def constraints(v):
    pos = v[:52].reshape(26,2)      # (x,y) pairs
    r   = v[52:]                   # radii

    # boundary terms – 4*26
    cons = np.empty(104, float)
    cons[ 0:26] = pos[:,0] - r     # left/right
    cons[26:52] = 1 - pos[:,0] - r
    cons[52:78] = pos[:,1] - r
    cons[78:104] = 1 - pos[:,1] - r

    # pairwise terms – 26 choose 2
    diffs = pos[:,None,:] - pos[None,:,:]   # (26,26,2)
    dists = np.sqrt((diffs**2).sum(axis=2))
    mask  = np.triu(np.ones((26,26), bool), k=1)
    pair_cons = dists[mask] - (r[:,None] + r[None,:])[mask]
    return np.concatenate([cons, pair_cons])

def ini_pos():
    gold_ratio = (1 + np.sqrt(5)) / 2
    inner_radius = 0.2
    outer_radius = 0.3
    rows = int(np.ceil(np.sqrt(26 / np.pi)))
    cols = int(np.ceil(26 / np.pi / rows))
    x_spread = 1 / (cols + 1)
    y_spread = 1 / (rows + 1)
    x_off = x_spread / 2
    y_off = y_spread / 2
    arr = np.empty((rows * cols, 2))
    cnt = 0
    for i in range(rows):
        for j in range(cols):
            if cnt < 26:
                arr[cnt, 0] = x_off + j * x_spread
                arr[cnt, 1] = y_off + i * y_spread
                cnt += 1
    return arr[:26]

bounds = [(0,1)]*52          # x
bounds += [(0,1)]*52          # y
bounds += [(0,0.5)]*26        # radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    nrestarts = 20
    best_r = None
    best_res = None

    pos = ini_pos()

    for _ in range(nrestarts):
        init = np.concatenate([pos.reshape(-1), 0.1 * np.random.rand(26)])
        res = minimize(fun=obj, x0=init, method='SLSQP',
                       bounds=bounds, constraints={'type':'ineq',
                                                   'fun': constraints},
                       options={'maxiter':4000, 'ftol':1e-9, 'disp':False})
        if res.success and (best_res is None or -res.fun > -best_res.fun):
            if np.all(constraints(res.x) >= -1e-10):
                best_res = res
                best_r = res.x[52:]

    if best_res is None or best_r is None:
        raise ValueError("Optimization failed to find a valid solution")

    centers = best_res.x[:52].reshape(26, 2)
    sum_radii = -best_res.fun
    return centers, best_r, sum_radii
