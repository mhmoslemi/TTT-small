import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog, minimize
from numpy.random import default_rng

def lp_oracle(p, ubA, ubb):
    r = p[26:]
    A = np.vstack((ubA, np.diag(-2 * r), np.ones((26, 26)) - np.eye(26)))
    b = np.hstack((ubb, np.zeros(26), -np.inf))
    res = linprog(-r, A_ub=A, b_ub=b, method='highs', tol=1e-9, bounds=(0, 0.5))
    if not res.success or (res.x < 0).any():
        return -1e6
    return np.sum(res.x)

def apollonius_circle(pos, center):
    dists = distance_matrix([center], pos)
    rs = dists + 0.001  # Avoid tiny negative radii
    lambda_ij = (longest_segment(pos, rs) / 2) * (1 - rs / dists)
    lambda_ij[np.isnan(lambda_ij)] = 0
    lambda_0j = (dists - rs) / 2
    lambda_0j[dists <= rs] = 0
    lambda_i0 = (dists + rs) / 2
    lambda_i0[dists >= rs] = 0
    lambda_i0j = np.minimum(lambda_0j, lambda_i0)
    lambda_ij = np.minimum(lambda_ij, lambda_i0j)
    max_lambda = np.max(lambda_ij, axis=0)
    return max_lambda, 1 / (1 + np.max(lambda_ij, axis=0))

def longest_segment(pos, rs):
    n = pos.shape[0]
    dists = distance_matrix(pos, pos)
    return np.sum(rs * (dists < 2 * rs / np.cos(np.pi / n)), axis=0) - 1e-6

def build_swarm(seed_layouts, num_particles, jitter_std):

            pos[1::2] = np.clip(pos[1::2], 0, 1)
            swarm.append(pos)
    return np.array(swarm)[:num_particles]
