import numpy as np
from scipy.optimize import linprog
import random

def solve_lp(centers):
    n = len(centers)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    m = len(pairs)
    
    # Extract x and y coordinates of centers
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Build the objective function coefficients
    c = -np.ones(n)
    
    # Build the A_ub matrix
    A_ub = np.zeros((4*n + m, n))
    
    # Boundary constraints r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    for i in range(n):
        A_ub[4*i, i] = 1
        A_ub[4*i+1, i] = 1
        A_ub[4*i+2, i] = 1
        A_ub[4*i+3, i] = 1
    
    # Pairwise constraints r_i + r_j <= d_{ij}
    for k, (i, j) in enumerate(pairs):
        idx = 4*n + k
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
    
    b_ub = np.hstack([x, 1 - x, y, 1 - y, np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)])
    b_ub[4*n:] -= np.concatenate([[r_i + r_j for r_i, r_j in solve_lp(centers).radii] for _ in pairs])

    # Bounds on radii (r_i >= 0 and r_i <= 0.5)
    bounds = [(0, 0.5) for _ in range(n)]

    # Solve the linear programming problem
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    # Extract the optimal radii
    radii = result.x
    sum_radii = -result.fun

    return sum_radii, radii

def solve_initial_configurations():
    n = 26
    candidates = []
    
    # Generate candidates from a 5x5 grid pattern
    grid = np.linspace(0.1, 0.9, 5)
    grid_centers = np.array([[x, y] for x in grid for y in grid])
    jittered_centers = (grid_centers + 0.02 * (np.random.rand(n, 2) - 0.5)) @ np.array([[1, 0], [0, 1]])
    jittered_centers = np.clip(jittered_centers, 0, 1)
    candidates.append((jittered_centers,))

    # Generate candidates from a hexagonal packing pattern
    h = 0.18
    v = np.sqrt(3)/2 * h
    for off in [0.5, -0.5]:
        for i in range(-1, 6):
            for j in range(-1, 6):
                if i + j >= -1 and i + j <= 6:
                    centers = np.array([(i*h + off/2 * v, j*h) for j in range(1, 7) for i in range(-j+1, j)])
                    jittered_centers = (centers + 0.02 * (np.random.rand(n, 2) - 0.5)) @ np.array([[1, 0], [0, 1]])
                    jittered_centers = np.clip(jittered_centers, 0, 1)
                    candidates.append((jittered_centers,))

    # Generate candidates from an offset corner pattern
    corner_centers = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]])
    for i in range(5):
        for j in range(5):
            centers = np.vstack((corner_centers, np.array([[x, y] for x in np.linspace(0.2, 0.8, 5) for y in np.linspace(0.2, 0.8, 5)])))
            jittered_centers = (centers + 0.02 * (np.random.rand(n, 2) - 0.5)) @ np.array([[1, 0], [0, 1]])
            jittered_centers = np.clip(jittered_centers, 0, 1)
            candidates.append((jittered_centers,))
    
    # Evaluate each candidate and select the best one
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    for centers, in candidates:
        sum_radii, radii = solve_lp(centers)
        if sum_radii > best_sum_radii:
            best_sum_radii = sum_radii
            best_centers = centers
            best_radii = radii
    
    return best_centers, best_radii

def refine_positions(centers):
    n = len(centers)
    current_sum_radii = 0
    sigma = 0.05
    n_iter = 5000
    alpha = 0.9995
    T = 0.1
    
    for _ in range(n_iter):
        # Randomly pick one circle
        i = random.randint(0, n-1)
        # Perturb the position
        dx, dy = sigma * (np.random.randn(2,) - 0.5)
        centers[i] += [dx, dy]
        centers[i] = np.clip(centers[i], 0, 1)
        
        # Compute radii for the new center configuration
        _, radii = solve_lp(centers)
        
        # Check the new sum of radii
        new_sum_radii = np.sum(radii)
        
        # Accept or reject the move
        if new_sum_radii > current_sum_radii or random.random() < np.exp((new_sum_radii - current_sum_radii) / T):
            current_sum_radii = new_sum_radii
        else:
            centers[i] -= [dx, dy]
        
        # Move the temperature and sigma scheduler
        T *= alpha
        sigma *= alpha
    
    return centers

def run_packing():
    # Generate an initial configuration and find the corresponding radii
    centers, radii = solve_initial_configurations()
    
    # Refine the positions using simulated annealing
    refined_centers = refine_positions(centers)
    
    # Compute the final radii for the refined centers
    _, final_radii = solve_lp(refined_centers)
    
    # Ensure all radii are non-negative and reclip to ensure they do not exceed boundaries
    final_radii = np.clip(final_radii, 0, 0.5)
    final_sum_radii = np.sum(final_radii)
    
    # Return the final positions, radii, and sum of radii
    return refined_centers, final_radii, final_sum_radii
