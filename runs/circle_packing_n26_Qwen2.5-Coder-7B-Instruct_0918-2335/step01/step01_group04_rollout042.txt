def create_hexagonal_seed():
    centers = np.array([
        [0.25 + 0.33 * np.cos(it * 2 * np.pi / 6), 0.25 + 0.33 * np.sin(it * 2 * np.pi / 6)]
        for it in range(18)
    ])
    centers = np.array([
        [0.5 + c for c in centers[i]]
        for i in range(5)
    ]).flatten()
    centers = np.array(centers).reshape(-1, 2)
    centers = np.concatenate([centers, np.random.rand(n - centers.shape[0], 2)])
    np.clip(centers, 0, 1)
    return centers

def create_uniform_random_seed():
    centers = np.random.rand(n, 2)
    np.clip(centers, 0, 1)
    return centers
