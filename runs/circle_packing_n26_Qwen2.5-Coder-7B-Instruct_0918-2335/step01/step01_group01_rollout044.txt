import numpy as np
from scipy.optimize import linprog, minimize

def lp_oracle(pos):
    n = pos.shape[0] // 2
    A_bounds = np.block([[np.eye(n), -np.eye(n)], [-np.eye(n), np.eye(n)],
                          [[-1] * n, np.zeros(n)], [[np.zeros(n)], [-1] * n],
                          np.ones(n, n)])
    b_bounds = np.hstack([pos[:, 0], 1 - pos[:, 0], pos[:, 1], 1 - pos[:, 1], np.zeros(2 * n + 2)])
    A_pair = pos[:, None, :] - pos[None, :, :]
    dists = np.linalg.norm(A_pair, axis=2)
    mask = np.triu(np.ones((n, n), bool), k=1)
    A_pair = dists[mask] - (pos[:, None, 2:] + pos[None, :, 2:])[mask]
    A = np.vstack([A_bounds, A_pair])
    b = np.hstack([b_bounds, np.zeros(A_pair.size)])
    result = linprog(-pos[:, 2:], A_ub=A, b_ub=b, bounds=(0, 1), method='highs', options={'tol': 1e-9})
    if result.success:
        r = result.x
        r[r < 1e-12] = 0
        return np.sum(r)
    else:
        return 1e6
