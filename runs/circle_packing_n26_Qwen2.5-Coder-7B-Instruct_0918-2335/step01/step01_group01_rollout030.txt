import numpy as np
from scipy.optimize import linprog, minimize

def compute_optimal_radii(pos):
    n = 26
    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    
    for i in range(n):
        A_ub[i, i] = -1
        b_ub[i] = pos[i]
    
    for i in range(n):
        A_ub[n + i*n + 0, i] = 1
        b_ub[n + i*n + 0] = 1 - pos[i]
    
        A_ub[n + i*n + 1, i] = -1
        b_ub[n + i*n + 1] = pos[i]
    
        A_ub[n + i*n + 2, i] = 1
        b_ub[n + i*n + 2] = 1 - pos[i]
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = pos[j] - pos[i]
            dist = np.sqrt(np.sum(diff**2))
            A_ub[n + n*(i+1) + j - (i+1), i] = 1
            A_ub[n + n*(i+1) + j - (i+1), j] = 1
            b_ub[n + n*(i+1) + j - (i+1)] = dist - 1e-9
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return -res.fun

def objective(pos_flat):
    pos = pos_flat.reshape(26, 2)
    radii = compute_optimal_radii(pos_flat)
    if radii is None:
        return 1e9
    return -np.sum(radii)

def constraints(v):
    pos = v[:52].reshape(26, 2)
    r = v[52:]
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r

    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])
