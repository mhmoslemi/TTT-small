import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def evaluate_positions(centers):
    n = centers.shape[0]
    dim = centers.shape[1]
    c = -np.ones(n)
    h = np.eye(n)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pairs)
    A_ub = np.empty((4 * n + 2 * m, n))
    b_ub = np.empty(4 * n + 2 * m)
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        A_ub[idx + 1, i] = 1
        b_ub[idx + 1] = 1 - centers[i, 0]
        A_ub[idx + 2, i] = 1
        b_ub[idx + 2] = centers[i, 1]
        A_ub[idx + 3, i] = 1
        b_ub[idx + 3] = 1 - centers[i, 1]
        idx += 4
    for i, j in pairs:
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = np.sqrt(((centers[i] - centers[j]) ** 2).sum())
        idx += 1
    result = linprog(c, A_ub, b_ub, method='highs')
    radii = -result.x
    return radii, np.sum(radii)

def run_packing():
    n = 26
    dim = 2
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pairs)

    # Pattern A: 5x5 grid
    grid_points = np.linspace(0.1, 0.9, 5)
    grid_rect = np.meshgrid(grid_points, grid_points)
    points_A = np.column_stack((grid_rect[0].flatten(), grid_rect[1].flatten()))
    points_A = np.vstack([points_A, np.array([0.5, 0.5])])
    while points_A.shape[0] > 26:
        points_A = points_A[np.random.choice(points_A.shape[0], 26, replace=False)]

    # Pattern B: 3x3 hexagonal packing
    h = 0.2
    v = np.sqrt(3)/2 * h
    points_B = []
    for i in range(4):
        for j in range(4):
            points_B.append([i*h + (j%2)*h/2, j*v])
    points_B = np.array(points_B)
    points_B = np.vstack([points_B, np.array([0.5, 0.5])])
    while points_B.shape[0] > 26:
        points_B = points_B[np.random.choice(points_B.shape[0], 26, replace=False)]

    # Pattern C: Offset corner pattern
    corners = np.array([[0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]])
    grid_rect = np.meshgrid(np.linspace(0.1, 0.9, 6), np.linspace(0.1, 0.9, 6))
    points_C = np.column_stack((grid_rect[0].flatten(), grid_rect[1].flatten()))
    points_C = np.vstack([points_C, corners, np.array([0.5, 0.5])])
    while points_C.shape[0] > 26:
        points_C = points_C[np.random.choice(points_C.shape[0], 26, replace=False)]

    # Random jitter
    jitter = 0.02
    for points in [points_A, points_B, points_C]:
        points += np.random.uniform(-jitter, jitter, points.shape)

    # Evaluate and keep best
    best_centers = None
    best_sum_radii = 0
    for points in [points_A, points_B, points_C]:
        radii, sum_radii = evaluate_positions(points)
        if sum_radii > best_sum_radii:
            best_centers = points
            best_sum_radii = sum_radii

    # Simulated annealing to refine
    sigma = 0.05
    alpha = 0.9995
    n_iter = 5000
    T = 1.0
    for _ in range(n_iter):
        i = np.random.randint(n)
        delta = np.random.normal(0, sigma, dim)
        new_centers = np.copy(best_centers)
        new_centers[i] += delta
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        new_radii, new_sum_radii = evaluate_positions(new_centers)
        if new_sum_radii > best_sum_radii:
            best_centers = new_centers
            best_sum_radii = new_sum_radii
        elif np.random.rand() < np.exp((new_sum_radii - best_sum_radii) / T):
            best_centers = new_centers
            best_sum_radii = new_sum_radii
        sigma *= alpha

    # FinalLP for accurate radii
    _, best_sum_radii = evaluate_positions(best_centers)
    return best_centers, np.zeros(n), best_sum_radii
