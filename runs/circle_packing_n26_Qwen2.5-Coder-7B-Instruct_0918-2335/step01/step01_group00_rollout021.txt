import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def compute_radii_and_value(centres):
    w_limits = np.minimum.reduce([centres[:, 0], 1 - centres[:, 0], centres[:, 1], 1 - centres[:, 1]])
    D = squareform(pdist(centres))
    n = len(centres)
    
    c = -np.ones(n)
    A_ub = np.zeros((n * (n - 1) // 2 + n, n))
    b_ub = np.zeros(n * (n - 1) // 2 + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = D[i, j]
            idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = w_limits[i]
        idx += 1
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, w_limits), method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        return -np.inf, np.zeros(n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def objective(x):
        centres = x[:52].reshape(26, 2)
        return -np.sum(compute_radii_and_value(centres)[0])

    def sa_move(centres, i, sigma_temp):
        centres_copy = centres.copy()
        centres_copy[i, :] += np.random.normal(0, sigma_temp, 2)
        centres_copy[i, centres_copy[i] < 0] = 0
        centres_copy[i, centres_copy[i] > 1] = 1
        return centres_copy

    def sa_acceptance(old_value, new_value, temperature):
        return True if new_value >= old_value else np.exp((new_value - old_value) / temperature)

    def cooling_schedule(t):
        return 0.97 * t

    n_points = 26
    temp = 0.1
    n_restarts = 150
    n_iter = 2000
    beta = 0.02

    best_value = -np.inf
    best_solution = None

    for restart in range(n_restarts):
        centres = np.random.rand(n_points, 2) * 0.5 + 0.25  # Centered hexagonal lattice layout
        np.random.shuffle(centres)

        for i in range(n_iter):
            i_rand = np.random.randint(n_points)
            new_centres = sa_move(centres, i_rand, beta)
            value, _ = compute_radii_and_value(new_centres)

            if sa_acceptance(best_value, value, temp):
                centres = new_centres
                best_value = value
                if value >= best_value:
                    best_solution = centres

            temp = cooling_schedule(temp)

    if best_solution is None:
        raise ValueError("Optimization failed to find valid solution")

    final_value, final_radii = compute_radii_and_value(best_solution)
    return best_solution.astype(np.float64), final_radii.astype(np.float64), final_value
