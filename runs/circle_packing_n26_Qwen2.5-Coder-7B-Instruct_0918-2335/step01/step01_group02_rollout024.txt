import numpy as np
from scipy.optimize import linprog, minimize

def lp_for_radii(centers):
    n = len(centers)
    c = -np.ones(n)
    A = []
    b = []
    for i in range(n):
        A.append([1] * n)
        b.append(centers[i][0])
        A.append([1] * n)
        b.append(1 - centers[i][0])
        A.append([1] * n)
        b.append(centers[i][1])
        A.append([1] * n)
        b.append(1 - centers[i][1])
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j][0] - centers[i][0]
            dy = centers[j][1] - centers[i][1]
            dist = np.sqrt(dx**2 + dy**2) - 1e-12
            A.append([-1] * n)
            A.append([1] * n)
            b.append(dist)
    A = np.array(A)
    b = np.array(b)
    res = linprog(c, A_ub=A, b_ub=b, bounds='nonneg', method='highs')
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)] * 26 * 2
    initial = np.random.rand(26 * 2)
    SA_iterations = 300
    moves_per_SA_iteration = 50
    initial_temperature = 0.1
    cool_down = lambda T, k: T / np.log(1 + k)

    best_centers = initial.reshape(26, 2)
    best_r = lp_for_radii(best_centers)
    best_sum = np.sum(best_r)

    for T in range(SA_iterations):
        current_centers = best_centers.copy()
        for _ in range(moves_per_SA_iteration):
            i = np.random.randint(26)
            delta = np.random.normal(0, 0.05 * T)
            current_centers[i][0] += delta
            current_centers[i][1] += delta
            current_centers[i] = np.clip(current_centers[i], 0, 1)
            current_r = lp_for_radii(current_centers)
            current_sum = np.sum(current_r)
            if current_sum >= best_sum or np.random.random() < np.exp((current_sum - best_sum) / T):
                best_centers = current_centers
                best_r = current_r
                best_sum = current_sum
        initial_temperature = cool_down(initial_temperature, T)

    def objective(vars):
        centers = vars[:26 * 2].reshape(26, 2)
        radii = lp_for_radii(centers)
        return -np.sum(radii)

    constraints = [
        {'type': 'ineq', 'fun': lambda vars: vars[26 * 2:i + 26 * 2] - (1 - vars[:26 * 2].reshape(26, 2))[:, 0] - 1e-12},
        {'type': 'ineq', 'fun': lambda vars: vars[26 * 2:i + 26 * 2] - (vars[:26 * 2].reshape(26, 2))[:, 1] - 1e-12},
        {'type': 'ineq', 'fun': lambda vars: vars[26 * 2:i + 26 * 2] - (1 - vars[:26 * 2].reshape(26, 2))[:, 1] - 1e-12},
        {'type': 'ineq', 'fun': lambda vars: vars[26 * 2:i + 26 * 2] - (vars[:26 * 2].reshape(26, 2))[:, 0] - 1e-12}
    ]

    bounds += [(0, 0.5)] * 26 * 2
    result = minimize(objective, best_centers.flatten(), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 8000, 'ftol': 1e-9, 'disp': False})
    return result.x[:26 * 2].reshape(26, 2), lp_for_radii(result.x[:26 * 2].reshape(26, 2)), -result.fun

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
