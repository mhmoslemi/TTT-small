import numpy as np
from scipy.optimize import minimize

def distance_matrix(centers):
    n = centers.shape[0]
    dists = np.zeros((n,n), np.double)
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = dists[j, i] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
    
    return dists

def run_packing():
    n = 26
    
    def obj(v):
        return -np.sum(v[52:])
    
    def constraints(v):
        pos = v[:52].reshape(n, 2)  # (x,y) pairs
        r = v[52:]  # radii

        # Boundary terms - 4 * n
        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        # Pairwise terms - n choose 2
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    def initialize_centers(n):
        hexagon_side = np.sqrt(7) / 3
        grid_side = np.sqrt(2)
        center_y = 0.5 * grid_side + 0.1 * np.sqrt(7)
        centers = np.array([
            [0.5 + 0.1 * j, 0.1 + 0.1 * k]
            for j in range(7)
            for k in range(int(7 - j * 0.5)) if (j + k) % 2 == 0
        ])

        centers = hexagon_side * centers
        centers += (1 - hexagon_side) * 0.5
        centers += np.array([
            [0.5 - x for x in centers[:, 0]],
            [1 - y for y in centers[:, 1]]
        ])

        centers = centers.flatten()
        return centers
    
    centers = initialize_centers(n)
    radii = np.ones(n) * 0.1 * np.sqrt(7)
    
    x0 = np.concatenate([centers, radii])
    bounds = [(0, 1)] * (3 * n)
    bounds[52:] = [(0, None)] * n

    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 4000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")
