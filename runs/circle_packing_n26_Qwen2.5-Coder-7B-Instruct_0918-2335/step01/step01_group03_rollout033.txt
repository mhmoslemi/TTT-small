import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
from scipy.spatial.distance import pdist, squareform

def generate_positions(base_layout, sigma=0.02):
    n = 26
    np.random.seed(42)
    if base_layout == "hexagon":
        m = 6
        centers = np.zeros((n, 2))
        centers[:m] = np.array([(np.arange(m) % 2) / (2 * m-1), np.arange(m) // (2 * m-1)]).T * 2 - 1
        centers[m:2*m+1] = np.array([(np.arange(m) % 2) / (2 * m), (np.arange(m) // 2) / m]).T * 2 - 1
        centers[2*m+1:] = np.array([(np.arange(m+1) % 2) / (2 * m+1), (np.arange(m+1) // (2 * m+1))>).T * 2 - 1
    elif base_layout == "5x5-grid":
        centers = np.array([(i // 5 + 0.5 / 5) * 2 - 1, (i % 5 + 0.5 / 5) * 2 - 1] for i in range(25)])
        centers = np.vstack([centers, np.array([0.0, 0.5, 1.0, 0.5, 0.5])])
    elif base_layout == "hybrid":
        centers = np.array([(0.0, 0.0), (0.0, 0.5), (0.0, 1.0), (0.5, 0.0), (0.5, 1.0), (1.0, 0.0), (1.0, 0.5), (1.0, 1.0)])
        center_indices = np.arange(8)
        np.random.shuffle(center_indices)
        centers = centers[center_indices]
        edge_centers_x = np.linspace(-1 + 0.05, 1 - 0.05, 5)[1:-1]
        edge_centers_y = np.linspace(-1 + 0.05, 1 - 0.05, 5)[1:-1]
        edge_centers = [(x, y) for x in edge_centers_x for y in edge_centers_y]
        centers = np.vstack((centers, edge_centers[-4:]))
        edge_centers_x = np.linspace(-1 + 0.1, 1 - 0.1, 5)
        edge_centers_y = np.linspace(-1 + 0.1, 1 - 0.1, 5)
        edge_centers = [(x, y) for x in edge_centers_x for y in edge_centers_y]
        centers = np.vstack((centers, edge_centers))
    elif base_layout == "random":
        centers = np.random.rand(n, 2) * 2 - 1
    centers += np.random.normal(0, sigma, centers.shape)
    centers = np.clip(centers, -1, 1)
    return centers

def lp_optimize_radii(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    c = -np.ones(n)
    A = np.hstack((np.eye(n), -np.eye(n), -dists, np.eye(n)))
    b = np.hstack((np.zeros(n), np.zeros(n), -np.ones(n)))
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success and res.message == 'Optimization terminated successfully.':
        return res.x
    else:
        return None

def evaluate(solution):
    centers, radii = solution[:52].reshape(26, 2), solution[52:]
    if not validate_packing(centers, radii)[0]:
        return -np.inf
    return -np.sum(radii)

def refine_with_sa(centers, radii, SA_iterations=10000):
    n = centers.shape[0]
    n_vars = n * 2 + n
    bounds = [(0, 1) for _ in range(n_vars)]

    def penalty(obj):
        centers, radii = obj[:52].reshape(26, 2), obj[52:]
        sum_r = np.sum(radii)
        overlaps = ((radii[:, np.newaxis] + radii - pdist(centers, 'euclidean'))).clip(max=-1e-6) + 1e-6
        over_penalty = overlaps.sum() * 1e6
        boundary_penalty = (np.clip(radii, 0, None) - np.clip(radii, 0, None)).sum() * 1e6
        return sum_r - over_penalty - boundary_penalty

    obj_fun = lambda x: evaluate((x[:52].reshape(26, 2), x[52:])) + penalty(x)
    initial_sol = np.hstack((centers.flatten(), radii))

    options = {
        'T_max': 1.0,
        'T_min': 1e-4,
        'cool_rate': 0.995
    }

    result = differential_evolution(obj_fun, bounds, options=options, popsize=20, strategy='best2bin', tol=1e-5, maxiter=SA_iterations)
    return result.x[:52].reshape(26, 2), result.x[52:]

def run_packing():
    candidates = []
    bases = ["hexagon", "5x5-grid", "hybrid", "random"]
    distances = pdist(np.array([[0, 0], [0, 1], [1, 0], [1, 1]]))
    max_radius = np.min(distances) / 2.
    sigmas = [0.05, 0.02]
    SA_iterations = [10000, 5000]

    for base in bases:
        for sigma in sigmas:
            for SA_iter in SA_iterations:
                centers0 = generate_positions(base, sigma)
                for _ in range(5):  # Restart about 5 times per configuration
                    centers = centers0 + np.random.normal(0, sigma / 2., centers0.shape)
                    centers = np.clip(centers, -1, 1)
                    radii = lp_optimize_radii(centers)
                    if radii is not None:
                        sum_r = np.sum(radii)
                        candidates.append((centers, radii, sum_r))

                        refined = refine_with_sa(centers, radii, SA_iter)
                        candidates.append(refined)

    best = max(candidates, key=lambda c: c[2])
   centers, radii = best[0], best[1]
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

centers, radii, sum_radii = run_packing()
print("Final result:", sum_radii)
