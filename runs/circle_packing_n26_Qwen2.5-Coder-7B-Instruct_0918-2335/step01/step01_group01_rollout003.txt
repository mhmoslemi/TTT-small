import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform
from random import random, seed
from itertools import combinations

# Static matrices for the LP problem - precomputed once
# A_ub, b_ub will be filled in dynamically for each evaluation
A_ub_static = np.ones((52, 26))
b_ub_static = np.ones(26)

def boundary_constraints(x):
    r = np.sqrt(np.sum(x[::2]**2 + x[1::2]**2, axis=0) + 0.5**2)
    return r

def pairwise_constraints(x):
    d = squareform(pdist(x.reshape(-1, 2)))
    n = len(d)
    A_eq = np.vstack([x.reshape(-1, 2)[i] - x.reshape(-1, 2)[j] for i, j in combinations(range(n), 2)])
    b_eq = d[np.triu_indices(n, k=1)]
    return A_eq, b_eq

def make_LP(x):
    global A_ub_static, b_ub_static
    A_ub = np.vstack([A_ub_static, boundary_constraints(x).reshape(-1, 1).T, pairwise_constraints(x)[0]])
    b_ub = np.hstack([b_ub_static, boundary_constraints(x), pairwise_constraints(x)[1]])
    c = -np.ones(26)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        r = np.maximum(result.x, 0.).clip(max=0.5)
        return r
    return np.zeros(26)

def lp_opt(x):
    r = make_LP(x)
    if np.any(r < 1e-12):
        r = np.clip(r, 0, 0.5)
    return r

def fitness(x):
    r = lp_opt(x)
    return np.sum(r)

def generate_initial_population():
    hex_grid = []
    random_points = []
    grid_points = []
    corner_heavy = []

    # Hex grid
    n_circles = 26
    R = 0.5 * np.tan(np.pi / 3)
    offset = 0.2 * random()
    y_offset = R + offset
    x_offset = 0.5 * R * np.sqrt(3)
    for i in range(6):
        for j in range(i % 2, 6, 2):
            x = x_offset * j + np.random.uniform(-0.01, 0.01)
            y = y_offset * i + np.random.uniform(-0.01, 0.01)
            hex_grid.append([x, y])
    if len(hex_grid) > n_circles:
        hex_grid = hex_grid[:n_circles]
    else:
        hex_grid += [np.random.rand(2) for _ in range(n_circles - len(hex_grid))]
    hex_grid = np.concatenate(hex_grid, axis=0)

    # Random points
    random_points = np.random.rand(n_circles, 2)

    # Grid points
    grid_points_1 = np.array([(i / 5. + 0.25, j / 5. + 0.25) for i in range(5) for j in range(5)])
    grid_points_2 = np.array([(1/2. - 0.05, 1/2. - 0.1), (1/2., 1/2. - 0.05), (1/2. + 0.05, 1/2. - 0.1), (1/2. - 0.05, 1/2.), (1/2., 1/2.), (1/2. + 0.05, 1/2. - 0.05), (1/2. - 0.05, 1/2. + 0.1), (1/2., 1/2. + 0.05), (1/2. + 0.05, 1/2.), (1/2. + 0.2, 1/2. + 0.2)])
    grid_points = np.vstack([grid_points_1, grid_points_2, [(1/2., 1/2.)]])
    if len(grid_points) > n_circles:
        grid_points = grid_points[:n_circles]
    else:
        grid_points += [np.random.rand(2) for _ in range(n_circles - len(grid_points))]
    grid_points = np.concatenate(grid_points, axis=0)

    # Corner-heavy points
    corner_heavy = [(0.1 + 0.1 * random(), 0.1 + 0.1 * random()) for _ in range(4)] + \
                    [(0.1 + 0.1 * random(), 0.9 - 0.1 * random()) for _ in range(4)] + \
                    [(0.9 - 0.1 * random(), 0.1 + 0.1 * random()) for _ in range(4)] + \
                    [(0.9 - 0.1 * random(), 0.9 - 0.1 * random()) for _ in range(4)] + \
                    [(0.5 + 0.1 * random(), 1/2.) for _ in range(7)] + \
                    [(0.05 + 0.1 * random(), 0.5) for _ in range(3)] + \
                    [(0.95 + 0.1 * random(), 0.5) for _ in range(3)] + \
                    [(0.5, 0.1 + 0.1 * random()) for _ in range(3)] + \
                    [(0.5, 0.9 - 0.1 * random()) for _ in range(3)]
    if len(corner_heavy) > n_circles:
        corner_heavy = corner_heavy[:n_circles]
    else:
        corner_heavy += [np.random.rand(2) for _ in range(n_circles - len(corner_heavy))]
    corner_heavy = np.concatenate(corner_heavy, axis=0)

    return [hex_grid, random_points, grid_points, corner_heavy]

def tournament_select(pop):
    i = np.random.choice(len(pop), 3).argmax()
    return pop[i][0]

def crossover(a, b):
    rng = np.random.uniform(0, 1, a.size)
    return np.where(rng < 0.5, a, b) + np.random.uniform(-0.005, 0.005, a.size)

def mutate(x):
    m = np.random.randint(1, 4, 1)[0]  # Number of circles to mutate
    while m > 0:
        i = np.random.randint(x.size // 2, len(x))
        dx = np.random.normal(0, 0.02)
        dy = np.random.normal(0, 0.02)
        if x[i] + dx >= 0 and x[i] + dx <= 1:
            x[i] += dx
        if x[i + 1] + dy >= 0 and x[i + 1] <= 1:
            x[i + 1] += dy
        m -= 1
    return x

def local_slsqp(x):
    x0 = np.copy(x)
    bnds = tuple((0, 1) for _ in range(len(x0)))
    constr = {'type': 'eq', 'fun': lambda x: fitness(x) - fitness(x0)}
    result = minimize(fitness, x0, constraints=constr, bounds=bnds, method='SLSQP', options={'maxiter': 25})
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    population = generate_initial_population()
    best_dist = 0.0
    best_center = np.zeros((26, 2))
    best_rad = np.zeros(26)
    EVAL = 4000
    POP = 30

    for _ in range(EVAL):
        a = tournament_select(population)
        b = tournament_select(population)
        c = crossover(a, b)
        mutate(c)
        c = local_slsqp(c)
        sum_rad = fitness(c)
        
        if sum_rad > best_dist:
            best_dist = sum_rad
            best_center = c.reshape(26, 2)
            best_rad = lp_opt(c)

    final_fit = minimize(fitness, best_center.flatten(), constraints={'type': 'eq', 'fun': lambda x: fitness(x) - best_dist}, bounds=[(0, 1)] * len(best_center), method='SLSQP')
    if final_fit.success:
        best_rad = lp_opt(best_center.flatten())
        best_center = best_center.flatten().reshape(26, 2)
    else:
        best_rad = lp_opt(best_center.flatten())
        
    return best_center, best_rad, best_dist + np.sum(best_rad)
