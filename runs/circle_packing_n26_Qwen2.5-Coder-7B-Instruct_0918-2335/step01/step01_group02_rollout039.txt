import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial import distance_matrix

def objective(x):
    return -np.sum(x[26:52])

def constraints(x):
    cx = x[:26].reshape(26, 2)
    cr = x[26:52]
    n = 52
    cons = np.empty(n + 100, float)
    cons[:n] = cx[:, 0] - cr
    cons[n:2*n] = 1 - cx[:, 0] - cr
    cons[2*n:3*n] = cx[:, 1] - cr
    cons[3*n:4*n] = 1 - cx[:, 1] - cr
    diffs = cx[:, None, :] - cx[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((26, 26), bool), k=1)
    pair_cons = (cr[:, None] + cr[None, :])[mask] - dists[mask]
    cons[4*n:] = pair_cons
    return cons

def sa_step(x, T):
    np.random.seed(12345)
    idx = np.random.choice(26)
    dT = 0.05 * T
    x_new = np.copy(x)
    x_new[:26] += dT * np.random.randn(2, 26)
    x_new = np.clip(x_new, 0, 1)
    theta = objective(x)
    eta = objective(x_new)
    if eta >= theta or np.exp((eta - theta) / T) > np.random.rand():
        return x_new
    else:
        return x

def run_packing():
    np.random.seed(12345)
    x0 = np.random.rand(52)
    T0 = 0.1
    kmax = 300
    m = 50
    best_x = np.copy(x0)
    best_theta = objective(x0)

    for k in range(kmax):
        for _ in range(m):
            x = sa_step(x0, T0)
            theta = objective(x)
            if theta > best_theta:
                best_x = np.copy(x)
                best_theta = theta

        T0 /= np.log(1 + k)

    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0, 1)] * 26 + [(0, 0.5)] * 26
    res = minimize(objective, best_x, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 8000, 'ftol': 1e-9, 'disp': False})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        return (res.x[:26].reshape(26, 2), res.x[26:], -res.fun)
    else:
        raise Exception("Optimization failed")

centers, radii, sum_radii = run_packing()
print(f"Total Radius Sum: {sum_radii}")
