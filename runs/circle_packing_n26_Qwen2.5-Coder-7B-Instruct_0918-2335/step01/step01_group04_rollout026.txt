import numpy as np
from scipy.optimize import linprog, minimize

def compute_radii(centers):
    n = centers.shape[0]
    max_radius = 0.5
    r = np.zeros(n)
    A = np.zeros((2*n + n*(n-1)//2, 2*n + n))
    b = np.zeros(2*n + n*(n-1)//2)
    
    # Wall constraints
    for i in range(n):
        j = i*2
        A[j, i*2:i*2+2] = [1, 0]
        b[j] = centers[i, 0]
        A[j+1, i*2:i*2+2] = [-1, 0]
        b[j+1] = 1 - centers[i, 0]
        A[j+2, i*2+2:i*2+4] = [0, 1]
        b[j+2] = centers[i, 1]
        A[j+3, i*2+2:i*2+4] = [0, -1]
        b[j+3] = 1 - centers[i, 1]
    
    # Pairwise constraints
    idx = 2*n
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A[idx, i*2:i*2+2] = [1, 0]
            A[idx, j*2:j*2+2] = [-1, 0]
            b[idx] = dist
            idx += 1
    
    res = linprog(-r, A_ub=A, b_ub=b, A_eq=None, b_eq=None, bounds=(None, max_radius), method='highs')
    return -res.x.reshape(-1)

def push_away(centers, radii):
    n = centers.shape[0]
    alpha = 0.01
    for _ in range(20):
        C_new = centers.copy()
        active_constraints = np.zeros(n, dtype=int)
        for i in range(n):
            dist_to_left = centers[i, 0] - radii[i]
            dist_to_right = 1 - centers[i, 0] - radii[i]
            dist_to_bottom = centers[i, 1] - radii[i]
            dist_to_top = 1 - centers[i, 1] - radii[i]
            distance_to_neighbours = []
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist == radii[i] + radii[j]:
                    distance_to_neighbours.append((dist, j))
            
            min_distance = float('inf')
            for dist, j in distance_to_neighbours:
                if dist < min_distance:
                    min_distance = dist
                    min_j = j
            
            if dist_to_left <= 0 and active_constraints[i] == 0:
                C_new[i, 0] += alpha
                active_constraints[i] = 1
            elif dist_to_right <= 0 and active_constraints[i] == 0:
                C_new[i, 0] -= alpha
                active_constraints[i] = 2
            elif dist_to_bottom <= 0 and active_constraints[i] == 0:
                C_new[i, 1] += alpha
                active_constraints[i] = 3
            elif dist_to_top <= 0 and active_constraints[i] == 0:
                C_new[i, 1] -= alpha
                active_constraints[i] = 4
            elif min_distance <= 0 and active_constraints[i] == 0:
                u_x = (centers[i, 0] - centers[min_j, 0]) / dist_to_neighbours[0][0]
                u_y = (centers[i, 1] - centers[min_j, 1]) / dist_to_neighbours[0][0]
                C_new[i] += alpha * np.array([u_x, u_y])
                active_constraints[i] = 5
        
        centers = np.clip(C_new, 0, 1)
        if np.sum(compute_radii(centers)) > np.sum(radii):
            centers = C_new
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    box_size = 1.0
    max_radius = box_size / 2
    
    # Initialize seeds
    seeds = [
        np.vstack([0.5 * np.cos(np.linspace(0, 2 * np.pi, 26, endpoint=False)), 0.5 * np.sin(np.linspace(0, 2 * np.pi, 26, endpoint=False))]).T,
        np.random.rand(n, 2),
        np.array([[0.2, 0.2], [0.8, 0.2], [0.5, 0.5]] + list(np.random.rand(n-3, 2)))
    ]
    
    best_sum radii = -np.inf
    best_centers = np.zeros((n, 2))
    
    for seed in seeds:
        radii = compute_radii(seed)
        cur_sum = np.sum(radii)
        if cur_sum > best_sum:
            best_centers = seed
            best_sum = cur_sum
        
        for _ in range(2):
            centers, radii = push_away(best_centers, radii)
            cur_sum = np.sum(radii)
            if cur_sum > best_sum:
                best_centers = centers
                best_sum = cur_sum
    
    best_r = compute_radii(best_centers)
    best_sum = np.sum(best_r)
    return best_centers, best_r, best_sum

def objective(x):
    return -np.sum(x[52:])

def constraints(x):
    pos = x[:52].reshape(n, 2)
    r = x[52:]
    
    cons = np.empty(104, float)
    cons[0:26] = pos[:, 0] - r
    cons[26:52] = 1 - pos[:, 0] - r
    cons[52:78] = pos[:, 1] - r
    cons[78:104] = 1 - pos[:, 1] - r
    
    diffs = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt((diffs ** 2).sum(axis=2))
    mask = np.triu(np.ones((n, n), bool), k=1)
    pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
    return np.concatenate([cons, pair_cons])

res = minimize(
    fun=objective,
    x0=np.concatenate((best_centers.flatten(), best_r)),
    method='SLSQP',
    bounds=[(0, box_size)] * n + [(0, box_size)] * n + [(0, max_radius)] * n,
    constraints={'type': 'ineq', 'fun': constraints},
    options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
)

if not res.success or np.any(constraints(res.x) < -1e-10):
    return np.zeros((n, 2)), np.zeros(n), 0.0

centers = res.x[:52].reshape(n, 2)
radii = res.x[52:]
sum_radii = np.sum(radii)

return centers, radii, sum_radii
