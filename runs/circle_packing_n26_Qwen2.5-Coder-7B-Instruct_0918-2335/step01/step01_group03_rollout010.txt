import numpy as np
from scipy.optimize import minimize
import scipy.spatial

def sample_speeds(base_speeds, multiplierfunc=np.random.lognormal):
    n = len(base_speeds)
    multipliers = multiplierfunc(mean=0., sigma=0.4, size=n)
    return multipliers * base_speeds

def inflate_discs(centers, geometry, speeds):
    n = len(speeds)
    active = set(range(n))
    t_stop = [None] * n
    dists = scipy.spatial.distance.pdist(centers)
    delta_t = (dists / (speeds[None, :] + speeds[:, None]))[:, np.triu_indices(n, k=1)]
    edge_t = np.minimum.accumulate(np.minimum.accumulate(np.minimum(centers[:, 0], 1 - centers[:, 0]), np.minimum(centers[:, 1], 1 - centers[:, 1])), axis=0)
    while active:
        t_min = min(delta_t.min(), edge_t.min())
        stop_set = {i for i in active if edge_t[i] - t_min < 1e-12 or np.any(delta_t[delta_t[:, 1] == t_min][:, 0] - t_min < 1e-12)}
        for i in stop_set:
            t_stop[i] = t_min
            active.remove(i)
    radii = np.array([s * ts for s, ts in zip(speeds, t_stop) if ts is not None])
    return centers, radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = scipy.spatial.distance.euclidean(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def simulate_packing(centers, target_sum, sample_size=3000, refine=True):
    best_centers, best_radii, best_sum = centers, np.zeros(26), target_sum + 1
    geometry = {
        "grid": np.array([
            [j/5, k/5] for j in range(6) for k in range(6)
        ]).reshape(26, 2)[:26],
        "hex": np.array([
            [1/3, 1/3], [2/3, 1/3], [1/2, 2/3], [0, 1/3], [1, 1/3], [1/2, 0],
            [1/6, 2/3], [5/6, 2/3], [1/6, 1/3], [5/6, 1/3], [1/6, 1/6], [5/6, 1/6],
            [1/3, 2/3], [2/3, 2/3], [1/2, 2/3], [1/3, 2/3], [2/3, 2/3], [1/2, 2/3],
            [1/6, 1/3], [5/6, 1/3], [1/6, 1/6], [5/6, 1/6], [1/6, 1/3], [5/6, 1/3]
        ]),
        "hybrid": np.array([
            [1, 1], [0, 0], [1, 0], [0, 1], [1/2, 1/2],
            [1/6, 1/6], [5/6, 1/6], [1/6, 5/6], [5/6, 5/6],
            [1/18, 5/9], [17/18, 5/9], [5/18, 17/18], [13/18, 17/18],
            [1/9, 10/9], [2/9, 1/9], [2/9, 10/9], [8/9, 1/9], [8/9, 10/9],
            [1/9, 2/9], [1/9, 17/18], [2/9, 17/18], [2/9, 2/9]
        ]),
        "random": np.random.rand(26, 2)
    }
    np.random.seed(0)
    for family, centers in geometry.items():
        for _ in range(sample_size):
            speeds = sample_speeds(1 / (np.minimum(centers[:, 0], 1 - centers[:, 0]) + np.minimum(centers[:, 1], 1 - centers[:, 1])) + 1)
            centers, radii = inflate_discs(centers, family, speeds)
            if radii.min() > 0:
                current_sum = np.sum(radii)
                if current_sum >= best_sum:
                    best_centers, best_radii, best_sum = centers, radii, current_sum
        if refine:
            x0 = np.concatenate((best_centers.flatten(), best_radii))
            res = minimize(fun=lambda x: -np.sum(x[26:]), x0=x0, method='SLSQP', bounds=[(0, 1)] * 52 + [(0, 0.5)] * 26, 
                           constraints={'type': 'ineq', 'fun': lambda x: np.diff([x[:52], x[52:]], axis=1).min()},
                           options={'maxiter': 2000, 'ftol': 1e-9})
            if res.success and validate_packing(res.x[:52].reshape((26, 2)), res.x[52:])[0]:
                best_centers, best_radii, best_sum = res.x[:52].reshape((26, 2)), res.x[52:], np.sum(res.x[52:])
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (52 + n)  # 52 centres + n radii
    x0 = np.concatenate((np.array([[0.5, 0.5] for _ in range(n)]).flatten(), np.zeros(n)))  # Initial guess for centres and radii

    def obj(v):
        return -np.sum(v[52:])

    def constraints(v):
        pos = v[:52].reshape(n, 2)  # (x,y) pairs
        r = v[52:]  # radii

        # Boundary terms - 4 * n
        cons = np.empty(4 * n, float)
        cons[0:n] = pos[:, 0] - r
        cons[n:2*n] = 1 - pos[:, 0] - r
        cons[2*n:3*n] = pos[:, 1] - r
        cons[3*n:4*n] = 1 - pos[:, 1] - r

        # Pairwise terms - n choose 2
        diffs = pos[:, None, :] - pos[None, :, :]  # (n,n,2)
        dists = np.sqrt((diffs ** 2).sum(axis=2))
        mask = np.triu(np.ones((n, n), bool), k=1)
        pair_cons = dists[mask] - (r[:, None] + r[None, :])[mask]
        return np.concatenate([cons, pair_cons])

    res = minimize(fun=obj, x0=x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 3000, 'ftol': 1e-9})

    if res.success and np.all(constraints(res.x) >= -1e-10):
        centers = res.x[:52].reshape(n, 2)
        radii = res.x[52:]
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")
