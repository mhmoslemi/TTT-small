import numpy as np
from scipy.optimize import linprog

def solve_radii(pos):
    n = pos.shape[0]
    A_ub = []
    b_ub = []

    # Boundary constraints
    for i in range(n):
        A_ub.append([1 if j == 2 * i else 0 for j in range(2 * n)])
        b_ub.append(pos[i, 0])

        A_ub.append([1 if j == 2 * i + 1 else 0 for j in range(2 * n)])
        b_ub.append(1 - pos[i, 0])

        A_ub.append([1 if j == 2 * i else 0 for j in range(2 * n)])
        b_ub.append(pos[i, 1])

        A_ub.append([1 if j == 2 * i + 1 else 0 for j in range(2 * n)])
        b_ub.append(1 - pos[i, 1])

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((pos[i, 0] - pos[j, 0]) ** 2 + (pos[i, 1] - pos[j, 1]) ** 2)
            A_ub.append([1 if k == 2 * i else 0 if k == 2 * i + 1 else 1 if k == 2 * j else 0 for k in range(2 * n)])
            b_ub.append(d)

    c = -np.ones(2 * n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    r_opt = result.x[:n]
    sum_r = -result.fun
    return r_opt, sum_r

def fitness(pos):
    r, s_r = solve_radii(pos)
    return -s_r

def crossover(parent1, parent2, rate):
    child = np.empty(52)
    for i in range(52):
        if np.random.rand() < rate:
            child[i] = parent2[i]
        else:
            child[i] = parent1[i]
    return child

def mutate(individual, rate, sigma):
    result = np.copy(individual)
    for i in range(52):
        if np.random.rand() < rate:
            result[i] += np.random.normal(0, sigma)
            result[i] = max(0, min(1, result[i]))
    return result

def tournament(pop, fitness_vals, k):
    indices = np.random.choice(np.arange(pop.shape[0]), size=k)
    return indices[np.argmax(fitness_vals[indices])]
