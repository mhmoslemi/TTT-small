import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.spatial.distance import squareform, pdist

def lp_radius_fixed_positions(centers):
    radii = np.zeros(26)
    for i in range(26):
        x, y = centers[i]
        min_radius = np.min([x, 1 - x, y, 1 - y])
        overlap_constraints = []
        for j in range(i + 1, 26):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist > 0:
                overlap_constraints.append(dist - (radii[j] + radii[i]))
            else:
                overlap_constraints.append(0)
        if any([c <= 0 for c in overlap_constraints]):
            min_radius = 0
        radii[i] = min_radius
    return -np.sum(radii)  # Return negative sum for minimization

def constrain(func):
    def constrained_func(x):
        if any(x < 0) or any(x >= 1):
            return np.inf
        return func(x)
    return constrained_func

def fitness_func(positions):
    centers = positions[:52].reshape(26, 2)
    dist_matrix = squareform(pdist(centers))
    A_ub = []
    b_ub = []
    A_ub.append(np.hstack([np.eye(26), -np.eye(26), np.zeros(26)]))
    b_ub.append(np.zeros(26))
    for i in range(26):
        x_range, y_range = centers[i]
        A_ub.append([-1*x_range, -1*y_range, -1])
        b_ub.append(-1)
        A_ub.append([1 - x_range, 1 - y_range, -1])
        b_ub.append(-1)
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub.append([1, 1, 0] - dist_matrix[i, j])
            b_ub.append(-1)
    return constrain(lp_radius_fixed_positions)(np.hstack((positions, np.zeros(26))))

def run_packing():
    bounds = [(0, 1)] * 52
    bounds += [(0, 1)] * 52
    bounds += [(0, 0.5)] * 52

    def obj_and_grad(x):
        obj_val = fitness_func(x)
        return obj_val, np.zeros_like(x)

    soln = differential_evolution(fitness_func, bounds=bounds, strategy='best1bin', popsize=50, maxiter=200,
                                    seed=42, polish=False)
    best_positions = soln.x[:52]
    centers = best_positions.reshape(26, 2)
    dist_matrix = squareform(pdist(centers))
    A_ub = []
    b_ub = []
    A_ub.append(np.hstack([np.eye(26), -np.eye(26), np.zeros(26)]))
    b_ub.append(np.zeros(26))
    for i in range(26):
        x_range, y_range = centers[i]
        A_ub.append([-1*x_range, -1*y_range, -1])
        b_ub.append(-1)
        A_ub.append([1 - x_range, 1 - y_range, -1])
        b_ub.append(-1)
    for i in range(26):
        for j in range(i + 1, 26):
            A_ub.append([1, 1, 0] - dist_matrix[i, j])
            b_ub.append(-1)
    cons = {'type': 'ineq', 'fun': lambda x: np.linalg.norm(fitness_func(x), ord=np.inf)}
    res = minimize(func=fitness_func, x0=best_positions, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-9})
    centers = res.x[:52].reshape(26, 2)
    radii = np.array([-lp_radius_fixed_positions(centers), radii])[0]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
