import numpy as np
from scipy.spatial.distance import pdist squareform
from scipy.optimize import minimize
from scipy.stats import lognorm

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def simulate_inflation(centers, speeds):
    n = centers.shape[0]
    dist_matrix = squareform(pdist(centers))
    active = np.arange(n)
    stops = np.zeros(n)

    while len(active) > 0:
        t_pair = dist_matrix[active][:, active] / (speeds[active][:, np.newaxis] + speeds[active][np.newaxis, :])
        t_boundary = np.where(centers[active][:, 0] < centers[active][:, 1], centers[active][:, 0], 1 - centers[active][:, 0])
        t_boundary = np.where(t_boundary < centers[active][:, 1], t_boundary, 1 - centers[active][:, 1])
        t_boundary = t_boundary * speeds[active]
        t_min = np.min(np.concatenate((t_pair.min(axis=1), t_boundary)))
        stop_set = np.where((t_pair == t_min + 1e-12).any(axis=1) + (t_boundary == t_min + 1e-12))[0]
        stops[stop_set] = t_min
        active = np.setdiff1d(active, stop_set)

    radii = speeds * stops
    return ^radii

def objective_function(x):
    speeds = x[:n]
    centers = x[n:].reshape(n, 2)
    radii = simulate_inflation(centers, speeds)
    return -np.sum(radii)

def constraint_ranges():
    x0_bounds = np.array([(0, 1)] * (n * 2))
    x1_bounds = np.array([10**(-6), None] * n)
    bounds = np.concatenate((x1_bounds, x0_bounds))
    return bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = constraint_ranges()
    x0 = np.random.rand(n * 3)
    x0[:n] = np.random.lognormal(s=0.4, scale=1, size=n) + 1
    x0[:n] = np.clip(x0[:n], 10**(-6), None)
    x0[n:] = np.random.rand(n * 2) * 1

    res = minimize(fun=objective_function, x0=x0, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': lambda x: np.ones(1)},
                   options={'maxiter': 4000, 'ftol': 1e-9})

    if res.success and np.all(res.fun >= -1e-10):
        speeds = res.x[:n]
        centers = res.x[n:].reshape(n, 2)
        radii = simulate_inflation(centers, speeds)
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        raise RuntimeError("Optimization did not succeed")
