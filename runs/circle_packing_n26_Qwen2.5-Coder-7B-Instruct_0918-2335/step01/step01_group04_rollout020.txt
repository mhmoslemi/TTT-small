import numpy as np
from scipy.optimize import linprog, minimize

def distance_matrix(centers):
    n = len(centers)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
            dists[j, i] = dists[i, j]
    return dists

def R_matrix(radii):
    n = len(radii)
    R = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            R[i, j] = radii[i] + radii[j]
            R[j, i] = R[i, j]
    return R

def solve_radii_LP(centers, R, b):
    n = len(centers)
    c = np.ones(n)
    A = R
    b1 = b
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b1, bounds=bounds, method='highs')
    return res.x

def active_constraints(centers, radii):
    n = len(centers)
    dists = distance_matrix(centers)
    Rs = R_matrix(radii)
    constraints = []
    for i in range(n):
        constraints.append(radii[i] - dists[i, :].clip(max=0))  # boundary constraints
        for j in range(i + 1, n):
            constraints.append(radii[i] + radii[j] - dists[i, j].clip(max=0))  # pairwise constraints
    active_indices = []
    for i in range(n):
        for j, constraint in enumerate(constraints):
            if abs(constraint[i]) < 1e-9:
                active_indices.append(j)
    return active_indices

def step_size(radii):
    return 0.01 * np.max(abs(radii))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_radius = 0.02
    max_iter = 200
    cooling_interval = 20
    epsilon = 1e-12
    patience = 10
    accept_rate = 0.5
    escape_interval = 50
    seeds = [
        np.random.rand(n, 2),  # random seed
    ]
    best_centers = None
    best_radii = None
    best_sum_radii = -np.inf

    for seed in seeds:
        centers = seed.copy()
        sum_radii = np.sum(initial_radius * np.ones(n))
        alpha = step_size(initial_radius * np.ones(n))

        for iteration in range(max_iter):
            r = solve_radii_LP(centers, R_matrix(radii), centers)
            radii = r
            active_indices = active_constraints(centers, radii)

            improved = np.zeros(n)
            for i in range(n):
                if len(active_indices) > 0:
                    if active_indices[0] < 2 * n:  # boundary constraint (0-25)
                        if active_indices[0] < n:  # left
                            df = np.array([1, 0, 0, 0])
                        elif active_indices[0] == n:  # right
                            df = np.array([-1, 0, 0, 0])
                        elif active_indices[0] == n + 1:  # bottom
                            df = np.array([0, 1, 0, 0])
                        elif active_indices[0] == n + 2:  # top
                            df = np.array([0, -1, 0, 0])
                    else:  # pairwise constraint (26-n-26)
                        j = active_indices[0] - 2 * n
                        df = (centers[i] - centers[j]) / np.linalg.norm(centers[i] - centers[j])
                else:
                    df = np.zeros(4)
                step = alpha * df
                new_centers = centers.copy()
                new_centers[i, 0] += step[0]
                new_centers[i, 1] += step[1]
                new_centers[i, 2] += step[2]
                new_centers[i, 3] += step[3]
                new_centers = np.clip(new_centers, 0, 1)
                new_r = solve_radii_LP(new_centers, R_matrix(radii), new_centers)
                new_sum_radii = np.sum(new_r)
                if new_sum_radii > sum_radii + epsilon:
                    centers = new_centers.copy()
                    sum_radii = new_sum_radii
                    improved[i] = 1
                else:
                    centers = centers.copy()

            alpha *= 0.5
            if iteration % cooling_interval == 0:
                alpha /= 2
            if sum_radii > best_sum_radii:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum_radii = sum_radii
            if np.sum(improved) / n < accept_rate and sum_radii - best_sum_radii < epsilon * patience:
                break

        for i in range(escape_interval):
            center_noise = np.random.randn(n, 2) * alpha / 2
            new_centers = best_centers + center_noise
            new_centers = np.clip(new_centers, 0, 1)
            r = solve_radii_LP(new_centers, R_matrix(best_radii), new_centers)
            new_radii = r
            new_sum_radii = np.sum(new_radii)
            if new_sum_radii > best_sum_radii:
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
                best_sum_radii = new_sum_radii

    final_centers = best_centers[:, :2]
    final_radii = best_radii
    final_sum_radii = best_sum_radii

    step_size = step_size(final_radii)
    bounds = [(0, None) for _ in range(2 * n)] + [(0, step_size) for _ in range(n)]
    constraints = ({'type': 'ineq', 'fun': lambda x: x[2 * n:]} for _ in range(n))
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: np.sqrt((x[2 * i] - x[2 * i + 2 * n]) ** 2 + (x[2 * i + 1] - x[2 * i + 1 + 2 * n]) ** 2) - x[2 * i + 2 * n * 2]})
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 2 * n * 2] - x[2 * i]} for i in range(n)]
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 2 * n * 2] - x[2 * i + 1]} for i in range(n)]
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 2 * n * 2] - (1 - x[2 * i + 1])} for i in range(n)]
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2 * i] - x[2 * i + 2 * n * 2]} for i in range(n)]
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2 * i + 1] - x[2 * i + 2 * n * 2]} for i in range(n)]
    res = minimize(lambda x: -np.sum(x[2 * n:]), np.concatenate((final_centers.flatten(), final_radii)), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9})
    final_centers = res.x[:2 * n].reshape(n, 2)
    final_radii = res.x[2 * n:]
    final_sum_radii = np.sum(final_radii)

    return final_centers, final_radii, final_sum_radii
