import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.special import hermite
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    
    def score(coeffs):
        nonlocal best_bound
        m = len(coeffs)
        coeffs = np.array(coeffs, dtype=float)
        H = np.zeros((m, m))
        for k in range(m):
            H[k] = hermite(4*k, np.linspace(-50, 50, 1000))
        
        Q = np.dot(H, coeffs)
        Q /= np.linalg.norm(Q)
        
        def find_r_max():
            def f(z):
                return Q[np.abs(z) < 100] / (z**2)
            
            try:
                root = root_scalar(lambda z: f(z).real, bracket=(-100, 100), method='brentq')
                return root.root**2
            except:
                return 10000
        
        r_max_sq = find_r_max()
        return r_max_sq / (2 * np.pi)
    
    def obj(coeffs):
        nonlocal best_bound
        val = score(coeffs)
        if val < best_bound:
            best_bound = val
            best_coeff = coeffs.copy()
        return val
    
    def optimize():
        initial_guess = np.random.rand(8)
        result = minimize(obj, initial_guess, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8})
        return result.x
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            coeffs = optimize()
            print(f"New best: {best_bound}")
        except:
            pass
    
    return best_coeff
