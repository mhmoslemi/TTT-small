def hermite_coefficients(c):
    """Evaluate Q(z) = sum c_k H_{4k}(z)"""
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        else:

    Q = np.zeros_like(z)
    for k in range(len(c)):
        Q += c[k] * hermite(4*k, z)
    return Q

def find_r_max(Q):
