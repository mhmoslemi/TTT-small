import numpy as np
import sympy
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8

    def obj_func(coeffs):
        try:
            # Construct polynomial
            z = sympy.symbols('z')
            Q = 0
            for k in range(len(coeffs)):
                Q += coeffs[k] * sympy.hermite(4*k, z)
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Ensure Q(z)/z^2 is positive at infinity
            Q = Q / z**2
            # Find largest positive real root
            roots = sympy.roots(Q)
            r_max = None
            for root in roots:
                if root.is_real and root > 0:
                    r_max = max(r_max, root)
            if r_max is None:
                return float('inf')
            # Return upper bound
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    # Initial guess
    initial_guess = np.random.rand(C4_MAX_COEFF_COUNT)

    # Optimization
    start_time = time.time()
    result = minimize(obj_func, initial_guess, method='L-BFGS-B', options={'maxiter': 1000})
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_guess  # fallback
    return result.x
