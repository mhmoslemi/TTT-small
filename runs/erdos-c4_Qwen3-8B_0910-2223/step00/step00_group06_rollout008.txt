import numpy as np
import sympy as sp
from scipy.optimize import minimize, root_scalar
from scipy.interpolate import interp1d
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    coeffs = np.random.rand(C4_MAX_COEFF_COUNT)
    
    def objective(coeffs):
        try:
            t0 = time()
            c = coeffs
            z = sp.symbols('z')
            H = [sp.hermite(n) for n in range(4*len(c)+1)]
            Q = sum(c[k] * H[4*k] for k in range(len(c)))
            Q = Q - Q.as_coefficients_dict()[z**2] * z**2
            Q = Q.subs(z, 0)
            if Q != 0:
                Q = Q * sp.symbols('s') / Q
                Q = Q.subs(sp.symbols('s'), 1)
            poly = Q.as_poly()
            def Q_func(x):
                return poly.eval(x)
            def find_r_max():
                try:
                    r = root_scalar(lambda x: Q_func(x), bracket=[1, 100], method='brentq')
                    return r.root
                except:
                    return 100
            r_max = find_r_max()
            U = r_max**2 / (2 * np.pi)
            elapsed = time() - t0
            print(f"Time: {elapsed:.2f}s, U: {U:.10f}")
            return U
        except:
            return float('inf')
    
    def numerical_objective(c):
        return objective(c)
    
    result = minimize(numerical_objective, x0=coeffs, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 100})
    best_coeffs = result.x
    return best_coeffs.tolist()
