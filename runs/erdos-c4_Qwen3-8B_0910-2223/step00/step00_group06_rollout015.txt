import numpy as np
from scipy.optimize import minimize, newton
from functools import lru_cache
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def hermite_coefficients(c):
        coeffs = np.array(c)
        H = np.array([np.polysum(4*k, 1) for k in range(len(coeffs))])
        return coeffs @ H
    
    def Q(z):
        return np.sum(c * [hermite_coefficients(c)[i] for i in range(len(c))]) / z**2
    
    def find_r_max(c):
        def f(z):
            return Q(z)
        
        try:
            r = newton(f, 1.0)
            return r
        except:
            return np.inf
    
    def evaluate_bound(c):
        r = find_r_max(c)
        return r**2 / (2 * np.pi)
    
    def objective(c):
        return evaluate_bound(c)
    
    def gradient_descent():
        result = minimize(objective, x0=np.random.rand(8), method='L-BFGS-B', bounds=[(0, 100) for _ in range(8)], options={'maxiter': 100})
        return result.x
    
    start_time = time.time()
    best_coeff = np.random.rand(8)
    best_value = evaluate_bound(best_coeff)
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        current = gradient_descent()
        current_value = evaluate_bound(current)
        if current_value < best_value:
            best_value = current_value
            best_coeff = current
    
    return best_coeff
