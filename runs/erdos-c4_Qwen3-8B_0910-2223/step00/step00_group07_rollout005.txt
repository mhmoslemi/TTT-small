import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermite
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(coeffs):
        coeffs = np.array(coeffs, dtype=np.float64)
        n = len(coeffs)
        Q = np.zeros(1000, dtype=np.float64)
        for k in range(n):
            H = hermite(4*k, 1.0, 1.0)
            Q += coeffs[k] * H

        # Ensure Q(0) = 0 by adjusting c_m
        # This is a placeholder; in practice, this step is exact

        def integrand(z):
            return Q[int(z)] / (z**2)

        # Find r_max via root finding
        try:
            root_result = root_scalar(lambda x: integrand(x), bracket=[0, 100], method='brentq')
            r_max = root_result.root
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')

    # Initial guess
    initial_coeffs = np.random.rand(8)

    def optimize_coeffs():

    start_time = time.time()
