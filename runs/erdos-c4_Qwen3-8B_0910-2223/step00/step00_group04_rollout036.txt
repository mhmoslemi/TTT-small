import numpy as np
from scipy.optimize import minimize
from scipy.optimize import fsolve
from scipy import special
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(c):
        m = len(c)
        H = [special.hermitium(n) for n in range(4*m)]
        Q = np.zeros(2*m+1)
        for k in range(m):
            Hk = H[4*k]
            Q += c[k] * Hk
        Q = np.flip(Q)
        
        def Q_z(z):
            val = 0
            for k in range(m):
                val += c[k] * special.her1(4*k, z)
            return val
        
        def Q_val(z):
            return Q_z(z) / (z**2)
        
        def f(z):
            return Q_val(z)
        
        try:
            root = fsolve(f, 1.0)
            if root[0] < 0:
                return float('inf')
            r_max = root[0]
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def objective(c):
        return evaluate_coefficients(c)
    
    def constraint(c):
        return np.dot(c, c) - 1
    
    def grad(c):
        return np.zeros_like(c)
    
    initial_guess = np.random.rand(8)
    bounds = [(0, 1) for _ in range(8)]
    
    best_coeff = initial_guess
    best_value = evaluate_coefficients(best_coeff)
    best_time = 0
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, initial_guess, bounds=bounds, jac=grad, method='L-BFGS-B', options={'maxiter': 100})
        if result.success:
            current_value = result.fun
            if current_value < best_value:
                best_value = current_value
                best_coeff = result.x
        initial_guess = (result.x + np.random.rand(8)*0.1) % 1
        time_used = time.time() - start_time
        best_time = time_used
    
    return best_coeff
