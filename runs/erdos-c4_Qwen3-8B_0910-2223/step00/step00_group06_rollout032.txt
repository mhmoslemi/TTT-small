import numpy as np
import random
from scipy.optimize import root_scalar
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    POP_SIZE = 32
    MAX_ITER = 100
    MUTATION_RATE = 0.1
    POP = [np.random.uniform(-1, 1, 8) for _ in range(POP_SIZE)]
    
    def evaluate_surrogate(c):
        Q = np.poly1d([0] + [0]*8)
        for k in range(8):
            H = np.polyspace.hermitenorm(4*k, 0, 100)
            Q = Q + c[k] * H
        Q = Q / np.poly1d([1, 0, 0])
        try:
            r_max = root_scalar(lambda z: Q(z), bracket=[0, 100], method='brentq').root
            return r_max**2 / (2 * np.pi)
        except:
            return np.inf
    
    best = min(POP, key=evaluate_surrogate)
    best_score = evaluate_surrogate(best)
    
    for _ in range(MAX_ITER):
        new_pop = []
        for c in POP:
            new_c = c.copy()
            for i in range(8):
                if random.random() < MUTATION_RATE:
                    new_c[i] += np.random.normal(0, 0.05)
                    new_c[i] = np.clip(new_c[i], -1, 1)
            new_pop.append(new_c)
        POP = new_pop
        for c in POP:
            score = evaluate_surrogate(c)
            if score < best_score:
                best = c
                best_score = score
    
    return best
