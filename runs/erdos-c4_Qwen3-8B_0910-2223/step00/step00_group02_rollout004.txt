import numpy as np
import random
from scipy.optimize import root_scalar
from sympy import symbols, hermite, Poly, roots

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Genetic algorithm parameters
    POP_SIZE = 64
    MAX_GEN = 10
    MUTATE_RATE = 0.3
    CROSS_RATE = 0.8
    ELITISM = 2
    
    def evaluate_coefficients(coeffs):
        try:
            z = symbols('z')
            Q = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
            Q = Q.as_poly().as_expr()
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Flip sign if needed for positive infinity
            if Q.as_poly().all_coeffs()[-1] < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q.as_poly().div(z**2)[0]
            # Find r_max
            def f(x):
                return Q.subs(z, x)
            try:
                res = root_scalar(f, bracket=[1, 100], method='brentq')
                r_max = res.root
                return r_max**2 / (2 * np.pi)
            except:
                return float('inf')
        except:
            return float('inf')
    
    def fitness(coeffs):
        return evaluate_coefficients(coeffs)
    
    def mutate(indv):
        new_indv = indv.copy()
        for i in range(len(new_indv)):
            if random.random() < MUTATE_RATE:
                new_indv[i] += np.random.uniform(-0.1, 0.1)
        return new_indv
    
    def crossover(a, b):
        c = [random.choice([a[i], b[i]]) for i in range(len(a))]
        return c
    
    # Initialize population
    pop = [np.random.uniform(-1, 1, 8) for _ in range(POP_SIZE)]
    best = min(pop, key=fitness)
    best_fitness = fitness(best)
    
    for gen in range(MAX_GEN):
        # Evaluate population
        fitnesses = [fitness(indv) for indv in pop]
        # Sort by fitness
        sorted_pop = [x for _, x in sorted(zip(fitnesses, pop))]
        sorted_fitness = sorted(fitnesses)
        # Select top ELITISM individuals
        elites = sorted_pop[:ELITISM]
        # Generate new population
        new_pop = elites.copy()
        while len(new_pop) < POP_SIZE:
            a, b = random.sample(sorted_pop, 2)
            child = crossover(a, b)
            if random.random() < MUTATE_RATE:
                child = mutate(child)
            new_pop.append(child)
        pop = new_pop
    
    # Final evaluation
    final_fitnesses = [fitness(indv) for indv in pop]
    best_index = np.argmin(final_fitnesses)
    best_coeffs = pop[best_index]
    return best_coeffs
