import random
from scipy.optimize import minimize

import time

def run(seed=42, budget_s=600, **kwargs):

            return np.sign(Q(a)) != np.sign(Q(b))

            if z == 0:
                return float('inf')

            high = 1000

                high *= 2

                    low = mid
