# Precomputed Hermite polynomials H_0 to H_32 (4k where k=0 to 8)
# H_n = 2^{n-1} * sum_{k=0}^{floor(n/2)} (-1)^k * n! / (k! (n - 2k)!)) * (x^{n-2k})

def hermite(n):
    x = sp.symbols('x')
    return sp.hermite(n, x)

def evaluate_Q(c, z):
    Q = 0
    for k in range(len(c)):
        Q += c[k] * hermite(4*k)
    return Q.subs(z)

def find_r_max(c):
    z = sp.symbols('z')
    Q = evaluate_Q(c, z)
    # Ensure Q(z)/z² changes sign
    expr = Q / z**2
    roots = sp.solveset(expr, z, domain=sp.Reals)
    real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
    real_roots = sorted(real_roots)
    for r in real_roots:
        if expr.subs(z, r).evalf().as_real_imag()[1] != 0:
            return r
    return None

def exact_U(c):
