def hermite_poly(n):
    return hermite(n, symbols('z'))

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    z = symbols('z')
    Q = sum(coeffs[k] * hermite_poly(4*k) for k in range(m))
    # Ensure Q(0) = 0

        Q = Q - Q.subs(z, 0)
    # Flip sign if needed

        Q = -Q
    # Divide by z^2
    Q = Q / z**2
    # Find largest positive real root
    try:
        roots_real = roots(Q, z, domain='R')
        r_max = None
        for root in roots_real:
            if root.is_real and root > 0:
                if r_max is None or root > r_max:
                    r_max = root
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    except:
        return float('inf')
