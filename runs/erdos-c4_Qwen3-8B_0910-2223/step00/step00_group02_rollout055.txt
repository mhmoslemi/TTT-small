import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.integrate import quad
from scipy.interpolate import interp1d
import time

def H4k(z):
    def hermite(n):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        elif n == 2:
            return 4*z**2 - 2
        elif n == 3:
            return 8*z**3 - 12*z
        elif n == 4:
            return 16*z**4 - 48*z**2 + 12
        return None
    return hermite(4*z)

def evaluate_Q(z, coeffs):
    return sum(coeffs[k] * H4k(z) for k in range(len(coeffs)))

def find_r_max(coeffs):
    def Q(z):
        return evaluate_Q(z, coeffs)
    def Qz2(z):
        return Q(z) / z**2 if z != 0 else 0
    try:
        r = root_scalar(Qz2, bracket=[0.1, 100], method='brentq', xtol=1e-10)
        return r.root**2
    except:
        return np.inf

def compute_U(coeffs):
    r_sq = find_r_max(coeffs)
    return r_sq / (2 * np.pi)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_coeffs = 8
    bounds = [(-10, 10) for _ in range(n_coeffs)]
    initial_guess = np.random.uniform(-10, 10, n_coeffs)
    
    def objective(x):
        return compute_U(x)
    
    start_time = time.time()
    result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time exceeded budget, returning last guess")
        return initial_guess
    return result.x
