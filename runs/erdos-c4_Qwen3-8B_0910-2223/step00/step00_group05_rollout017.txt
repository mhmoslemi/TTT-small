import numpy as np
import sympy as sp
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    z = sp.symbols('z')
    H = [sp.hermite(h) for h in range(4*8+1)]
    
    def eval_Q(c):
        Q = sum(c[k] * H[4*k] for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        return Q
    
    def compute_bound(c):
        Q = eval_Q(c)
        roots = sp.roots(Q)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        
        for r in real_roots:
            if Q.subs(z, r).as_numer_denom()[0].as_coeff_dict().get(z, 0) != 0:
                return (r**2) / (2*np.pi)
        return float('inf')
    
    def objective(c):
        return compute_bound(c)
    
    def grad_objective(c):
        return np.zeros_like(c)
    
    init_guess = np.random.rand(8)
    res = minimize(objective, init_guess, method='L-BFGS-B', jac=grad_objective, options={'maxiter': 100})
    return res.x.tolist()
