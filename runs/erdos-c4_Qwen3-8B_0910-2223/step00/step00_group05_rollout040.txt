import numpy as np
from scipy.optimize import minimize
from scipy.optimize._differentiable_functions import ScalarFunction
from sympy import symbols, expand, hermite, solve, simplify
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    best_coeffs = None
    best_value = float('inf')
    
    def evaluate(coeffs):
        z = symbols('z')
        Q = 0
        for k in range(len(coeffs)):
            Q += coeffs[k] * hermite(4*k, z)
        Q = Q.subs(z, 0)
        c_m = -Q / (hermite(4*len(coeffs), 0))
        Q = Q + c_m * hermite(4*len(coeffs), z)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def score(coeffs):
        nonlocal best_value, best_coeffs
        try:
            val = evaluate(coeffs)
            if val < best_value:
                best_value = val
                best_coeffs = coeffs.copy()
            return val
        except:
            return float('inf')
    
    def simplex_search():
        nonlocal best_value, best_coeffs
        initial_guess = np.random.rand(C4_MAX_COEFF_COUNT)
        result = minimize(score, initial_guess, method='Nelder-Mead', options={'xatol': 1e-8, 'fatol': 1e-8, 'disp': False})
        best_coeffs = result.x
        best_value = score(best_coeffs)
    
    def gradient_search():
        nonlocal best_value, best_coeffs
        def objective(x):
            return score(x)
        result = minimize(objective, np.random.rand(C4_MAX_COEFF_COUNT), method='L-BFGS-B', options={'gtol': 1e-5, 'maxfun': 1000})
        best_coeffs = result.x
        best_value = score(best_coeffs)
    
    def random_search():
        nonlocal best_value, best_coeffs
        for _ in range(100):
            coeffs = np.random.rand(C4_MAX_COEFF_COUNT)
            val = score(coeffs)
            if val < best_value:
                best_value = val
                best_coeffs = coeffs
    
    def hybrid_search():
        nonlocal best_value, best_coeffs
        random_search()
        gradient_search()
        simplex_search()
        return best_coeffs
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            best_coeffs = hybrid_search()
        except:
            pass
    return best_coeffs
