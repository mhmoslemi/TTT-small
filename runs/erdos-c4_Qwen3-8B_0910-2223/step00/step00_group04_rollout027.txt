import numpy as np
import sympy as sp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    best_coeff = np.random.rand(C4_MAX_COEFF_COUNT)
    best_score = float('inf')

    def numerical_surrogate(coeffs):
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*C4_MAX_COEFF_COUNT + 4)]
        Q = sum(coeffs[k] * H[4*k] for k in range(len(coeffs)))
        Q = Q.subs(z, z) / z**2
        roots = sp.solvers.solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots.sort()
        r_max = None
        for i in range(len(real_roots) - 1):
            if real_roots[i] < 0:
                continue
            if real_roots[i+1] > 0:
                r_max = real_roots[i+1]
                break
        return r_max**2 / (2 * np.pi)

    def exact_check(coeffs):
        z = sp.symbols('z')
        H = [sp.hermite(h) for h in range(4*C4_MAX_COEFF_COUNT + 4)]
        Q = sum(coeffs[k] * H[4*k] for k in range(len(coeffs)))
        Q = Q.subs(z, z)
        Q = sp.Poly(Q, z)
        Q = Q.as_expr()
        Q = Q / z**2
        roots = sp.solvers.solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted(r for r in real_roots if r > 0)
        r_max = real_roots[-1] if real_roots else None
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)

    def objective(coeffs):
        return exact_check(coeffs)

    for _ in range(1000):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        coeffs = np.random.rand(C4_MAX_COEFF_COUNT)
        try:
            score = numerical_surrogate(coeffs)
            if score < best_score:
                best_score = score
                best_coeff = coeffs.copy()
        except:
            pass

    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        coeffs = best_coeff + np.random.normal(0, 0.1, C4_MAX_COEFF_COUNT)
        coeffs = np.clip(coeffs, -1, 1)
        try:
            score = exact_check(coeffs)
            if score < best_score:
                best_score = score
                best_coeff = coeffs.copy()
        except:
            pass

    return best_coeff
