import numpy as np
import random
from scipy.optimize import minimize
from scipy.optimize._numdiff import approx_derivative
from sympy import symbols, diff, simplify, solve, HermiteH, Rational
from time import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    best_coeff = None
    best_value = float('inf')
    start_time = time()
    total_time = 0

    def evaluate_bound(coeffs):
        nonlocal total_time
        if total_time >= budget_s:
            return best_value
        t0 = time()
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        c_m = -Q / (HermiteH(4*m, 0))
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m)) + c_m * HermiteH(4*m, z)
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1
        value = r_max**2 / (2 * np.pi)
        total_time += time() - t0
        return value

    def obj(coeffs):
        return evaluate_bound(coeffs)

    for _ in range(100):
        if total_time >= budget_s:
            break
        coeffs = [Rational(random.uniform(-1, 1)) for _ in range(8)]
        res = minimize(obj, coeffs, method='L-BFGS-B', tol=1e-10, options={'disp': False})
        current_value = res.fun
        if current_value < best_value:
            best_value = current_value
            best_coeff = res.x

    return best_coeff
