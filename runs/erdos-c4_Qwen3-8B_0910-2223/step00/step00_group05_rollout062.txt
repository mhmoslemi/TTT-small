import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    coefficients = np.random.rand(8)
    
    def objective(x):
        c = x.copy()
        c[0] = 0  # Ensure Q(0) = 0
        # Approximate Q(z) = sum c_k H_{4k}(z)
        # Use numerical evaluation to find r_max
        def Q(z):
            val = 0
            for k in range(len(c)):
                val += c[k] * np.polynomial.hermite.hermval([z], np.polynomial.hermite.hermite_e(4*k))
            return val
        
        z = np.linspace(-100, 100, 10000)
        sign_changes = np.where(np.diff(np.sign(Q(z))) != 0)[0]
        r_max = 0
        for idx in sign_changes:
            if z[idx] > 0 and Q(z[idx]) > 0:
                r_max = z[idx]
        if r_max == 0:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    result = minimize(objective, coefficients, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)], options={'maxiter': 100, 'ftol': 1e-8})
    return result.x
