import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, Hermite, expand, solve, Eq, Rational
from sympy.polys.polytools import factor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_score = float('inf')
    
    def objective(coeffs):
        nonlocal best_score
        c = coeffs
        try:
            # Evaluate with exact symbolic computation
            z = symbols('z')
            Q = sum(c[k] * Hermite(4*k, z) for k in range(len(c)))
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Make Q positive at infinity by flipping sign if necessary
            if Q.as_coefficients_dict().get(z**0, 0) < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q / z**2
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            # Filter out even multiplicity roots
            unique_roots = []
            seen = set()
            for r in real_roots:
                if r not in seen:
                    seen.add(r)
                    unique_roots.append(r)
            # Find largest positive real root
            r_max = max(unique_roots)
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c
            return score
        except:
            return float('inf')
    
    # Initial guess
    initial_guess = np.random.rand(8)
    # Numerical optimization with restarts
    for _ in range(20):
        start_time = time.time()

            objective,
            initial_guess,
