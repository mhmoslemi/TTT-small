from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult

from sympy import symbols, Hermite, simplify, diff, solve, symbols
from sympy.polys.polytools import factor
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    for _ in range(100):
        start = time.time()

            c = x

            z = symbols('z')

            Q = Q / z**2
            # Find roots of Q(z)/z^2
            # Use symbolic computation to find real roots
            roots = solve(Q, z, domain='Real')
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            # Select r_max as the largest positive root
            if not real_roots:
                return float('inf')
            r_max = max(real_roots)
            # Compute U(c) = r_max^2 / (2*pi)
            return r_max**2 / (2 * np.pi)

        def surrogate(x):
            c = x
            z = np.linspace(-10, 10, 1000)
            Q = np.zeros_like(z)
