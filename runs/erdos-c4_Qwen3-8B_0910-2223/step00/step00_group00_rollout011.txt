import numpy as np
import scipy.optimize as optimize
from sympy import symbols, simplify, hermite, Eq, solve
from sympy.polys import Poly
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    best_c = c.copy()
    best_bound = float('inf')
    start_time = time()
    
    def compute_bound(coeffs):
        c_coeff = coeffs
        z = symbols('z')
        Q = sum(c_coeff[k] * hermite(4*k, z) for k in range(len(c_coeff)))
        Q = Q.as_poly()
        Q = Q.remove_term(1)  # Remove z^2 factor
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        real_roots = [r for r in real_roots if r not in [r for r in real_roots if real_roots.count(r) > 1]]
        real_roots.sort()
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return (r_max**2) / (2 * np.pi)

    def objective(x):
        nonlocal best_bound
        if time() - start_time > budget_s:
            return best_bound
        bound = compute_bound(x)
        if bound < best_bound:
            best_bound = bound
            best_c = x.copy()
        return bound

    res = optimize.minimize(objective, np.random.rand(8), method='L-BFGS-B', options={'maxiter': 100})
    return best_c
