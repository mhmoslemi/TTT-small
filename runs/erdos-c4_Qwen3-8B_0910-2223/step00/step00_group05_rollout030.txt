import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.optimize import Bounds
import time
import sys
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_score = float('inf')
    start_time = time.time()
    
    # Define Hermite polynomials
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        elif n == 2:
            return 4*z**2 - 2
        elif n == 3:
            return 8*z**3 - 12*z
        elif n == 4:
            return 16*z**4 - 48*z**2 + 12
        elif n == 5:
            return 32*z**5 - 160*z**3 + 120*z
        elif n == 6:
            return 64*z**6 - 384*z**4 + 480*z**2 - 120
        elif n == 7:
            return 128*z**7 - 1024*z**5 + 1920*z**3 - 1120*z
        elif n == 8:
            return 256*z**8 - 2560*z**6 + 6400*z**4 - 5600*z**2 + 1680
        return 0
    
    def evaluate_U(c):
        # Build polynomial Q(z) = sum c_k * H_{4k}(z)
        z = np.linspace(-100, 100, 20000)
        Q = np.zeros_like(z)
        for k in range(len(c)):
            Q += c[k] * hermite(4*k, z)
        
        # Ensure Q(0) = 0
        if Q[0] != 0:
            Q -= Q[0]
        
        # Flip sign if necessary to make it positive at infinity
        if Q[-1] < 0:
            Q *= -1
        
        # Remove z^2 factor
        Q /= z**2 + 1e-12
        
        # Find largest positive real root
        try:
            r_max = root_scalar(lambda x: Q[np.abs(x) < 100], bracket=[0, 100], method='brentq').root
        except:
            r_max = 100
        
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        try:
            return evaluate_U(c)
        except:
            return float('inf')
    
    # Initial random vectors
    for _ in range(10):
        c = np.random.rand(8)
        result = minimize(objective, c, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)], options={'maxiter': 100})
        if result.success:
            current_score = result.fun
            if current_score < best_score:
                best_score = current_score
                best_coeff = result.x
    
    # Refine with more accurate search
    for _ in range(10):
        if best_coeff is None:
            continue
        c = best_coeff.copy()
        result = minimize(objective, c, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)], options={'maxiter': 50})
        if result.success:
            current_score = result.fun
            if current_score < best_score:
                best_score = current_score
                best_coeff = result.x
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return best_coeff
    
    return best_coeff
