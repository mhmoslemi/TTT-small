import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, expand, solve, Poly, Rational, simplify
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_poly()
        Q = Q - Q.subs(z, 0)
        Q = Q.as_expr()
        Q = Q / z**2
        try:
            roots = solve(Q, z)
        except:
            return float('inf')
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([float(r) for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def surrogate_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_poly()
        Q = Q - Q.subs(z, 0)
        Q = Q.as_expr()
        Q = Q / z**2

            roots = solve(Q, z)

            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def fitness(coeffs):
        return evaluate_bound(coeffs)
    
    def surrogate_fitness(coeffs):
        return surrogate_bound(coeffs)
    
    def optimize():
        bounds = [(-10, 10) for _ in range(8)]
        initial_guess = np.random.uniform(-10, 10, 8)
        es = minimize(
            surrogate_fitness,
            initial_guess,
            bounds=bounds,
            method='L-BFGS-B',

        best_bound = fitness(best_coeffs)
        return best_coeffs
