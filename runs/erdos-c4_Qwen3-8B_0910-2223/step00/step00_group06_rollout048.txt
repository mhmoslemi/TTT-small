import numpy as np
from scipy.optimize import minimize, root_scalar
from scipy.optimize.optimize import OptimizeResult
from scipy.linalg import norm
from functools import partial
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    start_time = time.time()
    
    def hermite_poly(n):
        def H(x):
            if n == 0:
                return 1.0
            elif n == 1:
                return 2*x
            else:
                return 2*x*H(n-1) - 2*(n-1)*H(n-2)
        return H
    
    def compute_Q(z, coeffs):
        Q = 0.0
        for k, c in enumerate(coeffs):
            Q += c * hermite_poly(4*k)(z)
        return Q
    
    def find_r_max(coeffs):
        def Q(z):
            return compute_Q(z, coeffs)
        
        def find_root(start):
            try:
                res = root_scalar(Q, bracket=[start, start + 1], method='brentq', xtol=1e-8)
                return res.root
            except:
                return None
        
        def find_sign_change():
            start = 1.0
            while True:
                try:
                    r = find_root(start)
                    if r is not None:
                        return r
                    start *= 2
                except:
                    break
            return None
        
        r = find_sign_change()
        if r is None:
            return float('inf')
        return r
    
    def compute_bound(coeffs):
        try:
            r = find_r_max(coeffs)
            if r is None:
                return float('inf')
            return r**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(coeffs):
        return compute_bound(coeffs)
    
    def grad_objective(coeffs):
        return np.zeros_like(coeffs)
    
    def sample_coefficients():
        return np.random.uniform(-1, 1, 8)
    
    def optimize_with_restart():
        nonlocal best_coeff, best_bound
        for _ in range(20):
            coeffs = sample_coefficients()
            try:
                res = minimize(objective, coeffs, method='L-BFGS-B', jac=grad_objective, bounds=[(-1, 1)]*8, options={'maxiter': 100})
                current_coeff = res.x
                current_bound = objective(current_coeff)
                if current_bound < best_bound:
                    best_coeff = current_coeff
                    best_bound = current_bound
            except:
                pass
    
    optimize_with_restart()
    
    def evaluate_with_rational(coeffs):
        from sympy import symbols, Poly, hermite, Eq, solve
        z = symbols('z')
        Q = 0
        for k, c in enumerate(coeffs):
            Q += c * hermite(4*k, z)
        Q = Q.subs(z, 0)
        if Q != 0:
            Q += -Q
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        if not real_roots:
            return float('inf')
        real_roots.sort()
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def final_eval(coeffs):
        return evaluate_with_rational(coeffs)
    
    def refine_best():
        nonlocal best_coeff, best_bound
        for _ in range(50):
            coeffs = best_coeff + np.random.normal(0, 1e-3, 8)
            coeffs = np.clip(coeffs, -1, 1)
            bound = final_eval(coeffs)
            if bound < best_bound:
                best_bound = bound
                best_coeff = coeffs
    
    refine_best()
    
    if time.time() - start_time > budget_s:
        return best_coeff.tolist()
    return best_coeff.tolist()
