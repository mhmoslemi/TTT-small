import numpy as np
import random
from scipy.optimize import minimize_scalar
from sympy import symbols, Hermite, Poly, roots, N

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    
    def hermite_eval(coefficients):
        z = symbols('z')
        Q = sum(c * Hermite(4*k, z) for k, c in enumerate(coefficients))
        Q = Q.as_poly()
        z2 = z**2
        Q = Q / z2
        roots = roots(Q)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r) * Q.subs(z, r + 1e-6) < 0:
                r_max = r
                break
        else:
            r_max = 1.0
        return r_max**2 / (2 * np.pi)
    
    def eval_c4_coefficients(coeffs):
        return hermite_eval(coeffs)
    
    def fitness(coeffs):
        return eval_c4_coefficients(coeffs)
    
    def mutate(parent, mutation_rate=0.1):
        child = parent.copy()
        for i in range(len(child)):
            if random.random() < mutation_rate:
                child[i] += random.uniform(-0.5, 0.5)
        return child
    
    def crossover(parent1, parent2):
        point = random.randint(1, len(parent1)-1)
        child = parent1[:point] + parent2[point:]
        return child
    
    def generate_initial_population(size=50):
        return [np.random.uniform(-1, 1, C4_MAX_COEFF_COUNT).tolist() for _ in range(size)]
    
    population = generate_initial_population()
    best = min(population, key=fitness)
    best_fitness = fitness(best)
    
    for _ in range(100):
        if _ % 10 == 0:
            print(f"Generation {_}: best fitness = {best_fitness}")
        if _ > 50:
            budget_s -= 1
            if budget_s <= 0:
                break
        
        next_pop = []
        for _ in range(25):
            p1, p2 = random.choices(population, k=2)
            child = crossover(p1, p2)
            child = mutate(child)
            next_pop.append(child)
        
        population += next_pop
        population = list(set(tuple(p) for p in population))
        if len(population) > 100:
            population = random.sample(population, 100)
        
        current_best = min(population, key=fitness)
        if fitness(current_best) < best_fitness:
            best = current_best
            best_fitness = fitness(current_best)
    
    return best
