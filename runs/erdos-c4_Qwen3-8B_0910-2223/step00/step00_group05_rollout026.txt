import numpy as np
from scipy.optimize import minimize
from scipy.optimize import fsolve
from sympy import symbols, simplify, diff, Eq, solve, HermitePolynomial
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    max_coeff = 8
    best_score = float('inf')
    best_coeffs = None
    start_time = time.time()

    def fitness(coeffs):
        nonlocal best_score, best_coeffs
        t = time.time()
        if t - start_time > budget_s:
            return best_score

        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermitePolynomial(4*k, z) for k in range(len(c)))
        Q = Q.as_expr()
        Q = Q.subs(z, 0)
        Q = Q.as_poly()
        if Q.lc() < 0:
            Q = -Q
        Q = Q.as_expr()
        Q = Q / z**2
        roots = solve(Q, z)
        valid_roots = []
        for root in roots:
            if root.is_real and root > 0:
                if Q.subs(z, root).evalf() * Q.subs(z, root + 1e-6).evalf() < 0:
                    valid_roots.append(float(root))
        if not valid_roots:
            return float('inf')
        r_max = max(valid_roots)
        score = r_max**2 / (2 * np.pi)
        if score < best_score:
            best_score = score
            best_coeffs = coeffs.copy()
        return score

    def obj(coeffs):
        return fitness(coeffs)

    initial_guess = np.random.rand(max_coeff)
    result = minimize(obj, initial_guess, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(max_coeff)])
    return best_coeffs
