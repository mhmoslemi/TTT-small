import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, expand, diff, solve, simplify
from sympy.polys.polytools import factor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = -Q
        Q = Q + 1  # To ensure positivity at infinity
        Q = Q.subs(z, 0)
        Q = Q + 1  # To ensure positivity at infinity
        
        # Factor Q
        Q_poly = expand(Q)
        factors = factor(Q_poly)
        if factors.ascent != 1:
            Q_poly = factors.ascent

        # Find root r_max
        roots = solve(Q_poly, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted(real_roots)
        r_max = real_roots[-1] if real_roots else 1
        U = r_max**2 / (2 * np.pi)
        return U

    def fitness(coeffs):
        return evaluate_coefficients(coeffs)

    def obj_func(coeffs):
        return fitness(coeffs)

    # Initial guess: random coefficients
    initial_guess = np.random.rand(8)

    # Optimization
    start_time = time.time()
    result = minimize(obj_func, initial_guess, method='L-BFGS-B', options={'gtol': 1e-8, 'ftol': 1e-8, 'maxiter': 1000})
    best_coeffs = result.x

    # Check if we have enough time left for refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 200:
        # Refinement with Adam optimizer
        from torch import optim
        import torch

        # Convert to PyTorch tensor
        coeffs_tensor = torch.tensor(best_coeffs, requires_grad=True, dtype=torch.float64)
        optimizer = optim.Adam([coeffs_tensor], lr=0.1)

        for step in range(100):
            optimizer.zero_grad()
            loss = obj_func(coeffs_tensor.numpy())
            loss.backward()
            optimizer.step()
            if step % 10 == 0:
                print(f"Step {step}, Loss: {loss.item()}")

        best_coeffs = coeffs_tensor.numpy()

    # Final check and return best_coeffs
    return best_coeffs
