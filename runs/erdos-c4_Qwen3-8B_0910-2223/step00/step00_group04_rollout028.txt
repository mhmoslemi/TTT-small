from scipy.optimize import minimize, root_scalar
from scipy.optimize.optimize import Bounds
from scipy.special import hermite
import time

def run(seed=42, budget_s=600, **kwargs):

    best_bound = float('inf')

        nonlocal best_bound

            return val

            return Q(z) / (z**2)

            r_max = root_scalar(f, bracket=[1, 100], method='brentq').root
        except:
            return float('inf')
