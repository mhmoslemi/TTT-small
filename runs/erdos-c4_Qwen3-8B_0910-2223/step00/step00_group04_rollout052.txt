def run(seed=42, budget_s=600, **kwargs):

    def find_r_max(coeffs):
        from sympy import symbols, Eq, solve, N, roots, S
        from sympy.polys.polytools import factor, resultant
        from sympy.polys.roots import roots as sympy_roots
        from sympy import simplify, Rational
        
        z = symbols('z')

        Q = sum(c * h for c, h in zip(coeffs, H))
        
        # Enforce Q(0) = 0
        Q0 = Q.subs(z, 0)
        if Q0 != 0:
            raise ValueError("Q(0) != 0")
        
        # Make Q positive at infinity

        Q = Q / z**2

        r_max = max(real_roots)
