import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    
    # Symbolic variables
    z = sp.symbols('z')
    H = [sp.heritage.orthogonal_polynomials('Hermite', 4*k, z) for k in range(C4_MAX_COEFF_COUNT)]
    
    def evaluate_coefficients(c):
        Q = sum(c[k] * H[k] for k in range(len(c)))
        Q = Q.subs(z, 0)
        c_m = -Q / (H[-1].subs(z, 0))
        Q = sum(c[k] * H[k] for k in range(len(c))) + c_m * H[-1]
        Q = Q / z**2
        roots = sp.solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        r_max = max(real_roots) if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        return evaluate_coefficients(c)
    
    def numerical_surrogate(c):
        # Numerical approximation for fast inner loop
        Q = sum(c[k] * sp.herite.Hermite(4*k, z) for k in range(len(c)))
        Q = Q / z**2
        roots = np.roots([Q.as_poly().coeffs()[::-1]])
        real_roots = roots[roots > 0]
        r_max = np.max(real_roots) if real_roots.size else 1
        return r_max**2 / (2 * np.pi)
    
    # Initial guess
    c0 = np.random.rand(C4_MAX_COEFF_COUNT)
    best_c = c0
    best_value = objective(c0)
    
    start_time = time()
    while time() - start_time < budget_s:
        # Use numerical surrogate for gradient descent
        def surrogate(c):
            return numerical_surrogate(c)
        
        result = minimize(
            surrogate,
            best_c,
            method='L-BFGS-B',
            bounds=[(0, 10) for _ in range(C4_MAX_COEFF_COUNT)]
        )
        
        # Verify with exact calculation
        current_value = objective(result.x)
        if current_value < best_value:
            best_value = current_value
            best_c = result.x
    
    return best_c
