import numpy as np
import scipy.optimize as optimize
from sympy import symbols, HermiteH, simplify, solve, diff, expand
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        # Symbolic construction
        z = symbols('z')
        H = [HermiteH(k, z) for k in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        
        # Enforce Q(0) = 0

        if leading_term.coeff.is_negative:
            Q *= -1

        Q = Q.as_poly().as_expr()
        Q = Q / z**2
        Q = simplify(Q)
        
        # Find real positive roots with sign change
        roots = solve(Q, z)
        valid_roots = []
        for r in roots:
            if r.is_real and r > 0:
                # Check multiplicity
                if Q.as_poly().is_zero:
                    continue
                poly = Q.as_poly()
                if poly.degree() == 0:
                    continue
                if poly.coeff_monomial(r) == 0:
                    continue

                        continue
                    left = Q.subs(z, r_val - 1e-6)
                    right = Q.subs(z, r_val + 1e-6)

                        valid_roots.append(r_val)

                    pass

        r_max = max(valid_roots)

        return evaluate_coefficients(coeffs)
