from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, simplify
from sympy.polys.rationalfuncs import apart

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    
    def objective(x):
        coeffs = x.copy()
        coeffs = np.concatenate((coeffs, [0.0]))

            return float('inf')

            roots = solve(Q, z)
        except:
            return float('inf')
        
        r_max = 0
        for root in roots:
            if root.is_real and root > 0:
                m = root.as_numer.as_coeff_mul()[1]
                if m.is_negative:
                    root = -root
                r_max = max(r_max, root)
        
        if r_max == 0:
            return float('inf')
        
        return r_max**2 / (2 * np.pi)
    
    def surrogate(x):
        coeffs = x.copy()

    )
    
    best_coeffs = result.x
