import numpy as np
from scipy.optimize import minimize
from scipy.optimize import fsolve
from sympy import symbols, HermiteH, expand, simplify, solve, Eq
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = -Q
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        r_max = max(real_roots) if real_roots else 10
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        return evaluate_coefficients(c)
    
    best = None
    best_value = float('inf')
    
    for _ in range(10):
        c = np.random.rand(8)
        result = minimize(objective, c, method='L-BFGS-B', options={'maxiter': 100})
        if result.success:

            if val < best_value:

                best = result.x.copy()
    
    return best
