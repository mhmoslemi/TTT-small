import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve, Eq, simplify, Rational
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q.as_poly()
        Q = Q.subs(z, 0)
        c_m = -Q.as_coefficients_dict().get(z**0, 0)
        Q = Q - c_m

        # Make Q positive at infinity
        leading_term = Q.as_dict().get(z**m, 0)
        if leading_term < 0:
            Q *= -1

        # Remove z^2 factor
        Q = Q.as_poly()
        if Q.degree() < 2:
            return float('inf')

        # Check if z^2 divides Q
        if not Q.div(z**2)[1].is_zero:
            return float('inf')

        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real and r > 0]
        real_roots = [r for r in real_roots if not Q.subs(z, r).is_zero]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)

    def numerical_surrogate(coeffs):
        try:
            return evaluate_coefficients(coeffs)
        except:
            return float('inf')

    def objective(coeffs):
        return numerical_surrogate(coeffs)

    bounds = [(-10, 10) for _ in range(8)]

    # Initial guess
    init_guess = np.random.uniform(-1, 1, 8)

    start_time = time()
    result = minimize(objective, init_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
    elapsed = time() - start_time

    # Post-processing: refine top candidates
    candidates = [np.random.uniform(-1, 1, 8) for _ in range(10)]
    for c in candidates:
        res = minimize(objective, c, bounds=bounds, method='L-BFGS-B', options={'maxiter': 50})
        if res.fun < result.fun:
            result = res

    if elapsed > budget_s:
        result = minimize(objective, init_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': 10})
    
    return result.x
