def hermite_poly(n, z):
    """Approximate physicists' Hermite polynomial using recurrence"""
    if n == 0:
        return np.ones_like(z)
    elif n == 1:
        return 2*z
    H_prev = np.ones_like(z)
    H_curr = 2*z
    for _ in range(2, n+1):
        H_next = 2*z*H_curr - 2*(n-1)*H_prev
        H_prev, H_curr = H_curr, H_next
    return H_curr
