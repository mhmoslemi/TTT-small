import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, S, Interval, Function
from functools import lru_cache
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    budget = budget_s

    def evaluate_coefficients(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.as_poly()
        # Ensure Q(0) = 0
        if Q.subs(z, 0) != 0:
            Q -= Q.subs(z, 0)
        # Flip sign if needed so that Q(z)/z^2 tends to +infty as z approaches +/-infty
        if Q.lead_coeff < 0:
            Q *= -1
        # Remove z^2 factor
        try:
            Q = Q.as_poly().remove_terms([z**2])
        except:
            Q = Q.as_poly()
        # Find real roots of Q(z)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        # Find sign changes across real_roots
        def sign_at(z_val):
            try:
                return np.sign(Q.subs(z, z_val))
            except:
                return 0
        r_max = None
        for i in range(len(real_roots) - 1):
            r1 = real_roots[i]
            r2 = real_roots[i+1]
            if sign_at(r1) * sign_at(r2) < 0:
                r_max = r2
                break
        if r_max is None:
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)

    def cost(coeffs):
        return evaluate_coefficients(coeffs)

    def grid_search():
        nonlocal best_bound, best_coeff
        for _ in range(100):
            coeffs = np.random.uniform(-10, 10, size=8)
            bound = cost(coeffs)
            if bound < best_bound:
                best_bound = bound
                best_coeff = coeffs.copy()

    def local_search():
        nonlocal best_bound, best_coeff
        for _ in range(10):
            start = np.random.uniform(-10, 10, size=8)
            res = minimize(lambda x: cost(x), start, method='Nelder-Mead', options={'xatol': 1e-6, 'maxiter': 100})
            if res.success:
                bound = res.fun
                if bound < best_bound:
                    best_bound = bound
                    best_coeff = res.x

    start_time = time.time()
    grid_search()
    local_search()
    remaining_time = budget - (time.time() - start_time)
    if remaining_time > 0:
        for _ in range(10):
            start = np.random.uniform(-10, 10, size=8)
            res = minimize(lambda x: cost(x), start, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 100})
            if res.success:
                bound = res.fun
                if bound < best_bound:
                    best_bound = bound
                    best_coeff = res.x
    return best_coeff
