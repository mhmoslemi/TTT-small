import numpy as np
from scipy.optimize import minimize, root_scalar
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from scipy.special import hermite
from functools import lru_cache

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    def Q(z):
        res = 0
        for k in range(m):
            res += coeffs[k] * hermite(4*k, z)
        return res
    # Find r_max via root finding
    def f(z):
        return Q(z) / (z**2)
    try:
        # Find largest positive root where sign changes
        r_max = root_scalar(lambda x: f(x), bracket=(1, 1000), method='brentq')
        return r_max.root**2 / (2*np.pi)
    except:
        return np.inf

def run(seed=42, budget_s=600, **kwargs):

    n_coeff = 8
    bounds = [(-1, 1) for _ in range(n_coeff)]

    samples = [np.random.uniform(*b) for b in bounds]
    
    # Define GP surrogate
    kernel = ConstantKernel(1.0) * RBF(length_scale=1.0)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
    
    # Initial evaluations
    X = np.array(samples).reshape(-1, n_coeff)
    Y = np.array([evaluate_c4_coefficients(c) for c in samples])
    
    best_idx = np.argmin(Y)
    best_coeffs = samples[best_idx]

    # Bayesian optimization loop
    for _ in range(100):

            break

        gp.fit(X, Y)

        if new_y < best_value:
