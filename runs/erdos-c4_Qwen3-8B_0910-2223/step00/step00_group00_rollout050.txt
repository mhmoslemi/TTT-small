import numpy as np
from scipy.optimize import minimize
from sympy import symbols, diff, hermite, simplify, Poly, Rational

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(c):
        m = len(c)
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(m))
        Q = Q.as_poly()
        Q = Q * Rational(1, 1)  # Ensure rational coefficients
        
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        
        # Check sign at infinity
        sign = Q.LC().as_numer_denom()[0]
        if sign < 0:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q.as_poly().divide(z**2)[0]
        
        # Find real roots
        roots = np.real(np.roots(Q.all_coeffs()))
        roots = np.sort(roots)
        
        # Find r_max
        r_max = None
        for r in roots:
            if r > 0 and np.isclose(Q.evalf(subs={z: r}), 0):
                continue
            if r > 0 and np.sign(Q.evalf(subs={z: r - 1e-6})) != np.sign(Q.evalf(subs={z: r + 1e-6})):
                r_max = r
                break
        
        if r_max is None:
            return float('inf')
        
        return r_max**2 / (2 * np.pi)
    
    def surrogate(c):
        try:
            return evaluate_coefficients(c)
        except:
            return float('inf')
    
    # Initial guess
    c0 = np.random.uniform(-1, 1, 8)
    result = minimize(lambda c: surrogate(c), c0, method='L-BFGS-B', bounds=[(-1, 1)]*8, options={'ftol': 1e-8, 'gtol': 1e-8})
    
    # Refined search
    for _ in range(5):
        if result.fun < 0.3216:
            break

        new_c = np.clip(new_c, -1, 1)
        result = minimize(lambda c: surrogate(c), new_c, method='L-BFGS-B', bounds=[(-1, 1)]*8, options={'ftol': 1e-8, 'gtol': 1e1})
    
    best_coeffs = result.x
    return best_coeffs
