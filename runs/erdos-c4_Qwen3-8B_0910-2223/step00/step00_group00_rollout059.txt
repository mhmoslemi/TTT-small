import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from sympy import symbols, Eq, solve, Rational, simplify
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_value = float('inf')
    start_time = time()
    
    def objective(coeffs):
        nonlocal best_value
        if time() - start_time > budget_s - 1:
            return best_value
        try:
            # Numerical estimation of r_max
            def Q(z):
                return sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
            
            def f(z):
                return Q(z) / (z**2)
            
            try:
                r_max = root_scalar(f, bracket=[0.1, 100], method='brentq').root
                return r_max**2 / (2 * np.pi)
            except:
                return float('inf')
        except:
            return float('inf')
    
    def hermite(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        elif n == 2:
            return 4*z**2 - 2
        elif n == 3:
            return 8*z**3 - 12*z
        elif n == 4:
            return 16*z**4 - 48*z**2 + 12
        elif n == 5:
            return 32*z**5 - 160*z**3 + 120*z
        elif n == 6:
            return 64*z**6 - 384*z**4 + 480*z**2 - 120
        elif n == 7:
            return 128*z**7 - 1024*z**5 + 1920*z**3 - 1008*z
        elif n == 8:
            return 256*z**8 - 2560*z**6 + 6400*z**4 - 5760*z**2 + 1680
        return 0

    def exact_r_max(coeffs):
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        Q = Q - Q.subs(z, 0)
        Q = Q.as_numer_denom()[0]
        Q = Q.as_poly()
        roots = Q.all_roots()
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if r > 0:
                return r**2 / (2 * np.pi)
        return float('inf')

    for _ in range(10):
        coeffs = np.random.rand(8) * 2 - 1
        result = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        if time() - start_time > budget_s - 1:
            break
        if result.fun < best_value:
            best_value = result.fun
            best_coeff = result.x
    
    return best_coeff
