from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, S, N
from sympy.polys.polytools import factor
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    
    def evaluate_coefficients(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_numerical()
        Q = Q.subs(z, 0)
        Q = Q.as_poly()
        Q = Q * z**2
        Q = Q.as_poly()
        
        def find_r_max():
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = sorted(real_roots)
            for r in real_roots:
                if r > 0:
                    if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                        return r**2 / (2 * np.pi)
            return np.inf
        
        return find_r_max()
    
    def surrogate_objective(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))

        Q = Q * z**2
        Q = Q.as_poly()
        
        def find_r_max_surrogate():
