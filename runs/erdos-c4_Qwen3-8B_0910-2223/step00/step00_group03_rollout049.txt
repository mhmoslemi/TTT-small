import numpy as np
import random
from scipy.optimize import minimize
from sympy import symbols, Hermite, Poly, roots
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    def evaluate_bound(coeffs):
        c = np.array(coeffs)
        z = symbols('z')
        H = [Hermite(n) for n in range(len(c))]
        Q = sum(c[i] * H[i] for i in range(len(c)))
        Q = Q.subs(z, 0)
        Q = Q.as_poly()
        Q = Q * z**2
        roots = roots(Q)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def fast_objective(coeffs):
        return evaluate_bound(coeffs)
    
    best_coeffs = None
    best_value = float('inf')
    
    for _ in range(10):
        init = np.random.rand(8) * 2 - 1
        result = minimize(fast_objective, init, method='L-BFGS-B', options={'maxiter': 100})
        if result.success:
            coeffs = result.x
            value = fast_objective(coeffs)
            if value < best_value:
                best_value = value
                best_coeffs = coeffs
    
    return best_coeffs
