import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, simplify, solve, Eq, N
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c0 = np.random.rand(8)
    best_coeff = c0
    best_bound = float('inf')
    
    def evaluate_coeffs(coeffs):
        c = np.array(coeffs, dtype=np.float64)
        z = symbols('z')
        Q = sum(c[k] * Hermite(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0).as_numerical_value()
        Q = Q - Q.subs(z, 0)
        if Q < 0:
            Q = -Q
        Q = simplify(Q)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if len(real_roots) > 0 else 1
        U = r_max**2 / (2 * np.pi)
        return U
    
    def objective(x):
        return evaluate_coeffs(x)
    
    for _ in range(10):
        start_time = time()
        result = minimize(objective, c0, method='L-BFGS-B', bounds=[(0, 100) for _ in range(8)])
        if result.success:
            current_bound = result.fun
            if current_bound < best_bound:
                best_bound = current_bound
                best_coeff = result.x
        if time() - start_time > budget_s * 0.8:
            break
        c0 = result.x
    
    return best_coeff
