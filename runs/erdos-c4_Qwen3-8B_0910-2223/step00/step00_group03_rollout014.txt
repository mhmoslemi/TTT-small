from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_surrogate(coefficients):
        def Q(z):
            res = 0
            for k in range(len(coefficients)):
                res += coefficients[k] * hermite(4*k, z)
            return res
        
        def hermite(n, z):
            if n == 0:
                return 1.0
            elif n == 1:
                return 2*z
            else:
                return 2*z*hermite(n-1, z) - 2*(n-1)*hermite(n-2, z)
        
        def find_r_max():
            try:
                root = root_scalar(lambda z: Q(z)/z**2, bracket=[1, 1000], method='brentq')
                return root.root**2
            except:
                return 1000000000
        
        r_max_sq = find_r_max()
        return r_max_sq / (2 * np.pi)
    
    def fitness_func(x):
        return evaluate_surrogate(x)
    
    def num_coeffs():
        return 8
    
    def initial_guess():
        return np.random.rand(num_coeffs()) * 2 - 1
    
    result = minimize(fitness_func, initial_guess(), method='L-BFGS-B', bounds=[(-10, 10) for _ in range(num_coeffs())], options={'maxiter': 100, 'disp': False})
    
    return result.x
