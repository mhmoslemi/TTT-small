import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from sympy import symbols, hermite, Poly, solve, S
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coeffs(c):
        z = symbols('z')
        H = [hermite(k, z) for k in range(len(c))]
        Q = sum(c[k] * H[k] for k in range(len(c)))
        Q = Q.as_poly().as_expr()
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q / Q.subs(z, 0)
        Q = Q / z**2
        try:
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            r_max = max(real_roots)
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def grad(c):
        return np.random.randn(len(c))
    
    best = np.random.rand(8)
    best_score = evaluate_coeffs(best)
    
    for _ in range(100):
        start_time = time.time()
        res = minimize(lambda c: evaluate_coeffs(c), x0=best, jac=grad, method='L-BFGS-B', bounds=[(0, 10)]*8, options={'maxiter': 50})
        if time.time() - start_time > budget_s:
            break
        if evaluate_coeffs(res.x) < best_score:
            best = res.x
            best_score = evaluate_coeffs(best)
    
    return best
