import numpy as np
import scipy.optimize as opt
import sympy as sp
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def objective(c):
        # Numerical surrogate for root finding
        t = np.linspace(-10, 10, 1000)
        Q = np.zeros_like(t)
        for k in range(len(c)):
            H = sp.hermite(h=4*k)
            Q += c[k] * np.array([H.subs('z', ti) for ti in t])
        Q /= t**2
        roots = np.diff(np.sign(Q)).nonzero()[0]
        r_max = t[roots[0]] if roots.size else 10
        return r_max**2 / (2 * np.pi)
    
    def validate_coefficients(c):
        # Exact validation with sympy
        z = sp.symbols('z')
        Q = sum(c[k] * sp.hermite(4*k, z) for k in range(len(c)))
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = sp.solvers.solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    c0 = np.random.rand(8)
    best_c = c0.copy()
    best_bound = validate_coefficients(c0)
    
    for _ in range(200):
        if budget_s <= 0:
            break
        budget_s -= 1
        
        # Numerical optimization
        result = opt.minimize(
            objective,
            c0,
            method='L-BFGS-B',
            bounds=[(0, 10) for _ in range(8)],
            options={'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        c = result.x
        c = np.clip(c, 0, 10)
        
        # Validate with exact arithmetic
        try:
            bound = validate_coefficients(c)
        except:
            bound = float('inf')
        
        if bound < best_bound:
            best_bound = bound
            best_c = c.copy()
    
    return best_c
