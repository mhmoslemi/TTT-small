import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermiti

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coeffs(c):
        H = [hermiti(4*k, 0) for k in range(len(c))]
        Q = np.sum([c[k] * H[k] for k in range(len(c))], axis=0)
        Q = Q / np.polyint(Q, m=2)
        try:
            root = root_scalar(lambda x: Q(x), bracket=[0, 100], method='brentq')
            return root.root ** 2 / (2 * np.pi)
        except:
            return np.inf
    
    def objective(c):
        return evaluate_coeffs(c)
    
    results = []
    for _ in range(30):
        c = np.random.rand(8) * 2 - 1
        res = minimize(objective, c, method='L-BFGS-B', bounds=[(-5, 5)]*8, options={'gtol': 1e-8, 'ftol': 1e-8})
        results.append((res.fun, res.x))
    
    results.sort()
    return results[0][1]
