import matplotlib.pyplot as plt

from sympy import symbols, hermite, diff, solve, N, Rational

def run(seed=42, budget_s=600, **kwargs):

    def objective(c):

            return float('inf')

            # Find sign change
            candidates = []
            for i in range(len(real_roots)-1):
                r1, r2 = real_roots[i], real_roots[i+1]
                if r2 > 0 and Q.subs(z, (r1 + r2)/2).evalf() != 0:
                    candidates.append((r1 + r2)/2)
            if not candidates:
                return float('inf')

            return r_max**2 / (2 * np.pi)

            return float('inf')
