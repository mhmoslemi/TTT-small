from scipy.optimize import minimize
from sympy import symbols, hermite, poly, solve, Eq, simplify
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def numerical_surrogate(c):
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        Q = Q.as_coefficients_dict()
        c_m = -Q[1] if 1 in Q else 0
        Q = Q = Q = sum(c[k] * hermite(4*k, z) for k in range(len(c))) + c_m * hermite(4*len(c), z)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    def exact_eval(c):
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        Q = Q.as_coefficients_dict()
        c_m = -Q[1] if 1 in Q else 0
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c))) + c_m * hermite(4*len(c), z)

        real_roots = [r for r in real_roots if r > 0]

        r_max = real_roots[-1] if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        return exact_eval(c)
    
    best_coeff = None
    best_value = float('inf')
    
    start_time = time()
    
    for coeff_count in range(2, 9):
        print(f"Trying {coeff_count} coefficients...")
        initial_guess = np.random.rand(coeff_count)
