import numpy as np
import random
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, simplify, solve, roots, apart
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    best_coeffs = None
    best_score = float('inf')
    
    def score(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q.as_numerical()
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q.as_numerical().coeff(z**2)
        if Q.as_numerical().subs(z, 100) < 0:
            Q = -Q.as_numerical()
        Q = Q.as_numerical()
        
        def f(z_val):
            return Q.evalf(subs={z: z_val})
        
        try:
            r_max = root_scalar(f, bracket=[1, 1000], method='brentq').root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def generate_candidate(coeffs):
        return [coeffs[i] + random.uniform(-0.1, 0.1) for i in range(len(coeffs))]
    
    start_time = time.time()
    
    for _ in range(100):
        coeffs = [random.uniform(-1, 1) for _ in range(8)]
        current_score = score(coeffs)
        
        if current_score < best_score:
            best_score = current_score
            best_coeffs = coeffs
        
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_coeffs
