import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from sympy import symbols, HermiteH, roots, solve, simplify
from sympy.polys.polytools import factor
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    best_c = c.copy()
    best_bound = float('inf')
    start_time = time()
    
    def objective(c):
        nonlocal best_bound
        if time() - start_time > budget_s - 10:
            return best_bound
        
        c_list = c.tolist()
        z = symbols('z')
        Q = sum(c_list[k] * HermiteH(4*k, z) for k in range(8))
        
        # Force Q(0) = 0
        Q0 = Q.subs(z, 0)
        if Q0 != 0:
            Q = Q - Q0
        
        # Ensure positivity at infinity: flip sign if necessary
        coeff = Q.as_poly().all_coeffs()
        if coeff[-1] < 0:
            Q = -Q
        
        # Remove z^2 factor and find largest real root across sign change
        Q = Q.as_expr()
        Q = Q.subs(z, z**2).as_expr()  # Remove z^2 factor
        roots_sym = roots(Q, z, domain='Real')
        
        # Find r_max as the largest positive real root with sign change
        real_roots = [r for r in roots_sym if r.is_real and r > 0]
        real_roots = sorted(real_roots)
        r_values = []
        for i in range(len(real_roots)-1):
            if real_roots[i+1] - real_roots[i] > 1e-5:
                r_values.append(real_roots[i])
                r_values.append(real_roots[i+1])
        
        r_values = sorted(r_values)
        if not r_values:
            return float('inf')
        
        max_r = r_values[-1]
        bound = max_r**2 / (2 * np.pi)
        
        if bound < best_bound:
            best_bound = bound
            best_c = c.copy()
        
        return bound
    
    # Use gradient descent with constraints
    bounds = Bounds(0, 10)
    cons = LinearConstraint(np.ones(8), -np.inf, np.inf)

    result = minimize(objective, c, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)

    return best_c
