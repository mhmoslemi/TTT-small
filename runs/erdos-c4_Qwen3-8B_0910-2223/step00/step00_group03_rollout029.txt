from scipy.optimize import minimize
from scipy.optimize import Bounds

from sympy import symbols, hermite, expand, simplify, Eq, solve
import time

def run(seed=42, budget_s=600, **kwargs):

    z = symbols('z')

    c0 = 0.5
    c1 = -0.5
    c2 = 0.3
    c3 = -0.3
    c4 = 0.2
    c5 = -0.2
    c6 = 0.1

            Q += coefficients[k] * hermite(4*k, z)

        # Q(z) = sum c_k H_{4k}(z)
        # Compute the constant term: H_{0}(0) = 1
        # So we need sum c_k * H_{4k}(0) = 0
        # H_{4k}(0) for k >= 0 is non-zero only if 4k even, which it is
        # Compute H_{4k}(0) for k in range(len(coefficients))
        H0 = [hermite(4*k, 0) for k in range(len(coefficients))]
        const_term = sum([coefficients[k] * H0[k] for k in range(len(coefficients))])
        # Adjust c_m to make Q(0) = 0
        # Assume the last coefficient is the one to be adjusted
        # We will scale all coefficients so that Q(0) = 0
        scale = -const_term / H0[-1]
        adjusted_coeff = [coefficients[k] * scale for k in range(len(coefficients))]
        
        # Now form Q(z) = sum adjusted_coeff[k] * H_{4k}(z)
        Q = 0

            Q += adjusted_coeff[k] * hermite(4*k, z)

        Q_div_z2 = Q / z**2

        Q_poly = expand(Q_div_z2)

        roots = solve(Q_poly, z, domain='Real')

        if not sign_changes:
            return float('inf')
        
        r_max = max(sign_changes)
        return r_max**2 / (2 * np.pi)
    
    # Optimization with numerical surrogate

        return objective(x)
    
    result = minimize(surrogate_objective, x0=coefficients, method='L-BFGS-B', bounds=Bounds([-1, 1], [-1, 1]))
    best_coefficients = result.x
