import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import root_scalar
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    max_budget = budget_s
    start_time = time.time()
    
    def evaluate_coefficients(c):
        def hermite_4k(n):
            def hermite(n, x):
                if n == 0:
                    return 1
                elif n == 1:
                    return 2*x
                else:
                    return 2*x*hermite(n-1, x) - 2*(n-1)*hermite(n-2, x)
            return hermite(n, x)
        
        def Q(z):
            total = 0
            for k in range(len(c)):
                total += c[k] * hermite_4k(4*k)(z)
            return total
        
        def Q_z2(z):
            return Q(z) / (z**2)
        
        def find_r_max():
            try:
                r = root_scalar(Q_z2, bracket=[1, 100], method='brentq', rtol=1e-8)
                return r.root
            except:
                return 100
        
        r_max = find_r_max()
        U = r_max**2 / (2 * np.pi)
        return U
        
    def objective(c):
        return evaluate_coefficients(c)
    
    def bounds_generator():
        return [(-10, 10) for _ in range(C4_MAX_COEFF_COUNT)]
    
    def initial_guess():
        return np.random.uniform(-10, 10, C4_MAX_COEFF_COUNT)
    
    def refine_guess(c):
        for _ in range(10):
            c -= 0.1 * np.gradient(objective, c)
        return c
    
    def optimize():
        c = initial_guess()
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            c = refine_guess(c)
        best_c = c

            new_val = objective(new_c)
            if new_val < best_val:
                best_val = new_val
                best_c = new_c
        return best_c
    
    best_coeff = optimize()
