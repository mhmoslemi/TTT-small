import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    c = np.random.randn(8)
    
    def objective(x):
        c = x
        z = sp.symbols('z')
        H = [sp.hermite_poly(k) for k in range(9)]
        Q = sum(c[k] * H[4*k] for k in range(8))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = [r.evalf() for r in sp.solve(Q, z)]
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted(real_roots)
        sign_changes = 0
        for i in range(len(real_roots)-1):
            if real_roots[i] * real_roots[i+1] < 0:
                sign_changes += 1
        r_max = real_roots[-1] if sign_changes > 0 else None
        if r_max is None:
            return float('inf')
        U = (r_max**2) / (2 * np.pi)
        return U
    
    result = minimize(objective, c, method='L-BFGS-B', options={'maxiter': 100})
    return result['x'].tolist()
