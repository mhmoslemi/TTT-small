import numpy as np
from scipy.optimize import minimize
from scipy.optimize import fsolve
from scipy.integrate import quad
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    start_time = time.time()
    
    def hermite_poly(n):
        def H(x):
            return np.polysplines.pchip(np.linspace(-10, 10, 1000), [np.polysplines.spline(n, k=3)(x) for x in np.linspace(-10, 10, 1000)])(x)
        return H
    
    def evaluate_candidate(c):
        nonlocal best_bound
        m = len(c)
        H = [lambda x, k=k: hermite_poly(4*k)(x) for k in range(m)]
        
        def Q(x):
            return sum(c[k] * H[k](x) for k in range(m))
        
        def Q_z2(x):
            return Q(x) / (x**2 + 1e-12)
        
        def sign_change(x):
            return np.sign(Q_z2(x))
        
        def find_r_max():
            x_vals = np.linspace(-100, 100, 10000)
            sign_changes = np.sign(np.roll(np.sign(Q_z2(x_vals)), 1))
            idxs = np.where(sign_changes < 0)[0]
            if not idxs.size:
                return 1e10
            for idx in idxs:
                x = x_vals[idx]
                if np.isclose(Q_z2(x), 0):
                    continue
                if np.isclose(Q_z2(x), 0):
                    continue
                if np.sign(Q_z2(x)) != np.sign(Q_z2(x_vals[idx-1])):
                    return x
            return 1e10
        
        r_max = find_r_max()
        if r_max < 1e-6:
            return 1e10
        return r_max ** 2 / (2 * np.pi)
    
    def objective(c):
        nonlocal best_bound
        bound = evaluate_candidate(c)
        if bound < best_bound:
            best_bound = bound
            best_coeff = c.copy()
        return bound
    
    def surrogate_objective(c):
        return np.sum((c - np.random.normal(0, 1, len(c))) ** 2)
    
    initial_guess = np.random.normal(0, 1, 8)
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(surrogate_objective, initial_guess, method='L-BFGS-B', bounds=[(-5, 5) for _ in range(8)])
        initial_guess = result.x
        for _ in range(10):
            if time.time() - start_time > budget_s - 10:
                break
            try:
                result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(-5, 5) for _ in range(8)])
                initial_guess = result.x
            except:
                break
    
    return best_coeff
