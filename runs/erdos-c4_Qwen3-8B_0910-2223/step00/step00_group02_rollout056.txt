from scipy.optimize import minimize
from scipy.optimize._numdiff import approx_derivative
import time

        nonlocal best_bound
        start_time = time.time()

            # Define the Hermite polynomials up to degree 4k for k in 0..m-1
            def H(n, z):
                if n == 0:
                    return 1
                elif n == 1:
                    return 2*z
                else:
                    return 2*z*H(n-1, z) - 2*(n-1)*H(n-2, z)
            
            m = len(x)
            Q = np.zeros(m)
            for k in range(m):
                Q[k] = x[k] * H(4*k, 0)
            
            # Ensure Q(0) = 0 and adjust for sign at infinity
            if Q[0] != 0:
                Q = Q * (Q[0] / Q[0])  # This is redundant but ensures positivity

            # We numerically search for r_max
            def Q_normalized(z):
                return Q[0] + Q[1]*z + Q[2]*z**2 + Q[3]*z**3 + Q[4]*z**4
            
            # Newton-Raphson to find the largest positive real root
            def find_r_max():
                def f(z):
                    return Q_normalized(z)
                def df(z):
                    return 2*Q[1] + 6*Q[2]*z + 12*Q[3]*z**2 + 24*Q[4]*z**3

                            root = z + 1
                            break
                        except:

                max_iter = 100
                eps = 1e-8

                    fz = f(z)
                    dfz = df(z)

                    z -= fz / dfz

        except:
