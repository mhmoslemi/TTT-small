import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, diff, solve, Rational, simplify

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    total_time = 0

    def eval_upper_bound(c):
        c = np.array(c, dtype=np.float64)
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_independent(z)[1].as_coefficients_dict()[z**2] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for i in range(len(real_roots) - 1):
            r_max = real_roots[i]
            if (real_roots[i + 1] - r_max) < 1e-6:
                continue
            return r_max**2 / (2 * np.pi)
        return float('inf')

    def surrogate_bound(c):
        c = np.array(c, dtype=np.float64)
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_independent(z)[1].as_coefficients_dict()[z**2] * z**2
        Q = Q.as_poly()
        roots = [r.evalf() for r in Q.all_roots() if r.is_real]
        real_roots = sorted([r for r in roots if r > 0])
        for i in range(len(real_roots) - 1):
            r_max = real_roots[i]
            if (real_roots[i + 1] - r_max) < 1e-6:
                continue
            return r_max**2 / (2 * np.pi)
        return float('inf')

    def objective(c):
        return surrogate_bound(c)

    for _ in range(10):
        start_time = time.time()
        initial_guess = np.random.uniform(-1, 1, 8)
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(8)])
        if time.time() - start_time > budget_s:
            break
        if result.success:
            c = result.x
            bound = eval_upper_bound(c)
            if bound < best_bound:
                best_bound = bound
                best_coeff = c
        total_time += time.time() - start_time

    return best_coeff
