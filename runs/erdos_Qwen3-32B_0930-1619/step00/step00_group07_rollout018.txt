import numpy as np
import time

def run(seed=42, budget_s=1100):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    assert abs(np.sum(h) - target_sum) < 1e-8

    alpha = 0.1
    lambda_tv = 0.01
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    best_h = h.copy()

    start_time = time.time()
    max_iter = 10**6  # High upper bound, will break on time

    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()

        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)

        # Compute overlap gradient
        valid_mask = (np.arange(n_points) + k_max) >= 0
        valid_mask &= (np.arange(n_points) + k_max) < n_points
        valid_indices = np.where(valid_mask)[0]

        grad_overlap = np.zeros(n_points)
        if len(valid_indices) > 0:
            grad_overlap[valid_indices] = dx * ((1.0 - h[valid_indices + k_max]) - h[valid_indices])

        # TV gradient
        tv_grad = np.zeros(n_points)
        tv_grad[1:-1] = lambda_tv * (h[:-2] - 2 * h[1:-1] + h[2:])
        grad = grad_overlap + tv_grad

        # Update and project
        h -= alpha * grad
        np.clip(h, 0.0, 1.0, out=h)
        current_sum = h.sum()
        delta = (target_sum - current_sum) / n_points
        h += delta
        np.clip(h, 0.0, 1.0, out=h)

        # Exit on time
        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
