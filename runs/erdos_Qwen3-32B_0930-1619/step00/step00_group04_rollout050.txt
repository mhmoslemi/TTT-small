import numpy as np
import math
import time

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def perturb(h_current):
    n = len(h_current)
    indices = np.random.choice(n, size=2, replace=False)
    i, j = indices
    delta_plus = min(1.0 - h_current[i], h_current[j])
    delta_minus = min(h_current[i], 1.0 - h_current[j])
    if np.random.rand() < 0.5:
        delta = np.random.uniform(0, delta_plus)
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        delta = np.random.uniform(0, delta_minus)
        h_new = h_current.copy()
        h_new[i] -= delta
        h_new[j] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
