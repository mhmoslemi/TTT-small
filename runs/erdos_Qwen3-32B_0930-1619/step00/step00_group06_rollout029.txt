def project_onto_feasible(v, s, n_points):

    h_proj = cp.Variable(n_points)

    obj = cp.Minimize(cp.sum_squares(h_proj - v))

    constraints = [0 <= h_proj, h_proj <= 1]

    constraints.append(cp.sum(h_proj) == s)

    prob = cp.Problem(obj, constraints)

    prob.solve(solver=cp.ECOS)

        return np.array(h_proj.value).flatten()
