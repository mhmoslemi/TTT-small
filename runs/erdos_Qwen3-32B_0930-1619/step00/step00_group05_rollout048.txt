        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
            no_improvement_count = 0
            alpha = min(0.1, alpha * 1.5)
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                alpha = max(min_alpha, alpha * alpha_decay)
                no_improvement_count = 0

    return (best_h, best_c5, n_points)
