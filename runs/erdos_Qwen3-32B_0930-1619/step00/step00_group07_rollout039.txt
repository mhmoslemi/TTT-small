import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': constraint_sum},)

    def objective_func(h_vec):
        h = h_vec.reshape(n_points)
        max_c = 0.0
        for k in range(n_points):
            valid = n_points - k
            sum_term = np.sum(h[:valid] * (1.0 - h[k:k + valid]))
            c_k = dx * sum_term
            if c_k > max_c:
                max_c = c_k
        return max_c

    best_h = initial_h.copy()
    best_c5 = objective_func(best_h)

    options = {
        'maxiter': 100,
        'ftol': 1e-9,
        'disp': False
    }

    result = minimize(
        fun=objective_func,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    if result.success:
        final_c5 = result.fun
        if final_c5 < best_c5:
            best_h = result.x
            best_c5 = final_c5

    elapsed = time.time() - start_time
    print(f"Elapsed: {elapsed:.2f}s, Found C₅={best_c5:.7f}")

    return (best_h.tolist(), best_c5, n_points)
