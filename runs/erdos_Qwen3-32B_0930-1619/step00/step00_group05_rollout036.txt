import numpy as np
import time
import random

def run(seed=42, budget_s=1100):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    h_init = initial_h_values  # Assumes this is a valid global numpy array
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(h_candidate):
        h_candidate_clipped = np.clip(h_candidate, 0.0, 1.0)
        s = np.sum(h_candidate_clipped)
        if abs(s) < 1e-10:
            return h_candidate_clipped.copy()
        scale = target_sum / s
        return h_candidate_clipped * scale
    
    def evaluate_candidate(h_array):
        corr = np.correlate(h_array, 1.0 - h_array, mode='full')
        return np.max(corr) * dx
    
    def tournament_selection(pop, fitnesses, t_size=3):
        indices = np.random.choice(len(pop), t_size, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    pop_size = 30
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        population.append(repair(h_rand))
    
    best_h, best_c5 = None, float('inf')
    
    while time.time() - start_time < budget_s - 0.1:
        fitnesses = [evaluate_candidate(h) for h in population]
        improved = any(f < best_c5 for f in fitnesses)
        if improved:
            idx = np.argmin(fitnesses)
            best_c5 = fitnesses[idx]
            best_h = population[idx].copy()
        
        idxs = np.argsort(fitnesses)
        population = [population[i] for i in idxs]
        fit_sorted = [fitnesses[i] for i in idxs]
        
        new_pop = population[:int(pop_size * 0.2)]
        
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fit_sorted, 3)
            p2 = tournament_selection(population, fit_sorted, 3)
            
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:split], p2[split:]]).copy()
            if np.random.rand() < 0.15:

                child[m_idx] += np.random.uniform(-0.2, 0.2, len(m_idx))
                child = repair(child)
            child = repair(child)
            new_pop.append(child)
        
        population = new_pop
    
    return best_h, best_c5, n_points
