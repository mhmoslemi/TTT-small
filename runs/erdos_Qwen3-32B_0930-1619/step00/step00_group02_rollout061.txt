        if total_removable >= D + 1e-8:
            h -= (D / total_removable) * h[non_zero]
        else:
            h[non_zero] = 0
    h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_fitness(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr * dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    try:
        initial_h_values_g = np.array(initial_h_values)
        n_points = len(initial_h_values_g)
    except NameError:
        n_points = 100
    T = n_points / 2.0
    population_size = 50
    max_generations = 200
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    # Initialize population
    population = []
    try:
        h0 = np.array(initial_h_values_g).copy()
        h0 = repair(h0, T)
        population.append(h0)
    except NameError:
        pass
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair(h, T)
        population.append(h)
    population = np.array(population)

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 15:
            break
        # Evaluate fitness
        fitness = [evaluate_fitness(h, n_points) for h in population]
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices]
        current_best = population[0]
        current_best_fitness = fitness[0]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = current_best.copy()
        # Selection and crossover
        parent_pool = population[:population_size // 2]
        new_population = []
        while len(new_population) < population_size - parent_pool.shape[0]:
            p1, p2 = np.random.choice(parent_pool, 2)
            if np.random.rand() < 0.8:
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy()
            # Mutation
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = repair(child, T)
            else:
                child = repair(child, T)
            new_population.append(child)
        population = np.concatenate((parent_pool, np.array(new_population)))

    final_fitness = [evaluate_fitness(h, n_points) for h in population]
    best_idx = np.argmin(final_fitness)
    h_values = population[best_idx].tolist()
    c5_bound = final_fitness[best_idx]

    return (h_values, c5_bound, n_points)
