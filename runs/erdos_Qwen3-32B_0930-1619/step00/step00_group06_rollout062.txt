                    h_candidate[adjust_idx] = new_adj
                    # Enforce bounds
                    h_candidate[adjust_idx] = np.clip(h_candidate[adjust_idx], 0, 1)
                    if np.all((h_candidate >= 0) & (h_candidate <= 1)):
                        c5_new = evaluate_erdos_solution(h_candidate)
                        if c5_new < best_c5:
                            best_h[:] = h_candidate.copy()
                            best_c5 = c5_new
                            improved = True
                # Check time in inner loop
                if time.time() - start_time > budget_s:
                    return best_h, best_c5, n_points
            # End delta
        # End i
        if not improved:
            step_size *= 0.5
            steps_down += 1
            if steps_down > max_step_down:
                break
        else:
            steps_down = 0
        # Check time
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
