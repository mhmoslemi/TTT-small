    while time.time() - start_time < budget_s:
        # Compute current C5 and max indices
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points: 2*n_points - 1]
        current_c5 = np.max(valid_corr * dx)
        max_indices = np.flatnonzero(np.isclose(valid_corr, np.max(valid_corr), atol=1e-6))
        max_k_values = max_indices  # These are zero-based in the valid_corr array, corresponding to k=1 to k=n_points-1

        # Compute gradient
        gradient = np.zeros_like(h)
        for k_idx in max_k_values:
            k = k_idx + 1  # Convert zero-based index to actual shift
            for i in range(n_points):
                # Compute term1: derivative from h[i] affecting (1 - h[i +k])
                if i + k < n_points:
                    term1 = (1 - h[i + k]) * dx
                else:
                    term1 = 0.0
                # Compute term2: derivative from h[i -k] affecting h[i]
                if i - k >= 0:
                    term2 = -h[i - k] * dx
                else:
                    term2 = 0.0
                gradient[i] += term1 + term2

        if len(max_k_values) > 0:
            gradient /= len(max_k_values)

        # Update h
        h -= lr * gradient

        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if not np.isclose(current_sum, target_sum):
            scale = target_sum / current_sum
            h = h * scale
            # Clamp again after scaling
            h = np.clip(h, 0, 1)

        # Recompute C5 after update
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points: 2*n_points - 1]
        new_c5 = np.max(valid_corr * dx)

        # Update best_h if improvement
        if new_c5 < current_c5:
            best_h[:] = h[:]
            no_progress_count = 0
        else:
            no_progress_count += 1
            if no_progress_count > max_no_progress:
                # Perturb h
                h += np.random.normal(0, perturb_strength, size=h.shape)
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                if not np.isclose(current_sum, target_sum):
                    scale = target_sum / current_sum
                    h = h * scale
                    h = np.clip(h, 0, 1)
                no_progress_count = 0
                lr *= 0.5

        # Decay learning rate
        lr *= decay_rate

        # Early exit check
        if time.time() - start_time > budget_s - 1:
            break

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    valid_corr = corr[n_points: 2*n_points - 1]
    final_c5 = np.max(valid_corr * dx)
    return best_h, final_c5, n_points
