    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0

    start_time = time.time()
    max_iter = 1000
    no_improvement_count = 0
    patience = 100
    lr_decay_factor = 0.5
    lr_decay_steps = 100

    for iter in range(max_iter):
        t += 1
        current_time = time.time()
        if current_time - start_time > budget_s - 1:  # leave 1 sec buffer
            break

        # Compute cross-correlation
        c = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(c)
        max_idx = np.argmax(c)

        # Compute the lag k
        k = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += 1.0 - h[j]
                grad[j] -= h[i]

        # Multiply by dx since C_5 = correlation * dx
        grad = grad * dx

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = m_hat / (np.sqrt(v_hat) + epsilon)

        h -= learning_rate * step

        # Clip to [0,1]
        np.clip(h, 0, 1, out=h)

        # Project to satisfy sum constraint
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            # Fallback if sum is zero, but unlikely with proper clipping
            h[:] = 0.5

        # Compute current C₅
        c_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c_current < best_c5:
            best_c5 = c_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Learning rate decay
        if no_improvement_count >= lr_decay_steps:
            learning_rate *= lr_decay_factor
            no_improvement_count = 0

    return (best_h, best_c5, n_points)

import time

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    best_c5 = float('inf')

    learning_rate = 0.01
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    
    start_time = time.time()
    max_iter = 1000
    no_improvement = 0
    patience = 100
    lr_decay = 0.5
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        c = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c)
        k = max_idx - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += 1.0 - h[j]
                grad[j] -= h[i]
        grad *= dx
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = m_hat / (np.sqrt(v_hat) + epsilon)
        
        h -= learning_rate * step
        np.clip(h, 0, 1, out=h)
        total = np.sum(h)
        if total != 0:
            h *= (n_points / 2) / total
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            learning_rate *= lr_decay
            no_improvement = 0
    
    return (best_h, best_c5, n_points)
