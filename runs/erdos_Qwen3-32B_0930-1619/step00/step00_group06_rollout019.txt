    h_clipped = np.clip(h, 0, 1)
    sum_clipped = h_clipped.sum()
    if np.isclose(sum_clipped, target_sum):

    sum_clipped = h_clipped.sum()
    if np.isclose(sum_clipped, target_sum):

    h_final = np.clip(h_scaled, 0, 1)
    sum_final = h_final.sum()

    if addable > 0:
