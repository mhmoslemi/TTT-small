import numpy as np
import time

def project(h, target_sum, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = h_clipped.sum()
    if np.isclose(current_sum, target_sum, atol=1e-6):
        return h_clipped
    delta = target_sum - current_sum
    adjustment = delta / n_points
    h_projected = h_clipped + adjustment
    return np.clip(h_projected, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = None

    for iter in range(max_iter):

        # Compute correlation and find k_max
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)

        # Compute gradient
        grad_overlap = np.zeros(n_points)
        for i in range(n_points):
            j1 = i + k_max
            term1 = 0.0
            if 0 <= j1 < n_points:
                term1 = 1.0 - h[j1]
            j2 = i - k_max
            term2 = 0.0

            grad_overlap[i] = (term1 + term2) * dx

        # TV gradient
        grad_tv = np.zeros(n_points)
        for i in range(n_points):
            if i > 0:
                left = h[i] - h[i-1]
                grad_tv[i] -= np.sign(left)
            if i < n_points - 1:
                right = h[i+1] - h[i]
                grad_tv[i] += np.sign(right)

        # Update

        h = project(h - alpha_start * grad_total, target_sum, n_points)
