        perturbation = rng.uniform(-0.05, 0.05, size=n_points)
        h_start = np.clip(best_h + perturbation, 0, 1)
        h_sum = np.sum(h_start)
        if h_sum > tolerance:
            h_start = h_start * (n_points / 2.0) / h_sum
        h_start = np.clip(h_start, 0.0, 1.0)

        # Run constrained optimization
        result = scipy.optimize.minimize(
            fun=lambda x: compute_c5(x),
            x0=h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 50, 'ftol': 1e-4}
        )

        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_sum = np.sum(h_opt)
            if h_sum > tolerance:
                h_opt = h_opt * (n_points / 2.0) / h_sum
                h_opt = np.clip(h_opt, 0.0, 1.0)
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_h = h_opt
                best_c5 = current_c5

    # Final projection to satisfy constraints
    h_sum = np.sum(best_h)
    if h_sum > tolerance:
        best_h = best_h * (n_points / 2.0) / h_sum
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
