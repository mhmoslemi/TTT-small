import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=np.float64).copy()
    
    # Precompute cross-correlation indices for each k
    index_map = {}
    for k in range(-n_points + 1, n_points):
        indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                indices.append((i, j))
        index_map[k] = indices
    
    # Objective function to minimize C
    def objective(x):
        return x[-1]
    
    # Equality constraint for the integral
    def eq_constraint(x):
        return np.sum(x[:-1]) - n_points / 2
    
    # Constraint for each shift k
    def constraint_k(x, k):
        h = x[:-1]
        total = 0.0
        for i, j in index_map[k]:
            total += h[i] * (1 - h[j])
        total *= dx
        return total - x[-1]  # <= 0
    
    # Generate inequality constraints
    ineq_constraints = []
    for k in range(-n_points + 1, n_points):
        ineq_constraints.append({
            'type': 'ineq',
            'fun': lambda x, k=k: constraint_k(x, k)
        })
    
    # Combined constraints
    constraints = [
        {'type': 'eq', 'fun': eq_constraint}
    ] + ineq_constraints

    # Initial guess
    cross_corr_initial = np.correlate(h_initial, 1 - h_initial, mode='full')
    C_initial = cross_corr_initial.max() * dx
    x0 = np.concatenate([h_initial, [C_initial]])
    
    # Bounds: h in [0,1], C unbounded
    bounds = [(0.0, 1.0) for _ in range(n_points)] + [(None, None)]
    
    # Solver options with time check in loop
    start_time = time.time()
    result = minimize(
        objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-6, 'disp': False}
    )
    
    # Post-optimization validation
    h_opt = np.clip(result.x[:-1], 0.0, 1.0)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)  # Enforce integral constraint
    cross_corr_final = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = cross_corr_final.max()
    
    return h_opt, c5_bound, n_points
