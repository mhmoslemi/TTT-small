        max_contribution = np.max(conv)
        i_max = np.argmax(conv)
        k_max = i_max - (n_points - 1)

        # Gradient calculation
        grad = np.zeros(n_points)
        if k_max >= 0:
            valid = n_points - k_max
            for i in range(valid):
                grad[i] += (1.0 - h[i + k_max]) * dx
                grad[i + k_max] -= h[i] * dx
        else:
            k_val = -k_max
            for i in range(k_val, n_points):
                grad[i - k_val] += (1.0 - h[i]) * dx
                grad[i] -= h[i - k_val] * dx

        # Add TV regularization
        second_diff = np.zeros(n_points)
        for i in range(1, n_points - 1):
            second_diff[i] = h[i - 1] - 2 * h[i] + h[i + 1]
        grad += lambda_tv * second_diff

        # Update h
        h -= alpha * grad

        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2.0

        # Adjust sum constraint
        if abs(current_sum - target_sum) > 1e-6:
            free = h[(h > 0) & (h < 1)].copy()
            if free.size > 0:
                delta = (target_sum - current_sum) / free.size
                mask = (h > 0) & (h < 1)
                h[mask] += delta
                h = np.clip(h, 0, 1)

    # Final evaluation
    conv = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(conv)
    return h.tolist(), c5_bound, n_points
