import numpy as np
import time

def compute_gradient(h, k_max, dx, n_points):
    g = np.zeros(n_points)
    i = np.arange(n_points)
    j = i + k_max
    valid = (j >= 0) & (j < n_points)
    i_valid = i[valid].astype(int)
    j_valid = j[valid].astype(int)
    g[i_valid] += (1 - h[j_valid]) * dx
    g[j_valid] -= h[i_valid] * dx
    return g

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100):

    dx = 2.0 / n_points
    alpha = 0.1
    lambda_tv = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k_max = max_index - (n_points - 1)

        g = compute_gradient(h, k_max, dx, n_points)
        tv_grad = np.zeros(n_points)
