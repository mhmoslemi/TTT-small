    # Use provided initial_h_values if available
    try:
        h_initial = initial_h_values
        n_points = len(h_initial)
        target_sum = n_points / 2
        if not np.allclose(np.sum(h_initial), target_sum, atol=1e-8):
            raise ValueError("Initial h has incorrect sum")
    except:
        # Fallback construction for when initial_h_values is not available
        n_points = 45
        h_initial = np.full(n_points, 0.5)  # Start with uniform 0.5 values

    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    current_h = h_initial.copy()
    current_c5 = best_c5

    temperature = 1000.0
    cooling_rate = 0.995
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Perturb current_h keeping constraints
        candidate_h = perturb_h(current_h, target_sum, n_points)
        if candidate_h is None:
            continue  # Skip invalid candidates
        
        # Evaluate candidate
        try:
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            delta = candidate_c5 - current_c5
        except:
            continue  # Skip if evaluation fails

        # Acceptance decision
        if delta < 0:
            # Strict improvement
            current_h = candidate_h
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
        else:
            # Metropolis acceptance
            p = np.exp(-delta / max(temperature, 1e-5))
            if np.random.rand() < p:
                current_h = candidate_h
                current_c5 = candidate_c5

        # Cool the temperature
        temperature *= cooling_rate

        # Early stopping condition
        if temperature < 1e-8:
            break

    return best_h, best_c5, n_points

def perturb_h(h, target_sum, n_points):
    candidate_h = h.copy()
    perturb_count = min(5, n_points)  # Perturb 5% or 5 elements
    indices = np.random.choice(n_points, perturb_count, replace=False)
    
    # Small random perturbations in [-0.1, 0.1]
    perturb = np.random.uniform(-0.1, 0.1, size=perturb_count)
    candidate_h[indices] += perturb
    
    # Clip to [0, 1]
    candidate_h = np.clip(candidate_h, 0, 1)
    
    # Adjust sum without violating bounds
    current_sum = np.sum(candidate_h)
    error = target_sum - current_sum
    
    # Only proceed if error is significant
    if abs(error) < 1e-8:
        return candidate_h
    
    # Use proportional adjustment on non-bound elements
    adjustment_steps = 10
    step_size = error / adjustment_steps
    for _ in range(adjustment_steps):
        if abs(error) < 1e-8:
            break
            
        if error > 0:
            # Need to increase values
            non_upper = np.where(candidate_h < 1 - 1e-8)[0]
            if len(non_upper) == 0:
                continue
            idx = np.random.choice(non_upper)
            candidate_h[idx] = min(candidate_h[idx] + step_size, 1.0)
        else:
            # Need to decrease values
            non_lower = np.where(candidate_h > 1e-8)[0]
            if len(non_lower) == 0:
                continue
            idx = np.random.choice(non_lower)
            candidate_h[idx] = max(candidate_h[idx] + step_size, 0.0)
        current_sum = np.sum(candidate_h)
        error = target_sum - current_sum
    
    # Final validation of constraints
    if not np.all((candidate_h >= -1e-8) & (candidate_h <= 1 + 1e-8)):
        return None
    if not np.isclose(np.sum(candidate_h), target_sum, atol=1e-8):
        return None
        
    return candidate_h
