    start_time = time.time()
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h)
    
    while time.time() - start_time < budget_s - 1:
        i = random.randint(0, n_points-1)
        h_i = h[i]
        delta = max(-0.02, min(0.02, np.random.normal(0, 0.01)))
        h_i_new = np.clip(h[i] + delta, 0.0, 1.0)
        delta_total = h_i_new - h_i
        
        if abs(delta_total) < 1e-8:
            continue
        
        non_i = [j for j in range(n_points) if j != i]
        candidates = np.array(h[non_i])
        
        if delta_total > 0:
            j_candidate = non_i[np.argmax(candidates)]
            if h[j_candidate] >= delta_total:
                h_new = np.copy(h)
                h_new[i] = h_i_new
                h_new[j_candidate] -= delta_total
        else:
            amount = -delta_total
            j_candidate = non_i[np.argmin(candidates)]
            if h[j_candidate] + amount <= 1.0:
                h_new = np.copy(h)
                h_new[i] = h_i_new
                h_new[j_candidate] += amount
            else:
                continue
        
        if np.all((h_new >= 0) & (h_new <= 1)) and abs(np.sum(h_new) - target_sum) < 1e-8:
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
