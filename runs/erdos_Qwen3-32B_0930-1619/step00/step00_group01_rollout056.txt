                        new_affected_overlaps.append(new_overlap)
                    new_affected_max = max(new_affected_overlaps) if new_affected_overlaps else 0.0
                    new_max_candidate = max(non_affected_max, new_affected_max)
                    if new_max_candidate < best_new_max:
                        best_new_max = new_max_candidate
                        best_candidate = c
                h_current[i] = best_candidate

            # Normalize sum
            current_sum = np.sum(h_current)
            if np.isclose(current_sum, 0.0):
                scale = 1.0
            else:
                scale = (n_points / 2.0) / current_sum
            h_current = np.clip(h_current * scale, 0.0, 1.0)

            # Evaluate best_c5
            c5_curr = evaluate_erdos_solution(h_current)
            if c5_curr < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_curr
            coord_pass += 1

    return (best_h.tolist(), best_c5, n_points)
