import numpy as np
import time

def project(h, required_sum):
    h_clipped = np.clip(h, 0, 1).copy()
    current_sum = h_clipped.sum()
    if np.allclose(current_sum, required_sum):
        return h_clipped
    delta = required_sum - current_sum
    if delta > 0:
        candidates = h_clipped < 1.0
        if not candidates.any():
            return h_clipped
        amount_to_add = delta / candidates.sum()
        h_clipped[candidates] += amount_to_add
    else:
        candidates = h_clipped > 0.0
        if not candidates.any():
            return h_clipped
        amount_to_subtract = -delta / candidates.sum()
        h_clipped[candidates] -= amount_to_subtract
    return h_clipped

    start_time = time.time()
    alpha = 0.05  # Initial learning rate
    max_iter = 10_000
    best_c5 = np.inf
    best_h = h.copy()

    for iteration in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:
            break

        overlap = np.correlate(h, 1 - h, mode='full') * dx
