import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 200  # Number of samples
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    alpha = 0.1  # Step size for perturbation
    T = 0.1  # Initial temperature
    T_decay = 0.99  # Temperature decay factor every 10 iterations
    early_stop = 100  # Stop if no improvement for this many iterations

        h = np.zeros(n_points)
        for i in range(n_points):
            x = x_new[i]
            idx = np.searchsorted(x_original, x, side='right') - 1
            if idx < 0:
                idx = 0
            elif idx >= original_n:
                idx = original_n - 1
            h[i] = initial_h_values[idx]
    except:
        # Fallback to random initialization
        h = np.random.rand(n_points)
    
    def project(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_clipped)
        delta = target_sum - current_sum
        if abs(delta) < 1e-8:
            return h_clipped
        if delta > 0:
            mask = h_clipped < 1
            n_mask = np.sum(mask)
            if n_mask == 0:
                return h_clipped
            amount = delta / n_mask
            h_clipped[mask] += amount
        else:
            mask = h_clipped > 0
            n_mask = np.sum(mask)
            if n_mask == 0:
                return h_clipped
            amount = -delta / n_mask
            h_clipped[mask] -= amount
        return np.clip(h_clipped, 0, 1)
    
    # Ensure h has correct sum after initialization
    h = project(h)
    
    def compute_c5(h):
        correlations = np.correlate(h, 1 - h, mode='full')
        return np.max(correlations) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s:
