def project_to_feasible(h, D):

    h = np.clip(h, 0.0, 1.0)

    current_sum = h.sum()

    if abs(current_sum - D) < 1e-8:

        return h

    # Need to adjust h to get sum D.

    delta = current_sum - D

    # Depending on the sign of delta.

    if delta > 0:

        # Need to decrease by delta.

        # Sort the indices in descending order of h[i]

        indices_desc = np.argsort(h)[::-1]

        for i in indices_desc:

            if delta <= 0:

                break

            # How much can we decrease this element?

            max_decrease = h[i] - 0.0

            decrease = min(delta, max_decrease)

            h[i] -= decrease

            delta -= decrease

    else:

        # Need to increase by -delta.

        delta = -delta

        indices_asc = np.argsort(h)

        for i in indices_asc:

            if delta <= 0:

                break

            max_increase = 1.0 - h[i]

            increase = min(delta, max_increase)

            h[i] += increase

            delta -= increase

    return h

import numpy as np
import time

def project_to_feasible(h, D):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - D) < 1e-8:
        return h
    delta = current_sum - D
    if delta > 0:
        indices_desc = np.argsort(h)[::-1]
        for i in indices_desc:
            if delta <= 0:
                break
            max_decrease = h[i]
            decrease = min(delta, max_decrease)
            h[i] -= decrease
            delta -= decrease
    else:
        delta = -delta
        indices_asc = np.argsort(h)
        for i in indices_asc:
            if delta <= 0:
                break
            max_increase = 1.0 - h[i]
            increase = min(delta, max_increase)
            h[i] += increase
            delta -= increase
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
