import numpy as np
import time
from scipy.signal import correlate

def compute_c5(h, n_points):
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode='full')
    dx = 2.0 / n_points
    return np.max(corr) * dx

def project(h, target_sum):
    h = np.clip(h, 0, 1)
    adjustment = (target_sum - h.sum()) / len(h)
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0  # Enforces integral constraint

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    temp = 1.0
    T_decay = 0.995
    max_no_improve = 150

    start_time = time.time()
    no_improve = 0

    while time.time() - start_time < budget_s:
        # Perturb with temperature-scaled random step
        delta = np.random.uniform(-0.2, 0.2, size=n_points)
        h_candidate = project(h + delta, target_sum)

        # Enforce integral constraint more strictly
        current_sum = h_candidate.sum()
        if not np.allclose(current_sum, target_sum, rtol=1e-5):
            scale = np.sqrt((target_sum / current_sum))
            h_candidate = np.clip(h_candidate * scale, 0, 1)

        c5_candidate = compute_c5(h_candidate, n_points)
        delta_c5 = best_c5 - c5_candidate

        # Acceptance criterion
        if delta_c5 < 0 or np.exp(delta_c5 / temp) > np.random.uniform():
            h = h_candidate.copy()
            if delta_c5 < 0:
                best_c5 = c5_candidate
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        temp *= T_decay
        if no_improve > max_no_improve:
            break

    return best_h, best_c5, n_points
