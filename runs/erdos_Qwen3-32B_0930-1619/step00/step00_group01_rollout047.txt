import numpy as np
import time
import scipy.optimize

def run():
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0

    def adjust_sum(h_vec):
        s = h_vec.sum()
        if abs(s - target_sum) > 1e-6:
            adj = (target_sum - s) / n_points
            h_vec = h_vec + adj
            h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    h = adjust_sum(h)
    best_h = h.copy()

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx

    best_c5 = compute_c5(best_h)

    seed = 42
    np.random.seed(seed)
    start_time = time.time()
    T = 0.1
    T_min = 1e-4
    alpha = 0.999  # cooled per iter
    h_current = best_h.copy()

    # Simulated Annealing
    while time.time() - start_time < 900:  # leave 200s for gradient phase
        frac = min(1.0, (time.time() - start_time) / 900)
        perturb_rate = 0.05 + frac * 0.05  # increase perturbation as time progresses
        indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
        std_dev = T * (1.0 - frac) + 0.001 * frac  # adaptively scale noise
        delta = np.random.normal(0, std_dev, len(indices))

        h_new = h_current.copy()
        h_new[indices] = np.clip(h_new[indices] + delta, 0, 1)
        h_new = adjust_sum(h_new)

        current_c5 = compute_c5(h_current)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5:
            h_current = h_new.copy()

            h_current = h_new.copy()

        T *= alpha

    # Gradient descent with constraints
    def objective(x):
        return compute_c5(np.clip(x, 0, 1))

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
