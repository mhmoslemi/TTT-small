import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Adam parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m_t = np.zeros_like(h)
    v_t = np.zeros_like(h)
    iteration = 0
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(overlaps)
        m = max_index - n_points + 1

        # Compute gradient
        grad = np.zeros_like(h)
        for a in range(n_points):
            b = a + m
            if 0 <= b < n_points:
                grad[a] += (1 - h[b]) * dx
                grad[b] -= h[a] * dx

        # Update Adam parameters
        iteration += 1
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad ** 2)
        t = iteration
        m_hat = m_t / (1 - beta1 ** t)
        v_hat = v_t / (1 - beta2 ** t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        h_candidate = h - step
        # Project back to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        s = h_candidate.sum()
        if s != 0:
            scaling = target_sum / s
            h_candidate = h_candidate * scaling
        h_candidate = np.clip(h_candidate, 0, 1)

        h[:] = h_candidate[:]

        # Check if current solution is better
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points

def run(seed=42, budget_s=1100, **kwargs):

    beta1 = 0.9

    epsilon = 1e-8
    m_t = np.zeros_like(h)

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(overlaps)
        m = max_index - n_points + 1
        
        grad = np.zeros_like(h)
        for a in range(n_points):
            b = a + m
