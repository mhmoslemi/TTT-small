                temp_h_scaled = temp_h * scaling
                # Clip to bounds and adjust for sum
                temp_clipped = np.clip(temp_h_scaled, 0, 1)
                sum_clipped = np.sum(temp_clipped)
                if abs(sum_clipped - T) > 1e-6:  # Minor readjustment
                    delta = T - sum_clipped
                    temp_clipped[i] = np.clip(temp_clipped[i] + delta, 0, 1)

                # Evaluate C5
                current_c5 = evaluate_erdos_solution(temp_clipped)
                if current_c5 < temp_best_c5:
                    best_candidate = candidate
                    temp_best_c5 = current_c5
                    temp_best_h = temp_clipped.copy()

            if temp_best_c5 < best_c5:
                h = temp_best_h.copy()
                best_c5 = temp_best_c5
                improved = True

            if elapsed >= budget_s - 0.1:
                break

        if not improved:
            break  # Early stopping if no progress

    return h, best_c5, n_points
