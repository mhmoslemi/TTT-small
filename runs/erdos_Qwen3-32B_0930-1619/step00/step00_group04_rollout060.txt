                delta = correction / n_points
                new_h = np.clip(new_h + delta, 0, 1)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h[:] = new_h
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
