def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    start_time = time.time()

    h_current = initial_h_values.copy()

            h_candidate[j] = b_new
            
            c5_candidate = compute_c5(h_candidate)
