import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    np.random.seed(seed)

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        target_sum = n_points / 2
        if current_sum == 0:
            h.fill(target_sum / n_points)
            return h
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    h_initial = repair(h_initial)

    population_size = 40
    pop = [np.copy(h_initial)]
    for _ in range(population_size - 1):
        rand_h = np.random.rand(n_points)
        pop.append(repair(rand_h))
    pop = np.array(pop)

    best_h = np.copy(h_initial)
    best_c5 = float('inf')
    start_time = time.time()

    def compute_c5(h):
        h_flat = h.flatten()
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s:
        c5_list = np.array([compute_c5(ind) for ind in pop])
        current_idx = c5_list.argmin()
        current_c5 = c5_list[current_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = pop[current_idx].copy()

        new_pop = [np.copy(best_h)]  # Elitism
        while len(new_pop) < population_size:
            idxs = np.random.choice(population_size, 3, replace=False)
            targs = c5_list[idxs]
            targs_idx = np.argsort(targs)
            p1, p2 = pop[idxs[targs_idx[0]]], pop[idxs[targs_idx[1]]]

            # Crossover
            if np.random.rand() < 0.9:
                mid = np.random.randint(1, n_points - 1)
                c1, c2 = np.concatenate([p1[:mid], p2[mid:]]), np.concatenate([p2[:mid], p1[mid:]])
            else:
                c1, c2 = np.copy(p1), np.copy(p2)

            # Mutation
            for child in [c1, c2]:
                if np.random.rand() < 0.1:
                    mutate_indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                    for i in mutate_indices:
                        perturb = child[i] * (np.random.uniform(-0.15, 0.15))
                        child[i] += perturb
                        child[i] = np.clip(child[i], 0, 1)
                child[:] = repair(child)
                new_pop.append(np.copy(child))

        pop = np.array(new_pop[:population_size])

    return best_h, best_c5, n_points
