            constraints=constraints,
            callback=callback,
            options=options
        )
        if result.success:
            h_values = result.x
        else:
            h_values = best_h
    except:
        h_values = best_h
    
    c5_bound = objective_func(h_values)
    
    return h_values.tolist(), c5_bound, n_points
