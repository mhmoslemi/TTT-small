                grad[j] -= h[i]
        
        # Update h
        h -= alpha * grad
        
        # Project back to feasible set
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h *= target_sum / current_sum
    
    return best_h, best_c5, n_points
