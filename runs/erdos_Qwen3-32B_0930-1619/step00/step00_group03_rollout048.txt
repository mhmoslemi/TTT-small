        h = np.clip(h, 0, 1)

        current_sum = h.sum()

        if not np.isclose(current_sum, target_sum):

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    assert np.isclose(h.sum(), target_sum), "Initial h sum constraint violated"
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()
    
    lr = 0.02
    lr_decay = 0.999
    eval_interval = 100
    
    while time.time() - start_time < budget_s:
        current_c5 = None
        ones_h = 1.0 - h
        cross_corr = np.correlate(h, ones_h, mode='full')
        corrs = cross_corr * dx
        current_c5 = np.max(corrs)
        max_indices = np.where(corrs == current_c5)[0]
        
        grad = np.zeros(n_points)
        for idx in max_indices:
            lag = idx - (n_points - 1)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
        if len(max_indices) > 0:
            grad /= len(max_indices)
        h -= lr * grad
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            delta = (target_sum - current_sum) / n_points
            h += delta
            h = np.clip(h, 0, 1)
        
        lr *= lr_decay
        
        # Update best solution
        if current_c5 is not None and current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping if target is met
        if best_c5 < 0.38085 and time.time() - start_time > 10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
