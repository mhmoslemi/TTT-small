            best_c5 = new_c5
            current_h = new_h.copy()
            current_c5 = new_c5
        else:
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = new_h.copy()
                current_c5 = new_c5
        
        T *= alpha
    
    # Gradient Descent with constraint handling
    remaining_time = timeout - time.time()
    if remaining_time > 0:
        def objective(x):
            return compute_c5(x, dx)
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - desired_sum})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = minimize(
            objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(remaining_time * 1000), 'ftol': 1e-6}
        )
        
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    adjustment = (desired_sum - best_h.sum()) / n_points
    best_h += adjustment
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h, dx)
    
    return best_h, final_c5, n_points
