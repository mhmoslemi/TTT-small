import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h = cp.Variable(n_points, nonneg=True)
    C = cp.Variable()

    constraints = []

    # Integral constraint
    constraints.append(cp.sum(h) == n_points / 2.0)

    # Bounds for h
    constraints.append(h <= 1.0)
    constraints.append(h >= 0.0)

    # Initialize with the given starting point
    h.value = np.array(initial_h_values, dtype=np.float64)

    # Generate cross-correlation constraints for all valid shifts
    for k in range(-n_points + 1, n_points):

        end_i = min(n_points, n_points - k)
        if start_i >= end_i:

        terms = [h[i] * (1 - h[i + k]) for i in range(start_i, end_i)]
        if terms:  # Ensure there are terms to sum
            constraints.append(cp.sum(terms) <= C / dx)

    # Objective: minimize the maximum overlap
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)

    # Set solver options with time limit
    ipopt_time_limit = budget_s * 0.9 - (time.time() - start_time)
    solver_opts = {

        'print_level': 0
    }

    try:
        problem.solve(solver=cp.IPOPT, verbose=False, **solver_opts)
    except Exception:
        pass  # Fallback to original solution if solver fails

    # Ensure time budget is honored
    if time.time() - start_time > budget_s:
        h_values = initial_h_values.copy()

        h_values = np.nan_to_num(h.value, nan=0.5, posinf=1.0, neginf=0.0)

        h_values *= (n_points / 2.0) / np.sum(h_values)

    c5_bound = evaluate_erdos_solution(h_values)
