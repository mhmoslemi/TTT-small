def run(seed=42, budget_s=1100, **kwargs):

    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time()

    while time() - start_time < budget_s - 1:
        ones_minus_h = 1.0 - h
        overlaps = np.correlate(h, ones_minus_h, mode='full') * dx

        max_idx = np.argmax(overlaps)
        m_val = max_idx - (n_points - 1)

        grad = np.zeros_like(h)
        if m_val >= 0:
            right = n_points - m_val
            grad[:right] += (1.0 - h[m_val:]) * dx
            grad[m_val:] += -h[:right] * dx
        else:
            k = -m_val
            contrib = (1.0 - h[:n_points - k]) * dx
            grad[k:] += contrib
            contrib2 = -h[k:] * dx
            grad[:n_points - k] += contrib2

        t = int(1e6)  # Use dummy time step to avoid state
        m += (1 - beta1) * (grad - m)
        v += (1 - beta2) * (grad**2 - v)
        m_cap = m / (1 - beta1**t)
        v_cap = v / (1 - beta2**t)

        h_candidate = h - learning_rate * m_cap / (np.sqrt(v_cap) + epsilon)
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        s = np.sum(h_candidate)
        if abs(s - target_sum) > 1e-8 and s != 0:

            h_candidate *= alpha
            h_candidate = np.clip(h_candidate, 0.0, 1.0)

        current_c5 = overlaps.max()
