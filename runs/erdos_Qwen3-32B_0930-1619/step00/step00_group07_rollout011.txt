    start_time = time.time()

    eta = eta_initial

    while time.time() - start_time < budget_s and iter < max_iter:
        # Compute Correlations
        one_minus_h = 1.0 - h
        cross_corr = np.correlate(h, one_minus_h, mode='full')
        Corr = cross_corr * dx
        current_c5 = np.max(Corr)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find max index and corresponding lag
        max_idx = np.argmax(Corr)

        # Compute delta_h

            valid_i = np.arange(0, n_points - lag)
            delta_h[valid_i] = h[valid_i] * (1 - h[valid_i + lag])
        else:

            valid_i = np.arange(-lag, n_points)
            delta_h[valid_i] = h[valid_i] * (1 - h[valid_i + lag])

        # Update h

        eta = eta_initial / (1 + beta * iter)

        iter += 1
