    best_c5 = float('inf')
    max_iter = 100000
    alpha = 0.1
    patent_iter = 0
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1e-3:
            break
        
        # Compute correlation and identify max overlap k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        max_c5 = corr[max_idx]
        
        # Update best solution
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            patent_iter = 0
            alpha = min(alpha * 1.1, 0.5)
        else:
            patent_iter += 1
            alpha = max(alpha * 0.95, 0.001)
        
        # Gradient for max overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Forward term
            j = i + k
            term1 = (1.0 - h[j]) * dx if (0 <= j < n_points) else 0.0
            # Backward term
            m = i - k
            term2 = -h[m] * dx if (0 <= m < n_points) else 0.0
            gradient[i] = term1 + term2
        
        # Gradient step
        h[:] -= alpha * gradient

        h[:] = np.clip(h, 0, 1)

        if abs(sum_h - target_sum) > 1e-8:
