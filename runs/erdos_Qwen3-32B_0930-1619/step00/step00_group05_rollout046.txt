import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def repair_h(h):
        h = np.clip(h, 0, 1)
        target = n_points / 2
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        s_new = np.sum(h)
        if s_new == 0:
            return np.full_like(h, 0.5)
        h = h * (target / s_new)
        h = np.clip(h, 0, 1)
        return h

    h_initial = repair_h(h_initial)

    swarm_size = 30
    max_iter = 1000
    inertia = 0.8
    cognitive = 2.0
    social = 2.0

    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []

    p = h_initial.copy()
    particles.append(p)
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    personal_best.append(p)
    personal_best_c5.append(float('inf'))

    for _ in range(1, swarm_size):
        h = h_initial.copy() + np.random.uniform(-0.1, 0.1, n_points)
        h = repair_h(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        personal_best.append(h)
        personal_best_c5.append(float('inf'))

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    global_best = None
    global_best_c5 = float('inf')

    for i in range(swarm_size):
        c5 = compute_c5(particles[i])
        personal_best_c5[i] = c5
        if c5 < global_best_c5:
            global_best = particles[i].copy()
            global_best_c5 = c5

    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v_new = (inertia * v +
                     cognitive * r1 * (personal_best[i] - h) +
                     social * r2 * (global_best - h))

            h_new = h + v_new
            h_new = repair_h(h_new)
            particles[i] = h_new
            velocities[i] = v_new

            new_c5 = compute_c5(h_new)
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = h_new.copy()
                personal_best_c5[i] = new_c5
                if new_c5 < global_best_c5:
                    global_best = h_new.copy()
                    global_best_c5 = new_c5

        inertia *= 0.995

    return (global_best, global_best_c5, n_points)
