                else:
                    new_h[j] = candidate

        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if not np.allclose(new_sum, required_sum, atol=1e-5):
            for j in range(n_points):
                if 0 < new_h[j] < 1:
                    adjust = required_sum - new_sum
                    new_h[j] += adjust
                    new_h[j] = max(0.0, min(1.0, new_h[j]))
                    break

        current_c5 = evaluate_h(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = new_h[:]

    return (best_h.tolist(), best_c5, n_points)
