import numpy as np
import cvxpy as cp
import time

def compute_correlations(h, dx):
    n_points = len(h)
    if n_points == 0:
        return np.array([])
    corrs = np.correlate(h, h, mode='full')
    start_idx = n_points - 1
    end_idx = start_idx + n_points
    relevant_corrs = corrs[start_idx:end_idx]
    overlaps = 1.0 - relevant_corrs * dx
    return overlaps

def compute_linearized_overlap(h_current, h_var, k, dx):
    n_points = len(h_current)
    offset = k
    terms = []
    for i in range(n_points - offset):
        current_i_val = h_current[i]
        current_j_val = h_current[i + offset]
        term = current_i_val * h_var[i + offset] + h_var[i] * current_j_val - current_i_val * current_j_val
        terms.append(term)
    linearized_sum = sum(terms) * dx
    return 1 - linearized_sum  # Return the overlap integral approximation
