    α = 0.1
    max_iter = 100000

    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break

        ones_minus_h = 1.0 - h
        corrs = np.correlate(h, ones_minus_h, mode='full') * dx
        max_corr = np.max(corrs)
        m = np.argmax(corrs)
        k_max = m - (n - 1)

        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + k_max < n:
                grad[i] += (1.0 - h[i + k_max])
            if 0 <= i - k_max < n:
                grad[i] -= h[i - k_max]
        grad *= dx

        grad_tv = np.zeros(n)
        for i in range(n):

                grad_tv[i] = (h[1] if n > 1 else 0.0) - 2.0 * h[i]
