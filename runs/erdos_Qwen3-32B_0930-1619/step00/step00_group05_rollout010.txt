import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    dx = 2.0 / len(h)
    n_points = len(h)
    
    # Enforce constraints on initial h
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != 0:
        h = (h / total) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    # ADAM optimizer parameters
    alpha = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m, v = np.zeros_like(h), np.zeros_like(h)
    no_improve_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current correlation max
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(c)
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Find maximum lag index
        max_idx = np.argmax(c)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for current max lag
        i = np.arange(n_points)
        i_plus_k = i + lag
        mask = (i_plus_k >= 0) & (i_plus_k < n_points)
        part1 = np.where(mask, 1.0 - h[i_plus_k], 1.0)
        
        i_minus_k = i - lag
        mask2 = (i_minus_k >= 0) & (i_minus_k < n_points)
        part2 = np.where(mask2, -h[i_minus_k], 0.0)
        
        grad = (part1 + part2) * dx
        
        # ADAM step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**((time.time() - start_time) / (10**-6)))
        v_hat = v / (1 - beta2**((time.time() - start_time) / (10**-6)))
        h -= alpha * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible space
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-10:
            h.fill(0.5)
        else:
            h *= (n_points / 2) / total
        
        # Learning rate decay
        if no_improve_count > 100:
            alpha *= 0.5
            no_improve_count = 0
    
    return best_h.tolist(), best_c5, n_points
