    pbest_scores = np.full(n_particles, np.inf)
    gbest_position = np.zeros(n_points)
    gbest_score = np.inf

    # Initialize particles with perturbations
    for i in range(n_particles):
        if i == 0:
            particles[i] = initial_h.copy()
        else:
            pert = np.random.uniform(-0.05, 0.05, n_points)
            particle = initial_h.copy() + pert
            particle = np.clip(particle, 0., 1.)
            s = np.sum(particle)
            if s != 0:
                particle *= (n_points / 2.0) / s
                particle = np.clip(particle, 0., 1.)
            else:
                particle = initial_h.copy()
            particles[i] = particle
        # Evaluate and update pbest/gbest
        score_i = evaluate_erdos_solution(particles[i])
        if score_i < pbest_scores[i]:
            pbest_scores[i] = score_i
            pbest_positions[i] = particles[i].copy()
            if score_i < gbest_score:
                gbest_score = score_i
                gbest_position = particles[i].copy()

    # PSO parameters
    w, c1, c2 = 0.9, 2.0, 2.0
    start_time = time.time()

    # Main loop
    while time.time() - start_time < budget_s:
        no_imp = True
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_positions[i] - particles[i]) +
                             c2 * r2 * (gbest_position - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0., 1.)
            s = np.sum(particles[i])
            if s != 0:
                particles[i] *= (n_points / 2.0) / s
                particles[i] = np.clip(particles[i], 0., 1.)
            else:
                particles[i] = initial_h.copy()
            # Evaluate
            score_i = evaluate_erdos_solution(particles[i])
            if score_i < pbest_scores[i]:
                pbest_scores[i] = score_i
                pbest_positions[i] = particles[i].copy()
                if score_i < gbest_score:
                    gbest_score = score_i
                    gbest_position = particles[i].copy()
                    no_imp = False
        if no_imp:
            break  # Early stopping

    return (gbest_position, gbest_score, n_points)
