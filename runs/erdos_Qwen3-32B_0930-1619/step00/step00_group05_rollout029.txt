            corr = np.correlate(h, g, mode='full')
            current_score = np.max(corr) * dx
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = h.copy()
                # Update global best
                if current_score < global_best_score:
                    global_best_position = h.copy()
                    global_best_score = current_score
                    if current_score < best_c5:
                        best_h = h.copy()
                        best_c5 = current_score

        # Update velocities and positions
        for i in range(num_particles):
            h = particles[i]
            vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            # Update velocity
            vel_new = (inertia_weight * vel +
                       cognitive_weight * r1 * (personal_best_positions[i] - h) +
                       social_weight * r2 * (global_best_position - h))
            # Update position
            h_new = h + vel_new
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = h_new.sum()
            if sum_h_new == 0:
                h_new = best_h.copy()
            else:
                h_new *= required_sum / sum_h_new
            velocities[i] = vel_new
            particles[i] = h_new

        # Decrease inertia weight over time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        inertia_weight = max(0.4, 0.9 - 0.5 * (elapsed / budget_s))

    return best_h, best_c5, n_points
