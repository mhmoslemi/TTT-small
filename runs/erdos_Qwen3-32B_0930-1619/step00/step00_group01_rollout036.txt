    x0 = h_initial.ravel()

    start_time = time()
    timeout = 0.9 * budget_s  # Avoid last-minute I/O penalties

    result = minimize(
        fun=objective_function,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=linear_constraint,
        options={'maxiter': 100, 'disp': False}
    )

    h_opt = result.x.reshape(n_points)
    c5_bound = objective_function(h_opt)

    # Ensure the final solution satisfies constraints exactly
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_sum = h_opt.sum()
    if not np.isclose(h_sum, n_points / 2, atol=1e-8):
        scaling = (n_points / 2) / h_sum
        h_opt = np.clip(h_opt * scaling, 0.0, 1.0)

    c5_bound = objective_function(h_opt)
    return h_opt, c5_bound, n_points
