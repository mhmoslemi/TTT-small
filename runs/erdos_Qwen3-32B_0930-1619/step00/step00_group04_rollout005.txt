import numpy as np
import time

def compute_c5(h, dx):
    h = np.array(h, dtype=float)
    one_minus_h = 1 - h
    correlations = np.correlate(h, one_minus_h, mode='full')
    return max(correlations) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    T = n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    timeout = budget_s

    while time.time() - start_time < timeout:
        improved = False
        for i in range(n_points):
            current_value = h[i]
            current_c5 = compute_c5(h, dx)
            best_candidate_h = h.copy()
            best_c5_i = current_c5

            # Generate candidate values for h[i]
            steps = [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]
            candidates = [0.0, 1.0]
            for step in steps:
                c = max(0.0, min(1.0, current_value + step))
                candidates.append(c)

            for c in candidates:
                h_candidate_initial = h.copy()
                h_candidate_initial[i] = c
                sum_candidate_initial = np.sum(h_candidate_initial)
                if sum_candidate_initial == 0:
                    continue
                scale = T / sum_candidate_initial
                h_scaled = h_candidate_initial * scale
                h_clipped = np.clip(h_scaled, 0, 1)
                sum_clipped = np.sum(h_clipped)
                # Ensure sum is correct despite clipping
                if not np.isclose(sum_clipped, T, atol=1e-3):
                    # Renormalize only if necessary
                    h_clipped *= T / sum_clipped
                    h_clipped = np.clip(h_clipped, 0, 1)
                candidate_c5 = compute_c5(h_clipped, dx)
