    n = len(h)
    grad = np.zeros(n)
    for m in range(n):
        if 0 <= (m + k_max) < n:
            grad[m] += (1 - h[m + k_max])
        if 0 <= (m - k_max) < n:
            grad[m] -= h[m - k_max]
