    indices = np.argsort(h)
    if delta > 0:
        for i in indices:
            possible = 1.0 - h[i]
            if possible <= 0:
                continue
            if delta <= possible:
                h[i] += delta
                break
            else:
                h[i] = 1.0
                delta -= possible
    else:
        delta_abs = -delta
        indices = indices[::-1]
        for i in indices:
            possible = h[i]
            if possible <= 0:
                continue
            if delta_abs <= possible:
                h[i] -= delta_abs
                break
            else:
                h[i] = 0.0
                delta_abs -= possible
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    required_sum = n_points / 2.0
    assert np.abs(np.sum(h) - required_sum) < 1e-8, "Initial h must be feasible"

    current_c5 = evaluate_erdos_solution(h, n_points)
    best_c5 = current_c5
    best_h = h.copy()

    T_initial = 1000.0
    cooling_rate = 0.995
    T = T_initial
    perturb_ratio = 0.05
    perturb_amount = 0.1
    start_time = time.time()
    MAX_ITER = 100000  # High limit, time-based exit

    for iteration in range(MAX_ITER):
        if time.time() - start_time > budget_s:
            break

        h_prime = h.copy()
        n_perturb = max(1, int(n_points * perturb_ratio))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-perturb_amount, perturb_amount)
            h_prime[i] = max(0.0, min(1.0, h_prime[i] + delta))

        h_prime = adjust_sum(h_prime, required_sum)
        c5_prime = evaluate_erdos_solution(h_prime, n_points)

        if c5_prime < current_c5:
            h = h_prime.copy()
            current_c5 = c5_prime
            if c5_prime < best_c5:
                best_h = h.copy()
                best_c5 = c5_prime
        else:
            prob = np.exp((current_c5 - c5_prime) / T)
            if np.random.rand() < prob:
                h = h_prime.copy()
                current_c5 = c5_prime

        T *= cooling_rate

    return best_h.tolist(), best_c5, n_points
