import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize h
    try:
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    except NameError:
        n_points = 80
        h = 0.5 * np.ones(n_points, dtype=np.float64)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    alpha = 0.1
    lambda_tv = 0.01

    while time.time() - start_time < budget_s - 0.1:
        # Compute correlation and find k_max
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)

        # Gradient for overlap at k_max
        grad_overlap = np.zeros(n_points)
        for m in range(n_points):
            term1 = 0.0
            if 0 <= m + k_max < n_points:
                term1 = (1.0 - h[m + k_max]) * dx
            term2 = 0.0
            if 0 <= m - k_max < n_points:
                term2 = -h[m - k_max] * dx
            grad_overlap[m] = term1 + term2

        # TV regularization gradient
        grad_tv = np.zeros(n_points)
        for i in range(1, n_points - 1):
            grad_tv[i] = lambda_tv * (h[i - 1] - 2.0 * h[i] + h[i + 1])

        grad = grad_overlap + grad_tv
        h_new = h - alpha * grad

        # Projection step
        h_clipped = np.clip(h_new, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        target_sum = n_points / 2.0
        delta = target_sum - current_sum

        if abs(delta) > 1e-8:
            mask = (h_clipped > 0.0) & (h_clipped < 1.0)
            if np.any(mask):
                adjustment = delta / np.count_nonzero(mask)
                h_clipped[mask] += adjustment
                h_clipped = np.clip(h_clipped, 0.0, 1.0)

        h = h_clipped

        # Check improvement
        current_C5 = np.max(corr := np.correlate(h, 1 - h, mode='full') * dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

    return (best_h.tolist(), best_C5, n_points)
