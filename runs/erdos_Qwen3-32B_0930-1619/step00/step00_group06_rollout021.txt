import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values).copy().flatten()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(x):
        x = x.reshape(-1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr * dx)

    def project(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if abs(total - n_points / 2) > 1e-6:
            delta = (n_points / 2 - total) / n_points
            x += delta
            x = np.clip(x, 0, 1)
        return x

    best_h = project(h)
    best_c5 = objective(best_h)
    start_time = time.time()

    np.random.seed(seed)
    for _ in range(3):  # Limited perturbations to avoid local optima
        if time.time() - start_time >= budget_s:
            break
        # Perturb current best_h
        perturbed = np.random.normal(0, 0.1, n_points)
        x0 = project(best_h + perturbed)

        # Setup optimization
        bounds = [(0, 1)] * n_points
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

        # Run optimization
        try:
            result = minimize(
                objective, x0, method='SLSQP', 
                bounds=bounds, constraints=constraint,
                options={'maxiter': 200, 'ftol': 1e-6}
            )
            if result.success:
                candidate = project(result.x)
                c5_candidate = objective(candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = candidate
        except Exception:
            continue

    return best_h.tolist(), best_c5, n_points
