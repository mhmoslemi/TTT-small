import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 1000, 'ftol': 1e-8, 'disp': False, 'iprint': 0}
   
    start_time = time.time()

        fun=objective,

        constraints=constraints,
        options=options
    )

    elapsed = time.time() - start_time
