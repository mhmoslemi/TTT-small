    # Configuration
    n_points = 40
    dx = 2.0 / n_points
    initial_h = np.full(n_points, 0.5)  # Equal weights to meet ∫h dx = 1

    C_low, C_high = 0.0, 0.38085

    best_c5 = C_high
    start_time = time.time()

        constraints = [
            h >= 0,
