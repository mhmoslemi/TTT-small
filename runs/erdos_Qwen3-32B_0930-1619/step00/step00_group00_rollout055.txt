    target_sum = n_points / 2.0
    learning_rate = 0.01
    max_outer = 15000
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s and iteration < max_outer:
        iteration += 1

        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_C = np.max(overlaps)
        idx_max = np.argmax(overlaps)
        m = idx_max - (n_points - 1)

        grad = np.zeros_like(h, dtype=np.float64)
        indices = np.arange(n_points)
        valid_mask = (indices + m) >= 0
        valid_mask &= (indices + m) < n_points
        valid_indices = indices[valid_mask]

        grad[valid_indices] += (1 - h[valid_indices + m]) * dx
        grad[valid_indices + m] -= h[valid_indices] * dx

        h -= learning_rate * grad
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        delta = target_sum - current_sum
        if delta != 0:
            adjustment = delta / n_points
            h_adjusted = h_clipped + adjustment
            h = np.clip(h_adjusted, 0, 1)

    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return h.tolist(), np.max(overlaps), n_points
