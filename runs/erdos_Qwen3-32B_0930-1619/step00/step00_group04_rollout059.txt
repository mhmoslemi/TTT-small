import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    
    # Use the pre-imported initial_h_values if available
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        # Default initialization pattern
        n_points = 42
        h = 0.5 * np.ones(n_points)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = np.inf
    
    while time() - start_time < budget_s - 1.0:
        # Compute overlaps for all valid shifts
        g = 1.0 - h
        full_corr = np.correlate(h, g, mode='full')
        valid_slices = slice(n_points - 1, 2 * n_points - 1)
        overlaps = full_corr[valid_slices] * dx
        
        # Update best solution if improved
        current_max = np.max(overlaps)
        if current_max < best_c5:
            best_h[:] = h.copy()
            best_c5 = current_max
        
        # Select top 10% of shifts with highest overlap
        n_shifts = n_points
        M = max(2, int(0.1 * n_shifts))
        top_shifts = np.argpartition(overlaps, -M)[-M:]
        
        # Formulate convex approximation problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Create objective to minimize the maximum linearized overlap
        t = cp.Variable()
        for k in top_shifts:
            if k >= n_points:
                continue
            # Calculate valid indices for this shift
            valid_indices = np.arange(n_points - k)
            h_i = h_var[valid_indices]
            h_ki = h_var[valid_indices + k]
            h_t_i = h[valid_indices]
            h_t_ki = h[valid_indices + k]
            
            # Linearize the product terms
            linearized_term = cp.sum(
                h_t_i * h_ki + h_t_ki * h_i - h_t_i * h_t_ki
            ) * dx
            constraints.append(t >= linearized_term)
            
        obj = cp.Minimize(t)
        problem = cp.Problem(obj, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, max_iter=150, verbose=False)
            if problem.status == 'optimal':
                new_h = np.clip(h_var.value, 0, 1)
                # Ensure sum constraint is satisfied
                scaling = (n_points / 2) / (1e-10 + new_h.sum())
                new_h = np.clip(new_h * scaling, 0, 1)
                
                # Update only if improved
                if np.sum((new_h - h)**2) > 1e-10:  # avoid no-change cycles
                    h[:] = new_h
        except Exception:
            pass  # Skip if solve fails
    
    return best_h, best_c5, n_points
