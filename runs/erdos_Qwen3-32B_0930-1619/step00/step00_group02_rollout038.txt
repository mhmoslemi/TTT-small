import numpy as np
import time

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    scale = target_sum / current_sum
    h_scaled = h * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def compute_gradient(h, k_shift, n_points):
    grad = np.zeros(n_points)
    for x in range(n_points):
        x_plus_k = x + k_shift
        if 0 <= x_plus_k < n_points:
            grad[x] += (1 - h[x_plus_k])
        x_minus_k = x - k_shift
        if 0 <= x_minus_k < n_points:
            grad[x] -= h[x_minus_k]
    return grad
