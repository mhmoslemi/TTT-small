    n_points = len(initial_h_values)

    dx = 2.0 / n_points

    def adjust_sum(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if np.isclose(current_sum, target_sum) or np.isclose(current_sum, 0):
            return h_clipped
        delta = target_sum - current_sum
        adjustable_mask = (h_clipped > 0) & (h_clipped < 1)
        adjustable_indices = np.where(adjustable_mask)[0]
        if len(adjustable_indices) == 0:
            return h_clipped
        adjustment = delta / len(adjustable_indices)
        h_clipped[adjustable_indices] += adjustment
        return np.clip(h_clipped, 0, 1)

    def compute_fitness(h_vec):
        h = adjust_sum(h_vec.copy())
        g = 1 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr * dx)
