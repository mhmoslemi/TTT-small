            pos = particles[i]
            vel = velocities[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            new_vel = (inertia * vel +
                       cognitive * r1 * (personal_best[i] - pos) +
                       social * r2 * (global_best_pos - pos))
            new_pos = pos + new_vel
            new_pos = repair(new_pos)
            current_fitness = evaluate_erdos_solution(new_pos)

            if current_fitness < personal_best_fitness[i]:
                personal_best[i] = new_pos.copy()
                personal_best_fitness[i] = current_fitness

            if current_fitness < global_best_fitness:
                global_best_pos[:] = new_pos
                global_best_fitness = current_fitness

            particles[i] = new_pos
            velocities[i] = new_vel

        current_iter += 1

    return (global_best_pos.tolist(), global_best_fitness, n_points)
