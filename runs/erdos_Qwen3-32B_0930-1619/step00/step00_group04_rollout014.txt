import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h_best = initial_h_values.astype(float).copy()
    dx = 2.0 / n_points
    best_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full') * dx)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    options = {'maxiter': 50, 'ftol': 1e-8, 'iprint': 0}
    
    while time.time() - start_time < budget_s:
        # Perturbation and normalization
        rng = np.random.default_rng(seed)
        h0 = h_best + rng.normal(0, 0.05, size=n_points)
        h0 = np.clip(h0, 0, 1)
        h_sum = h0.sum()
        if h_sum != 0:
            h0 *= (n_points / 2.0) / h_sum
        h0 = np.clip(h0, 0, 1)
        
        # Optimization
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraint, options=options)
        if res.success:
            h_new = res.x.reshape(n_points)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                h_best[:] = h_new
                best_c5 = c5_new
        seed += 1
    
    return h_best, best_c5, n_points
