def generate_feasible_h(n, required_sum):
    h = np.random.rand(n)
    current_sum = h.sum()
    delta = required_sum - current_sum
    if abs(delta) < 1e-8:
        return h
    if delta > 0:
        headroom = 1 - h
        indices = np.argsort(headroom)[::-1]
        remaining = delta
        for i in indices:
            if remaining <= 1e-8:
                break
            add = min(remaining, headroom[i])
            h[i] += add
            remaining -= add
    else:

        indices = np.argsort(available)[::-1]

def tournament_selection(population, fitness, tournament_size):
