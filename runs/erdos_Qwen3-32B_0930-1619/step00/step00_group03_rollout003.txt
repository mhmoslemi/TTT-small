        elapsed = current_time - start_time
        if elapsed >= budget_s:
            break
        
        corr = correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-6)[0]
        
        valid_lags = []
        for idx in max_indices:
            lag = idx - (n_points - 1)
            if 1 <= lag <= n_points - 1:
                valid_lags.append(lag)
        
        grad = np.zeros_like(h)
        for k in valid_lags:
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])
                grad[i + k] += (-h[i])
        grad *= dx
        
        if len(valid_lags) > 0:
            grad /= (len(valid_lags) * n_points)
        
        lr = 0.01 * (1 - elapsed / budget_s)
        h = np.clip(h - lr * grad, 0, 1)
        h = project_to_sum(h, target_sum)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h.tolist(), final_c5, n_points
