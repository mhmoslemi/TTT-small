        for indiv in population:
            if np.all((0 <= indiv) & (indiv <= 1)):
                c5.append(evaluate_erdos_solution(indiv))
            else:
                c5.append(float('inf'))
        c5 = np.array(c5)
        
        # Update best
        idx = np.argmin(c5)
        if c5[idx] < best_c5:
            best_c5 = c5[idx]
            best_h = population[idx].copy()
        
        # Tournament selection
        parents = []
        for _ in range(len(population)):
            tournament = random.choices(range(len(population)), k=5)
            winner = min(tournament, key=lambda i: c5[i])
            parents.append(population[winner])
        parents = np.array(parents)
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            if random.random() < 0.8:
                alpha = random.random()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy()
            child = enforce_sum_and_bounds(child)
            # Mutation
            if random.random() < 0.1:
                child += np.random.normal(0, 0.05, size=n_points)
                child = enforce_sum_and_bounds(child)
            offspring.append(child)
        
        # Replace population
        idxs = np.random.choice(range(len(population)), size=len(offspring), replace=False)
        population[idxs] = offspring
    
    return best_h.tolist(), best_c5, n_points
