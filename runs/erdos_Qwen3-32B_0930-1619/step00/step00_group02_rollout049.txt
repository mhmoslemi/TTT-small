import numpy as np
import math
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_remaining = lambda: budget_s - (time.time() - start_time)

    # Determine initial configuration
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
        target_sum = n_points / 2
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-6:
            scaling = target_sum / current_sum
            h = np.clip(h * scaling, 0, 1)
    else:
        n_points = 200
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= target_sum / h.sum()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5

    T = 0.1
    alpha = 0.1
    cooling_rate = 0.999
    no_improvement = 0

    while budget_remaining() > 0:
        # Perturb
        perturbation = np.random.uniform(-alpha*T, alpha*T, size=n_points)
        candidate = current_h + perturbation
        candidate = np.clip(candidate, 0, 1)

        # Projection
        sum_cand = candidate.sum()
        if abs(sum_cand - target_sum) > 1e-6:
            candidate *= target_sum / sum_cand
            candidate = np.clip(candidate, 0, 1)

        # Acceptance
        new_c5 = compute_c5(candidate)
        if new_c5 < current_c5 or np.random.rand() < math.exp((current_c5 - new_c5)/max(T, 1e-6)):
            current_h = candidate.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        # Cool down
        T *= cooling_rate
        if no_improvement > 100:
            break

    return best_h.tolist(), best_c5, n_points
