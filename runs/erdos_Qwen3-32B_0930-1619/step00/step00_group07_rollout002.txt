        return np.max(corr)

    best_c5 = evaluate_C5(h)

        # Compute correlation and choose k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_indices = np.flatnonzero(corr == np.max(corr))
        k_index = np.random.choice(max_indices)
        k_actual = k_index - (n_points - 1)

        # Compute gradient for selected k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_actual
            if 0 <= j < n_points:
                grad[i] += dx * ((1 - h[j]) - h[i])

        # Add TV regularization
        for i in range(1, n_points-1):
