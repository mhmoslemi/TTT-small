import numpy as np
import time

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_1h = 1 - h
    corr = np.correlate(h, h_1h, mode='full')
    return corr.max() * dx

def adjust_sum(h, target_sum, max_iter=100, tol=1e-6):
    n_points = len(h)
    current_sum = h.sum()
    for _ in range(max_iter):
        error = current_sum - target_sum
        if abs(error) < tol:
            break
        adjust = error / n_points
        if error > 0:  # decrease
            h -= adjust
            h = np.clip(h, 0, 1)
        else:  # increase
            h += -adjust
            h = np.clip(h, 0, 1)
        current_sum = h.sum()
    return h
