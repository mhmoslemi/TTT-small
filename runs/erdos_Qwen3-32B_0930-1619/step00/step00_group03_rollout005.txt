from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        penalized_obj,

    if abs(current_sum - target_sum) > 1e-6:
