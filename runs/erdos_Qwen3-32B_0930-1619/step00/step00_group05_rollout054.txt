import numpy as np
import time
import random
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    import initial_h_values
    h_init = np.array(initial_h_values.initial_h_values)
    n_points = len(h_init)
    desired_sum = n_points / 2
    dx = 2.0 / n_points

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            return np.full(n_points, desired_sum / n_points)
        scale = desired_sum / current_sum
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        h_scaled += (desired_sum - h_scaled.sum()) / n_points
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    h_init = repair(h_init)

    pop_size = 30
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 3
    max_generations = 10000  # High limit, actual controlled by time
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_init.copy()

    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair(h)
        population.append(h)

    def evaluate_h(h):
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    while time.time() - start_time < budget_s - 0.1:
        scores = [evaluate_h(h) for h in population]
        idx = np.argmin(scores)
        if scores[idx] < best_c5:
            best_c5 = scores[idx]
            best_h = population[idx].copy()

        new_population = []
        for _ in range(pop_size // 2):
            # Tournament selection
            p1 = min(random.sample(population, tournament_size), key=evaluate_h)
            p2 = min(random.sample(population, tournament_size), key=evaluate_h)

            # Crossover
            if random.random() < crossover_rate:
                split = random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
            else:
                c1, c2 = p1.copy(), p2.copy()

            # Mutation
            for child in [c1, c2]:
                if random.random() < mutation_rate:
                    idx = random.randint(0, n_points - 1)
                    delta = random.uniform(-0.2, 0.2)
                    child[idx] = max(0, min(1, child[idx] + delta))
                new_population.append(repair(child))

        # Elitism and truncation
        population = sorted(population + new_population, key=evaluate_h)[:pop_size]

    return best_h.tolist(), best_c5, n_points
