        h_candidate = current_h.copy()
        for idx in indices:
            delta = np.random.uniform(-epsilon_perturb, epsilon_perturb)
            h_candidate[idx] = max(0.0, min(1.0, h_candidate[idx] + delta))
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_candidate)
        delta_sum = target_sum - current_sum
        if n_points > 0:
            h_candidate += delta_sum / n_points
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
        
        candidate_c5 = compute_c5(h_candidate)
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = h_candidate
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = candidate_c5
        
        # Cool down
        T *= alpha
    
    # Final adjustment to enforce exact sum constraint
    correction = (target_sum - np.sum(best_h)) / n_points
    best_h += correction
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return (best_h, best_c5, n_points)
