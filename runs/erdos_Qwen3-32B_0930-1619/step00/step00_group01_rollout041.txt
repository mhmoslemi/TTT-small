            i = np.random.randint(n_points)
            current_val = h[i]
            candidate_vals = []
            step = 0.1
            for dv in [-2*step, -1*step, 0, step, 2*step]:
                val = current_val + dv
                if 0 <= val <= 1 and val not in candidate_vals:
                    candidate_vals.append(val)
            if not candidate_vals:
                continue

            best_new_h = None
            best_new_c5 = float('inf')
            for val in candidate_vals:
                new_h = h.copy()
                new_h[i] = val
                delta = S - new_h.sum()
                adj = delta / n_points
                new_h += adj
                new_h = np.clip(new_h, 0.0, 1.0)
                c5 = _compute_c5(new_h, dx)
                if c5 < best_new_c5:
                    best_new_c5 = c5
                    best_new_h = new_h
            if best_new_h is not None and best_new_c5 < _compute_c5(h, dx):
                h[:] = best_new_h[:]

            if _compute_c5(h, dx) < best_c5:
                best_c5 = _compute_c5(h, dx)
                best_h[:] = h[:]
    return (best_h.tolist(), best_c5, n_points)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
