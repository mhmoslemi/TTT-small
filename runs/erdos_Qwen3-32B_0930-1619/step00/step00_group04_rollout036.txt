def compute_c5(h):
    dx = 2.0 / len(h)
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def adjust_sum(h, target_sum):
    tol = 1e-8
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < tol:
        return
    interior = np.flatnonzero((h > tol) & (h < 1 - tol))
    if not len(interior):
        return
    adj_per = delta / len(interior)
    h[interior] += adj_per
    np.clip(h, 0, 1, out=h)
