import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h)

    learning_rate = 0.05
    start_time = time.time()

    while time.time() - start_time < budget_s:
        correl = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correl)
        if current_c5 < best_c5:
            best_h = np.copy(h)
            best_c5 = current_c5

        m_idx = np.argmax(correl)
        m = m_idx - n_points + 1

        if not (0 <= m < n_points):
            continue

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + m
            if j < n_points:
                grad[i] += 1 - h[j]
            k = i - m
            if 0 <= k < n_points:
                grad[i] -= h[k]
        grad *= dx

        h_candidate = h - learning_rate * grad
        h_candidate = np.clip(h_candidate, 0, 1)

        s = np.sum(h_candidate)
        if s > 1e-8:
            scale = target_sum / s
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)

        h = h_candidate
