            k_min = max(-h_candidate[i], -(1 - h_candidate[j]))
            if k_min >= k_max:
                continue
            
            k = np.random.uniform(k_min, k_max)
            h_candidate[i] += k
            h_candidate[j] -= k
            h_candidate[i] = np.clip(h_candidate[i], 0, 1)
            h_candidate[j] = np.clip(h_candidate[j], 0, 1)
        
        current_c5 = evaluate_erdos_solution(h)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            delta = candidate_c5 - current_c5
            if delta < 0 and np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
        
        T *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
