import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        bounds=bounds,
        constraints=constraint_sum,
