import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair_individual(h):
        h_clipped = np.clip(h, 0., 1.)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            h_clipped = np.full(n_points, target_sum / n_points)
        else:
            scaling = target_sum / current_sum
            h_clipped *= scaling
            h_clipped = np.clip(h_clipped, 0., 1.)
        return h_clipped

    def evaluate_individual(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 30
    population = [repair_individual(np.random.rand(n_points)) for _ in range(population_size)]
    population[0] = initial_h_values.copy()

    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 1:
        children = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(population_size, 2)
            parent1, parent2 = population[p1], population[p2]

            if np.random.rand() < 0.8:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()

            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise

            child = repair_individual(child)
            children.append(child)

        all_candidates = population + children
        all_evals = [evaluate_individual(h) for h in all_candidates]
        best_indices = np.argsort(all_evals)[:population_size]
        population = [all_candidates[i] for i in best_indices]

        current_best_eval = min(all_evals)
        if current_best_eval < best_c5:
            best_c5 = current_best_eval
            best_h = population[0].copy()

    return best_h.tolist(), float(best_c5), n_points
