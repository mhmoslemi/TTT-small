import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    h_initial = initial_h_values

    h = cp.Variable(n_points)
    C = cp.Variable()

        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1,

        terms = [h[i] * (1 - h[i + k]) for i in indices]

    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)

    h.value = h_initial
    start_time = time.time()
    try:

            verbose=False,
            max_iter=1000,
            ipopt_options={"max_cpu_time": budget_s * 0.95},
        )
    except Exception:
        pass

    h_values = h.value if h.value is not None else h_initial
    cross_corr = np.correlate(h_values, 1 - h_values, mode="full")
    c5_bound = np.max(cross_corr) * dx

    return (h_values, c5_bound, n_points)
