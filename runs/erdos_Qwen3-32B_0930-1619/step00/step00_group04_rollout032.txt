        while steps_done < steps_per_T and time.time() - start_time < budget_s:
            candidate = perturb_h(current_h, n_points)
            candidate_c5 = compute_c5(candidate, n_points, dx)
            delta = candidate_c5 - best_c5
            if delta < 0:
                current_h[:] = candidate
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h[:] = candidate
            else:
                if np.random.rand() < np.exp(-delta / T):
                    current_h[:] = candidate
            steps_done += 1
        T *= alpha

    return (best_h, best_c5, n_points)
