import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points

    # Enforce initial constraints
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if current_sum > 1e-10:
        h *= (n_points / 2) / current_sum
    h = np.clip(h, 0.0, 1.0)

    # ADAM parameters
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100

    def compute_c5(h_eval):
        c_corr = np.correlate(h_eval, 1 - h_eval, mode='full') * dx
        return np.max(c_corr)

    best_c5 = compute_c5(h)

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        t += 1

        # Compute cross-correlation and max index
        c_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(c_corr)

        # Determine lag k
        k = max_index - (n_points - 1)

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx  # Negative term for (1-h) component

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # Apply step and project to feasible region
        h -= step
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if current_sum > 1e-10:
            scaling = (n_points / 2) / current_sum
            h *= scaling
        h = np.clip(h, 0.0, 1.0)
