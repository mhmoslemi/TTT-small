import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce sum constraint and bounds
    current_sum = h.sum()
    if not np.isclose(current_sum, target_sum):
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            corr = (target_sum - current_sum) / n_points
            h += corr
            h = np.clip(h, 0.0, 1.0)

    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        one_minus_x = 1 - x
        corr = np.correlate(x, one_minus_x, mode='full')
        return np.max(corr * dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    # Main optimization loop with time tracking
    while time.time() - start_time < budget_s - 10:  # 10s buffer
        result = scipy.optimize.minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100}
        )
        if result.success:
            x0 = np.clip(result.x, 0.0, 1.0)
            # Re-normalize after optimization
            current_sum = x0.sum()
            if not np.isclose(current_sum, target_sum):
                scale = target_sum / current_sum
                x0 = x0 * scale
                x0 = np.clip(x0, 0.0, 1.0)
        else:
            break  # Exit if stuck

    # Final validation and evaluation
    best_h = x0
    c5_bound = evaluate_erdos_solution(best_h)
    return (best_h, c5_bound, n_points)
