import numpy as np

def run(seed=42, budget_s=1100):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        target = n_points / 2.0
        if abs(total) < 1e-10:
            return np.full_like(h, 0.5)
        scale = target / total
        h = h * scale
        h = np.clip(h, 0, 1)
        adjustment = (target - np.sum(h)) / n_points
        h += adjustment
        return np.clip(h, 0, 1)

    h_initial = repair(h_initial)

    population_size = 40
    np.random.seed(seed)
    population = [h_initial]
    while len(population) < population_size:
        candidate = np.random.rand(n_points)
        candidate = repair(candidate)
        population.append(candidate)
    population = np.array(population)

    def evaluate(ind):
        cross_corr = np.correlate(ind, 1 - ind, mode='full')
        return np.max(cross_corr) * dx

    best_idx = 0
    best_fitness = evaluate(population[0])
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        fitnesses = np.array([evaluate(ind) for ind in population])
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_idx = current_best_idx

        new_population = []
        for _ in range(population_size // 2):
            def tournament():
                indices = np.random.choice(population_size, 3)
                return indices[np.argmin(fitnesses[indices])]

            p1_idx, p2_idx = tournament(), tournament()
            parent1 = population[p1_idx].copy()
            parent2 = population[p2_idx].copy()

            if np.random.rand() < 0.9:
