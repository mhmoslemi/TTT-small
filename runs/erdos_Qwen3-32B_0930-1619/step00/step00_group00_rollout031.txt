import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    try:
        h = initial_h_values.copy()
    except NameError:

        h = np.zeros(n_points)

        np.random.shuffle(h)
    else:
        n_points = len(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    T = 1000.0
    cooling_rate = 0.995
    
    while time.time() - start_time < budget_s - 1e-3:
        h_candidate = h.copy()
        perturbation_size = int(n_points * 0.05)
        indices = np.random.choice(n_points, size=perturbation_size, replace=False)
        subset = h_candidate[indices]
        n_zero = np.sum(subset == 0)
        n_one = len(subset) - n_zero
        m = min(n_zero, n_one)
        if m > 0:
            zero_idx = np.where(subset == 0)[0]
            one_idx = np.where(subset == 1)[0]

                swap_zero = np.random.choice(zero_idx, m, replace=False)
                swap_one = np.random.choice(one_idx, m, replace=False)
                h_candidate[indices[swap_zero]] = 1
                h_candidate[indices[swap_one]] = 0
        
        current_c5 = compute_c5(h, dx)
        new_c5 = compute_c5(h_candidate, dx)

    return best_h, c5_bound, n_points
