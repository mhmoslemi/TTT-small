def compute_c5(h, n_points):

def project_to_feasible(h, n_points):

    dx = 2.0 / n_points

        h *= 1.0 / sum_h
        h = np.clip(h, 0, 1)

def optimize_grid(n_points, h_init, remaining_time):

        return np.max(corr)

        return np.sum(h_flat) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
