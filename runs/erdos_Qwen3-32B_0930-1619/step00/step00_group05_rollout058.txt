                grad[i - m_lag] -= h[i]

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= alpha * m_hat / (np.sqrt(v_hat) + eps)

        # Project back to constraints
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != h.size:
            scaling = h.size / h_sum
            h *= scaling

        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= decay_steps:
            alpha *= lr_decay
            no_improvement = 0

    return best_h, best_c5, n_points

def compute_c5(h, dx):
    s = 1.0 - h
    c = np.correlate(h, s, mode='full')
    return np.max(c) * dx
