import numpy as np
import time

def run(seed=42, budget_s=1100):
    h = np.array(initial_h_values, dtype=np.float64).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0
    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()

    for iter_idx in range(5000):
        if time.time() - start_time > budget_s:
            break
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full') * dx
        current_max = corr.max()
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        k_max = np.argmax(corr)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = (i - k_max) % n_points
            grad[i] = ones_minus_h[idx] - h[i]
        
        gamma = 2.0 / (iter_idx + 2)
        h_new = h - gamma * grad
        h_clipped = np.clip(h_new, 0.0, 1.0)
        current_sum = h_clipped.sum()
        
        if abs(current_sum - T) > 1e-6:
            delta = T - current_sum
            non_clamped = np.where((h_clipped > 1e-8) & (h_clipped < 1 - 1e-8))[0]
            if len(non_clamped) > 0:
                amount_per = delta / len(non_clamped)
                h_clipped[non_clamped] += amount_per
                h_clipped = np.clip(h_clipped, 0.0, 1.0)
        h = h_clipped

    # Final evaluation
    ones_minus_best = 1.0 - best_h
    corr_final = np.correlate(best_h, ones_minus_best, mode='full') * dx
    final_c5 = corr_final.max()
    return best_h.tolist(), final_c5, n_points
