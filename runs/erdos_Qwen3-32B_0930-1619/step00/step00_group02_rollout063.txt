    h_repaired = np.clip(h, 0, 1)
    # Compute current sum after clipping
    sum_h = np.sum(h_repaired)

    delta = S - sum_h

    if abs(delta) < 1e-8:
        return h_repaired
    # Otherwise, adjust some elements
    # Find indices where h_repaired can be increased (if delta positive) or decreased (if delta negative)
    # For simplicity, let's spread the delta over all elements that are not at the bounds
    # This is a heuristic
    mask = (h_repaired < 1 - 1e-8) & (h_repaired > 1e-8)

    alpha = delta / mask.sum()
