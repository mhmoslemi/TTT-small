        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        noise = np.random.uniform(-0.05, 0.05, size=num_perturb)
        h_candidate[perturb_indices] += noise
        h_candidate = np.clip(h_candidate, 0.0, 1.0)

        current_sum = h_candidate.sum()
        delta = target_sum - current_sum
        if abs(delta) > 1e-8:
            if delta > 0:
                idx = np.where((h_candidate < 1.0 - 1e-6) & (h_candidate > 1e-6))[0]
                add_per = delta / len(idx)
                h_candidate[idx] += add_per
                h_candidate[idx] = np.clip(h_candidate[idx], 0.0, 1.0)
            else:
                idx = np.where((h_candidate > 0.0) & (h_candidate < 1.0 - 1e-6))[0]
                sub_per = (-delta) / len(idx)
                h_candidate[idx] -= sub_per
                h_candidate[idx] = np.clip(h_candidate[idx], 0.0, 1.0)

        def compute_c5(h):
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        current_c = compute_c5(h)
        candidate_c = compute_c5(h_candidate)

        if candidate_c < current_c or np.random.rand() < np.exp((current_c - candidate_c) / temperature):
            h = h_candidate.copy()
            current_c = candidate_c

        if current_c < best_c5:
            best_h = h.copy()
            best_c5 = current_c

        temperature *= cooling_rate

    c5_bound = evaluate_erdos_solution(best_h)
    return best_h, c5_bound, n_points
