import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    alpha = 0.1
    tv_lambda = 0.01
    max_iter = 10000
    np.random.seed(seed)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Find max C5 and corresponding k
        max_c5 = np.max(corr)
        idx = np.argmax(corr)
        k_max = idx - (n_points - 1)

        # Compute gradient from correlation
        grad_corr = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad_corr[i] += (1.0 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad_corr[i] -= h[i - k_max] * dx

        # Compute TV regularization gradient
        grad_tv = np.zeros(n_points)
        for i in range(1, n_points - 1):
            grad_tv[i] = h[i - 1] - 2 * h[i] + h[i + 1]
        grad_tv[0] = h[1] - h[0]
        grad_tv[-1] = h[-1] - h[-2]

        # Total gradient
        grad_total = grad_corr + tv_lambda * grad_tv

        # Update step
        h_new = h - alpha * grad_total

        # Project to feasible region
        h_new = np.clip(h_new, 0.0, 1.0)
        current_sum = np.sum(h_new)
        if abs(current_sum) > 1e-12:
            h_new = h_new * (target_sum / current_sum)
        h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new

        # Update best solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
