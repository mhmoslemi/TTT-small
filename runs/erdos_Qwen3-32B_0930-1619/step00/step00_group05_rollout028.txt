import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_parent = initial_h_values
    n_points = len(h_parent)
    dx = 2.0 / n_points

    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum == 0:
            return h
        target_sum = n_points / 2.0
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h

    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        return np.max(corr) * dx

    h_global = repair(h_parent)
    c5_global = compute_c5(h_global)

    n_particles = 30
    positions = np.zeros((n_particles, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    pbest_positions = np.zeros((n_particles, n_points))
    pbest_costs = np.zeros(n_particles)

    # Initialize positions
    positions[0] = h_global
    pbest_positions[0] = h_global.copy()
    pbest_costs[0] = c5_global
    for i in range(1, n_particles):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        pos = repair(h_global + noise)
        positions[i] = pos
        cost = compute_c5(pos)
        pbest_positions[i] = pos.copy()
        pbest_costs[i] = cost

    gbest_idx = np.argmin(pbest_costs)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_cost = pbest_costs[gbest_idx]
    no_improve = 0
    w_start, w_end = 0.9, 0.4

    start_time = time.time()
