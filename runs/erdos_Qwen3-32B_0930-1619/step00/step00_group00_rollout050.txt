import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    global initial_h_values
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    constraints = ({'type': 'eq', 'fun': constraint_sum},)
    bounds = [(0.0, 1.0)] * n_points

    options = {
        'maxiter': 1000,
        'ftol': 1e-8,
        'disp': False,
        'iprint': 0,
    }

    def time_check(xk):
        if time.time() - start_time >= budget_s:
            raise ValueError("Time limit exceeded")

    options['callback'] = time_check

    result = None
    try:
        result = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=options,
        )
    except ValueError:
        pass  # Time limit exceeded, use result from last iteration

    best_h = np.clip(result.x if result is not None else h_initial, 0.0, 1.0)
    c5_bound = evaluate_erdos_solution(n_points, best_h)

    return (best_h.tolist(), c5_bound, n_points)
