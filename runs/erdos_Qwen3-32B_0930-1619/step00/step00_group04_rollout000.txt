        delta = np.random.uniform(0, max_delta)
        if direction == 'i_to_j':
            h_new[i] -= delta
            h_new[j] += delta
        else:
            h_new[j] -= delta
            h_new[i] += delta
        
        # Clip to numerical stability
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new)
        delta_c5 = new_c5 - best_c5
        
        # Acceptance criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_new
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        
        # Temperature update
        temperature *= alpha
    
    return (best_h, best_c5, n_points)
