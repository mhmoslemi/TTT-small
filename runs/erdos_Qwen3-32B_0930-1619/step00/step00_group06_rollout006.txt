import numpy as np
from scipy.optimize import minimize

def evaluate_C5(h, n_points, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    h_init = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Ensure initial values sum exactly to required integral
    h_init = h_init.copy()
    h_init_sum = h_init.sum()
    if not np.isclose(h_init_sum, n_points / 2):
        scale = (n_points / 2) / h_init_sum
        h_init = np.clip(h_init * scale, 0, 1)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constr = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Objective function
    def objective_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)
    
    result = minimize(
        objective_func,
        h_init,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=sum_constr,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    # Post-process result
    best_h = np.clip(result.x, 0, 1)
    best_h = best_h.astype(np.float64)
    
    # Final constraint enforcement
    correction = (n_points / 2 - best_h.sum()) / n_points
    best_h += correction
    best_h = np.clip(best_h, 0, 1)
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s * 0.95:
        print("Early termination due to time constraints")
    
    c5_value = evaluate_C5(best_h, n_points, dx)
    return (best_h.tolist(), c5_value, n_points)
