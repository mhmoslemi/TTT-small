import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = np.array(initial_h_values, dtype=float)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full') * dx)
    start_time = time.time()

    # Phase 1: Simulated Annealing
    temp, alpha, perturb_ratio = 0.05, 0.995, 0.1
    n_perturb = max(1, int(n_points * perturb_ratio))
    while time.time() - start_time < budget_s * 0.7:
        h_new = h.copy()
        idxs = np.random.choice(n_points, n_perturb, replace=False)
        for i in idxs:
            delta = np.random.uniform(-temp, temp)
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new += (required_sum - h_new.sum()) / n_points
        h_new = np.clip(h_new, 0.0, 1.0)
        current_c5 = np.max(np.correlate(h, 1-h, mode='full') * dx)
        new_c5 = np.max(np.correlate(h_new, 1-h_new, mode='full') * dx)
        if new_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / max(temp, 1e-8)):
            h = h_new
        temp *= alpha

    # Phase 2: Gradient Descent
    lr, decay = 0.05, 0.99
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1-h, mode='full') * dx
        k_idx = np.argmax(corr)
        k_max = k_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            ipk = i + k_max
            imk = i - k_max
            term1 = (1 - h[ipk]) if 0 <= ipk < n_points else 0
