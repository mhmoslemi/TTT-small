        m = max_idx - (n_points - 1)  # Shift in array indices
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
        
        # Update h with gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        sum_new = h_new.sum()
        if abs(sum_new) > 1e-10:
            h_new = h_new * (n_points / 2) / sum_new
        
        # Update h if feasible
        h = h_new
        
        # Track improvements
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            counter = 0
        else:
            counter += 1
        
        # Adjust alpha
        if counter >= 50:
            alpha *= 0.8
            counter = 0
        if alpha < 1e-6:
            break
    
    return best_h.tolist(), best_c5, n_points
