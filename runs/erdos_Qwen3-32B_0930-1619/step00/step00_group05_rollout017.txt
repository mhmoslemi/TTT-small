    def project_h(h_vals, target_sum):
        h_clamped = np.clip(h_vals, 0, 1)
        current_sum = h_clamped.sum()
        if current_sum < 1e-10:
            return h_vals.copy()
        scale = target_sum / current_sum
        h_scaled = h_clamped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    h = project_h(h, target_sum)

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
