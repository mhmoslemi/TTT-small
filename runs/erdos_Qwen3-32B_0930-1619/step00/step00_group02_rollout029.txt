import numpy as np
from time import time

def repair_sum(h, n_points):
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if np.allclose(current_sum, target_sum):
        return h
    delta = target_sum - current_sum
    h[0] += delta
    return np.clip(h, 0, 1)

def perturb_h(h, mutation_rate=0.1, noise_std=0.05):
    h = h.copy()
    mutation_mask = np.random.rand(len(h)) < mutation_rate
    h[mutation_mask] += np.random.normal(0, noise_std, size=np.sum(mutation_mask))
    h = np.clip(h, 0, 1)
    return repair_sum(h, len(h))

def blend_crossover(parent1, parent2):
    alpha = np.random.rand()
    offspring = alpha * parent1 + (1 - alpha) * parent2
    return repair_sum(np.clip(offspring, 0, 1), len(offspring))

def tournament_selection(population, fitnesses, tournament_size=5):
    idx = np.random.randint(len(population), size=tournament_size)
    best_idx = idx[np.argmin(fitnesses[idx])]
    return population[best_idx].copy()

def evaluate_individual(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
