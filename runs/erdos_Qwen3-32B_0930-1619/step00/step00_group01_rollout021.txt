import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values, dtype=np.float64).copy()
    dx = 2.0 / n_points

    def objective_func(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def objective_grad(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        if k >= 0:
            valid_i = range(n_points - k)
        else:
            valid_i = range(-k, n_points)
        for i in valid_i:
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        return grad.flatten()

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(lb=0.0, ub=1.0)

    result = minimize(
        objective_func,
        initial_h,
        jac=objective_grad,
        constraints=sum_constraint,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-6}
    )

    best_h = result.x
    best_c5 = objective_func(best_h)

    # Try COBYLA with remaining time if available
    remaining_time = max(0, budget_s - (time() - start_time))
    if remaining_time > 10:
        result_extra = minimize(
            objective_func,
            best_h,
            constraints=sum_constraint,
            bounds=bounds,
            method='COBYLA',
            options={'maxiter': int(remaining_time / 10) + 1}
        )
        if result_extra.success:
            new_c5 = objective_func(result_extra.x)
            if new_c5 < best_c5:
                best_h = result_extra.x
                best_c5 = new_c5

    best_h = best_h.reshape(n_points)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-8), "Sum constraint violated"

    return (best_h.tolist(), best_c5, n_points)
