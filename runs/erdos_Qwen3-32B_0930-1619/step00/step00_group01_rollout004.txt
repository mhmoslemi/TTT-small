import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, n_points=47):
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full') * dx
        return np.max(corr)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    timeout = 0.9 * budget_s
    
    def callback(xk):
        nonlocal best_c5, best_h
        current_val = objective(xk)
        if current_val < best_c5:
            best_c5 = current_val
            best_h[:] = xk.copy()
        return False
    
    try:
        while time.time() - start_time < timeout:
            result = scipy.optimize.minimize(
                objective,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                callback=callback,

            )
            if result.success:
                best_h = result.x
                best_c5 = objective(best_h)
    except Exception:
        pass  # Gracefully handle interruptions
    
    return best_h, best_c5, n_points
