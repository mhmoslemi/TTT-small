import time
from scipy.interpolate import interp1d

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    return np.max(corr)

    h_clipped = np.clip(h_candidate, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    target_sum = n_points / 2.0
    if current_sum == 0.0:

    else:
        scaling_factor = target_sum / current_sum
        h_scaled = h_clipped * scaling_factor
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
    return h_clipped

        # Acceptance criteria
        if new_c5 < current_c5:
            current_h = h_projected.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:

            prob = np.exp(delta_energy / T)
            if np.random.rand() < prob:
                current_h = h_projected.copy()
                current_c5 = new_c5
            else:
                no_improvement_count += 1

        # Cool temperature
        T *= cooling_rate

        # Early stopping
        if no_improvement_count > max_no_improvement:
            break

    # Final check
    final_c5 = compute_c5(best_h, n_points)
    return best_h, final_c5, n_points
