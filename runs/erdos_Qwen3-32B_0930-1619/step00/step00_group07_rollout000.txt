    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 0.1:
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(cross_corr)
        current_c5 = cross_corr[max_index]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Determine lag k_max
        k_max = max_index - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k_max < n_points:
                term1 += (1.0 - h[i + k_max])
            term2 = 0.0
            if 0 <= i - k_max < n_points:
                term2 -= h[i - k_max]
            grad[i] += (term1 + term2) * dx

        # TV regularization
        for i in range(1, n_points - 1):
            grad[i] += lambda_tv * (h[i-1] - 2*h[i] + h[i+1])

        # Gradient descent step
        h -= learning_rate * grad
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Project to target sum
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
            h *= target_sum / current_sum
            h = np.clip(h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
