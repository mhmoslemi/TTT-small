import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = cp.Variable(n_points, nonneg=True)
    C = cp.Variable(nonneg=True)
    constraints = []
    constraints.append(cp.sum(h) == n_points / 2.0)
    constraints.append(h <= 1)
    constraints.append(h >= 0)
    
    for k in range(-n_points + 1, n_points):
        i_start = max(0, -k)
        i_end = min(n_points - 1, n_points - 1 - k)
        if i_start > i_end:
            continue
        terms = []
        for i in range(i_start, i_end + 1):
            j = i + k
            terms.append(h[i] * (1 - h[j]))
        sum_terms = cp.sum(terms) * dx
        constraints.append(sum_terms <= C)
    
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)
    
    # Warm start initialization
    h_init = np.array(initial_h_values).copy()
    total_h = np.sum(h_init)
    if abs(total_h - n_points / 2) > 1e-6:
        h_init = h_init * (n_points / 2) / total_h
    h.value = h_init
    
    try:
        problem.solve(solver=cp.IPOPT, max_iter=1000, verbose=False,
                      ipopt_options={"max_time": 900, "print_level": 0})
    except:
        pass  # Fall back to initial if solver fails
    
    h_values = h.value
    overlap = np.correlate(h_values, 1 - h_values, mode="full") * dx
    return (np.clip(h_values, 0, 1).tolist(), float(np.max(overlap)), n_points)
