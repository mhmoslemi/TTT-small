import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()

    def objective(x):
        x = x.reshape(n_points)
        corr = np.correlate(x, 1 - x, mode='full')
        valid_corr = corr[n_points : 2 * n_points - 1]
        return np.max(valid_corr) * dx

    bounds = tuple((0.0, 1.0) for _ in range(n_points))
    linear_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=linear_constraint,
        options={'maxiter': 1000}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        print(f"Time budget exceeded ({elapsed:.1f}s of {budget_s}s used). Returning best found so far.")
        h_opt = result.x
    else:
        h_opt = result.x if result.success else h

    final_c5 = evaluate_erdos_solution(h_opt)
    return h_opt, final_c5, n_points
