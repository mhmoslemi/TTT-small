def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values).astype(float)
    best_h = np.copy(h)
    best_c5 = evaluate_c5(h, n_points)
    
    max_iterations = 100000
    iteration = 0
    
    while time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        improved = False
        for i in range(n_points):
            # Sample candidates
            candidates = np.linspace(0, 1, 7)
            current_value = h[i]
            total_sum = np.sum(h)
            
            best_new = None
            min_c5 = evaluate_c5(h, n_points)
            
            for c in candidates:
                if np.isclose(c, current_value):
                    continue
                trial = np.copy(h)
                trial[i] = c
                trial_sum = np.sum(trial)
                if trial_sum != 0:
                    trial = trial * (n_points / 2) / trial_sum
                trial = np.clip(trial, 0, 1)
                c5 = evaluate_c5(trial, n_points)
                if c5 < min_c5:
                    min_c5 = c5
                    best_new = trial
            if best_new is not None and evaluate_c5(best_new, n_points) < evaluate_c5(h, n_points):
                h[:] = best_new
                improved = True
                if min_c5 < best_c5:
                    best_h[:] = h
                    best_c5 = min_c5
        
        if not improved:
            break
    
    c5_bound = best_c5
    return (best_h, c5_bound, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)
