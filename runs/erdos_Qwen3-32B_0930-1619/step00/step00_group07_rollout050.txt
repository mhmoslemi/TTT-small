def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.astype(float)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    alpha = 0.1
    lambda_tv = 0.01
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    max_iterations = 10000
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s * 0.99:
            break

        # Compute correlation and find k_max
        full_corr = np.correlate(h, 1 - h, mode='full')
        corr_values = full_corr * dx
        max_idx = np.argmax(corr_values)
        k_max = max_idx - (n_points - 1)

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx

        # Add TV regularization gradient
        for i in range(1, n_points - 1):
            grad[i] += lambda_tv * (h[i-1] - 2*h[i] + h[i+1])

        # Update step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)

        # Projection to maintain integral constraint
        current_sum = h_new.sum()
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum
            h_new = np.clip(h_new * scale, 0, 1)

        h = h_new

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
