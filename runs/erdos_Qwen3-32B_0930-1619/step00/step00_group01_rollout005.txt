import cvxpy as cp
import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 47  # default, can be adjusted in kwargs if needed

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_points = len(initial_h_values)
    else:
        raise ValueError("Initial h values required")

    # Create variables
    h = cp.Variable(n_points, nonneg=True)
    C = cp.Variable(nonnegative=True)
    dx = 2.0 / n_points

    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]

    # Add overlap constraints for each shift k
    for k in range(-(n_points - 1), n_points):
        valid_i = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_i.append(i)
        if valid_i:
            sum_expr = 0
            for i in valid_i:
                j = i + k
                sum_expr += h[i] * (1 - h[j])
            constraints.append(sum_expr * dx <= C)

    # Set initial guess if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h.value = np.array(initial_h_values)
    else:
        h.value = np.ones(n_points) * 0.5  # Feasible start

    # Objective is to minimize C
    problem = cp.Problem(cp.Minimize(C), constraints)
    
    # Try MOSEK first with time limit
    if 'MOSEK' in cp.installed_solvers():
        time_remaining = max(0, budget_s - (time.time() - start_time) - 10)
        problem.solve(solver=cp.MOSEK, mosek_params={b"MSK_DPAR_OPTIMIZER_TIME": max(1, time_remaining)},
                      verbose=True)
    else:
        # Fall back to ECOS
        problem.solve(solver=cp.ECOS, verbose=True)

    elapsed_time = time.time() - start_time

    # Extract solution
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
    else:
        # Fallback to feasible initial if optimization failed
        if 'initial_h_values' in globals():
            h_opt = np.array(initial_h_values)
        else:
            raise RuntimeError("No feasible solution found")

    # Compute final bound via evaluator
    from evaluate_erdos_solution import evaluate_erdos_solution
    c5_bound = evaluate_erdos_solution(h_opt, n_points)
    return h_opt, c5_bound, n_points
