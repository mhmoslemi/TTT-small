def evaluate_c5(h, dx):
    corrs = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corrs)

def repair_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0.0:
        return h_clipped.copy()
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
