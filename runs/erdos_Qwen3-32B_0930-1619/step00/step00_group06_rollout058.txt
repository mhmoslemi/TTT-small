import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0

    def compute_c5(h):
        ones_minus_h = 1.0 - h
        correlations = np.correlate(h, ones_minus_h, mode='full')
        return np.max(correlations) * dx

    np.random.seed(seed)
    max_iter = 2000
    tolerance = 1e-5
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        ones_minus_h = 1.0 - h
        correlations = np.correlate(h, ones_minus_h, mode='full')
        max_corr = np.max(correlations)
        idxs = np.where(correlations >= max_corr - 1e-6)[0]

        grad = np.zeros_like(h)
        for idx in idxs:
            if n_points - 1 <= idx <= 2 * n_points - 2:
                k_shift = idx - (n_points - 1)
                if k_shift == 0:
                    grad_k = (1.0 - h) - h
                else:
                    n_valid = n_points - k_shift
                    grad_k = np.zeros_like(h)
                    grad_k[:n_valid] = (1.0 - h[k_shift:k_shift + n_valid]) - h[:n_valid]
                grad += grad_k

        v = np.where(grad > 0, 0.0, 1.0)
        gamma = 2.0 / (it + 100)
        h_new = (1 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0.0, 1.0)

        current_sum = h_new.sum()
        delta = desired_sum - current_sum
        if delta != 0:
            adjustment = delta / n_points
            if delta > 0:
                h_new += adjustment
                h_new = np.minimum(h_new, 1.0)
            else:
                h_new -= adjustment
                h_new = np.maximum(h_new, 0.0)

        h = h_new

        if abs(compute_c5(h) - current_c5) < tolerance:
            break

    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_points
