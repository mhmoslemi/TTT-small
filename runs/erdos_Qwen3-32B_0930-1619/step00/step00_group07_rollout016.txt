import numpy as np
import time

# These functions are pre-imported by the harness:
# - evaluate_erdos_solution
# - initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.copy(initial_h_values)
    n_points = h.shape[0]
    dx = 2.0 / n_points
    alpha = 0.1
    lambda_tv = 0.01
    
    # Initialize best result tracking
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    
    while time.time() - start_time < budget_s:
        correlate_full = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(correlate_full)
        k_shift = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        
        if k_shift > 0:
            # Forward part (1 - h[i + k_shift])
            sl1 = slice(0, n_points - k_shift)
            sl2 = slice(k_shift, n_points)
            grad[sl1] += 1.0 - h[sl2]
            # Backward part (-h[i - k_shift])
            sl1 = slice(k_shift, n_points)
            sl2 = slice(0, n_points - k_shift)
            grad[sl1] += -h[sl2]
        elif k_shift < 0:
            k_shift_abs = -k_shift
            # Forward part (1 - h[i + k_shift] where i >=k_shift_abs)
            sl1 = slice(k_shift_abs, n_points)
            grad[sl1] += 1.0 - h[sl1]
            # Backward part (i < (n-k_shift_abs), add -h[i + k_shift_abs])
            sl1 = slice(0, n_points - k_shift_abs)
            sl2 = slice(k_shift_abs, k_shift_abs + n_points - k_shift_abs)
            grad[sl1] += -h[sl2]
        else:
            # k_shift = 0, gradient is (1 - h) - h
            grad += (1.0 - h) - h
        
        grad *= dx
        
        # Add TV regularization
        if n_points > 2:
            grad[1:-1] += lambda_tv * (h[:-2] - 2*h[1:-1] + h[2:])
        
        # Gradient update
        h -= alpha * grad
        
        # Projection: clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Enforce integral constraint
        target_sum = n_points / 2
        current_sum = h.sum()
        delta = (target_sum - current_sum) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        
        # Track best solution
        current_correlate = np.correlate(h, 1-h, mode='full')
        current_c5 = np.max(current_correlate) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
    return best_h, best_c5, n_points
