import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize the Erdős minimum overlap problem to find a step function h that minimizes C₅."""
    # Use provided initial_h_values
    h_start = np.array(initial_h_values)
    n_points = len(h_start)
    assert np.isclose(np.sum(h_start), n_points / 2), "Initial h is not feasible"
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    constraint_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    
    # Define objective function
    def objective(h_vec):
        dx = 2.0 / n_points
        h = np.array(h_vec)
        overlaps = []
        for k in range(1, n_points):
            slice1 = h[:n_points - k]
            slice2 = h[k:]
            overlap = np.dot(slice1, 1.0 - slice2) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Track best solution
    best_h = h_start.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run optimization
    result = minimize(
        objective,
        h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_sum,
        options={'maxiter': 200, 'ftol': 1e-6}
    )
    
    if result.success:
        improved_h = np.clip(result.x, 0.0, 1.0)
        # Rescale to ensure exact integrality
        diff = n_points / 2 - np.sum(improved_h)
        indices = np.argsort(np.abs(improved_h - 0.5))[::-1]
        for i in indices[:abs(int(np.round(diff * 10)))]:
            improved_h[i] += np.sign(diff) / 10
            improved_h[i] = np.clip(improved_h[i], 0.0, 1.0)
        improved_h = np.clip(improved_h, 0.0, 1.0)
        improved_c5 = objective(improved_h)
        if improved_c5 < best_c5:
            best_h = improved_h
            best_c5 = improved_c5
            
    end_time = time.time()
    elapsed = end_time - start_time
    if elapsed < budget_s:

                best_c5 = improved_c5
