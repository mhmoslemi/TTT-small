def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    delta = (target_sum - current_sum) / len(h)
    h_proj = h_clipped + delta
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

    lr = initial_lr

        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(corr)
        max_indices = np.where(np.isclose(corr, max_val, atol=1e-8))[0]
        shifts = [j - (n_points - 1) for j in max_indices]
