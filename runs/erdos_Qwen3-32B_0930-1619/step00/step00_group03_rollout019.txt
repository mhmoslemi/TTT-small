    n_points = len(initial_h_values)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Function to compute overlaps
    def compute_overlaps(h):
        ones_minus_h = 1.0 - h
        overlaps = np.correlate(h, ones_minus_h, mode='full') * dx
        return overlaps

    # Function to compute gradients for given shifts
    def compute_gradients_for_shifts(h, shifts, dx):
        n = h.shape[0]
        grad = np.zeros(n)
        for k_full in shifts:
            s = k_full - (n - 1)  # compute the shift in array indices
            if s >=0:
                valid_indices = np.arange(n - s)
            else:
                valid_indices = np.arange(-s, n)
            for i in valid_indices:
                j = i + s
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        return grad.mean(axis=0)  # average over the shifts

    # Projection function
    def project_to_feasible(h, target_sum):
        # Clip to [0, 1]
        h_proj = np.clip(h, 0.0, 1.0)
        current_sum = h_proj.sum()
        if np.abs(current_sum - target_sum) < 1e-6:
            return h_proj
        # Adjust the sum by adding a small value to all elements
        delta = target_sum - current_sum
        h_proj += delta / n_points
        # Clip again to avoid violating bounds
        return np.clip(h_proj, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = np.inf

    start_time = time.time()
    max_iterations = 10000  # Attempt a large number, but constrained by time
    alpha = 0.01  # Initial learning rate
    alpha_decay = 0.995  # Decay per iteration

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        overlaps = compute_overlaps(h)
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find all shifts k_full with max overlap
        max_overlap = np.max(overlaps)
        max_mask = np.isclose(overlaps, max_overlap)
        max_shifts = np.where(max_mask)[0]

        # Compute gradient for these shifts
        grad = compute_gradients_for_shifts(h, max_shifts, dx)

        # Update h with gradient descent
        h -= alpha * grad

        # Project back to feasible region
        h = project_to_feasible(h, target_sum)

        # Decay learning rate
        alpha *= alpha_decay
