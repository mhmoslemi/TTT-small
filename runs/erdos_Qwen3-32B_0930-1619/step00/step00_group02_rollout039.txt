    h_proj = np.clip(h, 0, 1)

    s_current = np.sum(h_proj)

    s_target = n_points / 2.0

import numpy as np
import time

def project(h, n_points):
    h_proj = np.clip(h, 0, 1)
    s_current = np.sum(h_proj)
    s_target = n_points / 2.0
    adjustment = (s_target - s_current) / n_points
    h_proj += adjustment
    h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
