    h = np.array(initial_h_values, copy=True).astype(np.float64)
    n_points = len(h)

    start_time = time()
    
    learning_rate = 0.01
    max_iter = 1000
    
    for _ in range(max_iter):
        if time() - start_time > budget_s:
            break
        
        g = 1.0 - h
        overlaps = []
        for m in range(n_points):
            if m == 0:
                overlap = np.sum(h * g) * dx
            else:
                valid = n_points - m
                overlap = np.sum(h[:valid] * g[m:]) * dx
            overlaps.append(overlap)
        max_overlap = max(overlaps)
        m_max = overlaps.index(max_overlap)
        
        grad = np.zeros(n_points)
        if m_max == 0:
            grad += (1 - 2 * h) * dx
        else:
            valid = n_points - m_max
            grad[:valid] += (1 - h[m_max:m_max + valid]) * dx
            grad[m_max:m_max + valid] -= h[:valid] * dx
        
        h_candidate = h - learning_rate * grad
        h_clipped = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_clipped)
        if not np.isclose(current_sum, T, rtol=1e-6, atol=1e-8):
            scale = T / current_sum
            h_scaled = h_clipped * scale
