import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values provided in the environment
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        overlay = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(overlay)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    current_h = h.copy()

    while time.time() - start_time < budget_s:
        # Compute overlaps for current h
        overlay = np.correlate(current_h, 1 - current_h, mode='full') * dx
        overlay_relevant = overlay[n_points - 1: 2 * n_points - 1]
        if len(overlay_relevant) == 0:
            break
        # Select top 10% or max 10 shifts
        top_indices = np.argpartition(-overlay_relevant, min(10, len(overlay_relevant)))[:, :10].flatten()
        selected_shifts = np.unique(top_indices[top_indices < len(overlay_relevant)])
        if len(selected_shifts) == 0:
            break

        # Setup convex problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        for shift in selected_shifts:
            coeffs = np.zeros(n_points)
            c = 0.0
            for i in range(n_points - shift):
                j = i + shift
                coeffs[j] += current_h[i]
                coeffs[i] += current_h[j]
                c += current_h[i] * current_h[j]
            constraints.append(dx * coeffs @ h_var - dx * c >= 1 - t)

        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
                new_h = h_var.value
                new_h = np.clip(new_h, 0, 1)
                total = np.sum(new_h)
                if abs(total) < 1e-10:
                    new_h = np.ones(n_points) * 0.5
                else:
                    new_h = new_h * (n_points / 2 / total)
                new_h = np.clip(new_h, 0, 1)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    current_h = new_h.copy()
        except:
            pass

    return best_h, best_c5, n_points
