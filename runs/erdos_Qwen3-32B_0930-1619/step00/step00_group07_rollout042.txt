    start_time = time.time()

    while time.time() - start_time < budget_s and max_iter > 0:
        max_iter -= 1

        # Compute cross-correlation

        c = np.correlate(h, 1 - h, mode='full')
        # Extract relevant shifts
        c5_candidates = c[n_points : 2 * n_points -1]
        current_c5 = np.max(c5_candidates) * dx

        if current_c5 < best_c5:

            best_h = h.copy()

        # Find the worst k (shift)
        max_index = np.argmax(c5_candidates)

        h = np.clip(h, 0.0, 1.0)  # clip again in case scaling pushed some beyond

    current_sum = np.sum(h)
