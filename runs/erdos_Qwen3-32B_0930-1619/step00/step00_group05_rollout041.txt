import numpy as np

def run(seed=42, budget_s=1100):

    n_points = len(h)
    dx = 2.0 / n_points

    # Clip and rescale h to meet constraints
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total) > 1e-10:

    best_c5 = np.inf

    m = np.zeros_like(h)
    v = np.zeros_like(h)
    no_improve = 0

    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:

        max_val = np.max(corr)
        current_c5 = max_val * dx

        if current_c5 < best_c5:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 100:

                no_improve = 0

                    dh[i] = 1 - h[i + lag]

            for i in range(n_points):
                if i + lag >= 0:
