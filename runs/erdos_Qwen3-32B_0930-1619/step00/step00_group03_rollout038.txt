import numpy as np
import scipy.optimize as opt
from time import time

def run(budget_s=1100):
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize to match sum constraint
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        scale = target_sum / current_sum
        h0 = np.clip(h0 * scale, 0.0, 1.0)

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        # Compute cross-correlation max
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)

    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum},)

    start_time = time()

        x0=h0,
        method='COBYLA',
        bounds=bounds,
        constraints=cons,
        options={'rhobeg': 0.1, 'maxiter': int(900.), 'tol': 1e-6}
    )
    elapsed = time() - start_time

        result.x = result.x  # Use solution from last iteration
