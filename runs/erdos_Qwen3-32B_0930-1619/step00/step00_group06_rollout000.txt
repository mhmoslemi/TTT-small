    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = 0.0
    
    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        S = corr[:n_points] * dx
        return np.max(S)
    
    def compute_gradient(h, max_index):
        m = max_index
        grad = np.zeros_like(h)
        for i in range(n_points - m):
            j = i + m
            grad[i] += 1 - h[j]
            grad[j] -= h[i]
        return grad
    
    def feasibility_repair(h, desired_sum):
        h_clipped = np.clip(h, 0, 1)
        cs = np.sum(h_clipped)
        if abs(cs - desired_sum) < 1e-8:
            return h_clipped.copy()
        changeable = np.where((h_clipped > 1e-8) & (h_clipped < 1 - 1e-8))[0]
        if len(changeable) == 0:
            return h_clipped.copy()
        delta = desired_sum - cs
        adj = delta / len(changeable)
        h_clipped[changeable] += adj
        return np.clip(h_clipped, 0, 1)
    
    best_c5 = evaluate_c5(h, dx)
    current_h = h.copy()
    
    while (time.time() - start_time) < budget_s:
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        S = corr[:n_points] * dx
        max_index = np.argmax(S)
        
        grad = compute_gradient(current_h, max_index)
        alpha = 0.1
        for _ in range(5):
            candidate = current_h - alpha * grad
            candidate_repaired = feasibility_repair(candidate, desired_sum)
            c5_candidate = evaluate_c5(candidate_repaired, dx)
            if c5_candidate < best_c5:
                current_h = candidate_repaired.copy()
                best_c5 = c5_candidate
                break
            alpha *= 0.5
        
        if best_c5 < evaluate_c5(best_h, dx):
            best_h = current_h.copy()
    
    return best_h.tolist(), best_c5, n_points
