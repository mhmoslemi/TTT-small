        delta_abs = -delta
        mask = h_clipped > 0.0
        if mask.any():
            amount = delta_abs / mask.sum()
            h_clipped[mask] -= amount
    return h_clipped

def evaluate_c5(h, dx):
    return np.max(compute_correlations(h, dx))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999

    while time.time() - start_time < budget_s:
        corr = compute_correlations(h, dx)
        max_idx = np.argmax(corr)
        k_shift = max_idx - (n_points - 1)
        grad = compute_gradient(h, k_shift)
        h = project(h - learning_rate * grad, target_sum)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    return best_h, best_c5, n_points
