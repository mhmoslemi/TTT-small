def interpolate_to_resolution(h, current_n, new_n):
    x_old = np.linspace(0, 2, current_n)
    x_new = np.linspace(0, 2, new_n)
    h_interpolated = np.interp(x_new, x_old, h)
    required_sum = new_n / 2.0
    h_interpolated *= required_sum / np.sum(h_interpolated)
    return h_interpolated

def compute_overlaps(h, n_points):
    dx = 2.0 / n_points
    h1 = h
    h2 = 1 - h
    corr = np.correlate(h1, h2, mode='full')
    return corr * dx

def optimize_resolution(h_initial, n_points, max_iter, start_time, budget_s):
