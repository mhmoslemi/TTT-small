import numpy as np
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration parameters
    n_points = 200  # Higher resolution for better accuracy
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Ensures ∫h(x)dx = 1 over [0, 2]
    
    # Initialize h from global initial_h_values using interpolation
    n_initial = len(initial_h_values)
    x_old = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    f = interp1d(x_old, initial_h_values, kind='linear')
    h = f(x_new)
    # Normalize to satisfy integral constraint
    h *= target_sum / np.sum(h)
    h = np.clip(h, 0, 1)  # Enforce domain constraint
    
    # C₅ evaluation function
    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return float(np.max(corr) * dx)
    
    best_c5 = best_h = evaluate_c5(h), h.copy()
    
    # Optimization parameters
    alpha = 0.1  # Step size
    temp = 0.1   # Initial temperature
    cool_rate = 0.99  # Exponential cooling
    max_noimprove = 100  # Early termination
    
    while time.time() - start_time < budget_s:
        h_candidate = h.copy()
        
        # Add random perturbation
        perturbation = np.random.uniform(-alpha, alpha, size=n_points)
        h_candidate = np.clip(h_candidate + perturbation, 0, 1)
        
        # Projection onto feasible region (sum constraint)
        current_sum = h_candidate.sum()
        if current_sum != 0:
            scale = target_sum / current_sum
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
        
        # Compute overlap for candidate
        new_c5 = evaluate_c5(h_candidate)
        
        # Acceptance criterion
        if new_c5 < best_c5[0] or np.random.rand() < np.exp((best_c5[0] - new_c5)/temp):
            h = h_candidate
            if new_c5 < best_c5[0]:
                best_c5 = (new_c5, h.copy())
                if time.time() - start_time >= budget_s:
                    break
        
        # Update temperature and check for termination
        temp *= cool_rate
        if time.time() - start_time >= budget_s:
            break
    
    return best_c5[1], best_c5[0], n_points
