        if s == 0:
            h_repaired = np.zeros_like(h_clipped)
            h_repaired[:n_points // 2] = 1.0
        else:
            scaling = (n_points / 2) / s
            h_scaled = h_clipped * scaling
            h_repaired = np.clip(h_scaled, 0, 1)

    pbest_positions = [p.copy() for p in particles]
