import numpy as np
import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=1100):
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    desired_sum = n_points / 2
    dx = 2.0 / n_points

    # Normalize h0 to meet constraints
    h0 = np.clip(h0, 0, 1)
    if np.sum(h0) != 0:
        h0 = h0 / h0.sum() * desired_sum
    h0 = np.clip(h0, 0, 1)

    # Objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # Constraints and bounds
    bounds = [(0, 1)] * n_points
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - desired_sum})

    # Optimization options
    options = {'maxiter': 600, 'ftol': 1e-7, 'disp': False}

    # Time tracking
    start_t = time()

    # Run optimization
    result = opt.minimize(
        objective, 
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    # Evaluate final solution
    c5_bound, h_final = evaluate_erdos_solution(result.x, n_points)

    return h_final, c5_bound, n_points
