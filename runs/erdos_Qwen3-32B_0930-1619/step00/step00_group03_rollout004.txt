    objective = cvxpy.Minimize(0.5 * cvxpy.sum_squares(h_proj - h))

    constraints = [0 <= h_proj, h_proj <= 1, cvxpy.sum(h_proj) == s]

    prob = cvxpy.Problem(objective, constraints)

    prob.solve(solver='ECOS')

    return np.array(h_proj.value).flatten()

def run(seed=42, budget_s=1100, **kwargs) -> Tuple[np.ndarray, float, int]:

    dx = 2.0 / n_points
    start_time = time.time()

    def project(y: np.ndarray, s: float) -> Optional[np.ndarray]:
        n = len(y)
        h_proj = cp.Variable(n)
        objective = cp.Minimize(0.5 * cp.sum_squares(h_proj - y))
        constraints = [0 <= h_proj, h_proj <= 1, cp.sum(h_proj) == s]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h_proj.value
        except:
            return None

    while True:
        # Time remaining check
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

        # Evaluate current max overlap and find best shift index
        cross_corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr_index = np.argmax(cross_corr)
        k_shift = max_corr_index - (n_points - 1)  # Convert correlate index to effective shift

        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from positive direction (i + k_shift)
            if 0 <= i + k_shift < n_points:
                grad[i] += (1.0 - h[i + k_shift]) * dx
            # Contribution from negative direction (i - k_shift)
            if 0 <= i - k_shift < n_points:
                grad[i] -= h[i - k_shift] * dx

        # Gradient descent with fixed learning rate
