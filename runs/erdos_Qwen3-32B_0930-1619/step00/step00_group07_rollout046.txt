    # Initial guess for C
    initial_c = compute_c5(h0)
    x0 = np.concatenate([h0, [initial_c]])

    def constraint_all(x):
        h = x[:-1]
        C = x[-1]
        n = len(h)
        dx_val = 2.0 / n
        constraints = []
        for k in range(-n + 1, n):
            start_i = max(0, -k)
            end_i = min(n - 1, n - 1 - k)
            val = 0.0
            if start_i <= end_i:
                slice_i = slice(start_i, end_i + 1)
                slice_j = slice(start_i + k, end_i + k + 1)
                val = np.sum(h[slice_i] * h[slice_j])
            overlap_integral = val * dx_val
            constraints.append(1.0 - overlap_integral - C)
        return np.array([-v for v in constraints])
