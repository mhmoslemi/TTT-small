import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    alpha = 0.1
    lambda_tv = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s:
        correlations = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlations)
        idx = np.unravel_index(np.argmax(correlations), correlations.shape)
        k_max = idx[0] - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                gradient[i] = dx * ((1 - h[j]) - h[i])

        for i in range(1, n_points - 1):
            gradient[i] += lambda_tv * (h[i-1] - 2*h[i] + h[i+1])

        h_new = h - alpha * gradient
        h_new = np.clip(h_new, 0, 1)
        current_sum = h_new.sum()
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            delta = target_sum - current_sum
            h_new += delta / n_points
            h_new = np.clip(h_new, 0, 1)

        new_correlations = np.correlate(h_new, 1 - h_new, mode='full') * dx
        new_c5 = np.max(new_correlations)

        if new_c5 < current_c5:
            h[:] = h_new[:]
            if new_c5 < best_c5:
                best_h[:] = h[:]
                best_c5 = new_c5

        else:
