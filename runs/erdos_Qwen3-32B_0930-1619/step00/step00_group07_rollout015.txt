        callback(x)
        return False

    from scipy.optimize import minimize
    result = minimize(
        objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        callback=time_callback,
        options={'maxiter': 100000, 'ftol': 1e-10}
    )

    final_c5 = run.best_c5
    final_h = run.best_h

    # Final constraint check and clipping
    final_h = np.clip(final_h, 0.0, 1.0)
    final_sum = np.sum(final_h)
    if not np.isclose(final_sum, n_points / 2):
        final_h *= (n_points / 2) / final_sum
    final_h = np.clip(final_h, 0.0, 1.0)

    return final_h, final_c5, n_points
