                child1 = np.concatenate([parent1[:mid], parent2[mid:]])
                child2 = np.concatenate([parent2[:mid], parent1[mid:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            for child in [child1, child2]:
                if np.random.rand() < mutation_prob:
                    mutation = np.random.normal(0, mutation_strength, n_points)
                    child += mutation
                child = repair_candidate(child, target_sum)
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
            if time.time() - start_time >= budget_s:
                break
        
        if not new_population:
            break
        
        combined = np.vstack([population, new_population])
        combined_fitness = np.array([compute_c5(h, dx) for h in combined])
        idx = np.argsort(combined_fitness)
        population = combined[idx[:population_size]]
        fitness = combined_fitness[idx[:population_size]]
    
    return (best_h.tolist(), best_c5, n_points)
