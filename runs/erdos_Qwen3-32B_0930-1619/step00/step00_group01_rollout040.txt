import time
import random

def generate_perturbed_candidate(h, n_points, n_perturb=5):
    h_perturbed = h.copy()
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        h_perturbed[i] = np.random.uniform(0.0, 1.0)
    target_sum = n_points / 2.0
    adjustment = (target_sum - np.sum(h_perturbed)) / n_points
    h_perturbed += adjustment
    h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
    return h_perturbed

def coordinate_descent(h, n_points, max_steps=200):
    current_h = h.copy()
    for _ in range(max_steps):
        i = np.random.randint(n_points)

        best_c5 = compute_c5(current_h, n_points)
        best_h = current_h.copy()
        for v in candidate_values:
            new_h = current_h.copy()
            new_h[i] = v
            current_sum = np.sum(new_h)
            delta = (n_points / 2.0) - current_sum
            adjustment = delta / n_points
            new_h += adjustment
            new_h_clipped = np.clip(new_h, 0.0, 1.0)

            if c5 < best_c5:

                best_h[:] = new_h_clipped

    return current_h

def compute_c5(h, n_points):

    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx
