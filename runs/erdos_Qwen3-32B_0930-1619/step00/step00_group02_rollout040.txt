import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    dx = 2.0 / 100  # Use 100 samples for balance of speed and accuracy
    n_points = 100
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    try:
        h = np.interp(np.linspace(0, 2, n_points),
                      np.linspace(0, 2, len(initial_h_values)),
                      initial_h_values)
    except:
        # Fallback to uniform distribution
        h = np.ones(n_points) / 2
    
    # Initial constraint projection
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    best_c5 = float('inf')
    best_h = h.copy()
    alpha = 0.1
    decay_rate = 0.99
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute correlation max
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        k = abs(lag)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
        
        # Compute gradient for the active k
        g = np.zeros(n_points)
        if k <= n_points:
            for i in range(n_points - k):
                j = i + k
                g[i] += (1 - h[j])
                g[j] -= h[i]
        
        g *= dx
        
        # Gradient update + projection
        h = np.clip(h - alpha * g, 0, 1)
        
        # Constraint correction
        current_sum = np.sum(h)
        h = h + (target_sum - current_sum) / n_points
        h = np.clip(h, 0, 1)
        
        alpha *= decay_rate
        
    return best_h, best_c5, n_points
