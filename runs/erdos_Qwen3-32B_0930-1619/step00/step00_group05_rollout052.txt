    child2 = np.concatenate([p2[:midpoint], p1[midpoint:]])
    child1 = repair(child1, target_sum)
    child2 = repair(child2, target_sum)
    return child1, child2

def mutate(h, p_mutate=0.1, max_perturb=0.2, target_sum=None):
    h_mut = h.copy()
    mask = np.random.rand(len(h)) < p_mutate
    perturb = np.random.uniform(-max_perturb, max_perturb, size=len(h)) * mask
    h_mut += perturb
    h_mut = repair(h_mut, target_sum)
    return h_mut

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    pop_size = 25
    tournament_size = 3
    p_mutate = 0.15
    max_perturb = 0.2

    # Initialize population
    population = []
    initial_h = repair(np.array(initial_h_values), target_sum)
    population.append(initial_h)

    for _ in range(pop_size - 1):
        h_random = np.random.rand(n_points)
        h_random = repair(h_random, target_sum)
        population.append(h_random)

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        # Evaluate all current population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            fitness.append(c5)

        # Select parents
        parents = select_parents(population, fitness, pop_size, tournament_size)

        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1 = parents[i]
                p2 = parents[i + 1]
                child1, child2 = crossover(p1, p2, target_sum)
                child1 = mutate(child1, p_mutate, max_perturb, target_sum)
                child2 = mutate(child2, p_mutate, max_perturb, target_sum)
                offspring.extend([child1, child2])
            else:
                p = parents[i]
                offspring.append(p.copy())

        # Combine population and offspring
        combined = population + offspring
        combined_fitness = [evaluate_erdos_solution(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        population = [combined[i] for i in sorted_indices[:pop_size]]

        if time.time() - start_time >= budget_s:
            break

    return (best_h, best_c5, n_points)
