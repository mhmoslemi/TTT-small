import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = -np.inf  # Just to start, will be replaced
    
    # Evaluate initial solution
    def compute_c5(h_current):
        max_overlap = -np.inf
        for k in range(n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                continue
            sum_val = np.sum(h_current[:valid_len] * (1 - h_current[k:k+valid_len]))
            over = sum_val * dx
            max_overlap = max(max_overlap, over)
        return max_overlap

    best_c5 = compute_c5(h)
    
    start_time = time.time()
    max_iterations = 1000
    for it in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                overlaps[k] = 0.0
            else:
                sum_val = np.sum(h[:valid_len] * (1 - h[k:k+valid_len]))
                overlaps[k] = sum_val * dx
        
        sorted_indices = np.argsort(overlaps)[::-1]
        top_k = 4  # Fixed small number to handle sparsity
        
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0
        ]
        
        for k in sorted_indices[:top_k]:
            if k >= n_points:
                continue
            linearized = 0.0
            for i in range(n_points - k):
                linearized += (h[i] * h_var[i + k] + h_var[i] * h[i + k]) - h[i] * h[i + k]
            constraints.append(t >= linearized)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                continue
        except Exception:
            continue
        
        # Clip and rescale
        h_new_clipped = np.clip(h_new, 0, 1)
        if np.sum(h_new_clipped) == 0:
            h_new_scaled = h_new_clipped
        else:
            scale = (n_points / 2.0) / np.sum(h_new_clipped)
            h_new_scaled = h_new_clipped * scale
        
        current_c5 = compute_c5(h_new_scaled)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h_new_scaled[:]
            h[:] = h_new_scaled[:]  # Update for next iteration
        
    return (best_h, best_c5, n_points)
