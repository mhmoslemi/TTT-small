def compute_gradient(h, k, dx, n_points):
    n = n_points
    grad = np.zeros(n)
    # Forward part: i + k in range
    indices_forward = np.arange(n) + k
    valid_forward = (indices_forward >= 0) & (indices_forward < n)
    contrib_forward = np.where(valid_forward, 1.0 - h[indices_forward], 0.0)
    grad += contrib_forward * dx

    # Backward part: i - k in range
    indices_backward = np.arange(n) - k
    valid_backward = (indices_backward >= 0) & (indices_backward < n)
    contrib_backward = np.where(valid_backward, h[indices_backward], 0.0)
    grad -= contrib_backward * dx
    return grad
