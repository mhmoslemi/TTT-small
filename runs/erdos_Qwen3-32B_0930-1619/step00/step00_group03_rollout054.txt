from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h0 = initial_h_values.copy()
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        return np.max(overlaps)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = Bounds(0.0, 1.0)
    cons = ({'type': 'eq', 'fun': constraint_sum})

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100}
    )
    elapsed = time.time() - start_time

    h_opt = result.x.reshape(n_points)
    h_opt_sum = np.sum(h_opt)
    if abs(h_opt_sum - target_sum) > 1e-6:
        scale = target_sum / h_opt_sum
        h_opt = np.clip(h_opt * scale, 0.0, 1.0)

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    overlaps = corr * dx
    c5_bound = np.max(overlaps)

    return h_opt.tolist(), c5_bound, n_points
