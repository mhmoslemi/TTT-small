    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h meets constraints
    h = np.clip(h, 0, 1)
    # Adjust sum
    def project(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = h_clipped.sum()
        if abs(sum_clipped - target_sum) < 1e-8:
            return h_clipped
        delta = target_sum - sum_clipped
        h_clipped += delta / n_points
        return np.clip(h_clipped, 0, 1)

    h = project(h, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h) # assume this returns C5

    alpha = 0.1
    start_time = time.time()
    no_improvement = 0
    max_no_improvement = 100

    while time.time() - start_time < budget_s:
        conv = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = conv.max()
        k_index = np.argmax(conv)
        k_max = k_index - (n_points - 1)  # actual lag

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        # Gradient step
        h -= alpha * grad
        # Project back
        h = project(h, target_sum)
        # Check if current norm is better
        current_c5_new = evaluate_erdos_solution(h)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = h.copy()
            alpha *= 1.05  # increase step if improvement
            no_improvement = 0
        else:
            alpha *= 0.9  # decrease step if no improvement
            no_improvement += 1
            if no_improvement > max_no_improvement:
                h = best_h.copy()  # restart from best
                no_improvement = 0

                perturb = np.random.normal(0, 0.01, h.shape)
                h += perturb
                h = project(h, target_sum)
        # Check time
    # Return
    return (best_h, best_c5, n_points)
