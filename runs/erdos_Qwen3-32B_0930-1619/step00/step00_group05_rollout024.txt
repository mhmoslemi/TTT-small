            if m_shift < n_points:

        else:
            m_abs = -m_shift
            if m_abs < n_points:
                grad_backward[:n_points - m_abs] = -h[m_abs:]

        grad = grad_forward + grad_backward

        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iter_count)
        v_hat = v / (1 - beta2**iter_count)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h = h - step
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, 0):
            h *= target_sum / current_sum

        current_c = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_c) * dx

        if current_c5 < best_c5 - 1e-8:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= 100:
                lr *= 0.5
                no_improvement = 0
                if lr < 1e-6:
                    break

    return (best_h.tolist(), best_c5, n_points)
