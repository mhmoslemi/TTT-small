import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    
    target_sum = n_points / 2.0
    assert np.isclose(np.sum(h), target_sum, atol=1e-6), "Initial h violates constraints."
    
    step_size = 0.1
    decay_rate = 0.999
    min_step_size = 1e-4
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current C5 and corresponding k
        h_1 = 1.0 - h
        corr = np.correlate(h, h_1, mode='full') * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        # Compute gradient for max shift k
        gradient = np.zeros(n_points)
        for j in range(n_points):
            # Negative part (j -k)
            i1 = j - k
            if 0 <= i1 < n_points:
                gradient[j] -= h[i1] * dx
            # Positive part (j +k)
            i2 = j + k
            if i2 < n_points:
                gradient[j] += (1.0 - h[i2]) * dx
        
        # Update h with gradient step
        h -= step_size * gradient
        
        # Clip to [0, 1]
        np.clip(h, 0.0, 1.0, out=h)
        
        # Adjust sum constraint
        current_sum = h.sum()
        delta = target_sum - current_sum
        adj = delta / n_points
        h += adj
        np.clip(h, 0.0, 1.0, out=h)
        
        # Step size decay
        step_size = max(step_size * decay_rate, min_step_size)
    
    return best_h.tolist(), best_c5, n_points
