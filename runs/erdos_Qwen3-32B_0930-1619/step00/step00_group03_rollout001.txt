import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points

    best_C5 = np.inf

    decay_rate = 0.999

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_C5 = np.max(corr) * dx

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h[:] = h.copy()

        max_corr = np.max(corr)
        max_indices = np.where(np.isclose(corr, max_corr, atol=1e-8))[0]

        grad = np.zeros(n_points)

            k = m - (n_points - 1)
            start = max(0, -k)
            end = min(n_points, n_points - k)
            valid_i = np.arange(start, end, dtype=int)
            j = valid_i + k
            grad[valid_i] += (1 - h[j])
            grad[j] += -h[valid_i]

        h -= learning_rate * grad
        np.clip(h, 0, 1, out=h)
