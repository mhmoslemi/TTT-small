            alpha = (n_points / 2) / s
            h_candidate = np.clip(h_candidate * alpha, 0, 1)
        h = h_candidate

        if t % 100 == 0:
            learning_rate *= 0.1

    return best_h, best_c5, n_points

def compute_c5(n, h):
    dx = 2.0 / n
    correl = np.correlate(h, 1 - h, mode='full')
    return np.max(correl * dx)
