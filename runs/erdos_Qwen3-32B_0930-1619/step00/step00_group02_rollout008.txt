    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped
    # If sum is way off, adjust
    if current_sum < target_sum:
        delta = target_sum - current_sum
        # Find how much we can add to each element without exceeding 1
        possible_increases = 1 - h_clipped
        total_possible = np.sum(possible_increases)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with upsampling from initial_h_values if available
    if 'initial_h_values' in globals():
        original_n = len(initial_h_values)
        dx_old = 2.0 / original_n
        h = np.zeros(n_points)
        for i in range(n_points):
            x_i_new = i * dx
            i_old = int(np.floor(x_i_new / dx_old))
            if i_old >= original_n:
                i_old = original_n - 1
            h[i] = initial_h_values[i_old]
    else:
        h = np.random.rand(n_points)
    
    def project(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if np.isclose(current_sum, target_sum):
            return h_clipped.copy()
        
        m = target_sum
        if current_sum < m:
            delta = m - current_sum
            possible = 1.0 - h_clipped
            total_possible = possible.sum()
            if total_possible > 0:
                alpha = delta / total_possible
                h_clipped += alpha * possible
            else:
                h_clipped[:] = 1.0
        else:
            delta = current_sum - m
            possible = h_clipped
            total_possible = possible.sum()
            if total_possible > 0:
                alpha = delta / total_possible
                h_clipped -= alpha * possible
            else:
                h_clipped[:] = 0.0
        return h_clipped
    
    def evaluate_c5(h):
        kernel = 1.0 - h
        full_correlate = np.correlate(h, kernel, mode='full') * dx
        return np.max(full_correlate)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = 0.05
    patience_counter = 0
    while time.time() - start_time < budget_s - 2:
        # Compute overlaps
        kernel = 1.0 - h
        full_correlate = np.correlate(h, kernel, mode='full') * dx
        max_index = np.argmax(full_correlate)
        k_max = max_index - (h.size - 1)
        
        # Compute gradient for current k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:

        # Update step with projection

        h = project(h, target_sum)
