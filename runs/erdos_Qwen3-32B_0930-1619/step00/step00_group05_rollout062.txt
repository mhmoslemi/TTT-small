import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = np.array(initial_h_values, copy=True)
    n_points = len(h0)
    dx = 2.0 / n_points

    def repair_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            h[:] = 0.5
        else:
            h *= (n_points / 2.0) / total
        return h

    h0 = repair_h(h0.copy())

    # PSO parameters
    swarm_size = 30
    particles_pos = np.zeros((swarm_size, n_points))
    particles_vel = 0.1 * np.random.randn(swarm_size, n_points)
    particles_pbest = np.zeros((swarm_size, n_points))
    particles_pbest_scores = np.full(swarm_size, np.inf)
    gbest_pos = np.zeros(n_points)
    gbest_score = np.inf

    # Initialize particles
    for i in range(swarm_size):
        pos = h0 + np.random.uniform(-0.1, 0.1, size=n_points)
        particles_pos[i] = repair_h(pos)
        particles_pbest[i] = particles_pos[i].copy()
        c5 = evaluate_erdos_solution(particles_pos[i])
        particles_pbest_scores[i] = c5
        if c5 < gbest_score:
            gbest_pos[...] = particles_pos[i].copy()
            gbest_score = c5

    inertia_start, inertia_end = 0.9, 0.4
    cognitive, social = 2.0, 2.0
    max_stall = 100
    stall_count = 0

    start_time = time.time()
    while time.time() - start_time < budget_s:
        progress = (time.time() - start_time) / budget_s
        inertia = inertia_start - progress * (inertia_start - inertia_end)

        # Update velocities and positions
        for i in range(swarm_size):
            pos = particles_pos[i]
            vel = particles_vel[i]
            pbest = particles_pbest[i]
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = (inertia * vel +
                   cognitive * r1 * (pbest - pos) +
                   social * r2 * (gbest_pos - pos))
            vel = np.clip(vel, -0.5, 0.5)  # Limit step size
            pos += vel
            particles_pos[i] = repair_h(pos)

        # Evaluate and update personal/global best
        updated = False
        for i in range(swarm_size):
            c5 = evaluate_erdos_solution(particles_pos[i])
            if c5 < particles_pbest_scores[i]:
                particles_pbest_scores[i] = c5
                particles_pbest[i] = particles_pos[i].copy()
                if c5 < gbest_score:
                    gbest_pos[...] = particles_pos[i].copy()
                    gbest_score = c5
                    stall_count = 0
                    updated = True
        if not updated:
            stall_count += 1
        if stall_count >= max_stall:
            break

    return gbest_pos.tolist(), gbest_score, n_points
