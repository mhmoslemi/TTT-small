import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()

    global initial_h_values
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            return 0.5 * np.ones(n_points)
        scaling = (n_points / 2.0) / current_sum
        h *= scaling
        h = np.clip(h, 0, 1)
        return h

    initial_h = repair_h(np.copy(initial_h))

    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h)
        population.append(h)

    best_c5 = float('inf')
    best_h = population[0].copy()

    def evaluate(h):
        one_minus_h = 1.0 - h
        correlations = np.correlate(h, one_minus_h, mode='full')
        return np.max(correlations) * dx

    while time.time() - start_time < budget_s:
        c5_values = [evaluate(chrom) for chrom in population]
        current_best_idx = np.argmin(c5_values)
        if c5_values[current_best_idx] < best_c5:
            best_c5 = c5_values[current_best_idx]
            best_h = np.copy(population[current_best_idx])

        new_population = [best_h.copy()]
        for _ in range(population_size - 1):
            # Tournament selection
            participants = np.random.choice(len(population), 5)
            candidates = [population[i] for i in participants]
            values = [c5_values[i] for i in participants]
            winner_idx = participants[np.argmin(values)]
            parent1 = population[winner_idx].copy()

            participants2 = np.random.choice(len(population), 5)
            candidates2 = [population[i] for i in participants2]
            values2 = [c5_values[i] for i in participants2]
            winner_idx2 = participants2[np.argmin(values2)]
            parent2 = population[winner_idx2].copy()

            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

            # Mutation
            if np.random.rand() < 0.15:
                n_mutate = max(1, int(0.1 * n_points))
                indices = np.random.choice(n_points, n_mutate, replace=False)
                for idx in indices:
                    perturbation = np.random.uniform(-0.2, 0.2)
                    child[idx] = np.clip(child[idx] + perturbation, 0.0, 1.0)

            child = repair_h(child)
            new_population.append(child)

        population = new_population

    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
