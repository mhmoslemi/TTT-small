            h[idx] = 1.0

            h = h * (n_points / 2.0) / s
        return h

    # Validate and repair initial_h
    best_h = repair_h(initial_h)
    pop_size = 20
    population = [best_h.copy()]
    for _ in range(pop_size - 1):
        rand_h = np.random.rand(n_points)
        population.append(repair_h(rand_h))

    best_c5 = evaluate_erdos_solution(best_h)
    no_improve = 0
    max_no_improve = 100

    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(c) for c in population]
        current_idx = np.argmin(fitness)
        if fitness[current_idx] < best_c5:
            best_c5 = fitness[current_idx]
            best_h = population[current_idx].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= max_no_improve:
            break

        # Tournament selection
        def tournament(pop, fit, size=3):
            indices = np.random.choice(pop_size, size)
            best = indices[np.argmin([fit[i] for i in indices])]
            return pop[best]

        p1 = tournament(population, fitness)

        # Crossover
