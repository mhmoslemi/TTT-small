import numpy as np
import time

def repair_h(h, desired_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, desired_sum, rtol=1e-5, atol=1e-6):
        return h_clipped
    flex_indices = np.where((h_clipped > 0) & (h_clipped < 1))[0]
    if len(flex_indices) == 0:
        return h_clipped
    delta = desired_sum - current_sum
    adjustment = delta / len(flex_indices)
    h_clipped[flex_indices] += adjustment
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

def crossover(parents):
    parent1, parent2 = parents
    split_idx = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:split_idx], parent2[split_idx:]])
    child2 = np.concatenate([parent2[:split_idx], parent1[split_idx:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, mutation_step=0.2):

        if np.random.rand() < mutation_rate:
            delta = mutation_step * np.random.uniform(-1, 1)
            h[i] += delta
    return np.clip(h, 0, 1)
