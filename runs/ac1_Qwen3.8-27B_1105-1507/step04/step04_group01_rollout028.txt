**Frequency domain approach:**

The constraint `a >= 0` becomes: `IFFT(A) >= 0` component-wise.
The constraint `sum(a) = 1` becomes: `A[0] = 1` (DC component).

We want to minimize the peak of `IFFT(|A|^2)`.

"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    conv = np.maximum(conv, 0)  # numerical safety
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def soft_max_obj_and_grad(z, tau):
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z
    
    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1
    
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    
    return soft_max_val, grad_z

def optimize_lfbs(a_init, tau_list, max_iter=200, time_budget=30.0):
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        try:
            result = minimize(soft_max_obj_and_grad, z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8,
                                      'maxls': 50})
            z_new = result.x
        except Exception:
            continue
        
        a_new = softmax(z_new)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            z = z_new
    
    return best_a, best_obj

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=15):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

def alternating_lp(a, max_iters=50, time_budget=40.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 30
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, _ = solve_lp_g(a, M, time_limit=10)
        if g is None:

        # Finer line search

            a_try = np.maximum(a_try, 0)

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            a = best_a_new

                s = np.sum(a)

                stall_count = 0
                patience = 15

def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau_start=3.0, tau_end=50.0):

def shape_init(n, shape_type, **kwargs):
    """Generate smooth shape initializations."""
    k = np.arange(n)
    t = 2.0 * k / (n - 1) - 1.0  # normalized to [-1, 1]
    
    if shape_type == 'hann':
        a = 0.5 * (1 + np.cos(np.pi * t))
    elif shape_type == 'raised_cos2':
        a = np.cos(np.pi * t / 2) ** 2
    elif shape_type == 'gaussian':
        sigma = kwargs.get('sigma', 0.4)
        a = np.exp(-t**2 / (2 * sigma**2))
    elif shape_type == 'plateau':
        # Flat top with cosine roll-off
        width = kwargs.get('width', 0.6)
        a = np.ones(n)
        roll = (1 - width) / 2
        for i in range(n):
            if t[i] < -width / 2:
                frac = (t[i] + 1) / (1 - width / 2) if (1 - width/2) > 0 else 0
                a[i] = 0.5 * (1 + np.cos(np.pi * frac))
            elif t[i] > width / 2:
                frac = (1 - t[i]) / (1 - width / 2) if (1 - width/2) > 0 else 0
                a[i] = 0.5 * (1 + np.cos(np.pi * frac))
    elif shape_type == 'polynomial':
        # (1 - t^2)^p
        p = kwargs.get('p', 2)
        a = np.maximum(0, 1 - t**2) ** p
    elif shape_type == 'fourier':
        # Sum of cosines
        a = np.ones(n)
        n_harm = kwargs.get('n_harm', 3)
        for h in range(1, n_harm + 1):
            amp = kwargs.get(f'amp_{h}', 0.3 / h)
            a += amp * np.cos(h * np.pi * t)
    elif shape_type == 'gaussian_mixture':
        # Sum of Gaussians
        a = np.zeros(n)
        n_comp = kwargs.get('n_comp', 3)
        for c in range(n_comp):
            center = kwargs.get(f'center_{c}', (c - (n_comp-1)/2) * 0.3)
            width = kwargs.get(f'width_{c}', 0.2)
            amp = kwargs.get(f'amp_{c}', 1.0)
            a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif shape_type == 'sqrt_hann':
        a = np.sqrt(0.5 * (1 + np.cos(np.pi * t)))
    elif shape_type == 'bump':
        # Sharp bump in the middle
        width = kwargs.get('width', 0.3)
        a = np.exp(-((t / width) ** 4))
    else:
        a = 0.5 * (1 + np.cos(np.pi * t))
    
    a = np.maximum(a, 0.001)
    s = np.sum(a)
    return a / s

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)

"""
Optimized step function search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses symmetric LP alternating, soft-max gradient optimization with adaptive tau,
shape-based initialization, and multi-scale refinement across sequence lengths.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    conv = np.maximum(conv, 0.0)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def soft_max_obj_and_grad(z, tau):
    """Soft-max smoothed objective and gradient w.r.t. z."""
    n = len(z)
    a = softmax_fn(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_fixed(a_fixed, time_limit=12):
    """
    Solve LP: min M s.t. (a_fixed * x)[k] <= M for all k, sum(x)=1, x>=0.
    Returns optimal x and M.
    """
    n = len(a_fixed)
    A = build_toeplitz(a_fixed)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            s = np.sum(x)
            if s > 1e-10:
                x = x / s
                M_opt = float(res.x[n])

    except Exception:

def subgradient_optimize(a, max_iters=2000, time_budget=20.0,
                         tau_start=3.0, tau_end=40.0):

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n

        a = a / s

    step = 0.03
    stall = 0
    best_step = step

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        shifted = np.clip(tau * (conv - max_conv), -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:

        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

            best_obj = obj
            best_a = a.copy()
            stall = 0

def shape_init(n, shape_type):
    """Generate smooth shape initializations."""
    k = np.arange(n, dtype=np.float64)
    t = 2.0 * k / max(1, n - 1) - 1.0

    if shape_type == 'hann':
        a = 0.5 * (1.0 + np.cos(np.pi * t))
    elif shape_type == 'raised_cos2':
        a = np.cos(0.5 * np.pi * t) ** 2
    elif shape_type == 'gaussian':
        a = np.exp(-t ** 2 / (2 * 0.35 ** 2))
    elif shape_type == 'plateau':
        width = 0.7
        a = np.ones(n)
        for i in range(n):
            if t[i] < -width / 2:
                frac = (t[i] + 1.0) / (1.0 - width / 2)
                a[i] = 0.5 * (1.0 + np.cos(np.pi * frac))
            elif t[i] > width / 2:
                frac = (1.0 - t[i]) / (1.0 - width / 2)
                a[i] = 0.5 * (1.0 + np.cos(np.pi * frac))
    elif shape_type == 'polynomial4':
        a = np.maximum(0, 1.0 - t ** 2) ** 2
    elif shape_type == 'fourier3':
        a = np.ones(n)
        a += 0.25 * np.cos(np.pi * t)
        a += 0.1 * np.cos(2 * np.pi * t)
        a += 0.05 * np.cos(3 * np.pi * t)
    elif shape_type == 'sqrt_hann':
        a = np.sqrt(np.maximum(0, 0.5 * (1.0 + np.cos(np.pi * t))))
    elif shape_type == 'bump4':
        a = np.exp(-(t / 0.4) ** 4)
    elif shape_type == 'triangle_flat':
        a = np.ones(n)
        decay = n // 8
        for i in range(n):
            if i < decay:
                a[i] *= i / decay
            elif i >= n - decay:
                a[i] *= (n - 1 - i) / decay
    elif shape_type == 'cos_mod':
        a = 0.5 * (1.0 + np.cos(np.pi * t)) + 0.1 * np.cos(3 * np.pi * t)
    else:
        a = 0.5 * (1.0 + np.cos(np.pi * t))

    a = np.maximum(a, 0.001)
    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_a = a_new.copy()
                if verbose:
                    print(f"    r{rnd} SymLP: {best_obj:.6f}")

        # L-BFGS-B
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            tau_list = [0.5 / n, 0.2 / n, max(0.05, 0.1 / n)]
            a_new, obj_new = optimize_lfbs(best_a, tau_list, max_iter=150,
                                           time_budget=min(remaining * 0.2, 20.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{rnd} L-BFGS: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.2, 18.0),
                                         tau_start=8.0, tau_end=35.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{rnd} SG: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-scale exploration with shape initialization,
    symmetric LP alternating, soft-max gradient refinement, and subgradient polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Lengths to explore
    n_values = [250, 350, 450, 550, 600, 700, 800, 900, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extras = [np_, max(150, np_ - 100), min(1200, np_ + 100), min(1200, np_ + 200)]
        n_values = sorted(set(n_values + extras))

    print(f"Lengths: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    candidates = []

    shapes = ['hann', 'raised_cos2', 'gaussian', 'plateau', 'polynomial4',
              'fourier3', 'sqrt_hann', 'bump4', 'triangle_flat', 'cos_mod']

    # Phase 1: Exploration
    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n <= 500:
            alloc = min(40.0, remaining * 0.11)
        elif n <= 800:
            alloc = min(55.0, remaining * 0.12)
        else:
            alloc = min(65.0, remaining * 0.13)
        alloc = max(18.0, alloc)

        print(f"\nN={n} ({remaining:.0f}s left, alloc={alloc:.0f}s)")

        n_shapes = min(len(shapes), max(3, int(alloc / 14)))

        for si in range(n_shapes):
            if time.monotonic() > deadline - 200:
                break
            rem_s = deadline - time.monotonic()
            if rem_s < 12:
                break

            shape = shapes[si]

            if si == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = shape_init(n, shape)

            # Quick subgradient
            sg_t = min(5.0, (alloc / n_shapes) * 0.2)
            a = subgradient_optimize(a, max_iters=int(sg_t * 60),
                                      time_budget=sg_t,
                                      tau_start=2.0, tau_end=12.0)

            # L-BFGS-B
            rem_s = deadline - time.monotonic()
            if rem_s > 12:
                lfbs_t = min((alloc / n_shapes) * 0.35, 22.0)
                tau_list = [4.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n]
                a, obj = optimize_lfbs(a, tau_list, max_iter=200, time_budget=lfbs_t)
            else:
                obj = compute_obj(a)

            # Symmetric LP
            rem_s = deadline - time.monotonic()
            if rem_s > 25 and n <= 800:
                a = symmetric_lp_alternate(a, max_iters=30,
                                           time_budget=min(rem_s * 0.3, 25.0))
                obj = compute_obj(a)

            # Final subgradient
            rem_s = deadline - time.monotonic()
            if rem_s > 8:
                a = subgradient_optimize(a, max_iters=int(rem_s * 12),
                                          time_budget=min(rem_s * 0.12, 8.0),
                                          tau_start=6.0, tau_end=25.0)
                obj = compute_obj(a)

            print(f"  {shape}: obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polish top candidates
    print(f"\n=== POLISH (best: {best_score:.6f}) ===")
    candidates.sort(key=lambda x: x[1])

    n_polish = min(5, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_c = candidates[ci][0]
        print(f"  Polish[{ci}]: n={len(a_c)}, obj={candidates[ci][1]:.6f}, {remaining:.0f}s")

        p_bud = min(remaining * 0.35, 80.0)
        a_pol, obj_pol = polish_sequence(a_c, time_budget=p_bud, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** NEW BEST: obj={best_score:.6f} **")
        else:
            # Perturbation restarts
            remaining = deadline - time.monotonic()
            if remaining > 35:
                rng = np.random.default_rng(ci * 787 + 55)
                for pt in range(4):
                    if time.monotonic() > deadline - 45:
                        break
                    sc = rng.uniform(0.02, 0.09)
                    a_p = a_c * np.exp(rng.normal(0, sc, len(a_c)))
                    a_p = np.maximum(a_p, 0)
                    sp = np.sum(a_p)
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    a_p = subgradient_optimize(a_p, max_iters=600, time_budget=3.5,
                                               tau_start=5.0, tau_end=20.0)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  ** NEW BEST (pert {pt}): obj={best_score:.6f} **")

    # Phase 3: Final intensive on global best
    if best_seq is not None and time.monotonic() < deadline - 25:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for fr in range(6):
            if time.monotonic() > deadline - 12:
                break
            remaining = deadline - time.monotonic()
            if remaining < 18:
                break

            # Symmetric LP
            if n_best <= 900:
                a_new = symmetric_lp_alternate(a_cur, max_iters=50,
                                               time_budget=min(remaining * 0.3, 40.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SymLP {fr}: {best_score:.6f}")

            # L-BFGS tiny tau
            remaining = deadline - time.monotonic()
            if remaining > 18:
                a_new, obj_new = optimize_lfbs(a_cur,
                                               [0.3/n_best, max(0.05, 0.1/n_best), max(0.03, 0.05/n_best)],
                                               max_iter=400,
                                               time_budget=min(remaining * 0.25, 22.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS {fr}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 12:
                a_new = subgradient_optimize(a_cur, max_iters=int(remaining * 12),
                                             time_budget=min(remaining * 0.2, 15.0),
                                             tau_start=12.0, tau_end=50.0)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG {fr}: {best_score:.6f}")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 200
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== DONE: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
