    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S*S)

def build_toeplitz(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    nnz = 0
    for j in range(n):
        nnz += min(n, 2*n-1-j)  # number of valid output positions for input j
    # Actually, for each j (column of T), the nonzero entries are at rows j to j+n-1 (clipped to 2n-2)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j+n, 2*n-1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    T = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return T

def solve_lp(a, M, time_limit=5.0):
    """Solve: max 1^T g s.t. T(a) @ g <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)  # minimize -1^T g = max 1^T g
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0), res.fun
    except:
        pass
    return None, None

def lp_iteration(a, time_budget_per_lp=3.0):

    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    if g_val is None:
        return None, M
    
    S_g = float(np.sum(g_val))
    if S_g < 1e-10:
        return None, M
    g = g_val / S_g
    
    # Line search
    best_a = a.copy()
    best_M = M
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = (1-t)*a + t*g
        a_try = np.maximum(a_try, 0)
        S_try = np.sum(a_try)
        if S_try < 1e-10:

        conv_try = compute_conv(a_try)
        M_try = float(np.max(conv_try))
        if M_try < best_M:

            best_a = a_try.copy()
    
    return best_a, best_M

def smooth_max_grad(a, beta):
    """Compute gradient of smooth max approximation of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    conv_shifted = conv - np.max(conv)  # numerical stability
    p = np.exp(beta * conv_shifted)
    p = p / np.sum(p)
    
    # Gradient: d(sum_k p[k]*conv[k])/da[i] (but we want d(max)/da[i] approx)
    # Actually, the gradient of (1/beta)*log(sum exp(beta*conv[k])) w.r.t. a[i]
    # = sum_k p[k] * d(conv[k])/da[i]
    # = sum_k p[k] * 2*a[k-i] (for valid k-i)
    # = 2 * (p convolved with a, appropriate slicing)
    
    # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    # So grad[i] = sum_{k=i}^{i+n-1} p[k] * 2*a[k-i] = 2 * sum_j p[i+j] * a[j]
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, p[i:i+n])
    
    return grad, conv, p

def projected_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Project onto sum(x) = target_sum, x >= 0
    # Equiv: project (v - target_sum/n) onto {y >= -target_sum/n, sum(y) = 0}
    # Or just: find theta s.t. sum(max(v - theta, 0)) = target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_gradient_descent(a, time_budget, beta_start=50, beta_end=500):
    """Gradient descent on smooth max with simplex projection."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)

    momentum = np.zeros(n)

        grad_centered = grad - np.mean(grad)

        a_new = a - lr * momentum

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.9
        
        # Increase beta periodically
        beta = min(beta * 1.05, beta_end)
        
        if obj < 1.5028:
            break
    
    return best_a, best_obj

Wait, I realize there's an issue with the smooth gradient. The smooth max `(1/beta)*log(sum exp(beta*conv[k]))` is an UPPER bound on `max conv[k]`. Its gradient points in the direction that increases the weighted average (with weights concentrated on the max). So we want to DECREASE this, meaning we should do gradient descent (subtract the gradient).

But the issue is that the smooth max gradient is not the same as the subgradient of the true max. For large beta, they should be close, but for small beta, the smooth version averages over all lags, which might not be what we want.

Let me reconsider. For the LP approach, the key benefit is that it gives a guaranteed improvement direction. Let me focus on making the LP approach more effective.

Actually, let me think about this more carefully. The LP:
max sum(g)
s.t. conv(a, g) <= M
     g >= 0

The optimal value is `sum(g*)`. If `sum(g*) > 1`, then there exists a feasible g with sum > 1, meaning the constraint conv(a,g) <= M is "loose" in some sense. If we normalize g* to sum 1, we get a direction g with sum 1 and conv(a,g) <= M/sum(g*) < M.

Now, if we set a_new = g* / sum(g*), then:
conv(a_new, a) = conv(g*/sum(g*), a) = conv(g*, a)/sum(g*) <= M/sum(g*) < M

This means the "cross-correlation" between the new and old a is bounded by M/sum(g*). But we need conv(a_new, a_new) <= M.

Hmm, this doesn't directly give us what we want. Let me think about the convex combination approach:

a_t = (1-t)*a + t*g (where g is normalized to sum 1)

conv(a_t, a_t) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)

We know conv(a,a) <= M and conv(a,g) <= M (from LP). But we don't know conv(g,g).

If conv(g,g) is also <= M (which is not guaranteed), then conv(a_t, a_t) <= M for all t in [0,1].

But conv(g,g) might be > M. In that case, we need t to be small enough.

Actually, for small t: conv(a_t, a_t) ≈ conv(a,a) + 2t*(conv(a,g) - conv(a,a)) + O(t^2)

If conv(a,g) < conv(a,a) at the peak, then for small t, the peak decreases. This is the key insight!

So the LP gives us a direction g such that conv(a,g) <= M everywhere. If the current peak of conv(a,a) is at position k*, and conv(a,g)[k*] < M (strictly), then moving in the direction of g will decrease the peak at k*.

The rate of decrease depends on how much conv(a,g)[k*] is below M. If it's equal to M, then the first-order change is zero (the peak is "flat" in the g direction).

For the overall algorithm to work well, we want the LP to give a g that is "furthest" from a in the direction that reduces the peak. The LP's convexity ensures we get the best such g (maximizing sum(g) is a proxy for getting g far from the boundary).

OK, I think the LP approach is sound. Let me now think about how to make it more efficient and effective than the previous implementation.

Key improvements:
1. **Tighter time management for LP**: For n=500, the LP has 500 vars and 999 constraints. HiGHS should solve this in < 0.1s typically. So we can afford many iterations.

2. **Multiple M values**: Instead of just using M = max(conv), also try M slightly below (e.g., 0.99*M, 0.98*M). If the LP is infeasible at the reduced M, it tells us the current M is tight. If feasible, we get a g that works even at the tighter M.

3. **Warm starting**: Use the previous LP solution as a warm start for the next one (if HiGHS supports it through scipy).

4. **Combination with smooth gradient**: When LP stagnates, use smooth gradient to escape.

5. **Better line search**: Use golden section or parabolic interpolation instead of fixed grid.

6. **Multi-M strategy**: Maintain a list of "active" M values (top few peaks) and solve the LP for the most constraining one.

The LP is:
max sum(g)
s.t. conv(a, g)[k] <= M for all k = 0, ..., 2n-2
     g >= 0

where M = max(conv(a, a)).

At the optimum, let g* be the optimizer. The key question is: what is sum(g*)?

If sum(g*) = 1, then g* = a is optimal (since a is feasible: conv(a,a) <= M). But that's a trivial solution.
