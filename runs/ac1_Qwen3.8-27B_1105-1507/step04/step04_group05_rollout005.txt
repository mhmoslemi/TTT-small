2. Fast coordinate descent with precomputed fixed-max (O(n) per evaluation)
3. Active-set LP (only top-K binding constraints)
4. Multi-length exploration
5. Intensive final polishing
Key improvements over previous: faster CD, smaller LPs, smooth optimization phase.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse_active(a, active_indices):
    """Build sparse constraint matrix for LP using only active constraints."""
    n = len(a)
    K = len(active_indices)
    A = lil_matrix((K, n))
    for idx, k in enumerate(active_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def active_set_lp(a, M, K=10):
    """LP: max sum(g) s.t. (a*g)_k <= M for top-K k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    # Get top K indices
    if K >= 2 * n - 1:
        active = list(range(2 * n - 1))
    else:
        top_idx = np.argsort(b)[::-1][:K]
        active = sorted(top_idx.tolist())
    
    c = -np.ones(n)
    A = build_A_sparse_active(a, active)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def cd_evaluate_x(x, ai, a, b, i, n, fixed_max, S):
    """Evaluate objective for coordinate i set to x, using precomputed data."""
    new_S = S - ai + x
    if new_S < 1e-15:
        return 1e10
    # Check affected convolution values
    max_c = fixed_max
    for k in range(max(0, i), min(2 * n - 1, i + n)):
        ki = k - i
        if 0 <= ki < n:
            if k == 2 * i:
                ck = b[k] - ai * ai + x * x
            else:
                ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            if ck > max_c:
                max_c = ck
    return 2.0 * n * max_c / (new_S * new_S)


def cd_optimize_coordinate(a, i, b, S):
    """Find optimal value for coordinate i using golden section search."""
    n = len(a)
    ai = a[i]
    
    # Precompute fixed max (convolution values not affected by changing a[i])
    lo_bound = max(0, i)
    hi_bound = min(2 * n - 2, i + n - 1)
    fixed_max = 0.0
    for k in range(2 * n - 1):
        if k < lo_bound or k > hi_bound:
            if b[k] > fixed_max:
                fixed_max = b[k]
    
    # Determine search bounds for x
    # Upper bound: where the quadratic term (if 2i < 2n-1) or linear terms hit fixed_max
    if 2 * i < 2 * n - 1 and fixed_max > 0:
        # x^2 term: b[2i] - ai^2 + x^2 = fixed_max => x = sqrt(fixed_max - b[2i] + ai^2)
        quad_val = fixed_max - b[2 * i] + ai * ai
        if quad_val > 0:
            x_quad = np.sqrt(quad_val)
        else:
            x_quad = 10.0
    else:
        x_quad = 10.0
    
    hi = max(3.0 * ai, S / n * 3.0, x_quad * 0.8, 0.1)
    lo = 0.0
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = cd_evaluate_x(c_pt, ai, a, b, i, n, fixed_max, S)
    fd = cd_evaluate_x(d_pt, ai, a, b, i, n, fixed_max, S)
    
    for _ in range(35):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = cd_evaluate_x(c_pt, ai, a, b, i, n, fixed_max, S)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = cd_evaluate_x(d_pt, ai, a, b, i, n, fixed_max, S)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def cd_pass(a, order=None):
    """One full pass of coordinate descent. Returns updated a and J."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)
    
    if order is None:
        order = np.arange(n)
    
    for i in order:
        new_val = cd_optimize_coordinate(a, i, b, S)
        if abs(new_val - a[i]) > 1e-12:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update convolution incrementally
            lo_k = max(0, i)
            hi_k = min(2 * n - 1, i + n - 1)
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
    
    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def cd_loop(a, n_passes, deadline, tracker):
    """Run multiple coordinate descent passes."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = cd_pass(a)
        if new_J < current_J - 1e-13:
            current_J = new_J
            tracker.update(a)
            if p % 3 == 0:
                print(f"    CD pass {p}: J={current_J:.8f} ({time.monotonic()-t0:.2f}s)", flush=True)
        else:
            if p > 2:
                break
    
    tracker.update(a)
    return a


def smooth_objective(a, beta):
    """Smooth approximation of J using log-sum-exp."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return 1e10
    n = len(a)
    b = fast_conv(a)
    # Log-sum-exp for numerical stability
    b_max = b.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b - b_max)))
    J_smooth = 2.0 * n * (log_sum_exp - b_max) / (S * S)  # This isn't quite right
    # Actually: J_smooth = 2n * (1/beta) * log(sum(exp(beta*b_k))) / S^2
    # = 2n * (b_max/beta + log(sum(exp(beta*(b_k - b_max))))/beta) / S^2
    # For the gradient, let's use: f(a) = (1/beta)*log(sum(exp(beta*b_k)))
    f = (1.0 / beta) * (b_max * beta + np.log(np.sum(np.exp(beta * (b - b_max)))))
    return 2.0 * n * f / (S * S)


def smooth_objective_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return 1e10, np.zeros_like(a)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp  # softmax weights
    
    f = (1.0 / beta) * (b_max * beta + np.log(sum_exp))
    J = 2.0 * n * f / (S * S)
    
    # Gradient of f w.r.t. a[i]:
    # df/da[i] = sum_k weights[k] * df_b_k/da[i]
    # df_b_k/da[i] = a[k-i] if 0 <= k-i < n (for k != 2i)
    #                = 2*a[i] if k == 2i
    # Actually: b[k] = sum_j a[j]*a[k-j]
    # db[k]/da[i] = a[k-i] (when 0<=k-i<n) + a[k-i] (when 0<=k-i<n, same term)
    # Wait: b[k] = sum_j a[j]*a[k-j]
    # db[k]/da[i]: the term a[i] appears when j=i (giving a[k-i]) and when k-j=i i.e. j=k-i (giving a[k-i])
    # If i == k-i (i.e. k=2i), these are the same term, so db[k]/da[i] = 2*a[i]
    # Otherwise db[k]/da[i] = 2*a[k-i] (if 0 <= k-i < n)
    
    grad_f = np.zeros(n)
    for k in range(2 * n - 1):
        ki = k - i
        # For each i, db[k]/da[i] = 2*a[|k-i|] if applicable
        pass
    
    # More efficient: for each i, grad_f[i] = sum_k weights[k] * db[k]/da[i]
    # db[k]/da[i] = 2*a[k-i] if 0<=k-i<n and k!=2i, or 2*a[i] if k=2i
    # Actually simpler: db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (when 0<=k-i<n)
    # But when k=2i, it's a[i]+a[i] = 2*a[i], which is the same formula!
    # So: db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    
    # grad_f[i] = sum_k weights[k] * 2*a[k-i] (where 0<=k-i<n)
    #          = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]  (for k = i+j, need i+j < 2n-1)
    
    for i in range(n):
        total = 0.0
        for j in range(n):
            k = i + j
            if k < 2 * n - 1:
                total += weights[k] * a[j]
        grad_f[i] = 2.0 * total
    
    # dJ/da[i] = 2n * (grad_f[i] * S^2 - f * 2*S) / S^4 = 2n * (grad_f[i] - 2*f/S) / S^2
    grad_J = 2.0 * n * (grad_f - 2.0 * f / S) / (S * S)
    
    return J, grad_J


def smooth_optimize(a, deadline, tracker, beta_start=10.0, beta_end=1000.0, n_steps=30):
    """Optimize using smooth (log-sum-exp) objective with continuation on beta."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Exponentially increase beta
        t_frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        # Use L-BFGS-B for several iterations at this beta
        for inner in range(5):
            if time.monotonic() > deadline:
                break
            J, grad = smooth_objective_and_grad(a, beta)
            
            # Manual gradient step with backtracking
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            
            # Try several step sizes
            best_a = None
            best_J = current_J
            for lr in [0.1 / gnorm, 0.05 / gnorm, 0.01 / gnorm, 0.005 / gnorm]:
                a_trial = a - lr * grad
                a_trial = np.maximum(a_trial, 0.001)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial
            
            if best_a is not None:
                a = best_a
                current_J = best_J
                tracker.update(a)
            else:
                break
        
        if step % 10 == 0:
            print(f"    Smooth step {step}: beta={beta:.1f}, J={compute_J(a):.8f}", flush=True)
    
    tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker, K=8, tighten=0.005):
    """LP mirror descent using active set (top K constraints)."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = active_set_lp(a, M_target, K=K)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = active_set_lp(a, M_target, K=K)
        if g is None:
            tighten *= 0.7
            if tighten < 1e-5:
                tighten = 0.005
            stall += 1
            if stall > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search: try multiple step sizes
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 1.0, 0.2, 0.8]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.001)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            stall += 1
            tighten *= 0.8
            if stall > 40:
                break
            continue
        
        a = best_a
        current_J = best_J
        stall = 0
        
        # Adaptively adjust
        improvement = M_target * tighten
        if improvement < 1e-8:
            tighten = max(0.001, tighten * 0.9)
        else:
            tighten = min(0.03, tighten * 1.02)
        
        if it % 20 == 0:
            tracker.update(a)
            print(f"    LP iter {it}: J={current_J:.8f} M={M:.6f}", flush=True)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def alternating_optimization(a, deadline, tracker, n_cycles=20):
    """Alternate between LP and coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for cyc in range(n_cycles):
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # LP phase
        lp_time = min(remaining * 0.5, 30)
        a = lp_mirror_descent(a, max_iters=20, deadline=time.monotonic() + lp_time,
                              tracker=tracker, K=8, tighten=0.004)
        
        if time.monotonic() > deadline:
            break
        
        # CD phase
        cd_time = deadline - time.monotonic()
        if cd_time > 5:
            n_passes = max(1, min(5, int(cd_time / 2)))
            a = cd_loop(a, n_passes=n_passes, deadline=time.monotonic() + cd_time - 2,
                        tracker=tracker)
        
        if cyc % 5 == 0:
            print(f"    Alt cycle {cyc}: J={compute_J(a):.8f}", flush=True)
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining smooth optimization, fast coordinate descent,
    active-set LP, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent found, using uniform n={parent.size}", flush=True)

    n_parent = parent.size

    # === Phase 1: Smooth optimization (gradient-based with continuation) ===
    print(f"\n{'='*50} Phase 1: Smooth Optimization (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    smooth_deadline = min(deadline - 800, t0 + 200)
    a = smooth_optimize(parent, smooth_deadline, tracker,
                        beta_start=20.0, beta_end=500.0, n_steps=20)
    tracker.report(f"Phase 1 Smooth: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Intensive coordinate descent ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 2: Fast Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        cd_deadline = min(deadline - 300, t0 + 300)
        a = cd_loop(a, n_passes=50, deadline=cd_deadline, tracker=tracker)
        tracker.report(f"Phase 2 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Active-set LP mirror descent ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: Active-Set LP {'='*50}", flush=True)
        t0 = time.monotonic()
        lp_deadline = min(deadline - 150, t0 + 200)
        a = lp_mirror_descent(a, max_iters=100, deadline=lp_deadline,
                              tracker=tracker, K=8, tighten=0.004)
        tracker.report(f"Phase 3 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Alternating LP + CD ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 4: Alternating LP+CD {'='*50}", flush=True)
        t0 = time.monotonic()
        a = alternating_optimization(tracker.best, deadline - 80, tracker, n_cycles=30)
        tracker.report(f"Phase 4 Alt: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try nearby lengths ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 5: Nearby Lengths {'='*50}", flush=True)
        t0 = time.monotonic()
        
        nearby_lengths = [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4,
                          n_parent - 6, n_parent + 6, n_parent + 8, n_parent - 8]
        nearby_lengths = [l for l in nearby_lengths if 200 <= l <= 600]
        seen = set()
        unique_lengths = []
        for l in nearby_lengths:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)
        
        for nl in unique_lengths:
            if time.monotonic() > deadline - 50:
                break
            
            a_resampled = resample_seq(tracker.best, nl)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()
            
            # Quick CD
            a_trial = cd_loop(a_resampled, n_passes=10,
                              deadline=time.monotonic() + 30, tracker=tracker)
            # Quick LP
            if time.monotonic() < deadline - 50:
                a_trial = lp_mirror_descent(a_trial, max_iters=15,
                                            deadline=time.monotonic() + 20,
                                            tracker=tracker, K=6, tighten=0.003)
            print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 5 Lengths: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation restarts ===
    if time.monotonic() < deadline - 40:
        print(f"\n{'='*50} Phase 6: Perturbation Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        
        for temp in [0.03, 0.02, 0.01, 0.005]:
            if time.monotonic() > deadline - 30:
                break
            
            a_pert = tracker.best.copy() + rng.normal(0, temp, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            
            a_pert = cd_loop(a_pert, n_passes=5,
                             deadline=time.monotonic() + 20, tracker=tracker)
        
        tracker.report(f"Phase 6 Restarts: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final intensive polish ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 7: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        
        # Final CD with many passes
        a_final = cd_loop(tracker.best, n_passes=30, deadline=deadline - 10,
                          tracker=tracker)
        # Final LP with tight parameters
        if time.monotonic() < deadline - 5:
            a_final = lp_mirror_descent(a_final, max_iters=30,
                                        deadline=deadline - 3,
                                        tracker=tracker, K=5, tighten=0.002)
        # One more CD
        if time.monotonic() < deadline - 1:
            a_final = cd_loop(a_final, n_passes=5, deadline=deadline - 0.5,
                              tracker=tracker)
        
        tracker.report(f"Phase 7 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
