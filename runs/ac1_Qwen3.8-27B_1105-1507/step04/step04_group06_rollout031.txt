    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    p = np.ones(m) / m
    current_max = fast_conv(a).max()
    best_max = current_max
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic reheat to escape local minima
        if it > 0 and it % reheat_every == 0:
            eta_d = min(eta_d * reheat_factor, 50.0)
            # Small random perturbation
            a = a + 0.001 * np.random.randn(n) * a
            a = np.maximum(a, 0)
            a = a / a.sum()

        # Compute autocorrelation
        b = fast_conv(a)
        b_max = b.max()

        if b_max < best_max:
            best_max = b_max
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)

        # Dual update: p ∝ p * exp(eta_d * b / b_max)
        # Use normalized b to keep exponents reasonable
        b_norm = b / max(b_max, 1e-30)
        log_update = eta_d * (b_norm - 1.0)
        log_update = np.clip(log_update, -100, 100)
        p = p * np.exp(log_update)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(m) / m
        else:
            p = p / p_sum

        # Compute force: force_i = sum_k p_k * 2*a[k-i] (correlation)
        force = 2.0 * np.correlate(p, a, mode='full')[:n]

        # Primal update: a_i ∝ a_i * exp(-eta_p * force_i / mean_force)
        force_mean = force.mean() + 1e-30
        log_update_p = -eta_p * (force / force_mean - 1.0)
        log_update_p = np.clip(log_update_p, -50, 50)
        a = a * np.exp(log_update_p)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-30:
            a = best_a.copy()
        else:
            a = a / a_sum

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def lbfgs_optimize(a0, deadline, betas=None, K=64):
    """L-BFGS-B with beta annealing for the smooth approximation."""
    if betas is None:
        betas = [(2.0, 30), (5.0, 40), (10.0, 50), (20.0, 50),
                 (50.0, 60), (100.0, 80), (200.0, 100), (500.0, 100)]

    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)
    a = a / a.sum()

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = J_and_grad(x2, beta, K)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = J_and_grad(x2, beta, K)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n (normalized to sum=1)."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = p / max(p.sum(), 1e-10)
        inits.append(p)
        p2 = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p2, 0.01) / max(np.maximum(p2, 0.01).sum(), 1e-10))
        p3 = p + rng.normal(0, 0.08, n) * p
        inits.append(np.maximum(p3, 0.01) / max(np.maximum(p3, 0.01).sum(), 1e-10))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (Hann window)
    w = 0.5 + 0.5 * np.cos(2.0 * np.pi * t_arr)
    inits.append(w / w.sum())

    # Hamming window
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * t_arr)
    inits.append(np.maximum(w, 0.01) / np.maximum(w, 0.01).sum())

    # Blackman window
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * t_arr) + 0.08 * np.cos(4.0 * np.pi * t_arr)
    inits.append(np.maximum(w, 0.01) / np.maximum(w, 0.01).sum())

    # Half-sine
    w = np.sin(np.pi * t_arr) + 0.1
    inits.append(w / w.sum())

    # Gaussian bump centered
    w = np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05
    inits.append(w / w.sum())

    # Two-peak
    w = 1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr)
    inits.append(np.maximum(w, 0.01) / np.maximum(w, 0.01).sum())

    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    noise = np.convolve(noise, kernel, mode='same')
    w = 1.0 + 0.3 * noise
    inits.append(np.maximum(w, 0.01) / np.maximum(w, 0.01).sum())

    # Exponential decay from center
    center = n // 2
    dist = np.abs(np.arange(n) - center) / center
    w = np.exp(-2.0 * dist) + 0.1
    inits.append(w / w.sum())

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def main_loop_mirror(a, deadline, tracker, n_eta_d_trials=None, n_eta_p_trials=None):
    """Run mirror descent with adaptive parameter search."""
    if n_eta_d_trials is None:
        n_eta_d_trials = [1.0, 2.0, 5.0, 10.0]
    if n_eta_p_trials is None:
        n_eta_p_trials = [0.1, 0.3, 0.5, 1.0]

    n = len(a)
    best_result = a.copy()
    best_max = fast_conv(a).max()

    # Try different parameter combinations
    time_per_comb = max(5.0, (deadline - time.monotonic()) / (len(n_eta_d_trials) * len(n_eta_p_trials)))

    for eta_d in n_eta_d_trials:
        for eta_p in n_eta_p_trials:
            if time.monotonic() > deadline:
                break
            sub_deadline = min(deadline, time.monotonic() + time_per_comb)
            a_try = a.copy()
            a_res = mirror_descent(a_try, sub_deadline, eta_d=eta_d, eta_p=eta_p,
                                   max_iters=200000, tracker=tracker)
            if a_res is not None:
                max_res = fast_conv(a_res).max()
                if max_res < best_max:
                    best_max = max_res
                    best_result = a_res.copy()

    return best_result


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
    
    Phase 1: L-BFGS-B with smooth objective (fast basin finding)
    Phase 2: Bidirectional mirror descent (main optimization, very fast)
    Phase 3: L-BFGS polish on mirror descent result
    Phase 4: Multi-restart with diverse perturbations
    Phase 5: Multi-length exploration
    Phase 6: Final fine-tuning
    """
    global _best_seq, _best_score
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()
    _t_start = time.monotonic()

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B quick basin finding (90s) ===
    t0 = time.monotonic()
    phase1_budget = 90.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'='*50} Phase 1: L-BFGS-B (budget {phase1_budget:.0f}s) {'='*50}", flush=True)

    for n in [1024, 512]:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), 5)

        for idx, a0 in enumerate(inits[:n_inits]):
            if time.monotonic() > deadline_p1:
                break
            a_opt = lbfgs_optimize(a0, deadline_p1,
                                   betas=[(5.0, 30), (20.0, 40), (50.0, 50), (100.0, 60)])
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Bidirectional Mirror Descent (600s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 600.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'='*50} Phase 2: Mirror Descent n={n_best} (budget {phase2_budget:.0f}s) {'='*50}", flush=True)

        a_work = resample_seq(tracker.best, n_best)
        a_work = a_work / max(a_work.sum(), 1e-10)

        # Run with different parameter sets
        a_work = main_loop_mirror(a_work, deadline_p2, tracker,
                                  n_eta_d_trials=[2.0, 5.0, 10.0, 20.0],
                                  n_eta_p_trials=[0.2, 0.5, 1.0, 2.0])

        # Additional runs from perturbed best
        for pert_scale in [0.05, 0.15]:
            if time.monotonic() > deadline_p2:
                break
            a_pert = tracker.best.copy().astype(np.float64)
            a_pert = a_pert + rng.normal(0, pert_scale, a_pert.size) * a_pert
            a_pert = np.maximum(a_pert, 0.01)
            a_pert = a_pert / a_pert.sum()
            sub_deadline = min(deadline_p2, time.monotonic() + 100.0)
            a_pert = mirror_descent(a_pert, sub_deadline, eta_d=5.0, eta_p=0.5,
                                    max_iters=100000, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: L-BFGS polish (200s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase3_budget = 200.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'='*50} Phase 3: L-BFGS Polish (budget {phase3_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        a_start = resample_seq(tracker.best, n_best)
        a_start = a_start / max(a_start.sum(), 1e-10)

        # Multiple L-BFGS runs with different beta schedules
        schedules = [
            [(10.0, 40), (30.0, 50), (100.0, 60), (300.0, 80), (1000.0, 100)],
            [(20.0, 30), (50.0, 40), (200.0, 50), (500.0, 60), (2000.0, 80)],
            [(50.0, 30), (100.0, 40), (500.0, 50), (1000.0, 60), (5000.0, 80)],
        ]

        for sched in schedules:
            if time.monotonic() > deadline_p3:
                break
            a_opt = lbfgs_optimize(a_start, deadline_p3, betas=sched, K=64)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                a_start = resample_seq(tracker.best, n_best)
                a_start = a_start / max(a_start.sum(), 1e-10)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-restart (250s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase4_budget = 250.0
        deadline_p4 = min(deadline, time.monotonic() + phase4_budget)
        print(f"\n{'='*50} Phase 4: Multi-Restart (budget {phase4_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        strategies = [
            ('gauss', 0.03), ('gauss', 0.05), ('gauss', 0.1),
            ('gauss', 0.02), ('spike', 5), ('spike', 10),
            ('smooth', 0.05), ('smooth', 0.1),
            ('edge', 0.1), ('alternating', 0.05),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline_p4:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gauss':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'spike':
                    k = min(int(param), n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    ks = max(3, n // 32)
                    kernel = np.ones(ks) / ks
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'edge':
                    # Perturb edges more
                    w = np.linspace(1, 3, n) * np.linspace(3, 1, n)
                    a = a + rng.normal(0, param, n) * a * w / w.max()
                elif pert_type == 'alternating':
                    sign = np.array([1 if (i % 2 == 0) else -1 for i in range(n)])
                    a = a + rng.normal(0, param, n) * sign * a

                a = np.maximum(a, 0.01)
                a = a / a.sum()

                # Quick mirror descent
                sub_deadline = min(deadline_p4, time.monotonic() + 15.0)
                a = mirror_descent(a, sub_deadline, eta_d=5.0, eta_p=0.5,
                                   max_iters=50000, tracker=tracker)
                # Quick L-BFGS
                if time.monotonic() < deadline_p4:
                    a = lbfgs_optimize(a, min(deadline_p4, time.monotonic() + 10.0),
                                       betas=[(20.0, 30), (100.0, 40), (500.0, 50)], K=48)
                    if a is not None and a.sum() > 0.01:
                        tracker.update(a)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 5: Multi-length exploration (150s) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase5_budget = 150.0
        deadline_p5 = min(deadline, time.monotonic() + phase5_budget)
        print(f"\n{'='*50} Phase 5: Length Exploration (budget {phase5_budget:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        lengths_to_try = []
        for n_try in [256, 384, 512, 768, 1024, 1280, 1536, 2048]:
            if abs(n_try - n_best) > 64 and n_try != n_best:
                lengths_to_try.append(n_try)
            elif n_try == n_best:
                continue
        # Sort by proximity to best
        lengths_to_try.sort(key=lambda x: abs(x - n_best))

        for n_try in lengths_to_try[:5]:
            if time.monotonic() > deadline_p5:
                break
            print(f"  Trying n={n_try}...", flush=True)
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            sub_deadline = min(deadline_p5, time.monotonic() + 25.0)
            a_res = mirror_descent(a_res, sub_deadline, eta_d=5.0, eta_p=0.5,
                                   max_iters=100000, tracker=tracker)
            if time.monotonic() < deadline_p5:
                a_res = lbfgs_optimize(a_res, min(deadline_p5, time.monotonic() + 10.0),
                                       betas=[(20.0, 30), (100.0, 40)], K=48)
                if a_res is not None and a_res.sum() > 0.01:
                    tracker.update(a_res)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final fine-tuning (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        print(f"\n{'='*50} Phase 6: Final Polish (budget {remaining:.0f}s) {'='*50}", flush=True)

        n_best = tracker.best.size
        a_start = resample_seq(tracker.best, n_best)
        a_start = a_start / max(a_start.sum(), 1e-10)

        # Very fine mirror descent
        sub_deadline = min(deadline, time.monotonic() + remaining * 0.4)
        a_start = mirror_descent(a_start, sub_deadline, eta_d=3.0, eta_p=0.3,
                                 max_iters=300000, reheat_every=10000,
                                 reheat_factor=1.1, tracker=tracker)

        # Final L-BFGS with high beta
        if time.monotonic() < deadline:
            a_start = lbfgs_optimize(a_start, deadline,
                                     betas=[(100.0, 50), (500.0, 60), (2000.0, 80), (10000.0, 100)],
                                     K=128)
            if a_start is not None and a_start.sum() > 0.01:
                tracker.update(a_start)

        # One more mirror descent from absolute best
        if tracker.best is not None and time.monotonic() < deadline:
            a_final = resample_seq(tracker.best, tracker.best.size)
            a_final = a_final / max(a_final.sum(), 1e-10)
            a_final = mirror_descent(a_final, deadline, eta_d=2.0, eta_p=0.2,
                                     max_iters=200000, reheat_every=20000,
                                     reheat_factor=1.05, tracker=tracker)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
