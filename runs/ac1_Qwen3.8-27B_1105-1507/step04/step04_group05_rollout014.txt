```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using exact QP formulation.
The key insight is that for fixed n, the problem is a convex QP:

  subject to (a*a)_k <= t for all k

Solving this gives the global optimum for that n, and we search over n.

import numpy as np
import cvxpy as cp
import time
import math


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    t = cp.Variable()

        expr = 0

        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            expr += a[j] * a[k - j]
        constraints.append(expr <= t)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using exact convex QP formulation.
Key insight: for fixed n, the problem is a convex QP (minimize t s.t. (a*a)_k <= t,
sum(a)=1, a>=0). Solving this gives the global optimum for that n. We search over n
and combine with iterative refinement (LP mirror descent + coordinate descent) for
larger n where the QP is harder to solve.
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def solve_qp_for_n(n, solver=cp.CLARABEL):
    """Solve the convex QP for given sequence length n.
    minimize t s.t. (a*a)_k <= t for all k, sum(a)=1, a>=0
    """
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()

    constraints = [cp.sum(a) == 1]

    for k in range(2 * n - 1):
        expr = cp.Constant(0)
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            expr = expr + a[j] * a[k - j]
        constraints.append(expr <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver)

    if a.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
        a_val = np.maximum(a.value, 0)
        t_val = float(t.value)
        return a_val, t_val
    return None, None


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A @ g <= M (convolution constraints)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tighten_base=0.003):
    """LP mirror descent iteration."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tighten = tighten_base
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-7:
                break
            continue
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.4 * (1.0 - progress))
        best_a = None
        best_J = current_J
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.8 * base_t]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall += 1
            if stall > 15:
                break
            tighten *= 0.7
            continue

        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.03, tighten * 0.98)

    return a


def coordinate_descent_pass(a):
    """One pass of coordinate descent with 1D optimization per coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)

    indices = list(range(n))
    np.random.shuffle(indices)

    for i in indices:
        ai = a[i]
        if ai < 1e-12:
            continue

        # Binary search + golden section for optimal value of a[i]
        def f_at_x(x):
            new_S = S - ai + x
            if new_S < 1e-8:
                return 1e10
            max_c = 0.0
            # Only need to check affected convolution entries
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        ck = b[k] - ai * ai + x * x
                    else:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                else:
                    ck = b[k]
                if ck > max_c:
                    max_c = ck
            # Also check unaffected entries for the global max
            for k in range(max(0, i + n), 2 * n - 1):
                if b[k] > max_c:
                    max_c = b[k]
            for k in range(0, min(2 * n - 1, max(0, i))):
                if b[k] > max_c:
                    max_c = b[k]
            return 2.0 * n * max_c / (new_S * new_S)

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = max(3.0 * ai, S / n * 3.0, 0.01)

        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        fc = f_at_x(c_pt)
        fd = f_at_x(d_pt)

        for _ in range(30):
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = f_at_x(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = f_at_x(d_pt)

        x_opt = max(0.0, (lo + hi) / 2.0)
        if abs(x_opt - ai) > 1e-12:
            a[i] = x_opt
            S = S - ai + x_opt
            # Update convolution
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai * ai + x_opt * x_opt
                    else:
                        b[k] = b[k] - 2.0 * ai * a[ki] + 2.0 * x_opt * a[ki]

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        update_best(a)
        print(f"  Final CD: J={best_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(400)
        best_J = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.10f}, n={best_seq.size}", flush=True)
    print(f"Total time: {time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
```
