    """Build sparse matrix A where row k represents (a*g)[k] = sum_j A[k,j]*g[j]."""
    rows = np.zeros(0, dtype=int)
    cols = np.zeros(0, dtype=int)

            all_cols.append(j)

"""

Combines LP dual-variable-guided multiplicative updates, accelerated subgradient
descent, CMA-ES on Fourier parameterization, multi-length search, and

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j] if valid."""
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            all_rows.append(k)
            all_cols.append(j)
            all_vals.append(a[k - j])
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n))


def solve_lp_with_dual(a, M, cache_key=None):
    """Solve LP: max sum(g) s.t. (a*g)[k]<=M, g>=0. Returns (g, lambda_k)."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            lambda_k = None
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                try:
                    lambda_k = np.maximum(np.asarray(r.ineqlin.marginals, dtype=np.float64), 0)
                except Exception:
                    lambda_k = None
            return g, lambda_k
    except Exception:
        pass
    return None, None


def dual_pressure(a, lambda_k):
    """Compute pressure p[i] = sum_k lambda_k * a[k-i] for valid indices."""
    n = len(a)
    p = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(lambda_k)):
        lk = lambda_k[k]
        if lk < 1e-14:
            continue
        ki = k - idx
        valid = (ki >= 0) & (ki < n)
        p[valid] += lk * a[ki[valid]]
    return p


def dual_multiplicative_update(a, lambda_k, t=1.0):
    """KKT-guided multiplicative update using dual variables."""
    n = len(a)
    p = dual_pressure(a, lambda_k)
    p = np.maximum(p, 1e-12)
    # We want a_new[i] proportional to a[i] / p[i] (damped by t)
    ratio = np.power(1.0 / p, t)
    ratio = ratio / ratio.max()  # normalize ratio
    a_new = a * ratio
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_mirror_descent_v2(a, max_iters, deadline, use_dual=True):
    """
    Enhanced LP mirror descent.
    Uses LP direction and/or dual-guided multiplicative update.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0
    momentum = np.zeros_like(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g, lambda_k = solve_lp_with_dual(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Build candidate directions
        directions = [g]
        if use_dual and lambda_k is not None and np.sum(lambda_k) > 1e-12:
            a_dual_t1 = dual_multiplicative_update(a, lambda_k, t=1.0)
            a_dual_t05 = dual_multiplicative_update(a, lambda_k, t=0.5)
            directions.append(a_dual_t1)
            directions.append(a_dual_t05)

        best_trial_J = current_J
        best_trial_a = None

        for direction in directions:
            if time.monotonic() > deadline:
                break
            t = 0.8
            for ls in range(25):
                a_trial = (1.0 - t) * a + t * direction
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    t *= 0.5
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J - 1e-13:
                    best_trial_J = J_trial
                    best_trial_a = a_trial.copy()
                    break
                t *= 0.65
                if t < 0.003:
                    break

        if best_trial_a is not None:
            # Add small momentum
            momentum = 0.7 * momentum + 0.3 * (best_trial_a - a)
            a = best_trial_a + 0.05 * momentum
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            current_J = compute_J(a)
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                break

    return best_a, best_J


def accelerated_subgradient(a, max_iters, deadline):
    """
    Accelerated subgradient method using LP dual weights for the gradient.
    Uses Nesterov-like momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    y = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    step_scale = 2.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_y = fast_conv(y)
        M_y = float(b_y.max())

        # Get subgradient using LP dual weights
        g_lp, lambda_k = solve_lp_with_dual(y, M_y)
        grad = np.zeros(n)

        if lambda_k is not None and np.sum(lambda_k) > 1e-12:
            # Weighted subgradient from dual
            lam_norm = lambda_k / np.sum(lambda_k)
            p = dual_pressure(y, lam_norm)
            grad = 2.0 * p
        else:
            # Fallback: uniform weights over active set
            tol = 1e-9 * M_y
            active = np.where(b_y >= M_y - tol)[0]
            for k in active:
                ki = k - np.arange(n)
                valid = (ki >= 0) & (ki < n)
                grad[valid] += 2.0 * y[ki[valid]]
            grad /= max(1, len(active))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        t_step = step_scale / (gnorm * np.sqrt(it + 1.0))
        t_step = min(t_step, 0.5)

        # Momentum update
        momentum = beta * momentum + (1 - beta) * (grad / gnorm)
        a_new = y - t_step * momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Nesterov-like update for y
        t_next = (1 + it) / (4 + it)  # ~ 1/(4*sqrt) schedule
        y = a_new + t_next * (a_new - a)
        y = np.maximum(y, 0)
        y = y / y.sum()
        a = a_new

        new_J = compute_J(a)
        if new_J < best_J:
            best_J = new_J
            best_a = a.copy()

    return best_a, best_J


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to non-negative sequence."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        return np.ones(n) / n
    return a / s


def seq_to_fourier(a, n_harm):
    """Extract Fourier coefficients from sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def cmaes_fourier_search(a_init, n_harm, max_gens, deadline, tracker):
    """CMA-ES on Fourier parameters with adaptive sigma."""
    n = len(a_init)
    dim = 2 * n_harm + 1
    lam = max(8, 4 + int(3 * np.log(dim)))
    mu = lam // 2

    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights /= weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))

    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)

    x = seq_to_fourier(a_init, n_harm)
    sigma = 0.03

    best_J = compute_J(a_init)
    best_x = x.copy()
    tracker.update(a_init)

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        # Generate offspring
        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            eigvals, eigvecs = np.linalg.eigh(C)
            eigvals = np.maximum(eigvals, 1e-14)
            y = (eigvecs * np.sqrt(eigvals)) @ (eigvecs.T @ z)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)

        # Selection
        idx = np.argsort(offspring_J)
        top_idx = idx[:mu]
        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * offspring[top_idx[i]]

        # Update sigma
        diff = (x_new - x) / (sigma * chi_N) if sigma > 1e-30 else np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(0.25 * c_sigma / d_sigma * (np.linalg.norm(p_sigma) ** 2 / chi_N ** 2 - 1))
        sigma = np.clip(sigma, 1e-8, 0.5)

        # Update covariance
        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-10)

        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = offspring[top_idx[i]] - x
            rank_mu += weights[i] * np.outer(d, d)

        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + (1 - h_sigma) * c_c * np.outer(p_c, p_c) + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c)) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + (1 - h_sigma) * c_c * np.outer(p_c, p_c) + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c)) + c_mu * rank_mu
        C = (C + C.T) / 2
        x = x_new

        # Track best
        if offspring_J[top_idx[0]] < best_J:
            best_J = offspring_J[top_idx[0]]
            best_x = offspring[top_idx[0]].copy()
            a_best = fourier_to_seq(best_x, n, n_harm)
            tracker.update(a_best)

    return best_x, best_J, fourier_to_seq(best_x, n, n_harm)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, ptype):
    """Apply structured perturbation."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    if ptype == 'shift':
        s = int(rng.integers(1, max(2, n // 8)))
        a_p = np.roll(a, s)
    elif ptype == 'modulate':
        freq = rng.uniform(0.5, 6.0)
        amp = rng.uniform(0.05, 0.2)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif ptype == 'smooth':
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif ptype == 'antismooth':
        # Sharpen: subtract smooth version
        k = int(rng.integers(3, max(4, n // 8)))
        kernel = np.ones(k) / k
        smooth = np.convolve(a, kernel, mode='same')
        a_p = a * 1.5 - 0.5 * smooth
    elif ptype == 'phase':
        # Add a small phase-shifted copy
        s = int(rng.integers(1, max(2, n // 16)))
        a_p = a + 0.3 * np.roll(a, s)
    else:
        a_p = a + rng.normal(0, 0.02, n)
    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences for the given length."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    # Structured candidates
    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t + 0.5), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.maximum(1.0 + 0.4 * np.cos(4 * np.pi * t), 0.05))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(6 * np.pi * t + 1.0), 0.05))
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.05))

    # Smooth random
    k = max(3, n // 12)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.3, 0.05))

    # Symmetric random
    half = n // 2
    rh = np.maximum(rng.uniform(0, 1, half) + 0.3, 0.1)
    sym = np.concatenate([rh, rh[::-1]])
    if len(sym) < n:
        sym = np.concatenate([sym, sym[:n - len(sym)]])
    inits.append(np.maximum(sym[:n], 0.05))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001).astype(np.float64)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):

        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  Current best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def annealing_restart(a, rng, temperature):
    """Simulated annealing-style perturbation."""
    n = len(a)
    a_new = a.copy()
    # Random component-wise perturbation
    noise = rng.normal(0, temperature, n)
    a_new = a_new + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s
