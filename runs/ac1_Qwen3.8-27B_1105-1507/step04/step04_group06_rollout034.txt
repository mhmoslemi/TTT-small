"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with softmax parameterization (handles non-negativity and scale invariance)
2. Log-sum-exp smoothed objective with adaptive beta annealing
3. FFT-based fast gradient computation (O(n log n) per step)
4. Multi-restart strategy with diverse perturbations
5. Nesterov-like momentum for accelerated convergence
6. Length exploration around best found solution
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_of_z(z):
    """Convert z to a = softmax(z) = exp(z - max(z)) / sum(exp(z - max(z)))."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    total = a.sum()
    if total < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / total


def smooth_obj_and_grad(a, beta):
    """
    Compute smooth approximation of J(a) and its gradient.
    Uses log-sum-exp to smooth the max operation.
    Returns (J_smooth, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
        G = b_max
    else:
        p = exp_shifted / Z
        G = b_max + np.log(Z) / beta

    S = float(a.sum())
    J = 2.0 * n * G / (S * S)

    # Gradient: dJ/da_i = 2n/S^3 * (S * dG/da_i - 2*G)
    # dG/da_i = sum_k p_k * db_k/da_i = sum_k p_k * 2*a_{k-i}
    # Compute cross-correlation of p and a for efficiency
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    cross = np.real(np.fft.ifft(P * np.conj(A)))[:n]
    dG = 2.0 * cross
    grad = 2.0 * n * (S * dG - 2.0 * G) / (S ** 3)

    return J, grad


def grad_to_z_space(grad_a, a):
    """Convert gradient from a-space to z-space given a = softmax(z)."""
    mean_grad = float(np.dot(grad_a, a))
    return a * (grad_a - mean_grad)


def adam_z_optimize(z0, deadline, beta_schedule, lr, stagnation_limit, tracker=None, label=""):
    """
    Adam optimizer over z where a = softmax(z).
    Uses log-sum-exp smoothed objective with beta annealing.
    """
    n = len(z0)
    z = z0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    a_curr = softmax_of_z(z)
    J_curr = compute_J(a_curr)
    best_a = a_curr.copy()
    best_J = J_curr
    stagnation = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    n_steps_total = 0

    if tracker is not None:
        tracker.update(a_curr)

    for beta, max_steps in beta_schedule:
        if time.monotonic() > deadline:
            break
        steps_in_phase = 0
        for step in range(max_steps):
            if time.monotonic() > deadline:
                break
            t += 1
            steps_in_phase += 1
            n_steps_total += 1

            J_smooth, grad_a = smooth_obj_and_grad(a_curr, beta)
            if not np.isfinite(J_smooth):
                break
            grad_z = grad_to_z_space(grad_a, a_curr)
            grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step_size = lr * m_hat / (np.sqrt(v_hat) + eps)
            z_new = z - step_size
            a_new = softmax_of_z(z_new)
            J_new = compute_J(a_new)

            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                if tracker is not None:
                    tracker.update(a_new)

            if J_new <= J_curr + 1e-14:
                z = z_new
                a_curr = a_new
                J_curr = J_new
                stagnation = 0
            else:
                # Try a smaller step
                z_half = z - 0.5 * step_size
                a_half = softmax_of_z(z_half)
                J_half = compute_J(a_half)
                if J_half < J_curr:
                    z = z_half
                    a_curr = a_half
                    J_curr = J_half
                    if J_half < best_J:
                        best_J = J_half
                        best_a = a_half.copy()
                        if tracker is not None:
                            tracker.update(a_half)
                    stagnation = 0
                else:
                    stagnation += 1

            if stagnation >= stagnation_limit:
                break

    return best_a, best_J

def a_to_z(a):
    """Convert a positive sequence to z-space (inverse softmax)."""
    a = np.maximum(a, 1e-15)
    a_norm = a / a.sum()
    return np.log(a_norm)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        # Perturbed parent
        p_pert = p + rng.normal(0, 0.05, n) * p
        p_pert = np.maximum(p_pert, 1e-4)
        inits.append(p_pert / p_pert.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay (asymmetric - mass on left)
    for rate in [0.01, 0.02, 0.03, 0.05]:
        e = np.exp(-rate * np.arange(n))
        inits.append(e / e.sum())

    # Exponential growth (mass on right)
    for rate in [0.01, 0.02, 0.03]:
        e = np.exp(rate * (np.arange(n) - n))
        inits.append(e / e.sum())

    # Power law
    for p in [0.5, 1.0, 2.0]:
        pl = (np.arange(n) + 1) ** (-p)
        inits.append(pl / pl.sum())

    # Cosine families (asymmetric)
    for freq in [1, 2, 3]:
        c = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Gaussian off-center
    for center in [0.2, 0.3, 0.7, 0.8]:
        g = np.exp(-20.0 * (t_arr - center) ** 2) + 0.01
        inits.append(g / g.sum())

    # Two-peak asymmetric
    two = np.exp(-30*(t_arr-0.2)**2) + 0.5*np.exp(-30*(t_arr-0.7)**2) + 0.01
    inits.append(two / two.sum())

    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Smoothed random
    r2 = rng.uniform(0.3, 1.0, n)
    kernel_size = max(3, n // 128)
    kernel = np.ones(kernel_size) / kernel_size
    r2 = np.convolve(r2, kernel, mode='same')
    inits.append(r2 / r2.sum())

    return inits


def perturb_z(z, rng, pert_type, param):
    """Perturb z-space vector for restarts."""
    n = len(z)
    z_new = z.copy()
    if pert_type == 'gaussian':
        z_new += rng.normal(0, param, n)
    elif pert_type == 'uniform':
        z_new += rng.uniform(-param, param, n)
    elif pert_type == 'spike':
        k = min(max(1, int(param * n)), n // 4)
        idx = rng.choice(n, size=k, replace=False)
        z_new[idx] += rng.uniform(-param, param, k)
    elif pert_type == 'smooth_noise':
        noise = rng.normal(0, param, n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        z_new += noise
    elif pert_type == 'scale':
        z_new = z * param
    return z_new


def exact_subgradient_step(a, max_iter=50, lr=0.001):
    """
    Exact subgradient descent (non-smooth) for final polishing.
    Uses the actual active set of max(conv(a,a)).
    """
    n = len(a)
    a = a.copy()
    S = a.sum()
    a = a / S

    for it in range(max_iter):
        b = fast_conv(a)
        b_max = b.max()
        # Active set: indices where b_k is within epsilon of max
        active = np.where(b >= b_max - 1e-10 * max(1.0, b_max))[0]
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)

        # Gradient: dJ/da_i = 2n/S^3 * (S * dG/da_i - 2G)
        # With exact G = b_max, dG/da_i = sum_k p_k * 2*a_{k-i}
        N = 1
        while N < 3 * n:
            N <<= 1
        P = np.fft.fft(p, N)
        A = np.fft.fft(a, N)
        cross = np.real(np.fft.ifft(P * np.conj(A)))[:n]
        dG = 2.0 * cross
        S = a.sum()
        grad = 2.0 * n * (S * dG - 2.0 * b_max) / (S ** 3)

        # Gradient step with projection
        a_new = np.maximum(a - lr * grad, 0.0)
        if a_new.sum() < 1e-6:
            break
        a_new = a_new / a_new.sum()

        J_old = 2.0 * n * b_max / (S * S)
        J_new = compute_J(a_new)
        if J_new < J_old:
            a = a_new
            S = a.sum()
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return a

        phase2_budget = min(500.0, deadline - time.monotonic())

        print(f"\n{'='*50} Phase 2: Extended Opt n={n_best} (budget {phase2_budget:.0f}s) {'='*50}", flush=True)

        # Resample to n_base for optimization
        a_work = resample_seq(tracker.best, n_base)
        a_work = np.maximum(a_work, 1e-10)
        a_work = a_work / a_work.sum()
        z_work = a_to_z(a_work)

        # Full beta annealing
        beta_schedule_full = [
            (1.0, 1000),
            (3.0, 1000),
            (10.0, 1500),
            (30.0, 2000),
            (100.0, 2000),
            (300.0, 2000),
            (1000.0, 2000),
        ]
        a_opt, J_opt = adam_z_optimize(z_work, deadline_p2, beta_schedule_full,
                                         lr=0.03, stagnation_limit=150,
                                         tracker=tracker, label="ext")

        if a_opt is not None:
            tracker.update(a_opt)

        # Second pass from best
        if time.monotonic() < deadline_p2:
            a_work = resample_seq(tracker.best, n_base)
            a_work = np.maximum(a_work, 1e-10)
            a_work = a_work / a_work.sum()
            z_work = a_to_z(a_work)
            beta_schedule_full2 = [(50.0, 2000), (200.0, 2000), (500.0, 2000), (2000.0, 2000)]
            a_opt2, J_opt2 = adam_z_optimize(z_work, deadline_p2, beta_schedule_full2,
                                               lr=0.02, stagnation_limit=100,
                                               tracker=tracker, label="ext2")
            if a_opt2 is not None:
                tracker.update(a_opt2)

        tracker.report(f"Phase 2 complete ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        phase3_budget = min(400.0, deadline - time.monotonic())
        deadline_p3 = time.monotonic() + phase3_budget
        print(f"\n{'='*50} Phase 3: Multi-Restart (budget {phase3_budget:.0f}s) {'='*50}", flush=True)

        perturbations = [
            ('gaussian', 0.3),
            ('gaussian', 0.15),
            ('gaussian', 0.5),
            ('uniform', 0.3),
            ('smooth_noise', 0.3),
            ('spike', 0.05),
            ('gaussian', 0.1),
            ('uniform', 0.5),
            ('smooth_noise', 0.5),
            ('spike', 0.1),
            ('gaussian', 0.8),
            ('uniform', 0.8),
        ]

        attempt = 0
        for pert_type, param in perturbations:
            for trial in range(8):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                z_base = a_to_z(tracker.best)
                z_pert = perturb_z(z_base, rng, pert_type, param)

                # Quick check with moderate beta
                a_check = softmax_of_z(z_pert)
                J_check = compute_J(a_check)
                if J_check < tracker.best_score:
                    tracker.update(a_check)

                # Adam refinement
                beta_sched = [(20.0, 800), (100.0, 1000), (500.0, 1000)]
                a_opt, J_opt = adam_z_optimize(z_pert, deadline_p3, beta_sched,
                                                 lr=0.03, stagnation_limit=60,
                                                 tracker=tracker, label=f"r{attempt}")
                if a_opt is not None:
                    tracker.update(a_opt)

        tracker.report(f"Phase 3 complete ({time.monotonic()-t0:.1f}s, {attempt} attempts)")

    # === Phase 4: Exact subgradient polishing ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        phase4_budget = min(150.0, deadline - time.monotonic())
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'='*50} Phase 4: Exact Polish (budget {phase4_budget:.0f}s) {'='*50}", flush=True)

        a_work = tracker.best.copy().astype(np.float64)
        a_work = np.maximum(a_work, 1e-10)
        a_work = a_work / a_work.sum()

        # Multiple exact subgradient passes with decreasing step size
        for lr_pg in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if time.monotonic() > deadline_p4:
                break
            a_polished = exact_subgradient_step(a_work, max_iter=200, lr=lr_pg)
            J_polished = compute_J(a_polished)
            if J_polished < tracker.best_score:
                tracker.update(a_polished)
                a_work = a_polished
            else:
                break

        # Very fine Adam polish with high beta
        if time.monotonic() < deadline_p4:
            z_work = a_to_z(tracker.best)
            beta_sched_fine = [(1000.0, 3000), (5000.0, 3000), (20000.0, 3000)]
            a_opt, J_opt = adam_z_optimize(z_work, deadline_p4, beta_sched_fine,
                                             lr=0.005, stagnation_limit=50,
                                             tracker=tracker, label="fine")
            if a_opt is not None:
                tracker.update(a_opt)

        tracker.report(f"Phase 4 complete ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Length exploration ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n{'='*50} Phase 5: Length Exploration ({remaining:.0f}s) {'='*50}", flush=True)

            n_best = tracker.best.size
            lengths_to_try = []
            for n_try in [512, 768, 1024, 1280, 1536, 2048]:
                if n_try != n_best and abs(n_try - n_best) >= 64:
                    lengths_to_try.append(n_try)

            for n_try in lengths_to_try:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = np.maximum(a_res, 1e-10)
                a_res = a_res / a_res.sum()
                z_res = a_to_z(a_res)

                budget_here = min(60.0, deadline - time.monotonic())
                beta_sched = [(10.0, 1000), (50.0, 1500), (200.0, 1500)]
                a_opt, J_opt = adam_z_optimize(z_res, time.monotonic() + budget_here,
                                                 beta_sched, lr=0.03, stagnation_limit=60,
                                                 tracker=tracker, label=f"n{n_try}")
                if a_opt is not None:
                    tracker.update(a_opt)

            tracker.report(f"Phase 5 complete ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
