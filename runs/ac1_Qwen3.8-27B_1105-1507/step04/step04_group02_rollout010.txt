import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def build_conv_matrix(a):
    """Build sparse matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    # conv(a,g)[k] = sum_i a[i]*g[k-i]
    # So T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    T = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return T

def solve_alternating_lp(a, M, n_trials=3):
    """Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    best_g = None
    best_obj = -np.inf
    
    for trial in range(n_trials):
        M_try = M * (1.0 + 0.02 * trial)
        try:
            res = linprog(c, A_ub=T, b_ub=np.full(2 * n - 1, M_try),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s = np.sum(g)
                if s > 1e-10 and s > best_obj:
                    best_g = g / s
                    best_obj = s
        except Exception:
            pass
    
    return best_g

def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S

def interpolate_sequence(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    try:
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f(x_new), 0)
    except Exception:
        a_new = np.ones(new_n) / new_n
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / S

            if obj_lp < best_obj:
                best_a = a_lp.copy()
                best_obj = obj_lp
        
        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for scale in [0.03, 0.06, 0.1]:
                if time.monotonic() - start > time_budget - 5:
                    break
                a_p = best_a * np.exp(rng.normal(0, scale, n))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p < 1e-10:
                    continue
                a_p = a_p / S_p
                # Quick refine
                a_p = active_set_subgradient(a_p, max_iters=500, time_budget=3.0)
                obj_p = compute_obj(a_p)
                if obj_p < best_obj:
                    best_a = a_p.copy()
                    best_obj = obj_p
    
    return best_a, best_obj
