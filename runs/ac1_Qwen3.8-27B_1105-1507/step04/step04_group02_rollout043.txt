import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0*n*M/(S*S)

def cross_corr_fft(a, p):
    """Compute h[i] = sum_j a[j]*p[j+i] for i=0,...,len(a)-1 using FFT."""
    n = len(a)
    N = 1
    while N < 3*n - 2:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    # We want h[i] = sum_j a[j]*p[j+i]
    # Using: h = convolve(flip(a), p)[n-1:2n-1]
    fa_flip = np.fft.rfft(np.flip(a), N)
    h = np.fft.irfft(fa_flip * fp, N)
    return h[n-1:2*n-1]

def lse_value_grad(b, tau):
    """Compute LSE value and gradient for normalized b (sum=1)."""
    n = len(b)
    conv = compute_conv(b)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    log_sum = np.log(np.sum(np.exp(shifted)))
    lse = tau * (max_c/tau + log_sum)
    obj_val = 2.0 * n * lse
    
    p = np.exp(shifted)
    p /= np.sum(p)
    
    h = cross_corr_fft(b, p)
    grad = 4.0 * n * h
    return obj_val, grad, conv

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_descent(b, tau_schedule, n_iters, verbose=False):
    """LSE-smoothed gradient descent with projection to simplex."""
    n = len(b)
    b = b.copy()
    b = project_simplex(b)
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    
    for it in range(n_iters):
        tau = tau_schedule(it)
        obj_val, grad, conv = lse_value_grad(b, tau)
        
        # Backtracking line search
        alpha = 1.0
        for ls in range(30):
            b_new = project_simplex(b - alpha * grad)
            obj_new = compute_obj(b_new)
            if obj_new < obj_val * 0.999 or alpha < 1e-12:
                break
            alpha *= 0.5
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
        
        if verbose and it % 100 == 0:
            print(f"    LSE it={it}, tau={tau:.5f}, obj={compute_obj(b):.6f}, best={best_obj:.6f}")
    
    return best_b, best_obj

**Coordinate Descent:**
For coordinate i, given all other coordinates fixed, the objective as a function of a_i is:

Let c_k be the convolution value at position k, where c_k = a_i * a_{k-i} + D_k for k where 0<=k-i<n and k≠2i, and c_{2i} = a_i^2 + D_{2i}.

The max is max(max_k (a_i * a_{k-i} + D_k) for valid k≠2i, a_i^2 + D_{2i}, max_k D_k for invalid k).

This is a piecewise-linear + quadratic function of a_i (specifically, the max of several lines and one parabola). It's convex in a_i.

To minimize: find the a_i >= 0 that minimizes this. We can do this by:
1. The function is convex, so the minimum is where the subgradient contains 0.
2. We can use bisection on the derivative (which is piecewise: the slope of the dominant linear term, or 2*a_i for the quadratic).

Actually, this is easier than I thought. The derivative of the max function at a_i is:
- If the max is achieved at a linear term: derivative = a_{k-i} (the coefficient of a_i in that term)
- If the max is achieved at the quadratic term: derivative = 2*a_i
- If multiple terms are tied: subgradient is the convex hull

The minimum is where the derivative changes from negative to positive (or the subgradient contains 0).

For implementation, I can use a simple approach:
1. Evaluate the objective at a_i = 0
2. Find the "break points" where the argmax changes
3. Search for the minimum among the candidate values

3. Multi-start with diverse n values and smooth initializations

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def normalize(a):
    S = np.sum(a)
    if S < 1e-10:
        return a
    return a / S


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def cross_corr_fft(a, p):
    """Compute h[i] = sum_j a[j]*p[j+i] for i=0,...,len(a)-1."""
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa_flip = np.fft.rfft(np.flip(a), N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fa_flip * fp, N)
    return h[n - 1:2 * n - 1]


def lse_value_grad(b, tau):
    """Compute LSE-smoothed objective value and gradient for normalized b."""
    n = len(b)
    conv = compute_conv(b)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    log_sum = np.log(np.sum(np.exp(shifted)))
    lse = tau * (max_c / tau + log_sum)
    obj_val = 2.0 * n * lse
    p = np.exp(shifted)
    p /= np.sum(p)
    h = cross_corr_fft(b, p)
    grad = 4.0 * n * h
    return obj_val, grad, conv

                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            S = np.sum(g)
            if S > 1e-10:
                g = g / S
            return g, res.x[n]
    except Exception:
        pass
    return None, np.inf


def lp_alternating(a, max_iters=60, time_budget=120.0):
    """LP alternating: minimize peak of conv(a,a) by alternating LP solves."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: find g minimizing max(conv(a, g))
        g, t_val = solve_lp_minpeak(a, time_limit=6)
        if g is None:
            break

        # Line search on blending parameter
        cur_obj = compute_obj(a)
        best_t = 0.0
        best_obj_ls = cur_obj
        for tau in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - tau) * a + tau * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls - 1e-12:
                best_obj_ls = obj_try
                best_t = tau

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            if best_obj_ls < best_obj:
                best_obj = best_obj_ls
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

        # Now solve the dual: find new a minimizing max(conv(g, a))
        a_new, t_val2 = solve_lp_minpeak(g, time_limit=6)
        if a_new is not None:
            best_t2 = 0.0
            best_obj_ls2 = best_obj
            for tau in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - tau) * a + tau * a_new
                a_try = np.maximum(a_try, 0)
                S_try = np.sum(a_try)
                if S_try < 1e-10:
                    continue
                a_try = a_try / S_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_ls2 - 1e-12:
                    best_obj_ls2 = obj_try
                    best_t2 = tau
            if best_t2 > 0:
                a = (1.0 - best_t2) * a + best_t2 * a_new
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                if best_obj_ls2 < best_obj:
                    best_obj = best_obj_ls2
                    best_a = a.copy()

    return best_a, best_obj


def lse_descent(b, time_budget=30.0, tau_start=0.05, tau_end=0.001):
    """LSE-smoothed gradient descent with adaptive step size."""
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    it = 0
    alpha = 0.5 / n

    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, it / max(1, int(time_budget * 10)))
        tau = tau_start * (tau_end / tau_start) ** progress

        obj_val, grad, conv = lse_value_grad(b, tau)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Try a few step sizes (backtracking)
        best_step = 0.0
        best_obj_ls = compute_obj(b)
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            b_new = b - scale * alpha * grad_centered
            b_new = project_simplex(b_new)
            obj_new = compute_obj(b_new)
            if obj_new < best_obj_ls - 1e-10:
                best_obj_ls = obj_new
                best_step = scale
            else:
                break

        if best_step > 0:
            b = b - best_step * alpha * grad_centered
            b = project_simplex(b)
            if best_obj_ls < best_obj - 1e-12:
                best_obj = best_obj_ls
                best_b = b.copy()
            alpha *= 1.01  # slight increase
        else:
            alpha *= 0.7
            if alpha < 1e-15:
                break

    return best_b, best_obj


def multiplicative_update(b, time_budget=20.0):
    """Multiplicative update: b_i <- b_i * exp(-eta * grad_i), renormalize."""
    n = len(b)
    b = normalize(b.copy().astype(np.float64))
    best_b = b.copy()
    best_obj = compute_obj(b)
    start = time.monotonic()
    eta = 1.0 / n

    while time.monotonic() - start < time_budget:
        tau = max(0.01, 0.05 * (time_budget - (time.monotonic() - start)) / time_budget)
        obj_val, grad, conv = lse_value_grad(b, tau)
        grad_centered = grad - np.mean(grad)

        b_new = b * np.exp(-eta * grad_centered)
        b_new = np.maximum(b_new, 1e-15)
        b_new = b_new / np.sum(b_new)
        obj_new = compute_obj(b_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_b = b_new.copy()
            b = b_new
        else:
            eta *= 0.8
            if eta < 1e-10:
                break

    return best_b, best_obj


def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.7, 0.7) / (m ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def raised_cosine_init(n, alpha=1.0):
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * x)) ** alpha
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def flat_with_perturb(n, seed=0, scale=0.1):
    rng = np.random.default_rng(seed)
    a = np.ones(n) + scale * rng.standard_normal(n)
    a = np.maximum(a, 0)
    return a / np.sum(a)


def get_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def refine_combined(a, time_budget=60.0):
    """Combined refinement: LP + LSE + multiplicative + perturbation."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # LP phase
        if remaining > 20:
            lp_budget = min(remaining * 0.4, 40.0)
            a_lp, obj_lp = lp_alternating(best_a, max_iters=30, time_budget=lp_budget)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                print(f"      LP: {best_obj:.6f}")

        # LSE gradient phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            lse_budget = min(remaining * 0.2, 15.0)
            a_lse, obj_lse = lse_descent(best_a, time_budget=lse_budget)
            if obj_lse < best_obj:
                best_obj = obj_lse
                best_a = a_lse.copy()
                print(f"      LSE: {best_obj:.6f}")

        # Multiplicative update
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            mu_budget = min(remaining * 0.15, 10.0)
            a_mu, obj_mu = multiplicative_update(best_a, time_budget=mu_budget)
            if obj_mu < best_obj:
                best_obj = obj_mu
                best_a = a_mu.copy()
                print(f"      MU: {best_obj:.6f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(round_num * 7919 + 13)
            for scale in [0.05, 0.1, 0.2]:
                if time.monotonic() - start > time_budget - 5:
                    break
                a_p = best_a * np.exp(rng.normal(0, scale, n))
                a_p = np.maximum(a_p, 0)
                a_p = a_p / np.sum(a_p)
                # Quick LP on perturbed
                a_q, obj_q = lp_alternating(a_p, max_iters=10, time_budget=min(10.0, remaining * 0.15))
                if obj_q < best_obj:
                    best_obj = obj_q
                    best_a = a_q.copy()
                    print(f"      Pert({scale}): {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start LP alternating + LSE + perturbation
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = get_parent()
    if parent is not None and len(parent) > 0:
        obj_parent = compute_obj(parent)
        best_score = obj_parent
        best_seq = parent.copy()
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, obj={obj_parent:.6f}")
    else:
        n_parent = 500
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)
        print(f"No parent. Using uniform n={n_parent}, obj={best_score:.6f}")

    # === Phase 1: Quick multi-start exploration ===
    print("\n=== Phase 1: Multi-start exploration ===")
    n_values = sorted(set([
        max(100, n_parent - 100),
        max(100, n_parent - 50),
        n_parent,
        n_parent + 50,
        n_parent + 100,
        n_parent + 200,
        min(1200, n_parent + 400)
    ]))
    n_values = [n for n in n_values if 100 <= n <= 1500]

    candidates = []
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        for init_type in ['cos', 'rcos', 'flat']:
            if time.monotonic() > deadline - 30:
                break
            if init_type == 'cos':
                a0 = cosine_init(n, n_modes=8, seed=seed + n)
            elif init_type == 'rcos':
                a0 = raised_cosine_init(n, alpha=1.5)
            else:
                a0 = flat_with_perturb(n, seed=seed + n, scale=0.05)

            obj0 = compute_obj(a0)
            # Quick LP
            a_q, obj_q = lp_alternating(a0, max_iters=15, time_budget=8.0)
            if obj_q < obj0:
                candidates.append((a_q, obj_q))
            else:
                candidates.append((a0, obj0))
            if obj_q < best_score:
                best_score = obj_q
                best_seq = a_q.copy()
                print(f"  n={n} {init_type}: {obj_q:.6f} ** NEW BEST **")

    # Also try parent-based interpolations
    if parent is not None:
        for n in [n_parent + 50, n_parent + 100, n_parent + 200]:
            if time.monotonic() > deadline - 30:
                break
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            try:
                f_interp = interp1d(x_old, parent, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
                a_interp = a_interp / np.sum(a_interp)
            except Exception:
                a_interp = np.ones(n) / n
            a_q, obj_q = lp_alternating(a_interp, max_iters=15, time_budget=8.0)
            if obj_q < best_score:
                best_score = obj_q
                best_seq = a_q.copy()
                print(f"  interp n={n}: {obj_q:.6f} ** NEW BEST **")

    # === Phase 2: Intensive refinement of best ===
    print(f"\n=== Phase 2: Intensive refinement (best={best_score:.6f}, n={len(best_seq)}) ===")
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    for round_num in range(10):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 80.0)
        print(f"  Round {round_num}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")
        a_new, obj_new = refine_combined(a_cur, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  Round {round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  Round {round_num}: {obj_new:.6f} (no improvement)")
            # Stronger perturbation
            rng = np.random.default_rng(round_num * 4444 + 777)
            n_cur = len(a_cur)
            for pscale in [0.15, 0.3, 0.5]:
                if time.monotonic() > deadline - 15:
                    break
                a_p = a_cur * np.exp(rng.normal(0, pscale, n_cur))
                a_p = np.maximum(a_p, 0)
                a_p = a_p / np.sum(a_p)
                alloc_p = min(remaining * 0.1, 20.0)
                a_pp, obj_pp = refine_combined(a_p, time_budget=alloc_p)
                if obj_pp < best_score:
                    best_score = obj_pp
                    best_seq = a_pp.copy()
                    a_cur = a_pp.copy()
                    print(f"  R{round_num} pert({pscale}): {best_score:.6f} ** NEW BEST **")

    # === Phase 3: Try different n for final push ===
    print(f"\n=== Phase 3: Multi-n final push ===")
    n_base = len(best_seq)
    n_tries = sorted(set([
        max(100, n_base - 100), max(100, n_base - 50),
        n_base + 50, n_base + 100, n_base + 150,
        n_base + 300
    ]))
    n_tries = [n for n in n_tries if 100 <= n <= 2000]

    for n_try in n_tries:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        x_old = np.linspace(0, 1, n_base)
        x_new = np.linspace(0, 1, n_try)
        try:
            f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
            a_interp = np.maximum(f_interp(x_new), 0)
            a_interp = a_interp / np.sum(a_interp)
        except Exception:
            a_interp = np.ones(n_try) / n_try

        alloc = min(remaining * 0.3, 40.0)
        a_new, obj_new = refine_combined(a_interp, time_budget=alloc)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 4: Final intensive LP polish ===
    print(f"\n=== Phase 4: Final LP polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Extended LP alternating
        a_lp, obj_lp = lp_alternating(a_cur, max_iters=80, time_budget=min(remaining * 0.7, 120.0))
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # One more combined round
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin, obj_fin = refine_combined(best_seq, time_budget=min(remaining * 0.8, 60.0))
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  Final combined: {best_score:.6f} ** NEW BEST **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
        print(f"  Validated: {best_score:.6f}")
    elif val == np.inf:
        print("  WARNING: invalid sequence, using fallback")
        result = [1.0] * 300
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
