```python
def conv_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = conv_fft(a)
    return 2.0 * n * np.max(conv) / (S * S)

def softmax_grad(a, T):
    """Compute soft-max objective and its gradient (projected to sum=0)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, np.zeros(n)
    conv = conv_fft(a)
    max_c = np.max(conv)
    # Soft-max
    log_weights = (conv - max_c) / T
    log_weights -= np.max(log_weights)  # extra stability
    w = np.exp(log_weights)
    w_sum = np.sum(w)
    w /= w_sum
    soft_max = max_c + T * (np.log(w_sum) + np.max(log_weights))

    # soft_max = max_c + T * log(sum_k exp(u_k))

    exp_u = np.exp(u)
    sum_exp_u = np.sum(exp_u)
    soft_max = max_c + T * np.log(sum_exp_u)
    w = exp_u / sum_exp_u

    # Full objective: obj = 2*n*soft_max/S^2
    obj = 2.0 * n * soft_max / (S * S)

    grad_proj = grad - np.mean(grad)
    
    return obj, grad_proj

def adam_ensemble(a, T_start, T_end, time_budget, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer with temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    n_steps_est = int(time_budget * 50)  # estimate steps
    for step in range(n_steps_est):
        if time.monotonic() - start > time_budget:
            break
        t += 1
        
        # Anneal temperature
        frac = t / n_steps_est
        T = T_start * (T_end / T_start) ** frac
        
        obj, grad = softmax_grad(a, T)
        if np.isinf(obj):
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj

                # Since we're changing a[i], let's just compute the relevant conv values
                # For efficiency, compute max of all conv values
                conv = conv_fft(a)
                return 2.0 * n * np.max(conv) / (S_new * S_new)
            
            # Golden section search on [0, 4*t_cur + 1e-6]
            a[i] = t_cur  # restore
            lo, hi = 0.0, max(4.0 * t_cur, 0.01)
            if t_cur < 1e-10:

            phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

            f1 = f_of_t(t1)

                    f2 = f1
                    t1 = hi - phi * (hi - lo)
                    f1 = f_of_t(t1)
                else:
                    lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = lo + phi * (hi - lo)
                    f2 = f_of_t(t2)
            
            t_opt = (lo + hi) / 2
            a[i] = t_opt
            S = S_0 + t_opt
            obj_cur = compute_obj(a)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_a = a.copy()
    
    return best_a, best_obj
```

```python
    if S < 1e-10:
        return np.inf
    conv = conv_fft(a)
    return 2.0 * n * float(np.max(conv)) / (S * S)

def softmax_obj_grad(a, T):
    """Compute soft-max objective and projected gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf, np.zeros(n)
    conv = conv_fft(a)
    max_c = np.max(conv)
    u = (conv - max_c) / T
    exp_u = np.exp(u)
    sum_eu = np.sum(exp_u)
    soft_max = max_c + T * np.log(sum_eu)
    w = exp_u / sum_eu
    
    # grad of soft_max: 2 * correlate(w, a, 'valid')
    grad_sm = 2.0 * np.correlate(w, a, mode='valid')
    
    obj = 2.0 * n * soft_max / (S * S)
    grad = (2.0 * n / S) * (grad_sm - 2.0 * soft_max / S)
    grad_proj = grad - np.mean(grad)
    
    return obj, grad_proj

def adam_smax(a, time_budget, T_start=None, T_end=None, lr=0.05, 
               beta1=0.9, beta2=0.999, eps=1e-8, check_interval=50):
    """Adam optimizer with soft-max and temperature annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S
    
    if T_start is None:
        conv = conv_fft(a)
        T_start = np.max(conv) * 0.5
    if T_end is None:
        T_end = T_start * 1e-4
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_obj = best_obj
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        frac = min(1.0, t / max(1, int(time_budget * 30)))
        T = T_start * (T_end / T_start) ** frac
        
        obj, grad = softmax_obj_grad(a, T)
        if not np.isfinite(obj):
            break
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        if t % check_interval == 0 or time.monotonic() - start > time_budget - 1:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new
        else:
            a = a_new
    
    # Final check
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    
    return best_a, best_obj
```

```python
    # Contribution is non-zero only at positions k in [i, i+n-1]
    contrib = np.zeros(2 * n - 1)
    k_min = i
    k_max = min(i + n - 1, 2 * n - 2)
    k_range = np.arange(k_min, k_max + 1)
    j_range = k_range - i  # j = k - i, ranges from 0 to n-1

    contrib[k_range] = 2.0 * t_val * a[j_range]

    if 2 * i <= 2 * n - 2:
        # At k = 2*i, the contribution should be t_val^2, not 2*t_val*a[i]
        # So subtract the linear and add the quadratic
        contrib[2 * i] = t_val * t_val
    
    conv_new = conv_base + contrib
    max_val = np.max(conv_new)
    return 2.0 * n * max_val / (S_new * S_new)
```

```python
"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: soft-max gradient with Adam optimizer, exact coordinate descent polish,
temperature annealing, multi-start with diverse structured initializations, and
multi-length exploration. Replaces top-K subgradient with smooth log-sum-exp surrogate
for more effective optimization.
"""

import numpy as np
import time


def conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = conv_fft(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def softmax_obj_grad(a, T):
    """
    Compute soft-max approximation of the objective and its gradient.
    Returns (objective_value, projected_gradient).
    The projection ensures the gradient is orthogonal to the all-ones vector
    (maintaining the sum constraint).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10 or T <= 0:
        return np.inf, np.zeros(n)
    conv = conv_fft(a)
    max_c = float(np.max(conv))
    # Numerically stable log-sum-exp
    u = (conv - max_c) / T
    exp_u = np.exp(u)
    sum_eu = float(np.sum(exp_u))
    soft_max = max_c + T * np.log(sum_eu)
    w = exp_u / sum_eu
    # Gradient of soft_max w.r.t. a[i] = 2 * sum_j a[j] * w[j+i]
    grad_sm = 2.0 * np.correlate(w, a, mode='valid')
    # Full objective: obj = 2*n*soft_max/S^2
    obj = 2.0 * n * soft_max / (S * S)
    # Gradient: d(obj)/d(a[i]) = (2n/S)*(grad_sm[i] - 2*soft_max/S)
    grad = (2.0 * n / S) * (grad_sm - 2.0 * soft_max / S)
    # Project to maintain sum constraint
    grad_proj = grad - np.mean(grad)
    return obj, grad_proj


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def adam_smax(a, time_budget, T_start=None, T_end=None, lr=0.03,
              beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer with soft-max surrogate and temperature annealing.
    Returns (best_sequence, best_objective).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))

    if T_start is None:
        conv = conv_fft(a)
        T_start = max(float(np.max(conv)) * 0.3, 1e-6)
    if T_end is None:
        T_end = T_start * 1e-5

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    lr_current = lr

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1.0, time_budget))
        # Exponential temperature annealing
        T = T_start * (T_end / T_start) ** frac

        obj_smooth, grad = softmax_obj_grad(a, T)
        if not np.isfinite(obj_smooth):
            break

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate: decay slightly over time
        lr_eff = lr_current * (1.0 - 0.3 * frac)

        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        S_new = float(np.sum(a_new))
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        a = a_new

        # Periodic evaluation with exact objective
        if t % 30 == 0 or elapsed > time_budget - 2.0:
            obj_exact = compute_obj(a)
            if obj_exact < best_obj - 1e-12:
                best_obj = obj_exact
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 20:
                    # Reduce learning rate and reset momentum
                    lr_current *= 0.7
                    m *= 0.5
                    v *= 0.5
                    stall_count = 0
                    if lr_current < 1e-6:
                        break

    # Final evaluation
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    return best_a, best_obj


def coord_descent(a, n_passes=3, time_budget=30.0):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    via golden section search. Highly effective for final polishing.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    S = float(np.sum(a))

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        improved_any = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            t_cur = a[i]
            S_0 = S - t_cur

            # Compute base convolution with a[i] = 0
            a[i] = 0.0
            conv_base = conv_fft(a)
            a[i] = t_cur

            # Precompute the linear slope contributions
            # For k in [i, min(i+n-1, 2n-2)]: conv[k] gets 2*t*a[k-i] added
            # Except at k=2*i where it's t^2 instead
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            def f_eval(t_val):
                S_new = S_0 + t_val
                if S_new < 1e-10:
                    return 1e10
                # Vectorized: compute max of modified convolution
                max_val = float(np.max(conv_base))
                if k_max >= k_min:
                    k_range = np.arange(k_min, k_max + 1)
                    j_range = k_range - i
                    contrib = 2.0 * t_val * a[j_range]
                    # Fix quadratic at k = 2*i
                    quad_idx = 2 * i
                    if k_min <= quad_idx <= k_max:
                        local_pos = quad_idx - k_min
                        contrib[local_pos] = t_val * t_val
                    modified = conv_base[k_range] + contrib
                    local_max = float(np.max(modified))
                    max_val = max(max_val, local_max)
                return 2.0 * n * max_val / (S_new * S_new)

            # Golden section search on [0, t_hi]
            t_lo = 0.0
            t_hi = max(4.0 * t_cur, 0.005)

            # Ensure bracket contains minimum
            f_mid = f_eval(t_cur) if t_cur < t_hi else f_eval(0.5 * t_hi)

            t1 = t_hi - phi * (t_hi - t_lo)
            t2 = t_lo + phi * (t_hi - t_lo)
            f1 = f_eval(t1)
            f2 = f_eval(t2)

            for _ in range(35):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - phi * (t_hi - t_lo)
                    f1 = f_eval(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + phi * (t_hi - t_lo)
                    f2 = f_eval(t2)

            t_opt = (t_lo + t_hi) / 2.0
            a[i] = t_opt
            S = S_0 + t_opt
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                improved_any = True

        if not improved_any:
            break

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    """Cosine series initialization with random coefficients."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2.0 * np.pi * m * k / n + phase)
    return normalize(a)


def arch_init(n, alpha=1.5):
    """Arch-shaped (circular) initialization - large in middle, small at edges."""
    k = np.arange(n, dtype=np.float64)
    x = 2.0 * k / (n - 1) - 1.0
    a = np.sqrt(np.maximum(1.0 - x * x, 0.0)) ** alpha
    return normalize(a)


def asym_arch_init(n, beta=0.3):
    """Asymmetric arch: shifts the peak to create multiplicative asymmetry."""
    k = np.arange(n, dtype=np.float64)
    center = 0.5 + beta
    x = (k / (n - 1) - center) / (0.5 - abs(beta))
    a = np.exp(-2.0 * x * x)
    return normalize(a)


def random_smooth_init(n, seed=0, smoothness=3):
    """Random initialization with Gaussian smoothing."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    # Apply smoothing
    kernel = np.ones(smoothness) / smoothness
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def linear_ramp_init(n, direction=1):
    """Linear ramp (increasing or decreasing)."""
    k = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = k + 1.0
    else:
        a = n - k
    return normalize(a)


def two_bump_init(n, gap=0.3):
    """Two separated Gaussian bumps (creates asymmetry)."""
    k = np.arange(n, dtype=np.float64)
    c1 = 0.5 - gap
    c2 = 0.5 + gap
    sigma = 0.15
    a = np.exp(-0.5 * ((k / n - c1) / sigma) ** 2) + 0.7 * np.exp(-0.5 * ((k / n - c2) / sigma) ** 2)
    return normalize(a)


def optimize_pipeline(a, time_budget):
    """
    Full optimization pipeline: Adam + coordinate descent interleaved.
    Returns (best_sequence, best_objective).
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 2.0:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break

        # Adam phase
        adam_time = min(remaining * 0.5, 40.0)
        if round_num > 3:
            adam_time = min(remaining * 0.3, 20.0)
        a_adam, obj_adam = adam_smax(best_a, time_budget=adam_time, lr=0.03 / (1 + 0.2 * round_num))
        if obj_adam < best_obj:
            best_obj = obj_adam
            best_a = a_adam.copy()

        # Coordinate descent phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8.0:
            cd_time = min(remaining * 0.4, 25.0)
            n_passes = max(1, int(cd_time / 10.0))
            a_cd, obj_cd = coord_descent(best_a, n_passes=n_passes, time_budget=cd_time)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()

        # Quick Adam polish if time remains
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5.0 and round_num <= 5:
            a_polish, obj_polish = adam_smax(best_a, time_budget=min(remaining * 0.2, 10.0), lr=0.01)
            if obj_polish < best_obj:
                best_obj = obj_polish
                best_a = a_polish.copy()

    return best_a, best_obj


def perturb_and_retry(best_a, best_obj, time_budget, round_num):
    """Perturb the best sequence and try to improve."""
    n = len(best_a)
    rng = np.random.default_rng(round_num * 1000 + 42)

    for pert_scale in [0.05, 0.1, 0.2]:
        if time.monotonic() > time_budget:
            break
        a_pert = best_a * np.exp(rng.normal(0, pert_scale, n))
        a_pert = normalize(a_pert)
        # Quick optimization
        a_new, obj_new = optimize_pipeline(a_pert, time_budget=min(20.0, (time_budget - time.monotonic()) * 0.4))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            S = np.sum(parent)
            if S > 0.01:
                obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines soft-max Adam, exact coordinate descent,
    multi-start with diverse initializations, and multi-length exploration
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
    else:
        n_default = 500
        best_seq = normalize(np.ones(n_default))
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_default}): obj={best_score:.6f}")

    print(f"Initial best: {best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: Multi-start exploration (20% of time) ===
    print("\n=== Phase 1: Multi-start exploration ===")
    phase1_time = 0.20 * (deadline - start_time)
    phase1_deadline = start_time + phase1_time

    n_base = len(best_seq)
    length_tries = sorted(set([
        n_base,
        max(200, n_base - 100),
        min(1200, n_base + 100),
        min(1200, n_base + 200),
    ]))

    candidate_pool = []

    for n_try in length_tries:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10.0:
            break

        inits = [
            ("cosine", cosine_init(n_try, n_modes=10, seed=seed + n_try)),
            ("arch", arch_init(n_try, alpha=1.5)),
            ("asym", asym_arch_init(n_try, beta=0.2)),
            ("rand", random_smooth_init(n_try, seed=seed + n_try * 7)),
            ("ramp", linear_ramp_init(n_try, direction=1)),
            ("2bump", two_bump_init(n_try, gap=0.25)),
        ]

        per_init_time = max(3.0, remaining / (len(inits) + 1))

        for name, a_init in inits:
            if time.monotonic() > phase1_deadline:
                break
            obj_init = compute_obj(a_init)
            # Quick Adam to improve
            a_quick, obj_quick = adam_smax(a_init, time_budget=min(per_init_time, 8.0), lr=0.05)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj, f"{name}_{n_try}"))

            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  {name} n={n_try}: {best_local_obj:.6f} ** NEW BEST **")

    # Also try parent with quick optimization
    if parent is not None and len(parent) > 0:
        a_p, obj_p = adam_smax(parent, time_budget=10.0, lr=0.04)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent refined: {obj_p:.6f} ** NEW BEST **")
        candidate_pool.append((a_p, obj_p, "parent_adam"))

    # === Phase 2: Top candidate refinement (50% of time) ===
    print("\n=== Phase 2: Refinement of top candidates ===")
    phase2_deadline = start_time + 0.70 * (deadline - start_time)
    candidate_pool.sort(key=lambda x: x[1])

    n_refine = min(4, len(candidate_pool))
    for idx in range(n_refine):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 20.0:
            break

        a_cand, obj_cand, name = candidate_pool[idx]
        if obj_cand > best_score + 0.03:
            break

        # Full pipeline optimization
        alloc = min(remaining * 0.4, 80.0)
        print(f"  Refining [{name}] n={len(a_cand)}, obj={obj_cand:.6f}, time={alloc:.0f}s")
        a_ref, obj_ref = optimize_pipeline(a_cand, time_budget=alloc)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={len(a_ref)}, obj={best_score:.6f} **")

    # === Phase 3: Intensive polishing (15% of time) ===
    print("\n=== Phase 3: Intensive polishing ===")
    phase3_deadline = start_time + 0.88 * (deadline - start_time)
    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    for round_num in range(8):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 15.0:
            break

        alloc = min(remaining * 0.3, 30.0)

        # Extended coordinate descent
        a_cd, obj_cd = coord_descent(a_cur, n_passes=5, time_budget=alloc * 0.6)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a_cur = a_cd.copy()
            print(f"  CD R{round_num}: {best_score:.6f} ** NEW BEST **")

        # Follow up with Adam
        remaining = phase3_deadline - time.monotonic()
        if remaining > 5.0:
            a_ad, obj_ad = adam_smax(a_cur, time_budget=min(remaining * 0.3, 10.0), lr=0.01)
            if obj_ad < best_score:
                best_score = obj_ad
                best_seq = a_ad.copy()
                a_cur = a_ad.copy()
                print(f"  Adam R{round_num}: {best_score:.6f} ** NEW BEST **")

        # Perturbation escape
        remaining = phase3_deadline - time.monotonic()
        if remaining > 10.0 and round_num % 2 == 1:
            a_cur, best_score = perturb_and_retry(a_cur, best_score, time.monotonic() + min(remaining * 0.3, 15.0), round_num)
            if best_score < np.inf:
                print(f"  Perturb R{round_num}: {best_score:.6f}")

    # === Phase 4: Final exploration (10% of time) ===
    print("\n=== Phase 4: Final exploration ===")
    remaining_final = deadline - time.monotonic()
    if remaining_final > 15.0:
        # Try a few more random perturbations with full pipeline
        rng = np.random.default_rng(99999 + seed)
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            a_pert = best_seq * np.exp(rng.normal(0, 0.08, n_cur))
            a_pert = normalize(a_pert)
            alloc = min(remaining_final * 0.25, 20.0)
            a_new, obj_new = optimize_pipeline(a_pert, time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final trial {trial}: {best_score:.6f} ** NEW BEST **")

        # Final coordinate descent on best
        remaining_final = deadline - time.monotonic()
        if remaining_final > 10.0:
            a_cd, obj_cd = coord_descent(best_seq, n_passes=5, time_budget=min(remaining_final * 0.7, 40.0))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.6f} ** NEW BEST **")

    # Final: one more Adam pass with very low lr
    remaining_final = deadline - time.monotonic()
    if remaining_final > 5.0:
        a_final, obj_final = adam_smax(best_seq, time_budget=min(remaining_final * 0.5, 10.0), lr=0.005)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Adam: {best_score:.6f} ** NEW BEST **")

    # Final validation and formatting
    if best_seq is None:
        best_seq = np.ones(500) / 500.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
