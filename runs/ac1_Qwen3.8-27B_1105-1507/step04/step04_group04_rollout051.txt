from scipy.optimize import minimize
from scipy.optimize import linprog

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    N = len(a)
    size = 1
    while size < 2 * N - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * N - 1]

def smooth_obj_grad(x, tau, N):
    """Compute smooth objective and gradient in log-parameterization."""
    # Convert x to a (softmax)
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    S = np.sum(a)
    a = a / S
    
    # Compute conv
    b = compute_conv(a)
    max_b = np.max(b)
    shifted = tau * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = np.sum(e)
    w = e / se
    
    # Objective
    sm_val = max_b + np.log(se) / tau
    obj = 2.0 * N * sm_val
    
    # Gradient w.r.t. a: g[i] = 2N * sum_k w[k] * a[k-i]
    c = np.correlate(w, a, mode='full')
    g = 2.0 * N * c[:N]

    grad = a * (g - np.dot(g, a))
    
    return obj, grad

    N = len(a0)
    a0 = a0.copy()
    S = np.sum(a0)
    if S < 1e-10:
        a0 = np.ones(N) / N
        S = 1.0
    else:
        a0 = a0 / S
    
    # Initialize x
    x = np.log(np.maximum(a0, 1e-30))
    
    best_a = a0.copy()
    best_obj = np.inf
    start = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - start > time_budget - 5:
            break

        def fun(x_val):
            obj, _ = smooth_obj_grad(x_val, tau, N)
            return obj
        
        def jac(x_val):
            _, grad = smooth_obj_grad(x_val, tau, N)
            return grad
        
        try:

import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import linprog

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    N = len(a)
    if N == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * N - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * N - 1]

def true_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    N = len(a)
    if N == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * N * max_b / (sum_a * sum_a)

def smooth_obj_grad(x, tau, N):
    """Compute smooth objective and gradient in log-parameterization.
    
    x is the log-parameter: a = softmax(x), so sum(a)=1, a>0.
    Returns (objective_value, gradient_wrt_x).
    """
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    S = np.sum(a)
    a = a / S
    
    b = compute_conv(a)
    max_b = np.max(b)
    shifted = tau * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = np.sum(e)
    
    if se < 1e-300:
        return np.inf, np.zeros(N)
    
    w = e / se
    sm_val = max_b + np.log(se) / tau
    obj = 2.0 * N * sm_val
    
    # g[i] = 2N * sum_k w[k] * a[k-i]
    c = np.correlate(w, a, mode='full')
    g = 2.0 * N * c[:N]
    
    # Gradient w.r.t. x: dx[i] = a[i] * (g[i] - dot(g, a))
    grad = a * (g - np.dot(g, a))
    
    return obj, grad

def optimize_smooth(a0, tau_list, time_budget, verbose=True):
    """Optimize using smooth gradient with L-BFGS-B and tau annealing."""
    N = len(a0)
    a = a0.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(N)
        S = N
    a = a / S
    
    x = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_true_obj = true_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget - 5:
            break
        
        def fun(x_val):
            obj, _ = smooth_obj_grad(x_val, tau, N)
            return obj
        
        def jac(x_val):
            _, grad = smooth_obj_grad(x_val, tau, N)
            return grad
        
        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=jac,
                          options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 1000})
            
            # Evaluate true objective at the result
            x_shifted = res.x - np.max(res.x)
            a_new = np.exp(x_shifted)
            a_new = a_new / np.sum(a_new)
            obj_new = true_obj(a_new)

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    N = len(a)
    A = np.zeros((2 * N - 1, N), dtype=np.float64)
    for j in range(N):
        k_start = j
        k_end = min(j + N, 2 * N - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    N = len(a)
    A = build_toeplitz(a)
    c = -np.ones(N)
    b_ub = np.full(2 * N - 1, M)
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def alternating_lp(a, max_iters=50, time_budget=30.0):
    """Alternating LP with line search."""
    N = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break
        
        g_norm = g / sum_g
        
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = true_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    
    return best_a, best_obj

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Step function optimization using smooth gradient (softmax approximation of max-conv)
with L-BFGS-B quasi-Newton method, tau annealing, LP polishing, and local search.
Key innovation: proper gradient via log-parameterization enables efficient
second-order optimization that outperforms pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import linprog


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    N = len(a)
    if N == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * N - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * N - 1]


def true_obj(a):
    """Compute true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    N = len(a)
    if N == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * N * max_b / (sum_a * sum_a)


def smooth_obj_grad(x, tau, N):
    """Compute smooth objective and gradient in log-parameterization.
    x is log-parameter: a = softmax(x), so sum(a)=1, a>0.
    Returns (objective_value, gradient_wrt_x)."""
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    S = np.sum(a)
    if S < 1e-300:
        return np.inf, np.zeros(N)
    a = a / S

    b = compute_conv(a)
    max_b = np.max(b)
    shifted = tau * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = np.sum(e)

    if se < 1e-300:
        return np.inf, np.zeros(N)

    w = e / se
    sm_val = max_b + np.log(se) / tau
    obj = 2.0 * N * sm_val

    # g[i] = 2N * sum_k w[k] * a[k-i] (truncated correlation)
    c = np.correlate(w, a, mode='full')
    g = 2.0 * N * c[:N]

    # Gradient w.r.t. x: dx[i] = a[i] * (g[i] - dot(g, a))
    grad = a * (g - np.dot(g, a))

    return obj, grad

    best_a = a.copy()
    best_true_obj = true_obj(a)
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget - 5:
            break

        def fun(x_val):
            obj, _ = smooth_obj_grad(x_val, tau, N)
            return obj

        def jac(x_val):
            _, grad = smooth_obj_grad(x_val, tau, N)
            return grad

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=jac,
                          options={'maxiter': 500, 'gtol': 1e-12, 'maxfun': 2000})

            x_shifted = res.x - np.max(res.x)
            a_new = np.exp(x_shifted)
            a_new = a_new / np.sum(a_new)
            obj_new = true_obj(a_new)

            if obj_new < best_true_obj:
                best_true_obj = obj_new
                best_a = a_new.copy()
                x = res.x.copy()

            if verbose:
                elapsed = time.monotonic() - start
                print(f"    tau={tau:.0f}: smooth={res.fun:.8f}, true={best_true_obj:.8f} ({elapsed:.1f}s)")

        except Exception as e:
            if verbose:
                print(f"    tau={tau:.0f}: error {type(e).__name__}")

    return best_a, best_true_obj


def multiplicative_sg_step(a, tau, step):
    """One multiplicative subgradient step (handles exact max, not smooth)."""
    N = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:N]
    grad_centered = grad - np.mean(grad)
    log_update = -step * grad_centered
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def sg_polish(a, time_budget=20.0):
    """Polish with exact subgradient (handles the true max, not smooth approx)."""
    N = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    step = 0.02
    tau = 50.0
    stall = 0
    patience = 50

    while time.monotonic() - start < time_budget:
        a = multiplicative_sg_step(a, tau, step)
        obj = true_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                tau *= 1.3
                if step < 0.002:
                    break
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = best_a * np.exp(rng.normal(0, 0.02, N))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience // 2, 20)

        if step > 0.005:
            step *= 0.9995

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    N = len(a)
    A = np.zeros((2 * N - 1, N), dtype=np.float64)
    for j in range(N):
        k_start = j
        k_end = min(j + N, 2 * N - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    N = len(a)
    A = build_toeplitz(a)
    c = -np.ones(N)
    b_ub = np.full(2 * N - 1, M)
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=50, time_budget=30.0):
    """Alternating LP with line search."""
    N = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = true_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break

    return best_a, best_obj


def local_search(a, time_budget=60.0, verbose=False):
    """Local search with perturbations and re-optimization."""
    N = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = true_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  LS start: obj={best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 10:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 12:
            break

        rng = np.random.default_rng(attempt * 7919 + seed_global)
        ptype = attempt % 4

        if ptype == 0:
            scale = 0.01 + 0.03 * (attempt % 5) / 5.0
            a_pert = best_a * np.exp(rng.normal(0, scale, N))
        elif ptype == 1:
            scale = 0.003 + 0.008 * (attempt % 5) / 5.0
            a_pert = best_a + rng.normal(0, scale, N)
            a_pert = np.maximum(a_pert, 0)
        elif ptype == 2:
            n_coords = max(2, N // 8)
            idx = rng.choice(N, size=min(n_coords, N), replace=False)
            a_pert = best_a.copy()
            for i in idx:
                a_pert[i] *= np.exp(rng.normal(0, 0.04))
        else:
            # Block perturbation
            block = max(3, N // 20)
            start_i = rng.integers(0, N - block)
            a_pert = best_a.copy()
            a_pert[start_i:start_i + block] *= np.exp(rng.normal(0, 0.03, block))

        a_pert = np.maximum(a_pert, 1e-15)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert

        sub_time = min(remaining * 0.35, 25.0)
        if sub_time > 8:
            a_opt, obj_opt = optimize_smooth(a_pert, [30.0, 80.0, 200.0], sub_time, verbose=False)
        else:
            a_opt = a_pert
            obj_opt = true_obj(a_pert)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    LS attempt {attempt}: obj={best_obj:.8f}")

    return best_a, best_obj


def resample(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):
    """Generate diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for scale in range(4):
            f = rng.uniform(0.3 * (2 ** scale), 1.2 * (2 ** scale))
            amp = 0.25 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = rng.random(max(5, n // 8))
        a = np.interp(k, np.arange(len(a)), a)
    elif choice == 6:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = 1.0 + 0.5 * rng.uniform(-1, 1, n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


seed_global = 42


def propose_candidate(seed=42, budget_s=1500):
    """
    Main optimization: smooth gradient (L-BFGS-B with softmax max approximation)
    + exact SG polish + LP polishing + local search.
    Explores n around parent length with focused time allocation.
    """
    global seed_global
    seed_global = seed

    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            p_obj = true_obj(parent)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = parent.copy()
    except Exception:
        pass

    n_parent = len(best_seq) if best_seq is not None else 330
    print(f"Initial: n={n_parent}, score={best_score:.6f}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # === Phase 1: Smooth gradient on parent (main workhorse) ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.30, 450.0)
        print(f"\n=== Phase 1: Smooth gradient (budget={p1_budget:.0f}s) ===")

        tau_list = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0, 600.0, 1200.0]
        a_opt, obj_opt = optimize_smooth(best_seq, tau_list, p1_budget, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1: {best_score:.8f}")

    # === Phase 2: Exact SG polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p2_budget = min(remaining * 0.10, 100.0)
        if p2_budget > 10:
            print(f"\n=== Phase 2: SG polish (budget={p2_budget:.0f}s) ===")
            a_opt = sg_polish(best_seq, p2_budget)
            obj_opt = true_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Phase 2: {best_score:.8f}")

    # === Phase 3: LP polishing ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p3_budget = min(remaining * 0.15, 150.0)
        if p3_budget > 15:
            print(f"\n=== Phase 3: LP polish (budget={p3_budget:.0f}s) ===")
            a_opt, obj_opt = alternating_lp(best_seq, max_iters=40, time_budget=p3_budget)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Phase 3: {best_score:.8f}")

    # === Phase 4: Smooth gradient refinement ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p4_budget = min(remaining * 0.12, 120.0)
        if p4_budget > 15:
            print(f"\n=== Phase 4: Smooth refinement (budget={p4_budget:.0f}s) ===")
            tau_list = [50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]
            a_opt, obj_opt = optimize_smooth(best_seq, tau_list, p4_budget, verbose=True)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Phase 4: {best_score:.8f}")

    # === Phase 5: Explore nearby n values ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 5: Nearby n exploration ===")

        n_candidates = []
        for delta in [-15, -10, -5, 5, 10, 15, -25, 25, -35, 35, -50, 50, 70, 90]:
            n_val = n_parent + delta
            if 150 <= n_val <= 800:
                n_candidates.append(n_val)
        for n_val in [280, 300, 320, 340, 350, 380, 400]:
            if n_val not in n_candidates:
                n_candidates.append(n_val)
        n_candidates = sorted(set(n_candidates))

        per_n = min(remaining / max(len(n_candidates), 1) * 1.2, 50.0)
        per_n = max(per_n, 12.0)

        for n_val in n_candidates:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if best_seq is not None and len(best_seq) != n_val:
                a_start = resample(best_seq, n_val)
            else:
                a_start = random_init(n_val, seed + n_val * 31)

            sub_budget = min(per_n, remaining * 0.3)
            a_opt, obj_opt = optimize_smooth(a_start, [15.0, 40.0, 100.0, 250.0], sub_budget, verbose=False)

            # Quick LP if promising
            if obj_opt < best_score + 0.03:
                lp_time = min(12.0, (deadline - time.monotonic()) * 0.1)
                if lp_time > 5:
                    a_opt2, obj_opt2 = alternating_lp(a_opt, max_iters=12, time_budget=lp_time)
                    if obj_opt2 < obj_opt:
                        a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_val}: {best_score:.8f} **")
            else:
                print(f"  n={n_val}: {obj_opt:.8f}")

    # === Phase 6: Final local search ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            p6_budget = min(remaining * 0.45, remaining - 15)
            print(f"\n=== Phase 6: Local search (budget={p6_budget:.0f}s) ===")
            a_opt, obj_opt = local_search(best_seq, p6_budget, verbose=True)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Phase 6: {best_score:.8f}")

    # === Phase 7: Final smooth + SG polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 25:
            p7_budget = min(remaining * 0.4, remaining - 10)
            print(f"\n=== Phase 7: Final polish (budget={p7_budget:.0f}s) ===")
            tau_list = [100.0, 250.0, 500.0, 1000.0, 2000.0]
            a_opt, obj_opt = optimize_smooth(best_seq, tau_list, p7_budget * 0.6, verbose=True)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
            # SG polish
            sg_time = p7_budget * 0.4
            if sg_time > 5:
                a_opt = sg_polish(best_seq, sg_time)
                obj_opt = true_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
            print(f"  Phase 7: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(330)
        best_score = true_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if final_val < best_score:
        best_score = final_val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
