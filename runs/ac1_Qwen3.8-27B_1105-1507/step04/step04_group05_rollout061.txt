    y = a.copy()
    tk = 1.0

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive temperature

        tau = tau_start * (tau_end / tau_start) ** progress

        # Gradient step on z

            tracker.update(x_new)

            tk = t_new

def golden_section_1d(f, lo, hi, n_iter=40):
    """Golden section search for minimum of unimodal function on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0

                ck = b[k] - ai * ai + x * x
            else:
                ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)

    # Find bracket
    lo = 0.0
    hi = max(2.0 * ai, 3.0 * S / n, 0.5)
    f_lo = eval_obj(lo)
    f_mid = eval_obj(hi * 0.5)
    if f_mid < f_lo:
        # Minimum might be above lo, expand hi
        f_hi = eval_obj(hi)
        if f_hi > f_mid:
            pass  # bracket [0, hi] is good
        else:
            hi *= 2.0
            while hi < 100.0 and eval_obj(hi) <= f_mid:
                hi *= 2.0
    x_opt = golden_section_1d(eval_obj, lo, hi, n_iter=35)
    return max(0.0, x_opt)


def coordinate_descent_pass(a, n_passes=1, deadline=None, tracker=None, verbose=False):
    """Multiple passes of exact coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    if tracker is not None:
        tracker.update(a)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        improvement_total = 0.0
        indices = list(range(n))
        np.random.shuffle(indices)

        for i in indices:
            ai_old = a[i]
            x_new = coordinate_descent_1d(a, i, b, S)
            if abs(x_new - ai_old) > 1e-12:
                # Update b incrementally
                for k in range(max(0, i), min(2 * n - 1, i + n)):
                    ki = k - i
                    if k == 2 * i:
                        b[k] = b[k] - ai_old * ai_old + x_new * x_new
                    else:
                        b[k] = b[k] - 2.0 * ai_old * a[ki] + 2.0 * x_new * a[ki]
                S = S - ai_old + x_new
                a[i] = x_new
                improvement_total += abs(x_new - ai_old)

        # Verify with full conv
        b = fast_conv(a)
        new_J = 2.0 * n * b.max() / (S * S)
        if new_J < current_J - 1e-12:
            current_J = new_J
            if tracker is not None:
                tracker.update(a)
            if verbose and p % 3 == 0:
                print(f"    CD pass {p}: J={current_J:.8f} imp={improvement_total:.4f}", flush=True)
        elif p > 5 and improvement_total < 1e-8:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def lp_escape(a, deadline, tracker, n_iters=20):
    """LP-based best response iteration for escape from local minima."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    S = a.sum()
    a = a / S
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try slightly tightened M
        for frac in [0.0, -0.001, -0.002, -0.005]:
            M_target = M * (1.0 + frac)
            if M_target <= 0:
                continue
            g = lp_best_response(a, M_target)
            if g is None:
                continue
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                continue
            g = g / g_sum

            # Try blending
            best_trial = None
            best_trial_J = current_J
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                trial = (1.0 - t) * a + t * g
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                J_trial = compute_J(trial)
                if J_trial < best_trial_J - 1e-13:
                    best_trial_J = J_trial
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                current_J = best_trial_J
                tracker.update(a)
            break
        else:
            # No LP success
            if it > 3:
                break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size < 2:
        return np.full(dest_n, src[0] if src.size == 1 else 0.5)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score - 1e-12:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} Best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    for freq in [2, 3, 5, 7]:
        inits.append(np.abs(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28))))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.15 * np.cos(2 * np.pi * 7 * t))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(0.3 + 0.7 * np.exp(-2.0 * (t - 0.5) ** 2))
    # Two-peak
    inits.append(0.3 + 0.5 * np.exp(-20 * (t - 0.3) ** 2) + 0.5 * np.exp(-20 * (t - 0.7) ** 2))
    # Ramp
    inits.append(0.5 + 0.5 * t)
    # Inverse ramp
    inits.append(0.5 + 0.5 * (1 - t))
    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses FISTA, coordinate descent, LP escape,
    and multi-length exploration to minimize J(a).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384", flush=True)

    n_parent = parent.size

    # === Phase 1: FISTA on parent ===
    print(f"\n{'='*50} Phase 1: FISTA on parent (n={n_parent}) {'='*50}", flush=True)
    t0 = time.monotonic()
    a1 = fista_smoothed(parent, deadline - 800, tracker, tau_start=1.0, tau_end=0.02, max_iters=3000)
    tracker.report(f"Phase 1 FISTA: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Coordinate descent polish ===
    if time.monotonic() < deadline - 600:
        print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        a2 = coordinate_descent_pass(tracker.best, n_passes=20, deadline=deadline - 600, tracker=tracker, verbose=True)
        tracker.report(f"Phase 2 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Escape ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 3: LP Escape {'='*50}", flush=True)
        t0 = time.monotonic()
        a3 = lp_escape(tracker.best, deadline - 400, tracker, n_iters=30)
        tracker.report(f"Phase 3 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: CD after LP ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 4: CD after LP {'='*50}", flush=True)
        t0 = time.monotonic()
        a4 = coordinate_descent_pass(tracker.best, n_passes=30, deadline=deadline - 300, tracker=tracker, verbose=True)
        tracker.report(f"Phase 4 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 5: Multi-Length {'='*50}", flush=True)
        t0 = time.monotonic()

        # Try several nearby lengths
        lengths_to_try = []
        base = tracker.best.size
        for dl in [-8, -6, -4, -2, 2, 4, 6, 8, 12, 16, -12, -16, 20, -20]:
            nl = base + dl
            if 200 <= nl <= 700 and nl not in lengths_to_try:
                lengths_to_try.append(nl)

        for nl in lengths_to_try:
            if time.monotonic() > deadline - 180:
                break

            a_res = resample_seq(tracker.best, nl)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()

            # Quick FISTA
            a_res = fista_smoothed(a_res, time.monotonic() + 15, tracker,
                                   tau_start=0.5, tau_end=0.05, max_iters=500)
            # Quick CD
            if time.monotonic() < deadline - 180:
                a_res = coordinate_descent_pass(a_res, n_passes=5,
                                                deadline=time.monotonic() + 10, tracker=tracker)
            print(f"    Length {nl}: J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 5 Multi: {time.monotonic()-t0:.1f}s")

    # === Phase 6: FISTA with different tau on best ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 6: FISTA refinement {'='*50}", flush=True)
        t0 = time.monotonic()
        # Try different temperature schedules
        for tau_s, tau_e in [(0.3, 0.01), (0.1, 0.005), (0.5, 0.003)]:
            if time.monotonic() > deadline - 100:
                break
            a_f = fista_smoothed(tracker.best, time.monotonic() + 30, tracker,
                                  tau_start=tau_s, tau_end=tau_e, max_iters=1000)
        tracker.report(f"Phase 6 FISTA ref: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Intensive coordinate descent ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50} Phase 7: Intensive CD {'='*50}", flush=True)
        t0 = time.monotonic()
        a7 = coordinate_descent_pass(tracker.best, n_passes=50, deadline=deadline - 50,
                                      tracker=tracker, verbose=True)
        tracker.report(f"Phase 7 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 8: LP + CD final ===
    if time.monotonic() < deadline - 20:
        print(f"\n{'='*50} Phase 8: Final LP+CD {'='*50}", flush=True)
        t0 = time.monotonic()
        a8 = lp_escape(tracker.best, time.monotonic() + 40, tracker, n_iters=20)
        if time.monotonic() < deadline - 15:
            a8 = coordinate_descent_pass(a8, n_passes=20, deadline=deadline - 15,
                                          tracker=tracker)
        tracker.report(f"Phase 8 Final: {time.monotonic()-t0:.1f}s")

    # === Phase 9: Last-resort CD ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 9: Last CD {'='*50}", flush=True)
        a9 = coordinate_descent_pass(tracker.best, n_passes=10, deadline=deadline - 5,
                                      tracker=tracker)
        tracker.report(f"Phase 9 Last: {time.monotonic()-time.monotonic():.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
