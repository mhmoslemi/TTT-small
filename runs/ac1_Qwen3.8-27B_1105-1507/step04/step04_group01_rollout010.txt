"""
Adaptive min-peak LP alternating optimization with dual-guided subgradient,
parametric family exploration, and multi-scale n-value search to minimize
2*n*max(conv(a,a))/(sum(a))^2. Key improvements: aggressive min-peak LP with
adaptive M reduction, conjugate-gradient polishing, and structured restarts.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_min_peak_lp(a, time_limit=30):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None


def subgradient_step(a, tau, step_size):
    """One subgradient step with smoothed max."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = np.clip(-tau * (conv - max_conv), -300, 300)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

    grad_centered = grad_a - np.mean(grad_a)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def subgradient_optimize(a, max_iters=2000, time_budget=20.0, tau_start=5.0, tau_end=50.0):

def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def soft_max_obj_and_grad_z(z, tau):
    """Compute soft-max of conv(softmax(z), softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_conv = np.exp(shifted)
    Z = float(np.sum(exp_conv))
    soft_max_val = cmax + tau * np.log(Z)
    w = exp_conv / Z

    w_rev = w[::-1]
    conv_fw = fft_convolve(a, w_rev)
    f1 = conv_fw[n - 1:2 * n - 1][::-1]
    grad_a = 2.0 * f1

    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    return soft_max_val, grad_z


def optimize_lbfgs(a_init, tau_list, max_iter=200, time_budget=30.0):

    n = len(a_init)

    s = np.sum(a)

        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    z = np.log(a)

    best_a = a.copy()

    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        try:
            result = minimize(lambda z_val: soft_max_obj_and_grad_z(z_val, tau),
                              z, jac=True, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
            z_new = result.x
        except Exception:
            continue

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)

            best_obj = obj
            best_a = a_new.copy()

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    a_new = np.maximum(a_new, 0)
    return a_new


def parametric_init(n, seed, family):
    """Generate parametric initial sequences."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if family == 0:
        # Cosh-shaped (high at edges)
        alpha = rng.uniform(0.5, 2.0)
        a = np.cosh(alpha * (k - center) / center)
    elif family == 1:
        # Inverse-Gaussian (high at edges)
        sigma = rng.uniform(0.2, 0.5)
        a = 1.0 / (1.0 + ((k - center) / (sigma * n)) ** 2 * rng.uniform(1, 10))
    elif family == 2:
        # Power-law edge enhancement
        p = rng.uniform(0.3, 1.5)
        a = (np.abs(k - center) / center + 0.1) ** p
    elif family == 3:
        # Cosine bell (moderate edges)
        a = 0.5 + 0.5 * np.cos(np.pi * k / (n - 1))
        a = a ** rng.uniform(0.5, 3.0)
    elif family == 4:
        # Two-peak
        f1 = rng.uniform(0.3, 0.45)
        f2 = rng.uniform(0.55, 0.7)
        a = 1.0 + rng.uniform(0.3, 1.0) * (np.exp(-((k/n - f1)/0.08)**2) + np.exp(-((k/n - f2)/0.08)**2))
    elif family == 5:
        # Periodic modulation
        f = rng.uniform(2, 8)
        amp = rng.uniform(0.2, 0.6)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif family == 6:
        # Log-edge: a[i] = 1 + log(1 + c * edge_dist)
        c = rng.uniform(1, 10)
        edge_dist = np.minimum(k, n - 1 - k)
        a = 1.0 + c * np.log1p(edge_dist / (n * 0.1))
    elif family == 7:
        # Step edge: enhanced at first and last m elements
        m = int(n * rng.uniform(0.05, 0.2))
        a = np.ones(n)
        a[:m] = rng.uniform(1.5, 4.0)

        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')

    s = np.sum(a)
    return a / s

            alloc = min(45.0, remaining * 0.12)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, min(4, int(alloc / 15)))

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                family = (seed + n * 7 + init_idx * 13) % 9
                a = parametric_init(n, seed + n * 104729 + init_idx * 79193, family)

            # Quick subgradient shaping
            sg_time = min(8.0, (alloc / n_inits) * 0.25)
            a, _ = subgradient_optimize(a, max_iters=int(sg_time * 30),
                                         time_budget=sg_time,
                                         tau_start=3.0, tau_end=25.0)

            # L-BFGS-B
            remaining_init = deadline - time.monotonic()
            if remaining_init > 10:
                lfbs_time = min((alloc / n_inits) * 0.35, 20.0)
                tau_list = [3.0/n, 1.5/n, 0.75/n, 0.4/n, 0.2/n]
                a, obj = optimize_lbfgs(a, tau_list, max_iter=150, time_budget=lfbs_time)
            else:
                obj = compute_obj(a)

            # Subgradient polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 8:
                a, obj = subgradient_optimize(a, max_iters=int(remaining_init * 10),
                                              time_budget=min(remaining_init * 0.15, 8.0),
                                              tau_start=8.0, tau_end=35.0)

            # Quick LP if n allows
            remaining_init = deadline - time.monotonic()
            if remaining_init > 15 and n <= 700:
                a = lp_alternating_advanced(a, max_iters=15,
                                            time_budget=min(remaining_init * 0.4, 25.0))
                obj = compute_obj(a)

            print(f"  init{init_idx} (fam={family if init_idx > 0 else 'parent'}): obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Intensive LP optimization on best candidates
    print(f"\n=== PHASE 2: INTENSIVE LP OPTIMIZATION ===")
    print(f"Best before Phase 2: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])
    n_polish = min(3, len(candidates))

    for ci in range(n_polish):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        cand_obj = candidates[ci][1]
        print(f"\n  Polishing candidate {ci} (n={n_cand}, obj={cand_obj:.6f}), {remaining:.0f}s left")

        # Heavy LP optimization
        lp_budget = min(remaining * 0.45, 120.0)
        if n_cand <= 800:
            a_pol = lp_alternating_advanced(a_cand, max_iters=50, time_budget=lp_budget)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  ** LP polish: obj={best_score:.6f} **")
            a_cand = a_pol

        # L-BFGS polish
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_cand <= 1000:
            tau_small = [0.5/n_cand, 0.2/n_cand, 0.1/n_cand, 0.05/n_cand]
            a_new, obj_new = optimize_lbfgs(a_cand, tau_small, max_iter=200,
                                            time_budget=min(remaining * 0.2, 25.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  ** L-BFGS polish: obj={best_score:.6f} **")
                a_cand = a_new

        # Subgradient final
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_new, obj_new = subgradient_optimize(a_cand, max_iters=int(remaining * 8),
                                                  time_budget=min(remaining * 0.2, 15.0),
                                                  tau_start=5.0, tau_end=25.0)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  ** SG polish: obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== PHASE 3: FINAL POLISH (n={n_best}, {remaining:.0f}s) ===")

        a_cur = best_seq.copy()
        for round_num in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP (most effective)
            if n_best <= 800 and remaining > 20:
                a_new = lp_alternating_advanced(a_cur, max_iters=25,
                                                time_budget=min(remaining * 0.3, 35.0))
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP r{round_num}: {best_score:.6f}")

            # L-BFGS with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 1000:
                tau_vsmall = 0.05 / n_best
                a_new, obj_new = optimize_lbfgs(a_cur, [0.3/n_best, 0.1/n_best, tau_vsmall],
                                                max_iter=200,
                                                time_budget=min(remaining * 0.25, 20.0))
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final L-BFGS r{round_num}: {best_score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new, obj_new = subgradient_optimize(a_cur, max_iters=int(remaining * 10),
                                                      time_budget=min(remaining * 0.2, 12.0),
                                                      tau_start=15.0, tau_end=60.0)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final SG r{round_num}: {best_score:.6f}")

            # Small perturbation + quick re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 997 + 313)
                scale = rng.uniform(0.01, 0.05)
                a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    if n_best <= 800 and remaining > 20:
                        a_pert = lp_alternating_advanced(a_pert, max_iters=10,
                                                         time_budget=min(remaining * 0.2, 15.0))
                    else:
                        a_pert, _ = subgradient_optimize(a_pert, max_iters=300,
                                                         time_budget=min(remaining * 0.15, 8.0),
                                                         tau_start=5.0, tau_end=20.0)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Final Perturb r{round_num}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
