import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = np.max(conv)
    return 2.0 * n * M / (S*S)

def build_toeplitz(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(2*n-1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()

def solve_lp(a, M, time_limit=5):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

        p = p / p.sum()

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_data(n):
    """Precompute row/col structure of the Toeplitz convolution matrix."""
    rows = []
    cols = []
    for j in range(n):
        for k in range(2 * n - 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
    return np.array(rows), np.array(cols)


def build_toeplitz_sparse(a, row_col_cache=None):
    n = len(a)
    if row_col_cache is not None:
        rows, cols = row_col_cache
        vals = np.zeros(len(rows))
        for idx, (r, c) in enumerate(zip(rows, cols)):
            vals[idx] = a[r - c]
    else:
        rows = []
        cols = []
        vals = []
        for j in range(n):
            for k in range(2 * n - 1):
                i = k - j
                if 0 <= i < n:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[i])
        rows, cols, vals = np.array(rows), np.array(cols), np.array(vals)
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp(a, M, row_col_cache=None, time_limit=5):
    n = len(a)
    A = build_toeplitz_sparse(a, row_col_cache)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0), res
    except Exception:
        pass
    return None, None


def lp_alternating(a, max_iters=300, time_budget=120.0, verbose=False):
    """LP-based alternating with adaptive M reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    row_col_cache = build_toeplitz_data(n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    m_reduction = 0.001
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Adaptive M: try slightly tighter than current peak
        M_target = M * (1.0 - m_reduction)
        g, res_lp = solve_lp(a, M_target, row_col_cache, time_limit=4)

        if g is None or np.sum(g) < 1e-10:
            # Try with current M
            g, res_lp = solve_lp(a, M, row_col_cache, time_limit=4)
            if g is None or np.sum(g) < 1e-10:
                break
            m_reduction = max(0.0, m_reduction - 0.0005)

        g = g / np.sum(g)

        # Line search with geometric progression
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            # Increase aggressiveness
            m_reduction = min(m_reduction + 0.0005, 0.05)
        else:
            stall += 1
            if stall > 10:
                # Perturb and reset
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.06, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                m_reduction = 0.001
                stall = 0
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                a = a / np.sum(a)

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: obj={best_obj:.6f}, M={M:.6f}, t={best_t:.2f}")

    return best_a, best_obj


def soft_subgradient(a, temperature=0.05, max_iters=2000, time_budget=30.0, verbose=False):
    """Soft subgradient descent with FFT-accelerated gradient and momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.9
    base_step = 0.08
    stall = 0

    # Precompute FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Soft-max weighting
        w = np.exp((conv - M) / temperature)
        w = w / w.sum()

        # Gradient via cross-correlation
        # grad[i] = sum_k w[k] * a[k-i] (a out of range = 0)
        w_full = np.zeros(size)
        w_full[:2*n-1] = w
        a_full = np.zeros(size)
        a_full[:n] = a
        w_ft = np.fft.rfft(w_full)
        a_ft = np.fft.rfft(a_full)
        # Cross-correlation: ifft(conj(w_ft) * a_ft)
        grad = np.fft.irfft(np.conj(w_ft) * a_ft, size)[:n]

        # Center gradient (preserve sum)
        grad_centered = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_centered
        step = base_step / (1.0 + 0.005 * it)

        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                rng = np.random.default_rng(it * 41 + 13)
                a = a * np.exp(rng.normal(0, 0.07, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0
                base_step *= 0.85
                if base_step < 1e-8:
                    break

        # Anneal temperature
        temperature = max(temperature * 0.9995, 0.01)

    return best_a, best_obj


def hankel_qp_solver(a_init, p, max_iters=50):
    """Solve min a^T Q a s.t. a >= 0, sum(a)=1 where Q = hankel(p)."""
    n = len(a_init)
    Q = hankel(p[:n], p[n-1:])

    def obj_fn(x):
        return x @ Q @ x

    def grad_fn(x):
        return 2.0 * Q @ x

    x0 = a_init.copy()
    x0 = x0 / np.sum(x0)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, None)] * n

    try:
        res = minimize(obj_fn, x0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iters, 'ftol': 1e-12})
        if res.success:
            x = np.maximum(res.x, 0)
            x = x / np.sum(x)
            return x
    except Exception:
        pass
    return a_init


def multiplicative_weights(a_init, max_iters=200, time_budget=60.0, eta=0.3):
    """Multiplicative weights approach using Hankel QP inner solver."""
    n = len(a_init)
    a = a_init.copy()
    a = a / np.sum(a)
    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve inner QP
        a_new = hankel_qp_solver(a, p, max_iters=30)
        if a_new is None or np.sum(a_new) < 1e-10:
            break

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # Update p via multiplicative weights
        conv = compute_conv(a_new)
        M = np.max(conv)
        if M < 1e-15:
            break
        log_p = np.log(np.maximum(p, 1e-30)) + eta * conv / M
        log_p -= np.max(log_p)
        p = np.exp(log_p)
        p = np.maximum(p, 1e-30)
        p = p / p.sum()

        a = a_new

    return best_a, best_obj


def cosine_init(n, n_modes=10, seed=0):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m ** 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def kaiser_init(n, alpha=5.0):
    """Kaiser window initialization."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    from scipy.special import i0
    a = i0(alpha * np.sqrt(np.maximum(1 - x**2, 0))) / i0(alpha)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def blackman_init(n):
    """Blackman window."""
    k = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * k / (n - 1)) + 0.08 * np.cos(4 * np.pi * k / (n - 1))
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def dolph_chebyshev_init(n, sidelobe_db=30):
    """Approximate Dolph-Chebyshev window."""
    k = np.arange(n)
    x = 2.0 * k / (n - 1) - 1.0
    # Approximate using raised cosine with sharp edges
    alpha = np.pi * sidelobe_db / 20.0
    a = np.cos(np.arccos(np.cos(np.pi * x)) / (np.pi + np.abs(x) * alpha))
    a = 0.5 * (1 + a)
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    return a / S


def refine_sequence(a, time_budget, n_lp=100, n_sg=100, verbose=False):
    """Full refinement pipeline."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP phase
        if remaining > 15 and n <= 800:
            lp_t = min(remaining * 0.35, 40.0)
            a_lp = lp_alternating(best_a, max_iters=int(lp_t * 3), time_budget=lp_t,
                                  verbose=verbose and round_num == 1)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    R{round_num} LP: {best_obj:.6f}")

        # Soft subgradient phase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_t = min(remaining * 0.25, 20.0)
            a_sg = soft_subgradient(best_a, temperature=0.05,
                                    max_iters=int(sg_t * 50), time_budget=sg_t)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    R{round_num} SG: {best_obj:.6f}")

        # MWU phase (for smaller n)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 400:
            mwu_t = min(remaining * 0.2, 15.0)
            a_mwu = multiplicative_weights(best_a, max_iters=int(mwu_t * 5),
                                           time_budget=mwu_t, eta=0.3)
            obj_mwu = compute_obj(a_mwu)
            if obj_mwu < best_obj:
                best_obj = obj_mwu
                best_a = a_mwu.copy()
                if verbose:
                    print(f"    R{round_num} MWU: {best_obj:.6f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            rng = np.random.default_rng(round_num * 97 + 11)
            for pert_idx in range(2):
                if time.monotonic() - start > time_budget - 5:
                    break
                scale = rng.uniform(0.04, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p < 1e-10:
                    continue
                a_pert = a_pert / S_p
                # Quick refinement
                a_pert = soft_subgradient(a_pert, temperature=0.08,
                                          max_iters=500, time_budget=min(3.0, remaining * 0.1))
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    R{round_num} Pert{pert_idx}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, soft subgradient, multiplicative weights,
    and multi-n exploration with perturbation restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 500

    print(f"Initial best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Multi-initialization search (20% of time) ===
    print("\n=== Phase 1: Multi-init search ===")
    phase1_budget = 0.20 * budget_s
    phase1_deadline = start_time + phase1_budget

    n_candidates = sorted(set([
        300, 400, n_parent, 600, 800
    ]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 1200]

    candidate_pool = []

    for n in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 15:
            break

        inits = [
            ("cos", cosine_init(n, n_modes=10, seed=seed + n * 7)),
            ("kaiser", kaiser_init(n, alpha=4.0)),
            ("kaiser2", kaiser_init(n, alpha=8.0)),
            ("blackman", blackman_init(n)),
        ]

        for name, a_init in inits:
            if time.monotonic() > phase1_deadline:
                break
            obj_init = compute_obj(a_init)
            # Quick subgradient
            a_quick = soft_subgradient(a_init, temperature=0.1,
                                       max_iters=500, time_budget=min(3.0, remaining * 0.1))
            obj_quick = compute_obj(a_quick)
            best_local = a_quick if obj_quick < obj_init else a_init
            best_local_obj = min(obj_quick, obj_init)
            candidate_pool.append((best_local, best_local_obj, n, name))
            if best_local_obj < best_score:
                best_score = best_local_obj
                best_seq = best_local.copy()
                print(f"  {name} (n={n}): {best_local_obj:.6f} ** NEW BEST **")
            else:
                print(f"  {name} (n={n}): {best_local_obj:.6f}")

    # Also try parent if available
    if parent is not None:
        candidate_pool.append((best_seq, best_score, n_parent, "parent"))

    # === Phase 2: Refine top candidates (35% of time) ===
    print("\n=== Phase 2: Refine top candidates ===")
    phase2_budget = 0.35 * budget_s
    phase2_deadline = start_time + phase1_budget + phase2_budget

    candidate_pool.sort(key=lambda x: x[1])
    n_refine = min(4, len(candidate_pool))

    for idx in range(n_refine):
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        a_cand, obj_cand, n_cand, name = candidate_pool[idx]
        if obj_cand > best_score + 0.1:
            break

        alloc = min(remaining * 0.35, 50.0)
        print(f"  Refining {name} (n={n_cand}, obj={obj_cand:.6f}), alloc={alloc:.0f}s")

        a_ref, obj_ref = refine_sequence(a_cand, time_budget=alloc, verbose=True)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"    ** NEW BEST: n={n_cand}, obj={best_score:.6f} **")

    # === Phase 3: Intensive LP + subgradient on best (25% of time) ===
    print("\n=== Phase 3: Intensive refinement ===")
    phase3_budget = 0.25 * budget_s
    phase3_deadline = start_time + phase1_budget + phase2_budget + phase3_budget

    a_cur = best_seq.copy()
    n_best = len(a_cur)

    for round_num in range(20):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.2, 30.0)

        # Alternate between LP and subgradient
        if round_num % 2 == 0 and n_best <= 800:
            a_new = lp_alternating(a_cur, max_iters=int(alloc * 4), time_budget=alloc)
        else:
            a_new = soft_subgradient(a_cur, temperature=0.04,
                                     max_iters=int(alloc * 40), time_budget=alloc)

        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  R{round_num}: {obj_new:.6f}")
            # Try perturbation
            rng = np.random.default_rng(round_num * 131 + 17)
            for pert_scale in [0.08, 0.15]:
                if time.monotonic() > phase3_deadline - 10:
                    break
                a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p > 1e-10:
                    a_pert = a_pert / S_p
                    a_pert = soft_subgradient(a_pert, temperature=0.06,
                                              max_iters=2000, time_budget=min(10.0, remaining * 0.15))
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        n_best = len(a_cur)
                        print(f"  R{round_num} pert({pert_scale}): {best_score:.6f} ** NEW BEST **")

    # === Phase 4: Multi-n exploration (15% of time) ===
    print("\n=== Phase 4: Multi-n exploration ===")
    phase4_budget = 0.15 * budget_s
    phase4_deadline = start_time + budget_s - 0.05 * budget_s

    n_base = len(best_seq)
    n_tries = sorted(set([
        max(100, n_base - 100), max(100, n_base - 50),
        n_base + 50, n_base + 100, n_base + 200,
        max(100, n_base + 300)
    ]))
    n_tries = [n for n in n_tries if 100 <= n <= 1500]

    for n_try in n_tries:
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 20:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, n_base)
        x_new = np.linspace(0, 1, n_try)
        try:
            f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
            a_interp = np.maximum(f_interp(x_new), 0)
        except Exception:
            a_interp = cosine_init(n_try, seed=seed + n_try)

        S_i = np.sum(a_interp)
        if S_i > 1e-10:
            a_interp = a_interp / S_i

        alloc = min(remaining * 0.3, 25.0)
        a_new = lp_alternating(a_interp, max_iters=int(alloc * 3), time_budget=alloc * 0.6)
        a_new = soft_subgradient(a_new, temperature=0.05,
                                 max_iters=int(alloc * 30), time_budget=alloc * 0.4)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 5: Final polish (5% of time) ===
    print("\n=== Phase 5: Final polish ===")
    a_cur = best_seq.copy()
    n_final = len(a_cur)
    remaining = deadline - time.monotonic()

    if remaining > 10:
        # Extended subgradient with low temperature
        a_sg = soft_subgradient(a_cur, temperature=0.02,
                                max_iters=int(remaining * 0.5 * 30),
                                time_budget=remaining * 0.4)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a_cur = a_sg.copy()
            print(f"  Final SG: {best_score:.6f} ** NEW BEST **")

        # Final LP
        remaining = deadline - time.monotonic()
        if remaining > 10 and n_final <= 600:
            a_lp = lp_alternating(a_cur, max_iters=int(remaining * 5),
                                  time_budget=min(remaining * 0.5, 20.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.6f} ** NEW BEST **")

        # Final perturbations
        remaining = deadline - time.monotonic()
        if remaining > 5:
            rng = np.random.default_rng(77777)
            for _ in range(3):
                if time.monotonic() > deadline - 5:
                    break
                a_p = a_cur * np.exp(rng.normal(0, 0.04, n_final))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p > 1e-10:
                    a_p = a_p / S_p
                    a_p = soft_subgradient(a_p, temperature=0.03,
                                           max_iters=3000, time_budget=min(remaining * 0.1, 8.0))
                    obj_p = compute_obj(a_p)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Perturb: {best_score:.6f} ** NEW BEST **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
