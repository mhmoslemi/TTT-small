from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def soft_max_objective(x, tau):
    """Compute soft-max J and its gradient."""
    a = np.maximum(x, 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_conv(a)
    b_scaled = b / tau
    b_max_s = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_s)
    log_sum_exp = b_max_s + np.log(exp_b.sum())
    F = tau * log_sum_exp  # soft-max value
    J = 2.0 * n * F / (s * s)
    
    # Gradient
    # softmax weights
    w = exp_b / exp_b.sum()
    
    # dF/da_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (valid indices)
    # This is a cross-correlation
    grad_F = np.zeros(n)
    for i in range(n):
        # w[i : i+max_j] dot a[:max_j]
        max_j = min(n, 2*n-1-i)
        if max_j > 0:
            grad_F[i] = 2.0 * np.dot(w[i:i+max_j], a[:max_j])
    
    grad_J = 2.0 * n * (grad_F * s - 2.0 * F) / (s ** 3)
    
    return J, grad_J

"""

1. Soft-max smoothed objective with L-BFGS-B (FFT-based gradients)
2. Fast coordinate descent with golden section search
3. LP-based best response refinement
4. Multi-length, multi-initialization exploration
5. Nesterov-accelerated subgradient as backup
Key innovation: soft-max L-BFGS-B navigates the non-smooth landscape differently

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_obj_and_grad(x, tau, n_target):
    """Compute soft-max smoothed J and gradient. x is the raw variable (a >= 0)."""
    a = np.maximum(x, 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_conv(a)
    b_scaled = b / tau
    b_max_s = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_s)
    sum_exp = exp_b.sum()
    log_sum_exp = b_max_s + np.log(sum_exp)
    F = tau * log_sum_exp
    J = 2.0 * n * F / (s * s)

    # Add penalty for deviating from target sum
    penalty_coeff = 1e-3
    J += penalty_coeff * (s - n_target) ** 2 / (n_target * n_target)

    # Softmax weights
    w = exp_b / sum_exp

    # Compute grad_F[i] = 2 * sum_j w[i+j] * a[j] via FFT cross-correlation
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(W * np.conj(A)).real[:n]
    grad_F = 2.0 * corr

    # dJ/da_i = 2n * (grad_F[i] * s - 2*F) / s^3 + penalty gradient
    grad_J = 2.0 * n * (grad_F * s - 2.0 * F) / (s ** 3)
    grad_J += 2.0 * penalty_coeff * (s - n_target) / (n_target * n_target)

    return J, grad_J

    """Optimize using soft-max L-BFGS-B with tau annealing."""
    a = np.maximum(a_init.astype(np.float64), 0.001)
    n = len(a)
    if n_target is None:
        n_target = n
    a = a * (n_target / a.sum())

    if tau_start is None:
        b = fast_conv(a)

    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_start
    plateau_count = 0

    for phase in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Define objective for L-BFGS-B
        def obj(x):
            val, _ = soft_max_obj_and_grad(x, tau, n_target)
            return val

        def grad(x):
            _, g = soft_max_obj_and_grad(x, tau, n_target)
            return g

        # Run L-BFGS-B

def coordinate_descent_1d(a, i, b, S, prefix_max, suffix_max):
    """Find optimal value for coordinate i."""
    n = len(a)
    ai = a[i]

    # Precompute unaffected max
    unaff_max = 0.0
    if i > 0:
        unaff_max = max(unaff_max, prefix_max[i - 1])
    if i + n < 2 * n - 1:
        unaff_max = max(unaff_max, suffix_max[i + n])

    # Precompute affected coefficients
    # For k in [i, i+n-1]: b[k] changes
    # k = 2i: quadratic (x^2 term)
    # k != 2i: linear, b[k] + 2*(x'-x)*a[k-i]
    lo_k = max(0, i)
    hi_k = min(2 * n - 1, i + n - 1)

    # Linear coefficients: for k != 2i, slope = 2*a[k-i], intercept = b[k] - 2*ai*a[k-i]
    linear_slopes = []
    linear_intercepts = []
    for k in range(lo_k, hi_k + 1):
        if k == 2 * i:
            continue
        ki = k - i
        if 0 <= ki < n:
            slope = 2.0 * a[ki]
            intercept = b[k] - 2.0 * ai * a[ki]
            linear_slopes.append(slope)
            linear_intercepts.append(intercept)

    linear_slopes = np.array(linear_slopes) if linear_slopes else np.array([0.0])
    linear_intercepts = np.array(linear_intercepts) if linear_intercepts else np.array([0.0])

    b_2i = b[2 * i] if 2 * i < 2 * n - 1 else 0.0

    def f(x):
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        # Quadratic part
        q_val = b_2i - ai * ai + x * x
        # Linear parts
        lin_vals = linear_intercepts + linear_slopes * x
        max_affected = max(q_val, lin_vals.max() if len(lin_vals) > 0 else 0.0)
        max_total = max(unaff_max, max_affected)
        return 2.0 * n * max_total / (new_S * new_S)

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai, S / n * 3.0, 0.1)

    c_point = hi - gr * (hi - lo)
    d_point = lo + gr * (hi - lo)
    fc = f(c_point)
    fd = f(d_point)

    for _ in range(45):
        if fc < fd:
            hi = d_point
            d_point = c_point
            fd = fc
            c_point = hi - gr * (hi - lo)
            fc = f(c_point)
        else:
            lo = c_point
            c_point = d_point
            fc = fd
            d_point = lo + gr * (hi - lo)
            fd = f(d_point)

    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a):
    """One full pass of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    n = len(a)
    b = fast_conv(a)

    # Precompute prefix and suffix maxima
    prefix_max = np.zeros(2 * n - 1)
    prefix_max[0] = b[0]
    for k in range(1, 2 * n - 1):
        prefix_max[k] = max(prefix_max[k - 1], b[k])
    suffix_max = np.zeros(2 * n - 1)
    suffix_max[2 * n - 2] = b[2 * n - 2]
    for k in range(2 * n - 3, -1, -1):
        suffix_max[k] = max(suffix_max[k + 1], b[k])

    indices = np.arange(n)
    np.random.shuffle(indices)

    for i in indices:
        new_val = coordinate_descent_1d(a, i, b, S, prefix_max, suffix_max)
        if abs(new_val - a[i]) > 1e-10:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update b incrementally
            lo_k = max(0, i)
            hi_k = min(2 * n - 1, i + n - 1)
            for k in range(lo_k, hi_k + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b_2i_old = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
            # Update prefix and suffix max (approximate - recompute in next pass)
            # For efficiency, skip updating prefix/suffix max here
            # (it's only used for the unaffected max, small error is OK)

    # Verify
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def coordinate_descent(a, n_passes, deadline, tracker):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum() * len(a)

    tracker.update(a)

    no_improve = 0
    for p in range(n_passes):

        a, new_J = coordinate_descent_pass(a)
        a = a / a.sum() * len(a)

        if new_J < current_J - 1e-12:
            current_J = new_J
            no_improve = 0
            tracker.update(a)
        else:
            no_improve += 1
            if no_improve >= 3:

    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def nesterov_subgradient(a, n_iters, deadline, tracker):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a.astype(np.float64), 0.001)
    n = len(a)
    a = a / a.sum() * n
    current_J = compute_J(a)
    tracker.update(a)

    v = np.zeros(n)
    a_prev = a.copy()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        S = a.sum()

        # Subgradient of max_b
        k_star = int(np.argmax(b))
        grad_b = np.zeros(n)
        for j in range(max(0, k_star - n + 1), min(n, k_star + 1)):
            ki = k_star - j
            if 0 <= ki < n:
                grad_b[j] = 2.0 * a[ki]

        # Gradient of J = 2n * b_max / S^2
        grad_J = 2.0 * n * (grad_b * S - 2.0 * b_max) / (S ** 3)

        # Nesterov step
        mu = 2.0 / (it + 2.0)
        a_new = a + mu * (a_prev - a)
        a_prev = a.copy()

        step = 1.0 / (it + 1)
        a_new = a_new - step * grad_J
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s * n

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            tracker.update(a)

        if it % 20 == 0:
            print(f"    Nesterov iter {it}: J={current_J:.8f}", flush=True)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.5, n))

    # Cosine modulations
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5]:
            inits.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.01))

    # Gaussian window
    center = n // 2
    for sigma in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n)) ** 2)
        inits.append(0.3 + 0.7 * g)

    # Inverted Gaussian (peaked at edges)
    for sigma in [0.1, 0.2]:
        g = np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n)) ** 2)
        inits.append(0.1 + 0.9 * (1.0 - g))

    # Sinusoidal
    for freq in [1, 2, 4]:
        inits.append(np.abs(np.sin(np.pi * freq * t)) + 0.1)

    # Two-tone
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt)
    inits.append(1.0 - 0.7 * alt)

    # Exponential decay from edges
    edge = np.minimum(np.arange(n), n - 1 - np.arange(n))
    inits.append(0.1 + 0.9 * (edge / (n / 2.0)) ** 0.5)

    # Random
    inits.append(rng.exponential(1.0, n) + 0.01)

    # From parent
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        # Perturbed parent
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))

    return inits

            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n={parent.size}", flush=True)

    n_parent = parent.size

    # === Phase 1: Multi-start exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-start exploration", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    lengths_to_try = [n_parent, n_parent - 2, n_parent + 2,
                      256, 320, 384, 400, 448, 512, 576, 640, 768]
    lengths_to_try = [l for l in lengths_to_try if 128 <= l <= 1024]
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)

    phase1_deadline = deadline - 600  # Reserve 600s for later phases
    best_from_phase1 = None
    best_J_phase1 = float('inf')

    for nl in unique_lengths:
        if time.monotonic() > phase1_deadline:
            break

        inits = generate_inits(nl, rng, parent)
        for init_idx, init in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break

            init = np.maximum(init, 0.001)
            init = init / init.sum() * nl

            # Quick soft-max optimization
            a_trial = soft_max_optimize(init, time.monotonic() + 30, tracker,
                                         tau_start=init.sum() / 20.0,
                                         tau_min=0.05, max_iter=20)

            J_trial = compute_J(a_trial)
            if J_trial < best_J_phase1:
                best_J_phase1 = J_trial
                best_from_phase1 = a_trial.copy()

            if init_idx % 3 == 0:
                print(f"    n={nl} init {init_idx}: J={J_trial:.8f}", flush=True)

    if best_from_phase1 is not None:
        tracker.update(best_from_phase1)

    tracker.report(f"Phase 1 done: {time.monotonic()-t0:.1f}s")

    # Use the best found so far
    a_best = tracker.best.copy() if tracker.best is not None else parent.copy()
    n_best = a_best.size

    # === Phase 2: Intensive optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Intensive optimization (n={n_best})", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    phase2_deadline = deadline - 300

    # Alternate between soft-max, CD, and LP
    for round_num in range(20):
        if time.monotonic() > phase2_deadline:
            break

        remaining = phase2_deadline - time.monotonic()
        if remaining < 10:
            break

        # Soft-max phase
        sm_budget = min(remaining * 0.4, 60)
        a_best = soft_max_optimize(a_best, time.monotonic() + sm_budget, tracker,
                                    tau_start=1.0, tau_min=0.01, max_iter=30)

        # Coordinate descent
        if time.monotonic() < phase2_deadline - 20:
            a_best = coordinate_descent(a_best, n_passes=5,
                                         deadline=min(time.monotonic() + 40, phase2_deadline - 20),
                                         tracker=tracker)

        # LP refinement (less frequent)
        if round_num % 3 == 2 and time.monotonic() < phase2_deadline - 30:
            a_best = lp_refine(a_best, n_iters=5,
                                deadline=min(time.monotonic() + 40, phase2_deadline - 30),
                                tracker=tracker)

        print(f"    Round {round_num}: J={compute_J(a_best):.8f}", flush=True)

    tracker.report(f"Phase 2 done: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Try different lengths from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Length exploration from best", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    phase3_deadline = deadline - 150

    nearby = [n_best - 4, n_best - 2, n_best + 2, n_best + 4, n_best + 8,
              n_best - 8, n_best + 16, n_best - 16,
              int(n_best * 0.8), int(n_best * 1.2)]
    nearby = [l for l in nearby if 100 <= l <= 1500]
    seen = set()
    unique_nearby = []
    for l in nearby:
        if l not in seen and l != n_best:
            seen.add(l)
            unique_nearby.append(l)

    for nl in unique_nearby:
        if time.monotonic() > phase3_deadline:
            break

        a_res = resample_seq(tracker.best, nl)
        a_res = np.maximum(a_res, 0.001)
        a_res = a_res / a_res.sum() * nl

        # Quick optimization
        a_trial = soft_max_optimize(a_res, time.monotonic() + 25, tracker,
                                     tau_start=1.0, tau_min=0.02, max_iter=15)
        if time.monotonic() < phase3_deadline - 10:
            a_trial = coordinate_descent(a_trial, n_passes=3,
                                          deadline=time.monotonic() + 20, tracker=tracker)

        print(f"    n={nl}: J={compute_J(a_trial):.8f}", flush=True)

    tracker.report(f"Phase 3 done: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Nesterov + perturbations ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Nesterov and perturbations", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    phase4_deadline = deadline - 60

    # Nesterov on best
    if time.monotonic() < phase4_deadline:
        a_nest = nesterov_subgradient(tracker.best, n_iters=100,
                                       deadline=min(time.monotonic() + 60, phase4_deadline),
                                       tracker=tracker)

    # Perturbation restarts
    for temp in [0.05, 0.02, 0.01, 0.005]:
        if time.monotonic() > phase4_deadline:
            break

        a_pert = tracker.best + rng.normal(0, temp, len(tracker.best))
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum() * len(a_pert)

        a_trial = soft_max_optimize(a_pert, time.monotonic() + 20, tracker,
                                     tau_start=0.5, tau_min=0.01, max_iter=15)
        if time.monotonic() < phase4_deadline - 5:
            a_trial = coordinate_descent(a_trial, n_passes=3,
                                          deadline=time.monotonic() + 15, tracker=tracker)

    tracker.report(f"Phase 4 done: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    # Intensive CD
    a_final = coordinate_descent(tracker.best, n_passes=20, deadline=deadline - 20, tracker=tracker)

    # Final soft-max with very small tau
    if time.monotonic() < deadline - 10:
        a_final = soft_max_optimize(a_final, deadline - 5, tracker,
                                     tau_start=0.1, tau_min=0.001, max_iter=20)

    # One more CD pass
    if time.monotonic() < deadline - 3:
        a_final, _ = coordinate_descent_pass(a_final)
        a_final = a_final / a_final.sum() * len(a_final)
        tracker.update(a_final)

    tracker.report(f"Phase 5 done: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
