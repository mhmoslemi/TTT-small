    tau_schedule: 'exp' (exponential decay), 'linear', 'fixed'

    best_J = compute_J(a)

    rho = 0.5

        if time.monotonic() > deadline:
            break

        if tau_schedule == 'exp':
            tau = max(tau_min, tau0 * np.exp(-0.005 * it))

Hybrid optimizer minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) fast smooth gradient descent with log-sum-exp, (2) LP best-response refinement,
(3) CMA-ES on Fourier parameters for global exploration, (4) multi-length search,
(5) diverse initialization and structured perturbations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient(a, conv, tau):
    """Compute gradient of log-sum-exp approximation of max(conv). Returns (grad, p)."""
    conv_max = conv.max()
    exp_conv = np.exp((conv - conv_max) / tau)
    sum_exp = exp_conv.sum()
    p = exp_conv / sum_exp
    # grad[i] = (2/tau) * sum_j p[i+j] * a[j]
    grad = (2.0 / tau) * np.correlate(p, a, mode='valid')
    return grad, p

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    total_nz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        total_nz += (j_max - j_min + 1)

    rows = np.zeros(total_nz, dtype=np.int32)
    cols = np.zeros(total_nz, dtype=np.int32)
    vals = np.zeros(total_nz, dtype=np.float64)

    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        cnt = j_max - j_min + 1
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(j_min, j_max + 1)
        vals[idx:idx+cnt] = a[k - j_min:k - j_min + cnt]
        idx += cnt

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 30:
                        break
            else:
                break
        except Exception:
            break

    return best_a, best_J


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):
    """Extract Fourier parameters from a sequence."""
    n = len(a)
    a_norm = a / a.sum()
    t = np.arange(n, dtype=np.float64) / n
    params = np.zeros(2 * n_harm + 1)
    params[0] = np.mean(a_norm)
    for k in range(1, n_harm + 1):
        params[2 * k - 1] = 2.0 * np.mean(a_norm * np.cos(2.0 * np.pi * k * t))
        params[2 * k] = 2.0 * np.mean(a_norm * np.sin(2.0 * np.pi * k * t))
    return params


def cmaes_eigen(C, z, dim):
    """Apply C^(1/2) @ z using eigendecomposition."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def run_cmaes_fourier(n, n_harm, params0, sigma0, deadline, tracker):
    """CMA-ES on Fourier parameters."""
    dim = 2 * n_harm + 1
    lam = 4 + int(3 * np.log(dim))
    mu = lam // 2

    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((dim + 2) ** 2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1 / (4 * dim) + 1 / (21 * dim ** 2))

    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)

    x = params0.copy()
    sigma = sigma0

    best_x = params0.copy()
    best_J = compute_J(fourier_to_seq(params0, n, n_harm))
    tracker.update(fourier_to_seq(params0, n, n_harm))

    for gen in range(5000):
        if time.monotonic() > deadline:
            break

        offspring = []
        offspring_J = []
        for i in range(lam):
            z = np.random.standard_normal(dim)
            y = cmaes_eigen(C, z, dim)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)

        idx = np.argsort(offspring_J)
        sorted_x = [offspring[i] for i in idx[:mu]]

        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * sorted_x[i]

        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-12)
        sigma = min(sigma, 0.5)

        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)

        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = sorted_x[i] - x
            rank_mu += weights[i] * np.outer(d, d)

        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (C + C.T) / 2
        x = x_new

        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()
            tracker.update(fourier_to_seq(best_x, n, n_harm))

    return best_x, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        # Slightly tapered
        taper = 1.0 - 0.3 * (2 * t - 1) ** 2
        inits.append(np.maximum(p * taper, 0.01))
        # Slightly edge-enhanced
        edge = 1.0 + 0.2 * (2 * t - 1) ** 2
        inits.append(np.maximum(p * edge, 0.01))

    # Standard shapes
    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    inits.append(1.0 / (1.0 + 0.5 * np.cos(2 * np.pi * t)))
    inits.append(np.maximum(1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.1 * np.cos(6 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 - 0.2 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))

    # Random smooth
    k = max(5, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # More structured
    for f in [2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)), 0.01))

    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: smooth gradient descent (primary), LP refinement,
    CMA-ES Fourier exploration, multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === Phase 1: Diversified smooth gradient descent ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Diversified smooth descent", flush=True)
    print(f"{'='*50}", flush=True)

    n_base = parent.size if parent.size <= 1024 else 640
    # Ensure reasonable length
    if n_base < 256:
        n_base = 512

    inits = generate_inits(n_base, rng, parent)
    # Limit number of inits for time management
    max_inits = 15
    inits = inits[:max_inits]

    per_init_time = max(3, min(15, int((deadline - t0 - 600) / max(1, len(inits)))))
    candidates = []

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 500:
            break
        a_init = np.maximum(a_init, 1e-10)
        a_init = a_init / a_init.sum()
        dl = time.monotonic() + per_init_time
        a_cand, J_cand = smooth_grad_descent(a_init, dl, tau0=0.05, tau_min=0.01,
                                             max_iters=5000, momentum_beta=0.8, tracker=tracker)
        candidates.append((J_cand, a_cand))
        print(f"  Init {i+1}/{len(inits)}: J={J_cand:.8f}", flush=True)

    # Sort by J
    candidates.sort(key=lambda x: x[0])
    print(f"  Best from Phase 1: J={candidates[0][0]:.8f}", flush=True)
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Long smooth descent on top candidates ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: Long smooth descent on top-3", flush=True)
    print(f"{'='*50}", flush=True)

    top_k = min(3, len(candidates))
    for i in range(top_k):
        if time.monotonic() > deadline - 350:
            break
        a_long, J_long = smooth_grad_descent(candidates[i][1],
                                             deadline - 300,
                                             tau0=0.03, tau_min=0.001,
                                             max_iters=500000,
                                             momentum_beta=0.85, tracker=tracker)
        print(f"  Top-{i+1}: J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_long < candidates[i][0]:
            candidates[i] = (J_long, a_long)

    candidates.sort(key=lambda x: x[0])
    best_a = candidates[0][1].copy()
    best_J = candidates[0][0]
    print(f"  Best after Phase 2: J={best_J:.8f}", flush=True)

    # === Phase 3: LP refinement ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: LP refinement (n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    a_lp = np.maximum(best_a, 1e-10)
    a_lp = a_lp / a_lp.sum()
    lp_dl = min(deadline - 200, time.monotonic() + 250)
    a_lp, J_lp = lp_mirror_descent(a_lp, lp_dl, max_iters=3000, use_multiplicative=False)
    tracker.update(a_lp)
    print(f"  LP linear: J={J_lp:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    if J_lp < best_J:
        best_a, best_J = a_lp.copy(), J_lp

    # Try multiplicative LP
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        a_lp2 = np.maximum(best_a, 1e-10)
        a_lp2 = a_lp2 / a_lp2.sum()
        a_lp2, J_lp2 = lp_mirror_descent(a_lp2, deadline - 130, max_iters=2000, use_multiplicative=True)
        tracker.update(a_lp2)
        print(f"  LP multiplicative: J={J_lp2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_lp2 < best_J:
            best_a, best_J = a_lp2.copy(), J_lp2

    # === Phase 4: Multi-length exploration ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Multi-length (base n={best_a.size})", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = best_a.size
    lengths_to_try = []
    for factor in [0.75, 1.25, 1.5, 2.0]:
        n_try = int(round(base_n * factor / 64.0) * 64)
        if 128 <= n_try <= 2048 and n_try != base_n:
            lengths_to_try.append(n_try)
    lengths_to_try = sorted(set(lengths_to_try))

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 100:
            break
        a_try = resample_seq(best_a, n_try)
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / a_try.sum()
        # Short smooth descent
        a_try, J_try = smooth_grad_descent(a_try, min(deadline - 80, time.monotonic() + 40),
                                           tau0=0.03, tau_min=0.005, max_iters=100000,
                                           momentum_beta=0.85, tracker=tracker)
        print(f"  n={n_try}: J={J_try:.8f}", flush=True)
        if J_try < best_J:
            best_a, best_J = a_try.copy(), J_try
            print(f"  -> New best: J={best_J:.8f} at n={n_try}", flush=True)

    print(f"  Phase 4 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: CMA-ES on Fourier params (if time remains) ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: CMA-ES Fourier (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        n_cma = best_a.size
        n_harm = min(20, max(5, n_cma // 8))
        params0 = seq_to_fourier(best_a, n_harm)
        params_best, J_cma = run_cmaes_fourier(n_cma, n_harm, params0, 0.03,
                                               deadline - 60, tracker)
        a_cma = fourier_to_seq(params_best, n_cma, n_harm)
        J_cma = compute_J(a_cma)
        print(f"  CMA-ES: J={J_cma:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_cma < best_J:
            best_a, best_J = a_cma.copy(), J_cma

        # LP from CMA result
        if time.monotonic() < deadline - 30 and J_cma <= best_J:
            a_cma_lp = np.maximum(a_cma, 1e-10)
            a_cma_lp = a_cma_lp / a_cma_lp.sum()
            a_cma_lp, J_cma_lp = lp_mirror_descent(a_cma_lp, deadline - 20,
                                                   max_iters=1000, use_multiplicative=False)
            tracker.update(a_cma_lp)
            print(f"  LP from CMA: J={J_cma_lp:.8f}", flush=True)
            if J_cma_lp < best_J:
                best_a, best_J = a_cma_lp.copy(), J_cma_lp

    # === Phase 6: Final polish with smooth gradient ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final smooth polish (n={best_a.size})", flush=True)
        print(f"{'='*50}", flush=True)

        a_final = np.maximum(best_a, 1e-10)
        a_final = a_final / a_final.sum()
        a_final, J_final = smooth_grad_descent(a_final, deadline - 5,
                                               tau0=0.01, tau_min=0.0005,
                                               max_iters=1000000,
                                               momentum_beta=0.9, tracker=tracker)
        print(f"  Final: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        if J_final < best_J:
            best_a, best_J = a_final.copy(), J_final

    # Ensure tracker has the best
    tracker.update(best_a)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = best_a

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
