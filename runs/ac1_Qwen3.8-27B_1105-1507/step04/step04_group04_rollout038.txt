"""
Step function optimization using:

2. Dual LP alternating (min-peak LP for both sequences)

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def lse_gradient_step(a, tau, step):
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    c = C[:2 * n - 1]
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_c / exp_sum
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(A * W.conj(), N)
    grad = np.zeros(n)
    grad[0] = 2.0 * cross[0]
    if n > 1:
        grad[1:] = cross[1:n] + cross[N - n + 1:N][::-1]
    # Gradient descent + projection
    v = a - step * grad
    a_new = project_simplex(v)
    return a_new

def lse_optimize(a, time_budget=30.0, tau_init=10.0, tau_max=200.0, step_init=0.1):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = step_init
    stall = 0

    while time.monotonic() - start < time_budget:
        a_new = lse_gradient_step(a, tau, step)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                if step > 0.005:
                    step *= 0.7
                else:
                    tau = min(tau * 1.3, tau_max)
                    step = step_init * 0.3
                stall = 0
                if tau > tau_max and step < 0.005:
                    break
        # Gradually increase tau
        tau = min(tau * 1.01, tau_max)
        step *= 0.9995

    return best_a

def solve_lp_minpeak_fixed(a, time_limit=10):
    """Solve: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def dual_lp_alternate(a, max_iters=50, time_budget=60.0):
    """Alternate min-peak LPs for both a and g."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    g = a.copy()  # Initialize g = a
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        # Step 1: Given a, find optimal g
        g_new, M_g = solve_lp_minpeak_fixed(a, time_limit=5)
        if g_new is None:
            break
        g = g_new

        # Step 2: Given g, find optimal a'
        a_new, M_a = solve_lp_minpeak_fixed(g, time_limit=5)
        if a_new is None:
            break
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        a_new = a_new / s

        # Check if a_new is "flat" (good for our objective)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            # Take a step toward a_new
            stall += 1
            if stall > 20:
                # Try blending
                t = 0.5
                a_blend = (1-t) * best_a + t * a_new
                a_blend = project_simplex(a_blend)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                break

        # Blend for stability
        a = 0.5 * a + 0.5 * a_new
        a = project_simplex(a)

    return best_a

def local_search(a, time_budget=30.0):
    """Local search with perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        rng = np.random.default_rng(attempt * 7919 + 42)
        scale = 0.01 + 0.05 * (attempt % 5) / 5.0
        a_pert = best_a * np.exp(rng.normal(0, scale, n))
        a_pert = project_simplex(a_pert)
        
        # Quick LSE optimization
        a_opt = lse_optimize(a_pert, time_budget=min(8.0, (time_budget - (time.monotonic() - start)) * 0.5),
                             tau_init=30.0, tau_max=100.0, step_init=0.05)
        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a

            amp = rng.uniform(0.2, 0.5)

def resample_sequence(a, new_n):

    if n == new_n:
        return a.copy()

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Step function optimization combining:
1. Log-sum-exp gradient with simplex projection (smooth min-max approach)
2. Dual LP alternating (min-peak LP solved for both a and g)
3. SCA-like linearized quadratic constraints
4. Intensive local search with diverse perturbations
5. Systematic n exploration with resampling from best parent
6. Sinusoidal multi-scale initialization for diversity
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def lse_gradient_step(a, tau, step):
    """One gradient step on log-sum-exp(tau * conv(a,a)) with simplex projection."""
    n = len(a)
    N = 1
    while N < 4 * n:
        N <<= 1
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    c = C[:2 * n - 1]
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_c / exp_sum
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(A * W.conj(), N)
    grad = np.zeros(n)
    grad[0] = 2.0 * cross[0]
    if n > 1:
        grad[1:] = cross[1:n] + cross[N - n + 1:N][::-1]
    v = a - step * grad
    a_new = project_simplex(v)
    return a_new


def lse_optimize(a, time_budget=30.0, tau_init=20.0, tau_max=200.0, step_init=0.1):
    """Optimize using log-sum-exp gradient with simplex projection and adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    tau = tau_init
    step = step_init
    stall = 0
    max_stall = 80

    while time.monotonic() - start < time_budget:
        a_new = lse_gradient_step(a, tau, step)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                if step > 0.005:
                    step *= 0.6
                else:
                    tau = min(tau * 1.5, tau_max)
                    step = step_init * 0.2
                stall = 0
                max_stall = max(30, max_stall // 2)
                if tau >= tau_max * 0.95 and step < 0.005:
                    break
        tau = min(tau * 1.005, tau_max)
        step *= 0.9997

    return best_a


def solve_lp_minpeak(fixed_seq, time_limit=8):
    """Solve: min M s.t. (fixed_seq * x)_k <= M for all k, sum(x)=1, x>=0.
    Returns (x_optimal, M_optimal) or (None, None)."""
    n = len(fixed_seq)
    A = build_toeplitz(fixed_seq)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            s = np.sum(x)
            if s < 1e-10:
                return None, None
            x = x / s
            return x, float(res.x[n])
    except Exception:
        pass
    return None, None


def dual_lp_alternate(a, max_iters=80, time_budget=60.0):
    """Alternate min-peak LPs: given a find best g, given g find best a."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        # Given current a, find optimal g
        g, M_g = solve_lp_minpeak(a, time_limit=5)
        if g is None:
            break

        # Given g, find optimal a_candidate
        a_cand, M_a = solve_lp_minpeak(g, time_limit=5)
        if a_cand is None:
            break
        a_cand = project_simplex(a_cand)

        obj = compute_obj(a_cand)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_cand.copy()
            stall = 0
            # Blend for stability
            a = 0.4 * a + 0.6 * a_cand
            a = project_simplex(a)
        else:
            stall += 1
            if stall > 30:
                # Try a few blends
                improved = False
                for t in [0.3, 0.5, 0.7]:
                    blend = (1 - t) * best_a + t * a_cand
                    blend = project_simplex(blend)
                    ob = compute_obj(blend)
                    if ob < best_obj:
                        best_obj = ob
                        best_a = blend.copy()
                        a = blend.copy()
                        improved = True
                        break
                if not improved:
                    break
            else:
                a = 0.7 * a + 0.3 * a_cand
                a = project_simplex(a)

    return best_a


def sca_lp_step(a, time_limit=8):
    """One SCA iteration: linearize quadratic constraints and solve LP.
    Min M s.t. linearized conv(a,a)_k <= M, sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    # Build linearized constraint matrix
    # Linearization: sum_j (a_j * x_{k-j} + a_{k-j} * x_j) - conv(a,a)_k <= M
    # This is: 2 * (a * x)_k - conv(a,a)_k <= M (by symmetry of convolution)
    # So constraint is: 2*(a*x)_k <= M + conv(a,a)_k
    A_toep = build_toeplitz(a)
    c_conv = compute_conv(a)
    n_constraints = 2 * n - 1

    # Variables: x_0,...,x_{n-1}, M
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = 2.0 * A_toep
    A_ub[:, n] = -1.0
    b_ub = c_conv.copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'prestore': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            s_x = np.sum(x)
            if s_x < 1e-10:
                return None
            x = x / s_x
            return x
    except Exception:
        pass
    return None


def sca_optimize(a, max_iters=100, time_budget=60.0):
    """SCA optimization: iteratively linearize and solve LP."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        a_new = sca_lp_step(a, time_limit=5)
        if a_new is None:
            break

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            # Damped update
            a = 0.5 * a + 0.5 * a_new
            a = project_simplex(a)
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
            # Try damped blend
            a = 0.8 * a + 0.2 * a_new
            a = project_simplex(a)

    return best_a


def local_search(a, time_budget=30.0):
    """Intensive local search with diverse perturbations and quick re-optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    attempt = 0

    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining_inner = time_budget - (time.monotonic() - start)
        if remaining_inner < 12:
            break

        rng = np.random.default_rng(attempt * 7919 + 12345)

        # Vary perturbation type
        ptype = attempt % 4
        if ptype == 0:
            scale = 0.02 + 0.04 * (attempt % 3) / 3.0
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
        elif ptype == 1:
            # Sparse perturbation
            a_pert = best_a.copy()
            n_pert = max(3, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_pert[idx] *= np.exp(rng.normal(0, 0.15, n_pert))
        elif ptype == 2:
            # Block perturbation
            a_pert = best_a.copy()
            block_size = max(5, n // 8)
            start_idx = rng.integers(0, n - block_size)
            a_pert[start_idx:start_idx + block_size] *= np.exp(rng.normal(0, 0.1, block_size))
        else:
            # Smooth perturbation via low-frequency noise
            freq = rng.uniform(0.5, 3.0)
            a_pert = best_a * (1.0 + 0.05 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi)))

        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = project_simplex(a_pert)

        # Quick optimization
        sub_time = min(10.0, remaining_inner * 0.4)
        a_opt = lse_optimize(a_pert, time_budget=sub_time, tau_init=40.0, tau_max=150.0, step_init=0.05)
        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.15, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.5 * (2 ** scale), 2.5 * (2 ** scale))
            amp = 0.35 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * k / n)
        a += 0.2 * np.cos(2 * np.pi * rng.uniform(3, 8) * k / n)
    elif choice == 7:
        # Two-center bump
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.2)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - c1) / w) ** 2) + 0.5 * np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 8:
        # Random smooth via interpolation
        a_coarse = rng.random(max(5, n // 8))
        a = np.interp(k, np.arange(len(a_coarse)), a_coarse)
        a = 0.5 + a  # Keep positive
    else:
        # Chirp-like
        f0 = rng.uniform(0.3, 1.0)
        f1 = rng.uniform(3.0, 10.0)
        t = k / n
        f = f0 + (f1 - f0) * t
        a = 1.0 + 0.4 * np.cos(2 * np.pi * np.cumsum(f))

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n, dtype=np.float64)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def optimize_single(a, time_budget=60.0):
    """Full optimization pipeline on a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LSE gradient (coarse to fine)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t = min(remaining * 0.25, 20.0)
        a_new = lse_optimize(best_a, time_budget=t, tau_init=15.0, tau_max=100.0, step_init=0.1)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 2: SCA LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t = min(remaining * 0.25, 25.0)
        a_new = sca_optimize(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 3: Dual LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t = min(remaining * 0.25, 25.0)
        a_new = dual_lp_alternate(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 4: LSE with high tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(remaining * 0.2, 15.0)
        a_new = lse_optimize(best_a, time_budget=t, tau_init=50.0, tau_max=300.0, step_init=0.03)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    # Phase 5: Local search
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t = min(remaining * 0.25, 20.0)
        a_new = local_search(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def optimize_n(n, seed, time_budget, parent_seq=None):
    """Run optimization for a given n with multiple initializations."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    n_inits = max(2, min(4, int(time_budget / 25)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 12:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        if init_idx == 0 and parent_seq is not None:
            a = resample_sequence(parent_seq, n)
        else:
            a = random_init(n, seed + init_idx * 79193)

        sub_budget = min(remaining * 0.6, 70.0)
        a_opt, obj = optimize_single(a, time_budget=sub_budget)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj
