import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient(a, tau):
    """
    Compute gradient of smooth max: (1/tau)*log(sum exp(tau*b_k)) w.r.t. a.
    Returns (gradient, softmax_weights, convolution).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, p, b


def exact_subgradient(a, tol_frac=1e-10):
    """Compute exact subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = max(tol_frac * M, 1e-12)
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, active, b


def run_smooth_grad(a, max_iters, deadline, tau_start=3.0, tau_end=60.0,
                    lr_start=0.25, lr_end=0.02, momentum=0.85):
    """Smooth gradient descent with momentum and adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = float(it) / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** frac

        grad, p, b = smooth_gradient(a, tau)
        M = b.max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def build_conv_matrix(n, a):
    """Build sparse matrix for conv(a, .) constraints."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, np.array(vals, dtype=np.float64)


def lp_best_response(a, M, time_limit=20.0):
    """
    LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    rows, cols, vals = build_conv_matrix(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, lp_time=15.0):
    """LP mirror descent with aggressive line search and convergence detection."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    iter_count = 0

    while iter_count < max_iters:
        if time.monotonic() > deadline:
            break
        iter_count += 1

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break

        # Aggressive line search
        improved = False
        base_t = max(0.1, 0.8 * (1.0 - 0.5 * iter_count / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.15, 0.1, 0.05, 0.02]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try a different strategy: use g directly with smaller step
            for t in [0.01, 0.005]:
                a_trial = (1 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-13:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                if no_improve >= 15:
                    break

    return best_a


def perturb_sequence(a, rng, mode='gaussian', scale=0.05):
    """Apply a structured perturbation to escape local optima."""
    n = len(a)
    a_p = a.copy().astype(np.float64)

    if mode == 'gaussian':
        a_p += rng.normal(0, scale, n)
    elif mode == 'swap':
        # Swap pairs of elements
        for _ in range(max(1, n // 32)):
            i, j = rng.integers(0, n, 2)
            a_p[i], a_p[j] = a_p[j], a_p[i]
    elif mode == 'block_shift':
        # Shift a random block
        block_size = rng.integers(n // 16, n // 4 + 1)
        start = rng.integers(0, n - block_size)
        a_p[start:start + block_size], a_p[start + block_size:start + 2 * block_size] = \
            a_p[start + block_size:start + 2 * block_size], a_p[start:start + block_size]
    elif mode == 'mirror_break':
        # Perturb the symmetry
        half = n // 2
        noise = rng.normal(0, scale * 0.5, half)
        a_p[:half] += noise
        a_p[half:] += noise[::-1] * rng.normal(0.5, 0.3, 1)
    elif mode == 'frequency':
        # Add a sinusoidal perturbation
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.linspace(0, 1, n, endpoint=False)
        a_p += scale * 0.3 * np.sin(2 * np.pi * freq * t + phase)
    elif mode == 'sparse':
        # Perturb only a few random positions
        k = rng.integers(n // 10, n // 3 + 1)
        idx = rng.choice(n, k, replace=False)
        a_p[idx] += rng.normal(0, scale * 2, k)

    a_p = np.maximum(a_p, 0.001)
    a_p = a_p / a_p.sum()
    return a_p


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import interp1d
    f = interp1d(t_old, src, kind='cubic', fill_value="extrapolate")
    return np.maximum(f(t_new), 0.0)


def generate_asymmetric_inits(n, rng, parent=None):
    """Generate diverse asymmetric initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Asymmetric perturbation
        asym = np.linspace(-0.15, 0.15, n)
        inits.append(np.maximum(p + asym, 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * asym), 0.01))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Linear ramp (highly asymmetric)
    inits.append(np.maximum(0.3 + 1.4 * t, 0.01))
    inits.append(np.maximum(1.7 - 1.4 * t, 0.01))

    # Quadratic
    inits.append(np.maximum(0.2 + 1.5 * t**2, 0.01))
    inits.append(np.maximum(1.7 - 1.5 * t**2, 0.01))

    # Exponential
    inits.append(np.maximum(0.1 + 1.5 * np.exp(2 * t), 0.01))
    inits.append(np.maximum(0.1 + 1.5 * np.exp(-2 * t), 0.01))

    # Sine (asymmetric phase)
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 3, 2 * np.pi / 3]:
            inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * freq * t + phase), 0.01))

    # Product of different frequencies
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 2 * t) + 0.3 * np.cos(2 * np.pi * 5 * t), 0.01))

    # Gaussian (off-center)
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.maximum(0.2 + 1.0 * np.exp(-8 * (t - center)**2), 0.01))

    # Two bumps (asymmetric)
    inits.append(np.maximum(0.1 + 0.8 * np.exp(-15 * (t - 0.25)**2) + 0.5 * np.exp(-15 * (t - 0.75)**2), 0.01))
    inits.append(np.maximum(0.1 + 0.5 * np.exp(-10 * (t - 0.2)**2) + 0.8 * np.exp(-12 * (t - 0.8)**2), 0.01))

    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Random with trend
    trend = rng.uniform(0.3, 1.0) * t
    noise = np.convolve(rng.normal(0, 0.2, n), kernel, mode='same')
    inits.append(np.maximum(0.3 + trend + noise, 0.01))

    # Power law
    inits.append(np.maximum(0.2 + t**0.5, 0.01))
    inits.append(np.maximum(0.2 + (1-t)**0.5, 0.01))

    # Cosine window (symmetric but good starting point)
    inits.append(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)))

    # Triangle (V-shape, asymmetric peak)
    for peak in [0.3, 0.5, 0.7]:
        inits.append(np.maximum(1.5 - 1.0 * np.abs(t - peak), 0.05))

    return inits

        self.best_score = float('inf')
        self.t0 = time.monotonic()
        self.n_updates = 0

    def update(self, a, verbose=True):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.n_updates += 1
            elapsed = time.monotonic() - self.t0
            if verbose:
                print(f"  *** J={score:.10f} n={a.size} updates={self.n_updates} t={elapsed:.1f}s", flush=True)

    def report(self, phase=""):
        if self.best is not None:
            print(f"  [{phase}] best: J={self.best_score:.10f} n={self.best.size} t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smooth gradient, LP mirror descent,
    and iterated local search with diverse perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, verbose=False)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, verbose=False)

    # Ensure we have a best to work with
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(parent)

    # === PHASE 1: Fast exploration across lengths ===
    phase1_deadline = time.monotonic() + 250.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fast Gradient Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [128, 192, 256, 320, 384, 512, 640, 768, 896, 1024]
    # Weight time allocation
    weights = {n: 1.0 / max(1, n / 128) for n in lengths}
    total_w = sum(weights.values())

    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break

        time_alloc = 250.0 * weights[n] / total_w
        time_alloc = max(15.0, min(60.0, time_alloc))
        local_deadline = min(time.monotonic() + time_alloc, phase1_deadline - 5)

        if time.monotonic() >= local_deadline:
            continue

        t0 = time.monotonic()
        inits = generate_asymmetric_inits(n, rng, tracker.best)
        best_M = float('inf')
        best_a = None

        for a0 in inits:
            if time.monotonic() > local_deadline - 2:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            iters = max(30, int(1500 / max(1, n // 64)))
            a = run_smooth_grad(a0, iters, local_deadline - 2,
                                tau_start=3.0, tau_end=50.0,
                                lr_start=0.2, lr_end=0.03)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

        if best_a is not None:
            J = compute_J(best_a)
            print(f"  n={n:4d}: M={best_M:.10f} J={J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_a, verbose=False)
            if compute_J(tracker.best) < compute_J(best_a):
                pass  # tracker already has better
            else:
                tracker.update(best_a)

    tracker.report("Phase 1")

    # === PHASE 2: LP Mirror Descent (main optimization) ===
    phase2_deadline = time.monotonic() + min(500.0, max(200.0, deadline - time.monotonic() - 250.0))
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    # Estimate LP time
    est_lp = 0.005 * (n_best / 128.0) ** 2
    lp_time = min(20.0, max(5.0, 300.0 / max(50, int(300.0 / est_lp))))
    print(f"  n={n_best}, est LP time={est_lp*1000:.0f}ms, lp_time_limit={lp_time:.1f}s", flush=True)

    a = tracker.best.copy()
    max_lp_iters = max(30, int(400.0 / max(0.05, est_lp)))
    print(f"  max LP iterations: {max_lp_iters}", flush=True)

    a = lp_mirror_descent(a, max_lp_iters, min(phase2_deadline, deadline - 200), lp_time=lp_time)
    tracker.update(a)

    # Second LP pass with slightly different LP time
    if time.monotonic() < deadline - 150:
        a = lp_mirror_descent(a, 50, min(time.monotonic() + 100, deadline - 150), lp_time=lp_time * 0.8)
        tracker.update(a)

    tracker.report("Phase 2")

    # === PHASE 3: Iterated Local Search ===
    phase3_deadline = min(time.monotonic() + 300.0, deadline - 100.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Iterated Local Search", flush=True)
    print(f"{'='*60}", flush=True)

    perturb_modes = ['gaussian', 'mirror_break', 'frequency', 'sparse', 'swap', 'block_shift']
    temp = 0.04
    stall_count = 0

    while time.monotonic() < phase3_deadline:
        a_p = perturb_sequence(tracker.best, rng, mode=perturb_modes[rng.integers(len(perturb_modes))],
                               scale=temp)

        # Quick gradient polish
        a_p = run_smooth_grad(a_p, 30, min(time.monotonic() + 5, phase3_deadline - 5),
                              tau_start=10, tau_end=40, lr_start=0.1, lr_end=0.02)
        tracker.update(a_p, verbose=False)

        # Short LP pass
        lp_t = min(lp_time * 0.5, max(3.0, (phase3_deadline - time.monotonic()) / 5.0))
        if time.monotonic() < phase3_deadline - lp_t - 2:
            a_p = lp_mirror_descent(a_p, 10, time.monotonic() + lp_t + 2, lp_time=lp_t)
            tracker.update(a_p, verbose=False)

        if tracker.n_updates % 10 == 0:
            stall_count = 0
        else:
            stall_count += 1
            temp *= 0.95
            if temp < 0.005:
                temp = 0.04  # Reset
                print(f"  [ILS] Temp reset at t={time.monotonic()-tracker.t0:.0f}s", flush=True)

    tracker.report("Phase 3")

    # === PHASE 4: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = tracker.best.size
    tried_n = {base_n}

    # Try nearby lengths
    for new_n in [base_n - 64, base_n - 32, base_n + 32, base_n + 64,
                  base_n * 2, base_n // 2 if base_n >= 256 else base_n * 3]:
        if time.monotonic() > deadline - 30:
            break
        if new_n < 64 or new_n > 2048 or new_n in tried_n:
            continue
        tried_n.add(new_n)

        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Gradient polish
        a_ext = run_smooth_grad(a_ext, 50, min(time.monotonic() + 20, deadline - 25),
                                tau_start=5, tau_end=40, lr_start=0.15, lr_end=0.03)
        tracker.update(a_ext, verbose=False)

        # Short LP
        if time.monotonic() < deadline - 30:
            est = 0.005 * (new_n / 128.0) ** 2
            lp_t = min(10.0, max(3.0, 50.0 / max(5, int(50.0 / est))))
            a_ext = lp_mirror_descent(a_ext, 15, min(time.monotonic() + lp_t * 15, deadline - 25), lp_time=lp_t)
            tracker.update(a_ext, verbose=False)

    tracker.report("Phase 4")

    # === PHASE 5: Final concentrated LP ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 10:
        n_f = tracker.best.size
        est_lp_f = 0.005 * (n_f / 128.0) ** 2
        remaining = deadline - time.monotonic() - 5
        max_iters_f = max(10, int(remaining / max(0.05, est_lp_f)))
        print(f"  n={n_f}, remaining={remaining:.0f}s, max_iters={max_iters_f}", flush=True)

        a = lp_mirror_descent(tracker.best, max_iters_f, deadline - 5, lp_time=min(20.0, est_lp_f * 5))
        tracker.update(a)

    tracker.report("Phase 5 / FINAL")

    # Fallback
    if tracker.best is None or tracker.best_score == float('inf'):
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.10f} n={tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t0:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
