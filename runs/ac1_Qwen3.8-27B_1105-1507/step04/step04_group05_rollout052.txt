1. Rapid LP mirror descent with adaptive tightening and exact line search
2. Efficient coordinate descent with O(n) per-coordinate updates
3. Dual-guided refinement using LP dual variables
4. Multi-length exploration and perturbation restarts
Key improvements: proper line search, faster CD, better time management.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_coo(a):
    """Build sparse constraint matrix for LP in COO format (efficient)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_coo(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None


def line_search_lp(a, g, current_J, max_iter=50):
    """Exact line search along direction g from a. Returns best t and new a."""
    n = len(a)
    S = a.sum()
    g_sum = g.sum()
    
    # f(t) = J((1-t)*a + t*g) normalized
    lo = 0.0
    hi = 1.0
    
    def f_of_t(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return 1e10
        b = fast_conv(a_t)
        return 2.0 * n * b.max() / (s * s)
    
    best_t = 0.0
    best_J = current_J
    
    # First check t=1
    J_at_1 = f_of_t(1.0)
    if J_at_1 < best_J:
        best_t = 1.0
        best_J = J_at_1
    
    # Bisection for minimum in [0, 1]
    # f(t) should be unimodal (or close to it)
    t_mid = 0.5
    f_mid = f_of_t(t_mid)
    
    # Try to find the minimum via ternary-like search
    for _ in range(max_iter):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        f1 = f_of_t(t1)
        f2 = f_of_t(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
        if hi - lo < 1e-8:
            break
    
    t_opt = (lo + hi) / 2.0
    f_opt = f_of_t(t_opt)
    
    if f_opt < best_J:
        best_t = t_opt
        best_J = f_opt
    
    if best_t < 1e-10:
        return a, current_J
    
    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a, current_J
    return a_new, best_J


def coordinate_descent_1d(a, i, b, S):
    """Find optimal value for coordinate i using golden section search.
    Only scans affected convolution indices for efficiency."""
    n = len(a)
    ai = a[i]
    
    # Precompute max of unaffected b values
    # Affected k range: [i, i+n-1] clipped to [0, 2n-2]
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        b_unaffected_max = b[mask].max() if mask.any() else 0.0
    else:
        b_unaffected_max = 0.0
    
    # Precompute coefficients for affected k
    affected_ks = list(range(k_lo, k_hi + 1))
    # For each affected k: c_k(x) = b[k] + 2*(x-ai)*a[k-i] (if k != 2i)
    #                      or  c_k(x) = b[k] - ai^2 + x^2 (if k == 2i)
    
    def f(x):
        """Compute J when a[i] = x."""
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        max_c = b_unaffected_max
        for k in affected_ks:
            ki = k - i
            if 0 <= ki < n:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ck = b[k] + 2.0 * (x - ai) * a[ki]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    # Upper bound: try current value * 3 or mean * 3
    hi = max(2.0 * ai, 2.0 * S / n, 0.01)
    
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c_pt)
    fd = f(d_pt)
    
    for _ in range(50):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
        if hi - lo < 1e-12:
            break
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a, max_coords=None, shuffle_seed=None):
    """One full pass of coordinate descent with incremental convolution update."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a, compute_J(a)
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    n_coords = n if max_coords is None else min(max_coords, n)
    if shuffle_seed is not None:
        rng = np.random.default_rng(shuffle_seed)
    else:
        rng = np.random.default_rng()
    indices = list(range(n))
    rng.shuffle(indices)
    indices = indices[:n_coords]
    
    improved = False
    for i in indices:
        new_val = coordinate_descent_1d(a, i, b, S)
        if abs(new_val - a[i]) > 1e-10:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            improved = True
            # Update convolution incrementally
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] + 2.0 * (new_val - old_val) * a[ki]
    
    if improved:
        # Verify with full convolution periodically
        b_check = fast_conv(a)
        new_J = 2.0 * n * b_check.max() / (S * S)
    else:
        new_J = current_J
    
    return a, new_J


def run_lp_iterations(a, max_iters, deadline, tracker, tighten_base=0.004):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try tight constraint first
        M_target = M * (1.0 - tighten)
        g, lp_result = lp_best_response(a, M_target)
        
        if g is None:
            # Relax
            M_target = M * (1.0 - tighten * 0.3)
            g, lp_result = lp_best_response(a, M_target)
        
        if g is None:
            tighten *= 0.5
            if tighten < 1e-5:
                tighten = tighten_base
            stall_count += 1
            if stall_count > 30:
                break
            continue
        
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        
        # Line search
        a_new, J_new = line_search_lp(a, g, current_J, max_iter=40)
        
        if J_new < current_J - 1e-12:
            a = a_new
            improvement = current_J - J_new
            current_J = J_new
            stall_count = 0
            # Adapt tightening
            if it % 50 == 0:
                tighten = min(0.02, tighten * 1.02)
            if improvement < 1e-8:
                tighten = max(0.001, tighten * 0.95)
            tracker.update(a)
        else:
            # Try smaller step
            a_half = 0.5 * a + 0.5 * g
            a_half = np.maximum(a_half, 0.0)
            if a_half.sum() > 1e-10:
                J_half = compute_J(a_half)
                if J_half < current_J - 1e-12:
                    a = a_half
                    current_J = J_half
                    stall_count = 0
                    tracker.update(a)
                else:
                    stall_count += 1
                    tighten *= 0.7
            else:
                stall_count += 1
                tighten *= 0.7
            
            if stall_count > 50:
                break
        
        if it % 50 == 0:
            print(f"    LP iter {it}: J={current_J:.10f}, tighten={tighten:.5f}", flush=True)
    
    tracker.update(a)
    return a


def dual_guided_cd(a, n_passes, deadline, tracker):
    """Coordinate descent guided by which convolution lags are active."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        S = a.sum()
        
        # Find active lags
        tol = 0.02 * b_max
        active_lags = np.where(b >= b_max - tol)[0]
        
        # Prioritize coordinates that affect active lags
        # Coordinate i affects lags k where 0 <= k-i < n, i.e., k in [i, i+n-1]
        priority = np.zeros(n)
        for k in active_lags:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                priority[i] += 1.0
        
        # Sort by priority (descending), then shuffle within same priority
        indices = list(range(n))
        # Weighted random selection biased toward high priority
        if priority.max() > 0:
            weights = priority + 0.1
            weights /= weights.sum()
            # Select coordinates proportional to priority
            for _ in range(n):
                idx = np.random.choice(n, p=weights)
                if idx not in indices or indices.count(idx) < 3:
                    pass
        else:
            np.random.shuffle(indices)
        
        # Standard CD pass with priority ordering
        a, new_J = coordinate_descent_pass(a, max_coords=n)
        
        if new_J < current_J - 1e-12:
            current_J = new_J
            tracker.update(a)
        else:
            # Try a few more passes if stalled
            for extra in range(3):
                a, new_J = coordinate_descent_pass(a)
                if new_J < current_J - 1e-12:
                    current_J = new_J
                    tracker.update(a)
                    break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s ***", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Cosine variations
    for freq in [1, 2, 3, 5]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))
    
    # Edge-concentrated
    inits.append(1.0 + 0.5 * np.cos(np.pi * t))
    
    # Exponential decay shape
    inits.append(np.exp(-2.0 * np.abs(t - 0.5) * 4))
    
    # Two-lobe
    inits.append(0.3 + 0.7 * (np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2)))
    
    # Synchronized sine product
    inits.append(np.abs(np.sin(np.pi * t)) ** 0.5)
    
    return [np.maximum(x, 0.01) for x in inits]


def subgradient_step(a, step_size, top_k=5):
    """One subgradient step focusing on top-k active lags."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Get top-k active lags
    top_indices = np.argsort(b)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        weight = b[k] / b_max  # Weight by how close to max
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            ki = k - j
            if 0 <= ki < n:
                grad[j] += weight * a[ki]
    
    # Subtract sum derivative term
    grad -= 2.0 * b_max / S * np.ones(n)
    
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a
    return a_new

        # One more LP
        if time.monotonic() < deadline - 5:
            a_final = run_lp_iterations(a_final, 20, min(deadline - 5, time.monotonic() + 15),
                                         tracker, tighten_base=0.001)
            a_final, _ = coordinate_descent_pass(a_final)
            tracker.update(a_final)
        
        tracker.report(f"Phase 7 Final ({time.monotonic()-t0:.1f}s)")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J = {tracker.best_score:.10f}, n = {tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"Target: 1.5027, Achieved: {tracker.best_score:.10f}", flush=True)
    
    return [float(x) for x in tracker.best]
