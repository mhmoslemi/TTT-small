"""

def fft_convolve(a, b):
    """Fast circular-avoiding full convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = fft_convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_value(a, beta):
    """Smooth approximation of the objective using log-sum-exp."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    log_sum_exp = beta * max_conv + np.log(np.sum(np.exp(shifted)))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * log_sum_exp / beta / (sum_a * sum_a)


def smooth_gradient(a, beta):
    """
    Compute gradient of smooth objective w.r.t. a.
    Returns (gradient, conv, weights).
    Gradient of 2n/(sum_a)^2 * (1/beta)*log(sum exp(beta*conv_k))

    shifted = beta * (conv - max_conv)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)

def multiplicative_descent(a, beta_start=5.0, beta_end=200.0,
                           step_start=0.1, max_iters=3000,
                           time_budget=30.0, line_search=True):
    """
    Multiplicative gradient descent with adaptive beta (log-sum-exp smoothing).
    Normalizes sum(a)=1 at each step. Uses backtracking line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 150

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        # Adaptive beta: start smooth, end sharp
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient
        grad, conv, w = smooth_gradient(a, beta)

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)

        # Adaptive step size
        step = step_start * (1.0 + 0.5 * (1.0 - progress))
        step = max(step * (0.95 ** (stall // 30)), 0.001)

        # Multiplicative update
        log_update = -step * grad_centered
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb to escape
                rng = np.random.default_rng(it * 7 + 3)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0
                patience = max(patience * 0.8, 40)
                step *= 0.7

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None:
                        g_val = np.maximum(g_var.value, 0)
                        return g_val, float(np.sum(g_val))
                except Exception:
                    continue
        except Exception:
            pass

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if HAS_CVXPY:

            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    if g_var.value is not None and M_var.value is not None:
                        g_val = np.maximum(g_var.value, 0)
                        s = np.sum(g_val)
                        if s < 1e-10:
                            return None, None
                        return g_val / s, float(M_var.value)
                except Exception:
                    continue
        except Exception:
            pass

    A = build_toeplitz(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=30, time_budget=40.0):
    """Alternating LP: fix a, solve for g that maximizes sum subject to conv bound."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        conv = fft_convolve(a, a)
        M = float(np.max(conv)) * 1.001  # slight relaxation for feasibility

        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break

        g_norm = g / sum_g

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t

        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_obj = best_obj_ls
            best_a = a_new.copy()
            a = a_new
        else:
            # Try min-peak LP
            g2, M2 = solve_lp_minpeak(a, time_limit=8)
            if g2 is not None:
                improved = False
                for t in [0.3, 0.5, 0.8]:
                    a_new = (1.0 - t) * a + t * g2
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new < 1e-10:
                        continue
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        break
                if not improved:
                    break

    return best_a


def coordinate_descent(a, max_sweeps=5, time_budget=30.0):
    """Coordinate-wise optimization with 1D line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        conv = fft_convolve(a, a)
        max_b = float(np.max(conv))
        sum_a = float(np.sum(a))

        for j in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Optimize a[j] in 1D
            # conv[k] = conv_old[k] + (new_val - old_val) * (a[k-j] + a[j-k])
            # where a[x] = 0 if x outside [0,n-1]
            old_val = a[j]
            # Precompute the "influence" vector
            infl = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                kj = k - j
                jk = j - k
                val = 0.0
                if 0 <= kj < n:
                    val += a[kj]
                if 0 <= jk < n:
                    val += a[jk]
                infl[k] = val

            # Try a few candidate values
            base_obj = 2.0 * n * max_b / (sum_a * sum_a)
            best_local = old_val
            best_local_obj = base_obj

            # Golden section search in [0, 3*old_val + 0.01]
            lo = 0.0
            hi = 3.0 * old_val + 0.05

            def f_1d(v):
                new_sum = sum_a - old_val + v
                if new_sum < 0.01:
                    return np.inf
                new_max = max_b
                if infl.any() != 0:
                    new_conv_diff = (v - old_val) * infl
                    new_conv = max_b + 0  # can't easily recompute full conv
                    # Use the fact that only elements where infl != 0 change
                    # Actually we need the full new conv to find new max
                    # For efficiency, check if the old max changes or a new max appears
                    # Approximate: new_max ≈ max(max_b + (v-old_val)*infl_at_old_max,
                    #                              max over k where infl[k] > 0 of (v-old_val)*infl[k])
                    # This is an upper bound; for the line search it's OK to be approximate
                    pass
                return 2.0 * n * new_max / (new_sum * new_sum)

            # Simpler: try a few values
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val,
                          old_val * 1.3, old_val * 1.5, old_val * 2.0,
                          old_val * 3.0]
            for v in candidates:
                if v < 0 or v > 1000:
                    continue
                a[j] = v
                new_sum = np.sum(a)
                if new_sum < 0.01:
                    a[j] = old_val
                    continue
                new_conv = fft_convolve(a, a)
                new_max = float(np.max(new_conv))
                obj = 2.0 * n * new_max / (new_sum * new_sum)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = v
            a[j] = best_local

        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def fine_tune(a, time_budget=20.0):
    """High-peta smooth gradient descent for final polish."""
    return multiplicative_descent(a, beta_start=50.0, beta_end=500.0,
                                  step_start=0.02, max_iters=5000,
                                  time_budget=time_budget)


def random_init(n, seed):
    """Generate diverse random initializations for non-negative sequences."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k / (n - 1)) if n > 1 else np.ones(1)
    elif choice == 1:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 2:
        # Two peaks
        c1 = n * rng.uniform(0.1, 0.4)
        c2 = n * rng.uniform(0.6, 0.9)
        w1 = n * rng.uniform(0.05, 0.2)
        w2 = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((k - c1) / w1) ** 2) + np.exp(-0.5 * ((k - c2) / w2) ** 2)
    elif choice == 3:
        # Sinusoidal amplitude
        f = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 4:
        # Multi-scale sinusoid
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.5 * (2 ** scale), 3.0 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Random smooth (interpolated)
        n_pts = max(3, n // 15)
        pts = rng.random(n_pts)
        a = np.interp(k, np.arange(n_pts), pts)
    elif choice == 6:
        # Triangle
        mid = n // 2
        a = 1.0 - np.abs(k - mid) / (mid + 1)
    elif choice == 7:
        # Banded (periodic large values)
        period = max(3, n // rng.integers(3, 8))
        a = np.zeros(n)
        for p in range(0, n, period):
            w = max(1, period // 4)
            a[max(0, p - w):min(n, p + w)] += 1.0
        a = a + 0.1
    elif choice == 8:
        # Chirp-like
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(4.0, 10.0)
        t = k / max(n - 1, 1)
        phase = 2 * np.pi * (f0 * t + (f1 - f0) * t * t / 2)
        a = 1.0 + 0.5 * np.cos(phase)
    elif choice == 9:
        # Product of two windows
        w1 = 0.5 + 0.5 * np.cos(2 * np.pi * k / n * rng.uniform(1, 3))
        w2 = np.exp(-0.5 * ((k - n * 0.5) / (n * 0.2)) ** 2)
        a = w1 * w2 + 0.2 * w1 + 0.2 * w2
    elif choice == 10:
        # Power law decay from center
        center = n * 0.5
        alpha = rng.uniform(1.0, 3.0)
        a = np.exp(-alpha * np.abs(k - center) / n)
    else:
        # Flat with random modulation
        a = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_load_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_pipeline(a, time_budget=60.0, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"    Init obj: {best_obj:.6f}")

    # Phase A: Smooth gradient (medium beta)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t = min(remaining * 0.3, 20.0)
        a_new = multiplicative_descent(best_a, beta_start=5.0, beta_end=80.0,
                                       step_start=0.08, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After SG1: {best_obj:.6f}")

    # Phase B: Smooth gradient (high beta, fine)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t = min(remaining * 0.25, 15.0)
        a_new = multiplicative_descent(best_a, beta_start=30.0, beta_end=300.0,
                                       step_start=0.03, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After SG2: {best_obj:.6f}")

    # Phase C: LP refinement
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t = min(remaining * 0.3, 30.0)
        a_new = alternating_lp(best_a, max_iters=20, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After LP: {best_obj:.6f}")

    # Phase D: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        t = min(remaining * 0.3, 20.0)
        a_new = coordinate_descent(best_a, max_sweeps=3, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After CD: {best_obj:.6f}")

    # Phase E: Final fine-tune
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t = min(remaining * 0.4, 15.0)
        a_new = fine_tune(best_a, time_budget=t)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    After FT: {best_obj:.6f}")

    return best_a, best_obj


def local_search(a, time_budget=40.0, verbose=False):
    """Perturbation-based local search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Local search start: {best_obj:.8f}")

    attempt = 0
    while time.monotonic() - start < time_budget - 5:
        attempt += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        rng = np.random.default_rng(attempt * 31337 + 42)
        scale = 0.03 + 0.04 * (attempt % 5) / 5.0

        for sub in range(2):
            if time.monotonic() - start > time_budget - 5:
                break

            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / np.sum(a_pert)

            sub_time = min(12.0, remaining * 0.35)
            a_opt, obj = optimize_pipeline(a_pert, time_budget=sub_time, verbose=False)

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_opt.copy()
                if verbose:
                    print(f"    LS attempt {attempt}.{sub}: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines adaptive smooth gradient descent, LP refinement,
    coordinate descent, and local search. Explores n in [250, 450] range.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330
        print("No parent available, starting fresh.")

    print(f"Initial best: {best_score:.6f} (n={n_parent})")
    print(f"Total budget: {budget_s}s")

    # === Phase 1: Quick optimization of parent (or n=330) ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        p1_budget = min(80.0, remaining * 0.12)
        print(f"\n=== Phase 1: Optimize parent (budget={p1_budget:.0f}s) ===")
        a_opt, obj = optimize_pipeline(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Improved: {best_score:.6f}")

    # === Phase 2: Explore nearby n values ===
    print(f"\n=== Phase 2: Explore n values ===")
    n_values = sorted(set([
        n_parent - 20, n_parent - 10, n_parent, n_parent + 10,
        n_parent + 20, n_parent + 30, n_parent + 40,
        max(200, n_parent - 50), min(500, n_parent + 60)
    ]))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_n = remaining / (len(n_values) + 1) * 1.3
        for n_val in n_values:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n, remaining * 0.25, 80.0)
            alloc = max(alloc, 15.0)

            print(f"  n={n_val} (alloc={alloc:.0f}s, remaining={remaining:.0f}s)")

            # Try 2-3 inits
            n_inits = 2 if alloc > 40 else 1
            for init_idx in range(n_inits):
                if time.monotonic() > deadline - 40:
                    break
                if init_idx == 0:
                    a_init = resample_sequence(parent, n_val) if parent is not None else random_init(n_val, seed + n_val)
                else:
                    a_init = random_init(n_val, seed + n_val * 1000 + init_idx * 7919)

                sub_budget = min(alloc * 0.6, 50.0)
                a_opt, obj = optimize_pipeline(a_init, time_budget=sub_budget,
                                               verbose=(n_val == n_parent and init_idx == 0))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

    # === Phase 3: Intensive optimization of best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            p3_budget = min(remaining * 0.3, 200.0)
            print(f"\n=== Phase 3: Intensive optimization (budget={p3_budget:.0f}s) ===")

            # Run full pipeline multiple times with different params
            for round_idx in range(3):
                if time.monotonic() > deadline - 30:
                    break
                sub_budget = min(p3_budget / (3 - round_idx), 80.0)

                # Different beta ranges for diversity
                betas = [(5, 50), (20, 200), (50, 500)]
                b_start, b_end = betas[round_idx % 3]
                a_new = multiplicative_descent(best_seq, beta_start=float(b_start),
                                               beta_end=float(b_end),
                                               step_start=0.05 * (0.5 ** round_idx),
                                               time_budget=sub_budget * 0.5)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Round {round_idx} SG: {best_score:.6f}")

                # LP
                if time.monotonic() > deadline - 20:
                    break
                a_new = alternating_lp(best_seq, max_iters=15,
                                       time_budget=sub_budget * 0.4)
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Round {round_idx} LP: {best_score:.6f}")

    # === Phase 4: Local search ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p4_budget = min(remaining * 0.35, 150.0)
            print(f"\n=== Phase 4: Local search (budget={p4_budget:.0f}s) ===")
            a_opt, obj = local_search(best_seq, time_budget=p4_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After local search: {best_score:.6f}")

    # === Phase 5: Final polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            p5_budget = min(remaining * 0.5, 60.0)
            print(f"\n=== Phase 5: Final polish (budget={p5_budget:.0f}s) ===")

            # Fine gradient
            a_new = fine_tune(best_seq, time_budget=p5_budget * 0.4)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Fine tune: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = coordinate_descent(best_seq, max_sweeps=3,
                                           time_budget=min(remaining * 0.5, 20.0))
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Coord descent: {best_score:.6f}")

            # One more LP
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = alternating_lp(best_seq, max_iters=10,
                                       time_budget=min(remaining * 0.6, 25.0))
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Final LP: {best_score:.6f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(330) / 330.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
