def spectral_flatten_init(n, rng, n_iters=20):
    """
    Initialize by iteratively flattening the spectrum.
    Start with random positive sequence, then:

    2. Replace magnitudes with their geometric mean (or some flat target)
    3. Keep phases, compute IFFT
    4. Clip to non-negative
    5. Repeat

    a = rng.uniform(0.1, 1.0, n)
    a /= a.sum()
    
    for _ in range(n_iters):

        mags = np.abs(A)

        # Target: geometric mean of magnitudes

        # Blend: partially flatten

    vals = np.empty(nnz)
    idx = 0
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            vals[idx] = a[k - j]
            idx += 1
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None

def multiplicative_descent(a, max_iters, deadline, tracker, eta_init=0.1, eta_decay=0.999):
    """
    Mirror descent with negative entropy (multiplicative updates).
    Update: a[j] *= exp(-eta * grad[j]) where grad[j] = d(max b)/da[j]
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    eta = eta_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        # Active set: indices where b is close to max
        threshold = b_max * (1 - 1e-6)
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient of max(b) w.r.t. a
        # d(b[k])/da[j] = a[k-j] if 0 <= k-j < n (for the term a[j]*a[k-j])
        #               + a[j-k] if 0 <= j-k < n (for the term a[j-k]*a[j])
        # But since b[k] = sum_j a[j]*a[k-j], we have d(b[k])/da[j] = a[k-j] (when valid)
        # Wait, b[k] = sum_j a[j] * a[k-j]. If we differentiate w.r.t. a[j]:
        #   d/da[j] of a[j]*a[k-j] = a[k-j] (holding other terms fixed)
        #   But a[k-j] also depends on a if j appears... no, a[k-j] is a different element.
        # Actually b[k] = sum_j a[j]*a[k-j]. The terms involving a[j] are:
        #   a[j]*a[k-j] (the j-th term in the sum)
        #   and a[k-j]*a[j] (the (k-j)-th term in the sum, if k-j != j and both valid)
        # So d(b[k])/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j (i.e., 2j != k)
        #                    = a[j]^2 if 2j == k (i.e., j = k/2)
        # Wait no. b[k] = sum_{j: valid} a[j] * a[k-j]. 
        # The term a[j] appears in:
        #   - The j-th term: a[j]*a[k-j]
        #   - The (k-j)-th term: a[k-j]*a[j] (same thing, just different index in sum)
        # So if j != k-j (i.e., 2j != k), then a[j] appears in two terms: a[j]*a[k-j] and a[k-j]*a[j]
        # But these are the SAME term in the sum (just viewed from different indices).
        # Let me re-examine: sum_j a[j]*a[k-j]. If I fix j, this is one term.
        # The (k-j)-th term is a[k-j]*a[j], which is the same value.
        # So in the sum, each unordered pair {j, k-j} appears once (if j ≠ k-j) or once (if j = k-j).
        # Wait no, the sum is over all valid j. If j is valid and k-j is also valid and j ≠ k-j,
        # then both a[j]*a[k-j] and a[k-j]*a[j] appear in the sum (as separate terms for j and k-j).
        # So d(b[k])/da[j] = a[k-j] (from the j-th term) + a[k-j] (from the (k-j)-th term) = 2*a[k-j]
        # UNLESS j = k-j, in which case it's just a[j] (from the single term a[j]*a[j]).
        
        # Actually, let me just do it numerically for correctness:
        # grad[j] = sum_{k in active} d(b[k])/da[j] / len(active)
        # d(b[k])/da[j] = a[k-j] if 0 <= k-j < n, plus a[j-(k)] if 0 <= j-k < n (but this is the same)
        
        # Let me be more careful:
        # b[k] = sum_{j: max(0,k-n+1) <= j <= min(k,n-1)} a[j] * a[k-j]
        # d(b[k])/da[j] = a[k-j] (from the j-th term in the sum)
        #                + a[j-k] (from the (j)-th... wait)
        # No. b[k] = sum_j a[j]*a[k-j]. 
        # d/da[j'] of b[k] = sum_j delta(j,j')*a[k-j] + sum_j a[j]*delta(k-j,j')
        #                   = a[k-j'] + a[j'-k]  (where the deltas select the right terms)
        # But wait, the second term: sum_j a[j]*delta(k-j, j') = a[k-j'] (if k-j' is in range)
        # Hmm, that gives the same thing. Let me re-derive:
        # b[k] = sum_j a[j]*a[k-j]
        # d(b[k])/da[m] = a[k-m] (from j=m term) + sum_j a[j] * d(a[k-j])/da[m]
        #                = a[k-m] + a[k-m] (from j=k-m term, if valid and k-m != m)
        #                = 2*a[k-m] if k-m != m and k-m is valid
        #                = a[k-m] if k-m == m (i.e., m = k/2)
        
        # OK so the gradient is:
        # grad[m] = (1/|S|) * sum_{k in S} [2*a[k-m] if 0<=k-m<n and k-m != m, 
        #                                   a[k-m] if 0<=k-m<n and k-m == m]
        
        grad = np.zeros(n)
        for k in active:
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                if j == k - j:
                    grad[j] += a[j]
                else:
                    grad[j] += a[k - j]
                    # The symmetric term (k-j) contributes a[j] to grad[k-j]
                    # We'll handle this in the k-j iteration... actually no.
                    # Let me just add a[k-j] to grad[j] for now.
                    # The full gradient should be:
                    # grad[m] = sum_{k in active, valid} (a[k-m] if m != k-m else a[m])
                    #         + sum_{k in active, valid} (a[m-(k)] ... no
                    # I'm overcomplicating this. Let me just use:
                    # (a*a)[k] = <a, shift_k(a)> where shift_k is the appropriate shift
                    # d/d a[m] of <a, shift_k(a)> = shift_k(a)[m] + a[m - shifted] 
                    # = a[k-m] + a[m-k] (with appropriate bounds)
                    # For our case: d(b[k])/da[m] = a[k-m] + a[m-k] where both terms are 0 if index out of [0,n)
                    pass
        
        # Let me redo this more carefully using a vectorized approach
        grad = np.zeros(n)
        for k in active:
            # b[k] = sum_{j=js}^{je} a[j]*a[k-j] where js=max(0,k-n+1), je=min(k,n-1)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            # d(b[k])/da[m] for m in [js, je]:
            #   = a[k-m] (from the m-th term)
            #   + (a[m-k] if m-k is in [js,je] range... no)
            # Actually, I'll use the fact that b is the linear convolution:
            # b[k] = sum_j a[j]*a[k-j]
            # If I perturb a[m] by epsilon, b[k] changes by:
            #   epsilon * a[k-m] (if 0<=k-m<n) [from the j=m term]
            #   + epsilon * a[m-k] (if 0<=m-k<n) [from the j=k-m term, if k-m != m]
            # So d(b[k])/da[m] = a[k-m]*I(0<=k-m<n) + a[m-k]*I(0<=m-k<n)
            # But a[k-m] and a[m-k] are the same element of a (just accessed differently)!
            # a[k-m] is the element at index k-m of the array a.
            # a[m-k] is the element at index m-k of the array a.
            # These are different elements unless k-m = m-k, i.e., k=2m.
            
            # For m in [js, je]:
            #   k-m is in [k-je, k-js] = [k-min(k,n-1), k-max(0,k-n+1)]
            #   If k <= n-1: k-m in [k-(n-1), k] → but k-m >= 0 since m <= k (from je=min(k,n-1))
            #                and k-m <= k <= n-1. So k-m is always valid when m <= k.
            #   If k >= n: k-m in [k-min(k,n-1), ...] = [k-(n-1), ...]
            #                k-m >= k-(n-1) >= 1 (since k>=n). And k-m <= k-js = k-max(0,k-n+1) = n-1.
            #                So k-m is in [k-n+1, n-1], which is valid.
            
            # So for all m in [js, je], k-m is valid (in [0, n-1]).
            # Now, is m-k valid? m-k ranges from js-k to je-k.
            #   If k <= n-1: js = 0, so m-k ranges from -k to min(k,n-1)-k <= 0.
            #     m-k >= 0 only if m >= k, i.e., m = k (the only case since m <= min(k,n-1) = k).
            #     So for k <= n-1, m-k >= 0 only when m = k, giving a[0].
            #   If k >= n: je = n-1, so m-k ranges from max(0,k-n+1)-k to (n-1)-k < 0.
            #     m-k >= 0 only if m >= k, but m <= n-1 < k. So never valid.
            #     Wait, for k >= n: js = k-n+1, je = n-1.
            #     m-k ranges from (k-n+1)-k = 1-n to (n-1)-k < 0.
            #     So m-k < 0 always. Not valid.
            
            # Therefore:
            # For k <= n-1: d(b[k])/da[m] = a[k-m] + a[m-k]*I(m>=k) = a[k-m] + a[m-k]*I(m=k)
            #   (since m <= k always when k <= n-1)
            #   = a[k-m] for m < k, and a[0] + a[0] = 2*a[0] for m = k
            #   Wait, that's not right either. Let me re-examine.
            #   For k <= n-1, m ranges from 0 to k.
            #   d(b[k])/da[m] = a[k-m] (always, since k-m >= 0 and <= k <= n-1)
            #                 + a[m-k] (only if m-k >= 0, i.e., m >= k, i.e., m = k)
            #   So for m < k: grad contribution = a[k-m]
            #      for m = k: grad contribution = a[0] + a[0] = 2*a[0]
            #   But a[k-m] when m=k is a[0]. So the total is a[0] + a[0] = 2*a[0]. ✓
            
            # For k >= n: d(b[k])/da[m] = a[k-m] (always valid as shown)
            #                 + a[m-k] (never valid since m < n <= k)
            #   So for all m in [k-n+1, n-1]: grad contribution = a[k-m]
            
            # Simplified:
            # d(b[k])/da[m] = a[k-m] + a[m-k]*I(0<=m-k<n)
            # For our valid range: = a[k-m] + (a[m-k] if 0<=m-k else 0)
            # Since m >= js >= 0 and m <= je <= n-1:
            #   m-k >= 0 iff m >= k (only possible when k <= n-1 and m = k)
            
            for m in range(js, je + 1):
                contrib = a[k - m]
                if 0 <= m - k < n:
                    contrib += a[m - k]
                grad[m] += contrib
        
        grad /= len(active)
        
        # Multiplicative update: a[j] *= exp(-eta * grad[j])
        # But we want to minimize, and grad points in the direction of increasing max(b)
        # So we decrease a where grad is large
        # Also need to maintain sum(a) = 1
        # Use: a_new[j] = a[j] * exp(-eta * (grad[j] - mean(grad))) / normalization
        # This preserves the sum approximately
        
        grad_centered = grad - grad.mean()
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.8
            if eta < 1e-8:
                break
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def autocorrelation_equalize(a, max_iters, deadline, tracker):
    """
    Directly try to equalize the autocorrelation by adjusting a.
    At each step, find the lag with maximum b and the lag with minimum b,
    and adjust a to reduce the former and increase the latter.
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    eta = 0.05
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        b_min = float(b.min())
        
        # Find the lag with max autocorrelation
        k_max = int(np.argmax(b))
        # Find a lag with low autocorrelation
        k_min = int(np.argmin(b))
        
        # To reduce b[k_max]: reduce contributions to it
        # b[k_max] = sum_j a[j]*a[k_max-j]
        # To increase b[k_min]: increase contributions to it
        
        # Simple approach: reduce a[j] where it contributes most to b[k_max]
        # and increase a[j] where it contributes to b[k_min]
        
        js_max = max(0, k_max - n + 1)
        je_max = min(k_max, n - 1)
        contrib_max = np.zeros(n)
        for j in range(js_max, je_max + 1):
            contrib_max[j] = a[k_max - j]
            if 0 <= j - k_max < n and j != k_max - j:
                contrib_max[j] += a[j - k_max]
        
        js_min = max(0, k_min - n + 1)
        je_min = min(k_min, n - 1)
        contrib_min = np.zeros(n)
        for j in range(js_min, je_min + 1):
            contrib_min[j] = a[k_min - j]
            if 0 <= j - k_min < n and j != k_min - j:
                contrib_min[j] += a[j - k_min]
        
        # Update: reduce where contrib_max is high, increase where contrib_min is high
        update = eta * (contrib_min / max(b_min, 1e-12) - contrib_max / b_max)
        update -= update.mean()  # maintain sum
        
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            eta = min(eta * 1.01, 0.5)
        else:
            eta *= 0.7
            if eta < 1e-10:
                break
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    """Log-sum-exp smoothing for differentiable optimization."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = 30.0
    lr = 0.5 / n
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        # LSE approximation of max
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a: 
        # LSE = b_max + log(sum(exp(tau*(b-b_max))))/tau
        # d(LSE)/da[j] = sum_k p[k] * d(b[k])/da[j]
        # d(b[k])/da[j] = a[k-j] + a[j-k] (with bounds)
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                contrib = a[k - j]
                if 0 <= j - k < n and j != k - j:
                    contrib += a[j - k]
                grad[j] += p[k] * contrib
        
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.005)
            lr = 0.5 / n
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    """LP best-response with fixed blending."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        
        g, r = lp_best_response(a, M_tight, time_limit=8)
        if g is None:
            g, r = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.03, 0.4 * (1.0 - progress))
        
        # Try multiple blend factors
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.9, 0.3, 0.1, 0.7]:
            if t > 1.0 or t <= 0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
        
        if best_a is None:
            # Try a smaller tighten
            if tighten > 0.001:
                tighten = tighten * 0.5
            continue
        
        a = best_a
        current_J = best_J
        tracker.update(a)
    
    return a

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        # Resample parent
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        p_resampled = np.maximum(np.interp(t_new, t_old, parent), 0.001)
        inits.append(p_resampled)
        # Reversed
        p_rev = parent[::-1]
        p_rev_r = np.maximum(np.interp(t_new, t_old, p_rev), 0.001)
        inits.append(p_rev_r)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Spectral flattening (multiple variants)
    for _ in range(3):
        inits.append(spectral_flatten_init(n, rng, n_iters=30))
    
    # Cosine modulated
    for freq in [1, 2, 3, 5, 8, 13]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi)))
    
    # Multi-frequency
    inits.append(1.0 + 0.3*np.sin(2*np.pi*t) + 0.2*np.sin(4*np.pi*t) + 0.1*np.sin(7*np.pi*t))
    inits.append(1.0 + 0.25*np.cos(3.7*2*np.pi*t) + 0.15*np.cos(7.3*2*np.pi*t))
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        width = rng.uniform(0.1, 0.3)
        inits.append(0.1 + np.exp(-((t-center)**2)/(2*width**2)))
    # Double bump
    inits.append(0.1 + np.exp(-((t-0.25)**2)/0.02) + np.exp(-((t-0.75)**2)/0.02))
    
    # Power law
    inits.append((t + 0.01)**rng.uniform(-0.5, 2.0))
    inits.append((1-t+0.01)**rng.uniform(-0.5, 2.0))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.1)
    
    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.exp(-np.arange(-kernel_size//2, kernel_size//2+1)**2 / (kernel_size//3)**2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Sine wave with random phase (ensures non-negative with offset)
    for _ in range(3):
        f = rng.uniform(1, 10)
        ph = rng.uniform(0, 2*np.pi)
        inits.append(0.5 + 0.5*np.sin(2*np.pi*f*t + ph))
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        s = x.sum()
        if s > 1e-10:
            x /= s
            result.append(x)
    return result

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def coordinate_descent_polish(a, max_steps, deadline, tracker):
    """Random coordinate descent for final polish."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        j = rng.integers(0, n)
        improved = False
        for scale in [0.5, 1.5, 0.7, 1.3, 0.3, 2.0, 0.85, 1.15]:
            a_trial = a.copy()
            a_trial[j] *= scale
            a_trial = np.maximum(a_trial, 1e-12)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        
        if step % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a
