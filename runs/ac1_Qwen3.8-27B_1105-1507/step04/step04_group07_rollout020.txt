"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening and golden-section line search
2. Multi-length exploration (512, 768, 1024, 1536)
3. Diverse initializations including spectral-shaping
4. Subgradient polishing
5. Coordinate descent final polish
6. Simulated annealing escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, max_time=5.0):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': max_time})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def golden_section_search(f, a, b, tol=1e-6, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01, use_golden=True):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight, max_time=3.0)
        if g is None:
            g = lp_best_response(a, M, max_time=3.0)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        if use_golden:
            # Golden section search for t
            def obj(t):
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    return float('inf')
                a_trial /= ss
                return compute_J(a_trial)
            t_best = golden_section_search(obj, 0.01, 0.95, tol=0.01, max_iter=15)
            a_new = (1.0 - t_best) * a + t_best * g
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
        else:
            # Fixed step sizes
            best_a = None
            best_J = current_J
            for t in [0.3, 0.5, 0.7, 0.9]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial
            if best_a is None:
                break
            a_new = best_a
            J_new = best_J
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 20 == 0:
                tracker.update(a)
        else:
            # Try smaller tightening
            if tighten > 0.001:
                tighten = max(0.001, tighten * 0.5)
            else:
                break
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.1 / n
    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_J = current_J
    stuck_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        active = np.where(b >= M - 1e-12)[0]
        n_active = len(active)
        
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= n_active
        
        # Projected subgradient: move opposite to grad, project onto simplex
        a_new = a - lr * grad
        # Project onto {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a
        else:
            a_new /= ss
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.05, 1.0 / n)
            stuck_count = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            lr *= 0.8
            stuck_count += 1
            if stuck_count > 200:
                break
    tracker.update(best_a)
    return best_a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Cyclic order with random restarts
        order = np.arange(n) if step % 10 == 0 else rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            for scale in [0.7, 1.3, 0.5, 1.5, 0.9, 1.1]:

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    n = len(a)
    rows, cols, _ = get_sparsity(n)

                    g /= gs
                    for t in [0.1, 0.3, 0.5]:
                        a_trial = (1.0 - t) * a + t * g
                        a_trial = np.maximum(a_trial, 0.0)
                        ss = a_trial.sum()
                        if ss < 1e-10:
                            continue
                        a_trial /= ss
                        J_trial = compute_J(a_trial)
                        if J_trial < current_J - 1e-15:
                            a = a_trial
                            current_J = J_trial
                            if J_trial < best_J:
                                best_J = J_trial
                                best_a = a.copy()
                                tracker.update(a)
                            break
                break

    tracker.update(best_a)
    return best_a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    lr = 0.3 / n
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        threshold = M - 1e-10
        active = np.where(b >= threshold)[0]
        n_active = len(active)
        if n_active == 0:
            break

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= n_active

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a
        else:
            a_new /= ss

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.03, 2.0 / n)
            stuck = 0
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            lr *= 0.75
            stuck += 1
            if stuck > 300:
                break

    tracker.update(best_a)
    return best_a

def coordinate_descent(a, max_steps, deadline, tracker):
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a
    best_J = current_J
    tracker.update(a)
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n) if step % 5 == 0 else rng.permutation(n)
        any_improve = False
        for j in order:
            if time.monotonic() > deadline:
                break
            best_scale = 1.0
            best_sJ = current_J
            for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
                a_trial = a.copy()
                a_trial[j] *= scale
                a_trial = np.maximum(a_trial, 1e-10)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                J_trial = compute_J(a_trial)
                if J_trial < best_sJ:
                    best_sJ = J_trial
                    best_scale = scale
            if best_scale != 1.0:
                a = a.copy()
                a[j] *= best_scale
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                current_J = best_sJ
                any_improve = True
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
        if not any_improve:
            break
        if step % 200 == 0:
            tracker.update(best_a)

    tracker.update(best_a)
    return best_a

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n)
        p = np.maximum(p, 0.001)
        p /= p.sum()
        inits.append(p)
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.1, n)
        p2 = np.maximum(p2, 0.001)
        p2 /= p2.sum()
        inits.append(p2)

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))

    # Chirp
    inits.append(1.0 + 0.3 * np.sin(np.pi * t ** 2))
    # Power law
    inits.append((t + 0.05) ** rng.uniform(-0.5, 0.5))
    # Gaussian mix
    inits.append(0.5 * np.exp(-4 * (t - 0.3) ** 2) + 0.5 * np.exp(-4 * (t - 0.7) ** 2) + 0.05)

    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0.001)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths to try
    if parent is not None:
        par_n = parent.size
        cand_lengths = [par_n, par_n * 2]
        if par_n > 256:
            cand_lengths.append(par_n // 2)
        # Filter valid lengths
        lengths = []
        for n in cand_lengths:
            if 256 <= n <= 2048 and n % 64 == 0:
                if n not in lengths:
                    lengths.append(n)
        if not lengths:
            lengths = [512]
        lengths.sort()
    else:
        lengths = [512, 1024]

    # Allocate time: give more to longer sequences, but cap
    reserve = 350  # reserve for polishing phases
    total_time = max(100, deadline - time.monotonic() - reserve)
    # Weight by sqrt(n) to favor longer
    w = np.array([np.sqrt(n) for n in lengths], dtype=np.float64)
    w /= w.sum()
    time_alloc = w * total_time

    print(f"Lengths: {lengths}, Time alloc: {[f'{t:.0f}s' for t in time_alloc]}", flush=True)

    # === Phase 1: Multi-length LP ===
    for idx, n in enumerate(lengths):
        t_start = time.monotonic()
        t_end = min(t_start + time_alloc[idx], deadline - reserve + 50)
        if time.monotonic() >= t_end - 30:
            continue

        print(f"\n{'='*50} Phase1 n={n} (budget {time_alloc[idx]:.0f}s) {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')
        n_inits = len(inits)

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > t_end - 20:
                break
            a0 = np.maximum(a0, 1e-8).astype(np.float64)
            a0 /= a0.sum()

            # LP phase
            t_avail = t_end - time.monotonic() - 30 - (n_inits - 1 - init_idx) * 10
            t_avail = max(10, t_avail)
            n_lp_iters = max(5, min(150, int(t_avail / max(0.3, n / 800.0))))

            t_lp_end = min(time.monotonic() + t_avail * 0.55, t_end - 40)
            a = run_lp_opt(a0, n_lp_iters, t_lp_end, tracker, tighten=0.02, use_golden=True)

            # Subgradient polish
            if time.monotonic() < t_end - 20:
                a = subgradient_descent(a, max(200, int(30000 / n)), t_end - 15, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
                print(f"    init {init_idx}: J={J:.8f}", flush=True)
            else:
                print(f"    init {init_idx}: J={J:.8f} (no improve)", flush=True)

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None:
                parent = best_for_n
            else:
                par_resampled = resample_seq(parent, best_for_n.size)
                if best_J_for_n < compute_J(par_resampled):
                    parent = best_for_n
            print(f"  Phase1 n={n} done: best J={best_J_for_n:.8f} ({time.monotonic()-t_start:.1f}s)", flush=True)

    # === Phase 2: Intensive LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2: Intensive LP n={n} {'='*50}", flush=True)
        a = tracker.best.copy().astype(np.float64)
        a /= a.sum()

        # Progressive tightening
        tighten_sched = [0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
        for tight in tighten_sched:
            if time.monotonic() > deadline - 50:
                break
            a = run_lp_opt(a, 40, deadline - 50, tracker, tighten=tight, use_golden=True)
            print(f"    tighten={tight}: J={compute_J(a):.8f}", flush=True)

        # Subgradient
        if time.monotonic() < deadline - 30:
            a = subgradient_descent(a, 5000, deadline - 25, tracker)

        print(f"  Phase2 done ({time.monotonic()-t0:.1f}s): J={compute_J(a):.8f}", flush=True)

    # === Phase 3: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3: Coordinate Descent n={tracker.best.size} {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, 3000, deadline - 15, tracker)

        # Final LP
        if time.monotonic() < deadline - 8:
            a = run_lp_opt(a, 20, deadline - 5, tracker, tighten=0.005, use_golden=False)

        print(f"  Phase3 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a = subgradient_descent(tracker.best, 2000, deadline - 2, tracker)
        a = run_lp_opt(a, 10, deadline - 1, tracker, tighten=0.003, use_golden=False)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    return [float(x) for x in tracker.best]
