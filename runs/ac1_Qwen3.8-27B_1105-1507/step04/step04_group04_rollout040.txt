```python
def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def softmax_grad_descent(a, time_budget=30.0, verbose=False):
    """
    Gradient descent on the smoothed objective using softmax approximation of max.
    Uses multiplicative updates (exponentiation gradient) with momentum.
    Anneals the temperature tau.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 20.0
    lr = 0.1
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = 50

    max_iters = int(time_budget * 100)
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute smoothed gradient
        conv = compute_conv(a)
        max_conv = np.max(conv)
        # Softmax weights
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            break
        p = p / p_sum

        # Gradient: d/da[i] of smoothed max
        # grad[i] = sum_{m=0}^{n-1} p[i+m] * a[m] * 2
        # (factor of 2 from the two terms in the convolution derivative)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Project gradient onto tangent space (zero-sum direction)
        # For multiplicative updates, we center the gradient
        grad_centered = grad - np.mean(grad)

        # Momentum
        momentum = beta * momentum + (1 - beta) * grad_centered

        # Multiplicative update
        update = -lr * momentum
        update = np.clip(update, -10, 10)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if stall > patience:
            lr *= 0.7
            if lr < 0.001:
                break
            # Perturb to escape
            rng = np.random.default_rng(it * 7919 + 13)
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            a = a / np.sum(a)
            stall = 0
            patience = max(patience // 2, 20)

        # Anneal tau (increase it to sharpen the max approximation)
        tau *= 1.005
        tau = min(tau, 200.0)

        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    SGD iter {it}, obj={obj:.8f}, best={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def exact_coordinate_descent(a, max_sweeps=20, time_budget=60.0, verbose=False):
    """
    Exact coordinate descent: for each coordinate i, find the optimal value
    of a[i] given all others fixed, using the 1D structure of the problem.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Save current value and compute objective contributions
            t_old = a[i]
            S = 1.0 - t_old  # sum of all other coordinates (since sum=1)

            if S < 1e-15:
                continue

            # Compute C_k = conv[k] without the contribution from a[i]
            conv = compute_conv(a)
            # Contribution of a[i] to conv[k]:
            # If 0 <= k-i < n: 2*a[i]*a[k-i] (or a[i]^2 if k=2i)
            C = conv.copy()
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                k_minus_i = k - i
                if k_minus_i == i:  # k = 2i
                    C[k] = conv[k] - t_old * t_old
                else:
                    C[k] = conv[k] - 2.0 * t_old * a[k_minus_i]

            # Now find optimal t >= 0
            # Candidate t values:
            # 1. t = 0
            # 2. For each k where linear coefficient > 0: t = S - C[k] / (2*a[k-i])
            # 3. For k = 2i: t = sqrt(C[2i]) if C[2i] > 0... 
            #    Actually t = C[2i] / S (from the quadratic case derivation)
            #    Wait, let me re-derive.
            #    f(t) = (C[2i] + t^2) / (S+t)^2
            #    f'(t) = [2t(S+t) - 2(C[2i]+t^2)(S+t)/(S+t)^2 ... ]
            #    Let me redo: f(t) = (C[2i] + t^2) / (S+t)^2
            #    f'(t) = [2t(S+t)^2 - (C[2i]+t^2)*2(S+t)] / (S+t)^4
            #          = [2t(S+t) - 2(C[2i]+t^2)] / (S+t)^3
            #          = [2tS + 2t^2 - 2C[2i] - 2t^2] / (S+t)^3
            #          = 2(tS - C[2i]) / (S+t)^3
            #    Setting to zero: t = C[2i] / S
            #    
            #    For linear: f(t) = (C[k] + 2*a[k-i]*t) / (S+t)^2
            #    f'(t) = [2*a[k-i]*(S+t)^2 - (C[k]+2*a[k-i]*t)*2(S+t)] / (S+t)^4
            #          = [2*a[k-i]*(S+t) - 2*(C[k]+2*a[k-i]*t)] / (S+t)^3
            #          = [2*a[k-i]*S + 2*a[k-i]*t - 2*C[k] - 4*a[k-i]*t] / (S+t)^3
            #          = [2*a[k-i]*S - 2*C[k] - 2*a[k-i]*t] / (S+t)^3
            #    Setting to zero: t = (a[k-i]*S - C[k]) / a[k-i] = S - C[k]/a[k-i]
            #    Wait, that's: 2*a[k-i]*S - 2*C[k] - 2*a[k-i]*t = 0
            #    => a[k-i]*S - C[k] - a[k-i]*t = 0
            #    => t = S - C[k]/a[k-i]

            candidates = [0.0]

            # Linear candidates
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                k_minus_i = k - i
                if k_minus_i == i:
                    continue  # quadratic case handled separately
                coeff = 2.0 * a[k_minus_i]
                if coeff < 1e-15:
                    continue
                # t = S - C[k] / (2*a[k-i]) ... wait
                # From above: t = S - C[k] / a[k-i]
                # But the linear term is 2*a[k-i]*t, so:
                # f(t) = (C[k] + 2*a[k-i]*t) / (S+t)^2
                # The derivative zero gives: t = S - C[k] / (2*a[k-i]) ... 
                # Let me recheck:
                # 2*a[k-i]*S - 2*C[k] - 2*a[k-i]*t = 0
                # a[k-i]*S - C[k] - a[k-i]*t = 0
                # a[k-i]*(S - t) = C[k]
                # t = S - C[k] / a[k-i]
                t_cand = S - C[k] / a[k_minus_i]
                if t_cand > 1e-10:
                    candidates.append(t_cand)

            # Quadratic candidate (k = 2i)
            if 2 * i < 2 * n - 1:
                t_quad = C[2 * i] / S
                if t_quad > 1e-10:
                    candidates.append(t_quad)

            # Also add the current value and some nearby values
            candidates.append(t_old)
            candidates.append(t_old * 0.5)
            candidates.append(t_old * 1.5)

            # Remove duplicates and sort
            candidates = sorted(set(max(0.0, c) for c in candidates))

            # Evaluate objective at each candidate
            best_t = t_old
            best_f = 2.0 * n * np.max(conv)  # current objective (sum=1)

            for t_cand in candidates:
                if t_cand < 0:
                    continue
                # Compute conv at this t
                # conv_new[k] = C[k] + linear/quadratic terms
                # For efficiency, just compute the max
                max_val = 0.0
                for k in range(2 * n - 1):
                    if k >= i and k < i + n:
                        k_minus_i = k - i
                        if k_minus_i == i:
                            val = C[k] + t_cand * t_cand
                        elif k_minus_i < n:
                            val = C[k] + 2.0 * t_cand * a[k_minus_i]
                        else:
                            val = C[k]
                    else:
                        val = C[k]
                    if val > max_val:
                        max_val = val
                f_cand = 2.0 * n * max_val  # since sum=1, (S+t)=1
                if f_cand < best_f:
                    best_f = f_cand
                    best_t = t_cand

            if best_t != t_old:
                a[i] = best_t
                improved = True

        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        elif not improved and sweep > 2:
            break

        if verbose and sweep % 2 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={obj:.8f}, best={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a, best_obj
```

The quotient of a convex function and a convex increasing function is not necessarily convex, but in practice it should be nearly unimodal for small changes.

```python
def optimize_single_coord(a, i, conv):
    """Find optimal value of a[i] given all others fixed. Returns new a[i]."""
    n = len(a)

    if S < 1e-15:
        return t
    
    # Compute C[k] = conv[k] without a[i]'s contribution
    C = conv.copy()
    L = np.zeros(2*n-1)  # linear coefficient
    Q = np.zeros(2*n-1)  # quadratic coefficient

            Q[k] = 1.0

            L[k] = 2.0*a[kmi]
    
    # Generate candidates
    candidates = [0.0, t]

        if kmi == i:
            if C[k] > 0:

    candidates.append(t * 0.5)
    candidates.append(t * 1.5)

    candidates = np.array([max(0.0, c) for c in candidates if np.isfinite(c) and c <= 1.0])
    if len(candidates) == 0:
        return t
    
    # Vectorized evaluation
    # conv_at_c[k] = C[k] + L[k]*t + Q[k]*t^2 for each candidate t
    cands = candidates[None, :]  # shape (1, m)
    conv_at_c = C[:, None] + L[:, None] * cands + Q[:, None] * cands**2  # shape (2n-1, m)
    max_vals = np.max(conv_at_c, axis=0)  # shape (m,)

    objectives = 2.0 * n * max_vals / denoms
    
    best_idx = np.argmin(objectives)
    best_t = candidates[best_idx]
    
    return max(0.0, best_t)
```

```python
"""
Step function optimization using:
1. Smoothed gradient descent with tau annealing and momentum
2. Exact vectorized coordinate descent (O(n^2) per coordinate via vectorized candidate evaluation)
3. LP-based alternating optimization
4. Simulated annealing for global exploration
5. Multi-phase strategy with careful time budgeting
Key novelty: fully vectorized exact 1D coordinate optimization using the piecewise-linear
structure of the max-convolution objective.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=5):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def smoothed_grad_descent(a, time_budget=60.0, verbose=False, tau_init=15.0, lr_init=0.08):
    """

    a = a / s

    lr = lr_init
    momentum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)

        if p_sum < 1e-300:
            break
        p = p / p_sum

        grad = 2.0 * np.correlate(p, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum + (1.0 - beta) * grad_centered
        update = -lr * momentum
        update = np.clip(update, -10, 10)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 1e-15)

def exact_coord_descent(a, max_sweeps=200, time_budget=120.0, verbose=False):
    """
    Exact vectorized coordinate descent. For each coordinate i, generates candidate
    optimal t values from the 1D structure and evaluates them all at once using
    vectorized numpy operations.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s_sum = np.sum(a)
    if s_sum < 1e-10:
        return a
    a = a / s_sum

    best_a = a.copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * np.max(conv)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        sweep_improved = False
        order = np.random.default_rng(sweep * 31 + 7 + n).permutation(n)

        for idx in range(n):
            if time.monotonic() - start > time_budget - 0.05:
                break
            i = int(order[idx])

            t = a[i]
            S = 1.0 - t  # sum of others (since sum=1)

            if S < 1e-12:
                continue

            # Compute C, L, Q arrays
            C = conv.copy()
            L = np.zeros(2 * n - 1)
            Q = np.zeros(2 * n - 1)

            k_lo = max(0, i)
            k_hi = min(2 * n - 1, i + n)
            for k in range(k_lo, k_hi):
                kmi = k - i
                if kmi == i:
                    C[k] = conv[k] - t * t
                    Q[k] = 1.0
                else:
                    C[k] = conv[k] - 2.0 * t * a[kmi]
                    L[k] = 2.0 * a[kmi]

                    if C[k] > 1e-12:
                        cands_list.append(C[k] / S)

                        cands_list.append(S - C[k] / a[kmi])

            cands_list.append(t * 0.5)
            cands_list.append(t * 1.5)
            cands_list.append(S / n)

                    candidates.append(c)

                for k in range(k_lo, k_hi):
                    kmi = k - i

                    else:
                        conv[k] += 2.0 * delta * a[kmi]

                sweep_improved = True

        if s_new > 1e-12:
            scale = 1.0 / s_new
            a = a * scale
            conv = conv * (scale * scale)

        obj = 2.0 * n * np.max(conv)

def alternating_lp(a, max_iters=40, time_budget=60.0, verbose=False):

            a = 0.9 * a + 0.1 * g_norm

            s = np.sum(a)

    return best_a, best_obj

def simulated_annealing(a, time_budget=120.0, verbose=False, n_restarts=5):
    """
    Simulated annealing with multiplicative perturbations.
    Multiple restarts from the current best with decreasing temperature.
    """
    n = len(a)
    a_best = a.copy().astype(np.float64)
    s = np.sum(a_best)
    if s < 1e-10:
        return a_best
    a_best = a_best / s
    best_obj = compute_obj(a_best)
    start = time.monotonic()

    for restart in range(n_restarts):
        if time.monotonic() - start > time_budget - 10:
            break

        a_curr = a_best.copy()
        T = 0.05 * (1.0 + 0.3 * np.exp(-restart * 0.5))
        T_min = 0.005
        cooling = 0.995
        restart_budget = (time_budget - 10) / (n_restarts - restart) if n_restarts > restart else 1e10
        restart_start = time.monotonic()

        stall = 0
        for it in range(50000):
            if time.monotonic() - restart_start > restart_budget:
                break
            if time.monotonic() - start > time_budget - 5:
                break

            rng = np.random.default_rng(hash((it, restart, n)) & 0xFFFFFFFF)
            # Multiplicative perturbation
            perturbation = rng.normal(0, T, n)
            a_prop = a_curr * np.exp(perturbation)
            a_prop = np.maximum(a_prop, 1e-12)
            s_prop = np.sum(a_prop)
            if s_prop < 1e-10:
                continue
            a_prop = a_prop / s_prop

            obj_prop = compute_obj(a_prop)
            delta = obj_prop - compute_obj(a_curr)

            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a_curr = a_prop
                if obj_prop < best_obj - 1e-14:
                    best_obj = obj_prop
                    a_best = a_curr.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            if stall > 500:
                T = max(T * 0.8, T_min)
                stall = 0
                if T <= T_min * 1.01:
                    break

            T = max(T * cooling, T_min)

    return a_best, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))
            amp = 0.3 / (scale + 1)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.5)
        a = 1.0 + 0.7 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 6:
        a = rng.random(max(5, n // 10))
        a = np.interp(k, np.arange(len(a)), a)
    else:
        a = np.ones(n)
        for _ in range(rng.integers(1, 3)):
            f = rng.uniform(0.5, 3.0)
            a += rng.uniform(0.2, 0.5) * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def full_pipeline(a, time_budget=100.0, verbose=False):
    """Run the full optimization pipeline on a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"  Start: obj={best_obj:.8f}")

    # Phase 1: Smoothed gradient descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        sg_time = min(remaining * 0.2, 25.0)
        a_new, obj = smoothed_grad_descent(best_a, time_budget=sg_time,
                                            verbose=verbose, tau_init=20.0, lr_init=0.06)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After SGD: {best_obj:.8f}")

    # Phase 2: Exact coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        cd_time = min(remaining * 0.3, 40.0)
        a_new, obj = exact_coord_descent(best_a, max_sweeps=500, time_budget=cd_time,
                                          verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD: {best_obj:.8f}")

    # Phase 3: LP optimization
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(remaining * 0.25, 30.0)
        a_new, obj = alternating_lp(best_a, max_iters=30, time_budget=lp_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP: {best_obj:.8f}")

    # Phase 4: More coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        cd_time = min(remaining * 0.3, 30.0)
        a_new, obj = exact_coord_descent(best_a, max_sweeps=1000, time_budget=cd_time,
                                          verbose=False)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After CD2: {best_obj:.8f}")

    # Phase 5: Min-peak LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        lp_time = min(remaining * 0.3, 25.0)
        a_new, obj = alternating_lp_minpeak(best_a, max_iters=25, time_budget=lp_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"  After LP-MP: {best_obj:.8f}")

    # Phase 6: Final CD polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.5, 20.0)
        a_new, obj = exact_coord_descent(best_a, max_sweeps=2000, time_budget=cd_time,
                                          verbose=False)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining smoothed gradient descent,
    exact coordinate descent, LP-based methods, and simulated annealing.
    Focuses on n around the parent length with exploration of nearby values.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 330
        best_seq = np.ones(n_parent) / n_parent
        best_score = compute_obj(best_seq)

    print(f"Initial best: {best_score:.8f} (n={n_parent})")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Intensive optimization of parent ===
    print("\n=== Phase 1: Intensive optimization of parent ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    a_opt, obj = full_pipeline(best_seq, time_budget=p1_budget, verbose=True)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  After Phase 1: {best_score:.8f}")

    # === Phase 2: Simulated annealing on best ===
    print("\n=== Phase 2: Simulated annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sa_budget = min(remaining * 0.2, 200.0)
        a_opt, obj = simulated_annealing(best_seq, time_budget=sa_budget, verbose=True, n_restarts=4)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After SA: {best_score:.8f}")
        # Polish after SA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            p2b_budget = min(remaining * 0.15, 100.0)
            a_opt, obj = full_pipeline(best_seq, time_budget=p2b_budget, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After SA polish: {best_score:.8f}")

    # === Phase 3: Try nearby n values ===
    print("\n=== Phase 3: Nearby n values ===")
    n_candidates = []
    for delta in range(-20, 21, 5):
        nv = n_parent + delta
        if 150 <= nv <= 800 and nv != n_parent:
            n_candidates.append(nv)
    # Add some specific values
    for nv in [n_parent - 10, n_parent + 10, n_parent - 15, n_parent + 15]:
        if 150 <= nv <= 800 and nv != n_parent and nv not in n_candidates:
            n_candidates.append(nv)
    n_candidates = sorted(set(n_candidates))[:12]

    remaining = deadline - time.monotonic()
    if remaining > 30 and n_candidates:
        per_n = max(15.0, (remaining * 0.25) / max(len(n_candidates), 1))
        for n_val in n_candidates:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n, remaining * 0.2, 60.0)
            alloc = max(alloc, 15.0)
            print(f"  n={n_val} (alloc={alloc:.0f}s)")
            a_init = resample_sequence(best_seq, n_val)
            a_opt, obj = full_pipeline(a_init, time_budget=alloc, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.8f} **")

    # === Phase 4: Second round of SA + polish on best ===
    print("\n=== Phase 4: Second round SA + polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sa2_budget = min(remaining * 0.15, 150.0)
        a_opt, obj = simulated_annealing(best_seq, time_budget=sa2_budget, verbose=False, n_restarts=3)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After SA2: {best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            p4b_budget = min(remaining * 0.2, 120.0)
            a_opt, obj = full_pipeline(best_seq, time_budget=p4b_budget, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 4 polish: {best_score:.8f}")

    # === Phase 5: Final intensive coordinate descent ===
    print("\n=== Phase 5: Final intensive CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        cd_final_budget = min(remaining * 0.3, remaining - 10.0)
        a_opt, obj = exact_coord_descent(best_seq, max_sweeps=5000, time_budget=cd_final_budget,
                                          verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After final CD: {best_score:.8f}")

    # === Phase 6: One more SA + CD cycle if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Final cycle ({remaining:.0f}s) ===")
        sa3_budget = min(remaining * 0.3, 100.0)
        a_opt, obj = simulated_annealing(best_seq, time_budget=sa3_budget, verbose=False, n_restarts=3)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

        remaining = deadline - time.monotonic()
        if remaining > 10:
            cd3_budget = min(remaining * 0.5, remaining - 5.0)
            a_opt, obj = exact_coord_descent(best_seq, max_sweeps=10000, time_budget=cd3_budget,
                                              verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  After Phase 6 CD: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
