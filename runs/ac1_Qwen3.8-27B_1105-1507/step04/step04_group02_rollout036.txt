    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

    n = len(a)

    n = len(a)
    rows, cols, vals = [], [], []
    for j in range(n):
        start = j
        end = min(j + n, 2*n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j]*length)
        vals.extend(a[:length].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

"""

import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d
from scipy.sparse import coo_matrix, csr_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T such that T@x = conv(a, x)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        length = end - start
        rows.extend(range(start, end))
        cols.extend([j] * length)
        vals.extend(a[:length].tolist())
    mat = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return mat


def solve_lp_maxsum(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, dual_vars)."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            dual = np.zeros(2 * n - 1)
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                    dual = np.maximum(res.ineqlin.marginals, 0)
            return g, dual
    except Exception:
        pass
    return None, None


def compute_subgradient_full(a, conv, M, rel_tol=0.05):
    """
    Compute subgradient of max(conv) using all 'near-active' constraints
    (those within rel_tol of the max). Returns (gradient, active_count).
    """
    n = len(a)
    threshold = M * (1.0 - rel_tol)
    active_mask = conv >= threshold
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(conv)])
    
    grad = np.zeros(n)
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        for m in range(lo, hi):
            j = k - m
            if 0 <= j < n:
                grad[m] += 2.0 * a[j]
    grad /= len(active_idx)
    return grad, len(active_idx)


def subgradient_descent(a, n_iters, time_budget, rel_tol=0.05, initial_step=0.08):
    """Momentum subgradient descent with adaptive step and stalling detection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    momentum = 0.85
    step = initial_step
    stall = 0
    best_stall_step = step
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        grad, n_active = compute_subgradient_full(a, conv, M, rel_tol)
        grad_centered = grad - np.mean(grad)
        
        v = momentum * v + (1.0 - momentum) * grad_centered
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                if stall > 200:
                    step *= 0.6
                    v *= 0.5
                    if step < 1e-8:
                        break
                elif stall > 100:
                    # Structured frequency-domain perturbation
                    rng = np.random.default_rng(it * 7919 + 13)
                    freq_pert = np.zeros(n)
                    n_pert_modes = rng.integers(1, 4)
                    for _ in range(n_pert_modes):

                        phase = rng.uniform(0, 2 * np.pi)

                        freq_pert += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                    a = a * np.exp(freq_pert)
                    a = np.maximum(a, 0)
                    S = np.sum(a)
                    if S < 1e-10:
                        break
                    a = a / S
                    v = np.zeros(n)
                    stall = 0
    
    return best_a


def alternating_lp_multiM(a, n_iters, time_budget):

        return a

    best_obj = compute_obj(a)

            break

        best_improvement = 0

            if time.monotonic() - start > time_budget:

            g, dual = solve_lp_maxsum(a, M_try, time_limit=3)

                continue
            g = g / Sg

                a_try = a_try / St

def guided_subgradient_lp(a, n_iters, time_budget):
    """
    Hybrid method: use LP dual variables to guide subgradient descent.
    At each step, solve LP to get dual vars, then do a subgradient step
    weighted by the dual information.
    """

    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.03
    stall = 0

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP to get dual variables
        g, dual = solve_lp_maxsum(a, M * 1.01, time_limit=4)
        
        if dual is not None and np.sum(dual) > 1e-10:
            # Weight convex constraints by dual variables
            # grad[m] = sum_k dual[k] * 2*a[k-m] / sum(dual)
            grad = np.zeros(n)
            active_k = np.where(dual > 1e-10)[0]
            for k in active_k:
                w = dual[k]
                lo = max(0, k - n + 1)
                hi = min(k + 1, n)
                for m in range(lo, hi):
                    j = k - m
                    if 0 <= j < n:
                        grad[m] += w * 2.0 * a[j]
            grad /= np.sum(dual[active_k])
        else:
            # Fallback to uniform active set
            grad, _ = compute_subgradient_full(a, conv, M, 0.05)
        
        grad_centered = grad - np.mean(grad)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                step *= 0.7
                if step < 1e-8:
                    break
    
    return best_a


def cosine_init(n, n_modes=10, seed=0):
    """Generate a smooth cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.8, 0.8) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def interpolate_sequence(a_base, n_target):
    """Interpolate a sequence to a new length using cubic spline."""
    x_old = np.linspace(0, 1, len(a_base))
    x_new = np.linspace(0, 1, n_target)
    try:
        f_interp = interp1d(x_old, a_base, kind='cubic', fill_value='extrapolate')
        a_new = np.maximum(f_interp(x_new), 0)
    except Exception:
        a_new = np.ones(n_target) / n_target
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(n_target) / n_target
    return a_new / S

            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                print(f"  C{cycle} LP: {best_score:.6f} ** NEW BEST **")
        
        # Perturbation escape
        remaining = deadline - time.monotonic()
        if remaining > 25:
            rng = np.random.default_rng(cycle * 31337 + seed)
            scales = [0.03, 0.06, 0.1]
            for scale in scales:
                if time.monotonic() > deadline - 20:
                    break
                a_p = a_cur * np.exp(rng.normal(0, scale, n_cur))
                a_p = np.maximum(a_p, 0)
                S_p = np.sum(a_p)
                if S_p < 1e-10:
                    continue
                a_p = a_p / S_p
                # Quick SG + LP
                a_p = subgradient_descent(a_p, n_iters=100000, time_budget=min(remaining * 0.05, 10.0))
                a_p = alternating_lp_multiM(a_p, n_iters=100000, time_budget=min(remaining * 0.05, 15.0))
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_cur = a_p.copy()
                    print(f"  C{cycle} Pert({scale}): {best_score:.6f} ** NEW BEST **")
    
    # === Phase 4: Try different n for final push ===
    print("\n=== Phase 4: Final multi-n exploration ===")
    if time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        n_final_tries = sorted(set([
            n_best - 50, n_best + 50, n_best - 100, n_best + 100, n_best + 150
        ]))
        n_final_tries = [n for n in n_final_tries if 100 <= n <= 1500]
        
        for n_try in n_final_tries:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            a_interp = interpolate_sequence(best_seq, n_try)
            # Refine
            a_new, obj_new = refine_sequence(a_interp, time_budget=min(remaining * 0.3, 40.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  n={n_try}: {best_score:.6f} ** NEW BEST **")
            else:
                print(f"  n={n_try}: {obj_new:.6f}")
    
    # === Phase 5: Final intensive polish ===
    print("\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = best_seq.copy()
        a_final = a_final / np.sum(a_final)
        
        # Extended LP
        lp_t = min(remaining * 0.5, 60.0)
        a_lp = alternating_lp_multiM(a_final, n_iters=1000000, time_budget=lp_t)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {best_score:.6f} ** NEW BEST **")
        
        # One last guided descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_gd = guided_subgradient_lp(best_seq, n_iters=1000000, time_budget=min(remaining * 0.4, 30.0))
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  Final GD: {best_score:.6f} ** NEW BEST **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(400) / 400
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"\n=== FINAL: n={len(result)}, evaluate={val:.6f}, internal={best_score:.6f} ===")
    
    if val != np.inf and val < best_score:
        best_score = val
    
    return result
