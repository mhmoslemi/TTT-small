def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

def smooth_max_and_grad(a, beta):

    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    p = exp_b / exp_b.sum()
    
    n = len(a)
    grad = np.zeros(n)

        pk = p[k]

        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]

    return smooth_val, grad, p, b


def normalize_simplex(a):

def build_A_matrix(n, a):

    cols = []

        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_best_response(a, M, time_limit=10):

        if r.success and r.x is not None:

            # Get dual variables (shadow prices)
            y = None

                if y_raw.size == 2 * n - 1:
                    y = np.maximum(-y_raw, 0)  # Duals for A_ub constraints

        pass

def smooth_max_grad(a, beta):
    """Compute (1/beta)*log(sum exp(beta*b_k)) and its gradient."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_b = np.exp(shifted)
    total = exp_b.sum()
    p = exp_b / total
    val = b_max + np.log(total) / beta
    
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
