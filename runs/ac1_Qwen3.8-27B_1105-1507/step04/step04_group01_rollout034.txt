1 - sum_k lambda_k = 0 (complementary slackness for M >= 0)
sum_k lambda_k * (b*b)[k] - mu*b - nu = 0 (stationarity)
lambda_k * ((b*b)[k] - M) = 0 (complementary slackness)
nu_i * b_i = 0 (complementary slackness)
lambda_k >= 0, nu_i >= 0, M >= 0

From complementary slackness: lambda_k > 0 only if (b*b)[k] = M.

From stationarity: b_i = (1/mu) * (sum_k lambda_k * d(b*b)[k]/db_i - nu_i)

For b_i > 0: nu_i = 0, so b_i = (1/mu) * sum_k lambda_k * d(b*b)[k]/db_i

This tells us that at the optimum, b_i is proportional to the "weighted gradient" of the active constraints.

This gives us a fixed-point iteration:
1. Given lambda (probabilities on active set), compute b_i = (1/mu) * sum_k lambda_k * d(b*b)[k]/db_i
2. Normalize b to sum to 1
3. Update lambda based on which constraints are active

This is similar to the mirror descent / softmax approach but with a different interpretation.

1. **Primary: Projected gradient / Frank-Wolfe with adaptive smoothing**
2. **Secondary: LP polishing (more aggressive)**
3. **Tertiary: Spectral-based initialization**
4. **Quaternary: Multiple lengths and restarts**

Key differences from previous approach:
- Use Frank-Wolfe instead of L-BFGS-B for the main optimization (avoids the softmax parametrization which can be unstable)
- More aggressive LP with better warm-starting
- Try to use the structure of the problem more directly

```python
"""
Hybrid Frank-Wolfe + LP + spectral initialization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses adaptive smoothing, aggressive LP
polishing, and spectral-based restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def smooth_max_val_and_weights(conv, tau):
    """Compute smoothed max and softmax weights."""
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    Z = float(np.sum(exp_vals))
    w = exp_vals / Z
    return cmax + tau * np.log(Z), w

def grad_smooth_max_from_weights(a, w):
    """
    Compute gradient of smoothed max w.r.t. a, given weights w on convolution.
    grad_a[i] = 2 * sum_{j} w[i+j] * a[j] (approximate via FFT)
    More precisely: grad_a[i] = sum_k w[k] * d(conv[k])/da[i]
    d(conv[k])/da[i] = a[k-i] + a[i] (if k-i in range, adjusted for k=2i)
    Using FFT: grad_a = 2 * iff(ifft(convolve(w_rev, a))[n-1:2n-1]) ... 
    Actually: sum_k w[k] * a[k-i] = (w_rev * a)[2n-2-i] where w_rev = w[::-1]
    And sum_k w[k] * a[i] (when k-i=i, i.e., k=2i) = a[i] * w[2i]
    So grad_a[i] = (w_rev * a)[2n-2-i] + (w_rev * a)[2n-2-i] + a[i]*w[2i]... 
    Hmm, let me just use the standard approach:
    grad_a[i] = 2 * sum_j w[i+j] * a[j] (this is the "twice the cross-correlation" formula)
    which equals 2 * (cross_corr(w, a))[i]
    """
    n = len(a)
    # cross_corr[i] = sum_j w[i+j] * a[j] for valid j
    # = sum_j w[i+j] * a[j]
    # Let's compute via FFT: this is the "anti-convolution" or cross-correlation
    # w has length 2n-1, a has length n
    # We want c[i] = sum_j w[i+j] * a[j] for i=0..n-1
    # This is like correlating w with a, shifted
    # c[i] = sum_j w[i+j]*a[j] = (w * a_flipped)[i + n - 1 - ...]
    # Let me just use: c = fft_convolve(w, a[::-1])[n-1:2n-1][::-1]
    # Actually: sum_j w[i+j]*a[j], let k = i+j, j = k-i
    # = sum_k w[k]*a[k-i] (for valid k and k-i)
    # This is the cross-correlation of w and a at lag i
    # cross_corr[i] = sum_k w[k]*a[k-i]
    # = (w * a_flipped)[i + n - 1] where a_flipped = a[::-1]
    # Let's verify: (w * a_rev)[m] = sum_k w[k]*a_rev[m-k] = sum_k w[k]*a[n-1-(m-k)] = sum_k w[k]*a[n-1-m+k]
    # We want sum_k w[k]*a[k-i]. Set n-1-m+k = k-i => n-1-m = -i => m = i+n-1
    # So cross_corr[i] = (w * a_rev)[i+n-1]
    
    w_len = len(w)  # 2n-1
    a_rev = a[::-1]
    conv_wa = fft_convolve(w, a_rev)
    # conv_wa has length (2n-1) + n - 1 = 3n-2
    # cross_corr[i] = conv_wa[i + n - 1] for i = 0..n-1
    cross_corr = conv_wa[n-1:2*n-1]
    grad_a = 2.0 * cross_corr
    return grad_a

def frank_wolfe_optimize(a_init, tau_list, max_iter=500, time_budget=30.0):
    """
    Frank-Wolfe (conditional gradient) method with adaptive smoothing.
    Optimizes on the probability simplex.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        a_cur = a_cur.copy() if 'a_cur' in dir() else a.copy()
        
        for it in range(max_iter):
            if time.monotonic() - start > time_budget:
                break
            
            conv = compute_conv(a_cur)
            sm_val, w = smooth_max_val_and_weights(conv, tau)
            grad_a = grad_smooth_max_from_weights(a_cur, w)
            
            # Frank-Wolfe: find s = e_{argmin grad_a}
            j_min = int(np.argmin(grad_a))
            
            # Step size: t = 2/(it+2) or line search
            t = 2.0 / (it + 2)
            
            # Update: a_new = (1-t)*a + t*e_j
            e_j = np.zeros(n)
            e_j[j_min] = 1.0
            a_new = (1.0 - t) * a_cur + t * e_j
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a_cur = a_new
            a = a_new
        
        # Refine with smaller steps
        # ...
    
    return best_a, best_obj
```

Hmm, the Frank-Wolfe with single vertex (one-hot) might be too slow. Let me use a modified version with "away steps" or use the full conditional gradient where the minimizer over the simplex of g^T v is just the vertex with smallest g_i.

Actually, for the simplex, the linear minimizer IS a vertex. So Frank-Wolfe always moves towards a single coordinate. This can be very slow.

A better approach: **Modified Frank-Wolfe (MF-W)** with away steps, or **25-step Frank-Wolfe** (take multiple small steps towards the same vertex).

Or even better: **Projected gradient descent** with the Euclidean projection onto the simplex.

The Euclidean projection onto the simplex {x : sum(x) = 1, x >= 0} can be done in O(n log n) time.

Let me use **L-R (Lloyd-Rudin) algorithm** or the standard sorting-based projection.

Actually, for the probability simplex, the projection is:
proj_simplex(v) = max(v - tau, 0) where tau is chosen so that sum(proj) = 1.

This can be found by sorting v in O(n log n).

```python
"""
Projected gradient + LP polishing + spectral restarts for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses adaptive tau smoothing, 
aggressive LP with warm starts, and spectral-based diversity.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

# --- Core utilities ---

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def smooth_max_and_weights(conv, tau):
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    Z = float(np.sum(exp_vals))
    w = exp_vals / Z
    val = cmax + tau * np.log(Z)
    return val, w

def grad_from_weights(a, w):
    """Gradient of smoothed max w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    conv_wa = fft_convolve(w, a_rev)
    cross_corr = conv_wa[n-1:2*n-1]
    grad_a = 2.0 * cross_corr
    return grad_a

def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

# --- Optimization methods ---

def projected_gradient_descent(a_init, tau_list, time_budget=30.0, max_iter_per_tau=500):
    """PGD on simplex with adaptive smoothing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_obj = best_obj
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        step = 1.0 / (2.0 * n * tau)  # Lipschitz-based step
        # Actually, the Lipschitz constant of the smoothed max gradient is bounded by 2*n (roughly)
        # So step = 1/(2*L) where L ~ 2*n*max(a) ~ 2
        step = 1.0 / max(2.0, 4.0 * n * np.max(a))
        
        momentum = 0.9
        m = np.zeros(n)
        
        for it in range(max_iter_per_tau):
            if time.monotonic() - start > time_budget:
                break
            
            conv = compute_conv(a)
            _, w = smooth_max_and_weights(conv, tau)
            grad = grad_from_weights(a, w)
            
            # Grad + momentum
            m = momentum * m + (1 - momentum) * grad
            a_new = a - step * m
            a_new = proj_simplex(a_new)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            # Adaptive step: if no progress, reduce step
            if it > 50 and obj_new >= prev_obj - 1e-10:
                step *= 0.8
            prev_obj = obj_new
            a = a_new
            
            if it > 10 and np.max(np.abs(a_new - a)) < 1e-10:
                break
        
        a = best_a.copy()
    
    return best_a, best_obj

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=15):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None

def lp_polish(a, max_iters=50, time_budget=30.0):
    """Alternating LP polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_g(a, M, time_limit=min(10, time_budget - (time.monotonic() - start)))
        if g is None:
            break
        
        # Try different blending factors
        best_t_obj = None
        best_a_new = a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if best_t_obj is None or obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
        
        if best_t_obj is not None and best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
        else:
            # Try min-peak LP
            g2, M2 = solve_lp_minpeak(a, time_limit=min(10, time_budget - (time.monotonic() - start)))
            if g2 is not None:
                for t in [0.3, 0.5, 0.8, 1.0]:
                    a_try = (1.0 - t) * a + t * g2
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
        
        # Check convergence
        if it > 5 and abs(best_obj - compute_obj(a)) < 1e-10:
            break
    
    return best_a

def spectral_init(n, seed):
    """Initialize with flat power spectrum (Zadoff-Chu like)."""
    rng = np.random.default_rng(seed)
    
    # Create a sequence with approximately flat power spectrum
    # In frequency domain: |A_k|^2 should be roughly constant
    # So A_k = exp(i*theta_k) for some phases theta_k
    # Then a = ifft(A) might have negative values, so we rectify
    
    # Method 1: Random phases, rectify
    # Method 2: Cosh transform
    
    choice = rng.integers(0, 5)
    
    if choice == 0:
        # Random phase, rectify
        phases = rng.uniform(0, 2 * np.pi, n)
        A = np.exp(1j * phases)
        a = np.real(ifft(A)).real
        a = np.maximum(a, 0.01)
    elif choice == 1:
        # Cosine with random frequency
        k = np.arange(n)
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Two-tone
        k = np.arange(n)
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        # Smooth random
        a = rng.random(n)
        # Low-pass filter
        from scipy.ndimage import gaussian_filter1d
        sigma = max(1.0, n / 30)
        a = gaussian_filter1d(a, sigma)
        a = np.maximum(a, 0.01)
    else:
        # Chirp
        k = np.arange(n)
        f0 = rng.uniform(1, 3)
        f1 = rng.uniform(3, 10)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
        a = np.maximum(a, 0.01)
    
    s = np.sum(a)
    return a / s

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
```

```python
"""
Hybrid projected gradient + multiplicative mirror descent + aggressive LP
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses adaptive tau smoothing,
geometric/Hadamard blend updates, multi-threshold LP, and spectral
initialization for diversity.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def smooth_max_val_and_weights(conv, tau):
    """Compute smoothed max value and softmax weights over convolution."""
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    Z = float(np.sum(exp_vals))
    w = exp_vals / Z
    val = cmax + tau * np.log(Z)
    return val, w


def grad_from_weights(a, w):
    """
    Compute gradient of smoothed max w.r.t. a, given weights w on convolution coeffs.
    grad_a[i] = sum_k w[k] * d(conv[k])/da[i]
    Using FFT cross-correlation: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    """
    n = len(a)
    a_rev = a[::-1]
    conv_wa = fft_convolve(w, a_rev)
    # conv_wa has length (2n-1) + n - 1 = 3n-2
    # cross_corr[i] = conv_wa[i + n - 1] for i = 0..n-1
    cross_corr = conv_wa[n - 1:2 * n - 1]
    grad_a = 2.0 * cross_corr
    return grad_a


def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def pgd_optimize(a_init, tau_list, time_budget=30.0, max_iter_per_tau=500, use_mirror=False):
    """
    Projected gradient descent (or mirror descent) on simplex with adaptive smoothing.
    minimize f_tau(a) = tau * log(sum_k exp((a*a)[k]/tau)) s.t. a in simplex
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        # Step size: for Lipschitz L = 2/tau, safe step = 1/L = tau/2
        # Use slightly smaller for stability
        step = tau / 3.0
        max_step = 1.0 / (4.0 * n)
        step = min(step, max_step)

        for it in range(max_iter_per_tau):
            if time.monotonic() - start > time_budget:
                break
            if stall > 200:
                break

            conv = compute_conv(a)
            _, w = smooth_max_val_and_weights(conv, tau)
            grad = grad_from_weights(a, w)

            if use_mirror:
                # Mirror descent: a_new[i] = a[i] * exp(-step * grad[i])
                a_new = a * np.exp(np.clip(-step * grad, -50, 50))
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    a_new = a.copy()
                else:
                    a_new = a_new / s_new
            else:
                # Projected gradient: a_new = proj_simplex(a - step * grad)
                a_new = proj_simplex(a - step * grad)

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall == 100:
                    step *= 0.7

            a = a_new

        # After each tau, reset to best
        a = best_a.copy()

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=12):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=12):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iters=60, time_budget=40.0, aggressive=False):
    """
    Alternating LP with multiple blending strategies including
    geometric mean and Hadamard product updates.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP with current M
        g = solve_lp_g(a, M, time_limit=min(10, max(3, remaining - 5)))
        if g is None:
            break

        # Try multiple blending strategies
        best_t_obj = best_obj
        best_a_new = best_a.copy()

        # Strategy 1: Linear blend
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()

        # Strategy 2: Hadamard product (emphasize where both are large)
        if aggressive:
            prod = a * g
            s_prod = np.sum(prod)
            if s_prod > 1e-10:
                a_try = prod / s_prod
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_new = a_try.copy()

        # Strategy 3: Geometric mean
        if aggressive:
            for alpha in [0.3, 0.5, 0.7]:
                a_try = np.power(np.maximum(a, 1e-30), 1 - alpha) * np.power(np.maximum(g, 1e-30), alpha)
                s_try = np.sum(a_try)
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_t_obj:
                        best_t_obj = obj_try
                        best_a_new = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a_new
            stall = 0
        else:
            stall += 1

            # Try min-peak LP as escape
            if stall > 5:
                g2, M2 = solve_lp_minpeak(a, time_limit=min(8, max(3, remaining - 5)))
                if g2 is not None:
                    for t in [0.3, 0.5, 0.8, 1.0]:
                        a_try = (1.0 - t) * a + t * g2
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try.copy()
                            stall = 0
                            break

            # Perturbation to escape
            if stall > 15:
                rng = np.random.default_rng(it * 7919 + 42)
                scale = rng.uniform(0.03, 0.1)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 5

    return best_a


def multiplicative_update_optimize(a_init, tau_list, time_budget=20.0, max_iter_per_tau=300):
    """
    Multiplicative (exponential) mirror descent on simplex.
    a_new[i] = a[i] * exp(-step * grad[i]), then normalize.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 1e-30)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break

        step = tau / 4.0
        max_step = 1.0 / (2.0 * n)
        step = min(step, max_step)

        for it in range(max_iter_per_tau):
            if time.monotonic() - start > time_budget:
                break
            if stall > 150:
                break

            conv = compute_conv(a)
            _, w = smooth_max_val_and_weights(conv, tau)
            grad = grad_from_weights(a, w)

            # Center gradient (to maintain sum constraint better)
            grad_centered = grad - np.dot(a, grad)

            a_new = a * np.exp(np.clip(-step * grad_centered, -30, 30))
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall == 80:
                    step *= 0.7
                    # Small perturbation
                    rng = np.random.default_rng(it * 31 + 7)
                    a = a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 1e-10)
                    a = a / np.sum(a)
                    stall = 30

            a = a_new

        a = best_a.copy()

    return best_a, best_obj


def spectral_init(n, seed):
    """Generate diverse spectral-based initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Random phase -> real part -> rectify
        phases = rng.uniform(0, 2 * np.pi, n)
        A = np.exp(1j * phases)
        a = np.real(ifft(A))
        a = np.maximum(a, 0.01)
    elif choice == 1:
        # Cosine with random freq
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Two-tone
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        # Smooth random (Gaussian filtered)
        a = rng.random(n)
        # Simple moving average
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Chirp
        f0 = rng.uniform(0.5, 3.0)
        f1 = rng.uniform(3.0, 10.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + (f1 - f0) * k**2 / (2 * n**2)))
        a = np.maximum(a, 0.01)
    elif choice == 5:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.6 * np.exp(-0.5 * ((k - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    else:
        # Power spectrum shaping: try to flatten |A|^2
        target = np.ones(n) * rng.uniform(0.5, 1.5)
        # a = ifft(sqrt(target) * exp(i*random_phase))
        phases = rng.uniform(0, 2 * np.pi, n)
        A = np.sqrt(target) * np.exp(1j * phases)
        a = np.real(ifft(A))
        a = np.maximum(a, 0.01)

    s = np.sum(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing: PGD + LP + mirror descent in alternating rounds."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # PGD with small tau
        if remaining > 15:
            pgd_budget = min(remaining * 0.3, 20.0)
            tau = 0.5 / n
            a_new, obj_new = pgd_optimize(best_a, [tau, tau * 0.4, tau * 0.15],
                                          time_budget=pgd_budget, max_iter_per_tau=300)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} PGD: {best_obj:.6f}")

        # LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 800:
            lp_budget = min(remaining * 0.35, 25.0)
            a_new = lp_alternate(best_a, max_iters=40, time_budget=lp_budget, aggressive=True)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} LP: {best_obj:.6f}")

        # Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            md_budget = min(remaining * 0.2, 12.0)
            tau_md = 1.0 / n
            a_new, obj_new = multiplicative_update_optimize(
                best_a, [tau_md, tau_md * 0.3],
                time_budget=md_budget, max_iter_per_tau=200)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    r{round_num} MD: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines projected gradient, mirror descent, and aggressive LP
    with geometric/Hadamard blending. Explores multiple lengths and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(best_seq)

    print(f"Initial best: {best_score:.6f}")

    # N values to explore (focus on sizes near parent)
    n_values = [500, 550, 600, 650, 700, 750, 800, 850]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(400, np_ - 50), min(1000, np_ + 50), min(1000, np_ + 100), min(1000, np_ + 150)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")

    candidates = []  # (a, obj, n)

    # Phase 1: Multi-init exploration
    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation per n
        if n <= 600:
            alloc = min(70.0, remaining * 0.14)
        elif n <= 800:
            alloc = min(60.0, remaining * 0.12)
        else:
            alloc = min(50.0, remaining * 0.10)
        alloc = max(25.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        n_inits = max(2, min(4, int(alloc / 18)))
        per_init = alloc / n_inits

        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 250:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 20:
                break

            # Initialization
            if init_idx == 0 and best_seq is not None:
                a = resample_sequence(best_seq, n)
            else:
                a = spectral_init(n, seed + n * 1009 + init_idx * 7919)

            # Step 1: PGD (additive) with tau annealing
            alloc1 = min(per_init * 0.35, 20.0)
            tau_list1 = [8.0/n, 4.0/n, 2.0/n, 1.0/n, 0.5/n]
            a, obj = pgd_optimize(a, tau_list1, time_budget=alloc1, max_iter_per_tau=300)

            # Step 2: Mirror descent (multiplicative)
            remaining_init = deadline - time.monotonic()
            if remaining_init > 30:
                alloc2 = min(per_init * 0.25, 15.0)
                tau_list2 = [3.0/n, 1.5/n, 0.75/n, 0.375/n]
                a, obj = multiplicative_update_optimize(a, tau_list2, time_budget=alloc2,
                                                        max_iter_per_tau=250)

            # Step 3: LP polish
            remaining_init = deadline - time.monotonic()
            if remaining_init > 30 and n <= 850:
                alloc3 = min(per_init * 0.30, 20.0)
                a = lp_alternate(a, max_iters=35, time_budget=alloc3, aggressive=True)
                obj = compute_obj(a)

            # Step 4: Fine PGD
            remaining_init = deadline - time.monotonic()
            if remaining_init > 25:
                alloc4 = min(per_init * 0.15, 10.0)
                tau_fine = [0.3/n, 0.15/n]
                a, obj = pgd_optimize(a, tau_fine, time_budget=alloc4, max_iter_per_tau=500)

            print(f"  init{init_idx}: n={n}, obj={obj:.6f}")

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

            candidates.append((a.copy(), obj, n))

    # Phase 2: Polish top candidates
    print(f"\n=== POLISH PHASE ===")
    print(f"Best before polish: {best_score:.6f}")

    candidates.sort(key=lambda x: x[1])

    n_polish = min(4, len(candidates))
    for ci in range(n_polish):
        if time.monotonic() >= deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][0]
        n_cand = candidates[ci][2]
        polish_budget = min(remaining * 0.35, 90.0)

        print(f"  Polishing candidate {ci} (n={n_cand}, obj={candidates[ci][1]:.6f}), budget={polish_budget:.0f}s")

        a_pol, obj_pol = polish_sequence(a_cand, time_budget=polish_budget, verbose=True)

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE after polish: obj={best_score:.6f} **")

        # Perturbation-based restarts
        remaining = deadline - time.monotonic()
        if remaining > 40:
            rng = np.random.default_rng(ci * 555 + 999)
            for ptry in range(4):
                if time.monotonic() > deadline - 50:
                    break
                # Different perturbation scales
                scale = rng.uniform(0.04, 0.15)
                a_pert = a_cand * np.exp(rng.normal(0, scale, n_cand))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s

                # Quick optimization cycle
                tau_p = 2.0 / n_cand
                a_new, obj_new = pgd_optimize(a_pert, [tau_p, tau_p * 0.4],
                                              time_budget=min(remaining * 0.1, 10.0),
                                              max_iter_per_tau=300)

                # Quick LP
                remaining_p = deadline - time.monotonic()
                if remaining_p > 50 and n_cand <= 800:
                    a_new = lp_alternate(a_new, max_iters=15, time_budget=min(remaining_p * 0.1, 10.0))
                    obj_new = compute_obj(a_new)

                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  ** UPDATE (perturb {ptry}, scale={scale:.3f}): obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()

        for final_round in range(5):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Very small tau PGD
            tau_vsmall = 0.05 / n_best
            a_new, obj_new = pgd_optimize(
                a_cur, [0.4/n_best, 0.1/n_best, tau_vsmall],
                time_budget=min(remaining * 0.25, 18.0), max_iter_per_tau=800)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final PGD {final_round}: {best_score:.6f}")

            # Mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_new, obj_new = multiplicative_update_optimize(
                    a_cur, [0.5/n_best, 0.1/n_best],
                    time_budget=min(remaining * 0.15, 10.0), max_iter_per_tau=500)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final MD {final_round}: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and n_best <= 850:
                a_new = lp_alternate(a_cur, max_iters=40,
                                     time_budget=min(remaining * 0.3, 30.0), aggressive=True)
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Final LP {final_round}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
