def normalize_sequence(a):
    """Normalize the sequence to have unit energy, but ensuring non-negative and sum condition."""
    n_val = len(a)
    if n_val <= 0:

    a = [min(1000.0, x) for x in a]
    s = sum(a)
    if s < 0.01:
        # Scale up to satisfy sum condition, without violating cap
        scale_factor = 0.01 / s
        a = [x * scale_factor for x in a]
    return a
