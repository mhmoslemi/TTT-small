def active_set_solve(a_init, tol_positive=1e-6, tol_active=0.01, max_iters=1000):
    a = a_init.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    
    for it in range(max_iters):
        b = fast_conv(a)
        b_max = b.max()
        
        # Active set of convolution positions
        S = np.where(b >= b_max * (1 - tol_active))[0]
        
        # Active set of coordinates
        A = np.where(a > tol_positive)[0]
        
        if len(A) == 0 or len(S) == 0:
            break
        
        m = len(A)
        # Build M: M[row, col] = 1 if (A[row] + A[col]) in S
        S_set = set(S.tolist())
        M = np.zeros((m, m))
        for rs in range(m):
            for cs in range(m):
                if (A[rs] + A[cs]) in S_set:
                    M[rs, cs] = 1
        
        # Solve: M @ a_A = c * 1, sum(a_A) = 1
        # [M, 1; 1^T, 0] @ [a_A; c] = [1; 1] ... wait no
        # M @ a_A = c * 1_m
        # 1_m^T @ a_A = 1
        # => [M, -1; 1^T, 0] @ [a_A; c] = [0; 1]
        # or: [M, 1; 1^T, 0] @ [a_A; c'] = [0; 1] where c' = -c
        # Let me use: M @ a_A - c * 1 = 0
        #            1^T @ a_A = 1
        # Augmented system: [M, -1; 1^T, 0] [a_A; c] = [0; 1]
        
        Aug = np.zeros((m + 1, m + 1))
        Aug[:m, :m] = M
        Aug[:m, m] = -1.0
        Aug[m, :m] = 1.0
        
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(Aug, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(Aug, rhs, rcond=None)[0]
        
        a_new_A = sol[:m]
        c_val = sol[m]
        
        # Check non-negativity
        if np.min(a_new_A) < -1e-10:
            # Remove the most negative coordinate from A
            worst = np.argmin(a_new_A)
            A = np.delete(A, worst)
            m = len(A)
            if m < 2:
                break
            continue
        
        a_new = np.zeros(n)
        a_new[A] = np.maximum(a_new_A, 0)
        
        # Check if sum is close to 1
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        
        a = a_new
    
    return a

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def active_set_solve(a_init, tol_positive=1e-6, tol_active_frac=0.02, max_iters=500, deadline=None):
    """Solve using KKT conditions: at optimum, for all active i,
    sum_{k in S} a[k-i] = constant, and sum(a) = 1."""
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Active set of convolution positions (those close to max)
        threshold = b_max * (1.0 - tol_active_frac)
        S = np.where(b >= threshold)[0]
        S_list = S.tolist()
        S_set = set(S_list)
        
        # Active set of coordinates
        A_mask = a > tol_positive
        A = np.where(A_mask)[0]
        m = len(A)
        
        if m < 2 or len(S) < 1:
            break
        
        # Build M matrix: M[row, col] = 1 if (A[row] + A[col]) in S
        # For efficiency, use vectorized approach
        A_sum = A[:, None] + A[None, :]  # m x m matrix of sums
        M = ((A_sum[:, :, None] == S[None, None, :]).any(axis=2)).astype(np.float64)
        
        # Actually this is memory intensive for large m. Let's use a different approach.
        # M[row, col] = 1 iff A[row] + A[col] in S_set
        M = np.zeros((m, m), dtype=np.float64)
        for rs in range(m):
            targets = S + A[rs]  # values of A[col] that make A[rs]+A[col] in S
            M[rs, :] = np.isin(A, targets)
        
        # Solve: M @ a_A = c * 1, sum(a_A) = 1
        Aug = np.zeros((m + 1, m + 1))
        Aug[:m, :m] = M
        Aug[:m, m] = -1.0
        Aug[m, :m] = 1.0
        
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(Aug, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(Aug, rhs, rcond=1e-12)[0]
            except:
                break
        
        a_new_A = sol[:m]
        
        # Check non-negativity
        min_val = np.min(a_new_A)
        if min_val < -1e-8:
            # Remove the most negative coordinate
            worst_local = int(np.argmin(a_new_A))
            A = np.delete(A, worst_local)
            m = len(A)
            if m < 2:
                break
            continue
        
        # Construct new solution
        a_new = np.zeros(n)
        a_new[A] = np.maximum(a_new_A, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        # Check if this is actually better
        b_new = fast_conv(a_new)
        if b_new.max() > b_max * 1.05:
            # Got significantly worse, blend
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a /= a.sum()
        else:
            # Check convergence
            if np.max(np.abs(a_new - a)) < 1e-12:
                a = a_new
                break
            a = a_new
    
    return a


def lp_iteration(a, M_target, step_candidates=None):
    """One LP iteration: find best response and try step sizes."""
    n = len(a)
    g = lp_best_response(a, M_target)
    if g is None:
        return None, None
    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, None
    g /= g_sum
    
    if step_candidates is None:
        step_candidates = [0.3, 0.2, 0.1, 0.5, 0.05]
    
    best_a = None
    best_J = compute_J(a)
    
    for t in step_candidates:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_a = a_trial
    
    return best_a, best_J


def coordinate_descent_pass_fast(a, deadline=None, max_iterations=None):
    """Fast coordinate descent with incremental convolution updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S  # normalize
    S = 1.0
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max()
    
    max_iter = max_iterations if max_iterations is not None else n * 3
    
    for iteration in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Select coordinate (cyclic or random)
        i = iteration % n
        
        ai = a[i]
        if ai < 1e-10:
            continue
        
        # The affected b positions are k in [i, i+n-1]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        
        # Unaffected max
        if k_lo > 0:
            b_before = b[:k_lo].max() if k_lo > 0 else 0
        else:
            b_before = 0
        if k_hi < 2 * n - 2:
            b_after = b[k_hi+1:].max() if k_hi + 1 < 2 * n - 1 else 0
        else:
            b_after = 0
        M_unaffected = max(b_before, b_after)
        
        # For the affected positions, b[k] changes linearly with x (except k=2i)
        # b[k]_new = b[k] + 2*a[k-i]*(x - ai) for k != 2i
        # b[2i]_new = b[2i] - ai^2 + x^2
        # J(x) = 2n * max(M_unaffected, max_affected(x)) / (1 + x - ai)^2
        
        # Find optimal x via golden section on [0, 3*ai]
        x_lo = 0.0
        x_hi = max(3.0 * ai, 0.1)
        
        def eval_J(x):
            new_S = S - ai + x
            if new_S < 1e-10:
                return 1e10
            max_b = M_unaffected
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        bk = b[k] - ai * ai + x * x
                    else:
                        bk = b[k] + 2.0 * a[ki] * (x - ai)
                else:
                    bk = b[k]
                if bk > max_b:
                    max_b = bk
            return 2.0 * n * max_b / (new_S * new_S)
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = x_hi - gr * (x_hi - x_lo)
        d_pt = x_lo + gr * (x_hi - x_lo)
        fc = eval_J(c_pt)
        fd = eval_J(d_pt)
        
        for _ in range(25):
            if fc < fd:
                x_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = x_hi - gr * (x_hi - x_lo)
                fc = eval_J(c_pt)
            else:
                x_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = x_lo + gr * (x_hi - x_lo)
                fd = eval_J(d_pt)
        
        x_opt = (x_lo + x_hi) / 2.0
        x_opt = max(0.0, x_opt)
        
        if abs(x_opt - ai) > 1e-12:
            # Update a[i] and b
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - ai * ai + x_opt * x_opt
                    else:
                        b[k] = b[k] + 2.0 * a[ki] * (x_opt - ai)
            a[i] = x_opt
            S = S - ai + x_opt
    
    # Normalize
    S = a.sum()
    if S > 1e-10:
        a /= S
    return a


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine windows
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.cos(np.pi * (t - 0.5))**2 + 0.1)
    inits.append(np.maximum(np.cos(np.pi * t), 0) + 0.05)
    
    # Triangular
    inits.append(1 - np.abs(2 * t - 1))
    
    # Gaussian
    inits.append(np.exp(-2 * (t - 0.5)**2 / (0.1 * 0.1)) * 0.3 + 0.7)
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.2)
    
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    
    # Two-peak
    inits.append(0.5 * np.exp(-30*(t-0.25)**2) + 0.5 * np.exp(-30*(t-0.75)**2) + 0.1)
    
    # Random smooth
    for _ in range(3):
        f = 1.0 + 0.3 * np.cos(2*np.pi*rng.uniform(1,5)*t + rng.uniform(0,6.28))
        f += 0.2 * np.cos(2*np.pi*rng.uniform(5,15)*t + rng.uniform(0,6.28))
        inits.append(np.maximum(f, 0.01))
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    return inits


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_phase(a, deadline, tracker, max_iters=200, tighten_start=0.005):
    """Run LP best-response iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_start
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-5:
                break
            continue
        
        g = np.maximum(g, 0.0)
        g /= g.sum()
        
        # Try step sizes with line search
        best_a = None
        best_J = current_J
        steps = [0.5, 0.3, 0.2, 0.1, 0.7]
        for t in steps:
            a_t = (1-t)*a + t*g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10: continue
            a_t /= s
            J_t = compute_J(a_t)
            if J_t < best_J - 1e-14:
                best_J = J_t
                best_a = a_t
        
        if best_a is None:
            stall += 1
            tighten *= 0.7
            if stall > 30:
                break
            continue
        
        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.02, tighten * 0.995)
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows_list.extend([k] * cnt)
            cols_list.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals_list.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows_list)):
        A[rows_list[idx], cols_list[idx]] = vals_list[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def active_set_solve(a_init, tol_positive=1e-5, tol_active_frac=0.03, max_iters=300, deadline=None):
    """KKT-based active set iteration.
    At optimum: for all active i (a_i>0), sum_{k in S} a[k-i] = constant.
    Solves the resulting linear system iteratively."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Active set of convolution positions
        threshold = b_max * (1.0 - tol_active_frac)
        S = np.where(b >= threshold)[0]
        # Boolean mask for fast lookup
        S_bool = np.zeros(2 * n - 1, dtype=bool)
        S_bool[S] = True

        # Active set of coordinates
        A = np.where(a > tol_positive)[0]
        m = len(A)

        if m < 2 or len(S) < 1:
            break

        # Build M matrix: M[row, col] = 1 if (A[row] + A[col]) in S
        A_sum = A[:, None] + A[None, :]
        M = S_bool[A_sum].astype(np.float64)

        # Solve: M @ a_A = c * 1_m, sum(a_A) = 1
        # Augmented: [M, -1; 1^T, 0] [a_A; c] = [0; 1]
        Aug = np.zeros((m + 1, m + 1))
        Aug[:m, :m] = M
        Aug[:m, m] = -1.0
        Aug[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0

        try:
            sol = np.linalg.solve(Aug, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(Aug, rhs, rcond=1e-12)[0]
            except Exception:
                break

        a_new_A = sol[:m]

        # Handle non-negativity violations
        min_val = np.min(a_new_A)
        if min_val < -1e-8:
            # Remove the most negative coordinate and retry
            worst_local = int(np.argmin(a_new_A))
            A = np.delete(A, worst_local)
            m = len(A)
            if m < 2:
                break
            continue

        # Construct new solution
        a_new = np.zeros(n)
        a_new[A] = np.maximum(a_new_A, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new

        # Check if this is actually an improvement
        b_new = fast_conv(a_new)
        if b_new.max() > b_max * 1.02:
            # Got worse, blend
            a = 0.6 * a + 0.4 * a_new
            a = np.maximum(a, 0)
            a /= a.sum()
        else:
            # Check convergence
            if np.max(np.abs(a_new - a)) < 1e-12:
                a = a_new
                break
            a = a_new

    return a


def mirror_descent(a_init, eta0=0.05, deadline=None, max_iters=5000, tracker=None):
    """Mirror descent (exponential weights) on the simplex for min max(b)."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-8).copy()
    a /= a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    eta = eta0
    patience = 0
    improvements = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Smoothed active set via softmax
        temp = 0.005 * b_max
        if temp < 1e-15:
            temp = 1e-15
        weights = np.exp((b - b_max) / temp)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.zeros(2 * n - 1)
            weights[np.argmax(b)] = 1.0
        else:
            weights /= wsum

        # Compute subgradient: g[j] = sum_k weights[k] * a[k-j]
        g = np.zeros(n)
        for j in range(n):
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                g[j] = np.dot(weights[k_lo:k_hi + 1], a[idx - j])

        # Mirror descent: multiplicative update
        log_a = np.log(a + 1e-30) - eta * g
        # Numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_sum = a_new.sum()
        if a_sum < 1e-10:
            break
        a_new /= a_sum
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            patience = 0
            improvements += 1
            if tracker is not None and improvements % 10 == 0:
                tracker.update(a_new)
        else:
            patience += 1

        # Adapt learning rate
        if patience == 100:
            eta *= 0.7
            patience = 0
        elif patience == 200:
            eta *= 0.5
            patience = 0

        a = a_new

        if patience > 1000:
            break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def run_lp_phase(a, deadline, tracker, max_iters=150, tighten_start=0.004):
    """Run LP best-response iterations with adaptive tightening."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.001).copy()
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    tighten = tighten_start
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-5:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search over multiple step sizes
        best_a = None
        best_J = current_J
        steps = [0.4, 0.25, 0.15, 0.5, 0.1, 0.6]
        for t in steps:
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            J_t = compute_J(a_t)
            if J_t < best_J - 1e-14:
                best_J = J_t
                best_a = a_t

        if best_a is None:
            stall += 1
            tighten *= 0.6
            if stall > 40:
                break
            continue

        a = best_a
        current_J = best_J
        stall = 0
        tighten = min(0.015, tighten * 0.998)

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_pass(a, deadline=None, n_passes=3):
    """Fast coordinate descent with incremental convolution updates."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    n = len(a)
    b = fast_conv(a).copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        for i in range(n):
            ai = a[i]
            if ai < 1e-12:
                # Try setting to a small value
                x_candidates = [0.01 / n, 0.02 / n]
                best_x = 0.0
                best_J = compute_J(a)
                for x in x_candidates:
                    a[i] = x
                    S_tmp = S - ai + x
                    b_tmp = b.copy()
                    for k in range(max(0, i), min(2 * n - 1, i + n)):
                        ki = k - i
                        if 0 <= ki < n:
                            if k == 2 * i:
                                b_tmp[k] = b[k] - ai * ai + x * x
                            else:
                                b_tmp[k] = b[k] + 2.0 * a[ki] * (x - ai)
                    J_tmp = 2.0 * n * b_tmp.max() / (S_tmp * S_tmp)
                    if J_tmp < best_J:
                        best_J = J_tmp
                        best_x = x
                    b = b_tmp
                if best_x > 0:
                    a[i] = best_x
                    S = S - ai + best_x
                    for k in range(max(0, i), min(2 * n - 1, i + n)):
                        ki = k - i
                        if 0 <= ki < n:
                            if k == 2 * i:
                                b[k] = b[k] - ai * ai + best_x * best_x
                            else:
                                b[k] = b[k] + 2.0 * a[ki] * (best_x - ai)
                continue

            # Golden section search for optimal value of a[i]
            x_lo = 0.0
            x_hi = max(3.0 * ai, 1.0 / n)

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Precompute unaffected max
            M_unaff = 0.0
            if k_lo > 0:
                M_unaff = max(M_unaff, b[:k_lo].max())
            if k_hi + 1 < 2 * n - 1:
                M_unaff = max(M_unaff, b[k_hi + 1:].max())

            c_pt = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)

            def eval_x(x):
                new_S = S - ai + x
                if new_S < 1e-15:
                    return 1e10
                max_b = M_unaff
                for k in range(k_lo, k_hi + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            bk = b[k] - ai * ai + x * x
                        else:
                            bk = b[k] + 2.0 * a[ki] * (x - ai)
                    else:
                        bk = b[k]
                    if bk > max_b:
                        max_b = bk
                return 2.0 * n * max_b / (new_S * new_S)

            fc = eval_x(c_pt)
            fd = eval_x(d_pt)

            for _ in range(30):
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = x_hi - gr * (x_hi - x_lo)
                    fc = eval_x(c_pt)
                else:
                    x_lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = eval_x(d_pt)

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            if abs(x_opt - ai) > 1e-12:
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            b[k] = b[k] - ai * ai + x_opt * x_opt
                        else:
                            b[k] = b[k] + 2.0 * a[ki] * (x_opt - ai)
                a[i] = x_opt
                S = S - ai + x_opt

    # Final normalization
    S_final = a.sum()
    if S_final > 1e-10:
        a /= S_final
    return a


def generate_structured_inits(n, rng):
    """Generate structured initial sequences based on theory."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Theoretical continuous optimum: raised cosine (cos^2)
    inits.append(np.cos(np.pi * (t - 0.5)) ** 2 + 1e-4)

    # Half-cosine (zero at edges)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05)

    # Triangular
    inits.append(1.0 - np.abs(2.0 * t - 1.0) + 0.05)

    # Gaussian centered
    inits.append(0.3 * np.exp(-4.0 * (t - 0.5) ** 2) + 0.7)
    inits.append(0.5 * np.exp(-6.0 * (t - 0.5) ** 2) + 0.5)

    # Super-Gaussian
    inits.append(0.3 * np.exp(-8.0 * (t - 0.5) ** 2) + 0.7)

    # Cosine with slight asymmetry
    inits.append(np.cos(np.pi * (t - 0.45)) ** 2 + 0.05)

    # Two-lobe
    inits.append(0.4 * np.exp(-20 * (t - 0.3) ** 2) + 0.4 * np.exp(-20 * (t - 0.7) ** 2) + 0.2)

    # Cosine with frequency modulation
    for freq in [0.5, 1.0, 1.5]:
        inits.append(0.5 * (1 + np.cos(2 * np.pi * freq * t)) + 0.05)

    # Random smooth
    for _ in range(3):
        f = 0.8 + 0.2 * np.cos(2 * np.pi * rng.uniform(0.5, 2) * t + rng.uniform(0, 6.28))
        inits.append(np.maximum(f, 0.01))

    # Uniform (baseline)
    inits.append(np.ones(n))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic-like interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        n = 386

        tracker.update(parent)
        print(f"Default raised cosine: n={n}, J={tracker.best_score:.8f}", flush=True)

        print(f"{'='*60}", flush=True)
