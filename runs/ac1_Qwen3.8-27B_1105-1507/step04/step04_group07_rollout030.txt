    """Compute LSE-smoothed objective and its gradient w.r.t. log-parameterization."""
    # Softmax parameterization
    p = params - params.max()
    exp_p = np.exp(p)
    a = exp_p / exp_p.sum()
    
    # Compute autocorrelation
    b = fast_conv(a)  # length 2n-1
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p_dist = exp_b / exp_b.sum()
    
    # LSE value
    lse_val = b_max + np.log(exp_b.sum()) / tau
    obj = 2.0 * n * lse_val  # sum(a) = 1
    
    # Gradient of LSE w.r.t. a:
    # d(LSE)/d(a_j) = sum_k p_dist[k] * d(b_k)/d(a_j)
    # = (a ⋆ p_dist)[j] + (a * p_dist)[j]  [cross-correlation + convolution]
    
    s = 1
    while s < 3 * n:
        s <<= 1
    fa = np.fft.fft(a, s)
    fp = np.fft.fft(p_dist, s)
    
    # Cross-correlation: ifft(conj(fa) * fp)
    # Convolution: ifft(fa * fp)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_objective(params, tau, n):
    """LSE-smoothed objective. params -> a = softmax(params). Returns (f, grad)."""
    a = softmax(params)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p_dist = exp_b / exp_sum
    lse_val = b_max + np.log(exp_sum) / tau
    obj = 2.0 * n * lse_val
    
    # Gradient of LSE w.r.t. a_j:
    # = (a cross-correlate p_dist)[j] + (a correlate p_dist)[j]
    s = 1
    while s < 3 * n:
        s <<= 1
    fa = np.fft.fft(a, s)
    fp = np.fft.fft(p_dist, s)
    grad_a = np.fft.ifft(fp * (np.conj(fa) + fa), s).real[:n]
    
    # Chain rule through softmax
    s_grad = np.dot(grad_a, a)
    grad_params = a * (grad_a - s_grad)
    
    return obj, grad_params

def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-10)

_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:

        cols = []

            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def lp_best_response(a, M):
    """Find g >= 0 maximizing sum(g) s.t. (a*g)[k] <= M for all k."""
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)

        j = cols[i]

    b_ub = np.full(2 * n - 1, M)

    return None
