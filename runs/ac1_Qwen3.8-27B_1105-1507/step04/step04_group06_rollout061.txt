    Returns (smooth_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
        smooth_val = b_max
    else:
        p = exp_shifted / Z
        smooth_val = b_max + np.log(Z) / beta
    
    # Gradient = 2 * correlate(p, a) via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P_hat = np.fft.rfft(p, N)
    A_hat = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_hat * np.conj(A_hat), N)[:n]
    grad = 2.0 * corr
    return smooth_val, grad

def J_and_grad_smooth(a, beta):
    """Compute J_smooth(a) and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    F_val, grad_F = smooth_max_and_grad(a, beta)
    J_val = 2.0 * n * F_val / (s * s)
    # dJ/da_i = 2n * (s * dF/da_i - 2*F) / s^3
    grad_J = 2.0 * n * (s * grad_F - 2.0 * F_val) / (s ** 3)
    return J_val, grad_J

# ====== Multiplicative Exponential Descent ======

def med_step(a, beta, eta):
    """
    One multiplicative exponential descent step.
    Returns (a_new, J_new).
    """
    J_val, grad_J = J_and_grad_smooth(a, beta)
    # Multiplicative update: a * exp(-eta * grad)
    log_update = -eta * grad_J
    # Clip to prevent overflow/underflow
    log_update = np.clip(log_update, -50, 50)
    update = np.exp(log_update)
    a_new = a * update
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a, J_val
    a_new = a_new / s
    J_new = compute_J(a_new)
    return a_new, J_new

def med_optimize(a, beta_schedule, deadline, tracker=None):
    """
    Multiplicative Exponential Descent with adaptive step size.
    beta_schedule: list of (beta, n_iters) tuples
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    for beta, n_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        eta = 1.0  # Initial step size
        stall = 0
        
        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            
            # Try the step
            a_new, J_new = med_step(a, beta, eta)
            
            if J_new < current_J - 1e-15:
                a = a_new
                current_J = J_new
                stall = 0
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(a)
            else:
                # Backtrack: reduce eta
                eta = max(eta * 0.5, 1e-10)
                stall += 1
                if stall > 30:
                    break
        
        # After each beta phase, try to increase eta slightly for next phase
        eta = min(eta * 2.0, 5.0)
    
    return best_a

# ====== LP Mirror Descent (from previous, kept for polish) ======

def get_A_structured(n):
    """Get pre-computed (row, col) for convolution constraint matrix."""
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32),
                       np.array(cols, dtype=np.int32))
    return _A_cache[n]

def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    rows_arr, cols_arr = get_A_structured(n)
    vals = a[rows_arr - cols_arr]
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=15, stall_patience=30):
    """LP-based mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        n = len(a)
        
        g = lp_solve(a, M, n)
        if g is None:
            break
        
        # Line search
        best_t = 0.0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < current_J - 1e-14:
                a = a_try
                current_J = J_try
                best_t = t
                stall_count = 0
                if J_try < best_J:
                    best_J = J_try
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(a)
                break
        
        if best_t == 0.0:
            stall_count += 1
            if stall_count >= stall_patience:
                break
    
    return best_a

def lbfgs_optimize(a0, beta, max_iters=100):
    """L-BFGS-B with smooth objective."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-8)
    
    def func(x):
        x2 = np.maximum(x, 0)
        f, _ = J_and_grad_smooth(x2, beta)
        return f if np.isfinite(f) else 1e10
    
    def jac(x):
        x2 = np.maximum(x, 0)
        _, g = J_and_grad_smooth(x2, beta)
        return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    
    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a0.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iters, 'max_fun': max_iters*10})
        if res.fun < 1e10:
            return np.maximum(res.x, 1e-10)
    except Exception:
        pass
    return a0

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. Multiplicative Exponential Descent (MED) - O(n log n) per step via FFT
2. Log-sum-exp smoothing with adaptive beta annealing
3. LP mirror descent for final precision polish
4. L-BFGS-B for basin finding
5. Multi-restart with diverse perturbations
6. Multi-length exploration
Key innovation: MED is 10-20x faster per iteration than LP, enabling far more
optimization steps within the time budget.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

_A_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a.
    Returns (smooth_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
        smooth_val = b_max
    else:
        p = exp_shifted / Z
        smooth_val = b_max + np.log(Z) / beta

    # Gradient = 2 * correlate(p, a, 'full')[:n] via FFT
    N = 1
    while N < 3 * n:
        N <<= 1
    P_hat = np.fft.rfft(p, N)
    A_hat = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_hat * np.conj(A_hat), N)[:n]
    grad = 2.0 * corr
    return smooth_val, grad


def J_and_grad_smooth(a, beta):
    """Compute J_smooth(a) = 2n*F_beta(a)/s^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    F_val, grad_F = smooth_max_and_grad(a, beta)
    J_val = 2.0 * n * F_val / (s * s)
    grad_J = 2.0 * n * (s * grad_F - 2.0 * F_val) / (s ** 3)
    return J_val, grad_J


def med_step(a, beta, eta):
    """One multiplicative exponential descent step with adaptive clipping."""
    J_val, grad_J = J_and_grad_smooth(a, beta)
    # Center the gradient for better numerical behavior
    grad_centered = grad_J - np.mean(grad_J * a) / max(a.sum(), 1e-10)
    log_update = -eta * grad_centered
    log_update = np.clip(log_update, -40, 40)
    update = np.exp(log_update)
    a_new = a * update
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a, J_val
    a_new = a_new / s
    return a_new, J_val


def med_optimize(a, beta_schedule, deadline, tracker=None, verbose=True):
    """
    Multiplicative Exponential Descent with adaptive step size and beta annealing.
    beta_schedule: list of (beta, n_iters) tuples.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()

    for phase_idx, (beta, n_iters) in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break

        eta = 1.0
        stall = 0
        improved_this_phase = 0

        for it in range(n_iters):
            if time.monotonic() > deadline:
                break

            a_new, _ = med_step(a, beta, eta)
            J_new = compute_J(a_new)

            if J_new < current_J - 1e-15:
                a = a_new
                current_J = J_new
                stall = 0
                improved_this_phase += 1
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(a)
                # Occasionally increase eta
                if stall == 0 and it % 50 == 0:
                    eta = min(eta * 1.1, 5.0)
            else:
                eta = max(eta * 0.7, 1e-8)
                stall += 1
                if stall > 50:
                    break

        if verbose:
            elapsed = time.monotonic() - (deadline - (n_iters * 0.01))
            print(f"  MED beta={beta:.0f}: J={current_J:.8f} (improved {improved_this_phase} steps)", flush=True)

def lbfgs_optimize(a0, beta, max_iters=100, deadline=None):
    """L-BFGS-B with smooth objective for basin finding."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-8)

    def func(x):
        x2 = np.maximum(x, 0)
        f, _ = J_and_grad_smooth(x2, beta)
        return f if np.isfinite(f) else 1e10

    def jac(x):
        x2 = np.maximum(x, 0)
        _, g = J_and_grad_smooth(x2, beta)
        return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

    bounds = [(0.0, None)] * n
    try:
        opts = {'maxiter': max_iters, 'max_fun': max_iters * 15}
        if deadline is not None:
            opts['iprint'] = -1
        res = minimize(func, a0.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds, options=opts)
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 1e-10)
            if new_a.sum() > 1e-6:
                return new_a
    except Exception:
        pass
    return a0


def get_A_structured(n):
    """Get pre-computed (row, col) arrays for convolution constraint matrix."""
    if n not in _A_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_cache[n] = (np.array(rows, dtype=np.int32),
                       np.array(cols, dtype=np.int32))
    return _A_cache[n]


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    rows_arr, cols_arr = get_A_structured(n)
    vals = a[rows_arr - cols_arr]
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None, stall_patience=30):
    """LP-based mirror descent with line search for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    stall_count = 0
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(a, M, n)
        if g is None:
            break

        # Line search with multiple t values
        improved = False
        for t in [0.3, 0.5, 0.2, 0.7, 0.1, 0.8, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < current_J - 1e-14:
                a = a_try
                current_J = J_try
                stall_count = 0
                improved = True
                if J_try < best_J:
                    best_J = J_try
                    best_a = a.copy()
                    if tracker is not None:
                        tracker.update(a)
                break

        if not improved:
            stall_count += 1
            if stall_count >= stall_patience:
                break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            # Short LP polish
            budget_lp = min(25.0, dl4 - time.monotonic())
            if budget_lp > 5:
                a = lp_mirror_descent(a, int(budget_lp * 20),
                                      time.monotonic() + budget_lp,
                                      tracker=tracker, stall_patience=15)

    # === Phase 5: Length exploration ===
    if tracker.best is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*50} Phase 5: Length Exploration ({remaining:.0f}s) {'='*50}", flush=True)
            n_best = tracker.best.size
            candidates = [n_best - 256, n_best - 128, n_best + 128, n_best + 256,
                          768, 1536, 2048, 3072]
            seen = set()
            for n_try in candidates:
                if n_try < 128 or n_try > 4096 or n_try in seen:
                    continue
                if abs(n_try - n_best) < 64:
                    continue
                if time.monotonic() > deadline:
                    break
                seen.add(n_try)
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline - time.monotonic())
                if budget_here < 5:
                    break
                short_sched = [(20.0, 200), (100.0, 300), (500.0, 500), (2000.0, 800)]
                a_res = med_optimize(a_res, short_sched, time.monotonic() + budget_here,
                                     tracker=tracker, verbose=False)

    # === Phase 6: Final intense polish ===
    if tracker.best is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n{'='*50} Phase 6: Final Polish ({remaining:.0f}s) {'='*50}", flush=True)
            # Very high beta MED for final refinement
            final_sched = [
                (1000.0, 500),
                (2000.0, 1000),
                (5000.0, 2000),
            ]
            a = tracker.best.copy()
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            a = med_optimize(a, final_sched, deadline, tracker=tracker, verbose=False)
            tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    return [float(x) for x in result]
