import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv_fft(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def soft_max_grad(conv, p):
    """Compute gradient of soft-max approximation (sum(exp(p*conv)/N))^(1/p)"""
    # Use log-sum-exp for numerical stability
    max_c = np.max(conv)
    shifted = p * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights, max_c + np.log(np.sum(exp_shifted)) / p

def compute_smooth_subgradient(a, p, topk=None):

    Compute subgradient of soft-max approximation of max(conv(a,a)).
    Returns gradient vector of length n.
    """
    n = len(a)

    weights, soft_max_val = soft_max_grad(conv, p)

        sub_weights = np.zeros(len(conv))
        sub_weights[idx] = weights[idx]
        sub_weights /= sub_weights.sum()
        weights = sub_weights

    # d(conv[k])/da[i] = a[k-i] + a[i-k] ... wait, no.
    # conv[k] = sum_j a[j]*a[k-j]
    # d(conv[k])/da[i] = a[k-i] if 0<=k-i<n (from the first factor)

    # Actually: conv[k] = sum_j a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
    # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=i<n, from second factor where j=k-i)
    # Wait, let me be more careful.
    # conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * a[k-j]
    # d(conv[k])/da[i] = sum over terms where j=i or k-j=i
    #   = a[k-i] (if 0<=k-i<n, from j=i term: a[i]*a[k-i])
    #   + a[i] * ... no wait
    #   From j=i: a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
    #   From k-j=i, i.e., j=k-i: a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
    # But if i = k-i, i.e., k=2i, then it's a[i]^2, derivative is 2*a[i]
    # So in general: d(conv[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if k!=2i
    #                          = 2*a[i] if k=2i (which is the same formula: 2*a[k-i] = 2*a[i])
    # Wait, that's always the same! d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n

    # ∂conv[k]/∂a[i] = sum_j [δ(j,i)*a[k-j] + a[j]*δ(k-j,i)]
    #   = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)
    #   = 2*a[k-i] (if 0<=k-i<n)
    # Yes! So the gradient is always 2*a[k-i] when 0<=k-i<n.
    
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-15:
            continue

        # d(conv[k])/da[i] = 2*a[k-i] for valid i

def lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Return g or None."""
    n = len(a)
    # Build Toeplitz matrix: (conv(a,g))[k] = sum_i a[max(0,k-n+1):min(k,n-1)] * g[..., ]
    # Actually: conv[k] = sum_j a[j]*g[k-j] for valid j,k-j
    # For fixed a, this is a linear function of g.
    # conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
    # Let i = k-j, so j = k-i, i ranges from k-min(k,n-1) to k-max(0,k-n+1)
    # i ranges from max(0, k-n+1) to min(k, n-1)
    # conv[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * g[i]
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def subgradient_descent(a, time_budget, topk=10, lr_init=0.05):
    """Projected subgradient descent on simplex with exact max subgradient."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.7
    stagnation = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)
        M = np.max(conv)
        
        # Find active set (top-k)
        active = np.argsort(conv)[-topk:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            idx = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end+1] += 2.0 * a[idx]
        grad /= len(active)
        
        # Center
        grad_c = grad - np.mean(grad)
        v = momentum * v + (1 - momentum) * grad_c
        
        # Step
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            break

            stagnation = 0

            stagnation += 1
            if stagnation > 30:
                lr *= 0.85
                v *= 0.5
                stagnation = 0

def simulated_annealing_escape(a, best_obj, time_budget, n_rounds=5):

def random_structured_init(n, seed=0):
    """Generate a random structured initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    n_modes = rng.integers(3, 10)
    for m in range(1, n_modes + 1):
        amp = rng.uniform(-0.5, 0.5) / (m * 0.7)
        phase = rng.uniform(0, 2*np.pi)
        a += amp * np.cos(2*np.pi*m*k/n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n)/n
    return a / S

def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    from scipy.interpolate import interp1d
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = np.maximum(f(x_new), 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return np.ones(n_new)/n_new
    return a_new / S

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max mirror descent (adaptive temperature), exact subgradient with momentum,
alternating LP refinement, frequency-domain perturbations, and multi-length exploration.
Key novelty over previous: adaptive soft-max schedule, dual-informed subgradient,
and Fourier-domain structured perturbations for better escape from local optima.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
from scipy.interpolate import interp1d
import time


def compute_conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv_fft(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def soft_max_weights(conv, p):
    """Compute softmax weights over convolution values with temperature p."""
    max_c = np.max(conv)
    shifted = p * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv), max_c
    weights = exp_shifted / total
    smv = max_c + np.log(total) / p
    return weights, smv


def subgradient_from_weights(a, weights):
    """
    Compute (weighted) subgradient of sum_k weights[k]*conv[k] w.r.t. a.
    d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w * a[idx]
    return grad


def soft_max_mirror_descent(a, p, time_budget, lr_init=0.1):
    """
    Mirror descent (multiplicative weights) with soft-max smoothing.
    Works on normalized sequences (sum=1).
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    stagnation = 0
    decay = 0.995

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)
        weights, smv = soft_max_weights(conv, p)
        grad = subgradient_from_weights(a, weights)
        grad_c = grad - np.mean(grad)

        # Adaptive learning rate
        lr_eff = lr * decay ** stagnation
        if lr_eff < 1e-8:

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

            best_a = a_new.copy()

            stagnation = 0

            stagnation += 1
            if stagnation > 40:

                if lr < 1e-7:

def exact_subgradient_descent(a, time_budget, topk=8, lr_init=0.05):
    """Projected subgradient with momentum, using top-k active set."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.75
    stagnation = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(a)
        M = np.max(conv)
        active = np.argsort(conv)[-topk:]

        # Weighted subgradient (weight by how close to max)
        active_vals = conv[active]
        weights = (active_vals - M) + M  # = active_vals
        w_norm = weights / (weights.sum() + 1e-30)

        grad = np.zeros(n)
        for ii, k in enumerate(active):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                idx = k - np.arange(i_start, i_end + 1)
                grad[i_start:i_end + 1] += 2.0 * w_norm[ii] * a[idx]

        grad_c = grad - np.mean(grad)
        v = momentum * v + (1 - momentum) * grad_c

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)

            best_a = a_new.copy()
            a = a_new
            stagnation = 0
        else:
            a = a_new
            stagnation += 1
            if stagnation > 30:
                lr *= 0.85
                v *= 0.5
                stagnation = 0
                if lr < 1e-8:
                    break

    return best_a, best_obj


def build_toeplitz_sparse(a):
    """Build sparse matrix A where (A*g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-10:
                return g
    except Exception:
        pass
    return None


def alternating_lp(a, time_budget, n_iters=30):
    """Alternating LP with line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stagnation = 0

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        M = float(np.max(conv))

        g = lp_direction(a, M, time_limit=4)
        if g is None:
            break
        g = g / np.sum(g)

        # Line search
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = normalize(a)
            best_obj = best_t_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 12:
                break
            a = 0.95 * a + 0.05 * g
            a = normalize(a)

    return best_a, best_obj


def fourier_perturb(a, n_perturb=3, scale=0.05, seed=0):
    """Perturb a few random Fourier coefficients of the sequence."""
    n = len(a)
    rng = np.random.default_rng(seed)
    fa = np.fft.rfft(a)
    # Don't perturb DC or Nyquist
    interior = np.arange(1, len(fa) - (1 if n % 2 == 0 else 0))
    if len(interior) < n_perturb:
        n_perturb = max(1, len(interior) // 2)
    idx = rng.choice(interior, size=min(n_perturb, len(interior)), replace=False)
    for i in idx:
        fa[i] *= np.exp(rng.normal(0, scale, 2) + 1j * rng.normal(0, scale, 1))
    # Make conjugate symmetric
    a_new = np.fft.irfft(fa, n)
    return normalize(a_new)


def cosine_init(n, n_modes=8, seed=0):
    """Cosine-series initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        c = rng.uniform(-0.5, 0.5) / (m * 0.8)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * m * k / n + phase)
    return normalize(a)


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = np.maximum(f(x_new), 0)
    return normalize(a_new)


def combined_refine(a, time_budget, use_lp=True, verbose=False):
    """Run a full cycle of refinement methods on a sequence."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Soft-max mirror descent
        t1 = min(remaining * 0.2, 12.0)
        a_new, obj_new = soft_max_mirror_descent(a, p=16, time_budget=t1, lr_init=0.08)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"    SM: {best_obj:.6f}")

        # Exact subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t2 = min(remaining * 0.2, 10.0)
            a_new, obj_new = exact_subgradient_descent(a, time_budget=t2, topk=6, lr_init=0.04)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose:
                    print(f"    SG: {best_obj:.6f}")

        # LP
        if use_lp and n <= 500:
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 8:
                t3 = min(remaining * 0.25, 25.0)
                a_new, obj_new = alternating_lp(a, time_budget=t3, n_iters=15)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"    LP: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(budget_s=1500):
    """
    Main search: multi-phase optimization combining soft-max mirror descent,
    exact subgradient, LP, Fourier perturbations, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        best_seq = normalize(parent.copy())
        best_score = compute_obj(best_seq)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, obj={best_score:.6f}")
    except Exception:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
        n_parent = 300
        print(f"No parent. Uniform n={n_parent}, obj={best_score:.6f}")

    # === Phase 1: Adaptive soft-max schedule ===
    print("\n=== Phase 1: Adaptive soft-max mirror descent ===")
    a_cur = best_seq.copy()
    for p in [2, 4, 8, 16, 32, 64, 128]:
        if time.monotonic() > deadline - 120:
            break
        t_budget = min(20.0, (deadline - time.monotonic()) * 0.1)
        a_new, obj_new = soft_max_mirror_descent(a_cur, p=p, time_budget=t_budget, lr_init=0.07)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  p={p}: {best_score:.6f} ** IMPROVED **")
        else:
            print(f"  p={p}: {obj_new:.6f}")
            a_cur = a_new.copy()

    # === Phase 2: Exact subgradient with varying topk ===
    print("\n=== Phase 2: Exact subgradient ===")
    a_cur = best_seq.copy()
    for topk in [3, 5, 8, 15, 30]:
        if time.monotonic() > deadline - 100:
            break
        t_budget = min(18.0, (deadline - time.monotonic()) * 0.1)
        a_new, obj_new = exact_subgradient_descent(a_cur, time_budget=t_budget, topk=topk, lr_init=0.04)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  topk={topk}: {best_score:.6f} ** IMPROVED **")
        else:
            print(f"  topk={topk}: {obj_new:.6f}")

    # === Phase 3: LP refinement ===
    print("\n=== Phase 3: LP refinement ===")
    n_cur = len(best_seq)
    if n_cur <= 550:
        a_cur = best_seq.copy()
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_budget = min(remaining * 0.2, 60.0)
            a_new, obj_new = alternating_lp(a_cur, time_budget=t_budget, n_iters=35)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  LP: {best_score:.6f} ** IMPROVED **")
            else:
                print(f"  LP: {obj_new:.6f}")

    # === Phase 4: Fourier-domain structured perturbations ===
    print("\n=== Phase 4: Fourier perturbations ===")
    a_cur = best_seq.copy()
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_pert = min(12, int(remaining * 0.12))
        for i in range(n_pert):
            if time.monotonic() > deadline - 30:
                break
            rng_seed = i * 2654435761 + 7
            scale = 0.03 + 0.04 * (i % 3)
            a_pert = fourier_perturb(a_cur, n_perturb=2, scale=scale, seed=rng_seed)
            # Quick refine
            t_q = min(6.0, (deadline - time.monotonic()) * 0.05)
            a_q, obj_q = exact_subgradient_descent(a_pert, time_budget=t_q, topk=5, lr_init=0.04)
            if obj_q < best_score:
                best_score = obj_q
                best_seq = a_q.copy()
                a_cur = a_q.copy()
                print(f"  Pert{i} (scale={scale:.3f}): {best_score:.6f} ** IMPROVED **")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length exploration ===")
    n_base = len(best_seq)
    n_tries = sorted(set([
        max(60, n_base - 100), max(60, n_base - 50),
        n_base + 50, n_base + 100, n_base + 200,
    ]))
    n_tries = [n for n in n_tries if 50 <= n <= 2000]

    for n_try in n_tries:
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        a_interp = interpolate_seq(best_seq, n_try)
        t_budget = min(remaining * 0.2, 25.0)
        a_new, obj_new = combined_refine(a_interp, time_budget=t_budget,
                                         use_lp=(n_try <= 450), verbose=False)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.6f} ** IMPROVED **")
        else:
            print(f"  n={n_try}: {obj_new:.6f}")

    # === Phase 6: Intensive combined refinement ===
    print("\n=== Phase 6: Intensive refinement ===")
    a_cur = best_seq.copy()
    for round_num in range(25):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        t_budget = min(remaining * 0.15, 20.0)
        use_lp = len(a_cur) <= 500 and remaining > 20
        a_new, obj_new = combined_refine(a_cur, time_budget=t_budget,
                                         use_lp=use_lp, verbose=False)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_cur = a_new.copy()
            print(f"  R{round_num}: {best_score:.6f} ** IMPROVED **")
        else:
            # Try perturbation escape
            if remaining > 15:
                rng = np.random.default_rng(round_num * 15485863 + 3)
                a_pert = a_cur * np.exp(rng.normal(0, 0.06, len(a_cur)))
                a_pert = normalize(a_pert)
                t_q = min(8.0, remaining * 0.1)
                a_q, obj_q = combined_refine(a_pert, time_budget=t_q,
                                             use_lp=(len(a_cur) <= 500), verbose=False)
                if obj_q < best_score:
                    best_score = obj_q
                    best_seq = a_q.copy()
                    a_cur = a_q.copy()
                    print(f"  R{round_num} pert: {best_score:.6f} ** IMPROVED **")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final polish ===")
    a_cur = best_seq.copy()
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Extended subgradient with very small steps
        t_final = min(remaining * 0.4, 30.0)
        a_new, obj_new = exact_subgradient_descent(a_cur, time_budget=t_final, topk=5, lr_init=0.015)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Final SG: {best_score:.6f} ** IMPROVED **")

        remaining = deadline - time.monotonic()
        if remaining > 8 and len(best_seq) <= 450:
            a_new, obj_new = alternating_lp(a_cur, time_budget=min(remaining * 0.5, 25.0), n_iters=20)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final LP: {best_score:.6f} ** IMPROVED **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = min(best_score, val)
    else:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
