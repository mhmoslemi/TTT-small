def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    
    # Build sparse matrix: row k, col j -> a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        jl = max(0, k - n + 1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

        if violated:
            active_ks = active_ks + violated[:max_new_constraints]
            return lp_with_active_set(a, M, active_ks, max_new_constraints)
    
    return g, active_ks

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
1. Intensive LP mirror descent (best response) with adaptive parameters
2. Fast coordinate descent with incremental convolution updates
3. Multi-length exploration
4. Structured perturbation restarts
5. Final polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 2
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_sparse_conv(a):
    """Build sparse matrix A where (A @ g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_best_response(a, M):
    """Solve LP: max 1^T g s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_sparse_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def cd_optim_1d(a, i, b, S):
    """Optimize a[i] given b and S using golden section."""
    n = len(a)
    old = a[i]
    
    # Precompute affected values
    k_lo = max(0, i)
    k_hi = min(2*n-2, i+n-1)
    affected = np.arange(k_lo, k_hi+1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        unaff_max = b[mask].max() if mask.any() else 0.0
    else:
        unaff_max = 0.0
    
    def obj(x):
        new_S = S - old + x
        if new_S < 1e-15:
            return 1e30
        max_c = unaff_max
        for k in affected:
            ki = k - i
            if k == 2*i:
                ck = b[k] - old*old + x*x
            else:
                ck = b[k] - 2.0*old*a[ki] + 2.0*x*a[ki]
            if ck > max_c:
                max_c = ck
        return 2.0*n*max_c / (new_S*new_S)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(4.0*old, 4.0*S/n, 0.01)
    
    c_pt = hi - gr*(hi-lo)
    d_pt = lo + gr*(hi-lo)
    fc, fd = obj(c_pt), obj(d_pt)
    
    for _ in range(60):
        if fc < fd:
            hi, d_pt, fd = d_pt, c_pt, fc
            c_pt = hi - gr*(hi-lo)
            fc = obj(c_pt)
        else:
            lo, c_pt, fc = c_pt, d_pt, fd
            d_pt = lo + gr*(hi-lo)
            fd = obj(d_pt)
    
    return max(0.0, (lo+hi)/2.0)


def cd_pass(a, b, S):
    """One full pass of coordinate descent."""
    n = len(a)
    for i in range(n):
        old = a[i]
        new = cd_optim_1d(a, i, b, S)
        if abs(new - old) > 1e-14:
            # Update b incrementally
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            for k in range(k_lo, k_hi+1):
                ki = k - i
                if k == 2*i:
                    b[k] -= old*old - new*new
                else:
                    b[k] += 2.0*(new - old)*a[ki]
            a[i] = new
            S += new - old
    return a, b, S
