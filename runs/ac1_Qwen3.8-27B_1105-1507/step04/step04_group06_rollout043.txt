"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast log-sum-exp gradient descent with adaptive beta for rapid basin finding
2. LP-based mirror descent with pre-computed sparse structures for refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


# Global cache for sparse matrix structures
_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_structure(n):
    """Get or pre-compute the (row, col) arrays for the LP constraint matrix."""
    if n not in _structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _structure_cache[n] = (np.array(rows, dtype=np.int32),
                               np.array(cols, dtype=np.int32))
    return _structure_cache[n]


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    rows_arr, cols_arr = get_structure(n)
    vals = a[rows_arr - cols_arr]
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def smooth_obj_grad(a, beta):
    """
    Compute smooth approximation of J(a) and its gradient using log-sum-exp.
    Returns (F_value, gradient_array).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    p = exp_s / Z
    M = c_max + np.log(max(Z, 1e-300)) / beta

    # Compute dM/da using the reversal-convolution trick
    # dM/da[j] = (2/beta) * sum_k p[k] * a[k-j]
    # = (2/beta) * (p_rev * a)[2n-2-j]  where p_rev = p[::-1]
    p_rev = p[::-1]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    P = np.fft.rfft(p_rev, N)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(P * A, N)

    # dM/da[j] = (2/beta) * conv[2n-2-j] for j = 0, ..., n-1
    # So dM_da = (2/beta) * conv[n-1 : 2n-1][::-1]

def gradient_descent_phase(a, deadline, tracker, max_time=60.0):

    n = len(a)

    step = 0.1
    stall = 0
    best_F = current_F

    t_start = time.monotonic()

        else:

                N <<= 1

        for _ in range(20):

            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:

                continue
            a_try = a_try / s_try
            F_try = compute_J(a_try)
            if F_try < current_F:
                a = a_try
                current_F = F_try
                improved = True
                break
            step *= 0.5

        if improved:
            stall = 0
            if current_F < best_F:
                best_F = current_F
                best_a = a.copy()
                tracker.update(a)
            # Slowly increase beta to sharpen the max approximation

            step = min(step * 1.02, 0.5)
        else:

            step *= 0.5

                # Increase beta for sharper max approximation
                beta *= 2.0
                beta = min(beta, 1000.0)
                step = 0.1
                stall = 0
                if beta > 1000:
                    break

    return a if current_F < best_F else best_a


def lp_mirror_descent(a, max_iters, deadline, tracker, n_line_evals=12,

    gr = (np.sqrt(5.0) - 1.0) / 2.0

        M = float(b.max())

        if g is None:

        best_J = current_J
        lo = 0.0
        hi = 1.0

            best_J = J_at_1
            best_t = 1.0

            t2 = lo + gr * (hi - lo)
            a1 = (1.0 - t1) * a + t1 * g
            a2 = (1.0 - t2) * a + t2 * g
            J1 = compute_J(a1)
            J2 = compute_J(a2)
            if J1 < J2:
                hi = t2
            else:
                lo = t1
            t_mid = (lo + hi) / 2.0
            a_mid = (1.0 - t_mid) * a + t_mid * g
            J_mid = compute_J(a_mid)
            if J_mid < best_J:
                best_J = J_mid
                best_t = t_mid

        if best_J < current_J - 1e-13:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            current_J = best_J
            stall_count = 0
            if current_J < best_J_so_far:
                best_J_so_far = current_J
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # Try small fixed steps
            found_improve = False
            for t_try in [0.01, 0.005, 0.02, 0.05]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    found_improve = True
                    if current_J < best_J_so_far:
                        best_J_so_far = current_J
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            if not found_improve:
                stall_count += 1
                if stall_count >= stall_patience:
                    break

        if tracker is not None and it % 100 == 0 and it > 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.08, n) * p
        inits.append(np.maximum(p, 0.01))
        # Anti-symmetric perturbation of parent
        p_anti = p * (1.0 + 0.3 * np.sin(np.pi * t))
        inits.append(np.maximum(p_anti, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # One-sided (heavy on first half)
    inits.append(np.maximum(1.0 + 0.8 * (0.5 - t), 0.1))

    # Anti-symmetric (Barker-like)
    inits.append(1.0 + 0.5 * np.sin(np.pi * t))

    # Strong anti-symmetric
    inits.append(np.maximum(0.3 + 0.7 * (1.0 - 2.0 * t).clip(-1, 1), 0.05))

    # Cosine families (various frequencies)
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t - np.pi / 2))

    # Gaussian offset from center
    for mu in [0.25, 0.4, 0.6, 0.75]:
        inits.append(np.exp(-8.0 * (t - mu) ** 2) + 0.05)

    # Random smooth
    coeff = rng.normal(0, 1, min(n, 50))
    t_coarse = np.linspace(0, 1, len(coeff))
    smooth = np.interp(t, t_coarse, coeff)
    inits.append(1.0 + 0.3 * smooth)
    inits.append(np.maximum(1.0 + 0.5 * smooth, 0.05))

    # Sawtooth patterns
    for period in [4, 8, 16]:
        saw = (t * period) % 1.0
        inits.append(1.0 + 0.3 * (2.0 * saw - 1.0))

    # Random binary-like
    inits.append(rng.choice([0.2, 1.0], size=n))

    # Exponential decay + reverse
    inits.append(0.5 * np.exp(-3.0 * t) + 0.5 * np.exp(-3.0 * (1 - t)) + 0.1)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def peak_deflation(a, n_iterations=50, deadline=None):
    """
    Direct peak deflation: iteratively reduce a_i at positions that
    contribute most to the maximum convolution peak.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(n_iterations):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        # Find top contributing lags
        top_lags = np.argsort(b)[-min(10, 2 * n - 1):]

        # For each top lag k, the contribution to c_k from a_i is a_{k-i}
        # We want to reduce a at positions that appear frequently
        reduce_mask = np.zeros(n)
        for k in top_lags:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    reduce_mask[i] += b[k]

        # Reduce a where reduce_mask is high, but keep non-negative
        if reduce_mask.max() > 0:
            reduction = 0.02 * reduce_mask / reduce_mask.max()
            a_new = a - reduction * a
            a_new = np.maximum(a_new, 1e-8)
            s = a_new.sum()
            if s < 1e-6:
                break
            a_new = a_new / s
            J_new = compute_J(a_new)
            if J_new < current_J:
                a = a_new
                current_J = J_new
            else:
                # Try smaller reduction
                a_new = a - 0.005 * reduction * a
                a_new = np.maximum(a_new, 1e-8)
                s = a_new.sum()
                if s < 1e-6:
                    break
                a_new = a_new / s
                J_new = compute_J(a_new)
                if J_new < current_J:
                    a = a_new
                    current_J = J_new
                else:
                    break

    return a, current_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 0: Fast gradient descent (log-sum-exp) for rapid basin finding
    Phase 1: Peak deflation heuristic
    Phase 2: LP mirror descent (main refinement)
    Phase 3: Multi-restart with diverse perturbations
    Phase 4: Final LP polish + length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 0: Fast gradient descent (new) ===
    t0 = time.monotonic()
    phase0_budget = 150.0
    deadline_p0 = time.monotonic() + phase0_budget
    print(f"\n{'=' * 50} Phase 0: Gradient Descent (budget {phase0_budget:.0f}s) {'=' * 50}", flush=True)

    lengths_p0 = [512, 1024]
    if parent is not None:
        lengths_p0 = [parent.size] + lengths_p0

    for n in lengths_p0:
        if time.monotonic() > deadline_p0:
            break
        t_start = time.monotonic()
        print(f"  Gradient descent n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Try a subset of initializations
        max_inits = max(4, min(len(inits), 8))

        for idx in range(max_inits):
            if time.monotonic() > deadline_p0:
                break
            a0 = inits[idx].astype(np.float64)
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a_opt = gradient_descent_phase(a0, deadline_p0, tracker, max_time=10.0)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 0 complete")

    # === Phase 1: Peak deflation heuristic ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        phase1_budget = 60.0
        deadline_p1 = time.monotonic() + phase1_budget
        print(f"\n{'=' * 50} Phase 1: Peak Deflation (budget {phase1_budget:.0f}s) {'=' * 50}", flush=True)

        a_def, J_def = peak_deflation(tracker.best, n_iterations=200, deadline=deadline_p1)
        tracker.update(a_def)

        # Also try on perturbed versions
        for pert_scale in [0.05, 0.1, 0.2]:
            if time.monotonic() > deadline_p1:
                break
            a_pert = tracker.best + rng.normal(0, pert_scale, tracker.best.size) * tracker.best
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()
            a_def2, _ = peak_deflation(a_pert, n_iterations=100, deadline=deadline_p1)
            tracker.update(a_def2)

        tracker.report(f"Phase 1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP mirror descent (MAIN PHASE) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 750.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'=' * 50} Phase 2: LP Mirror Descent n={n_best} (budget {phase2_budget:.0f}s) {'=' * 50}", flush=True)

        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 40)
        max_iters = min(max_iters, 8000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker,
                              n_line_evals=14, stall_patience=40)

        # Try at different lengths
        for n_try in [512, 1024, 2048]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(150.0, deadline_p2 - time.monotonic())
            if budget_here < 20:
                continue
            max_iters_here = int(budget_here * 30)
            max_iters_here = min(max_iters_here, 3000)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker,
                                      n_line_evals=12, stall_patience=30)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 300.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'=' * 50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.1),
            ('gaussian', 0.2),
            ('smooth_noise', 0.1),
            ('spike_up', 5),
            ('spike_down', 5),
            ('anti_sym', 0.3),
            ('gaussian', 0.02),
            ('uniform', 0.15),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(5):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)
                t_arr = np.linspace(0, 1, n, endpoint=False)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n) * a
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    ks = max(3, n // 32)
                    kernel = np.ones(ks) / ks
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a
                elif pert_type == 'spike_up':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(1.5, 4.0, k)
                elif pert_type == 'spike_down':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.1, 0.5, k)
                elif pert_type == 'anti_sym':
                    a = a * (1.0 + param * np.sin(np.pi * t_arr + rng.uniform(0, np.pi)))

                a = np.maximum(a, 1e-6)
                s = a.sum()
                if s < 1e-6:
                    continue
                a = a / s

                # Quick check
                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # LP refinement
                budget_here = min(40.0, deadline_p3 - time.monotonic())
                if budget_here < 10:
                    continue
                max_iters = int(budget_here * 35)
                max_iters = min(max_iters, 1000)
                a = lp_mirror_descent(a, max_iters, deadline_p3, tracker,
                                      n_line_evals=12, stall_patience=20)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 30:
            phase4_budget = min(150.0, remaining)
            deadline_p4 = time.monotonic() + phase4_budget
            print(f"\n{'=' * 50} Phase 4: Final Polish n={n_best} (budget {phase4_budget:.0f}s) {'=' * 50}", flush=True)

            max_iters = int(phase4_budget * 35)
            max_iters = min(max_iters, 5000)
            a = lp_mirror_descent(tracker.best, max_iters, deadline_p4, tracker,
                                  n_line_evals=16, stall_patience=50)

            # Second pass
            if tracker.best is not None and time.monotonic() < deadline_p4:
                remaining2 = deadline_p4 - time.monotonic()
                if remaining2 > 20:
                    max_iters2 = int(remaining2 * 35)
                    max_iters2 = min(max_iters2, 3000)
                    a = lp_mirror_descent(tracker.best, max_iters2, deadline_p4, tracker,
                                          n_line_evals=16, stall_patience=40)

            # Length exploration
            for n_try in [n_best - 64, n_best + 64, 768, 1536]:
                if n_try < 128 or n_try > 2500 or time.monotonic() > deadline_p4:
                    continue
                if n_try == n_best:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(30.0, deadline_p4 - time.monotonic())
                if budget_here < 10:
                    continue
                max_iters = int(budget_here * 30)
                max_iters = min(max_iters, 800)
                a_res = lp_mirror_descent(a_res, max_iters, deadline_p4, tracker,
                                          n_line_evals=12, stall_patience=20)

            # Final peak deflation
            if tracker.best is not None and time.monotonic() < deadline_p4:
                a_final, _ = peak_deflation(tracker.best, n_iterations=100,
                                            deadline=deadline_p4)
                tracker.update(a_final)

            tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
