    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    # conv(p, a_rev)[k] = sum_j p_j * a_rev[k-j] = sum_j p_j * a[n-1-(k-j)]
    # Hmm, let me think more carefully.
    # We want g_i = sum_k p_k * a_{k-i} where 0 <= k-i < n
    # Let's define: g_i = sum_{k=i}^{i+n-1} p_k * a_{k-i} (clipped to valid range)
    # This is the cross-correlation (p, a) at lag i
    # In terms of convolution: cross_corr(p, a, lag=i) = conv(p_flipped, a)[i + len(p) - 1]
    
    # Simpler: use the fact that this is the "sliding dot product"
    # g = np.correlate(p, a, mode='full')[n-1 : n-1+n]
    # But for large arrays, use FFT
    p_flipped = p[::-1]
    s2 = 1
    while s2 < len(p_flipped) + n:
        s2 <<= 1
    corr = np.fft.irfft(np.fft.rfft(p_flipped, s2) * np.fft.rfft(a, s2), s2)
    # corr[k] = sum_j p_flipped[j] * a[k-j] = sum_j p[len(p)-1-j] * a[k-j]
    # Let m = len(p)-1-j, then j = len(p)-1-m
    # corr[k] = sum_m p[m] * a[k-(len(p)-1-m)] = sum_m p[m] * a[k-len(p)+1+m]
    # We want g_i = sum_k p_k * a_{k-i}
    # Let k' = k - i + n - 1... this is getting complicated.
    
    # Let me just use the direct formula with FFT:
    # g_i = sum_k p_k * a_{k-i} for 0 <= k-i < n, 0 <= k < len(p)
    # = sum_k p_k * a_{k-i} * [0 <= k-i < n]
    # Define q[j] = a[j] for 0<=j<n, 0 otherwise (already the case)
    # g_i = sum_k p_k * q[k-i] = (p * q_flipped)[len(p) - 1 + i]
    # where * denotes convolution
    q_flipped = a[::-1]
    s3 = 1
    while s3 < len(p) + n - 1:
        s3 <<= 1
    conv_pq = np.fft.irfft(np.fft.rfft(p, s3) * np.fft.rfft(q_flipped, s3), s3)
    # conv_pq[k] = sum_j p[j] * q_flipped[k-j] = sum_j p[j] * a[n-1-(k-j)]
    # We want sum_j p[j] * a[j-i]
    # Let j-i = m, j = m+i: sum_m p[m+i] * a[m]
    # conv_pq[i + n - 1] = sum_j p[j] * a[n-1-(i+n-1-j)] = sum_j p[j] * a[j-i]
    # Yes! So g_i = conv_pq[i + n - 1]
    grad_B = 2.0 * conv_pq[n - 1:n - 1 + n]
    
    # Full gradient of J = 2n * sm_val / s^2
    # dJ/da_i = 2n * (s * grad_B_i - 2*sm_val) / s^3
    grad_J = 2.0 * n * (s * grad_B - 2.0 * sm_val) / (s ** 3)
    return grad_J


def multiplicative_peak_suppression(a, top_k=3, lr=0.1, n_iters=1):
    """Apply multiplicative peak suppression for n_iters steps."""
    a = np.maximum(a.copy(), 1e-10)
    n = len(a)
    for _ in range(n_iters):
        B = fast_conv(a)
        # Weight peaks by their value (emphasis on larger peaks)
        B_sorted = np.sort(B)[::-1]
        # Use top-k peaks with geometric weighting
        weights = np.zeros(len(B))
        for idx in range(min(top_k, len(B))):
            k = np.argsort(B)[-top_k:][idx]
            weights[k] = (0.5 ** idx)
        w_sum = weights.sum()
        if w_sum > 0:
            weights /= w_sum
        # Compute weighted subgradient direction
        # direction_i = sum_k weights[k] * a[k-i]
        s_fft = 1
        while s_fft < len(B) + n - 1:
            s_fft <<= 1
        w_flipped = weights[::-1]
        a_flipped = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(weights, s_fft) * np.fft.rfft(a_flipped, s_fft), s_fft)
        direction = conv_w[n - 1:n - 1 + n]
        # Multiplicative update: a_i *= exp(-lr * direction_i)
        a = a * np.exp(-lr * direction / max(direction.max(), 1e-10))
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    return a


def accelerated_fw(a, deadline, max_iters=5000, top_k=3, tracker=None):
    """
    Accelerated mirror descent with multiplicative peak suppression.
    Uses Nesterov acceleration and adaptive learning rate.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    if tracker is not None:
        tracker.update(a)
    
    # Nesterov momentum
    y = a.copy()  # momentum variable
    lr = 0.05
    best_lr = lr
    stall_count = 0
    improvements = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute peaks at momentum point
        B = fast_conv(y)
        B_max = B.max()
        
        # Weight peaks
        top_indices = np.argsort(B)[-top_k:]
        weights = np.zeros(len(B))
        for idx, k in enumerate(top_indices):
            weights[k] = (0.6 ** idx)
        w_sum = weights.sum()
        if w_sum > 0:
            weights /= w_sum
        
        # Compute direction: d_i = sum_k weights[k] * a[k-i]
        s_fft = 1
        while s_fft < len(B) + n - 1:
            s_fft <<= 1
        a_flipped = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(weights, s_fft) * np.fft.rfft(a_flipped, s_fft), s_fft)
        direction = conv_w[n - 1:n - 1 + n]
        
        # Multiplicative update with Nesterov
        # y_new = (1-t)*y + t*e_{i*} (FW step on y)
        # But we use multiplicative: a_new = a * exp(-lr * direction)
        
        # Adaptive learning rate: try a range of steps
        best_step_J = current_J
        best_step_a = a.copy()
        
        for t in [lr, lr * 0.5, lr * 2, lr * 0.25, lr * 4]:
            a_try = a * np.exp(-t * direction / max(direction.max() * n / len(B), 1e-10))
            a_try = np.maximum(a_try, 1e-10)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            J_try = compute_J(a_try)
            if J_try < best_step_J:
                best_step_J = J_try
                best_step_a = a_try
                best_lr = t
        
        if best_step_J < current_J - 1e-14:
            # Nesterov update
            t_nesterov = 2.0 / (it + 3)  # Polyak step size
            y = best_step_a + (t_nesterov / (it + 3)) * (best_step_a - a) if it > 0 else best_step_a
            a = best_step_a
            current_J = best_step_J
            stall_count = 0
            improvements += 1
            
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # No improvement - reduce learning rate
            lr *= 0.7
            if lr < 1e-6:
                lr = 0.05  # Reset
            stall_count += 1
            # Try a small random perturbation to escape
            if stall_count > 20:
                a = a + rng_global.normal(0, 0.02, n) * a
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                y = a.copy()
                current_J = compute_J(a)
                stall_count = 0
        
        if it % 100 == 0 and tracker is not None:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a if best_J <= compute_J(a) else a


def smooth_gradient_descent(a, deadline, max_iters=2000):
    """Gradient descent on smoothed objective with momentum and adaptive beta."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    momentum = np.zeros(n)
    beta = 5.0
    lr = 0.01
    v = np.zeros(n)  # for Nesterov
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad = gradient_smooth_J(a, beta)
        if not np.all(np.isfinite(grad)):
            grad = np.nan_to_num(grad, nan=0.0)
        
        # Nesterov momentum
        v = 0.9 * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-8:
                break
        
        # Anneal beta
        if it % 100 == 0:
            beta *= 1.5
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n).copy())
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine families
    for freq in [2, 3, 4, 5, 6, 7, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))
    
    # Sine
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * 3 * t_arr), 0.01))
    
    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    
    # Two-peak
    inits.append(1.0 + 0.4 * np.cos(4.0 * np.pi * t_arr) * np.cos(2.0 * np.pi * t_arr))
    
    # Power law
    inits.append((t_arr + 0.05) ** 0.3 * (1 - t_arr + 0.05) ** 0.3)
    
    # Exponential decay
    inits.append(np.exp(-1.0 * t_arr) + 0.2)
    
    # Random
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Frequency-shaped: concentrated at specific frequencies
    for k_center in [n // 4, n // 3, n // 2, 2 * n // 3]:
        f = np.fft.rfftfreq(n)
        mask = np.exp(-((f - k_center / n) ** 2) / (2 * (1.0 / n) ** 2))
        inits.append(np.maximum(np.fft.irfft(np.fft.rfft(np.ones(n)) * mask, n), 0.01))
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

            for _ in range(2000):
                if time.monotonic() > deadline:
                    break
                a = multiplicative_peak_suppression(a, top_k=5, lr=lr, n_iters=1)
                # Occasionally try smoothed gradient
                if _ % 50 == 0:
                    a = smooth_gradient_descent(a, time.monotonic() + 2.0, max_iters=50)
                    tracker.update(a)
            tracker.update(a)
            
            # Very fine multiplicative steps
            if time.monotonic() < deadline:
                a = tracker.best.copy()
                a = a / a.sum()
                for _ in range(3000):
                    if time.monotonic() > deadline:
                        break
                    a = multiplicative_peak_suppression(a, top_k=1, lr=0.01, n_iters=1)
                tracker.update(a)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
