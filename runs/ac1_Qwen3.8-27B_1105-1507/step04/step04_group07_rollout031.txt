        cumulative = np.cumsum(v_sorted)

        best_lambda = 0.0

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. KKT water-filling fixed-point iteration (primary method)
2. Nesterov-accelerated LSE gradient descent with FFT-based gradients
3. LP-based best-response polishing with progressive tightening
4. Multi-scale initialization (small n -> large n via upsampling)
5. Diversified restarts and simulated annealing escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# ============ Core Computation ============

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, p

def kkt_subgrad(a, active_mask):
    n = len(a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(active_mask.astype(np.float64), s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    return 2.0 * corr[:n]

def simplex_project(v):
    n = len(v)
    v_max = v.max()
    if v_max <= 0:
        return np.ones(n) / n
    lo = 0.0
    hi = float(v_max)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    a = np.maximum(v - lam, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        return np.ones(n) / n
    return a / a_sum

def kkt_fixed_point(a, max_iters, deadline, tracker, relax_start=0.8, relax_min=0.05):
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    relax = relax_start
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-10 * M, 1e-14)
        active_mask = (b >= M - tol).astype(np.float64)
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask = np.zeros(2 * n - 1)
            active_mask[np.argmax(b)] = 1.0
            n_active = 1
        active_mask /= n_active
        sg = kkt_subgrad(a, active_mask)
        v = sg / (2.0 * M)
        v_sum = np.maximum(v, 0.0).sum()
        if v_sum < 1e-10:
            break
        a_new = simplex_project(v)
        a_try = (1.0 - relax) * a + relax * a_new
        a_try = np.maximum(a_try, 1e-10)
        a_try /= a_try.sum()
        J_try = compute_J(a_try)
        if J_try < current_J - 1e-15:
            a = a_try
            current_J = J_try
            stall = 0
            relax = min(1.0, relax * 1.05)
        else:
            relax = max(relax_min, relax * 0.7)
            stall += 1
            if stall > 100:
                break
        if it % 30 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def nesterov_lse(a, max_iters, deadline, tracker, tau_start=30.0, tau_end=500.0):
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    tau = tau_start
    lr = 1.0 / (n * 4.0)
    t_k = 1.0
    a_prev = a.copy()
    y = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, _ = lse_grad(y, tau)
        grad_mean = grad.mean()
        grad_centered = grad - grad_mean
        a_cand = y - lr * grad_centered
        a_new = simplex_project(a_cand + grad_mean * n / n)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_new
            a_prev = a.copy()
            a = a_new
            y = a + beta * (a - a_prev)
            current_J = J_new
            t_k = t_new
            stall = 0
            tau = min(tau_end, tau * 1.005)
            lr = 1.0 / (n * 4.0 * np.sqrt(tau / tau_start))
        else:
            lr *= 0.5
            stall += 1
            if stall > 50:
                tau = max(tau_start, tau * 0.9)
                lr = 1.0 / (n * 4.0 * np.sqrt(tau / tau_start))
                stall = 0
                if stall > 200:
                    break
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def plain_lse(a, max_iters, deadline, tracker, tau=100.0):
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 1.0 / (n * 10.0)
    momentum = 0.8
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, _ = lse_grad(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 2.0 / n)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                break
        if it % 30 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lp_polish(a, deadline, tracker, max_rounds=30):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a /= a.sum()

    def build_conv_matrix():
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                rows.append(k)
                cols.append(j)
        return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

    rows, cols = build_conv_matrix()
    nnz = len(rows)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - 0.005 * (rnd + 1))
        vals = np.empty(nnz)
        for i in range(nnz):
            k = rows[i]
            j = cols[i]
            vals[i] = a[k - j]
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_tight)
        bounds = [(0.0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 5})
            if not r.success or r.x is None:
                b_ub = np.full(2 * n - 1, M)
                r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                            options={'presolve': True, 'time_limit': 5})
                if not r.success or r.x is None:
                    break
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g /= gs
        except Exception:
            break
        best_a = None
        best_J = compute_J(a)
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        tracker.update(a)
    tracker.update(a)
    return a

# ============ Initialization ============

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def flat_spectrum_init(n, rng):
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t))
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.cos(np.pi * t) ** 0.5)
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

        if np.isfinite(score) and score < self.best_score:

            self.best_score = score

def diversify_restart(a, deadline, tracker, n_restarts=20):
    rng = np.random.default_rng(12345)
    n = a.size
    current = a.copy()
    current_J = compute_J(current)
    tracker.update(current)
    temp = 0.1

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        noise = rng.normal(0, temp, n)
        a_pert = current + noise
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        a_refined = kkt_fixed_point(a_pert, 200, deadline - 5, tracker, relax_start=0.6)
        a_refined = nesterov_lse(a_refined, 500, deadline - 5, tracker, tau_start=50.0)
        J_ref = compute_J(a_refined)
        delta = J_ref - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            current = a_refined
            current_J = J_ref
        temp *= 0.93
    tracker.update(current)
    return current
